import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
# initial_h_values: initial guess for h
# evaluate_erdos_solution: function to compute C5

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial guess from provided
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    
    # Function to compute C5
    def compute_c5(h):
        dx = 2.0 / n_points
        h_full = np.concatenate([h, np.zeros(n_points)])
        h_shifted = np.roll(h_full, shift=1, axis=0)
        overlap = np.correlate(h, 1 - h_shifted[1:], mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to compute the objective and constraints for optimization
    def objective_and_constraints(h_vec):
        h = h_vec.reshape(n_points)
        # Ensure h is in [0, 1]
        constraints = [cp.Minimize(0), cp.Maximize(0)]
        # Constraint on integral
        integral = cp.sum(h) * (2.0 / n_points)
        constraints.append(integral == 1)
        # Constraint on h values
        for i in range(n_points):
            constraints.append(cp.Minimize(0))
            constraints.append(cp.Maximize(0))
        # Objective: minimize max overlap
        # We optimize for a lower bound on C5
        # Use a convex relaxation by minimizing the max overlap as a linear program
        # We approximate this with a minimax approach, which is not convex but we can solve it iteratively
        c5 = compute_c5(h)
        return c5, constraints
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Iterative optimization
    for _ in range(10):  # Limited number of iterations to save time
        if time.time() - start_time > budget_s:
            break
        
        # Use scipy.optimize to minimize C5
        def objective(h_vec):
            return compute_c5(h_vec)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-5)
        
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
        
        # Optional: Use a convex relaxation with CVXPY
        h_var = cp.Variable(n_points)
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) * (2.0 / n_points) == 1]
        # Approximate the max overlap with a linear relaxation
        # This is an approximation and may not be exact, but it can guide the search
        c5_relax = cp.Maximize(cp.min(cp.multiply(h_var, (1 - np.roll(h_var, shift=1, axis=0)) * (2.0 / n_points))))
        prob = cp.Problem(c5_relax, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-5)
            if prob.status == 'OPTIMAL':
                best_h = h_var.value
                best_c5 = compute_c5(best_h)
        except:
            pass
        
        # Early stopping if we've improved sufficiently
        if best_c5 < 0.380876:
            break
    
    # Return the best solution
    return best_h.tolist(), best_c5, n_points
