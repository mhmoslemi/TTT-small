        return c5

    def objective_function(h):
        return compute_c5(h)

        elapsed = 0
        best_c5 = float('inf')
        best_h = h_values.copy()
        num_iterations = 100
        for i in range(num_iterations):
            start_time = time.time()
            # Use local optimization
            result = optimize.minimize(
                objective_function,
                h_values,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'ftol': 1e-6, 'maxiter': 100}
            )
            elapsed += time.time() - start_time
            if elapsed >= budget_s:
                break
            if result.success:
                h_values = result.x
                current_c5 = compute_c5(h_values)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_values.copy()
