import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Define the run function
def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and precision
    dx = 2.0 / n_points
    h_values = initial_h_values.copy()[:n_points]  # Start with initial construction
    
    start_time = time.time()
    
    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points - n_points % 2]])  # Ensure even length for full convolution
        shifted_h = np.zeros(2 * n_points)
        shifted_h[n_points:] = h
        shifted_1_minus_h = 1 - h
        correlate = np.correlate(shifted_h, shifted_1_minus_h, mode="full")
        c5 = np.max(correlate) * dx
        return c5
    
    def objective_function(h_values):
        return evaluate_c5(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Use SLSQP method for optimization
    result = optimize.minimize(
        objective_function,
        h_values,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
    )
    
    # Check if optimization completed within budget
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        print("Time budget exceeded. Returning best solution so far.")
        return (h_values, evaluate_c5(h_values), n_points)
    
    # Refine the solution with more points if time allows
    if elapsed_time < budget_s - 5:
        n_points += 50
        dx = 2.0 / n_points
        h_values = result.x
        h_values = np.pad(h_values, (0, n_points - len(h_values)), 'constant')[:n_points]
        
        # Re-evaluate with new points
        result = optimize.minimize(
            objective_function,
            h_values,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-6, 'disp': False}
        )
    
    # Final evaluation
    c5_bound = result.fun
    return (result.x, c5_bound, n_points)
