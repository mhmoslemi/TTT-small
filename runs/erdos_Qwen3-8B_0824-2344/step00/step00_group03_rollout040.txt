import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points for computational efficiency
    dx = 2.0 / n_points
    
    # Initial guess
    h = initial_h_values.copy()
    h = h[:n_points]  # Ensure we have exactly n_points samples
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the integral constraint
    
    # Evaluate initial solution
    def compute_c5(h_vals):
        h_vals = np.array(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    c5_initial = compute_c5(h)
    
    # Optimize h to minimize C5
    def objective_func(h_vals):
        h_vals = np.array(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define optimization problem
    x = cp.Variable(n_points, bounds=bounds)
    objective = cp.Minimize(cp.Max(cp.sum(cp.multiply(x, 1 - x[k:k + n_points])) for k in range(n_points)))
    constraints = [cp.sum(x) == n_points / 2]
    prob = cp.Problem(objective, constraints)
    
    # Solve using ECOS
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
    except:
        pass
    
    # Extract the optimal solution
    h_opt = x.value
    
    # Evaluate the optimized solution
    c5_opt = compute_c5(h_opt)
    
    # Return the best solution found
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Perform additional fine-tuning if time allows
        h_refined = np.copy(h_opt)
        for _ in range(10):
            # Perturb the solution slightly
            perturbation = np.random.normal(0, 0.05, n_points)
            h_refined += perturbation
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            c5_refined = compute_c5(h_refined)
            if c5_refined < c5_opt:
                c5_opt = c5_refined
                h_opt = h_refined
    
    return h_opt, c5_opt, n_points
