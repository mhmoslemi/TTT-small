```python
import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_points = 100  # Starting with 100 points for speed
    dx = 2.0 / n_points
    
    # Initial guess from the provided initial_h_values
    initial_h = initial_h_values.copy()
    initial_h = np.resize(initial_h, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Optimization function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Gradient-based optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    result = optimize.minimize(objective, initial_h, bounds=bounds, method='L-BFGS-B', tol=1e-6)
    best_h = result.x
    best_c5 = result.fun
    
    # Try increasing n_points if time allows
    while time.time() - start_time < budget_s - 1:
        n_points += 10
        dx = 2.0 / n_points
        new_h = np.zeros(n_points)
        # Interpolate best_h to new points
        for i in range(n_points):
            new_h[i] = np.interp(i * dx, np.linspace(0, 2, len(best_h)), best_h)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Objective function for new n_points
        def new_objective(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(new_objective, new_h, bounds=bounds, method='L-BFGS-B', tol=1e-6)
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h, c5_bound, len(best_h))
```
