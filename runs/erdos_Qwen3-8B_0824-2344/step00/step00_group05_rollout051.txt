import numpy as np

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose n_points (smaller is better for speed)
    n_points = 100  # Try with 100 points for faster optimization
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[initial_h_values[:n_points]] = 1.0
    h = h[:n_points]  # Keep only the first n_points samples
    
    # Ensure the integral is exactly 1
    total = np.sum(h) * dx
    if abs(total - 1.0) > 1e-6:
        h = h / total * 1.0
    
    # Function to compute the overlap integral for a given h
    def compute_c5(h):
        h = np.array(h)
        h_ext = np.concatenate([h, np.zeros(n_points)])
        h_shifted = np.zeros(2 * n_points)
        for k in range(1, n_points + 1):
            h_shifted[k:] = h[:n_points - k]
        shift = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(shift * dx)
        return max_overlap
    
    # Optimization: minimize the maximum overlap integral
    # We will use a gradient-based approach with a penalty function
    def objective_function(h_values):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        # Ensure integral is exactly 1
        total = np.sum(h) * dx
        if abs(total - 1.0) > 1e-6:
            h = h / total * 1.0
        return compute_c5(h)
    
    # Use scipy optimization
    result = optimize.minimize(
        fun=objective_function,
        x0=h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        tol=1e-5,
        options={'maxiter': 100}
    )
    
    # Check if we are still within the budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 5:
        # If time is running out, return the best result so far
        return (h, compute_c5(h), n_points)
    
    # If we have time, perform a more refined optimization
    result = optimize.minimize(
        fun=objective_function,
        x0=result.x,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        tol=1e-8,
        options={'maxiter': 50}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Final check for the best result
    return (best_h, best_c5, n_points)
