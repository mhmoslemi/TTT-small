import numpy as np

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the target C5 value we want to beat
    target_c5 = 0.380876
    
    # Start with the initial solution provided
    initial_h_values = np.array(initial_h_values)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Function to evaluate the C5 value for a given h
    def evaluate_c5(h):
        h = np.array(h)
        h = np.tile(h, (len(h), 1))
        h = np.roll(h, shift=np.arange(len(h)), axis=1)
        overlap = np.sum(h * (1 - h[:, 1:]), axis=1)
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to compute the cost (C5) for a given h
    def objective(h):
        return evaluate_c5(h)
    
    # Function to compute the constraints (sum(h) * dx == 1)
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization settings
    max_iterations = 100
    tolerance = 1e-5
    best_c5 = float('inf')
    best_h = initial_h_values
    
    # Start the timer
    start_time = time.time()
    
    for _ in range(max_iterations):
        # Define the optimization variables
        h = cp.Variable(n_points)
        # Define the objective
        obj = cp.Minimize(objective(h))
        # Define the constraints
        constraints = [constraint(h) == 0]
        
        # Solve the problem using SCS solver
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=tolerance)
            if prob.status == 'optimal':
                current_h = np.array(h.value)
                current_c5 = evaluate_c5(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
                if best_c5 <= target_c5:
                    break  # Early stopping if target is achieved
        except:
            pass
        
        # Check for time budget
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
