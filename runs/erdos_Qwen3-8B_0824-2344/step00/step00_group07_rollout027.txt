import numpy as np
import scipy.optimize as opt
import cvxpy as cp

import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Number of points
    n_points = 80  # Start with 80 points, which is manageable and fast
    
    # Initial guess from provided initial_h_values
    h_values = initial_h_values.copy()
    h_values = h_values[:n_points]  # Ensure it's of correct length
    
    # Normalize to satisfy the integral constraint
    total_integral = np.sum(h_values) * (2.0 / n_points)
    scaling_factor = 1.0 / total_integral
    h_values *= scaling_factor
    
    # Function to compute C5 bound
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Find the maximum overlap integral
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Function to optimize h using gradient descent with constraints
    def optimize_h(h_values, max_iter=100, learning_rate=0.01):
        h = np.array(h_values)
        n = len(h)
        dx = 2.0 / n
        
        for _ in range(max_iter):
            # Compute current C5
            current_c5 = compute_c5(h)
            
            # Compute the gradient of C5 with respect to h
            # For simplicity, we use a numerical approximation
            grad = np.zeros(n)
            eps = 1e-6
            for i in range(n):
                h_plus = h.copy()
                h_plus[i] += eps
                h_minus = h.copy()
                h_minus[i] -= eps
                grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * eps)
            
            # Compute the integral constraint gradient
            grad_integral = np.ones(n) * (2.0 / n)
            
            # Project to feasible region
            h = np.clip(h - learning_rate * grad / np.linalg.norm(grad), 0, 1)
            
            # Enforce integral constraint
            total_integral = np.sum(h) * dx
            scaling_factor = 1.0 / total_integral
            h = h * scaling_factor
            
        return h
    
    # Run the optimization
    best_h = copy.deepcopy(h_values)
    best_c5 = compute_c5(best_h)
    
    # Run for a few iterations to improve
    for _ in range(10):
        h = optimize_h(best_h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # If we have time, increase the number of points to potentially get a better solution
    if time.time() - start_time < budget_s - 5:
        n_points = 100
        h_values = np.zeros(n_points)
        # Initialize with a better guess
        for i in range(n_points):
            h_values[i] = 0.5 if i % 3 == 0 else 0.8 if i % 3 == 1 else 0.2
        total_integral = np.sum(h_values) * (2.0 / n_points)
        scaling_factor = 1.0 / total_integral
        h_values *= scaling_factor
        
        # Run the optimization
        best_h = copy.deepcopy(h_values)
        best_c5 = compute_c5(best_h)
        
        for _ in range(20):
            h = optimize_h(best_h, max_iter=20, learning_rate=0.005)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
    
    # Return the result
    return (best_h, best_c5, n_points)
