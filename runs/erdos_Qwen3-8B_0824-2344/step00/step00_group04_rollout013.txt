import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Try different numbers of points to find a good balance between accuracy and speed
    n_points_list = [100, 200, 300, 400, 500]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        h_initial = initial_h_values[:n_points] if len(initial_h_values) >= n_points else np.random.rand(n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Ensure integral is 1
        
        # Function to evaluate C5 for given h
        def evaluate_c5(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1
            correlation = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(correlation) * dx
            return c5
        
        # Optimization settings
        max_iter = 100
        tol = 1e-4
        time_used = 0
        
        # Start optimization
        h = h_initial.copy()
        for _ in range(max_iter):
            start_time = time.time()
            
            # Perturb h slightly
            perturbation = np.random.normal(0, 0.05, n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1
            
            # Evaluate C5
            c5 = evaluate_c5(h)
            
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n_points
            
            # Check time budget
            time_used = time.time() - start_time
            if time_used > budget_s:
                break
            
        # Early termination if best solution is already found with sufficient accuracy
        if best_c5 <= 0.380876:
            break
    
    # Final evaluation
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (best_n / 2) / np.sum(best_h)  # Ensure integral is 1
        c5 = evaluate_c5(best_h)
    else:
        # Default to initial_h_values if no improvement was found
        best_h = initial_h_values
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (len(best_h) / 2) / np.sum(best_h)
        c5 = evaluate_c5(best_h)
    
    return (best_h, c5, len(best_h))
