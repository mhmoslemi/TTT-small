import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Initial construction from initial_h_values (70 samples)
    n_points = 70
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]
    # Ensure the sum of h_initial satisfies the constraint
    total_sum = np.sum(h_initial)
    required_sum = n_points / 2
    if abs(total_sum - required_sum) > 1e-6:
        # Normalize the initial values
        h_initial = h_initial * (required_sum / total_sum)
    
    # Define a function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate([h, np.zeros_like(h)])
        shifted = np.roll(h, 1)
        overlap = correlate(h, 1 - shifted, mode='full')
        dx = 2 / len(h)
        c5 = np.max(overlap * dx)
        return c5
    
    # Objective function for optimization
    def objective(x):
        return compute_c5(x)
    
    # Bounds for each point in h (between 0 and 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using the Nelder-Mead algorithm
    start_time = time.time()
    result = opt.minimize(
        objective,
        h_initial,
        method='Nelder-Mead',
        bounds=bounds,
        options={'xatol': 1e-4, 'maxiter': 100}
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Check if we need to refine with a more precise optimization
    if time.time() - start_time < budget_s - 10:
        # Use CVXPY for more precise optimization
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        # Objective is to minimize the maximum overlap
        overlap = correlate(np.concatenate([h.value, np.zeros_like(h.value)]), 1 - np.roll(h.value, 1), mode='full')
        dx = 2 / len(h.value)
        c5 = cp.Max(overlap * dx)
        prob = cp.Problem(cp.Minimize(c5), constraints)
        prob.solve(solver='SCIP', verbose=False)
        
        best_h = h.value
        best_c5 = prob.value
    
    # Ensure the final solution respects the constraints
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if abs(final_sum - n_points / 2) > 1e-6:
        final_h = final_h * (n_points / 2) / final_sum
    final_h = np.clip(final_h, 0, 1)
    
    return final_h, best_c5, n_points
