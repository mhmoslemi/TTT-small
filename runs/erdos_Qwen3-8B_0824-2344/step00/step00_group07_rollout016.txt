```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 80  # Start with the initial n_points

    # Try increasing n_points if time allows
    for n_points in [80, 100, 120, 140, 160, 180, 200]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate initial h_values based on n_points
        if n_points == 80:
            h_initial = initial_h_values
        else:
            h_initial = np.zeros(n_points)
            dx = 2.0 / n_points
            # Create a piecewise constant function that integrates to 1
            # Example: spread the integral evenly across the interval
            h_initial = np.repeat(1.0 / n_points, n_points)
        
        # Optimize h to minimize the maximum overlap integral
        # Convert to a decision variable
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2]
        
        # Evaluate the overlap for a given h
        def objective_func(h_values):
            h_values = np.array(h_values)
            dx = 2.0 / n_points
            # Compute the overlap integral for all k
            overlap = []
            for k in range(n_points):
                corr = np.correlate(h_values, 1 - h_values, mode="full")
                overlap.append(corr[k + n_points - 1] * dx)
            return np.max(overlap)
        
        # Use a numerical optimizer
        result = optimize.minimize(
            objective_func,
            h_initial,
            bounds=[(0, 1) for _ in range(n_points)],
            method='L-BFGS-B',
            options={'ftol': 1e-6, 'gtol': 1e-6, 'maxiter': 100}
        )
        
        h_opt = result.x
        c5 = objective_func(h_opt)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        
        # Optionally, print progress
        print(f"n_points={n_points}, C5={c5:.6f}, best={best_c5:.6f}")

    # Evaluate the best solution found
    final_h = best_h
    final_c5 = best_c5
    print(f"Final best C5: {final_c5:.6f} with n_points={len(final_h)}")
    
    return final_h, final_c5, len(final_h)
```
