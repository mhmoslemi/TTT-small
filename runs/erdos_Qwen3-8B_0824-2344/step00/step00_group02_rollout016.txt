```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported helper functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 200  # Start with 200 points for efficiency
    
    # Initial guess from the given starting point
    h_initial = np.array(initial_h_values)
    h_initial = h_initial[:n_points]  # Truncate or extend as needed
    h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant') if len(h_initial) < n_points else h_initial[:n_points]
    
    # Normalize to ensure the integral is 1
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        # Reshape h to ensure it's 1D and of length n_points
        h = h.reshape(n_points)
        # Ensure the integral is exactly 1
        h_norm = h * (n_points / 2) / np.sum(h)
        # Compute the overlap integral using convolution
        overlap = signal.correlate(h_norm, 1 - h_norm, mode="full") * (2 / n_points)
        # Return the maximum overlap as the objective
        return np.max(overlap)
    
    # Define the constraint that the integral is 1 (already handled in objective)
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    start_time = time.time()
    result = opt.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        jac=None,
        options={'ftol': 1e-6, 'maxiter': 100}
    )
    
    # Extract the best solution
    best_h = result.x
    best_h = best_h.reshape(n_points)
    best_h = best_h * (n_points / 2) / np.sum(best_h)  # Ensure integral is 1
    
    # Compute the C₅ value
    c5_bound = evaluate_erdos_solution(best_h)
    
    # Check time and refine if needed
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Try with more points if time allows
        n_points = int(n_points * 1.5)
        h_initial = np.array(best_h)
        h_initial = h_initial[:n_points]  # Truncate or extend as needed
        h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant') if len(h_initial) < n_points else h_initial[:n_points]
        
        # Normalize
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Re-define the objective with more points
        def objective_large(h):
            h = h.reshape(n_points)
            h_norm = h * (n_points / 2) / np.sum(h)
            overlap = signal.correlate(h_norm, 1 - h_norm, mode="full") * (2 / n_points)
            return np.max(overlap)
        
        # Optimization with more points
        result = opt.minimize(
            objective_large,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            jac=None,
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        
        best_h = result.x
        best_h = best_h.reshape(n_points)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        c5_bound = evaluate_erdos_solution(best_h)
    
    return (best_h, c5_bound, n_points)
```
