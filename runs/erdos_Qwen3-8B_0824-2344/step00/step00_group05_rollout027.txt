def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Determine number of points
    n_points = 256  # Good balance between speed and resolution
    n_half = n_points // 2
    dx = 2.0 / n_points
    
    # Initial guess
    h = initial_h_values[:n_points].copy()
    
    # Normalize to satisfy volume constraint
    h = h / np.sum(h) * (n_points / 2)
    
    # Function to evaluate C_5
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Define objective function
    def objective(h):
        return evaluate_c5(h)
    
    # Define constraints
    def constraints(h):
        return [np.sum(h) - n_points / 2]
    
    # Optimization using gradient descent with constraints
    def optimize_h(h):
        for _ in range(1000):
            # Compute current C5
            current_c5 = evaluate_c5(h)
            if current_c5 < 0.3808:
                break
            # Compute the correlation
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr_index = np.argmax(corr)
            # Adjust h to reduce the maximum correlation
            shift = max_corr_index - n_points + 1  # Shift of the maximum overlap
            # Apply a small perturbation to reduce overlap at the maximum point
            h[shift:shift + n_points // 2] = np.clip(h[shift:shift + n_points // 2] - 0.01, 0, 1)
            h = np.clip(h, 0, 1)
            # Ensure volume constraint
            h = h / np.sum(h) * (n_points / 2)
        return h
    
    # Optimize
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Run optimization
    for _ in range(50):
        h = optimize_h(h.copy())
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        if time.time() - start_time > budget_s:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n_points)
