import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=60, **kwargs):

    dx = 2.0 / n_points
    h_initial = initial_h_values(n_points)  # Use the provided initial construction
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Create 1 - h
        one_minus_h = 1.0 - h
        # Compute the correlation
        corr = np.correlate(h, one_minus_h, mode="full")
        # Find the maximum value and multiply by dx
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to optimize h to minimize C5
    def optimize_h(h_initial, n_points, budget_s):
        # Convert to numpy array
        h = np.array(h_initial)
        # Objective: minimize the max overlap integral
        # Use a local optimization method to find a better solution
        
        # Define optimization variables
        x = cp.Variable(n_points, nonneg=True)
        # Constraint: sum of h must equal n_points / 2
        constraint = [cp.sum(x) == n_points / 2]
        
        # Objective function: minimize the maximum overlap integral
        # We'll use a convex relaxation by penalizing large values
        # Here, we use a gradient-based approach with scipy.optimize
        
        # Define the objective function for scipy
        def objective(x):
            h = x
            one_minus_h = 1.0 - h
            corr = np.correlate(h, one_minus_h, mode="full")
            return np.max(corr) * dx
        
        # Set up bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run the optimization
        start_time = time.time()
        result = opt.minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 100})

        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            print("Time budget exceeded, returning initial solution.")

        # Get the optimized solution
        h_opt = result.x
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt, n_points
    
    # Run the optimization
    best_h, best_c5, n = optimize_h(h_initial, n_points, budget_s)

    if budget_s > 10:
        n_points_refine = 200

        start_time = time.time()

        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s:
            best_h = result_refine.x
            best_c5 = compute_c5(best_h)
    
    # Final check
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n
