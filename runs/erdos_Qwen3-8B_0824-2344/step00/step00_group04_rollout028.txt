```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set the random seed
    np.random.seed(seed)
    
    # Start the timer
    start_time = time.time()
    
    # We'll use 42 samples as a starting point
    n_points = 42
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    # Ensure the initial solution satisfies the constraints
    while True:
        # Ensure the integral constraint is satisfied
        if np.abs(np.sum(h) * dx - 1) > 1e-8:
            h = np.random.uniform(0, 1, n_points)
            continue
        
        # Check if the current h is valid
        if np.all(h >= 0) and np.all(h <= 1):
            break
    
    best_c5 = evaluate_erdos_solution(h, n_points, dx)
    best_h = h.copy()
    
    # Try a few random perturbations to escape local minima
    for _ in range(5):
        # Perturb the h values slightly
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h_perturbed = np.clip(h + perturbation, 0, 1)
        
        # Ensure the integral constraint is satisfied
        while True:
            if np.abs(np.sum(h_perturbed) * dx - 1) > 1e-8:
                h_perturbed = np.random.uniform(0, 1, n_points)
                continue
        
            if np.all(h_perturbed >= 0) and np.all(h_perturbed <= 1):
                break
        
        # Evaluate the C5 value
        c5 = evaluate_erdos_solution(h_perturbed, n_points, dx)
        
        # Update the best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()
    
    # Try to refine with a gradient-based optimizer
    def objective(x):
        h = x.reshape(n_points)
        # Ensure the integral constraint is satisfied
        if np.abs(np.sum(h) * dx - 1) > 1e-8:
            return np.inf
        
        # Compute the C5 value
        c5 = evaluate_erdos_solution(h, n_points, dx)
        return c5
    
    # Define bounds and initial guess
    bounds = [(0, 1) for _ in range(n_points)]
    initial_guess = best_h.copy()
    
    # Perform optimization
    result = opt.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 100},
        jac=None
    )
    
    # Update best solution if improvement is found
    if result.success:
        h_opt = result.x
        c5_opt = evaluate_erdos_solution(h_opt, n_points, dx)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Check time remaining
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # If there is time left, try with more samples
    if remaining > 1.0:
        n_points = min(100, int(remaining * 10))
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        
        # Ensure the integral constraint is satisfied
        while True:
            if np.abs(np.sum(h) * dx - 1) > 1e-8:
                h = np.random.uniform(0, 1, n_points)
                continue
        
            if np.all(h >= 0) and np.all(h <= 1):
                break
        
        # Evaluate the C5 value
        c5 = evaluate_erdos_solution(h, n_points, dx)
        
        # Update the best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Return the best solution found
    return best_h, best_c5, n_points
```
