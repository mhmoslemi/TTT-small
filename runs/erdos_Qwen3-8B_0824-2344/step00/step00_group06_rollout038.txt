```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import sys

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assume this is a numpy array of shape (95,)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points
    max_iterations = 10  # Limit iterations to stay within budget
    best_c5 = float('inf')
    best_h = initial_h_values[:n_points].copy()
    
    for _ in range(max_iterations):
        # Convert to CVXPY variables
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # Evaluate the overlap integrals for all shifts
        overlap_values = []
        for k in range(1, n_points):
            shifted_h = np.roll(h.value, k)
            overlap = np.dot(h.value, 1 - shifted_h) * (2.0 / n_points)
            overlap_values.append(overlap)
        
        # Objective is to minimize the maximum of the overlap values
        obj = cp.Minimize(cp.Max(overlap_values))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == 'optimal':
                current_h = h.value
                current_c5 = max(overlap_values)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            else:
                break
        except:
            break
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
```
