def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    correlation = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 samples for better resolution

    # Try different numbers of points
    for n_points in [100, 150, 200, 250, 300]:
        if time.time() - start_time > budget_s:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)
        
        # Initial guess (from provided initial solution scaled)
        h_initial = initial_h_values()[:n_points] if len(initial_h_values()) >= n_points else initial_h_values()
        h_initial = np.resize(h_initial, n_points)
        h_initial = np.clip(h_initial, 0, 1)
        
        # Objective function: maximize the overlap integral (we minimize the maximum overlap)
        def objective(h):
            h = np.clip(h, 0, 1)
            ones_minus_h = 1 - h
            correlation = np.correlate(h, ones_minus_h, mode="full")
            c5 = np.max(correlation) * dx
            return c5

        # Constraints: sum(h) * dx = 1
        def constraint_sum(h):
            return np.sum(h) - n_points / 2

        # Optimization with L-BFGS-B
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}]
        )

        if result.success:
            h = result.x
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
                print(f"Found better solution with n={n_points}, C5={current_c5:.6f}")
        else:
            print(f"Optimization failed for n={n_points}, status={result.message}")

    # Try fine-tuning the best solution found
    if best_h is not None:
        dx = 2.0 / len(best_h)
        n_points = len(best_h)
        def fine_tune_objective(h):
            h = np.clip(h, 0, 1)
            ones_minus_h = 1 - h
            correlation = np.correlate(h, ones_minus_h, mode="full")
            c5 = np.max(correlation) * dx
            return c5

        def fine_tune_constraint(h):
            return np.sum(h) - n_points / 2

        # Perform a small-scale optimization around the best solution
        result = minimize(
            fine_tune_objective,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': fine_tune_constraint}]
        )

        if result.success:
            best_h = result.x
            current_c5 = fine_tune_objective(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                print(f"Fine-tuned solution with n={n_points}, C5={current_c5:.6f}")

    # Return the best solution found
    return (best_h, best_c5, n_points)
