import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.concatenate([h[-1:], h[:-1]])
    overlap = np.correlate(h, 1 - shifted_h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    return [0.5] * 45  # Placeholder for initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 samples for balance between speed and accuracy
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values())
    initial_h = initial_h[:n_points]
    
    # Ensure the initial h satisfies the integral constraint
    initial_h = initial_h / (np.sum(initial_h) * dx) * (n_points / 2)
    
    def objective(h_values):
        h = np.array(h_values)
        shifted_h = np.concatenate([h[-1:], h[:-1]])
        overlap = np.correlate(h, 1 - shifted_h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def constraints(h_values):
        h = np.array(h_values)
        return [np.sum(h) - n_points / 2]  # Enforce integral constraint
    
    # Use constrained optimization
    h = cp.Variable(n_points)
    obj = cp.Minimize(objective(h.value))
    cons = [cp.sum(h) == n_points / 2]
    prob = cp.Problem(obj, cons)
    try:
        start_time = time.time()
        result = prob.solve(solver=cp.CVXOPT, verbose=False)
        if result is not None:
            best_h = h.value
            best_c5 = objective(best_h)
        else:
            best_h = initial_h
            best_c5 = evaluate_erdos_solution(initial_h)
    except:
        best_h = initial_h
        best_c5 = evaluate_erdos_solution(initial_h)
    
    # Simple local refinement using gradient descent
    def local_refinement(h_values, max_iter=100, learning_rate=0.01):
        h = np.array(h_values)
        for _ in range(max_iter):
            shifted_h = np.concatenate([h[-1:], h[:-1]])
            overlap = np.correlate(h, 1 - shifted_h, mode="full")
            max_overlap = np.max(overlap) * dx
            grad = np.zeros_like(h)
            for i in range(n_points):
                for k in range(n_points):
                    if k < n_points - 1:
                        grad[i] -= (overlap[k + i] * dx) * (1 - shifted_h[k + i]) * dx
                    if k > 0:
                        grad[i] += (overlap[k + i] * dx) * h[k + i] * dx
            grad = grad / np.max(np.abs(grad)) * learning_rate
            h = h - grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)  # Enforce integral constraint
        return h
    
    best_h = local_refinement(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure we do not exceed the budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # Additional refinement if time permits
    for _ in range(5):
        best_h = local_refinement(best_h)
        best_c5 = evaluate_erdos_solution(best_h)
        if elapsed_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, n_points)
