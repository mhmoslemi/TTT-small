import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import sys
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 100  # Start with 100 points for speed

    # Initial guess
    h_initial = initial_h_values(n_points)
    best_c5 = evaluate_erdos_solution(h_initial, n_points)

    # Try increasing n_points up to a maximum of 500
    max_n_points = 500
    for n_points in range(100, max_n_points + 1, 50):
        # Ensure the total integral over [0, 2] is 1
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)

        # Optimization: minimize the maximum overlap integral
        # This is a challenging problem, so we use a heuristic approach

        # Initial solution
        h_initial = h.copy()
        c5 = evaluate_erdos_solution(h_initial, n_points)
        best_c5 = min(best_c5, c5)
        best_h = h_initial.copy()

        # Refine with gradient descent
        learning_rate = 0.01
        for _ in range(100):
            # Compute current overlap integral
            corr = signal.correlate(h, 1 - h, mode="full")
            c5_val = np.max(corr) * dx

            # Compute gradient of c5 with respect to h
            # This is an approximation using finite differences
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = signal.correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5_val) / 1e-5

                h_minus = h.copy()
                h_minus[i] -= 1e-5
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                corr_minus = signal.correlate(h_minus, 1 - h_minus, mode="full")
                c5_minus = np.max(corr_minus) * dx
                grad[i] = (c5_val - c5_minus) / 1e-5

            # Update h
            h -= learning_rate * grad

            # Ensure h stays within [0, 1] and has unit integral
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

            # Evaluate
            c5 = evaluate_erdos_solution(h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

            # Check time budget
            if time.time() - start_time > budget_s:
                break

    # Return the best solution found
    return best_h, best_c5, n_points
