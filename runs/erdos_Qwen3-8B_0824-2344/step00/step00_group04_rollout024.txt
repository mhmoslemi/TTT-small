import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Initialize parameters
    n_points = 200  # Start with 200 points; can be adjusted if needed
    dx = 2.0 / n_points
    max_iterations = 50  # Limit iterations to stay within time budget

    # Initial guess from provided initial_h_values (rescaled to fit n_points)
    initial_h = initial_h_values
    initial_h = np.array(initial_h)
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)

    # Ensure the integral constraint is satisfied
    total_integral = np.sum(initial_h) * dx
    scaling_factor = 1.0 / total_integral
    initial_h *= scaling_factor
    initial_h = np.clip(initial_h, 0, 1)

    # Define the function to minimize C₅
    def evaluate_solution(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        total_integral = np.sum(h) * dx
        if np.abs(total_integral - 1) > 1e-6:
            return float('inf')
        # Compute 1 - h(x + k)
        corr = np.correlate(h, 1 - h, mode="full")
        # Compute the max overlap integral
        c5 = np.max(corr) * dx
        return c5

    # Define the optimization problem using CVXPY
    def optimize_h(h_initial):
        h = cp.Variable(n_points, nonneg=True, boolean=False)
        # Integral constraint
        integral = cp.sum(h) * dx
        constraints = [integral == 1.0]
        # Objective: minimize the maximum overlap integral
        # We approximate this by evaluating C₅ and minimizing it
        def objective_func(h_val):
            h_val = np.array(h_val)
            h_val = np.clip(h_val, 0, 1)
            corr = np.correlate(h_val, 1 - h_val, mode="full")
            return np.max(corr) * dx

        # Use a local optimization method to minimize the objective
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        return result.x

    # Run optimization
    best_h = initial_h.copy()
    best_c5 = evaluate_solution(best_h)

    for iteration in range(max_iterations):
        start_time = time.time()
        # Solve the optimization problem
        optimized_h = optimize_h(best_h)
        # Compute the current C₅
        current_c5 = evaluate_solution(optimized_h)
        # Update the best solution
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5
        # Check if time is up
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:  # Leave 1 second buffer
            break

    # Final evaluation for the best solution
    final_c5 = evaluate_solution(best_h)
    return best_h, final_c5, n_points
