import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
# evaluate_erdos_solution() is used to compute the overlap integral
# initial_h_values is a starting point with n_points = 45

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Function to evaluate C5 given h_values
    def evaluate_c5(h_values, n_points, dx):
        h = np.array(h_values)
        one_minus_h = 1 - h
        # Compute the full convolution
        corr = np.correlate(h, one_minus_h, mode="full")
        # Only consider the relevant part of the convolution
        # For x in [0,2], we need h(x) * (1 - h(x+k)) for k in [-1, 1]
        # So we take the middle part of the convolution result
        # The length of the full convolution is 2 * n_points - 1
        # We need to focus on the parts where k is in [-1, 1]
        start_idx = n_points - 2
        end_idx = n_points
        relevant_corr = corr[start_idx:end_idx]
        # Scale by dx
        c5 = np.max(relevant_corr) * dx
        return c5

    # Function to optimize h_values to minimize C5
    def optimize_h_values(h_initial, n_points, dx, max_iter=100):
        # Optimization variables
        h = np.copy(h_initial)
        best_h = np.copy(h)
        best_c5 = evaluate_c5(h, n_points, dx)

        for _ in range(max_iter):
            # Compute current C5
            current_c5 = evaluate_c5(h, n_points, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = np.copy(h)

            # Gradient descent step
            # Compute the gradient of C5 w.r.t. h_values
            # For simplicity, we use a numerical gradient
            grad = np.zeros_like(h)
            for i in range(n_points):
                # Perturb each h[i] slightly and compute the change in C5
                h_perturb = h.copy()
                h_perturb[i] += 1e-5
                c5_perturb = evaluate_c5(h_perturb, n_points, dx)
                grad[i] = (c5_perturb - current_c5) / 1e-5

            # Update h
            h -= 0.1 * grad
            h = np.clip(h, 0, 1)

        return best_h, best_c5

    # Initial configuration
    initial_h_values = np.array(initial_h_values)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Get the current C5 value
    current_c5 = evaluate_c5(initial_h_values, n_points, dx)

    # Try to optimize the initial configuration
    optimized_h, optimized_c5 = optimize_h_values(initial_h_values, n_points, dx)

    # Try a few other starting points (e.g., uniform, periodic)
    # Uniform starting point
    h_uniform = np.zeros(n_points)
    h_uniform[::2] = 1  # Alternate 1s and 0s
    h_uniform = np.clip(h_uniform, 0, 1)
    h_uniform = h_uniform / np.sum(h_uniform) * (n_points / 2)
    c5_uniform = evaluate_c5(h_uniform, n_points, dx)

    # Periodic starting point
    h_periodic = np.sin(np.linspace(0, 2 * np.pi, n_points)) * 0.5 + 0.5
    h_periodic = np.clip(h_periodic, 0, 1)
    h_periodic = h_periodic / np.sum(h_periodic) * (n_points / 2)
    c5_periodic = evaluate_c5(h_periodic, n_points, dx)

    # Compare all candidates
    candidates = [
        (initial_h_values, current_c5),
        (optimized_h, optimized_c5),
        (h_uniform, c5_uniform),
        (h_periodic, c5_periodic),
    ]

    # Find the best candidate
    best_h, best_c5 = min(candidates, key=lambda x: x[1])

    # Check if we can find a better solution with fewer points
    # Try with n_points = 100
    n_points = 100
    dx = 2.0 / n_points

    # Uniform starting point for n=100
    h_uniform = np.zeros(n_points)
    h_uniform[::2] = 1  # Alternate 1s and 0s
    h_uniform = np.clip(h_uniform, 0, 1)
    h_uniform = h_uniform / np.sum(h_uniform) * (n_points / 2)
    c5_uniform = evaluate_c5(h_uniform, n_points, dx)

    # Periodic starting point for n=100
    h_periodic = np.sin(np.linspace(0, 2 * np.pi, n_points)) * 0.5 + 0.5
    h_periodic = np.clip(h_periodic, 0, 1)
    h_periodic = h_periodic / np.sum(h_periodic) * (n_points / 2)
    c5_periodic = evaluate_c5(h_periodic, n_points, dx)

    # Evaluate the best candidate
    best_candidate = min(
        [(h_uniform, c5_uniform), (h_periodic, c5_periodic)],
        key=lambda x: x[1],
    )
    best_h, best_c5 = best_candidate

    # Return the best solution
    return best_h, best_c5, n_points
