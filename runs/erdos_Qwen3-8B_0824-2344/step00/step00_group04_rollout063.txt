import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    max_n_points = 256  # Start with 256 points for faster evaluation
    n_points = max_n_points
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy() if hasattr(initial_h_values, '__iter__') else np.random.rand(n_points) * 0.5
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def evaluate(h):
        return objective(h)
    
    def optimize_solution(h_initial, n_points, budget_s):
        start_time = time.time()
        best_h = h_initial.copy()
        best_c5 = evaluate(best_h)
        best_time = time.time() - start_time
        if best_time >= budget_s:
            return best_h, best_c5, n_points
        
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        for shift in range(1, n_points // 2):
            def shifted_overlap(h_val):
                h_val = h_val.value if h_val.is_scaled else h_val
                shifted = np.roll(h_val, shift)
                overlap = np.correlate(h_val, 1 - shifted, mode="full")
                c5 = np.max(overlap * dx)
                return c5
            
            objective_func = cp.Minimize(cp.norm(cp.multiply(cp.ones(n_points), shifted_overlap(h)), 1))
            prob = cp.Problem(objective_func, constraints)
            
            try:
                result = prob.solve(solver=cp.ECOS, max_iter=100, verbose=False)
                if result is not None:
                    h.value = h.value * (n_points / 2) / np.sum(h.value)
                    best_h = h.value.copy()
                    best_c5 = evaluate(best_h)
                    best_time = time.time() - start_time
                    if best_time >= budget_s:
                        return best_h, best_c5, n_points
            except:
                pass
        
        return best_h, best_c5, n_points
    
    best_h, best_c5, n_points = optimize_solution(h_initial, n_points, budget_s)
    return best_h, best_c5, n_points
