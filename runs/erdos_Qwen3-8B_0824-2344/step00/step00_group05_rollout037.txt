import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Define the function to compute the overlap integral
def compute_c5(h_values, n_points, dx):
    h = np.array(h_values)
    one_minus_h = 1.0 - h
    correlation = signal.correlate(h, one_minus_h, mode="full")
    c5 = np.max(correlation) * dx
    return c5

# Define the run function
def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Start from initial_h_values
    h_initial = initial_h_values
    h_initial = h_initial[:n_points]  # Ensure it fits the current n_points
    
    # Ensure the initial h satisfies the constraint sum(h) == n_points / 2
    sum_h = np.sum(h_initial)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
        h_initial = np.clip(h_initial, 0, 1)
    
    # Function to evaluate and return the c5 value
    def objective(x):
        h = x
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        c5 = compute_c5(h, n_points, dx)
        return c5
    
    # Function to return the current h_values, c5_bound, n_points
    def get_result(h):
        return h, compute_c5(h, n_points, dx), n_points
    
    # Optimization using differential evolution
    result = optimize.differential_evolution(
        objective,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        args=(),
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.8,
        tol=1e-6,
        maxiter=100,
        polish=True,
        disp=False,
        seed=seed
    )
    
    # Get the best solution found
    best_h = result.x
    best_c5 = result.fun
    
    # Check if we have time to try a higher resolution
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 5:
        n_points_new = min(200, n_points + 50)
        dx_new = 2.0 / n_points_new
        h_refined = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, n_points), best_h)
        h_refined = np.clip(h_refined, 0, 1)
        sum_h_refined = np.sum(h_refined)
        if np.abs(sum_h_refined - n_points_new / 2) > 1e-6:
            h_refined = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, len(h_initial)), h_initial)
            h_refined = np.clip(h_refined, 0, 1)
        
        def objective_refined(x):
            h = x
            sum_h = np.sum(h)
            if np.abs(sum_h - n_points_new / 2) > 1e-6:
                return np.inf
            c5 = compute_c5(h, n_points_new, dx_new)
            return c5
        
        result_refined = optimize.differential_evolution(
            objective_refined,
            bounds=[(0.0, 1.0) for _ in range(n_points_new)],
            args=(),
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-6,
            maxiter=100,
            polish=True,
            disp=False,
            seed=seed
        )
        
        best_h = result_refined.x
        best_c5 = result_refined.fun
    
    # Return the best solution
    return get_result(best_h)

# Define the function to evaluate the solution
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        return np.inf
    correlation = signal.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(correlation) * dx
    return c5
