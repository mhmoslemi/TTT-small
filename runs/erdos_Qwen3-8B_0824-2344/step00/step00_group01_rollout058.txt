import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial setup
    n_points = 100  # Start with 100 points; can be increased if needed
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Use initial construction as starting point
    h /= np.sum(h) * dx  # Normalize to meet the integral constraint
    h *= n_points / 2  # Ensure sum(h) == n_points / 2 exactly

    # Function to compute C₅
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        h_norm = h / np.sum(h) * dx  # Normalize to [0, 1]
        shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted[k:] = h[:-k]
        overlap = np.correlate(h_norm, 1 - h_norm, mode="full")
        return np.max(overlap) * dx

    # Optimization strategy: Gradient-based with random restarts
    def optimize_h(h_initial, max_iter=100, tol=1e-5):
        h = np.copy(h_initial)
        best_h = np.copy(h)
        best_c5 = compute_c5(h)
        best_c5_value = best_c5

        for _ in range(max_iter):
            # Random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
            h_candidate = np.clip(h + perturbation, 0, 1)
            h_candidate /= np.sum(h_candidate) * dx
            h_candidate *= n_points / 2

            # Evaluate
            c5 = compute_c5(h_candidate)
            if c5 < best_c5_value:
                best_h = np.copy(h_candidate)
                best_c5_value = c5

            # Early stopping
            if abs(best_c5_value - best_c5) < tol:
                break
            h = np.copy(best_h)
            best_c5 = best_c5_value

        return best_h, best_c5_value

    # Run optimization
    h_opt, c5_opt = optimize_h(h, max_iter=50)

    # Final evaluation
    c5_final = compute_c5(h_opt)
    n_points_final = len(h_opt)

    # Ensure we don't exceed the budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        print("Time budget exceeded, returning best found so far")
        return h_opt, c5_final, n_points_final

    return h_opt, c5_final, n_points_final
