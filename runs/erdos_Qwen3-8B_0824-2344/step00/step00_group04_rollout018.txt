import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros_like(h)
    for k in range(1, n_points):
        shifted_h = np.roll(h, k)
        overlap = np.dot(h, 1 - shifted_h)
        overlap *= dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

def initial_h_values():
    return np.array([0.5] * 42)

# Function to compute the overlap integral for a given h
def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h)
    max_overlap = 0.0
    for k in range(1, n_points):
        shifted_h = np.roll(h, k)
        overlap = np.dot(h, 1 - shifted_h)
        overlap *= dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

# Optimization function to minimize C5
def optimize_erdos_solution(initial_h, n_points, budget_s):
    h = cp.Variable(n_points)
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]
    objective = cp.Minimize(cp.Max([cp.sum(cp.multiply(h, 1 - cp.roll(h, k))) for k in range(1, n_points)]))
    
    start_time = time.time()
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, max_iter=1000, verbose=False)
    except:
        pass
    
    if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
        h_values = np.round(np.array(h.value), 4)
        c5 = compute_c5(h_values)
        return h_values, c5
    else:
        # Fallback to gradient descent
        h_values = np.array(initial_h)
        learning_rate = 0.1
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
            for k in range(1, n_points):
                shifted_h = np.roll(h_values, k)
                overlap = np.dot(h_values, 1 - shifted_h)
                grad = np.zeros_like(h_values)
                for i in range(n_points):
                    grad[i] = 1 - shifted_h[i] - np.dot(h_values, np.roll(1 - shifted_h, -i))
                h_values -= learning_rate * grad
            h_values = np.clip(h_values, 0, 1)
            h_values = (h_values * n_points / 2).astype(float)
            h_values = h_values / np.sum(h_values) * n_points / 2
            c5 = compute_c5(h_values)
        return h_values, c5

def run(seed=42, budget_s=60, **kwargs):
    initial_h = initial_h_values()
    n_points = len(initial_h)
    h_values = np.copy(initial_h)
    best_c5 = compute_c5(h_values)
    best_h = h_values
    
    # Try with different n_points
    for n_points in [42, 60, 80, 100, 120, 140, 160, 180, 200]:
        if time.time() > budget_s:
            break
        initial_h = np.random.rand(n_points)
        initial_h = (initial_h * n_points / 2).astype(float)
        initial_h = initial_h / np.sum(initial_h) * n_points / 2
        h_values, c5 = optimize_erdos_solution(initial_h, n_points, budget_s)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values
    
    return best_h, best_c5, len(best_h)
