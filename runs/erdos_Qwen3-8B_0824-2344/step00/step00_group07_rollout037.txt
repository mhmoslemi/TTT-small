```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy import signal
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Start with initial guess
    n_points = 80
    h = np.array(initial_h_values)  # Initial guess with n_points samples
    
    # Define dx
    dx = 2.0 / n_points
    
    # Objective function to minimize the maximum overlap integral
    def objective(h_values):
        h = h_values.copy()
        # Ensure h is within [0,1]
        h = np.clip(h, 0, 1)
        # Reshape for convolution (full mode)
        h_reshaped = h.reshape(-1, 1)
        one_minus_h = 1 - h
        # Compute convolution
        conv = signal.correlate(h_reshaped, one_minus_h, mode='full')
        # Compute max overlap integral
        max_overlap = np.max(conv) * dx
        return max_overlap

    # Optimization strategy: use gradient-free optimization with random restarts and perturbations
    def optimize_h(h_initial, n_points, max_iter=100, perturbation=0.05, dx=2.0 / n_points):
        h = h_initial.copy()
        best_h = h.copy()
        best_c5 = objective(h)
        for _ in range(max_iter):
            # Random perturbation
            h = np.clip(h + np.random.uniform(-perturbation, perturbation, size=n_points), 0, 1)
            c5 = objective(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        return best_h, best_c5

    # Run the optimization
    best_h, best_c5 = optimize_h(h, n_points, max_iter=50, perturbation=0.05)

    # Evaluate the result
    result = evaluate_erdos_solution(best_h)
    c5_bound = result['c5']
    n_points = len(best_h)

    # Print results
    elapsed_time = time.time() - start_time
    print(f"Best C5 bound: {c5_bound}, n_points: {n_points}, time: {elapsed_time:.2f}s")

    return best_h, c5_bound, n_points
```
