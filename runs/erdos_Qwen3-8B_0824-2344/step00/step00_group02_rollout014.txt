```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.roll(h, shift=1, axis=0)
    overlap = correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values(n_points):
    # Placeholder for initial h_values, n_points is the number of samples
    # This is a basic uniform distribution, may not be optimal
    return np.ones(n_points) * (n_points / 2) / 2

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points up to 500 for faster evaluation
    n_points_list = [50, 100, 150, 200, 250, 300, 350, 400, 450, 500]
    best_c5 = float('inf')
    best_h = np.array([])
    
    for n_points in n_points_list:
        # Generate a random initial solution
        h_initial = initial_h_values(n_points)
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = h_initial * (n_points / 2) / 2  # Ensure sum is n_points / 2

        # Define optimization variables
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # Objective function to minimize the maximum overlap
        # Compute the overlap integral for all k
        c5_values = []
        for k in range(1, n_points):
            # Shift h by k and compute the overlap
            h_shifted = np.roll(h.value, shift=k)
            overlap = np.dot(h.value, 1 - h_shifted)
            c5_values.append(overlap)
        
        # Use an approximation of the maximum overlap for optimization
        max_overlap = cp.Max(cp.sum(cp.multiply(np.array(c5_values), cp.ones_like(h)))) / n_points

        # Define the optimization problem
        problem = cp.Problem(cp.Minimize(max_overlap), constraints)
        
        # Solve the problem
        start_time = time.time()
        try:
            result = problem.solve(max_iters=100, verbose=False)
            if result is not None:
                h_values = h.value
                c5 = evaluate_erdos_solution(h_values)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
        except:
            pass
        
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, len(best_h)
```
