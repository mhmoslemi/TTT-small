import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(70) * 2 - 1  # Placeholder for initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster evaluation

    # Function to evaluate and return C5
    def evaluate(h):
        nonlocal best_c5
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Initial guess
    h0 = np.random.rand(n_points) * 2 - 1
    h0 = np.clip(h0, 0, 1)
    h0 = h0 * (n_points / 2) / np.sum(h0)  # Ensure integral is 1

    # Optimization: Use gradient descent with momentum
    learning_rate = 0.01
    momentum = 0.9
    max_iterations = 1000
    best_h = h0.copy()
    best_c5 = evaluate(best_h)

    for iteration in range(max_iterations):
        # Compute the overlap integral for current h
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if time.time() - start_time > budget_s:
            break

        # Compute gradient using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)

        # Update using gradient descent with momentum
        velocity = momentum * velocity + learning_rate * (-grad)
        best_h += velocity
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)

        # Early stopping if time is up
        if time.time() - start_time > budget_s:
            break

    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h)

    # Return the best result
    return final_h, final_c5, n_points
