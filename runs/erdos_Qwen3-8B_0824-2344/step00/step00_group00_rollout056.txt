import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial n_points as 45 or lower
    n_points = 45
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h = h / (np.sum(h) * dx) * (n_points / 2)  # Ensure sum(h) == n_points / 2
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Trial: reduce to n_points = 30
    n_points = 30
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h / (np.sum(h) * dx) * (n_points / 2)
    c5 = evaluate_erdos_solution(h, dx)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    
    # Trial: reduce to n_points = 20
    n_points = 20
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h / (np.sum(h) * dx) * (n_points / 2)
    c5 = evaluate_erdos_solution(h, dx)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    
    # Trial: reduce to n_points = 15
    n_points = 15
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h / (np.sum(h) * dx) * (n_points / 2)
    c5 = evaluate_erdos_solution(h, dx)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    
    # Trial: reduce to n_points = 10
    n_points = 10
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h / (np.sum(h) * dx) * (n_points / 2)
    c5 = evaluate_erdos_solution(h, dx)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
