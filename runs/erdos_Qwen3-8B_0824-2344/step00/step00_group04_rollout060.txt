def evaluate_erdos_solution(h_values: np.ndarray) -> float:
    """Evaluates the Erdős minimum overlap problem constant C₅."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = h_values.astype(float)
    correlate = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(correlate) * dx
    return c5

def run(seed=42, budget_s=60, **kwargs) -> Tuple[np.ndarray, float, int]:
    np.random.seed(seed)
    n_points = 100  # Use 100 points for a balance of performance and resolution
    dx = 2.0 / n_points
    initial_h = initial_h_values[:n_points].copy()
    initial_h = np.clip(initial_h, 0.0, 1.0)
    initial_h = initial_h * (n_points / 2)  # Ensure the integral is 1
    
    # Try to optimize using a direct optimization approach
    def objective(h: np.ndarray) -> float:
        h = np.clip(h, 0.0, 1.0)
        correlate = np.correlate(h, 1 - h, mode="full")
        return np.max(correlate) * dx

    # Define bounds and initial guess
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h.copy()

    # Use a L-BFGS-B optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        tol=1e-6,
        options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-5}
    )
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (initial_h, evaluate_erdos_solution(initial_h), n_points)
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try a convex optimization approach with CVXPy
    # Define variables
    h = cp.Variable(n_points, bounds=(0.0, 1.0))
    # Objective function: maximize the overlap integral
    correlate = np.correlate(h.value, 1 - h.value, mode="full")
    c5 = cp.Maximize(cp.Max(correlate * dx))
    
    # Add constraint for integral
    constraint = [cp.sum(h) == n_points / 2]
    
    prob = cp.Problem(c5, constraint)
    # Solve the problem
    start_time = time.time()
    result = prob.solve(solver=cp.SCS, eps=1e-6)
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (best_h, best_c5, n_points)
    
    if result == cp.OPTIMAL:
        best_h = h.value
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
