def compute_overlap_integral(h, dx):
    h_extended = np.concatenate([h, h, h, h])
    corr = np.correlate(h_extended, 1 - h, mode="full")
    overlap = np.max(corr) * dx
    return overlap

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    dx = 2.0 / n_points
    initial_h = initial_h_values[:n_points] / (n_points / 2)
    
    best_h = initial_h.copy()
    best_c5 = compute_overlap_integral(best_h, dx)
    best_h_values = best_h.copy()
    
    # Try increasing n_points up to 200 to find better solution
    for n_points in [100, 150, 200]:
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1
        
        # Use a genetic algorithm to find good solution
        def objective(h_flat):
            h = h_flat.reshape(-1)
            h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1
            return compute_overlap_integral(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h.flatten(),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-5}
        )
        
        h_result = result.x.reshape(n_points)
        h_result = h_result * (n_points / 2) / np.sum(h_result)
        current_c5 = compute_overlap_integral(h_result, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_result.copy()
            best_h_values = best_h.copy()
    
    return best_h_values, best_c5, n_points
