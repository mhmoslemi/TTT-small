```python
import numpy as np
import scipy.optimize as optimize
import scipy.interpolate as interp
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200  # Start with 200 points for balance between speed and accuracy
    dx = 2.0 / n_points

    # Initial guess: use the initial_h_values with some small random perturbations
    h_initial = np.array(initial_h_values[:n_points])
    h_initial = np.clip(h_initial + 0.01 * np.random.randn(n_points), 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0, 1)

    # Define the objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h_pad = np.concatenate([h, np.zeros(n_points)])
        one_minus_h = 1 - h_pad
        correlate = np.correlate(h_pad, one_minus_h, mode="full")
        c5 = np.max(correlate) * dx
        return c5

    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    x0 = h_initial

    # Use differential evolution for optimization
    result = optimize.differential_evolution(objective, bounds, strategy='best1bin', 
                                             popsize=15, mutation=(0.5, 1), recombination=0.8, 
                                             maxiter=100, tol=1e-6, seed=seed, 
                                             callback=lambda x: None)

    best_h = np.clip(result.x, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)

    # Evaluate the best solution
    c5_bound = evaluate_erdos_solution(best_h, n_points, dx)

    # Check if we can increase n_points for better accuracy
    if time.time() - start_time < budget_s - 2:
        n_points = 250
        dx = 2.0 / n_points
        best_h = np.linspace(0, 1, n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_h = np.clip(best_h, 0, 1)

        # Use a more efficient optimization method
        result = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                                   options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6})
        best_h = np.clip(result.x, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_h = np.clip(best_h, 0, 1)

        c5_bound = evaluate_erdos_solution(best_h, n_points, dx)

    return best_h, c5_bound, n_points
```
