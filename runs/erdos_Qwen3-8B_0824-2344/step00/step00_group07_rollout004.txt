import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Start with the initial construction
    n_points = 80
    h_values = initial_h_values.copy()
    dx = 2.0 / n_points
    # Ensure sum(h) * dx = 1
    h_values = h_values / np.sum(h_values) * (n_points / 2)
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        # Compute the full cross-correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Only consider the relevant part of the correlation
        # The relevant part corresponds to k from -len(h)+1 to len(h)-1
        # We need to find the maximum over the overlap integrals for each k
        # The maximum C5 is the maximum of corr[k] * dx where k is in the range
        # that corresponds to the valid overlaps
        # The overlap integrals are valid for k in -len(h)+1 to len(h)-1
        # For a function defined on [0,2], valid k are those for which x + k is still in [0,2]
        # So we only consider k such that the overlap is within [0,2]
        # Here, we just use all correlation values and take the max
        c5_bound = np.max(corr) * dx
        return c5_bound

    # Function to optimize the h values
    def optimize_h():
        nonlocal h_values, n_points, dx
        # Initial guess
        h = h_values.copy()
        # Create a vector of h values
        h_vec = cp.Variable(n_points)
        # Constraints
        constraints = [
            cp.sum(h_vec) == n_points / 2,  # Ensure the integral is 1
            0 <= h_vec,
            h_vec <= 1
        ]
        # Objective: minimize the maximum overlap integral
        # We need to compute the maximum overlap for each k and minimize it
        # For simplicity, we can use a convex optimization with a large lambda
        # This is a simplification; a better approach would involve a max-min formulation
        # For the purpose of this problem, we approximate it as minimizing the maximum
        # overlap integral using a convex relaxation
        # We approximate the maximum with a sum of constraints
        # For each k, the overlap integral is the correlation at k, scaled by dx
        # We can approximate the maximum as the largest value among all k
        # For a small n_points, we compute it directly
        # We use a simple approach here to avoid overcomplicating
        # This is a simplified optimization to get a good starting point
        # For the purpose of speed and simplicity
        
        # We use a gradient-based optimization to minimize the C5
        # Convert h to a numpy array
        h = h_values.copy()
        # Define the objective function
        def objective_func(h):
            return evaluate_overlap(h)
        
        # Use scipy's optimization
        result = optimize.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        h_values = result.x
        return result.fun

    # Run the optimization
    c5_bound = optimize_h()
    
    # Additional fine-tuning
    for _ in range(3):
        # Run a few more iterations of optimization
        result = optimize.minimize(
            evaluate_overlap,
            h_values,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        h_values = result.x
        c5_bound = result.fun

    # Ensure we don't exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (h_values, c5_bound, n_points)

    return (h_values, c5_bound, n_points)
