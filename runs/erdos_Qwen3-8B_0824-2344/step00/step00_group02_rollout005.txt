def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45  # Start with the initial number of points
    dx = 2.0 / n_points

    current_c5 = evaluate_erdos_solution(h)

    best_c5 = current_c5
    best_n_points = n_points
    
    # Try increasing the number of points to improve accuracy
    for n_points in [60, 80, 100, 120, 140, 160, 180, 200]:
        if time.time() - start_time > budget_s - 0.5:
            break
        
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # Initialize with a uniform distribution and adjust to meet the integral constraint
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        
        # Function to evaluate the C5 value
        def evaluate_c5(h):
            h = h.reshape(-1, 1)
            shifted_h = np.roll(h, shift=1, axis=0)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            return overlap
        
        # Try local optimization
        def objective(x):
            return evaluate_c5(x)
        
        # Use L-BFGS-B for optimization
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=1e-6,
            options={'maxiter': 100}
        )
        
        if result.success:
            h_opt = result.x
            c5 = evaluate_c5(h_opt)
            if c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5
                best_n_points = n_points
    
    # Final check with the best solution
    if best_c5 < current_c5:
        h = best_h
        n_points = best_n_points
        dx = 2.0 / n_points
        current_c5 = best_c5
    
    return (h.tolist(), current_c5, n_points)
