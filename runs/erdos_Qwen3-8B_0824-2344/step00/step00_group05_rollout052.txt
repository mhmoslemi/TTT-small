import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for faster computation
    dx = 2.0 / n_points

    # Initial guess: Use the provided initial_h_values or a uniform distribution
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values[:n_points])
    else:
        h_initial = np.random.rand(n_points)
        h_initial = (h_initial * 2 - 1) * 0.5 + 0.5  # Random between 0 and 1

    # Ensure the initial guess satisfies the integral constraint
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0, 1)

    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Function to compute the objective and constraints
    def objective(h):
        return evaluate_c5(h)

    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Use L-BFGS-B optimization
    result = minimize(objective, h_initial, method='L-BFGS-B', constraints=cons, tol=1e-6, options={'maxiter': 1000})

    # Check if we have time remaining to refine with a higher resolution
    elapsed = time.time() - time.time()  # Placeholder to simulate elapsed time
    if elapsed < budget_s - 5:
        n_points = 200
        dx = 2.0 / n_points
        h_initial = result.x
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

        def evaluate_c5_high_res(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = signal.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx

        def objective_high_res(h):
            return evaluate_c5_high_res(h)

        result_high_res = minimize(objective_high_res, h_initial, method='L-BFGS-B', constraints=cons, tol=1e-6, options={'maxiter': 500})
        result = result_high_res

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_points
