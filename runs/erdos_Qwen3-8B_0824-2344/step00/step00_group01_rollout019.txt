import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(47) * 2 - 1  # Placeholder for initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    max_n_points = 500  # Use up to 500 points for better resolution
    n_points = min(500, max_n_points)
    dx = 2.0 / n_points

    # Initial guess - start from the given initial_h_values and scale to [0, 2]
    h_initial = np.interp(initial_h_values, [-1, 1], [0, 2])
    h_initial = np.interp(h_initial, [0, 2], [0, 1])

    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h_initial)
    h_initial = h_initial * (n_points / 2) / sum_h

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        correlation = signal.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    # Optimization using gradient descent with momentum
    def objective(h):
        return compute_c5(h)

    h = h_initial.copy()
    learning_rate = 0.01
    momentum = 0.9
    decay = 0.999
    eps = 1e-8

    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    iterations = 0

    while time.time() - start_time < budget_s:
        # Compute gradient numerically
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * 1e-6)
        
        # Update with momentum and decay
        velocity = momentum * velocity + (grad / np.sqrt(np.mean(grad**2) + eps))
        h = h - learning_rate * velocity

        # Clip to [0, 1]
        h = np.clip(h, 0, 1)

        # Enforce the integral constraint
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        iterations += 1

        # Reduce learning rate every 100 iterations
        if iterations % 100 == 0:
            learning_rate *= 0.99

    return best_h, best_c5, n_points
