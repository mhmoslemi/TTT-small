import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define a function to compute the overlap integral
    def compute_c5(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        # Compute the convolution (correlation)
        correlation = np.correlate(h, 1 - h, mode="full")
        # Only the overlapping parts are relevant
        # The relevant part of the correlation is from index n_points - 1 to 2 * n_points - 2
        relevant_corr = correlation[n_points - 1:2 * n_points - 1]
        # Compute the maximum overlap integral
        max_overlap = np.max(relevant_corr) * dx
        return max_overlap
    
    # Define a function to compute the cost for optimization
    def objective_function(h_values):
        return compute_c5(h_values)
    
    # Define the optimization problem
    def optimize_h_values(n_points):
        # Initial guess from the provided initial_h_values
        initial_guess = np.array(initial_h_values)
        # Resample to the desired number of points
        if len(initial_guess) != n_points:
            # Interpolate to get a smooth initial guess
            initial_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_guess)), initial_guess)
        # Ensure the integral constraint is satisfied
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        # Objective: minimize the maximum overlap integral
        # We use a convex relaxation to minimize the maximum value
        # This is an approximation to the original non-convex problem
        # We can use a convex optimization approach to approximate it
        # Use a convex relaxation with a tolerance
        # For this we use a convex optimization problem where we minimize the maximum of the overlap integrals
        # This is a non-convex optimization, but we use a convex relaxation to estimate it
        # Instead of minimizing the max of the overlap, we minimize the max over the relevant correlations
        # This is a convex problem: minimize max over convex functions
        # We will use a convex optimization to find a lower bound
        # Here we use the average value of the overlap integrals as a surrogate
        # This is a heuristic, but it is computationally feasible
        # For the purpose of this problem, we assume this gives a good enough result
        objective = cp.Minimize(cp.norm(relevant_corr * dx, 1) / n_points)
        problem = cp.Problem(objective, constraints)
        start_time = time.time()
        result = problem.solve()
        elapsed_time = time.time() - start_time
        # Extract the optimized h values
        h_values = h.value
        # Compute the actual C5
        c5_bound = compute_c5(h_values)
        return h_values, c5_bound, n_points, elapsed_time
    
    # Find the optimal number of points within the budget
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    best_elapsed = 0
    # Start with a smaller number of points to save time
    # Try a few values of n_points and find the best one
    n_points_list = [100, 200, 300, 400, 500]
    for n_points in n_points_list:
        start_time = time.time()
        h_values, c5, _, elapsed = optimize_h_values(n_points)
        elapsed = time.time() - start_time
        if c5 < best_c5:
            best_h = h_values
            best_c5 = c5
            best_n_points = n_points
            best_elapsed = elapsed
        if best_elapsed >= budget_s:
            break
    # If we still have time, try to refine the solution with higher n_points
    if best_elapsed < budget_s:
        n_points = min(1000, int(budget_s * 10))
        start_time = time.time()
        h_values, c5, _, elapsed = optimize_h_values(n_points)
        elapsed = time.time() - start_time
        if c5 < best_c5:
            best_h = h_values
            best_c5 = c5
            best_n_points = n_points
            best_elapsed = elapsed
    
    # Return the best solution found
    return best_h, best_c5, best_n_points
