import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Initial solution (n=70 samples)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = initial_h * (n_points / 2) / sum_h

    def evaluate_c5(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate([h, h[:1]])  # pad for convolution
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def objective_function(h_values):
        return evaluate_c5(h_values)

    def compute_inequalities(h_values, max_k, dx):
        n = len(h_values)
        h = np.array(h_values, dtype=np.float64)
        constraints = []
        for k in range(1, max_k + 1):
            if k >= n:
                break
            for i in range(n):
                j = (i + k) % n
                if h[i] + (1 - h[j]) > 1 + 1e-6:
                    constraints.append(cp.constraints.Minimize(h[i] + (1 - h[j])))
        return constraints

    # Start with the initial solution
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    # Try a few optimization steps with gradient descent
    for _ in range(5):
        h = best_h.copy()
        h = np.clip(h + np.random.randn(len(h)) * 0.05, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Try a local optimization with scipy
    result = optimize.minimize(
        objective_function,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-5}
    )
    if result.success:
        best_h = result.x
        best_c5 = evaluate_c5(best_h)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)

    # Try a few more gradient steps
    for _ in range(5):
        h = best_h.copy()
        h = np.clip(h + np.random.randn(len(h)) * 0.05, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Final check for time
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 5:
        return (best_h.tolist(), best_c5, n_points)

    # Try a small convex optimization
    h = cp.Variable(n_points)
    objective = cp.Minimize(evaluate_c5(h))
    constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
    prob = cp.Problem(objective, constraints)
    try:
        result = prob.solve(solver=cp.SCS, max_iters=100, eps=1e-5)
        if result is not None and result.success:
            best_h = h.value
            best_c5 = evaluate_c5(best_h)
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
