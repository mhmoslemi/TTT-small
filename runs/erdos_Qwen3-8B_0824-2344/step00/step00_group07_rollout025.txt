```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 80  # Starting with 80 points, can be adjusted if needed
    dx = 2.0 / n_points

    # Initial solution
    h = np.array(initial_h_values)  # Ensure it's of length n_points
    h = h[:n_points]  # Trim if necessary
    h = h / np.sum(h) * (n_points / 2)  # Ensure sum(h) == n_points / 2
    h = np.clip(h, 0, 1)  # Ensure within [0, 1]
    c5 = evaluate_erdos_solution(h, dx)
    best_c5 = c5
    best_h = h.copy()

    # Try perturbing the initial solution
    for _ in range(10):  # Try a few perturbations
        if time.time() - start_time > budget_s - 1:
            break
        h_perturb = h.copy()
        idx = np.random.choice(n_points, 5, replace=False)
        h_perturb[idx] = np.random.uniform(0, 1, size=5)
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_h = np.sum(h_perturb)
        h_perturb = h_perturb * (n_points / 2) / sum_h
        c5_perturb = evaluate_erdos_solution(h_perturb, dx)
        if c5_perturb < best_c5:
            best_c5 = c5_perturb
            best_h = h_perturb.copy()

    # Try a few more perturbations with different strategies
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        h_perturb = h.copy()
        idx = np.random.choice(n_points, 5, replace=False)
        h_perturb[idx] = np.random.uniform(0, 1, size=5)
        h_perturb = np.clip(h_perturb, 0, 1)
        sum_h = np.sum(h_perturb)
        h_perturb = h_perturb * (n_points / 2) / sum_h
        c5_perturb = evaluate_erdos_solution(h_perturb, dx)
        if c5_perturb < best_c5:
            best_c5 = c5_perturb
            best_h = h_perturb.copy()

    # Try a local optimization
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5_val = evaluate_erdos_solution(h, dx)
        return c5_val

    bounds = [(0, 1) for _ in range(n_points)]
    result = optimize.minimize(
        objective,
        x0=best_h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 50, 'ftol': 1e-6}
    )

    if result.success:
        best_h = result.x
        best_c5 = result.fun

    # Final check
    h_final = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_final)
    h_final = h_final * (n_points / 2) / sum_h
    c5_final = evaluate_erdos_solution(h_final, dx)
    best_c5 = c5_final
    best_h = h_final

    return (best_h, best_c5, n_points)
```
