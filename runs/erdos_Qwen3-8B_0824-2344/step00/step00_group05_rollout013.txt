import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Choose a number of points (smaller is faster)
    n_points = 100  # Reduced from 70 to balance speed and accuracy
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        dx = 2.0 / n_points
        h = np.repeat(h_values, 2)  # Repeat for full convolution
        one_minus_h = 1 - h
        # Compute the full convolution
        conv = np.correlate(h, one_minus_h, mode="full")
        # Return the maximum value scaled by dx
        return np.max(conv) * dx
    
    # Function to compute the objective and constraints
    def objective_and_constraints(h_values):
        # Objective: minimize the maximum overlap integral
        c5 = compute_c5(h_values)
        return c5
    
    # Define the initial guess from provided initial_h_values
    h_initial = np.array(initial_h_values)
    h_initial = np.interp(h_initial, [0, len(h_initial)-1], [0, n_points-1])
    h_initial = h_initial[:n_points]
    
    # Ensure the initial guess meets the constraints
    sum_h_initial = np.sum(h_initial) * (2.0 / n_points)
    if abs(sum_h_initial - 1.0) > 1e-6:
        h_initial = np.array([1.0 / n_points] * n_points)
    
    # Optimization: Use a local optimizer to improve the initial guess
    # We use the SLSQP method with bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(
        objective_and_constraints,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Evaluate the solution
    print(f"Best C₅ value: {best_c5}")
    
    # Return the result
    return (best_h, best_c5, n_points)
