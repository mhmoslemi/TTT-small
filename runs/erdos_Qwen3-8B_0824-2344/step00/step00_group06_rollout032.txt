import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Define parameters
    n_points = 1000  # Optimal number of points for performance and accuracy
    dx = 2.0 / n_points
    max_iterations = 100  # Limit to avoid infinite loops
    tolerance = 1e-4  # Convergence tolerance
    max_time = budget_s  # Time budget in seconds

    # Initial guess
    h = initial_h_values.copy()
    h = np.resize(h, n_points)
    h = h / (np.sum(h) * dx)  # Normalize to satisfy the integral constraint

    # Ensure the sum constraint is exactly satisfied
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h

    # Function to compute the overlap integral
    def compute_c5(h):
        h_reshaped = np.resize(h, (n_points, 1))
        one_minus_h = 1 - h_reshaped
        conv = np.correlate(h_reshaped, one_minus_h, mode="full")
        c5_values = conv * dx
        return np.max(c5_values)

    # Function to compute the objective function for optimization
    def objective_function(h_flat):
        h = h_flat.reshape(n_points)
        # Enforce the integral constraint
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            return 1e10  # Penalize constraint violation
        return compute_c5(h)

    # Function to compute the derivative for optimization
    def objective_derivative(h_flat):
        h = h_flat.reshape(n_points)
        # Use finite differences to approximate the derivative
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (objective_function(h_plus) - objective_function(h_minus)) / (2 * eps)
        return grad.flatten()

    # Optimizer settings
    optimizer = optimize.minimize(
        fun=objective_function,
        x0=h.flatten(),
        method='L-BFGS-B',
        jac=objective_derivative,
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': max_iterations, 'ftol': tolerance, 'gtol': tolerance}
    )

    best_h = optimizer.x.reshape(n_points)
    best_c5 = compute_c5(best_h)

    # Ensure the sum constraint is exactly satisfied
    sum_h = np.sum(best_h)
    best_h = best_h * (n_points / 2) / sum_h

    # Final evaluation
    final_c5 = compute_c5(best_h)

    # Ensure we do not exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > max_time:
        return (h, final_c5, n_points)

    return (best_h, final_c5, n_points)

# Initialize start time
start_time = time.time()
