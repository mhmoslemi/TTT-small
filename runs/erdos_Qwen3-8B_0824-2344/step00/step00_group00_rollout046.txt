def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = h_values.copy()
    h = np.concatenate((h, h))  # Double the array to handle full convolution
    corr = np.correlate(h, (1 - h[:n_points]), mode="full")
    c5 = np.max(corr) * dx
    return c5

def optimize_h(n_points, initial_guess=None):
    h = np.zeros(n_points)
    if initial_guess is not None:
        h = initial_guess
    else:
        h = np.full(n_points, 1.0 / n_points)  # Uniform distribution
    h = np.clip(h, 0.0, 1.0)
    h = h * (n_points / 2)  # Ensure the integral is 1
    h = np.clip(h, 0.0, n_points)  # Clamp to [0, n_points]

    def objective(x):
        h_new = x.copy()
        h_new = np.clip(h_new, 0.0, n_points)
        return evaluate_erdos_solution(h_new)

    bounds = [(0.0, n_points) for _ in range(n_points)]
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, tol=1e-5, options={'maxiter': 100})
    return result.x

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100  # Start with a manageable number of points
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]
    initial_h = np.clip(initial_h, 0.0, n_points)
    initial_h = initial_h * (n_points / 2)
    initial_h = np.clip(initial_h, 0.0, n_points)

    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)

    # Try different numbers of points
    for n_points in [100, 200, 300, 400]:
        if time.time() - start_time > budget_s - 2:
            break
        # Optimize for this number of points
        h_opt = optimize_h(n_points, initial_guess=initial_h)
        h_opt = np.clip(h_opt, 0.0, n_points)
        h_opt = h_opt * (n_points / 2)  # Ensure the integral is 1
        h_opt = np.clip(h_opt, 0.0, n_points)
        c5 = evaluate_erdos_solution(h_opt)
        # Update best solution
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
        print(f"n_points={n_points}, C5={best_c5:.6f}, time={time.time()-start_time:.2f}s")

    print(f"Final C5: {best_c5:.6f}")
    return best_h.tolist(), best_c5, len(best_h)
