import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from the pre-imported variable
    initial_h_values = np.array(initial_h_values)
    n_points_initial = len(initial_h_values)
    n_points = max(100, n_points_initial)  # Start with 100 points
    
    best_c5 = float('inf')
    best_h_values = initial_h_values.copy()
    
    # Evaluate initial solution
    def evaluate_h(h):
        nonlocal best_c5
        h = h.reshape(-1)
        h = np.concatenate([h, h[:len(h)//2]])  # Mirror for full convolution
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / len(h))
        if c5 < best_c5:
            best_c5 = c5
            best_h_values = h[:len(h)//2]
        return c5
    
    # Start with the initial construction
    best_c5 = evaluate_h(initial_h_values)
    
    # Try a few points with different n_points
    for n_points in [100, 200, 300, 400]:
        if time.time() > time.time() - budget_s:
            break
        
        # Generate random initial h values
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = h.astype(float)
        
        # Use a local optimization method
        def objective(x):
            x = x.reshape(-1)
            x = np.concatenate([x, x[:len(x)//2]])
            corr = signal.correlate(x, 1 - x, mode="full")
            c5 = np.max(corr) * (2.0 / len(x))
            return c5
        
        # Optimization with L-BFGS-B
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        h_opt = result.x
        c5 = evaluate_h(h_opt)
    
    # Final evaluation
    best_h_values = best_h_values[:n_points_initial]
    best_c5 = evaluate_h(best_h_values)
    
    return (best_h_values, best_c5, len(best_h_values))
