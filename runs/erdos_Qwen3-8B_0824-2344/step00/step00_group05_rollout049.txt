import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for actual initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    max_attempts = 100  # Limit the number of attempts to stay within time
    attempt = 0

    while attempt < max_attempts and time.time() - start_time < budget_s:
        attempt += 1
        # Perturb the current solution
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_perturbed = np.clip(best_h + perturbation, 0, 1)
        # Ensure the integral constraint is satisfied
        integral = np.sum(h_perturbed) * (2.0 / n_points)
        if np.abs(integral - 1.0) > 1e-5:
            # Adjust the perturbed values to meet the integral constraint
            delta = (1.0 - integral) / n_points
            h_perturbed += delta
        # Evaluate the C5 value
        c5 = evaluate_erdos_solution(h_perturbed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()
        # Optional: try a different number of points if needed
        # We can consider reducing the number of points to speed up the search
        # For simplicity, we stick with the initial number of points for now
        # Consider exploring a grid of n_points values for further optimization

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
