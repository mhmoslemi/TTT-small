```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100  # Start with 100 samples
    max_iter = 100  # Maximum number of iterations
    dx = 2.0 / n_points
    
    # Initial guess from provided initial_h_values (resized to n_points)
    if len(initial_h_values) != n_points:
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    h = np.copy(initial_h_values)
    h = h / np.sum(h) * (n_points / 2)  # Ensure the integral is 1
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Define the optimization variables
        x = cp.Variable(n_points, boolean=True)
        # Ensure the sum constraint is satisfied
        constraint_sum = cp.sum(x) == (n_points / 2)
        
        # Define the objective function to minimize the max overlap
        # To avoid CVXPY's difficulty with max, we reformulate using auxiliary variables
        # Create auxiliary variables for each shift
        shifts = np.arange(1, n_points + 1)  # Shifts from 1 to n_points
        aux_vars = [cp.Variable(n_points) for _ in range(len(shifts))]
        
        # Objective: minimize the maximum of the overlap integrals
        objective = cp.Minimize(cp.Max([cp.sum(aux_vars[i]) * dx for i in range(len(shifts))]))
        
        # Constraints
        constraints = [constraint_sum]
        for i, shift in enumerate(shifts):
            # Ensure the overlap integral is correctly represented
            for j in range(n_points):
                if shift + j < n_points:
                    overlap_j = x[j] * (1 - x[j + shift])
                else:
                    overlap_j = x[j] * (1 - x[j - n_points + shift])
                constraints.append(aux_vars[i][j] >= overlap_j)
                constraints.append(aux_vars[i][j] <= overlap_j)
        
        # Solve the optimization problem
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.ECOS, eps=1e-6)
        except:
            # Fallback to a simpler approach if ECOS fails
            pass
        
        # Extract the optimized binary values
        optimized_x = np.round(x.value).astype(int)
        
        # Convert binary values to h values (0 or 1)
        h = optimized_x
        
        # Check the constraints again and ensure the sum is correct
        if np.sum(h) != (n_points / 2):
            # Adjust to ensure the sum constraint
            diff = (n_points / 2) - np.sum(h)
            if diff > 0:
                # Flip zeros to ones
                zeros = np.where(h == 0)[0]
                if len(zeros) > 0:
                    h[zeros[0]] = 1
                    diff -= 1
            elif diff < 0:
                # Flip ones to zeros
                ones = np.where(h == 1)[0]
                if len(ones) > 0:
                    h[ones[0]] = 0
                    diff += 1
        
        # Calculate the current C5
        current_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
```
