```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster initial evaluation
    
    # Initial setup based on initial_h_values
    initial_h = initial_h_values
    initial_h = np.array(initial_h)
    initial_h = np.interp(initial_h, (initial_h.min(), initial_h.max()), (0, 1))
    initial_h = np.interp(initial_h, (initial_h.min(), initial_h.max()), (0, 1))
    initial_h = initial_h[:n_points]
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def evaluate_h(h_values):
        h = np.array(h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) > 0 else h
        dx = 2.0 / n_points
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Initial evaluation
    current_c5 = evaluate_h(initial_h)
    best_c5 = current_c5
    best_h = initial_h.copy()
    
    # Optimization loop
    for _ in range(30):  # Limit iterations to fit within time budget
        if time.time() - start_time > budget_s - 5:
            break
        
        # Use gradient descent with random perturbations
        h = best_h.copy()
        h += np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) > 0 else h
        c5 = evaluate_h(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Try increasing n_points if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        c5 = evaluate_h(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
```
