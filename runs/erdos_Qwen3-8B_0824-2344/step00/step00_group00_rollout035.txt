import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):

    def objective(x):
        h = x.reshape((n_points,))
        # Normalize h to ensure ∫h(x)dx = 1
        h_norm = h / (np.sum(h) * dx)
        # Compute the convolution for all shifts
        conv = np.correlate(h_norm, 1 - h_norm, mode="full")
        # Multiply by dx and take the maximum
        c5 = np.max(conv * dx)
        return c5
    
    # Constraints: 0 <= h[i] <= 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    max_iter = 100
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Use L-BFGS-B with initial guess
        result = opt.minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            bounds=bounds,
