def run(seed=42, budget_s=60, **kwargs):

    best_c5 = np.inf

    max_n_points = 100  # Maximum number of samples to try

    for n in range(n_points, max_n_points + 1):
        dx = 2.0 / n
        # Initialize h with the initial values, scaled to fit the current n
        if n == n_points:
            h = initial_h_values.copy()
            h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h)), h)
        else:
            h = np.random.rand(n)  # Random initialization for new n
            # Ensure the integral constraint is satisfied
            total = np.sum(h) * dx
            scale = 1.0 / total
            h *= scale
        
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        
        # Function to compute C₅
        def compute_c5(h):
            h = np.clip(h, 0, 1)
            # Compute 1 - h
            one_minus_h = 1 - h
            # Compute cross-correlation (mode="full")
            corr = np.correlate(h, one_minus_h, mode="full")
            # Integrate over [0, 2] (dx is 2.0 / n)
            c5 = np.max(corr * dx)
            return c5
        
        # Use a local optimization to improve h
        def objective(x):
            h = x.reshape((-1, 1))
            h = np.clip(h, 0, 1)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5
        
        # Define variables and constraints
        x = cp.Variable((n, 1))
        constraints = [
            cp.sum(x) == n / 2,  # Integral constraint
            cp.Minimize(cp.max(cp.multiply(np.correlate(x, 1 - x, mode="full"), dx)))
        ]
        prob = cp.Problem(cp.Minimize(cp.max(cp.multiply(np.correlate(x, 1 - x, mode="full"), dx))), constraints)
        
        # Solve the problem with a time limit
        prob.solve(solver=cp.ECOS, verbose=False, time_limit=budget_s - (time.time() - start_time))
        
        if prob.status == "optimal" or prob.status == "infeasible":
            h_opt = np.round(x.value).flatten()

        # Check time budget

    # Return best solution found
    return (best_h, best_c5, len(best_h))
