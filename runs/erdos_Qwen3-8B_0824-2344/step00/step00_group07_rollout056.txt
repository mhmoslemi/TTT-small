```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    max_n_points = 500
    n_points = 500  # Start with 500 points for better resolution
    dx = 2.0 / n_points

    # Initial guess
    h = np.array(initial_h_values)  # Start with the initial construction
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    h = np.clip(h, 0, 1)
    
    # Normalize to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    
    elapsed_time = 0
    
    # Optimization loop
    for _ in range(100):  # Run multiple iterations
        start_time = time.time()
        
        # Reshape to a continuous function for numerical integration
        x = np.linspace(0, 2, n_points)
        h = np.clip(h, 0, 1)  # Ensure h stays within [0,1]
        
        # Define 1 - h(x)
        one_minus_h = 1 - h
        
        # Compute correlations (overlap integrals)
        corr = np.correlate(h, one_minus_h, mode="full")
        overlaps = corr * dx
        
        # Find the maximum overlap integral
        current_c5 = np.max(overlaps)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # If time is up, return the best solution
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s - 1:
            break
        
        # Use optimization to improve h
        # Define variables for constrained optimization
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var <= 1,
            h_var >= 0
        ]
        
        # Objective: minimize the maximum overlap integral
        # This is a difficult objective to optimize directly, so we use a heuristic
        
        # Heuristic: minimize the maximum overlap integral by balancing the overlap integrals
        # For simplicity, we minimize the sum of the overlaps (this is not optimal but provides improvement)
        overlaps = np.correlate(h, one_minus_h, mode="full") * dx
        max_overlap = np.max(overlaps)
        objective = cp.Minimize(max_overlap)
        
        # Solve the problem
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver='ECOS', eps=1e-4)
            h = h_var.value
        except:
            pass
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Return the best solution
    return (best_h, best_c5, n_points)
```
