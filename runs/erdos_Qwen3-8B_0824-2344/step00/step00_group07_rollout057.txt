import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.zeros_like(h)
    for k in range(1, n_points):
        h_shifted[k:] = h[:-k]
    overlap = np.correlate(h, 1 - h_shifted, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    # A simple initial guess based on a triangular function
    n_points = 80
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 1:
            h[i] = x
        elif x > 1:
            h[i] = 2 - x
        else:
            h[i] = 1
    return h

# Strategy: We will use a combination of local optimization and global search
# to find a h_values array that minimizes the maximum overlap integral C5.
# We start with the initial_h_values and refine it using scipy.optimize functions,
# possibly switching to a more global optimization technique if needed.

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100  # Use 100 points to balance speed and precision
    dx = 2.0 / n_points
    
    # Initial guess
    h_initial = initial_h_values()
    h_initial = h_initial[:n_points]  # Ensure it's of correct length
    
    # Objective function: maximize the overlap integral
    # We need to minimize this maximum value, so we use a negative objective
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        # Compute the overlap integral
        h_shifted = np.zeros_like(h)
        for k in range(1, n_points):
            h_shifted[k:] = h[:-k]
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraints: sum(h) = n_points / 2
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Use a local optimization method to refine the initial guess
    h_flat = h_initial.copy()
    result = opt.minimize(
        fun=objective,
        x0=h_flat,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100}
    )
    
    # If we have time left, perform a more global search
    if time.time() - start_time < budget_s - 10:
        # Try a more global optimization approach
        h_flat_global = result.x
        result_global = opt.minimize(
            fun=objective,
            x0=h_flat_global,
            method='differential_evolution',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 30}
        )
        h_flat = result_global.x
        c5_bound = result_global.fun
    else:
        h_flat = result.x
        c5_bound = result.fun
    
    h_values = h_flat
    return h_values, c5_bound, n_points
