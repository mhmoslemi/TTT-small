import numpy as np
import scipy.optimize as opt
import math
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Start time tracking
    start_time = time.time()
    
    # Parameters
    max_n_points = 500  # Maximum number of points to consider
    n_points = 100  # Start with 100 points, can increase if time allows
    dx = 2.0 / n_points
    target_c5 = 0.380876
    
    # Initial guess
    h_initial = initial_h_values.copy()
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_values = h_initial.copy()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = np.pad(h, (0, 1), mode='constant')  # Pad for full convolution
        conv = np.convolve(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Function to compute constraints
    def constraints(h):
        return np.sum(h) - n_points / 2

    # Optimization setup
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints}
    
    # Optimization
    result = opt.minimize(
        fun=objective,
        x0=h_values,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Check if optimization succeeded
    if result.success:
        h_values = result.x
    else:
        # If optimization fails, try with more points
        n_points = min(max_n_points, n_points * 2)
        dx = 2.0 / n_points
        h_values = np.random.rand(n_points)
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    # Ensure the integral constraint is satisfied
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    # Evaluate the final C5
    c5_bound = compute_c5(h_values)
    
    # Output the best solution found
    return h_values, c5_bound, n_points
