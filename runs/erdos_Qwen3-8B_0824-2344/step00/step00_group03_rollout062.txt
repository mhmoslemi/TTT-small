import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

    # Start time
    start_time = time.time()
    
    # Initial setup
    n_points = 70  # Start with the initial number of points
    h = initial_h_values.copy()
    
    # Function to evaluate C5 for a given h_values
    def evaluate_C5(h_values):
        # Ensure h_values is normalized
        h = h_values / (n_points / 2)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Take the maximum value
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Function to compute the objective for optimization
    def objective_func(h_values):
        return evaluate_C5(h_values)
    
    # Function to compute the constraints
    def constraint_func(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Initial evaluation
    initial_c5 = evaluate_C5(h)
    best_c5 = initial_c5
    best_h = h.copy()
    
    # Attempt to refine the initial solution
    # Use CVXOPT to optimize the function with constraints
    # Convert h to a CVXOPT variable
    h_cvx = cp.Variable(n_points)
    # Objective function
    obj = cp.Minimize(objective_func(h_cvx))
    # Constraint: sum(h) = n_points / 2
    constraint = cp.Constraint(h_cvx, cp.EqualTo(n_points / 2))
    # Solve the problem
    prob = cp.Problem(obj, [constraint])
    try:
        result = prob.solve(solver=cp.CVXOPT, verbose=False)
        if result is not None:
            h_opt = h_cvx.value
            c5_opt = evaluate_C5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Check if we can try a smaller number of points
    # Try n_points = 60
    n_points = 60
    h = np.zeros(n_points)

    h[n_points // 3: 2 * n_points // 3] = 0.0
    h[2 * n_points // 3:] = 1.0
    h = h / (n_points / 2)
    c5 = evaluate_C5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    for _ in range(10):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        h = h / (n_points / 2)
        c5 = evaluate_C5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s:

    h = best_h
    learning_rate = 0.01
    for _ in range(100):
        # Compute the gradient (approximate using finite differences)
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / (n_points / 2)
            c5_plus = evaluate_C5(h_plus)
            
            h_minus = h.copy()

            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus / (n_points / 2)
            c5_minus = evaluate_C5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
        
        # Update the h_values
        h = h - learning_rate * grad

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final evaluation
    return (best_h, best_c5, n_points)
