import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 80  # Start with the initial n_points
    dx = 2.0 / n_points
    
    # Initial guess
    h = np.array(initial_h_values)  # Use the initial guess provided
    
    # Ensure sum(h) * dx == 1
    sum_h = np.sum(h)
    if not np.isclose(sum_h * dx, 1):
        h = h / sum_h * (1 / dx)
    
    # Function to calculate the overlap integral C_5
    def calculate_c5(h_values):
        h = h_values.copy()
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral using numpy correlate
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr)
        # Multiply by dx to get the integral value
        return max_overlap * dx
    
    # Function to optimize h to minimize C_5
    def optimize_h(h_initial, budget_s):
        best_h = h_initial.copy()
        best_c5 = calculate_c5(best_h)
        best_time = 0
        
        # Define optimization variables
        x = cp.Variable(h_initial.shape)
        constraints = [0 <= x, x <= 1]
        constraints.append(cp.sum(x) == n_points / 2)
        
        # Try optimization with different solvers
        solvers = ['ECOS', 'CVXOPT', 'SCIP', 'SLSQP', 'COBYLA']
        for solver in solvers:
            try:
                # Define objective to minimize the max overlap (approximate with a quadratic)
                # Use a conservative upper bound for max(corr) using quadratic approximation
                corr = np.correlate(x.value, 1 - x.value, mode='full')
                max_corr = np.max(corr)
                # Since we can't directly minimize max_corr in CVXPy, use a heuristic
                # Approximate by minimizing the sum of squared differences from the mean
                # This is not exact but can guide the optimization
                obj = cp.Minimize(cp.sum(cp.square(x - np.mean(x))))
                prob = cp.Problem(obj, constraints)
                start_time = time.time()
                prob.solve(solver=solver, verbose=False)
                elapsed_time = time.time() - start_time
                best_time = max(best_time, elapsed_time)
                
                if prob.status == 'optimal':
                    h_candidate = np.clip(x.value, 0, 1)
                    c5_candidate = calculate_c5(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
            except:
                continue
            
            if best_time >= budget_s:
                break
        
        # Hybrid optimization with gradient descent
        # Use the best h from optimization as the starting point
        h_optimized = best_h.copy()
        learning_rate = 0.01
        for _ in range(1000):
            # Compute correlations
            corr = np.correlate(h_optimized, 1 - h_optimized, mode='full')
            max_corr = np.max(corr)
            # Gradients (approximate using finite differences)
            grads = np.zeros_like(h_optimized)
            for i in range(len(h_optimized)):
                h_plus = h_optimized.copy()
                h_plus[i] += 1e-5
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                grads[i] = np.max(corr_plus) - max_corr
            
            # Update h_optimized
            h_optimized -= learning_rate * grads
            h_optimized = np.clip(h_optimized, 0, 1)
            # Ensure sum(h) * dx == 1
            h_optimized = h_optimized / np.sum(h_optimized) * (1 / dx)
            
            # Stop if we run out of time
            if time.time() - start_time >= budget_s:
                break
        
        c5_final = calculate_c5(h_optimized)
        return h_optimized, c5_final
    
    # Run the optimization
    best_h, best_c5 = optimize_h(h, budget_s)
    
    # Ensure best_h is in [0, 1]
    best_h = np.clip(best_h, 0, 1)
    
    # Ensure the sum constraint is satisfied
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h * dx, 1):
        best_h = best_h / sum_h * (1 / dx)
    
    # Final evaluation
    final_c5 = calculate_c5(best_h)
    
    return best_h, final_c5, n_points
