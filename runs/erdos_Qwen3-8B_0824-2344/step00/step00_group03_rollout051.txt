import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Start timing
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Initial guess: use the provided initial_h_values
    initial_h = initial_h_values
    n_points_initial = len(initial_h)
    dx = 2.0 / n_points_initial
    h_values = initial_h.copy()

    # Function to compute C5 from h_values
    def compute_c5(h):
        h = np.array(h)
        n_points = len(h)
        dx = 2.0 / n_points
        shifted = np.zeros(2 * n_points - 1)
        for i in range(n_points):
            shifted[i:i + n_points] += h[i] * (1 - h[i + n_points:] if i + n_points < len(shifted) else 0)
        overlap_integral = np.sum(shifted) * dx
        return overlap_integral

    # Function to evaluate the solution
    def evaluate_solution(h):
        nonlocal best_c5, best_h, best_n_points
        n_points = len(h)
        dx = 2.0 / n_points
        if np.sum(h) * dx < 1 - 1e-6 or np.sum(h) * dx > 1 + 1e-6:
            return float('inf')
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n_points
        return c5

    # Function to run a single optimization step
    def optimize_step(h, n_points, max_iter=100, tol=1e-4):
        # Define variables
        x = cp.Variable(n_points)
        constraints = [
            0 <= x,
            x <= 1,
            cp.sum(x) == n_points / 2
        ]
        # Objective function: minimize the maximum overlap integral
        # We use a convex relaxation by minimizing the sum of absolute values
        # This is an approximation, but helps to guide the optimization
        obj = cp.Minimize(cp.sum(cp.abs(x - x[:, np.newaxis] * (1 - x[:, np.newaxis]))))
        problem = cp.Problem(obj, constraints)
        result = problem.solve(solver=cp.SCS, max_iter=max_iter, eps=tol, verbose=False)
        if result is not None:
            return x.value
        return h

    # Optimization loop with decreasing n_points
    n_points_list = [n_points_initial // 2, n_points_initial // 3, n_points_initial // 4]
    for n_points in n_points_list:
        if n_points < 2:
            continue
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # Start with initial guess
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), initial_h)
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        if np.sum(h) * dx < 1 - 1e-6 or np.sum(h) * dx > 1 + 1e-6:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx)) * 1
        # Run optimization
        for _ in range(10):
            h = optimize_step(h, n_points)
            if time.time() - start_time > budget_s - 1:
                break
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n_points = n_points
        if time.time() - start_time > budget_s - 1:
            break

    # Final evaluation
    if best_h is None:
        best_h = initial_h.copy()
        best_c5 = compute_c5(best_h)
        best_n_points = len(best_h)

    # Ensure the solution satisfies the constraints
    if np.sum(best_h) * (2.0 / best_n_points) < 1 - 1e-6 or np.sum(best_h) * (2.0 / best_n_points) > 1 + 1e-6:
        best_h = np.random.rand(best_n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (1 / (np.sum(best_h) * (2.0 / best_n_points))) * 1
        best_c5 = compute_c5(best_h)

    # Final check and return
    best_c5 = min(best_c5, evaluate_erdos_solution(best_h))
    return (best_h, best_c5, best_n_points)
