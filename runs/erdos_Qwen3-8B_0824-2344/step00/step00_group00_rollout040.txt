def run(seed=42, budget_s=60, **kwargs):

    start_time = time.time()

    sum_h = np.sum(h_initial)

        # Adjust the initial guess to satisfy the constraint
        excess = sum_h - n_points / 2
        if excess > 0:
            # Reduce the sum by adjusting some values
            indices = np.argsort(h_initial)
            for i in indices:
                if h_initial[i] > 0:
                    h_initial[i] -= excess
                    excess = 0
                    break
        else:
            # Increase the sum by adjusting some values
            indices = np.argsort(h_initial)
            for i in indices[::-1]:
                if h_initial[i] < 1:
                    h_initial[i] += -excess
                    excess = 0
                    break

    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)  # Ensure values are within [0, 1]
        dx = 2.0 / len(h)
        # Compute (1 - h[x + k]) for all k such that x + k is in [0, 2]
        # Use convolution to compute the overlap
        shifted_h = np.zeros_like(h)
        for i in range(len(h)):
            shifted_h[i] = 1 - h[i]
        correlation = np.correlate(h, shifted_h, mode="full")
        # Only consider valid shifts where the overlap is meaningful
        # We consider shifts where x + k is within [0, 2]
        # The maximum overlap corresponds to the maximum value in the correlation
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    # Objective function for optimization: minimize the overlap
    def objective_function(h_values):
        return evaluate_overlap(h_values)
    
    # Optimization: use simulated annealing to minimize the overlap
    result = optimize.basinhopping(objective_function, h_initial,
