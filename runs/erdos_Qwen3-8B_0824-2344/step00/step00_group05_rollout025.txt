import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Try different n_points (smaller n_points for faster computation)
    n_points_list = [100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Initial guess from provided initial_h_values or random
        if n_points == 100:
            h = initial_h_values[:100]
        elif n_points == 150:
            h = initial_h_values[:150]
        elif n_points == 200:
            h = initial_h_values[:200]
        else:
            h = np.random.rand(n_points)
        
        # Ensure sum(h) * dx = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Function to compute C5 for a given h
        def compute_c5(h):
            h = np.array(h)
            h = np.concatenate((h, h[:1]))  # Extend to handle circular convolution
            corr = np.correlate(h, 1 - h, mode="full")
            overlap = np.max(corr) * dx
            return overlap
        
        # Optimization: minimize C5 using gradient-based optimization
        def objective(x):
            h = x
            h = np.concatenate((h, h[:1]))  # Extend to handle circular convolution
            corr = np.correlate(h, 1 - h, mode="full")
            overlap = np.max(corr) * dx
            return overlap
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use L-BFGS-B for optimization
        result = opt.minimize(
            objective,
            h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'disp': False, 'maxiter': 100}
        )
        
        # Check if result is successful
        if result.success:
            h_opt = result.x
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n_points = n_points
        
        # Early stopping if budget exhausted
        if time.time() - start_time > budget_s:
            break
    
    # Return best solution found
    return (best_h, best_c5, best_n_points)
