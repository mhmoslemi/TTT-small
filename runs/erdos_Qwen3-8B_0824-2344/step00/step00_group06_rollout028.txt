import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        dx = 2.0 / n_points
        # Compute the cross-correlation with (1 - h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    current_c5 = evaluate_overlap(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Optimization settings
    max_iterations = 100
    for _ in range(max_iterations):
        # Convert h to a variable for optimization
        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        constraints = [cp.sum(h_var) == n_points / 2]
        
        # Objective is to minimize the maximum overlap
        overlap = np.correlate(h_var, 1 - h_var, mode="full")
        max_overlap = cp.Max(overlap * 2.0 / n_points)
        objective = cp.Minimize(max_overlap)
        
        # Solve the optimization problem
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status != 'OPTIMAL':
                break
            h_new = h_var.value
            new_c5 = evaluate_overlap(h_new)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
        except:
            # If solver fails, use a local optimization step
            h_new = h.copy()
            dx = 2.0 / n_points
            # Random perturbation
            perturbation = np.random.normal(0, 0.05, n_points)
            h_new += perturbation
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_overlap(h_new)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h)
    return (best_h.tolist(), final_c5, n_points)
