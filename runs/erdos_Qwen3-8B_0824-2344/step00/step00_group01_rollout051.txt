import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    
    # We'll try a few different grid sizes starting from the initial one
    grid_sizes = [n_points, n_points * 2, n_points * 4]
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for n in grid_sizes:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create a grid of points
        dx = 2.0 / n
        x = np.linspace(0, 2, n)
        
        # Initial guess from the current best
        if n > len(best_h):
            # Interpolate to a finer grid
            h_fine = np.interp(x, np.linspace(0, 2, len(best_h)), best_h)
        else:
            h_fine = best_h.copy()
        
        # Bounds for h
        h_bounds = [(0, 1) for _ in range(n)]
        
        # Objective: minimize the maximum overlap
        # Define variables for the function h
        h_var = cp.Variable(n)
        
        # Compute the correlation for all k
        # We'll compute for k from -n+1 to n-1
        max_overlap = 0.0
        
        # Since we're using a fixed dx, we'll compute the integral as sum(h(x) * (1 - h(x + k))) * dx
        # To avoid negative indices, we'll wrap the x values using modulo
        for k in range(1, n):
            h_shifted = np.roll(h_var, k)
            overlap = cp.sum(h_var * (1 - h_shifted))
            max_overlap = cp.Max(max_overlap, overlap)
        
        # Minimize the maximum overlap
        prob = cp.Problem(cp.Minimize(max_overlap))
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status == 'OPTIMAL':
            h_new = h_var.value
            # Ensure the integral constraint is satisfied (sum(h) * dx = 1)
            integral = np.sum(h_new) * dx
            if abs(integral - 1.0) > 1e-5:
                h_new = h_new * (1.0 / integral)
            
            # Calculate the C5 value
            c5 = max(np.correlate(h_new, 1 - h_new, mode="full") * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        
        # Optional: try to improve with local search
        # We'll use a simple gradient descent-style approach
        for _ in range(10):
            step_size = 0.01
            h_temp = h_new.copy()
            for i in range(n):
                # Compute the overlap for all k
                temp_h = h_temp.copy()
                temp_h[i] += step_size
                overlap = np.correlate(temp_h, 1 - temp_h, mode="full") * dx
                max_overlap_temp = np.max(overlap)
                
                temp_h = h_temp.copy()
                temp_h[i] -= step_size
                overlap = np.correlate(temp_h, 1 - temp_h, mode="full") * dx
                max_overlap_temp = np.min([max_overlap_temp, np.max(overlap)])
                
                if max_overlap_temp < best_c5:
                    h_new = temp_h.copy()
            
            # Calculate the C5 value
            c5 = max(np.correlate(h_new, 1 - h_new, mode="full") * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        
    # Evaluate the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure the solution is valid
    if np.sum(best_h) * 2.0 / len(best_h) != 1.0:
        final_c5 = float('inf')
    
    # Return the best solution
    return best_h, final_c5, n_points
