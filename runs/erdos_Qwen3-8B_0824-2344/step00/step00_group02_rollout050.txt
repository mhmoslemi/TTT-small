import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted[k:] = h[:len(h) - k]
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Function to compute the overlap for all shifts
    def compute_all_overlaps(h):
        h = np.array(h)
        h_shifted = np.zeros((len(h), len(h)))
        for k in range(len(h)):
            h_shifted[k, :] = h[:len(h) - k] if k < len(h) else np.zeros(len(h))
        h_shifted = h_shifted[:len(h), :]
        h_ones = np.ones_like(h_shifted)
        overlaps = h_shifted * (h_ones - h_shifted)
        return np.max(overlaps) * dx
    
    # Initial C5 value
    c5_initial = compute_c5(h_initial)
    best_h = h_initial.copy()
    best_c5 = c5_initial
    
    # Try a few different numbers of points, starting from the initial
    n_candidates = [n_points // 2, n_points, n_points * 2, n_points * 3]
    for n in n_candidates:
        if n < 10:
            continue
        if n > 500:
            continue
        
        # Create a new h with n points
        h = np.random.rand(n) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        
        # Optimize with L-BFGS-B
        start_time = time.time()
        result = optimize.minimize(
            compute_all_overlaps,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n)],
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    
    # Final check with the best solution
    final_c5 = compute_all_overlaps(best_h)
    return best_h, final_c5, len(best_h)
