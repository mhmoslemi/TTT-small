```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points values
    n_points_list = [100, 200, 300, 400]
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in n_points_list:
        if time.time() > budget_s - 1:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)
        
        # Start with initial guess
        h_initial = initial_h_values.copy()
        h_initial = np.interp(x, np.linspace(0, 2, len(h_initial)), h_initial)
        h_initial = np.clip(h_initial, 0, 1)
        
        # Optimization variables
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Objective function: minimize the maximum overlap integral
        # We approximate the maximum by using a Monte Carlo approach
        # To avoid overfitting, we use a small number of samples
        num_samples = 100
        max_overlap = 0.0
        for i in range(num_samples):
            k = np.random.uniform(-2, 2)
            h_shifted = np.roll(h.value, int(k * n_points / 2))
            overlap = np.sum(h.value * (1 - h_shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        
        # Define the objective as minimizing the maximum overlap
        # We use a tolerance to handle numerical issues
        tolerance = 1e-4
        obj = cp.Minimize(max_overlap)
        
        # Solve the optimization problem
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4, max_iter=100)
        except:
            print("Solver failed, trying with another initial guess...")
            h_initial = np.random.uniform(0, 1, n_points)
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
            h.value = h_initial[:]
            prob.solve(solver=cp.SCS, eps=1e-4, max_iter=100)
        
        # Evaluate the solution
        if prob.status == 'optimal':
            h_values = h.value
            # Compute the maximum overlap
            c5 = 0.0
            for k in np.linspace(-2, 2, 100):
                h_shifted = np.roll(h_values, int(k * n_points / 2))
                overlap = np.sum(h_values * (1 - h_shifted)) * dx
                if overlap > c5:
                    c5 = overlap
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values
                best_n_points = n_points
        
        # Optional: add random perturbations to escape local minima
        h_values = np.clip(h.value + np.random.normal(0, 0.05, n_points), 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h.value = h_values[:]
    
    # Final evaluation
    if best_h is None:
        best_h = initial_h_values
    
    c5_bound = evaluate_erdos_solution(best_h)
    
    return best_h, c5_bound, best_n_points
```
