```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 800  # Chosen to balance speed and accuracy
    dx = 2.0 / n_points
    target_c5 = 0.380876  # Target to beat
    
    # Initial guess (start from a good initial construction)
    h = np.copy(initial_h_values)
    h = np.resize(h, n_points)
    h /= np.sum(h) * dx  # Ensure integral is 1
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h_vals):
        h = h_vals.copy()
        h[h < 0] = 0
        h[h > 1] = 1
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Optimization function to minimize C5
    def objective_func(h_vals):
        return evaluate_overlap(h_vals)
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run optimization
    result = optimize.minimize(
        objective_func,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Evaluate the best solution found
    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    
    # Early stopping if time is up
    if time.time() - start_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    # Try a few random perturbations if we have time
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        c5 = evaluate_overlap(perturbed_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = perturbed_h
    
    return (best_h, best_c5, n_points)
```
