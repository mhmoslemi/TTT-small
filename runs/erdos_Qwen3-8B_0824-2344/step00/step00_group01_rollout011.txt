import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial samples (n_points = 47)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    
    # Function to compute the C5 value
    def compute_c5(h):
        dx = 2.0 / n_points
        h_extended = np.concatenate([h, h[1:]])
        correlated = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlated) * dx
        return c5
    
    # Function to evaluate and return C5 value
    def objective(h):
        return compute_c5(h)
    
    # Function to evaluate and return the C5 value with constraints
    def evaluate_with_constraints(h):
        # Check constraints
        if not (np.all(h >= 0) and np.all(h <= 1)):
            return np.inf
        if abs(np.sum(h) - n_points / 2) > 1e-8:
            return np.inf
        return compute_c5(h)
    
    # Optimization with gradient descent
    def optimize_solution(h):
        learning_rate = 0.01
        max_iterations = 1000
        for _ in range(max_iterations):
            # Compute the overlap integral
            c5 = evaluate_with_constraints(h)
            if c5 < 0.38092:
                break  # Early stopping if target is reached
            # Compute the gradient
            h_extended = np.concatenate([h, h[1:]])
            correlated = np.correlate(h, 1 - h, mode="full")
            max_index = np.argmax(correlated)
            # Determine the direction to adjust h
            if max_index < n_points:
                # Overlap with h[x] and 1 - h[x + k]
                # Adjust h[x] to reduce the overlap
                h[max_index] -= 0.01
            else:
                # Overlap with h[x] and 1 - h[x + k]
                # Adjust h[x - k] to reduce the overlap
                k = max_index - n_points
                h[k] -= 0.01
            # Clip values to [0, 1]
            h = np.clip(h, 0, 1)
        return h
    
    # Run optimization
    best_h = optimize_solution(h)
    best_c5 = compute_c5(best_h)
    
    # If time remains, try a local optimization
    if time.time() - start_time < budget_s:
        # Local optimization with CVXPY
        # Define variables
        h_var = cp.Variable(n_points)
        # Objective
        dx = 2.0 / n_points
        h_extended = cp.vstack([h_var, h_var[1:]])
        correlated = cp.correlate(h_var, 1 - h_var, mode="full")
        c5 = cp.max(correlated) * dx
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]
        # Solve the optimization
        problem = cp.Problem(cp.Minimize(c5), constraints)
        try:
            result = problem.solve(solver=cp.SCS, time_limit=budget_s - (time.time() - start_time))
            if result is not None:
                best_h = h_var.value
                best_c5 = c5.value
        except:
            pass
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Return the best solution
    return (best_h, final_c5, n_points)
