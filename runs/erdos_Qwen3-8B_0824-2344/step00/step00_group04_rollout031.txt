import numpy as np
import scipy.optimize as opt

# Pre-imported functions

# evaluate_erdos_solution(h_values) is used to compute the C₅ value

# Define the run function
def run(seed=42, budget_s=60, **kwargs):
    # Set a random seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Parameters
    max_points = 1000  # Max number of points to consider
    n_points = 100  # Start with 100 points, can increase if time allows
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        h = np.array(h_values)
        shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted[k:] = h[:len(h) - k]
        integrals = np.zeros(len(h) - 1)
        for k in range(len(integrals)):
            overlap = np.sum(h * (1 - shifted[k:]))
            integrals[k] = overlap * dx
        return np.max(integrals)
    
    # Initial point (starting from provided initial_h_values)
    h_values = initial_h_values.copy()
    n_points_initial = len(h_values)
    
    # Scale the initial point to the desired number of points
    if n_points_initial != n_points:
        h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), h_values)
    
    # Check if the initial point satisfies the constraint
    if np.sum(h_values) * dx != 1:
        h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), h_values)
        h_values = (h_values * (n_points / 2.0)) / (np.sum(h_values) * dx)
    
    # Start with the initial point
    best_c5 = compute_c5(h_values)
    best_h = h_values.copy()
    
    # Try a few different starting points with different numbers of points
    for n_points in [100, 150, 200]:
        dx = 2.0 / n_points
        
        # Start with a uniform distribution (for comparison)
        h_values = np.zeros(n_points)
        h_values[:int(n_points / 2)] = 1.0
        if np.sum(h_values) * dx != 1:
            h_values = h_values / (np.sum(h_values) * dx)
        
        # Check if the initial point is valid
        if np.sum(h_values) * dx != 1:
            h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), h_values)
            h_values = (h_values * (n_points / 2.0)) / (np.sum(h_values) * dx)
        
        # Use local optimization to improve the solution
        def objective(x):
            h = x.reshape(n_points)
            if np.sum(h) * dx < 1 - 1e-6 or np.sum(h) * dx > 1 + 1e-6:
                return np.inf
            return compute_c5(h)
        
        # Bounds for the optimization
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use the initial guess
        x0 = h_values.copy()
        
        # Perform optimization
        result = opt.minimize(objective, x0=x0, bounds=bounds, method='L-BFGS-B', tol=1e-6)
        
        # Update the best solution
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    
    # Final check: ensure the best solution satisfies the constraint
    if np.sum(best_h) * dx != 1:
        best_h = best_h / (np.sum(best_h) * dx) * (n_points / 2.0)
    
    # Ensure the values are within [0, 1]
    best_h = np.clip(best_h, 0.0, 1.0)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Check elapsed time
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, final_c5, n_points)
    
    return (best_h, final_c5, n_points)
