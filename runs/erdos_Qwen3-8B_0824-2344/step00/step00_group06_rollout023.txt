```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 samples
    dx = 2.0 / n_points
    max_iterations = 100  # Maximum number of iterations
    time_remaining = budget_s
    
    # Initialize with the initial H values
    h_values = initial_h_values.copy()
    h_values = h_values[:n_points]  # Ensure it's the correct length
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(h_values) * dx
    if abs(total_integral - 1.0) > 1e-6:
        # Normalize to satisfy the integral constraint
        h_values = h_values * (1.0 / total_integral)
    
    best_h = h_values.copy()
    best_c5 = evaluate_erdos_solution(h_values)
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > time_remaining:
            break
        
        # Create a copy for perturbation
        h = h_values.copy()
        
        # Perturb the function slightly
        perturbation = np.random.normal(0, 0.05, n_points)
        h += perturbation
        h = np.clip(h, 0.0, 1.0)
        
        # Ensure the integral constraint is satisfied
        total_integral = np.sum(h) * dx
        if abs(total_integral - 1.0) > 1e-6:
            h = h * (1.0 / total_integral)
        
        # Evaluate the C5 value
        c5 = evaluate_erdos_solution(h)
        
        # Update the best solution
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Print progress
        print(f"Iteration {iteration + 1}: Current C5 = {best_c5:.6f}")
    
    # If time is remaining, try with a higher number of points (up to 500)
    if time_remaining - (time.time() - start_time) > 10:
        n_points = 500
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        h_values[::2] = 1.0  # Start with a simple alternating pattern
        
        # Normalize to satisfy the integral constraint
        total_integral = np.sum(h_values) * dx
        h_values = h_values * (1.0 / total_integral)
        
        # Evaluate the C5 value
        c5 = evaluate_erdos_solution(h_values)
        
        if c5 < best_c5:
            best_h = h_values.copy()
            best_c5 = c5
    
    # Final check with a convex optimization approach
    if time_remaining - (time.time() - start_time) > 5:
        # Define variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,  # Ensure the integral constraint
            cp.Minimize(cp.max(cp.multiply(h, 1 - cp.roll(h, 1))))
        ]
        
        # Solve the optimization problem
        problem = cp.Problem(cp.Minimize(cp.max(cp.multiply(h, 1 - cp.roll(h, 1)))), constraints)
        result = problem.solve()
        
        # Extract the solution
        h_values = h.value
        h_values = np.clip(h_values, 0.0, 1.0)
        
        # Evaluate the C5 value
        c5 = evaluate_erdos_solution(h_values)
        
        if c5 < best_c5:
            best_h = h_values.copy()
            best_c5 = c5
    
    # Final evaluation to check if the best solution is within the target
    if best_c5 < 0.380876:
        print(f"Improved upper bound: C5 ≤ {best_c5:.6f}")
    else:
        print(f"Current bound: C5 ≤ {best_c5:.6f}")
    
    return (best_h, best_c5, n_points)
```
