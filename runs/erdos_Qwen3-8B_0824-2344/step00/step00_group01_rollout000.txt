import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Define the initial number of points and initial guess
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points]) * (n_points / 2)  # Normalize to meet sum(h) == n_points / 2
    
    # Function to compute C_5
    def compute_c5(h_values):
        h = h_values / (n_points / 2)  # Normalize back to [0, 1]
        h = np.clip(h, 0, 1)  # Ensure values are within [0, 1]
        h_pad = np.concatenate([h, np.zeros_like(h)])
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    # Function to compute the objective for optimization
    def objective_func(h_values):
        return compute_c5(h_values)
    
    # Function to compute the constraints for optimization
    def constraint_func(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Set up the optimization problem
    h_var = cp.Variable(n_points)
    objective = cp.Minimize(objective_func(h_var))
    constraints = [constraint_func(h_var) == 0, 0 <= h_var, h_var <= 1]
    
    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    try:
        result = prob.solve(solver=cp.SCS, verbose=False, time_limit=budget_s - (time.time() - start_time))
    except:
        result = None
    
    if result is not None and h_var.value is not None:
        h_values = h_var.value
        c5_bound = compute_c5(h_values)
    else:
        # If optimization fails, use the initial guess
        h_values = h
        c5_bound = compute_c5(h_values)
    
    # Ensure the solution satisfies constraints
    h_values = np.clip(h_values, 0, 1)
    if np.sum(h_values) != n_points / 2:
        h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    # Final check for validity
    h_values = np.clip(h_values, 0, 1)
    final_c5 = compute_c5(h_values)
    
    return h_values, final_c5, n_points
