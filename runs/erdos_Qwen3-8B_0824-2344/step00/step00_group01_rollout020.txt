import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial guess with n_points = 47
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values, dtype=np.float64)
    
    # Try different n_points (lower than 1k preferred)
    n_points_list = [47, 64, 128, 256, 512]
    best_c5 = float('inf')
    best_h_values = None
    
    for n_points in n_points_list:
        # Ensure we don't exceed the time budget
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create evenly spaced points in [0, 2]
        x = np.linspace(0, 2, n_points, endpoint=False)
        dx = 2.0 / n_points
        
        # Initial guess: interpolate from initial_h_values
        if n_points < n_points_initial:
            # Interpolate to lower resolution
            h = np.interp(x, np.linspace(0, 2, n_points_initial), h_initial)
        else:
            # Extrapolate to higher resolution
            h = np.interp(x, np.linspace(0, 2, n_points_initial), h_initial)
        
        # Scale h to meet the constraint sum(h) == n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Define objective function
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            # Compute overlap using convolution
            overlap = signal.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlap)
        
        # Define constraints
        def constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - n_points / 2
        
        # Set up optimization problem
        h_flat = cp.Variable(n_points)
        obj = cp.Minimize(objective(h_flat))
        cons = [constraint(h_flat) == 0]
        
        # Solve using SCS
        prob = cp.Problem(obj, cons)
        try:
            result = prob.solve(solver=cp.SCS, eps=1e-4, time_limit=10)
            if result is not None and result < best_c5:
                best_c5 = result
                best_h_values = h_flat.value
        except:
            pass
        
        # Run gradient descent as a fallback
        h_flat = np.copy(h)
        for _ in range(100):
            # Compute overlap
            overlap = signal.correlate(h_flat, 1 - h_flat, mode='full') * dx
            c5 = np.max(overlap)
            
            # Compute gradient
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Perturb h_flat[i]
                h_perturb = h_flat.copy()
                h_perturb[i] += 1e-5
                overlap_perturb = signal.correlate(h_perturb, 1 - h_perturb, mode='full') * dx
                grad[i] = (np.max(overlap_perturb) - c5) / 1e-5
            
            # Adjust h_flat
            h_flat -= 0.01 * grad
            h_flat = np.clip(h_flat, 0, 1)
            h_flat = h_flat * (n_points / 2) / np.sum(h_flat)
        
        # Compute final c5
        overlap = signal.correlate(h_flat, 1 - h_flat, mode='full') * dx
        c5 = np.max(overlap)
        if c5 < best_c5:
            best_c5 = c5
            best_h_values = h_flat
        
    # Convert best_h_values to a list
    best_h_values = best_h_values.tolist()
    
    # Ensure the sum constraint is met
    sum_h = sum(best_h_values) * (2.0 / len(best_h_values))
    if abs(sum_h - 1.0) > 1e-6:
        # Adjust to meet the constraint
        factor = 1.0 / sum_h
        best_h_values = [x * factor for x in best_h_values]
    
    return best_h_values, best_c5, len(best_h_values)
