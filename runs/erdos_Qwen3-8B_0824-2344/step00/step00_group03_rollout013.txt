import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
# evaluate_erdos_solution() is already imported
# initial_h_values is already imported as a numpy array

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = h[:n_points]  # Ensure we use the correct number of points
    
    # Objective: minimize the maximum overlap
    def objective(h_flat):
        h = h_flat.reshape((n_points,))
        # Compute the correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        # Scale appropriately
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    # Bounds: 0 <= h[i] <= 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    h_initial = h.copy()

    # Optimization
    result = optimize.minimize(
        objective,
        h_initial,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": constraint_sum}],
        method="SLSQP",
        options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8},
    )
    
    # Check if we've used up the budget
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 1:
        return (h_initial, objective(h_initial), n_points)

    # Use the optimized result
    h_optimized = result.x
    c5_bound = result.fun
    return (h_optimized, c5_bound, n_points)
