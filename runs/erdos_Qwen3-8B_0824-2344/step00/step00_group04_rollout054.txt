import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    max_iterations = 50
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    # Ensure the initial h values satisfy the constraint
    initial_h = initial_h_values[:n_points] / (n_points / 2)
    initial_h[initial_h > 1] = 1
    initial_h[initial_h < 0] = 0
    best_h = initial_h.copy()
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        h_pad = np.zeros(n_points * 2 + 1)
        h_pad[:n_points] = h
        h_pad[n_points:] = np.zeros(n_points)
        result = np.correlate(h_pad, 1 - h, mode="full")
        return np.max(result) * (2.0 / n_points)
    
    # Function to optimize h
    def optimize_h():
        nonlocal best_c5, best_h
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Objective is to minimize the maximum overlap
        h_pad = cp.vstack([h, np.zeros(n_points)])
        h_pad = cp.vstack([h_pad, np.zeros(n_points)])
        h_pad = h_pad.reshape(-1)
        obj = cp.Maximize(cp.Minimize(cp.norm(np.correlate(h_pad, 1 - h, mode="full"), 1)))
        
        # Solve using ECOS
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS)
            if prob.status == 'OPTIMAL':
                h_val = h.value
                c5 = evaluate_overlap(h_val)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_val.copy()
        except:
            pass
    
    for _ in range(max_iterations):
        optimize_h()
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h, best_c5, n_points)
