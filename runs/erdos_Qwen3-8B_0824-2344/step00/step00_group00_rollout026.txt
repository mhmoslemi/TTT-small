import numpy as np

# Initial construction from global variable
initial_h_values = np.array([...])  # Assume this is preloaded

# Helper function to compute the overlap integral
def compute_c5(h, dx):
    h_extended = np.zeros(2 * len(h))
    h_extended[:len(h)] = h
    h_1_minus = 1 - h_extended
    correlation = np.correlate(h_extended, h_1_minus, mode="full")
    return np.max(correlation) * dx

# Function to generate a random initial guess
def generate_random_guess(n_points):
    h = np.random.rand(n_points)
    h = (h * 2 - 1) * 0.5 + 0.5  # Scale to [0, 1]
    h = h / np.sum(h) * n_points / 2
    return h

# Function to minimize the overlap integral using gradient descent
def optimize_h(n_points, initial_h=None, max_iter=100, learning_rate=0.01, seed=42):
    np.random.seed(seed)
    if initial_h is None:
        h = generate_random_guess(n_points)
    else:
        h = initial_h.copy()
    
    best_h = h
    best_c5 = compute_c5(h, 2.0 / n_points)
    
    for _ in range(max_iter):
        # Compute the overlap integral
        c5 = compute_c5(h, 2.0 / n_points)
        
        # Compute the derivative to update h
        # For simplicity, approximate with finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            c5_plus = compute_c5(h_plus, 2.0 / n_points)
            grad[i] = (c5_plus - c5) / 1e-5
        
        # Update h
        h -= learning_rate * grad
        
        # Ensure h stays in [0, 1]
        h = np.clip(h, 0, 1)
        
        # Enforce the integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Update best solution
        c5_new = compute_c5(h, 2.0 / n_points)
        if c5_new < best_c5:
            best_h = h
            best_c5 = c5_new
    
    return best_h, best_c5

# Function to find the best solution within the budget

    start_time = time.time()

        if c5_opt < best_c5:
