```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the full convolution (correlation)
    corr = np.correlate(h, one_minus_h, mode='full')
    # Scale by dx and find the maximum
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    # Initial construction with n=42 samples
    return np.array([0.5, 0.5, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75,
                     0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75,
                     0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75,
                     0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75,
                     0.75, 0.75, 0.75, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Try with different n_points
    n_points_list = [50, 60, 70, 80, 90, 100]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        dx = 2.0 / n_points
        h = initial_h_values()
        h = np.resize(h, n_points)
        h = h * (n_points / 2) / np.sum(h)
        
        def objective(x):
            h_values = x
            h_values = np.clip(h_values, 0, 1)
            corr = np.correlate(h_values, 1 - h_values, mode='full')
            return np.max(corr) * dx
        
        constraints = [
            cp.sum(x) == n_points / 2
        ]
        
        x = cp.Variable(n_points)
        prob = cp.Problem(cp.Minimize(objective(x)), constraints)
        
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-4, max_iters=1000)
            if result is not None:
                h_values = x.value
                c5 = evaluate_erdos_solution(h_values)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
                    best_n = n_points
        except:
            pass
        
        print(f"n_points={n_points}, c5={best_c5}")
    
    return best_h, best_c5, best_n
```
