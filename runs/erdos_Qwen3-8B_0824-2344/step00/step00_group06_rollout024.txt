import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial heuristic: start from the initial construction
    h_initial = initial_h_values
    
    # Parameters
    n_points_initial = len(h_initial)
    n_points = n_points_initial  # Start with the initial number of points
    
    # Try reducing number of points for faster evaluation
    # Start with the initial n_points and try reducing if needed
    max_attempts = 5
    best_c5 = float('inf')
    best_h = h_initial.copy()
    best_n_points = n_points_initial
    
    # Function to evaluate C5 for a given h
    def evaluate_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        # Compute the overlap integral using convolution
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization function to minimize the worst-case overlap
    def objective_func(h_vals):
        h = np.array(h_vals)
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Constraint: integral of h over [0, 2] is 1
    def constraint_func(h_vals):
        return np.sum(np.array(h_vals)) - len(h_vals) / 2
    
    # Try different number of points to find a good balance
    for attempt in range(max_attempts):
        start_time = time.time()
        n_points = max(20, n_points_initial - 20)  # Ensure at least 20 points
        
        # Initialize h with the initial values, rescaled to fit n_points
        h = np.zeros(n_points)
        if len(initial_h_values) > n_points:
            h = initial_h_values[:n_points]
        else:
            h = np.resize(initial_h_values, n_points)
            # Normalize to ensure the integral is 1
            h = h * (len(initial_h_values) / 2) / (n_points / 2)
        
        # Perform optimization to minimize C5
        # We'll use a local optimization method with random restarts
        for _ in range(10):  # 10 random restarts
            # Initialize h with some perturbation
            h_perturb = h + np.random.normal(0, 0.1, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            # Normalize to ensure integral is 1
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            
            # Run optimization
            result = optimize.minimize(objective_func, h_perturb, method='L-BFGS-B', 
                                       bounds=[(0, 1) for _ in range(n_points)],
                                       tol=1e-10, options={'maxiter': 100})
            
            # Update best solution
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
                best_n_points = n_points
        
        # Check if we've exceeded the budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Evaluate the best solution
    final_c5 = evaluate_c5(best_h)
    
    # Return the result
    return best_h, final_c5, best_n_points
