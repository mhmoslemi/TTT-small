import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Predefine some parameters
    max_points = 100  # Maximum number of points to consider
    initial_points = 70  # Initial number of points from the initial solution
    initial_h_values = np.load("initial_h_values.npy")  # Assuming initial_h_values is provided
    n_points_initial = len(initial_h_values)
    n_points = min(max_points, n_points_initial)  # Use initial points if they are within the limit

    # Start with initial solution
    h = initial_h_values[:n_points].copy()
    # Normalize to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)

    def evaluate_c5(h_values):
        h = h_values.copy()
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        # Use convolution with mode "full"
        corr = np.correlate(h, 1 - h, mode="full")
        # The overlap integral is the maximum of the correlation scaled by dx
        dx = 2.0 / len(h)
        c5 = np.max(corr) * dx
        return c5

    def optimize_h(h_initial, max_iter=100, tolerance=1e-5):
        best_h = h_initial.copy()
        best_c5 = evaluate_c5(best_h)
        best_iter = 0
        start_time = time.time()

        for iter in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            # Use a simple gradient-based optimization
            h = best_h.copy()
            # Create an optimization variable
            x = cp.Variable(len(h))
            # Define the objective function: minimize the maximum overlap
            # We use a convex relaxation to approximate the max operation
            # This is a simplified approach for demonstration purposes
            corr = np.correlate(h, 1 - h, mode="full")
            dx = 2.0 / len(h)
            # We'll use a small epsilon to avoid division by zero
            eps = 1e-8
            # Create a list of possible correlation values
            corr_values = corr * dx
            # We approximate the max using a sum of convex terms
            # This is a simplified relaxation; a better approach would involve more sophisticated convex programming
            objective = cp.Minimize(cp.sum([cp.max(corr_values[i] - x[i] + x[i + len(h)] - 1) for i in range(len(h))]))
            constraints = [
                cp.sum(x) == len(h) / 2,  # Ensure integral constraint
                0 <= x,
                x <= 1
            ]

            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.SCS, eps=1e-6)
                # Update the best solution
                best_h = x.value
                best_c5 = evaluate_c5(best_h)
                best_iter = iter
            except:
                # If optimization fails, use a simpler gradient descent
                h = best_h.copy()
                dx = 2.0 / len(h)
                # Simple gradient descent
                for _ in range(100):
                    corr = np.correlate(h, 1 - h, mode="full")
                    dx = 2.0 / len(h)
                    overlap = corr * dx
                    max_overlap = np.max(overlap)
                    # Compute gradient approximation
                    grad = np.zeros_like(h)
                    for i in range(len(h)):
                        # Perturb h[i]
                        h_perturb = h.copy()
                        h_perturb[i] += 1e-5
                        corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode="full")
                        dx_perturb = 2.0 / len(h)
                        overlap_perturb = corr_perturb * dx_perturb
                        grad[i] = (np.max(overlap_perturb) - max_overlap) / 1e-5
                    # Update h using gradient descent
                    h -= 0.01 * grad
                    h = np.clip(h, 0, 1)
                    # Enforce integral constraint
                    h = h * (len(h) / 2) / np.sum(h)
                best_h = h
                best_c5 = evaluate_c5(h)
            # Early stopping if improvement is minimal
            if np.abs(best_c5 - evaluate_c5(best_h)) < tolerance:
                break
        return best_h, best_c5

    # Optimize the initial solution
    optimized_h, optimized_c5 = optimize_h(h, max_iter=50, tolerance=1e-5)

    for n in range(initial_points - 10, initial_points + 10):
        if n < 20:
            continue
        # Create a new h with n samples
        h = np.zeros(n)
        # Distribute the integral constraint
        h = np.random.rand(n)
        h = h * (n / 2) / np.sum(h)
        # Optimize the new h
        optimized_h_new, optimized_c5_new = optimize_h(h, max_iter=50, tolerance=1e-5)
        # If the new solution is better, update
        if optimized_c5_new < optimized_c5:
            optimized_h = optimized_h_new
            optimized_c5 = optimized_c5_new

    # Return the final solution
    return optimized_h, optimized_c5, len(optimized_h)
