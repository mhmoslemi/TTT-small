import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can be increased if time allows

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        nonlocal n_points
        h = np.array(h)
        h = np.concatenate((h, h[:n_points//2]))  # Pad for full convolution
        h_shifted = np.roll(h, -n_points//2)
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        return max(overlap) * (2.0 / n_points)

    # Function to compute the total integral
    def total_integral(h):
        return np.sum(h) * (2.0 / n_points)

    # Initial guess
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]  # Ensure it's the right length
    h_initial = np.array(h_initial, dtype=np.float64)

    # Optimization: Use gradient descent with constraints
    def objective_function(h):
        nonlocal n_points
        h = np.array(h, dtype=np.float64)
        h = np.concatenate((h, h[:n_points//2]))  # Pad for full convolution
        h_shifted = np.roll(h, -n_points//2)
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = max(overlap) * (2.0 / n_points)
        return c5

    # Minimize the overlap integral
    def optimize_h(h0):
        nonlocal n_points, best_c5, best_h
        h = h0.copy()
        for _ in range(1000):  # Max iterations
            if time.time() - start_time > budget_s:
                break
            # Evaluate current C5
            current_c5 = evaluate_overlap(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            # Gradient descent step
            h_shifted = np.roll(h, -n_points//2)
            overlap = np.correlate(h, 1 - h_shifted, mode="full")
            max_overlap_idx = np.argmax(overlap)
            h[max_overlap_idx] = np.clip(h[max_overlap_idx] - 0.01, 0, 1)
            # Regularization: adjust for total integral constraint
            total = np.sum(h) * (2.0 / n_points)
            if abs(total - 1.0) > 0.01:
                h = h * (1.0 / total) * (2.0 / n_points)

    optimize_h(h_initial)

    # Try a slightly higher number of points if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (2.0 / n_points)
        optimize_h(h_initial)

    # Final evaluation
    if best_h is not None:
        best_h = np.array(best_h, dtype=np.float64)
        best_c5 = evaluate_overlap(best_h)
    else:
        best_h = h_initial
        best_c5 = evaluate_overlap(best_h)

    n_points = len(best_h)
    return best_h, best_c5, n_points
