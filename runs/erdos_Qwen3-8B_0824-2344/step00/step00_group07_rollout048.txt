import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
from functools import lru_cache

# Pre-imported helpers
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values as initial_h

# Set the initial number of points
n_points = 200
dx = 2.0 / n_points

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial guess based on provided initial_h_values
    h = np.array(initial_h).copy()
    
    # Rescale to fit the number of points (if needed)
    if len(h) != n_points:
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    h = h[:n_points]
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2)  # Ensure the integral equals 1
    
    # Function to compute C5
    def compute_c5(h_values):
        h = np.array(h_values)
        # Ensure the integral is 1
        integral = np.sum(h) * dx
        if np.abs(integral - 1) > 1e-6:
            return np.inf
        # Compute correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        # The max overlap integral is the maximum value of the correlation
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute the gradient of C5
    def compute_gradient(h_values):
        h = np.array(h_values)
        integral = np.sum(h) * dx
        if np.abs(integral - 1) > 1e-6:
            return np.zeros_like(h)
        # Compute correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        max_index = np.argmax(corr)
        # Extract the relevant part of the correlation
        corr_segment = corr[max_index - n_points:max_index + n_points]
        # Compute derivative of the correlation with respect to h
        # Use central difference for gradient estimation
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            c5_plus = compute_c5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            c5_minus = compute_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * 1e-6)
        return grad
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Run optimization
    t_remaining = budget_s - (time.time() - start_time)
    if t_remaining > 0:
        # Use Nelder-Mead for optimization
        result = optimize.minimize(
            compute_c5,
            h,
            method='L-BFGS-B',
            jac=compute_gradient,
            options={'maxiter': int(t_remaining * 5), 'ftol': 1e-6, 'gtol': 1e-6}
        )
        best_h = result.x
        best_c5 = result.fun
    
    # Ensure the integral is 1
    integral = np.sum(best_h) * dx
    if np.abs(integral - 1) > 1e-6:
        # Normalize the function
        normalization_factor = 1 / (integral)
        best_h = best_h * normalization_factor
    
    # Final check
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
