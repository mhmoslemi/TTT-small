    start_time = time.time()
    iterations = 0

        h = h * dx
        conv = correlate(h, 1 - h, mode="full")
        return np.max(conv)

        nonlocal best_c5, best_h
        c5_current = compute_c5(h_current)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
        return best_c5

    # Initial evaluation
    current_c5 = compute_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()

    # Gradient descent with momentum for faster convergence
    learning_rate = 0.01
    momentum = 0.9
    grad = np.zeros(n_points)
    prev_grad = np.zeros(n_points)

    while time.time() - start_time < budget_s:
        iterations += 1
        if iterations > 300:
            break
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        h = h * dx
        conv = correlate(h, 1 - h, mode="full")
        c5 = np.max(conv)
        if c5 >= best_c5:
            break  # No improvement, exit early
        # Compute gradients
        for i in range(n_points):
            h[i] += 1e-5
            conv = correlate(h, 1 - h, mode="full")
            grad[i] = (np.max(conv) - c5) / 1e-5
            h[i] -= 1e-5
        # Update weights
        grad = grad * learning_rate + momentum * prev_grad
        prev_grad = grad
        h += grad
        h = np.clip(h, 0, 1)
        h = h * dx
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h = best_h.copy() + perturbation
        h = np.clip(h, 0, 1)
        h = h * dx
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5

        h = np.clip(x.value, 0, 1)
        h = h * dx
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:

            best_h = h.copy()
    except:
        pass

    h_values = best_h.copy() * dx
    return h_values, best_c5, n_points
