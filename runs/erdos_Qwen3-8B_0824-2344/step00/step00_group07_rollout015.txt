import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 points, adjust if needed
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Use initial sample
    h = h / np.sum(h) * (n_points / 2)  # Normalize to meet integral constraint
    
    # Function to compute C5
    def compute_c5(h_values):
        h = h_values.copy()
        h = h / np.sum(h) * (n_points / 2)  # Ensure integral constraint
        h = np.roll(h, 1)  # Shift to avoid edge effects
        k_values = np.arange(-n_points + 1, n_points)
        overlaps = []
        for k in k_values:
            if k < 0:
                overlap = np.sum(h[:n_points + k] * (1 - h[-k:]))
            elif k > 0:
                overlap = np.sum(h[:n_points - k] * (1 - h[k:]))
            else:
                overlap = np.sum(h * (1 - h))
            overlaps.append(overlap * dx)
        return max(overlaps)
    
    # Function to optimize h
    def objective_func(h_values):
        return compute_c5(h_values)
    
    # Optimization
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h
    
    # Try different n_points values to find optimal resolution
    for n_points in [100, 150, 200, 250]:
        dx = 2.0 / n_points
        h = np.copy(initial_h_values[:n_points])
        h = h / np.sum(h) * (n_points / 2)
        
        # Define variables
        x = cp.Variable(n_points, integer=True)
        x_bounds = [(0, 1) for _ in range(n_points)]
        constraints = [x >= 0, x <= 1, cp.sum(x) == n_points / 2]
        
        # Objective
        def obj(x_val):
            h_vals = x_val
            h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
            h_vals = np.roll(h_vals, 1)
            k_values = np.arange(-n_points + 1, n_points)
            overlaps = []
            for k in k_values:
                if k < 0:
                    overlap = np.sum(h_vals[:n_points + k] * (1 - h_vals[-k:]))
                elif k > 0:
                    overlap = np.sum(h_vals[:n_points - k] * (1 - h_vals[k:]))
                else:
                    overlap = np.sum(h_vals * (1 - h_vals))
                overlaps.append(overlap * dx)
            return max(overlaps)
        
        # Use a heuristic to find the minimum
        def objective(x_val):
            return obj(x_val)
        
        # Try simulated annealing
        x_opt = opt.minimize(objective, h, method='L-BFGS-B', bounds=x_bounds, tol=1e-6)
        h_opt = x_opt.x
        
        # Compute C5
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
