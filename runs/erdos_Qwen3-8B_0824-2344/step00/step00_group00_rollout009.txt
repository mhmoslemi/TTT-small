import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 45  # Start with the initial n_points
    
    # Define the initial h_values
    h = initial_h_values.copy() / (2.0 / n_points)  # Normalize to match the constraint
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        h = h_values / (2.0 / n_points)
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * (2.0 / n_points)
        return c5
    
    # Function to optimize h using gradient descent
    def optimize_h(h0, max_iter=100, learning_rate=0.01, tol=1e-5):
        h = h0.copy()
        for i in range(max_iter):
            # Evaluate current C5
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Compute gradient
            grad = np.zeros_like(h)
            for k in range(len(h)):
                for j in range(len(h)):
                    if k + j < len(h):
                        h_k = h[k]
                        one_minus_h_j = 1 - h[j]
                        grad[k] += one_minus_h_j * (np.max([0, 1 - h_k]) - np.max([0, h_k]))
                    else:
                        break
            
            # Update h
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            
            # Check for convergence
            if np.abs(np.sum(h) - len(h)) < tol:
                break
        
        return h
    
    # Try optimizing from initial guess
    optimized_h = optimize_h(h)
    current_c5 = compute_c5(optimized_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = optimized_h
    
    # Try reducing n_points if possible
    for n_points in range(30, 46, 5):
        # Create new h with n_points samples
        h_new = np.zeros(n_points)
        h_new[:len(h)] = h / (2.0 / n_points)
        h_new = np.clip(h_new, 0, 1)
        
        # Try optimizing
        optimized_h_new = optimize_h(h_new)
        current_c5 = compute_c5(optimized_h_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h_new
    
    # Final check with the best solution
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)
    
    # Ensure the integral constraint is satisfied
    integral = np.sum(best_h) * (2.0 / len(best_h))
    if np.abs(integral - 1) > 1e-5:
        best_h = best_h * (1 / integral)
    
    # Final check of the constraint
    integral = np.sum(best_h) * (2.0 / len(best_h))
    print(f"Integral: {integral}")
    
    elapsed_time = time.time() - start_time
    print(f"Time elapsed: {elapsed_time:.2f} seconds")
    
    return best_h, best_c5, len(best_h)
