import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial solution
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values)
    
    # Function to compute the C5 value
    def compute_C5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points//2]])  # Extend to full correlation length
        corr = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        return np.max(corr * dx)
    
    # Function to compute the overlap integral for a given shift k
    def overlap_integral(h, k):
        h = np.array(h)
        return np.trapz(h * (1 - np.roll(h, k)), dx=2.0 / n_points)
    
    # Function to evaluate the worst-case overlap
    def evaluate_solution(h):
        max_overlap = 0.0
        for k in range(1, n_points):
            overlap = overlap_integral(h, k)
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Define optimization problem using CVXPY
    def optimize_h(h_values):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        obj = cp.Minimize(cp.max([overlap_integral(h, k) for k in range(1, n_points)]))
        
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.SCS, eps=1e-4, time_limit=60 - (time.time() - start_time))
        
        if result is not None:
            h_values = np.array(h.value)
            h_values = np.clip(h_values, 0, 1)
            return h_values
        return h_values
    
    # Optimization loop
    best_h = h_values
    best_c5 = compute_C5(best_h)
    
    for _ in range(5):  # Perform multiple rounds of optimization
        # Local search with gradient descent
        h = best_h.copy()
        h += np.random.normal(0, 0.05, size=n_points)
        h = np.clip(h, 0, 1)
        h = optimize_h(h)
        
        # Evaluate current solution
        current_c5 = compute_C5(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
    
    # Final evaluation
    final_c5 = compute_C5(best_h)
    return best_h, final_c5, n_points
