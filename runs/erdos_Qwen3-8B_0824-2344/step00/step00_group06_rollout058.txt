```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * 2 / len(h)
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h_values = initial_h_values.copy()
    n_points = len(h_values)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    best_c5 = current_c5

    # Define the objective function to minimize
    def objective(h_values):
        if len(h_values) != n_points:
            raise ValueError("h_values must have length n_points")
        c5 = evaluate_erdos_solution(h_values)
        return c5

    # Define constraints for optimization
    def constraints(h_values):
        if len(h_values) != n_points:
            raise ValueError("h_values must have length n_points")
        return np.array([np.sum(h_values) - n_points / 2])

    # Optimization parameters
    max_iterations = 100
    step_size = 0.05
    tolerance = 1e-4
    perturbation = 0.1

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break

        # Perturb the current solution
        perturbed = h_values.copy()
        indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
        perturbed[indices] += np.random.uniform(-perturbation, perturbation, size=len(indices))
        perturbed = np.clip(perturbed, 0.0, 1.0)

        # Evaluate the perturbed solution
        new_c5 = objective(perturbed)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = perturbed.copy()
            print(f"Improved C5: {best_c5:.6f}")

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    print(f"Final C5: {final_c5:.6f}")
    return best_h, final_c5, n_points
```
