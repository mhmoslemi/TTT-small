def run(seed=42, budget_s=60, **kwargs):

    h = np.zeros(n_points)

        h[i] = np.random.rand() * 2 - 1

    h /= (np.sum(h) / (n_points / 2))

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_n_points = n_points

    for n_points in [150, 200, 250]:

        h = np.zeros(n_points)

        # Normalize to satisfy the integral constraint
        h /= (np.sum(h) / (n_points / 2))

        # Evaluate
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n_points = n_points

        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    for n_points in [250, 300]:

        h = np.random.rand(n_points) * 2 - 1

            if c5 < best_c5:
