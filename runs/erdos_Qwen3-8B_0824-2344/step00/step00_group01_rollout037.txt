def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = signal.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    
    # Try smaller n_points first for speed
    for n in [47, 64, 128, 256, 512, 1024]:
        if time.time() - start_time > budget_s:
            break
        
        h = np.random.uniform(0, 1, n)
        h = h / np.sum(h) * (n / 2)
        
        # Use L-BFGS-B optimization with bounds and constraints
        def objective(x):
            h_vals = x.reshape(n)
            corr = signal.correlate(h_vals, 1 - h_vals, mode='full')
            return np.max(corr) * (2.0 / n)
        
        bounds = [(0.0, 1.0) for _ in range(n)]
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, tol=1e-5, options={'maxiter': 100})
        
        if result.success:
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        # Try a few local refinements with different starting points
        for _ in range(3):
            if time.time() - start_time > budget_s:
                break
            perturbation = np.random.normal(0, 0.1, n)
            perturbation = np.clip(perturbation, -0.2, 0.2)
            h = best_h + perturbation
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n / 2)
            
            def objective(x):
                h_vals = x.reshape(n)
                corr = signal.correlate(h_vals, 1 - h_vals, mode='full')
                return np.max(corr) * (2.0 / n)
            
            result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, tol=1e-5, options={'maxiter': 50})
            
            if result.success:
                current_h = result.x
                current_c5 = evaluate_erdos_solution(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
    
    # Try CVXOPT for more precise optimization on the best result
    if time.time() - start_time < budget_s:
        n = len(best_h)
        h = cp.Variable(n)
        constraints = [cp.sum(h) == n / 2, 0 <= h, h <= 1]
        objective = cp.Maximize(cp.Max(signal.correlate(h.value, 1 - h.value, mode='full') * (2.0 / n)))
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CVXOPT, eps=1e-6)
        best_h = h.value
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, len(best_h))
