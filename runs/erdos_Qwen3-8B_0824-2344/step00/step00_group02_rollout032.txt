def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    # Set the number of points
    n_points = 100  # Start with 100 points for efficiency

    # Optimization function
    def optimize_h():
        nonlocal best_c5, best_h
        h = cp.Variable(n_points, low=0.0, up=1.0)
        constraints = [cp.sum(h) == n_points / 2]
        objective = cp.Minimize(0)  # No direct objective, but we'll evaluate the C5 value
        prob = cp.Problem(objective, constraints)
        prob.solve(solver='ECOS', eps=1e-6)
        if prob.status == 'optimal':
            h_values = h.value
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values.copy()
                print(f"New best C5: {best_c5:.6f}")
        else:
            print("Problem not solved optimally.")
    
    # Run the optimization
    for _ in range(10):
        optimize_h()
        if time.time() - start_time > budget_s - 1:
            break
    
    # Post-optimization refinement
    for _ in range(3):
        h = np.array(best_h)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr_index = np.argmax(corr)
        max_corr_value = corr[max_corr_index]
        # Modify the function at the position of the maximum correlation
        if max_corr_index < len(h):
            h[max_corr_index] = max(0.0, h[max_corr_index] - 0.05)
        else:
            h[max_corr_index - len(h)] = max(0.0, h[max_corr_index - len(h)] - 0.05)
        # Ensure the sum condition is satisfied
        adjustment = (n_points / 2 - np.sum(h)) / n_points
        h += adjustment
        h = np.clip(h, 0.0, 1.0)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
