```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
from initial_h_values import initial_h_values
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Starting with 100 points for faster evaluation

    # Try different numbers of points up to a maximum of 500
    for n_points in [100, 200, 300, 400, 500]:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        # Generate initial guess from the provided initial_h_values
        initial_guess = initial_h_values.copy()
        # Scale to fit the current n_points
        scaled_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_guess)), initial_guess)
        scaled_initial = np.clip(scaled_initial, 0, 1)

        # Define the optimization variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h) == n_points / 2,  # Ensure the integral is 1
            h <= 1,
            h >= 0
        ]

        # Define the overlap integral for all shifts k
        def compute_C5(h_values):
            h_func = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points), h_values)
            conv = correlate(h_func, 1 - h_func, mode='full')
            c5 = np.max(conv * (2.0 / n_points))
            return c5

        # Define objective function to minimize the maximum overlap
        def objective(h_values):
            return compute_C5(h_values)

        # Perform optimization
        try:
            result = optimize.minimize(
                objective,
                scaled_initial,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                tol=1e-6,
                options={'maxiter': 100}
            )

            h_values = result.x
            h_values = np.clip(h_values, 0, 1)
            c5 = compute_C5(h_values)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values

        except Exception as e:
            print(f"Optimization failed for {n_points} points: {e}")

    # Evaluate the best solution found
    if best_h is not None:
        final_c5 = compute_C5(best_h)
        print(f"Best C5 bound: {final_c5} with {len(best_h)} points")
    else:
        final_c5 = best_c5

    return best_h, final_c5, len(best_h)
```
