def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial solution
    h_values = initial_h_values.copy()
    n_points = len(h_values)
    best_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    
    # Try a few random perturbations and improvements
    for _ in range(10):
        # Randomly perturb some values
        idxs = np.random.choice(n_points, size=20, replace=False)
        h_values[idxs] = np.random.uniform(0, 1, size=20)
        # Re-normalize
        h_values = h_values / (np.sum(h_values) / (n_points / 2))
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
    
    # Try an optimization approach with CVXOPT
    # We'll use a mixed-integer programming approach
    # To simplify, we'll use a greedy approach with CVXOPT to minimize the max overlap
    
    # Convert h_values to the CVXOPT variable
    x = cp.Variable(n_points, boolean=True)
    objective = cp.Minimize(cp.Max(cp.sum(cp.multiply(x, 1 - x)), cp.sum(cp.multiply(x, 1 - x[1:]))) * (2.0 / n_points))
    constraints = [cp.sum(x) == n_points / 2]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == 'optimal':
            h_values = np.array(x.value).flatten()
            current_c5 = evaluate_erdos_solution(h_values)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_values.copy()
    except:
        pass
    
    # Try a more refined optimization with gradient-based methods
    def objective_func(h_values):
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values / (np.sum(h_values) / (n_points / 2))
        return evaluate_erdos_solution(h_values)
    
    def gradient_func(h_values):
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values / (np.sum(h_values) / (n_points / 2))
        grad = np.zeros_like(h_values)
        for i in range(len(h_values)):
            h_values[i] = 1.0
            grad[i] = (evaluate_erdos_solution(h_values) - evaluate_erdos_solution(h_values.copy())) / 1e-6
            h_values[i] = 0.0
        return grad
    
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(objective_func, best_h, method='L-BFGS-B', jac=gradient_func, bounds=bounds, tol=1e-4)
    if result.success:
        h_values = result.x
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values / (np.sum(h_values) / (n_points / 2))
        current_c5 = evaluate_erdos_solution(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / (np.sum(best_h) / (n_points / 2))
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, n_points
