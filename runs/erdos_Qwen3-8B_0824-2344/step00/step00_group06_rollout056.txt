```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Assume this is a pre-loaded initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 samples (adjust as needed)
    max_iterations = 100  # Early stopping to respect time budget
    
    # Initial guess
    h_initial = initial_h_values[:n_points] if len(initial_h_values) > n_points else initial_h_values
    h_initial = np.clip(h_initial, 0, 1)
    total_area = np.sum(h_initial) * (2.0 / n_points)
    h_initial = h_initial * (1 / (total_area / (2.0 / n_points)))
    h_initial = np.clip(h_initial, 0, 1)
    
    for iteration in range(max_iterations):
        # Convert to CVXPY variables
        h = cp.Variable(n_points, lower=0.0, upper=1.0)
        objective = cp.Maximize(cp.sum(h * (2.0 / n_points)))
        constraints = [cp.sum(h) * (2.0 / n_points) == 1.0]
        
        # Optimization
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, max_iter=100)
        
        if prob.status != 'OPTIMAL':
            break
        
        h_values = h.value
        c5 = evaluate_erdos_solution(h_values)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values
        
        # Adjust n_points if needed
        if iteration % 10 == 0 and n_points < 500:
            n_points += 50
            h_initial = np.random.rand(n_points)
            h_initial = np.clip(h_initial, 0, 1)
            total_area = np.sum(h_initial) * (2.0 / n_points)
            h_initial = h_initial * (1 / (total_area / (2.0 / n_points)))
            h_initial = np.clip(h_initial, 0, 1)
    
    # Final evaluation
    if best_h is None:
        best_h = h_initial
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return best_h, best_c5, len(best_h)
    
    return best_h, best_c5, len(best_h)
```
