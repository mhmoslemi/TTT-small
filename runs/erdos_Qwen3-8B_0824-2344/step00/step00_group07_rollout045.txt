import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 80  # Start with 80 points as initial point
    h_values = initial_h_values.copy()
    
    # Function to compute C5
    def compute_c5(h):
        h_np = np.array(h)
        dx = 2.0 / n_points
        # Compute (1 - h) shifted by k
        shifted_h = np.zeros((n_points, n_points))
        for k in range(n_points):
            shifted_h[:, k] = h_np[-k:] if k < n_points else h_np
        # Compute overlap integrals
        integrals = np.zeros(n_points)
        for k in range(n_points):
            integrals[k] = np.sum(h_np * (1 - shifted_h[:, k])) * dx
        return np.max(integrals)
    
    # Function to evaluate and return the C5 value
    def evaluate(h):
        return compute_c5(h)
    
    # Function to enforce the integral constraint
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy
    def optimize_h_values(h_values, budget_s):
        start_time = time.time()
        best_c5 = np.inf
        best_h = h_values.copy()
        best_iter = 0
        
        # Use a local optimization approach with random restarts
        for _ in range(50):  # Number of restarts
            if time.time() - start_time > budget_s:
                break
            # Local optimization step
            h = h_values.copy()
            h = np.clip(h, 0, 1)
            
            # Use BFGS to minimize C5
            result = optimize.minimize(
                lambda x: compute_c5(x),
                x0=h,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if result.success:
                c5_val = result.fun
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = result.x.copy()
                    best_iter += 1
            else:
                # If optimization fails, perturb the solution
                perturbation = np.random.normal(0, 0.05, size=n_points)
                h = np.clip(h + perturbation, 0, 1)
                h_values = h.copy()
        
        # Final evaluation
        final_c5 = compute_c5(best_h)
        return best_h, final_c5, n_points
    
    # Run the optimization
    best_h, best_c5, n_points = optimize_h_values(h_values, budget_s)
    
    # Apply final clipping
    best_h = np.clip(best_h, 0, 1)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, n_points
