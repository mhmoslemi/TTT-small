```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial solution
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Function to compute C_5
    def compute_c5(h):
        h = np.array(h)
        h_1_minus = 1 - h
        cross_corr = np.correlate(h, h_1_minus, mode="full")
        return np.max(cross_corr) * dx
    
    # Function to optimize h
    def optimize_h(h_initial, n_points, max_iter=100, tol=1e-4):
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        
        # Define the objective function
        def objective(h):
            h_val = np.array(h)
            h_1_minus = 1 - h_val
            cross_corr = np.correlate(h_val, h_1_minus, mode="full")
            return cp.Max(cross_corr) * (2.0 / n_points)
        
        # Use a local optimizer
        result = optimize.minimize(
            lambda x: objective(x).value,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': max_iter, 'ftol': tol}
        )
        return result.x
    
    # Run the optimization
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    best_n_points = n_points
    
    # Try different numbers of points to find the best
    for n in [n_points, n_points - 10, n_points - 20, n_points - 30]:
        if n < 10:
            continue
        # Generate initial guess
        h = np.zeros(n)
        h[0::2] = 0.5
        h[1::2] = 1.0
        h = h[:n]
        h = h * (n / 2.0) / np.sum(h)
        # Optimize
        optimized_h = optimize_h(h, n)
        c5 = compute_c5(optimized_h)
        if c5 < best_c5:
            best_h = optimized_h
            best_c5 = c5
            best_n_points = n
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n_points
```
