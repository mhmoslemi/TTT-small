```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Pre-loaded initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set the number of points (smaller for faster optimization)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial construction
    h = np.copy(initial_h_values[:n_points])
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), h)
    h = np.clip(h, 0.0, 1.0)
    
    # Ensure the integral is 1
    sum_h = np.sum(h)
    if sum_h != n_points / 2:
        scale_factor = (n_points / 2) / sum_h
        h *= scale_factor
    
    # Define the function to minimize
    def objective(h):
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        integral = evaluate_erdos_solution(h)
        return integral
    
    # Optimization parameters
    max_iter = 100
    tolerance = 1e-4
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    # Perform optimization using gradient descent
    for _ in range(max_iter):
        # Evaluate current objective
        current_c5 = objective(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        # Perturb the solution
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        # Evaluate perturbed solution
        perturbed_c5 = objective(perturbed_h)
        
        # Decide whether to accept the perturbed solution
        if perturbed_c5 < current_c5:
            best_h = perturbed_h
        elif np.random.rand() < np.exp((current_c5 - perturbed_c5) / 0.1):
            best_h = perturbed_h
        
        # Check for early stopping
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
```
