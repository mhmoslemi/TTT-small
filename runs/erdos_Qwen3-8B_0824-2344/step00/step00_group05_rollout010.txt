import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions and variables
# evaluate_erdos_solution and initial_h_values are available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define initial configuration from initial_h_values
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values, dtype=np.float64)
    
    # Evaluate initial C5
    def compute_c5(h):
        dx = 2.0 / len(h)
        h_pad = np.zeros(len(h) * 2)
        h_pad[:len(h)] = h
        h_pad[len(h):] = h
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap * dx)
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Set up optimization with bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define a function to run the optimization
    def optimize_h(h_initial):
        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x
    
    # Run optimization
    best_h = h_values.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    elapsed_time = 0
    
    # Simple hill-climbing with random perturbations
    for _ in range(100):
        if elapsed_time >= budget_s:
            break
        # Perturb the current h_values
        h_perturbed = copy.deepcopy(best_h)
        idx = np.random.randint(n_points)
        h_perturbed[idx] += np.random.uniform(-0.1, 0.1)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Check constraint
        if np.abs(np.sum(h_perturbed) - n_points / 2) < 1e-6:
            current_c5 = compute_c5(h_perturbed)
            if current_c5 < best_c5:
                best_h = h_perturbed
                best_c5 = current_c5
        
        # Update elapsed time
        elapsed_time = time.time() - start_time
    
    return best_h.tolist(), best_c5, n_points
