```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load("initial_h.npy")  # Assuming a precomputed initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster computation
    
    # Function to evaluate a candidate h
    def evaluate_candidate(h):
        nonlocal best_c5
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Initial guess from the provided initial_h_values
    initial_h = initial_h_values.copy()
    if len(initial_h) > n_points:
        initial_h = initial_h[:n_points]
    else:
        # Pad with zeros if needed
        initial_h = np.pad(initial_h, (0, n_points - len(initial_h)), mode='constant')
    
    # Optimize using a gradient-based approach with constraints
    x = cp.Variable(n_points)
    objective = cp.Minimize(evaluate_erdos_solution(x))
    constraints = [
        cp.sum(x) == n_points / 2,  # Ensure integral is 1
        0 <= x,
        x <= 1
    ]
    
    prob = cp.Problem(objective, constraints)
    try:
        result = prob.solve(solver=cp.SCS, max_iter=100, time_limit=budget_s - (time.time() - start_time))
        if result is not None:
            best_h = x.value
    except:
        pass
    
    # If optimization fails, use initial guess
    if best_h is None:
        best_h = initial_h
    
    # Refine using local search
    for _ in range(5):  # A few rounds of refinement
        if time.time() - start_time >= budget_s:
            break
        # Perturb the solution slightly
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = np.concatenate([perturbed_h[:n_points // 2], perturbed_h[n_points // 2 + 1:]])
        perturbed_h = np.pad(perturbed_h, (0, n_points - len(perturbed_h)), mode='constant')
        c5 = evaluate_erdos_solution(perturbed_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = perturbed_h
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
```
