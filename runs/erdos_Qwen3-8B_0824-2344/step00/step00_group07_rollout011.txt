import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the overlap integral
    # Use convolution with mode='full' and then take the max
    conv = signal.correlate(h_values, 1 - h_values, mode='full')
    c5 = np.max(conv) * dx
    return c5

def initial_h_values():
    # Initial construction (n=80 samples)
    # This is an example; actual initial values will be provided by the environment
    return np.zeros(80)

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Start with the initial construction
    n_points = 80
    h_values = initial_h_values()
    
    # Try improving with smaller n_points first to save time
    for n in [40, 50, 60, 70, 80]:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Use a smaller number of samples
        h = np.zeros(n)
        # Randomly assign values to the h array
        # Ensure the sum is n/2 to satisfy the constraint
        # Use a simple random construction as a starting point
        # We will optimize this with a more efficient approach later
        
        # Simple random construction
        # Assign 1's and 0's randomly
        # Calculate how much we need to adjust to get the correct sum
        num_ones = int(n / 2)
        indices = np.random.choice(n, num_ones, replace=False)
        h[indices] = 1
        
        # Use scipy.optimize to minimize the max overlap
        def objective(x):
            h = x.reshape(n)
            conv = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(conv) * (2.0 / n)
            return c5
        
        # Constraints: sum(h) == n/2, 0 <= h <= 1
        bounds = [(0, 1) for _ in range(n)]
        constraints = [cp.sum(h) == n / 2]
        
        # Use CVXPY to solve the optimization problem
        h = cp.Variable(n, bounds=(0, 1))
        obj = cp.Minimize(objective(h.value))
        prob = cp.Problem(obj, constraints)
        try:
            result = prob.solve(solver=cp.SCS)
            if result is not None:
                h_values = h.value
                c5_bound = evaluate_erdos_solution(h_values)
        except:
            pass
        
        # After optimization, evaluate and track the best solution
        c5_bound = evaluate_erdos_solution(h_values)
        print(f"n={n}, C5={c5_bound}")
    
    # Final evaluation with the best solution found
    best_c5 = evaluate_erdos_solution(h_values)
    return (h_values, best_c5, len(h_values))
