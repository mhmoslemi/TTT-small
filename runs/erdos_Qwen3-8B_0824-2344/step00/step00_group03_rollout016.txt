import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Starting with 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points])  # Start with the initial construction
    
    # Ensure the initial h satisfies the constraint sum(h) == n_points / 2
    h = h / np.sum(h) * (n_points / 2)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = h.reshape(-1)
        h_shifted = np.zeros_like(h)
        for i in range(len(h)):
            h_shifted[i:] = h[:len(h)-i]
        shifted = np.roll(h_shifted, 1)  # Shift by 1
        overlap = np.correlate(h, 1 - shifted, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Optimization with L-BFGS-B for faster convergence
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        return compute_c5(h)
    
    def constraint(x):
        h = x.copy()
        return np.sum(h) - n_points / 2
    
    # Initial optimization
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        h,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": constraint}],
        method="L-BFGS-B",
        tol=1e-6,
        options={"maxiter": 100}
    )
    
    # Update h with the best result
    h = result.x
    c5 = result.fun
    
    # Try increasing the number of points if budget allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        n_points = 200
        dx = 2.0 / n_points
        h = np.array(initial_h_values[:n_points]) / np.sum(initial_h_values[:n_points]) * (n_points / 2)
        
        def compute_c5_200(h):
            h = h.reshape(-1)
            h_shifted = np.zeros_like(h)
            for i in range(len(h)):
                h_shifted[i:] = h[:len(h)-i]
            shifted = np.roll(h_shifted, 1)  # Shift by 1
            overlap = np.correlate(h, 1 - shifted, mode="full")
            c5 = np.max(overlap) * dx
            return c5
        
        def objective_200(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            return compute_c5_200(h)
        
        def constraint_200(x):
            h = x.copy()
            return np.sum(h) - n_points / 2
        
        result = opt.minimize(
            objective_200,
            h,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_200}],
            method="L-BFGS-B",
            tol=1e-6,
            options={"maxiter": 100}
        )
        
        h = result.x
        c5 = result.fun
    
    # Final check for best result
    h = np.clip(h, 0, 1)
    c5 = compute_c5(h)
    
    return h, c5, n_points
