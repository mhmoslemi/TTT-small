import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Start with the initial construction
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points
    sum_h = np.sum(h_values) * dx
    if not np.isclose(sum_h, 1.0):
        h_values = h_values * (1.0 / sum_h) * dx

    def compute_c5(h):
        h = np.array(h, dtype=float)
        dx = 2.0 / len(h)
        h = h * dx  # Scale to ensure the integral is 1
        h = np.clip(h, 0, 1)
        # Compute (1 - h[x + k])
        # To compute the correlation, we need to pad h with zeros
        h_padded = np.zeros(2 * len(h) - 1)
        h_padded[:len(h)] = h
        # Compute the convolution of h and (1 - h)
        conv = np.convolve(h, 1 - h, mode='full')
        # The maximum value corresponds to the maximum overlap
        c5 = np.max(conv)
        return c5

    # Optimization parameters
    max_iterations = 50
    learning_rate = 0.1
    temperature = 1.0
    cooling_rate = 0.99

    best_c5 = compute_c5(h_values)
    best_h = h_values.copy()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Create a copy of current h_values
        h = best_h.copy()
        # Add small random noise
        noise = np.random.normal(0, 0.05, len(h))
        h += noise
        h = np.clip(h, 0, 1)
        # Normalize to ensure the integral is 1
        h = h * (1.0 / (np.sum(h) * 2.0 / len(h)))

        # Compute the cost
        c5 = compute_c5(h)
        # Accept with probability based on simulated annealing
        if np.random.rand() < np.exp((best_c5 - c5) / temperature):
            best_c5 = c5
            best_h = h.copy()

        # Cool down the temperature
        temperature *= cooling_rate

    # Evaluate the best solution
    c5_bound = best_c5
    n_points = len(best_h)

    return (best_h, c5_bound, n_points)
