import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different n_points values starting from 100 (smaller than 1k)
    n_points_list = [100, 200, 300, 400, 500]
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Start with initial_h_values, adjusted to fit n_points
        if n_points == initial_h_values.shape[0]:
            h_initial = initial_h_values.copy()
        else:
            # Simple interpolation from initial_h_values to fit new n_points
            h_initial = np.interp(x, np.linspace(0, 2, initial_h_values.shape[0]), initial_h_values)
        
        # Ensure sum(h) is n_points / 2
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define optimization variables
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # Define objective function: minimize the maximum overlap of h with 1 - h shifted
        # We approximate the max using a convex relaxation (max over shifted versions)
        # Here we try a few shifts (k in [-1, 1])
        c5 = 0
        for k in range(-1, 2):
            shifted_h = np.roll(h.value, k) if k != 0 else h.value
            # Use a simple approximation of the integral using dx
            overlap = np.dot(h.value, 1 - shifted_h) * dx
            c5 = max(c5, overlap)
        
        # We minimize the max overlap using a convex relaxation
        # Here we use a simple approach to minimize the max by minimizing the max of several overlaps
        # This is an approximate and simplified method for demonstration
        overlap_values = []
        for k in range(-1, 2):
            shifted_h = np.roll(h, k) if k != 0 else h
            overlap = cp.sum(h * (1 - shifted_h)) * dx
            overlap_values.append(overlap)
        c5_obj = cp.Max(overlap_values)
        
        # Optimization problem
        prob = cp.Problem(cp.Minimize(c5_obj), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4)
            if prob.status == "optimal":
                h_opt = h.value
                # Evaluate the overlap integral
                c5_bound = 0
                for k in range(-1, 2):
                    shifted_h = np.roll(h_opt, k) if k != 0 else h_opt
                    overlap = np.dot(h_opt, 1 - shifted_h) * dx
                    c5_bound = max(c5_bound, overlap)
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_opt
                    best_n = n_points
        except:
            pass
        
        # Optionally use local optimization to fine-tune the solution
        # Use a local optimization method like SLSQP
        def objective_func(h_vec):
            h = h_vec.reshape(n_points)
            c5_bound = 0
            for k in range(-1, 2):
                shifted_h = np.roll(h, k) if k != 0 else h
                overlap = np.dot(h, 1 - shifted_h) * dx
                c5_bound = max(c5_bound, overlap)
            return c5_bound
        
        def constraint_func(h_vec):
            h = h_vec.reshape(n_points)
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        
        if result.success:
            h_opt = result.x
            c5_bound = 0
            for k in range(-1, 2):
                shifted_h = np.roll(h_opt, k) if k != 0 else h_opt
                overlap = np.dot(h_opt, 1 - shifted_h) * dx
                c5_bound = max(c5_bound, overlap)
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_opt
                best_n = n_points
    
    # Return the best solution found
    return (best_h, best_c5, best_n)
