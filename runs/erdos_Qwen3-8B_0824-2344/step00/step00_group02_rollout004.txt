```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Parameters
    max_points = 200  # Max number of points to try
    max_iterations = 100  # Max number of optimization iterations
    
    # Start timing
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Start from the initial construction
    h_initial = initial_h_values.copy()
    h_initial = np.array(h_initial)
    
    # Try varying number of points
    for n_points in range(50, max_points + 1, 50):
        dx = 2.0 / n_points
        h = h_initial.copy()
        h = np.resize(h, (n_points,))
        h /= np.sum(h) * dx  # Normalize to ensure integral is 1
        
        # Optimization: use genetic algorithm to find better h
        def objective_function(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h / (np.sum(h) * dx)  # Ensure integral is 1
            # Compute the overlap integral for all k
            overlap = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(overlap)
        
        # Use a genetic algorithm with some randomness
        result = optimize.differential_evolution(
            objective_function,
            bounds=[(0, 1) for _ in range(n_points)],
            strategy='best1bin',
            maxiter=max_iterations,
            popsize=15,
            tol=1e-4,
            mutation=(0.5, 1),
            recombination='binomial',
            seed=seed
        )
        
        h_optimized = result.x
        c5_bound = result.fun
        
        # Update best solution
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_optimized.copy()
            print(f"Best C5 found: {best_c5:.6f} (n_points={n_points})")
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure the final h is properly scaled and valid
    final_h = np.array(best_h)
    final_h = np.clip(final_h, 0, 1)
    final_h /= np.sum(final_h) * (2.0 / len(final_h))  # Ensure integral is 1
    
    return final_h, final_c5, len(final_h)
```
