```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assuming initial construction is saved

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Starting with 100 points
    dx = 2.0 / n_points
    initial_h = initial_h_values[:n_points]  # Use only the first n_points samples if available
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    best_n_points = n_points
    
    # Time tracking
    start_time = time.time()
    
    # Try increasing number of points up to 500
    for n_points in range(100, 501, 50):
        dx = 2.0 / n_points
        h = np.random.rand(n_points) * 0.5  # Initial random guess
        # Ensure the integral constraint is satisfied
        h = h * (n_points / 2) / np.sum(h)
        
        # Optimization: minimize max overlap
        # We'll use a convex relaxation with a tolerance
        eps = 1e-3
        max_overlap = cp.Variable()
        constraints = [max_overlap >= correlate(h, 1 - h, mode="full") * dx]
        
        # Solve the optimization problem
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        try:
            prob.solve(solver='ECOS', verbose=False)
            if prob.status == 'OPTIMAL':
                c5 = max_overlap.value
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n_points = n_points
        except:
            pass
        
        # Early exit if time is up
        if time.time() - start_time > budget_s:
            break
    
    # Evaluate the best solution found
    best_c5 = evaluate_erdos_solution(best_h)
    return (best_h, best_c5, best_n_points)
```
