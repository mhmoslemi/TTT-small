import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, n_points=200):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Define the number of points
    n = n_points
    dx = 2.0 / n
    
    # Initial guess (start with the provided initial solution)
    h = initial_h_values[:n]  # Ensure we do not exceed the length of initial_h_values
    
    # Ensure the initial solution satisfies the constraint sum(h) == n / 2
    # If initial_h_values is longer than n, take the first n values
    if len(h) > n:
        h = h[:n]
    # If initial_h_values is shorter than n, pad with zeros
    elif len(h) < n:
        h = np.pad(h, (0, n - len(h)), mode='constant')
    
    # Enforce the constraint that sum(h) == n / 2
    # This is done by adjusting the last value
    sum_h = np.sum(h)
    if abs(sum_h - n / 2) > 1e-6:
        diff = (n / 2) - sum_h
        h[-1] += diff
    
    # Define the function to minimize
    def objective(h):
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        h_shifted = np.roll(h, 1)
        overlap = np.correlate(h, 1 - h_shifted, mode='full')
        c5 = np.max(overlap * dx)
        return c5
    
    # Define the bounds for h
    bounds = Bounds(0, 1)
    
    # Set up optimization problem
    # We use the L-BFGS-B optimizer for efficiency
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        jac=None,  # No analytical gradient
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Extract the best solution found
    best_h = result.x
    # Ensure the constraint is satisfied
    sum_best_h = np.sum(best_h)
    if abs(sum_best_h - n / 2) > 1e-6:
        # Adjust the last value to meet the constraint
        diff = (n / 2) - sum_best_h
        best_h[-1] += diff
    
    # Compute the final C5 value
    best_h = np.clip(best_h, 0, 1)
    h_shifted = np.roll(best_h, 1)
    overlap = np.correlate(best_h, 1 - h_shifted, mode='full')
    c5_bound = np.max(overlap * dx)
    
    # Return the best solution and the computed C5 value
    return (best_h, c5_bound, n)
