```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try with different numbers of points
    n_points_list = [300, 200, 150, 120, 100]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        total = n_points / 2
        h[0] = 1.0
        
        for i in range(1, n_points):
            # Randomly set h[i] to 0 or 1
            if np.random.rand() < 0.5:
                h[i] = 1.0
                total -= 1.0
            else:
                h[i] = 0.0
        
        # Adjust to meet the constraint
        if total > 0:
            remaining = total
            idxs = np.where(h == 0)[0]
            if len(idxs) > 0:
                for idx in idxs[:int(remaining)]:
                    h[idx] = 1.0
        elif total < 0:
            remaining = -total
            idxs = np.where(h == 1)[0]
            if len(idxs) > 0:
                for idx in idxs[:int(remaining)]:
                    h[idx] = 0.0
        
        def objective(x):
            h = x
            if np.any(h < 0) or np.any(h > 1):
                return float('inf')
            # Compute the overlap integral
            overlap = correlate(h, 1 - h, mode="full") * dx
            return np.max(overlap)
        
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        ]
        
        result = minimize(
            objective,
            h,
            method='SLSQP',
            constraints=constraints,
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        if result.success:
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                best_n = n_points
    
    # Fallback to initial_h_values if no improvement found
    if best_h is None or best_c5 > evaluate_erdos_solution(initial_h_values):
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
        best_n = len(best_h)
    
    return best_h, best_c5, best_n
```
