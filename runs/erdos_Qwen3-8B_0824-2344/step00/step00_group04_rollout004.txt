def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.pad(h_values, (0, 1), mode='constant', constant_values=0)
    h_1_minus = 1 - h
    corr = np.correlate(h, h_1_minus, mode='full')
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()

    h = np.array(initial_h_values, dtype=float)
    best_c5 = evaluate_erdos_solution(h)

    dx = 2.0 / n_points

    n_points_list = [50, 60, 70, 80, 90, 100]
    
    for n in n_points_list:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create a new h with n points
        h_new = np.zeros(n)
        # Start with a simple pattern
        h_new[:(n//2)] = 1.0
        h_new[(n//2):] = 0.0
        # Ensure the integral is 1
        total = np.sum(h_new) * dx
        h_new = h_new * (1.0 / total) * (n / 2)
        
        # Evaluate initial c5
        c5 = evaluate_erdos_solution(h_new)
        if c5 < best_c5:

    # Try to optimize the best_h
    h_opt = best_h.copy()
    n_opt = len(h_opt)
    dx_opt = 2.0 / n_opt
    
    # Optimization problem: minimize max_k ∫ h(x)(1 - h(x+k)) dx
    # Convert to discrete variables
    h = cp.Variable(n_opt)
    constraints = [0 <= h, h <= 1]
    constraints.append(cp.sum(h) == n_opt / 2)
    
    # Objective: minimize the maximum overlap
    overlap = np.zeros(n_opt * 2)
    for k in range(1, n_opt):
        if k >= n_opt:
            break
        for i in range(n_opt):
            if i + k < n_opt:
                overlap[k + i] += h[i] * (1 - h[i + k])
            else:
                overlap[k + i] += h[i] * (1 - h[i + k - n_opt])
    
    max_overlap = cp.Max(overlap)
    problem = cp.Problem(cp.Minimize(max_overlap), constraints)
    
    try:
        problem.solve(solver=cp.SCS)
        if problem.status == 'optimal':
            h_opt = np.array(h.value)
            c5_opt = evaluate_erdos_solution(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, len(best_h))
