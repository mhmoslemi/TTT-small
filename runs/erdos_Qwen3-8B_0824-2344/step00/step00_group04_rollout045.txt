import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points values, starting with 200 (smaller is faster to optimize)
    n_points = 200
    dx = 2.0 / n_points
    
    # Initial construction
    h = np.array(initial_h_values)
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = np.pad(h, (0, 1), mode='constant')  # For full convolution
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    # Function to optimize h
    def optimize_h(h, budget_s):
        start_time = time.time()
        
        # Try local optimization using gradient descent
        def objective(h_vec):
            h = h_vec.reshape(n_points)
            return compute_c5(h)
        
        def grad_objective(h_vec):
            h = h_vec.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = np.pad(h, (0, 1), mode='constant')
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr * dx)
            # Compute gradient numerically
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                obj_plus = compute_c5(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                obj_minus = compute_c5(h_minus)
                grad[i] = (obj_plus - obj_minus) / (2 * 1e-5)
            return grad.flatten()
        
        # Optimization
        h_opt = h.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            grad = grad_objective(h_opt)
            h_opt -= 0.01 * grad
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = (h_opt * n_points / 2).astype(np.float64)
            h_opt /= (n_points / 2)
        
        # Try CVXPY optimization
        if time.time() - start_time < budget_s * 0.8:
            # CVXPY setup
            h_var = cp.Variable(n_points)
            constraints = [
                cp.sum(h_var) == n_points / 2,
                0 <= h_var,
                h_var <= 1
            ]
            objective = cp.Maximize(cp.min(cp.multiply(h_var, 1 - cp.roll(h_var, 1))))
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                h_opt = h_var.value
            except:
                pass
        
        return h_opt
    
    # Optimize the initial construction
    h = optimize_h(h, budget_s)
    
    # Compute the final C5 value
    c5_bound = compute_c5(h)
    
    return h, c5_bound, n_points
