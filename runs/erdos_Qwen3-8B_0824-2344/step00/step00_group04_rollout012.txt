```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points
    max_points = 1000  # Max points to try
    
    # Function to compute the overlap integral
    def compute_C5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h_extended = np.concatenate([h, np.zeros_like(h)])
        result = np.correlate(h_extended, 1 - h, mode="full")
        return np.max(result * dx)
    
    # Function to generate a random initial solution
    def generate_initial_solution(n_points):
        h = np.random.rand(n_points)
        h = (h / np.sum(h) * n_points / 2).astype(np.float64)
        return h
    
    # Start with the initial solution
    h = initial_h_values
    if len(h) < n_points:
        h = np.pad(h, (0, n_points - len(h)), 'constant')
    elif len(h) > n_points:
        h = h[:n_points]
    h = h[:n_points]
    current_c5 = compute_C5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Try increasing number of points
    for n_points in range(100, max_points + 1, 50):
        # Generate a new solution with this number of points
        h_new = generate_initial_solution(n_points)
        h_new = h_new / np.sum(h_new) * n_points / 2
        
        # Compute C5
        c5 = compute_C5(h_new)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    
    # Run optimization with CVXPY for the best n_points
    n_points = len(best_h)
    h = cp.Variable(n_points)
    dx = 2.0 / n_points
    objective = cp.Minimize(cp.Max(cp.sum(h * (1 - h[1:]))) * dx)
    constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == 'OPTIMAL':
            best_h = h.value
            best_c5 = compute_C5(best_h)
    except:
        pass
    
    # Final check
    best_c5 = compute_C5(best_h)
    return best_h.tolist(), best_c5, len(best_h)
```
