import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    # Compute the overlap integrals and return the maximum
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros(2 * n_points)
    shifted_h[:n_points] = h
    shifted_h[n_points:] = h[::-1]  # Mirror the function
    # Compute the product (1 - h(x + k)) for all k
    one_minus_h = 1 - h
    full_convolution = signal.correlate(one_minus_h, h, mode='full')
    # Scale by dx and take the maximum
    c5 = np.max(full_convolution * dx)
    return c5

def initial_h_values():
    # Returns an initial guess for h_values (n=70 samples)
    return np.load('initial_h.npy')

# Strategy:
# 1. Start with an initial guess from the initial_h_values function
# 2. Use a local optimization approach to minimize the maximum overlap integral
# 3. Try various initializations and perturbations to avoid local optima
# 4. Return the best solution found within the time budget

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Try multiple initial points
    for i in range(3):
        # Start with the initial guess
        initial_h = initial_h_values()
        n_points = len(initial_h)
        h = initial_h.copy()
        # Perturb slightly to avoid local optima
        h += np.random.uniform(-0.05, 0.05, size=n_points)
        h = np.clip(h, 0, 1)

        # Optimize using a local optimization approach
        def objective(x):
            # x is a vector of h values
            h = x.copy()
            h = np.clip(h, 0, 1)
            # Compute the overlap integral
            one_minus_h = 1 - h
            full_convolution = signal.correlate(one_minus_h, h, mode='full')
            c5 = np.max(full_convolution * (2.0 / n_points))
            return c5

        # Define bounds and constraints
        bounds = Bounds(0, 1)
        constraints = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])

        # Run optimization
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-8, 'maxiter': 1000}
        )

        # Check if the optimization succeeded

        # Optimize using a local optimization approach

            one_minus_h = 1 - h
            full_convolution = signal.correlate(one_minus_h, h, mode='full')
