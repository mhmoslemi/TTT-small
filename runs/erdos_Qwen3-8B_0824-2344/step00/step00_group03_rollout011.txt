```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100
    h = initial_h_values.copy()
    h = h[:n_points]
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = (h * (n_points / 2) / sum_h)
    
    def compute_c5(h_values):
        h = h_values.copy()
        h = h[:n_points]
        dx = 2.0 / n_points
        conv = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    def objective(x):
        return compute_c5(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Optimization: use gradient descent with constraints
    x0 = h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-6, 'disp': False}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Check if we have time left to further refine
    if time.time() - start_time < budget_s - 2:
        # Use a small perturbation to avoid getting stuck
        perturbation = 0.01 * np.random.randn(n_points)
        perturbation = np.clip(perturbation, -0.05, 0.05)
        best_h += perturbation
        best_h = np.clip(best_h, 0, 1)
        best_h = (best_h * (n_points / 2)) / np.sum(best_h)
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
```
