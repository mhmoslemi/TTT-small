```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported helper functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Initial guess from pre-imported values
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values, dtype=np.float64)
    h_initial = h_initial / np.sum(h_initial) * (n_points_initial / 2)
    
    # Try different numbers of points, starting from the initial one
    for n_points in [n_points_initial, n_points_initial - 2, n_points_initial - 4, n_points_initial - 6]:
        if n_points < 6:
            continue
        
        # Define the grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Construct the initial guess
        h = h_initial[:n_points].copy()
        h = h / np.sum(h) * (n_points / 2)
        
        # Define the objective function (max overlap integral)
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            # Compute 1 - h(x + k)
            # Use padding to handle wrap-around
            h_shifted = np.zeros((n_points, n_points))
            for i in range(n_points):
                h_shifted[i, :] = h[(i + np.arange(n_points)) % n_points]
            # Compute the correlation
            corr = np.correlate(h, 1 - h, mode="full")
            # Multiply by dx to get the integral
            overlap = corr * dx
            # Return the maximum overlap
            return np.max(overlap)
        
        # Define constraints
        def constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - (n_points / 2)
        
        # Optimization with L-BFGS-B
        result = opt.minimize(
            fun=objective_func,
            x0=h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            tol=1e-8,
            options={'maxiter': 1000, 'disp': False}
        )
        
        if result.success:
            h_opt = result.x
            c5 = objective_func(h_opt)
            
            # Early exit if we have enough time
            if time.time() - start_time > budget_s - 5:
                break
            
            # Check if this is the best solution so far
            print(f"n_points={n_points}, C5={c5}")
            return (h_opt.tolist(), c5, n_points)
    
    # If no better solution found, return the initial one
    return (h_initial.tolist(), evaluate_erdos_solution(h_initial), len(h_initial))
```
