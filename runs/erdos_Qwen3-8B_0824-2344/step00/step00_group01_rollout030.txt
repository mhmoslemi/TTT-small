```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import matplotlib.pyplot as plt

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    best_n_points = 0

    # Try initial solution
    initial_h = initial_h_values
    n_points = len(initial_h)
    c5 = evaluate_erdos_solution(initial_h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = initial_h.copy()
        best_n_points = n_points

    # Try smaller n_points
    for n_points in range(100, 300, 50):
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n_points

    # Try more refined optimization
    for n_points in [200, 150, 100]:
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = h.copy()

        # Perform gradient descent with some regularization
        for _ in range(200):
            # Compute overlaps
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx

            # Compute gradient
            # We approximate the gradient numerically
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / 1e-5

                h_minus = h.copy()
                h_minus[i] -= 1e-5
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full")
                c5_minus = np.max(corr_minus) * dx
                grad[i] = (c5 - c5_minus) / 1e-5

            # Update h
            h -= 0.1 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

            # Early stopping
            if time.time() - start_time > budget_s - 1:
                break

        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n_points

    # Return best solution
    return best_h, best_c5, best_n_points
```
