```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial number of points
    n_points = 50
    max_n_points = 1000  # Upper limit for n_points to keep within time budget
    
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h)):
            shifted[k:] = h[:-k]
        overlap = np.sum(h * (1 - shifted)) * dx
        return overlap
    
    def objective_function(h, n_points):
        h = np.array(h)
        # Ensure h is within [0,1]
        h = np.clip(h, 0, 1)
        # Enforce integral condition
        integral = np.sum(h) * (2.0 / n_points)
        # Small tolerance to avoid division by zero
        if abs(integral - 1.0) > 1e-6:
            return np.inf
        # Compute all possible overlaps
        overlaps = []
        for k in range(1, n_points):
            shifted = np.zeros_like(h)
            shifted[k:] = h[:-k]
            overlap = np.sum(h * (1 - shifted)) * (2.0 / n_points)
            overlaps.append(overlap)
        # Return the maximum overlap
        return max(overlaps)
    
    def optimize_h(n_points):
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * (2.0 / n_points)
        if abs(integral - 1.0) > 1e-6:
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            integral = np.sum(h) * (2.0 / n_points)
            if abs(integral - 1.0) > 1e-6:
                h = np.array([1.0 / n_points] * n_points)
        # Use differential evolution for optimization
        result = optimize.differential_evolution(
            lambda x: objective_function(x, n_points),
            bounds=[(0, 1) for _ in range(n_points)],
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            disp=False,
            callback=lambda x: None
        )
        best_h = result.x
        best_c5 = compute_c5(best_h)
        return best_h, best_c5
    
    # Start with initial_h_values
    h_initial = initial_h_values
    n_initial = len(h_initial)
    c5_initial = compute_c5(h_initial)
    
    # Try increasing n_points up to max_n_points
    best_h = h_initial
    best_c5 = c5_initial
    best_n_points = n_initial
    
    for n_points in range(n_initial + 1, max_n_points + 1):
        if time.time() - start_time > budget_s - 1:
            break
        print(f"Trying n_points={n_points}")
        h_optimized, c5_optimized = optimize_h(n_points)
        if c5_optimized < best_c5:
            best_h = h_optimized
            best_c5 = c5_optimized
            best_n_points = n_points
    
    # Final check with the initial h
    if compute_c5(best_h) < best_c5:
        best_c5 = compute_c5(best_h)
    
    # Evaluate the best solution
    evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, best_n_points
```
