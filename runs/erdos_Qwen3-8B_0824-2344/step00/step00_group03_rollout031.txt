import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions and variables
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Strategy:
# 1. Start from the initial_h_values and use a greedy approach to refine it.
# 2. Use a local optimization strategy to minimize the overlap integral C5.
# 3. Use a stepwise refinement to improve the solution iteratively within the time budget.

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial solution
    h_values = initial_h_values.copy()
    n_points = len(h_values)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    # Function to compute overlap integral for a given shift k
    def compute_overlap(h, k):
        h_k = np.roll(h, k)
        return np.sum(h * (1 - h_k)) * dx
    
    # Function to compute C5 for all k
    def compute_all_overlaps(h):
        n_points = len(h)
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            overlaps[k] = compute_overlap(h, k)
        return np.max(overlaps)
    
    # Local optimization using gradient descent
    def optimize_h(h, max_iter=100, learning_rate=0.01):
        h = h.copy()
        for _ in range(max_iter):
            # Compute overlaps
            overlaps = compute_all_overlaps(h)
            # Find the k that maximizes the overlap
            max_k = np.argmax(overlaps)
            # Adjust h to reduce the overlap at max_k
            # Shift the function to reduce overlap at max_k
            h_shifted = np.roll(h, max_k)
            # Compute gradient
            grad = (h - h_shifted) * (1 - h_shifted)
            # Update h
            h = h - learning_rate * grad
            # Ensure h stays within [0, 1]
            h = np.clip(h, 0, 1)
        return h
    
    # Refine the initial solution
    start_time = time.time()
    for _ in range(5):
        h_values = optimize_h(h_values)
        current_c5 = compute_c5(h_values)
        print(f"Current C5: {current_c5}")
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = compute_c5(h_values)
    print(f"Final C5: {final_c5}")
    
    return h_values, final_c5, n_points
