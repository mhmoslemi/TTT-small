import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Start timing
    start_time = time.time()
    
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Number of samples
    n_points = 100  # A reasonable size for speed and accuracy
    
    # Initialize h with the initial guess
    h = np.array(initial_h_values[:n_points])
    h = h / np.sum(h) * (n_points / 2)  # Normalize to meet the integral constraint
    
    # Function to evaluate C5
    def evaluate_C5(h):
        h = np.array(h)
        # Normalize to meet the integral constraint
        h = h / np.sum(h) * (n_points / 2)
        # Compute the overlap integral
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Function to minimize C5
    def objective(x):
        return evaluate_C5(x)
    
    # Bounds for each h[i] (0 to 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        tol=1e-6,
        options={'maxiter': 100, 'disp': False}
    )
    
    # Extract best solution
    best_h = result.x
    
    # Evaluate the best solution
    best_c5 = evaluate_C5(best_h)
    
    # Check if we have time left to refine further
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10:
        # Perform a few more iterations with a smaller tolerance
        result = optimize.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            tol=1e-8,
            options={'maxiter': 50, 'disp': False}
        )
        best_h = result.x
        best_c5 = evaluate_C5(best_h)
    
    return (best_h, best_c5, n_points)
