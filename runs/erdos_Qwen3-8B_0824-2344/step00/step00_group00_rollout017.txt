```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Import helper functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Start with the initial n_points
    
    # Use the initial solution as a starting point
    h_values = np.array(initial_h_values)
    
    # Check if the initial solution already satisfies the integral constraint
    if np.abs(np.sum(h_values) - n_points / 2) > 1e-6:
        # Scale to meet the constraint
        h_values = h_values / np.sum(h_values) * (n_points / 2)
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Function to compute the objective and constraints
    def objective(h):
        return evaluate_overlap(h)
    
    # Optimizer settings
    max_iter = 100
    tolerance = 1e-5
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Local optimization using scipy's minimize
        res = opt.minimize(
            fun=objective,
            x0=h_values,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=tolerance,
            options={'maxiter': 10}
        )
        
        h_values = res.x
        
        # Check if the integral constraint is satisfied
        if np.abs(np.sum(h_values) - n_points / 2) > 1e-6:
            h_values = h_values / np.sum(h_values) * (n_points / 2)
        
        # Evaluate the current solution
        c5_bound = evaluate_overlap(h_values)
        
        # Consider increasing n_points if time allows
        if time.time() - start_time < budget_s - 5:
            n_points += 10
            h_values = np.linspace(0, 2, n_points)
            h_values = h_values / np.sum(h_values) * (n_points / 2)
    
    # Final evaluation
    c5_bound = evaluate_overlap(h_values)
    return h_values, c5_bound, n_points
```
