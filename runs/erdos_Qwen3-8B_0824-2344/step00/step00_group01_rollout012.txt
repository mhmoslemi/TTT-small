import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Starting with n_points = 100 for speed
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess from provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]  # Truncate to match n_points
    
    # Ensure sum(h_initial) * dx == 1
    sum_h = np.sum(h_initial)
    if abs(sum_h * dx - 1) > 1e-8:
        # Normalize if needed
        h_initial = h_initial * (1 / (sum_h * dx))
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Ensure h is within [0,1]
        h = np.clip(h, 0, 1)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Normalize by dx
        c5 = np.max(corr * dx)
        return c5
    
    # Optimization loop
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    
    # Try increasing n_points if time allows
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (1 / (np.sum(h) * dx))  # Normalize
        
        # Use a simple gradient descent approach to minimize c5
        max_iterations = 100
        learning_rate = 0.01
        for _ in range(max_iterations):
            # Compute current c5
            current_c5 = compute_c5(h)
            
            # If current c5 is better than best, update
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Compute the gradient numerically (simplified)
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = np.clip(h_plus, 0, 1)
                grad[i] = (compute_c5(h_plus) - current_c5) / 1e-5
            
            # Update h in the direction of the negative gradient
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))  # Normalize
        
        # Evaluate best so far
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Return the best solution found
    return best_h, final_c5, n_points
