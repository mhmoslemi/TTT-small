import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define number of points
    n_points = 1000  # We will reduce if needed
    dx = 2.0 / n_points
    
    # Initial guess
    h = np.array(initial_h_values[:n_points])  # Use initial construction
    h = h / (dx * n_points) * n_points / 2  # Normalize to satisfy the integral constraint
    h = np.clip(h, 0, 1)
    
    # Function to evaluate overlap integral
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        h = h.reshape(-1, 1)
        one_minus_h = 1 - h
        overlap = np.correlate(h.flatten(), one_minus_h.flatten(), mode="full")
        return np.max(overlap) * dx
    
    # Optimization via gradient descent with constraints
    def objective_func(h):
        return evaluate_overlap(h)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Gradient descent with constraints
    h_opt = np.copy(h)
    learning_rate = 0.05
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute overlap
        c5 = evaluate_overlap(h_opt)
        
        # Compute gradient (approximate)
        grad = np.zeros_like(h_opt)
        for i in range(n_points):
            h_plus = h_opt.copy()
            h_plus[i] += 1e-5
            c5_plus = evaluate_overlap(h_plus)
            grad[i] = (c5_plus - c5) / 1e-5
        
        # Compute constraint gradient
        constraint_grad = np.ones_like(h_opt)
        
        # Update with gradient descent
        h_opt -= learning_rate * grad
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)  # Enforce constraint
        
        # Early stopping if improvement is minimal
        if _ > 100 and np.abs(evaluate_overlap(h_opt) - evaluate_overlap(h_opt - grad * learning_rate)) < 1e-6:
            break
    
    # Final evaluation
    best_h = h_opt
    best_c5 = evaluate_overlap(best_h)
    
    # Reduce n_points if possible
    if n_points > 200:
        n_points = 200
        dx = 2.0 / n_points
        best_h = np.resize(best_h, n_points)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_h = np.clip(best_h, 0, 1)
        best_c5 = evaluate_overlap(best_h)
    
    return (best_h, best_c5, n_points)
