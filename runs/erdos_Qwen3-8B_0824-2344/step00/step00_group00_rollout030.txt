```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = 45
    h_values = initial_h_values.copy()
    dx = 2.0 / n_points
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h_expanded = np.concatenate([h, np.zeros_like(h)])
        overlap = signal.correlate(h_expanded, 1 - h, mode='full')
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to evaluate the objective and constraints
    def objective_function(h):
        return compute_c5(h)
    
    # Function to enforce constraints
    def constraint_function(h):
        return np.sum(h) - (n_points / 2)
    
    # Initial optimization with scipy
    def optimize_with_scipy(h_initial):
        result = opt.minimize(
            objective_function,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(len(h_initial))],
            options={'maxiter': 100}
        )
        return result.x
    
    # Refine with CVXPY for better accuracy
    def optimize_with_cvxpy(h_initial):
        h = cp.Variable(len(h_initial))
        objective = cp.Maximize(cp.max(signal.correlate(h, 1 - h, mode='full') * (2.0 / len(h))))
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.OSQP, eps=1e-6)
            return h.value
        except:
            return h_initial
    
    # Refine with multiple iterations
    h_best = h_values
    c5_best = compute_c5(h_best)
    for _ in range(3):
        h_best = optimize_with_scipy(h_best)
        c5_best = compute_c5(h_best)
        h_best = optimize_with_cvxpy(h_best)
        c5_best = compute_c5(h_best)
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = compute_c5(h_best)
    return h_best, final_c5, len(h_best)
```
