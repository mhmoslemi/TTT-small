import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Initial parameters
    n_points = 95  # Starting with the initial construction
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)  # Initial guess
    h /= np.sum(h) * dx  # Normalize to ensure the integral is 1
    h = np.clip(h, 0, 1)  # Ensure values are within [0,1]

    start_time = time.time()

    # Define objective function for optimization
    def objective(x):
        nonlocal h
        h = x.copy()
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1) > 1e-8:
            return np.inf  # Penalize for not satisfying the constraint
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)
        return c5

    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    # If optimization failed, return the initial solution
    if result.success:
        h_opt = result.x
    else:
        h_opt = h

    # Evaluate the solution
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / (np.sum(h_opt) * dx)  # Normalize again in case of numerical issues
    h_opt = np.clip(h_opt, 0, 1)
    c5_bound = evaluate_erdos_solution(h_opt)

    # If within budget, try to refine with more points
    if time.time() - start_time < budget_s - 5:
        n_points = 127  # Try a slightly larger number of points
        dx = 2.0 / n_points
        # Create a new initial guess with more points
        h_new = np.zeros(n_points)
        h_new[:n_points // 2] = 1  # Start with a simple binary-like pattern
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / (np.sum(h_new) * dx)  # Normalize
        h_new = np.clip(h_new, 0, 1)

        def objective_new(x):
            nonlocal h_new
            h_new = x.copy()
            h_new = np.clip(h_new, 0, 1)
            sum_h = np.sum(h_new) * dx
            if abs(sum_h - 1) > 1e-8:
                return np.inf
            overlap = np.correlate(h_new, 1 - h_new, mode="full") * dx
            c5 = np.max(overlap)
            return c5

        bounds_new = [(0, 1) for _ in range(n_points)]
        result_new = opt.minimize(
            objective_new,
            x0=h_new,
            bounds=bounds_new,
            method='L-BFGS-B',
            jac=None,
            options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
        )

        if result_new.success:
            h_opt = result_new.x
        else:
            h_opt = h_new

        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt / (np.sum(h_opt) * dx)
        h_opt = np.clip(h_opt, 0, 1)
        c5_bound = evaluate_erdos_solution(h_opt)

    # Final check and return
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / (np.sum(h_opt) * dx)
    h_opt = np.clip(h_opt, 0, 1)
    c5_bound = evaluate_erdos_solution(h_opt)

    return h_opt, c5_bound, n_points
