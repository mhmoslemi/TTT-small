import numpy as np

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Number of samples
    n_points = 100  # Start with 100 samples, adjust if needed
    
    # Define the initial guess
    initial_h = np.array(initial_h_values[:n_points]) if len(initial_h_values) > n_points else np.random.rand(n_points)
    initial_h = (initial_h * 2 - 1) * 0.5 + 0.5  # Normalize to [0, 1]
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2)  # Ensure the integral is 1
    
    # Define dx
    dx = 2.0 / n_points
    
    def objective(h):
        # Compute the overlap integral
        h = h.reshape(-1)
        h_extended = np.concatenate([h, h, h])
        overlap = np.correlate(h_extended, 1 - h, mode='full')
        return np.max(overlap * dx)
    
    # Define constraints
    def constraints(h):
        # Ensure the integral is 1
        integral = np.sum(h) * dx
        return np.array([integral - 1.0])
    
    # Optimization using scipy.optimize
    result = optimize.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    # Get the best solution found
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2)  # Ensure the integral is 1
    
    # Compute the best C5 bound
    best_c5 = objective(best_h)
    
    # Return the result
    return (best_h, best_c5, n_points)
