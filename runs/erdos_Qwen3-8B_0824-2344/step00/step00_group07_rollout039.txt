```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 80  # Start with 80 samples as the initial construction
    
    # Try different n_points values to find a balance between accuracy and computation time
    for n_points in [80, 160, 320, 400]:
        if time.time() - start_time > budget_s - 2:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)
        
        # Generate initial guess
        h_initial = initial_h_values.copy()
        h_initial = np.resize(h_initial, (n_points,))
        
        # Bounds for h[i]
        h_low = np.zeros(n_points)
        h_high = np.ones(n_points)
        
        # Optimization variables
        h = cp.Variable(n_points, nonnegativ=True, bounded=(h_low, h_high))
        
        # Constraint: integral of h over [0, 2] is 1
        constraint = cp.sum(h) == n_points / 2
        
        # Objective: minimize the maximum overlap integral
        # Compute all shifts k in [-n_points + 1, n_points - 1]
        max_shift = n_points - 1
        shifts = np.arange(-max_shift, max_shift + 1)
        overlap_integrals = []
        
        for k in shifts:
            if k < 0:
                shifted_h = np.roll(h.value, k)
            else:
                shifted_h = np.roll(h.value, k)
            overlap = cp.sum(h * (1 - shifted_h)) * dx
            overlap_integrals.append(overlap)
        
        # Objective is to minimize the maximum overlap integral
        objective = cp.Minimize(cp.Max(overlap_integrals))
        
        # Solve the optimization problem
        problem = cp.Problem(objective, [constraint])
        try:
            result = problem.solve(solver=cp.SCS, verbose=False, max_iters=100)
        except:
            continue
        
        if result is not None and result < best_c5:
            best_c5 = result
            best_h = h.value
            print(f"n_points={n_points}, C5={best_c5}")
    
    # Convert the best_h to a numpy array for output
    best_h_values = best_h if best_h is not None else initial_h_values
    best_h_values = np.resize(best_h_values, (n_points,))
    
    return (best_h_values, best_c5, n_points)
```
