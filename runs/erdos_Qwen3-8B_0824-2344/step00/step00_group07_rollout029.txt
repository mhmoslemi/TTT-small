import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    h_values = initial_h_values.copy()
    n_points = len(h_values)
    dx = 2.0 / n_points
    
    # Function to compute C_5
    def compute_c5(h):
        h = np.array(h)
        h_shifted = np.roll(h, shift=1, axis=0)
        overlap = h * (1 - h_shifted)
        return np.max(overlap) * dx
    
    # Function to evaluate the solution
    def evaluate(h):
        return compute_c5(h)
    
    # Function to compute the overlap integral for all shifts
    def compute_overlap_integral(h):
        h = np.array(h)
        n = len(h)
        overlap = np.zeros(n)
        for k in range(1, n):
            h_k = np.roll(h, shift=k, axis=0)
            overlap_k = h * (1 - h_k)
            overlap += np.max(overlap_k) * dx
        return np.max(overlap)
    
    # Function to optimize with CVXPY
    def optimize_h():
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        obj = cp.Minimize(cp.Max(cp.multiply(h, 1 - cp.roll(h, shift=1, axis=0))) * dx)
        prob = cp.Problem(obj, constraints)
        prob.solve(solver='ECOS')
        return h.value
    
    # Try different numbers of points to find the best solution within budget
    best_c5 = float('inf')
    best_h = h_values
    best_n_points = n_points
    
    # Try smaller grids first for faster evaluation
    for n_points in [100, 200, 400, 800]:
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h_values = h.copy()
        
        # Perform a few iterations of optimization
        for _ in range(5):
            h_vals = optimize_h()
            h_values = h_vals
            c5 = compute_c5(h_values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values.copy()
                best_n_points = n_points
        
        if time.time() - start_time > budget_s - 1:
            break
    
    # Evaluate the best solution
    final_c5 = compute_overlap_integral(best_h)
    
    return (best_h, final_c5, best_n_points)
