```python
import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points

    # Function to evaluate the C5 value for a given h
    def evaluate_c5(h):
        dx = 2.0 / n_points
        h_array = np.array(h)
        shifted = np.zeros_like(h_array)
        for k in range(1, n_points):
            shifted[k:] = h_array[:-k]
        overlaps = np.correlate(h_array, 1 - h_array, mode="full")
        overlap_values = overlaps[n_points - 1 : -n_points + 1]  # Only valid shifts
        c5_current = np.max(overlap_values) * dx
        return c5_current

    # Function to compute the integral constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Optimization with scipy
    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        result = optimize.minimize(
            lambda h: evaluate_c5(h),
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            tol=1e-5,
            options={'maxiter': 100}
        )
        return result.x

    # Start with the initial guess
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]  # Ensure it's the right length
    h_initial = np.array(h_initial)
    h_initial = optimize_h(h_initial)

    # Evaluate the result
    c5_current = evaluate_c5(h_initial)
    best_c5 = c5_current
    best_h = h_initial

    # Try increasing the number of points if time allows
    while time.time() - start_time < budget_s - 5:
        n_points += 20
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2)  # Scale to meet the integral constraint
        h_initial = h_initial[:n_points]
        h_initial = optimize_h(h_initial)
        c5_current = evaluate_c5(h_initial)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_initial

    # Ensure best_h is the correct length
    best_h = best_h[:n_points]
    return best_h, best_c5, n_points
```
