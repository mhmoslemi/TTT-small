import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 100
    
    # Function to compute C5 for a given h_values
    def compute_c5(h_values):
        dx = 2.0 / n_points
        h = np.array(h_values)
        shifted = np.zeros_like(h)
        for k in range(1, n_points):
            shifted[k:] = h[:-k]
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Initial guess: use provided initial_h_values or random
    if hasattr(initial_h_values, "__len__") and len(initial_h_values) == n_points:
        h_initial = initial_h_values
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Objective function with bounds and constraints
    def objective_function(h):
        c5 = compute_c5(h)
        return c5
    
    # Optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    start_time = time.time()
    result = opt.minimize(
        objective_function,
        h_initial,
        method="L-BFGS-B",
        bounds=bounds,
        options={"ftol": 1e-6, "gtol": 1e-6, "maxiter": 100}
    )
    
    # Check elapsed time and finish if budget is exceeded
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 1:
        return (result.x, compute_c5(result.x), n_points)
    
    # Use CVXOPT for more precise optimization
    try:
        h_cvx = cp.Variable(n_points)
        constraints = [cp.sum(h_cvx) == n_points / 2, 0 <= h_cvx, h_cvx <= 1]
        obj = cp.Minimize(compute_c5(h_cvx))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CVXOPT, eps=1e-6)
        h_optimized = h_cvx.value
    except:
        h_optimized = result.x
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 1:
        return (h_optimized, compute_c5(h_optimized), n_points)
    
    # Final result
    return (h_optimized, compute_c5(h_optimized), n_points)
