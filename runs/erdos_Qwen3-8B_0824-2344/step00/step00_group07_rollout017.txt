```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points (we'll use 400 for better resolution)
    n_points = 400
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()[:n_points]  # Ensure we don't exceed n_points
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    
    # Ensure the integral of h is 1
    sum_h = np.sum(h_initial) * dx
    if abs(sum_h - 1) > 1e-6:
        # Normalize
        h_initial = h_initial * (1 / sum_h)

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = np.pad(h, (0, n_points), mode='constant', constant_values=0)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    # Optimization objective: minimize the maximum overlap
    def objective(h):
        return compute_c5(h)

    # We'll use a gradient-based optimization with some constraints
    # We'll use a modified version of the initial solution and perturb it

    # Initial guess
    h = h_initial.copy()

    # Define the optimization variables and constraints
    h_var = cp.Variable(n_points, nonneg=True, bounds=(0, 1))
    constraints = [
        cp.sum(h_var) == n_points / 2,
    ]

    # Optimization problem
    prob = cp.Problem(cp.Minimize(objective(h_var)), constraints)

    start_time = time.time()
    try:
        # If we have enough time, perform the optimization
        if time.time() - start_time < budget_s - 10:
            prob.solve(solver='MOSEK', verbose=False)
            if prob.status == 'optimal':
                h_opt = h_var.value
            else:
                h_opt = h_initial.copy()
        else:
            # If we run out of time, use the initial guess
            h_opt = h_initial.copy()
    except:
        # In case of error, use the initial guess
        h_opt = h_initial.copy()

    # Ensure h_opt is within [0, 1]
    h_opt = np.clip(h_opt, 0, 1)

    # Compute the final C5 value
    c5_bound = compute_c5(h_opt)

    return h_opt, c5_bound, n_points
```
