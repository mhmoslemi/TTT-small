```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
from time import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Example initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can increase if time allows

    # Function to evaluate the C5 value of a given h
    def evaluate(h):
        nonlocal best_c5
        if len(h) != n_points:
            h = np.resize(h, n_points)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Initial guess from provided initial_h_values
    initial_h = np.resize(initial_h_values, n_points)
    initial_h = np.clip(initial_h, 0, 1)

    # Try optimizing with different n_points
    for n_points in [100, 150, 200, 250, 300]:
        if time() - start_time > budget_s - 1:
            break

        # Define the problem
        h = cp.Variable(n_points)
        # Constraint: sum(h) == n_points / 2
        constraints = [cp.sum(h) == n_points / 2]
        # Objective: minimize the maximum overlap (using a convex relaxation)
        # We use a convex relaxation to avoid non-convex min-max problems
        # By minimizing the worst-case overlap we approximate the min-max
        # Note: This is an approximation and might not give the exact minimum
        # But it's computationally feasible for large n_points

        # For each shift k, compute overlap integral
        # We approximate the max overlap by using convex constraints
        overlap = cp.Parameter((n_points, n_points))
        for i in range(n_points):
            for j in range(n_points):
                overlap[i, j] = cp.multiply(h[i], 1 - h[(i + j) % n_points]) * (2.0 / n_points)
        max_overlap = cp.Maximize(cp.Minimize(cp.sum(overlap)))

        # Optimization problem
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)

        # Solve the problem
        try:
            prob.solve(solver='CVXOPT', eps=1e-6)
            if prob.status == 'optimal':
                h_opt = h.value
                h_opt = np.clip(h_opt, 0, 1)
                c5 = evaluate(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
        except:
            pass

        # Also try a gradient-based optimization
        def objective(x):
            x = np.clip(x, 0, 1)
            return evaluate(x)

        result = optimize.minimize(objective, initial_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            c5 = evaluate(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    return best_h, best_c5, n_points
```
