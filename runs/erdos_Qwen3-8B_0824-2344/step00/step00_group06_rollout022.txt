```python
import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # start with 100 points (can increase if necessary)
    h = initial_h_values.copy()  # start from the initial construction
    h = h[:n_points]  # truncate to n_points if needed
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)  # resample to n_points
    
    # Normalize h to ensure the integral is 1
    h /= np.sum(h) * 2 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    max_iter = 100
    learning_rate = 0.01
    num_samples = 50  # number of samples to generate in each iteration
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Generate random perturbations
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Normalize h_perturbed
        h_perturbed /= np.sum(h_perturbed) * 2 / n_points
        
        # Evaluate the solution
        c5 = evaluate_erdos_solution(h_perturbed)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()
        
        # Move towards the best solution
        h = best_h * (1 - learning_rate) + h * learning_rate
    
    # Try increasing n_points if time allows
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        h = initial_h_values.copy()
        h = h[:n_points]
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h /= np.sum(h) * 2 / n_points
        best_h = h.copy()
        best_c5 = evaluate_erdos_solution(h)
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturbed = h + perturb
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed /= np.sum(h_perturbed) * 2 / n_points
            c5 = evaluate_erdos_solution(h_perturbed)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_perturbed.copy()
            h = best_h * (1 - learning_rate) + h * learning_rate
    
    return best_h, best_c5, n_points
```
