import numpy as np

import time

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Use initial_h_values if available
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        h = np.interp(h, [0, len(h)], [0, n_points])
        h = np.clip(h, 0, n_points)
        h = h.astype(float)
    
    # Ensure the sum of h is exactly n_points / 2
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h / h_sum * (n_points / 2)
    
    # Define function to compute C5
    def compute_c5(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals.reshape(-1)
        h_vals = np.interp(h_vals, [0, len(h_vals)], [0, 2])
        dx = 2.0 / len(h_vals)
        h_vals = h_vals * dx
        conv = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5 = np.max(conv)
        return c5
    
    # Define function to evaluate and track best solution
    def evaluate_and_track(h_vals, best_h, best_c5):
        c5 = compute_c5(h_vals)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()
        return best_h, best_c5
    
    # Initial best solution
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    def objective_func(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        return compute_c5(h_vals)
    
    # Run optimization
    result = optimize.minimize(
        fun=objective_func,
        x0=best_h,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'disp': False}
    )
    
    # Update best solution
    best_h = result.x
    best_c5 = compute_c5(best_h)

    for _ in range(5):

        if c5_perturb < best_c5:
            best_h = h_perturb
            best_c5 = c5_perturb
    
    # Check elapsed time
    elapsed_time = time.time() - start_time

        # If time allows, try a higher resolution
        n_points = 200

        # Ensure the sum of h is exactly n_points / 2

        best_h_high_res = h_high_res.copy()

        result_high_res = optimize.minimize(

            bounds=[(0, 1) for _ in range(n_points)],

        )
        
        # Update best solution
        best_h_high_res = result_high_res.x
        best_c5_high_res = compute_c5(best_h_high_res)
        
        # Update best solution if higher resolution gives better result
        if best_c5_high_res < best_c5:
            best_h = best_h_high_res
            best_c5 = best_c5_high_res
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
