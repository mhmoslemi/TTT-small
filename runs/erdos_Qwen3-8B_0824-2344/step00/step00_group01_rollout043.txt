import numpy as np
import numpy.random as npr
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points for faster optimization
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points])  # Initial guess with 100 samples
    
    # Ensure the initial h has the correct integral
    h = h * (n_points / 2) / np.sum(h)  # Scale to ensure ∫h(x)dx = 1
    
    def compute_c5(h_values):
        """
        Compute the maximum overlap integral C5 given a step function h_values.
        """
        h = h_values.copy()
        h = h * (n_points / 2) / np.sum(h)  # Scale for correct integral
        dx = 2.0 / n_points
        # Compute the cross-correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        # Apply the dx scaling
        corr = corr * dx
        # Find the maximum value
        return np.max(corr)
    
    def objective(x, n_points, dx):
        """
        Objective function for optimization: minimize the maximum overlap.
        """
        h = x.reshape(n_points)
        # Scale to ensure ∫h(x)dx = 1
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)
    
    def constraint(x, n_points):
        """
        Constraint: ∫h(x)dx = 1
        """
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    x = cp.Variable(n_points)
    objective_func = cp.Minimize(objective(x, n_points, dx))
    constraints = [constraint(x, n_points) == 0]
    problem = cp.Problem(objective_func, constraints)
    
    # Run the optimization
    try:
        result = problem.solve(solver=cp.SCS, eps=1e-4, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
        if result is not None:
            h_opt = x.value
            c5_bound = compute_c5(h_opt)
        else:
            h_opt = h
            c5_bound = compute_c5(h)
    except:
        h_opt = h
        c5_bound = compute_c5(h)
    
    # If optimization failed or didn't improve, try a local search
    if c5_bound > 0.380876:
        # Local search with random perturbations
        for _ in range(10):
            perturbation = np.random.normal(0, 0.05, size=n_points)
            h_perturbed = h_opt + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Scale to ensure ∫h(x)dx = 1
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            c5_perturbed = compute_c5(h_perturbed)
            if c5_perturbed < c5_bound:
                h_opt = h_perturbed
                c5_bound = c5_perturbed
    
    # If not enough improvement, try a larger n_points
    if c5_bound > 0.380876:
        n_points = 200
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        c5_bound = compute_c5(h)
    
    # Ensure we don't exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 1:
        # Further small-scale optimization
        for _ in range(10):
            perturbation = np.random.normal(0, 0.05, size=n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            c5_perturbed = compute_c5(h_perturbed)
            if c5_perturbed < c5_bound:
                h = h_perturbed
                c5_bound = c5_perturbed
    
    return h, c5_bound, n_points
