import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Parameters
    n_points = 100  # Start with 100 points, can adjust if needed
    dx = 2.0 / n_points
    max_iterations = 100  # Maximum iterations for optimization
    tolerance = 1e-5  # Tolerance for convergence
    initial_h = initial_h_values.copy()  # Start from initial construction
    initial_h = initial_h[:n_points]  # Ensure we have the right number of points

    # Convert initial_h to a numpy array
    h_values = np.array(initial_h, dtype=np.float64)
    h_values = h_values[:n_points]  # Ensure the length matches n_points
    h_values = np.clip(h_values, 0, 1)  # Ensure values are within [0,1]
    h_values = h_values * (n_points / 2)  # Ensure the integral is 1

    best_c5 = float('inf')
    best_h = h_values.copy()

    # Define objective function to minimize max overlap
    def objective_function(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)
        return c5

    # Define optimization problem
    h = cp.Variable(n_points)
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]

    # Optimization loop
    start_time = time.time()
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Objective: minimize the max overlap
        obj = cp.Minimize(cp.Max(cp.multiply(dx, np.correlate(h, 1 - h, mode="full"))))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=tolerance, max_iter=100)
        except:
            # If the solver fails, use gradient descent
            h_values = h_values + np.random.normal(0, 0.01, n_points)
            h_values = np.clip(h_values, 0, 1)
            h_values = h_values * (n_points / 2)
            continue

        if prob.status in ['optimal', 'reoptimal']:
            h_values = h.value
            h_values = np.clip(h_values, 0, 1)
            h_values = h_values * (n_points / 2)
            c5 = objective_function(h_values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values.copy()

    # Final evaluation
    final_c5 = objective_function(best_h)
    return (best_h, final_c5, n_points)
