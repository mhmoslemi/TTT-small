import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h_shifted = np.zeros(2 * n_points)
    h_shifted[:n_points] = h
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    return np.load("initial_h_values.npy")

# Strategy: Use a gradient-based optimization approach with a piecewise constant function
# and a penalty method to enforce the integral constraint. We optimize over h_values
# to minimize the maximum overlap integral. We start from the initial solution and
# use a local optimization method (L-BFGS-B) to improve it.

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial solution
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = sum(h_initial)
    
    # Adjust for integral constraint (sum(h) * dx = 1)
    if abs(sum_h * dx - 1) > 1e-6:
        h_initial = h_initial * (1 / (sum_h * dx))
    
    def objective(h):
        # Objective function to minimize: max overlap integral
        n_points = len(h)
        dx = 2.0 / n_points
        h_shifted = np.zeros(2 * n_points)
        h_shifted[:n_points] = h
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr) * dx
        return max_corr
    
    def constraint(h):
        # Constraint: sum(h) * dx = 1
        n_points = len(h)
        dx = 2.0 / n_points
        return sum(h) * dx - 1
    
    # Define optimization variables
    h_vars = cp.Variable(n_points, name='h')
    
    # Define objective and constraints
    obj = cp.Minimize(cp.Max(cp.multiply(cp.Constant(2.0 / n_points), cp.correlate(h_vars, 1 - h_vars, mode="full"))))
    constraints = [cp.sum(h_vars) == n_points / 2]
    
    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, eps=1e-6)
    
    # Extract the optimal solution
    h_opt = h_vars.value
    c5_bound = evaluate_erdos_solution(h_opt)
    
    return h_opt, c5_bound, len(h_opt)
