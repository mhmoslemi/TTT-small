import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h_values.npy')  # Assume this is the initial construction with n=95 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Parameters
    max_n_points = 500  # We'll try up to 500 points for better accuracy
    n_points = 100  # Start with 100 points for faster initial optimization
    best_c5 = float('inf')
    best_h = None
    best_n = n_points

    # Start with the initial construction
    h = initial_h_values[:n_points].copy()
    h = np.minimum(np.maximum(h, 0), 1)
    sum_h = np.sum(h)
    if sum_h != n_points / 2:
        h = h * (n_points / 2) / sum_h
    
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n_points

    # Try increasing n_points in steps
    for n_points in range(150, max_n_points + 1, 50):
        # Generate random initial guess
        h = np.random.rand(n_points)
        h = np.minimum(np.maximum(h, 0), 1)
        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            h = h * (n_points / 2) / sum_h

        # Use gradient descent to minimize the max overlap
        def objective(h_flat):
            h = h_flat.reshape(-1)
            integral = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(integral) * (2.0 / n_points)
            return c5

        def grad_objective(h_flat):
            h = h_flat.reshape(-1)
            integral = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(integral) * (2.0 / n_points)
            # Gradient is approximate using finite differences
            eps = 1e-6
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.minimum(np.maximum(h_plus, 0), 1)
                c5_plus = evaluate_erdos_solution(h_plus)
                grad[i] = (c5_plus - c5) / eps
            return grad

        # Optimization
        result = optimize.minimize(objective, h, method='L-BFGS-B', jac=grad_objective, options={'maxiter': 100, 'ftol': 1e-6})
        h_opt = result.x
        h_opt = np.minimum(np.maximum(h_opt, 0), 1)
        sum_h = np.sum(h_opt)
        if sum_h != n_points / 2:
            h_opt = h_opt * (n_points / 2) / sum_h

        c5_opt = evaluate_erdos_solution(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_points

        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break

    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, best_n
