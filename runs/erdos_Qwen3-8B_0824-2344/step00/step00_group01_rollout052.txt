    for i in range(n_points):

    # Define the function to compute C5 bound
    def compute_c5(h_values):
        h = np.array(h_values)
        shifted_h = np.zeros_like(h)

            shifted_h = np.roll(h, k)

                c5_bound = overlap_integral
        return c5_bound

    # Optimization: minimize the maximum overlap integral
    # We'll use a gradient-based approach with a penalty function
    def objective_function(h_values):
        h = np.array(h_values)
        # Compute all shifts and find the maximum overlap
        max_overlap = 0.0
        for k in range(1, n_points):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    initial_guess = initial_h_values.copy()

    start_time = time.time()
    result = opt.minimize(
        objective_function,
        initial_guess,
        method='L-BFGS-B',

    )

        # Recompute C5 with finer grid
