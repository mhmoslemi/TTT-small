def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    # Try multiple initial configurations
    for _ in range(3):
        n_points = 100  # Reduced from 80 to 100 for better resolution
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy the area constraint
        
        # Optimize the function to minimize the maximum overlap
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            overlaps = signal.correlate(h, 1 - h, mode="full") * dx
            return np.max(overlaps)
        
        # Use L-BFGS-B for optimization
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        h = result.x
        c5 = objective(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, len(best_h)
