import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros((n_points, n_points))
    for i in range(n_points):
        shifted_h[i, :] = np.roll(h, i)
    overlap = np.sum(h * (1 - shifted_h), axis=1) * dx
    return np.max(overlap)

def initial_h_values():
    # This is a placeholder for an initial construction, which would be provided in actual code
    # For demonstration, we'll use a simple uniform h
    n_points = 70
    return np.array([0.5] * n_points)

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Initial setup
    initial_n_points = 70
    initial_h = initial_h_values()
    best_h = copy.deepcopy(initial_h)
    best_c5 = evaluate_erdos_solution(initial_h)
    best_n_points = initial_n_points
    
    # Try smaller n_points (e.g., 50) for faster evaluation
    for n_points in [50, 60, 70, 80, 90, 100]:
        if time.time() - start_time > budget_s:
            break
        
        # Create initial guess
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        
        # Define variables
        x = cp.Variable(n_points, nonneg=True)
        constraints = [0 <= x, x <= 1]
        constraints += [cp.sum(x) == n_points / 2]
        
        # Objective: minimize the maximum overlap
        overlap = []
        for k in range(n_points):
            shifted_h = np.roll(x, k)
            overlap_k = cp.sum(x * (1 - shifted_h))
            overlap.append(overlap_k)
        c5 = cp.Max(overlap) * dx
        
        # Solve
        prob = cp.Problem(cp.Minimize(c5), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=1000, eps=1e-6)
            if prob.status == 'optimal':
                h_candidate = x.value
                c5_candidate = prob.objective.value
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    best_n_points = n_points
        except:
            pass
        
        # Evaluate current best
        if time.time() - start_time > budget_s:
            break
    
    # If we have a better solution, return it
    if best_n_points < initial_n_points and best_c5 < best_c5:
        return (best_h, best_c5, best_n_points)
    
    # Try a more refined search on the best n_points
    final_n_points = best_n_points
    dx = 2.0 / final_n_points
    h = np.random.rand(final_n_points)
    h = h * (final_n_points / 2) / np.sum(h)
    
    # Define variables
    x = cp.Variable(final_n_points, nonneg=True)
    constraints = [0 <= x, x <= 1]
    constraints += [cp.sum(x) == final_n_points / 2]
    
    # Objective: minimize the maximum overlap
    overlap = []
    for k in range(final_n_points):
        shifted_h = np.roll(x, k)
        overlap_k = cp.sum(x * (1 - shifted_h))
        overlap.append(overlap_k)
    c5 = cp.Max(overlap) * dx
    
    # Solve
    prob = cp.Problem(cp.Minimize(c5), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iter=1000, eps=1e-6)
        if prob.status == 'optimal':
            h_candidate = x.value
            c5_candidate = prob.objective.value
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
                best_n_points = final_n_points
    except:
        pass
    
    return (best_h, best_c5, best_n_points)
