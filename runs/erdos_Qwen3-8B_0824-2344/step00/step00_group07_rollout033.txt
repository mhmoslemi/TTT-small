import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from pre-imported variable
    initial_h_values = evaluate_erdos_solution()
    n_points_initial = len(initial_h_values)
    
    # Start with initial n_points
    n_points = n_points_initial
    
    # Define a function to compute the overlap integral
    def compute_c5(h_values):
        h = np.array(h_values, dtype=np.float64)
        shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted[k:] = h[:-k]
        overlap = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * (2.0 / len(h))
        return max_overlap
    
    # Define a function to compute the function value for optimization
    def objective_function(h_values):
        return compute_c5(h_values)
    
    # Function to generate a random initial guess
    def generate_initial_guess(n_points):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    best_h_values = initial_h_values.copy()
    best_c5 = compute_c5(best_h_values)
    
    start_time = time.time()
    
    # Try increasing n_points up to a certain limit to explore the space
    for n_points in range(n_points_initial, min(1000, n_points_initial + 200)):
        print(f"Trying with n_points = {n_points}")
        
        # Generate a random initial guess
        h = generate_initial_guess(n_points)
        
        # Set up a constrained optimization problem using CVXPY
        h_var = cp.Variable(n_points)
        constraint = (cp.sum(h_var) == n_points / 2)
        problem = cp.Problem(cp.Minimize(objective_function(h_var)), [constraint])
        
        # Solve the optimization problem
        try:
            result = problem.solve(solver=cp.SCS, max_iters=1000, verbose=False)
            if result is not None and cp.value(problem.objective) is not None:
                h_opt = h_var.value
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h_values = h_opt.copy()
        except:
            pass
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Try optimizing with the best found h_values
    print(f"Optimizing with n_points = {len(best_h_values)}")
    h = best_h_values.copy()
    h_var = cp.Variable(len(h))
    constraint = (cp.sum(h_var) == len(h) / 2)
    problem = cp.Problem(cp.Minimize(objective_function(h_var)), [constraint])
    
    try:
        result = problem.solve(solver=cp.SCS, max_iters=1000, verbose=False)
        if result is not None and cp.value(problem.objective) is not None:
            h_opt = h_var.value
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h_values = h_opt.copy()
    except:
        pass
    
    # Final check with initial configuration
    c5_initial = compute_c5(initial_h_values)
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h_values = initial_h_values.copy()
    
    return best_h_values, best_c5, len(best_h_values)
