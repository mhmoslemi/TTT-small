```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    max_n_points = 1000  # Maximum number of points to consider
    n_start = 200  # Starting number of points
    n_end = 500  # Maximum number of points to try
    improvement_threshold = 0.0001  # Stop if improvement is below this
    max_iterations = 10  # Limit the number of iterations to prevent infinite loops
    max_time_per_n = 30  # Time per n_points in seconds
    
    best_h = initial_h_values.copy()  # Start with the initial construction
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = len(best_h)
    
    elapsed_time = 0
    
    for n_points in range(n_start, n_end + 1, 10):
        if elapsed_time >= budget_s:
            break
        
        # Calculate dx
        dx = 2.0 / n_points
        # Create a grid
        x = np.linspace(0, 2, n_points)
        
        # Define the objective function with constraints
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # Define the overlap integral for all shifts
        c5_values = []
        for k in range(1, n_points - 1):  # Shifts from 1 to n_points - 2
            h_shifted = np.roll(h.value, k) if hasattr(h, 'value') else np.roll(h, k)
            integral = cp.sum(h * (1 - h_shifted)) * dx
            c5_values.append(integral)
        
        # Maximize the overlap integral (objective is to minimize maximum)
        objective = cp.Minimize(cp.Max(c5_values))
        problem = cp.Problem(objective, constraints)
        
        # Solve the problem
        start_time = time.time()
        try:
            result = problem.solve(solver=cp.ECOS, verbose=False)
            if result is not None and result < best_c5:
                best_c5 = result
                best_h = h.value
                best_n_points = n_points
                elapsed_time = time.time() - start_time
                print(f"Improved c5 to {best_c5:.6f} with n_points={best_n_points}")
            else:
                elapsed_time = time.time() - start_time
        except:
            elapsed_time = time.time() - start_time
        
        # Check if we should continue
        if elapsed_time >= max_time_per_n:
            print(f"Reached time limit for n_points={n_points}")
            break
    
    # Final check to ensure best_h is properly formatted
    best_h = np.clip(best_h, 0, 1)
    return best_h, best_c5, best_n_points
```
