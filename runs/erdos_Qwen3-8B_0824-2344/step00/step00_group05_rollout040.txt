import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1-h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    n_points = 100  # Start with 100 points for faster evaluation
    h = initial_h_values[:n_points].copy()
    h = h / np.sum(h) * (n_points / 2)  # Ensure integral is 1
    h = np.clip(h, 0, 1)  # Ensure values are in [0,1]
    
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Try different numbers of points up to 500
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate initial random samples with proper normalization
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        
        # Define variables
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        
        # Evaluate the C5 metric as a function of h
        def objective_func(h):
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * (2.0 / len(h))
        
        # Use gradient descent to minimize the C5 metric
        def objective(x_vec):
            h = x_vec
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * (2.0 / len(h))
        
        # Optimization with gradient descent
        def optimize_h(h_initial):
            h = h_initial.copy()
            for _ in range(100):
                # Compute the current C5
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                
                # Compute the gradient (numerically)
                grad = np.zeros_like(h)
                for i in range(len(h)):
                    h_plus = h.copy()
                    h_plus[i] += 1e-4
                    c5_plus = objective(h_plus)
                    h_minus = h.copy()
                    h_minus[i] -= 1e-4
                    c5_minus = objective(h_minus)
                    grad[i] = (c5_plus - c5_minus) / (2 * 1e-4)
                
                # Update h using gradient descent
                h -= 0.01 * grad
                
                # Enforce constraints
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)
            
            return h
        
        h = optimize_h(h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5, len(best_h)
