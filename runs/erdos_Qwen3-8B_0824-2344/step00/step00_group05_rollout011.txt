import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial solution
    n_points = 70
    h_values = initial_h_values.copy()
    h_values = h_values[:n_points]
    
    # Ensure the integral is exactly 1
    total = np.sum(h_values) * (2.0 / n_points)
    if abs(total - 1.0) > 1e-6:
        h_values = np.array(h_values, dtype=np.float64)
        h_values /= total
    
    best_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    
    # Try smaller n_points
    for n_points in [50, 40, 30, 20, 10]:
        h_values = np.zeros(n_points)
        h_values[0] = 1.0  # Initialize a single point at 1
        h_values = h_values / (np.sum(h_values) * (2.0 / n_points))  # Normalize
        
        # Use a greedy approach to construct a function that minimizes overlap
        for _ in range(1000):
            # Randomly select positions to flip
            idx = np.random.randint(0, n_points)
            h_values[idx] = 1.0 - h_values[idx]
            
            # Ensure the integral is exactly 1
            total = np.sum(h_values) * (2.0 / n_points)
            if abs(total - 1.0) > 1e-6:
                h_values /= total
        
        # Evaluate the solution
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values.copy()
    
    # Use a local optimization method
    def objective(h):
        h = np.array(h, dtype=np.float64)
        h = h / (np.sum(h) * (2.0 / len(h)))  # Normalize
        return evaluate_erdos_solution(h)
    
    # Use a gradient-free optimization method
    result = optimize.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(len(best_h))],
        options={'ftol': 1e-6, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure the integral is exactly 1
    total = np.sum(best_h) * (2.0 / len(best_h))
    if abs(total - 1.0) > 1e-6:
        best_h /= total
    
    return best_h, best_c5, len(best_h)
