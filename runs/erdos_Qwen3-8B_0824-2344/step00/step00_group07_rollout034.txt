import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Helper function to compute the overlap integral
def compute_c5(h_values, dx):
    h = np.array(h_values)
    one_minus_h = 1 - h
    corr = signal.correlate(h, one_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

# Function to generate an initial guess based on the current record
def generate_initial_guess(n_points):
    # Start with a random initial guess
    h = np.random.rand(n_points)
    # Normalize to satisfy the integral constraint
    h = h / np.sum(h) * (n_points / 2)
    return h.tolist()

# Function to optimize the step function
def optimize_step_function(n_points, max_iterations=100, tolerance=1e-4):
    h = np.random.rand(n_points)
    h = h / np.sum(h) * (n_points / 2)
    
    for _ in range(max_iterations):
        dx = 2.0 / n_points
        c5 = compute_c5(h, dx)
        if c5 < tolerance:
            break
        
        # Objective: minimize the maximum overlap integral
        # Use CVXPY to minimize the maximum overlap
        # We use a convex relaxation for this
        h_var = cp.Variable(n_points, name='h')
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]
        # Estimate the maximum overlap using a convex underapproximation
        corr = signal.correlate(h_var.value, 1 - h_var.value, mode='full')
        max_overlap = cp.Max(cp.vstack([cp.sum(corr[i:i+n_points] * h_var) for i in range(n_points)]))
        
        # Objective: minimize the maximum overlap
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        prob.solve(solver=cp.MOSEK, eps=1e-6)
        if prob.status != 'OPTIMAL':
            break
        h = h_var.value
    
    return h

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Try different number of points from 100 to 500
    n_points_list = [100, 200, 300, 400, 500]
    
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess
        h = generate_initial_guess(n_points)
        
        # Optimize
        h_optimized = optimize_step_function(n_points, max_iterations=50, tolerance=1e-5)
        
        # Compute C5
        dx = 2.0 / n_points
        c5 = compute_c5(h_optimized, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_optimized
            print(f"Found better solution with n_points={n_points}, C5={c5}")
    
    # Return the best solution found
    return (best_h, best_c5, len(best_h))
