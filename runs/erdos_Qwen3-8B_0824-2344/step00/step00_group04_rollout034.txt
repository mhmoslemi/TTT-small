import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    start_time = time.time()
    
    # Starting parameters
    n_points = 100  # Initial number of points, can be adjusted if needed
    dx = 2.0 / n_points
    h = initial_h_values.copy()  # Start with the initial construction
    h = h[:n_points]  # Ensure it's the correct length
    
    # Ensure the integral constraint is satisfied
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-8:
        h = initial_h_values.copy()
        h = h[:n_points]
        sum_h = np.sum(h)
    
    best_h = h
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    max_iter = 100
    step_size = 0.1
    tolerance = 1e-6
    
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Evaluate current C5
        current_c5 = evaluate_erdos_solution(h)
        
        # Update best solution if needed
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Attempt small perturbations
        for i in range(n_points):
            # Randomly perturb h[i] within bounds
            h[i] += np.random.uniform(-step_size, step_size)
            h[i] = np.clip(h[i], 0.0, 1.0)
        
        # Ensure the area constraint is approximately satisfied
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-4:
            # Adjust h to meet the constraint
            excess = sum_h - n_points / 2
            if excess > 0:
                # Reduce values where h is high
                h += np.random.uniform(-excess / n_points, 0.0, n_points)
            else:
                # Increase values where h is low
                h += np.random.uniform(0.0, -excess / n_points, n_points)
            h = np.clip(h, 0.0, 1.0)
    
    # Evaluate the best solution found
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
