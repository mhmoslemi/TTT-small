```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Strategy:
# 1. Start with the initial_h_values as a baseline.
# 2. Use a combination of gradient-based optimization and simulated annealing to refine the function.
# 3. Perform a grid search over a few different n_points (number of samples) to find the best balance between accuracy and computational efficiency.
# 4. Evaluate the overlap integral for each candidate h and select the one with the smallest C₅.
# 5. Use random restarts and perturbations to avoid local optima.

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Try a few different n_points values
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() > time.time() + budget_s:
            break

        # Initialize h with the initial values, scaled appropriately
        h_initial = np.array(initial_h_values) * (n_points / len(initial_h_values))
        h_initial = np.resize(h_initial, n_points)
        h_initial = np.clip(h_initial, 0, 1)

        # Ensure the integral constraint is exactly met
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

        # Define the objective function
        def objective(h):
            dx = 2.0 / n_points
            h = np.clip(h, 0, 1)
            integral = 0.0
            for k in range(1, int(2 * dx) + 1):
                shift = k * dx
                shifted = np.roll(h, int(shift / dx))
                integral += np.dot(h, 1 - shifted) * dx
            return integral

        # Use a simple optimization method (e.g., Nelder-Mead)
        result = optimize.minimize(
            objective,
            h_initial,
            method='Nelder-Mead',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'xatol': 1e-5, 'fatol': 1e-5}
        )

        # Evaluate the best candidate found
        h_candidate = result.x
        c5 = evaluate_erdos_solution(h_candidate, n_points)

        # Check if this is the best solution found so far
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate
            best_n_points = n_points

    # Use CVXPY to refine the solution further with more accurate optimization
    # This is a simplified convex relaxation for demonstration; in practice, more sophisticated formulations may be used
    if best_n_points > 0:
        n_points = best_n_points
        h = cp.Variable(n_points)
        dx = 2.0 / n_points

        # Objective: minimize the maximum overlap integral
        # We approximate it by considering all k in a reasonable range
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]

        objective = cp.Minimize(cp.max([cp.sum(h[i] * (1 - h[(i + k) % n_points])) for k in range(1, int(2 * dx) + 1)]))
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.SCS, eps=1e-5)

        if result is not None:
            h_final = np.array(result).flatten()
            c5 = evaluate_erdos_solution(h_final, n_points)

            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final
                best_n_points = n_points

    return best_h, best_c5, best_n_points
```
