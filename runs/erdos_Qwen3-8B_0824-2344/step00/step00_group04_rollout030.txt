import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.0, 0.0, 0.0, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333,
                             0.3333333333333333, 0.3333333333333333, 0.3333333333333333, 0.3333333333333333])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial h values
    h = np.copy(initial_h_values)
    n_points = len(h)
    
    # Try a few different n_points to find a good balance between resolution and speed
    for n in [100, 150, 200, 250, 300]:
        # Generate a random initial guess
        h_initial = np.random.rand(n) * 2.0 - 1.0  # Random values in [-1, 1]
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n / 2)  # Adjust to meet the volume constraint
        h_initial = h_initial / (n / 2)  # Normalize to [0, 1]
        
        # Optimization using CVXPY
        x = cp.Variable(n)
        constraints = [
            cp.sum(x) == n / 2,
            0 <= x,
            x <= 1
        ]
        obj = cp.Maximize(cp.multiply(x, 1 - cp.roll(x, 1)) + cp.multiply(x, 1 - cp.roll(x, 2)) +
                          cp.multiply(x, 1 - cp.roll(x, 3)) + cp.multiply(x, 1 - cp.roll(x, 4)) +
                          cp.multiply(x, 1 - cp.roll(x, 5)))
        prob = cp.Problem(obj, constraints)
        start_time = time.time()
        try:
            result = prob.solve(solver=cp.SCS, time_limit=budget_s - (time.time() - start_time))
            if result is not None:
                h_candidate = np.array(x.value)
                c5_candidate = evaluate_erdos_solution(h_candidate)
                if c5_candidate < 0.380876:
                    h = h_candidate
                    n_points = n
                    print(f"Found better solution with n={n}, C5={c5_candidate}")
        except:
            pass
        
        # Try a gradient descent approach
        h_opt = np.copy(h_initial)
        for _ in range(100):
            # Compute the correlation with the shifted h
            corr = np.correlate(h_opt, 1 - h_opt, mode="full")
            # Compute the derivative and update h_opt
            grad = np.zeros_like(h_opt)
            for i in range(n):
                for shift in range(1, 6):
                    if i + shift < n:
                        grad[i] += (1 - h_opt[i + shift]) * corr[i + shift]
                    if i - shift >= 0:
                        grad[i] += (1 - h_opt[i - shift]) * corr[i - shift]
            grad = grad / np.linalg.norm(grad)
            h_opt -= 0.01 * grad
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n / 2) / np.sum(h_opt)
        
        c5_candidate = evaluate_erdos_solution(h_opt)
        if c5_candidate < 0.380876:
            h = h_opt
            n_points = n
            print(f"Found better solution with n={n}, C5={c5_candidate}")
    
    return h, evaluate_erdos_solution(h), n_points
