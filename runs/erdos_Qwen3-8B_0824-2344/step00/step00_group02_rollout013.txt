```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    # Start timing
    start_time = time.time()

    # Initial parameters
    n_points_start = len(initial_h_values)
    n_points = n_points_start
    h_values = np.array(initial_h_values)

    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / len(h)
        # Compute the convolution (correlation)
        corr = signal.correlate(h, 1 - h, mode="full")
        # The maximum overlap is the maximum value of the correlation
        # The integral is the correlation value multiplied by dx
        c5 = np.max(corr) * dx
        return c5

    # Function to check the constraints
    def check_constraints(h):
        if not np.all((h >= 0) & (h <= 1)):
            return False
        if np.abs(np.sum(h) - len(h) / 2) > 1e-6:
            return False
        return True

    # Function to optimize h
    def optimize_h(h_initial):
        n_points = len(h_initial)
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        obj = cp.Minimize(cp.max(signal.correlate(h.value, 1 - h.value, mode="full") * 2.0 / n_points))
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.SCS, max_iter=1000)
        if result is not None and check_constraints(h.value):
            return h.value
        return h_initial

    # Start with the initial solution
    best_h = h_values
    best_c5 = compute_c5(best_h)

    # Try increasing n_points to see if we can improve
    for n_points in range(n_points_start + 1, 100):
        # Create a new initial guess by upsampling
        h_upsampled = np.zeros(n_points)
        h_upsampled[::2] = best_h
        h_upsampled = (h_upsampled + np.random.rand(n_points) * 0.01) / 2.0
        h_upsampled = np.clip(h_upsampled, 0, 1)
        h_upsampled = optimize_h(h_upsampled)

        # Compute the new c5
        c5 = compute_c5(h_upsampled)
        if c5 < best_c5:
            best_h = h_upsampled
            best_c5 = c5

        # Check time budget
        if time.time() - start_time > budget_s:
            break

    # Return the best solution
    return (best_h, best_c5, len(best_h))
```
