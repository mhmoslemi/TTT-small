```python
import numpy as np
import scipy.optimize as opt
import time
import warnings

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Constants and parameters
MAX_ITER = 100
TIME_BUDGET = 60
N_POINTS = 400  # Default number of points; can be adjusted based on time

def run(seed=42, budget_s=TIME_BUDGET, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial guess from provided initial_h_values
    h_initial = initial_h_values
    n_points_initial = len(h_initial)
    
    # If the initial number of points is more than our target, reduce it
    if n_points_initial > N_POINTS:
        # Downsample initial_h_values to N_POINTS samples
        h_initial = np.interp(np.linspace(0, 2, N_POINTS), np.linspace(0, 2, n_points_initial), h_initial)
        h_initial = np.clip(h_initial, 0, 1)
    
    # Ensure the initial h satisfies the integral constraint
    def constraint_sum(h):
        return np.sum(h) - N_POINTS / 2
    
    # Function to compute the overlap integral (C5)
    def objective_func(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / N_POINTS)
        return c5
    
    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(N_POINTS)]
    
    # Run a constrained optimization to minimize C5
    result = opt.minimize(
        objective_func,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': MAX_ITER, 'disp': False}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = objective_func(best_h)
    
    # Final check and return
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        warnings.warn(f"Time budget exceeded: {elapsed_time:.2f} seconds")
    
    return (best_h, best_c5, N_POINTS)
```
