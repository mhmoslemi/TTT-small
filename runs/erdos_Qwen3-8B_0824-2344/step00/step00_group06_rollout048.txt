```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Define the run function
def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = 95
    h = initial_h_values()
    
    # Try a few different n_points values to find the best
    n_points_list = [95, 128, 192, 256, 384, 512]  # Try increasing sizes for better resolution
    best_c5 = float('inf')
    best_h = h
    
    for n in n_points_list:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate a grid
        dx = 2.0 / n
        x = np.linspace(0, 2, n, endpoint=False)
        
        # Define the optimization variables (h_values)
        h_vars = cp.Variable(n, nonneg=True, integer=False)
        constraints = [cp.sum(h_vars) == n / 2]
        
        # Compute the overlap integral for all shifts k from 0 to n-1
        # For each shift k, compute max_k ∫ h(x)(1 - h(x+k)) dx
        max_overlap = 0.0
        for k in range(n):
            # Ensure we stay within the domain [0, 2]
            k_shifted = k % n
            h_shifted = np.roll(h_vars.value, k_shifted) if hasattr(h_vars, "value") else h_vars
            overlap = cp.sum(cp.multiply(h_vars, 1 - h_shifted)) * dx
            max_overlap = cp.Max(max_overlap, overlap)
        
        # Objective: minimize the maximum overlap
        objective = cp.Minimize(max_overlap)
        
        # Solve the optimization problem
        try:
            prob = cp.Problem(objective, constraints)
            result = prob.solve(solver=cp.CVXOPT, verbose=False)
            if result is not None and prob.status == 'optimal':
                h_values = h_vars.value
                c5 = max_overlap.value
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
                    print(f"Found better solution with n={n}, C5={c5:.6f}")
        except:
            print(f"Optimization failed for n={n}")
    
    # Evaluate the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution found
    return best_h, final_c5, len(best_h)
```
