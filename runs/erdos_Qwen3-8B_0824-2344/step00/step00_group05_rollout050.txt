```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose n_points for the grid
    n_points = 200  # Adjust based on budget and performance
    dx = 2.0 / n_points
    
    # Start with an initial guess
    h_initial = initial_h_values.copy()
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h = h_initial.copy()
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        overlap = signal.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Run the optimization
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Check if we ran out of time
    elapsed = time.time() - start_time
    if elapsed > budget_s - 2:
        # If time is limited, use the best solution found so far
        best_h = h
        best_c5 = evaluate_erdos_solution(best_h)
    else:
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
```
