import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for actual data
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1.0 - h
    # Compute the full convolution
    conv = signal.correlate(h, one_minus_h, mode='full')
    # Extract the relevant part of the convolution
    # We are interested in the overlap for shifts k = 0, 1, ..., n_points - 1
    # The full convolution has length 2*n_points - 1
    # The relevant part starts from index n_points - 1 to 2*n_points - 2
    relevant_conv = conv[n_points - 1:2*n_points - 1]
    # Compute the max overlap integral
    c5 = np.max(relevant_conv * dx)
    return c5

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial construction
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Initial C5 value
    current_c5 = evaluate_erdos_solution(initial_h)
    best_h = initial_h.copy()
    best_c5 = current_c5
    
    # We'll perform a local search with random perturbations
    # and gradient descent on the C5 objective
    # We'll also try some heuristic improvements
    
    # Parameters for the optimization
    max_iterations = 100
    learning_rate = 0.01
    num_perturbations = 20
    num_random_trials = 10
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the current solution
        perturbed_h = best_h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                perturbed_h[i] += np.random.uniform(-0.1, 0.1)
                perturbed_h[i] = np.clip(perturbed_h[i], 0, 1)
        
        # Normalize to ensure integral is 1
        perturbed_h = perturbed_h / (np.sum(perturbed_h) * dx) * (n_points / 2)
        
        # Evaluate the new C5
        new_c5 = evaluate_erdos_solution(perturbed_h)
        if new_c5 < best_c5:
            best_h = perturbed_h.copy()
            best_c5 = new_c5
        
        # Try some random initializations
        for _ in range(num_random_trials):
            random_h = np.random.uniform(0, 1, n_points)
            random_h = random_h / (np.sum(random_h) * dx) * (n_points / 2)
            random_c5 = evaluate_erdos_solution(random_h)
            if random_c5 < best_c5:
                best_h = random_h.copy()
                best_c5 = random_c5
        
        # If the best_c5 is already very low, we can stop earlier
        if best_c5 < 0.3808:
            break
    
    # We can also try reducing the number of points if needed
    # For faster evaluation, consider reducing n_points
    # However, since the initial construction is already at n=95, we'll keep it as is
    
    # Return the best solution
    return (best_h.tolist(), best_c5, n_points)
