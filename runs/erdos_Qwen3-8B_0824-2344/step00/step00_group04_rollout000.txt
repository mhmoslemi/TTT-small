```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Initial guess
    n_points = 100  # Start with 100 points for faster exploration
    h = initial_h_values.copy()
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    h = np.clip(h, 0, 1)
    h /= (np.sum(h) / n_points)  # Ensure integral is 1
    
    best_h = h
    best_c5 = evaluate_erdos_solution(h)
    
    # Define the objective function to minimize C5
    def objective_func(h_vec):
        h = h_vec.reshape(-1)
        # Ensure integral is 1
        h = h / (np.sum(h) / n_points)
        # Compute C5
        c5 = evaluate_erdos_solution(h)
        return c5
    
    # Optimization using L-BFGS-B
    result = optimize.minimize(
        objective_func,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        tol=1e-6,
        options={'maxiter': 100}
    )
    
    # Update best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Check if we can increase n_points for better precision
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h = np.clip(h, 0, 1)
        h /= (np.sum(h) / n_points)
        
        # Refine with a more accurate optimization
        result = optimize.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=1e-8,
            options={'maxiter': 100}
        )
        
        best_h = result.x
        best_c5 = result.fun
    
    # Final check
    if time.time() - start_time < budget_s:
        n_points = 300
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h = np.clip(h, 0, 1)
        h /= (np.sum(h) / n_points)
        
        result = optimize.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=1e-8,
            options={'maxiter': 100}
        )
        
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, n_points
```
