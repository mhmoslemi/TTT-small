import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Initial parameters
    n_points = 95  # Starting with the initial sample size
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    
    # Ensure the initial h satisfies the constraint
    h = h / np.sum(h) * (n_points / 2)
    
    # Function to compute the overlap integral for a given h
    def compute_c5(h_values):
        h = h_values / np.sum(h_values) * (n_points / 2)
        # Ensure h is in [0,1]
        h = np.clip(h, 0, 1)
        # Compute the correlation and take the maximum
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Function to minimize the maximum overlap
    def objective(h_values):
        return compute_c5(h_values)
    
    # Function to check constraints
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Optimization using scipy.optimize
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        jac=False,
        options={'maxiter': 100, 'gtol': 1e-5}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Update n_points if needed, but keep it under 1k
    if best_c5 > 0.380876:
        # Try a slightly larger n_points
        n_points = 128
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # Distribute the mass evenly for a more refined search
        h[:int(n_points / 2)] = 1
        h[int(n_points / 2):] = 0
        # Perform another optimization
        def compute_c5_new(h_values):
            h = h_values / np.sum(h_values) * (n_points / 2)
            h = np.clip(h, 0, 1)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx
        
        def objective_new(h_values):
            return compute_c5_new(h_values)
        
        result_new = optimize.minimize(
            objective_new,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=False,
            options={'maxiter': 50, 'gtol': 1e-5}
        )
        
        best_h = result_new.x
        best_c5 = compute_c5_new(best_h)
    
    # Ensure we do not exceed the budget
    elapsed_time = time.time() - time.time()
    if elapsed_time >= budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
