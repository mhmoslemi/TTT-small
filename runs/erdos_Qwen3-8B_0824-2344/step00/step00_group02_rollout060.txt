def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)

    def objective(h):
        h = np.clip(h, 0, 1)
        correlation = signal.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(correlation) * (2.0 / len(h))
        return max_overlap

    def optimize_h(h_initial, n_points, max_iter=100):
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints += [cp.sum(h) == n_points / 2]
        objective = cp.Maximize(cp.min(cp.multiply(h, 1 - h)))
        prob = cp.Problem(objective, constraints)
        result = prob.solve(max_iter=100, eps=1e-6)
        if result is not None:
            return h.value
        return h_initial

    def refine_solution(h, n_points, iterations=5):
        for _ in range(iterations):
            h = optimize_h(h, n_points)
            correlation = signal.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(correlation) * (2.0 / len(h))
            print(f"Refinement step: C5 = {max_overlap}")
        return h

    best_h = h_initial
    best_c5 = evaluate_erdos_solution(h_initial)

    # Try with smaller n_points to find a good starting point
    for n_points in [40, 50, 60, 70, 80, 90, 100]:
        h = np.copy(initial_h_values[:n_points])
        h = h / np.sum(h) * (n_points / 2)
        h = refine_solution(h, n_points)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Try with higher n_points for better accuracy
    for n_points in [150, 200, 250, 300]:
        h = np.array([0.0] * n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = refine_solution(h, n_points)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Final refinement
    best_h = refine_solution(best_h, len(best_h))
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Try further refinement if time is left
        for _ in range(3):
            best_h = optimize_h(best_h, len(best_h))
            best_c5 = evaluate_erdos_solution(best_h)
            if best_c5 < 0.38080:
                break

    print(f"Best C5 found: {best_c5}")
    return best_h, best_c5, len(best_h)
