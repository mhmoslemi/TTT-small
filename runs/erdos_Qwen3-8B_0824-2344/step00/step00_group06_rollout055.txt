import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
from scipy.optimize.slsqp import minimize_slsqp
from scipy.optimize import minimize
import warnings

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

# Initial construction
initial_h_values = np.load('initial_h.npy')  # Load initial construction
n_points_initial = len(initial_h_values)
dx_initial = 2.0 / n_points_initial

# Convert initial h to a function to ensure it fits constraints
def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Set up parameters
    max_n_points = 500  # Maximum number of samples to consider
    start_time = time.time()
    
    # Define a function to evaluate C5
    def evaluate_c5(h):
        if len(h) < 2:
            return np.inf
        # Ensure h is on [0,1]
        h = np.clip(h, 0, 1)
        # Ensure integral is 1
        if np.abs(np.sum(h) * (2.0 / len(h)) - 1.0) > 1e-8:
            return np.inf
        # Compute overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / len(h))
        return c5
    
    # Define a function to optimize with constraints
    def objective(h):
        return evaluate_c5(h)
    
    # Define bounds for h
    bounds = Bounds(0, 1)
    
    # Start from initial solution and refine it
    h = initial_h_values.copy()
    n_points = len(h)
    
    # Try increasing n_points up to max_n_points
    for n_points in range(len(h), max_n_points + 1):
        # Check if we've exceeded time budget
        if time.time() - start_time > budget_s - 1:
            break
        
        # Resample h to n_points
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        h_new = np.zeros(n_points)
        for i in range(len(h)):
            idx = int((x[i] / 2) * n_points)
            h_new[idx] += h[i]
        h_new /= np.sum(h_new)
        
        # Now perform gradient descent with constraints
        result = minimize(
            objective,
            h_new,
            method='L-BFGS-B',
            bounds=bounds,
            jac=None,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        h = result.x
        current_c5 = result.fun
        
        # Print progress
        print(f"n_points={n_points}, C5={current_c5}")
    
    # Final evaluation
    final_c5 = evaluate_c5(h)
    print(f"Final C5={final_c5}")
    
    return h, final_c5, len(h)

# Run the function
if __name__ == "__main__":
    h, c5_bound, n_points = run()
    print(f"Final solution: h={h}, C5={c5_bound}, n_points={n_points}")
