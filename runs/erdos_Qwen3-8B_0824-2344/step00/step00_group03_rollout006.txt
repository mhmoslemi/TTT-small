import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
from scipy.optimize import minimize, Bounds
from scipy.sparse import diags
from scipy.sparse.linalg import spsolve
from functools import lru_cache
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is pre-loaded with a suitable initial guess

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess
    n_points = 200  # Start with 200 points for efficiency
    h = initial_h_values[:n_points].copy()
    h = np.minimum(np.maximum(h, 0), 1)
    
    # Ensure the integral condition
    total_integral = np.sum(h) * (2.0 / n_points)
    if abs(total_integral - 1) > 1e-6:
        h = np.random.rand(n_points)
        h = np.minimum(np.maximum(h, 0), 1)
        h /= (2.0 / n_points)
        total_integral = np.sum(h) * (2.0 / n_points)
    
    def objective(x):
        h = x.copy()
        h = np.minimum(np.maximum(h, 0), 1)
        c5 = evaluate_erdos_solution(h)
        return c5
    
    def constraint_integral(x):
        return np.sum(x) * (2.0 / n_points) - 1
    
    # Bounds for each h[i]
    bounds = Bounds(0, 1)
    
    # Optimization settings
    method = 'L-BFGS-B'
    max_iter = 50
    tol = 1e-5
    
    # Try multiple starting points
    best_c5 = float('inf')
    best_h = h
    
    # Run optimization
    for _ in range(3):  # Try multiple initializations
        h_initial = h.copy()
        np.random.shuffle(h_initial)
        h_initial = np.minimum(np.maximum(h_initial, 0), 1)
        h_initial /= (2.0 / n_points)
        h_initial = np.random.rand(n_points)
        h_initial = np.minimum(np.maximum(h_initial, 0), 1)
        h_initial /= (2.0 / n_points)
        
        result = minimize(objective, h_initial, method=method, bounds=bounds, tol=tol, options={'maxiter': max_iter})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    
    # Try increasing n_points if budget allows
    if time.time() - start_time < budget_s - 10:
        n_points = 300
        h = np.random.rand(n_points)
        h = np.minimum(np.maximum(h, 0), 1)
        h /= (2.0 / n_points)
        
        best_c5 = float('inf')
        best_h = h
        
        for _ in range(2):  # Try multiple initializations
            h_initial = h.copy()
            np.random.shuffle(h_initial)
            h_initial = np.minimum(np.maximum(h_initial, 0), 1)
            h_initial /= (2.0 / n_points)
            h_initial = np.random.rand(n_points)
            h_initial = np.minimum(np.maximum(h_initial, 0), 1)
            h_initial /= (2.0 / n_points)
            
            result = minimize(objective, h_initial, method=method, bounds=bounds, tol=tol, options={'maxiter': max_iter})
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
