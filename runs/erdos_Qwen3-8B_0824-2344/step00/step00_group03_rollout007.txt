import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Define initial number of points
    n_points_start = 100  # Start with 100 points, can increase later if time allows
    
    # Start with the initial construction
    h_initial = initial_h_values
    n_points = len(h_initial)
    
    # Rescale initial h to match the constraint sum(h) == n_points / 2
    h_initial = np.array(h_initial)
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Function to evaluate the C5 value
    def evaluate_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        # Compute the cross-correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # We are interested in the maximum value of the product with dx
        c5 = np.max(corr) * dx
        return c5
    
    # Function to evaluate the cost for optimization
    def objective(x):
        h = x.copy()
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the integral constraint
        integral = np.sum(h) * (2.0 / len(h))
        # Adjust h to satisfy the constraint
        if integral < 1.0:
            # Normalize
            scale = 1.0 / integral
            h = h * scale
        elif integral > 1.0:
            # Clip and scale
            h = np.clip(h, 0, 1)
            scale = 1.0 / np.sum(h) * (len(h) / 2.0)
            h = h * scale
        # Compute C5
        c5 = evaluate_c5(h)
        return c5
    
    # Function to optimize using scipy
    def optimize_h(start_h, max_iterations=100, tol=1e-5):
        h = start_h.copy()
        for _ in range(max_iterations):
            # Evaluate current C5
            current_c5 = evaluate_c5(h)
            # Compute gradient numerically
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                c5_plus = evaluate_c5(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                c5_minus = evaluate_c5(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            # Use gradient descent with momentum
            learning_rate = 0.1
            h = h - learning_rate * grad
            # Clip to [0, 1]
            h = np.clip(h, 0, 1)
            # Normalize to satisfy the integral constraint
            integral = np.sum(h) * (2.0 / len(h))
            if integral < 1.0:
                scale = 1.0 / integral
                h = h * scale
            elif integral > 1.0:
                h = np.clip(h, 0, 1)
                scale = 1.0 / np.sum(h) * (len(h) / 2.0)
                h = h * scale
            # Check for convergence
            if np.abs(evaluate_c5(h) - current_c5) < tol:
                break
        return h
    
    # Try optimizing with different starting points
    best_h = h_initial
    best_c5 = evaluate_c5(best_h)
    best_n_points = n_points
    
    # Attempt to optimize with different n_points
    for n_points in [100, 200, 300, 400, 500]:
        # Create a random starting point
        h_start = np.random.rand(n_points)
        h_start = h_start / np.sum(h_start) * (n_points / 2)
        # Optimize
        h_opt = optimize_h(h_start)
        c5 = evaluate_c5(h_opt)
        # Update best solution
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
            best_n_points = n_points
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Optionally, try using convex optimization
    # Define variables
    n = best_n_points
    h = cp.Variable(n)
    # Objective: minimize the maximum overlap (approximate via constraints)
    # We use an approximate approach to maximize the minimum overlap
    # For simplicity, we minimize the maximum value of the cross-correlation
    # We compute the cross-correlation and find its maximum
    # For convex optimization, we can use a convex relaxation
    # Since this is computationally expensive, we only do this if time allows
    if time.time() - start_time < budget_s - 5:
        # Define the cross-correlation and compute the maximum
        corr = cp.builtins.convolution(h, 1 - h)
        # The maximum is a non-convex function, so we use a convex approximation
        # Approximate maximum by adding constraints for all possible shifts
        # Since we can't compute all shifts, we approximate by checking the full convolution
        # This is a rough approximation
        constraints = [cp.sum(h) * (2.0 / n) == 1.0, 0 <= h, h <= 1.0]
        # We use a convex relaxation by setting a target C5 value
        target_c5 = best_c5
        # Add constraints to ensure all cross-correlation values are below target_c5
        for i in range(n):
            for j in range(n):
                corr_ij = cp.builtins.convolution(h, 1 - h)[i + j]
                constraints.append(corr_ij <= target_c5)
        # Solve the optimization problem
        problem = cp.Problem(cp.Minimize(0), constraints)
        try:
            problem.solve(solver=cp.MOSEK)
            if problem.status == 'OPTIMAL':
                h_opt = h.value
                c5 = evaluate_c5(h_opt)
                if c5 < best_c5:
                    best_h = h_opt
                    best_c5 = c5
                    best_n_points = n
        except:
            pass
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, best_n_points)
