```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Assuming this is a pre-loaded initial solution

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Set parameters
    n_points_list = [100, 200, 300]  # Try different grid sizes
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    start_time = time.time()
    
    for n_points in n_points_list:
        # Generate initial guess based on the provided initial solution
        if n_points < len(initial_h_values):
            h = np.zeros(n_points)
            h[:n_points] = initial_h_values[:n_points]
        else:
            h = np.zeros(n_points)
            h[:len(initial_h_values)] = initial_h_values
            # Fill remaining points with random values
            h[len(initial_h_values):] = np.random.uniform(0, 1, n_points - len(initial_h_values))
        
        # Normalize the integral to meet the constraint
        scale = (n_points / 2) / np.sum(h)
        h *= scale
        
        # Define the objective function and constraints
        def objective_function(h):
            # Evaluate the overlap integral for all k
            c5_values = []
            for k in range(1, len(h) + 1):
                shifted_h = np.roll(h, k)
                overlap = np.dot(h, 1 - shifted_h)
                c5_values.append(overlap)
            return max(c5_values)
        
        # Optimization using gradient descent
        def optimize_h(h_initial):
            h = h_initial.copy()
            learning_rate = 0.01
            max_iterations = 1000
            for _ in range(max_iterations):
                # Compute the objective function
                c5_values = []
                for k in range(1, len(h) + 1):
                    shifted_h = np.roll(h, k)
                    overlap = np.dot(h, 1 - shifted_h)
                    c5_values.append(overlap)
                current_c5 = max(c5_values)
                
                # Gradient descent step
                gradient = np.zeros_like(h)
                for k in range(1, len(h) + 1):
                    shifted_h = np.roll(h, k)
                    overlap = np.dot(h, 1 - shifted_h)
                    for i in range(len(h)):
                        gradient[i] += (1 - shifted_h[i]) * h[i] * (overlap - current_c5)
                h -= learning_rate * gradient
                h = np.clip(h, 0, 1)
                
                # Check if we're within the budget
                if time.time() - start_time > budget_s - 1:
                    break
            return h
        
        # Run optimization
        h = optimize_h(h)
        c5_value = evaluate_erdos_solution(h)
        
        # Update the best solution
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h
            best_n_points = n_points
        
        # Check if we're within the budget
        if time.time() - start_time > budget_s - 1:
            break
    
    return best_h, best_c5, best_n_points
```
