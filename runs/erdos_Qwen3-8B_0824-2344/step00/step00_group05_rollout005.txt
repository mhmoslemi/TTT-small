```python
import numpy as np
import scipy.optimize as optimize
import scipy.integrate as integrate
import time
import copy
import sys

# Global pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h_values.npy')  # Assuming this path is correct and contains the initial 70-point solution

# Define the run function
def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        return evaluate_erdos_solution(h)

    # Function to create an initial guess
    def create_initial_guess(n_points):
        if n_points == len(initial_h_values):
            return copy.copy(initial_h_values)
        else:
            # Start from a uniform distribution and perturb it
            h = np.ones(n_points) * 0.5
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            return h

    # Function to optimize the h_values
    def optimize_h(n_points, max_iterations=100):
        # Initial guess
        h_initial = create_initial_guess(n_points)
        # Ensure the integral constraint is met
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        # Objective: minimize the maximum overlap
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            fun=evaluate_overlap,
            x0=h_initial,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': max_iterations, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        return result.x

    # Function to try different n_points
    def try_n_points(n_points):
        start_time = time.time()
        h_optimized = optimize_h(n_points)
        # Check the constraint
        if np.abs(np.sum(h_optimized) - n_points / 2) > 1e-5:
            print(f"Constraint violation for n={n_points}, skipping.")
            return None, None
        c5 = evaluate_overlap(h_optimized)
        time_taken = time.time() - start_time
        return h_optimized, c5, time_taken

    best_h = None
    best_c5 = float('inf')
    best_n = 0
    elapsed_time = 0

    # Try different number of points
    for n_points in [70, 100, 150, 200, 250, 300, 350, 400]:
        if elapsed_time + 10 > budget_s:
            break
        h, c5, t = try_n_points(n_points)
        if h is not None:
            elapsed_time += t
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
                best_n = n_points
            print(f"n={n_points}, c5={c5:.6f}, time={t:.2f}s, total={elapsed_time:.2f}s")
        else:
            print(f"n={n_points} failed, time={t:.2f}s, total={elapsed_time:.2f}s")

    # Try a few more refined points if time allows
    if elapsed_time < budget_s:
        for n_points in [450, 500, 550, 600]:
            if elapsed_time + 10 > budget_s:
                break
            h, c5, t = try_n_points(n_points)
            if h is not None:
                elapsed_time += t
                if c5 < best_c5:
                    best_h = h
                    best_c5 = c5
                    best_n = n_points
                print(f"n={n_points}, c5={c5:.6f}, time={t:.2f}s, total={elapsed_time:.2f}s")
            else:
                print(f"n={n_points} failed, time={t:.2f}s, total={elapsed_time:.2f}s")

    # Return the best solution found
    return best_h, best_c5, best_n
```
