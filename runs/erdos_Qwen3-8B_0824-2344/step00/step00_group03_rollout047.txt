import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()[:n_points]
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        h = np.pad(h, (0, n_points - len(h)), 'constant')
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    # Function to compute objective for optimization
    def objective(h):
        return compute_c5(h)
    
    # Function to compute gradient for optimization
    def gradient(h):
        h = np.array(h)
        h = np.pad(h, (0, n_points - len(h)), 'constant')
        # Numerical gradient using finite differences
        h_plus = h.copy()
        h_plus[0] += 1e-5
        h_minus = h.copy()
        h_minus[0] -= 1e-5
        grad = (objective(h_plus) - objective(h_minus)) / (2 * 1e-5)
        return grad
    
    # Optimization settings
    max_iterations = 100
    tolerance = 1e-5
    
    # Initial guess
    h = h_initial
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for _ in range(max_iterations):
        # Compute current C5
        current_c5 = compute_c5(h)
        
        # Check if we are within budget
        if time.time() - start_time > budget_s:
            break
        
        # Use gradient descent
        grad = gradient(h)
        step_size = 0.1
        h_new = h - step_size * grad
        
        # Enforce constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new) if np.sum(h_new) != 0 else h_new
        
        # Update best solution
        if compute_c5(h_new) < best_c5:
            best_h = h_new.copy()
            best_c5 = compute_c5(h_new)
        
        h = h_new
    
    # Final check
    final_c5 = compute_c5(best_h)
    
    # Return the best solution found
    return best_h, final_c5, n_points
