```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess
    n_points = 100  # Start with 100 points for faster computation
    h = initial_h_values.copy()
    h = h[:n_points]  # Trim to n_points
    
    # Ensure the sum of h is n_points / 2
    if np.sum(h) != n_points / 2:
        h = h / np.sum(h) * (n_points / 2)
    
    # Define the function to evaluate C5
    def evaluate_C5(h_values):
        h = h_values.copy()
        dx = 2.0 / len(h)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Find the maximum value
        max_corr = np.max(corr)
        # Return the maximum overlap integral
        return max_corr * dx

    # Define the optimization problem
    def optimize_h(h_values):
        h = h_values.copy()
        h = h / np.sum(h) * (n_points / 2)  # Ensure the integral constraint is met
        return evaluate_C5(h)

    # Define the optimization function
    def optimize_func(x):
        # x is a vector of h values
        h_values = x
        return optimize_h(h_values)

    # Bounds for h values (0 to 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run the optimization
    start_time = time.time()
    result = optimize.minimize(
        optimize_func,
        h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    elapsed_time = time.time() - start_time

    # If time allows, increase the number of points
    if elapsed_time < budget_s - 5:
        n_points_new = min(2 * n_points, 1000)
        h_new = result.x
        # Resample to n_points_new
        h_resampled = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, n_points), h_new)
        # Ensure sum is correct
        h_resampled = h_resampled / np.sum(h_resampled) * (n_points_new / 2)
        # Evaluate
        c5_bound = evaluate_C5(h_resampled)
        return (h_resampled, c5_bound, n_points_new)
    
    # If time is up, return the best solution
    c5_bound = evaluate_C5(result.x)
    return (result.x, c5_bound, n_points)
```
