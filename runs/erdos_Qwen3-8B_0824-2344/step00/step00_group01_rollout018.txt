```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Predefined initial solution
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    
    # We will try various n_points values to find the best result within the budget
    n_points_options = [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]
    best_c5 = float('inf')
    best_h = h_initial
    best_n_points = n_points_initial
    
    for n_points in n_points_options:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Define the grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Objective: Minimize max_k ∫ h(x)(1 - h(x + k)) dx
        # We evaluate the correlation and find the maximum
        
        def objective_function(h):
            # Pad h to double the length for full convolution
            h_padded = np.zeros(2 * n_points)
            h_padded[:n_points] = h
            h_padded[n_points:] = h
            
            # Compute the correlation (full mode)
            corr = np.correlate(h_padded, 1 - h, mode="full")
            
            # Compute the integral (multiply by dx)
            corr_integrated = corr * dx
            
            # Find the maximum overlap integral
            max_overlap = np.max(corr_integrated)
            
            return max_overlap
        
        # Constraints:
        # 1. h[i] ∈ [0, 1]
        # 2. sum(h) * dx = 1 → sum(h) = n_points / 2
        
        # Initial guess: use the initial solution rescaled to the current n_points
        h_initial_rescaled = np.interp(x, np.linspace(0, 2, n_points_initial), initial_h_values)
        
        # Define the optimization problem
        h = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # We will optimize the objective using gradient-based methods since it's non-convex
        def optimize_h(h_initial_guess):
            result = optimize.minimize(
                objective_function,
                h_initial_guess,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
            )
            return result.x
        
        # Run the optimization
        h_opt = optimize_h(h_initial_rescaled)
        
        # Evaluate the result
        c5_bound = objective_function(h_opt)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt
            best_n_points = n_points
        
        # Early stopping if we have found the target (or better)
        if best_c5 <= 0.380876:
            break
    
    # Final evaluation
    final_c5 = objective_function(best_h)
    
    # Return the best solution
    return (best_h.tolist(), final_c5, best_n_points)
```
