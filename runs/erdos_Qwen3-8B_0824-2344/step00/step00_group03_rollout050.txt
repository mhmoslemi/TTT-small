```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted[k:] = h[:len(h) - k]
        overlap = np.correlate(h, 1 - shifted, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to optimize h
    def optimize_h(h_initial, n_points, budget_s):
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Define objective function to minimize the maximum overlap
        def objective(h_val):
            h_val = np.array(h_val)
            shifted = np.zeros_like(h_val)
            for k in range(1, len(h_val) - 1):
                shifted[k:] = h_val[:len(h_val) - k]
            overlap = np.correlate(h_val, 1 - shifted, mode="full")
            return np.max(overlap) * dx
        
        # Use a numerical optimization method
        best_h = None
        best_c5 = float('inf')
        for _ in range(100):  # Run multiple optimizations
            result = opt.minimize(
                lambda x: objective(x),
                x0=h_initial,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'ftol': 1e-6, 'maxiter': 100}
            )
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
            # Random perturbation to escape local optima
            perturb = np.random.normal(0, 0.05, size=n_points)
            h_initial = np.clip(best_h + perturb, 0, 1)
            elapsed = time.time() - start_time
            if elapsed > budget_s - 1:
                break
        return best_h, best_c5
    
    # Optimize the initial h values
    optimized_h, c5_bound = optimize_h(initial_h, n_points, budget_s)
    
    # Check if we need to reduce n_points further
    if n_points > 500:
        # Reduce to a smaller number of points
        n_points_new = 500
        dx_new = 2.0 / n_points_new
        # Resample the optimized h values
        optimized_h_resampled = np.zeros(n_points_new)
        for i in range(n_points_new):
            idx = int(i * n_points / n_points_new)
            optimized_h_resampled[i] = optimized_h[idx]
        # Normalize
        optimized_h_resampled = (optimized_h_resampled / np.sum(optimized_h_resampled)) * (n_points_new / 2)
        # Compute new C5
        c5_bound = compute_c5(optimized_h_resampled)
        optimized_h = optimized_h_resampled
        n_points = n_points_new
    
    # Final evaluation
    final_c5 = compute_c5(optimized_h)
    return optimized_h, final_c5, n_points
```
