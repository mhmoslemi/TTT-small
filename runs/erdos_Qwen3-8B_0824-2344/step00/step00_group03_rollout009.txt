```python
import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 samples, can be increased if needed
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]  # Ensure it fits the n_points

    # Constraint: sum(h) = n_points / 2
    sum_h = np.sum(h_initial)
    h_initial = h_initial * (n_points / 2) / sum_h

    def objective_function(h_values):
        # Convert to numpy array
        h = np.array(h_values, dtype=np.float64)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        h_1_minus_h = np.concatenate([1 - h, np.zeros_like(h)])
        correlation = np.correlate(h, h_1_minus_h, mode="full")
        c5 = np.max(correlation) * dx
        return c5

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    # Bounds: each h[i] ∈ [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    # Optimization setup
    result = opt.minimize(
        objective_function,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    # Get the best solution found
    best_h = result.x
    best_c5 = result.fun

    # If we have time, try increasing n_points and refine
    if time.time() - start_time < budget_s - 5:
        n_points = 150
        dx = 2.0 / n_points
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_initial = np.clip(h_initial, 0, 1)

        def objective_function_large(h_values):
            h = np.array(h_values, dtype=np.float64)
            h = np.clip(h, 0, 1)
            h_1_minus_h = np.concatenate([1 - h, np.zeros_like(h)])
            correlation = np.correlate(h, h_1_minus_h, mode="full")
            c5 = np.max(correlation) * dx
            return c5

        def constraint_sum_large(h_values):
            return np.sum(h_values) - n_points / 2

        bounds_large = [(0, 1) for _ in range(n_points)]

        result_large = opt.minimize(
            objective_function_large,
            h_initial,
            method='SLSQP',
            bounds=bounds_large,
            constraints={'type': 'eq', 'fun': constraint_sum_large},
            options={'ftol': 1e-8, 'maxiter': 50}
        )

        best_h = result_large.x
        best_c5 = result_large.fun

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    return best_h, final_c5, n_points
```
