import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Define the number of points; we'll start with 100 for balance between accuracy and speed
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h based on the current best known solution
    h = np.array(initial_h_values, dtype=float)[:n_points]
    h = h / np.sum(h) * (n_points / 2)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.pad(h, (0, 1), 'constant')  # Pad for correlation
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Function to evaluate the overlap for a given h
    def evaluate(h):
        return compute_c5(h)
    
    # Function to update h using a gradient descent-like approach
    def update_h(h, learning_rate=0.01, max_iter=100):
        for _ in range(max_iter):
            # Compute current C5
            current_c5 = evaluate(h)
            if current_c5 <= 0.380876:
                break
            
            # Compute the gradients by perturbing h
            h_perturb = h.copy()
            h_perturb += 1e-5
            h_perturb = np.clip(h_perturb, 0, 1)
            c5_perturb = evaluate(h_perturb)
            grad = (c5_perturb - current_c5) / 1e-5
            
            # Update h to minimize the overlap
            h -= learning_rate * grad
            
            # Ensure h remains within bounds
            h = np.clip(h, 0, 1)
        
        return h
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    # Try refining with different starting points and learning rates
    for _ in range(3):
        # Random perturbation of h
        h = best_h + np.random.uniform(-0.05, 0.05, size=n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Update h
        h = update_h(h)
        
        # Evaluate
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Check the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 1:
        return (best_h, best_c5, n_points)
    
    # Try another optimization strategy: CVXPY optimization
    # Define variables
    x = cp.Variable(n_points)
    constraints = [
        cp.sum(x) == n_points / 2,
        0 <= x,
        x <= 1
    ]
    
    # Objective function: minimize the maximum overlap
    # We approximate the maximum by a convex relaxation
    obj = cp.Minimize(cp.Max(cp.sum(cp.multiply(x, 1 - cp.roll(x, 1))) * dx))
    
    # Solve
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == 'OPTIMAL':
            best_h = x.value
            best_c5 = evaluate(best_h)
    except:
        pass
    
    return (best_h, best_c5, n_points)
