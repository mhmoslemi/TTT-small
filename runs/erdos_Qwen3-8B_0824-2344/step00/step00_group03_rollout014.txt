import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Set number of points for the function h
    n_points = 200  # A reasonable number for balancing speed and accuracy
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Initial guess from the provided initial_h_values (scaled to n_points)
    initial_h_values_scaled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    h = initial_h_values_scaled.copy()
    
    # Ensure the integral of h is exactly 1
    integral = np.sum(h) * dx
    if np.abs(integral - 1.0) > 1e-6:
        h = initial_h_values_scaled.copy()
        # Normalize the integral to 1
        norm_factor = 1.0 / integral
        h *= norm_factor
    
    def objective_function(h_vec):
        h_vec = h_vec.reshape(n_points)
        # Ensure h is in [0, 1]
        h_clamped = np.clip(h_vec, 0, 1)
        # Compute the overlap integral
        h_array = h_clamped.reshape(n_points)
        ones_minus_h = 1 - h_array
        # Compute the correlation
        correlation = np.correlate(h_array, ones_minus_h, mode="full")
        # Multiply by dx to get the integral
        c5 = np.max(correlation) * dx
        return c5
    
    def objective_function_with_gradient(h_vec):
        h_vec = h_vec.reshape(n_points)
        # Ensure h is in [0, 1]
        h_clamped = np.clip(h_vec, 0, 1)
        # Compute the overlap integral
        h_array = h_clamped.reshape(n_points)
        ones_minus_h = 1 - h_array
        # Compute the correlation
        correlation = np.correlate(h_array, ones_minus_h, mode="full")
        c5 = np.max(correlation) * dx
        
        # Compute the gradient of c5 with respect to h
        # Note: This is an approximation since the maximum may not be differentiable
        # We assume the maximum occurs at a specific shift
        max_index = np.argmax(correlation)
        shift = max_index - n_points + 1
        if shift < 0:
            shift = 0
        elif shift >= n_points:
            shift = n_points - 1
        
        # Compute the derivative of c5 with respect to h
        h_shifted = np.roll(h_array, shift)
        grad = np.zeros(n_points)
        grad += np.roll(ones_minus_h, shift) * dx
        grad -= h_shifted * dx
        return c5, grad.flatten()
    
    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    max_iter = 100
    tol = 1e-6
    
    # Start optimization
    result = opt.minimize(
        objective_function,
        h,
        method="L-BFGS-B",
        bounds=bounds,
        jac=None,
        options={"maxiter": max_iter, "ftol": tol}
    )
    
    # If optimization failed, try using a gradient-based method
    if result.success:
        best_h = result.x
    else:
        result = opt.minimize(
            objective_function_with_gradient,
            h,
            method="L-BFGS-B",
            bounds=bounds,
            jac=objective_function_with_gradient,
            options={"maxiter": max_iter, "ftol": tol}
        )
        best_h = result.x
    
    # Ensure the integral of h is exactly 1
    integral = np.sum(best_h) * dx
    if np.abs(integral - 1.0) > 1e-6:
        best_h = initial_h_values_scaled.copy()
        norm_factor = 1.0 / integral
        best_h *= norm_factor
    
    # Compute the final c5 value
    c5_bound = objective_function(best_h)
    
    # Ensure we stop before the budget runs out
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 1:
        return (best_h, c5_bound, n_points)
    
    # Optionally perform further refinement
    # For example, perform a local search
    local_search_result = opt.minimize(
        objective_function,
        best_h,
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 20, "ftol": 1e-6}
    )
    local_best_h = local_search_result.x
    local_c5 = objective_function(local_best_h)
    if local_c5 < c5_bound:
        best_h = local_best_h
        c5_bound = local_c5
    
    return (best_h, c5_bound, n_points)
