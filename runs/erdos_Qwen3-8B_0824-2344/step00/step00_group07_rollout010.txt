import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time
from scipy.signal import correlate

# Pre-imported functions
# evaluate_erdos_solution() is not used
# initial_h_values is available

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 80
    h_initial = initial_h_values
    
    # Ensure the initial h_initial has the correct number of points
    if len(h_initial) != n_points:
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        # Compute overlap integral
        result = correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        # Take maximum of the result
        return np.max(result)

    # Optimization problem: minimize C5
    def optimize_c5(h_values):
        h = np.array(h_values)
        # Compute C5
        c5 = compute_c5(h)
        return c5

    # Local optimization using scipy.optimize
    def optimize_local(h_initial):
        result = optimize.minimize(
            optimize_c5,
            h_initial,
            bounds=[(0, 1)] * n_points,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-7, 'eps': 1e-8}
        )
        return result.x, result.fun

    # Global optimization using simulated annealing
    def optimize_global(h_initial):
        result = optimize.differential_evolution(
            optimize_c5,
            bounds=[(0, 1)] * n_points,
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-7,
            maxiter=100,
            polish=True,
            disp=False
        )
        return result.x, result.fun

    # Try local optimization
    best_h, best_c5 = optimize_local(h_initial)

    # Try global optimization with a smaller population for speed
    # We'll use a smaller budget for global search
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Run global optimization
        global_h, global_c5 = optimize_global(best_h)
        
        # Update best if better
        if global_c5 < best_c5:
            best_h, best_c5 = global_h, global_c5
        
        # Run local optimization on the global result
        local_h, local_c5 = optimize_local(global_h)
        if local_c5 < best_c5:
            best_h, best_c5 = local_h, local_c5
        
        # If no improvement, break
        if np.abs(local_c5 - best_c5) < 1e-10:
            break

    # Ensure sum(h) * dx == 1
    dx = 2.0 / n_points
    sum_h = np.sum(best_h) * dx
    if np.abs(sum_h - 1.0) > 1e-10:
        # Normalize h
        best_h = best_h * (1.0 / sum_h)
    
    return best_h, best_c5, n_points
