import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Initial solution is provided as a global variable
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)

    # Function to compute overlap integral
    def compute_c5(h):
        h = np.array(h)
        overlap = signal.correlate(h, 1 - h, mode='full')
        return np.max(overlap) * (2.0 / len(h))

    # Function to optimize h using gradient descent
    def optimize_h(h_initial, max_iter=100, learning_rate=0.01, tolerance=1e-4):
        h = h_initial.copy()
        for _ in range(max_iter):
            # Compute current C5
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                if best_c5 < 0.380876:
                    break

            # Compute gradients (approximate using finite differences)
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                c5_plus = compute_c5(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                c5_minus = compute_c5(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)

            # Update h
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            # Ensure integral constraint is satisfied
            h[-1] = (1 - np.sum(h) * 2.0 / len(h)) / 2.0
            h = np.clip(h, 0, 1)

            # Early stopping if improvement is minimal
            if np.abs(current_c5 - compute_c5(h)) < tolerance:
                break

        return h

    # Try different numbers of points
    for n in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 5:
            break
        # Generate random initial guess
        h = np.random.rand(n)
        h[-1] = (1 - np.sum(h) * 2.0 / n) / 2.0
        h = np.clip(h, 0, 1)

        # Optimize
        h_opt = optimize_h(h, max_iter=50, learning_rate=0.01)

        # Compute final C5
        c5_val = compute_c5(h_opt)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            if best_c5 < 0.380876:
                break

    # Final check with initial solution
    initial_c5 = compute_c5(initial_h_values)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        best_h = initial_h_values.copy()

    # Ensure the best_h has the correct number of points
    if len(best_h) != n_points:
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)

    return (best_h, best_c5, len(best_h))
