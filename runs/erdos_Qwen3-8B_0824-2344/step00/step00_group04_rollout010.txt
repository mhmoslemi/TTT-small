def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    shifted_h = np.zeros(2 * n_points)
    shifted_h[:n_points] = h
    shifted_h[n_points:] = h[::-1]
    corr = np.correlate(h, 1 - h, mode="full")
    overlap = corr * dx
    return np.max(overlap)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial values
    h_values = np.copy(initial_h_values)
    n_points = len(h_values)
    dx = 2.0 / n_points
    
    # Helper function to compute C5 for a given h_values
    def compute_c5(h):
        return evaluate_erdos_solution(h)
    
    # Optimization objective: minimize C5
    def objective_func(h_values):
        return compute_c5(h_values)
    
    # Optimization constraints
    def constraints_func(h_values):
        # Ensure h_values is within [0, 1]
        lb = np.zeros_like(h_values)
        ub = np.ones_like(h_values)
        return [np.all(h_values >= lb), np.all(h_values <= ub)]
    
    # Gradient-based optimization
    def optimize():
        # Initial guess
        h = np.copy(h_values)
        
        # Optimization settings
        max_iter = 100
        tol = 1e-3
        history = []
        
        for _ in range(max_iter):
            start_time = time.time()
            # Gradient descent
            c5 = compute_c5(h)
            history.append(c5)
            
            # Stop if convergence
            if len(history) > 2 and np.abs(history[-1] - history[-2]) < tol:
                break
            
            # Gradient step (approximate using finite differences)
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(len(h)):
                h_plus = np.copy(h)

                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Adjust h to reduce C5
            h -= 0.01 * grad
            
            # Ensure bounds
            h = np.clip(h, 0, 1)
            
            # Check time budget
            elapsed_time = time.time() - start_time
            if elapsed_time > budget_s:
                break
        
        return h, compute_c5(h)
    
    # Run optimization
    best_h, best_c5 = optimize()
