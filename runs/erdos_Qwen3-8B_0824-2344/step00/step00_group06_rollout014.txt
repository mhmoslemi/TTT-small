```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Example initial construction (n=95 samples)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided data
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    total_area = h_initial.sum() * dx
    if not np.isclose(total_area, 1.0):
        h_initial = h_initial / total_area * 1.0

    # Try smaller n_points for faster computation
    for n_points in [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]:
        if time.time() - start_time > budget_s - 1.0:
            break
        
        # Generate random initial guess
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = h_initial / (h_initial.sum() * dx) * 1.0
        h_initial = np.clip(h_initial, 0, 1)
        
        # Define variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2]
        for k in range(1, n_points):
            overlap = cp.sum(h * (1 - h[k:])) * dx
            constraints.append(overlap <= cp.Variable())
        
        # Objective: minimize the maximum overlap
        max_overlap = cp.Variable()
        constraints += [max_overlap >= cp.sum(h * (1 - h[k:])) * dx for k in range(n_points)]
        objective = cp.Minimize(max_overlap)
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.ECOS, verbose=False)
            if result is not None and result < 0.380876:
                h_values = h.value
                c5_bound = result
                return (h_values, c5_bound, n_points)
        except:
            pass
    
    # If no improvement found, return initial construction
    h_values = initial_h_values
    c5_bound = evaluate_erdos_solution(h_values)
    return (h_values, c5_bound, len(h_values))
```
