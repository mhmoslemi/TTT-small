```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assume this is pre-loaded with n=70 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default parameters
    n_points = 200  # Starting with 200 samples for a balance between speed and accuracy
    max_iter = 100  # Maximum number of iterations for optimization
    tol = 1e-4  # Tolerance for convergence
    dx = 2.0 / n_points
    
    # Initial guess - start with the provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = np.resize(h_initial, n_points)  # Ensure it has the correct length
    
    # Constraint: sum(h) = n_points / 2
    sum_h = np.sum(h_initial) * dx
    if abs(sum_h - 1) > 1e-6:
        h_initial = np.resize(h_initial, n_points)
        h_initial = (h_initial / sum_h) * 1  # Normalize to meet the constraint
    
    # Objective function to minimize - we want to minimize the maximum overlap
    def objective_func(h):
        h = np.resize(h, n_points)  # Ensure it has the correct length
        # Normalize to satisfy the constraint
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1) > 1e-6:
            h = (h / sum_h) * 1  # Normalize
        # Compute overlap integral for all shifts
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Optimization using COBYLA
    result = optimize.minimize(
        objective_func,
        h_initial,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        tol=tol,
        options={'maxiter': max_iter}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # If the optimization takes too long, try reducing n_points
    if time.time() - start_time > budget_s - 5:
        n_points = 150
        dx = 2.0 / n_points
        h_initial = np.resize(best_h, n_points)
        h_initial = (h_initial / np.sum(h_initial) * dx) * 1  # Normalize
        # Re-evaluate with smaller n_points
        def objective_func_small(h):
            h = np.resize(h, n_points)
            sum_h = np.sum(h) * dx
            if abs(sum_h - 1) > 1e-6:
                h = (h / sum_h) * 1  # Normalize
            overlap = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(overlap)
        result_small = optimize.minimize(
            objective_func_small,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=tol,
            options={'maxiter': max_iter}
        )
        best_h = result_small.x
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Final check
    best_h = np.resize(best_h, n_points)
    sum_h = np.sum(best_h) * dx
    if abs(sum_h - 1) > 1e-6:
        best_h = (best_h / sum_h) * 1
    
    elapsed_time = time.time() - start_time
    print(f"Optimized in {elapsed_time:.2f} seconds")
    
    return best_h, best_c5, n_points
```
