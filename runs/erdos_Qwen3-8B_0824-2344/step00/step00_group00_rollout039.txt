import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
import matplotlib.pyplot as plt

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros((n_points, n_points))
    for i in range(n_points):
        shifted_h[i, :] = np.roll(h, i)
    overlap = np.mean(np.abs(h[:, np.newaxis] * (1 - shifted_h)), axis=0)
    return np.max(overlap)

def initial_h_values():
    # Initial construction with 45 points
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    best_n_points = 0
    
    # Start with initial construction
    n_points = len(initial_h_values())
    h = initial_h_values().copy()
    c5 = evaluate_erdos_solution(h)
    best_c5 = c5
    best_h = h
    best_n_points = n_points
    
    # Try increasing n_points up to a maximum of 1000
    for n_points in range(50, 1001, 50):
        # Ensure the function is smooth by regularizing the h_values
        h = np.random.rand(n_points) * 2 - 1
        h = (h + 1) / 2  # Normalize to [0, 1]
        h = h * (n_points / 2)  # Adjust to satisfy the integral constraint
        h = h / np.sum(h)
        
        # Optimization via gradient descent
        def objective(x):
            h = x.reshape((n_points,))
            dx = 2.0 / n_points
            if np.any(h < 0) or np.any(h > 1):
                return np.inf  # Penalize values outside [0, 1]
            shifted_h = np.zeros((n_points, n_points))
            for i in range(n_points):
                shifted_h[i, :] = np.roll(h, i)
            overlap = np.mean(np.abs(h[:, np.newaxis] * (1 - shifted_h)), axis=0)
            return np.max(overlap) * dx
        
        # Initial guess
        x0 = h.copy()
        
        # Perform optimization
        result = minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], options={'disp': False, 'maxiter': 100})
        
        # Check if the solution is valid
        if result.success:
            h_opt = result.x
            c5_opt = result.fun
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n_points = n_points
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, best_n_points
