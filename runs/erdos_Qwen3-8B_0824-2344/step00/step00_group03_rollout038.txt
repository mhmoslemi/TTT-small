```python
import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Start the timer
    start_time = time.time()
    
    # Initial number of points
    n_points = 100  # Starting with 100 points, can be adjusted if needed
    # We will try different n_points to find the best balance between accuracy and speed
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points
    
    # Try different n_points
    for n_points in [100, 200, 300, 400, 500, 600]:
        if time.time() - start_time > budget_s:
            break
        
        # Initial guess from the provided initial_h_values (interpolated to n_points)
        initial_h = initial_h_values
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        h = np.interp(x, np.linspace(0, 2, len(initial_h)), initial_h)
        
        # Define variables
        h_var = opt.Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1, np.sum(h_var) == n_points / 2]
        
        # Objective function
        def objective(h_vals):
            # Compute overlap integral
            h = h_vals
            shifted_h = np.roll(h, 1)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            return overlap
        
        # Use scipy optimization
        result = opt.minimize(objective, h, constraints=constraints, method='SLSQP', tol=1e-6)
        h_opt = result.x
        
        # Compute C5
        c5 = evaluate_erdos_solution(h_opt, n_points, dx)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n_points = n_points
    
    # Final check: try further refinement with best_n_points
    if time.time() - start_time < budget_s:
        n_points = best_n_points
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Refine the solution
        initial_h = best_h
        result = opt.minimize(objective, initial_h, constraints=constraints, method='SLSQP', tol=1e-8)
        h_opt = result.x
        c5 = evaluate_erdos_solution(h_opt, n_points, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    return best_h, best_c5, best_n_points
```
