import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can be increased if needed
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Initial guess based on initial_h_values if available
    if hasattr(globals(), 'initial_h_values'):
        h = initial_h_values[:n_points]  # Use only the first n_points samples
    else:
        # Default initial guess: 1/2 in the first half, 0 in the second half
        h = np.concatenate([np.ones(n_points // 2), np.zeros(n_points // 2)])
    
    # Ensure the integral constraint is satisfied
    sum_h = np.sum(h)
    if sum_h != n_points / 2:
        h = h * (n_points / 2) / sum_h
    
    def compute_c5(h_values):
        h = h_values.copy()
        # Pad h with zeros to handle full correlation
        h_padded = np.zeros(2 * len(h) - 1)
        h_padded[:len(h)] = h
        # Compute the full correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Multiply by dx and take maximum
        return np.max(corr * dx)
    
    # Objective: minimize C5
    def objective(x):
        return compute_c5(x)
    
    # Constraints: sum(x) == n_points / 2, 0 <= x[i] <= 1
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def constraint_low(x):
        return x - np.zeros_like(x)
    
    def constraint_high(x):
        return np.ones_like(x) - x
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    x0 = h.copy()
    
    # Optimization settings
    max_iter = 100
    tol = 1e-6
    
    # Use L-BFGS-B for optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        tol=tol,
        options={'maxiter': max_iter, 'ftol': 1e-8}
    )
    
    # Track elapsed time
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        print("Time budget exceeded, returning initial guess")
        return (h, compute_c5(h), n_points)
    
    # If optimization improved the result
    if result.success:
        best_h = result.x
        best_c5 = compute_c5(best_h)
    else:
        best_h = h
        best_c5 = compute_c5(h)
    
    # Try refining with a local optimizer
    # Use Sequential Quadratic Programming (SLSQP)
    result_sqp = optimize.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': constraint_low},
            {'type': 'ineq', 'fun': constraint_high}
        ],
        tol=1e-6,
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    if result_sqp.success:
        best_h = result_sqp.x
        best_c5 = compute_c5(best_h)
    else:
        best_h = best_h
        best_c5 = best_c5
    
    # Final check
    if np.sum(best_h) != n_points / 2:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h, best_c5, n_points)
