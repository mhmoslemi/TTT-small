```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define n_points and initial guess
    n_points = 100  # Start with 100 samples for optimization efficiency
    dx = 2.0 / n_points
    h = initial_h_values.copy()[:n_points]  # Start with initial guess
    
    # Function to compute C5
    def compute_c5(h_vals):
        h_val = np.array(h_vals)
        h_expanded = np.concatenate((h_val, h_val[:n_points//2]))
        one_minus_h = 1 - h_expanded
        corr = np.correlate(h_expanded, one_minus_h, mode="full")
        return np.max(corr) * dx
    
    # Optimization settings
    max_iter = 100  # Limit iterations to fit within budget
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Optimization loop
    start_time = time.time()
    for i in range(max_iter):
        # Convert to CVXPY variables for optimization
        h_var = cp.Variable(n_points, lb=0, ub=1)
        constraints = [cp.sum(h_var) == n_points / 2]
        
        # Objective: minimize the maximum overlap
        h_expanded = cp.vstack((h_var, h_var[:n_points//2]))
        one_minus_h = 1 - h_expanded
        corr = cp.multiply(h_expanded, one_minus_h)
        corr = cp.sum(corr, axis=1)
        corr = cp.max(corr)
        obj = cp.Minimize(corr)
        
        # Solve the optimization problem
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4, max_iters=100)
            if prob.status == 'optimal':
                h_new = h_var.value
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break

    # Final evaluation
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
```
