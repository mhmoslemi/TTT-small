def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility

    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        dx = 2.0 / len(h)
        shifted = np.zeros(2 * len(h) - 1)
        shifted[len(h)-1:] = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Function to optimize with a small step size and gradient descent
    def optimize_h(h, max_iterations=100, learning_rate=0.01, tolerance=1e-4):
        h_vals = np.copy(h)
        best_c5 = compute_c5(h_vals)
        best_h = h_vals
        for _ in range(max_iterations):
            # Compute the overlap integral for all k
            c5_values = []
            for k in range(len(h_vals) - 1):
                if k < len(h_vals):
                    overlap = np.sum(h_vals * (1 - h_vals[k:])) * dx
                    c5_values.append(overlap)
            current_c5 = np.max(c5_values)
            
            # Compute gradients for each position
            grad = np.zeros_like(h_vals)
            for i in range(len(h_vals)):
                if i < len(h_vals):
                    for k in range(len(h_vals)):
                        if k < len(h_vals):
                            overlap = np.sum(h_vals * (1 - h_vals[k:])) * dx
                            grad[i] += (overlap - current_c5) * 2 * h_vals[i] * (1 - h_vals[k:])
            
            # Update h_vals with gradient descent
            h_vals -= learning_rate * grad
            
            # Clip values to [0, 1]
            h_vals = np.clip(h_vals, 0, 1)
            
            # Check for convergence
            if np.abs(current_c5 - best_c5) < tolerance:
                break
            best_c5 = current_c5
            best_h = h_vals
        return best_h, best_c5
