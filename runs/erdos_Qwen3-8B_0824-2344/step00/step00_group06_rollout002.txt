```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h.npy')  # Sample initial solution with n=95 points

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Number of points in the step function
    dx = 2.0 / n_points
    
    # Initial guess based on the provided initial solution
    h = np.zeros(n_points)
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        # If initial solution is not available, initialize with uniform distribution
        h = np.random.rand(n_points)
        h /= np.sum(h) * dx  # Ensure ∫h(x)dx = 1
    
    # Define the overlap function and objective
    def compute_overlap(h):
        # Compute the overlap integral
        overlaps = np.correlate(h, 1 - h, mode="full")
        return np.max(overlaps) * dx
    
    # Optimization function
    def objective(h):
        return compute_overlap(h)
    
    # Bounds for h: [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization using scipy's minimize
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    )
    
    # Check if optimization was successful
    if result.success:
        h_values = result.x
    else:
        h_values = h
    
    # Evaluate the best solution found
    c5_bound = compute_overlap(h_values)
    
    # Stop early if time is up
    if time.time() - start_time > budget_s - 0.5:
        return h_values, c5_bound, n_points
    
    # Try a few more refinements with different starting points
    for _ in range(3):
        # Add small random perturbations to avoid local minima
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = h_values + perturbation
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)  # Ensure ∫h(x)dx = 1
        
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        )
        
        if result.success:
            h_values = result.x
        else:
            h_values = h
        
        # Check time budget
        if time.time() - start_time > budget_s - 0.5:
            break
    
    # Final evaluation
    c5_bound = compute_overlap(h_values)
    
    return h_values, c5_bound, n_points
```
