import numpy as np
import scipy.optimize as optimize
import time
import sys
import random
import cvxpy as cp
import copy

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial solution
    h_values = copy.deepcopy(initial_h_values)
    n_points = len(h_values)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h_values)
    
    # Try optimizing with different starting points
    for _ in range(3):
        # Random perturbation to avoid local optima
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_values = np.clip(h_values + perturb, 0.0, 1.0)
        # Ensure the integral is maintained
        h_values = h_values * (n_points / 2.0) / np.sum(h_values)
        c5 = evaluate_erdos_solution(h_values)
        if c5 < current_c5:
            current_c5 = c5
            best_h = h_values.copy()
    
    # Try a coarser grid for faster optimization
    n_points_coarse = int(n_points / 2)
    h_coarse = np.zeros(n_points_coarse)
    for i in range(n_points_coarse):
        h_coarse[i] = np.mean(h_values[2*i:2*i+2])
    h_coarse = h_coarse * (n_points_coarse / 2.0) / np.sum(h_coarse)
    c5_coarse = evaluate_erdos_solution(h_coarse)
    if c5_coarse < current_c5:
        current_c5 = c5_coarse
        best_h = h_coarse.copy()
    
    # Try local optimization using gradient descent
    def objective_func(h):
        h = h.reshape(-1)
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        return evaluate_erdos_solution(h)
    
    result = optimize.minimize(objective_func, h_values, method='L-BFGS-B', options={'maxiter': 100})
    if result.success:
        h_values = result.x
        h_values = np.clip(h_values, 0.0, 1.0)
        h_values = h_values * (n_points / 2.0) / np.sum(h_values)
        c5 = evaluate_erdos_solution(h_values)
        if c5 < current_c5:
            current_c5 = c5
            best_h = h_values.copy()
    
    # Final check with the best solution
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Return the result
    return (best_h, best_c5, len(best_h))
