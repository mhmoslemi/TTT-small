import numpy as np

from scipy.signal import correlate
import random

# Import pre-defined initial solution
initial_h_values = np.array([...])  # Placeholder for the actual initial h values (n=47)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    max_n_points = 256  # Maximum number of points for the function h
    n_points = 128  # Start with 128 points (can be adjusted if time permits)
    dx = 2.0 / n_points
    target_c5 = 0.380876
    
    # Initial guess
    h = np.zeros(n_points)
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)  # Ensure ∫h(x)dx = 1
    
    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h = h.reshape(-1)
        h_extended = np.concatenate([h, np.zeros(n_points)])
        one_minus_h = 1 - h
        one_minus_h_extended = np.concatenate([one_minus_h, np.zeros(n_points)])
        corr = correlate(h_extended, one_minus_h_extended, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute the objective and constraints
    def objective(h):
        return evaluate_c5(h)
    
    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - (n_points / 2)
    
    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': constraint_sum})
    
    # Run optimization
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
    best_h = result.x
    best_c5 = evaluate_c5(best_h)
    
    # Try a few more configurations with different n_points if time remains
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        n_points = 256
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
        best_h = result.x
        best_c5 = evaluate_c5(best_h)
    
    # Return best solution found
    return (best_h, best_c5, n_points)
