evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    h_values = np.zeros(n_points)
    h_values[:] = 0.5  # Initial guess

    # Evaluate initial solution
    c5_initial = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    best_c5 = c5_initial

    # Define objective and constraints
    def objective_function(h):
        return evaluate_erdos_solution(h)

    # Use a gradient descent approach with momentum
    learning_rate = 0.01
    momentum = 0.9
    max_iterations = 1000
    tolerance = 1e-6

    for _ in range(max_iterations):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break

        # Evaluate current solution
        current_c5 = objective_function(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()

        # Random perturbation to escape local minima
        perturbation = np.random.normal(0, 0.05, n_points)
        h_values += perturbation
        h_values = np.clip(h_values, 0, 1)

        # Ensure the integral constraint is satisfied
        total_integral = np.sum(h_values) * dx
        if abs(total_integral - 1) > 1e-6:
            scale_factor = 1.0 / total_integral
            h_values *= scale_factor

    # Return the best solution found
    return (best_h, best_c5, n_points)
