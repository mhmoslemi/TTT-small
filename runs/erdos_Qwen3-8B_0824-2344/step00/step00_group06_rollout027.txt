import numpy as np

def run(seed=42, budget_s=60, **kwargs):

    start_time = time.time()

    # Function to compute C₅ for a given h_values
    def compute_c5(h_values, dx):
        h = np.array(h_values)
        h_shifted = np.concatenate([np.zeros_like(h), h[:-1]])
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    # Initial construction from initial_h_values

    initial_n_points = len(initial_h_values)
    initial_dx = 2.0 / initial_n_points
    initial_c5 = compute_c5(initial_h_values, initial_dx)
    
    # Start with the initial solution
    best_h = initial_h_values.copy()
    best_c5 = initial_c5
    best_n_points = initial_n_points

    # Try with a smaller number of points (start from 100 and go up to 500)
    for n_points in range(100, 501):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        h[0] = 1.0  # Start with a simple initialization

        # Define variables
        x = cp.Variable(n_points, integer=True)
        constraints = [
            cp.sum(x) == n_points / 2,  # Normalize to 1 in integral (sum(x) * dx = 1)
            0 <= x,
            x <= n_points
        ]

        # Objective: Minimize max overlap
        # We approximate the max overlap by discretizing possible shifts
        # and minimizing the maximum of the overlap integrals
        # For efficiency, sample a subset of shifts
        shift_indices = np.arange(0, n_points - 1)
        shift_indices = shift_indices[::10]  # Sample every 10th shift for efficiency

        obj_list = []
        for shift in shift_indices:
            shifted_h = np.roll(h, shift)
            overlap = np.correlate(h, 1 - shifted_h, mode='full')
            obj_list.append(overlap * dx)

        obj = cp.Minimize(cp.Max(obj_list))

        # Solve the problem
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.GUROBI, time_limit=budget_s - (time.time() - start_time))
        except:
            pass  # Skip if solver fails

        if prob.status == 'optimal':
            h_values = x.value
            if np.max(h_values) > n_points:
                h_values = np.clip(h_values, 0, n_points)
            h_values = h_values / n_points  # Normalize to [0, 1]
            current_c5 = compute_c5(h_values, dx)

            if current_c5 < best_c5:
                best_h = h_values.copy()
                best_c5 = current_c5
                best_n_points = n_points

    # Final check with the best solution
    best_c5 = compute_c5(best_h, 2.0 / best_n_points)
    
    return best_h, best_c5, best_n_points
