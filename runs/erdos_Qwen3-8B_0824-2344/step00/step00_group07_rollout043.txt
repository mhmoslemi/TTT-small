import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Starting point
    n_points = 80
    h = initial_h_values.copy()
    h = h[:n_points]
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = np.array([0.5] * n_points)
    
    # Function to compute C5
    def compute_c5(h_values):
        dx = 2.0 / len(h_values)
        h = np.array(h_values)
        shift = np.arange(len(h))
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    # Function to optimize h for minimal C5
    def optimize_h(h_values, n_points, budget_s):
        h = h_values.copy()
        h = h[:n_points]
        dx = 2.0 / n_points
        sum_h = np.sum(h)
        
        # Constraint: sum(h) * dx = 1
        # Objective: minimize max_k ∫ h(x)(1 - h(x + k)) dx
        
        # We use a heuristic optimization approach due to complexity of the problem
        # We try a few optimization steps with different initializations
        
        # Try initial optimization
        for _ in range(3):
            if time.time() - start_time > budget_s:
                break
            
            # Use L-BFGS-B with bounds and constraints
            def objective(x):
                h_new = x[:n_points]
                if np.any(h_new < 0) or np.any(h_new > 1):
                    return np.inf
                return compute_c5(h_new)
            
            def constraint(x):
                h_new = x[:n_points]
                return np.sum(h_new) - n_points / 2
            
            x0 = h.copy()
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint}
            
            result = optimize.minimize(
                objective, x0, bounds=bounds, constraints=cons,
                method='L-BFGS-B', tol=1e-5, options={'ftol': 1e-5, 'maxiter': 100}
            )
            
            h = result.x[:n_points]
        
        return h
    
    # Optimize
    h_opt = optimize_h(h, n_points, budget_s)
    
    # Compute final C5
    c5 = compute_c5(h_opt)
    
    # Check if we can reduce n_points and improve further
    # Try with n_points = 60
    n_points = 60
    dx = 2.0 / n_points
    h_red = np.array([0.5] * n_points)
    h_red = optimize_h(h_red, n_points, budget_s - (time.time() - start_time))
    c5_red = compute_c5(h_red)
    
    # Choose the better solution
    if c5_red < c5:
        h_opt = h_red
        c5 = c5_red
        n_points = 60
    
    # Try with n_points = 40
    n_points = 40
    dx = 2.0 / n_points
    h_red = np.array([0.5] * n_points)
    h_red = optimize_h(h_red, n_points, budget_s - (time.time() - start_time))
    c5_red = compute_c5(h_red)
    
    if c5_red < c5:
        h_opt = h_red
        c5 = c5_red
        n_points = 40
    
    # Final check if we can reduce further
    if time.time() - start_time < budget_s:
        n_points = 30
        dx = 2.0 / n_points
        h_red = np.array([0.5] * n_points)
        h_red = optimize_h(h_red, n_points, budget_s - (time.time() - start_time))
        c5_red = compute_c5(h_red)
        
        if c5_red < c5:
            h_opt = h_red
            c5 = c5_red
            n_points = 30
    
    # Final check if we can reduce further
    if time.time() - start_time < budget_s:

        h_red = np.array([0.5] * n_points)
        h_red = optimize_h(h_red, n_points, budget_s - (time.time() - start_time))
        c5_red = compute_c5(h_red)
        
        if c5_red < c5:
            h_opt = h_red
            c5 = c5_red
            n_points = 20
    
    return h_opt, c5, n_points
