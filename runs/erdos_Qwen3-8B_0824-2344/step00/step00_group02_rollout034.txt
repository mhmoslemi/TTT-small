import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import scipy.sparse as sp
import scipy.sparse.linalg as sla
import time
import sys

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Define parameters
    max_points = 1000  # Maximum number of samples
    n_points = 100  # Start with 100 points, will increase if needed
    max_iterations = 100  # Maximum number of optimization iterations
    tol = 1e-4  # Tolerance for optimization
    max_time = budget_s  # Time budget in seconds
    start_time = time.time()
    
    # Initialize h_values
    h_values = initial_h_values.copy()
    n_points = len(h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h_values)
    if abs(sum_h - n_points / 2) > 1e-6:
        h_values = initial_h_values.copy()
        h_values = h_values / np.sum(h_values) * (n_points / 2)
    
    def compute_c5(h):
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, h[:1]])  # Pad for full convolution
        conv = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    def objective(x):
        # Objective function to minimize C5
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.concatenate([h, h[:1]])  # Pad for full convolution
        conv = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    def update_h_values(h):
        # Ensure sum of h is exactly n_points / 2
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            h = h / sum_h * (n_points / 2)
        return h
    
    # Optimization using L-BFGS-B
    result = opt.minimize(
        fun=objective,
        x0=h_values,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        tol=tol,
        options={'maxiter': max_iterations, 'disp': False}
    )
    
    best_c5 = compute_c5(result.x)
    best_h = update_h_values(result.x)
    
    # Check if we can improve with more points
    if time.time() - start_time < max_time - 5:
        n_points = 200
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        # Initial guess: evenly spaced points
        h_values[::2] = 1.0  # Place 1s at even indices
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        h_values = np.clip(h_values, 0, 1)
        
        def objective_high_res(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            h = np.concatenate([h, h[:1]])  # Pad for full convolution
            conv = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(conv) * dx
            return c5
        
        result_high_res = opt.minimize(
            fun=objective_high_res,
            x0=h_values,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=tol,
            options={'maxiter': max_iterations, 'disp': False}
        )
        
        high_res_c5 = compute_c5(result_high_res.x)
        high_res_h = update_h_values(result_high_res.x)
        
        if high_res_c5 < best_c5:
            best_c5 = high_res_c5
            best_h = high_res_h
            n_points = 200
    
    # Further refinement with even higher resolution if time allows
    if time.time() - start_time < max_time - 5:
        n_points = 300
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        h_values[::3] = 1.0  # Initial guess with 1s every third point
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        h_values = np.clip(h_values, 0, 1)
        
        def objective_higher_res(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            h = np.concatenate([h, h[:1]])  # Pad for full convolution
            conv = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(conv) * dx
            return c5
        
        result_higher_res = opt.minimize(
            fun=objective_higher_res,
            x0=h_values,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=tol,
            options={'maxiter': max_iterations, 'disp': False}
        )
        
        higher_res_c5 = compute_c5(result_higher_res.x)
        higher_res_h = update_h_values(result_higher_res.x)
        
        if higher_res_c5 < best_c5:
            best_c5 = higher_res_c5
            best_h = higher_res_h
            n_points = 300
    
    # Final check for time
    if time.time() - start_time < max_time - 5:
        n_points = 400
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        h_values[::4] = 1.0  # Initial guess with 1s every fourth point
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        h_values = np.clip(h_values, 0, 1)
        
        def objective_final(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            h = np.concatenate([h, h[:1]])  # Pad for full convolution
            conv = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(conv) * dx
            return c5
        
        result_final = opt.minimize(
            fun=objective_final,
            x0=h_values,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=tol,
            options={'maxiter': max_iterations, 'disp': False}
        )
        
        final_c5 = compute_c5(result_final.x)
        final_h = update_h_values(result_final.x)
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
            n_points = 400
    
    # Final output
    return (best_h, best_c5, n_points)
