import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can be increased if needed
    
    # Initial guess
    h = initial_h_values[:]
    h = np.array(h)
    h = h[:n_points]  # Ensure it fits the n_points
    h = h / np.sum(h) * (n_points / 2)  # Ensure the integral is 1

    # Function to evaluate the overlap integral
    def evaluate_c5(h_values):
        h = h_values.copy()
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Extract the relevant part (shifts from -len(h)+1 to len(h)-1)
        # The max overlap corresponds to the maximum of the correlation
        c5 = np.max(corr) * dx
        return c5

    # Optimization function
    def optimize_h(h_values):
        nonlocal best_c5, best_h
        h = h_values.copy()
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Simulated annealing
    def simulated_annealing(h_initial, temp=1000, cooling_rate=0.995, iterations=1000):
        h = h_initial.copy()
        h = h / np.sum(h) * (n_points / 2)  # Ensure the integral is 1
        best_h = h.copy()
        best_c5 = evaluate_c5(h)
        for i in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
            # Perturb the solution
            idx = np.random.randint(0, len(h))
            h[idx] += np.random.uniform(-0.1, 0.1)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)  # Ensure the integral is 1
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            # Accept worse solution with probability
            if np.random.rand() < np.exp((best_c5 - current_c5) / temp):
                best_h = h.copy()
                best_c5 = current_c5
            temp *= cooling_rate
        return best_h, best_c5

    # Run the optimization
    h_opt, c5_opt = simulated_annealing(h, temp=1000, cooling_rate=0.995, iterations=1000)
    n_points = len(h_opt)
    elapsed_time = time.time() - start_time

    # If time remains, try more points
    if elapsed_time < budget_s - 2:
        n_points = 200
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)  # Ensure the integral is 1
        h_opt, c5_opt = simulated_annealing(h, temp=1000, cooling_rate=0.995, iterations=1000)
        n_points = len(h_opt)
        elapsed_time = time.time() - start_time

    # Final check
    if elapsed_time < budget_s - 2:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)  # Ensure the integral is 1
        h_opt, c5_opt = simulated_annealing(h, temp=1000, cooling_rate=0.995, iterations=1000)
        n_points = len(h_opt)

    return best_h, best_c5, n_points
