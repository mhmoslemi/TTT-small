    best_n = n_points_initial
    
    for n_points in n_points_list:
        start_time = time.time()
        
        # Generate grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initial guess: interpolate the initial function
        if n_points > n_points_initial:
            # Interpolate initial h to new grid
            h_interp = np.interp(x, np.linspace(0, 2, n_points_initial), h_initial)
        else:
            h_interp = h_initial[:n_points]
        
        # Define bounds for h
        bounds = Bounds(0.0, 1.0)
        
        # Objective function: minimize max overlap
        def objective(h):
            h = h.reshape(-1)
            # Compute overlaps for all shifts k
            overlaps = np.zeros(n_points)
            for k in range(1, n_points):
                # Compute (1 - h(x + k)) for x in [0, 2 - k]
                shifted_h = np.roll(h, k)
                shifted_h = np.where(shifted_h > 2, 0, shifted_h)
                shifted_h = np.where(shifted_h < 0, 0, shifted_h)
                overlap = np.sum(h * (1 - shifted_h)) * dx
                overlaps[k - 1] = overlap
            return np.max(overlaps)
        
        # Optimization: minimize the max overlap using L-BFGS-B
        result = optimize.minimize(
            objective,
            h_interp,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        # Evaluate the result
        h_opt = result.x
        c5_opt = objective(h_opt)
        
        # Check if this is the best solution found
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_points
        
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    return best_h, best_c5, best_n
