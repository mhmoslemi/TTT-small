```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial h_values (n=70)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h) * dx
    if np.abs(sum_h - 1) > 1e-6:
        h = h * (1 / sum_h) * dx
    
    def compute_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        shifted = np.zeros(2 * n - 1)
        for i in range(n):
            shifted[i:i + n] += h[i] * (1 - h[i + n - 1: n + i - 1])
        return np.max(shifted) * dx
    
    c5 = compute_c5(h)
    
    # Try improving by adjusting h
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb h slightly
        perturbation = np.random.rand(len(h)) * 0.1 - 0.05
        h_perturbed = np.clip(h + perturbation, 0, 1)
        sum_h_perturbed = np.sum(h_perturbed) * dx
        if np.abs(sum_h_perturbed - 1) > 1e-6:
            scale = 1 / sum_h_perturbed
            h_perturbed = h_perturbed * scale * dx
        c5_perturbed = compute_c5(h_perturbed)
        
        if c5_perturbed < c5:
            h = h_perturbed
            c5 = c5_perturbed
    
    # Try a finer grid
    if time.time() - start_time < budget_s - 1:
        n_points = 100
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 0.5
            else:
                h[i] = 0.5
        
        # Optimize using CVXPY
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var <= 1,
            h_var >= 0
        ]
        
        objective = cp.Minimize(cp.Max(cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, 1, axis=0)))) * dx)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            h = h_var.value
            c5 = compute_c5(h)
        except:
            pass
    
    # Final check
    if time.time() - start_time > budget_s - 1:
        return h, c5, n_points
    
    return h, c5, n_points
```
