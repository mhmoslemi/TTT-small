import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions for evaluation and initial solution
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial number of points
    n_points = 80  # Starting with the provided initial solution
    
    # Initial h_values
    h = initial_h_values.copy()
    
    # Objective function to minimize the maximum overlap integral
    def objective_function(h_values):
        # Ensure h_values is in [0, 1] and sum to n_points / 2
        h_values = np.clip(h_values, 0, 1)
        sum_h = np.sum(h_values)
        if abs(sum_h - n_points / 2) > 1e-6:
            return np.inf  # Penalize if sum is not correct
        
        # Compute the overlap integral
        dx = 2.0 / n_points
        h_array = np.array(h_values)
        conv = signal.correlate(h_array, 1 - h_array, mode='full')
        c5 = np.max(conv) * dx
        
        return c5
    
    # Optimize using COBYLA (Constraint Optimization BY Linear Approximation)
    result = opt.minimize(
        objective_function,
        h,
        method='COBYLA',
        options={'maxiter': 100, 'disp': False}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = result.fun
    
    # Check if we can reduce n_points further
    # Try smaller n_points, but keep at least 40 for stability
    for n_points in range(40, 80, 10):  # Try 40, 50, 60, 70
        h_new = np.random.rand(n_points)
        sum_h = np.sum(h_new)
        h_new = h_new * (n_points / 2) / sum_h if sum_h != 0 else h_new
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate the new solution
        dx = 2.0 / n_points
        h_array = h_new
        conv = signal.correlate(h_array, 1 - h_array, mode='full')
        c5_new = np.max(conv) * dx
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            n_points = n_points
    
    # Final check on the best solution
    h_array = np.array(best_h)
    dx = 2.0 / n_points
    conv = signal.correlate(h_array, 1 - h_array, mode='full')
    best_c5 = np.max(conv) * dx
    
    # Ensure final h is in [0, 1] and sum condition is satisfied
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    best_h = best_h * (n_points / 2) / sum_h if sum_h != 0 else best_h
    
    # Calculate final C5
    final_c5 = best_c5
    
    # Ensure we don't exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, final_c5, n_points)
    
    # Optional: Run a more refined optimization step
    result = opt.minimize(
        objective_function,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 50, 'disp': False}
    )
    best_h = result.x
    best_c5 = result.fun
    
    # Final evaluation
    h_array = np.array(best_h)
    dx = 2.0 / n_points
    conv = signal.correlate(h_array, 1 - h_array, mode='full')
    final_c5 = np.max(conv) * dx
    
    # Ensure final h is valid
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    best_h = best_h * (n_points / 2) / sum_h if sum_h != 0 else best_h
    
    return (best_h, final_c5, n_points)
