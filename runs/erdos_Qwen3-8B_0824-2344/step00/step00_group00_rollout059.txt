import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import sys

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 45  # Starting with the initial n_points
    max_n_points = 100  # Max n_points to consider
    
    # Try a few different n_points to find a good value
    for n_points in range(45, max_n_points + 1):
        if time.time() - start_time > budget_s - 1:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initial guess: use the initial_h_values scaled to the current n_points
        initial_h = np.zeros(n_points)
        initial_h_values_scaled = initial_h_values * (n_points / 45)
        initial_h = np.round(np.interp(x, np.linspace(0, 2, 45), initial_h_values_scaled)).astype(float)
        
        # Ensure the sum of h is n_points / 2
        initial_sum = np.sum(initial_h)
        if abs(initial_sum - n_points / 2) > 1e-5:
            # Adjust the initial guess to meet the constraint
            adjustment = (n_points / 2 - initial_sum) / n_points
            initial_h += adjustment
        
        # Set up variables for optimization
        h = cp.Variable(n_points, nonneg=True, boolean=False)
        constraints = [cp.sum(h) == n_points / 2]
        obj = cp.Minimize(0)  # We minimize by evaluating the cost function directly
        
        # Define the cost function: max over all k of the overlap integral
        # Use a discrete approximation of the integral
        def compute_c5(h):
            h_values = h.value
            h_values = np.clip(h_values, 0, 1)
            h_values = np.pad(h_values, (0, 1), mode='constant')  # Add a zero to handle full convolution
            conv = signal.correlate(h_values, 1 - h_values, mode='full')
            return np.max(conv) * dx
        
        # Use a simple gradient-based optimization approach
        try:
            result = optimize.minimize(compute_c5, initial_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
            h_values = result.x
            current_c5 = compute_c5(h_values)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_values
                print(f"Found better solution with n={n_points}, c5={best_c5:.6f}")
        except:
            pass
        
        # Early stopping if we've found a good enough solution
        if best_c5 < 0.380876:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    print(f"Final solution with n={n_points}, c5={final_c5:.6f}")
    
    return best_h, final_c5, n_points
