import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h_values = np.copy(initial_h_values)
    best_c5 = evaluate_erdos_solution(h_values)
    
    # Define the number of points for the current search
    n_points_search = n_points
    if n_points_search < 100:
        n_points_search = 100
    
    # Perform a local optimization
    def objective_function(h):
        # Ensure h is in [0,1] and has integral 1
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * (2.0 / n_points_search)
        if abs(integral - 1.0) > 1e-6:
            return np.inf
        # Compute the overlap integral
        h = np.concatenate((h, h[:2]))
        shifted = np.roll(h, shift=1)
        overlap = np.sum(h * (1 - shifted)) * (2.0 / n_points_search)
        return overlap
    
    def objective_function_with_gradient(h):
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * (2.0 / n_points_search)
        if abs(integral - 1.0) > 1e-6:
            return np.inf, np.zeros_like(h)
        
        # Compute the overlap integral
        h = np.concatenate((h, h[:2]))
        shifted = np.roll(h, shift=1)
        overlap = np.sum(h * (1 - shifted)) * (2.0 / n_points_search)
        
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points_search):
            if i < n_points_search - 1:
                shifted_i = np.roll(h, shift=i+1)
                grad[i] = (1 - shifted_i[i+1]) - (1 - shifted_i[i])
            else:
                shifted_i = np.roll(h, shift=i+1)
                grad[i] = (1 - shifted_i[i+1]) - (1 - shifted_i[i])
        
        # Normalize gradient
        grad = (grad * (2.0 / n_points_search)) / (np.sum(h) * (2.0 / n_points_search))
        return overlap, grad
    
    # Perform a gradient descent optimization
    h_opt = np.copy(h_values)
    learning_rate = 0.01
    max_iterations = 1000
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute objective and gradient
        obj, grad = objective_function_with_gradient(h_opt)
        # Update h_opt
        h_opt = h_opt - learning_rate * grad
        # Clip to [0, 1]
        h_opt = np.clip(h_opt, 0, 1)
        # Ensure integral constraint
        integral = np.sum(h_opt) * (2.0 / n_points_search)
        if abs(integral - 1.0) > 1e-6:
            h_opt = h_opt * (1.0 / integral) * (2.0 / n_points_search)
        
        # Update best solution
        current_c5 = evaluate_erdos_solution(h_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_values = h_opt
    
    # Perform a grid search with perturbations
    perturbations = [0.01, 0.02, 0.05, 0.1]
    for perturbation in perturbations:
        if time.time() - start_time > budget_s:
            break
        
        for i in range(n_points_search):
            # Perturb the value at index i
            h_perturbed = np.copy(best_h_values)
            h_perturbed[i] += np.random.uniform(-perturbation, perturbation)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            integral = np.sum(h_perturbed) * (2.0 / n_points_search)
            if abs(integral - 1.0) > 1e-6:
                h_perturbed = h_perturbed * (1.0 / integral) * (2.0 / n_points_search)
            
            current_c5 = evaluate_erdos_solution(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_values = h_perturbed
    
    # Return the best solution
    return (best_h_values, best_c5, n_points_search)
