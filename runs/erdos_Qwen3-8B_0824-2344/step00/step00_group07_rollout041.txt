import scipy.optimize as opt
import time

    def evaluate_c5(h):

        h = np.pad(h, (0, n_points), mode='constant', constant_values=0)
        h = h[:n_points]
        h = np.pad(h, (0, n_points), mode='constant', constant_values=0)
        h_overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(h_overlap) * dx
        return c5

    def objective_func(h_values):
        h = np.array(h_values)
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        return evaluate_c5(h)

    def constraint_func(h_values):
        h = np.array(h_values)
        return np.sum(h) - n_points / 2

    # Initial guess
    h_initial = initial_h.copy()
    h_initial = h_initial[:n_points]
    
    # Simple local optimization
    def local_optimize(h_initial):
        result = opt.minimize(objective_func, h_initial, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-6)
        return result.x
    
    start_time = time.time()
    best_h = h_initial
    best_c5 = evaluate_c5(best_h)
    best_n_points = n_points
    
    # Try local optimization
    h_local = local_optimize(h_initial)
    c5_local = evaluate_c5(h_local)
    if c5_local < best_c5:
        best_h = h_local
        best_c5 = c5_local
