import numpy as np
import scipy.optimize as optimize
import time
import copy

# Import pre-defined initial solution
from evaluate_erdos_solution import initial_h_values, evaluate_erdos_solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100  # Start with 100 points for a good balance between speed and accuracy
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values[:n_points])
    initial_h = initial_h / (np.sum(initial_h) * dx) * (n_points / 2)  # Normalize to satisfy integral constraint
    
    def objective(h_values):
        # Ensure h_values is within [0, 1]
        h_values = np.clip(h_values, 0, 1)
        # Ensure integral constraint is satisfied
        if np.sum(h_values) != n_points / 2:
            return float('inf')
        # Compute the overlap integral
        h = h_values.reshape(-1, 1)
        one_minus_h_shifted = 1 - h[1:]
        overlap = np.dot(h[:-1], one_minus_h_shifted)
        c5 = np.max(overlap) * dx
        return c5
    
    def objective_with_gradient(h_values):
        h_values = np.clip(h_values, 0, 1)
        if np.sum(h_values) != n_points / 2:
            return float('inf'), np.zeros_like(h_values)
        h = h_values.reshape(-1, 1)
        one_minus_h_shifted = 1 - h[1:]
        overlap = np.dot(h[:-1], one_minus_h_shifted)
        c5 = np.max(overlap) * dx
        # Compute gradient (approximate)
        grad = np.zeros_like(h_values)
        for i in range(n_points - 1):
            # For each point, compute how it affects the max overlap
            h_i = h_values[i]
            h_i_plus_1 = h_values[i + 1]
            # Compute derivative contribution
            contribution = 0.0
            for j in range(n_points - 1):
                if i == j:
                    contribution += (1 - h_values[j + 1]) * dx
                elif i == j + 1:
                    contribution += (h_values[j]) * dx
                else:
                    contribution += 0
            grad[i] = contribution
        grad = grad * dx
        return c5, grad
    
    # Optimization with scipy
    result = optimize.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False},
        bounds=[(0, 1) for _ in range(n_points)]
    )
    
    # Check if we can improve with more points
    if time.time() < time.time() + budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        initial_h = np.array(initial_h_values[:n_points])
        initial_h = initial_h / (np.sum(initial_h) * dx) * (n_points / 2)
        
        result = optimize.minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            jac=None,
            options={'maxiter': 100, 'ftol': 1e-6, 'disp': False},
            bounds=[(0, 1) for _ in range(n_points)]
        )
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
