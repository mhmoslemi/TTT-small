import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values, n_points: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / n_points)
initial_h_values = np.random.rand(42) * 2 - 1  # Placeholder for actual initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100  # Start with 100 points
    
    # Initial guess
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate current h
        h_vals = h.copy()
        h_vals = np.clip(h_vals, 0, 1)
        c5 = evaluate_erdos_solution(h_vals, n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()
        
        # Update h using gradient descent
        # Compute correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        corr = corr * (2.0 / n_points)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(len(corr)):
                if k < n_points:
                    grad[i] += corr[k] * (1 - h_vals[i])
                if k >= n_points:
                    grad[i] += corr[k] * (h_vals[i])
        
        # Update h
        h = h - 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    h_vals = best_h.copy()
    h_vals = np.clip(h_vals, 0, 1)
    c5 = evaluate_erdos_solution(h_vals, n_points)
    
    return (h_vals, c5, n_points)
