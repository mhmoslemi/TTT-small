def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    h_shifted = np.zeros(n * 2 - 1)
    h_shifted[:n] = h
    h_shifted[n:] = h[::-1]
    h_shifted = h_shifted * dx
    shifted_ones_minus_h = 1 - h_shifted
    correlation = np.correlate(h_shifted, shifted_ones_minus_h, mode="full")
    c5 = np.max(correlation)
    return c5

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define initial number of points
    n_points = 100  # Start with 100 points, can be increased if time allows
    
    # Optimize function to minimize max overlap
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1
        c5 = evaluate_erdos_solution(h)
        return c5

    # Define an optimizer
    def optimize_h(n_points, max_iter=100):
        x0 = np.random.rand(n_points)
        x0 = x0 * (n_points / 2) / np.sum(x0)
        x0 = np.clip(x0, 0, 1)
        
        # Setup bounds and constraints for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use a simple gradient descent with adaptive step size
        x = x0.copy()
        best_x = x.copy()
        best_c5 = objective(x)
        iter = 0
        while iter < max_iter and time.time() - start_time < budget_s:
            grad = np.zeros(n_points)
            for i in range(n_points):
                x_plus = x.copy()
                x_plus[i] += 1e-5
                x_plus = np.clip(x_plus, 0, 1)
                x_plus = x_plus * (n_points / 2) / np.sum(x_plus)
                c5_plus = objective(x_plus)
                x_minus = x.copy()
                x_minus[i] -= 1e-5
                x_minus = np.clip(x_minus, 0, 1)
                x_minus = x_minus * (n_points / 2) / np.sum(x_minus)
                c5_minus = objective(x_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            
            # Update x using gradient descent
            step_size = 0.1 / (np.linalg.norm(grad) + 1e-8)
            x = x - step_size * grad
            x = np.clip(x, 0, 1)
            x = x * (n_points / 2) / np.sum(x)
            
            # Evaluate
            current_c5 = objective(x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_x = x.copy()
            iter += 1
        
        return best_x, best_c5

    # Start with initial point and optimize
    h_values, c5_bound = optimize_h(n_points)
    
    # Increase n_points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        n_points = 200
        h_values, c5_bound = optimize_h(n_points)
    
    return h_values, c5_bound, n_points
