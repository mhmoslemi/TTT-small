```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
import math

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Try with different numbers of points
    n_points_list = [100, 150, 200]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Discretize the interval [0, 2]
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initialize using the current best or random
        if best_h is None:
            h = initial_h_values[:n_points]  # Use initial values if available
        else:
            h = best_h.copy()
        
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        
        # Ensure the integral is 1
        h_sum = np.sum(h)
        if h_sum == 0:
            h = np.full(n_points, 0.5)
        elif h_sum != 0:
            h = h * (n_points / 2) / h_sum
        
        # Optimization: minimize the maximum overlap integral
        # Formulate as an optimization problem
        h_var = cp.Variable(n_points, nonneg=True)
        obj = cp.Minimize(cp.Max(cp.sum(h_var * (1 - h_var[1:])), cp.sum(h_var * (1 - h_var[:-1]))))
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]
        
        # Use ECOS solver for efficiency
        problem = cp.Problem(obj, constraints)
        try:
            result = problem.solve(solver=cp.ECOS, verbose=False)
            if result is not None:
                h_opt = np.round(h_var.value, 4)
                h_opt = np.clip(h_opt, 0, 1)
                h_opt_sum = np.sum(h_opt)
                if h_opt_sum == 0:
                    h_opt = np.full(n_points, 0.5)
                elif h_opt_sum != 0:
                    h_opt = h_opt * (n_points / 2) / h_opt_sum
                h_opt = np.clip(h_opt, 0, 1)
                
                # Compute C5
                c5 = evaluate_erdos_solution(h_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        except:
            pass
        
        # Also try some random perturbations
        for _ in range(5):
            if time.time() - start_time > budget_s:
                break
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            h_perturb = h + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb_sum = np.sum(h_perturb)
            if h_perturb_sum == 0:
                h_perturb = np.full(n_points, 0.5)
            elif h_perturb_sum != 0:
                h_perturb = h_perturb * (n_points / 2) / h_perturb_sum
            h_perturb = np.clip(h_perturb, 0, 1)
            
            c5 = evaluate_erdos_solution(h_perturb, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_perturb
    
    # Final check
    if best_h is None:
        best_h = np.full(n_points, 0.5)
        c5 = evaluate_erdos_solution(best_h, dx)
    
    return best_h, best_c5, len(best_h)
```
