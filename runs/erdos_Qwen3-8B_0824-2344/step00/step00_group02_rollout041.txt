```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Constants
MAX_ITERATIONS = 100
MAX_EVALUATIONS = 50
INITIAL_NUM_POINTS = 45
STEP = 0.05  # Smaller step for better resolution

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = INITIAL_NUM_POINTS
    
    # Initialize from provided starting point
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)  # Ensure values are within [0, 1]
    h = np.round(h, 5)  # Avoid numerical issues
    h = np.concatenate([h, np.zeros(n_points - len(h))])  # Ensure consistent length
    h = h[:n_points]
    h = np.clip(h, 0, 1)
    h = np.round(h, 5)
    
    # Compute initial C5
    c5 = evaluate_erdos_solution(h)
    best_c5 = c5
    best_h = h
    
    # Optimization loop
    for _ in range(MAX_ITERATIONS):
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial perturbation
        perturb = np.random.normal(0, STEP, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = np.round(h_perturbed, 5)
        
        # Compute C5 for perturbed solution
        c5_perturbed = evaluate_erdos_solution(h_perturbed)
        
        # Keep better solution
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed
            print(f"Found better solution: C5 = {best_c5:.6f}")
        
        # Try local optimization around current best
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2,
            0 <= h_opt,
            h_opt <= 1
        ]
        
        # Objective: minimize max overlap
        def objective_func(h_opt):
            h = h_opt.value
            # Compute overlap integrals
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            max_overlap = np.max(overlap)
            return max_overlap
        
        # Use a numerical optimization approach
        result = optimize.minimize(
            fun=lambda x: objective_func(x),
            x0=h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)]
        )
        
        if result.success:
            h_optimized = result.x
            h_optimized = np.clip(h_optimized, 0, 1)
            h_optimized = np.round(h_optimized, 5)
            c5_optimized = evaluate_erdos_solution(h_optimized)
            
            if c5_optimized < best_c5:
                best_c5 = c5_optimized
                best_h = h_optimized
                print(f"Found better solution via optimization: C5 = {best_c5:.6f}")
    
    # Evaluate final solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure best_h has n_points entries
    if len(best_h) < n_points:
        best_h = np.concatenate([best_h, np.zeros(n_points - len(best_h))])
    elif len(best_h) > n_points:
        best_h = best_h[:n_points]
    
    return (best_h, final_c5, n_points)
```
