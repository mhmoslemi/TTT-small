```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(95)  # Placeholder for initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    
    # Define the objective function to minimize
    def objective(h):
        nonlocal best_c5
        # Ensure h is within [0, 1] and sum is correct
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            h = h / sum_h * (n_points / 2)
        # Compute the overlap integral
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Define the optimization problem
    def optimize_h(n_points, budget_s):
        h = cp.Variable(n_points)
        # Objective: minimize the maximum overlap integral
        # Use a convex relaxation by approximating max with a convex function
        obj = cp.Minimize(cp.Max(cp.log_sum_exp(cp.multiply(1 - h, h - h.shift(1, constant=0)))))  # Approximation for max

        # Constraints
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]

        # Solve the problem
        prob = cp.Problem(obj, constraints)
        # Use a solvers that can handle large problems
        prob.solve(solver=cp.SCS, max_iter=1000, time_limit=budget_s - (time.time() - start_time))

        if prob.status == 'optimal':
            return h.value
        else:
            return np.random.rand(n_points)

    # Try different n_points
    for n_points in [100, 200, 300]:
        if time.time() - start_time > budget_s - 1:
            break
        h = optimize_h(n_points, budget_s)
        if h is not None:
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    return best_h, best_c5, len(best_h)
```
