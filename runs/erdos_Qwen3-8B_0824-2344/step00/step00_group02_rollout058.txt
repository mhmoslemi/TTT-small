import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    elapsed_time = 0
    
    # Initial guess
    n_points_initial = len(initial_h_values)
    h_initial = initial_h_values.copy()
    h_initial = h_initial / (np.sum(h_initial) * 2.0 / n_points_initial)  # Normalize
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        h_shifted = np.zeros((n, n))
        for i in range(n):
            h_shifted[i, :] = np.roll(h, i)
        overlap = np.zeros(n)
        for k in range(1, n):
            overlap[k] = np.sum(h * (1 - h_shifted[k, :])) * dx
        return np.max(overlap)
    
    # Function to compute the objective and constraints for optimization
    def objective(x):
        return compute_c5(x)
    
    # Optimization settings
    n_points = n_points_initial
    h = h_initial.copy()
    
    # Perform optimization
    while elapsed_time < budget_s:
        # Optimization settings
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=False,
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        # Update h
        h = result.x
        
        # Compute current C5
        current_c5 = compute_c5(h)
        print(f"Current C5: {current_c5:.6f}")
        
        # Check if we can reduce n_points
        if n_points > 100:
            n_points = max(50, int(n_points * 0.9))
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        
        # Check if we should increase n_points
        if n_points < 500:
            n_points = min(500, int(n_points * 1.1))
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        
        # Update elapsed time
        elapsed_time = time.time() - start_time
    
    # Final evaluation
    final_c5 = compute_c5(h)
    return h, final_c5, len(h)
