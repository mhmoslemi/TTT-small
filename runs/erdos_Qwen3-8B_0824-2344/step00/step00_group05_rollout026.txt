import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all shifted versions of 1-h
    shifted = [np.roll(1 - h, k) for k in range(1, n_points)]
    # Compute overlaps
    overlaps = [np.sum(h * shifted[i]) * dx for i in range(len(shifted))]
    return np.max(overlaps)

def initial_h_values():
    return np.array([0.5] * 70)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for balance between speed and accuracy
    
    # Initial guess
    h_initial = initial_h_values()
    h_initial = np.resize(h_initial, n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h = h_initial.copy()
    
    # Function to evaluate C5
    def evaluate(h):
        nonlocal n_points
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        dx = 2.0 / n_points
        shifted = [np.roll(1 - h, k) for k in range(1, n_points)]
        overlaps = [np.sum(h * shifted[i]) * dx for i in range(len(shifted))]
        return np.max(overlaps)
    
    # Optimization loop
    for _ in range(10):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Try simulated annealing
        temp = 100.0
        cooling_rate = 0.995
        for step in range(1000):
            if elapsed > budget_s:
                break
            # Perturb h
            idx = np.random.randint(n_points)
            h[idx] += np.random.uniform(-0.1, 0.1)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate new C5
            new_c5 = evaluate(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            else:
                # Accept with probability based on temperature
                delta = new_c5 - best_c5
                if np.random.rand() < np.exp(-delta / temp):
                    best_c5 = new_c5
                    best_h = h.copy()
            temp *= cooling_rate
        
        # Try gradient descent
        h = best_h.copy()
        learning_rate = 0.01
        for step in range(100):
            if elapsed > budget_s:
                break
            # Compute gradient
            h_current = h.copy()
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            dx = 2.0 / n_points
            shifted = [np.roll(1 - h_current, k) for k in range(1, n_points)]
            overlaps = [np.sum(h_current * shifted[i]) * dx for i in range(len(shifted))]
            c5 = np.max(overlaps)
            
            # Compute gradient of C5 with respect to h
            # This is approximate and computationally expensive
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_test = h_current.copy()
                h_test[i] += 1e-6
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                shifted = [np.roll(1 - h_test, k) for k in range(1, n_points)]
                overlaps_test = [np.sum(h_test * shifted[i]) * dx for i in range(len(shifted))]
                c5_test = np.max(overlaps_test)
                grad[i] = (c5_test - c5) / 1e-6
            
            # Update h
            h = h_current - learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5_new = evaluate(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate(best_h)
    
    return best_h.tolist(), final_c5, n_points
