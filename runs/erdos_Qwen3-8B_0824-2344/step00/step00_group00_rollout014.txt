def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Initial number of points
    n_points = 45  # Starting with initial sample size
    # We'll try increasing n_points if needed, but keep it below 1k
    
    # Function to evaluate the overlap integral
    def evaluate(h):
        # Normalize h (should already be normalized)
        dx = 2.0 / len(h)
        return max(np.correlate(h, 1 - h, mode="full")) * dx
    
    # Function to generate a random perturbation to h
    def perturb(h, perturb_strength=0.1):
        h_perturbed = h.copy()
        indices = np.random.choice(len(h_perturbed), size=int(len(h_perturbed) * 0.1), replace=False)
        h_perturbed[indices] += np.random.uniform(-perturb_strength, perturb_strength, size=len(indices))
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        return h_perturbed
    
    # Function to apply a small mutation
    def mutate(h):
        h_mutated = h.copy()
        indices = np.random.choice(len(h_mutated), size=int(len(h_mutated) * 0.1), replace=False)
        h_mutated[indices] += np.random.uniform(-0.1, 0.1, size=len(indices))
        h_mutated = np.clip(h_mutated, 0.0, 1.0)
        return h_mutated
    
    # Function to find the best solution in a local neighborhood
    def local_search(h, max_iter=100):
        best_h = h.copy()
        best_c5 = evaluate(best_h)
        
        for _ in range(max_iter):
            # Try perturbation
            perturbed_h = perturb(best_h)
            c5 = evaluate(perturbed_h)
            if c5 < best_c5:
                best_h = perturbed_h
                best_c5 = c5
            # Try mutation
            mutated_h = mutate(best_h)
            c5 = evaluate(mutated_h)
            if c5 < best_c5:
                best_h = mutated_h
                best_c5 = c5
        
        return best_h, best_c5
    
    # Function to find the best solution using simulated annealing
    def simulated_annealing(h, max_iter=1000, temp=1.0, cooling_rate=0.995):
        best_h = h.copy()
        best_c5 = evaluate(best_h)
        current_h = h.copy()
        current_c5 = best_c5
        
        for _ in range(max_iter):
            # Generate a perturbed version
            perturbed_h = perturb(current_h)
            perturbed_c5 = evaluate(perturbed_h)
            
            # Acceptance probability
            if perturbed_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - perturbed_c5) / temp):
                current_h = perturbed_h
                current_c5 = perturbed_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            
            # Cool down
            temp *= cooling_rate
        
        return best_h, best_c5
    
    # Function to optimize using a convex relaxation
    def convex_relaxation(n_points):
        # Use CVXPY to solve a convex relaxation of the optimization problem
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        # Objective: minimize the maximum overlap
        # To approximate, we use a convex relaxation
        # We use a semidefinite programming approach
        # For simplicity, we just optimize the average overlap
        obj = cp.Minimize(cp.sum(cp.multiply(x, 1 - x)))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.SCS)
        return x.value
    
    # Initial solution
    h = initial_h_values.copy()
    
    # Try local search
    h, c5 = local_search(h)
    best_h = h.copy()
    best_c5 = c5
    
    # Try simulated annealing
    h, c5 = simulated_annealing(h)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    
    # Try convex relaxation
    n_points = 100
    h = convex_relaxation(n_points)
    h = np.clip(h, 0.0, 1.0)
    # Normalize h
    h = h / np.sum(h) * (n_points / 2)
    c5 = evaluate(h)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    
    # Perform a few more rounds of local search and simulated annealing
    for _ in range(5):
        h, c5 = local_search(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        h, c5 = simulated_annealing(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Try increasing n_points if within budget
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h = np.random.uniform(0.0, 1.0, n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0.0, 1.0)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Ensure the solution is within the budget
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return best_h, best_c5, len(best_h)
    
    # Final return
    return best_h, best_c5, len(best_h)
