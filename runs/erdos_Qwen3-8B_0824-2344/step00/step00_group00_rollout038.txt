```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial construction
    n_points = 45
    h_initial = initial_h_values.copy()
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    
    # Function to evaluate overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        ones_minus_h = 1 - h
        corr = signal.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Function to optimize h
    def optimize_h(h0, n_points, max_iter=100, tol=1e-5):
        h = cp.Variable(n_points)
        obj = cp.Minimize(cp.Max([cp.sum(cp.multiply(h, 1 - h[k:])) for k in range(n_points)]))
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.SCS, max_iter=max_iter, eps=tol, verbose=False)
        if result is None:
            return h0
        return h.value
    
    # Initial optimization
    h_opt = optimize_h(h_initial, n_points)
    c5_initial = evaluate_overlap(h_opt)
    best_h = h_opt
    best_c5 = c5_initial
    
    # Try a few different starting points
    for _ in range(3):
        # Perturb the initial solution
        h_perturbed = h_opt + 0.1 * np.random.randn(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Optimize perturbed solution
        h_opt = optimize_h(h_perturbed, n_points)
        c5_current = evaluate_overlap(h_opt)
        
        # Update best solution
        if c5_current < best_c5:
            best_h = h_opt
            best_c5 = c5_current
    
    # Try a smaller n_points
    n_points = 40
    h_small = np.zeros(n_points)
    h_small[:int(n_points/2)] = 1.0
    h_small = h_small * (n_points / 2) / np.sum(h_small)
    
    # Optimize small solution
    h_opt_small = optimize_h(h_small, n_points)
    c5_small = evaluate_overlap(h_opt_small)
    
    # Update best solution
    if c5_small < best_c5:
        best_h = h_opt_small
        best_c5 = c5_small
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # If time remains, perform a final fine-tuning
        h_final = best_h + 0.05 * np.random.randn(len(best_h))
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (len(best_h) / 2) / np.sum(h_final)
        c5_final = evaluate_overlap(h_final)
        
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return best_h, best_c5, len(best_h)
```
