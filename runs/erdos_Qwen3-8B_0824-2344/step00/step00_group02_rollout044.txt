```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Try different number of points
    n_points_list = [100, 200, 300, 400, 500]
    
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Create uniform grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initial guess
        h_initial = initial_h_values.copy()
        h_initial = np.interp(x, np.linspace(0, 2, len(h_initial)), h_initial)
        h_initial = np.clip(h_initial, 0, 1)
        
        # Rescale to meet the integral constraint
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define variables
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Objective function
        # We minimize the maximum overlap
        # This is a max-min problem, so we need to minimize the maximum value of the overlap
        # The overlap is the convolution of h and (1 - h)
        # To approximate this, we minimize the maximum value over a set of shifts
        
        # Sample shifts
        shifts = np.linspace(-2, 2, 100)
        
        # Define objective
        # We minimize the maximum overlap over the sampled shifts
        # To do this, we define an auxiliary variable for the maximum value
        max_overlap = cp.Variable()
        constraints += [max_overlap >= 0]
        
        for k in shifts:
            # Compute the overlap for this shift
            # The overlap integral is approximately the convolution of h and (1 - h) multiplied by dx
            # To avoid computing the full convolution, we sample a few points
            overlap = 0
            for i in range(n_points):
                for j in range(n_points):
                    if 0 <= x[i] - k + x[j] <= 2:
                        overlap += h[i] * (1 - h[j]) * dx
            constraints += [max_overlap >= overlap]
        
        # Objective is to minimize the max_overlap
        objective = cp.Minimize(max_overlap)
        problem = cp.Problem(objective, constraints)
        
        # Solve the problem
        try:
            result = problem.solve(solver='ECOS')
            if result is not None:
                h_values = np.round(h.value, 4)
                h_values = np.clip(h_values, 0, 1)
                # Rescale to meet the integral constraint
                h_values = h_values * (n_points / 2) / np.sum(h_values)
                # Compute the C5 value
                c5_value = np.max(signal.correlate(h_values, 1 - h_values, mode="full")) * dx
                if c5_value < best_c5:
                    best_c5 = c5_value
                    best_h = h_values
        except:
            pass
    
    return best_h, best_c5, len(best_h)
```
