import numpy as np

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100  # Start with 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    
    # Initial guess: start from the provided initial solution
    h_values = initial_h_values.copy()
    h_values = h_values[:n_points]  # Ensure it's the correct size
    
    # Normalize h_values to satisfy the integral constraint
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    # Ensure h_values are within [0, 1]
    h_values = np.clip(h_values, 0, 1)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h_fft = np.fft.fft(h)
        shifted_h = np.roll(h, n_points // 2)
        shifted_1_minus_h = 1 - shifted_h
        shifted_1_minus_h_fft = np.fft.fft(shifted_1_minus_h)
        conv_fft = np.conj(h_fft) * shifted_1_minus_h_fft
        conv = np.fft.ifft(conv_fft).real
        return np.max(conv)
    
    # Function to compute the overlap integral
    def objective_function(h):
        return compute_c5(h)
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-4
    
    # Optimization loop with early stopping
    best_c5 = float('inf')
    best_h = h_values.copy()
    start_time = time.time()
    
    for _ in range(max_iter):
        # Convert h to a numpy array
        h = np.array(h_values)
        
        # Check if we have exceeded the time budget
        if time.time() - start_time > budget_s:
            break
        
        # Compute the current C5 value
        current_c5 = compute_c5(h)
        
        # Update the best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Random perturbation to escape local minima
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Evaluate the perturbed solution
        current_c5_perturbed = compute_c5(h_perturbed)
        
        # Accept the perturbed solution if it's better
        if current_c5_perturbed < current_c5:
            h_values = h_perturbed.copy()
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Return the best solution found
    return best_h, final_c5, n_points
