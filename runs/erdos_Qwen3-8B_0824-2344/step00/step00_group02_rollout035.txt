    h_values = np.array(initial_h_values)
    best_c5 = np.inf
    best_h = h_values.copy()

    # Try a few different starting points to escape local optima
    for _ in range(3):
        # Perturb the initial guess
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_values = np.clip(h_values + perturbation, 0, 1)

        # Ensure the integral constraint is satisfied
        total_sum = np.sum(h_values)
        scale = (n_points / 2) / total_sum if total_sum != 0 else 1.0
        h_values = h_values * scale

        # Function to evaluate C5 given h_values
        def evaluate_c5(h_values):
            h = h_values.copy()
            h = np.concatenate([h, h])  # Extend to full length for correlation
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * (2.0 / n_points)
            return c5

        # Optimize using gradient descent with line search
        start_time = time.time()
        for iteration in range(100):
            if time.time() - start_time > budget_s:
                break
            # Compute current C5
            current_c5 = evaluate_c5(h_values)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_values.copy()

            # Compute gradient (numerical approximation)
            grad = np.zeros_like(h_values)
            for i in range(n_points):
                h_perturbed = h_values.copy()
                h_perturbed[i] += 1e-4
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
                c5_perturbed = evaluate_c5(h_perturbed)
                grad[i] = (c5_perturbed - current_c5) / 1e-4

            # Gradient descent step
            alpha = 0.1
            h_values -= alpha * grad
            h_values = np.clip(h_values, 0, 1)
            total_sum = np.sum(h_values)
            scale = (n_points / 2) / total_sum if total_sum != 0 else 1.0
            h_values = h_values * scale

        # Check if current best is better than previous
        if evaluate_c5(best_h) < best_c5:
            best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_points
