import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Starting with a reasonable number of points
    n_points = 100
    # Initial guess based on existing construction
    h_initial = initial_h_values.copy()
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n_points)

    # Function to optimize h with constraints
    def optimize_h(h_initial):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        obj = cp.Minimize(evaluate_c5(h))
        prob = cp.Problem(obj, constraints)
        start_time = time.time()
        result = prob.solve(solver=cp.SCS, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
        if result is not None and prob.status == 'OPTIMAL':
            return h.value
        return h_initial

    # Try to improve the initial solution
    best_h = h_initial
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()

    # Perform a few iterations of optimization
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h_candidate = optimize_h(best_h)
        c5_candidate = evaluate_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n_points
