import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for speed and scalability
    
    # Helper function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / n_points
        h_array = np.array(h)
        h_shifted = np.zeros((n_points, n_points))
        for i in range(n_points):
            h_shifted[i, :] = h_array[(i + n_points - 1) % n_points]
        overlap = np.correlate(h_array, 1 - h_array, mode='full')
        c5 = np.max(overlap) * dx
        return c5
    
    # Initial guess from pre-defined initial_h_values
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = initial_h[:n_points]  # Truncate or extend if needed
    h = np.array(initial_h)
    
    # Optimization: minimize the maximum overlap
    # Constraint: sum(h) == n_points / 2
    # Objective: minimize the maximum value of integral of h(x)(1 - h(x + k))
    
    # Try a few different n_points values to find the best
    n_points_list = [100, 150, 200]
    for n_points in n_points_list:
        dx = 2.0 / n_points
        # Rescale initial_h to fit n_points
        h_resampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        h_resampled = np.clip(h_resampled, 0, 1)
        h_resampled = h_resampled * (n_points / 2) / np.sum(h_resampled)
        
        # Optimization using CVXPY
        h_var = cp.Variable(n_points, nonneg=True, boolean=False)
        constraints = [cp.sum(h_var) == n_points / 2]
        objective = cp.Minimize(cp.Max(cp.sum(cp.multiply(h_var, 1 - h_var[n_points - 1::-1]))))
        
        # Try to minimize the maximum overlap
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.SCS, max_iter=1000, time_limit=60 - (time.time() - start_time))
        except:
            pass
        
        h_opt = np.round(h_var.value, 4)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    # Also try a simple random initialization for diversity
    h_rand = np.random.rand(n_points)
    h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
    c5 = compute_c5(h_rand)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_rand
    
    # Final evaluation
    final_c5 = best_c5
    return (best_h.tolist(), final_c5, n_points)
