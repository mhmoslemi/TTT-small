```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from scipy.optimize.optimize import OptimizeResult
from scipy.optimize._differentiable_functions import ScalarFunction

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Define a function to compute the overlap integral
def compute_c5(h_values, dx):
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

# Define a function to evaluate the overlap integral and return the maximum value
def objective_function(h_values, dx):
    return compute_c5(h_values, dx)

# Define a function to compute the value of the integral and the constraints
def objective_and_constraints(h_values, dx):
    h = np.array(h_values)
    # Ensure h is in [0, 1]
    bounds = Bounds(0, 1)
    # Define the objective function
    def objective(x):
        return compute_c5(x, dx)
    # Define the constraint: sum(h) == n_points / 2
    n_points = len(h_values)
    def constraint(x):
        return np.sum(x) - n_points / 2
    # Return the function to minimize
    return objective, constraint

# Define the run function
def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Start time
    start_time = time.time()
    # Initial number of points
    n_points = 100
    # Initial h values (from initial_h_values)
    h = initial_h_values
    # Ensure h has exactly n_points samples
    if len(h) != n_points:
        h = h[:n_points]
    # Adjust the initial h to meet the integral constraint
    h = h / (np.sum(h) * 2)  # scale to ensure sum(h) * dx = 1
    # dx
    dx = 2.0 / n_points
    # Compute the initial C5 value
    initial_c5 = compute_c5(h, dx)
    # Best solution found so far
    best_h = h.copy()
    best_c5 = initial_c5
    # Optimization parameters
    max_iterations = 50
    for _ in range(max_iterations):
        # Set up the optimization problem
        h_values = best_h.copy()
        # Define the objective function
        def objective(x):
            return compute_c5(x, dx)
        # Define constraints
        def constraint(x):
            return np.sum(x) - n_points / 2
        # Bounds for h
        bounds = Bounds(0, 1)
        # Optimization settings
        options = {"maxiter": 100, "ftol": 1e-6}
        # Perform optimization
        result = opt.minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options=options
        )
        # Check if optimization succeeded
        if result.success:
            best_h = result.x.copy()
            best_c5 = compute_c5(best_h, dx)
            # Check time budget
            elapsed_time = time.time() - start_time
            if elapsed_time >= budget_s:
                break
        else:
            # If optimization failed, break
            break
    # Return the best solution found
    return best_h, best_c5, len(best_h)
```
