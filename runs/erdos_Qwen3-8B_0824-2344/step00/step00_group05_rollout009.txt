import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load("initial_h_values.npy")  # Assuming this file exists with the initial construction

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Initial n_points (start with 100 points for faster computation)
    n_points = 100
    
    # Time tracking
    start_time = time.time()
    
    # Define the function to evaluate the overlap integral
    def evaluate_overlap(h_values):
        return evaluate_erdos_solution(h_values)
    
    # Define the function to optimize the overlap integral
    def optimize_overlap(n_points):
        # Create the grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Define variables
        h = cp.Variable(n_points, nonneg=True, boolean=True)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,  # Normalize to have integral 1
            h <= 1,
            h >= 0
        ]
        
        # Objective: Minimize the maximum overlap integral
        # Since we can't directly minimize the maximum, we use a bilevel optimization approach
        # Here, we minimize the maximum overlap by using a binary search on the maximum value
        # and checking feasibility for each value
        
        # Initial guess (from initial_h_values)
        h_initial = np.zeros(n_points)
        if len(initial_h_values) == n_points:
            h_initial = initial_h_values.copy()
        else:
            h_initial = np.random.rand(n_points)
            h_initial = np.clip(h_initial, 0, 1)
        
        # Define the objective function for binary search
        def is_feasible(max_overlap):
            # We want to find a h such that the maximum overlap is <= max_overlap
            # This is a feasibility problem
            # Use a convex relaxation to approximate it
            # Note: This is a simplified approach due to time constraints
            
            # Convex relaxation (using mean instead of max)
            # This is a heuristic to find a solution with overlap <= max_overlap
            # In practice, more sophisticated methods would be required
            overlap = np.correlate(h.value, 1 - h.value, mode="full") * dx
            return np.max(overlap) <= max_overlap
        
        # Binary search to find the minimal max_overlap
        low = 0.0
        high = 1.0
        max_iterations = 50
        tolerance = 1e-5
        
        for _ in range(max_iterations):
            mid = (low + high) / 2
            # Check feasibility
            h.value = h_initial.copy()
            h.value = np.clip(h.value, 0, 1)
            
            # Evaluate the overlap
            overlap = np.correlate(h.value, 1 - h.value, mode="full") * dx
            if np.max(overlap) <= mid:
                high = mid
            else:
                low = mid
        
        # Return the best solution found so far
        return h.value, high
    
    # Try increasing n_points if time permits
    best_h = initial_h_values
    best_c5 = float('inf')
    
    # Start with n_points = 100
    h_values, c5_bound = optimize_overlap(n_points)
    best_h = h_values
    best_c5 = c5_bound
    
    # Try increasing n_points if time permits
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        n_points = 200
        h_values, c5_bound = optimize_overlap(n_points)
        if c5_bound < best_c5:
            best_h = h_values
            best_c5 = c5_bound
    
    # Try increasing n_points if time permits
    if remaining_time > 10:
        n_points = 300
        h_values, c5_bound = optimize_overlap(n_points)
        if c5_bound < best_c5:
            best_h = h_values
            best_c5 = c5_bound
    
    # Return the best solution found
    return (best_h, best_c5, len(best_h))
