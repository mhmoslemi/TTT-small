import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.optimize import Bounds, minimize, differential_evolution
from scipy.signal import correlate

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Global variables
initial_h_values = initial_h_values
initial_n_points = len(initial_h_values)
initial_h = np.array(initial_h_values)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from the provided construction
    h = np.copy(initial_h)
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    target_sum = n_points / 2  # Ensure sum(h) == n_points / 2
    if np.abs(sum_h - target_sum) > 1e-8:
        h = initial_h / (sum_h / target_sum)
    
    # Function to evaluate C5
    def compute_c5(h_vals):
        n_points = len(h_vals)
        dx = 2.0 / n_points
        h = np.array(h_vals)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral using convolution
        conv = correlate(h, 1 - h, mode="full")
        c5 = np.max(conv * dx)
        return c5
    
    # Function to optimize h
    def optimize_h(h_vals):
        n_points = len(h_vals)
        dx = 2.0 / n_points
        sum_h = np.sum(h_vals)
        # Constraints: sum(h) = n_points / 2, 0 <= h <= 1
        # Objective: minimize the maximum overlap integral
        # We use a heuristic approach due to time constraints
        # To optimize, we use differential evolution on the h vector
        bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(
            compute_c5,
            bounds,
            popsize=15,
            strategy='best1bin',
            maxiter=200,
            tol=1e-4,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed,
            callback=None,
            disp=False,
            polish=True,
            keep_history=False
        )
        return result.x
    
    # Run optimization
    start_time = time.time()
    best_h = h
    best_c5 = compute_c5(h)
    for _ in range(3):  # Run multiple times with different initializations
        h = optimize_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
        if time.time() - start_time > budget_s - 5:
            break
    
    n_points = len(best_h)
    # Ensure sum of h is n_points / 2
    sum_h = np.sum(best_h)
    if np.abs(sum_h - n_points / 2) > 1e-8:
        best_h = best_h / (sum_h / (n_points / 2))
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), final_c5, n_points
