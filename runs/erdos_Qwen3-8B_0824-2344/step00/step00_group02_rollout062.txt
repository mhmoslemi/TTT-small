```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define parameters
    n_points = 45  # Start with the initial n_points
    max_n_points = 100  # Maximum points to consider
    max_iterations = 20  # Max iterations for optimization
    
    # Initial guess
    h_initial = initial_h_values.copy()
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h)
        if len(h) != n_points:
            raise ValueError("h must have length n_points")
        
        # Normalize h to satisfy the integral constraint
        h_norm = h / (np.sum(h) * (2.0 / n_points))
        
        # Compute the correlation
        corr = np.correlate(h_norm, 1 - h_norm, mode="full")
        overlap = np.max(corr) * (2.0 / n_points)
        return overlap
    
    # Function to optimize h using gradient descent
    def optimize_h(h, n_points, max_iterations=20, learning_rate=0.01):
        h = h.copy()
        for _ in range(max_iterations):
            # Compute the overlap
            overlap = compute_overlap(h)
            
            # Compute gradient
            # We use a finite difference approximation
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                overlap_plus = compute_overlap(h_plus)
                grad[i] = (overlap_plus - overlap) / 1e-5
            
            # Update h
            h -= learning_rate * grad
            
            # Clamp h to [0, 1]
            h = np.clip(h, 0, 1)
            
            # Normalize h to satisfy the integral constraint
            h = h / (np.sum(h) * (2.0 / n_points))
        
        return h
    
    # Start optimization
    start_time = time.time()
    for n_points in range(45, max_n_points + 1):
        if time.time() - start_time > budget_s:
            break
        
        # Initialize h with previous best
        h = best_h.copy()
        if n_points > len(h):
            # Pad with zeros
            h = np.pad(h, (0, n_points - len(h)), mode='constant')
        else:
            # Truncate
            h = h[:n_points]
        
        # Optimize h
        h = optimize_h(h, n_points)
        
        # Evaluate overlap
        current_c5 = compute_overlap(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Final evaluation
    final_c5 = compute_overlap(best_h)
    return best_h, final_c5, len(best_h)
```
