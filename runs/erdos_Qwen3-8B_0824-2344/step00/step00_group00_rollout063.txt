def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial guess
    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    initial_c5 = evaluate_erdos_solution(h)
    
    # Try increasing n_points up to 1000
    for n_points in [45, 90, 180, 360, 720, 1000]:
        if time.time() - start_time > budget_s:
            break
        
        # Create a new h with n_points samples
        h = np.zeros(n_points)
        h[::2] = 1.0  # Initial pattern: alternate 1 and 0
        h = h + np.random.normal(0, 0.1, n_points)  # Add some randomness
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        h = h * (n_points / 2) / total  # Normalize
        
        # Define the variables
        x = cp.Variable(n_points)
        constraints = [0 <= x, x <= 1]
        constraints.append(cp.sum(x) == n_points / 2)
        
        # Objective function to minimize maximum overlap
        overlaps = []
        for k in range(1, n_points):
            overlap = cp.sum(cp.multiply(x, 1 - cp.roll(x, k)))
            overlaps.append(overlap)
        max_overlap = cp.Max(overlaps)
        
        # Solve the optimization problem
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        start_time = time.time()
        result = prob.solve(solver="MOSEK", verbose=False)
        if result is not None:
            h = x.value
        else:
            h = np.copy(initial_h_values[:n_points])
        
        c5 = evaluate_erdos_solution(h)
        if c5 < initial_c5:
            initial_c5 = c5
            initial_h_values = h
    
    return initial_h_values, initial_c5, n_points
