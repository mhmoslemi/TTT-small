import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 70
    h_initial = initial_h_values.copy()
    
    # Function to compute the C5 value
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h_extended = np.concatenate([h, np.zeros_like(h)])
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Define the objective and constraints
    def objective_function(h_values):
        return compute_c5(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Optimization: Use a local optimizer (e.g., SLSQP)
    # We'll use a gradient-free optimizer for simplicity and speed
    def optimize_h(h_initial, max_iter=100, tol=1e-5):
        h = h_initial.copy()
        for _ in range(max_iter):
            # Compute current C5
            current_c5 = compute_c5(h)
            # Perturb the function values slightly
            perturbation = np.random.uniform(-0.05, 0.05, size=len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            # Compute C5 for the perturbed function
            c5_perturbed = compute_c5(h_perturbed)
            # Accept the perturbation if it improves the C5
            if c5_perturbed < current_c5:
                h = h_perturbed
            # Check for convergence
            if np.abs(c5_perturbed - current_c5) < tol:
                break
        return h, current_c5
    
    # Run the optimization
    optimized_h, best_c5 = optimize_h(h_initial)
    
    # Evaluate the best solution
    best_c5 = compute_c5(optimized_h)
    
    # Check if we have time to run a more thorough optimization
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 10:
        # Run a more refined optimization using a local solver
        h = cp.Variable(len(optimized_h))
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        objective = cp.Minimize(cp.Max(cp.correlate(h, 1 - h, mode="full") * 2.0 / len(h)))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-5)
            optimized_h = h.value
            best_c5 = compute_c5(optimized_h)
        except:
            pass
    
    return optimized_h, best_c5, n_points
