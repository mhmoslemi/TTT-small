import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial h values from the provided construction
    initial_h_values = np.array(initial_h_values)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points//2]])  # Duplicate to handle wrap-around
        corr = np.correlate(h, 1 - h, mode='full')
        overlap = np.max(corr * dx)
        return overlap
    
    # Function to compute the objective
    def objective(h):
        return evaluate_overlap(h)
    
    # Function to compute the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for h values
    bounds = Bounds(0, 1)
    
    # Optimization settings
    options = {
        'maxiter': 100,
        'disp': False,
        'ftol': 1e-6
    }
    
    # Initial guess
    h0 = initial_h_values.copy()
    h0 = (h0 - np.min(h0)) / (np.max(h0) - np.min(h0))  # Normalize to [0,1]
    
    # Run optimization
    start_time = time.time()
    result = opt.minimize(
        objective,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options=options
    )
    
    # Track time and return best solution
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        # If time is up, return the best found so far
        return (initial_h_values, evaluate_overlap(initial_h_values), n_points)
    
    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    
    # Ensure h is within [0, 1] and sum is correct
    best_h = np.clip(best_h, 0, 1)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = initial_h_values.copy()
        best_c5 = evaluate_overlap(best_h)
    
    return (best_h, best_c5, n_points)
