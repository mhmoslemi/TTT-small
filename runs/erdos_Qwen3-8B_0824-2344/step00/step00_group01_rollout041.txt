```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set up random seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 0
    
    # Try different number of points (start with 100, then 200, 400)
    for n in [100, 200, 400]:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Define dx
        dx = 2.0 / n
        
        # Initialize h with the initial construction or random
        h = initial_h_values.copy()[:n]
        if len(h) < n:
            h = np.concatenate([h, np.zeros(n - len(h))])
        h = h[:n]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet the integral constraint
        total = np.sum(h) * dx
        h *= (1 / total) * (n / 2)
        
        # Function to compute the max overlap
        def compute_c5(h):
            h = np.clip(h, 0, 1)
            h_expanded = np.zeros_like(h)
            h_expanded[:len(h)] = h
            h_expanded = np.concatenate([h_expanded, np.zeros(n)])
            corr = np.correlate(h_expanded, 1 - h_expanded, mode='full')
            return np.max(corr * dx)
        
        # Optimization objective: minimize max overlap
        def objective_func(x):
            x = np.clip(x, 0, 1)
            x_expanded = np.zeros_like(x)
            x_expanded[:len(x)] = x
            x_expanded = np.concatenate([x_expanded, np.zeros(n)])
            corr = np.correlate(x_expanded, 1 - x_expanded, mode='full')
            return np.max(corr * dx)
        
        # Constraint: integral of h is 1
        def constraint_func(x):
            return np.sum(x) * dx - 1
        
        # Initial guess
        h_initial = h.copy()
        
        # Use scipy optimization with bounds and constraints
        bounds = [(0, 1) for _ in range(n)]
        cons = {'type': 'eq', 'fun': constraint_func}
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        # Evaluate the result
        if result.success:
            h_opt = result.x
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n_points = n
        
        # Optional: try a few random perturbations for further optimization
        for _ in range(3):
            if time.time() - start_time > budget_s - 5:
                break
            perturb = np.random.uniform(-0.1, 0.1, n)
            h_perturb = np.clip(h_initial + perturb, 0, 1)
            c5_perturb = compute_c5(h_perturb)
            if c5_perturb < best_c5:
                best_c5 = c5_perturb
                best_h = h_perturb
                n_points = n
    
    # Return the best solution found
    return best_h, best_c5, n_points
```
