import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial number of points
    n_points = 95
    h_values = initial_h_values.copy()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / len(h)
        h_rolled = np.roll(h, shift=1, axis=0)
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Function to compute the objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Function to perform optimization using gradient descent
    def optimize_function(h, max_iter=100, learning_rate=0.01):
        for _ in range(max_iter):
            # Compute current C5
            current_c5 = compute_c5(h)
            
            # Compute the gradient (approximate using finite differences)
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                grad[i] = (compute_c5(h_plus) - current_c5) / 1e-5
            
            # Update h
            h -= learning_rate * grad
            
            # Ensure h is within [0, 1] and normalize
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            h = h * (n_points / 2) / total if total != 0 else h
            
            # Check for early stopping
            if time.time() - start_time > budget_s - 1:
                break
        
        return h
    
    # Initial optimization
    h_values = optimize_function(h_values)
    
    # Refinement using local optimization
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        h_values = optimize_function(h_values, max_iter=50, learning_rate=0.001)
    
    # Evaluate the result
    c5_bound = compute_c5(h_values)
    
    # Return the result
    return (h_values, c5_bound, len(h_values))
