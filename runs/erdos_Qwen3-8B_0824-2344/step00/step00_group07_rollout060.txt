```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assuming initial_h.npy is provided

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 80  # Start with 80 samples as provided
    max_n_points = 200  # Max points to consider for optimization
    step = 2.0 / n_points
    current_h = initial_h_values[:n_points].copy()
    
    best_c5 = evaluate_erdos_solution(current_h)
    best_h = current_h.copy()
    
    # Function to evaluate overlap integral
    def evaluate_overlap(h):
        return evaluate_erdos_solution(h)
    
    # Optimization function to minimize C₅
    def optimize_h(h_values):
        nonlocal best_c5, best_h
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        c5 = evaluate_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Start optimization loop
    for n_points in range(80, max_n_points + 1, 20):
        dx = 2.0 / n_points
        # Initialize with previous best or random
        if n_points > len(best_h):
            h = np.random.rand(n_points) * 2.0 - 1.0  # Random initial guess
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)  # Scale to satisfy the integral constraint
        else:
            h = best_h.copy()
            h = np.pad(h, (0, n_points - len(h)), mode='constant', constant_values=0.5)
            h = h * (n_points / 2) / np.sum(h)
        
        # Use gradient descent to minimize C₅
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            return evaluate_overlap(h)
        
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        if result.success:
            current_h = result.x.copy()
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break
    
    return best_h, best_c5, len(best_h)
```
