import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 100  # Start with 100 points for faster initial evaluation
    dx = 2.0 / n_points

    # Initial solution from provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]  # Ensure it fits our n_points
    h_initial = np.clip(h_initial, 0.0, 1.0)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Normalize to meet the integral constraint

    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.clip(h, 0.0, 1.0)
        # Ensure the integral constraint is satisfied
        integral = np.sum(h) * dx
        if np.abs(integral - 1.0) > 1e-6:
            return np.inf  # Penalize if integral constraint is violated
        # Compute the overlap integral for all k
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Function to optimize h using gradient descent
    def optimize_h(h_initial, n_points, max_iter=100, learning_rate=0.01):
        h = h_initial.copy()
        best_h = h.copy()
        best_c5 = evaluate_c5(h)
        for _ in range(max_iter):
            # Compute the gradient of C5 with respect to h
            # To approximate the gradient, perturb h slightly
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_perturb = h.copy()
                h_perturb[i] += eps
                h_perturb[i] = np.clip(h_perturb[i], 0.0, 1.0)
                c5_perturb = evaluate_c5(h_perturb)
                h_perturb = h.copy()
                h_perturb[i] -= eps
                h_perturb[i] = np.clip(h_perturb[i], 0.0, 1.0)
                c5_perturb_neg = evaluate_c5(h_perturb)
                grad[i] = (c5_perturb - c5_perturb_neg) / (2 * eps)
            # Update h using gradient descent
            h -= learning_rate * grad
            h = np.clip(h, 0.0, 1.0)
            # Ensure the integral constraint is satisfied
            h = h * (n_points / 2) / np.sum(h)
            # Evaluate C5
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            # Early stopping if improving less
            if _ > 50 and np.abs(current_c5 - best_c5) < 1e-6:
                break
        return best_h, best_c5

    # Run optimization
    best_h, best_c5 = optimize_h(h_initial, n_points)
    
    # Try increasing n_points to improve the solution
    for n_points in [200, 300, 400]:
        if time.time() - start_time > budget_s - 5:
            break
        dx = 2.0 / n_points
        # Create initial h from best_h
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_initial = np.clip(h_initial, 0.0, 1.0)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        # Run optimization
        best_h, best_c5 = optimize_h(h_initial, n_points)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n_points
