import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with a moderate number of points
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h = h / np.sum(h) * (n_points / 2)
    
    # Evaluate initial C5
    def compute_C5(h_vals):
        h = h_vals.copy()
        h = np.concatenate([h, h[:1]])  # Pad for full convolution
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_C5(best_h)
    
    # Iterative refinement
    for _ in range(30):  # Limit iterations to stay within budget
        if time.time() - start_time > budget_s - 1:
            break
        
        # Use convex optimization to adjust h for better C5
        n_points = min(200, n_points + 20)  # Incrementally increase resolution
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Define variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2]
        
        # Objective: minimize the maximum overlap
        def objective_func(h_vals):
            h_vals = h_vals.copy()
            h_vals = np.concatenate([h_vals, h_vals[:1]])
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            return np.max(corr) * dx
        
        # Use a simple gradient-based approach
        def obj_func(h_vals):
            return objective_func(h_vals)
        
        # Use a numerical derivative method
        def grad_func(h_vals):
            eps = 1e-6
            grad = np.zeros_like(h_vals)
            for i in range(len(h_vals)):
                h_plus = h_vals.copy()
                h_plus[i] += eps
                h_minus = h_vals.copy()
                h_minus[i] -= eps
                grad[i] = (obj_func(h_plus) - obj_func(h_minus)) / (2 * eps)
            return grad
        
        # Perform gradient descent
        h_vals = np.random.rand(n_points) * 2
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            grad = grad_func(h_vals)
            h_vals -= 0.01 * grad
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        
        # Evaluate
        current_c5 = compute_C5(h_vals)
        if current_c5 < best_c5:
            best_h = h_vals.copy()
            best_c5 = current_c5
    
    # Use the best found solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (len(best_h) / 2)
    return best_h, best_c5, len(best_h)
