import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported helper functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Parameters
    max_points = 100  # Max number of points to consider
    n_points = 50  # Start with 50 points
    
    # Initial guess from provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = np.array(h_initial)
    h_initial = h_initial[:n_points]  # Truncate to n_points if needed
    
    # Ensure sum of h_initial is n_points / 2
    sum_h_initial = np.sum(h_initial)
    if abs(sum_h_initial - n_points / 2) > 1e-6:
        h_initial = h_initial / sum_h_initial * (n_points / 2)
    
    def compute_c5(h):
        dx = 2.0 / len(h)
        h_full = np.concatenate([h, np.zeros_like(h)])
        h_shifted = np.concatenate([np.zeros_like(h), h])
        correlate = np.correlate(h_full, 1 - h_shifted, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def objective(x):
        h = x.copy()
        h[h < 0] = 0
        h[h > 1] = 1
        return compute_c5(h)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Optimization setup
    x = cp.Variable(n_points)
    objective_func = cp.Minimize(objective(x))
    constraints = [constraint_sum(x) == 0, 0 <= x, x <= 1]
    problem = cp.Problem(objective_func, constraints)
    
    # Solve the optimization problem
    start_time = time.time()
    result = problem.solve(solver=cp.SCS, eps=1e-4, max_iter=1000)
    
    # Check if the problem was solved successfully
    if result is None or not np.isfinite(result):
        # If optimization failed, use the initial guess
        h_best = h_initial.copy()
    else:
        h_best = x.value
    
    # Evaluate the best solution
    c5_bound = compute_c5(h_best)
    
    # Return the best solution
    return h_best, c5_bound, n_points
