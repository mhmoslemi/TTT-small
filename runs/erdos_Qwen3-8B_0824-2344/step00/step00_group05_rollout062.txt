    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Start with 100 points, can be increased if needed
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()  # Use the provided initial values
    initial_h = initial_h[:n_points]  # Ensure it matches the number of points
    
    # Ensure the initial_h has the correct integral
    sum_h = np.sum(initial_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        # Adjust the initial_h to meet the constraint
        initial_h = initial_h / sum_h * (n_points / 2)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        shifted_h = np.zeros_like(h)
        for k in range(1, n_points):
            shifted_h[k:] = h[:-k]
        overlap = np.sum(h * (1 - shifted_h)) * dx
        return np.max(overlap)
    
    # Optimization function
    def objective_function(x):
        h = np.clip(x, 0, 1)
        return compute_c5(h)
    
    # Optimization constraints
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Setup optimization
    x = cp.Variable(n_points)
    objective = cp.Minimize(objective_function(x))
    constraints = [constraint(x) == 0, 0 <= x, x <= 1]
    prob = cp.Problem(objective, constraints)
    
    # Run optimization
    start_time = time.time()
    result = prob.solve(solver=cp.SCS, eps=1e-4, max_iter=100)
    
    # If optimization fails, fall back to initial values
    if result is None or np.isnan(result):
        h_opt = initial_h
    else:
        h_opt = x.value
    
    # Ensure the final h_opt is valid
    h_opt = np.clip(h_opt, 0, 1)
    sum_h = np.sum(h_opt)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h_opt = h_opt / sum_h * (n_points / 2)
    
    # Compute final C5 value
    c5_bound = compute_c5(h_opt)
    
    # Return the result
    return (h_opt, c5_bound, n_points)
