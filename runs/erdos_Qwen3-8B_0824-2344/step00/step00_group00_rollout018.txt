import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 45
    h_values = initial_h_values.copy()
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        # Compute (1 - h[x + k]) for all k
        h_shifted = np.zeros((len(h), len(h)))
        for k in range(len(h)):
            h_shifted[:, k] = np.roll(h, k)
        shifted_1_minus_h = 1.0 - h_shifted
        # Compute the overlap integral for each shift
        overlap_integrals = np.zeros(len(h))
        for k in range(len(h)):
            overlap_integrals[k] = np.sum(h * shifted_1_minus_h[:, k]) * dx
        return np.max(overlap_integrals)
    
    # Function to evaluate the solution
    def evaluate_solution(h):
        return compute_c5(h)
    
    # Define the optimization problem
    def optimize_h(h_guess, n_points, budget_s):
        start_time = time.time()
        best_c5 = float('inf')
        best_h = h_guess.copy()
        
        # Try different numbers of points
        for n in range(n_points, n_points + 10):
            if time.time() - start_time > budget_s - 1:
                break
            h = np.copy(h_guess)
            h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h_guess)), h)
            h = h / np.sum(h) * (n / 2)
            h = np.clip(h, 0, 1)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Try simple gradient descent refinement
        for _ in range(10):
            if time.time() - start_time > budget_s - 1:
                break
            h = best_h.copy()
            for i in range(len(h)):
                h[i] += np.random.normal(0, 0.01)
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (len(h) / 2)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        return best_h, best_c5
    
    # Run the optimization
    best_h, best_c5 = optimize_h(h_values, n_points, budget_s)
    
    # Refine with a small local search
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        h = best_h.copy()
        for i in range(len(h)):
            h[i] += np.random.normal(0, 0.01)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (len(h) / 2)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final check
    final_c5 = compute_c5(best_h)
    final_h_values = best_h.copy()
    final_n_points = len(final_h_values)
    
    return final_h_values, final_c5, final_n_points
