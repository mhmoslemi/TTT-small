import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set the seed for reproducibility
    np.random.seed(seed)
    
    # Define the initial h_values
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    
    # We'll try different sample numbers to find the best solution
    n_points_list = [50, 75, 100, 125, 150, 175, 200]
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    # Try different number of samples
    for n_points in n_points_list:
        # Compute dx
        dx = 2.0 / n_points
        
        # Initialize h with some perturbation from the initial solution
        h = np.copy(h_initial)
        if n_points > n_points_initial:
            # Use a uniform distribution to fill the extra points
            h = np.concatenate([h, np.random.rand(n_points - n_points_initial)])
        else:
            # Use a uniform distribution to fill the missing points
            indices = np.random.choice(n_points_initial, n_points, replace=False)
            h = np.zeros(n_points)
            h[indices] = h_initial
        
        # Ensure sum(h) * dx == 1
        h /= (np.sum(h) * dx)
        h *= (n_points / 2.0)
        
        # Optimize the function to minimize C5
        def objective(x):
            h_values = x.reshape((n_points,))
            dx = 2.0 / n_points
            # Ensure h_values are in [0, 1]
            h_values = np.clip(h_values, 0, 1)
            # Ensure sum(h) * dx == 1
            h_values = h_values * (n_points / 2.0) / np.sum(h_values)
            # Compute the convolution
            conv = signal.correlate(h_values, 1 - h_values, mode="full")
            # Compute the maximum value
            c5 = np.max(conv) * dx
            return c5
        
        def constraint(x):
            h_values = x.reshape((n_points,))
            return np.sum(h_values) - n_points / 2.0
        
        # Set up optimization
        x0 = h.copy()
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint}]
        
        # Perform optimization with L-BFGS-B
        result = optimize.minimize(
            objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-7}
        )
        
        # Extract the best solution
        best_h_candidate = result.x
        best_h_candidate = np.clip(best_h_candidate, 0, 1)
        best_h_candidate = best_h_candidate * (n_points / 2.0) / np.sum(best_h_candidate)
        
        # Compute the C5 value
        conv = signal.correlate(best_h_candidate, 1 - best_h_candidate, mode="full")
        c5 = np.max(conv) * (2.0 / n_points)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h_candidate
        
        # Print progress
        print(f"n_points={n_points}, C5={c5}")
    
    # Return best solution found
    return (best_h, best_c5, len(best_h))
