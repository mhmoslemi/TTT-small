```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    h_values = initial_h_values.copy()
    n_points = len(h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h_values) * dx
    # Adjust to ensure sum(h) * dx = 1
    h_values = h_values / sum_h * 1.0
    
    # Function to compute the C5 bound
    def compute_c5(h):
        h = np.array(h)
        n_points = len(h)
        dx = 2.0 / n_points
        # Ensure values are within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the convolution: correlate(h, 1-h, mode='full')
        corr = signal.correlate(h, 1 - h, mode='full')
        # Normalize by dx
        c5 = np.max(corr) * dx
        return c5
    
    # Function to optimize h to minimize C5
    def objective(h):
        return compute_c5(h)
    
    # Function to enforce the constraint sum(h) * dx = 1
    def constraint_sum(h):
        return np.sum(h) - (n_points / 2)  # sum(h) * dx = 1 => sum(h) = n_points / 2
    
    # Setup optimization problem with bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Use L-BFGS-B optimizer for faster convergence
    result = opt.minimize(
        objective,
        h_values,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 500, 'disp': False}
    )
    
    # Check if we have a feasible solution and improve it
    if result.success:
        best_h = result.x
        best_c5 = compute_c5(best_h)
    else:
        best_h = h_values
        best_c5 = compute_c5(best_h)
    
    # Try a few random perturbations to avoid local minima
    for _ in range(3):
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        perturb_h = best_h + perturb
        perturb_h = np.clip(perturb_h, 0, 1)
        perturb_c5 = compute_c5(perturb_h)
        if perturb_c5 < best_c5:
            best_h = perturb_h
            best_c5 = perturb_c5
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    print(f"Final C5: {final_c5}")
    
    # Ensure we don't exceed time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (best_h, final_c5, n_points)
    
    # Try adjusting the number of points for faster optimization
    # Start with n_points = 100, adjust based on previous results
    for n_test in [100, 150, 200, 250, 300]:
        test_h = np.random.uniform(0, 1, n_test)
        test_h = test_h * (n_test / 2) / np.sum(test_h)
        test_c5 = compute_c5(test_h)
        if test_c5 < final_c5:
            best_h = test_h
            best_c5 = test_c5
            n_points = n_test
    
    return (best_h, best_c5, n_points)
```
