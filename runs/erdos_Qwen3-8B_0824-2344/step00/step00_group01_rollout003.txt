import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from collections import deque

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided construction
    initial_h_values = np.array(initial_h_values)
    n_points_initial = len(initial_h_values)
    n_points = n_points_initial
    
    # Define the overlap integral function
    def compute_c5(h_values):
        h = h_values
        n = len(h)
        dx = 2.0 / n
        # Compute (1 - h(x + k)) for all k
        h_shifted = np.zeros((n, n))
        for i in range(n):
            h_shifted[i, :] = h[(i + np.arange(n)) % n]
        # Compute the product and integrate
        product = h * (1 - h_shifted)
        # Compute the maximum row sum
        max_row_sum = np.max(np.sum(product, axis=1))
        return max_row_sum * dx
    
    # Function to optimize
    def objective(x):
        return compute_c5(x)
    
    # Function to evaluate the quality
    def evaluate(x):
        return objective(x)
    
    # Optimization loop
    best_h = initial_h_values.copy()
    best_c5 = evaluate(best_h)
    best_n_points = n_points_initial
    start_time = time.time()
    
    # Try reducing n_points to find a better solution
    for n_points in range(50, n_points_initial, -1):
        dx = 2.0 / n_points
        # Initial guess: scale the initial_h_values to fit the new n_points
        # Interpolate to maintain the same shape
        h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), initial_h_values)
        # Ensure the integral is exactly 1
        sum_h = np.sum(h_values)
        h_values = h_values * (n_points / 2) / sum_h
        
        # Define the optimization variables
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Objective: minimize the maximum overlap
        max_overlap = cp.Variable()
        constraints += [max_overlap >= h[i] * (1 - h[(i + np.arange(n_points)) % n_points]) for i in range(n_points)]
        objective = cp.Minimize(max_overlap)
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-6, time_limit=20)
            if prob.status == 'optimal':
                h_values_opt = h.value

            # If optimization fails, use the initial guess
            pass
        
        # Check if time is up
        if time.time() - start_time > budget_s:
