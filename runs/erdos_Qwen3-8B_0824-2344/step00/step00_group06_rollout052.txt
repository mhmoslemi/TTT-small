import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assuming a pre-downloaded initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Define number of points based on budget (smaller sequences are preferred)
    n_points = 100  # A reasonable number that balances speed and accuracy
    
    # Initial guess: use the pre-downloaded construction or random
    if len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Convert to float64 for numerical stability
    h_initial = h_initial.astype(np.float64)
    
    # Define optimization problem
    h = cp.Variable(n_points)
    
    # Objective: minimize the maximum overlap integral
    # We'll use a convex relaxation by minimizing the maximum via a convex surrogate
    # Here, we directly minimize the maximum using a scalar decision variable
    # This is a convex problem and can be solved efficiently
    # We'll use a convex relaxation approach
    
    # For the optimization, we'll minimize the max overlap
    # We'll convert the correlation into a form that can be optimized
    max_overlap = cp.Variable()
    
    # Compute the correlation using convolution
    # Since we have a fixed dx, we can compute this as a dot product
    # We'll use a convex relaxation approach
    # Here, we use a scalar variable `max_overlap` to track the maximum overlap
    
    # We'll use a loop to evaluate every possible shift k and ensure the maximum overlap is minimized
    # Due to the computational cost, we'll use a small number of shifts for efficiency
    # We'll compute for k in [-n_points//2, n_points//2]
    
    # For the optimization, we'll directly minimize the maximum value of the overlap at each shift
    constraints = []
    for k in range(-n_points // 2, n_points // 2 + 1):
        if k == 0:
            continue  # Skip the case where k=0, as it's not relevant for overlap
        # Compute the overlap for shift k
        # For this, we'll use a finite difference approach
        overlap = 0.0
        for i in range(n_points):
            if i + k < 0 or i + k >= n_points:
                continue
            overlap += h[i] * (1 - h[i + k])
        # We minimize the maximum of all overlaps
        constraints.append(max_overlap >= overlap)
    
    # Additional constraints
    constraints.append(cp.sum(h) == n_points / 2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Objective: minimize the maximum overlap
    objective = cp.Minimize(max_overlap)
    
    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    
    # Solve within the budget
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining < 0:
        return h_initial, evaluate_erdos_solution(h_initial), n_points
    
    try:
        result = prob.solve(solver=cp.ECOS, max_iter=1000, verbose=False, time_limit=remaining)
        if result is not None:
            best_h = h.value
            best_c5 = max_overlap.value
        else:
            best_h = h_initial
            best_c5 = evaluate_erdos_solution(h_initial)
    except:
        best_h = h_initial
        best_c5 = evaluate_erdos_solution(h_initial)
    
    # Return the best solution
    return best_h, best_c5, n_points
