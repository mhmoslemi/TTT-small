def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    c5_values = []
    for k in range(1, n_points):
        overlap = np.correlate(h_values, 1 - h_values, mode="full")
        c5_values.append(overlap[k] * dx)
    return max(c5_values)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Start with a base solution
    h_values = initial_h_values()
    n_points = len(h_values)
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    
    # We'll try a few different strategies
    
    # 1. Random perturbations
    def perturb(h, magnitude=0.1):
        h = h.copy()
        for i in range(len(h)):
            h[i] += np.random.uniform(-magnitude, magnitude)
            h[i] = np.clip(h[i], 0, 1)
        return h
    
    # 2. Try optimizing with scipy.optimize
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # 3. Try a few random restarts
    for _ in range(5):
        # Initial perturbation
        h_values = perturb(h_values, magnitude=0.05)
        # Try gradient descent
        result = optimize.minimize(objective, h_values, method='L-BFGS-B',
                                  bounds=[(0, 1)]*n_points, tol=1e-6)
        if result.success:
            h_values = result.x
            current_c5 = evaluate_erdos_solution(h_values)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_values.copy()
        # Try simulated annealing
        result = optimize.minimize(objective, h_values, method='SLSQP',
                                  bounds=[(0, 1)]*n_points, tol=1e-6)
        if result.success:
            h_values = result.x
            current_c5 = evaluate_erdos_solution(h_values)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_values.copy()
        # Try random perturbation
        h_values = perturb(h_values, magnitude=0.05)
    
    # 4. Use CVXPY to optimize
    # Since we're constrained to [0,1] and sum(h) = n_points / 2,
    # We can cast this as a convex optimization problem
    # But this is more complex. Let's try to use it for refining the solution
    h = cp.Variable(n_points, lb=0, ub=1)
    objective = cp.Maximize(cp.max(signal.correlate(h.value, 1 - h.value, mode="full") * dx))
    problem = cp.Problem(objective)
    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status == 'optimal':
            h_values = h.value
            current_c5 = evaluate_erdos_solution(h_values)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_values.copy()
    except:
        pass
    
    # 5. Final perturbations
    best_h = perturb(best_h, magnitude=0.05)
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, n_points
