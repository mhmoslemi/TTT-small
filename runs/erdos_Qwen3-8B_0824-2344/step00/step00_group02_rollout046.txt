def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()

    np.random.seed(seed)
    
    # Initial sample size

    def evaluate_overlap(h):
        return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
    
    # Function to compute the total integral
    def total_integral(h):
        return np.sum(h) * (2.0 / len(h))
    
    # Initial solution
    h = initial_h_values.copy()
    
    # Evaluate the initial C5 value
    c5 = evaluate_overlap(h)
    
    # Function to perform gradient descent optimization
    def optimize_h(h, max_iterations=100, learning_rate=0.01, tolerance=1e-5):
        for _ in range(max_iterations):
            # Compute the gradient of the overlap integral
            # We use a numerical gradient approximation
            
            # Compute the current overlap values
            overlap = np.correlate(h, 1 - h, mode="full")
            max_overlap = np.max(overlap)
            idx = np.argmax(overlap)
            
            # Perturb h slightly to estimate the gradient
            delta = 1e-4
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_pert = h.copy()
                h_pert[i] += delta
                overlap_pert = np.correlate(h_pert, 1 - h_pert, mode="full")
                max_overlap_pert = np.max(overlap_pert)
                grad[i] = (max_overlap_pert - max_overlap) / delta
            
            # Compute the gradient of the total integral constraint
            grad_total = np.ones_like(h) * (2.0 / len(h))
            
            # Use a penalty method to enforce the constraint
            penalty = 100.0
            grad_penalty = penalty * (total_integral(h) - 1.0) * grad_total
            
            # Update h using gradient descent
            h -= learning_rate * (grad + grad_penalty)
            
            # Clamp h to [0, 1]
            h = np.clip(h, 0.0, 1.0)
            
            # Check for convergence
            if np.abs(total_integral(h) - 1.0) < tolerance:
                break
            
            # Check for time budget
            if time.time() - start_time > budget_s - 1:
                break
        
        return h
    
    # Optimize the initial solution
    h_optimized = optimize_h(h)

    c5_optimized = evaluate_overlap(h_optimized)

    for _ in range(5):

            c5_perturbed = evaluate_overlap(h_perturbed)

                h_optimized = h_perturbed
                c5_optimized = c5_perturbed

    return (h_optimized, c5_optimized, len(h_optimized))
