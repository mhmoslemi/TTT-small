```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assuming the initial construction is loaded

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster optimization
    max_iter = 100
    
    # Initial guess
    h_initial = initial_h_values[:n_points]  # Use initial construction
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Ensure sum(h) == n_points / 2

    for _ in range(max_iter):
        # Define the objective function to minimize the max overlap
        def objective(h):
            nonlocal best_c5
            h = np.clip(h, 0, 1)
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
            c5 = np.max(overlap)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            return c5
        
        # Define bounds for h
        bounds = Bounds(0, 1)
        
        # Optimization using L-BFGS-B
        result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, tol=1e-8, options={'maxiter': 10})
        
        # Check if we're running out of time
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 10:
            break
        
        h_initial = result.x  # Update initial guess for next iteration
    
    # Try with a slightly higher number of points
    n_points = 200
    h_initial = best_h
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Ensure sum(h) == n_points / 2
    bounds = Bounds(0, 1)
    
    # Final optimization with more points
    def final_objective(h):
        nonlocal best_c5
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        c5 = np.max(overlap)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    result = minimize(final_objective, h_initial, method='L-BFGS-B', bounds=bounds, tol=1e-8, options={'maxiter': 10})
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure sum(h) == n_points / 2
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h, best_c5, n_points)
```
