import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all shifted versions of 1 - h
    shifted_1_minus_h = []
    for k in range(1, n_points):
        shifted = np.roll(h, k)
        shifted_1_minus_h.append(1 - shifted)
    # Compute the correlation
    corr = np.zeros(n_points)
    for k in range(n_points):
        corr[k] = np.sum(h * (1 - np.roll(h, k)))
    # Normalize by dx
    corr = corr * dx
    # Return the maximum value
    return np.max(corr)

def initial_h_values():
    # Initial solution with 45 samples
    return [0.5 if i % 3 == 0 else 0.0 for i in range(45)]

# Strategy:
# 1. Start from a known initial solution (45 samples)
# 2. Use a local optimization approach to improve the solution
# 3. Use a genetic algorithm to explore the solution space
# 4. Track the best solution found and return it within the time budget

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Start with 100 samples
    dx = 2.0 / n_points
    max_iterations = 100  # Maximum number of iterations
    population_size = 30  # Population size for genetic algorithm
    mutation_rate = 0.1  # Mutation rate for genetic algorithm
    crossover_rate = 0.8  # Crossover rate for genetic algorithm
    
    # Initial solution
    initial_h = initial_h_values()
    initial_h = [initial_h[i] for i in range(n_points)]
    initial_h = [initial_h[i] * (n_points / 2) / sum(initial_h) for i in range(n_points)]
    h_values = initial_h
    
    # Optimize using a combination of gradient descent and genetic algorithm
    best_h = h_values
    best_c5 = evaluate_erdos_solution(h_values)
    start_time = time.time()
    
    # Iterative optimization
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Local optimization using gradient descent
        h = np.array(best_h)
        constraints = [
            cp.sum(h) == n_points / 2,
            cp.Minimize(cp.Max([cp.sum(h * (1 - np.roll(h, k))) for k in range(1, n_points)]))
        ]
        prob = cp.Problem(constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-3)
            h = np.array(prob.variables()[0].value)
            h = np.maximum(0, np.minimum(1, h))
            current_c5 = evaluate_erdos_solution(h)
        except:
            h = best_h
            current_c5 = best_c5
        
        # Genetic algorithm
        population = [np.random.rand(n_points) * 2 - 1 for _ in range(population_size)]
        population = [np.maximum(0, np.minimum(1, p * (n_points / 2) / sum(p))) for p in population]
        for p in population:
            p = np.array(p)
        
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                mutation = np.random.rand(n_points) * 0.2 - 0.1
                population[i] = np.array(population[i]) + mutation
                population[i] = np.maximum(0, np.minimum(1, population[i]))
        
        for i in range(population_size):
            if np.random.rand() < crossover_rate:
                j = np.random.randint(population_size)
                crossover_point = np.random.randint(n_points)
                population[i] = np.concatenate([population[i][:crossover_point], population[j][crossover_point:]])
                population[i] = np.maximum(0, np.minimum(1, population[i]))
        
        population = [np.array(p) for p in population]
        population = [p * (n_points / 2) / sum(p) for p in population]
        
        # Evaluate all solutions
        c5_values = [evaluate_erdos_solution(p) for p in population]
        best_index = np.argmin(c5_values)
        best_h = population[best_index]
        current_c5 = c5_values[best_index]
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h
    
    # Convert best_h to list and return
    best_h = np.array(best_h).tolist()
    return best_h, best_c5, n_points
