    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Evaluate the current C5
    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points - 1]])
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Objective function to minimize
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return evaluate_c5(h)

    # Constraint: sum(h) == n_points / 2
    def constraint(h_flat):
        return np.sum(h_flat) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization function
    def optimize_h(h0):
        result = minimize(objective, h0, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxfun': 10000})
        return result.x

    # Start with the initial solution
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    best_n_points = n_points

    # Try different numbers of points (start with lower ones for speed)
    for n_points in [40, 60, 80, 100, 120, 140, 160, 180, 200]:
        # Generate initial points
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # Distribute evenly with some randomness
        h = np.random.rand(n_points)
        # Normalize to ensure sum is n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Optimize
        start_time = time.time()
        h_opt = optimize_h(h)
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            continue
        
        c5 = evaluate_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
            best_n_points = n_points

    # Refine with a few more steps
    for _ in range(3):
        h_opt = optimize_h(best_h)
        c5 = evaluate_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5

    return best_h.tolist(), best_c5, best_n_points
