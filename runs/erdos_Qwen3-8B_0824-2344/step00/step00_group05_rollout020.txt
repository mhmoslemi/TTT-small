```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Example placeholder; actual data may differ

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 samples, adjust if needed based on time budget
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Use initial values as a starting point
    h = h / np.sum(h) * (n_points / 2)  # Ensure sum(h) * dx = 1

    # Evaluate initial C5
    def compute_c5(h):
        h = np.atleast_2d(h)
        shifted = np.zeros((n_points, n_points))
        for i in range(n_points):
            shifted[i, :] = h[i, :] * (1 - h[np.roll(range(n_points), i), :])
        overlap = np.sum(shifted) * dx
        return np.max(overlap)

    c5_initial = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_initial

    # Optimization loop
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Define variables
        h_var = cp.Variable(n_points, boole=True)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]

        # Objective: minimize the maximum overlap
        # We use a convex relaxation by minimizing the maximum over all shifts
        # Here, we use a heuristic to approximate the max
        shift_vars = []
        for k in range(n_points):
            shift_var = cp.Variable(n_points, boole=True)
            shift_vars.append(shift_var)
            constraints.append(cp.sum(shift_var) == n_points / 2)
            constraints.append(0 <= shift_var)
            constraints.append(shift_var <= 1)

        # Objective: minimize max over k
        obj = cp.Minimize(cp.Max([cp.sum(h_var * (1 - shift_var[i, :])) for i in range(n_points)]))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-4)
            if prob.status == 'optimal':
                h = h_var.value
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        except:
            pass

        # Early termination if no improvement
        if np.abs(best_c5 - c5_initial) < 1e-6:
            break

    return (best_h, best_c5, n_points)
```
