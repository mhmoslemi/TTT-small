import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.random.rand(42) * 2 - 1  # Example initial values (scaled to [0,2])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Determine the number of points, starting with 100 and increasing if needed
    n_points = 100
    max_n_points = 500
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Initial guess based on the provided initial_h_values
    initial_h = initial_h_values.copy()
    initial_h = np.interp(initial_h, [initial_h.min(), initial_h.max()], [0, 2])
    initial_h = np.interp(initial_h, [0, 2], [0, 1])
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = np.resize(initial_h, (n_points,))
    initial_h = np.repeat(initial_h, 2 // n_points)  # Ensure the length is 2
    
    # Function to evaluate the solution
    def evaluate(h):
        nonlocal best_c5
        h = np.clip(h, 0, 1)
        h = np.resize(h, (n_points,))
        h = np.interp(h, [0, 1], [0, 2])
        h = np.interp(h, [0, 2], [0, 1])
        h = np.repeat(h, 2 // n_points)
        integral = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        c5 = np.max(integral)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        return c5
    
    # Use a simple optimization method with bounds
    def optimize_h():
        nonlocal best_c5
        h = np.random.rand(n_points)
        h = np.interp(h, [0, 1], [0, 2])
        h = np.interp(h, [0, 2], [0, 1])
        h = np.repeat(h, 2 // n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        result = optimize.minimize(evaluate, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(len(h))])
        return result
    
    # Run optimization with multiple iterations
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        result = optimize_h()
        if result.success:
            best_c5 = result.fun
            best_h = result.x
        else:
            break
    
    # If not enough points, increase n_points and try again
    while n_points < max_n_points and time.time() - start_time < budget_s:
        n_points *= 2
        best_h = np.random.rand(n_points)
        best_h = np.interp(best_h, [0, 1], [0, 2])
        best_h = np.interp(best_h, [0, 2], [0, 1])
        best_h = np.repeat(best_h, 2 // n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate(best_h)
    
    # Ensure best_h is of length 2
    best_h = np.interp(best_h, [0, 1], [0, 2])
    best_h = np.interp(best_h, [0, 2], [0, 1])
    best_h = np.repeat(best_h, 2 // n_points)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h, best_c5, len(best_h))
