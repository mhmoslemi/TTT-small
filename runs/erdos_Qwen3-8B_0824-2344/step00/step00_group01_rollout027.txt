import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points for a balance between resolution and speed
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h = h[:n_points]
    if sum(h) * 2.0 / n_points != 1:
        h = h / sum(h) * (n_points / 2.0)
    
    # Function to compute C5
    def compute_c5(h):
        dx = 2.0 / n_points
        h_corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(h_corr) * dx
        return c5
    
    # Function to optimize h
    def optimize_h(h_initial):
        h = h_initial.copy()
        h = h / sum(h) * (n_points / 2.0)
        h = np.clip(h, 0.0, 1.0)
        return h
    
    # Optimization loop
    best_h = h
    best_c5 = compute_c5(h)
    
    # Try different numbers of points
    for n_points in [100, 200, 300, 400, 500]:
        # If time is exhausted, break
        if time.time() - start_time > budget_s - 5:
            break
        
        # Reset h
        h = np.random.rand(n_points)
        h = h / sum(h) * (n_points / 2.0)
        h = np.clip(h, 0.0, 1.0)
        
        # Optimization using gradient descent
        for _ in range(100):
            # Compute current C5
            current_c5 = compute_c5(h)
            
            # Compute gradients numerically
            grad = np.zeros(n_points)
            dx = 2.0 / n_points
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = h_plus / sum(h_plus) * (n_points / 2.0)
                h_plus = np.clip(h_plus, 0.0, 1.0)
                c5_plus = compute_c5(h_plus)
                
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                h_minus = h_minus / sum(h_minus) * (n_points / 2.0)
                h_minus = np.clip(h_minus, 0.0, 1.0)
                c5_minus = compute_c5(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            
            # Update h
            h -= 0.01 * grad
            h = np.clip(h, 0.0, 1.0)
            h = h / sum(h) * (n_points / 2.0)
            
            # Check if we have a better solution
            if compute_c5(h) < current_c5:
                best_h = h
                best_c5 = compute_c5(h)
        
        # Update best solution
        if compute_c5(h) < best_c5:
            best_h = h
            best_c5 = compute_c5(h)
    
    # Final check
    current_c5 = compute_c5(best_h)
    if current_c5 < best_c5:
        best_h = best_h
        best_c5 = current_c5
    
    # Return the best solution
    return (best_h.tolist(), best_c5, n_points)
