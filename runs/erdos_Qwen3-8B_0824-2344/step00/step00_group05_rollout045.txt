import numpy as np
import numpy.random as npr
import scipy.optimize as opt
import time
import copy
from scipy.optimize import minimize

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Choose a reasonable number of points for speed and accuracy
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()  # Use the initial construction as a starting point
    initial_h = initial_h[:n_points]  # Ensure it's the correct length

    # Ensure the initial_h satisfies the integral constraint
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        # Adjust if necessary (e.g., by adding a small delta to some elements)
        delta = (n_points / 2 - sum_h) / n_points
        initial_h += delta

    # Objective function: max of the correlation
    def objective(h):
        h_array = np.array(h)
        # Ensure h_array is within [0, 1]
        h_array = np.clip(h_array, 0, 1)
        # Compute the correlation
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        # Multiply by dx for the integral
        corr *= dx
        # Return the maximum value
        return np.max(corr)

    # Constraints: sum(h) == n_points / 2
    def constraint_func(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_func}]
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization settings
    max_iter = 100
    tol = 1e-5
    options = {'maxiter': max_iter, 'ftol': tol, 'disp': False}

    # Run optimization
    start_time = time.time()
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options=options)
    elapsed_time = time.time() - start_time

    # Check if the optimization was successful
    if result.success:
        best_h = result.x
    else:
        # If optimization failed, use the initial guess
        best_h = initial_h

    # Evaluate the best solution
    best_c5 = objective(best_h)
    print(f"Optimized C₅ value: {best_c5}")

    return best_h, best_c5, n_points
