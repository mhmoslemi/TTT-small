import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assume this is a 1D numpy array

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided values
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial) * dx
    if np.abs(sum_h - 1.0) > 1e-6:
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * 2.0  # Normalize to sum to 1 over [0,2]
    
    # Function to evaluate C5 given h_values
    def evaluate_C5(h_values):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        # Compute the overlap integral for all shifts
        corr = np.correlate(h, 1 - h, mode="full")
        # The full correlation has length 2n_points - 1
        # We are interested in the maximum integral over all shifts
        # The correlation values are multiplied by dx (2/n_points)
        c5 = np.max(corr * (2.0 / n_points))
        return c5
    
    # Objective function for optimization
    def objective_func(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        return evaluate_C5(h)
    
    # Optimization parameters
    max_iter = 100
    tol = 1e-5
    best_h = h_initial.copy()
    best_c5 = evaluate_C5(best_h)
    
    # Simple gradient descent with momentum
    learning_rate = 0.01
    momentum = 0.8
    grad = np.zeros(n_points)
    for i in range(max_iter):
        # Evaluate objective and gradient numerically
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        
        # Compute forward and backward differences for gradient
        grad_forward = np.zeros(n_points)
        grad_backward = np.zeros(n_points)
        for j in range(n_points):
            h_forward = h.copy()
            h_forward[j] += 1e-6
            h_forward = np.clip(h_forward, 0, 1)
            grad_forward[j] = (evaluate_C5(h_forward) - evaluate_C5(h)) / 1e-6
            
            h_backward = h.copy()
            h_backward[j] -= 1e-6
            h_backward = np.clip(h_backward, 0, 1)
            grad_backward[j] = (evaluate_C5(h) - evaluate_C5(h_backward)) / 1e-6
            
        grad = (grad_forward + grad_backward) / 2.0
        
        # Update
        descent = -learning_rate * grad
        best_h += momentum * descent
        best_h = np.clip(best_h, 0, 1)
        
        # Check if we are improving
        current_c5 = evaluate_C5(best_h)
        if current_c5 < best_c5 - tol:
            best_c5 = current_c5
        elif current_c5 > best_c5 + tol:
            # If we are getting worse, reset
            best_h = h_initial.copy()
            best_c5 = evaluate_C5(best_h)
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Optionally, run a convex optimization to fine-tune
    # This is a simplified version of the problem using convex optimization
    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1]
    constraints += [cp.sum(h) == n_points / 2]  # Ensure the integral is 1 over [0,2]
    
    # Define the objective: minimize the maximum overlap integral
    # This is a bit complex to model directly, so we'll approximate by minimizing the average overlap
    # We use the correlation result as an estimate of the worst-case overlap
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = cp.Max(corr * (2.0 / n_points))
    objective = cp.Minimize(c5)
    
    # Solve the convex optimization
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == "optimal":
            best_h = h.value
            best_c5 = evaluate_C5(best_h)
    except:
        pass
    
    # Ensure the final best_h is valid
    best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_C5(best_h)
    
    return best_h, best_c5, n_points
