# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Reshape h_values to 1D array
    h = np.array(h_values).reshape(-1)
    # Compute 1-h
    one_minus_h = 1 - h
    # Compute the cross-correlation
    corr = correlate(h, one_minus_h, mode="full")
    # Extract the relevant part (the overlap integrals)
    # The overlap integrals are between positions -n+1 to n-1
    # But we need to check all k such that the overlap is valid
    # The max k corresponds to the maximum overlap
    # We take the maximum over the correlation values, and multiply by dx
    c5 = np.max(corr) * dx
    return c5
