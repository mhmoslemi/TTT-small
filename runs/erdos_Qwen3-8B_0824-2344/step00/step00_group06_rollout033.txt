```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is a pre-loaded initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can be increased if time permits
    dx = 2.0 / n_points
    h_initial = initial_h_values[:n_points] if len(initial_h_values) > n_points else initial_h_values
    
    # Ensure the initial h_initial has the correct sum
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Optimization function to minimize the maximum overlap integral
    def objective_function(h_values):
        h = np.array(h_values)
        # Compute the overlap integral for all lags
        overlaps = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlaps) * dx
        return c5

    # Constraint: sum(h) == n_points / 2
    def constraint_func(h_values):
        return np.sum(h_values) - n_points / 2

    # Optimization using COBYLA
    result = optimize.minimize(
        objective_function,
        h_initial,
        method='COBYLA',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_func}],
        options={'maxiter': 100, 'disp': False}
    )

    best_h = result.x
    best_c5 = objective_function(best_h)
    
    # If time remains, try increasing n_points and optimize again
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        n_points = 200
        dx = 2.0 / n_points
        h_initial = np.zeros(n_points)
        # Simple initial guess: alternate 0 and 1
        h_initial[::2] = 1.0
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        result = optimize.minimize(
            objective_function,
            h_initial,
            method='COBYLA',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_func}],
            options={'maxiter': 100, 'disp': False}
        )
        
        best_h = result.x
        best_c5 = objective_function(best_h)

    # Final check with CVXPY for potential better solution
    if remaining_time > 5:
        h = cp.Variable(n_points)
        objective = cp.Maximize(cp.multiply(h, 1 - h))
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        prob = cp.Problem(objective, constraints)
        prob.solve()
        best_h = h.value
        best_c5 = evaluate_erdos_solution(best_h)

    # Ensure the final best_c5 is within the budget time
    if time.time() - start_time > budget_s - 1:
        print("Time budget exceeded, returning best so far")
    
    return best_h, best_c5, n_points
```
