```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Choose a reasonable number of points for balance between accuracy and speed
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = initial_h[:n_points]
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2)  # Ensure the integral constraint is satisfied
    initial_h = initial_h / (n_points / 2)  # Normalize back to [0, 1]

    def objective(h):
        h = np.clip(h, 0, 1)
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    # Start optimization
    start_time = time.time()
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        tol=1e-8,
        options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
    )

    # Extract the best solution found
    best_h = result.x
    best_c5 = objective(best_h)

    # Ensure the integral constraint is satisfied
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)

    # Evaluate the final solution
    final_c5 = evaluate_erdos_solution(best_h)

    # Return the best solution found
    return best_h, final_c5, n_points
```
