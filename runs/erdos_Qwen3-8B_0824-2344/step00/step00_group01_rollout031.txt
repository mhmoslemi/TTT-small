```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions and initial construction
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 100  # Start with 100 points, can increase if time allows
    h_values = initial_h_values[:n_points].copy()
    
    # Constraint: sum(h) == n_points / 2
    sum_h = np.sum(h_values)
    h_values = h_values * (n_points / 2) / sum_h
    
    # Function to evaluate the overlap integral
    def evaluate_c5(h):
        dx = 2.0 / len(h)
        h_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(h_corr) * dx
        return c5
    
    # Function to optimize with constraints
    def optimize_h(h_initial, max_iter=100, tol=1e-6):
        h = cp.Variable(len(h_initial))
        obj = cp.Minimize(evaluate_c5(h))
        constraints = [cp.sum(h) == len(h_initial) / 2, 0 <= h, h <= 1]
        prob = cp.Problem(obj, constraints)
        start_time = time.time()
        try:
            result = prob.solve(solver=cp.SCS, max_iter=max_iter, eps=tol, verbose=False)
            if result is not None and time.time() - start_time < budget_s:
                return h.value
            else:
                return h_initial
        except:
            return h_initial
    
    # Run optimization
    best_h = h_values.copy()
    best_c5 = evaluate_c5(best_h)
    best_n_points = n_points
    
    # Try increasing n_points if time allows
    for n_points in [100, 200, 300, 400, 500]:
        h_vals = np.random.rand(n_points) * 2 - 1  # Random initial guess
        h_vals = (h_vals - np.min(h_vals)) / (np.max(h_vals) - np.min(h_vals))
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        
        # Optimize
        h_opt = optimize_h(h_vals)
        c5 = evaluate_c5(h_opt)
        
        # Update best solution
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
            best_n_points = n_points
        
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, best_n_points
```
