import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
import cvxpy as cp

# Pre-imported functions
# evaluate_erdos_solution is not used in this approach
# initial_h_values is provided as initial guess

def run(seed=42, budget_s=60, **kwargs):
    # Use the initial guess as a starting point if available
    initial_h_values = np.array([0.0] * 45)  # default fallback
    if 'initial_h_values' in globals():
        initial_h_values = np.array(globals()['initial_h_values'])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Objective function: minimize the maximum overlap integral
    def objective_function(h_values):
        h = h_values.reshape(-1, 1)
        h_shifted = np.roll(h, shift=1, axis=0)
        # Compute the overlap integrals for all k
        overlaps = np.dot(h, 1 - h_shifted)
        overlap_integral = overlaps * dx
        return np.max(overlap_integral)

    # Constraints:
    # 1. h[i] ∈ [0, 1]
    # 2. sum(h) * dx = 1 => sum(h) == n_points / 2
    bounds = Bounds(0, 1)

    # Use a local optimization method with a starting point
    h_values = initial_h_values.copy()
    start_time = time.time()
    result = optimize.minimize(
        fun=objective_function,
        x0=h_values,
        method='L-BFGS-B',
        bounds=bounds,
        tol=1e-5,
        options={'maxiter': 1000}
    )
    elapsed_time = time.time() - start_time

    # If optimization succeeded, use the result; otherwise, use the initial guess
    if result.success:
        h_values = result.x
    else:
        h_values = initial_h_values

    # Ensure sum(h_values) is exactly n_points / 2
    h_values = np.clip(h_values, 0, 1)
    total = np.sum(h_values)
    if np.abs(total - n_points / 2) > 1e-5:
        # Correct the sum by scaling
        scaling_factor = (n_points / 2) / total
        h_values = h_values * scaling_factor
        h_values = np.clip(h_values, 0, 1)

    # Recompute the C5 value for the optimized h_values
    h = h_values.reshape(-1, 1)
    h_shifted = np.roll(h, shift=1, axis=0)
    overlaps = np.dot(h, 1 - h_shifted)
    overlap_integral = overlaps * dx
    c5_bound = np.max(overlap_integral)

    # Return the best solution found
    return (h_values.tolist(), c5_bound, len(h_values))
