def run(seed=42, budget_s=80, **kwargs):

        correlations = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        return np.max(correlations)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    learning_rate = 0.1
    max_iterations = 100000
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 1.0:
            break
        
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] += (1 - h[i + k])
            if 0 <= i - k < n_points:
                grad[i] -= h[i - k]
        grad *= dx
        
        h -= learning_rate * grad
        
        h_clamped = np.clip(h, 0, 1)
        s = h_clamped.sum()

            h_clamped = np.clip(h_clamped, 0, 1)
