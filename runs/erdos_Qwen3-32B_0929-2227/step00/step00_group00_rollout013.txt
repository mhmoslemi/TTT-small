import numpy as np
import time

def run(seed=42, budget_s=80, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global initial_h_values
    h = np.copy(initial_h_values)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) == n_points/2
    
    # Best solution tracker
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    learning_rate = 0.01
    learning_rate_decay = 0.995
    max_iterations = 1000  # Conservative cap to ensure time fits within 80s
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute overlaps and find maximum overlap index
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        k_idx = np.argmax(overlap)
        current_c5 = overlap[k_idx]
        
        # Update best solution
        if current_c5 < best_c5:
            best_h[:] = h.copy()
            best_c5 = current_c5
        
        # Determine positive k_shift from k_idx
        k_shift = k_idx - (n_points - 1)
        if k_shift < 0:
            k_shift = abs(k_shift)
        else:
            k_shift = k_shift
        if k_shift >= n_points:
            continue  # No overlapping terms for this shift
        
        # Compute gradient for this k_shift
        grad = np.zeros_like(h)
        for x in range(n_points - k_shift):
            grad[x] += (1 - h[x + k_shift])  # Contribution to h[x]
            grad[x + k_shift] -= h[x]  # Contribution to h[x + k_shift]
        
        # Gradient descent update
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Enforce sum constraint (project onto feasible region)
        current_sum = h.sum()
        if not np.isclose(current_sum, target_sum):
            active_mask = (h > 0) & (h < 1)
            active_idx = np.where(active_mask)[0]
            if len(active_idx) > 0:
                delta = target_sum - current_sum
                # Uniform adjustment
                h[active_idx] += delta / len(active_idx)
                # Re-clip after adjustment
                h = np.clip(h, 0, 1)
        
        # Decay learning rate
        learning_rate *= learning_rate_decay
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.astype(np.float64), final_c5, n_points)
