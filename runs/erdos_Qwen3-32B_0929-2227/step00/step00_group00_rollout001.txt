    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    # Ensure initial h is feasible
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) > 1e-6:
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = (target_sum - current_sum) / n_points
        h = np.clip(h + delta, 0, 1)

    learning_rate = 0.01
    max_iter = 10000
    iter_count = 0

    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Compute correlation to find max overlap
        corr = correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(corr)
        k_shift = max_index - (n_points - 1)
        k = abs(k_shift)

        # Compute gradient for the current max shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += 1 - h[i + k]
            if i - k >= 0:
                grad[i] += -h[i - k]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm

        # Update h with gradient descent and projection
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        delta = (target_sum - current_sum) / n_points
        h_new += delta
        h_new = np.clip(h_new, 0, 1)
        h[:] = h_new[:]  # Ensure h remains feasible

        # Update best solution
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = h[:]

        iter_count += 1

    return (best_h.tolist(), best_c5, n_points)
