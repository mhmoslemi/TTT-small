def run(seed=42, budget_s=80):

    start_time = time.time()
    # Optimization parameters
    learning_rate = 0.01

    best_h = h.copy()
    best_c5 = evaluate(max_overlap(h, dx))

    while time.time() - start_time < budget_s:
        # Compute overlaps
        overlaps, k_indices = compute_overlaps(h, n_points, dx)
        current_c5 = overlaps.max()
        # Find the index where max occurs and the corresponding k
        max_idx = np.argmax(overlaps)
        k_max = k_indices[max_idx]

        # Compute gradient for the k_max
        grad = compute_gradient(h, k_max, dx)

        # Gradient step
        h = h - learning_rate * grad

        # Clip to [0,1]
        h = np.clip(h, 0, 1)

        # Renormalize
        sum_h = h.sum()
        delta = target_sum - sum_h
        h += delta / n_points
        h = np.clip(h, 0, 1)

        # Evaluate the new C5
        overlaps_new, _ = compute_overlaps(h, n_points, dx)
        current_c5_new = overlaps_new.max()

        # Check if this is the best solution so far
        if current_c5_new < best_c5:
            best_h = h.copy()
            best_c5 = current_c5_new

        # Adaptive learning rate (optional)
        if current_c5_new < current_c5:
            learning_rate *= 1.05  # increase
        else:
            learning_rate *= 0.5  # decrease

    h_values = best_h
    c5_bound = best_c5
    return h_values.tolist(), c5_bound, n_points
