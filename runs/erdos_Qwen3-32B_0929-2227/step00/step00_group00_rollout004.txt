import numpy as np
import time

def run(seed=42, budget_s=80, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    budget_time = budget_s

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial feasibility check
    if not np.allclose(np.sum(h), target_sum):
        # Normalize if initial sum is not correct
        h = h / np.sum(h) * target_sum

    best_h = h.copy()
    best_c5 = None

    learning_rate = 0.1
    max_iterations = 10000  # High upper bound; actual iterations governed by time
    it = 0

    while time.time() - start_time < budget_time and it < max_iterations:
        it += 1

        # Compute overlaps and find the max shift
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        i_max = np.argmax(overlaps)
        k_shift = i_max - (n_points - 1)

        # Compute gradient for the maximum shift
        gradient = np.zeros(n_points)
        for j in range(n_points):
            term1 = 0.0
            term2 = 0.0
            if 0 <= (j + k_shift) < n_points:
                term1 = 1.0 - h[j + k_shift]
            if 0 <= (j - k_shift) < n_points:
                term2 = h[j - k_shift]
            gradient[j] = term1 - term2
        gradient *= dx  # Multiply by dx for the integral scaling

        # Gradient descent step with clipping and renormalization
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0.0, 1.0)
        sum_h_new = np.sum(h_new)
        if not np.isclose(sum_h_new, 0.0):
            scaling = target_sum / sum_h_new
            h_new *= scaling
            h_new = np.clip(h_new, 0.0, 1.0)
        else:
            h_new = h.copy()

        h[:] = h_new

        # Compute current C5 and update best
        overlaps_current = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlaps_current)
        if best_c5 is None or current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h.copy()

        # Learning rate schedule (adaptive)
        learning_rate *= 0.997  # Exponential decay

    # Final validation
    if np.allclose(np.sum(best_h), target_sum) and np.all(0.0 <= best_h) and np.all(best_h <= 1.0):
        return (best_h, best_c5, n_points)
    else:
        # Fallback to initial if constraints are violated
        return (initial_h_values, np.max(np.correlate(initial_h_values, 1 - initial_h_values, mode='full') * dx), n_points)
