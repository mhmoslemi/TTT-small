import numpy as np
import time

def run(seed=42, budget_s=80, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)

    start_time = time.time()

    while time.time() - start_time < budget_s:
        O_vals = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(O_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        max_index = np.argmax(O_vals)
        k = max_index - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            term1 = 0.0
            if k >= 0:
                if i + k < n_points:
                    term1 = 1.0 - h[i + k]
            else:
                if i + k >= 0:
                    term1 = 1.0 - h[i + k]

            term2 = 0.0
            if k > 0:
                if i - k >= 0:
                    term2 = -h[i - k]
            else:
                if (i - k) < n_points and (i - k) >= 0:
                    term2 = -h[i - k]

            grad[i] = dx * (term1 + term2)

        grad_norm = np.linalg.norm(grad)
        learning_rate = 0.1 / (grad_norm + 1e-8)

        h -= learning_rate * grad

        clipped_h = np.clip(h, 0, 1)
        current_sum = np.sum(clipped_h)
        if current_sum != 0:
            scale = target_sum / current_sum
            h = np.clip(clipped_h * scale, 0, 1)
        else:
            h[:] = clipped_h

    return best_h, best_c5, n_points
