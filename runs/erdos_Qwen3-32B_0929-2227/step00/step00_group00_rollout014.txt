                    overlap = np.sum(h_scaled[:overlap_len] * (1.0 - h_scaled[effective_k:effective_k + overlap_len])) * dx
                if overlap < max_c5:
                    h = h_scaled.copy()
                    updated = True
                    break
            alpha *= 0.5

        if updated:
            ones_minus_h_current = 1.0 - h
            corr_current = np.correlate(h, ones_minus_h_current, mode='full')
            current_c5 = np.max(corr_current * dx)
            if best_c5 is None or current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    current_sum = np.sum(best_h)
    if not np.isclose(current_sum, target_sum):
        best_h = best_h * target_sum / current_sum
        best_h = np.clip(best_h, 0.0, 1.0)

    def evaluate_c5(h_eval):
        ones_minus_h = 1.0 - h_eval
        corr = np.correlate(h_eval, ones_minus_h, mode='full')
        return np.max(corr * dx)

    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n_points
