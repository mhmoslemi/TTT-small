            term1 = (1 - h[pos]) if 0 <= pos < n_points else 0.0
            neg = j - k
            term2 = h[neg] if 0 <= neg < n_points else 0.0
            grad[j] = dx * (term1 - term2)

        alpha = 0.1
        h -= alpha * grad
        h = np.clip(h, 0, 1)

        h_proj = cp.Variable(n_points)
        constraints = [cp.sum(h_proj) == target_sum, h_proj >= 0, h_proj <= 1]
        prob = cp.Problem(cp.Minimize(cp.sum_squares(h_proj - h)), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        h[:] = h_proj.value

        current_c5 = compute_C5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h.copy()

    return best_h, best_c5, n_points
