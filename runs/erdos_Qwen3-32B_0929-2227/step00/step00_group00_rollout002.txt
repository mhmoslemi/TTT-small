import numpy as np
import time

def run(seed=42, budget_s=80, **kwargs):
    np.random.seed(seed & 0xFFFFFFFF)  # Ensure 32-bit seed
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    desired_sum = n_points / 2
    assert np.allclose(h.sum(), desired_sum), "Initial h is not feasible"

    best_h = h.copy()
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = np.max(overlaps)
    start_time = time.time()

    def compute_gradient(h, k):
        n = len(h)
        grad = np.zeros(n)
        if k >= 0 and k < n:
            valid_x = np.arange(n - k)
            grad[valid_x] += (1 - h[valid_x + k]) * dx
            grad[k:] -= h[valid_x] * dx
        elif k < 0 and k >= -n + 1:
            k_neg = -k
            valid_x = np.arange(k_neg, n)
            grad[valid_x] += (1 - h[valid_x - k_neg]) * dx
            upper = n - k_neg
            grad[:upper] -= h[k_neg:] * dx
        return grad

    lr = 0.1
    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        current_max = np.max(overlaps)
        m = np.argmax(overlaps)
        k = m - (n_points - 1)

        grad = compute_gradient(h, k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-8:
            grad = grad / grad_norm

        h -= lr * grad
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
