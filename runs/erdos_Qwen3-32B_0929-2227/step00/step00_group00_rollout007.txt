def project_h(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if np.isclose(current_sum, target_sum):
        return h_clipped
    delta = target_sum - current_sum
    if delta > 0:
        # Need to increase
        # Find positions where h_clipped < 1
        indices = h_clipped < 1
        available = 1 - h_clipped[indices]
        if np.sum(available) == 0:
            return h_clipped
        alpha = delta / np.sum(available)
        h_proj = h_clipped.copy()
        h_proj[indices] += alpha * available
    else:
        # Need to decrease
        delta = -delta
        indices = h_clipped > 0
        available = h_clipped[indices]
        if np.sum(available) == 0:
            return h_clipped
        alpha = delta / np.sum(available)
        h_proj = h_clipped.copy()
        h_proj[indices] -= alpha * available
    return h_proj
