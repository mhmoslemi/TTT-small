import time
import numpy as np

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    """Clip to [0,1] and enforce exact sum == n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    # iterative redistribution
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            add = diff / idx.size
            add = np.minimum(add, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            sub = (-diff) / idx.size
            sub = np.minimum(sub, h[idx])
            h[idx] -= sub
        diff = target - h.sum()
    # final safety clip
    return np.clip(h, 0.0, 1.0)

def random_walk(h, sigma, rng):
    """Add Gaussian noise and project back to feasible set."""
    h_new = h + rng.normal(scale=sigma, size=h.shape)
    return project_feasible(h_new, len(h))

def interpolate_solution(prev_h, new_n):
    """Linear interpolation from previous grid to a new grid size."""
    old_n = len(prev_h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, prev_h)

def optimize_grid(init_h, n, time_limit, rng):
    """Simple hill‑climbing / simulated‑annealing search on a fixed grid."""
    start = time.time()
    h = project_feasible(np.asarray(init_h, dtype=float), n)
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    # scale of perturbations; larger grids need finer steps
    sigma = 0.05 * (n / 100.0) ** 0.5
    while time.time() - start < time_limit:
        h_candidate = random_walk(h, sigma, rng)
        cand_score = compute_c5(h_candidate)

        # accept if better (deterministic hill climb)
        if cand_score < cur_score:
            h = h_candidate
            cur_score = cand_score

        # keep global best
        if cand_score < best_score:
            best_h = h_candidate.copy()
            best_score = cand_score
            # stop early if we beat target
            if best_score <= 0.3808:
                break

        # slowly shrink step size
        sigma *= 0.9995
        # avoid sigma becoming too tiny
        if sigma < 1e-6:
            sigma = 1e-6

    return best_h, best_score
