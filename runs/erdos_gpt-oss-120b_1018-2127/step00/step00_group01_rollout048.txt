import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def _compute_c5(h):
    """Return the overlap integral C5 for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _make_initial(old_h, new_n):
    """Interpolate old_h (defined on [0,2)) to a new grid of size new_n."""
    old_n = old_h.size
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    # enforce the exact integral constraint by rescaling
    new_h *= (new_n / 2) / new_h.sum()
    # clip back into [0,1] after rescaling
    np.clip(new_h, 0.0, 1.0, out=new_h)
    # renormalize again if clipping changed the sum
    new_h *= (new_n / 2) / new_h.sum()
    return new_h
