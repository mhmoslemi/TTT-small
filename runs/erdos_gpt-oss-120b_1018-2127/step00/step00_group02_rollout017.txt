import time
import numpy as np

def _compute_c5(h):
    """Return the C5 value for a discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_pair(h, i, j, delta):
    """Move mass delta from i to j respecting bounds [0,1]."""
    hi, hj = h[i], h[j]
    # feasible delta range
    lo = max(-hi, -(1 - hj))          # decreasing i, increasing j
    hi_bound = min(1 - hi, hj)       # increasing i, decreasing j
    # clamp delta
    delta = max(lo, min(delta, hi_bound))
    if delta == 0:
        return None
    h_new = h.copy()
    h_new[i] = hi - delta
    h_new[j] = hj + delta
    return h_new

def _initial_feasible(n, seed=0):
    """Create a random feasible vector of length n."""
    rng = np.random.default_rng(seed)
    # start with uniform mass
    h = rng.random(n)
    h = np.clip(h, 0, 1)
    # enforce sum = n/2 by scaling and then projecting
    target = n / 2.0
    # simple iterative scaling + clipping
    for _ in range(10):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        scale = target / s if s != 0 else 0
        h = np.clip(h * scale, 0, 1)
    # final adjustment of the tiny residual
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # add residual to a random index that stays within bounds
        idx = rng.integers(0, n)
        delta = np.clip(residual, -h[idx], 1 - h[idx])
        h[idx] += delta
    return h

    rng = np.random.default_rng(seed)

    # Determine discretisation size
    if 'initial_h_values' in globals():
        base_h = np.asarray(initial_h_values, dtype=float)
        base_n = base_h.shape[0]
        # upscale to a modestly larger grid for more flexibility, capped at 120 pts
        max_n = 120
        n = min(max_n, max(base_n, 80))
        if n != base_n:
            # linear interpolation onto new grid
            old_x = np.linspace(0, 2, base_n, endpoint=False)
            new_x = np.linspace(0, 2, n, endpoint=False)
            h_current = np.interp(new_x, old_x, base_h)
            # enforce bounds and exact integral
            h_current = np.clip(h_current, 0.0, 1.0)
            # project to exact sum = n/2
            target = n / 2.0
            cur_sum = h_current.sum()
            if cur_sum != 0:
                h_current *= target / cur_sum
            # final clipping & correction
            h_current = np.clip(h_current, 0.0, 1.0)
            residual = target - h_current.sum()
            if abs(residual) > 1e-12:
                idx = rng.integers(0, n)
