import time
import numpy as np
import random
from math import exp

def compute_c5(h):
    """Discrete C5 evaluation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def normalize_to_sum(h, target_sum, max_iter=1000):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # distribute diff uniformly, respecting bounds
        add = diff / h.size
        h += add
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-10:
        # distribute remaining error proportionally to free capacity
        free_up = 1.0 - h
        free_down = h
        up = free_up.sum()
        down = free_down.sum()
        if diff > 0 and up > 0:
            h += (free_up / up) * diff
        elif diff < 0 and down > 0:
            h += (free_down / down) * diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h, new_n):
    """Linear interpolation of step function onto a finer grid."""
    old_n = h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    new_h = np.interp(new_x, old_x, h, left=0.0, right=0.0)
    target_sum = new_n / 2.0
    return normalize_to_sum(new_h, target_sum)

def local_search(initial_h, budget, seed=42):
    """Simple stochastic local search on a fixed grid size."""

    np.random.seed(seed)
    n = initial_h.size
    target_sum = n / 2.0
    h = normalize_to_sum(initial_h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()
    max_move = 0.2  # maximal mass transferred in one move
    temp0 = 0.01
    decay = 0.9995

    it = 0
    while time.time() - start < budget:
        i, j = random.sample(range(n), 2)
        # allowable delta respecting bounds
        hi, hj = h[i], h[j]
        delta_max = min(max_move, hi, 1.0 - hj)
        if delta_max <= 0:

        delta = random.uniform(-delta_max, delta_max)
        # ensure we move mass from i to j (delta positive means i loses)

                continue
        else:

                continue

        h_new[j] += delta
        # bounds already respected by construction

            prob = exp(-(c5_new - best_c5) / (t + 1e-12))

        it += 1
    return best_h, best_c5
