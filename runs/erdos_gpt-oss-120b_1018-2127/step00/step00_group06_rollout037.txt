import time
import math
import numpy as np

def _objective(h):
    """Maximum overlap integral for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_bounds_sum(x, target_sum):
    """
    Project vector x onto the box [0,1] while enforcing sum(x) = target_sum.
    Simple water-filling iterative scheme.
    """
    lower, upper = 0.0, 1.0
    x = np.clip(x, lower, upper)
    # iterative redistribution
    for _ in range(10):
        total = x.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        x[free] += delta
        x = np.clip(x, lower, upper)
    # final fix for any tiny residual using a simple scaling if needed
    residual = target_sum - x.sum()
    if abs(residual) > 1e-12:
        # distribute the residual uniformly over non‑saturated entries
        free = (x > lower) & (x < upper)
        n_free = free.sum()
        if n_free > 0:
            x[free] += residual / n_free
            x = np.clip(x, lower, upper)
    return x

def _random_walk(h, rng, scale):
    """Perturb h with Gaussian noise of given scale and repair constraints."""
    h_new = h + rng.normal(scale=scale, size=h.shape)
    target = h.shape[0] / 2.0
    h_new = _project_bounds_sum(h_new, target)
    return h_new

def _interpolate(h, new_n):
    """Linear interpolation of a periodic step function from len(h) to new_n points."""
    old_n = len(h)
    # use points on [0,2) for interpolation
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def _optimize_on_grid(initial_h, time_limit, rng):
    """
    Hill‑climbing / simulated‑annealing search on a fixed grid size.
    Returns best vector found within the allotted wall‑clock time.
    """
    n = len(initial_h)
    target_sum = n / 2.0
    best_h = _project_bounds_sum(initial_h.copy(), target_sum)
    best_score = _objective(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    start = time.time()
    iter_cnt = 0
    temperature = 1.0
    # scale proportional to grid size; adapt later
    base_scale = 0.05

    while time.time() - start < time_limit:
        # annealing schedule for scale
        scale = base_scale * (1.0 / (1.0 + iter_cnt * 1e-4))
        h_candidate = _random_walk(cur_h, rng, scale)
        cand_score = _objective(h_candidate)

            cur_h, cur_score = h_candidate, cand_score

        if cand_score < best_score:
            best_h, best_score = h_candidate.copy(), cand_score

        temperature *= 0.9995
        iter_cnt += 1

    return best_h, best_score
