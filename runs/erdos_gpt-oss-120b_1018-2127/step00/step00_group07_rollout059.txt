import time, math, random
import numpy as np

def _compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _feasible_random(n, rng):
    """Generate a random feasible vector of length n (values in [0,1], sum=n/2)."""
    # start with uniform random numbers, then project onto the simplex with box constraints
    x = rng.random(n)
    # scale to required sum
    x *= (n / 2) / x.sum()
    # clip to [0,1] and renormalize if needed (iterative projection)
    for _ in range(5):
        x = np.clip(x, 0.0, 1.0)
        diff = (n / 2) - x.sum()
        if abs(diff) < 1e-12:
            break
        # distribute the difference uniformly over non‑boundary entries
        mask = (x > 0.0) & (x < 1.0)
        if not np.any(mask):
            break
        x[mask] += diff / np.count_nonzero(mask)
    # final clip to guarantee feasibility
    x = np.clip(x, 0.0, 1.0)
    # enforce exact sum by a tiny proportional adjustment
    s = x.sum()
    if s > 0:
        x *= (n / 2) / s
    return x

def _perturb(h, rng):
    """Transfer a small amount of mass between two coordinates, preserving sum."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # amount to move, up to 0.2 but limited by bounds
    eps = rng.random() * 0.2
    eps = min(eps, h[i])           # cannot go below 0 at i
    eps = min(eps, 1.0 - h[j])     # cannot exceed 1 at j
    if eps <= 0:
        return h.copy()
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

    """Return (h_values, c5_bound, n_points) for a low‑overlap step function."""

    rng = np.random.default_rng(seed)

    best_h = h0.copy()
