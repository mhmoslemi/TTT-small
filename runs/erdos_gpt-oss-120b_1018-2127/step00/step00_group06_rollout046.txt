import time, math, numpy as np

def _objective(h, dx):
    # max correlation with 1-h, multiplied by dx
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _repair(h, target_sum, max_iter=20):
    """Project h onto [0,1] and enforce exact sum = target_sum."""
    for _ in range(max_iter):
        np.clip(h, 0.0, 1.0, out=h)
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # distribute equally; after adding, clipping may change sum, so iterate
        h += diff / h.size
    # final clip to be safe
    np.clip(h, 0.0, 1.0, out=h)
    # tiny residual correction on a non‑boundary element if needed
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # find an index not at a bound
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:
            idx = free[0]
            h[idx] += diff
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _random_neighbor(h, scale, rng):
    """Create a perturbed neighbor while conserving the sum (before repair)."""
    n = h.size
    # pick two distinct indices
    i, j = rng.choice(n, 2, replace=False)
    delta = rng.normal(scale=scale)
    # transfer delta from i to j
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
