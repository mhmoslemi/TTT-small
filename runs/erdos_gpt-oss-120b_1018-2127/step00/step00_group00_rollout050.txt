import time
import numpy as np
from scipy.optimize import minimize

def _compute_score(h):
    """Return C5 for step vector h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _is_feasible(h):
    """Loose feasibility test used before optimization."""
    if np.any(h < 0) or np.any(h > 1):
        return False
    # sum should equal n/2 (integral 1)
    target = h.size / 2.0
    return abs(h.sum() - target) < 1e-2


def _resample(old_h, new_n):
    """Linear interpolation of old_h (defined on [0,2]) to length new_n."""
    old_n = old_h.size
    # grid points at cell mid‑points
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0 / new_n
    new_h = np.interp(new_x, old_x, old_h, left=old_h[0], right=old_h[-1])
    # clip to admissible interval
    np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h


def _optimize_start(start_h, seed):
    """Run a constrained SLSQP optimisation starting from start_h."""
    n = start_h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - n / 2.0}
    options = {"maxiter": 500, "ftol": 1e-6, "disp": False}
    res = minimize(
        _compute_score,
        start_h,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options=options,
    )
    return res
