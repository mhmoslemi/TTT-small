def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random

    np.random.seed(seed)
    random.seed(seed)

    # ---- initial vector -------------------------------------------------
    try:
        init = initial_h_values  # provided by the harness
        n_points = int(len(init))
        h_current = np.array(init, dtype=float)
    except Exception:
        n_points = 70
        h_current = np.random.rand(n_points)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ---- projection onto feasible set -----------------------------------
    def repair(vec):
        vec = np.clip(vec, 0.0, 1.0)
        for _ in range(20):          # water‑filling style projection
            s = vec.sum()
            diff = target_sum - s
            if abs(diff) < 1e-12:
                break
            if diff > 0:              # need to increase total mass
                idx = vec < 1.0
                if not idx.any():
                    break
                inc = diff / idx.sum()
                vec[idx] = np.clip(vec[idx] + inc, 0.0, 1.0)
            else:                     # need to decrease total mass
                idx = vec > 0.0
                if not idx.any():
                    break
                dec = (-diff) / idx.sum()
                vec[idx] = np.clip(vec[idx] - dec, 0.0, 1.0)
        return vec

    h_current = repair(h_current)

    # ---- objective -------------------------------------------------------
    def evaluate(vec):
        corr = np.correlate(vec, 1.0 - vec, mode="full")
        return np.max(corr) * dx

    cur_score = evaluate(h_current)
    best_h = h_current.copy()
    best_score = cur_score

    # ---- simulated annealing parameters ---------------------------------
    start_time = time.time()
    temp = 0.12                # initial temperature
    sigma = 0.07               # initial noise scale
    decay = 0.99993            # per‑iteration decay
    restart_interval = 8000   # iterations before a random restart

    it = 0
    while True:
        if time.time() - start_time > budget_s - 0.5:
            break

        # propose a new candidate
        proposal = h_current + np.random.randn(n_points) * sigma
        proposal = repair(proposal)
        prop_score = evaluate(proposal)

        # keep global best
        if prop_score < best_score:
            best_score = prop_score
            best_h = proposal.copy()

        # Metropolis acceptance
        if prop_score < cur_score or np.random.rand() < math.exp(-(prop_score - cur_score) / max(temp, 1e-15)):
            h_current = proposal
            cur_score = prop_score

        # schedule updates
        temp *= decay
        sigma *= decay
        it += 1

        # occasional random restart to escape stagnation
        if it % restart_interval == 0:
            rnd = np.random.rand(n_points)
            h_current = repair(rnd)
            cur_score = evaluate(h_current)

    return best_h, best_score, n_points
