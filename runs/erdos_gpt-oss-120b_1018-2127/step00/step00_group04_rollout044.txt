import numpy as np
import time

def _compute_c5(h, dx):
    # correlation of h with (1-h); mode='full' gives length 2*n-1
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _interpolate_initial(initial, target_n):
    # linear interpolation to target size preserving integral
    src_n = len(initial)
    x_src = np.linspace(0.0, 2.0, src_n, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_tgt = np.interp(x_tgt, x_src, initial)
    # renormalize to exact integral 1
    dx = 2.0 / target_n
    total = h_tgt.sum() * dx
    if total != 0:
        h_tgt *= (1.0 / total)
    h_tgt = np.clip(h_tgt, 0.0, 1.0)
    return h_tgt
