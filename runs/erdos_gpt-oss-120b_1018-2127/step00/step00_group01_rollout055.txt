import numpy as np
from scipy.optimize import differential_evolution

def _c5_of_h(h):
    """Compute C5 for a discrete step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation h * (1-h shifted)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_onto_constraints(h, target_sum, max_iter=10):
    """
    Clip to [0,1] and enforce sum == target_sum by simple iterative scaling.

    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        scale = target_sum / s if s != 0 else 0.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)

    diff = target_sum - h.sum()
