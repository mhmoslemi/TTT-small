def _evaluate_c5(h):
    """Return the overlap integral C5 for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation with zero padding (mode='full')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
