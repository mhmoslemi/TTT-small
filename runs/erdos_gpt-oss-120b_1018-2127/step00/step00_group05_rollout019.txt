import time
import numpy as np

def _dx(n):
    return 2.0 / n

def _evaluate(h):
    n = h.shape[0]
    dx = _dx(n)
    # full convolution length = 2*n-1, shift index - (n-1) corresponds to lag k
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_box_sum(v, target):
    """
    Project vector v onto the box [0,1] while forcing sum(v)=target.
    Uses a simple iterative redistribution; converges quickly for our use.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # indices that can receive/take mass
        if diff > 0:
            free = (v < 1.0)
        else:
            free = (v > 0.0)
        count = free.sum()
        if count == 0:
            break
        delta = diff / count
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final scaling if still off (rare)
    s = v.sum()
    if s != 0:
        v *= target / s
        v = np.clip(v, 0.0, 1.0)
    return v

def _make_initial(n_points):
    """Create a feasible start vector. Use the provided initial_h_values if present."""
    target = n_points / 2.0
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        m = init.shape[0]
        # interpolate to new size
        src_x = np.linspace(0, 2, m, endpoint=False)
        dst_x = np.linspace(0, 2, n_points, endpoint=False)
        h = np.interp(dst_x, src_x, init)
    else:
        # uniform start (average value = 0.5)
        h = np.full(n_points, 0.5, dtype=float)
    h = _project_box_sum(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimise the
    Erdős overlap constant C5.

    Returns:
        h_values (np.ndarray): optimal values, length n_points
        c5_bound (float): the achieved C5 value
        n_points (int): discretisation size used
    """
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s * 0.95  # safety margin

    # choose discretisation (even number, not too large)
    n_points = int(kwargs.get('n_points', 140))
    if n_points % 2:
        n_points += 1
    target_sum = n_points / 2.0

    # initialise
    h_cur = _make_initial(n_points)
    c5_cur = _evaluate(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur

    # parameters for stochastic search
    base_eps = 0.2 * target_sum / n_points   # typical perturbation magnitude
    it = 0
    while time.time() - start_time < max_time:
        it += 1
        # annealing style decay
        eps = base_eps * (0.995 ** it)

        # propose a random perturbation with zero-mean to keep sum near target
        delta = (np.random.rand(n_points) - 0.5) * eps
        h_prop = h_cur + delta
        h_prop = _project_box_sum(h_prop, target_sum)
