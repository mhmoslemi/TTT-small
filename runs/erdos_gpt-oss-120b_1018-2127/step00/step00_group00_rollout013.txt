import time
import numpy as np
from scipy import optimize

def _compute_c5(h):
    """Maximum overlap integral for discrete step function h."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _project(h, target_sum):
    """Clip to [0,1] and adjust to meet the target sum."""
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative redistribution
    for _ in range(10):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0 + 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(free):
            # distribute uniformly respecting bounds
            free = np.ones_like(h, dtype=bool)
        delta = diff / free.sum()
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction if still off (rare)
    if abs(h.sum() - target_sum) > 1e-9:
        scale = target_sum / h.sum()
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    return h

def _resize_h(old_h, new_N):
    """Resize old_h to new_N points keeping the integral constraint."""
    old_N = old_h.size
    old_x = np.linspace(0.0, 2.0, old_N, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_N, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    target = new_N / 2.0
    return _project(new_h, target)

def _optimize_h(init_h, max_iter=200):
    """Run SLSQP to minimize C5 with bounds and sum constraint."""
    N = init_h.size
    target_sum = N / 2.0
    bounds = [(0.0, 1.0)] * N
    cons = {"type": "eq", "fun": lambda h: h.sum() - target_sum}

    def obj(h):
        return _compute_c5(h)

    res = optimize.minimize(
        obj,
        init_h,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": max_iter, "ftol": 1e-10, "disp": False},
    )
    h_opt = _project(res.x, target_sum)
    return h_opt, _compute_c5(h_opt)
