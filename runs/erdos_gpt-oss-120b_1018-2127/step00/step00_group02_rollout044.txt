import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return discrete C5 for step values h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(x, n):
    """Clamp to [0,1] and enforce sum == n/2 by a simple affine shift."""
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    cur = x.sum()
    if cur != target:
        # shift uniformly, then re‑clamp
        shift = (target - cur) / n
        x = x + shift
        # after shift some entries may violate bounds – fix them iteratively
        for _ in range(5):
            low = x < 0.0
            high = x > 1.0
            if not low.any() and not high.any():
                break
            excess = x[low].sum() - 0.0 * low.sum()
            deficit = 1.0 * high.sum() - x[high].sum()
            # redistribute excess/deficit proportionally
            if low.any():
                x[low] = 0.0
            if high.any():
                x[high] = 1.0
            cur = x.sum()
            shift = (target - cur) / (n - low.sum() - high.sum())
            x[~low & ~high] += shift
    return np.clip(x, 0.0, 1.0)

def optimize_at_resolution(n, init_h, time_deadline):
    """Run a series of SLSQP attempts until time runs out."""
    best_h = init_h.copy()
    best_val = compute_c5(best_h)

    # optimizer settings

    while time.time() < time_deadline:
