import time
import numpy as np

# Helpers ---------------------------------------------------------------

def ensure_feasibility(h):
    """Project h onto the box [0,1] and enforce sum(h)=len(h)/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 0:
        h = h * (target / s)
    # a second clipping to guard against floating drift
    h = np.clip(h, 0.0, 1.0)
    # renormalize if clipping changed the sum
    s = h.sum()
    if s > 0:
        h = h * (target / s)
    return h

def compute_c5(h):
    """Return the overlap constant C5 for a discretised step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_candidate(base, scale):
    """Perturb base vector with Gaussian noise of given scale and re‑project."""
    cand = base + np.random.normal(scale=scale, size=base.shape)
    return ensure_feasibility(cand)

# Main optimization -----------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h=1 that minimizes
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.

    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise
    if 'initial_h_values' in globals():
        h = ensure_feasibility(np.array(initial_h_values, copy=True))
    else:
        n = 70
        h = np.full(n, 0.5)                 # sum = n/2 already satisfies ∫h=1
        h = ensure_feasibility(h)

    n = h.size
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # parameters for the stochastic search
    max_iter = 500000                     # safety cap
    iter_count = 0
    scale = 0.2                           # initial perturbation size
    scale_decay = 0.9995                  # exponential decay per iteration
    temp = 0.01                           # tiny temperature for occasional uphill moves

    while True:
        now = time.time()
        if now - start_time >= budget_s:
            break
        if iter_count >= max_iter:
            break

        # generate a candidate
        cand = random_candidate(best_h, scale)

        # evaluate
        score = compute_c5(cand)

        # accept if better, else maybe accept uphill with tiny prob
        if score < best_score or np.random.rand() < np.exp(-(score - best_score) / max(temp, 1e-9)):
            best_h = cand
            best_score = score

        # occasional random restart to escape flat regions
        if iter_count % 20000 == 0 and iter_count > 0:
            # start from a fresh random feasible point
            rnd = np.random.rand(n)
            rnd = ensure_feasibility(rnd)
            if compute_c5(rnd) < best_score:
                best_h = rnd
                best_score = compute_c5(rnd)

        # anneal parameters
        scale = max(scale * scale_decay, 1e-4)
        iter_count += 1

    # Ensure final output respects constraints exactly
    best_h = ensure_feasibility(best_h)
    final_score = compute_c5(best_h)

    return best_h, final_score, n
