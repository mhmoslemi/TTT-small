import time, math, random
import numpy as np

def _project_box_sum(v, target, eps=1e-12, max_iter=50):
    """Project vector v onto [0,1]^n with exact sum = target using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < eps:
            break
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _c5_value(h):
    """Maximum overlap integral for step function h."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx

def _anneal(start_h, target_sum, time_limit):
    """Simple simulated‑annealing optimiser for the C5 objective."""
    n = start_h.size
    cur = start_h.copy()
    cur_val = _c5_value(cur)
    best = cur.copy()
    best_val = cur_val
    end = time.time() + time_limit
    T = 1.0
    it = 0
    while time.time() < end:
        # decreasing perturbation magnitude
        sigma = 0.15 * (1.0 - it / 20000.0)
        prop = cur + sigma * np.random.randn(n)
        prop = _project_box_sum(prop, target_sum)
        val = _c5_value(prop)
        if val < cur_val or random.random() < math.exp(-(val - cur_val) / max(T, 1e-12)):
            cur, cur_val = prop, val
            if val < best_val:
                best, best_val = prop.copy(), val
        # exponential cooling
        T *= 0.9995
        it += 1
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C5 constant."""
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    # Determine initial vector if provided
    try:
        init = np.asarray(initial_h_values, dtype=float)
        init_n = init.size
    except Exception:
        init = None
        init_n = 0

    # Resolution schedule (must stay <1000 points)
    sizes = [95, 190, 380]
    # Rough time split among stages
    time_splits = [0.20, 0.30, 0.50]

    best_h = None
    best_val = float('inf')

    for size, portion in zip(sizes, time_splits):
        remaining = budget_s - (time.time() - start_time)
        if remaining <= 0.0:
            break
        stage_budget = min(remaining * portion, remaining)

        target_sum = size / 2.0

        if best_h is None:
            # First stage: use supplied construction or random start
            if init is not None and init_n == size:
                h0 = init.copy()
            else:
                h0 = _project_box_sum(np.random.rand(size), target_sum)
        else:
            # Upscale previous best by integer factor if possible
            if size % best_h.size == 0:
                factor = size // best_h.size
                h0 = np.repeat(best_h, factor)
                h0 = _project_box_sum(h0, target_sum)
            else:
                # Otherwise interpolate linearly
                x_old = np.linspace(0, 2, best_h.size, endpoint=False)
                x_new = np.linspace(0, 2, size, endpoint=False)
                h0 = np.interp(x_new, x_old, best_h)
                h0 = _project_box_sum(h0, target_sum)

        # Run annealing on this grid
        h_opt, val_opt = _anneal(h0, target_sum, stage_budget)

        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

        # If time is nearly exhausted, break early
        if time.time() - start_time > budget_s * 0.99:
            break

    # Final evaluation (ensure constraints are met)
    n_final = best_h.size
    best_h = _project_box_sum(best_h, n_final / 2.0)
    final_val = _c5_value(best_h)

    return best_h, final_val, n_final
