import numpy as np
import time
import scipy.optimize as opt

def _smooth_max(values, alpha=50.0):
    """Smooth approximation of max using log-sum-exp."""
    z = alpha * values
    z_max = np.max(z)
    # stable log-sum-exp
    return (z_max + np.log(np.sum(np.exp(z - z_max)))) / alpha

def _objective(h, dx):
    """Objective: smooth max of the overlap integral."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return _smooth_max(corr)

def _evaluate_c5(h, dx):
    """Exact C5 value (max overlap) for a given h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

        Number of discretization points used.
    """

    np.random.seed(seed)

    # ------------------------------------------------------------------
