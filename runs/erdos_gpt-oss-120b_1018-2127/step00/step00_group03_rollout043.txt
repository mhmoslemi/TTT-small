import time
import numpy as np

def _project_h(h, target_sum):
    """Project vector h onto the box [0,1] and the affine sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        if diff > 0:
            h[mask] = np.minimum(1.0, h[mask] + delta)
        else:
            h[mask] = np.maximum(0.0, h[mask] - delta)
    return h

def _c5_of_h(h):
    """Return the C5 value for a discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _initial_h(n):
    """Generate an initial feasible h with length n."""
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.asarray(initial_h_values, dtype=np.float32)
        if init.size != n:
            # linear interpolation to required size
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
    else:
        # simple half‑interval block
        init = np.zeros(n, dtype=np.float32)
        init[: n // 2] = 1.0
    init = _project_h(init, target_sum=n / 2.0)
    return init

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a low C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Optimised values of h on a uniform grid.
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a moderate grid size (70 is the provided size, 200 gives more freedom)
    base_n = len(initial_h_values) if 'initial_h_values' in globals() and initial_h_values is not None else 70
    n = max(base_n, 200)                     # ensure at least 200 points for flexibility
    target_sum = n / 2.0

    h = _initial_h(n)
    best_h = h.copy()
    best_score = _c5_of_h(h)

    # Simulated‑annealing parameters
    max_iters = 50000
    temp_start = 0.05
    temp_end = 5e-4
    for it in range(max_iters):
        # time‑budget check
        if time.time() - start_time > budget_s * 0.95:
            break

        # linear temperature schedule
        t = it / max_iters
        temp = temp_start * (1 - t) + temp_end * t

        # propose a new candidate by perturbing a random subset
        h_prop = h.copy()
        # number of coordinates to modify (adapt with temperature)
        k = max(1, int(0.02 * n * (1 - t) + 1))
        idx = rng.choice(n, size=k, replace=False)
        # Gaussian perturbation
        perturb = rng.normal(loc=0.0, scale=0.1 * temp, size=k)
        h_prop[idx] += perturb
        h_prop = _project_h(h_prop, target_sum)

        cand_score = _c5_of_h(h_prop)

        # Metropolis acceptance
        if cand_score < best_score:
            best_h, best_score = h_prop, cand_score
        if cand_score <= _c5_of_h(h) or rng.random() < np.exp((_c5_of_h(h) - cand_score) / (temp + 1e-12)):
            h = h_prop

    # Final polishing: a few deterministic gradient‑free refinements
    for _ in range(200):
        if time.time() - start_time > budget_s * 0.97:
            break
        # small local swaps between two points
        i, j = rng.choice(n, 2, replace=False)
        h_swap = h.copy()
        h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
        h_swap = _project_h(h_swap, target_sum)
        swap_score = _c5_of_h(h_swap)
        if swap_score < best_score:
            best_h, best_score = h_swap, swap_score
            h = h_swap

    return best_h, best_score, n
