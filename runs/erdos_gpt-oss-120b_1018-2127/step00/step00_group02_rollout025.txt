import numpy as np
import time

def _project_onto_feasible(y, target_sum, max_iter=60, tol=1e-9):
    """
    Project vector y onto the set {x | 0<=x<=1 , sum(x)=target_sum}.
    Uses bisection on the Lagrange multiplier lambda.
    """
    lo = -np.max(y)          # ensures y+lambda <= 0 for all -> clip to 0
    hi = 1.0 - np.min(y)     # ensures y+lambda >= 1 for all -> clip to 1
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        # monotone: increasing lambda raises the sum
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def _c5_and_shift(h, dx):
    """Return (C5 value, shift index) for discrete h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = int(np.argmax(corr))
    return corr[idx], idx - (len(h) - 1)

def _c5_gradient(h, shift):
    """
    Subgradient of max correlation wrt h when the maximizing shift is `shift`.
    Handles zero padding outside the domain.
    """
    n = len(h)
    grad = np.zeros_like(h)
    if shift >= 0:
        # term (1 - h[i+shift])
        grad[:n - shift] += 1.0 - h[shift:]
        # term -h[i-shift]
        grad[shift:] += -h[:n - shift]
    else:
        s = -shift
        grad[s:] += 1.0 - h[:n - s]
        grad[:n - s] += -h[s:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimise the
    Erdős C5 overlap constant.

    Returns
    -------
    h_values : np.ndarray
        Optimised values in [0,1] with sum = n_points/2.
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s - 0.5  # small safety margin

    # Determine problem size
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.size
    else:
        n = 100  # fallback size
        h = rng.random(n)
        target = n / 2.0
        h = _project_onto_feasible(h, target)

    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = h.copy()
    best_c5, best_shift = _c5_and_shift(best_h, dx)

    it = 0
    step0 = 0.4
    while time.time() - start_time < time_limit:
        # ---- gradient step ----
        c5, shift = _c5_and_shift(h, dx)
        grad = _c5_gradient(h, shift)

        # diminishing step size
        step = step0 / (1.0 + it * 0.001)

        y = h - step * grad
        h_new = _project_onto_feasible(y, target_sum)

        # evaluate
        c5_new, _ = _c5_and_shift(h_new, dx)

        # accept if improvement, else keep old
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            shift = shift  # unchanged variable
        else:
            # occasional exploratory perturbation
            if rng.random() < 0.02:
                pert = rng.normal(scale=0.05, size=n)
                h_pert = _project_onto_feasible(h + pert, target_sum)
                c5_pert, _ = _c5_and_shift(h_pert, dx)
                if c5_pert < c5:
                    h = h_pert
                    c5 = c5_pert

        # keep global best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # restart with a fresh random feasible point every few hundred iterations
        if it % 500 == 0 and it > 0:
            rnd = rng.random(n)
            h = _project_onto_feasible(rnd, target_sum)

        it += 1

    # final evaluation on best solution
    final_c5, _ = _c5_and_shift(best_h, dx)
    return best_h, float(final_c5), n
