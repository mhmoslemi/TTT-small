import time
import numpy as np
from scipy.optimize import minimize

# global variable used by objective & constraint (required to avoid closures)
_opt_n = None

def compute_score_global(x):
    """Return max overlap integral for step function represented by x."""
    n = _opt_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return np.max(corr) * dx

def eq_sum_constraint(x):
    """Equality constraint: integral equals 1 (sum = n/2)."""
    return np.sum(x) - _opt_n / 2.0

def correlation_score(x):
    """Convenient wrapper used when n is known (e.g., for quick checks)."""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return np.max(corr) * dx

def resample_h(old_h, new_n):
    """Linear interpolation of a step function sampled on [0,2)."""
    old_n = len(old_h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # enforce feasibility (bounds and integral)
    new_h = np.clip(new_h, 0.0, 1.0)
    total = new_h.sum()
    if total > 0:
        new_h *= (new_n / 2.0) / total
    new_h = np.clip(new_h, 0.0, 1.0)
    # final renormalisation (small adjustments keep it feasible)
    deficit = (new_n / 2.0) - new_h.sum()
    if abs(deficit) > 1e-12:
        # distribute deficit uniformly while staying in bounds
        free = np.where((new_h > 0) & (new_h < 1))[0]
        if free.size > 0:
            add = deficit / free.size
            new_h[free] += add
            new_h = np.clip(new_h, 0.0, 1.0)
        # final scaling if still off
        total = new_h.sum()
        if total > 0:
            new_h *= (new_n / 2.0) / total
    return new_h

def make_initial(n):
    """Construct a feasible starting vector of length n."""
    if 'initial_h_values' in globals():
        base = initial_h_values
        if len(base) == n:
            h = base.copy()
        else:
            h = resample_h(base, n)
    else:
        # simple half‑ones, half‑zeros pattern
        h = np.zeros(n)
        half = n // 2
        h[:half] = 1.0
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total > 0:
        h *= (n / 2.0) / total
    h = np.clip(h, 0.0, 1.0)
    return h

def optimize_h(h0, n, seed):
    """Run a local SLSQP optimisation starting from h0."""
    global _opt_n
    _opt_n = n
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': eq_sum_constraint}
    # SLSQP is robust for small problems; limit iterations for time safety
    res = minimize(
        compute_score_global,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    )
    return res

def random_feasible(n):
    """Generate a random feasible vector (uniform in [0,1] then scaled)."""
    rnd = np.random.rand(n)
    rnd = np.clip(rnd, 0.0, 1.0)
    total = rnd.sum()
    if total > 0:
        rnd *= (n / 2.0) / total
    rnd = np.clip(rnd, 0.0, 1.0)
    return rnd

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    np.random.seed(seed)
    start = time.time()
    # list of resolutions to explore (coarse → fine)
    n_list = [45, 60, 90, 180, 300]
    best_h = None
    best_score = np.inf
    best_n = None

    for n in n_list:
        # time check before heavy work
        if time.time() - start > budget_s:
            break

        # start from a transferred solution; also try a random restart
        h_start = make_initial(n)
        candidates = [h_start, random_feasible(n)]

        for h0 in candidates:
            # ensure we don't exceed budget inside the inner loop
            if time.time() - start > budget_s:
                break

            # quick feasibility guard
            if np.any(h0 < -1e-12) or np.any(h0 > 1 + 1e-12):
                continue
            if not np.isclose(h0.sum(), n / 2.0, atol=1e-6):
                continue

            res = optimize_h(h0, n, seed)
            if not res.success:
                continue
            h_opt = np.clip(res.x, 0.0, 1.0)  # numerical safety
            score = correlation_score(h_opt)

            if score < best_score - 1e-9:
                best_score = float(score)
                best_h = h_opt.copy()
                best_n = n

        # optionally stop early if very good bound reached
        if best_score < 0.3808:
            break

    # fallback: if nothing succeeded, use the provided initial vector
    if best_h is None:
        if 'initial_h_values' in globals():
            best_h = initial_h_values.copy()
            best_n = len(best_h)
            best_score = correlation_score(best_h)
        else:
            # construct a trivial feasible vector
            best_n = 45
            best_h = make_initial(best_n)
            best_score = correlation_score(best_h)

    return (best_h, best_score, best_n)
