import time
import numpy as np

def _project_bounds_sum(x, target):
    """Project vector x onto [0,1] bounds with sum equal to target."""
    x = np.clip(x, 0.0, 1.0)
    # Iteratively fix sum while respecting bounds
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # cannot adjust further
            break
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    return x

def _evaluate_c5(h, dx):
    """Compute the C5 value for a given discrete h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (corr.max()) * dx

def _random_feasible(n, rng):
    """Generate a random feasible h (bounds [0,1], sum = n/2)."""
    x = rng.random(n)
    target = n / 2.0
    return _project_bounds_sum(x, target)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap constant.

    Returns:
        h_values (numpy.ndarray): optimized samples of h
        c5_bound (float): corresponding C5 value
        n_points (int): number of samples used (len(h_values))
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s * 0.95  # keep a safety margin

    # Determine problem size
    try:
        n = int(len(initial_h_values))
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        n = 45
        h0 = _random_feasible(n, rng)

    dx = 2.0 / n
    target_sum = n / 2.0

    # Initial best
    best_h = _project_bounds_sum(h0, target_sum)
    best_c5 = _evaluate_c5(best_h, dx)

    # Parameters for the stochastic search
    steps = [0.4, 0.2, 0.1, 0.05, 0.02]  # decreasing perturbation amplitudes
    max_iter_per_step = 2000

    # Main search loop
    for step in steps:
        iter_count = 0
        while time.time() - start_time < time_limit and iter_count < max_iter_per_step:
            # Propose a new candidate
            delta = rng.normal(scale=step, size=n)
            delta -= delta.mean()                     # enforce zero sum perturbation
            cand = best_h + delta
            cand = _project_bounds_sum(cand, target_sum)
            cand_c5 = _evaluate_c5(cand, dx)

            # Accept if improvement
            if cand_c5 < best_c5:
                best_h = cand
                best_c5 = cand_c5
                # reset iteration counter for this step size to explore locally
                iter_count = 0
            else:
                iter_count += 1

        # occasional random restart to escape local minima
        if time.time() - start_time < time_limit:
            rand_h = _random_feasible(n, rng)
            rand_c5 = _evaluate_c5(rand_h, dx)
            if rand_c5 < best_c5:
                best_h = rand_h
                best_c5 = rand_c5

    # Final projection to guarantee feasibility (numerical safety)
    best_h = _project_bounds_sum(best_h, target_sum)
    best_c5 = _evaluate_c5(best_h, dx)

    return best_h, float(best_c5), n
