import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine grid size
    # ------------------------------------------------------------------
    base_h = globals().get("initial_h_values", None)
    if base_h is not None:
        base_n = len(base_h)
    else:
        base_n = 70
    # Target size: try a modest increase for more flexibility but keep cost low
    n_points = int(kwargs.get("n_points", min(140, max(70, base_n * 2))))
    dx = 2.0 / n_points
    target_sum = n_points / 2.0            # because sum(h)*dx == 1

    # ------------------------------------------------------------------
    # Helper: enforce bounds and exact integral
    # ------------------------------------------------------------------
    def repair(vec):
        h = np.clip(vec, 0.0, 1.0)

        # iterative rescaling to hit the target sum while staying in [0,1]
        for _ in range(5):
            s = target_sum / h.sum()
            h = np.clip(h * s, 0.0, 1.0)

        # final tiny adjustment on a single coordinate if needed
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            # choose an index that can absorb the diff without leaving [0,1]
            idx = rng.integers(0, n_points)
            allowed = (1.0 - h[idx]) if diff > 0 else -h[idx]
            delta = np.clip(diff, -allowed, allowed)
            h[idx] += delta

        # one more global scaling if rounding errors remain
        if abs(h.sum() - target_sum) > 1e-12:
            h = np.clip(h * (target_sum / h.sum()), 0.0, 1.0)

        return h

    # ------------------------------------------------------------------
    # Objective evaluation
    # ------------------------------------------------------------------
    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    # ------------------------------------------------------------------
    # Initial solution (use provided construction if any)
    # ------------------------------------------------------------------
    if base_h is not None:
        # Upsample / downsample to desired n_points via linear interpolation
        x_old = np.linspace(0.0, 2.0, len(base_h), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_old, base_h)
    else:
        h0 = rng.random(n_points)

    best_h = repair(h0)
    best_score = evaluate(best_h)

    # ------------------------------------------------------------------
    # Simple population based hill‑climbing
    # ------------------------------------------------------------------
    pop_size = 12
    population = [best_h] + [repair(rng.random(n_points)) for _ in range(pop_size - 1)]
    scores = [evaluate(ind) for ind in population]

    start_time = time.time()
    iteration = 0
    sigma = 0.05                       # initial mutation magnitude

    while time.time() - start_time < budget_s - 0.5:
        iteration += 1
        # gradually shrink sigma
        sigma = max(0.005, sigma * 0.995)

        new_pop = []
        new_scores = []

        for ind, sc in zip(population, scores):
            # propose a mutated copy
            mutant = ind + rng.normal(scale=sigma, size=n_points)
            mutant = repair(mutant)
            mscore = evaluate(mutant)

            # keep the better of parent / mutant
            if mscore < sc:
                new_pop.append(mutant)
                new_scores.append(mscore)
                if mscore < best_score:
                    best_h, best_score = mutant.copy(), mscore
            else:
                new_pop.append(ind)
                new_scores.append(sc)

        # keep only the top half of the pool (elitism)
        combined = list(zip(new_scores, new_pop))
        combined.sort(key=lambda x: x[0])
        keep = pop_size // 2
        population = [c[1] for c in combined[:keep]]
        scores = [c[0] for c in combined[:keep]]

        # refill to full size with fresh random individuals
        while len(population) < pop_size:
            cand = repair(rng.random(n_points))
            population.append(cand)
            scores.append(evaluate(cand))

        # early exit if we already beat the target bound comfortably
        if best_score < 0.350:
            break

    return best_h, float(best_score), n_points
