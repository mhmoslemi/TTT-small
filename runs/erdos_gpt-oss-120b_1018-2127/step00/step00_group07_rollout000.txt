import time, random, math
import numpy as np

def _repair_h(h):
    """Clip to [0,1] and enforce sum(h)=len(h)/2 by iterative scaling."""
    n = h.shape[0]
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        s = h.sum()
        if s == 0:
            h[:] = target / n
            break
        if abs(s - target) < 1e-12:
            break
        factor = target / s
        h = np.clip(h * factor, 0.0, 1.0)
    return h

def _c5_score(h):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _anneal(start_h, time_limit):
    """Simple simulated‑annealing optimizer respecting the time budget."""
    rnd = random.Random(0)          # deterministic internal RNG
    np_rng = np.random.default_rng(0)

    h_cur = start_h.copy()
    best_h = h_cur.copy()
    best_score = _c5_score(best_h)

    n = h_cur.shape[0]
    T = 0.05                         # initial temperature
    cooling = 0.9995
    pop = 0.07                       # perturbation magnitude (fraction of unit)

    it = 0
    start = time.time()
    while True:
        now = time.time()
        if now - start > time_limit:
            break

        # propose perturbation
        noise = (np_rng.random(n) - 0.5) * 2.0 * pop
        h_new = h_cur + noise
        h_new = _repair_h(h_new)

        score_new = _c5_score(h_new)
        score_cur = _c5_score(h_cur)

        if score_new < score_cur or rnd.random() < math.exp(-(score_new - score_cur) / max(T, 1e-12)):
            h_cur = h_new
            if score_new < best_score:
                best_score = score_new
                best_h = h_new.copy()

        T *= cooling
        it += 1
        if it % 200 == 0:        # occasional re‑scale of pop to keep moves effective
            pop = max(0.02, pop * 0.99)

    return best_h, best_score

def _scale_up(h, factor):
    """Upsample by integer factor using simple repetition."""
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    """Return an improved step function for the Erdős C₅ problem.
    Output: (h_values, c5_bound, n_points)
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    remaining = budget_s

    # ------------------------------------------------------------------
    # 1. Base resolution (use provided initial_h_values if present)
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 80
        h0 = np.full(n0, 0.5)          # uniform satisfies integral exactly
    h0 = _repair_h(h0)

    # allocate time slice for base optimisation
    base_time = min(remaining * 0.4, 300.0)
    h_best, score_best = _anneal(h0, base_time)
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    # ------------------------------------------------------------------
    # 2. Try higher resolutions if time permits
    # ------------------------------------------------------------------
    for factor in (2, 2):               # 80→160→320
        if remaining < 30.0:
            break
        # upscale current best
        h_up = _scale_up(h_best, factor)
        # tiny random shake to break symmetry
        h_up = _repair_h(h_up + (np.random.rand(h_up.size) - 0.5) * 0.02)
        # allocate time slice (decrease with each upscale)
        slice_time = min(remaining * 0.4, 250.0)
        h_up_opt, score_up = _anneal(h_up, slice_time)
        if score_up < score_best:
            h_best, score_best = h_up_opt, score_up
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed

    # ------------------------------------------------------------------
    # 3. Final safety check: ensure constraints exactly hold
    # ------------------------------------------------------------------
    h_best = _repair_h(h_best)
    final_score = _c5_score(h_best)

    return h_best, final_score, h_best.shape[0]
