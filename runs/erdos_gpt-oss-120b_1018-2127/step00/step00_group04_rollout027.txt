import time
import numpy as np

def _c5_from_h(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx using FFT correlation."""
    n = h.size
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    rev = (1.0 - h)[::-1]
    H = np.fft.rfft(h, L)
    R = np.fft.rfft(rev, L)
    conv = np.fft.irfft(H * R, L)[: 2 * n - 1]
    return np.max(conv) * dx


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (samples on a uniform grid) that
    minimizes the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Array of samples (0/1) of the best found h.
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of grid points used.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Grid definition – moderate resolution for fast evaluation
    n_points = 400                     # can be increased later if time permits
    dx = 2.0 / n_points

    # Use the provided initial vector if it exists; otherwise start from uniform split.
    try:
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n_points:
            # resample to our grid via simple nearest‑neighbor copy
            idx = np.linspace(0, init.size - 1, n_points).astype(int)
            init_h = init[idx]
        else:
            init_h = init.copy()
    except Exception:
        init_h = np.concatenate([np.ones(n_points // 2), np.zeros(n_points // 2)])

    best_h = init_h.copy()
    best_c5 = _c5_from_h(best_h, dx)

    # ----------------------------------------------------------------------
    # Parameterised search: two disjoint intervals of total length 1.
    # a = length of first interval, placed at the left end.
    # b = start of second interval (must be ≥ a).
    # ----------------------------------------------------------------------
    a_vals = np.arange(0.02, 0.99, 0.02)           # step 0.02 for speed
    for a in a_vals:
        # feasible range for b given a
        b_min = a
        b_max = 2.0 - (1.0 - a)
        if b_max < b_min:
            continue
        b_vals = np.arange(b_min, b_max + 1e-9, 0.02)
        for b in b_vals:
            # construct h
            x = (np.arange(n_points) + 0.5) * dx
            h = ((x <= a) | ((x >= b) & (x <= b + (1.0 - a)))).astype(float)

            c5 = _c5_from_h(h, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, h

        # time guard
        if time.time() - start_time > budget_s * 0.85:
            break

    # ----------------------------------------------------------------------
    # Simple stochastic refinement around the best found configuration.
    # Flip a few random points while keeping the total integral exactly 1.
    # ----------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.95:
        # propose a random swap of a 1↔0 pair to preserve the sum
        ones = np.flatnonzero(best_h > 0.5)
        zeros = np.flatnonzero(best_h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_new = best_h.copy()
        h_new[i], h_new[j] = 0.0, 1.0
        c5_new = _c5_from_h(h_new, dx)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new

    return best_h, float(best_c5), n_points
