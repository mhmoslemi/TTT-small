import time
import numpy as np

# Helper: enforce box [0,1] and integral constraint exactly
def repair_h(h, target_sum, max_iter=20):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        # scale towards target
        if s > 0:
            h *= target_sum / s
        else:
            h[:] = target_sum / len(h)
        h = np.clip(h, 0.0, 1.0)
    # final tiny adjustment to hit target_sum exactly (within float eps)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff to a position that stays feasible
        for i in range(len(h)):
            new_val = h[i] + diff
            if 0.0 <= new_val <= 1.0:
                h[i] = new_val
                break
    return h

# Compute C5 for a given h vector
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose discretisation size (trade‑off between quality and speed)
    n_points = 200
    target_sum = n_points / 2.0

    # Build initial candidate
    if 'initial_h_values' in globals():
        init_old = np.asarray(initial_h_values, dtype=float)
        # linear interpolation to the new grid

        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_old, init_old)
    else:
        # simple half‑ones / half‑zeros pattern
        h = np.zeros(n_points, dtype=float)
        h[:n_points // 2] = 1.0

    h = repair_h(h, target_sum)
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Simulated‑annealing / stochastic descent parameters
    max_iter = 20000
    temp0 = 0.05

    for it in range(max_iter):

        # temperature schedule
