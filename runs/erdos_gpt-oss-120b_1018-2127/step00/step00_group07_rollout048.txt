import numpy as np
import time
import math

def _evaluate_c5(h):
    """Return C5 for sample array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def _project(h, target):
    """
    Project vector h onto the box [0,1] and the affine constraint sum(h)=target.
    Uses bisection on a scalar lambda for the operation  clip(h - λ,0,1).
    """
    # initial bounds for lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(h - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def _simulated_annealing(start_h, budget, start_time):
    """
    Simple SA that respects the time budget.
    Returns best_h, best_score found.
    """
    n = start_h.shape[0]
    target = n / 2.0
    cur = _project(start_h.copy(), target)
    cur_score = _evaluate_c5(cur)
    best_h, best_score = cur.copy(), cur_score

    # algorithmic parameters
    temp = 1.0
    cooling = 0.9995
    step_scale = 0.08  # typical perturbation magnitude
    rng = np.random.default_rng(12345)

    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget:
            break
        iteration += 1

        # generate candidate

        cand = cur + perturb
        cand = _project(cand, target)

        cand_score = _evaluate_c5(cand)

        # keep global best

        # SA acceptance
        if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / max(temp, 1e-12)):
            cur, cur_score = cand, cand_score

        # temperature decay
        temp *= cooling
        # gradual reduction of step size
        if iteration % 500 == 0:
            step_scale *= 0.9
            step_scale = max(step_scale, 0.005)

    return best_h, best_score


def _increase_resolution(h, new_n):
    """Linearly interpolate h to new_n samples over [0,2]."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
