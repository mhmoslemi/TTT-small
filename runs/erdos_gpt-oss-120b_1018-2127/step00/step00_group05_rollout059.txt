import time
import numpy as np
import random

# Helpers (pre‑imported in the harness):
# evaluate_erdos_solution(h)  – returns a dict with at least a 'c5' entry
# initial_h_values             – optional numpy array of a starting construction

def _dx(n):
    """Step size for n samples on [0,2]."""
    return 2.0 / n

def _corr_max(h):
    """Maximum overlap integral C5 for a discrete step function h."""
    n = h.size
    dx = _dx(n)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _repair(h):
    """Project h onto the feasible set: 0≤h≤1, sum(h)=n/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # Simple proportional scaling with iterative slack handling
    for _ in range(10):
        total = h.sum()
        delta = target - total
        if abs(delta) < 1e-12:
            break
        if delta > 0:
            slack = 1.0 - h
            slack_sum = slack.sum()
            if slack_sum == 0:
                break
            inc = delta * (slack / slack_sum)
            h += inc
        else:
            slack = h
            slack_sum = slack.sum()
            if slack_sum == 0:
                break
            dec = (-delta) * (slack / slack_sum)
            h -= dec
        h = np.clip(h, 0.0, 1.0)
    return h

def _initial_vector():

    if 'initial_h_values' in globals():

def _perturb(h, step):
    """Return a copy of h with a small random perturbation respecting bounds."""
    n = h.size
    h_new = h.copy()
    # modify a small random subset (1‑3 indices)
    k = random.randint(1, 3)
    idx = random.sample(range(n), k)

        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
    h_new = _repair(h_new)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a low‑C5 step function.

    Returns
    -------

    np.random.seed(seed)

    h = _initial_vector()
    best_h = h.copy()
    best_c5 = _corr_max(h)

    # Simple stochastic descent
    step = 0.3          # initial perturbation magnitude
    iter_cnt = 0

        iter_cnt += 1
        # gradually shrink step size

            cand_c5 = _corr_max(h_candidate)
