import time
import numpy as np

# evaluate_erdos_solution and initial_h_values are provided by the harness.

def _compute_c5(h, dx):
    """Return the C5 value for a step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _repair_h(h, target):
    """Project h onto the box [0,1] and enforce sum(h)=target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        h[:] = target / h.size
    else:
        h = h * (target / cur)
        h = np.clip(h, 0.0, 1.0)

    # Distribute any remaining mass while staying in bounds
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        idx = np.arange(h.size)
        np.random.shuffle(idx)
        for i in idx:
            if diff == 0.0:
                break
            if diff > 0.0:
                avail = 1.0 - h[i]
                delta = min(avail, diff)
            else:
                avail = h[i]
                delta = -min(avail, -diff)
            h[i] += delta
            diff -= delta
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 constant.

    Returns:
        (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    start_time = time.time()

    # -------------------------------------------------
    # Problem dimensions
    # -------------------------------------------------
    # Use a modest resolution that balances quality and speed.
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum(h)*dx = 1

    # -------------------------------------------------
    # Initial construction
    # -------------------------------------------------
    if "initial_h_values" in globals():
        # Upsample the provided parent to our resolution.
        parent = np.asarray(initial_h_values, dtype=float)
        x_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_parent, parent)
    else:
        # Uniform start (integral = 1)
        h = np.full(n_points, 0.5)

    h = _repair_h(h, target_sum)
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    # -------------------------------------------------
    # Simple stochastic local search
    # -------------------------------------------------
    iteration = 0
    max_iter = 200000  # soft cap; real stop is time budget
    sigma_initial = 0.2
    sigma_final = 0.005

    while True:
        now = time.time()
        if now - start_time > budget_s * 0.98:
            break
        if iteration >= max_iter:
            break

        # annealing schedule for step size
        t = iteration / max_iter
        sigma = sigma_initial * (1 - t) + sigma_final * t

        # propose a candidate
        cand = best_h + sigma * np.random.randn(n_points)
        cand = _repair_h(cand, target_sum)

        cand_c5 = _compute_c5(cand, dx)

        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand
        iteration += 1

    # Final safeguard: ensure feasibility
    best_h = _repair_h(best_h, target_sum)
    best_c5 = _compute_c5(best_h, dx)

    return best_h, float(best_c5), n_points
