import time
import numpy as np

def _compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_swap(h, i, j, delta):
    """Return a copy of h after moving `delta` from index j to i,
    clipped to keep all entries in [0,1]."""
    hi, hj = h[i], h[j]
    # feasible delta limits
    max_up_i = 1.0 - hi          # how much we can add to i
    max_down_j = hj             # how much we can remove from j
    max_up_j = 1.0 - hj
    max_down_i = hi

    if delta > 0:
        delta = min(delta, max_up_i, max_down_j)
    else:
        delta = -min(-delta, max_up_j, max_down_i)

    if delta == 0.0:
        return None  # no feasible move
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h
