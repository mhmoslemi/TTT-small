import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h, dx):
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_sum(vec, target):
    """Scale vector to have given sum while keeping non‑negativity."""
    s = np.sum(vec)
    if s == 0:
        return vec
    return vec * (target / s)

def _make_start(n, target_sum, base=None, rng=None):
    """Create a feasible start vector."""
    if base is not None:
        # perturb a copy of base
        h = np.array(base, copy=True)
        # add small random noise
        h += rng.uniform(-0.05, 0.05, size=n)
        h = np.clip(h, 0.0, 1.0)
    else:
        # random uniform start
        h = rng.uniform(0.0, 1.0, size=n)
    # project to required sum (exactly)
    h = _project_sum(h, target_sum)
    # ensure bounds after scaling
    h = np.clip(h, 0.0, 1.0)
    # final projection to sum (possible tiny deviation)
    h = _project_sum(h, target_sum)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    Optimize a step function h on [0,2] discretised to n points

        c5_bound (float): achieved C5 bound
        n_points (int): number of discretisation points
    """

    rng = np.random.default_rng(seed)

    # Determine problem size

        h0 = np.asarray(initial_h_values, dtype=float)

    else:
        n = 100

    dx = 2.0 / n

    # Baseline evaluation

        best_h = np.clip(h0, 0.0, 1.0)
        best_h = _project_sum(best_h, target_sum)
        best_c5 = _compute_c5(best_h, dx)
    else:
        best_h = _make_start(n, target_sum, rng=rng)
        best_c5 = _compute_c5(best_h, dx)

    # Equality constraint for SLSQP
    cons = ({'type': 'eq',

    bounds = [(0.0, 1.0) for _ in range(n)]
