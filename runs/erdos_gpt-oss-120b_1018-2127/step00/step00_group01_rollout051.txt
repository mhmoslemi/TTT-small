import time, numpy as np
from scipy.optimize import differential_evolution, minimize

def _c5_value(h):
    """Compute the overlap integral C5 for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _feasible_initial(n, seed):
    """Create a feasible start vector (sum = n/2) when no parent is given."""
    rng = np.random.default_rng(seed)
    # random Dirichlet scaled to n/2 keeps each entry ≤1 with high prob
    a = rng.random(n) + 0.1
    x = rng.dirichlet(a) * (n / 2)
    # clip to [0,1] and renormalise if necessary
    x = np.clip(x, 0.0, 1.0)
    excess = x.sum() - n / 2
    if abs(excess) > 1e-12:
        # adjust uniformly within bounds
        adj = excess / n
        x -= adj
        x = np.clip(x, 0.0, 1.0)
        # final renorm
        x *= (n / 2) / x.sum()
    return x
