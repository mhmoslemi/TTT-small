import time
import numpy as np

def _compute_c5(h):
    """Return C5 for a discrete step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _min_overlap(h):
    """Maximum of C5 equals 1 - (2/n)*min_overlap, so we use the min overlap."""
    n = h.size
    # autocorrelation of h with itself
    auto = np.correlate(h, h, mode="full")
    # central index corresponds to shift 0
    center = n - 1
    # consider all non‑zero shifts
    overlaps = np.concatenate((auto[:center], auto[center + 1:]))
    return overlaps.min()

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # decide discretisation size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        # start from the given vector but project to binary while keeping the sum
        h = (initial_h_values > 0.5).astype(float)
        # if the sum is not exactly n/2, fix by random swaps
        target_sum = n // 2
        cur_sum = int(h.sum())
        while cur_sum != target_sum:
            if cur_sum > target_sum:
                idx = rng.choice(np.where(h == 1)[0])
                h[idx] = 0
                cur_sum -= 1
            else:
                idx = rng.choice(np.where(h == 0)[0])
                h[idx] = 1
                cur_sum += 1
    else:
        # default size, even number for easy balancing
        n = 200
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        rng.shuffle(h)

    # keep track of best solution
    best_h = h.copy()
    best_min_ov = _min_overlap(best_h)
    best_c5 = 1.0 - (2.0 / n) * best_min_ov

    # annealing parameters
    temperature = 0.5
    cooling = 0.9995

    # pre‑compute indices of ones and zeros for fast swaps
    while True:
        now = time.time()
        if now - start > budget_s - 0.5:
            break

        # pick a random 1‑index and 0‑index to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # proposal: swap values
        h[i], h[j] = h[j], h[i]

        # evaluate
        cur_min_ov = _min_overlap(h)
        if cur_min_ov > best_min_ov:
            # improvement – keep
            best_min_ov = cur_min_ov
            best_h = h.copy()
            best_c5 = 1.0 - (2.0 / n) * best_min_ov
        else:
            # accept with probability exp(Δ/ T) where Δ = cur - best (negative)
            delta = cur_min_ov - best_min_ov
            if rng.random() >= np.exp(delta / temperature):
                # reject: undo swap
                h[i], h[j] = h[j], h[i]

        # temperature decay
        temperature *= cooling
        if temperature < 1e-6:
            temperature = 1e-6

    # final evaluation
    final_c5 = _compute_c5(best_h)

    return best_h, float(final_c5), n
