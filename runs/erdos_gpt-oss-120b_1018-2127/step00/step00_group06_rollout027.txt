import time
import numpy as np

def _evaluate(h):
    """Return C5 for a step function h on [0,2] with uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation h * (1-h shifted)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(h, target_sum, max_iter=20, tol=1e-9):
    """Project h into [0,1]^n with sum equal to target_sum."""
    # initial clipping
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        # indices that are not at a bound
        free = (h > 0.0) & (h < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds, cannot adjust further
            break
        # distribute the needed correction uniformly over free variables
        delta = diff / n_free
        h[free] += delta
        # re‑clip because adding may have pushed some out of bounds
        h = np.clip(h, 0.0, 1.0)
    # final correction by scaling the interior if needed
    s = h.sum()
    if abs(s - target_sum) > tol:

        interior = (h > 0.0) & (h < 1.0)
        if interior.any():
            h[interior] *= target_sum / s
            h = np.clip(h, 0.0, 1.0)
    return h

    rnd = np.random.rand(n)
    return _project_to_constraints(rnd, target_sum)
