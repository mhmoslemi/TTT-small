import time
import math
import random
import numpy as np

def _compute_c5(h):
    """Return the C5 value for a discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _project_constraints(v, n):
    """Project vector v onto the box [0,1]^n with sum(v)=n/2."""
    target = n * 0.5
    h = np.clip(v, 0.0, 1.0)

    # iterative water‑filling style repair
    for _ in range(20):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(free.sum())
        if cnt == 0:
            # no interior points – distribute equally then clip
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / cnt
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)

    # final scaling if still off (rare)
    s = h.sum()
    if abs(s - target) > 1e-9:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)

    return h
