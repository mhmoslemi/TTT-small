import numpy as np
import time

def compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def binary_adjust(h, target_ones, rng):
    """Flip entries to obtain exactly target_ones ones, preserving bounds."""
    cur_ones = int(np.round(np.sum(h)))
    if cur_ones == target_ones:
        return h
    indices_one = np.where(h > 0.5)[0]
    indices_zero = np.where(h <= 0.5)[0]
    while cur_ones != target_ones:
        if cur_ones > target_ones:
            # turn a 1 into 0
            i = rng.choice(indices_one)
            h[i] = 0.0
            indices_one = np.delete(indices_one, np.where(indices_one == i))
            indices_zero = np.append(indices_zero, i)
            cur_ones -= 1
        else:
            # turn a 0 into 1
            i = rng.choice(indices_zero)
            h[i] = 1.0
            indices_zero = np.delete(indices_zero, np.where(indices_zero == i))
            indices_one = np.append(indices_one, i)
            cur_ones += 1
    return h

def upsample_to_n(h0, n, rng):
    """Resize h0 (array of length m) to length n by linear interpolation
    and then binarize with the correct number of ones."""
    m = len(h0)
    x0 = np.linspace(0.0, 2.0, m, endpoint=False)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x, x0, h0)
    # initial binarisation
    h = (h >= 0.5).astype(float)
    target_ones = n // 2
    h = binary_adjust(h, target_ones, rng)
    return h

def random_binary(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones_idx = rng.choice(n, n // 2, replace=False)
    h[ones_idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.

    Returns:
        h_values (np.ndarray) : length n_points, values in [0,1]

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # Simulated‑annealing parameters
    max_iters = 50000
    init_temp = 0.02
    final_temp = 0.0005

    # Pre‑compute indices of ones and zeros for fast swaps
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h <= 0.5)[0]

    for itr in range(1, max_iters + 1):
        # time‑budget check
        if time.time() - start_time > budget_s * 0.95:
            break

        # linear temperature schedule
        temp = init_temp + (final_temp - init_temp) * (itr / max_iters)

        # propose a swap
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
