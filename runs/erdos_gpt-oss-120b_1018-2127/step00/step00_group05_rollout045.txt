import time
import numpy as np

def _repair(h, target_sum):
    """Project vector h onto [0,1] box while enforcing sum == target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative scaling+clipping
    for _ in range(5):
        cur_sum = h.sum()
        if cur_sum == 0:
            h[:] = target_sum / h.size
            break
        scale = target_sum / cur_sum
        h = np.clip(h * scale, 0.0, 1.0)
    # Final adjustment to hit target exactly (if tiny error)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly over non‑saturated entries
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _evaluate(h, dx):
    """Return max overlap C5 for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Desired resolution
    n_points = kwargs.get("n_points", 200)
    if n_points < 50:
        n_points = 50
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum(h)*dx = 1

    # Initialise h
    if 'initial_h_values' in globals() and initial_h_values is not None:
        # Upsample / interpolate to required size
        base = np.asarray(initial_h_values, dtype=float)
        base_x = np.linspace(0.0, 2.0, base.size, endpoint=False)
        new_x = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(new_x, base_x, base)
    else:
        # simple uniform start
        h = np.full(n_points, 0.5)

    h = _repair(h, target_sum)
    best_h = h.copy()
    best_score = _evaluate(best_h, dx)

    # Adaptive parameters
    max_iter = 10_000_000  # safety cap
    temp_start = 0.2
    temp_end = 0.001
    decay = (temp_end / temp_start) ** (1.0 / max_iter)

    temperature = temp_start
    iter_count = 0

    while True:
        # Time check
        if time.time() - start_time > budget_s - 0.5:
            break
        if iter_count >= max_iter:
            break

        # Propose a perturbation
        proposal = best_h + rng.normal(scale=0.07, size=n_points)
        proposal = _repair(proposal, target_sum)

        score = _evaluate(proposal, dx)

        # Acceptance rule (simulated annealing)
        if score < best_score:
            best_h, best_score = proposal, score
        else:
            # maybe accept worse
            if rng.random() < np.exp(-(score - best_score) / temperature):
                best_h, best_score = proposal, score

        # Cool down
        temperature *= decay
        iter_count += 1

        # Occasionally restart from a random feasible point to escape
        if iter_count % 5000 == 0:
            rnd = rng.random(n_points)
            rnd = _repair(rnd, target_sum)
            rnd_score = _evaluate(rnd, dx)
            if rnd_score < best_score:
                best_h, best_score = rnd, rnd_score

    return best_h, float(best_score), n_points
