import time
import numpy as np

def _objective(h):
    """Return C5 for vector h (samples over [0,2])."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair(h):
    """Clip to [0,1] and enforce sum(h) == n/2 by uniform redistribution."""
    n = h.shape[0]
    target = n / 2.0
    # 1. clip
    h = np.clip(h, 0.0, 1.0)
    # 2. adjust the sum iteratively
    for _ in range(10):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction to hit the target exactly (within floating error)
    cur = h.sum()
    if abs(cur - target) > 1e-10:
        h += (target - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _interpolate(prev, n_new):
    """Linear interpolation of a solution vector to a new length."""
    x_old = np.linspace(0.0, 2.0, len(prev), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, prev)

def _random_walk(h, rng, sigma):
    """Perturb h by Gaussian noise with standard deviation sigma."""
    h_new = h + rng.normal(scale=sigma, size=h.shape)
    return h_new

def _anneal(initial_h, n, time_limit):
    """Simple simulated‑annealing loop respecting the time limit."""
    rng = np.random.default_rng(12345)
    best = _repair(initial_h.copy())
    best_score = _objective(best)

    # schedule parameters
    T = 1.0
    cooling = 0.99995
    scale_factor = 0.05  # magnitude of raw perturbation relative to n

    start = time.time()
    while True:
        now = time.time()
        if now - start > time_limit:
            break

        sigma = scale_factor * np.sqrt(T) / np.sqrt(n)  # smaller sigma for larger n
        cand = _random_walk(best, rng, sigma)
        cand = _repair(cand)

        cand_score = _objective(cand)

        # accept if improvement; otherwise probabilistic acceptance
        if cand_score < best_score:
            best, best_score = cand, cand_score
        else:
            # Metropolis criterion
            accept_prob = np.exp(-(cand_score - best_score) / max(T, 1e-12))
            if rng.random() < accept_prob:
                best, best_score = cand, cand_score

        T *= cooling
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the allotted budget."""
    start_total = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a progression of grid sizes.
    # ------------------------------------------------------------------
    grid_sizes = [95, 190, 380, 760]

    # Allocate portions of the total budget proportionally to grid size.
    # Larger grids are more expensive, so give them less absolute time.
    total_weight = sum(1.0 / s for s in grid_sizes)
    time_alloc = {s: (budget_s * (1.0 / s) / total_weight) for s in grid_sizes}

    best_overall_h = None
    best_overall_score = np.inf
    best_overall_n = None

    # ------------------------------------------------------------------
    # 2. Iterate over the grids, carrying the solution forward.
    # ------------------------------------------------------------------
    parent = None
    for n in grid_sizes:
        # respect remaining wall‑clock time
        elapsed = time.time() - start_total
        if elapsed >= budget_s:
            break
        remaining = budget_s - elapsed
        allowed = min(time_alloc[n], remaining)

        # starting vector for this grid
        if parent is None:
            # use provided initial_h_values if it exists and has correct length
            try:
                init = np.array(initial_h_values, copy=False)
                if init.shape[0] != n:
                    init = _interpolate(init, n)
            except NameError:
                # no parent supplied – create a simple balanced step function
                init = np.zeros(n)
                half = n // 2
                init[:half] = 1.0
        else:
            init = _interpolate(parent, n)

        # run annealing on this grid
        h_opt, score_opt = _anneal(init, n, allowed)

        # keep the best across grids
        if score_opt < best_overall_score:
            best_overall_h = h_opt.copy()
            best_overall_score = score_opt
            best_overall_n = n

        parent = h_opt  # feed forward to finer grid

    # ------------------------------------------------------------------
    # 3. Final polishing on the best grid (optional extra time)
    # ------------------------------------------------------------------
    elapsed = time.time() - start_total
    if elapsed < budget_s and best_overall_h is not None:
        extra = budget_s - elapsed
        h_final, score_final = _anneal(best_overall_h, best_overall_n, extra * 0.5)
        if score_final < best_overall_score:
            best_overall_h, best_overall_score = h_final, score_final

    # Ensure feasibility before returning
    best_overall_h = _repair(best_overall_h)

    return best_overall_h, float(best_overall_score), int(best_overall_n)
