    h = np.clip(h, 0.0, 1.0)

        h = np.clip(h * factor, 0.0, 1.0)

            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

import time, math
import numpy as np

def _repair(h, target_sum):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    # iterative scaling & clipping
    for _ in range(5):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        if s == 0:
            break
        factor = target_sum / s
        h = np.clip(h * factor, 0.0, 1.0)
    # final redistribution for tiny residuals
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _score(h, dx):
    """Maximum overlap integral for a given step vector."""
    # correlation of h with (1-h) over all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _optimize(h_start, n, time_limit, seed):
    """Simple simulated‑annealing style optimizer for a fixed resolution."""
    target_sum = n / 2.0
    dx = 2.0 / n
    rng = np.random.default_rng(seed)

    h = _repair(h_start.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_score = _score(best_h, dx)

    start = time.time()
    it = 0
    # anticipated iteration budget; will stop earlier if out of time
    max_iters = 50000
    while True:
        if time.time() - start > time_limit:
            break
        it += 1
        # linear cooling from T=1 to Tmin
        T = max(1e-4, 1.0 * (1.0 - it / max_iters))
        step = 0.2 * T  # maximum perturbation magnitude
        perturb = (rng.random(n) - 0.5) * step
        h_new = _repair(h + perturb, target_sum)
        s_new = _score(h_new, dx)

        if s_new < best_score:
            best_score = s_new
            best_h = h_new.copy()

        # accept if improvement or with Metropolis probability
        curr_score = _score(h, dx)
        if s_new < curr_score or rng.random() < math.exp((curr_score - s_new) / (T + 1e-12)):
            h = h_new
    return best_h, best_score
