import time
import numpy as np

def _project_to_constraints(v, target_sum):
    """Project vector v onto the box [0,1] and the affine sum constraint."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(15):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final fix if needed
    s = v.sum()
    if abs(s - target_sum) > 1e-8:
        scale = target_sum / s if s != 0 else 0.0
        v = np.clip(v * scale, 0.0, 1.0)
    return v

def _random_feasible(n):
    """Generate a random feasible h vector of length n."""
    v = np.random.rand(n)
    return _project_to_constraints(v, n / 2.0)

def _evaluate_c5(h):
    """Compute the C5 value for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 bound.

    Returns:
        h_values (np.ndarray): optimal step values (length n_points)
        c5_bound (float):   corresponding C5 value
        n_points (int):     number of discretisation points used
    """
    np.random.seed(seed)
    start_time = time.time()

    # --- Choose discretisation size -------------------------------------------------
    # Small enough to stay fast, large enough to allow improvement.
    n_points = 140                     # 2× the default 70‑sample parent
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # --- Initialise from parent if present -----------------------------------------
    try:
        parent = initial_h_values          # supplied by harness when available
        # linear interpolation to new resolution
        x_old = np.linspace(0.0, 2.0, len(parent), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h_current = np.interp(x_new, x_old, parent)
        h_current = _project_to_constraints(h_current, target_sum)
    except Exception:
        h_current = _random_feasible(n_points)

    best_h = h_current.copy()
    best_c5 = _evaluate_c5(best_h)

    # --- Optimisation parameters ----------------------------------------------------
    max_time = budget_s * 0.95               # leave margin for finalisation
    sigma = 0.15                              # initial perturbation scale
    it = 0
    while time.time() - start_time < max_time:
        it += 1
        # create candidate by perturbing the current best
        candidate = best_h + np.random.normal(scale=sigma, size=n_points)
        candidate = np.clip(candidate, 0.0, 1.0)
        candidate = _project_to_constraints(candidate, target_sum)

        cand_c5 = _evaluate_c5(candidate)

        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = candidate
            # tighten step size after improvement
            sigma = max(0.02, sigma * 0.95)

        # occasional random restart to escape plateaus
        if it % 1200 == 0:
            rand_candidate = _random_feasible(n_points)
            rand_c5 = _evaluate_c5(rand_candidate)
            if rand_c5 < best_c5:
                best_c5 = rand_c5
                best_h = rand_candidate
                sigma = 0.15

        # gradual cooling of sigma
        if it % 500 == 0:
            sigma = max(0.02, sigma * 0.97)

    # final safety projection (should already be feasible)
    best_h = _project_to_constraints(best_h, target_sum)
    best_c5 = _evaluate_c5(best_h)

    return best_h, best_c5, n_points
