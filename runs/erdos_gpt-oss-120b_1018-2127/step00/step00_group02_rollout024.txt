import time
import numpy as np

def compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    # linear correlation with zero padding (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def make_binary_feasible(n):
    """Binary feasible vector: exactly n/2 ones (or nearest integer)."""
    ones = n // 2
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, ones, replace=False)
    h[idx] = 1.0
    # if n is odd, adjust the remaining mass uniformly
    if n % 2:
        remaining = 0.5
        zero_idx = np.where(h == 0)[0]
        h[zero_idx[0]] = remaining
    return h

def perturb(h):
    """Transfer a small amount of mass between two coordinates while staying feasible."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # amount that can be added to i without exceeding 1
    max_inc = min(0.2, 1.0 - h[i])
    # amount that can be removed from j without going below 0
    max_dec = min(0.2, h[j])
    if max_inc <= 0 or max_dec <= 0:
        return h.copy()
    delta = np.random.uniform(0.0, min(max_inc, max_dec))
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def project_to_constraints(h):
    """Clip to [0,1] and enforce the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    target_sum = h.size / 2.0
    current_sum = h.sum()
    if np.isclose(current_sum, target_sum, atol=1e-12):
        return h
    # Simple redistribution of the difference uniformly on free coordinates
    diff = target_sum - current_sum
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size == 0:
        # fallback: scale whole vector (will stay inside bounds because of clipping)
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0.0, 1.0)
        # re‑adjust any tiny residual error
        diff = target_sum - h.sum()
    # distribute diff equally on free entries, respecting bounds
    while abs(diff) > 1e-12 and free.size:
        inc = diff / free.size
        for idx in free:
            new_val = h[idx] + inc
            if new_val < 0.0:
                diff += h[idx]  # we set it to 0
                h[idx] = 0.0
            elif new_val > 1.0:
                diff -= (1.0 - h[idx])  # we set it to 1
                h[idx] = 1.0
            else:
                h[idx] = new_val
        diff = target_sum - h.sum()
        free = np.where((h > 0.0) & (h < 1.0))[0]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h minimizing the Erdős overlap constant C₅."""
    np.random.seed(seed)
    start = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = h_init.size
        # Enforce exact sum constraint on the supplied start
        h_init = project_to_constraints(h_init)
    else:
        n = 60                 # default grid size when no parent is provided
        h_init = make_binary_feasible(n)

    dx = 2.0 / n
    best_h = h_init
    best_c5 = compute_c5(best_h, dx)

    # Main stochastic improvement loop
    while time.time() - start < budget_s * 0.95:
        # occasional full random restart
        if np.random.rand() < 0.02:
            candidate = make_binary_feasible(n)
        else:
            candidate = perturb(best_h)

        candidate = project_to_constraints(candidate)
        cand_c5 = compute_c5(candidate, dx)

        if cand_c5 < best_c5:
            best_h, best_c5 = candidate, cand_c5

    return best_h, best_c5, n
