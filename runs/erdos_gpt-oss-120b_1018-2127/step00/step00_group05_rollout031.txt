import numpy as np
import time
import random

# Helper to ensure feasibility: 0<=x<=1 and sum(x)=target (target = n/2)
def make_feasible(v):
    n = len(v)
    target = n / 2.0
    # clip first
    v = np.clip(v, 0.0, 1.0)
    # if already satisfies sum, return
    current = v.sum()
    if abs(current - target) < 1e-12:
        return v
    # project using a simple quadratic program (cvxpy may be unavailable,
    # so we implement a capped simplex projection)
    # This is a standard algorithm: first ignore upper bound, then enforce it.
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # enforce upper bound 1 by similar procedure
    over = w - 1.0
    if np.any(over > 1e-12):
        # simple iterative clipping
        for _ in range(10):
            excess = w[w > 1.0] - 1.0
            excess_sum = excess.sum()
            if excess_sum < 1e-12:
                break
            mask = w < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            delta = excess_sum / cnt
            w[mask] = np.minimum(w[mask] + delta, 1.0)
            w[w > 1.0] = 1.0
    # final scaling to correct any tiny residual
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # distribute diff proportionally to free variables
        free = (w > 0) & (w < 1)
        cnt = free.sum()
        if cnt > 0:
            w[free] += diff / cnt
    # final clip
    w = np.clip(w, 0.0, 1.0)
    return w

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # obtain starting vector
    try:
        h = np.array(initial_h_values, dtype=float)  # may be provided by harness
    except NameError:
        # fallback: uniform distribution satisfying integral
        n = 70

        h[:n // 2] = 1.0
    h = make_feasible(h)
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    max_iter = 50000
    epsilon_max = 0.08
    temp_start = 0.02
    temp_end = 1e-4
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        # temperature schedule (linear decay)
        temp = temp_start + (temp_end - temp_start) * (it / max_iter)

        # propose a transfer between two random indices
        i, j = np.random.choice(n, 2, replace=False)
        # maximum we can move from i to j while staying in [0,1]
        max_move = min(epsilon_max, best_h[i], 1.0 - best_h[j])
        if max_move <= 0:

        delta = np.random.uniform(0, max_move)

        new_h[j] += delta
