import numpy as np
import time

def compute_c5(h):
    """Return the max overlap integral for discretized step function h."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with values in [0,1] and integral 1.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Determine problem size and initialise h
    # ----------------------------------------------------------------------
    try:

    except NameError:
        # fall back to a moderate resolution if no parent is provided
        n = 200
        h = np.full(n, 0.5, dtype=float)  # sum = n/2 satisfies the integral

    target_sum = n / 2.0
    # enforce the exact sum (might be slightly off because of float noise)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute the discrepancy uniformly while respecting bounds
        adjust = diff / n
        h += adjust
        h = np.clip(h, 0.0, 1.0)
        # final corrective scaling
        h *= target_sum / h.sum()

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ----------------------------------------------------------------------
    # Simple simulated‑annealing style search
    # ----------------------------------------------------------------------
    total_time = budget_s * 0.95  # leave a small margin
    max_iter = 10_000_000  # safety upper bound
    step_initial = 0.05
    temp_initial = 0.02

    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= total_time:
            break

        # linear schedule for step size and temperature
        frac = elapsed / total_time
        step = step_initial * (1.0 - frac) + 0.001 * frac
        temp = temp_initial * (1.0 - frac) + 1e-5 * frac

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        # propose a delta that keeps both variables in [0,1]
        delta_min_i = max(-step, -h[i])
        delta_max_i = min(step, 1.0 - h[i])
        if delta_min_i >= delta_max_i:
            continue
        delta = rng.uniform(delta_min_i, delta_max_i)

        # ensure the companion index can absorb -delta
        if -delta < -h[j] or -delta > (1.0 - h[j]):
            # adjust delta to the feasible range for j
            delta = np.clip(delta, - (1.0 - h[j]), h[j])

        h_new[j] -= delta
