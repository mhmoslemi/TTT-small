import time, math
import numpy as np
from scipy import stats  # just to ensure scipy is available

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretized step function h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def project_to_bounds_and_sum(v: np.ndarray,
                              total: float,
                              lower: float = 0.0,
                              upper: float = 1.0,
                              max_iter: int = 200) -> np.ndarray:
    """Clip to [lower,upper] and adjust so that sum(v) == total."""
    v = np.clip(v, lower, upper)
    for _ in range(max_iter):
        cur = v.sum()
        diff = cur - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v > lower
        else:
            free = v < upper
        if not np.any(free):
            break
        delta = diff / free.sum()
        v[free] -= delta
        v = np.clip(v, lower, upper)
    # final tiny correction if needed
    cur = v.sum()
    if abs(cur - total) > 1e-9:
        residual = total - cur
        free = np.ones_like(v, dtype=bool)
        v[free] += residual / free.sum()
        v = np.clip(v, lower, upper)
    return v


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretised) to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # Determine starting vector and size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 84  # fallback size
        h0 = np.full(n, 0.5, dtype=float)  # satisfies the integral automatically

    target_sum = n / 2.0
    h = project_to_bounds_and_sum(h0, target_sum)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    start_time = time.time()
    # Simple simulated‑annealing schedule
    T = 0.6
    Tmin = 1e-4
    decay = 0.998

    while time.time() - start_time < budget_s * 0.95:
        # propose a perturbation
        noise = rng.normal(scale=0.12, size=n)
        cand = h + noise
        cand = project_to_bounds_and_sum(cand, target_sum)

        cand_val = compute_c5(cand)
        cur_val = compute_c5(h)

        # accept if better or by Metropolis criterion
        if cand_val < cur_val or rng.random() < math.exp((cur_val - cand_val) / max(T, 1e-12)):
            h = cand
            cur_val = cand_val

        # keep global best
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # temperature decay
        T = max(Tmin, T * decay)

    return (best_h, best_val, n)
