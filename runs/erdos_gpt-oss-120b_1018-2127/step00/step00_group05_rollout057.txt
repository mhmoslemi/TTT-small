import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that minimizes
    C5 = max_x ∫ h(t) (1-h(t+x)) dt.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # Determine number of discretization points.
    # Prefer to use the length of the provided initial_h_values if present.
    try:
        base_len = len(initial_h_values)  # pre‑imported global
    except Exception:
        base_len = 70
    n_points = kwargs.get("n_points", base_len)
    n_points = int(max(30, min(200, n_points)))  # keep size modest
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # ∑ h[i] must equal this

    # ----------------------------------------------------------------------
    # Helper: produce a feasible binary vector (0/1) with exactly target_sum ones.
    def binary_initial():
        h = np.zeros(n_points, dtype=float)
        ones = int(round(target_sum))
        # place ones uniformly as a starting pattern
        idx = np.linspace(0, n_points - 1, ones, dtype=int)
        h[idx] = 1.0
        rng.shuffle(h)
        return h

    # Helper: evaluate C5 for a candidate vector (assumes feasibility)
    def evaluate(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    # ----------------------------------------------------------------------
    # Start from the provided construction if it matches the desired size,
    # otherwise fall back to a binary pattern.
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h_current = np.clip(initial_h_values.astype(float), 0.0, 1.0)
        # Repair to satisfy the integral exactly (simple scaling + clipping loop)
        for _ in range(5):
            s = h_current.sum()
            if abs(s - target_sum) < 1e-12:
                break
            h_current = np.clip(h_current * (target_sum / s), 0.0, 1.0)
    else:
        h_current = binary_initial()

    best_h = h_current.copy()
    best_score = evaluate(best_h)

    # ----------------------------------------------------------------------
    # Hill‑climbing with swaps that preserve the binary sum.
    start_time = time.time()
    no_improve = 0
    while True:
        # Time‑budget guard
        if time.time() - start_time > budget_s - 0.5:
            break

        # Pick a random 1 and a random 0 to swap
        ones_idx = np.flatnonzero(best_h > 0.5)
        zeros_idx = np.flatnonzero(best_h < 0.5)

        score = evaluate(h_candidate)

        if score < best_score - 1e-12:
            best_h = h_candidate
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        # If stuck for a while, perform a larger perturbation
        if no_improve >= 5000:
