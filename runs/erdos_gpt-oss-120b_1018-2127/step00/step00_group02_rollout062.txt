import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact discrete C5 value for step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def project_bounds_sum(h, target_sum, iters=10):
    """
    Clip to [0,1] and then adjust to hit the required sum.
    Simple iterative proportional redistribution.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(iters):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:  # fallback to all entries
            free = np.arange(h.size)
        if diff > 0:
            caps = 1.0 - h[free]
        else:
            caps = h[free]
        total_cap = caps.sum()
        if total_cap <= 0:
            break
        delta = diff * (caps / total_cap)
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction (adds uniform amount, may violate bounds minimally)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h


def perturb(h, sigma, rng):
    """Add small Gaussian noise on a random subset and re‑project."""
    n = h.size
    new = h.copy()
    # pick ~10 % of indices
    k = max(1, n // 10)
    idx = rng.choice(n, size=k, replace=False)
    new[idx] += rng.normal(scale=sigma, size=k)
    target = n / 2.0
    return project_bounds_sum(new, target)


def smooth_max_objective(h, beta):
    """Smooth approximation of max using log‑sum‑exp."""

    return (np.log(np.sum(np.exp(beta * (corr - m)))) / beta) + m


def slsqp_optimize(h0, beta=12.0, maxiter=250):
    """Try to improve h0 with SLSQP on the smooth‑max objective."""
    n = h0.size
    target = n / 2.0

    def obj(x):
        return smooth_max_objective(x, beta)

    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
    )
    if res.success:
        return np.clip(res.x, 0.0, 1.0)
    return h0
