import time
import numpy as np

# evaluate_erdos_solution and initial_h_values are provided in the execution environment.

def compute_c5(h):
    """Return the C5 bound for a given step function h."""
    return evaluate_erdos_solution(h)

def feasible_initial(n, seed):
    """Create a feasible initial vector of length n with values in [0,1] and sum = n/2."""
    rng = np.random.default_rng(seed)
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        if h0.size == n:
            h = h0.copy()
        else:
            # fall back to random construction if size mismatch
            h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
            rng.shuffle(h)
    else:
        # start from a binary pattern (half ones, half zeros)
        h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        rng.shuffle(h)

    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # adjust first entry while respecting bounds
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        # if still off due to clipping, distribute the remaining error uniformly
        residual = target - h.sum()
        if abs(residual) > 1e-12:
            # spread residual over all entries proportionally (small correction)
            h += residual / n
            np.clip(h, 0.0, 1.0, out=h)
            # final renorm to exact sum (tiny adjustment on first element)
            h[0] += target - h.sum()
    return h

def propose_move(h, rng):
    """Generate a new feasible h by moving mass between two positions."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]

    # allowable delta so that both stay in [0,1]
    delta_max = min(1.0 - hi, hj)
    delta_min = -min(hi, 1.0 - hj)
    if delta_max <= delta_min:
        return h  # no feasible move (should be rare)

    # scale magnitude with a random factor
    delta = rng.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h to lower the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h on the discretised grid.
    c5_bound : float
        Corresponding maximal overlap integral.
    n_points : int
        Number of discretisation points used.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a modest grid size (even ensures binary feasibility)
    n_points = 124 if "initial_h_values" not in globals() else len(initial_h_values)
    if n_points % 2 == 1:
        n_points += 1  # enforce even length for convenience

    # Initial feasible vector
    h = feasible_initial(n_points, seed)
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Simulated‑annealing schedule
    elapsed = lambda: time.time() - start_time
    max_elapsed = budget_s * 0.95  # leave a safety margin
    init_temp = best_score * 0.5 + 1e-3
    final_temp = 1e-5

    iteration = 0
    while elapsed() < max_elapsed:
        fraction = elapsed() / max_elapsed
        temperature = init_temp * (1 - fraction) + final_temp * fraction

        h_candidate = propose_move(h, rng)
        cand_score = compute_c5(h_candidate)

        # Accept if better or with probability exp(-(Δ)/T)
        if cand_score < best_score:
            best_h, best_score = h_candidate.copy(), cand_score
        delta = cand_score - compute_c5(h)
        if delta < 0 or rng.random() < np.exp(-delta / temperature):
            h = h_candidate

        iteration += 1
        # occasional re‑randomisation to escape deep local minima
        if iteration % 5000 == 0:
            h = feasible_initial(n_points, rng.integers(0, 2**31 - 1))

    # Final evaluation to guarantee consistency
    final_score = compute_c5(best_h)
    return best_h, final_score, n_points
