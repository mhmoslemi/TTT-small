import time
import numpy as np

def _c5_of_h(h):
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discrete h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _random_move(rng, h, total, max_step=0.02):
    """Propose a move that preserves the total sum and respects bounds."""
    n = h.size
    # pick two distinct indices
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # compute allowable delta so both stay in [0,1]
    hi, hj = h[i], h[j]
    delta_max = min(max_step, 1.0 - hi, hj)      # increase i, decrease j
    delta_min = -min(max_step, hi, 1.0 - hj)    # decrease i, increase j
    if delta_max <= delta_min:
        return None  # no feasible move
    delta = rng.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    # sum is preserved automatically
    return h_new
