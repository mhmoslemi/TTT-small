import time
import numpy as np

def _project_to_constraints(h, target_sum, eps=1e-9, max_iter=10):
    """
    Project vector h onto the box [0,1] and the affine hyperplane sum(h)=target_sum.
    Simple successive projection: clip, then shift, repeat.
    """
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) <= eps:
            break
        # shift uniformly to fix the sum
        h += (target_sum - cur_sum) / h.size
    # final clip to guarantee box constraints
    h = np.clip(h, 0.0, 1.0)
    # final exact correction of the sum by distributing the tiny residual
    residual = target_sum - h.sum()
    if abs(residual) > eps:
        # add residual to the first element that is not at a bound
        free = np.where((h > 0.0 + eps) & (h < 1.0 - eps))[0]
        if free.size == 0:
            # all at bounds, spread over all entries (should be tiny)
            h += residual / h.size
        else:
            h[free[0]] += residual
    return h

def _compute_c5(h, dx):
    """Return max correlation value C5 for vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def _gradient_at_lag(h, lag, dx):
    """
    Gradient of the correlation at a given lag (integer) with respect to h.
    Handles both positive and negative lags; out‑of‑range contributions are zero.
    """
    n = h.size
    if lag >= 0:
        part1 = np.empty_like(h)
        part1[: n - lag] = 1.0 - h[lag:]
        part1[n - lag :] = 0.0
        part2 = np.empty_like(h)
        part2[:lag] = 0.0
        part2[lag:] = -h[: n - lag]
    else:  # lag < 0, use symmetry
        l = -lag
        part1 = np.empty_like(h)
        part1[l:] = 1.0 - h[: n - l]
        part1[:l] = 0.0
        part2 = np.empty_like(h)
        part2[: n - l] = -h[l:]
        part2[n - l :] = 0.0
    return (part1 + part2) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step‑function h on [0,2] to minimise the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # Determine number of sample points
    if 'initial_h_values' in globals():
        init = globals()['initial_h_values']
        n = int(init.shape[0])
        # initialise from the supplied vector, projected onto constraints
        target_sum = n / 2.0
        h = _project_to_constraints(init.astype(float).copy(), target_sum)
    else:
        # fallback: use 80 points
        n = 80
        target_sum = n / 2.0
        # random start respecting constraints
        h = rng.random(n)
        h = _project_to_constraints(h, target_sum)

    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    start_time = time.time()
    # optimisation parameters
    max_iters = 200000
    step0 = 0.2
    decay = 0.9995

    step = step0
    it = 0
    while it < max_iters:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # current correlation values
        corr_vals = np.correlate(h, 1.0 - h, mode='full') * dx
        lag_max = np.argmax(corr_vals) - (n - 1)  # shift index range -(n-1)..(n-1)

        # gradient descent step at the worst lag
        grad = _gradient_at_lag(h, lag_max, dx)
        h_new = h - step * grad
        h_new = _project_to_constraints(h_new, target_sum)

        c5_new = _compute_c5(h_new, dx)

        # accept if improvement, otherwise keep old (simple greedy)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < _compute_c5(h, dx):
            h = h_new.copy()
        else:
            # occasional random restart to escape plateaus
            if rng.random() < 0.001:
                h = rng.random(n)
                h = _project_to_constraints(h, target_sum)

        # anneal step size
        step *= decay
        it += 1

    # final projection to guarantee exact constraints
    best_h = _project_to_constraints(best_h, target_sum)

    final_c5 = _compute_c5(best_h, dx)
    return best_h, float(final_c5), int(n)
