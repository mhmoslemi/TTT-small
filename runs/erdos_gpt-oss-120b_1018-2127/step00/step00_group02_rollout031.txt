import time
import numpy as np

# evaluation helper may be provided; fallback to own implementation
def _compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _evaluate(h):
    """Use provided evaluator if present, otherwise internal."""
    try:
        # evaluate_erdos_solution is injected by the harness if available
        return float(evaluate_erdos_solution(h))
    except Exception:
        return _compute_c5(h)


def _project_bounds_and_sum(h):
    """Clip to [0,1] and renormalize to exact sum = n/2 while staying in bounds."""
    n = h.size
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if np.isclose(cur_sum, target_sum):
        return h
    # Simple redistribution: add/subtract uniformly where possible
    diff = target_sum - cur_sum
    # Distribute the diff proportionally to distance from bounds
    free_up = 1.0 - h          # capacity to increase
    free_down = h              # capacity to decrease
    total_free = free_up.sum() if diff > 0 else free_down.sum()
    if total_free == 0:
        return h  # can't adjust
    if diff > 0:
        adjust = diff * free_up / total_free
        h = h + adjust
    else:
        adjust = (-diff) * free_down / total_free
        h = h - adjust
    # Clip again to be safe
    return np.clip(h, 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap constant.

    Returns:
        h_values : list of floats, length n_points, each in [0,1]
        c5_bound : float, achieved C5 value
        n_points : int, number of discretisation points
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise solution
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals() and isinstance(initial_h_values, (list, np.ndarray)):
        h = np.asarray(initial_h_values, dtype=float)
        n = h.size
        # Ensure feasibility
        h = _project_bounds_and_sum(h)
    else:
        # fallback: 45 points, half ones half zeros
        n = 45
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        h = _project_bounds_and_sum(h)

    best_h = h.copy()
    best_c5 = _evaluate(best_h)

    # ------------------------------------------------------------------
    # 2. Simulated annealing parameters
    # ------------------------------------------------------------------
    max_iters = 10_000_000  # safety cap
    temp_initial = 0.02
    temp_final = 1e-5
    temp = temp_initial
    step_initial = 0.15
    step_final = 0.005
    step = step_initial

    # schedule based on elapsed proportion
    def _update_schedule(elapsed, total):
        frac = elapsed / total
        # exponential cooling for temperature
        t = temp_initial * (temp_final / temp_initial) ** frac
        s = step_initial * (step_final / step_initial) ** frac
        return t, s

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # leave a small margin
            break
        if it >= max_iters:
            break
        it += 1

        # update cooling schedule
        temp, step = _update_schedule(elapsed, budget_s)

        # propose a move that preserves the sum exactly
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = rng.normal(scale=step)

        # bound eps to keep both entries within [0,1]
        eps = max(min(eps, 1.0 - h[i]), -h[j])
        if eps == 0:
            continue

        h_candidate = h.copy()
        h_candidate[i] += eps
        h_candidate[j] -= eps

        # projection is not needed because sum unchanged and bounds enforced above
        # but clip just in case numerical errors
        h_candidate = np.clip(h_candidate, 0.0, 1.0)

        cand_c5 = _evaluate(h_candidate)

        # acceptance criterion (Metropolis)
        if cand_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = cand_c5
            h = h_candidate
        else:
            accept_prob = np.exp(-(cand_c5 - best_c5) / max(temp, 1e-12))
            if rng.random() < accept_prob:
                h = h_candidate

        # occasional random restart to avoid stagnation
        if it % 200000 == 0:
            # perturb best solution slightly
            pert = rng.normal(scale=step, size=n)
            pert -= pert.mean() * (2.0 / n)  # keep sum zero
            h_restart = best_h + pert
            h_restart = _project_bounds_and_sum(h_restart)
            h = h_restart
            # recompute its C5
            _ = _evaluate(h)  # discard

    # Final projection to guarantee feasibility
    best_h = _project_bounds_and_sum(best_h)

    return best_h.tolist(), best_c5, int(n)
