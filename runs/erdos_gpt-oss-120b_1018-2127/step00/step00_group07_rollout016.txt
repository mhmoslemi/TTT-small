import time
import numpy as np
import math

def _project_to_feasible(h, target_sum):
    """Clip h to [0,1] and adjust to meet exact sum target."""
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative redistribution
    for _ in range(10):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # indices that can move in needed direction
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        # distribute proportionally
        available = mask.sum()
        step = diff / available
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # adjust a single free element if possible
        idx = np.where((h > 0.0) & (h < 1.0))[0]
        if idx.size:
            i = idx[0]
            h[i] = min(1.0, max(0.0, h[i] + diff))
    return h

def _evaluate_c5(h, dx):
    """Return max overlap integral for step function h."""
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h on [0,2] to minimize the Erdős C5 bound.

    Returns:
        h_values (np.ndarray): optimized values on the grid, sum = n_points/2.
        c5_bound (float): corresponding maximal overlap integral.
        n_points (int): number of discretisation points used.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine grid size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
        # Ensure feasibility
        target = n / 2.0
        h0 = _project_to_feasible(h0, target)
    else:
        n = 80
        target = n / 2.0
        # random feasible start
        rand = rng.random(n)
        h0 = _project_to_feasible(rand, target)

    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = _evaluate_c5(best_h, dx)

    # Simulated annealing parameters
    init_temp = 0.02
    min_temp = 1e-5

    # Main loop
    iteration = 0
    while True:

        if elapsed >= budget_s * 0.98:  # leave tiny margin
