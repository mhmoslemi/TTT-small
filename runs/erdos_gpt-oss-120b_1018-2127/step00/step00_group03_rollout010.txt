import time
import numpy as np

def _evaluate_h(h):
    """Return the Erdős overlap constant C5 for a given vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_feasible(h, target_sum):
    """Adjust a vector so that its entries lie in [0,1] and sum equals target_sum.
    This is a simple clipping+redistribution that preserves feasibility."""
    # Clip first
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # Distribute the remaining diff uniformly over indices that are not at a bound
    mask = (h > 0.0) & (h < 1.0)
    if np.any(mask):
        adjustment = diff / mask.sum()
        h[mask] += adjustment
        # after adjustment some values may violate bounds; clip again
        h = np.clip(h, 0.0, 1.0)
    # Finally renormalize any tiny residual error
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # add residual to a random non‑saturated coordinate
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            free = np.arange(h.size)
        idx = np.random.choice(free)
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h

    Returns

            Optimised function values in [0,1].
        c5_bound : float
            The attained max overlap integral.
        n_points : int
            Number of discretisation points (length of h_values).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:

        h = np.asarray(initial_h_values, dtype=float).copy()
    except NameError:
        # fallback: simple half‑ones / half‑zeros pattern
        n_default = 70
        h = np.concatenate([np.ones(n_default // 2), np.zeros(n_default - n_default // 2)])
        h = h.astype(float)

    n = h.shape[0]
    target_sum = n / 2.0
    # ensure exact feasibility of the start vector
    h = _project_feasible(h, target_sum)

    best_h = h.copy()
    best_score = _evaluate_h(best_h)

    # ------------------------------------------------------------------
    # simulated‑annealing style random walk
    # ------------------------------------------------------------------
    # schedule parameters
    max_iter = 10_000_000  # upper bound; loop also stops by time
    temp_start = 0.5
    temp_end = 0.001
    decay = (temp_end / temp_start) ** (1.0 / max_iter)

    temp = temp_start
    it = 0
    while True:
        # time check
        if time.time() - start_time > budget_s - 0.2:
            break
        if it >= max_iter:
            break

        # propose a move: transfer mass delta from i (donor) to j (receiver)
        i, j = np.random.choice(n, 2, replace=False)

        # maximum transferable amount without violating bounds
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0.0:
            it += 1
            continue

        delta = np.random.rand() * max_transfer * (0.5 + 0.5 * temp)
