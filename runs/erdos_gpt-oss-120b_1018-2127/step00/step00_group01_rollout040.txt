def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, scipy.optimize as opt
    start = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n_points = 64
        h0 = rng.random(n_points)
        h0 = h0 / h0.sum() * (n_points/2)  # enforce integral
    n_points = h0.size
    dx = 2.0 / n_points

    def overlap(h):
        # assume h is numpy array
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    target_sum = n_points / 2.0
    penalty = 1e3

    def obj(x):
        # x may be list, convert
        h = np.clip(x, 0.0, 1.0)
        val = overlap(h)
        dev = np.sum(h) - target_sum
        if dev != 0:
            val += penalty * abs(dev)
        return val

    bounds = [(0.0,1.0)] * n_points

    # differential evolution
    maxiter = 200
    # Ensure we don't exceed budget
    remaining = budget_s - (time.time() - start)
    # small safety
    if remaining < 5:
        # fallback to current h0
        best = h0.copy()
    else:
        de_result = opt.differential_evolution(obj, bounds,
                                               seed=seed,
                                               maxiter=int(maxiter),
                                               popsize=15,
                                               polish=False,
                                               updating='deferred',
                                               workers=1)
        best = np.clip(de_result.x,0,1)

    # local refinement if time permits
    if time.time() - start < budget_s * 0.9:
        cons = {'type':'eq', 'fun': lambda x: np.sum(x) - target_sum}
        # use SLSQP
        loc_res = opt.minimize(lambda x: overlap(np.clip(x,0,1)),
                               best,
                               method='SLSQP',
                               bounds=bounds,
                               constraints=cons,
                               options={'maxiter':200, 'ftol':1e-9})
        if loc_res.success:
            best = np.clip(loc_res.x,0,1)

    # final evaluation
    c5 = overlap(best)
    # optional: use provided evaluator for accuracy
    try:
        c5 = evaluate_erdos_solution(best)
    except Exception:
        pass

    return best, float(c5), n_points

import time
import numpy as np
import scipy.optimize as opt

# globals for objective helpers (will be set in run)
_DX = 1.0
_TARGET_SUM = 0.0
_PENALTY = 1e3

def _overlap(h: np.ndarray) -> float:
    """Return max overlap integral for a given step function h."""
    # full convolution gives all shifts; multiply by dx
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * _DX

def _objective(x: np.ndarray) -> float:
    """Penalized objective for differential evolution."""
    h = np.clip(x, 0.0, 1.0)
    val = _overlap(h)
    dev = np.sum(h) - _TARGET_SUM
    if dev != 0.0:
        val += _PENALTY * abs(dev)
    return val

def _eq_sum(x: np.ndarray) -> float:
    """Equality constraint: integral must be exactly 1."""
    return np.sum(x) - _TARGET_SUM

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h on [0,2] minimizing the Erdős C₅ overlap."""
    global _DX, _TARGET_SUM, _PENALTY

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # starting vector
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        n_start = 60
        h0 = rng.random(n_start)
        h0 = h0 / h0.sum() * (n_start / 2.0)  # enforce integral constraint

    n_points = h0.size
    _DX = 2.0 / n_points
    _TARGET_SUM = n_points / 2.0

    bounds = [(0.0, 1.0)] * n_points

    # ------------------------------------------------------------------
    # Global search via Differential Evolution
    # ------------------------------------------------------------------
    time_elapsed = lambda: time.time() - start_time
    maxiter_de = 250                     # modest iterations to stay within budget
    popsize_de = 15

    # If remaining time is very low, skip DE and keep current h0
    if time_elapsed() < budget_s * 0.2:
        best_h = h0.copy()
    else:
        de_res = opt.differential_evolution(
            _objective,
            bounds,
            seed=seed,
            maxiter=maxiter_de,
            popsize=popsize_de,
            polish=False,
            updating="deferred",
            workers=1,
        )
        best_h = np.clip(de_res.x, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Local refinement with SLSQP (if time permits)
    # ------------------------------------------------------------------
    if time_elapsed() < budget_s * 0.85:
        constraints = {"type": "eq", "fun": _eq_sum}
        # SLSQP respects bounds and a single equality constraint
        slsq_res = opt.minimize(
            _overlap,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 300, "ftol": 1e-12},
        )
        if slsq_res.success:
            best_h = np.clip(slsq_res.x, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Final evaluation
    # ------------------------------------------------------------------
    try:
        c5_val = float(evaluate_erdos_solution(best_h))
    except Exception:
        c5_val = float(_overlap(best_h))

    return best_h, c5_val, n_points
