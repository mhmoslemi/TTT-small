import time
import numpy as np

def _repair(vec, target_sum, max_iter=5):
    """Project vec onto [0,1] box and enforce sum == target_sum."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        if abs(s - target_sum) <= 1e-12:
            break
        # scale towards target, then re‑clip
        factor = target_sum / (s if s != 0 else 1e-12)
        v *= factor
        v = np.clip(v, 0.0, 1.0)
    # final tiny adjustment to hit target exactly (if possible)
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly over non‑boundary entries
        free = (v > 0.0) & (v < 1.0)
        if free.any():
            v[free] += diff / free.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _evaluate(h, dx):
    """Return C5 for a given step vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] satisfying
        0 ≤ h ≤ 1,   ∑ h * dx = 1,
    to minimise C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    t_start = time.time()

    # ----------------------------------------------------------------------
    # Determine number of sample points and initial vector
    # ----------------------------------------------------------------------
    if 'n_points' in kwargs:
        n_points = int(kwargs['n_points'])
    else:
        # use the size of the provided parent if present, else a modest default
        n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70

    dx = 2.0 / n_points
    target_sum = n_points / 2.0                     # because ∫ h = 1

    # initialise from parent if possible, otherwise from uniform guess
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        if len(parent) != n_points:
            # linear interpolation / resampling to new resolution
            x_old = np.linspace(0.0, 2.0, len(parent), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
            parent = np.interp(x_new, x_old, parent)
        parent = _repair(parent, target_sum)
    else:
        parent = _repair(rng.random(n_points), target_sum)

    best_h = parent.copy()
    best_score = _evaluate(best_h, dx)

    # ----------------------------------------------------------------------
    # Simple stochastic hill‑climbing with occasional random restarts
    # ----------------------------------------------------------------------
    # parameters (tuned for speed vs quality)
    max_no_improve = 2000            # stop if no improvement for this many trials
    no_improve_cnt = 0
    trial = 0

    while time.time() - t_start < budget_s - 0.5:   # tiny safety margin
        trial += 1
        # generate a mutant
        noise = rng.normal(scale=0.07, size=n_points)
        candidate = best_h + noise
        candidate = _repair(candidate, target_sum)

        cand_score = _evaluate(candidate, dx)

        if cand_score < best_score - 1e-12:
            best_h, best_score = candidate, cand_score
            no_improve_cnt = 0
        else:
            no_improve_cnt += 1

        # occasional random restart to escape plateaus
