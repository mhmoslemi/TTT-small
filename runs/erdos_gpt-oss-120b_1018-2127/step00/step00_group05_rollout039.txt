import time
import numpy as np

# ----------------------------------------------------------------------
def _evaluate(h):
    """Return C5 for a step vector h on [0,2] (dx = 2/n)."""
    try:
        # pre‑imported helper, if available
        return evaluate_erdos_solution(h)
    except NameError:
        n = h.size
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        return dx * np.max(corr)


def _project_to_simplex(v, target, eps=1e-12):
    """Clip to [0,1] and then adjust to hit exact sum = target."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= eps:
        return v
    # simple redistribution: shift uniformly among non‑bound entries
    free = (v > eps) & (v < 1 - eps)
    if free.any():
        delta = (target - s) / free.sum()
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    else:
        # fallback – scale whole vector then clip again
        v = v * (target / max(s, eps))
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction
    diff = target - v.sum()
    if abs(diff) > eps:
        # add diff to the first non‑saturated entry
        idx = np.where((v > eps) & (v < 1 - eps))[0]
        if idx.size:
            v[idx[0]] += diff
            v = np.clip(v, 0.0, 1.0)
    return v


def _random_feasible(n, target, rng):
    """Generate a random feasible vector (sum = target, 0‑1 bounds)."""
    # start from Dirichlet then clip/scale
    x = rng.random(n)
    x = x / x.sum() * target
    return _project_to_simplex(x, target)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h minimizing the Erdős C5 overlap.

    Returns
    -------
    (h_values, c5_bound, n_points) :
        h_values – NumPy 1‑D array of length n_points satisfying the constraints
        c5_bound – the achieved C5 value (float)
        n_points – number of discretisation points used
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Choose discretisation size.
    # If a parent construction exists, reuse its length; otherwise start with 70.
    if "initial_h_values" in globals():
        h_curr = np.array(initial_h_values, dtype=float)  # use directly
        n = h_curr.size
        target = n / 2.0
        h_curr = _project_to_simplex(h_curr, target)
    else:
        n = 70
        target = n / 2.0
        h_curr = np.zeros(n, dtype=float)
        h_curr[:int(target)] = 1.0  # binary feasible start

    best_h = h_curr.copy()
    best_c5 = _evaluate(best_h)

    # ------------------------------------------------------------------
    # Main stochastic improvement loop.
    # Simple pairwise mass‑transfer moves keep the sum invariant.
    max_iter = 10 ** 7  # safety cap
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Occasionally restart with a fresh random feasible vector
        if rng.random() < 0.005:
            cand = _random_feasible(n, target, rng)
        else:
            cand = best_h.copy()
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # propose moving a small amount δ from i to j
                delta = (rng.random() - 0.5) * 0.2  # in [-0.1,0.1]
                delta = np.clip(delta, -cand[i], 1.0 - cand[j])
                cand[i] -= delta
                cand[j] += delta
        # Ensure feasibility (mainly for the random restart case)
        cand = _project_to_simplex(cand, target)

        c5 = _evaluate(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand
        iter_count += 1

    # ------------------------------------------------------------------
    # If time permits, try a modest refinement with a higher resolution.
    # This optional phase uses the current best as a template.
    if time.time() - start_time < budget_s * 0.90:
        # double points (must stay even for integer target)
        n2 = n * 2
        target2 = n2 / 2.0
        # interpolate current best onto the finer grid
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_fine = np.interp(x_new, x_old, best_h)
        h_fine = _project_to_simplex(h_fine, target2)

        # local search on fine grid (lighter iteration count)
        best_h_f = h_fine.copy()
        best_c5_f = _evaluate(best_h_f)
        for _ in range(2000):
            i, j = rng.integers(0, n2, size=2)
            if i == j:
                continue
            delta = (rng.random() - 0.5) * 0.05
            delta = np.clip(delta, -best_h_f[i], 1.0 - best_h_f[j])
            cand_f = best_h_f.copy()
            cand_f[i] -= delta
            cand_f[j] += delta
            cand_f = _project_to_simplex(cand_f, target2)
            c5_f = _evaluate(cand_f)
            if c5_f < best_c5_f:
                best_c5_f = c5_f
                best_h_f = cand_f

        # Accept fine solution if it improves overall bound
        if best_c5_f < best_c5:
            best_h = best_h_f
            best_c5 = best_c5_f
            n = n2

    # ------------------------------------------------------------------
    return (best_h, float(best_c5), int(n))
