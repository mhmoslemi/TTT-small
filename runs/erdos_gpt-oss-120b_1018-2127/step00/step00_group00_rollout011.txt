import numpy as np
import time

# The harness provides evaluate_erdos_solution and possibly initial_h_values
# We will compute the objective ourselves for robustness.

def _c5_from_h(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discrete samples."""
    n = h.shape[0]
    dx = 2.0 / n
    # full discrete cross‑correlation
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _propose_move(h, rng):
    """Return a new h obtained by moving a small amount of mass between two indices."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    # amount we can take from i and give to j while respecting bounds
    max_take = min(h[i], 1.0 - h[j])
    if max_take <= 0:
        return None
    # limit step size to keep search fine‑grained
    step = min(0.05, max_take)          # maximal transfer per move
    delta = rng.random() * step        # 0 <= delta <= step
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h
