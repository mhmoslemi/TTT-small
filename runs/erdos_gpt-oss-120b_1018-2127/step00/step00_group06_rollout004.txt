import time
import numpy as np

def _c5_of_h(h, dx):
    """Return the C5 value for a step function h sampled on a uniform grid."""
    # correlation of h with (1-h) over all integer shifts
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(conv)

def _project_bounds(arr):
    """Clamp to [0,1] and return a copy."""
    return np.clip(arr, 0.0, 1.0)

def _initial_vector(n):
    """Create a feasible start: uniform values with exact integral."""
    # start with 0.5 everywhere, then adjust a few entries to satisfy sum=n/2
    h = np.full(n, 0.5, dtype=float)
    target = n / 2.0
    current = h.sum()
    diff = target - current
    # add diff to first element (diff will be integer multiple of 0.5, so stay in bounds)
    h[0] += diff
    h = _project_bounds(h)
    # after clipping we may be slightly off; fix by scaling uniformly
    excess = h.sum() - target
    if abs(excess) > 1e-12:
        # distribute the excess over all entries that are not at a bound
        mask = (h > 0.0) & (h < 1.0)
        if np.any(mask):
            h[mask] -= excess / mask.sum()
            h = _project_bounds(h)
    # final exactness adjustment (tiny)
    h -= (h.sum() - target) / n
    return h
