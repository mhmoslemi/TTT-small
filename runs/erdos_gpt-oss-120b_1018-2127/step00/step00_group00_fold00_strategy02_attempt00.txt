   class CriticNetwork:
        def __init__(self, input_size=..., hidden_layers=[64,64,64]):
            self.model = self.build_model(input_size, hidden_layers)
            self.optimizer = Adam(self.model.parameters(), lr=0.0001)
            self.loss_fn = MSELoss()

        def build_model(self, input_size, hidden_layers):
            # Build MLP architecture
            layers = [input_size] + hidden_layers + [1]
            net = Sequential()
            for i in range(len(layers)-1):
                net.add(Dense(layers[i], layers[i+1], activation='relu'))
            return net

        def train(self, h_batch, target_batch, epochs=10, batch_size=32):
            # Convert to tensors
            for epoch in range(epochs):
                for i in range(0, len(h_batch), batch_size):
                    x = torch.tensor(h_batch[i], dtype=torch.float32).view(-1, input_size)
                    y = torch.tensor(target_batch[i]).view(-1, 1)
                    optimizer.zero_grad()
                    output = self.model(x)
                    loss = self.loss_fn(output, y)
                    loss.backward()
                    optimizer.step()
