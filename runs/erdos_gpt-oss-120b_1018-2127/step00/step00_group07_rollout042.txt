import time, math, random
import numpy as np

def _repair(vec):
    """Clip to [0,1] and enforce sum(vec)=len(vec)/2."""
    n = len(vec)
    target = n / 2.0
    vec = np.clip(vec, 0.0, 1.0)
    for _ in range(5):
        total = vec.sum()
        if total == 0:
            vec = np.full_like(vec, target / n)
            break
        scale = target / total
        vec = vec * scale
        vec = np.clip(vec, 0.0, 1.0)
    # tiny residual correction
    diff = target - vec.sum()
    if abs(diff) > 1e-12:
        mask = (vec > 0) & (vec < 1)
        if mask.any():
            vec[mask] += diff / mask.sum()
            vec = np.clip(vec, 0.0, 1.0)
        else:
            vec = np.full_like(vec, target / n)
    return vec

def _score(vec, dx):
    """Maximum overlap integral for step function vec."""
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx
