import time
import numpy as np
from math import inf

def _c5_of_h(h, dx):
    """Return the C5 value for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _feasible_move(h, i, j, delta):
    """Return a new array after moving delta from i to j, keeping bounds."""
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def _random_delta(max_delta):
    """Sample a positive delta up to max_delta."""
    # use a uniform draw, but shrink a bit to keep away from exact bounds
    return np.random.rand() * max_delta * 0.9 + 1e-8

def _local_search(h, dx, budget_s, start_time):
    """Simple pair‑swap hill‑climb within remaining budget."""
    best_h = h.copy()
    best_val = _c5_of_h(best_h, dx)
    n = len(h)

    while time.time() - start_time < budget_s:
        i, j = np.random.choice(n, 2, replace=False)

        # how much we can transfer from i to j while staying in [0,1]
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 1e-12:
            continue
        delta = _random_delta(max_delta)

        cand = _feasible_move(best_h, i, j, delta)
        val = _c5_of_h(cand, dx)

        if val < best_val - 1e-9:               # accept improvement
            best_h, best_val = cand, val
        # else reject (simple hill‑climb)

    return best_h, best_val

def _interpolate_h(old_h, new_n):
