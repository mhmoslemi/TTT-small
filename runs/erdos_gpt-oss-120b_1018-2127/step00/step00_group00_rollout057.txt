import time
import numpy as np
from scipy.optimize import minimize

# Helper: compute the C5 bound for a given step vector
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# Helper: check feasibility (within tolerances)
def feasible(h):
    n = h.size
    if h.min() < -1e-9 or h.max() > 1 + 1e-9:
        return False
    return abs(h.sum() - n / 2) <= 1e-6

# Interpolate an existing solution to a new resolution
def transfer_solution(old_h, new_n):
    old_n = old_h.size
    # sample points at left edges of each subinterval
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    # bring sum to target by scaling, then clamp and renormalize once
    target = new_n / 2.0
    scale = target / new_h.sum() if new_h.sum() != 0 else 0.0
    new_h = new_h * scale
    new_h = np.clip(new_h, 0.0, 1.0)
    # final tiny adjustment to meet the integral exactly (add uniform offset)
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        new_h += diff / new_n
        new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

# Optimization wrapper for a given resolution
def optimize_h(start_h, rng):
    n = start_h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - n / 2.0}

    # objective expects a flat array
    def obj(x):
        return compute_c5(x)

    best_res = None
    best_val = np.inf

    # a few random restarts to escape poor local minima
    for _ in range(3):
        if _ == 0:
            x0 = start_h.copy()
        else:
            noise = rng.uniform(-0.05, 0.05, size=n)
            x0 = np.clip(start_h + noise, 0.0, 1.0)
            # renormalize to satisfy integral (simple scaling)
            x0 = x0 * (n / 2.0) / x0.sum()
        res = minimize(
            obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 800, "ftol": 1e-8, "disp": False},
        )
        if res.success:
            val = obj(res.x)
            if val < best_val:
                best_val = val
                best_res = res

    return best_res

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # resolutions to try (must be modest to stay within budget)
    n_list = [30, 45, 60, 90, 180]

    # locate initial vector if supplied
    try:
        init_vec = np.array(initial_h_values, copy=True)
    except NameError:
        # fallback: uniform distribution
        init_vec = np.full(45, 0.5)  # sum = 22.5 = 45/2, respects constraints
    # ensure proper shape
    if init_vec.ndim != 1:
        init_vec = init_vec.ravel()

    best_h = None
    best_score = np.inf
    best_n = None

    for n in n_list:
        if time.time() - start_time > budget_s:
            break

        if n == init_vec.size:
            h0 = init_vec.copy()
        else:
            h0 = transfer_solution(init_vec, n)

        # sanity check / project if needed
        if not feasible(h0):
            # simple projection onto bounds and integral
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = h0 * (n / 2.0) / h0.sum()

        # evaluate current
        cur_score = compute_c5(h0)
        if cur_score < best_score:
            best_score, best_h, best_n = cur_score, h0.copy(), n

        # run local optimization
        res = optimize_h(h0, rng)
        if res is not None:

                    best_score, best_h, best_n = opt_score, h_opt.copy(), n

        best_h = init_vec.copy()
