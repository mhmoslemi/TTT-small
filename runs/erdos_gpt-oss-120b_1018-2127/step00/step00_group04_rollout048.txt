import numpy as np
import time
import cvxpy as cp

def _project_to_constraints(v, target_sum):
    """Project vector v onto the box [0,1] and linear equality sum = target_sum."""
    n = v.shape[0]
    w = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(w - v)),
                      [w >= 0, w <= 1, cp.sum(w) == target_sum])
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=200)
    if w.value is None:
        # fallback: simple clipping and scaling
        w = np.clip(v, 0, 1)
        diff = target_sum - w.sum()
        w += diff / n
        w = np.clip(w, 0, 1)
    else:
        w = np.array(w.value).flatten()
    return w

def _evaluate_c5(h):
    """Return C5 for a step function sampled in array h."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _initial_h(n_points):
    """Create an initial guess of length n_points."""
    # Use provided initial_h_values if present, otherwise start with 0.5 everywhere.
    try:
        init = np.asarray(initial_h_values, dtype=float)
        # interpolate to required length
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_old, init)
    except NameError:
        h = np.full(n_points, 0.5, dtype=float)
    # enforce constraints exactly
    target = n_points / 2.0
    h = _project_to_constraints(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (sampled) to minimize the
    Erdős overlap constant C5.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Choose a modest resolution: enough to capture structure but fast to evaluate.
    n_points = 200  # 0.01 spacing
    target_sum = n_points / 2.0

    # Initial vector
    h = _initial_h(n_points)
    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)

    # Pre‑compute index pairs for moves
    idx = np.arange(n_points)

    # Main stochastic local‑search loop
    while time.time() - start_time < budget_s * 0.95:
        # pick two distinct indices
        i, j = np.random.choice(idx, 2, replace=False)
        # decide a transfer amount delta from i to j
        max_delta = min(0.02, h[i], 1.0 - h[j])
        if max_delta <= 1e-8:
            continue
        delta = np.random.uniform(0, max_delta)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # no need to re‑project: sum preserved and bounds respected by construction
        c5_new = _evaluate_c5(h_new)

        # Accept if better (greedy). Occasionally accept worse to escape local minima.
        if c5_new < best_c5 or np.random.rand() < 0.001:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        # Periodically re‑project to correct any numeric drift
        if np.random.rand() < 0.01:
            h = _project_to_constraints(h, target_sum)

    # Final projection to guarantee feasibility
    best_h = _project_to_constraints(best_h, target_sum)
    best_c5 = _evaluate_c5(best_h)

    return best_h, float(best_c5), n_points
