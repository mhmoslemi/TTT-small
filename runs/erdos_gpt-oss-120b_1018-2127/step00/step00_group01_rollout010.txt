import time
import numpy as np

def _evaluate_c5(h):
    """Return the max overlap C5 for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _initial_guess(n):
    """Create a feasible starting vector for given n."""
    target = n / 2.0
    h = np.zeros(n, dtype=float)
    full = int(target)                     # number of entries set to 1
    h[:full] = 1.0
    rem = target - full
    if full < n:
        h[full] = rem
    return h

def _random_move(h, step):
    """Propose a new h by moving a small amount of mass between two positions."""
    n = h.size
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None, None
    delta = (np.random.rand() - 0.5) * 2 * step  # signed delta in [-step, step]
    # ensure bounds after move
    if delta > 0:
        max_delta = min(delta, 1.0 - h[j], h[i])   # take from i, give to j
    else:
        max_delta = max(delta, -h[j], -(1.0 - h[i]))  # negative delta means opposite direction
    if max_delta == 0:
        return None, None
    h_new = h.copy()
    h_new[i] -= max_delta
    h_new[j] += max_delta
    return h_new, max_delta

def _upgrade_grid(h):

    current = h_new.sum()
    diff = target - current
    if abs(diff) > 1e-12:

            free = np.arange(n_new)
