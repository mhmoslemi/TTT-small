def _compute_c5(h: np.ndarray) -> float:
    """Exact C5 evaluation for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _project_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Projects vector v onto the box [0,1]^n while enforcing sum(v)=target_sum.
    Uses a simple water-filling / iterative redistribution scheme.
    """
    w = np.clip(v, 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - w.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((w > 0.0) & (w < 1.0))[0]
        if free.size == 0:          # all at bounds – cannot satisfy exactly, break
            break
        inc = diff / free.size
        w[free] += inc
        w = np.clip(w, 0.0, 1.0)
    # Minor correction for possible tiny residual
    residual = target_sum - w.sum()
    if abs(residual) > 1e-12:
        # add/subtract residual to the first non‑boundary entry
        idx = np.where((w > 0.0) & (w < 1.0))[0]
        if idx.size > 0:
            w[idx[0]] = np.clip(w[idx[0]] + residual, 0.0, 1.0)
    return w
