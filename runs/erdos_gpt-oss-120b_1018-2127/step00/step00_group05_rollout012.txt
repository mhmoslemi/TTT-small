import time
import numpy as np

def _project_bounds_sum(v, target, lo=0.0, hi=1.0, iters=60):
    """Project vector v onto the box [lo,hi] with a fixed sum = target.
    Returns w = clip(v + λ, lo, hi) where λ is chosen so that sum(w)=target.
    """
    # Binary search on λ
    low = -np.max(v) - 1.0
    high = hi - np.min(v) + 1.0
    for _ in range(iters):
        mid = (low + high) * 0.5
        w = np.clip(v + mid, lo, hi)
        s = w.sum()
        if s > target:
            high = mid
        else:
            low = mid
    λ = (low + high) * 0.5
    return np.clip(v + λ, lo, hi)


def _c5_score(h):
    """Return C5 = max correlation of h with (1-h) multiplied by dx."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _make_initial(n_points):
    """Create a feasible starting vector of length n_points."""
    # Use the provided initial_h_values if present, otherwise a simple shape.
    try:
        base = np.array(initial_h_values, dtype=float)
        base_n = base.size
        # Interpolate linearly onto the new grid
        x_old = np.linspace(0.0, 2.0, base_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_old, base)
    except NameError:
        # No parent construction: use a symmetric two‑block shape
        h = np.zeros(n_points)
        half = n_points // 4
        h[:half] = 1.0
        h[-half:] = 1.0
    # Project onto feasible set
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    h = _project_bounds_sum(h, target)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap constant.

    Returns
    -------
    h_values : np.ndarray
        Feasible vector of length n_points with values in [0,1] and sum == n_points/2.
    c5_bound : float
        The achieved C5 value (max overlap integral).
    n_points : int
        Number of discretisation points used.
    """
    start_time = time.time()
    np.random.seed(seed)

    # Choose a moderate dimension (under 1k). 200 gives enough flexibility.
    n_points = 200
    target_sum = n_points / 2.0

    # Initial feasible solution
    current = _make_initial(n_points)
    best = current.copy()
    best_score = _c5_score(best)

    # Simple stochastic hill‑climbing with occasional larger jumps
    iteration = 0
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1

        # Propose a perturbation
        proposal = best.copy()
        # Number of coordinates to touch decays slowly
        k = max(1, int(n_points * 0.02 * (0.99 ** (iteration // 1000))))
        idx = np.random.choice(n_points, size=k, replace=False)
        noise = np.random.normal(scale=0.05, size=k)
        proposal[idx] += noise

        # Enforce box constraints and the integral constraint
        proposal = np.clip(proposal, 0.0, 1.0)
        proposal = _project_bounds_sum(proposal, target_sum)

        # Evaluate
        score = _c5_score(proposal)

        # Accept if better
        if score < best_score:
            best = proposal
            best_score = score

        # Occasionally perform a larger random restart to escape stagnation
        if iteration % 2000 == 0:
            restart = _make_initial(n_points)
            restart_score = _c5_score(restart)
            if restart_score < best_score:
                best = restart
                best_score = restart_score

    # Final check: ensure feasibility (project once more)
    best = _project_bounds_sum(np.clip(best, 0.0, 1.0), target_sum)

    return best, best_score, n_points
