def compute_max_overlap(h):
    n = len(h)
    hx = h.astype(np.float64)
    one_minus = 1.0 - hx
    fft_len = 1 << (2*n -1).bit_length()
    H = np.fft.fft(hx, fft_len)
    O = np.fft.fft(one_minus, fft_len)
    corr = np.fft.ifft(H * np.conj(O)).real
    # correlate mode='full' yields length 2n-1 with shift -n+1..n-1
    corr = corr[:2*n-1]  # because cyclic conv; but due to zero padding it matches linear.
    return np.max(corr) * (2.0 / n)

import time, random, math
import numpy as np

def _max_overlap(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for discretized h."""
    n = h.size
    dx = 2.0 / n
    # linear convolution via FFT (zero‑padded to avoid wrap‑around)
    m = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, m)
    O = np.fft.fft(1.0 - h, m)
    corr = np.fft.ifft(H * np.conj(O)).real[: 2 * n - 1]
    return np.max(corr) * dx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------
    # Initialise h – use provided construction if available, otherwise a flat start.
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        # upscale by simple repetition (95 → 190 points, keeps the integral exact)
        h = np.repeat(h0, 2)
    else:
        # flat start with exact integral (value 0.5 gives sum = n/2)
        n_target = 190
        h = np.full(n_target, 0.5, dtype=float)

    n = h.size
    assert abs(h.sum() - n / 2) < 1e-12, "initial sum constraint violated"
    # ----------------------------------------------------------------------
    # Simulated‑annealing loop
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_score = _max_overlap(best_h)

    cur_h = h.copy()
    cur_score = best_score

    # initial temperature loosely based on current objective
    T = max(best_score * 0.5, 0.01)

    iteration = 0
    # main optimisation cycle – stop a bit before the hard budget limit
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1

        # propose a mass transfer between two distinct positions
        i, j = random.sample(range(n), 2)

        # maximum feasible transfer respecting bounds
        max_from_i = cur_h[i]                 # can take at most this much away
        max_to_j = 1.0 - cur_h[j]             # can add at most this much
        max_move = min(max_from_i, max_to_j, 0.1)  # cap step size for stability
        if max_move <= 0:
            continue

        delta = random.random() * max_move
        if random.random() < 0.5:
            delta = -delta  # opposite direction

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        # safety – bounds (should already hold)
        if (new_h < 0).any() or (new_h > 1).any():
            continue

        new_score = _max_overlap(new_h)
        dE = new_score - cur_score

        # Metropolis acceptance
        if dE < 0 or random.random() < math.exp(-dE / T):
            cur_h, cur_score = new_h, new_score
            if new_score < best_score:
                best_score, best_h = new_score, new_h.copy()

        # simple exponential cooling every few thousand steps
        if iteration % 2000 == 0:
            T *= 0.9
            if T < 1e-6:
                T = 1e-6

    # ----------------------------------------------------------------------
    # Return the best solution found within the time budget.
    # ----------------------------------------------------------------------
    return (best_h, best_score, n)
