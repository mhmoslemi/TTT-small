import numpy as np, time, math, random, sys, warnings
from numpy.fft import fft, ifft

# helper to compute C5 using FFT (full correlation) for speed
def compute_c5(h):
    n = h.size
    size = 2 * n
    dx = 2.0 / n
    fh = fft(h, size)
    f1mh = fft(1.0 - h, size)
    corr = ifft(fh * np.conj(f1mh)).real * dx
    # correlation length is size; the valid lags are -n+1 .. n-1
    # np.max over all entries gives the required max overlap
    return float(np.max(corr))

# project vector onto [0,1]^n with exact sum = n/2
def normalize_h(h):
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # simple iterative redistribution
    for _ in range(1000):
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            inc = diff / np.count_nonzero(mask)
            inc = min(inc, 0.5)
            h[mask] += inc
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            dec = (-diff) / np.count_nonzero(mask)
            dec = min(dec, 0.5)
            h[mask] -= dec
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    return h

# single‑temperature annealing step, preserving sum exactly
def propose_swap(h, temperature):
    n = h.size
    i, j = random.sample(range(n), 2)
    hi, hj = h[i], h[j]
    delta_min = max(-hi, hj - 1.0)
    delta_max = min(1.0 - hi, hj)
    if delta_min >= delta_max:
        return None, None, None
    # scale interval by temperature (cooling)
    width = delta_max - delta_min
    delta = delta_min + width * random.random()
    # optionally shrink moves as temperature drops
    delta *= temperature
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h, i, j

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    best_overall_h = None
    best_overall_c5 = math.inf
    # try a few different discretizations; allocate time roughly equally
    candidate_ns = [80, 120, 200]
    time_per_n = budget_s / len(candidate_ns) * 0.95

    for n in candidate_ns:
        # initialise h
        if 'initial_h_values' in globals():
            init = np.asarray(initial_h_values, dtype=float)
            m = init.size
            x_init = np.linspace(0, 2, m, endpoint=False)
            x_grid = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_grid, x_init, init)
        else:
            h = np.random.rand(n)
        h = normalize_h(h)

        current_c5 = compute_c5(h)
        best_h = h.copy()
        best_c5 = current_c5

        # annealing parameters
        temp0 = 0.5
        temp_end = 0.001
        iter_cnt = 0
        max_iter = 200000  # upper bound; will break on time

        while time.time() - start_time < budget_s - 0.5 and iter_cnt < max_iter:
            elapsed = time.time() - start_time
            # linear temperature schedule per n
            prog = min(1.0, (elapsed) / time_per_n)
            temperature = temp0 * (1 - prog) + temp_end * prog

            proposal, i, j = propose_swap(h, temperature)
            if proposal is None:
                continue
            new_c5 = compute_c5(proposal)

            accept = False
            if new_c5 < current_c5:
                accept = True
            else:
                # Metropolis criterion
                delta = new_c5 - current_c5
                if math.exp(-delta / (temperature + 1e-12)) > random.random():
                    accept = True

            if accept:
                h = proposal
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            iter_cnt += 1

            # occasional early exit if time scarce
            if time.time() - start_time > budget_s - 0.5:
                break

        # keep global best
        if best_c5 < best_overall_c5:
            best_overall_c5 = best_c5
            best_overall_h = best_h.copy()

    # final sanity: ensure constraints
    if best_overall_h is None:
        # fallback uniform
        n = 100
        best_overall_h = np.full(n, 0.5)
        best_overall_c5 = compute_c5(best_overall_h)

    best_overall_h = normalize_h(best_overall_h)
    best_overall_c5 = compute_c5(best_overall_h)

    return best_overall_h, best_overall_c5, best_overall_h.size
