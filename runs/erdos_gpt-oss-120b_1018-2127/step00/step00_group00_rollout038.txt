import time
import numpy as np

def _project_to_constraints(v, target_sum):
    """Clip v to [0,1] and adjust to have exact sum = target_sum."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(20):  # enough iterations for convergence
        cur = v.sum()
        diff = cur - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:  # sum too large, reduce from positive entries
            pos = v > 0
            total = v[pos].sum()
            if total == 0:
                break
            reduction = diff * (v[pos] / total)
            v[pos] -= reduction
        else:          # sum too small, add to entries below 1
            under = v < 1
            capacity = (1 - v[under]).sum()
            if capacity == 0:
                break
            addition = (-diff) * ((1 - v[under]) / capacity)
            v[under] += addition
        v = np.clip(v, 0.0, 1.0)
    # final guard against tiny violations
    v = np.clip(v, 0.0, 1.0)
    return v

def _random_feasible(n, target_sum):
    """Generate a random vector in [0,1]^n with sum = target_sum."""
    # start from exponential draws
    x = np.random.exponential(scale=1.0, size=n)
    x = x / x.sum() * target_sum
    return _project_to_constraints(x, target_sum)

def _evaluate_c5(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize the
    Erdős overlap constant C5.

    Returns
    -------
    h_values : ndarray
        Optimized samples of h, length n_points.
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretization points used.
    """
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        n = int(len(initial_h_values))
        h_start = np.array(initial_h_values, dtype=float)
    else:
        n = 45                     # default size when no parent solution
        h_start = _random_feasible(n, n / 2.0)

    target_sum = n / 2.0           # because integral over [0,2] must be 1
    dx = 2.0 / n

    # Ensure starting point respects constraints
    h_start = _project_to_constraints(h_start, target_sum)
    best_h = h_start.copy()
    best_score = _evaluate_c5(best_h, dx)

    start_time = time.time()
    time_limit = budget_s * 0.95   # leave a small safety margin

    # Parameters for stochastic local search
    step_scale = 0.4               # initial noise magnitude
    min_step = 1e-4
    decay = 0.9995                 # gradual cooling

    iteration = 0
    while time.time() - start_time < time_limit:
        iteration += 1

        # Occasionally restart from a fresh random point
        if np.random.rand() < 0.02:
            cand = _random_feasible(n, target_sum)
        else:
            cand = best_h + step_scale * np.random.randn(n)
            cand = _project_to_constraints(cand, target_sum)

        score = _evaluate_c5(cand, dx)

        if score < best_score:
            best_score = score
            best_h = cand
            # tighten step size when improvement found
            step_scale = max(step_scale * 0.97, min_step)

        # slowly cool down the search
        step_scale = max(step_scale * decay, min_step)

        # safety break if extremely good bound reached (theoretical limit ~0.38)
        if best_score < 0.379:
            break

    return best_h, float(best_score), n
