import numpy as np
import random

def _project_to_feasible(h, target_sum, eps=1e-12, max_iter=1000):
    """Adjust h in-place so that sum(h)=target_sum while keeping 0<=h<=1."""
    cur = h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(cur - target_sum) <= eps:
            break
        if cur > target_sum:
            # need to decrease total mass
            idx = np.where(h > 0)[0]
            if idx.size == 0:
                break
            i = random.choice(idx)
            delta = min(h[i], (cur - target_sum) / 1.0)
            h[i] -= delta
            cur -= delta
        else:
            # need to increase total mass
            idx = np.where(h < 1)[0]
            if idx.size == 0:
                break
            i = random.choice(idx)
            delta = min(1 - h[i], (target_sum - cur) / 1.0)
            h[i] += delta
            cur += delta
    return h

def _c5_value(h, dx):
    """Compute the C5 bound for a given h vector."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _feasible_swap(h, step):
    """Perform a random swap that preserves the sum exactly and respects bounds."""
    n = h.size
    i, j = random.sample(range(n), 2)
    hi, hj = h[i], h[j]
    # allowable delta so that both stay in [0,1]
    lower = max(-hi, hj - 1.0)
    upper = min(1.0 - hi, hj)
    if lower >= upper:
        return False  # no feasible move
    delta = random.uniform(lower, upper) * step
    # ensure delta stays within tighter bounds
    delta = max(lower, min(upper, delta))
    h[i] = hi + delta
    h[j] = hj - delta
    return True
