import time
import numpy as np

def _objective(h, dx):
    """Maximum overlap integral for a given step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_bounds_sum(v, target):
    """Clip to [0,1] and adjust to satisfy sum(v)=target (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(20):
        cur = v.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # tiny residual correction
    residual = target - v.sum()
    if abs(residual) > 1e-12:
        for i in range(len(v)):
            if residual > 0 and v[i] < 1:
                add = min(residual, 1.0 - v[i])
                v[i] += add
                residual -= add
            elif residual < 0 and v[i] > 0:
                sub = min(-residual, v[i])
                v[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return v

def _interpolate(h, new_n):
    """Linear interpolation of h from its current length to new_n points."""
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def _hill_climb(h_start, dx, deadline):
    """Simple simulated‑annealing hill‑climber respecting the constraints."""
    n = len(h_start)
    target = n / 2.0
    rng = np.random.default_rng()

    best_score = _objective(best, dx)
    cur = best.copy()
    cur_score = best_score
    T = 1.0
    it = 0

        # step size adapts mildly with problem size
