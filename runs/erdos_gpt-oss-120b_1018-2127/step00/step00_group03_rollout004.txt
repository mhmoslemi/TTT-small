import time, random, numpy as np

def _c5_via_correlation(arr):
    """Fallback C5 computation if evaluate_erdos_solution is unavailable."""
    dx = 2.0 / arr.size
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * dx)

def _compute_c5(arr):
    """Try the provided evaluator, otherwise fall back."""
    try:
        return float(evaluate_erdos_solution(arr))
    except Exception:  # pragma: no cover
        return _c5_via_correlation(arr)

def _project_to_feasible(vec):
    """Project a vector onto the simplex {0≤x_i≤1, sum x_i = n/2}."""
    n = vec.size
    target = n / 2.0
    # Clip first
    vec = np.clip(vec, 0.0, 1.0)
    # Simple iterative correction
    excess = vec.sum() - target
    if abs(excess) < 1e-12:
        return vec
    # Work only on interior entries to keep bounds
    interior = (vec > 0.0) & (vec < 1.0)
    if interior.any():
        vec[interior] -= excess / interior.sum()
        vec = np.clip(vec, 0.0, 1.0)
    # Final tiny fix on the first element
    diff = target - vec.sum()
    vec[0] = np.clip(vec[0] + diff, 0.0, 1.0)
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C₅ problem."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # ---- initial construction --------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64)
    else:
        n = 70
        h = np.full(n, 0.5, dtype=np.float64)  # already satisfies sum = n/2
    n = h.size
    h = _project_to_feasible(h)

    best_h = h.copy()
    best_score = _compute_c5(best_h)

    # Early exit if the given construction already beats the target
    TARGET = 0.3808
    if best_score <= TARGET:
        return (best_h.tolist(), float(best_score), n)

    # ---- stochastic hill‑climbing -----------------------------------------
    while time.time() - start_time < budget_s * 0.95:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # allowed move range keeping both entries in [0,1]
        delta_min = -min(best_h[j], 1.0 - best_h[i])
        delta_max =  min(best_h[i], 1.0 - best_h[j])
        if delta_min >= delta_max:
            continue
        # propose a modest move
        delta = np.random.uniform(delta_min, delta_max) * 0.5
        if abs(delta) < 1e-8:

        candidate = best_h.copy()
        candidate[i] -= delta
        candidate[j] += delta

        score = _compute_c5(candidate)

        if score < best_score:
            best_h = candidate
            best_score = score
