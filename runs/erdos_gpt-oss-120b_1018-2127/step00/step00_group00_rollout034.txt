import time
import numpy as np
from scipy.optimize import minimize

def _c5_from_h(h):
    """Return C5 value for a discretized h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def _interpolate_h(old_h, new_n):
    """Resize a step vector to a new length using linear interpolation."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    # enforce integral constraint exactly
    target_sum = new_n / 2.0
    cur_sum = new_h.sum()
    if cur_sum > 0:
        new_h *= target_sum / cur_sum
    new_h = np.clip(new_h, 0.0, 1.0)
    # re‑adjust any leftover mass
    diff = target_sum - new_h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly over interior points that are not at bounds
        free = (new_h > 0.0) & (new_h < 1.0)
        if free.any():
            new_h[free] += diff / free.sum()
            new_h = np.clip(new_h, 0.0, 1.0)
        else:
            # fallback: add to first element respecting bound
            new_h[0] = min(1.0, max(0.0, new_h[0] + diff))
    return new_h

def _optimize_at_resolution(init_h, max_iter=200, max_time=None):
    """Run a few SLSQP restarts starting from init_h."""
    n = init_h.shape[0]
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    best_h = init_h.copy()
    best_val = _c5_from_h(best_h)

    rnd = np.random.RandomState(0)

    start = time.time()
    for restart in range(5):
        # random perturbation respecting bounds
        pert = best_h + 0.05 * rnd.randn(n)
        pert = np.clip(pert, 0.0, 1.0)
        # enforce the sum constraint by a simple scaling + shift
        cur = pert.sum()
        if cur != 0:
            pert = pert * (target_sum / cur)
        pert = np.clip(pert, 0.0, 1.0)

        res = minimize(
            fun=lambda x: _c5_from_h(x),
            x0=pert,
            method='SLSQP',
            bounds=bounds,
            constraints=[eq_cons],
            options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False}
        )
        if not res.success:
            continue
        val = _c5_from_h(res.x)
        if val < best_val:
            best_val = val
            best_h = res.x.copy()
        if max_time is not None and (time.time() - start) > max_time:
            break
    return best_h, best_val
