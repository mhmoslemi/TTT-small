import time
import numpy as np
import random

def _repair(vec, target_sum):
    """Clip to [0,1] and scale to satisfy sum == target_sum."""
    v = np.clip(vec, 0.0, 1.0)
    total = v.sum()
    if total == 0:
        return np.full_like(v, target_sum / v.size)
    v = v * (target_sum / total)
    # repeat clipping+scaling a few times to respect bounds
    for _ in range(3):
        v = np.clip(v, 0.0, 1.0)
        total = v.sum()
        if abs(total - target_sum) < 1e-9:
            break
        v = v * (target_sum / total)
    # tiny residual correction
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        mask = (v < 1.0) if diff > 0 else (v > 0.0)
        idx = np.where(mask)[0]
        if idx.size:
            v[idx] += diff / idx.size
            v = np.clip(v, 0.0, 1.0)
    return v

def _evaluate(vec, dx):
    """Return C5 for a given step vector."""
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return dx * np.max(corr)

def _mutate(parent, target_sum, strategy):
    """Create a child from parent using strategy A (resample) or B (perturb)."""
    n = parent.size
    child = parent.copy()
    if strategy == "A":  # resample a fraction of entries
        frac = 0.20
        k = max(1, int(frac * n))
        idx = np.random.choice(n, k, replace=False)
        child[idx] = np.random.uniform(0.0, 1.0, size=k)
    else:               # perturb a fraction with small noise
        mask = np.random.rand(n) < 0.30
        noise = np.random.uniform(-0.02, 0.02, size=mask.sum())
        child[mask] += noise
    child = _repair(child, target_sum)
    return child

def _random_feasible(n, target_sum):
    """Generate a random feasible vector."""
    vec = np.random.rand(n)
    return _repair(vec, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) minimizing the Erdős C5 overlap."""
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    init = globals().get("initial_h_values", None)
    if init is not None:
        n = int(init.size)
        best = _repair(np.array(init, dtype=float), n / 2.0)
    else:
        n = 200
        best = _random_feasible(n, n / 2.0)

    target_sum = n / 2.0
    dx = 2.0 / n
    best_score = _evaluate(best, dx)

    start = time.time()
    max_time = budget_s * 0.95  # keep a small safety margin

    # Simple stochastic hill‑climbing with occasional restarts
    while time.time() - start < max_time:
        # choose mutation type
        strat = "A" if random.random() < 0.5 else "B"
        child = _mutate(best, target_sum, strat)
        score = _evaluate(child, dx)

        if score < best_score:
            best, best_score = child, score

        # tiny chance to restart from a fresh random point
        if random.random() < 0.008:
            cand = _random_feasible(n, target_sum)
            cand_score = _evaluate(cand, dx)
            if cand_score < best_score:
                best, best_score = cand, cand_score

    return best, float(best_score), n
