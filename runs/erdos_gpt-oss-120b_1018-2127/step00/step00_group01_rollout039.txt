import time
import numpy as np

def _compute_c5(h):
    """Return C5 for a step function given by array h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _feasible_initial(n, rng):
    """Create a feasible initial vector of length n."""
    if "initial_h_values" in globals():
        h0 = globals()["initial_h_values"]
        if h0.size == n:
            return h0.astype(float).copy()
        # interpolate to new grid
        x_old = np.linspace(0.0, 2.0, h0.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h_new = np.interp(x_new, x_old, h0)
        h_new = np.clip(h_new, 0.0, 1.0)
        total = h_new.sum()
        if total > 0:
            h_new *= (n / 2.0) / total
        h_new = np.clip(h_new, 0.0, 1.0)
        return h_new
    # fall back: random feasible vector
    vals = rng.random(n)
    vals = np.clip(vals, 0.0, 1.0)
    vals *= (n / 2.0) / vals.sum()
    return vals

def _propose_move(h, step, rng):
    """Transfer a small amount of mass between two coordinates."""
    n = h.size
    i, j = rng.choice(n, 2, replace=False)
    max_from_i = min(step, h[i])
    max_to_j = min(step, 1.0 - h[j])
    eps = rng.uniform(0.0, min(max_from_i, max_to_j))
    if eps == 0.0:
        return h
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new
