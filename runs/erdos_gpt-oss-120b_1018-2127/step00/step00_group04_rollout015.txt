import time
import numpy as np
import random

def _c5_of_h(h, dx):
    """Compute C5 = max correlation of h with (1-h)."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _initial_binary(n, rng):
    """Create a random binary vector of length n with sum = n/2."""
    ones = n // 2
    h = np.zeros(n, dtype=float)
    idx = rng.permutation(n)[:ones]
    h[idx] = 1.0
    return h

def _make_binary_from_initial(initial, n, rng):
    """Upsample or downsample an initial vector to length n and binarize."""
    init = np.asarray(initial, dtype=float)
    if len(init) == n:
        h = (init > 0.5).astype(float)
    else:
        # linear interpolation to target size
        xs = np.linspace(0, 2, len(init), endpoint=False)
        xt = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(xt, xs, init)
        h = (h_interp > 0.5).astype(float)
    # enforce exact half ones
    cur_ones = int(h.sum())
    target = n // 2
    if cur_ones > target:
        # turn some ones to zeros
        ones_idx = np.where(h == 1)[0]
        to_flip = rng.choice(ones_idx, cur_ones - target, replace=False)
        h[to_flip] = 0.0
    elif cur_ones < target:
        zeros_idx = np.where(h == 0)[0]
        to_flip = rng.choice(zeros_idx, target - cur_ones, replace=False)
        h[to_flip] = 1.0
    return h
