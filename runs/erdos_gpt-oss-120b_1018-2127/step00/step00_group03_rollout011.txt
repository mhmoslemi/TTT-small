import time
import numpy as np
import cvxpy as cp

# The harness provides evaluate_erdos_solution and optionally initial_h_values
# evaluate_erdos_solution(h) -> float or (float, ...) where the first element is C5

def _project_feasible(vec, target_sum):
    """Return the Euclidean projection of vec onto {0<=x<=1, sum(x)=target_sum}."""
    n = vec.shape[0]
    x = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(x - vec))
    constraints = [x >= 0, x <= 1, cp.sum(x) == target_sum]
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    if x.value is None:
        # fallback: simple clipping and scaling (rare)
        proj = np.clip(vec, 0, 1)
        s = proj.sum()
        if s == 0:
            proj[:] = target_sum / n
        else:
            proj *= target_sum / s
        proj = np.clip(proj, 0, 1)
        # Renormalize once more if needed
        diff = target_sum - proj.sum()
        if abs(diff) > 1e-9:
            # distribute the tiny difference uniformly
            proj += diff / n
            proj = np.clip(proj, 0, 1)
        return proj
    return np.array(x.value, dtype=np.float64)


def _c5_of(h):
    """Compute the overlap bound C5 for a candidate h."""
    val = evaluate_erdos_solution(h)
    if isinstance(val, (list, tuple, np.ndarray)):
        return float(val[0])
    return float(val)


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with area 1 that minimizes C5.

    Returns:
        h_values (np.ndarray): best found vector of length n_points.
        c5_bound (float): corresponding C5 value.
        n_points (int): length of the vector.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
        target_sum = n / 2.0
        h_current = _project_feasible(h0, target_sum)
    else:
        n = 70  # fallback size; reasonable trade‑off between quality and speed
        target_sum = n / 2.0
        # start from a simple uniform distribution respecting bounds
        h_current = _project_feasible(rng.random(n), target_sum)

    best_h = h_current.copy()
    best_c5 = _c5_of(best_h)

    # Annealing parameters
    sigma = 0.2               # initial std dev of perturbations
    sigma_min = 0.005
    decay = 0.995
    iter_count = 0

    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:
            break

        # generate a candidate by perturbing a random subset
        cand = best_h.copy()
        # pick a random number of coordinates to modify (at least 1)
        k = rng.integers(1, max(2, n // 10))
        idx = rng.choice(n, size=k, replace=False)
        cand[idx] += sigma * rng.standard_normal(k)

        # project back to feasible region
        cand = _project_feasible(cand, target_sum)

        c5 = _c5_of(cand)
        iter_count += 1

        # Accept if better
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand
            # optional early exit if we beat target
            if best_c5 < 0.3808:
                break

        # Occasionally try a completely random restart to escape local minima
        if iter_count % (n * 5) == 0:
            random_vec = rng.random(n)
            random_vec = _project_feasible(random_vec, target_sum)
            c5_rand = _c5_of(random_vec)
            if c5_rand < best_c5:
                best_c5, best_h = c5_rand, random_vec

        # Reduce perturbation scale gradually
        sigma = max(sigma * decay, sigma_min)

    # Final evaluation to ensure consistency
    best_c5 = _c5_of(best_h)
