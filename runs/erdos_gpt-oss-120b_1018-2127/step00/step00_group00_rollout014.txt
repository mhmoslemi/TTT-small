import time
import numpy as np
import cvxpy as cp  # imported for potential fallback, not used in core loop

# pre-imported in the harness:
# evaluate_erdos_solution, initial_h_values

def _project_onto_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the box [0,1]^n and the affine
    hyperplane sum(x) = target using a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate(h: np.ndarray) -> float:
    """Wrapper around the provided evaluator; fall back to manual computation if missing."""
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        # manual computation: correlation of h with (1-h)
        n = h.size
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        return float(corr.max())


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Stochastic local search for a step function h on [0,2] that minimises the
    Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best found vector (values in [0,1], sum = n/2).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Length of the vector (grid resolution).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine initial size and vector
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 45
        h0 = np.full(n0, 0.5)  # uniform satisfies sum = n/2
    n = h0.size
    target_sum = n / 2.0

    # Ensure feasibility of the start
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_onto_bounds_sum(h0, target_sum)

    best_h = h0.copy()
    best_score = _evaluate(best_h)

    # Parameters for stochastic search
    max_iters = 200000  # large upper bound; real stop is time budget
    initial_sigma = 0.15
    final_sigma = 0.005

    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed + 0.2 >= budget_s:
            break
        it += 1
        # linearly anneal sigma
        sigma = initial_sigma - (initial_sigma - final_sigma) * (it / max_iters)
        if sigma < final_sigma:
            sigma = final_sigma

        # propose a new candidate
        cand = best_h + sigma * np.random.randn(n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_onto_bounds_sum(cand, target_sum)

        score = _evaluate(cand)

        if score < best_score:
            best_h = cand
            best_score = score

        # occasional random restart to avoid stagnation
        if it % 5000 == 0:
            rnd = np.random.rand(n)
            rnd = _project_onto_bounds_sum(rnd, target_sum)
            rnd_score = _evaluate(rnd)
            if rnd_score < best_score:
                best_h = rnd
                best_score = rnd_score

    return best_h, best_score, n
