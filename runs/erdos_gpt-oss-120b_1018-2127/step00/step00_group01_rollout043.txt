import time
import numpy as np
from scipy.optimize import differential_evolution

def _correlation_max(h):
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx for discrete h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # full convolution gives length 2n-1, centre at shift 0 at index n-1
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _project_to_simplex(v, target_sum):
    """Project vector v onto [0,1] box and then adjust to have given sum."""
    # clip to box
    v = np.clip(v, 0.0, 1.0)
    current = v.sum()
    if np.isclose(current, target_sum):
        return v
    # simple scaling and redistribution
    if current == 0:
        # distribute evenly
        return np.full_like(v, target_sum / v.size)
    scale = target_sum / current
    v *= scale
    # after scaling some entries may exceed 1 or go below 0 again
    # iterate a few times to fix
    for _ in range(5):
        over = v > 1
        under = v < 0
        if not over.any() and not under.any():
            break
        excess = v[over] - 1
        deficit = -v[under]
        v[over] = 1
        v[under] = 0
        total_excess = excess.sum()
        total_deficit = deficit.sum()
        net = total_excess - total_deficit
        if np.abs(net) < 1e-12:
            break
        # redistribute remaining discrepancy uniformly among interior points
        interior = (~over) & (~under)
        if interior.any():
            v[interior] -= net / interior.sum()
        else:
            # all at bounds, just distribute to all
            v -= net / v.size
    # final clip
    return np.clip(v, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine grid size
    try:
        init_vec = np.asarray(initial_h_values, dtype=float)
        n = init_vec.size
        # ensure even size for exact sum target
        if n % 2 == 1:
            # pad with a zero to make even (adds negligible effect)
            init_vec = np.append(init_vec, 0.0)
            n += 1
    except NameError:
        n = 100  # fallback modest size, even by construction
        init_vec = None

    target_sum = n / 2  # because ∑h = n/2 gives integral 1

    # Helper to evaluate with penalty for feasibility
    penalty_weight = 1e6

    def objective(x):
        # x is raw vector, may violate sum constraint
        # evaluate penalty
        pen = penalty_weight * abs(x.sum() - target_sum)
        # compute correlation max (no scaling needed)
        return _correlation_max(x) + pen

    # Prepare DE bounds
    bounds = [(0.0, 1.0) for _ in range(n)]

    # Build initial population
    popsize = 15
    init_pop = []
    # include the provided initial vector (projected)
    if init_vec is not None:
        proj = _project_to_simplex(init_vec, target_sum)
        init_pop.append(proj)
    # fill rest with random feasible vectors
    while len(init_pop) < popsize:
        rnd = rng.random(n)
        proj = _project_to_simplex(rnd, target_sum)
        init_pop.append(proj)
    init_pop = np.array(init_pop)

    # Run differential evolution with time guard
    max_iter = 200
    best_h = None
    best_val = np.inf

    def time_exceeded():
        return (time.time() - start_time) >= budget_s * 0.65  # reserve 35% for refinement

    de_result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=max_iter,
        popsize=popsize,
        init=init_pop,
        tol=1e-7,
        updating='deferred',
        polish=False,
        disp=False,
        seed=seed,
        callback=lambda xk, convergence: None
    )
    # capture best from DE
    best_h = _project_to_simplex(de_result.x, target_sum)
    best_val = _correlation_max(best_h)

    # Simple stochastic refinement (hill climbing) within remaining budget
    while not time_exceeded():
        # perturb current best
        step_scale = 0.05
        cand = best_h + rng.normal(scale=step_scale, size=n)
        cand = _project_to_simplex(cand, target_sum)
        val = _correlation_max(cand)
        if val < best_val:
            best_h, best_val = cand, val
        # occasionally reduce step size
        if rng.random() < 0.01:
            step_scale *= 0.9
        # break if already very low (the known target)
        if best_val <= 0.3808:
            break

    # Final projection to guarantee exact sum
    best_h = _project_to_simplex(best_h, target_sum)
    final_c5 = _correlation_max(best_h)

    return best_h, float(final_c5), int(n)
