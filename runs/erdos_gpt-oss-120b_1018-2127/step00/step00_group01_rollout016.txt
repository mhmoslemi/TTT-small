import numpy as np
import time

def _compute_c5(h):
    """Return C5 for a given h (array of length n)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
