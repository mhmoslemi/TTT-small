import numpy as np
import time

def _c5_score(h, dx):
    """Compute C5 = max_k ∫ h(x)(1−h(x+k)) dx via discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_move(h, step):
    """Propose a feasible move that keeps the total sum unchanged."""
    n = h.size
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    # delta can be positive (transfer from j to i) or negative.
    delta = (np.random.rand() - 0.5) * 2.0 * step
    if delta > 0:
        delta = min(delta, 1.0 - h[i], h[j])
    else:
        delta = -min(-delta, h[i], 1.0 - h[j])
    if abs(delta) < 1e-12:
        return None, None, None
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, i, j

def _hill_climb(h, dx, budget, start_time):
    """Simple simulated‑annealing style hill climbing."""
    best_h = h.copy()
    best_score = _c5_score(best_h, dx)

    # parameters
    max_step = 0.2          # maximal transferred mass at start
    min_step = 0.001        # never go below this
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget - 0.5:
            break
        # linear cooling for step size
        t = 1.0 - elapsed / budget
        step = max(min_step, max_step * t)

        h_new, _, _ = _random_move(h, step)
        if h_new is None:
            continue
        new_score = _c5_score(h_new, dx)

        # greedy accept if better, otherwise Metropolis with temperature proportional to t
        if new_score < best_score:
            h = h_new
            best_h = h_new.copy()
            best_score = new_score
        else:
            # temperature schedule (scale 0.01·t to keep acceptance low)
            prob = np.exp((best_score - new_score) / (0.01 * max(t, 1e-3)))
            if np.random.rand() < prob:
                h = h_new

    return best_h, best_score

def _refine_resolution(h_coarse, n_target, budget, start_time):
    """Interpolate a coarse solution to a finer grid and improve it."""
    n_coarse = h_coarse.size
    x_coarse = np.linspace(0.0, 2.0, n_coarse, endpoint=False)
    x_fine = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)

    # enforce feasibility: clip to [0,1] then adjust to exact sum using a simple redistribution
    h_fine = np.clip(h_fine, 0.0, 1.0)
    target_sum = n_target / 2.0
    diff = target_sum - h_fine.sum()
    # distribute the difference uniformly while staying in bounds
    while abs(diff) > 1e-8:
        add = diff / n_target
        h_fine += add
        np.clip(h_fine, 0.0, 1.0, out=h_fine)
        new_diff = target_sum - h_fine.sum()
        if abs(new_diff - diff) < 1e-12:
            break
        diff = new_diff

    dx = 2.0 / n_target
    return _hill_climb(h_fine, dx, budget, start_time)
