import time
import numpy as np

def _repair(h, target, max_iter=15):
    """Clip to [0,1] and adjust to reach the required sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (h < 1.0)
            if not free.any():
                break
            add = diff / free.sum()
            h[free] += add
        else:
            free = (h > 0.0)
            if not free.any():
                break
            sub = diff / free.sum()
            h[free] += sub
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h, dx):
    """Compute the C5 value for a given step function."""
    # correlation h with 1-h, full mode
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _anneal(start_h, dx, target, time_budget, rng):
    """Simple simulated‑annealing on the feasible region."""
    best_h = start_h.copy()
    best_c = _c5_value(best_h, dx)
    cur_h = best_h.copy()
    cur_c = best_c

    start = time.time()
    # annealing schedule parameters
    init_temp = 0.02
    final_temp = 1e-4
    init_step = 0.15

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_budget:
            break
        # temperature linearly decreasing
        t = init_temp * (1 - elapsed / time_budget) + final_temp * (elapsed / time_budget)
        step = init_step * (1 - elapsed / time_budget) + 1e-4 * (elapsed / time_budget)

        # propose perturbation
        proposal = cur_h + rng.normal(scale=step, size=cur_h.shape)
        proposal = _repair(proposal, target)

        prop_c = _c5_value(proposal, dx)

        # acceptance test
        if prop_c < cur_c or rng.random() < np.exp(-(prop_c - cur_c) / max(t, 1e-12)):
            cur_h, cur_c = proposal, prop_c

        if prop_c < best_c:
            best_h, best_c = proposal.copy(), prop_c

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
    h_values is a numpy array of length n_points satisfying the constraints,
    and c5_bound is the achieved maximal overlap integral.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: 70 uniformly spaced points
        h0 = np.full(70, 0.5, dtype=float)
    n = h0.size
    dx = 2.0 / n
    target_sum = n / 2.0  # because ∑h = target_sum  ↔  integral = 1

    # Ensure starting point is feasible
    h0 = _repair(h0, target_sum)

    best_h, best_c = h0.copy(), _c5_value(h0, dx)

    start_time = time.time()
    # allocate time for a few independent runs
    total_budget = budget_s - 0.5  # small safety margin
    # number of restarts (including the initial one)
    n_restarts = 5
    # budget per restart (equally split, last may get leftovers)
    for i in range(n_restarts):
        now = time.time()
        remaining = total_budget - (now - start_time)
        if remaining <= 0:
            break
        # allocate a fraction; ensure at least 5 seconds per run
        run_budget = max(5.0, remaining / (n_restarts - i))

        if cand_c < best_c:
            best_h, best_c = cand_h, cand_c

    return best_h, float(best_c), n
