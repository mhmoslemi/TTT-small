def run(seed=42, budget_s=1100, **kwargs):

    # Number of discretization points

        n_points = 70

    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate(h):
        """Return C5 for a given step function h."""
        corr = np.correlate(h, 1.0 - h, mode="full")
        return dx * np.max(corr)

    # ---------- initial binary candidate ----------
    def random_binary():
        """Generate a random binary vector with exactly target_sum ones."""
        h = np.zeros(n_points, dtype=float)
        h[:int(target_sum)] = 1.0
        rng.shuffle(h)
        return h

    current_h = random_binary()
    current_c5 = evaluate(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # ---------- optimisation parameters ----------
    start_time = time.time()
    max_time = budget_s * 0.95          # keep a safety margin
    max_no_improve = 4000               # restart after this many failed tries
    no_improve = 0
    temp_initial = 0.02                 # simulated‑annealing temperature

    # Pre‑compute index pools for speed
    all_indices = np.arange(n_points)

    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break

        # simulated‑annealing temperature schedule
        temperature = temp_initial * (1.0 - elapsed / max_time)

        # pick a 1‑position and a 0‑position and swap them
        ones = np.where(current_h > 0.5)[0]
        zeros = np.where(current_h <= 0.5)[0]
        if ones.size == 0 or zeros.size == 0:

        i = rng.choice(ones)
