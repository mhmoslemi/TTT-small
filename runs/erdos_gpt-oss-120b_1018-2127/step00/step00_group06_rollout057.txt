import time
import numpy as np

def compute_c5(h):
    """Return max overlap integral approximation for step function h."""
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all lags
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair(h):
    """Project h onto [0,1] box with exact sum = n/2."""
    n = len(h)
    target = n / 2.0
    # initial clipping
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        # indices that are not at a bound
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds: distribute uniformly then re‑clip
            h += diff / n
        else:
            h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def random_walk(h, scale):
    """Add Gaussian noise scaled to step size."""
    return h + np.random.randn(len(h)) * scale

def interpolate_solution(old_h, new_n):
    """Linear interpolation of old_h (length m) to new_n points."""
    m = len(old_h)
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # ensure feasibility after interpolation
    new_h = repair(new_h)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function to minimise the Erdős C5 overlap bound."""
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # initialise first grid
    try:
        init = initial_h_values  # provided by harness if any
        n0 = len(init)
        start_h = repair(np.array(init, dtype=float))
    except NameError:
        n0 = 100  # fallback size
        start_h = np.full(n0, 0.5)  # uniform feasible start
    best_h = start_h.copy()
    best_score = compute_c5(best_h)
    # ------------------------------------------------------------------
    # grid progression (doubling up to a reasonable cap)
    grid_sizes = [n0, n0 * 2, n0 * 4, n0 * 8]
    max_n = 800  # hard cap to keep work manageable
    grid_sizes = [n for n in grid_sizes if n <= max_n]
    # ------------------------------------------------------------------
    for n in grid_sizes:
        # time check before each grid
        if time.time() - start_time > budget_s * 0.98:
            break
        # obtain starting solution for this grid
        if n == n0:
            cur_h = best_h.copy()
        else:
            cur_h = interpolate_solution(best_h, n)
        cur_score = compute_c5(cur_h)
        # simulated‑annealing style search
        temp = 1.0
        scale = 0.02  # base perturbation magnitude
        # adapt scale to grid size
        scale *= (n / n0) ** 0.2
        # number of iterations proportional to remaining time and inversely to n
        remaining = budget_s - (time.time() - start_time)
        iters = int(min(200000, max(2000, remaining * 200 / n)))
        for _ in range(iters):
            if time.time() - start_time > budget_s * 0.99:
                break
            cand = random_walk(cur_h, scale)
            cand = repair(cand)
            cand_score = compute_c5(cand)
            # accept if better or with Metropolis probability
            if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / temp):
                cur_h, cur_score = cand, cand_score
            # keep global best
            if cur_score < best_score:
                best_h = cur_h.copy()
                best_score = cur_score
            # cool temperature
            temp *= 0.9997
        # after finishing this grid, set best_h as the seed for next
    # final safety: ensure exact integral constraint
    best_h = repair(best_h)
    best_score = compute_c5(best_h)
    return best_h, best_score, len(best_h)
