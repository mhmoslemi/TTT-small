import time
import numpy as np

def _repair(vec, target):
    """Project vec onto [0,1] with sum == target (simple iterative scheme)."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        if abs(s - target) < 1e-12:
            break
        # distribute the deficit/surplus uniformly
        v += (target - s) / v.size
        v = np.clip(v, 0.0, 1.0)
    # final scaling if still off (rare)
    s = v.sum()
    if s > 0:
        v *= target / s
        v = np.clip(v, 0.0, 1.0)
    # tiny residual fix
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        for i in range(v.size):
            if diff > 0 and v[i] < 1.0:
                add = min(diff, 1.0 - v[i])
                v[i] += add
                diff -= add
            elif diff < 0 and v[i] > 0.0:
                sub = min(-diff, v[i])
                v[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return v

def _c5_value(h):
    """Compute C5 for a discrete step function h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h:[0,2]→[0,1] with ∫h=1 to minimize
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    # Determine resolution
    base_h = globals().get('initial_h_values')
    if base_h is not None:
        n0 = int(base_h.size)
    else:
        n0 = 70
    n_points = kwargs.get('n_points', n0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0          # because ∫h = 1 ⇒ sum(h)*dx = 1

    # Build initial candidate
    if base_h is not None:
        # Upsample / downsample linearly to required size
        xs_src = np.linspace(0, 2, base_h.size, endpoint=False)
        xs_tgt = np.linspace(0, 2, n_points, endpoint=False)
        h_curr = np.interp(xs_tgt, xs_src, base_h)
    else:
        h_curr = np.full(n_points, 0.5)   # satisfies sum automatically

    h_curr = _repair(h_curr, target_sum)
    best_h = h_curr.copy()
    best_score = _c5_value(best_h)

    # Parameters for stochastic search
    sigma = 0.2                     # initial noise scale
    min_sigma = 1e-4
    decay = 0.9995
    patience = 2000                # iterations without improvement before random restart
    no_improve = 0
    t_start = time.time()

    while True:
        if time.time() - t_start > budget_s - 0.5:
            break

        # Propose new candidate
        cand = best_h + rng.normal(0.0, sigma, n_points)
        cand = _repair(cand, target_sum)

        score = _c5_value(cand)

        if score < best_score:
            best_h = cand
            best_score = score
            sigma = max(min_sigma, sigma * 0.98)   # tighten after improvement
            no_improve = 0
        else:
            no_improve += 1
            sigma = max(min_sigma, sigma * decay)

        # Random restart if stuck
        if no_improve >= patience:
            cand = rng.random(n_points)          # fresh random point
            cand = _repair(cand, target_sum)
            cand_score = _c5_value(cand)
            if cand_score < best_score:
                best_h = cand
                best_score = cand_score
                sigma = 0.2
            no_improve = 0

    return best_h, float(best_score), int(n_points)
