import numpy as np
import time

def _project_to_constraints(h, target_sum, max_iter=10, atol=1e-10):
    """Clip h to [0,1] and enforce sum == target_sum (approximately)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target_sum - h.sum()
        if abs(diff) < atol:
            break
        # indices that are not at a bound can absorb the adjustment
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
        else:
            # everything is at a bound – distribute uniformly (will be re‑clipped)
            h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h):
    """Return the C5 bound for a given step vector h."""
    # correlation length = 2*len(h)-1, mode='full' gives all shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / h.size
    return corr.max() * dx

def _random_neighbor(h, step_scale=0.1):
    """Create a small random perturbation of h and re‑project."""
    n = h.size
    # pick a few indices to modify
    k = max(1, n // 20)                     # ~5% of entries
    idx = np.random.choice(n, k, replace=False)
    delta = np.random.randn(k) * step_scale
    h_new = h.copy()
    h_new[idx] += delta
    return h_new

def _optimize_one_grid(init_h, budget, start_time):
    """Hill‑climbing / simulated‑annealing on a single grid size."""
    n = init_h.size
    target_sum = n / 2.0
    h = _project_to_constraints(init_h, target_sum)
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # annealing schedule
    temp_start = 0.5
    temp_end = 0.001
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget:
            break
        # linear cooling
        temp = temp_start + (temp_end - temp_start) * (elapsed / budget)
        # generate neighbor
        cand = _random_neighbor(h, step_scale=0.2 * temp)
        cand = _project_to_constraints(cand, target_sum)
        cand_c5 = _c5_value(cand)
        # accept if better or with probability exp(-(Δ)/temp)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand.copy(), cand_c5
        if cand_c5 < _c5_value(h) or np.random.rand() < np.exp(-(cand_c5 - _c5_value(h)) / max(temp, 1e-12)):
            h = cand
        it += 1
    return best_h, best_c5
