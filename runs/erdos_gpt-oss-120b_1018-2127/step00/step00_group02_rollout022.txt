import time
import numpy as np
import random

def compute_c5(h):
    """Return the overlap bound C5 for a discrete step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def repair(h):
    """Project h back onto the feasible set 0≤h≤1 and sum(h)=n/2."""
    n = len(h)
    target = n / 2.0
    # clip to bounds
    np.clip(h, 0.0, 1.0, out=h)
    # adjust the total mass iteratively
    for _ in range(10):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        # distribute diff uniformly over non‑saturated entries
        mask = (h > 0.0) & (h < 1.0)
        cnt = np.count_nonzero(mask)
        if cnt == 0:          # all at bounds – distribute to all entries
            mask[:] = True
            cnt = n
        add = diff / cnt
        h[mask] += add
        np.clip(h, 0.0, 1.0, out=h)
    # final clip to be safe
    np.clip(h, 0.0, 1.0, out=h)
    return h

def upscale(h, new_n):
    """Linearly interpolate h to a new length new_n."""
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős overlap constant C5.

    Returns:
        (h_values, c5_bound, n_points)
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # --------------------------------------------------------------
    # 1. Initialise a feasible vector
    # --------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        init_n = len(h0)
    else:
        init_n = 45
        h0 = np.zeros(init_n)
        h0[:init_n // 2] = 1.0

    # target resolution (trading quality vs speed)
    target_n = 180
    h = upscale(h0, target_n)
    h = repair(h)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # --------------------------------------------------------------
    # 2. Simple stochastic local search
    # --------------------------------------------------------------
    max_time = budget_s - 1.0      # a small safety margin
    it = 0
    while time.time() - start_time < max_time:
        it += 1
        # Adaptive noise level – slowly cools
        sigma = 0.12 * (0.995 ** (it // 1000))
        cand = best_h + sigma * np.random.randn(target_n)
        cand = repair(cand)
        score = compute_c5(cand)

        if score < best_score:

            best_h = cand

        # occasional diversification

            cand = repair(cand)
            s2 = compute_c5(cand)
            if s2 < best_score:
                best_score = s2
                best_h = cand
