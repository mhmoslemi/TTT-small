import time
import numpy as np
import random

def _objective(h):
    """Return C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h):
    """Project h onto the feasible set 0≤h≤1 and sum(h)=n/2."""
    n = h.shape[0]
    desired = n / 2.0

    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)

    # enforce the sum constraint by redistributing the excess uniformly
    # over the non‑boundary entries; iterate a few times for stability.
    for _ in range(5):
        total = h.sum()
        diff = desired - total
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt_free = free.sum()
        if cnt_free == 0:
            # all entries are at a bound – rescale globally and re‑clip
            if total > 0:
                h *= desired / total
            h = np.clip(h, 0.0, 1.0)
            continue
        adjust = diff / cnt_free
        h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
    # final clean‑up: one global scaling if still off by a tiny amount
    total = h.sum()
    if abs(total - desired) > 1e-10 and total > 0:
        h *= desired / total
        h = np.clip(h, 0.0, 1.0)
    return h

def _random_walk(h, sigma):
    """Perturb h with Gaussian noise of given sigma."""
    return h + np.random.randn(h.shape[0]) * sigma

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h to minimise the Erdős C5 overlap bound."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # initialise from the provided parent if it exists
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # fallback: simple half‑filled step function (integral = 1)
        n_fallback = 100
        h = np.concatenate([np.ones(n_fallback // 2), np.zeros(n_fallback // 2)])
        h = h.astype(float)

    n = h.shape[0]
    h = _repair(h)
    cur_score = _objective(h)

    best_h = h.copy()
    best_score = cur_score

    # hyper‑parameters
    sigma = 0.02          # base perturbation magnitude
    max_iters = 200000   # upper bound; loop will break earlier by time

    it = 0
    while it < max_iters and (time.time() - start_time) < budget_s * 0.95:
        # gradually shrink sigma as we progress
        cur_sigma = sigma * (0.99 ** (it // 1000))
        h_new = _random_walk(h, cur_sigma)
        h_new = _repair(h_new)
        new_score = _objective(h_new)

        # accept if improvement (simple hill‑climbing)
        if new_score < cur_score:
            h = h_new
            cur_score = new_score

        # store the global best
        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score

        it += 1

    # final safety check – ensure feasibility of the returned vector
    best_h = _repair(best_h)
    best_score = _objective(best_h)

    return best_h, best_score, best_h.shape[0]
