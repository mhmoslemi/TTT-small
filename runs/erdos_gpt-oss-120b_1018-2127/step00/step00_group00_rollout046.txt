import time
import numpy as np
from scipy.optimize import minimize

def _dx(n):
    return 2.0 / n

def compute_score(h, n):
    """Return C5 for a given step vector h of length n."""
    dx = _dx(n)
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _make_feasible(x, n):
    """Clip to [0,1] and enforce sum(x)=n/2 by a simple redistribution."""
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    cur = x.sum()
    if np.isclose(cur, target, atol=1e-8):
        return x
    # distribute the difference uniformly, respecting bounds
    diff = target - cur
    # iterative projection
    for _ in range(10):
        if np.abs(diff) < 1e-10:
            break
        inc = diff / n
        x += inc
        # enforce bounds again
        x = np.clip(x, 0.0, 1.0)
        new_cur = x.sum()
        diff = target - new_cur
    return x

def transfer_solution(old_h, new_n):
    """Resample old_h (defined on len(old_h)) to length new_n using linear interpolation."""
    old_n = len(old_h)
    # original grid points in [0,2)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h, period=2.0)
    return _make_feasible(new_h, new_n)

def _optimize_start(h0, n, rng):
    """Run SLSQP from h0 and return (h, score)."""
    bounds = [(0.0, 1.0)] * n
    constraints = {"type": "eq", "fun": lambda x: x.sum() - n / 2.0}
    opts = {"maxiter": 600, "ftol": 1e-9, "disp": False}
    res = minimize(
        lambda x: compute_score(x, n),
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    if not res.success:
        return None, None
    h_opt = _make_feasible(res.x, n)
    return h_opt, compute_score(h_opt, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_score = np.inf
    best_h = None
    best_n = None

    # possible resolutions (keep ≤180 to stay small)
    n_list = [45, 60, 90, 120, 180]

    # detect if a parent construction is supplied
    has_parent = "initial_h_values" in globals()
    parent_h = globals().get("initial_h_values", None)

    for n in n_list:
        if time.time() - start > budget_s:
            break

        # ----- initial guess -----
        if has_parent and len(parent_h) == 45 and n == 45:
            h0 = parent_h.copy()
        elif has_parent:
            # use parent as base and resample
            h0 = transfer_solution(parent_h, n)
        else:
            # random feasible vector
            rand = rng.random(n)
            rand = _make_feasible(rand, n)
            h0 = rand

        # evaluate and possibly improve
        cur_score = compute_score(h0, n)
        if cur_score < best_score:
            best_score, best_h, best_n = cur_score, h0.copy(), n

        # main local optimisation
        h_opt, sc_opt = _optimize_start(h0, n, rng)
        if h_opt is not None and sc_opt < best_score:
            best_score, best_h, best_n = sc_opt, h_opt.copy(), n

        # a few cheap random restarts
        for _ in range(3):
            if time.time() - start > budget_s:
                break
            rand = rng.random(n)
            rand = _make_feasible(rand, n)
            h_opt, sc_opt = _optimize_start(rand, n, rng)
            if h_opt is not None and sc_opt < best_score:
                best_score, best_h, best_n = sc_opt, h_opt.copy(), n

    # safety fallback: if nothing succeeded, fall back to uniform half‑interval
    if best_h is None:
        n = 45
        h = np.zeros(n)
        half = n // 2
        h[:half] = 1.0
        best_h = _make_feasible(h, n)
        best_score = compute_score(best_h, n)
        best_n = n

    return best_h, best_score, best_n
