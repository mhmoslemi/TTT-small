import numpy as np
import time
import math

def evaluate_h(h):
    """Return C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using zero padding (full mode)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_feasible(h, target_sum):
    """Clip to [0,1] and renormalise to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # equally distribute the mass
        return np.full_like(h, target_sum / h.size)
    # simple scaling; after scaling we may exceed bounds once more, so iterate a couple times
    for _ in range(3):
        scale = target_sum / cur
        h = np.clip(h * scale, 0.0, 1.0)
        cur = h.sum()
    # final tiny adjustment to hit target sum exactly (within floating error)
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # distribute diff uniformly on indices not at a bound
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def initial_vector(n):
    """Create a simple feasible start: half the entries 1, half 0 (shuffled)."""
    h = np.zeros(n)
    ones = n // 2
    h[:ones] = 1.0
    np.random.shuffle(h)
    return h.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine discretisation size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = kwargs.get('n_points', 100)
        h0 = initial_vector(n)

    target_sum = n / 2.0          # because sum(h) must equal n/2
    h_current = repair_feasible(h0, target_sum)
    best_h = h_current.copy()
    best_score = evaluate_h(best_h)

    # Parameters for the stochastic search
    max_iter = 2000000           # safety upper bound
    max_delta_initial = 0.2      # initial mass moved in a single step
    min_delta = 1e-4

    # Temperature schedule (simulated annealing style)
    def temperature(frac):
        # start high, cool to near zero
        return max(0.01, (1.0 - frac) ** 3)

    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if iter_count >= max_iter:
            break
        iter_count += 1
        # gradually shrink the move size
        frac = iter_count / max_iter
        max_delta = max_delta_initial * (1.0 - frac) + min_delta * frac

        # propose a move that conserves total mass
        i, j = rng.integers(0, n, size=2)
        # ensure i has something to give and j can receive
        attempts = 0
        while (h_current[i] <= 1e-12 or h_current[j] >= 1.0 - 1e-12) and attempts < 5:
            i, j = rng.integers(0, n, size=2)
            attempts += 1
        if h_current[i] <= 1e-12 or h_current[j] >= 1.0 - 1e-12:
            # fallback: random perturbation of a single coordinate and re‑repair
            h_new = h_current + (rng.random(n) - 0.5) * max_delta * 0.1
            h_new = repair_feasible(h_new, target_sum)
        else:
            delta = rng.random() * max_delta
            delta = min(delta, h_current[i], 1.0 - h_current[j])
            h_new = h_current.copy()
            h_new[i] -= delta
            h_new[j] += delta

        # quick feasibility check (bounds are already respected)
        score_new = evaluate_h(h_new)

        # acceptance rule
        if score_new < best_score:
            best_score = score_new
            best_h = h_new.copy()
        # Metropolis acceptance
        if score_new < evaluate_h(h_current):
            accept = True
        else:
            temp = temperature(frac)
            prob = math.exp(-(score_new - evaluate_h(h_current)) / (temp + 1e-12))
            accept = rng.random() < prob
        if accept:
            h_current = h_new

        # occasional random restart to escape stagnation
        if iter_count % 20000 == 0:
            if rng.random() < 0.3:
                h_current = repair_feasible(initial_vector(n), target_sum)

    # final repair to guarantee exact feasibility
    best_h = repair_feasible(best_h, target_sum)

    return (best_h, best_score, n)
