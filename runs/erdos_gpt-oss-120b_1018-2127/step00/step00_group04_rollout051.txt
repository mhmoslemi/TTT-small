import time
import numpy as np

def _compute_max_overlap(h):
    """Return the maximal overlap sum (without dx scaling)."""
    # correlation h with (1-h)
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimizes a binary step function h on [0,2] with ∫h = 1.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        # Ensure binary
        h0 = (h0 > 0.5).astype(float)
    else:
        n = 200                      # default size, even
        # start with half ones uniformly spaced
        h0 = np.zeros(n, dtype=float)
        h0[::2] = 1.0                # simple alternating pattern

    # Enforce exact sum constraint (half ones)
    half = n // 2
    ones_idx = np.flatnonzero(h0 == 1.0)
    zeros_idx = np.flatnonzero(h0 == 0.0)

    # If count mismatched, randomly adjust
    if ones_idx.size > half:
        drop = rng.choice(ones_idx, size=ones_idx.size - half, replace=False)
        h0[drop] = 0.0
        ones_idx = np.flatnonzero(h0 == 1.0)
        zeros_idx = np.flatnonzero(h0 == 0.0)
    elif ones_idx.size < half:
        add = rng.choice(zeros_idx, size=half - ones_idx.size, replace=False)
        h0[add] = 1.0

    # Current state
    h_cur = h0.copy()
    cur_score = _compute_max_overlap(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score

    # Simple greedy hill‑climbing with occasional random restarts
    max_iters = 10_000_000  # upper bound; actual stops by time
    it = 0
    while time.time() - start_time < budget_s * 0.95 and it < max_iters:
        it += 1
        # pick a 1 and a 0 to swap
        ones = np.flatnonzero(h_cur == 1.0)
        zeros = np.flatnonzero(h_cur == 0.0)
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        h_candidate = h_cur.copy()
        h_candidate[i] = 0.0
        h_candidate[j] = 1.0

        cand_score = _compute_max_overlap(h_candidate)

        # Accept only if improves (greedy)
        if cand_score < cur_score:
            h_cur = h_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_h = h_candidate.copy()
                best_score = cand_score

        # occasional random restart to avoid stagnation
        if it % 50000 == 0:
            # perturb a small fraction of bits
            perturb = rng.choice(n, size=n // 20, replace=False)
            h_cur[perturb] = 1.0 - h_cur[perturb]
            # re‑balance to exact half ones
            ones_now = np.flatnonzero(h_cur == 1.0)
            if ones_now.size > half:
                drop = rng.choice(ones_now, size=ones_now.size - half, replace=False)
                h_cur[drop] = 0.0
            elif ones_now.size < half:
                zeros_now = np.flatnonzero(h_cur == 0.0)
                add = rng.choice(zeros_now, size=half - ones_now.size, replace=False)
                h_cur[add] = 1.0
            cur_score = _compute_max_overlap(h_cur)

    # Compute final C5 bound with dx scaling
    dx = 2.0 / n
    c5_bound = best_score * dx
    return best_h, float(c5_bound), n
