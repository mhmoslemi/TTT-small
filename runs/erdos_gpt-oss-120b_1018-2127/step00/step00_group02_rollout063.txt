import time
import numpy as np
from math import inf

def compute_c5(h):
    """Return C5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_box_simplex(v, target_sum, max_iter=100, tol=1e-10):
    """
    Euclidean projection of v onto {x | 0 <= x <= 1, sum x = target_sum }.
    Uses bisection on the Lagrange multiplier theta.
    """
    lo = -np.max(v)          # makes v+theta <= 0  -> proj = 0
    hi = 1.0 - np.min(v)     # makes v+theta >= 1  -> proj = 1
    for _ in range(max_iter):
        theta = (lo + hi) / 2.0
        x = np.clip(v + theta, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s < target_sum:
            lo = theta
        else:
            hi = theta
    # final projection
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # tiny correction to hit target_sum exactly (within numeric limits)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        # try to distribute diff to a non‑saturated coordinate
        free = np.where((x > 0.0 + 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size == 0:
            free = np.arange(x.size)
        idx = free[0]
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def random_feasible(n, target_sum, rng):
    """Generate a random vector in [0,1]^n with sum == target_sum."""
    # start from exponential variables (Dirichlet) then project
    v = rng.exponential(scale=1.0, size=n)
    v = v / v.sum() * target_sum
    return project_box_simplex(v, target_sum)

def hill_climb(start_h, target_sum, budget_s, rng):
    """Simple stochastic hill‑climbing on C5."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_score = compute_c5(best_h)
    current_h = best_h.copy()
    current_score = best_score

    start_time = time.time()
    iter_cnt = 0
    no_improve = 0
    max_no_improve = 2000

    # step size schedule (in absolute h-value units)
    max_step = 0.2
    min_step = 0.001

    while time.time() - start_time < budget_s:
        elapsed = time.time() - start_time
        # linear decay of step
        step = max_step - (max_step - min_step) * (elapsed / budget_s)
        # propose a move: pick two indices and transfer delta
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(-step, step)
        # enforce box constraints in the transfer
        hi_i = current_h[i]
        lo_i = -hi_i          # can decrease at most hi_i
        hi_j = 1.0 - current_h[j]
        lo_j = -current_h[j]  # can decrease at most current_h[j]
        delta = np.clip(delta, max(lo_i, -hi_j), min(hi_i, hi_j))
        if delta == 0.0:
            continue
        proposal = current_h.copy()
        proposal[i] -= delta
        proposal[j] += delta
        # projection is not needed because we respected bounds and sum
        score = compute_c5(proposal)
        iter_cnt += 1
        if score < current_score - 1e-12:
            current_h = proposal
            current_score = score
            no_improve = 0
            if score < best_score - 1e-12:
                best_h = proposal.copy()
                best_score = score
        else:
            no_improve += 1

        # random restart if stuck
        if no_improve >= max_no_improve:
            current_h = random_feasible(n, target_sum, rng)
            current_score = compute_c5(current_h)
            no_improve = 0

    return best_h, best_score

def upscale_vector(old_h, new_n):
    """Linear interpolation of a periodic step function onto a finer grid."""
    old_n = old_h.shape[0]
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xs_new, xs_old, old_h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine initial resolution and vector
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        target_sum = n0 / 2.0
        # ensure feasibility
        h0 = project_box_simplex(h0, target_sum)
    else:
        n0 = 45
        target_sum = n0 / 2.0
        h0 = random_feasible(n0, target_sum, rng)

    # Allocate time slices: 70% for first resolution, 30% for upscale (if any)
    t1 = budget_s * 0.7
    best_h, best_score = hill_climb(h0, target_sum, t1, rng)

    # Try a higher resolution if time permits
    remaining = budget_s - (time.time() - (time.time() - budget_s + t1))  # approximate
    if remaining > 30.0:
        # choose a finer grid (e.g., 60 points)
        n1 = max(60, int(n0 * 1.4))
        upscale = upscale_vector(best_h, n1)
        target_sum1 = n1 / 2.0
        upscale = project_box_simplex(upscale, target_sum1)
        t2 = remaining
        best_h, best_score = hill_climb(upscale, target_sum1, t2, rng)
        n0 = n1

    # Return as list for compatibility
    return best_h.tolist(), float(best_score), int(n0)
