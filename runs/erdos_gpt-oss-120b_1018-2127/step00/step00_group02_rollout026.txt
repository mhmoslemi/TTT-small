import numpy as np
import time

def _compute_c5(h):
    """Return C5 for a given step function sampled on [0,2]."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return conv.max()

def _project_to_constraints(h, target_sum):
    """Project h onto [0,1] box and enforce sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # iterative redistribution among non‑boundary entries
    for _ in range(200):
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # cannot move any further; break to avoid division by zero
            break
        delta = diff / cnt
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
    # tiny residual correction by uniform shift (may violate bounds negligibly)
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _neighbor_candidate(best, step, rng):
    """Generate a random neighbor of best respecting constraints via projection."""
    cand = best + rng.normal(scale=step, size=best.size)
    target_sum = best.size / 2.0
    cand = _project_to_constraints(cand, target_sum)
    return cand
