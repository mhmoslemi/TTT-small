import time
import numpy as np

def evaluate_erdos(h):
    """Compute C5 for a discretized step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def random_exchange(h, rng, max_move=0.1):
    """Propose a new h by moving a feasible amount between two random indices."""
    n = len(h)
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    hi, hj = h[i], h[j]

    # feasible delta so that 0<=hi+delta<=1 and 0<=hj-delta<=1
    delta_min = max(-hi, hj - 1.0)          # most negative allowed
    delta_max = min(1.0 - hi, hj)           # most positive allowed
    if delta_min >= delta_max:
        return None  # no feasible move (rare)
    # limit magnitude to keep steps moderate
    bound = min(max_move, delta_max - delta_min)
    delta = rng.uniform(delta_min, delta_max)
    if delta > bound:
        delta = bound
    elif delta < -bound:
        delta = -bound

    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

def project_to_constraints(h, target_sum):
    """Project onto box [0,1] and exact sum by simple clipping & redistribution."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # redistribute the difference uniformly (small adjustments)
    if abs(diff) > 1e-12:
        # find indices not at bounds in the direction of needed adjustment
        if diff > 0:
            eligible = np.where(h < 1.0)[0]
        else:
            eligible = np.where(h > 0.0)[0]
        if eligible.size > 0:
            add = diff / eligible.size
            h[eligible] += add
            h = np.clip(h, 0.0, 1.0)  # final clip
    return h

    else:
        # start from a simple half‑filled block
        h = np.zeros(n_points, dtype=float)
