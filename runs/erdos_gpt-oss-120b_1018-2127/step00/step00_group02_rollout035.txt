import numpy as np
import time

def evaluate_h(h):
    """Return the C5 value for a step function h sampled on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum_and_bounds(h, target_sum):
    """Project h onto [0,1] and enforce exact sum by uniform redistribution."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # add uniform correction; may violate bounds slightly, so iterate once more
    h += diff / h.size
    h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    cur = h.sum()
    if abs(cur - target_sum) > 1e-12:
        correction = (target_sum - cur) / h.size
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_max_shift(h):
    """Sub‑gradient of the maximal overlap with respect to h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    idxs = np.where(corr == max_val)[0]
    grad = np.zeros_like(h)
    for k in idxs:
        shift = k - (n - 1)        # shift index: positive => h_i pairs with h_{i+shift}
        if shift >= 0:
            # i from 0 to n-shift-1
            grad[: n - shift] += (1.0 - h[shift:]) - h[: n - shift]
        else:
            s = -shift
            # i from s to n-1
            grad[s:] += (1.0 - h[: n - s]) - h[s:]
    grad /= max(1, idxs.size)
    return grad

def local_optimize(h0, max_iter, step0, rng):
    """Projected sub‑gradient descent on the max‑overlap objective."""
    n = h0.size
    target_sum = n / 2.0
    h = h0.copy()
    for it in range(max_iter):
        grad = gradient_max_shift(h)
        # simple step‑size schedule
        step = step0 * (0.995 ** it)
        h -= step * grad
        h = project_sum_and_bounds(h, target_sum)
    return h, evaluate_h(h)

def upscale(h, new_n):
    """Linear interpolation of a sampled function to a finer grid."""
    old_n = h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, h)
    # enforce exact integral after interpolation
    target_sum = new_n / 2.0
    new_h = project_sum_and_bounds(new_h, target_sum)
    return new_h
