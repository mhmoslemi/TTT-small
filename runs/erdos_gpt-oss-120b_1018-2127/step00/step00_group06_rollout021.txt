import numpy as np
import time
import math

def _evaluate_h(h):
    """Return C5 bound for a given h vector."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) in full mode
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _binary_from_continuous(v):
    """Convert a continuous vector with sum = n/2 to a binary {0,1} vector with same sum."""
    n = v.shape[0]
    half = n // 2
    # indices of the largest half entries become 1
    idx = np.argpartition(v, -half)[-half:]
    bin_vec = np.zeros(n, dtype=np.float64)
    bin_vec[idx] = 1.0
    return bin_vec

def _random_binary(n, rng):
    """Create a random binary vector with exactly n/2 ones."""
    half = n // 2
    vec = np.zeros(n, dtype=np.float64)
    vec[:half] = 1.0
    rng.shuffle(vec)
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine grid size
    init = globals().get("initial_h_values", None)
    if init is not None:
        n_points = int(init.shape[0])
        # Ensure even size for exact integral
        if n_points % 2 == 1:
            n_points += 1
    else:
        n_points = 200  # default even size

    # Build initial candidate
    if init is not None:
        h_curr = np.array(init, dtype=np.float64)
        if h_curr.shape[0] != n_points:
            # interpolate/resize if needed (simple linear interpolation)
            x_old = np.linspace(0, 2, h_curr.shape[0], endpoint=False)
            x_new = np.linspace(0, 2, n_points, endpoint=False)
            h_curr = np.interp(x_new, x_old, h_curr)
        # enforce bounds and integral exactly
        h_curr = np.clip(h_curr, 0.0, 1.0)
        target_sum = n_points / 2.0
        diff = target_sum - h_curr.sum()
        # adjust uniformly within bounds
        if abs(diff) > 1e-9:
            # simple projection: add/subtract a tiny constant then re‑clip
            delta = diff / n_points
            h_curr += delta
            h_curr = np.clip(h_curr, 0.0, 1.0)
            # final fix by scaling the free entries
            free = (h_curr > 0) & (h_curr < 1)
            if free.any():
                scale = (target_sum - h_curr[~free].sum()) / h_curr[free].sum()
                h_curr[free] *= scale
        # make binary for easier search
        h_curr = _binary_from_continuous(h_curr)
    else:
        h_curr = _random_binary(n_points, rng)

    # Simulated annealing parameters
    temp = 1.0
