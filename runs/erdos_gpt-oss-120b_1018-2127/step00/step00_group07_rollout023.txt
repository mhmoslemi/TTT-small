import time, math, random
import numpy as np

def _evaluate(h):
    """Return C5 for step function h on [0,2] using correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _repair_sum(h, target):
    """Adjust h so that sum(h) == target while staying in [0,1]."""
    max_iter = 10
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            h[idx] = np.minimum(1.0, h[idx] + inc)
        else:
            # need to remove mass
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            dec = diff / idx.size  # diff is negative
            h[idx] = np.maximum(0.0, h[idx] + dec)  # adding negative dec
    # final clip (safety)
    np.clip(h, 0.0, 1.0, out=h)
    # tiny correction by scaling if still off (rare)
    s = h.sum()
    if abs(s - target) > 1e-9:
        if s > 0:
            h *= target / s
        else:
            # all zeros – distribute evenly
            h[:] = target / h.size
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _perturb(h, scale=0.1):
    """Create a nearby feasible h by moving a small amount between two indices."""
    n = h.shape[0]
    i, j = random.randint(0, n - 1), random.randint(0, n - 1)
    while j == i:
        j = random.randint(0, n - 1)
    delta = (random.random() - 0.5) * 2.0 * scale  # in [-scale,scale]
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    np.clip(h_new, 0.0, 1.0, out=h_new)
    target = n / 2.0
    h_new = _repair_sum(h_new, target)
    return h_new

def _simulated_annealing(start_h, time_limit, start_temp=1.0, cooling=0.9995, scale=0.1):
    """Run SA until time_limit seconds have elapsed, return best (h,score)."""
    best_h = start_h.copy()
    best_score = _evaluate(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    T = start_temp
    start = time.time()
    while time.time() - start < time_limit:
        cand = _perturb(cur_h, scale=scale)
        cand_score = _evaluate(cand)
        if cand_score < cur_score or random.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_h, cur_score = cand, cand_score
            if cur_score < best_score:
                best_h, best_score = cur_h.copy(), cur_score
        T *= cooling
        # occasional scale adaptation
        if random.random() < 0.01:
            scale = max(0.01, min(0.3, scale * (1.05 if cand_score < cur_score else 0.95)))
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ----- initial construction -----
