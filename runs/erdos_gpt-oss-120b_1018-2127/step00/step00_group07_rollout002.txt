import time
import numpy as np

def _compute_c5(h, scale):
    """Return C5 for the step function h with given dx scale (dx = 2/n)."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * scale

def _repair(h, target_sum, max_iter=5):
    """Project h onto the box [0,1] and enforce sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        # scale proportionally, then re‑clip
        if s == 0:
            # distribute uniformly if everything zero
            h[:] = target_sum / len(h)
            break
        h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
    # final adjustment for tiny residual error
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # add/subtract diff to a non‑boundary element
        idx = np.argmax((h > 0) & (h < 1))
        if diff > 0:
            h[idx] = min(1.0, h[idx] + diff)
        else:
            h[idx] = max(0.0, h[idx] + diff)
    return h

def _initial_h():
    """Obtain a feasible starting vector."""
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        # no parent construction – start with a simple half‑filled box
        n = 80
        h0 = np.concatenate([np.ones(n // 2), np.zeros(n // 2)])
    n = len(h0)
    target = n / 2.0
    h0 = _repair(h0, target)
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 bound.

    Returns
    -------
    h_values : ndarray
        Optimised samples (length n_points) in [0,1] with sum == n_points/2.
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points.
    """
    np.random.seed(seed)
    start_time = time.time()

    h = _initial_h()
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = h.copy()
    best_score = _compute_c5(best_h, dx)

    # Parameters for the stochastic search
    step_scale = 0.2          # initial magnitude of perturbations
    cooling_rate = 0.995
    min_step = 1e-4
    max_iters = 10_000_000   # safety upper bound

    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:      # leave margin for finalisation
            break
        if iter_count >= max_iters:
            break

        # generate candidate by adding Gaussian noise
        cand = best_h + np.random.normal(scale=step_scale, size=n)
        cand = _repair(cand, target_sum)

        # evaluate
        score = _compute_c5(cand, dx)

        # keep improvement
        if score < best_score:
            best_score = score
            best_h = cand.copy()
            # if we already beat the target, stop early
            if best_score <= 0.3808:
                break

        # simulated‑annealing style acceptance
        else:
            # probability to accept worse move
            prob = np.exp((best_score - score) / (step_scale + 1e-12))
            if np.random.rand() < prob:
                best_h = cand.copy()
                best_score = score

        # cooling of step size
        step_scale = max(min_step, step_scale * cooling_rate)

    best_h = _repair(best_h, target_sum)
