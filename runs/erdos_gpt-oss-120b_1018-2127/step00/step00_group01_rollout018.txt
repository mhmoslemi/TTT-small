import time, math
import numpy as np

def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Return the overlap constant C5 for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] (discretised) to minimise the
    Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimised samples of h, length n_points.
    c5_bound : float
        The achieved C5 value.
    n_points : int
        Number of discretisation points.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # 1. Initialise the discretisation size and the starting vector.
    # ------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = h.size
    else:
        n = 200                     # fallback size – still cheap to evaluate
        h = rng.random(n)
        # scale to satisfy the integral constraint exactly (sum = n/2)
        h *= (n / 2) / h.sum()
        # clip to [0,1] while preserving the sum (very small adjustments)
        np.clip(h, 0.0, 1.0, out=h)
        excess = (h.sum() - n / 2)
        if abs(excess) > 1e-12:
            # distribute the excess uniformly, respecting bounds
            for _ in range(10):
                adjust = excess / n
                h += adjust
                np.clip(h, 0.0, 1.0, out=h)
                excess = (h.sum() - n / 2)
                if abs(excess) < 1e-12:
                    break

    dx = 2.0 / n

    cur_c5 = _compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------
    # 2. Simulated‑annealing style local search.
    # ------------------------------------------------------------
    max_iter = 250000
    init_temp = 0.02                # initial temperature for Metropolis criterion
    step_scale = 0.05               # maximal mass moved in a single step

    for it in range(max_iter):
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        # decide direction of transfer: from i to j
        give = h[i]
        receive = 1.0 - h[j]
        max_transfer = min(step_scale, give, receive)
        if max_transfer <= 0.0:
            continue

        eps = rng.random() * max_transfer

        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps

        new_c5 = _compute_c5(h_new, dx)
        delta = new_c5 - cur_c5

        # temperature schedule (linear cooling)
        temp = init_temp * (1.0 - it / max_iter)
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / (temp + 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

        # occasional random perturbation to escape plateaus
        if (it + 1) % 50000 == 0:
            # replace a small random block with fresh noise then renormalise
            block_start = rng.integers(0, n)
            block_len = min(rng.integers(5, 20), n - block_start)
            h[block_start:block_start + block_len] = rng.random(block_len)
            # renormalise to exact sum while keeping bounds
            h *= (n / 2) / h.sum()
            np.clip(h, 0.0, 1.0, out=h)
            cur_c5 = _compute_c5(h, dx)
            if cur_c5 < best_c5:
                best_c5, best_h = cur_c5, h.copy()

    # ------------------------------------------------------------
    # 3. Return the best found solution.
    # ------------------------------------------------------------
    return best_h, best_c5, n
