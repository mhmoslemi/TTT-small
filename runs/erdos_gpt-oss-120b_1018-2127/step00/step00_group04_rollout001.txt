import time
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_to_feasible(v, target):
    """Project vector v onto the box [0,1]^n with exact sum = target."""
    v = np.clip(v, 0.0, 1.0)
    n = v.size
    for _ in range(200):  # enough for convergence
        excess = v.sum() - target
        if abs(excess) < 1e-12:
            break
        free = (v > 0.0 + 1e-14) & (v < 1.0 - 1e-14)
        n_free = free.sum()
        if n_free == 0:
            # all variables at a bound, cannot fix excess -> break
            break
        adjust = excess / n_free
        v[free] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    diff = v.sum() - target
    if abs(diff) > 1e-10:
        v += -diff / n
        v = np.clip(v, 0.0, 1.0)
    return v

def random_feasible(rng, n, target):
    """Generate a random feasible vector."""
    # start with uniform random values, then project
    v = rng.random(n)
    return project_to_feasible(v, target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine number of sample points
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        # fallback: use a modest size if no parent is provided
        n_points = 42
        h0 = None

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because integral must be 1

    # Initialise best solution
    if h0 is not None:
        cur = project_to_feasible(h0.copy(), target_sum)
    else:
        cur = random_feasible(rng, n_points, target_sum)

    best_h = cur.copy()
    best_c5 = compute_c5(best_h, dx)

    # Optimization parameters
    start_time = time.time()
    step = 0.25          # initial perturbation magnitude
    step_min = 1e-5
    decay = 0.995        # geometric decay per iteration
    iter_count = 0
    max_iters = 10_000_000  # safety upper bound

    while time.time() - start_time < budget_s * 0.99 and iter_count < max_iters:
        iter_count += 1

        # Occasionally perform a full random restart
        if rng.random() < 0.001:
            cand = random_feasible(rng, n_points, target_sum)
        else:
            # Gaussian perturbation around current best
            perturb = rng.normal(scale=step, size=n_points)
            cand = project_to_feasible(best_h + perturb, target_sum)

        cand_c5 = compute_c5(cand, dx)

        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            # reset step size to allow larger moves after improvement
            step = max(step, 0.01)

        # Gradually shrink the step size
        step = max(step * decay, step_min)

    return best_h, best_c5, n_points
