import time
import numpy as np
from scipy.optimize import minimize

# compute the C5 value for a discrete h (full correlation, scaled by dx)
def _c5_of_h(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

# project a vector onto the feasible simplex with box [0,1] and given total mass
def _make_feasible(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    # simple iterative scaling + clipping
    for _ in range(5):
        cur = v.sum()
        if cur == 0:
            v[:] = target_sum / v.size
            break
        scale = target_sum / cur
        v *= scale
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if still off (distribute uniformly on non‑boundary entries)
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        free = (v > 0.0) & (v < 1.0)
        if free.any():
            v[free] += diff / free.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

# resize h to a new length using linear interpolation then make feasible
def _resize_h(old_h, new_len):
    old_len = old_h.size
    if old_len == new_len:
        return old_h.copy()
    # points in [0,1) for interpolation
    old_x = np.linspace(0.0, 1.0, old_len, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_len, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    target = new_len / 2.0
    return _make_feasible(new_h, target)

# run a single SLSQP optimization, respecting a time budget
def _optimize_once(init_h, budget):
    n = init_h.size
    target = n / 2.0
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target}
    bounds = [(0.0, 1.0)] * n

    opts = {"maxiter": 250, "ftol": 1e-9, "disp": False}
    # ensure we don't exceed the allotted time
    start = time.time()
    res = minimize(_c5_of_h, init_h, method="SLSQP",
                   bounds=bounds, constraints=[cons],
                   options=opts)
    elapsed = time.time() - start
    if elapsed > budget:
        # if optimizer ran too long, just return the current point
        return init_h, _c5_of_h(init_h)
    return res.x, _c5_of_h(res.x)
