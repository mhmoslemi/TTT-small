import time
import numpy as np
from scipy.optimize import minimize

# Helper: compute the C5 value for a discretized h
def compute_c5(h):
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

# Project a vector onto the feasible set {0≤h_i≤1, sum h_i = N/2}
def project_feasible(v, target_sum):
    # Clip to box
    v = np.clip(v, 0.0, 1.0)
    # Adjust the sum by a uniform shift, iterating a few times
    for _ in range(5):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # distribute diff equally among non‑boundary entries
        free = (v > 0.0) & (v < 1.0)
        n_free = free.sum()
        if n_free == 0:  # all at bounds, just spread over all
            v += diff / v.size
        else:
            v[free] += diff / n_free
        v = np.clip(v, 0.0, 1.0)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        N = h0.size
    else:
        N = 45
        h0 = np.full(N, 0.5, dtype=float)  # satisfies the integral constraint

    target_sum = N / 2.0
    dx = 2.0 / N

    # Objective: C5 + penalty for violating the sum constraint
    penalty_coeff = 1e3

    def objective(x):
        h = np.clip(x, 0.0, 1.0)
        c5 = compute_c5(h)
        sum_pen = (h.sum() - target_sum) ** 2
        return c5 + penalty_coeff * sum_pen

    bounds = [(0.0, 1.0)] * N
    best_h = None
    best_c5 = np.inf

    # Perform a few independent restarts within the time budget
    max_restarts = 8
    for restart in range(max_restarts):
        # Time check
        if time.time() - start_time > budget_s * 0.95:
            break

        if restart == 0 and 'initial_h_values' in globals():
            x0 = h0.copy()
        else:
            # Random perturbation of the initial feasible point
            noise = np.random.uniform(-0.2, 0.2, size=N)
            x0 = h0 + noise
            x0 = project_feasible(x0, target_sum)

        # Run a gradient‑free L‑BFGS‑B (uses numeric gradients)
        res = minimize(
            objective,
            x0,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxfun": 2000, "ftol": 1e-12, "maxiter": 500},
        )

        h_candidate = project_feasible(res.x, target_sum)
        c5_candidate = compute_c5(h_candidate)

        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate

    # Fallback to the initial construction if optimization failed to improve
    if best_h is None:
        best_h = project_feasible(h0, target_sum)
        best_c5 = compute_c5(best_h)

    # Final sanity clip (should already be feasible)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure exact sum by a tiny uniform adjustment
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        best_h += diff / N
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, float(best_c5), N
