import time, math, random
import numpy as np

def _c5_value(h):
    """Compute C5 = max correlation of h with (1-h) times dx."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_feasible(h):
    """Clip to [0,1] and enforce sum(h)=n/2 (dx*sum=1)."""
    n = h.shape[0]
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    # simple iterative adjustment
    for _ in range(5):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # final tweak on non‑bound entries
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def _init_candidates(n):
    """Generate a few diverse feasible starters."""
    cand = []

    # 1. use supplied initial vector if present
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if len(h0) == n:
            cand.append(_project_feasible(h0))

    # 2. sinusoidal pattern (average exactly 0.5)
    x = np.arange(n)
    sinusoid = 0.5 + 0.5 * np.sin(2.0 * math.pi * x / n)
    cand.append(_project_feasible(sinusoid))

    # 3. random binary with exactly n/2 ones
    bin_vec = np.zeros(n)
    idx = np.random.choice(n, n // 2, replace=False)
    bin_vec[idx] = 1.0
    cand.append(_project_feasible(bin_vec))

    # 4. random continuous vector, then project
    rand_vec = np.random.rand(n)
    cand.append(_project_feasible(rand_vec))

    return cand


def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    where h_values satisfies the constraints and c5_bound = max_k ∫ h(x)(1-h(x+k))dx.
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # resolution: use length of the given construction if any,
    # otherwise fall back to 45.
    n = len(initial_h_values) if "initial_h_values" in globals() else 45

    # generate starting candidates

    best_h = None
    best_score = float("inf")

    # evaluate initial candidates
    for h in candidates:
        score = _c5_value(h)
        if score < best_score:

            best_h = h.copy()

    # hill‑climbing parameters
    max_iters = 200000          # generous upper bound, will be cut by time
    report_interval = 5000

    it = 0
