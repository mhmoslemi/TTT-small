import time
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp

def _c5_of_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_feasible(y, n):
    """Euclidean projection of y onto {0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    x = cp.Variable(len(y))
    prob = cp.Problem(
        cp.Minimize(cp.sum_squares(x - y)),
        [x >= 0, x <= 1, cp.sum(x) == target],
    )
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    if x.value is None:
        # fallback: simple scaling and clipping
        z = np.clip(y, 0.0, 1.0)
        s = np.sum(z)
        if s == 0:
            z[: int(target)] = 1.0
        else:
            z *= target / s
        return np.clip(z, 0.0, 1.0)
    return np.asarray(x.value).flatten()

def _upsample(old, n_new):
    """Linear interpolation of old values on [0,2) to n_new points."""
    n_old = len(old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, old)

def _optimize(initial, n, max_iter, rng):
    """Run SLSQP from a feasible start."""
    bounds = [(0.0, 1.0)] * n
    cons = ({
        "type": "eq",
        "fun": lambda x: np.sum(x) - n / 2.0,
    },)

    def obj(x):
        return _c5_of_h(x)

    res = minimize(
        obj,
        initial,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    if not res.success:
        # try a quick repair and return the repaired point
        x = _project_feasible(res.x, n)
        return x, _c5_of_h(x)
    return res.x, res.fun

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points)
    where h_values is a 1‑D NumPy array representing the step function.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Target resolution – small enough for speed, large enough for quality
    n_points = 200

    # Build initial feasible vector
    if "initial_h_values" in globals():
        h0 = _upsample(initial_h_values, n_points)
    else:
        h0 = rng.random(n_points)
    h0 = _project_feasible(h0, n_points)
    best_h = h0
    best_c5 = _c5_of_h(best_h)

    # Keep the very first feasible solution as a fallback
    fallback = (best_h.copy(), best_c5)

    # Main search loop
    max_restarts = 1000  # hard cap; actual count limited by time
    for _ in range(max_restarts):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # Random perturbation around current best
        perturb = rng.normal(scale=0.15, size=n_points)
        candidate = best_h + perturb
        candidate = _project_feasible(candidate, n_points)

        # Local optimization (few SLSQP iterations)
        max_iter = 120
        opt_h, opt_c5 = _optimize(candidate, n_points, max_iter, rng)

        if opt_c5 < best_c5:
            best_h, best_c5 = opt_h, opt_c5

    # If nothing improved, return fallback
    if best_c5 >= fallback[1]:
        best_h, best_c5 = fallback

    # Final safety projection (ensuring exact sum)
    best_h = _project_feasible(best_h, n_points)

    return best_h, float(best_c5), n_points
