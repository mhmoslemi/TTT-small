import numpy as np
import time
from scipy.optimize import minimize

# helper to compute C5 for a given h and dx
def _c5_value(h, dx):
    # full cross-correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# project a vector onto [0,1] box and the affine hyperplane sum = target
def _repair(h, target, eps=1e-12, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) < eps:
            break
        # indices that can move (not at a bound)
        free = (h > eps) & (h < 1.0 - eps)
        nfree = free.sum()
        if nfree == 0:        # nothing can move, just distribute uniformly
            h += -diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / nfree
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction to meet sum exactly (may break bounds by <1e-12)
    s = h.sum()
    if abs(s - target) > eps:
        h += (target - s) / len(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h on [0,2], its C5 bound, and the number of points."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 70
        h0 = np.full(n, 0.5)  # dummy; will be repaired to satisfy sum
    dx = 2.0 / n
    target_sum = n / 2.0  # because sum * dx = 1

    # Ensure feasible starting point
    h0 = _repair(h0, target_sum)

    # Store best solution
    best_h = h0.copy()
    best_c5 = _c5_value(best_h, dx)

    # SLSQP setup
    bounds = [(0.0, 1.0)] * n
    constraints = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target_sum
    },)

    # Helper wrapper for optimizer (needs to return scalar)
    def objective(x):
        return _c5_value(x, dx)

    # Number of restarts we can afford
    restart = 0
    while time.time() - start_time < budget_s * 0.9:
        # Random perturbation of current best (or initial for first restart)
        if restart == 0:
            x0 = best_h.copy()
        else:
            perturb = rng.normal(scale=0.05, size=n)
            x0 = best_h + perturb
        x0 = _repair(x0, target_sum)

        # Run local optimization
        res = minimize(
            objective,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        )

        if res.success:

                best_c5 = c5_candidate
                best_h = h_candidate
