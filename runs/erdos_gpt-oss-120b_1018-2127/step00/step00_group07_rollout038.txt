import time
from math import isfinite

def evaluate_c5(h):
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx via discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def project_h(h):
    """Enforce h∈[0,1] and ∫h = 1 (i.e., sum = n/2)."""
