import time, math, numpy as np

def _compute_c5(vec, dx):
    # autocorrelation of vec with (1-vec), max over all shifts
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return float(np.max(corr) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Initialise a binary step function satisfying the integral constraint.
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        N = h0.size

        h = (h0 >= 0.5).astype(float)
        target = N // 2                     # required number of ones

            zeros = np.where(h == 0)[0]
            flip = rng.choice(zeros, size=diff, replace=False)
            h[flip] = 1.0
        elif diff < 0:
            ones = np.where(h == 1)[0]
            flip = rng.choice(ones, size=-diff, replace=False)
            h[flip] = 0.0
    else:
        N = 80                               # modest size, good speed
        target = N // 2
        h = np.zeros(N, dtype=float)
        ones_idx = rng.choice(N, size=target, replace=False)
        h[ones_idx] = 1.0

    dx = 2.0 / N
    cur_c5 = _compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5

    # ----------------------------------------------------------------------
    # Simulated annealing: keep the number of ones constant, swap a 1 with a 0.
    # ----------------------------------------------------------------------
    T_start = 0.12
    T_end = 1e-5
    max_iter = 300000
