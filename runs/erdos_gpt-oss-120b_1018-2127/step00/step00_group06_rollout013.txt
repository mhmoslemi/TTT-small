def _project_to_feasible(h, target_sum):
    """Project h onto the box [0,1] with equality sum(target_sum)."""
    # Simple capped simplex projection (iterative water-filling)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # indices that are strictly inside the box
        mask = (h > 0.0) & (h < 1.0)
        cnt = mask.sum()

            # redistribute uniformly (rare)
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / cnt
        h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_score(h, dx):
    """Compute C5 for a given h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _random_feasible(n, rng):

    # start with uniform random, then project
    h = rng.random(n)
    target = n / 2.0
    h = _project_to_feasible(h, target)
    return h
