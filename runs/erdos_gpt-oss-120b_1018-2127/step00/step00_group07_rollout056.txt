import time
import numpy as np

def _binarize(vector):
    """Return a binary vector with the same length and sum = n/2 (rounded down if odd)."""
    n = len(vector)
    target = n // 2
    # argsort descending
    idx = np.argsort(-vector)
    bin_vec = np.zeros(n, dtype=np.float64)
    bin_vec[idx[:target]] = 1.0
    return bin_vec

def _compute_c5(h):
    """Compute the C5 value for a binary step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def _local_search(initial_h, time_limit):
    """Simple hill‑climbing with occasional random restarts."""
    start = time.time()
    n = len(initial_h)
    h = initial_h.copy()
    best_h = h.copy()
    best_score = _compute_c5(h)

    # keep index sets for quick swaps
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())

    # annealing temperature schedule
    T0 = 1.0
    T = T0
    cooling = 0.9995

    iter_count = 0
    while time.time() - start < time_limit:
        iter_count += 1
        # pick a random 1 and 0 to swap
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))

        # perform swap
        h[i], h[j] = 0.0, 1.0
        new_score = _compute_c5(h)

        accept = False
        if new_score < best_score:
            accept = True
        else:
            # Metropolis criterion
            delta = new_score - best_score
            prob = np.exp(-delta / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # update sets
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0

        # temperature decay
        T *= cooling

        # occasional random restart to avoid stagnation
        if iter_count % 2000 == 0:
            if np.random.rand() < 0.3:
                # restart keeping best as seed
                h = best_h.copy()
                ones = set(np.where(h == 1)[0].tolist())
                zeros = set(np.where(h == 0)[0].tolist())
                T = T0

    return best_h, best_score

def _upgrade_resolution(h, factor):
    """Create a higher‑resolution binary vector by repeating each entry `factor` times
    and then re‑balancing to keep exactly half ones."""
    n_small = len(h)
    n_big = n_small * factor
    # repeat entries
    big = np.repeat(h, factor)
    # current count of ones
    cur_ones = int(big.sum())
    target_ones = n_big // 2
    if cur_ones > target_ones:
        # turn some ones to zeros
        ones_idx = np.where(big == 1)[0]
        drop = cur_ones - target_ones
        drop_idx = np.random.choice(ones_idx, size=drop, replace=False)
        big[drop_idx] = 0.0
    elif cur_ones < target_ones:
        zeros_idx = np.where(big == 0)[0]
        add = target_ones - cur_ones
        add_idx = np.random.choice(zeros_idx, size=add, replace=False)
        big[add_idx] = 1.0
    return big
