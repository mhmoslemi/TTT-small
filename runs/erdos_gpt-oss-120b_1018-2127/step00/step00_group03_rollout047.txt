import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 value for a given step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _random_shift(h: np.ndarray, rng: np.random.Generator, max_step: float = 0.2) -> np.ndarray:
    """Return a copy of h with a small mass transfer between two positions."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # allowable delta so that both stay in [0,1]
    if rng.random() < 0.5:
        # move from i to j
        delta_max = min(max_step, h[i], 1.0 - h[j])
        if delta_max <= 0:
            return h  # no feasible move
        delta = rng.uniform(0.0, delta_max)
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:
        # move from j to i
        delta_max = min(max_step, h[j], 1.0 - h[i])

            return h
        delta = rng.uniform(0.0, delta_max)

        h_new[i] += delta
        h_new[j] -= delta
    return h_new
