import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection of a vector onto the box [0,1] with exact sum
# ------------------------------------------------------------
def _project_box_sum(y, target, max_iter=60, tol=1e-10):
    """
    Solve: min ||x-y||^2  s.t. 0<=x<=1, sum(x)=target
    using a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def _c5_value(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _anneal(h0, budget_s, start_time):
    n = h0.size
    target = n / 2.0
    best_h = h0.copy()
    best_val = _c5_value(best_h)

    cur_h = best_h.copy()
    cur_val = best_val

    # Annealing schedule parameters
    T_init = 0.2
    T_final = 1e-4
    max_iters = 200000  # generous upper bound; will stop by time
    it = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.97:   # keep a little margin
            break
        if it >= max_iters:
            break
        it += 1

        # temperature linear in log‑scale
        theta = it / max_iters
        T = T_init * (T_final / T_init) ** theta

        # propose a zero‑sum perturbation
        sigma = 0.02 * (1.0 - theta) + 0.001
        delta = np.random.randn(n) * sigma
        delta -= delta.mean()  # enforce zero sum
        cand = cur_h + delta
        cand = _project_box_sum(cand, target)

        cand_val = _c5_value(cand)
        dE = cand_val - cur_val

            cur_h, cur_val = cand, cand_val

        # occasional restart if stuck
