import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project(h, target_sum):
    """Clip to [0,1] and renormalize so that sum(h) == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        return np.full_like(h, target_sum / h.size)
    # scale to hit the target sum, then re‑clip a few times
    for _ in range(5):
        h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
    return h

def _run_optimization(start_h, dx, budget, start_time):
    """Run a short SLSQP optimisation, respecting the time budget."""
    n = start_h.size
    target_sum = n / 2.0  # because integral must equal 1 (dx = 2/n)

    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = {"type": "eq", "fun": lambda x: x.sum() - target_sum}

    best_h = start_h.copy()
    best_val = _evaluate_c5(best_h, dx)

    # wrapper that also respects the wall‑clock budget
    def objective(x):
        if time.time() - start_time > budget * 0.9:
            # early exit signal for the optimizer
            raise StopIteration
        return _evaluate_c5(x, dx)

    # SLSQP may raise StopIteration; catch it and return current best
    try:
        res = minimize(
            objective,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            cand = _project(res.x, target_sum)
            cand_val = _evaluate_c5(cand, dx)
            if cand_val < best_val:
                best_h, best_val = cand, cand_val
    except StopIteration:
        pass

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    np.random.seed(seed)
    start_time = time.time()

    # Determine sampling resolution
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 80                     # fallback resolution
        h0 = np.full(n_points, 0.5)       # simple uniform start

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h0 = _project(h0, target_sum)

    best_h, best_val = _run_optimization(h0, dx, budget_s, start_time)

    # Simple stochastic refinement if time permits
    max_iters = 2000
    iter_count = 0
    while time.time() - start_time < budget_s * 0.98 and iter_count < max_iters:
        cand = best_h.copy()
        # perturb a random subset
        k = max(1, n_points // 20)
        idx = np.random.randint(0, n_points, size=k)
        cand[idx] += np.random.normal(scale=0.08, size=k)
        cand = _project(cand, target_sum)
        cand_val = _evaluate_c5(cand, dx)
        if cand_val < best_val:
            best_h, best_val = cand, cand_val
        iter_count += 1

    # Final safety projection (should already hold)
    best_h = _project(best_h, target_sum)
    best_val = _evaluate_c5(best_h, dx)

    return best_h, best_val, n_points
