import time
import numpy as np
from scipy import optimize

def _compute_c5(h):
    """Return the overlap constant C5 for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _objective(x, n, penalty):
    """DE objective: max overlap + quadratic penalty for the integral constraint."""
    s = np.sum(x)
    pen = penalty * (s - n / 2.0) ** 2
    return _compute_c5(x) + pen

def _refine_with_slsqp(x0, n, time_left):
    """Local refinement using SLSQP if time permits."""
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    opts = {"maxiter": 200, "ftol": 1e-9}
    start = time.time()
    res = optimize.minimize(
        lambda x: _compute_c5(x),
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    if not res.success and (time.time() - start) > time_left:
        # fallback to original point if optimisation ran out of time
        return x0
    return res.x

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples.

    The function respects the time budget and tries to improve upon an
    optional `initial_h_values` supplied by the harness.
    """
    start_time = time.time()
    # ------------------------------------------------------------------
    # Determine problem size and starting vector
    # ------------------------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        h0 = init.copy()
    except NameError:
        n = 47  # a modest size when no parent is provided
        h0 = np.full(n, 0.5)  # satisfies the integral exactly
    # Ensure feasibility of the start (tiny numerical adjustments)
    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = h0.sum()
    if cur_sum != n / 2.0:
        # simple scaling while keeping within [0,1]
        scale = (n / 2.0) / cur_sum
        h0 = np.clip(h0 * scale, 0.0, 1.0)
        # final correction by adding/subtracting a tiny amount to the first element
        diff = n / 2.0 - h0.sum()
        h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Global search with Differential Evolution
    # ------------------------------------------------------------------
    penalty_factor = 1.0e6
    bounds = [(0.0, 1.0)] * n
    # allocate most of the budget to DE; stop earlier if time is low
    max_iter = 200
    popsize = 15
    de_kwargs = dict(
        func=lambda x: _objective(x, n, penalty_factor),
        bounds=bounds,
        seed=seed,
        polish=False,
        updating="deferred",
        maxiter=max_iter,
        popsize=popsize,
        workers=1,

    result = optimize.differential_evolution(**de_kwargs)
    best = np.clip(result.x, 0.0, 1.0)
