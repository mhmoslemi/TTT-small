import numpy as np
import time, math, random

def _evaluate(h, dx):
    """Return max overlap for step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

def _propose_move(h, eps_max=0.05):
    """
    Return a new vector where a small amount epsilon is transferred
    from index i (where h[i] > 0) to index j (where h[j] < 1).
    """
    n = h.size
    # pick source index with positive mass
    i = np.random.randint(n)
    while h[i] <= 1e-12:
        i = np.random.randint(n)
    # pick destination index with room
    j = np.random.randint(n)
    while h[j] >= 1.0 - 1e-12:
        j = np.random.randint(n)
    eps = min(eps_max, h[i], 1.0 - h[j])
    eps = np.random.random() * eps
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # ---- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 200
        h = np.full(n_points, 0.5, dtype=float)

    n_points = h.size
    dx = 2.0 / n_points

    # enforce exact integral constraint ∑h·dx = 1
    total = h.sum() * dx
    if abs(total - 1.0) > 1e-12:
        # simple scaling with clipping, iterated a few times
        for _ in range(10):
            h = np.clip(h / total, 0.0, 1.0)
            total = h.sum() * dx
            if abs(total - 1.0) < 1e-12:
                break
        # final renormalisation preserving bounds
        deficit = 1.0 - total
        if abs(deficit) > 1e-12:
            mask = (h > 0) & (h < 1)
            if mask.any():
                adjust = deficit / (mask.sum() * dx)
                h[mask] += adjust
                h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_val = _evaluate(best_h, dx)

    # ---- simulated annealing -------------------------------------------
    T = 0.02                        # initial temperature
    decay = 0.9995                  # geometric cooling
    iter_cnt = 0

    while time.time() - start < budget_s - 0.5:
        h_candidate = _propose_move(best_h)
        val = _evaluate(h_candidate, dx)
        delta = val - best_val
        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            best_h = h_candidate
            best_val = val
        T *= decay
        iter_cnt += 1

        # occasional random restart to escape stagnation
        if iter_cnt % 5000 == 0:

            h_rand = np.clip(h_rand, 0.0, 1.0)

            val_rand = _evaluate(h_rand, dx)
