import time
import math
import random
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_h(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and renormalise to the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        h[:] = target_sum / h.size
        return h
    # simple scaling; a few refinement passes keep it feasible
    for _ in range(4):
        h *= target_sum / h.sum()
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction to hit the sum as close as possible
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny difference uniformly over interior points
        interior = (h > 0) & (h < 1)
        if interior.any():
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def _anneal(initial_h: np.ndarray, max_iter: int, time_limit: float, rng_seed: int):
    """Simple simulated‑annealing optimiser for the max‑correlation objective."""
    np.random.seed(rng_seed)
    random.seed(rng_seed)

    n = initial_h.size
    target = n / 2.0
    h = initial_h.copy()
    cur_score = _compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    T = 1.0                     # initial temperature
    start = time.time()
    i = 0
    while i < max_iter and (time.time() - start) < time_limit:
        # step size shrinks with temperature
        step = 0.1 * T
        perturb = (np.random.rand(n) - 0.5) * 2.0 * step
        h_new = _project_h(h + perturb, target)
        new_score = _compute_c5(h_new)

        # Metropolis acceptance
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            h = h_new
            cur_score = new_score

        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()

        T *= 0.9995            # geometric cooling
        i += 1

    return best_h, best_score
