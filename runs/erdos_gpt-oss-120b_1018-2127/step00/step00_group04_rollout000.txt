import time
import numpy as np

def _project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0<=x<=1, sum x = target} via bisection (water-filling)."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute the overlap bound C5 for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with prescribed sum."""
    v = rng.random(n)
    return _project_to_feasible(v, target)
