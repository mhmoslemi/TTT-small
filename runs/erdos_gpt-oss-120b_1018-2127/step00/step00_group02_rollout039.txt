import time
import numpy as np
import random

def compute_c5(h):
    """Return max overlap integral for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_to_constraints(h, target_sum, max_iter=20):
    """Clip to [0,1] and adjust to hit the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # all at bounds – spread uniformly and re‑clip
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-9:
        for i in range(len(h)):
            if cur == target_sum:
                break
            if cur < target_sum:
                avail = 1.0 - h[i]
                delta = min(avail, target_sum - cur)
            else:
                avail = h[i]
                delta = -min(avail, cur - target_sum)
            h[i] += delta
            cur += delta
    return h

def upsample(vec, n_target):
    """Linearly interpolate a vector defined on [0,2) to n_target points."""
    n_src = len(vec)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_tgt, x_src, vec)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function minimizing the Erdős C5 bound."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    best_h = None
    best_score = float("inf")
    best_n = None

    # Candidate grid sizes – keep them modest for speed.
    grid_sizes = [60, 80, 100]

    for n in grid_sizes:
        # Time allocation for this grid size.
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left <= 0.0:
            break
        # Rough share of remaining time for the current size.
        remaining_sizes = len(grid_sizes) - grid_sizes.index(n)
        sub_budget = time_left / remaining_sizes

        # Initial vector.
        if "initial_h_values" in globals():
            h0 = upsample(initial_h_values, n)
        else:
            h0 = np.full(n, 0.5)
        h0 = project_to_constraints(h0, n / 2)

        # Local stochastic improvement.
        best_local_h = h0
        best_local_score = compute_c5(best_local_h)

        step = 0.2  # magnitude of random perturbations
        local_start = time.time()
        # Use most of the allocated sub‑budget for this size.
        while time.time() - local_start < 0.85 * sub_budget:
            cand = best_local_h.copy()
            # Perturb a random subset of coordinates.
            m = max(1, int(0.1 * n))
            idx = np.random.choice(n, m, replace=False)
            cand[idx] += np.random.uniform(-step, step, size=m)
            cand = project_to_constraints(cand, n / 2)

            score = compute_c5(cand)
            if score < best_local_score:
                best_local_h, best_local_score = cand, score
                step = max(0.01, step * 0.95)  # tighten search after improvement
            else:
                step = min(0.5, step * 1.02)   # occasionally broaden search

        # Keep the best across all grid sizes.
        if best_local_score < best_score:
            best_h, best_score, best_n = best_local_h, best_local_score, n

    # Ensure we return something even if no improvement happened.
    if best_h is None:
        # Fallback to a simple balanced vector.
        best_n = 60
        best_h = np.full(best_n, 0.5)
        best_h = project_to_constraints(best_h, best_n / 2)
        best_score = compute_c5(best_h)

    return best_h, best_score, best_n
