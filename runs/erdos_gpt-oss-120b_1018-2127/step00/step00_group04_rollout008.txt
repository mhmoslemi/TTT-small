import time, math, random
import numpy as np

def _objective(h, dx):
    """Compute C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _project_bounds_and_sum(h, target_sum):
    """Clip to [0,1] and then adjust a single coordinate to meet the exact sum."""
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return
    # try to fix the difference by modifying a random feasible index
    idxs = np.arange(h.size)
    random.shuffle(idxs)
    for i in idxs:
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    # If still off (numerical), distribute uniformly (should be tiny)
    if abs(diff) > 1e-12:
        h += diff / h.size

def _initial_vector(n):
    """Create a feasible starting vector (binary, sum = n/2)."""
    h = np.zeros(n, dtype=float)
    ones = n // 2
    h[:ones] = 1.0
    np.random.shuffle(h)
    return h

def _upsample_initial(old_h, new_n):
    """Interpolate an existing initial vector onto a larger grid."""
    old_n = old_h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # enforce exact integral after interpolation
    target = new_n / 2.0
    _project_bounds_and_sum(new_h, target)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized step function sampled on a uniform grid over [0,2].
    c5_bound : float
        Corresponding maximal overlap value.
    n_points : int
        Number of samples used.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # Choose resolution (trade‑off between quality & speed)
    n_points = 200                     # < 1k as required
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ------------------------------------------------------------------
    # Initialise h
    if 'initial_h_values' in globals():
        init = globals()['initial_h_values']
        if isinstance(init, np.ndarray):
            h = _upsample_initial(init, n_points)
        else:  # list‑like
            h = _upsample_initial(np.asarray(init, dtype=float), n_points)
    else:
        h = _initial_vector(n_points)

    best_h = h.copy()
    best_val = _objective(best_h, dx)

    # ------------------------------------------------------------------
    # Simple stochastic hill‑climbing with occasional restarts
    iteration = 0
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        # Occasionally restart from a fresh random binary vector
        if iteration % 3000 == 0:
            cand = _initial_vector(n_points)
        else:
            cand = best_h.copy()
            # Perturb two coordinates while preserving the sum
            i, j = np.random.randint(0, n_points, size=2)
            eps = 0.12  # step size
            delta = (np.random.rand() - 0.5) * 2 * eps
            cand[i] += delta
            cand[j] -= delta

        _project_bounds_and_sum(cand, target_sum)
        val = _objective(cand, dx)

        # Accept if improvement
        if val < best_val:
            best_val, best_h = val, cand.copy()

        # Simple simulated‑annealing style acceptance to escape local minima
        else:
            # temperature decays logarithmically
            elapsed = time.time() - start_time
            temp = max(1e-4, (budget_s - elapsed) / budget_s)
            prob = math.exp(-(val - best_val) / (temp + 1e-12))
            if random.random() < prob:
                best_val, best_h = val, cand.copy()

    return best_h, float(best_val), n_points
