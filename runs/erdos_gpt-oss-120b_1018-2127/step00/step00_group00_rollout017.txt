import time
import numpy as np

def _c5_of_h(h):
    """Compute C5 for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_feasible(h, target_sum, max_iter=20):
    """Clip to [0,1] and adjust to hit the target sum (∑h = target_sum)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free_mask = (h > 0.0) & (h < 1.0)
        if free_mask.any():
            delta = diff / free_mask.sum()
            h[free_mask] += delta
            h = np.clip(h, 0.0, 1.0)
        else:
            # all variables at a bound; fall back to proportional scaling
            if s == 0:
                break
            h = h * (target_sum / s)
            h = np.clip(h, 0.0, 1.0)
    # final safeguard
    s = h.sum()
    if abs(s - target_sum) > 1e-8 and s > 0:
        h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def _evaluate(h):

    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
