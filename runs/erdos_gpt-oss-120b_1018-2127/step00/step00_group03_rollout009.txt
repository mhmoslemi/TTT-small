import time
import numpy as np

def _project_to_sum(v, target):
    """Clip to [0,1] and adjust to meet sum(target) using water-filling."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(1000):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    s = v.sum()
    if abs(s - target) > 1e-9:
        # distribute remaining uniformly within bounds
        inc = (target - s) / v.size
        v = np.clip(v + inc, 0.0, 1.0)
    return v


def _c5_score(h):
    """Fallback C5 computation (used if evaluate_erdos_solution is unavailable)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns:
        h_values (np.ndarray) – optimized values in [0,1] of length n_points
        c5_bound (float)   – achieved C5 value
        n_points (int)     – number of discretisation points used
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------ #
    # Initialise h ----------------------------------------------------- #
    # ------------------------------------------------------------------ #
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        n = 70
        h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        rng.shuffle(h)

    n = h.shape[0]
    target_sum = n / 2.0
    h = _project_to_sum(h, target_sum)

    # ------------------------------------------------------------------ #
    # Scoring utilities ------------------------------------------------ #
    # ------------------------------------------------------------------ #
    if 'evaluate_erdos_solution' in globals():
        def score(vec):
            return evaluate_erdos_solution(vec)
    else:
        def score(vec):
            return _c5_score(vec)

    best_h = h.copy()
    best_score = score(best_h)

    # ------------------------------------------------------------------ #
    # Simulated‑annealing loop ------------------------------------------ #
    # ------------------------------------------------------------------ #
    iteration = 0
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        # Temperature schedule (slow cooling)
        T = max(0.005, 0.5 * (1.0 - iteration / 5000.0))

        # Propose a move that preserves the sum exactly:
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        delta = rng.uniform(0.0, max_transfer)

        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # Evaluate
        new_score = score(h_new)

        # Acceptance test
        if new_score < best_score or rng.random() < np.exp((best_score - new_score) / T):
            best_h = h_new
            best_score = new_score

        # Periodic diversification
        if iteration % 2000 == 0:
            # Small Gaussian perturbation followed by projection
            pert = rng.normal(scale=0.03, size=n)
            h_pert = np.clip(best_h + pert, 0.0, 1.0)
            h_pert = _project_to_sum(h_pert, target_sum)
            pert_score = score(h_pert)
            if pert_score < best_score:
                best_h = h_pert
                best_score = pert_score

    return best_h, float(best_score), n
