def _c5_from_h(h):
    """Compute C5 for a discretized h using the definition."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def _project_onto_constraints(h):
    """
    Clip to [0,1] and then adjust to satisfy sum(h) = n/2 exactly.
    Simple iterative projection onto the affine hyperplane.
    """
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # indices that are not at a bound can receive the adjustment
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:  # all at bounds – distribute to the whole vector
            h += diff / n
        else:
            h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix if needed
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.where((h > 0.0) & (h < 1.0))[0]
        if idx.size:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def _random_feasible(n, rng):
    """Generate a random feasible vector of length n."""
    # start with uniform random numbers
    h = rng.random(n)
    # scale to meet the sum constraint
    h *= (n / 2.0) / h.sum()
    h = np.clip(h, 0.0, 1.0)
    # project again to clean up any tiny violation
    h = _project_onto_constraints(h)
    return h
