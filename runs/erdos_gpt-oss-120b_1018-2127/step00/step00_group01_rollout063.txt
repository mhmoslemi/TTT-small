import numpy as np
import time
import math
import random
from scipy.optimize import differential_evolution  # optional, may be used later

# -------------------------------------------------------------
# Helper functions
# -------------------------------------------------------------
def project_to_feasible(h):
    """
    Project vector h onto the feasible set:
        0 <= h_i <= 1
        sum(h) = n/2   (so that integral = 1 with dx = 2/n)
    Uses a simple iterative clipping + uniform correction.
    """
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(8):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-10:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction to hit the target sum as closely as possible
    s = h.sum()
    if abs(s - target) > 1e-8:
        # distribute the remaining discrepancy over non‑boundary entries
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            correction = (target - s) / free.sum()
            h[free] += correction
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    """
    Compute the overlap constant C5 for a discrete step function h.
    """
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def interpolate_initial(initial, n):
    """
    Interpolate a given initial vector (length m) to length n,
    preserving the integral constraint as closely as possible.
    """
    m = len(initial)
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_old, initial)
    return project_to_feasible(h)

# -------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points)
    where h_values is a NumPy array of length n_points representing the
    step function, and c5_bound is the achieved C5 value.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # Choose a modest grid size; higher resolution improves bound but costs time.
    n_points = 200

    # -----------------------------------------------------------------
    # Initialise h
    # -----------------------------------------------------------------
    if 'initial_h_values' in globals():
        h_current = interpolate_initial(np.array(initial_h_values, dtype=float), n_points)
    else:
        # start from a random feasible point
        h_current = np.random.rand(n_points)
        h_current = project_to_feasible(h_current)

    best_h = h_current.copy()
    best_val = compute_c5(best_h)

    # -----------------------------------------------------------------
    # Parameters for the stochastic local search
    # -----------------------------------------------------------------
    max_iter = 5000                # safety cap; loop may exit earlier because of time
    sigma_initial = 0.2            # magnitude of random perturbations
    sigma_final = 0.001            # final perturbation size
    temp_initial = 0.02            # temperature for occasional uphill moves
    temp_final = 0.0001

    iter_count = 0
    stagnant = 0                  # count iterations without improvement

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        if iter_count >= max_iter:
            break

        # linear schedules for sigma and temperature
        frac = elapsed / (budget_s * 0.95)
        sigma = sigma_initial * (1 - frac) + sigma_final * frac
        temperature = temp_initial * (1 - frac) + temp_final * frac

        # propose a new candidate
        candidate = best_h + sigma * np.random.randn(n_points)
        candidate = project_to_feasible(candidate)

        cand_val = compute_c5(candidate)

        # accept if better, or with a small Metropolis probability
        if cand_val < best_val:
            best_h, best_val = candidate, cand_val
            stagnant = 0
        else:
            if np.random.rand() < math.exp((best_val - cand_val) / max(temperature, 1e-12)):

                stagnant = 0
            else:
                stagnant += 1

        # occasional random restart if we are stuck

            restart = project_to_feasible(restart)
            restart_val = compute_c5(restart)
            if restart_val < best_val:
                best_h, best_val = restart, restart_val
            stagnant = 0

        iter_count += 1

    # -----------------------------------------------------------------
    # Final polishing with a short differential evolution run (optional)
    # -----------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:  # give DE at least a few seconds

            vec = project_to_feasible(vec)   # enforce constraints inside the objective

    # Ensure the final vector satisfies the integral exactly (tiny drift correction)
    best_h = project_to_feasible(best_h)
