import time
import numpy as np
import random

def _objective(h):
    """Maximum overlap integral for a given step function."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)

def _repair(h):
    """Project h onto [0,1]^n with the exact integral constraint sum(h)=n/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Tiny residual fix
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-9:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            idx = free[0]
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def _interpolate(old_h, new_n):
    """Linear interpolation of an existing solution onto a finer grid."""
    old_n = old_h.size
    # sample points uniformly on [0,2)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, old_h)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) after a timed stochastic search."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # 1. Starting solution
    # --------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h0 = np.asarray(init, dtype=float)
        n0 = h0.size
    else:
        n0 = 100                         # fallback size (even)
        h0 = np.zeros(n0, dtype=float)
        h0[: n0 // 2] = 1.0               # simple step, integral = n/2
    h0 = _repair(h0)
    best_h = h0.copy()
    best_score = _objective(best_h)
    best_n = n0

    # --------------------------------------------------------------
    # 2. Multi‑grid schedule (doubling until ~800 points)
    # --------------------------------------------------------------
    sizes = [n0]
    while sizes[-1] * 2 <= 800:
        sizes.append(sizes[-1] * 2)

    for idx, n in enumerate(sizes):
        # time guard
        if time.time() - start_time > budget_s:
            break

        # 2a. Initialise current grid
        if idx == 0:
            h = best_h.copy()
        else:
            h = _interpolate(best_h, n)
        h = _repair(h)

        cur_best_h = h.copy()
        cur_best_score = _objective(cur_best_h)

        # 2b. Allocate iterations proportionally to remaining budget
        elapsed = time.time() - start_time
        remaining = max(0.0, budget_s - elapsed)
        # rough estimate: 30 k iterations per second on modest hardware,
        # limit to a few × 10⁴ to stay safe.
        iters = int(min(30000, remaining * 30))
        if iters < 1:
            iters = 1

        # 2c. Stochastic local search
        T = 1.0
        scale = 0.02 / np.sqrt(n)   # step magnitude

        for i in range(iters):
            # periodic time check
            if (i & 0xFF) == 0 and time.time() - start_time > budget_s:
                break

            h_new = h + np.random.normal(scale=scale, size=n)
            h_new = _repair(h_new)

            score = _objective(h_new)

            if score < cur_best_score:
                cur_best_h = h_new
                cur_best_score = score
                h = h_new
            else:
                # occasional uphill move (simulated annealing)
                if np.random.rand() < np.exp(-(score - cur_best_score) / (T + 1e-12)):
                    h = h_new
            T *= 0.9999   # exponential cooling

        # 2d. Record best found on this grid
        if cur_best_score < best_score:
            best_h = cur_best_h
            best_score = cur_best_score
            best_n = n

    # --------------------------------------------------------------
    # 3. Return final answer
    # --------------------------------------------------------------
    return best_h, best_score, best_n
