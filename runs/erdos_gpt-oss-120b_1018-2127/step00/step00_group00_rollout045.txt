import time
import numpy as np
from scipy.optimize import minimize

# helpers --------------------------------------------------------------------
def compute_score(h):
    """Maximum overlap integral for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def proj_bounds(x):
    """Clip to [0,1] and renormalize to satisfy sum(x)=n/2 via simple scaling."""
    n = x.size
    x = np.clip(x, 0.0, 1.0)
    total = np.sum(x)
    target = n / 2.0
    if total > 0:
        x *= target / total
    # second clip may be needed after scaling
    return np.clip(x, 0.0, 1.0)

def optimize_h(x0, n, budget, start):
    """Run SLSQP from x0, respecting remaining time budget."""
    # early exit if not enough time left
    if time.time() - start > budget:
        return None, None
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    opts = {"maxiter": 500, "ftol": 1e-6, "disp": False}
    res = minimize(compute_score, x0, method="SLSQP",
                   bounds=bounds, constraints=cons, options=opts)
    if not res.success:
        return None, None
    h_opt = proj_bounds(res.x)
    return h_opt, compute_score(h_opt)

def interpolate_solution(old_h, new_n):
    """Linear interpolation of old_h (size m) to new_n points on [0,2)."""
    m = old_h.size
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # enforce exact integral constraint by simple scaling+clipping
    new_h = proj_bounds(new_h)
    return new_h

# main -----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where h_values is a numpy array
    representing a step function on [0,2] with integral 1, and c5_bound is the
    achieved maximal overlap integral.
    """
    np.random.seed(seed)
    start_time = time.time()
    # initial data: may be provided by the harness
    try:
        init_h = np.array(initial_h_values, copy=True)
    except NameError:
        # fallback uniform construction if parent not present
        init_n = 45
        init_h = np.full(init_n, 0.5)  # sum = 0.5*init_n = init_n/2
    init_n = init_h.size

    best_h = init_h
    best_score = compute_score(best_h)

    # candidate resolutions (including the original)
    n_list = [init_n, 60, 90, 120, 180]
    for n in n_list:
        if time.time() - start_time > budget_s:
            break
        if n == init_n:
            x0 = best_h
        else:
            x0 = interpolate_solution(best_h, n)
        # ensure feasible start
        x0 = proj_bounds(x0)
        h_opt, score_opt = optimize_h(x0, n, budget_s, start_time)
        if h_opt is not None and score_opt is not None:
            if score_opt < best_score - 1e-8:
