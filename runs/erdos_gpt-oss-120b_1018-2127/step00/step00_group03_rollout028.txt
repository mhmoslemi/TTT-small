import numpy as np
import time

def _compute_c5(h):
    """Compute the C5 value for a discretized step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _random_initial(n, rng):
    """Generate a random feasible h with length n."""
    # start with uniform random values, then project to the simplex with bounds [0,1]
    h = rng.random(n)
    # enforce sum = n/2 by scaling and clipping iteratively
    target = n / 2.0
    for _ in range(20):
        s = h.sum()
        if np.isclose(s, target):
            break
        # simple affine scaling then clip
        h = h * (target / s)
        np.clip(h, 0.0, 1.0, out=h)
    # final correction to exact sum by moving mass between two points
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # find indices not at bounds to adjust
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            free = np.arange(n)
        i = free[0]
        h[i] += diff
        h[i] = np.clip(h[i], 0.0, 1.0)
    return h

def _anneal(h, budget_s, rng):
    """Simple simulated‑annealing improvement of h."""
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    # annealing schedule
    start_time = time.time()
    T = 0.05          # initial temperature
    decay = 0.99995   # per‑iteration decay
    max_iter = 10_000_000  # hard guard

    it = 0
    while True:
        it += 1
        if it > max_iter:
            break
        now = time.time()
        if now - start_time > budget_s * 0.95:  # leave a little margin
            break

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)

        # propose a transfer delta respecting bounds
        max_transfer = min(T, h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = (rng.random() - 0.5) * 2.0 * max_transfer
        # ensure after move stays inside [0,1]
        if not (0.0 <= h[i] - delta <= 1.0 and 0.0 <= h[j] + delta <= 1.0):
            continue

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        val_new = _compute_c5(h_new)

        # acceptance rule
        accept = False
        if val_new < best_val:
            accept = True
        else:
            prob = np.exp((best_val - val_new) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            h = h_new
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()

        # temperature decay
        T *= decay
        if T < 1e-6:
            T = 1e-6

    return best_h, best_val
