import time
import numpy as np
from scipy.optimize import differential_evolution

# evaluate_erdos_solution and initial_h_values are injected by the harness.

def _c5_of(h: np.ndarray) -> float:
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _repair(h: np.ndarray) -> np.ndarray:
    """Project h onto the feasible box [0,1] and enforce sum(h)=n/2 exactly."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    cur = h.sum()
    if abs(cur - target) <= 1e-12:
        return h

    # Simple iterative redistribution to hit the exact sum.
    # First distribute excess/deficit uniformly over mutable entries.
    for _ in range(5):
        diff = h.sum() - target
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            mask = h > 0.0
            if not mask.any():
                break
            reduction = diff / mask.sum()
            h[mask] -= reduction
        else:
            mask = h < 1.0
            if not mask.any():
                break
            addition = -diff / mask.sum()
            h[mask] += addition
        h = np.clip(h, 0.0, 1.0)

    # Final tweak of a single coordinate to eliminate residual error.
    residual = h.sum() - target
    if abs(residual) > 1e-12:
        # find an index that can absorb the residual without leaving [0,1]
        for idx in range(n):
            if 0.0 <= h[idx] - residual <= 1.0:
                h[idx] -= residual
                break
    return h


def _local_search(h0: np.ndarray, rng: np.random.Generator,
                  time_limit: float) -> tuple[np.ndarray, float]:
    """Simple hill‑climbing on the feasible set."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_of(h)
    n = h.size
    eps = 0.02
    decay = 0.9995

    while time.time() - start < time_limit:
        # pick two distinct indices with room to move mass
        i_candidates = np.where(h > 1e-12)[0]
        j_candidates = np.where(h < 1.0 - 1e-12)[0]
        if i_candidates.size == 0 or j_candidates.size == 0:
            break
        i = rng.choice(i_candidates)
        j = rng.choice(j_candidates)
        if i == j:
            continue

        delta = min(eps, h[i], 1.0 - h[j])
        if delta <= 0:
            continue

        h[i] -= delta
        h[j] += delta

        val = _c5_of(h)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta

        eps *= decay
        if eps < 1e-5:
            eps = 0.02  # reset to keep exploring

    return best_h, best_val


def _objective_flat(x, n):
    """DE objective: penalized C5 for a flattened vector of length n."""
    h = np.array(x)
    h = _repair(h)
    return _c5_of(h)


def _run_global_de(n, rng, time_budget):
    """Run a lightweight differential evolution to get a promising candidate."""
    bounds = [(0.0, 1.0) for _ in range(n)]

    # DE respects bounds, we enforce the equality via penalty in the objective.
    de_opts = {
        "maxiter": 30,
        "popsize": 8,
        "mutation": (0.5, 1.0),
        "recombination": 0.7,
        "seed": int(rng.integers(0, 2**32 - 1)),
        "tol": 1e-8,
        "polish": False,
        "disp": False,
    }

    result = differential_evolution(
        func=lambda x: _objective_flat(x, n),
        bounds=bounds,
        **de_opts,
    )
    h = _repair(result.x)
    return h, _c5_of(h)
