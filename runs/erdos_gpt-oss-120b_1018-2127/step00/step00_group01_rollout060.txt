import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def _evaluate(h, dx):
    """Return max overlap integral C5 for discrete h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _penalized_obj(h, target, dx, penalty=1e8):
    """Objective for DE: C5 plus large penalty for mass violation."""
    s = np.sum(h)
    val = _evaluate(h, dx)
    if abs(s - target) > 1e-6:
        val += penalty * abs(s - target)
    return val

def _project_feasible(h, target):
    """Clip to [0,1] and then shift uniformly to meet the mass constraint."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    # distribute diff evenly, then re‑clip (iterate a few times)
    if abs(diff) > 1e-12:
        step = diff / len(h)
        h += step
        h = np.clip(h, 0.0, 1.0)
        # adjust residual once more
        residual = target - np.sum(h)
        if abs(residual) > 1e-12:
            h += residual / np.count_nonzero((h > 0) & (h < 1))
            h = np.clip(h, 0.0, 1.0)
    return h

def _local_refine(x0, target, dx, time_limit):
    """SLSQP refinement respecting bounds and linear mass equality."""
    bounds = [(0.0, 1.0)] * len(x0)
    cons = {"type": "eq", "fun": lambda h: np.sum(h) - target}
    opts = {"maxiter": 500, "ftol": 1e-12}
    start = time.time()
    res = minimize(
        lambda h: _evaluate(h, dx),
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    # enforce feasibility in case of tiny violation
    if time.time() - start < time_limit:
        h = _project_feasible(res.x, target)
        return h
    return _project_feasible(x0, target)

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of points."""
    np.random.seed(seed)
    start_time = time.time()
    # ----- initial discretisation -----

    n0 = len(h0)
    target_sum = n0 / 2.0
    dx = 2.0 / n0

    # ----- global DE phase (short) -----
    remaining = budget_s - (time.time() - start_time)
    de_max_iter = max(1, int(remaining * 0.5 / (n0 * 0.001)))  # rough scaling
    bounds = [(0.0, 1.0)] * n0

        return _penalized_obj(h, target_sum, dx)

    result_de = differential_evolution(

    best = _project_feasible(result_de.x, target_sum)
