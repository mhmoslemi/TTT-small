import time
import numpy as np

def _project_to_constraints(vec):
    """Clip to [0,1] and enforce sum(vec)=len(vec)/2."""
    n = vec.size
    target = n / 2.0
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(10):
        total = v.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        if not np.any(free):
            # all at bounds, distribute uniformly (will stay at bounds)
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / free.sum()
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    return v

def _evaluate_c5(h):
    """Return max overlap integral for discretised h."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _initial_vector(n_target):
    """Create a starting vector of length n_target."""
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        n0 = base.size
        # linear interpolation to new grid
        src_x = np.linspace(0.0, 2.0, n0, endpoint=False)
        tgt_x = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h = np.interp(tgt_x, src_x, base)
    else:
        # simple half‑ones, half‑zeros pattern
        h = np.zeros(n_target, dtype=float)
        h[: n_target // 2] = 1.0
    h = _project_to_constraints(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return (h_values, c5_bound, n_points) where
    - h_values: ndarray of length n_points satisfying the constraints
    - c5_bound: the achieved maximum overlap integral
    - n_points: number of discretisation points used
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # choose a moderate size (fast enough for many iterations)
    n_points = 240  # 120 samples per unit length, well under 1k
    h_current = _initial_vector(n_points)
    best_h = h_current.copy()
    best_val = _evaluate_c5(best_h)

    # Simulated‑annealing style parameters
    max_iter = 200000
    max_temp = 0.02
    min_temp = 0.0005
    step_base = 0.08  # typical mass moved per proposal

    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # temperature schedule (linear)
        temp = max_temp - (max_temp - min_temp) * (it / max_iter)

        # propose a simple mass‑transfer move
        i, j = rng.choice(n_points, size=2, replace=False)
        # decide direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        max_out = h_current[src]          # how much can be taken
        max_in = 1.0 - h_current[dst]     # how much can be received
        if max_out <= 0.0 or max_in <= 0.0:
            continue
        # magnitude scales with iteration progress
        step = step_base * (1.0 - it / max_iter)
        amt = rng.random() * min(step, max_out, max_in)

        h_prop = h_current.copy()
        h_prop[src] -= amt
        h_prop[dst] += amt

        # Ensure constraints (projection is cheap)
        h_prop = _project_to_constraints(h_prop)

        val_prop = _evaluate_c5(h_prop)

        # Acceptance rule
        if val_prop < best_val:
            best_h = h_prop
            best_val = val_prop
        if val_prop < _evaluate_c5(h_current):
            accept = True
        else:
            # Metropolis criterion
            accept = rng.random() < np.exp(-(val_prop - best_val) / (temp + 1e-12))
        if accept:
            h_current = h_prop

        # occasional random shake to escape stagnation
        if (it & 0x3FFF) == 0:  # every ~16384 iterations
            mask = rng.random(n_points) < 0.02
            h_current[mask] = rng.random(mask.sum())
            h_current = _project_to_constraints(h_current)

    # final projection for safety
    best_h = _project_to_constraints(best_h)
    final_val = _evaluate_c5(best_h)

    return best_h, float(final_val), n_points
