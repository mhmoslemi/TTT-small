import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_max_overlap(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_vector(h, target_sum):
    """Clip to [0,1] and adjust to meet the linear equality sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # distribute the difference to non‑boundary entries
    free = (h > 0.0) & (h < 1.0)
    if free.any():
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    # final scaling to guarantee exact sum (tiny adjustment)
    cur = h.sum()
    if cur != 0:
        h *= target_sum / cur
    # ensure still within bounds after scaling
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h on [0,2] to lower the Erdős C5 bound.

    Returns
    -------
    h_values : ndarray
        Optimized values sampled uniformly over [0,2].
    c5_bound : float
        Corresponding maximal overlap integral.
    n_points : int
        Number of discretisation points (length of h_values).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.size
    else:
        n = 80
        h0 = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    h0 = _project_vector(h0, target_sum)

    best_h = h0
    best_val = _evaluate_max_overlap(best_h)

    # Linear equality constraint for the integral
    constraints = [{"type": "eq", "fun": lambda h: h.sum() - target_sum}]
    bounds = [(0.0, 1.0) for _ in range(n)]

    # Optimization options
    opts = {"maxiter": 2000, "ftol": 1e-12, "disp": False}

    # Main search loop – perform several restarts while time permits
    while time.time() - start_time < budget_s - 5.0:
        # Random perturbation of the current best solution
        perturb = rng.normal(scale=0.1, size=n)
        x0 = best_h + perturb
        x0 = _project_vector(x0, target_sum)

        res = minimize(
            fun=_evaluate_max_overlap,
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options=opts,
        )

        if not res.success:
            # If the optimizer failed, fall back to a fresh random start
            x0 = rng.random(n)
            x0 = _project_vector(x0, target_sum)
            res = minimize(
                fun=_evaluate_max_overlap,
                x0=x0,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options=opts,
            )

        if res.success:
            val = _evaluate_max_overlap(res.x)
            if val < best_val:
                best_val = val
                best_h = np.clip(res.x, 0.0, 1.0)

        # Safety check: if we have already beaten the target, exit early
        if best_val < 0.3808:
            break

    return best_h, best_val, n
