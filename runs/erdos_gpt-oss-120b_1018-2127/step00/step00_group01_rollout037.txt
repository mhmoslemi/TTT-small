import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

    """

    start_time = time.time()

            # renormalise while preserving bounds

            excess = h0.sum() - target
            if excess > 0:
                # scale down uniformly
