import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Exact C5 evaluation using full correlation as described."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def project_to_feasible(h, target_sum):
    """Project h onto [0,1] and adjust to meet the exact sum constraint."""
    # Simple iterative projection: clip then shift uniformly, repeat a few times
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        h += diff / h.size
    # final clip to guarantee bounds
    h = np.clip(h, 0.0, 1.0)
    # tiny residual correction by scaling inside bounds if needed
    current = h.sum()
    if abs(current - target_sum) > 1e-10:
        # distribute remaining error proportionally where interior (not at bounds)
        interior = (h > 0) & (h < 1)
        if interior.any():
            correction = (target_sum - current) / interior.sum()
            h[interior] += correction
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_active_shift(h, k):
    """Gradient of the active (max) shift term."""
    n = h.size
    coeff = 2.0 / n
    h_plus = np.roll(h, -k)   # h[(i+k) % n]
    h_minus = np.roll(h, k)   # h[(i-k) % n]
    return coeff * (1.0 - h_plus - h_minus)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size and initial vector

    else:
        n = 60
        h = np.random.rand(n)
        target = n / 2.0
        h = project_to_feasible(h, target)

    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Parameters for the iterative scheme
    max_iters = 20000
    step0 = 0.5
    step_min = 1e-4
    decay = 0.9995

    step = step0
    it = 0
    while it < max_iters and (time.time() - start_time) < budget_s * 0.95:
        # full correlation to find active shift
        dx = 2.0 / n
        corr = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
        idx_max = np.argmax(corr)
        k = idx_max - (n - 1)  # shift index (can be negative)

        # gradient of the active term
        grad = gradient_active_shift(best_h, k)

        # gradient descent step
        h_new = best_h - step * grad

        # enforce feasibility
        h_new = project_to_feasible(h_new, target_sum)

        # evaluate
        c5_new = compute_c5(h_new)

        # accept if improvement; otherwise keep current (simple greedy)

            best_c5 = c5_new

        # schedule step size
        step = max(step * decay, step_min)

    # Final polishing with a few SLSQP runs from the best point
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = ({
        'type': 'eq',
        'fun': lambda x: x.sum() - target_sum

        return compute_c5(x)
