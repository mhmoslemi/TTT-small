import time
import numpy as np
import random

# assume the harness provides these globals

def _initial_h(n):
    """Return a feasible starter vector of length n (average 0.5)."""
    h = np.full(n, 0.5, dtype=np.float64)
    return h

def _swap_move(h, i, j, delta):
    """Add delta to h[i] and subtract from h[j]; assumes feasibility after call."""
    h[i] += delta
    h[j] -= delta

def _propose_swap(h):
    """Generate a feasible swap move (i,j,delta)."""
    n = h.size
    i, j = random.sample(range(n), 2)
    # direction: decide whether to move mass from j to i or opposite
    if random.random() < 0.5:
        i, j = j, i
    # maximum allowed delta respecting box constraints
    max_delta = min(1.0 - h[i], h[j])
    if max_delta <= 0:
        return None
    # pick a random delta up to max_delta
    delta = random.random() * max_delta
    return i, j, delta
