def is_feasible(h, n):
    return min(h) >= 0 and max(h) <= 1 and np.isclose(sum(h), n/2, atol=1e-3)

def compute_score(h, n):
    corr = np.correlate(h, 1-h, mode="full")
    scale = 2.0 / n
    return np.max(corr) * scale

def minimize_score(h, n, seed):
