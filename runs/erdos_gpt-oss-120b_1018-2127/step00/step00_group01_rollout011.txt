import numpy as np
import time
import random
from math import inf

def _evaluate_c5(h):
    """Return the C5 value for a step vector h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _project_pairwise(h, i, j, delta):
    """Adjust h[i] and h[j] by +delta and -delta respecting [0,1] bounds.
    Returns new vector and the actual delta applied (may be clipped)."""
    hi, hj = h[i], h[j]
    # allowable delta range
    max_pos = min(1.0 - hi, hj)          # increase hi, decrease hj
    max_neg = min(hi, 1.0 - hj)          # decrease hi, increase hj
    max_delta = min(delta, max_pos)
    min_delta = max(delta, -max_neg)
    actual = max(min_delta, min(max_delta, delta))
    if actual == 0.0:
        return h, 0.0
    h_new = h.copy()
    h_new[i] = hi + actual
    h_new[j] = hj - actual
    return h_new, actual
