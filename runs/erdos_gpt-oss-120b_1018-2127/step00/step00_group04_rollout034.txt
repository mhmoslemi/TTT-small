import time
import numpy as np

def _compute_c5(h, dx):
    """Return the overlap integral C5 for a binary vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _initial_binary(n, seed, provided):
    """Create a feasible binary start vector of length n."""
    rng = np.random.default_rng(seed)
    if provided is not None:
        # ensure exactly n/2 ones by taking the largest entries
        h = np.asarray(provided, dtype=float)
        if h.size != n:
            # resample to desired size by interpolation
            x_old = np.linspace(0, 2, h.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, h)
        ones = n // 2
        idx = np.argsort(-h)  # descending
        bin_h = np.zeros(n, dtype=float)
        bin_h[idx[:ones]] = 1.0
        return bin_h
    # no parent: random binary with correct sum
    ones = n // 2
    bin_h = np.zeros(n, dtype=float)
    bin_h[:ones] = 1.0
    rng.shuffle(bin_h)
    return bin_h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] (discretised) that lowers
    the Erdős C5 overlap bound.

    Returns
    -------
    h_values : np.ndarray
        The best binary vector found (values in {0,1}).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Length of the discretisation.
    """
    start_time = time.time()
    # Use the globally supplied initial_h_values if present
    provided = globals().get("initial_h_values", None)

    # Choose a modest grid size (prefer < 200 for speed)
    n_points = 200 if provided is None else len(provided)
    if n_points % 2 != 0:
        n_points += 1  # ensure even for exact half‑integral sum

    dx = 2.0 / n_points

    # Initialise binary feasible vector
    h_cur = _initial_binary(n_points, seed, provided)
    best_h = h_cur.copy()
    best_c5 = _compute_c5(best_h, dx)

    # Pre‑compute indexes of ones and zeros for fast swaps
    rng = np.random.default_rng(seed)
    temperature = 0.5
    cooling = 0.9995
    min_temp = 1e-5

    # Main simulated‑annealing loop
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:   # leave a margin for finalisation
            break
        # pick a random 1‑position and a random 0‑position and swap them
        ones_idx = np.flatnonzero(h_cur == 1.0)
        zeros_idx = np.flatnonzero(h_cur == 0.0)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        h_new = h_cur.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0

        new_c5 = _compute_c5(h_new, dx)

        # Acceptance criterion
        delta = new_c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temperature, min_temp)):
            h_cur = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

        # Cool down
        temperature = max(temperature * cooling, min_temp)

    # Final evaluation to ensure feasibility
    best_c5 = _compute_c5(best_h, dx)

    return best_h, float(best_c5), int(n_points)
