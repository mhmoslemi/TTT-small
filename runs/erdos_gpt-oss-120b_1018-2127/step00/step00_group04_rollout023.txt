import numpy as np
import time
import random

def compute_c5(h, dx):
    """Maximum overlap integral for discrete h."""
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return conv.max()

def binary_initialize(n, rng):
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def binarize_from_initial(h_init):
    """Convert an arbitrary initial vector to a binary one with correct sum."""
    n = h_init.size
    # indices of largest values become 1
    order = np.argsort(h_init)[::-1]
    h = np.zeros(n, dtype=float)
    h[order[: n // 2]] = 1.0
    return h

def hill_climb(initial_h, dx, time_limit, rng):
    """Simple hill‑climbing with occasional swaps."""
    n = initial_h.size
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    # indices of ones and zeros for quick sampling
    ones = set(np.flatnonzero(h == 1.0))
    zeros = set(np.flatnonzero(h == 0.0))

    # parameters
    max_iter = 20000
    accept_prob = 0.0  # pure greedy; can be raised for simulated annealing

    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))

        # propose swap
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h, dx)

        improved = new_c5 < best_c5
        accept = improved or (rng.random() < accept_prob)

        if accept:
            # update sets
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)

            if improved:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0

        # optional cooling of accept_prob (not used here)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine number of sample points
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n_points = init.size
        # Ensure even length for exact half‑integral constraint
        if n_points % 2 == 1:
            n_points += 1
        # Build a binary starting vector from the given one
        init_bin = binarize_from_initial(init[:n_points])
    else:
        n_points = 200  # default moderate size
        init_bin = None

    dx = 2.0 / n_points
    best_overall_h = None
    best_overall_c5 = np.inf

    # Allocate time budget: reserve a small margin for final return
    total_budget = budget_s * 0.95
    # time for each restart (including its hill‑climb)
    restart_budget = total_budget / 5.0

    for _ in range(5):
        remaining = total_budget - (time.time() - start_time)
        if remaining <= 0:
            break
        # Choose starting point
        if init_bin is not None and rng.random() < 0.3:
            start_h = init_bin.copy()
        else:
            start_h = binary_initialize(n_points, rng)

        # Run a hill‑climb within its allocated slice of time
        time_allowed = min(restart_budget, remaining)
        h_opt, c5_opt = hill_climb(start_h, dx, time_allowed, rng)

        if c5_opt < best_overall_c5:
            best_overall_c5 = c5_opt
            best_overall_h = h_opt.copy()

    # Fallback: if nothing succeeded, use a simple alternating pattern
    if best_overall_h is None:
        best_overall_h = binary_initialize(n_points, rng)
        best_overall_c5 = compute_c5(best_overall_h, dx)

    return best_overall_h, float(best_overall_c5), int(n_points)
