import time
import numpy as np

def _project_to_constraints(h, target_sum, max_iter=20):
    """Clip to [0,1] and adjust to hit the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        total = h.sum()
        diff = target_sum - total
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            free = 1.0 - h
            cap = free.sum()
            if cap == 0:
                break
            h += diff * free / cap
        else:
            free = h
            cap = free.sum()
            if cap == 0:
                break
            h -= (-diff) * free / cap
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction (rare)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-8:
        h += residual / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with the required sum."""
    h = rng.random(n)
    target = n / 2.0
    return _project_to_constraints(h, target)

def _compute_c5(h):
    """Discrete evaluation of the overlap constant."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _upscale(h_small, n_large):
    """Linearly interpolate a coarse vector to a finer grid."""
    m = h_small.size
    x_small = np.linspace(0.0, 2.0, m, endpoint=False)
    x_large = np.linspace(0.0, 2.0, n_large, endpoint=False)
    h_large = np.interp(x_large, x_small, h_small)
    target = n_large / 2.0
    return _project_to_constraints(h_large, target)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Choose a working resolution (balance quality vs speed)
    # ------------------------------------------------------------
    n_points = 380  # 2× the provided 95-point construction
    target_sum = n_points / 2.0

    # ------------------------------------------------------------
    # Initialise best solution
    # ------------------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None and isinstance(init, (list, np.ndarray)):
        init = np.asarray(init, dtype=float)
        if init.size != n_points:
            best_h = _upscale(init, n_points)
        else:
            best_h = _project_to_constraints(init.copy(), target_sum)
    else:
        best_h = _random_feasible(n_points, rng)

    best_score = _compute_c5(best_h)

    # Simple population-less stochastic hill‑climber with occasional restarts
    max_iter = 200000  # generous upper bound; real stop comes from time
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.96:
            break

        # Adaptive step size: start larger, decay over iterations
        sigma = 0.2 * (1.0 - iteration / max_iter)
        sigma = max(sigma, 0.005)

        # Propose a mutated candidate
        cand = best_h + sigma * rng.standard_normal(n_points)
        cand = _project_to_constraints(cand, target_sum)

        cand_score = _compute_c5(cand)
        if cand_score < best_score:
            best_h, best_score = cand, cand_score

        # Occasionally inject a completely random feasible vector
        if rng.random() < 0.005:
            rand_h = _random_feasible(n_points, rng)
            rand_score = _compute_c5(rand_h)
            if rand_score < best_score:
                best_h, best_score = rand_h, rand_score

        iteration += 1

    # ------------------------------------------------------------
    # Return the best found solution
    # ------------------------------------------------------------
    return best_h, float(best_score), n_points
