import numpy as np, time, math, random

def _evaluate(h: np.ndarray) -> float:
    """Return C5 for a discrete step function h."""
    n = h.size
    N = 2 * n                     # zero‑padding length for full correlation
    H = np.fft.rfft(h, n=N)
    G = np.fft.rfft(1.0 - h, n=N)
    corr = np.fft.irfft(H * G, n=N)          # length N, first 2n‑1 entries are the full correlate
    dx = 2.0 / n
    return np.max(corr[: 2 * n - 1]) * dx


def _resample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of the parent construction to n points."""
    x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, init)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h : [0,2] → [0,1] with integral 1
    to obtain a low upper bound for the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimised samples of h on a uniform grid.
    c5_bound : float
        The achieved value of C5.
    n_points : int
        Number of samples used (grid size).
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------
    # 1. Choose a modest grid size (fast evaluation, enough resolution)
    # ------------------------------------------------------------
    n_points = 200                     # < 1k as required

    # ------------------------------------------------------------
    # 2. Initialise a feasible h (respecting sum == n/2)
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = _resample_initial(initial_h_values, n_points)
    else:
        h = np.zeros(n_points)
        idx = np.random.choice(n_points, n_points // 2, replace=False)
        h[idx] = 1.0

    # Clip to [0,1] and enforce exact sum by a two‑index transfer
    h = np.clip(h, 0.0, 1.0)
    target_sum = n_points / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # find one index that can be increased and one that can be decreased
        if diff > 0:
            inc_candidates = np.where(h < 1.0 - 1e-12)[0]
            dec_candidates = np.where(h > 1e-12)[0]
        else:
            inc_candidates = np.where(h > 1e-12)[0]
            dec_candidates = np.where(h < 1.0 - 1e-12)[0]
        if inc_candidates.size and dec_candidates.size:
            i = inc_candidates[0]
            j = dec_candidates[0]
            delta = np.sign(diff) * min(abs(diff), 1.0 - h[i] if diff > 0 else h[i],
                                         h[j] if diff > 0 else 1.0 - h[j])
            h[i] += delta
            h[j] -= delta
    h = np.clip(h, 0.0, 1.0)           # safety

    # ------------------------------------------------------------
    # 3. Local stochastic optimisation (simulated annealing)
    # ------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _evaluate(best_h)

    current_h = h.copy()
    current_c5 = best_c5

    # annealing schedule parameters
    T0 = 0.02
    end_fraction = 0.95               # stop a little before the hard limit

    while True:
        if time.time() - start > budget_s * end_fraction:
            break

        # ----- propose a move that preserves the total sum -----
        i, j = np.random.choice(n_points, 2, replace=False)
        hi, hj = current_h[i], current_h[j]

        # delta must keep both entries inside [0,1]
        delta_min = max(-hi, -(1.0 - hj))
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue

        delta = np.random.uniform(delta_min, delta_max)
        new_h = current_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        new_c5 = _evaluate(new_h)

        # ----- acceptance rule -----
        if new_c5 < current_c5:
            accept = True
        else:
            elapsed = time.time() - start
            fraction = elapsed / (budget_s * end_fraction)
            T = T0 * (1.0 - fraction)          # linear cooling
            prob = math.exp(-(new_c5 - current_c5) / (T + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

    return (best_h, float(best_c5), n_points)
