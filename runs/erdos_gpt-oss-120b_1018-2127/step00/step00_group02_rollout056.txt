import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Return the discrete overlap bound for step function h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(h, target_sum):
    """Clip to [0,1] and adjust linearly to hit the exact sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    # distribute the deficit/surplus uniformly among non‑boundary vars
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size == 0:
        # nothing free – fallback to scaling then clipping
        h = h * (target_sum / cur)
        return np.clip(h, 0.0, 1.0)
    delta = (target_sum - cur) / free.size
    h[free] += delta
    # second pass to enforce bounds
    return np.clip(h, 0.0, 1.0)

def _optimize_start(h_start, budget, start_time):
    """Run a short SLSQP optimisation starting from h_start."""
    n = h_start.size
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = ({
        "type": "eq",
        "fun": lambda x: x.sum() - target_sum,
    },)

    # small wrapper so we can abort if time is out
    def obj(x):
        return _compute_c5(x)

    opts = {"maxiter": 300, "ftol": 1e-9, "disp": False}
    res = minimize(
        obj,
        h_start,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    # ensure feasibility
    h_opt = _project_to_constraints(res.x, target_sum)
    return h_opt, _compute_c5(h_opt)
