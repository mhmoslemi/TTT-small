import numpy as np
import time

def _project_bounds_sum(v, target, lo=0.0, hi=1.0, eps=1e-12):
    """Project vector v onto [lo,hi] with given sum=target."""
    # Initial clipping
    w = np.clip(v, lo, hi)
    cur = w.sum()
    if abs(cur - target) <= eps:
        return w
    # Determine indices that can still move
    free = np.ones_like(w, dtype=bool)
    # Iterative water‑filling / draining
    for _ in range(10):
        if cur < target:
            # need to add mass, distribute to those below hi
            mask = free & (w < hi - eps)
            if not np.any(mask):
                break
            add = (target - cur) / mask.sum()
            w[mask] = np.minimum(w[mask] + add, hi)
        else:
            # need to remove mass, take from those above lo
            mask = free & (w > lo + eps)
            if not np.any(mask):
                break
            sub = (cur - target) / mask.sum()
            w[mask] = np.maximum(w[mask] - sub, lo)
        cur = w.sum()
        if abs(cur - target) <= eps:
            break
    # tiny residual fix
    if abs(cur - target) > eps:
        diff = target - cur
        idx = np.argmax(w if diff > 0 else -w)
        w[idx] = np.clip(w[idx] + diff, lo, hi)
    return w

def _evaluate(h, dx):
    """Return max overlap integral for step vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
