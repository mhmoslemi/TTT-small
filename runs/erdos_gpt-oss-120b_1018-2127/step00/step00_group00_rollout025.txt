import time
import numpy as np
from scipy.optimize import minimize

def _compute_score(h, n):
    """Maximum overlap integral for step vector h of length n."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _feasible_vector(vec, n):
    """Clip to [0,1] and enforce sum = n/2 (integral = 1)."""
    vec = np.clip(vec, 0.0, 1.0)
    total = np.sum(vec)
    if total == 0:
        # fallback uniform distribution
        vec[:] = 0.5
        return vec
    vec = vec / total * (n / 2)
    vec = np.clip(vec, 0.0, 1.0)
    # small residual correction
    diff = np.sum(vec) - n / 2
    if abs(diff) > 1e-8:
        vec -= diff / n
        vec = np.clip(vec, 0.0, 1.0)
    return vec


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.

    Returns
    -------
    (h_values, c5_bound, n_points) : tuple
        h_values – numpy array of the best step values found
        c5_bound – the corresponding maximal overlap integral
        n_points – number of discretization points used
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_score = np.inf
    best_n = None

    # resolutions to try (keep arrays small)
    n_list = [45, 60, 90]

    # Grab any parent construction if present
    parent_h = globals().get("initial_h_values", None)

    for n in n_list:
        if time.time() - start_time > budget_s:
            break

        # Prepare a small set of starting candidates
        candidates = []

        # 1) transferred parent solution if available
        if parent_h is not None:
            parent_arr = np.asarray(parent_h, dtype=float)
            if len(parent_arr) == n:
                candidates.append(parent_arr.copy())
            else:
                # linear interpolation onto new grid
                x_old = np.linspace(0.0, 2.0, len(parent_arr), endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                interp = np.interp(x_new, x_old, parent_arr)
                candidates.append(interp)

        # 2) a few random feasible vectors
        for _ in range(5):
            rand = rng.random(n)
            rand = _feasible_vector(rand, n)
            candidates.append(rand)

        # bounds and equality constraint (integral = 1)
        bounds = [(0.0, 1.0)] * n
        eq_con = {"type": "eq", "fun": lambda x, nn=n: np.sum(x) - nn / 2}

        for init_vec in candidates:
            if time.time() - start_time > budget_s:
                break

            init_vec = _feasible_vector(init_vec, n)

            # Local optimization via SLSQP
            res = minimize(
                lambda x: _compute_score(x, n),
                init_vec,
                method="SLSQP",
                bounds=bounds,
                constraints=eq_con,
                options={"maxiter": 400, "ftol": 1e-9, "disp": False},
            )

            if not res.success:
                continue

            h_opt = _feasible_vector(res.x, n)
            score = _compute_score(h_opt, n)

            if score < best_score - 1e-9:
                best_score = score
                best_h = h_opt.copy()
                best_n = n

    # Fallback: use parent or a simple uniform vector
    if best_h is None:
        if parent_h is not None:
            best_h = np.asarray(parent_h, dtype=float)
            best_n = len(best_h)
            best_score = _compute_score(best_h, best_n)
        else:
            best_n = 45
            best_h = np.full(best_n, 0.5)
            best_score = _compute_score(best_h, best_n)

    return (best_h, float(best_score), int(best_n))
