import time
import numpy as np
from scipy.optimize import minimize

def compute_score(h):
    """Maximum overlap integral for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def is_feasible(h, target_sum, atol=1e-6):
    """Check bounds and linear constraint."""
    if np.any(h < -1e-9) or np.any(h > 1 + 1e-9):
        return False
    return abs(h.sum() - target_sum) <= atol

def project_feasible(h, target_sum):
    """Scale and clip to satisfy 0≤h≤1 and sum≈target_sum."""
    # simple scaling then clipping loop
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - target_sum) < 1e-9:
            break
        scale = target_sum / cur if cur != 0 else 0.0
        h *= scale
    # final clip to guarantee bounds
    h = np.clip(h, 0.0, 1.0)
    # tiny correction for sum
    diff = target_sum - h.sum()
    if abs(diff) > 1e-9:
        # distribute diff uniformly on non‑boundary entries
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.any():
            inc = diff / free.sum()
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
    return h

def transfer_solution(old_h, new_n):
    """Resize old_h (length old_n) to new_n points via linear interpolation."""
    old_n = old_h.size
    if old_n == new_n:
        return old_h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    target_sum = new_n / 2.0
    new_h = project_feasible(new_h, target_sum)
    return new_h

def minimize_h(start_h, seed):
    """Run SLSQP to reduce the max overlap, preserving feasibility."""
    n = start_h.size
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target_sum}
    # deterministic seed for reproducibility of any internal randomness
    rng = np.random.default_rng(seed)

    # small random perturbation to escape flat regions
    pert = rng.normal(scale=0.02, size=n)
    x0 = np.clip(start_h + pert, 0.0, 1.0)
    x0 = project_feasible(x0, target_sum)

    res = minimize(
        compute_score,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 500, "ftol": 1e-7, "disp": False},
    )
    if not res.success:
        # fallback to the start point if optimizer failed
        return start_h, compute_score(start_h)
    h_opt = project_feasible(res.x, target_sum)
    return h_opt, compute_score(h_opt)
