import time
import random
import numpy as np

# Helper: compute C5 for a discrete step function h on [0,2]
def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

# Project onto box [0,1] and linear equality sum(h)=target
def _project_constraints(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            break
        adj = diff / free.sum()
        h[free] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

# Upscale/downscale a vector to a new length preserving the integral
def _resample(h, new_n):
    old_n = h.size
    # positions of sample mid‑points
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(new_x, old_x, h, left=0.0, right=0.0)
    # force the correct total mass
    target = new_n / 2.0
    h_new = _project_constraints(h_new, target)
    return h_new

# Simple coordinate‑wise hill‑climbing with a sum‑preserving move
def _optimize_local(initial_h, time_deadline):
    h = initial_h.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = _compute_c5(h)

    while time.time() < time_deadline:
        # pick indices that can donate / receive mass
        donors = np.where(h > 1e-12)[0]
        receivers = np.where(h < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = int(random.choice(donors))
        j = int(random.choice(receivers))
        max_eps = min(0.05, h[i], 1.0 - h[j])
        if max_eps <= 0.0:
            continue
        eps = random.random() * max_eps
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        c_new = _compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        # occasional random reset to escape plateaus
        if random.random() < 0.001:
            h = best_h.copy()
    return best_h, best_c
