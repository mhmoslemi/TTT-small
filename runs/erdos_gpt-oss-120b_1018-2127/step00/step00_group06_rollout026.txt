import time
import numpy as np
import random
import math

def objective(h, n):
    """Compute C5 upper bound for discrete h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_bounds_sum(x, target):
    """
    Euclidean projection of vector x onto the box [0,1]^n with sum == target.
    Implements a simple water‑filling iteration.
    """
    x = np.clip(x, 0.0, 1.0)
    # Iterate until the sum constraint is met within tolerance
    for _ in range(1000):
        total = x.sum()
        diff = target - total
        if abs(diff) <= 1e-12:
            break
        # free indices are those not at a bound
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        nfree = int(free.sum())
        if nfree == 0:
            # all variables are at a bound; spread diff uniformly and re‑clip
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
            continue
        adjust = diff / nfree
        x[free] += adjust
        x = np.clip(x, 0.0, 1.0)
    return x


def random_neighbor(h, step_scale, target):
    """Create a neighbour by Gaussian perturbation followed by projection."""
    delta = np.random.randn(len(h)) * step_scale
    h_new = h + delta
    return project_bounds_sum(h_new, target)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] that minimizes the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    # ------------------------------------------------------------------
    # Initialise base solution
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: uniform density with correct integral
        n0 = 100  # even number for convenience
        h0 = np.full(n0, 0.5, dtype=float)

    n = len(h0)
    target = n / 2.0
    h0 = project_bounds_sum(h0, target)

    best_h = h0.copy()
    best_score = objective(best_h, n)

    # SA parameters
    step_scale = 0.25          # initial perturbation magnitude
    cooling = 0.9997           # geometric cooling factor
    min_step = 5e-6            # lower bound on step size
    # ------------------------------------------------------------------
    # Main optimisation loop
    while time.time() - start < budget_s * 0.95:
        h_candidate = random_neighbor(best_h, step_scale, target)
        cand_score = objective(h_candidate, n)

        if cand_score < best_score:
            best_h = h_candidate
            best_score = cand_score

        # occasional larger jumps to escape plateaus
        if random.random() < 0.001:
            step_scale = max(step_scale * 1.5, 0.05)

        # cool
        step_scale = max(step_scale * cooling, min_step)

    # ------------------------------------------------------------------
    # Optional refinement on a finer grid if time permits
    # (simple up‑sampling without re‑optimisation)
    remaining = budget_s - (time.time() - start)
    if remaining > 2.0 and n < 760:
        # double resolution
        n2 = min(760, n * 2)
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_fine = np.interp(x_new, x_old, best_h)
        target_fine = n2 / 2.0
        h_fine = project_bounds_sum(h_fine, target_fine)
        fine_score = objective(h_fine, n2)

        if fine_score < best_score:
            best_h = h_fine
            best_score = fine_score
            n = n2

    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_score), int(n))
