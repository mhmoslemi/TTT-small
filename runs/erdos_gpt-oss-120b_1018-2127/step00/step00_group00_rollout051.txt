import time
import numpy as np
from scipy.optimize import minimize

def _compute_score(h, n):
    """Maximum overlap integral for discretized h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _feasible(h, n, atol=1e-6):
    """Return True if h respects bounds and integral constraint."""
    if np.any(h < -atol) or np.any(h > 1.0 + atol):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-4

def _transfer(old_h, new_n):
    """Linearly interpolate old_h (of length m) onto a grid of length new_n."""
    m = len(old_h)
    if m == new_n:
        return old_h.copy()
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # Scale to satisfy the integral constraint before returning
    target = new_n / 2.0
    cur = np.sum(new_h)
    if cur > 0:
        new_h *= target / cur
    # Clip to [0,1] (tiny violations may appear after scaling)
    np.clip(new_h, 0.0, 1.0, out=new_h)
    # Final renormalisation to meet the sum exactly (project onto hyperplane)
    diff = target - np.sum(new_h)
    if abs(diff) > 1e-12:
        # distribute the correction uniformly on the interior (where 0<h<1)
        mask = (new_h > 1e-12) & (new_h < 1.0 - 1e-12)
        if np.any(mask):
            new_h[mask] += diff / np.count_nonzero(mask)
            np.clip(new_h, 0.0, 1.0, out=new_h)
    return new_h

def _objective(x, n):
    return _compute_score(x, n)

def _optimize(start_h, n, seed):
    """Run SLSQP from start_h respecting bounds and the integral equality."""
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    # deterministic behaviour for reproducibility
    rng = np.random.default_rng(seed)
    # Slight perturbation to avoid exact degeneracy
    x0 = start_h + rng.uniform(-1e-4, 1e-4, size=n)
    x0 = np.clip(x0, 0.0, 1.0)
    # Re‑normalise after perturbation
    x0 *= (n / 2.0) / np.sum(x0)
    opts = {"maxiter": 400, "ftol": 1e-8, "disp": False}
    res = minimize(_objective, x0, args=(n,), method="SLSQP",
                   bounds=bounds, constraints=cons, options=opts)
    return res

    best_score = np.inf
    best_n = None

        base_n = 45
        base_h = np.full(base_n, 0.5, dtype=float)

    for n in n_candidates:
        if time.time() - start_time > budget_s:
            break

        # Obtain a starting vector of length n
        if n == base_n:
            h0 = base_h.copy()
        else:
            h0 = _transfer(base_h, n)

        # Ensure feasibility before optimisation
        np.clip(h0, 0.0, 1.0, out=h0)
        # Adjust sum exactly
        target = n / 2.0
        h0 *= target / np.sum(h0)

        cur_score = _compute_score(h0, n)
