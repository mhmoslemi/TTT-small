import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_score_vec(x):
    """C5 value for step function represented by vector x."""
    n = x.shape[0]
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * dx

def project_onto_box_simplex(v, desired_sum, lo=0.0, hi=1.0, tol=1e-9, maxit=60):
    """
    Project vector v onto the set { y | lo <= y_i <= hi, sum(y)=desired_sum }.
    Uses a bisection on the common shift lambda: y = clip(v+lambda, lo, hi).
    """
    # feasible range for lambda
    low, high = -1.0, 1.0
    for _ in range(maxit):
        lam = (low + high) * 0.5
        y = np.clip(v + lam, lo, hi)
        s = y.sum()
        if abs(s - desired_sum) <= tol:
            return y
        if s < desired_sum:
            low = lam
        else:
            high = lam
    return np.clip(v + (low + high) * 0.5, lo, hi)

def transfer_solution(old_h, new_n):
    """Linear interpolation of old_h (any length) to length new_n."""
    n_old = old_h.shape[0]
    old_x = np.linspace(0.0, 2.0, n_old, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    return new_h

def feasible_score(x):
    """Score after forcing feasibility (used for safety)."""
    n = x.shape[0]
    x = np.clip(x, 0.0, 1.0)
    x = project_onto_box_simplex(x, n / 2.0)
    return compute_score_vec(x), x

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    h_values : numpy array of step heights on a uniform grid of [0,2]
    c5_bound : the achieved upper bound (smaller is better)
    n_points : length of h_values
    """
    np.random.seed(seed)
    start = time.time()

    # Candidate resolutions (must include the size of initial_h_values if present)
    n_list = [30, 45, 60, 90, 180]

    best_h = None
    best_score = None
    best_n = None

    # Determine whether an initial construction is supplied
    has_initial = 'initial_h_values' in globals()
    init_h = None
    if has_initial:
        init_h = np.asarray(initial_h_values, dtype=float)

    for n in n_list:
        # time guard
        if time.time() - start > budget_s:
            break

        # ----- starting vector -------------------------------------------------
        if has_initial and init_h.shape[0] == n:
            h0 = init_h.copy()
        elif has_initial:
            h0 = transfer_solution(init_h, n)
        else:
            # no parent construction – generate a random feasible start
            rand = np.random.rand(n)
            h0 = project_onto_box_simplex(rand, n / 2.0)

        # Ensure feasibility (may be slightly off after interpolation)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_simplex(h0, n / 2.0)

        # ----- optimization settings --------------------------------------------
        bounds = [(0.0, 1.0)] * n
        eq_cons = {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}
        opt_options = {'maxiter': 500, 'ftol': 1e-9, 'disp': False}

        # Helper to run a single SLSQP call
        def run_opt(start_vec):
            res = minimize(
                compute_score_vec,
                start_vec,
                method='SLSQP',
                bounds=bounds,
                constraints=eq_cons,
                options=opt_options,
            )
            if not res.success:
                # Even if not successful, try to project the result
                proj_x = project_onto_box_simplex(res.x, n / 2.0)
                return compute_score_vec(proj_x), proj_x
            proj_x = project_onto_box_simplex(res.x, n / 2.0)
            return compute_score_vec(proj_x), proj_x

        # ----- first deterministic run (starting vector) -----------------------
        score, h_opt = run_opt(h0)

        if best_score is None or score < best_score:
            best_score, best_h, best_n = score, h_opt.copy(), n

        # ----- few stochastic restarts -----------------------------------------
        # number of restarts adapts to remaining budget
        remaining = budget_s - (time.time() - start)
        restarts = 2 if remaining > 30 else 0
        for _ in range(restarts):
            if time.time() - start > budget_s:
                break
            rand = np.random.rand(n)
            rand = project_onto_box_simplex(rand, n / 2.0)
            r_score, r_h = run_opt(rand)
            if r_score < best_score:
                best_score, best_h, best_n = r_score, r_h.copy(), n

    # If nothing succeeded (unlikely), fall back to a uniform split
    if best_h is None:
        n = 45
        best_h = np.full(n, 0.5)
        best_h[0] = 0.5  # keep sum = n/2
        best_score = compute_score_vec(best_h)
        best_n = n

    return best_h, float(best_score), int(best_n)
