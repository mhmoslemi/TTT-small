import time
import numpy as np
from math import inf

def compute_c5(h):
    """Return the overlap bound C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_constraints(h):
    """Clip to [0,1] and enforce exact integral = n/2."""
    n = h.shape[0]
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # Adjust to meet the exact sum while staying in [0,1]
    current = h.sum()
    diff = target_sum - current
    if abs(diff) < 1e-12:
        return h

    # Distribute the difference uniformly, then re‑clip and iterate a few times
    for _ in range(10):
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        current = h.sum()
        diff = target_sum - current
        if abs(diff) < 1e-12:
            break
    return h

def hill_climb(initial_h, time_limit, rng):
    """Simple stochastic hill‑climbing keeping the sum constraint."""
    best_h = project_to_constraints(initial_h.copy())
    best_val = compute_c5(best_h)
    n = best_h.size
    start = time.time()
    max_iter = 200000

    # parameters
    delta_max = 0.1          # largest amount moved in one step
    patience = 0
    while time.time() - start < time_limit and patience < 5000 and max_iter > 0:
        max_iter -= 1
        # pick two distinct indices where a transfer is possible
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]

        # direction: try moving from higher to lower (to keep bounds)
        if hi <= 0.0 or hj >= 1.0:
            continue
        delta = rng.random() * min(delta_max, hi, 1.0 - hj)
        if delta <= 0.0:
            continue

        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_val = compute_c5(new_h)
        if new_val < best_val - 1e-9:
            best_h, best_val = new_h, new_val
            patience = 0
        else:
            patience += 1

    return best_h, best_val

def upscale(h, new_n):
    """Interpolate a vector h to length new_n preserving the integral."""
    old_n = h.size
    x_old = np.linspace(0, 2, old_n, endpoint=False) + 1.0/old_n
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    # enforce exact integral after interpolation
    return project_to_constraints(h_new)
