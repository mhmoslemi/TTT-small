import numpy as np
import time
import random

# Helper to compute the overlap constant C5 for a given h
def evaluate_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

# Project a vector onto the feasible set:
#   0 ≤ h_i ≤ 1 and sum(h) = n/2
def project_feasible(h):
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(15):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            # all at bounds – distribute uniformly and re‑clip
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            adjust = diff / cnt
            h[mask] += adjust
            h = np.clip(h, 0.0, 1.0)
    # final scaling if a tiny residual remains
    s = h.sum()
    if s > 0:
        h *= target / s
    return np.clip(h, 0.0, 1.0)


def random_feasible(n):
    """Create a random feasible h of length n."""
    h = np.random.rand(n)
    return project_feasible(h)
