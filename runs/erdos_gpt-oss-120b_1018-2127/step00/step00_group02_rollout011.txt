import numpy as np
import time
import random

def _c5_score(h, dx):
    """Compute C5 = max correlation of h with (1-h)."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _random_move(h, step, rng):
    """Propose a new h by moving a small amount between two positions,
    preserving the sum and staying in [0,1]. Returns new array (copy)."""
    n = h.shape[0]
    i, j = rng.sample(range(n), 2)
    hi, hj = h[i], h[j]

    # decide direction
    if rng.random() < 0.5:
        # try to increase i, decrease j
        max_inc = min(step, 1.0 - hi, hj)
        if max_inc <= 0:
            return None
        delta = rng.uniform(0, max_inc)
    else:
        # try to increase j, decrease i
        max_inc = min(step, 1.0 - hj, hi)
        if max_inc <= 0:
            return None
        delta = -rng.uniform(0, max_inc)

    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h (as a NumPy array), its C5 bound, and n_points."""
    rng = random.Random(seed)
    np_rng = np.random.default_rng(seed)

    # Determine initial resolution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        # fallback: 45 points binary random
        n = 45
        h0 = np.zeros(n, dtype=float)
        ones = n // 2
        idx = np_rng.choice(n, ones, replace=False)
        h0[idx] = 1.0

    dx = 2.0 / n
    target_sum = n / 2.0  # because integral should be 1

    # Project initial vector to feasible set (ensure sum exact)
    cur = h0.copy()
    cur_sum = cur.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Simple scaling then clipping
        cur = cur * (target_sum / cur_sum)
        cur = np.clip(cur, 0.0, 1.0)
        # Adjust residual by adding 1 to first entries not at 1
        residual = target_sum - cur.sum()
        for i in range(n):
            if residual <= 0:
                break
            add = min(1.0 - cur[i], residual)
            cur[i] += add
            residual -= add

    best_h = cur.copy()
    best_score = _c5_score(best_h, dx)

    start_time = time.time()
    time_limit = budget_s * 0.94  # keep a safety margin

    # Parameters for stochastic local search
    step_size = 0.2          # maximal transfer amount
    temp_start = 0.02
    temp_end = 1e-5
    decay = 0.9995

    iteration = 0
    temp = temp_start

    while time.time() - start_time < time_limit:
        iteration += 1
        candidate = _random_move(best_h, step_size, rng)
        if candidate is None:
            continue
        cand_score = _c5_score(candidate, dx)

        # Accept if better, else maybe accept by SA probability
        if cand_score < best_score or rng.random() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
            best_h = candidate
            best_score = cand_score

        # Temperature schedule
        temp = max(temp * decay, temp_end)

        # Occasionally shrink step size to fine‑tune
        if iteration % 5000 == 0:
            step_size = max(step_size * 0.8, 0.01)

    # Final evaluation (ensure feasibility)
    # Clip once more and renormalize sum exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h = best_h * (target_sum / cur_sum)
        # tiny correction loop
        residual = target_sum - best_h.sum()
        for i in range(n):
            if abs(residual) < 1e-12:
                break
            adj = np.sign(residual) * min(abs(residual), 1.0 - best_h[i] if residual > 0 else best_h[i])
            best_h[i] += adj
            residual -= adj

    final_score = _c5_score(best_h, dx)

    return best_h, final_score, n
