import time
import numpy as np

def _objective(h):
    """Return C5 for a given h using FFT convolution."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    conv = np.fft.irfft(H * G, m)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def _random_swap_move(h, rng):
    """Transfer a random amount of mass between two indices, preserving sum and bounds."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    hi, hj = h[i], h[j]

    # feasible transfer from j to i
    max_from_j = min(1.0 - hi, hj)
    # feasible transfer from i to j
    max_from_i = min(hi, 1.0 - hj)

    if max_from_j == 0 and max_from_i == 0:
        return h.copy()

    if rng.random() < 0.5 and max_from_j > 0:
        delta = rng.random() * max_from_j
        new = h.copy()
        new[i] = hi + delta
        new[j] = hj - delta
    else:
        delta = rng.random() * max_from_i
        new = h.copy()
        new[i] = hi - delta
        new[j] = hj + delta
    return new


def _optimize(h_start, time_budget, rng):
    """Simple simulated‑annealing style hill‑climb within a time budget."""
    end_time = time.time() + time_budget
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_val = _objective(h_cur)

    # temperature schedule
    T = 0.02
    decay = 0.9999

    while time.time() < end_time:
        h_new = _random_swap_move(h_cur, rng)
        val_new = _objective(h_new)
        delta = val_new - best_val

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-9)):
            h_cur = h_new
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()

        T *= decay
    return best_h, best_val


def _interpolate(h, new_n):
    """Linear interpolation of h onto a finer grid of size new_n."""
    old_n = h.shape[0]
    x_old = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(x_new, x_old, h)
    # enforce sum constraint exactly
    target_sum = new_n / 2.0
    cur_sum = h_new.sum()
    if cur_sum != 0:
        h_new *= target_sum / cur_sum
    # clip bounds and re‑project sum if clipping altered it
    np.clip(h_new, 0.0, 1.0, out=h_new)
    cur_sum = h_new.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h_new += (target_sum - cur_sum) / new_n
        np.clip(h_new, 0.0, 1.0, out=h_new)
    return h_new
