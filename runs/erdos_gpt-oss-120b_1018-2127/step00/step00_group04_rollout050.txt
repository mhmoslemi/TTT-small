import time
import numpy as np
import random

def compute_c5(h, dx):
    """Return max overlap integral for step vector h."""
    # correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def binary_swap(h):
    """Swap a random 1 with a random 0, returning a new copy."""
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h <= 0.5)
    if len(ones) == 0 or len(zeros) == 0:
        return h.copy()
    i = random.choice(ones)
    j = random.choice(zeros)
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    return h_new

def normalize_binary(n):
    """Create a binary vector of length n with exactly n/2 ones."""
    assert n % 2 == 0, "n must be even for binary normalization"
    h = np.zeros(n, dtype=float)
    ones_idx = random.sample(range(n), n // 2)
    h[ones_idx] = 1.0
    return h
