import time
import numpy as np

def _c5_of_h(h, dx):
    """Compute C₅ for a discrete step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _normalize_h(h, target_sum):
    """Clamp to [0,1] and adjust to hit the required sum."""
    np.clip(h, 0.0, 1.0, out=h)
    residual = target_sum - h.sum()
    # simple uniform correction, repeat a few times
    for _ in range(10):
        if abs(residual) < 1e-12:
            break
        # indices that can still move in the needed direction
        if residual > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if free.size == 0:
            break
        delta = residual / free.size
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
        residual = target_sum - h.sum()
    return h

def _random_swap(h, max_step):
    """Perform a small mass‑transfer between two positions while preserving the sum."""
    low = np.where(h < 1.0 - 1e-12)[0]
    high = np.where(h > 1e-12)[0]
    if low.size == 0 or high.size == 0:
        return h
    i = np.random.choice(low)
    j = np.random.choice(high)
    delta = min(max_step, 1.0 - h[i], h[j])
    if delta <= 0:
        return h
    delta *= np.random.rand()
    h[i] += delta
    h[j] -= delta
    return h

def _search(initial, n, time_deadline, best_global_c5):
    """Greedy local search for a given resolution."""
    target = n / 2.0
    h = _normalize_h(initial.astype(float), target)
    dx = 2.0 / n
    cur_c5 = _c5_of_h(h, dx)

    best_h_local = h.copy()
    best_c5_local = cur_c5
    step = 0.1
    it = 0
    max_iter = 30000

    while time.time() < time_deadline and it < max_iter:
        it += 1
        h_candidate = h.copy()
        for _ in range(5):
            h_candidate = _random_swap(h_candidate, step)
        c5 = _c5_of_h(h_candidate, dx)
        if c5 < cur_c5:
            h = h_candidate
            cur_c5 = c5
            if c5 < best_c5_local:
                best_h_local = h_candidate.copy()
                best_c5_local = c5
                if c5 < best_global_c5:
                    # early exit if we beat the current global best dramatically
                    pass
        if it % 2000 == 0:
            step *= 0.9
    return best_h_local, best_c5_local

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C₅ bound, and number of samples."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s * 0.99  # a little safety margin

    # resolutions to try (must stay modest for speed)
    resolutions = [80, 120, 160, 200]

    # keep the best across all resolutions
    best_h = None
    best_c5 = float("inf")
    best_n = None

    for n in resolutions:
        # initialise h either from the provided construction or a simple pattern
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            h0 = initial_h_values.astype(float).copy()
        elif "initial_h_values" in globals():
            # linear interpolation to new size
            old_n = len(initial_h_values)
            x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, initial_h_values)
        else:
            # alternating 1/0 pattern
            h0 = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)], dtype=float)

        # run local optimisation
        h_opt, c5_opt = _search(h0, n, deadline, best_c5)

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n

        if time.time() >= deadline:
            break

    # fallback – very unlikely to be needed
    if best_h is None:
        n = resolutions[0]
        best_h = np.full(n, 0.5)
        best_c5 = _c5_of_h(best_h, 2.0 / n)
        best_n = n

    # final sanity‑check (force constraints)
    best_h = _normalize_h(best_h, best_n / 2.0)
    best_c5 = _c5_of_h(best_h, 2.0 / best_n)

    return best_h, float(best_c5), int(best_n)
