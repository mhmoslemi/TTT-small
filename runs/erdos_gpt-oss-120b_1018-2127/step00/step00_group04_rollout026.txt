import time, math, random
import numpy as np

def _evaluate(h, dx):
    """Return max overlap integral for step function h (discretized)."""
    conv = np.convolve(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def _project_sum_one_half(h, target_sum):
    """Adjust h to satisfy sum constraint while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    if abs(current - target_sum) < 1e-12:
        return h
    # simple proportional scaling within bounds, then fix residual with a sweep
    scale = target_sum / current if current > 0 else 0.0
    h = np.clip(h * scale, 0.0, 1.0)
    diff = target_sum - h.sum()
    # distribute the remaining diff greedily
    if diff > 0:
        # add to entries not at 1
        idx = np.where(h < 1.0)[0]
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    elif diff < 0:
        # remove from entries not at 0
        idx = np.where(h > 0.0)[0]
        for i in idx:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h

def _anneal(initial_h, dx, time_limit):
    """Simple simulated‑annealing optimizer for the max‑correlation objective."""
    n = len(initial_h)
    target_sum = n / 2.0
    best_h = _project_sum_one_half(initial_h.copy(), target_sum)
    best_val = _evaluate(best_h, dx)

    curr_h = best_h.copy()
    curr_val = best_val

    start = time.time()
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break

        # temperature linearly decreasing to a small floor
        temp = max(1e-5, (1.0 - elapsed / time_limit) * 0.02)

        # propose a move that preserves the sum
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # determine feasible delta range
        max_up = min(0.1, 1.0 - curr_h[j])
        max_down = min(0.1, curr_h[i])
        delta = np.random.uniform(-max_down, max_up)
        if abs(delta) < 1e-12:
            continue

        new_h = curr_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        # keep within [0,1] (numerical safety)
        if new_h[i] < -1e-12 or new_h[i] > 1.0 + 1e-12:
            continue
        if new_h[j] < -1e-12 or new_h[j] > 1.0 + 1e-12:
            continue

        new_val = _evaluate(new_h, dx)

        # Metropolis acceptance
        accept = False
        if new_val < curr_val:
            accept = True
        else:
            prob = math.exp(-(new_val - curr_val) / temp)
            if random.random() < prob:
                accept = True

        if accept:
            curr_h = new_h
            curr_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap bound C5.

    Returns
    -------
    h_values : ndarray
        Optimized values of h sampled uniformly on [0,2].
    c5_bound : float
        Corresponding maximal overlap integral (the bound).
    n_points : int
        Number of discretisation points used.
    """
    random.seed(seed)
    np.random.seed(seed)

    # Determine discretisation size
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 200                     # reasonable default under 1k
        h0 = np.full(n, 0.5)        # uniform satisfies integral constraint

    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure the starting vector respects constraints
    h_start = _project_sum_one_half(h0, target_sum)

    # Allocate most of the budget to optimisation, keep a small safety margin
    start_time = time.time()
    remaining = budget_s - 0.5   # leave half‑second for finalisation
    if remaining < 5.0:
        remaining = max(0.0, budget_s - 0.1)

    # Run a single annealing pass; can be extended with restarts if time permits
    h_opt, c5_opt = _anneal(h_start, dx, remaining)

    # Final safety projection (should be unnecessary but guards against drift)
    h_opt = _project_sum_one_half(h_opt, target_sum)
    c5_opt = _evaluate(h_opt, dx)

    return h_opt, c5_opt, n
