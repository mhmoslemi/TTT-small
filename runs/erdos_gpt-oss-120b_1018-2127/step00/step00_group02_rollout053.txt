import time, math, numpy as np, random, sys
from scipy import optimize

def _compute_c5(h):
    """Discrete approximation of C5 for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair(h):
    """Clip to [0,1] and enforce the integral constraint exactly."""
    h = np.clip(h, 0.0, 1.0)
    n = h.shape[0]
    target = n / 2.0          # because ∫ h = 1  ⇒  sum(h)*dx = 1  ⇒ sum = n/2
    s = h.sum()
    if s == 0:
        h[:] = target / n
    else:
        h *= target / s
    # a final clip may slightly break the sum; fix by a tiny redistribution
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
    return np.clip(h, 0.0, 1.0)

def _random_step(h, rng, eps_scale=0.02):
    """Propose a small random redistribution that respects the box bounds."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None
    eps = rng.normal(scale=eps_scale)
    new_h = h.copy()
    # move eps from i to j
    new_h[i] = np.clip(new_h[i] - eps, 0.0, 1.0)
    new_h[j] = np.clip(new_h[j] + eps, 0.0, 1.0)
    # enforce exact sum
    target = n / 2.0
    cur = new_h.sum()
    if cur != 0:
        new_h *= target / cur
    new_h = np.clip(new_h, 0.0, 1.0)
    # tiny redistribution to kill any remaining rounding error
    diff = target - new_h.sum()
    if abs(diff) > 1e-12:
        new_h += diff / n
    return np.clip(new_h, 0.0, 1.0)

    Optimize a step function h on [0,2] with integral 1 to minimize
    C5 = max_k ∫ h(x) (1-h(x+k)) dx.

        c5_bound : float   (the attained C5 value)
        n_points : int     (number of samples, len(h_values))
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)

        n = 60                       # a modest default size
        h0 = np.full(n, 0.5, dtype=float)
    h = _repair(h0)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Simple stochastic hill‑climbing / simulated annealing loop
    # ------------------------------------------------------------------
    max_iters = 300000
    eps_scale = 0.03
    for it in range(max_iters):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        cand = _random_step(h, rng, eps_scale=eps_scale)
        if cand is None:
            continue

        cand_c5 = _compute_c5(cand)

        if cand_c5 < best_c5 - 1e-8:
