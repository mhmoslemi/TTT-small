import time
import numpy as np
import math
import random

def _evaluate(h):
    """Return C5 for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _repair(h, target_sum, max_iters=15):
    """Project h onto [0,1]^n with exact sum constraint (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iters):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        # distribute the deficit/surplus uniformly
        h += diff / h.shape[0]
        h = np.clip(h, 0.0, 1.0)
    # final small correction on free variables
    diff = target_sum - h.sum()
    if abs(diff) > 1e-9:
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # ------------------------------------------------------------------
    # Initialise problem size and starting vector
    # ------------------------------------------------------------------
    try:
        # parent construction may be provided
        init = initial_h_values  # type: ignore
        n = len(init)
        h_current = np.array(init, dtype=float)
    except Exception:
        # fallback size – a compromise between speed and resolution
        n = 200
        h_current = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    h_current = _repair(h_current, target_sum)

    best_h = h_current.copy()
    best_c = _evaluate(best_h)

    # ------------------------------------------------------------------
    # Simple simulated‑annealing / randomized local search
    # ------------------------------------------------------------------
    iter_count = 0
    max_iter = 10_000_000  # safety cap; actual stop is time
    while time.time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # temperature schedule
        t = max(0.001, 0.5 * (0.995 ** (iter_count // 200)))
        # proposal step size adapts with temperature
        step = 0.3 * t
        cand = best_h + step * np.random.randn(n)
        cand = _repair(cand, target_sum)
        cand_c = _evaluate(cand)

        if cand_c < best_c:
            best_c = cand_c
            best_h = cand
        else:
            # occasional uphill moves
            prob = math.exp(-(cand_c - best_c) / (t + 1e-12))
            if random.random() < prob:
                best_h = cand
                best_c = cand_c

        iter_count += 1

    # ------------------------------------------------------------------
    # Return result in required format
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_c), n)
