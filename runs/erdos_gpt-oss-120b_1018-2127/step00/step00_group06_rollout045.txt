import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the discrete C5 value for step function h on [0,2]."""
    # correlation of h with (1-h), full mode gives all lags
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def _repair(h: np.ndarray) -> np.ndarray:
    """Project h onto the feasible set: 0≤h≤1 and sum(h)=n/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # simple iterative projection onto the affine hyperplane
    for _ in range(5):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _random_walk(h: np.ndarray, rng: np.random.Generator, scale: float) -> np.ndarray:
    """Perturb h by Gaussian noise of given scale and repair."""
    proposal = h + rng.normal(loc=0.0, scale=scale, size=h.shape)
    return _repair(proposal)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Stochastic search for a step function h on [0,2] that minimizes C5.

    Returns
    -------
    h_values : np.ndarray
        The best step‑function sample values found.
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = h0.size
    else:
        n0 = 100                     # a modest even size if no parent is given
        h0 = np.full(n0, 0.5, dtype=float)

    if n0 % 2 == 1:                 # enforce even length for symmetry of the problem
        n0 += 1
        h0 = np.resize(h0, n0)

    best_h = _repair(h0)
    best_score = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    iteration = 0
    temperature = 1.0               # for a mild simulated‑annealing acceptance rule
    # We will allocate the bulk of the budget to this single grid;
    # multi‑grid refinements are optional and only attempted if time permits.
    max_iter_without_check = 5000   # check the clock only every so many steps

    while (time.time() - start_time) < budget_s * 0.95:
        # gradually shrink the perturbation scale as the search proceeds
        scale = 0.03 * (1.0 - iteration / 2e6)
        scale = max(scale, 0.001)   # avoid vanishing steps

        cand = _random_walk(best_h, rng, scale)
        cand_score = _compute_c5(cand)

        # Accept if better, otherwise maybe accept according to temperature
        if cand_score < best_score:
            best_h, best_score = cand, cand_score
        else:
            # Metropolis criterion
            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-9))
            if rng.random() < prob:
                best_h, best_score = cand, cand_score

        # cool down
        temperature *= 0.999995
        iteration += 1

        # periodic time check
        if iteration % max_iter_without_check == 0:
            if (time.time() - start_time) >= budget_s * 0.95:
                break

    # ------------------------------------------------------------------
    # Optional refinement on a finer grid if time remains
    # ------------------------------------------------------------------
    time_left = budget_s - (time.time() - start_time)
    if time_left > 30.0:
        # try doubling the resolution once
        n_fine = best_h.size * 2
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_fine, endpoint=False)
        h_fine = np.interp(x_new, x_old, best_h)  # linear interpolation
        h_fine = _repair(h_fine)

        # run a short local search on the finer grid
        fine_start = time.time()
        fine_iter = 0
        while (time.time() - fine_start) < min(20.0, time_left * 0.5):
            scale_f = 0.01 * (1.0 - fine_iter / 1e5)
            scale_f = max(scale_f, 0.0005)
            cand_f = _random_walk(h_fine, rng, scale_f)
            cand_f_score = _compute_c5(cand_f)
            if cand_f_score < best_score:
                best_h, best_score = cand_f, cand_f_score
                h_fine = cand_f
            fine_iter += 1

    return best_h, best_score, best_h.size
