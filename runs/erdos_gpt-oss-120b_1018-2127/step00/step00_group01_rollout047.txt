import time
import numpy as np
from math import ceil

def _evaluate(h, dx):
    """Return max overlap C5 for discrete h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _repair(h, target_sum):
    """Project h onto [0,1] and enforce exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    # simple redistribution of the sum error
    for _ in range(3):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # distribute equally, but stay in bounds
        inc = diff / len(h)
        h += inc
        h = np.clip(h, 0.0, 1.0)
    # final tiny adjustment if still off (should be tiny)
    cur = h.sum()
    if abs(cur - target_sum) > 1e-10:
        # scale uniformly then clip again
        scale = target_sum / cur if cur != 0 else 0.0
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    return h

def _perturb(h, rng, magnitude=0.05):
    """Add Gaussian noise and repair."""
    noisy = h + rng.normal(scale=magnitude, size=h.shape)
    return noisy

def _run_one_grid(seed, n, init_vec, time_limit):
    rng = np.random.default_rng(seed)
    dx = 2.0 / n
    target_sum = n / 2.0
    # start from init_vec interpolated to size n
    if init_vec is not None and len(init_vec) != n:

        h = np.interp(x_new, x_old, init_vec)
    elif init_vec is not None:
        h = init_vec.copy()
    else:
        # simple uniform start (satisfies sum)
        h = np.full(n, 0.5)
    h = _repair(h, target_sum)
    best_h = h.copy()
    best_val = _evaluate(best_h, dx)
