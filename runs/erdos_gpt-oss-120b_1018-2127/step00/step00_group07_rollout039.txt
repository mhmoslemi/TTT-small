import time, math, random
import numpy as np

def compute_c5(h):
    """Maximum overlap integral for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, target_sum):
    """Clip to [0,1] and renormalize to exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        # distribute uniformly
        h[:] = target_sum / h.size
        return h
    h *= target_sum / s
    # after scaling may slightly exceed bounds; clip again and renormalize a few times
    for _ in range(3):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        h *= target_sum / s
    return h

def anneal_one_resolution(h0, budget, seed=42):

    random.seed(seed)
    np.random.seed(seed)

    cur_score = best_score

def upscale(h, new_n):
    """Linearly interpolate h (defined on [0,2]) to new_n points."""
