import numpy as np
import random
import math
import time

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_feasible(h: np.ndarray) -> np.ndarray:
    """Clamp to [0,1] and enforce exact integral sum(h)=n/2."""
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    # distribute the small difference uniformly; then re‑clamp
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    # if still off because of clipping, fix by adjusting first element
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        adjust = target - cur
        h[0] = min(max(h[0] + adjust, 0.0), 1.0)
    return h

def _propose(h: np.ndarray, max_step: float) -> np.ndarray:
    """Generate a feasible neighbour by moving mass between two positions."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    # decide direction of transfer
    if random.random() < 0.5:
        # move from i to j
        max_transfer = min(max_step, hi, 1.0 - hj)
        delta = random.random() * max_transfer
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:
        # move from j to i
        max_transfer = min(max_step, hj, 1.0 - hi)
        delta = random.random() * max_transfer
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
    return h_new

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimized values (length n_points) satisfying the constraints.
    c5_bound : float
        The achieved C5 value for the returned h.
    n_points : int
        Number of discretisation points.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------------ #
    # Initialise h
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.size
        h = _project_feasible(h)
    else:
        n = 80
        h = np.zeros(n, dtype=float)
        ones = n // 2
        idx = np.random.choice(n, ones, replace=False)
        h[idx] = 1.0
    cur_score = _compute_c5(h)

    best_h = h.copy()
    best_score = cur_score

    # ------------------------------------------------------------------ #
    # Simulated annealing parameters
    T = 0.05                     # initial temperature
    alpha = 0.9995               # cooling factor
    max_step_initial = 0.2      # initial maximal transfer amount
    iteration = 0

    # ------------------------------------------------------------------ #
    # Main optimisation loop
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1

        # shrink step size together with temperature
        step = max_step_initial * (T / 0.05)

        h_candidate = _propose(best_h, step)
        # candidate is already feasible (sum unchanged, bounds respected)
        cand_score = _compute_c5(h_candidate)

        delta = cand_score - best_score
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            best_h = h_candidate
            best_score = cand_score

        T *= alpha
        if T < 1e-6:
            T = 1e-6

        # occasional gentle random perturbation to escape plateaus
        if iteration % 5000 == 0:
            jitter = np.random.uniform(-step, step, size=n)
            pert = best_h + jitter
            pert = _project_feasible(pert)
            pert_score = _compute_c5(pert)
            if pert_score < best_score:
                best_h = pert
                best_score = pert_score

    # ------------------------------------------------------------------ #
    # Final projection (ensures exact feasibility)
    best_h = _project_feasible(best_h)
    final_score = _compute_c5(best_h)

    return best_h, final_score, n
