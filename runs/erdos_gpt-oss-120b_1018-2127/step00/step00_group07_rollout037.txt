def project_bounds_sum(v, target):
    v = np.clip(v, 0.0, 1.0)
    current = v.sum()
    if abs(current - target) < 1e-12:
        return v
    # iterative water-filling
    # compute excess/deficit
    excess = target - current
    # indices not at bound in direction of needed adjustment
    # We'll distribute excess equally among free indices, repeat until convergence
    free = np.ones_like(v, dtype=bool)
    while abs(excess) > 1e-12:
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        delta = excess / free_idx.size
        # cap delta so that values stay within [0,1]
        for idx in free_idx:
            if delta > 0:
                max_inc = 1.0 - v[idx]
                inc = min(delta, max_inc)
                v[idx] += inc
                excess -= inc
                if v[idx] >= 1.0 - 1e-12:
                    free[idx] = False
            else:
                max_dec = v[idx]  # can subtract up to itself
                dec = max(delta, -max_dec)
                v[idx] += dec
                excess -= dec
                if v[idx] <= 1e-12:
                    free[idx] = False
        if np.all(~free):
            break
    # final clip
    return np.clip(v, 0.0, 1.0)

import numpy as np
import time
import math

def _evaluate(h):
    """Return C5 for the step function represented by array h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimise the
    Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimised samples of h on the grid.
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of samples (grid size).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        # simple feasible start: half the points set to 1, rest 0
        h0 = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        h0[idx] = 1.0

    n = h0.size
    target_sum = n / 2.0

    # Ensure the integral constraint exactly (tiny numerical drift possible)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # distribute the needed adjustment uniformly over non‑boundary entries
        free = (h0 > 0.0) & (h0 < 1.0)
        if free.any():
            h0[free] += (target_sum - cur_sum) / free.sum()
        else:
            # fallback: clip and then add a tiny correction to the first element
            h0 = np.clip(h0, 0.0, 1.0)
            h0[0] += target_sum - h0.sum()
    h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_c5 = _evaluate(best_h)

    current_h = best_h.copy()
    current_c5 = best_c5

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search
    # ------------------------------------------------------------------
    start_time = time.time()
    end_time = start_time + budget_s - 0.5  # safety margin

    temp_initial = 0.02
    temp_final = 1e-5

    while time.time() < end_time:
        # linear temperature schedule
        elapsed = time.time() - start_time
        temp = max(temp_final,
                   temp_initial * (1.0 - elapsed / budget_s))

        # propose a mass transfer between two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # amount that can be moved from i to j while respecting bounds
        max_transfer = min(current_h[i], 1.0 - current_h[j])
        if max_transfer <= 0.0:
            continue
        delta = max_transfer * rng.random() * 0.5  # restrict step size

        new_h = current_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        # evaluate candidate
        cand_c5 = _evaluate(new_h)

        # acceptance test
        if cand_c5 < current_c5:
            accept = True
        else:
            prob = math.exp(-(cand_c5 - current_c5) / temp)
            accept = rng.random() < prob

        if accept:
            current_h = new_h
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = new_h.copy()

    # ------------------------------------------------------------------
    # Return the best found solution
    # ------------------------------------------------------------------
    return best_h, float(best_c5), n
