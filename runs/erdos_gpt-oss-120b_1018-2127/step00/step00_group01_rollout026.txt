import time
import numpy as np
import random

def compute_c5(h):
    """Return the discrete C5 value for a step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def binarize_preserve_sum(v):
    """Convert any vector to a binary vector with exactly sum(v)=len(v)/2."""
    n = v.shape[0]
    half = n // 2
    idx = np.argsort(v)[::-1]          # descending order
    h = np.zeros(n, dtype=float)
    h[idx[:half]] = 1.0
    return h

def neighbor_swap(h):
    """Generate a neighbor by swapping a 0 and a 1 entry."""
    zeros = np.flatnonzero(h == 0.0)
    ones = np.flatnonzero(h == 1.0)
    if zeros.size == 0 or ones.size == 0:
        return h.copy()
    i = random.choice(zeros)
    j = random.choice(ones)
    h2 = h.copy()
    h2[i], h2[j] = 1.0, 0.0
    return h2

def hill_climb(initial_h, time_limit):
    """Simple stochastic hill‑climb with occasional uphill moves."""
    start = time.time()
    best_h = initial_h.copy()
    best_val = compute_c5(best_h)
    n = best_h.size
    # annealing schedule parameters
    temp_start = 0.02
    temp_end = 0.0005
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # linear cooling
        t = temp_start + (temp_end - temp_start) * (time.time() - start) / time_limit
        cand = neighbor_swap(best_h)
        cand_val = compute_c5(cand)
        delta = cand_val - best_val
        if delta < 0 or random.random() < np.exp(-delta / (t + 1e-12)):
            best_h, best_val = cand, cand_val
        # occasional random restart if stagnating
        if it % 5000 == 0:
            # with small probability, re‑initialize from the original seed
            if random.random() < 0.1:
                best_h = binarize_preserve_sum(np.random.rand(n))
                best_val = compute_c5(best_h)
    return best_h, best_val

def upscale_binary(h, factor):
    """Upscale a binary pattern by integer factor (repeat each entry)."""
    return np.repeat(h, factor)

    random.seed(seed)

        np.random.shuffle(h_bin)

    t0 = budget_s * 0.70
    best_h, best_val = hill_climb(h_bin, t0)

    remaining = budget_s - time_used

        h_up = upscale_binary(best_h, upscale_factor)
        # Re‑balance to exact sum after upscaling (sum doubles automatically)

        h_up_opt, val_up = hill_climb(h_up, remaining)
        if val_up < best_val:
