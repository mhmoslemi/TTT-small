import time
import numpy as np

# Helper: compute C5 for a discrete h
def _c5_of_h(h):
    n = h.size
    dx = 2.0 / n
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

# If the harness provides a specialized evaluator, use it; otherwise fall back.
def _evaluate(h):
    try:
        # evaluate_erdos_solution is pre‑imported by the harness
        return evaluate_erdos_solution(h)
    except Exception:
        return _c5_of_h(h)

# Perform a simple projected random walk (hill‑climbing / simulated‑annealing)
def _optimize(initial, time_budget, rng):
    n = initial.size
    # ensure feasibility
    h = np.clip(initial, 0.0, 1.0)
    total_needed = n / 2.0
    cur_sum = h.sum()
    if cur_sum != 0:
        h *= (total_needed / cur_sum)
    # after scaling some values may leave [0,1]; project again
    h = np.clip(h, 0.0, 1.0)
    # final correction to exact sum by adding/subtracting a tiny amount uniformly
    deficit = total_needed - h.sum()
    if abs(deficit) > 1e-12:
        h += deficit / n
        h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_score = _evaluate(best_h)

    start = time.time()
    iteration = 0
    # step size schedule
    step_max = 0.05
    while time.time() - start < time_budget * 0.95:
        iteration += 1
        # gradually shrink step size
        step = step_max * (0.5 ** (iteration // 2000))
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] <= 0.0 and h[j] >= 1.0:
            continue
        # decide transfer direction
        if h[i] > h[j]:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        max_transfer = min(step, h[donor], 1.0 - h[receiver])
        if max_transfer <= 0:
            continue
        h_new = h.copy()
        h_new[donor] -= max_transfer
        h_new[receiver] += max_transfer
        # feasibility already guaranteed by construction
        score = _evaluate(h_new)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_new
            h = h_new
        # occasional random restart to avoid stagnation
        if iteration % 5000 == 0:
            if rng.random() < 0.1:
                h = best_h.copy()
                # small random jitter respecting bounds & sum
                perturb = rng.uniform(-step, step, size=n)
                h += perturb
                # project back to [0,1]
                h = np.clip(h, 0.0, 1.0)
                # enforce exact sum
                deficit = total_needed - h.sum()
                h += deficit / n
                h = np.clip(h, 0.0, 1.0)

    return best_h, best_score

    h_opt, c5_best = _optimize(init_vec, budget_s, rng)
