import time
import numpy as np
from scipy.optimize import OptimizeResult

# helper: evaluate C5 for a given step function h
def _evaluate_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


# simple move that keeps total mass fixed and respects bounds
def _move_mass(h, i, j, amount):
    """move `amount` from index i to j, clipping to [0,1]. Returns new array."""
    hi, hj = h[i], h[j]
    move = min(amount, hi)                     # can't take more than we have
    move = min(move, 1.0 - hj)                  # can't exceed upper bound at j
    if move <= 0:
        return None
    new_h = h.copy()
    new_h[i] -= move
    new_h[j] += move
    return new_h


def _local_search(initial_h, time_budget, start_time, rng):
    """hill‑climbing improvement for a fixed grid size."""
    best_h = initial_h.copy()
    best_val = _evaluate_c5(best_h)
    n = best_h.size

    # parameters controlling search intensity
    max_step = 0.02          # maximal mass moved in one trial
    report_interval = 5.0   # seconds between optional debug prints (silently ignored)

    while time.time() - start_time < time_budget:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # try moving a random amount up to max_step
        amt = rng.random() * max_step
        new_h = _move_mass(best_h, i, j, amt)
        if new_h is None:
            continue
        val = _evaluate_c5(new_h)
        if val < best_val - 1e-9:          # accept only strict improvements
            best_h, best_val = new_h, val
        # occasional random reshuffle to escape plateaus
        if rng.random() < 0.001:
            # pick three indices and perform a small cyclic transfer
            a, b, c = rng.choice(n, size=3, replace=False)
            eps = rng.random() * max_step
            h_tmp = _move_mass(best_h, a, b, eps)
            if h_tmp is not None:
                h_tmp = _move_mass(h_tmp, b, c, eps)
                if h_tmp is not None:
                    h_tmp = _move_mass(h_tmp, c, a, eps)
                    if h_tmp is not None:
                        val = _evaluate_c5(h_tmp)
                        if val < best_val - 1e-9:
                            best_h, best_val = h_tmp, val
    return best_h, best_val
