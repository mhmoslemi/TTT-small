import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Discrete approximation of C5 for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum_and_bounds(h, target):
    """Project h onto [0,1] box and enforce exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    # simple iterative correction
    for _ in range(10):
        cur = h.sum()
        if abs(cur - target) < 1e-12:
            break
        delta = target - cur
        # distribute delta to interior points (not at bounds)
        interior = (h > 0.0) & (h < 1.0)
        if interior.any():
            h[interior] += delta / interior.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # all points at a bound, distribute uniformly
            h += delta / h.size
            h = np.clip(h, 0.0, 1.0)
    return h

def local_optimize(start):
    """Run a constrained SLSQP optimization from a given start vector."""
    n = start.size
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = ({
        "type": "eq",
        "fun": lambda x: x.sum() - target_sum,
    },)

    res = minimize(
        compute_c5,
        start,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": 2000, "ftol": 1e-12, "disp": False},
    )
    h = project_sum_and_bounds(res.x, target_sum)
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
    h_values is a NumPy array describing the step function,
    c5_bound is the achieved upper bound,
    n_points is the number of discretization samples.
    """
    rnd = np.random.RandomState(seed)
    start_time = time.time()

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 45
        h0 = np.full(n, 0.5, dtype=float)

    # Ensure feasibility of the initial vector
    h0 = project_sum_and_bounds(h0, n / 2.0)

    best_h, best_c5 = local_optimize(h0)

    # Simple multi‑restart loop while time permits
    while time.time() - start_time < budget_s * 0.95:
        # perturb current best solution
        noise = rnd.normal(scale=0.1, size=n)
        pert = best_h + noise
        pert = project_sum_and_bounds(pert, n / 2.0)

        h_candidate, c5_candidate = local_optimize(pert)

        if c5_candidate < best_c5:
            best_h, best_c5 = h_candidate, c5_candidate

        # limit number of attempts
        if time.time() - start_time > budget_s * 0.9:
            break

    return best_h, float(best_c5), int(n)
