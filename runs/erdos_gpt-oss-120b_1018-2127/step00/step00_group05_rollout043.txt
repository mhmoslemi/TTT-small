import numpy as np
import time

def _project_feasible(v, target):
    """Project vector v onto the box [0,1] and the affine sum constraint."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(15):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = v < 1.0 - 1e-12
        else:
            idx = v > 1e-12
        if not np.any(idx):
            break
        adjust = diff / idx.sum()
        v[idx] += adjust
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction
    s = v.sum()
    if abs(s - target) > 1e-10:
        v += (target - s) / len(v)
        v = np.clip(v, 0.0, 1.0)
    return v

def _evaluate(h, dx):
    """Return C5 for a feasible h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _initial_h(n_points):
    """Build an initial guess from the global initial_h_values (if present)."""
    try:
        base = np.asarray(initial_h_values, dtype=float)
        if len(base) == n_points:
            return _project_feasible(base, n_points / 2.0)
        # linear interpolation / nearest‑neighbor up‑sample
        x_old = np.linspace(0, 2, len(base), endpoint=False) + 1.0 / len(base)
        x_new = np.linspace(0, 2, n_points, endpoint=False) + 1.0 / n_points
        up = np.interp(x_new, x_old, base)
        return _project_feasible(up, n_points / 2.0)
    except NameError:
        # No parent provided – start from a simple two‑block shape
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        return _project_feasible(h, n_points / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with integral 1 to minimise
    C5 = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Choose resolution (prefer a modest size for speed)
    n_points = kwargs.get('n_points', 100)
    if n_points < 30:
        n_points = 30
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial population (one elite)
    best_h = _initial_h(n_points)
    best_score = _evaluate(best_h, dx)

    # Parameters for stochastic search
    sigma = 0.2                     # initial noise magnitude
    decay = 0.9995                  # exponential decay per iteration
    max_iter = 200000               # generous upper bound; will stop by time
    report_interval = 5000

    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s - 5.0:
            break

        # generate candidate by Gaussian perturbation
        cand = best_h + sigma * np.random.randn(n_points)
        cand = _project_feasible(cand, target_sum)

        score = _evaluate(cand, dx)

        # accept if improved
        if score < best_score:
            best_h, best_score = cand, score
            # occasional aggressive reset if we get very low score
            if best_score < 0.34:
                sigma = 0.2
        # decay noise
        sigma *= decay
        if sigma < 1e-5:
            sigma = 1e-5

        # occasional random restart to escape stagnation

            cand_score = _evaluate(cand, dx)
            if cand_score < best_score:
                best_h, best_score = cand, cand_score

        # optional progress check (no output, just internal)
        if it % report_interval == 0 and it > 0:
            # if no improvement in the last interval, boost sigma
            pass

    return best_h, float(best_score), n_points
