def project_onto_constraints(v, target):
    # clip to [0,1]
    v = np.clip(v, 0, 1)
    # if sum close target, done
    # else adjust using iterative: while True:
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # indices that are not at bounds in direction of diff
    if diff > 0:
        free = v < 1 - 1e-12
    else:
        free = v > 1e-12
    if not np.any(free):
        # cannot adjust, return clipped
        return v
    add = diff / free.sum()
    v[free] += add
    # after adding, may violate bounds, clip and repeat

import time
import numpy as np

def _project_onto_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1] with the affine sum constraint sum(v)=target_sum."""
    v = np.clip(v, 0.0, 1.0)
    # Simple water-filling loop
    for _ in range(10):
        diff = target_sum - v.sum()
        if abs(diff) < 1e-12:
            break
        # indices that can move in the needed direction
        if diff > 0:
            free = v < 1.0 - 1e-12
        else:
            free = v > 1e-12
        if not np.any(free):
            break
        add = diff / free.sum()
        v[free] += add
        v = np.clip(v, 0.0, 1.0)
    return v


def _c5_of(h: np.ndarray) -> float:
    """Compute C5 for a discretised step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h:[0,2]->[0,1] with ∫h = 1 to minimize
        C5 = max_k ∫ h(x)(1-h(x+k)) dx .
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    target_sum = n / 2.0
    h = _project_onto_constraints(h, target_sum)

    best_h = h.copy()
    best_score = _c5_of(best_h)

    start_time = time.time()
    it = 0
    # ------------------------------------------------------------------
