import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Discrete C5 for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(x):
    """Project vector to [0,1] and enforce mean 0.5 (integral=1)."""
    x = np.clip(x, 0.0, 1.0)
    mean = x.mean()
    if mean != 0.5:
        x = x - (mean - 0.5)
        x = np.clip(x, 0.0, 1.0)
    # after clipping the mean may drift a bit; perform a final tiny correction
    diff = 0.5 - x.mean()
    if abs(diff) > 1e-12:
        x += diff
        x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the C5 bound."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size

        n = h0.size
        # ensure it respects constraints (might be slightly off)
        h0 = project_to_constraints(h0)
    else:
        n = 45
        h0 = np.full(n, 0.5, dtype=float)  # uniform satisfies integral

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # SLSQP constraints
    cons = ({
        'type': 'eq',
        'fun': lambda x: x.mean() - 0.5,
        'jac': lambda x: np.full_like(x, 1.0 / n)
    },)

    bounds = [(0.0, 1.0)] * n

    attempt = 0
    max_attempts = 30

    while time.time() - start_time < budget_s - 5 and attempt < max_attempts:
        # perturb current best

        x0 = best_h + perturb
        x0 = project_to_constraints(x0)

        res = minimize(
