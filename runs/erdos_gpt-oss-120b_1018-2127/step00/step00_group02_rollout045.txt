import numpy as np
import time

def _c5_from_h(h):
    """Compute C5 for a discretized step function h."""
    dx = 2.0 / h.size
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_bounds_sum(x, target):
    """Project vector x onto [0,1]^n while forcing sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(15):  # water‑filling iterations
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        if not np.any(free):
            # all entries at a bound – cannot adjust further
            break
        adjust = diff / free.sum()
        x[free] += adjust
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction to hit the target exactly (may introduce tiny bound violation)
    correction = (target - x.sum()) / x.size
    x += correction
    return np.clip(x, 0.0, 1.0)

    Returns
    -------
    h_values : numpy.ndarray
        The best discretized function found (values in [0,1]).
    c5_bound : float
        Corresponding C₅ upper bound.
    n_points : int
        Number of samples (len(h_values)).
    """
    rnd = np.random.RandomState(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
