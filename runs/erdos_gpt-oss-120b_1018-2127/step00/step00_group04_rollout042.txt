import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for discretized h."""
    n = h.size
    dx = 2.0 / n
    m = 2 * n
    # linear convolution via FFT (equivalent to np.correlate(full))
    H = np.fft.rfft(h, n=m)
    G = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(H * np.conj(G), n=m)[: 2 * n - 1]
    return np.max(conv) * dx


def _make_initial_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a feasible starting vector (values in [0,1], sum = n/2)."""
    # start from the constant 0.5 (which already satisfies the sum)
    h = np.full(n, 0.5, dtype=float)
    # inject a small random perturbation while keeping feasibility
    for _ in range(n // 2):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a random amount from i to j
        delta = rng.random() * 0.2
        delta = min(delta, h[i])          # cannot go below 0
        delta = min(delta, 1.0 - h[j])    # cannot exceed 1
        h[i] -= delta
        h[j] += delta
    return h

    Returns a tuple (h_values, c5_bound, n_points)

    h_values : list of float, length n_points, each in [0,1] and sum = n_points/2

    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # choose a modest size – large enough to give flexibility, small enough for speed
    n = 80                       # must be even so that sum = n/2 is integer
    target_sum = n / 2.0
    dx = 2.0 / n

    # initialise h (use provided initial_h_values if present)
    try:

        # interpolate to length n
        x_src = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_tgt = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_tgt, x_src, init)
        # project to the feasible set (clip, then adjust sum by a simple redistribution)
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        if cur_sum == 0:
            h = np.full(n, 0.5)
        else:
            h *= target_sum / cur_sum
            # after scaling we may violate bounds by a tiny amount; fix it
            overflow = h - 1.0
            underflow = -h

                h[overflow > 0] = 1.0

                h[underflow > 0] = 0.0

                if free > 0:

        # final safety renorm (preserves bounds)
        h = np.clip(h, 0.0, 1.0)

        if abs(cur_sum - target_sum) > 1e-8:
