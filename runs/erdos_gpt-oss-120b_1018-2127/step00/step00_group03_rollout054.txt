import time
import numpy as np

def _project_feasible(v, target_sum):
    """Clip to [0,1] and adjust to achieve the exact target sum."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if np.isclose(s, target_sum):
        return v
    if s < target_sum:          # need to increase some entries
        idx = np.where(v < 1.0)[0]
        while s < target_sum and idx.size:
            room = 1.0 - v[idx]
            delta = target_sum - s
            inc = np.minimum(delta / idx.size, room)
            v[idx] += inc
            s = v.sum()
            idx = np.where(v < 1.0 - 1e-12)[0]
    else:                        # need to decrease some entries
        idx = np.where(v > 0.0)[0]
        while s > target_sum and idx.size:
            room = v[idx]
            delta = s - target_sum
            dec = np.minimum(delta / idx.size, room)
            v[idx] -= dec
            s = v.sum()
            idx = np.where(v > 1e-12)[0]
    return v

def _c5_score(h):
    """Return the overlap integral C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _random_feasible(n, target_sum, rng):
    """Draw a random feasible vector (values in [0,1] summing to target_sum)."""
    # start from Dirichlet-like sample
    v = rng.random(n)
    v = _project_feasible(v, target_sum)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function on [0,2] with small C5."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():

        n = h.size

        h = _random_feasible(n, n / 2.0, rng)

    target_sum = n / 2.0
    h = _project_feasible(h, target_sum)

    best_h = h.copy()
    best_score = _c5_score(best_h)

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search
    # ------------------------------------------------------------------
    # schedule parameters
    T0 = 0.02
    T_min = 1e-5
    decay = 0.9995

    temperature = T0
    iteration = 0
    max_iter_without_improve = 20000
    since_best = 0

    while time.time() - start_time < budget_s - 0.5:
        iteration += 1
        # propose a transfer between two coordinates
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # ensure i can give and j can receive
        max_give = min(0.02, h[i])          # limit step size for stability
        max_receive = min(0.02, 1.0 - h[j])
        eps = min(max_give, max_receive) * rng.random()
        if eps <= 0:
            continue

        h_new = best_h.copy()
        h_new[i] -= eps
        h_new[j] += eps

        # already respects bounds, sum unchanged, but numerical drift possible
        h_new = _project_feasible(h_new, target_sum)

        score_new = _c5_score(h_new)

        # acceptance rule
        if score_new < best_score or rng.random() < np.exp(-(score_new - best_score) / temperature):
            best_h = h_new
            best_score = score_new
            since_best = 0
        else:
            since_best += 1

        # temperature cooling

            h_restart = _random_feasible(n, target_sum, rng)
            score_restart = _c5_score(h_restart)
            if score_restart < best_score:
                best_h = h_restart
                best_score = score_restart
            since_best = 0

    # Return result
    # ------------------------------------------------------------------
    return best_h.tolist(), float(best_score), n
