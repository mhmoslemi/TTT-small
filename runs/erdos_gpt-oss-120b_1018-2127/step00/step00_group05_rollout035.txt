import time
import numpy as np

def _c5_value(h, dx):
    """Compute C5 for a step vector h given dx."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _binary_initial(n):
    """Return a binary vector of length n with exactly n/2 ones, shuffled."""
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h

def _hill_climb_binary(n, time_budget, start_best=None):
    """Simple hill‑climbing on binary vectors with exactly n/2 ones."""
    dx = 2.0 / n
    # start from provided vector if feasible, else a fresh random binary vector
    if start_best is not None and np.isclose(np.sum(start_best), n / 2):
        h_curr = start_best.copy()
    else:
        h_curr = _binary_initial(n)

    best_h = h_curr.copy()
    best_c = _c5_value(best_h, dx)

    # Pre‑compute index arrays for ones and zeros to make swaps cheap
    ones_idx = np.flatnonzero(best_h == 1.0)
