import time, math, random
import numpy as np

def _compute_score(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto the box [0,1]^n with prescribed sum = target."""
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative redistribution
    for _ in range(5):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # free variables (not at a bound)
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt == 0:
            # all at bounds – stretch uniformly then re‑clip
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    # Final scaling if tiny residual error remains
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    return h

    Returns a tuple (h_values, c5_bound, n_points) where

    # enforce feasibility

    best_h = h_cur.copy()

    # ---- simulated‑annealing parameters --------------------------------------
    T0 = 1.0                 # initial temperature
    T = T0
    cooling = 0.995

    def _perturb(h):
