import time
import numpy as np
import random
import math

def compute_c5(h):
    """Maximum overlap integral C5 for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def project_onto_box_simplex(v, target):
    """
    Project vector v onto the feasible set:
        0 <= x_i <= 1  and  sum x_i = target
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def ensure_feasible(h):
    """Clip to [0,1] and adjust sum to required value."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-12:
        h = project_onto_box_simplex(h, target)
    return h

def initial_vector():
    """Create an initial feasible vector when no parent is supplied."""
    n = 70
    h = np.full(n, 0.5)          # sum = n/2 already
    return ensure_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # Obtain starting vector
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values, dtype=float)
    else:
        h_current = initial_vector()
    h_current = ensure_feasible(h_current)
    n = h_current.size
    target_sum = n / 2.0

    best_h = h_current.copy()
    best_score = compute_c5(best_h)

    start_time = time.time()
    time_limit = budget_s * 0.95  # safety margin
    # Parameters for the stochastic search
    sigma = 0.05          # std dev of Gaussian perturbation
    improve_target = 0.3808

    # Main stochastic hill‑climbing loop
    while time.time() - start_time < time_limit:
        # Propose a new candidate by Gaussian perturbation
        candidate = best_h + np.random.normal(scale=sigma, size=n)
        candidate = ensure_feasible(candidate)

        score = compute_c5(candidate)

        # Accept if improvement
        if score < best_score:
            best_score = score
            best_h = candidate

        # Early exit if we beat the target bound
        if best_score <= improve_target:
            break

        # Occasionally shrink perturbation to fine‑tune
        if random.random() < 0.01:
            sigma *= 0.9
            sigma = max(sigma, 1e-4)

    # Return as a Python list for compatibility
    return (best_h.tolist(), best_score, n)
