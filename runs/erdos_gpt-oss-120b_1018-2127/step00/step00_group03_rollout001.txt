import numpy as np
import time

def _c5_score(h):
    """Return the C5 value for a step function defined by samples h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_to_feasible(v, target_sum):
    """
    Project vector v onto the box [0,1]^n while enforcing sum(v)=target_sum.
    Uses a simple bisection on the Lagrange multiplier lambda:
        x_i = clip(v_i - lambda, 0, 1)
    """
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    # final projection
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # numerical cleanup: ensure exact sum by a tiny redistribution if needed
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny residual uniformly over interior points
        interior = (x > 0.0) & (x < 1.0)
        if interior.any():
            x[interior] += diff / interior.sum()
            x = np.clip(x, 0.0, 1.0)
    return x
