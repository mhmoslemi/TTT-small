import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that minimizes
    C5 = max_k ∫ h(x)*(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    t_start = time.time()

    # Determine an initial resolution; use initial_h_values length if available
    try:
        init_len = len(initial_h_values)  # provided by harness when present
        n_points = max(100, min(300, init_len * 2))
    except Exception:
        n_points = 200  # default resolution (even)

    # Ensure even number for exact half integral
    if n_points % 2 == 1:
        n_points += 1

    dx = 2.0 / n_points
    target_sum = n_points // 2  # because sum(h) must equal n_points/2

    # Evaluation of C5 for a candidate vector h (float array in [0,1])
    def evaluate(h_vec):
        # h_vec is assumed feasible (0/1), but we tolerate small numeric noise
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * dx

    # Helper: generate a random binary vector with exact half ones
    def random_binary():
        h = np.zeros(n_points, dtype=np.float64)
        ones = np.random.choice(n_points, target_sum, replace=False)
        h[ones] = 1.0
        return h

    # Optional: start from a repaired version of initial_h_values if it exists
    best_h = None
    best_score = np.inf

    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.shape[0] != n_points:
            # Resample via nearest‑neighbor interpolation
            idx = (np.linspace(0, init.shape[0]-1, n_points)).astype(int)
            init = init[idx]
        # Clip and enforce integral by a simple scaling‑and‑clipping loop
        h0 = np.clip(init, 0.0, 1.0)
        for _ in range(5):
            s = h0.sum()
            if s == 0:
                h0[:] = 1.0 / n_points
                break
            factor = target_sum / s
            h0 = np.clip(h0 * factor, 0.0, 1.0)
        # Finally force binary by thresholding at 0.5 (keeps sum close to target)
        h0 = (h0 >= 0.5).astype(np.float64)
        # Adjust if sum deviates
        cur = int(h0.sum())
        while cur > target_sum:
            zeros = np.where(h0 == 1.0)[0]
            h0[np.random.choice(zeros)] = 0.0
            cur -= 1
        while cur < target_sum:
            zeros = np.where(h0 == 0.0)[0]
            h0[np.random.choice(zeros)] = 1.0
            cur += 1
        score0 = evaluate(h0)
        best_h, best_score = h0, score0
    except Exception:
        # No usable initial vector; continue with random search
        pass

    # Main random‑restart hill‑climbing loop
    # Allocate most of the budget to the search; keep a small safety margin
    while time.time() - t_start < budget_s - 0.5:
        # generate a fresh random binary start
        h = random_binary()
        cur_score = evaluate(h)

        # Simple local improvement: try a limited number of swaps
        for _ in range(80):
            # pick a 1‑index and a 0‑index to swap
            ones_idx = np.flatnonzero(h == 1.0)
            zeros_idx = np.flatnonzero(h == 0.0)
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            h[i], h[j] = h[j], h[i]          # swap
            new_score = evaluate(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]      # revert

        # Occasionally keep the best found from this restart without local search
        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()

    # As a final safeguard, if nothing was found (should not happen), fall back to uniform 0.5
    if best_h is None:
        best_h = np.full(n_points, 0.5, dtype=np.float64)
        best_score = evaluate(best_h)

    return best_h, float(best_score), n_points
