import time
import numpy as np
from math import exp

def _project_to_feasible(h, target_sum):
    """Project vector h onto [0,1] box with exact sum = target_sum."""
    # Simple iterative water-filling
    for _ in range(10):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # distribute uniformly (rare) and clip again
            h += diff / len(h)
            continue
        delta = diff / cnt
        h[free] += delta
    # final clip to iron out tiny violations
    h = np.clip(h, 0.0, 1.0)
    # tiny residual correction by adding to first non‑saturated entry
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        idx = np.where((h > 0.0) & (h < 1.0))[0]
        if idx.size:
            h[idx[0]] += residual
            h = np.clip(h, 0.0, 1.0)
    return h

def _initial_guess(n):
    """Construct a smooth cosine‑shaped start that respects the constraints."""
    dx = 2.0 / n
    x = np.arange(n) * dx
    h = 0.5 + 0.5 * np.cos(np.pi * x)          # values lie in [0,1]
    target = n / 2.0
    h = _project_to_feasible(h, target)
    return h

def _evaluate(h):
    """Return the C5 value for the given step function."""
    # evaluate_erdos_solution is provided by the harness
    return evaluate_erdos_solution(h)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h minimizing the Erdős C5 overlap bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized samples of h on [0,2].
    c5_bound : float
        Corresponding C5 upper bound (the objective value).
    n_points : int
        Number of discretization points used.
    """
    rnd = np.random.RandomState(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals() and isinstance(initial_h_values, np.ndarray):
        h_curr = np.array(initial_h_values, dtype=float)
        n = h_curr.size
        # enforce exact sum constraint (may be slightly off due to rounding)
        h_curr = _project_to_feasible(h_curr, n / 2.0)
    else:
        n = 42  # default modest size
        h_curr = _initial_guess(n)

    target_sum = n / 2.0
    best_h = h_curr.copy()
    best_val = _evaluate(best_h)

    # Simulated annealing parameters
    temp = 0.2
    temp_decay = 0.9995
    min_temp = 1e-5
    it = 0

    while time.time() - start_time < budget_s:
        it += 1
        # Propose a move: transfer a small amount between two random indices
        i, j = rnd.choice(n, 2, replace=False)
        eps = rnd.uniform(-0.05, 0.05)
        h_new = h_curr.copy()
        h_new[i] -= eps
        h_new[j] += eps
        h_new = _project_to_feasible(h_new, target_sum)

        val_new = _evaluate(h_new)

        # Accept if better or according to Metropolis criterion
        improve = val_new < best_val
        if improve or rnd.rand() < exp(-(val_new - best_val) / max(temp, 1e-12)):
            h_curr = h_new
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()

        # Temperature schedule
