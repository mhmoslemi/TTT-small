import time
import random
import numpy as np

def _c5_value(h):
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _initial_vector(n):
    """Fallback initial vector: first n/2 entries 1, rest 0 (integral =1)."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[:half] = 1.0
    return h
