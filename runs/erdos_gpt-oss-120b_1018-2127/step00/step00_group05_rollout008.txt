import time
import numpy as np
import random

def _repair(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and enforce sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2):
        s = h.sum()
        if s == 0:
            h = np.full_like(h, target_sum / h.size)
            break
        scale = target_sum / s
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
    # final normalization (tiny residual)
    s = h.sum()
    if s != 0:
        h *= target_sum / s
    return h

def _evaluate(h: np.ndarray) -> float:
    """Return C5 = dx * max correlation of h with (1-h)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(corr.max())

def _mutate_A(parent: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Resample 20% of entries uniformly."""
    n = parent.size
    child = parent.copy()
    k = max(1, n // 5)
    idx = rng.choice(n, size=k, replace=False)
    child[idx] = rng.random(k)  # uniform [0,1)
    return _repair(child, target_sum)

def _mutate_B(parent: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Add small noise to ~30% of entries."""
    n = parent.size
    child = parent.copy()
    mask = rng.random(n) < 0.30
    noise = (rng.random(mask.sum()) * 2.0 - 1.0) * 0.015
    child[mask] += noise
    child = np.clip(child, 0.0, 1.0)
    return _repair(child, target_sum)
