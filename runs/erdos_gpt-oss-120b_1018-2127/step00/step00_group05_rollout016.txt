import time
import numpy as np

def _evaluate_c5(h, dx):
    """Return C5 for a given h vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h, target_sum):
    """Clip to [0,1] and enforce exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    if abs(current - target_sum) <= 1e-12:
        return h
    excess = current - target_sum
    if excess > 0.0:
        # need to reduce total mass
        for idx in np.argsort(-h):  # start with largest entries
            dec = min(h[idx], excess)
            h[idx] -= dec
            excess -= dec
            if excess <= 1e-12:
                break
    else:
        deficit = -excess
        for idx in np.argsort(h):   # start with smallest entries
            inc = min(1.0 - h[idx], deficit)
            h[idx] += inc
            deficit -= inc
            if deficit <= 1e-12:
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Feasible vector (length n_points) with values in [0,1] and sum = n_points/2.
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretization points.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n_points = init.shape[0]
    except NameError:
        n_points = 120  # fallback size
        init = np.full(n_points, 0.5, dtype=float)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Make sure the starting vector is feasible
    h_current = _repair(init.copy(), target_sum)
    best_h = h_current.copy()
    best_score = _evaluate_c5(best_h, dx)

    # Simulated‑annealing parameters
    max_iter = 200000
    temp = 0.02
    temp_decay = 0.99995
    max_delta = 0.12   # maximal amount moved between two coordinates

    iteration = 0
    while iteration < max_iter and (time.time() - start_time) < budget_s * 0.95:
        # propose a move: transfer delta from i to j
        i, j = np.random.choice(n_points, 2, replace=False)
        # feasible delta range respecting box constraints
        max_take = min(h_current[i], max_delta)
        max_give = min(1.0 - h_current[j], max_delta)
        delta_max = min(max_take, max_give)
        if delta_max <= 0:
            iteration += 1
            continue
        delta = np.random.rand() * delta_max
        h_candidate = h_current.copy()
        h_candidate[i] -= delta
        h_candidate[j] += delta

        # evaluate
        cand_score = _evaluate_c5(h_candidate, dx)

        # accept with SA rule
        if cand_score < best_score or np.random.rand() < np.exp(-(cand_score - best_score) / (temp + 1e-12)):
            h_current = h_candidate
            if cand_score < best_score:
                best_h = h_candidate
                best_score = cand_score

        temp *= temp_decay
        iteration += 1

    # Final repair to guarantee feasibility
    best_h = _repair(best_h, target_sum)
