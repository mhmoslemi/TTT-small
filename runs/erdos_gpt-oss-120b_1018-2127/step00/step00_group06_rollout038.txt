import time, math, random
import numpy as np

def _project_sum(vec, target):
    """Clip to [0,1] and adjust to hit exact target sum."""
    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    if abs(cur - target) < 1e-12:
        return vec
    diff = target - cur
    # distribute proportionally to available capacity
    if diff > 0:
        cap = 1.0 - vec
    else:
        cap = vec
    cap_sum = cap.sum()
    if cap_sum == 0:
        return vec
    vec += diff * (cap / cap_sum)
    vec = np.clip(vec, 0.0, 1.0)
    # tiny residual fix
    residual = target - vec.sum()
    if abs(residual) > 1e-12:
        free = (vec > 0) & (vec < 1)
        if free.any():
            vec[free] += residual / free.sum()
            vec = np.clip(vec, 0.0, 1.0)
    return vec
