import time
import numpy as np
import random

def _evaluate_c5(h):
    """Return the C5 bound for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives linear convolution
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _initial_binary(n, rng):
    """Create a random binary vector of length n with exactly n/2 ones."""
    ones = np.ones(n // 2, dtype=float)
    zeros = np.zeros(n - n // 2, dtype=float)
    h = np.concatenate([ones, zeros])
    rng.shuffle(h)
    return h

    Optimize a step function h on [0,2] to minimize the Erdős C5 bound.

        n_points (int): number of discretisation points.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # Determine discretisation size
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        # enforce even n for exact half‑mass binary representation
        if n % 2:
            n += 1
    else:
        n = 120  # modest size, even already

    # Prepare initial candidate
    if 'initial_h_values' in globals():
        h = np.clip(np.asarray(initial_h_values, dtype=float), 0.0, 1.0).copy()
        # enforce exact integral constraint by simple scaling then rounding
        total = h.sum()
        target = n / 2
        if abs(total - target) > 1e-8:
            # replace with binary start if scaling would violate bounds
            h = _initial_binary(n, rng)
    else:
        h = _initial_binary(n, rng)

    # Ensure exact mass constraint
    # For binary vectors this holds automatically; otherwise project
    if not np.isclose(h.sum(), n / 2):
        # project by sorting and thresholding
        idx = np.argsort(-h)  # descending
        h[:] = 0.0
        h[idx[: n // 2]] = 1.0

    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)

    # Keep index lists for fast swapping
    ones_idx = np.flatnonzero(best_h == 1.0).tolist()
    zeros_idx = np.flatnonzero(best_h == 0.0).tolist()

    # Simulated‑annealing parameters
    T = 0.02  # initial temperature
    decay = 0.9995
    min_T = 1e-6

    # Main optimisation loop
    while time.time() - start_time < budget_s * 0.94:
        # pick a random 1 and 0 to swap
        i = random.choice(ones_idx)
        j = random.choice(zeros_idx)

        h[i], h[j] = 0.0, 1.0
        new_c5 = _evaluate_c5(h)

        if delta < 0:
            accept = True

            if T > min_T and random.random() < np.exp(-delta / T):
