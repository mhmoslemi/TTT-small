import time
import numpy as np

def _project_to_constraints(vec, target_sum, max_iter=15):
    """Clip to [0,1] and adjust to achieve exact sum."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(max_iter):
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-10:
            break
        free = (v > 0.0) & (v < 1.0)
        nfree = free.sum()
        if nfree == 0:
            break
        delta = diff / nfree
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    cur = v.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-8:
        for i in range(len(v)):

                v[i] -= sub
                diff += sub

    return v

def _score(h):
    """Compute C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
