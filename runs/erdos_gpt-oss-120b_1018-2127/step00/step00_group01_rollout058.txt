import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def _project_to_simplex_box(y, target):
    """Project y onto {x | 0<=x_i<=1, sum x_i = target} using water-filling."""
    # clip to [0,1] first to keep within bounds
    y = np.clip(y, 0.0, 1.0)
    # binary search for lambda
    lo, hi = -1.0, 2.0  # wide enough
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def _c5_value(h):
    """Return C5 for a feasible h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _objective(vec, target):
    """Objective for DE: feasibility projection + C5."""
    h = _project_to_simplex_box(vec, target)
    return _c5_value(h)

def _interpolate(old, new_n):
    """Linear interpolation of old vector onto new_n points over [0,2]."""
    old_n = old.size
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xs_new, xs_old, old)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    # ----- Determine grid size -----

    # aim for a finer grid but keep it modest
    n_target = max(100, min(200, n0 * 2))
    h0 = _interpolate(h0, n_target)

    # ensure feasibility
    target_sum = n_target / 2.0
    h0 = _project_to_simplex_box(h0, target_sum)

    # ----- Global search with Differential Evolution -----
    bounds = [(0.0, 1.0)] * n_target
    max_iter = 500  # will be cut by time callback if needed

    def de_callback(xk, convergence):
        elapsed = time.time() - start_time
        # stop DE after ~2/3 of budget
        return elapsed > budget_s * 0.66

    result_de = differential_evolution(
        _objective,
        bounds,
        args=(target_sum,),
        strategy='best1exp',
        maxiter=max_iter,
        popsize=15,

    best_vec = result_de.x
    best_h = _project_to_simplex_box(best_vec, target_sum)
    best_c5 = _c5_value(best_h)

    # ----- Local refinement (SLSQP) -----
    remaining = budget_s - (time.time() - start_time)
