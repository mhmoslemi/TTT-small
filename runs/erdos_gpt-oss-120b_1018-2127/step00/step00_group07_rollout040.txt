import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.size
    dx = 2.0 / n
    # correlation via np.convolve (full) which gives same as correlate with flipped second arg
    conv = np.convolve(h, 1.0 - h, mode='full') * dx
    return float(conv.max())

    if cur_sum != 0:
        h *= target_sum / cur_sum
    h = np.clip(h, 0.0, 1.0)
    # after clipping, re‑normalize to keep exact sum
    excess = h.sum() - target_sum
