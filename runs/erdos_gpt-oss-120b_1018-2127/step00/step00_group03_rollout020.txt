import time, math, random
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a discretized step function."""
    # full cross‑correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # scale by dx = 2 / n_points
    return float(corr.max() * (2.0 / h.size))


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] (sampled) to minimise the
    Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best discovered vector (values in [0,1]).
    c5_bound : float
        The corresponding C5 value.
    n_points : int
        Number of samples used.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h ------------------------------------------------------
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = h.size
    else:
        n = 70                     # default size if no parent is provided
        h = np.full(n, 0.5, dtype=float)  # satisfies the integral constraint

    target_sum = n / 2.0           # because ∫ h = 1 => sum(h)*dx = 1, dx = 2/n
    # enforce box constraints and exact sum (projected)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # simple redistribution of the residual while respecting bounds
    while abs(diff) > 1e-12:
        idx = random.randrange(n)
        if diff > 0:
            avail = 1.0 - h[idx]
            delta = min(diff, avail)
        else:
            avail = h[idx]
            delta = max(diff, -avail)
        h[idx] += delta
        diff = target_sum - h.sum()

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ------------------------------------------------------------------
    # simulated‑annealing search ----------------------------------------
    # ------------------------------------------------------------------
    start = time.time()
    max_iter = 200_000
    # temperature schedule (log‑linear)
    T0, T1 = 5e-3, 5e-6

    for it in range(max_iter):
        if time.time() - start > budget_s * 0.95:
            break

        # temperature decreasing linearly with iteration count
        T = T0 + (T1 - T0) * (it / max_iter)

        # propose a mass‑transfer move between two indices
        i, j = random.randrange(n), random.randrange(n)
        if i == j:
            continue

        # maximum transferable amount respecting bounds
        max_up = 1.0 - h[i]
        max_down = h[j]
        max_transfer = min(max_up, max_down, 0.05)   # limit step size
        if max_transfer <= 0:
            continue

        delta = (random.random() * 2.0 - 1.0) * max_transfer
        # enforce sign: positive delta moves mass from j→i, negative opposite
        if delta > 0:
            delta = min(delta, max_up, max_down)
        else:
            delta = -min(-delta, h[i], 1.0 - h[j])

        if abs(delta) < 1e-12:
            continue

        h[i] += delta
        h[j] -= delta

        cur_c5 = _c5_value(h)

        # accept if improvement or by Metropolis criterion
        improve = cur_c5 < best_c5 - 1e-12
        if improve or random.random() < math.exp(-(cur_c5 - best_c5) / (T + 1e-12)):
            if improve:
                best_c5 = cur_c5
                best_h = h.copy()
            # keep the move
        else:
            # revert
            h[i] -= delta
            h[j] += delta

    # ------------------------------------------------------------------
    # return best found solution ----------------------------------------
    # ------------------------------------------------------------------
    return (best_h, float(best_c5), n)
