import time
import numpy as np
from scipy import optimize

def compute_c5(h):
    """Discrete approximation of C5 for step function h on [0,2]."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_bounds_and_sum(h, target_sum):
    """Clip to [0,1] and adjust to meet the integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # iterative water‑filling / draining
    for _ in range(20):
        free = (h > 0.0) & (h < 1.0)
        if not free.any():
            # forced uniform shift (rare)
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            break
        adjust = diff / free.sum()
        h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
    # final tiny correction to eliminate floating error
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function h (as a NumPy array), its C5 bound, and sample count.
    The function respects the time budget and tries to improve the known bound.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: uniform distribution satisfying the integral
        n_fallback = 45
        h0 = np.full(n_fallback, 0.5)
    n = h0.size
    target_sum = n / 2.0  # because integral over [0,2] must be 1

    # Ensure feasibility of the starting point
    h_best = project_bounds_and_sum(h0.copy(), target_sum)
    best_c5 = compute_c5(h_best)

    start_time = time.time()
    end_time = start_time + budget_s * 0.97  # safety margin

    # Parameters for the stochastic search
    sigma_initial = 0.2   # initial std. dev. of perturbations
    sigma_final = 0.005   # final std. dev.
    decay_rate = (sigma_final / sigma_initial) ** (1 / (budget_s + 1e-12))

    sigma = sigma_initial

    # Main stochastic improvement loop
    while time.time() < end_time:
        # generate a candidate by Gaussian perturbation
        cand = h_best + sigma * rng.standard_normal(n)
        cand = project_bounds_and_sum(cand, target_sum)

        cand_c5 = compute_c5(cand)

        # accept if better (strict improvement)
        if cand_c5 < best_c5 - 1e-8:
            h_best = cand
            best_c5 = cand_c5

        # occasional random restarts to escape plateaus
        if rng.random() < 0.001:
            restart = rng.uniform(0.0, 1.0, n)
            restart = project_bounds_and_sum(restart, target_sum)
            restart_c5 = compute_c5(restart)
            if restart_c5 < best_c5:
                h_best = restart
                best_c5 = restart_c5

        # decay perturbation size
        sigma *= decay_rate
        if sigma < sigma_final:
            sigma = sigma_final

    # Final refinement with a few SLSQP steps (optional but cheap)
    def obj_vec(x):
        return compute_c5(x)

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target_sum,
        'jac': lambda x: np.ones_like(x)
    },)
    bounds = [(0.0, 1.0)] * n

    try:
        res = optimize.minimize(
            obj_vec,
            h_best,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-12, 'maxiter': 200, 'disp': False}
        )
        if res.success:
            h_best = project_bounds_and_sum(res.x, target_sum)
            best_c5 = compute_c5(h_best)
    except Exception:
        pass  # If optimizer fails, keep current best

    return (h_best, float(best_c5), n)
