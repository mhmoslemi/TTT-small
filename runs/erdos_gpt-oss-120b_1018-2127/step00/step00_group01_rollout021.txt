import numpy as np
import time

def _compute_c5(h):
    """Return the discrete C5 value for a given step vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def _project_to_constraints(h, target):
    """Clip to [0,1] and adjust to meet the linear sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction to ensure exactness when possible
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute the remaining error on a random subset
        idx = np.arange(h.size)
        np.random.shuffle(idx)
        for i in idx:
            add = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

    """
    Optimize a step function for the Erdős C5 problem.

    Returns

    h_values : np.ndarray

    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of samples used (grid size).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # 1. Grid size and initial vector
    # ------------------------------------------------------------------

    target_sum = n / 2.0
    h = _project_to_constraints(h0, target_sum)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Simple stochastic local search
    # ------------------------------------------------------------------
    max_iter = 2000000          # upper bound, will be cut by time budget

        i, j = rng.integers(0, n, size=2)
