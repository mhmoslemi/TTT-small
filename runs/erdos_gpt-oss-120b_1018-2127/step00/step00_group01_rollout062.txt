def _project_onto_bounds_sum(v, target_sum, lower=0.0, upper=1.0, max_iter=100):
    """Project vector v onto the box [lower,upper] with a prescribed sum."""
    v = np.asarray(v, dtype=float)
    for _ in range(max_iter):
        v = np.clip(v, lower, upper)
        cur_sum = v.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        free = (v > lower + 1e-12) & (v < upper - 1e-12)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # all at bounds; distribute remaining diff uniformly (will violate bounds minimally)
            v += diff / len(v)
            continue
        adjust = diff / n_free
        v[free] += adjust
    # final clip to guarantee feasibility

    return v

def _c5_of_h(h, dx):
    """Compute the C5 value for a discrete step function h."""
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(conv)

def _initial_triangle(n):
