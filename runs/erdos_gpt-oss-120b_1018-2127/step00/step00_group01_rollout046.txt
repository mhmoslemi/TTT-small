import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

# Helper to compute the C5 value for a discrete h
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

# Project a vector onto the box [0,1] and the hyperplane sum = n/2
def project_feasible(v):
    n = v.size
    target_sum = n / 2.0
    # clip to box first
    x = np.clip(v, 0.0, 1.0)
    # water-filling to meet the sum constraint
    deficit = target_sum - x.sum()
    # indices that can still move (not at a bound)
    free = np.arange(n)
    while abs(deficit) > 1e-10 and free.size > 0:
        # maximal uniform adjustment without breaking bounds
        if deficit > 0:
            # how much each free entry can increase
            caps = 1.0 - x[free]
            max_inc = caps.min()
            inc = min(deficit / free.size, max_inc)
            x[free] += inc
            deficit -= inc * free.size
            # remove those that hit the upper bound
            free = free[x[free] < 1.0 - 1e-12]
        else:

            max_dec = caps.min()
