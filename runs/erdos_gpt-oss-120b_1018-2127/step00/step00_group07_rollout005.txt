import time
import numpy as np

def evaluate_c5(h):
    """Return max overlap C5 for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def upsample(h, new_n):
    """Linear interpolation of h (defined on [0,2]) to a new length."""
    old_n = h.shape[0]
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xs_new, xs_old, h)

def local_search(initial_h, budget, start_time):
    """Simple simulated‑annealing style improvement."""
    n = initial_h.shape[0]
    target_sum = n / 2.0          # because ∑h = n/2  ⇔  integral =1
    h = initial_h.copy()
    # enforce exact sum (might drift due to numeric issues)
    h *= target_sum / h.sum()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

    # temperature schedule parameters
    T0 = 0.1
    Tmin = 1e-5

    it = 0
    while True:
        now = time.time()
        if now - start_time >= budget * 0.95:
            break
        # linear cooling
        frac = (now - start_time) / (budget * 0.95)
        T = max(Tmin, T0 * (1 - frac))

        # propose a mass transfer between two indices
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = h[i], h[j]
        # transferable amount without violating bounds
        max_transfer = min(hi, 1.0 - hj)
        if max_transfer <= 0.0:
            continue
        delta = np.random.rand() * max_transfer
        h[i] = hi - delta
        h[j] = hj + delta

        c5 = evaluate_c5(h)
        # acceptance test (Metropolis)
        if c5 < best_c5 or np.random.rand() < np.exp((best_c5 - c5) / (T + 1e-12)):
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # revert move
            h[i] = hi
            h[j] = hj

        it += 1
        # occasional re‑normalize to kill drift
        if it % 5000 == 0:
            s = h.sum()
            if abs(s - target_sum) > 1e-9:
                h *= target_sum / s

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points) : tuple
        h_values – numpy array of the best step heights found,
        c5_bound – the corresponding maximal overlap,
        n_points – number of discretisation points used.
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # 1. Choose an initial discretisation
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        # force exact integral constraint
        target_sum0 = n0 / 2.0
        h0 *= target_sum0 / h0.sum()
        resolutions = [n0, 2 * n0, 4 * n0]          # 80 → 160 → 320 if 80 given
    else:
        n0 = 100
        h0 = np.full(n0, 0.5, dtype=float)        # uniform start, already satisfies sum=n/2
        resolutions = [n0, 2 * n0, 4 * n0]

    best_overall_h = None
    best_overall_c5 = np.inf
    best_overall_n = None

    # ------------------------------------------------------------------
    # 2. Optimise each resolution in increasing order, re‑using the best result
    # ------------------------------------------------------------------
    for n in resolutions:
        # up‑sample previous best or initial vector to current size
        if best_overall_h is not None:
            h_start = upsample(best_overall_h, n)
        else:
            h_start = upsample(h0, n) if n != n0 else h0.copy()

        # ensure feasibility after interpolation
        h_start = np.clip(h_start, 0.0, 1.0)
        target_sum = n / 2.0
        h_start *= target_sum / h_start.sum()

        # allocate portion of remaining budget (simple equal share)
        elapsed = time.time() - start_time
        remaining = max(0.0, budget_s - elapsed)
        # give at least a few seconds to each level
        level_budget = remaining / (len(resolutions) - resolutions.index(n) + 0.1)

        h_opt, c5_opt = local_search(h_start, level_budget, start_time)

        if c5_opt < best_overall_c5:
            best_overall_c5 = c5_opt
            best_overall_h = h_opt
            best_overall_n = n

        # early exit if time is exhausted
        if time.time() - start_time >= budget_s * 0.99:
            break

    # ------------------------------------------------------------------
    # 3. Return the best found solution
    # ------------------------------------------------------------------
    return best_overall_h, best_overall_c5, int(best_overall_n)
