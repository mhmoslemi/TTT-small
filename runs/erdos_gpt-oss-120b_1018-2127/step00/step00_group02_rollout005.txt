import numpy as np
import time
import math
import random
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair(h):
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    target = len(h) / 2.0
    if abs(total - target) > 1e-12:
        # adjust uniformly to hit the exact sum
        diff = target - total
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # final re‑scale if still off due to clipping
        total = np.sum(h)
        if abs(total - target) > 1e-12:
            h *= target / total
    return h

def interpolate_h(old_h, new_n):
    """Linear interpolation of a step function defined on [0,2]."""
    old_n = len(old_h)
    x_old = np.linspace(0, 2, old_n, endpoint=False) + 1.0/old_n
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    new_h = np.interp(x_new, x_old, old_h)
    return repair(new_h)

def local_search(initial_h, budget, start_time):
    """Simple pairwise transfer hill‑climb with occasional annealing."""
    n = len(initial_h)
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # parameters
    max_step = 0.2
    temp0 = 0.01
    it = 0
    while time.time() - start_time < budget:
        it += 1
        # linear cooling
        temp = temp0 * max(0.0, 1.0 - (time.time() - start_time) / budget)
        i, j = random.sample(range(n), 2)
        delta = (random.random() - 0.5) * max_step
        # propose transfer delta from i to j
        hi, hj = h[i], h[j]
        if hi - delta < 0.0 or hi - delta > 1.0:
            continue
        if hj + delta < 0.0 or hj + delta > 1.0:
            continue
        h[i] = hi - delta
        h[j] = hj + delta
        c5_new = compute_c5(h)
        improve = c5_new < best_c5 - 1e-8
        if improve or (temp > 0 and random.random() < math.exp(-(c5_new - best_c5) / temp)):
            if improve:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
        # occasional full repair to cure drift
        if it % 5000 == 0:
            h = repair(h)
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5
