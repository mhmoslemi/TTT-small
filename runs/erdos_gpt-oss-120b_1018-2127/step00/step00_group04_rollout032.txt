import time
import numpy as np
import random

def _c5_of_h(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for discretized h."""
    # correlation h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def _hill_climb(h, dx, max_passes=200):
    """Greedy swap hill‑climb on a binary vector with fixed sum."""
    n = h.size
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    cur_c5 = _c5_of_h(h, dx)

    for _ in range(max_passes):
        improved = False
        best_swap = None
        best_c5 = cur_c5

        # try all 1↔0 swaps
        for i1 in ones:
            # early break if already perfect (theoretical lower bound ~0.333)
            for i0 in zeros:
                h[i1], h[i0] = 0, 1  # perform swap
                new_c5 = _c5_of_h(h, dx)
                h[i1], h[i0] = 1, 0  # revert

                if new_c5 < best_c5 - 1e-12:
