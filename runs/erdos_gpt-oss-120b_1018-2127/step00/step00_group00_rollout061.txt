import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_score(h):
    """Return C5 value for step function samples h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def make_initial(n):
    """Create an initial feasible vector of length n."""
    # Use the parent construction if available
    if "initial_h_values" in globals():
        old = np.asarray(initial_h_values, dtype=float)
        old_n = old.size
        if old_n == n:
            h = old.copy()
        else:
            # linear interpolation onto new grid
            old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(new_x, old_x, old)
    else:
        # simple deterministic start: first half ones, second half zeros
        h = np.zeros(n, dtype=float)
        ones = int(np.floor(n / 2))
        h[:ones] = 1.0
        rem = (n / 2) - ones
        if rem > 1e-12:
            h[ones] = min(1.0, rem * 2.0)  # never exceeds 1

    # enforce feasibility (clip & renormalize)
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        # fallback uniform distribution
        h[:] = 0.5
        total = h.sum()
    h *= (n / 2) / total
    # final clip in case scaling pushed values slightly out of bounds
    h = np.clip(h, 0.0, 1.0)
    return h


def feasible(h):
    """Check bounds and integral constraint."""
    if np.any(h < -1e-9) or np.any(h > 1 + 1e-9):
        return False
    return abs(h.sum() - h.size / 2) <= 1e-6


def optimize_h(h0, seed):
    """Run SLSQP from start vector h0; return OptimizeResult."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2}
    rng = np.random.default_rng(seed)
    # small random perturb before optimization to escape symmetry
    pert = rng.normal(scale=0.02, size=n)
    x0 = np.clip(h0 + pert, 0.0, 1.0)
    # renormalize to maintain sum constraint exactly (SLSQP will enforce it)
    x0 *= (n / 2) / x0.sum()
    options = {"maxiter": 600, "ftol": 1e-7, "disp": False}
    res = minimize(compute_score, x0, method="SLSQP",
                   bounds=bounds, constraints=cons, options=options)
    return res
