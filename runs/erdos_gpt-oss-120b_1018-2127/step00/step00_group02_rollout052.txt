def _project_bounds_sum(x, target_sum):
    """Clip x to [0,1] and adjust to meet exact sum = target_sum."""
    x = np.clip(x, 0.0, 1.0)
    # Simple iterative redistribution
    for _ in range(10):
        s = x.sum()
        if abs(s - target_sum) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds – distribute uniformly and re‑clip
            delta = (target_sum - s) / len(x)
            x += delta
            x = np.clip(x, 0.0, 1.0)
            continue
        delta = (target_sum - s) / n_free
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny adjustment to hit the sum exactly (may slightly violate bounds)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        # spread diff over non‑saturated entries
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
        else:
            # fallback: distribute over all and re‑clip
            x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
    return x

def _corr_full(h):
    """Full (non‑circular) correlation of h with (1‑h)."""
    return np.correlate(h, 1.0 - h, mode='full')

def _max_corr_and_grad(h, dx):

    corr = _corr_full(h) * dx
    idx = np.argmax(corr)
    max_val = corr[idx]

    n = len(h)
    lag = idx - (n - 1)

    # Gradient wrt h_j: (1 - h_{j-lag}) - h_{j+lag}
