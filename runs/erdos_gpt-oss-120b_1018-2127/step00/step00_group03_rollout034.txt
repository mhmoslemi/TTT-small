def project_to_feasible(v):
    import cvxpy as cp
    n = len(v)
    h = cp.Variable(n)
    target = n/2.0
    prob = cp.Problem(cp.Minimize(cp.sum_squares(h - v)),
                      [h >= 0, h <= 1, cp.sum(h) == target])
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Load or create initial h vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fallback: uniform guess (will be projected)
        n_default = 70
        h0 = np.full(n_default, 0.5, dtype=float)

    n = len(h0)
    target_sum = n / 2.0          # because dx = 2/n and integral must be 1

    # ------------------------------------------------------------------
    # Projection onto the bounded simplex: 0 ≤ x_i ≤ 1, sum x_i = target_sum
    # Uses bisection on the shifting parameter λ in x_i = clip(v_i-λ,0,1)
    # ------------------------------------------------------------------
    def project_to_feasible(v):
        lo, hi = -1e5, 1e5
        for _ in range(60):
            lam = (lo + hi) * 0.5
            x = np.clip(v - lam, 0.0, 1.0)
            s = x.sum()
            if abs(s - target_sum) <= 1e-10:
                return x
            if s > target_sum:
                lo = lam
            else:
                hi = lam
        return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Objective: compute C5 for a candidate h
    # ------------------------------------------------------------------
    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # ------------------------------------------------------------------
    # Initialise best solution
    # ------------------------------------------------------------------
    best_h = project_to_feasible(h0)
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Simple stochastic local search with occasional random restarts
    # ------------------------------------------------------------------
    iteration = 0
    base_step = 0.5          # initial perturbation magnitude
    while time.time() - start_time < budget_s - 0.5:
        # decay step size slowly
        step = base_step * (0.995 ** iteration)

        # propose a new point by Gaussian perturbation
        cand = best_h + np.random.randn(n) * step
        cand = project_to_feasible(cand)
        cand_c5 = compute_c5(cand)

        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5

        # every few hundred iterations try a fresh random start
        if iteration % 250 == 0:
            rand = np.random.rand(n)
            rand = project_to_feasible(rand)
            rand_c5 = compute_c5(rand)
            if rand_c5 < best_c5:
                best_h, best_c5 = rand, rand_c5

        iteration += 1

    return (best_h.tolist(), float(best_c5), n)
