import time
import numpy as np
from scipy import optimize

def _compute_c5(h, dx):
    """Return the max overlap integral for discrete step h."""
    # correlation of h with (1-h), zero-padded (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _hill_climb(h, dx, time_limit, start_time):
    """Simple hill‑climbing on binary step functions.

    Parameters
    ----------
    h : np.ndarray
        Current binary array (0/1) satisfying sum(h)=n/2.
    dx : float
        Sample spacing.
    time_limit : float
        Remaining seconds allowed for this phase.
    start_time : float
        Global start time (for time checks).

    Returns
    -------
    best_h, best_c5
    """
    n = h.size
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    # indices of zeros and ones for quick swapping
    zeros = np.flatnonzero(best_h == 0)
    ones = np.flatnonzero(best_h == 1)

    # pre‑compute a mask of possible lags (full mode size)
    while time.time() - start_time < time_limit:
        # pick a random zero and a random one to swap
        if zeros.size == 0 or ones.size == 0:
            break
        i = np.random.choice(zeros)
        j = np.random.choice(ones)

        # perform swap
        h_new = best_h.copy()
        h_new[i], h_new[j] = 1, 0

        c5_new = _compute_c5(h_new, dx)

        # accept if better
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            # update zero/one lists
            zeros = np.where(best_h == 0)[0]
            ones = np.where(best_h == 1)[0]

    return best_h, best_c5

def _optimize_at_resolution(initial_h, n_target, budget_s, start_time):
    """Optimize a binary step function at a given resolution."""
    # interpolate initial_h to the new number of points
    x_old = np.linspace(0.0, 2.0, initial_h.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_interp = np.interp(x_new, x_old, initial_h)

    # convert to binary while preserving total integral
    half = n_target // 2
    # take the largest half values as 1, rest 0
    idx_desc = np.argsort(-h_interp)
    h_bin = np.zeros(n_target, dtype=float)
    h_bin[idx_desc[:half]] = 1.0

    dx = 2.0 / n_target
    remaining = budget_s - (time.time() - start_time)
    if remaining <= 0:
        return h_bin, _compute_c5(h_bin, dx)

    # devote half of the remaining budget to this resolution
    phase_time = remaining * 0.5
    best_h, best_c5 = _hill_climb(h_bin, dx, phase_time, start_time)
    return best_h, best_c5
