import time, math
import numpy as np

def _c5_value(h):
    """Return the C5 value for a discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def _project_to_feasible(h, target_sum):
    """
    Adjust vector h in-place so that its entries lie in [0,1] and sum equals target_sum.
    Simple iterative redistribution, sufficient for small corrections.
    """
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    n = h.size
    rnd = np.random.RandomState(0)
    # Distribute the remaining difference greedily
    while abs(diff) > 1e-12:
        idx = rnd.randint(0, n)
        if diff > 0:
            room = 1.0 - h[idx]
            if room <= 0:
                continue
            delta = min(diff, room)
        else:
            room = h[idx]
            if room <= 0:
                continue
            delta = max(diff, -room)
        h[idx] += delta
        diff = target_sum - h.sum()
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimise the
    Erdős overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points) :
        h_values – list of floats, the optimiser's best h,
        c5_bound – float, the achieved C5 value,
        n_points – int, the discretisation size.
    """
    rnd = np.random.RandomState(seed)
    # ----------------------------------------------------------------------
    # Determine problem size and initialise a feasible h
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
    else:
        n = 45                           # fallback size if no parent supplied
        init = None

    target_sum = n / 2.0                # ∫ h = 1  ⇒  sum(h)*dx = 1  ⇒  sum(h)=n/2

    if init is not None:
        # Use a rounded version of the supplied vector while keeping the total sum.
        # Take the highest values as 1, the lowest as 0, and adjust one entry if needed.
        h = np.zeros(n, dtype=float)
        order = np.argsort(-init)       # descending
        ones = int(math.floor(target_sum))
        h[order[:ones]] = 1.0
        remainder = target_sum - ones
        if remainder > 0:
            h[order[ones]] = remainder
        h = _project_to_feasible(h, target_sum)
    else:
        # Random binary (or half‑binary) start
        h = np.zeros(n, dtype=float)
        ones = int(math.floor(target_sum))
        h[:ones] = 1.0
        remainder = target_sum - ones
        if remainder > 0:
            h[ones] = remainder
        rnd.shuffle(h)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ----------------------------------------------------------------------
    # Simple simulated‑annealing / hill‑climbing loop
    # ----------------------------------------------------------------------
    start_time = time.time()
    end_time = start_time + budget_s * 0.95   # leave a margin
    iteration = 0

    while time.time() < end_time:
        # Propose a move that keeps the sum exactly unchanged:
        # transfer a small amount epsilon from index i to index j.
        i, j = rnd.choice(n, 2, replace=False)
        epsilon = 0.05 * rnd.rand()  # random step size up to 0.05
        hi, hj = h[i], h[j]

        # Determine feasible direction
        if hi - epsilon >= 0.0 and hj + epsilon <= 1.0:
            h_new = h.copy()
            h_new[i] -= epsilon
            h_new[j] += epsilon
        elif hi + epsilon <= 1.0 and hj - epsilon >= 0.0:
            h_new = h.copy()
            h_new[i] += epsilon
            h_new[j] -= epsilon
        else:
            continue  # infeasible move, try next iteration

        c5_new = _c5_value(h_new)

        if c5_new < best_c5 - 1e-9:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Simulated‑annealing acceptance
            elapsed = time.time() - start_time
            remaining = max(end_time - time.time(), 1e-6)
            # Linear temperature schedule
            temp = remaining / (budget_s * 0.95)
            prob = math.exp(-(c5_new - best_c5) / (temp * 0.01 + 1e-12))
            if rnd.rand() < prob:
                h = h_new

        iteration += 1

    # Final sanity projection (should already be feasible)
    best_h = _project_to_feasible(best_h, target_sum)

    return (best_h.tolist(), float(best_c5), n)
