import time
import numpy as np
from scipy.optimize import minimize

# Helper: compute C5 for a discrete step function h on [0,2]
def compute_c5(h, dx):
    # convolution of h with (1-h)
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return conv.max()

# Project a vector onto the box [0,1] and enforce the integral constraint exactly.
def make_feasible(h, target_sum, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) < 1e-12:
            break
        # indices that are not at a bound can receive the adjustment
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # all at bounds – distribute uniformly and re‑clip
            delta = (target_sum - cur_sum) / h.size
            h += delta
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = (target_sum - cur_sum) / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if still off (rare)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # distribute remaining error uniformly and clip again
        h += (target_sum - cur_sum) / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Choose a modest number of samples (fast evaluation, yet expressive)
    n_points = 200                     # can be changed if time permits
    dx = 2.0 / n_points
    target_sum = n_points / 2.0        # integral = 1  => sum(h) = n/2

    # Build an initial feasible vector.
    if 'initial_h_values' in globals():
        # interpolate the provided parent solution to the new grid
        parent = np.asarray(initial_h_values, dtype=float)
        m = parent.size
        x_parent = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_parent, parent)
        h0 = make_feasible(h0, target_sum)
    else:
        # simple deterministic start: half ones, half zeros shuffled
        h0 = np.zeros(n_points, dtype=float)
        h0[:n_points // 2] = 1.0
        np.random.shuffle(h0)
        h0 = make_feasible(h0, target_sum)

    # Record best solution found so far
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Constraint for the optimizer (equality on the sum)
    cons = ({
        'type': 'eq',
        'fun': lambda h: np.sum(h) - target_sum
    },)

    bounds = [(0.0, 1.0)] * n_points

    # Local optimizer wrapper
    def local_opt(start_vec):
        res = minimize(
            fun=lambda h: compute_c5(h, dx),
            x0=start_vec,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 400, 'ftol': 1e-9, 'disp': False}
        )
        if not res.success:
            return None, np.inf
        h_opt = make_feasible(res.x, target_sum)  # safety projection
        return h_opt, compute_c5(h_opt, dx)

    # First attempt from the interpolated parent
    h_opt, c5_val = local_opt(best_h)
    if c5_val < best_c5:
        best_h, best_c5 = h_opt, c5_val

    # Stochastic restarts while time permits
    while time.time() - start_time < budget_s - 5.0:  # keep a small safety margin
        # Random feasible start: start from best_h plus small noise
        noise = np.random.randn(n_points) * 0.1
        trial = best_h + noise
        trial = make_feasible(trial, target_sum)

        h_opt, c5_val = local_opt(trial)
        if c5_val < best_c5:
            best_h, best_c5 = h_opt, c5_val

        # Occasionally try a completely random binary pattern
        if np.random.rand() < 0.1:
            rand_bin = np.zeros(n_points)
            rand_bin[:n_points // 2] = 1.0
            np.random.shuffle(rand_bin)
            rand_bin = make_feasible(rand_bin, target_sum)
            h_opt, c5_val = local_opt(rand_bin)
            if c5_val < best_c5:
                best_h, best_c5 = h_opt, c5_val

    # Ensure final vector respects constraints before returning
    best_h = make_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)

    return best_h, float(best_c5), n_points
