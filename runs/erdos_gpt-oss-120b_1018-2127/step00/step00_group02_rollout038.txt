import time
import numpy as np
from math import ceil

# helpers that will be injected by the harness
# evaluate_erdos_solution(h) -> float  (optional)

def _compute_c5(h):
    """Exact C5 for sampled step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _evaluate(h):
    """Wrapper that prefers the harness evaluator if present."""
    try:
        return float(evaluate_erdos_solution(h))
    except Exception:  # pragma: no cover
        return _compute_c5(h)

def _repair_sum(h, target_sum):
    """Adjust h so that sum equals target_sum while staying in [0,1]."""
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute diff proportionally to distance from bounds
    free = np.where((h > 0) & (h < 1))[0]
    if free.size == 0:
        # fallback: blunt scaling then clipping
        h = np.clip(h + diff / len(h), 0.0, 1.0)
        # final correction
        excess = target_sum - h.sum()
        if abs(excess) > 1e-12:
            h += excess / len(h)
        return h
    # simple linear redistribution
    delta = diff / free.size
    h[free] += delta
    h = np.clip(h, 0.0, 1.0)
    # recurse a few times to fix residual
    for _ in range(3):
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        diff = target_sum - cur
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def _local_search(h, budget, start, rng):
    """Simple pairwise perturbation hill‑climber."""
    n = h.shape[0]

    # time split

        hi = new_h[i] + delta
        hj = new_h[j] - delta

            hi = 0.0

def _interpolate(h, new_n):
    """Linear interpolation of periodic step function."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    # enforce integral exactly
    target_sum = new_n / 2.0
    h_new = _repair_sum(h_new, target_sum)
    return h_new
