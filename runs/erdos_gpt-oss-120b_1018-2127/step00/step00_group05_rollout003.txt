import time
import numpy as np
import random

# Helper: evaluate overlap C5 using the provided evaluator if available,
# otherwise compute directly.
def compute_c5(h):
    try:
        return evaluate_erdos_solution(h)
    except Exception:
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return dx * np.max(corr)


def repair(h, target_sum):
    """Clip to [0,1] and scale to meet the integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        return np.full_like(h, target_sum / len(h))
    scale = target_sum / s
    h = h * scale
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        return np.full_like(h, target_sum / len(h))
    scale = target_sum / s
    h = h * scale
    h = np.clip(h, 0.0, 1.0)

    # Tiny redistribution to hit the sum exactly
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def mutate_resample(parent, target_sum):
    """Replace a fraction of entries by fresh uniform samples."""
    n = len(parent)
    child = parent.copy()
    k = max(1, n // 5)  # 20% of points
    idx = np.random.choice(n, k, replace=False)
    child[idx] = np.random.rand(k)
    return repair(child, target_sum)


def mutate_perturb(parent, target_sum):
    """Add small zero‑mean noise to a subset of entries."""
    n = len(parent)
    child = parent.copy()
    mask = np.random.rand(n) < 0.3  # 30% of points
    noise = np.random.randn(mask.sum()) * 0.02
    child[mask] += noise
    return repair(child, target_sum)


def initial_vector(n):
    """Create a feasible start when no parent is supplied."""
    rnd = np.random.rand(n)
    return repair(rnd, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # Determine dimension
    try:
        h0 = np.array(initial_h_values, copy=False)
        n = len(h0)
    except NameError:
        n = 70  # fallback size
        h0 = initial_vector(n)

    target_sum = n / 2.0  # because dx = 2/n and ∫h = 1
    best_h = repair(h0, target_sum)
    best_c5 = compute_c5(best_h)

    # Simple evolutionary loop
    population_size = 10
    pop = [best_h.copy() for _ in range(population_size)]

    while time.time() - start_time < budget_s * 0.95:
        new_pop = []
        for parent in pop:
            # pick mutation strategy probabilistically
            if random.random() < 0.6:
                child = mutate_resample(parent, target_sum)
            else:
                child = mutate_perturb(parent, target_sum)

            c5 = compute_c5(child)
            if c5 < best_c5:
                best_c5 = c5
                best_h = child.copy()
            new_pop.append(child)

        # keep the best individuals plus a few random ones for diversity
        fitness = [(compute_c5(ind), ind) for ind in new_pop]
        fitness.sort(key=lambda x: x[0])
        pop = [ind for _, ind in fitness[:population_size // 2]]
        # fill remaining slots with random perturbations of the current best
        while len(pop) < population_size:
            if random.random() < 0.5:
                cand = mutate_resample(best_h, target_sum)
            else:
                cand = mutate_perturb(best_h, target_sum)
            pop.append(cand)

    return best_h, float(best_c5), n
