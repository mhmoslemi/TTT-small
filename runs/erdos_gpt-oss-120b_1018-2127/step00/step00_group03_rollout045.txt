import numpy as np
import time

# helper: compute C5 by full correlation
def _c5_score(h):
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_to_simplex(h):
    """Return a feasible version of h with sum(h)=n/2 and 0<=h<=1.
    Simple heuristic: keep the largest n/2 entries as 1, rest 0."""
    n = h.size
    target = n // 2
    idx = np.argpartition(h, -target)[-target:]
    proj = np.zeros_like(h)
    proj[idx] = 1.0
    return proj
