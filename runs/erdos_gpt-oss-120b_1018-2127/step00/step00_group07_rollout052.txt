import time, math, random, numpy as np

def compute_c5(h):
    """Return C5 for the step function values h (numpy array)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def propose_swap(h, rng):
    """Return a new h after swapping two random indices (preserves sum)."""
    i, j = rng.integers(0, h.size, size=2)
    if i == j:
        return h.copy()
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def anneal_search(h, time_budget, rng):
    """Simple simulated‑annealing on binary‑swap moves."""
    start = time.time()
    best_h = h.copy()
    best_score = compute_c5(best_h)
    cur_h = h.copy()
    cur_score = best_score

    # annealing schedule parameters
    T0 = 0.05  # initial temperature
    Tmin = 1e-6
    decay = 0.9995

    T = T0
    it = 0
    # pre‑measure a cheap iteration time to limit loops
    while True:
        now = time.time()
        if now - start >= time_budget:
            break
        # propose
        cand = propose_swap(cur_h, rng)
        cand_score = compute_c5(cand)

        # acceptance
        if cand_score < cur_score:
            cur_h, cur_score = cand, cand_score
            if cand_score < best_score:
                best_h, best_score = cand.copy(), cand_score
        else:
            delta = cand_score - cur_score
            if T > 0 and random.random() < math.exp(-delta / T):
                cur_h, cur_score = cand, cand_score

        # temperature update
        T = max(Tmin, T * decay)
        it += 1

        # occasional fast jump if stuck

            idx = rng.choice(cur_h.size, size=4, replace=False)
            cur_h[idx] = 1 - cur_h[idx]  # flip bits
            cur_score = compute_c5(cur_h)

    return best_h, best_score
