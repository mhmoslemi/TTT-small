import time
import numpy as np

def evaluate_c5(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(conv.max() * dx)

def project_to_constraints(v, target):
    """Clip to [0,1] and adjust to meet sum == target (bounded simplex projection)."""
    v = np.clip(v, 0.0, 1.0)
    # Simple water‑filling style adjustment
    for _ in range(10):  # enough iterations for convergence on modest sizes
        excess = v.sum() - target
        if abs(excess) < 1e-12:
            break
        if excess > 0:
            mask = v > 1e-12          # can decrease
        else:
            mask = v < 1.0 - 1e-12    # can increase
        if not np.any(mask):
            break
        adjust = -excess / mask.sum()
        v[mask] += adjust
        v = np.clip(v, 0.0, 1.0)
    return v

def anneal(h, target, dx, time_allow):
    """Simple simulated‑annealing loop respecting the time allowance."""
    best = h.copy()
    best_score = evaluate_c5(best, dx)
    cur = h.copy()
    cur_score = best_score
    start = time.time()
    it = 0
    # temperature schedule: start proportional to current score
    T0 = max(0.01, best_score * 0.5)
    while time.time() - start < time_allow:
        it += 1
        T = T0 * (0.995 ** it)
        idx = np.random.randint(len(cur))
        delta = (np.random.rand() - 0.5) * 0.2   # up to +-0.1
        cand = cur.copy()
        cand[idx] += delta
        cand = project_to_constraints(cand, target)
        cand_score = evaluate_c5(cand, dx)

        # keep best
        if cand_score < best_score:
            best, best_score = cand, cand_score

        # metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur, cur_score = cand, cand_score
    return best, best_score

def upscale(h, new_n):
    """Resize h to length new_n by simple repetition and projection."""
    old_n = len(h)
    factor = new_n // old_n
    remainder = new_n % old_n
    h_up = np.repeat(h, factor)
    if remainder:
        h_up = np.concatenate([h_up, h[:remainder]])
    target = new_n / 2.0
    h_up = project_to_constraints(h_up, target)
    return h_up
