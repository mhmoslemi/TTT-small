import numpy as np
import time
import random
import math

def _project_h(v, target_sum):
    """Project vector v onto [0,1] bounds with given sum using simple iterative redistribution."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target_sum) < 1e-12:
        return v
    for _ in range(15):
        diff = target_sum - s
        free = (v > 0.0) & (v < 1.0)
        nfree = int(free.sum())
        if nfree == 0:
            break
        v[free] += diff / nfree
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        if abs(s - target_sum) < 1e-9:
            break
    # final proportional fix if needed
    if abs(s - target_sum) > 1e-8:
        v *= target_sum / s
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_score(h):
    """Compute C5 for a given step function h (array)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _neighbor(h, sigma, rng):
    """Generate a neighbor by perturbing a few coordinates."""
    n = h.shape[0]
    new_h = h.copy()
    # perturb a random subset (≈5% of entries)
    k = max(1, n // 20)
    idx = rng.sample(range(n), k)
    noise = rng.uniform(-sigma, sigma, size=k)
    new_h[idx] += noise
    # project back to feasible set
    target = n / 2.0
    new_h = _project_h(new_h, target)
    return new_h
