def repair(h, target):
    h = np.clip(h,0,1)
    s = h.sum()
    if s == 0:
        h = np.full_like(h, target/len(h))
    else:
        h = h * (target / s)
    # after scaling may exceed bounds, clip again and renorm once more
    h = np.clip(h,0,1)
    s = h.sum()
    if s != target:
        h = h * (target / s)
    return h

import time, random, math
import numpy as np

def _repair(vec, target_sum):
    """Clip to [0,1] and rescale to achieve the required sum."""
    v = np.clip(vec, 0.0, 1.0)
    s = v.sum()
    if s == 0.0:
        v.fill(target_sum / len(v))
    else:
        v = v * (target_sum / s)
    # second pass in case scaling pushed values outside [0,1]
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if s != target_sum:
        v = v * (target_sum / s)
    return v

def _evaluate(h, dx):
    """Return C₅ for a given h (full-mode correlation, scaled by dx)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * float(corr.max())

def _mutate(parent, target_sum):
    """Produce a child by random reset or jitter, then repair."""
    n = len(parent)
    child = parent.copy()
    if random.random() < 0.5:                     # reset a fraction of entries
        k = max(1, n // 5)
        idx = np.random.choice(n, size=k, replace=False)
        child[idx] = np.random.rand(k)
    else:                                         # jitter a subset
        mask = np.random.rand(n) < 0.30
        delta = (np.random.rand(mask.sum()) * 2.0 - 1.0) * 0.02
        child[mask] += delta
    return _repair(child, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C₅ bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : 1‑D NumPy array with entries in [0,1] and sum = n/2
        c5_bound : float, the achieved max overlap
        n_points : integer, number of discretization samples
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # --------------------------------------------------------------
    # Determine discretisation size and initialise a feasible h0
    # --------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 200                     # a modest size when no parent is supplied
        h0 = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0            # because ∫ h = 1 ⇒ Σh * (2/n) = 1 ⇒ Σh = n/2
    dx = 2.0 / n

    h0 = _repair(h0, target_sum)

    best_h = h0
    best_c5 = _evaluate(best_h, dx)

    # --------------------------------------------------------------
    # Simple evolutionary loop
    # --------------------------------------------------------------
    pop_size = 12
    population = [best_h]
    while len(population) < pop_size:
        rnd = np.random.rand(n)
        population.append(_repair(rnd, target_sum))

    # Main optimisation loop, respecting the time budget
    while time.time() - start_time < budget_s * 0.98:
        new_candidates = []
        for indiv in population:
            child = _mutate(indiv, target_sum)
            c5 = _evaluate(child, dx)

            if c5 < best_c5:
                best_c5 = c5
                best_h = child

            new_candidates.append((c5, child))

        # keep the best individuals for the next generation
        new_candidates.sort(key=lambda tup: tup[0])
        population = [c[1] for c in new_candidates[:pop_size]]

    # Final sanity check (repair once more just in case)
    best_h = _repair(best_h, target_sum)
    best_c5 = _evaluate(best_h, dx)

    return (best_h, float(best_c5), n)
