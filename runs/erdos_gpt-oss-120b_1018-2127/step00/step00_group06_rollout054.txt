import time
import numpy as np
import cvxpy as cp

def _compute_c5(h):
    """Return the max overlap integral for a discrete h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h, target_sum):
    """Project h onto the feasible set {0≤h≤1, sum(h)=target_sum}."""
    n = h.size
    hh = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(hh - h))
    cons = [hh >= 0, hh <= 1, cp.sum(hh) == target_sum]
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    except Exception:  # pragma: no cover
        prob.status = cp.INFEASIBLE
    if hh.value is not None:
        return np.asarray(hh.value).astype(float).ravel()
    # fallback: simple clipping + scaling
    h_clip = np.clip(h, 0.0, 1.0)
    s = h_clip.sum()
    if s == 0:
        return np.full(n, target_sum / n)
    return h_clip * (target_sum / s)

def _interpolate(old_h, new_n):
    """Linear interpolation of a periodic step function on [0,2)."""
    old_n = old_h.size
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)

def _search(initial_h, budget, rng):
    """Simulated‑annealing style refinement for a fixed grid size."""
    n = initial_h.size
    target = n / 2.0
    h = _repair(initial_h, target)
    best_h = h.copy()
    best_score = _compute_c5(best_h)

    T = 1.0
    start = time.time()
    while time.time() - start < budget:
        # magnitude adapts to grid size
        sigma = 0.03 / np.sqrt(n)
        cand = best_h + rng.normal(scale=sigma, size=n)
        cand = _repair(cand, target)
        sc = _compute_c5(cand)
        if sc < best_score or rng.random() < np.exp((best_score - sc) / max(T, 1e-12)):
            best_h, best_score = cand, sc
        T *= 0.9995
    return best_h, best_score
