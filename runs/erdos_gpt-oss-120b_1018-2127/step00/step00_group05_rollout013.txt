import numpy as np
import time

def _repair_vector(v, target_sum, max_iter=20, eps=1e-9):
    """Clip to [0,1] and adjust to exact sum target."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) <= eps:
            break
        # indices that can move in the needed direction
        if diff > 0:
            movable = v < 1.0 - eps
        else:
            movable = v > eps
        idx = np.where(movable)[0]
        if idx.size == 0:
            break
        # distribute the needed correction uniformly
        delta = diff / idx.size
        v[idx] += delta
        v = np.clip(v, 0.0, 1.0)
    # final scaling if still off (unlikely)
    s = v.sum()
    if abs(s - target_sum) > 1e-6:
        scale = target_sum / s
        v = np.clip(v * scale, 0.0, 1.0)
        # a last clip‑and‑scale round
        v = np.clip(v, 0.0, 1.0)
        v *= target_sum / v.sum()
    return v

def _evaluate(v, dx):
    """Compute C5 = max_{k} ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def _upscale(initial, n_target):
    """Resample initial vector (length m) to length n_target by linear interpolation."""
    m = len(initial)
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_new, x_old, initial)
