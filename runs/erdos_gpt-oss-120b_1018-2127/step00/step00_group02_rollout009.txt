import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h):
    """Return the overlap constant C5 for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_initial(init_vals, n):
    """Interpolate initial array (of length m) to length n on [0,2)."""
    m = init_vals.size
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(new_x, old_x, init_vals)

def slsqp_optimize(h0, n, maxiter, time_deadline):
    """Run a limited SLSQP optimization, aborting if time runs out."""
    # Equality: sum = n/2
    constraints = ({'type': 'eq',
                    'fun': lambda x: np.sum(x) - n / 2.0},)

    bounds = [(0.0, 1.0) for _ in range(n)]

    def obj(x):
        return compute_c5(x)

    # Callback to stop early if out of time
    def callback(xk):
        if time.time() > time_deadline:
            raise RuntimeError("time limit")

    try:
        res = minimize(obj, h0, method="SLSQP",
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False},
                       callback=callback)
    except RuntimeError:
        # timed out inside optimizer
        return None
    if not res.success:
        return None
    return res.x

def random_local_search(h, best_c5, budget_end, rng):
    """Simple hill‑climbing with random pair moves preserving the sum."""
    n = h.size
    step_scale = 0.02  # typical delta magnitude
    while time.time() < budget_end:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # maximum admissible delta to stay in [0,1]
        max_inc = min(1.0 - h[i], h[j])
        max_dec = min(h[i], 1.0 - h[j])
        # pick a signed delta within allowed range
        delta = (rng.random() - 0.5) * 2.0 * step_scale
        delta = np.clip(delta, -max_dec, max_inc)

        h_new[j] -= delta
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
    return h, best_c5
