import time
import numpy as np
import math
import random

# The harness provides `initial_h_values` when a parent solution exists.
# We will use it directly if present, otherwise start from a uniform distribution.
def _initial_h():
    try:
        hv = initial_h_values  # type: ignore
        h = np.array(hv, dtype=float)
    except NameError:
        n = 80
        h = np.full(n, 0.5)  # satisfies integral because sum = n/2
    return h

def _repair(h, target_sum, max_iter=5):
    """Project h onto [0,1] with exact sum = target_sum (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        # Scale uniformly then clip again
        if cur == 0:
            h += target_sum / h.size
        else:
            h *= (target_sum / cur)
        h = np.clip(h, 0.0, 1.0)
    # Final tiny correction by distributing the residual uniformly
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        h += residual / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_score(h):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] with integral 1 that minimizes C5."""
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    h_cur = _initial_h()
    n = h_cur.size
    target_sum = n / 2.0
    h_cur = _repair(h_cur, target_sum)
    cur_score = _c5_score(h_cur)

    best_h = h_cur.copy()
    best_score = cur_score

    # Simulated‑annealing parameters
    temp0 = 0.05
    temp = temp0
    cooling = 0.9995
    min_temp = 1e-7
    # step magnitude relative to typical value (0‑1 range)
    step_scale = 0.08

    # Main loop respecting the time budget
    while True:
        if time.time() - start_time > budget_s * 0.98:
            break

        # Propose a new candidate by Gaussian perturbation
        perturb = (np.random.rand(n) - 0.5) * 2.0 * step_scale
        h_prop = h_cur + perturb
        h_prop = _repair(h_prop, target_sum)

        prop_score = _c5_score(h_prop)

        # Accept if better, otherwise with SA probability
        if prop_score < cur_score:
            h_cur, cur_score = h_prop, prop_score
        else:
            delta = cur_score - prop_score
            prob = math.exp(delta / max(temp, min_temp))
            if random.random() < prob:
                h_cur, cur_score = h_prop, prop_score

        # Keep global best
        if cur_score < best_score:
            best_h, best_score = h_cur.copy(), cur_score

        # Temperature schedule
        temp = max(temp * cooling, min_temp)

        # Occasionally restart from the best found so far to diversify
        if random.random() < 0.005:
            h_cur = best_h.copy()
            cur_score = best_score
            temp = temp0

    # Return as Python list for compatibility with the evaluator
    return (best_h.tolist(), float(best_score), n)
