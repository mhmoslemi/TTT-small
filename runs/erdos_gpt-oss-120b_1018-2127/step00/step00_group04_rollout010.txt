import numpy as np
import time
import random
import math

def compute_c5(h):
    """Maximum overlap integral for a given step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def propose_move(h, rng, step_scale=0.3):
    """Transfer a small amount of mass from i to j, keeping sum fixed."""
    n = h.size
    i, j = rng.sample(range(n), 2)
    # amount that can be moved from i and received by j
    max_from = h[i]
    max_to = 1.0 - h[j]
    if max_from <= 0 or max_to <= 0:
        return None  # infeasible move
    delta_max = min(max_from, max_to)
    # random fraction of the allowable transfer
    delta = rng.random() * delta_max * step_scale
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
