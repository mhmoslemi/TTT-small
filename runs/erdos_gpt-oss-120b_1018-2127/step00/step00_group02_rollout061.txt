import time, math, random
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Return the discrete C5 value for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x):
    """Project a vector onto the box [0,1] and the affine sum constraint sum= n/2."""
    n = x.shape[0]
    target = n / 2.0
    # First clip to box
    x = np.clip(x, 0.0, 1.0)

    # Use a few simple redistribution steps to satisfy the sum.
    for _ in range(20):
        current = x.sum()
        diff = target - current
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        if not np.any(free):
            # all at bounds – distribute uniformly and clip again
            x += diff / n
            x = np.clip(x, 0.0, 1.0)
            continue
        adjustment = diff / free.sum()
        x[free] += adjustment
        x = np.clip(x, 0.0, 1.0)
    return x

def neighbour(h, rng):
    """Generate a feasible neighbour of h by moving a small amount between two coordinates."""
    n = h.shape[0]
    i, j = rng.sample(range(n), 2)
    # propose a delta respecting bounds
    max_delta = min(0.2, h[j])               # how much we can take from j
    max_delta = min(max_delta, 1.0 - h[i])   # how much we can give to i
    if max_delta <= 0:
        return h.copy()
    delta = rng.uniform(-max_delta, max_delta)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # numerical safety
    new_h = np.clip(new_h, 0.0, 1.0)
    # sum is preserved up to round‑off; fix if needed
    new_h = project_feasible(new_h)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = random.Random(seed)
    np_rng = np.random.default_rng(seed)

    # Determine dimension
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        h0 = project_feasible(h0)
    else:
        n = 60                     # a modest even size if no starter is given
        # random feasible start (uniform then project)
        h0 = np_rng.random(n)
        h0 = project_feasible(h0)

    best_h = h0
    best_c = compute_c5(best_h)

    # Simulated‑annealing parameters
    init_temp = 0.02
    cooling_rate = 0.9995

    temp = init_temp
    it = 0
    while time.time() - start_time < budget_s * 0.95:
        it += 1
        cand_h = neighbour(best_h, rng)
        cand_c = compute_c5(cand_h)

        if cand_c < best_c - 1e-8:
            best_h, best_c = cand_h, cand_c
        else:
            # occasional uphill moves
            prob = math.exp(-(cand_c - best_c) / max(temp, 1e-12))
            if rng.random() < prob:
                best_h, best_c = cand_h, cand_c

        # temperature schedule
        temp *= cooling_rate

        # periodic random restart if stagnation detected
        if it % 5000 == 0:
            # keep the best found so far, but try a fresh start
            new_start = np_rng.random(n)
            new_start = project_feasible(new_start)
            new_c = compute_c5(new_start)
            if new_c < best_c:
                best_h, best_c = new_start, new_c

    # Final safety projection (ensures feasibility)
    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)

    return best_h.tolist(), best_c, n
