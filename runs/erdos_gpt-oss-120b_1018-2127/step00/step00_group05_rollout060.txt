import time
import numpy as np

def _enforce_constraints(v, target_sum):
    """Clip to [0,1] and adjust to obtain the exact required sum."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        if abs(s - target_sum) < 1e-12:
            break
        if s == 0.0:
            v = np.full_like(v, target_sum / v.size)
            break
        factor = target_sum / s
        v = np.clip(v * factor, 0.0, 1.0)
    # minor residual correction
    s = v.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            v += diff / v.size
        else:
            add = diff / free.size
            v[free] += add
            v = np.clip(v, 0.0, 1.0)
            # final renormalisation if needed
            v *= target_sum / v.sum()
    return v


def _evaluate(vec):
    """Return the C5 value for a candidate step function."""
    n = vec.size
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return corr.max() * dx


def _resample(vec, n):
    """Linear interpolation of a vector to a new length n."""
    vec = np.asarray(vec, dtype=float)
    if vec.size == n:
        return vec.copy()
    x_old = np.linspace(0.0, 2.0, vec.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, vec)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.

    Returns
    -------
    h_values : ndarray
        Optimised values of h on a uniform grid.
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size -------------------------------------------------
    if 'n_points' in kwargs:
        n_points = int(kwargs['n_points'])
    elif 'initial_h_values' in globals():
        n_points = max(60, min(200, len(initial_h_values) * 2))
    else:
        n_points = 100

    target_sum = n_points / 2.0  # because integral must be 1 over length 2

    # Build initial candidate -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = _resample(initial_h_values, n_points)
    else:
        h0 = rng.random(n_points)
    h0 = _enforce_constraints(h0, target_sum)

    best_vec = h0
    best_score = _evaluate(best_vec)

    start_time = time.time()
    max_iters = 200000
    step_scale = 0.3  # initial perturbation amplitude

    it = 0
    while it < max_iters and (time.time() - start_time) < (budget_s - 0.5):
        # generate a neighbour by adding uniform noise
        noise = (rng.random(n_points) - 0.5) * step_scale
        cand = best_vec + noise
        cand = _enforce_constraints(cand, target_sum)

        cand_score = _evaluate(cand)

        if cand_score < best_score:
            best_vec, best_score = cand, cand_score
            # gradually shrink step size for finer search
            step_scale = max(0.02, step_scale * 0.995)
        else:
            # occasional random restart to escape plateaus
            if rng.random() < 0.005:
                rand_vec = rng.random(n_points)
                rand_vec = _enforce_constraints(rand_vec, target_sum)
                rand_score = _evaluate(rand_vec)
                if rand_score < best_score:
                    best_vec, best_score = rand_vec, rand_score

        it += 1

    # Final evaluation (ensuring constraints once more)
    best_vec = _enforce_constraints(best_vec, target_sum)
    best_score = _evaluate(best_vec)

    return best_vec, float(best_score), n_points
