import numpy as np
import time
import random
from math import exp

# Helper to compute the C5 value for a given step function h
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

# Simple projection to keep the sum constraint exact after numerical noise
def normalize_sum(h):
    n = h.size
    target = n / 2.0
    cur = h.sum()
    if cur == 0:
        return np.full(n, target / n)
    h = h * (target / cur)
    # Clip and re‑adjust any slight violations
    h = np.clip(h, 0.0, 1.0)
    excess = h.sum() - target
    if abs(excess) > 1e-12:
        # distribute excess uniformly while respecting bounds
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            delta = excess / mask.sum()
            h[mask] -= delta
            h = np.clip(h, 0.0, 1.0)
    return h

def initial_vector(n):
    # if a parent construction is provided use it (it should already have length n)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size == n:
            return normalize_sum(h0)
    # otherwise start from the uniform feasible point 0.5
    h0 = np.full(n, 0.5, dtype=float)
    return normalize_sum(h0)

def simulated_annealing(h0, budget, start_time):
    n = h0.size
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget:
            break

        # temperature decays linearly to near zero
        frac = elapsed / budget
        temp = max(0.001, (1.0 - frac) * 0.02)   # small temperature scale
        step_scale = max(0.001, (1.0 - frac) * 0.2)

        i, j = random.sample(range(n), 2)
        delta = (random.random() - 0.5) * step_scale
        new_i = np.clip(cur_h[i] + delta, 0.0, 1.0)
        actual_delta = new_i - cur_h[i]
        new_j = cur_h[j] - actual_delta
        if not (0.0 <= new_j <= 1.0):
            continue

        cand = cur_h.copy()
        cand[i] = new_i
        cand[j] = new_j
        cand = normalize_sum(cand)   # re‑enforce exact sum (cheap for n≈47)

        cand_c5 = compute_c5(cand)

        if cand_c5 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_h, best_c5 = cand.copy(), cand_c5
        else:
            # Metropolis acceptance
            if random.random() < exp(-(cand_c5 - cur_c5) / temp):
                cur_h, cur_c5 = cand, cand_c5

        iteration += 1

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    budget = budget_s

    # start with the provided resolution (or 47 if absent)
    n = 47
    h, c5 = simulated_annealing(initial_vector(n), budget, start)

    # progressive refinement: double the grid a few times if time permits
    while True:
        elapsed = time.time() - start
        remaining = budget - elapsed
        if remaining < 30.0:      # need some time for another round
            break
        # double resolution
        h = np.repeat(h, 2)
        n = h.size
        # ensure feasibility after repeating (sum is automatically correct)
        h = np.clip(h, 0.0, 1.0)
        # run a shorter SA on the finer grid
        h, c5 = simulated_annealing(h, remaining, start)

        # stop after a few refinements to stay safe
        if n >= 376:   # 47 * 8
            break

    return h, float(c5), n
