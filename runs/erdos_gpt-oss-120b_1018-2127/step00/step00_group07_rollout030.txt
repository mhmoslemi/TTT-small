import time
import numpy as np

def _compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return conv.max()

def _project_bounds(delta, hi, lo):
    """Clip delta so that hi+delta∈[0,1] and lo-delta∈[0,1]."""
    if delta > 0:
        delta = min(delta, 1.0 - hi, lo)  # lo - delta >=0 → delta <= lo
    else:
        delta = max(delta, -(hi), -(1.0 - lo))
    return delta

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function for the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized vector on [0,2] with values in [0,1].
    c5_bound : float
        Corresponding maximal overlap integral.
    n_points : int
        Number of discretisation points.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine initial size / vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        # enforce exact integral (may be slightly off due to float noise)
        target_sum = n / 2.0
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
    else:
        n = 80  # fallback size
        h = rng.random(n)
        # scale to satisfy integral exactly while staying in [0,1]
        # simple rescaling then clipping iterative
        target_sum = n / 2.0
        scale = target_sum / h.sum()
        h *= scale
        # clip and fix sum if needed
        excess = h.sum() - target_sum
        while abs(excess) > 1e-12:
            mask = (h > 0) & (h < 1)
            if not np.any(mask):
                break
            adjust = excess / mask.sum()
            h[mask] -= adjust
            np.clip(h, 0.0, 1.0, out=h)
            excess = h.sum() - target_sum

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # parameters for stochastic search
    max_iter = 200000
    temperature = 0.02
    temp_decay = 0.9995
    step_scale = 0.04  # typical delta magnitude

    it = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        if it >= max_iter:
            break
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # propose delta
        delta = (rng.random() * 2.0 - 1.0) * step_scale
        delta = _project_bounds(delta, best_h[i], best_h[j])
        if delta == 0.0:
            it += 1
            continue
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # compute new bound
        new_c5 = _compute_c5(h_new)
        improve = new_c5 < best_c5
        if improve or rng.random() < np.exp(-(new_c5 - best_c5) / temperature):
            best_h = h_new
            best_c5 = new_c5
        # temperature schedule
        temperature *= temp_decay
        # occasionally shrink step scale to fine‑tune
        if it % 5000 == 0 and it > 0:
            step_scale *= 0.95
        it += 1

    return best_h, float(best_c5), int(best_h.size)
