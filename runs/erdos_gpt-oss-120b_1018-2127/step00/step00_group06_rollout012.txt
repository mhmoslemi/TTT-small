import time, math, random
import numpy as np
import scipy.optimize

def _c5_of(h):
    """Return C5 for a discretized step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project(h, target):
    """Clamp to [0,1] and adjust to meet the exact sum `target`."""
    h = np.clip(h, 0.0, 1.0)
    excess = np.sum(h) - target
    # simple water‑filling loop
    for _ in range(15):
        if abs(excess) < 1e-12:
            break
        if excess > 0:
            mask = h > 0.0
        else:
            mask = h < 1.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        adjust = excess / cnt
        h[mask] -= adjust
        h = np.clip(h, 0.0, 1.0)
        excess = np.sum(h) - target
    return h

def _anneal(initial_h, budget, seed):
    """Simple simulated‑annealing improvement on a fixed grid size."""
    rng = np.random.default_rng(seed)
    n = initial_h.size
    target_sum = n / 2.0
    h = initial_h.copy()
    best_h = h.copy()
    best_val = _c5_of(h)
    T = 1.0 + best_val          # initial temperature
    start = time.time()
    it = 0
    while time.time() - start < budget:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # propose a transfer amount
        delta = (rng.random() - 0.5) * 0.2  # scale of change
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = _project(h_new, target_sum)
        val = _c5_of(h_new)
        if val < best_val or rng.random() < math.exp(-(val - best_val) / T):
            h = h_new
            if val < best_val:
                best_val = val
                best_h = h_new.copy()
        # cooling
        it += 1
        if it % (n * 10) == 0:
            T *= 0.995
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where
    h_values is a list of numbers in [0,1] with sum = n_points/2
    and c5_bound is the achieved max overlap integral.
    """
    rng = np.random.default_rng(seed)
    start_all = time.time()
    # base grid
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        base_n = h0.size
        # enforce exact sum just in case
        h0 = _project(h0, base_n / 2.0)
    else:
        base_n = 95
        h0 = rng.random(base_n)
        h0 = _project(h0, base_n / 2.0)

    # resolution schedule (grow up to ~380 points)
    resolutions = [base_n]
    while resolutions[-1] * 2 <= 380:
        resolutions.append(resolutions[-1] * 2)

    best_h_overall = h0
    best_val_overall = _c5_of(h0)

    remaining_budget = budget_s
    for idx, n in enumerate(resolutions):
        # allocate a share of remaining time
        if idx == len(resolutions) - 1:
            alloc = remaining_budget
        else:
            # give a bit more to finer grids
            alloc = remaining_budget * 0.35
        remaining_budget -= alloc

        if n == base_n:
            h_curr = h0.copy()
        else:
            # upsample previous best solution linearly
            x_old = np.linspace(0.0, 2.0, base_n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_curr = np.interp(x_new, x_old, best_h_overall)
            h_curr = _project(h_curr, n / 2.0)

        # run annealing on this grid
        h_opt, val_opt = _anneal(h_curr, alloc, seed + idx + 1)

        if val_opt < best_val_overall:
            best_val_overall = val_opt
            best_h_overall = h_opt
            base_n = n  # future up‑sampling uses the improved size

        # safety break if time runs out
        if time.time() - start_all > budget_s * 0.98:
            break

    # final check to guarantee feasibility
    best_h_overall = _project(best_h_overall, best_h_overall.size / 2.0)
    best_val_overall = _c5_of(best_h_overall)

    return best_h_overall.tolist(), float(best_val_overall), best_h_overall.size
