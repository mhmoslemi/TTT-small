import time
import math
import numpy as np

def _compute_c5(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h at the discretisation points.
    c5_bound : float
        Corresponding max overlap value.
    n_points : int
        Number of discretisation points (length of h_values).
    """
    np.random.seed(seed)

    # -------------------------------------------------------------
    # Starting point
    # -------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: half ones, half zeros (integral = 1)
        n = 95
        h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)

    n = h.size
    dx = 2.0 / n

    # -------------------------------------------------------------
    # Helper to keep feasibility (sum = n/2, 0 ≤ h ≤ 1)
    # -------------------------------------------------------------
    target_sum = n / 2.0
    assert abs(h.sum() - target_sum) < 1e-12, "initial sum mismatch"

    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    # -------------------------------------------------------------
    # Simulated‑annealing style local search
    # -------------------------------------------------------------
    start_time = time.time()
    # algorithmic parameters
    temperature = 0.1
    step_scale = 0.1
    min_temp = 1e-6

    while time.time() - start_time < budget_s * 0.95:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        hi, hj = best_h[i], best_h[j]

        # feasible delta that moves mass from i to j while staying in [0,1]
        low = max(hi - 1.0, -hj)          # delta lower bound
        high = min(hi, 1.0 - hj)          # delta upper bound
        if low >= high:
            continue

        # sample a delta; scale to encourage smaller moves as we cool
        delta = np.random.uniform(low, high) * step_scale

        # form candidate
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate
        cand_c5 = _compute_c5(cand, dx)

        # accept if better, else with Metropolis probability
        improve = cand_c5 < best_c5
        prob = math.exp((best_c5 - cand_c5) / max(temperature, min_temp))
        if improve or np.random.rand() < prob:
            best_h = cand
            best_c5 = cand_c5

            # early exit if we already beat target
            if best_c5 <= 0.3808:
                break

        # cooling schedule
        temperature *= 0.9995
        step_scale *= 0.9999
        if temperature < min_temp:
            temperature = min_temp

    # -------------------------------------------------------------
    # Final sanity check (project back onto the exact simplex if needed)
    # -------------------------------------------------------------
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # rescale uniformly then clip and renormalise the tiny residual
        best_h *= target_sum / cur_sum
        # clip to [0,1] and fix any leftover mass
        excess = best_h.sum() - target_sum
        if abs(excess) > 1e-12:
            # distribute correction equally among interior points
            mask = (best_h > 0) & (best_h < 1)
            if mask.any():
                best_h[mask] -= excess / mask.sum()
                best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _compute_c5(best_h, dx)
    return best_h, final_c5, n
