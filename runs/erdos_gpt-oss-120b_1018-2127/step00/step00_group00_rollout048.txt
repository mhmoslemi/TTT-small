import time
import numpy as np
from scipy.optimize import minimize

# Helper: compute the C5 bound for a given step vector h
def _compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    return np.max(corr) * dx

# Helper: produce a feasible vector of length n from a source vector src
def _resample_feasible(src, n):
    src = np.asarray(src, dtype=float)
    m = src.size
    # linear interpolation onto new grid
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(new_x, old_x, src)
    # enforce bounds and the integral constraint sum(h)=n/2
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    # after scaling some values may have left [0,1]; clip and renorm iteratively
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - target) < 1e-9:
            break
        # distribute the deficit/excess uniformly where possible
        diff = target - cur
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # fallback: add diff/n everywhere then clip again
            h += diff / n
        else:
            inc = diff / free.size
            h[free] += inc
    # final clip to guarantee feasibility
    h = np.clip(h, 0.0, 1.0)
    # small correction to meet sum exactly (within float tolerance)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    return h

# Perform a local SLSQP optimisation for a given length n and start vector x0
def _optimize_h(x0, n, maxiter=800, ftol=1e-7):
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    opts = {"maxiter": maxiter, "ftol": ftol, "disp": False}
    res = minimize(_compute_c5, np.asarray(x0, dtype=float),
                   method="SLSQP", bounds=bounds,
                   constraints=cons, options=opts)
    return res

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine a base length; use the length of initial_h_values if present,
    # otherwise fall back to a modest size.

        base_n = base_h.size
    except Exception:
        base_n = 45
        base_h = np.full(base_n, 0.5)  # uniform feasible start

    # Candidate lengths to explore (must stay small for speed)
    n_list = [30, base_n, 60, 90, 120]
    best_h = None
    best_score = np.inf
    best_n = None

    for n in n_list:
        # Time check
        if time.time() - start_time > budget_s:
            break

        # Obtain initial guess for this resolution
        if n == base_n:
            h0 = base_h.copy()
        else:
            h0 = _resample_feasible(base_h, n)

        # Quick feasibility/score check before optimisation
        if not (h0 >= 0.0).all() or not (h0 <= 1.0).all():
            continue
        if not np.isclose(h0.sum(), n / 2.0, atol=1e-6):
            continue
        cur_score = _compute_c5(h0)
        if cur_score < best_score:
            best_score, best_h, best_n = cur_score, h0.copy(), n
