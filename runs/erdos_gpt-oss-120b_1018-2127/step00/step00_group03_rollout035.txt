import time
import math
import random

import numpy as np
import cvxpy as cp

# ------------------------------------------------------------
# Helper: project onto the capped simplex {0<=x<=1, sum x = target}
def _project_feasible(v, target):
    n = len(v)
    x = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(x - v)),
                      [x >= 0,
                       x <= 1,
                       cp.sum(x) == target])
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    return np.asarray(x.value).reshape(-1)

# ------------------------------------------------------------
def _initial_h(n_points):
    target_sum = n_points / 2.0
    if 'initial_h_values' in globals():
        h0 = globals()['initial_h_values']
        n0 = len(h0)
        if n0 != n_points:
            # linear interpolation onto the new grid
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x1 = np.linspace(0.0, 2.0, n_points, endpoint=False)
            h = np.interp(x1, x0, h0)
        else:
            h = h0.astype(float).copy()
    else:
        # simple deterministic start: first half 1, rest 0
        h = np.zeros(n_points, dtype=float)
        h[: n_points // 2] = 1.0
    # enforce feasibility
    h = _project_feasible(h, target_sum)
    return h

# ------------------------------------------------------------
def _c5_score(h):
    """max_k ∫ h(x)(1-h(x+k)) dx   (discrete version)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

# ------------------------------------------------------------
def _random_move(h, max_delta):
    """Pick two indices i!=j and transfer a small amount delta."""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    delta = np.random.uniform(-max_delta, max_delta)
    hi = h[i] + delta
    hj = h[j] - delta
    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
        h_new = h.copy()
        h_new[i] = hi
        h_new[j] = hj
        return h_new
    return h  # reject move that would violate box constraints

# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # ---- parameters ------------------------------------------------
    n_points = 120                     # modest size, fast evaluation
    target_sum = n_points / 2.0
    start_h = _initial_h(n_points)

    best_h = start_h.copy()
    best_score = _c5_score(best_h)

    current_h = best_h.copy()
    current_score = best_score

    max_delta = 0.2                    # initial perturbation magnitude
    start_time = time.time()

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # propose a new feasible vector by a tiny redistribution
        cand_h = _random_move(current_h, max_delta)
        cand_score = _c5_score(cand_h)

        # deterministic improvement
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h.copy()
            current_h = cand_h.copy()
            current_score = cand_score
        else:
            # simple simulated‑annealing acceptance
            # temperature falls linearly with remaining time
            T = max(0.001, (budget_s - elapsed) / budget_s)
            prob = math.exp(-(cand_score - best_score) / (0.01 * T))
            if random.random() < prob:
                current_h = cand_h.copy()
                current_score = cand_score

        # anneal the step size
        max_delta = max(0.001, max_delta * 0.9995)

    # final evaluation (just to be safe)
    final_score = _c5_score(best_h)

    return best_h, final_score, n_points
