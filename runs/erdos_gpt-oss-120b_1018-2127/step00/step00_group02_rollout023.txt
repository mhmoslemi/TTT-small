import numpy as np
import time
import random
from math import inf

def compute_c5(h):
    """Discrete C5 for step function sampled on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def perturb_pair(h, step):
    """Return a new copy of h after moving a small amount between two entries."""
    n = len(h)
    i, j = random.sample(range(n), 2)
    # direction: move amount from j to i (i increases, j decreases)
    max_inc = 1.0 - h[i]
    max_dec = h[j]
    # also allow opposite direction
    if random.random() < 0.5:
        max_inc, max_dec = max_dec, max_inc
        i, j = j, i  # swap so that i will increase, j will decrease
    if max_inc <= 0 or max_dec <= 0:
        return None  # no feasible move
    delta = step * random.random() * min(max_inc, max_dec)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def local_search(start_h, max_seconds):
    """Simple hill‑climbing with pairwise transfers, time‑bounded."""
    best_h = start_h.copy()
    best_val = compute_c5(best_h)
    n = len(best_h)
    start_time = time.time()
    init_step = 0.05  # maximum transfer per move
    while time.time() - start_time < max_seconds:
        elapsed = time.time() - start_time
        # linearly shrink step size
        step = init_step * (1.0 - elapsed / max_seconds)
        cand = perturb_pair(best_h, step)
        if cand is None:
            continue
        val = compute_c5(cand)
        if val < best_val - 1e-7:  # strict improvement
            best_h, best_val = cand, val
        # occasional random reset to escape plateaus
        if random.random() < 1e-4:
            best_h = start_h.copy()
            best_val = compute_c5(best_h)
    return best_h, best_val

def interpolate_h(old_h, new_n):
    """Interpolate old_h (length m) onto a grid of length new_n, preserving integral."""
    m = len(old_h)
    x_old = np.linspace(0.0, 2.0, m, endpoint=False) + 1.0 / m
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    new_h = np.interp(x_new, x_old, old_h)
    # repair bounds
    new_h = np.clip(new_h, 0.0, 1.0)
    # enforce exact integral by a uniform shift (preserving bounds)
    target_sum = new_n / 2.0
    diff = target_sum - new_h.sum()
    new_h += diff / new_n
    new_h = np.clip(new_h, 0.0, 1.0)
    # final rescale to hit the sum exactly (tiny adjustment)
    cur_sum = new_h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        new_h *= target_sum / cur_sum
        new_h = np.clip(new_h, 0.0, 1.0)
    return new_h
