import time
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

def _project_feasible(x, target):
    """Project vector x onto {0<=y<=1, sum(y)=target} via a small QP."""
    n = x.shape[0]
    y = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(y - x)),
                      [cp.sum(y) == target,
                       y >= 0,
                       y <= 1])
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    return np.array(y.value).reshape(-1)

def _c5_value(h):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _optimize_h(start_h, target, time_deadline):
    """Run a bounded SLSQP optimisation starting from start_h."""
    n = start_h.size
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq',
             'fun': lambda hh: float(np.sum(hh) - target)},)

    # SLSQP may be stopped by time; we catch the exception and fall back.
    res = minimize(fun=_c5_value,
                   x0=start_h,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=cons,
                   options={'maxiter': 500,
                            'ftol': 1e-9,
                            'disp': False})
    h_opt = _project_feasible(res.x, target)
    return h_opt, _c5_value(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95   # keep a safety margin

    # -------------------------------------------------------------
    # Determine starting vector and problem size
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        n = 70                     # fallback size
        h0 = np.full(n, 0.5)       # already satisfies the integral
    target = n / 2.0                # because ∫h = 1 → sum(h)=n/2

    # Project the initial vector onto the feasible set
    h0 = _project_feasible(h0, target)
    best_h = h0
    best_c5 = _c5_value(best_h)

    # -------------------------------------------------------------
    # Simple multi‑restart local search
    max_restarts = 6
    restarts = 0
    while time.time() < deadline and restarts < max_restarts:
        # Perturb current best (or original) solution
        sigma = 0.08
        trial = best_h + sigma * np.random.randn(n)
        trial = np.clip(trial, 0.0, 1.0)
        trial = _project_feasible(trial, target)

        # Optimize locally
        h_opt, c5_val = _optimize_h(trial, target, deadline)

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt

        restarts += 1

    # Ensure final feasibility and recompute exact C5
    best_h = _project_feasible(best_h, target)
    best_c5 = _c5_value(best_h)

    return best_h, best_c5, n
