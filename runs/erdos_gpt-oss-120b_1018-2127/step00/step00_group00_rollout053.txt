import time
import numpy as np
from scipy.optimize import minimize

def _compute_score(h):
    """Return C5 for step values h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _feasible(h, n, atol=1e-6):
    """Check bounds and integral constraint."""
    if np.any(h < -atol) or np.any(h > 1 + atol):
        return False
    return abs(h.sum() - n / 2) <= atol

def _project_to_feasible(h, n):
    """Scale to meet the integral, then clip to [0,1]."""
    target = n / 2.0
    cur = h.sum()
    if cur != 0:
        h = h * (target / cur)
    # clip and renormalize a tiny amount if needed
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the remaining mass uniformly where room exists
        room_up = 1.0 - h
        room_down = h
        # simple proportional redistribution
        if diff > 0:
            inc = np.minimum(room_up, diff / np.sum(room_up > 0))
            h += inc
        else:
            dec = np.minimum(room_down, -diff / np.sum(room_down > 0))
            h -= dec
    return h

def _minimize_score(h0, n, seed):
    """Run SLSQP from h0, respecting bounds and exact-sum constraint."""
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    opts = {"maxiter": 800, "ftol": 1e-9, "disp": False}
    rng = np.random.default_rng(seed)
    # add a small random perturbation to avoid degenerate start
    x0 = np.clip(h0 + 1e-4 * rng.standard_normal(n), 0.0, 1.0)
    res = minimize(_compute_score, x0, method="SLSQP",
                   bounds=bounds, constraints=cons, options=opts)
    return res

def _resample(old_h, new_n):
    """Linear interpolation of old_h (length m) to new_n points."""
    m = old_h.size
    old_x = np.linspace(0.0, 2.0, m, endpoint=False) + 1.0 / m
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    new_h = np.interp(new_x, old_x, old_h)
    return new_h
