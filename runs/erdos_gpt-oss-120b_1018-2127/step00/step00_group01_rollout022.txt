import time
import math
import numpy as np
import scipy.optimize as opt

def _evaluate(h):
    """Return the C5 value for a given step function sampled vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _make_feasible(x, target_sum):
    """Project vector x into the box [0,1] while enforcing sum == target_sum."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(5):
        s = x.sum()
        delta = (target_sum - s) / x.size
        x = x + delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction (might be tiny overflow beyond bounds)
    s = x.sum()
    if abs(s - target_sum) > 1e-12:
        # distribute the remaining error uniformly to interior points
        interior = (x > 0.0) & (x < 1.0)
        if interior.any():
            x[interior] += (target_sum - s) / interior.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _local_optimize(h0):
    """Run a small constrained local optimisation starting from h0."""
    n = h0.size
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda h: h.sum() - target_sum}
    # objective wrapper (no gradient supplied)
    def obj(h):
        return _evaluate(h)
    opts = {"maxiter": 200, "ftol": 1e-9, "disp": False}
    res = opt.minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options=opts,
    )
    if res.success:
        return res.x, res.fun
    # fallback: return the original point's evaluation
    return h0, obj(h0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (sampled uniformly) that yields a
    small value of C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Determine number of sample points
    init = globals().get("initial_h_values", None)
    n = int(init.size) if isinstance(init, (list, np.ndarray)) else 47
    target_sum = n / 2.0

    best_h = None
    best_score = math.inf

    # Helper to possibly update the best solution
    def _maybe_update(h, score):
        nonlocal best_h, best_score
        if score < best_score:
            best_h = h.copy()
            best_score = float(score)

    # 1. Try the provided initial construction if present
    if isinstance(init, (list, np.ndarray)):
        h0 = np.array(init, dtype=float)
        h0 = _make_feasible(h0, target_sum)
        h_opt, sc = _local_optimize(h0)
        _maybe_update(h_opt, sc)

    # 2. Uniform start (always feasible)
    h0 = np.full(n, 0.5, dtype=float)
    h_opt, sc = _local_optimize(h0)
    _maybe_update(h_opt, sc)

    # 3. Random restarts within remaining time
    while time.time() - start_time < budget_s - 5.0:
        # generate a random vector and project to feasibility
        rnd = np.random.rand(n)
        rnd = _make_feasible(rnd, target_sum)
        h_opt, sc = _local_optimize(rnd)
        _maybe_update(h_opt, sc)

    # Ensure we always return a feasible vector
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)

    # Final evaluation (in case numerical drift occurred)
    final_score = _evaluate(best_h)

    return best_h, final_score, n
