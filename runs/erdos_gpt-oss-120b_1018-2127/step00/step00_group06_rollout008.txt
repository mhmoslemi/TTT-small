import time
import numpy as np

def _c5_score(h, dx):
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated by correlation."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_balanced(n):
    """Return a binary (float) array of length n with exactly n/2 ones."""
    arr = np.zeros(n, dtype=np.float64)
    arr[: n // 2] = 1.0
    np.random.shuffle(arr)
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points).
    h_values is a numpy array of length n_points with entries in [0,1],
    satisfying sum(h)*dx == 1, where dx = 2.0/n_points.
    The goal is to minimize the computed C5 bound.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_budget = budget_s

    # choose a modest grid size (even) – larger grids give finer resolution but cost more time
    n_points = 120  # even
    dx = 2.0 / n_points

    # initialise with a random balanced binary vector; if a parent is provided, use it as a seed
    if 'initial_h_values' in globals():
        # interpolate the parent to current grid size, then binarise while preserving the count of ones
        parent = np.asarray(initial_h_values, dtype=np.float64)
        # linear interpolation to target size
        x_parent = np.linspace(0, 2, len(parent), endpoint=False)
        x_target = np.linspace(0, 2, n_points, endpoint=False)
        h0 = np.interp(x_target, x_parent, parent)
        # binarise by taking the top n/2 values as 1, rest 0
        thresh = np.partition(h0, -n_points // 2)[-n_points // 2]
        h = (h0 >= thresh).astype(np.float64)
        # if we happened to get more/less ones due to ties, fix it
        diff = int(h.sum()) - n_points // 2
        if diff > 0:
            # too many ones – turn some of the smallest ones to 0
            idx = np.where(h == 1)[0]
            rng.shuffle(idx)
            h[idx[:diff]] = 0.0
        elif diff < 0:
            idx = np.where(h == 0)[0]
            rng.shuffle(idx)
            h[idx[: -diff]] = 1.0
    else:
        h = _random_balanced(n_points)

    best_h = h.copy()
    best_score = _c5_score(best_h, dx)

    # current state for hill‑climbing
    cur_h = h.copy()
    cur_score = best_score

    # parameters for the simple stochastic local search
    no_improve = 0
    max_no_improve = 5000
    restart_interval = 20000

    iteration = 0
    while time.time() - start_time < time_budget - 0.5:
        iteration += 1

        # propose a swap of a 1 and a 0 (preserves the sum constraint)
        i, j = rng.integers(0, n_points, size=2)
        while cur_h[i] == cur_h[j]:
            i, j = rng.integers(0, n_points, size=2)
        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        new_score = _c5_score(cur_h, dx)

        # accept if improvement; otherwise revert
        if new_score < cur_score:
            cur_score = new_score
            no_improve = 0
            # update global best if needed
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            no_improve += 1

        # occasional random restart to escape possible plateaus
        if no_improve >= max_no_improve or iteration % restart_interval == 0:
            cur_h = _random_balanced(n_points)
            cur_score = _c5_score(cur_h, dx)
            no_improve = 0

    # final safeguard: ensure the returned vector satisfies the constraints exactly
    # (numerical rounding may have introduced tiny drift)
    cur_sum = best_h.sum()
    target_sum = n_points / 2
    if cur_sum != target_sum:
        # adjust by flipping the smallest |value-0.5| entries
        diff = int(target_sum - cur_sum)
        if diff > 0:
            # need more ones

            rng.shuffle(zero_idx)
            best_h[zero_idx[:diff]] = 1.0
        elif diff < 0:
            one_idx = np.where(best_h == 1)[0]
            rng.shuffle(one_idx)
            best_h[one_idx[: -diff]] = 0.0

    final_score = _c5_score(best_h, dx)
    return best_h, float(final_score), int(n_points)
