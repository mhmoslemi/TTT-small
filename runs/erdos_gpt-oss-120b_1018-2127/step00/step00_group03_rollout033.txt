import time
import numpy as np
import random

def compute_c5(h, dx):
    # full linear correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def binary_initial(N):
    # simple deterministic start: first N/2 ones then zeros
    h = np.zeros(N, dtype=np.float64)
    h[: N // 2] = 1.0
    np.random.shuffle(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # Determine number of points
    if "initial_h_values" in globals():
        init_arr = np.asarray(initial_h_values, dtype=np.float64)
        N = len(init_arr)
        # Binarise while preserving sum = N/2
        half = N // 2
        idx = np.argpartition(-init_arr, half)[:half]
        h = np.zeros(N, dtype=np.float64)
        h[idx] = 1.0
    else:
        N = 200  # moderate size, even
        h = binary_initial(N)

    dx = 2.0 / N
    best_h = h.copy()
    best_score = compute_c5(best_h, dx)

    current_h = h.copy()
    current_score = best_score

    # annealing parameters
    max_iters = 500000
    T0 = 0.3
    Tmin = 1e-4
    decay = (Tmin / T0) ** (1.0 / max_iters)

    temperature = T0
    iteration = 0

    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:
            break
        if iteration >= max_iters:
            break

        # propose a swap of two positions (maintains binary and sum)
        i, j = np.random.randint(0, N, size=2)
        while i == j or current_h[i] == current_h[j]:
            i, j = np.random.randint(0, N, size=2)

        # swap
        current_h[i], current_h[j] = current_h[j], current_h[i]

        new_score = compute_c5(current_h, dx)

        delta = new_score - current_score
        if delta < 0 or random.random() < np.exp(-delta / temperature):
            # accept
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = current_h.copy()
        else:
            # reject: swap back
            current_h[i], current_h[j] = current_h[j], current_h[i]

        temperature *= decay
        iteration += 1

        # occasional random restart to escape local minima
        if iteration % 50000 == 0:
            # keep best, but re‑shuffle rest
            current_h = best_h.copy()
            np.random.shuffle(current_h)

    return (best_h.tolist(), best_score, N)
