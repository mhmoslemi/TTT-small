import time
import numpy as np

def _objective(h, dx):
    """Maximum overlap integral approximation."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _project_capped_simplex(v, target, cap=1.0, iters=5):
    """
    Project vector v onto {x : 0 <= x <= cap, sum x = target}.
    Simple iterative redistribution – sufficient for our heuristic.
    """
    x = np.clip(v, 0.0, cap)
    for _ in range(iters):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Distribute diff uniformly among non‑saturated entries
        free = (x > 0.0) & (x < cap)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds, just scale globally then clip again
            x += diff / len(x)
            x = np.clip(x, 0.0, cap)
        else:
            inc = diff / cnt
            x[free] += inc
            x = np.clip(x, 0.0, cap)
    # Final tiny correction to meet exact sum (may violate bounds by eps)
    s = x.sum()
    if abs(s - target) > 1e-12:
        correction = (target - s) / len(x)
        x += correction
        x = np.clip(x, 0.0, cap)
        # renormalize one last time
        delta = target - x.sum()
        if abs(delta) > 1e-12:
            # distribute remaining delta uniformly
            x += delta / len(x)
            x = np.clip(x, 0.0, cap)
    return x
