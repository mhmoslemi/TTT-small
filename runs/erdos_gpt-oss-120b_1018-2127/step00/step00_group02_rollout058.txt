import time
import numpy as np
from scipy import optimize

def compute_c5(h):
    """Discrete C5 using full correlation as in the evaluator."""
    dx = 2.0 / h.size
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_bounds_and_sum(x, target_sum):
    """Clip to [0,1] and adjust to hit target_sum by a simple affine shift."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    if cur == 0:
        # avoid division by zero, fall back to uniform
        x = np.full_like(x, target_sum / x.size)
    else:
        shift = (target_sum - cur) / x.size
        x = x + shift
    # second clip may break the sum a little; redo once if needed
    x = np.clip(x, 0.0, 1.0)
    # final renorm to exact sum (preserving bounds by scaling of interior points)
    cur = x.sum()
    if cur != 0:
        scale = target_sum / cur
        x = x * scale
    return np.clip(x, 0.0, 1.0)

def slsqp_optimize(h0, budget, start_time):
    """Run SLSQP from a starting point respecting time budget."""
    n = h0.size
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = ({
        'type': 'eq',
        'fun': lambda x: x.sum() - target
    },)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    # optimizer options – keep iteration counts modest
    opts = {'maxiter': 2000, 'ftol': 1e-9, 'disp': False}

    # single run; if it exceeds overall budget we abort early

        if time.time() - start_time > budget:
            raise TimeoutError

    try:
        res = optimize.minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options=opts,
            callback=callback
        )
    except TimeoutError:
        return best_h, best_val

    if res.success and res.fun < best_val:
        best_h = project_bounds_and_sum(res.x, target)
        best_val = compute_c5(best_h)

    return best_h, best_val

def interpolate_h(old_h, new_n):
    """Linear interpolation of a step function from old grid to new grid."""
    old_n = old_h.size
    dx_old = 2.0 / old_n
    dx_new = 2.0 / new_n
