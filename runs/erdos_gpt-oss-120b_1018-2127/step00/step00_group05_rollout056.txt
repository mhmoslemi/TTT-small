import time
import numpy as np
import random

def _compute_c5(arr, dx):
    """Maximum overlap integral for a discrete step function."""
    # correlation h with (1-h); np.correlate already does the required flip
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(corr.max() * dx)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with integral 1 that minimizes
    C₅ = max_k ∫ h(x)(1‑h(x+k)) dx.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : list of floats, length n_points, each in [0,1]
        c5_bound : float, the achieved C₅ value
        n_points : int, number of discretisation points used
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    try:
        # parent construction may be present
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    except NameError:
        # fall back to a simple half‑ones construction
        n = 120                       # modest size, fast enough for the budget
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0

    dx = 2.0 / n
    target_sum = n / 2.0               # because ∫h = 1 ⇒ Σh * dx = 1

    # enforce box constraints
    h = np.clip(h0, 0.0, 1.0)

    # repair integral (simple redistribution, preserving bounds)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # try to add the whole diff to the first element that can absorb it
        for idx in range(n):
            available = 1.0 - h[idx] if diff > 0 else h[idx]
            change = min(abs(diff), available)
            change = change if diff > 0 else -change
            h[idx] += change
            diff -= change
            if abs(diff) < 1e-12:
                break
        # if still off due to rounding, renormalise uniformly
        if abs(h.sum() - target_sum) > 1e-9:
            h *= target_sum / h.sum()

    # ------------------------------------------------------------------
    # Local search (pairwise mass transfer)
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    max_iter = 200000
    iter_done = 0

    while (time.time() - start_time) < budget_s * 0.95 and iter_done < max_iter:
        iter_done += 1

        # decreasing step size
        eps = 0.08 * (1.0 - iter_done / max_iter) + 0.001

        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() * 2.0 - 1.0) * eps   # propose transfer

        hi, hj = best_h[i], best_h[j]
        new_i, new_j = hi + delta, hj - delta

        # keep within [0,1]; otherwise reject
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue

        cand = best_h.copy()
        cand[i] = new_i
        cand[j] = new_j

        c5 = _compute_c5(cand, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand

    # ------------------------------------------------------------------
    # Return solution
    # ------------------------------------------------------------------
    return (best_h.tolist(), best_c5, int(n))
