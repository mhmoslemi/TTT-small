from scipy.optimize import minimize

# Helper: compute the C5 bound for a given step function vector h
def compute_score(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

# Helper: resample a vector defined on [0,2) to a different resolution
def resample_h(old_h, new_n):
    old_n = old_h.size
    if old_n == new_n:
        return old_h.copy()
    # grid points (start at 0, step dx, exclude the endpoint 2)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
