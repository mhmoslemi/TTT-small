import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Discrete C5 for step function sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target, max_iter=40):
    """
    Project vector x onto the box [0,1] while forcing sum == target.
    Uses a simple bisection on the additive shift.
    """
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 45
        h0 = np.full(n, 0.5)  # uniform feasible start

    target_sum = n / 2.0
    # Ensure feasibility of initial vector
    h0 = project_to_sum_box(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}

    # Primary local refinement of the provided initial solution
    res = minimize(
        compute_c5,
        best_h,
        method="SLSQP",
        bounds=bounds,
        constraints=[eq_cons],
        options={"maxiter": 2000, "ftol": 1e-10, "disp": False},
    )
    if res.success:
        cand = project_to_sum_box(res.x, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5

    # Random restarts within remaining time
    iteration = 0
    while time.time() - start_time < budget_s - 5.0 and iteration < 30:
        iteration += 1

        # Random feasible start via Dirichlet‑like scaling
        rnd = np.random.rand(n)
        rnd = rnd / rnd.sum() * target_sum
        rnd = project_to_sum_box(rnd, target_sum)

        res = minimize(
            compute_c5,
            rnd,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 2000, "ftol": 1e-10, "disp": False},
        )
        if not res.success:
            continue

        cand = project_to_sum_box(res.x, target_sum)
        cand_c5 = compute_c5(cand)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5

    return best_h, best_c5, n
