import numpy as np
import random
import time

# helpers --------------------------------------------------------------

def repair(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and scale to meet the exact sum target."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        h = np.full_like(h, target_sum / h.size)
        return np.clip(h, 0.0, 1.0)
    h = h * (target_sum / s)
    # one more clip–scale pass to fix possible bound violations after scaling
    for _ in range(2):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        h = h * (target_sum / s)
    return h

def c5_score(h: np.ndarray) -> float:
    """Return the C5 value for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def mutate_resample(parent: np.ndarray, frac: float, target_sum: float) -> np.ndarray:
    """Replace a fraction of entries by fresh uniform samples."""
    n = parent.size
    child = parent.copy()
    k = max(1, int(frac * n))
    idx = np.random.choice(n, k, replace=False)
    child[idx] = np.random.rand(k)
    return repair(child, target_sum)

def mutate_noise(parent: np.ndarray, frac: float, amp: float, target_sum: float) -> np.ndarray:
    """Add small uniform noise to a fraction of entries."""
    n = parent.size
    child = parent.copy()
    mask = np.random.rand(n) < frac
    if mask.any():
        noise = (np.random.rand(mask.sum()) * 2.0 - 1.0) * amp
        child[mask] += noise
    return repair(child, target_sum)

# main entry -----------------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Optimised discretisation (values in [0,1], sum·dx = 1).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points.
    """
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------------
    # 1. Determine problem size and initialise population
    # --------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n_points = init.size
        target_sum = n_points / 2.0
        # Ensure feasibility
        init = repair(init, target_sum)
    else:
        n_points = 70                     # reasonable default
        target_sum = n_points / 2.0
        init = np.full(n_points, 0.5)     # uniform start, will be repaired
        init = repair(init, target_sum)

    pop_size = 12
    population = [init]
    while len(population) < pop_size:
        rand = np.random.rand(n_points)
        population.append(repair(rand, target_sum))

    scores = [c5_score(ind) for ind in population]
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # --------------------------------------------------------------
    # 2. Evolution loop respecting the time budget
    # --------------------------------------------------------------
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    max_time = budget_s * 0.99   # leave a tiny margin

    generation = 0
    while elapsed() < max_time:
        generation += 1
        new_population = []
        new_scores = []

        # elitism: keep current best
        new_population.append(best_h)
        new_scores.append(best_score)

        # generate offspring
        for _ in range(pop_size - 1):
            parent = random.choice(population)
            # pick mutation type probabilistically
            if random.random() < 0.55:
                child = mutate_resample(parent, frac=0.20, target_sum=target_sum)
            else:
                child = mutate_noise(parent, frac=0.30, amp=0.015, target_sum=target_sum)
            score = c5_score(child)

            new_population.append(child)
            new_scores.append(score)

        # select the top pop_size individuals for next generation
        idx_sorted = np.argsort(new_scores)[:pop_size]
        population = [new_population[i] for i in idx_sorted]
        scores = [new_scores[i] for i in idx_sorted]

        # update global best
        if scores[0] < best_score:

                    population[i] = repair(rand, target_sum)
                    scores[i] = c5_score(population[i])

    # --------------------------------------------------------------
    # 3. Return result
    # --------------------------------------------------------------
    # Ensure final feasibility
    best_h = repair(best_h, target_sum)
    best_score = c5_score(best_h)
    return best_h, best_score, n_points
