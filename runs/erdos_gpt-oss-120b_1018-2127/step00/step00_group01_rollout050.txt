import time
import numpy as np
from scipy.optimize import differential_evolution

def _compute_c5(h):
    """Return max overlap integral for discrete step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _objective(x, n, target_sum):
    """
    DE objective.  x holds the first n-1 entries of h.
    The last entry is forced to satisfy the integral constraint.
    """
    h = np.empty(n, dtype=float)
    h[:-1] = x
    last = target_sum - np.sum(x)
    # penalty if last out of [0,1]
    if last < 0.0 or last > 1.0:
        return 1e6 + abs(last) * 1e6
    h[-1] = last
    # small box‑constraint penalty for safety
    if np.any(h < 0.0) or np.any(h > 1.0):
        return 1e6
    return _compute_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h=1 and 0≤h≤1.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine number of discretisation points.

        init_h = np.asarray(initial_h_values, dtype=float)
        n = init_h.size
    else:
        n = 47  # fallback size
        init_h = np.full(n, 0.5, dtype=float)  # satisfies the integral exactly

    target_sum = n / 2.0  # because dx = 2/n

    # Bounds for the first n-1 variables.
    bounds = [(0.0, 1.0)] * (n - 1)

    # Differential Evolution options.
    maxiter = 200  # modest to stay well within the time budget
    popsize = 15   # default * dimension scaling

    result = differential_evolution(
        _objective,
        bounds,
        args=(n, target_sum),
        seed=seed,
        maxiter=maxiter,
        popsize=popsize,
        polish=True,
        updating="deferred",
        disp=False,
        workers=1,
    )

    # Reconstruct the full h vector from the DE solution.
    x_best = result.x
    h_best = np.empty(n, dtype=float)
    h_best[:-1] = x_best
    h_best[-1] = target_sum - np.sum(x_best)

    # Clip to [0,1] just in case of tiny numerical drift.
    np.clip(h_best, 0.0, 1.0, out=h_best)

    c5_bound = _compute_c5(h_best)

    elapsed = time.time() - start_time
    # If we somehow exceeded the budget (unlikely), return the best we have.
    if elapsed > budget_s:
