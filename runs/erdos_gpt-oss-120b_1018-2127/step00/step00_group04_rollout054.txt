import time
import numpy as np
import random

def _evaluate(h, dx):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _pairwise_delta(h, i, j):
    """Return a feasible random delta to add to h[i] and subtract from h[j]."""
    # max increase at i without exceeding 1
    max_inc_i = 1.0 - h[i]
    # max decrease at j without dropping below 0
    max_dec_j = h[j]
    # feasible delta is limited by both directions
    max_pos = min(max_inc_i, max_dec_j)
    # max decrease at i, increase at j
    max_dec_i = h[i]
    max_inc_j = 1.0 - h[j]
    max_neg = -min(max_dec_i, max_inc_j)
    if max_pos == 0 and max_neg == 0:
        return 0.0
    return random.uniform(max_neg, max_pos)

def _anneal(h, dx, time_limit, start_time):
    """Simple simulated annealing on a fixed-size vector."""
    n = h.size
    best_h = h.copy()
    best_score = _evaluate(h, dx)
    cur_score = best_score
    temperature = 0.02  # initial temperature
    decay = 0.9995

    while time.time() - start_time < time_limit:
        # pick two distinct indices
        i, j = random.sample(range(n), 2)
        delta = _pairwise_delta(h, i, j)
        if delta == 0.0:
            continue
        h[i] += delta
        h[j] -= delta

        new_score = _evaluate(h, dx)
        improve = new_score < cur_score
        accept = improve or (random.random() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)))
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta

        temperature *= decay
        if temperature < 1e-5:
            temperature = 1e-5
    return best_h, best_score

def _initialize_h(n):

                h[mask] -= adjust
