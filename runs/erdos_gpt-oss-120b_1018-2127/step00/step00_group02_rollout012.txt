# Helper to compute the C5 bound for a discrete step function h on [0,2]
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

# Obtain a feasible starting vector of length n
def get_initial_h(n, seed=0):
    rng = np.random.RandomState(seed)
    # use the provided parent construction if present
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        m = parent.size
        if n == m:
            return parent.copy()
        # simple up‑sampling by repetition (preserves sum exactly)
        rep = n // m
        extra = n - rep * m
        h = np.repeat(parent, rep)
        if extra:
            h = np.concatenate([h, parent[:extra]])
        # enforce exact integral: sum must be n/2
        target = n / 2.0
        cur = h.sum()
        if abs(cur - target) > 1e-9:
            h += (target - cur) / n
            np.clip(h, 0.0, 1.0, out=h)
        return h
    except NameError:
        # no parent: create a binary vector with exactly n/2 ones
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        rng.shuffle(h)
        return h

# Perform a simple projected random walk (simulated annealing)
def optimize_h(h0, budget_s, seed=0):
    rng = np.random.RandomState(seed)
    n = h0.size
    best_h = h0.copy()
    best_score = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    start = time.time()
    # temperature schedule parameters
    T0 = max(0.01, (best_score - 0.3) if best_score > 0.3 else 0.01)
    t = 0
    while time.time() - start < budget_s * 0.94:
        t += 1
        # exponential cooling
        T = T0 * math.exp(-t / 2000.0)

        # pick two distinct indices
        i, j = rng.choice(n, 2, replace=False)
        hi, hj = cur_h[i], cur_h[j]

        # decide direction of transfer
        if rng.rand() < 0.5:
            # move from j to i
            max_transfer = min(1.0 - hi, hj)
        else:
            # move from i to j
            max_transfer = min(hi, 1.0 - hj)
            i, j = j, i  # swap so we always add to i, subtract from j
            hi, hj = cur_h[i], cur_h[j]

        if max_transfer <= 0:
            continue

        # random delta, limited to a fraction of the allowed transfer
        delta = rng.rand() * max_transfer * 0.5
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_score = compute_c5(new_h)

        # accept if better or with Metropolis probability
        if new_score < cur_score or rng.rand() < math.exp((cur_score - new_score) / max(T, 1e-12)):
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score

        # occasional random restart to escape flat regions
        if t % 5000 == 0 and rng.rand() < 0.1:
            cur_h = best_h.copy()
            cur_score = best_score

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) respecting the problem constraints."""
    np.random.seed(seed)

    # start with a modest resolution; we will try a few sizes within the budget
    candidates_n = [60, 90, 120]  # all even, keep evaluation cheap
    best_overall = None
    best_score = float("inf")
    start_all = time.time()

    for n in candidates_n:
        # allocate part of the time budget proportionally
        remaining = budget_s - (time.time() - start_all)
        if remaining <= 0:
            break
        # reserve at least 5 seconds per size
        alloc = max(5.0, remaining / (len(candidates_n) - candidates_n.index(n)))
        h0 = get_initial_h(n, seed=seed + n)
        h_opt, score = optimize_h(h0, alloc, seed=seed + n * 7)

        # keep the best across resolutions
        if score < best_score:
            best_score = score
            best_overall = (h_opt, score, n)

        # early exit if we already beat the target
        if best_score <= 0.3808:
            break

    # safety: if nothing found, fallback to the initial parent (or binary)
    if best_overall is None:
        h0 = get_initial_h(60, seed=seed)
        best_overall = (h0, compute_c5(h0), 60)

    h_vals, c5_bound, n_pts = best_overall

    # final sanity checks
    assert np.all(h_vals >= -1e-12) and np.all(h_vals <= 1 + 1e-12), "bounds violated"
    assert abs(h_vals.sum() - n_pts / 2.0) < 1e-9, "integral constraint violated"

    return (h_vals, c5_bound, n_pts)
