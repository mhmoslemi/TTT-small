import numpy as np
import time
from scipy.optimize import differential_evolution

# helper: project onto [0,1] box and enforce exact sum = target (capped simplex)
def _project_box_simplex(v, target, max_iter=60, tol=1e-9):
    # water-filling with binary search on lambda
    lo = -np.max(v) - 1.0  # ensures all v+lambda <=0
    hi = 1.0 - np.min(v) + 1.0  # ensures all v+lambda >=1
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            hi = lam
        else:
            lo = lam
    # final clipping
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _initial_vector(n):
    # if a global initial_h_values exists and matches size, use it
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size == n:
            return init.copy()
        # otherwise interpolate/extrapolate
        xi = np.linspace(0, init.size - 1, n)
        return np.interp(xi, np.arange(init.size), init)
    # fallback: uniform distribution satisfying integral
    h = np.full(n, 0.5)
    return h

def _local_search(h0, time_limit, best_val):
    n = h0.size
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = best_val
    start = time.time()
    while time.time() - start < time_limit:
        # propose Gaussian perturbation
        cand = best_h + np.random.normal(scale=0.05, size=n)
        cand = _project_box_simplex(cand, target)
        c5 = _c5_value(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_all = time.time()
    # candidate grid sizes (keep modest)
    grids = [47, 94, 188]
    overall_best_h = None
    overall_best_c5 = np.inf
    overall_n = None

    # allocate roughly equal time per grid, but ensure at least some minimal time
    per_grid_budget = max(1.0, budget_s / len(grids))

    for n in grids:
        if time.time() - start_all > budget_s:
            break
        # construct feasible start
        h_start = _initial_vector(n)
        target = n / 2.0
        h_start = _project_box_simplex(h_start, target)
        # quick global improvement with DE (few generations)
        bounds = [(0.0, 1.0) for _ in range(n)]

        # penalty function embeds sum constraint via projection inside objective
        def de_obj(x):
            x = _project_box_simplex(x, target)
            return _c5_value(x)

        # limit DE evaluations to keep within time
        max_iter = 30
        try:
            result = differential_evolution(
                de_obj,
                bounds,
                maxiter=max_iter,
                popsize=10,
                polish=False,
                seed=seed,
                atol=0,
                updating='deferred',
                workers=1,
                disp=False,
                init='latinhypercube',
            )
            h_candidate = _project_box_simplex(result.x, target)
            c5_candidate = _c5_value(h_candidate)

            h_candidate = h_start
            c5_candidate = _c5_value(h_candidate)
