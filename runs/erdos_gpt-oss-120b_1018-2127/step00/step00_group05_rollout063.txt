import time
import numpy as np

def _repair_binary(vec, target):
    """Ensure vec is binary (0/1) and sum equals target."""
    vec = np.where(vec > 0.5, 1.0, 0.0)
    cur = int(vec.sum())
    if cur > target:
        ones = np.flatnonzero(vec)
        to_zero = np.random.choice(ones, cur - target, replace=False)
        vec[to_zero] = 0.0
    elif cur < target:
        zeros = np.flatnonzero(1 - vec)
        to_one = np.random.choice(zeros, target - cur, replace=False)
        vec[to_one] = 1.0
    return vec

def _evaluate(h, dx):
    """Compute C5 for a given step vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    # resolution
    n_points = kwargs.get("n_points", 120)
    dx = 2.0 / n_points
    target_sum = n_points // 2          # exactly half the samples should be 1

    # ------------------------------------------------------------------
    # Build an initial binary vector, using the supplied initial_h_values
    # if they exist, otherwise start from a simple block pattern.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        old_n = init.size
        xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        xs_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        interp = np.interp(xs_new, xs_old, init)
        h0 = (interp > 0.5).astype(float)
    else:
        # simple block: first half ones, second half zeros
        h0 = np.concatenate([np.ones(target_sum), np.zeros(n_points - target_sum)])
        np.random.shuffle(h0)

    h0 = _repair_binary(h0, target_sum)
    best_h = h0.copy()
    best_score = _evaluate(best_h, dx)

    # ------------------------------------------------------------------
    # Simple stochastic hill‑climbing in the space of binary vectors.
    # ------------------------------------------------------------------
    no_improve = 0
    max_no_improve = 2000          # restart after stagnation
    while True:
        if time.time() - t_start > budget_s - 0.5:
            break

        # flip a few bits while preserving the total number of ones
        flip_cnt = np.random.randint(1, 4)   # 1,2 or 3 bits
        ones = np.flatnonzero(best_h)
        zeros = np.flatnonzero(1.0 - best_h)

        if ones.size == 0 or zeros.size == 0:
            candidate = np.random.choice([0.0, 1.0], size=n_points)
            candidate = _repair_binary(candidate, target_sum)
        else:
            # choose indices to toggle
            flip_from_one = np.random.choice(ones, min(flip_cnt, ones.size), replace=False)
            flip_from_zero = np.random.choice(zeros, min(flip_cnt, zeros.size), replace=False)
            candidate = best_h.copy()
            candidate[flip_from_one] = 0.0
            candidate[flip_from_zero] = 1.0

        # evaluate
        score = _evaluate(candidate, dx)

        if score < best_score:
            best_h = candidate
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        # random restart if stuck
        if no_improve >= max_no_improve:
            rand_vec = np.random.rand(n_points)
            best_h = _repair_binary(rand_vec, target_sum)
            best_score = _evaluate(best_h, dx)
            no_improve = 0

    # ensure exact constraint before returning
    best_h = _repair_binary(best_h, target_sum)
    final_score = _evaluate(best_h, dx)

    return best_h.astype(float), float(final_score), int(n_points)
