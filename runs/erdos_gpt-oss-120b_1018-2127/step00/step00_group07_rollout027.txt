import time, random, math
import numpy as np

def _project_to_bounds_sum(x, target, lower=0.0, upper=1.0, max_iter=20):
    """Clamp to [lower,upper] and enforce sum == target by iterative redistribution."""
    x = np.clip(x, lower, upper)
    for _ in range(max_iter):
        total = x.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds – distribute uniformly then re‑clip
            x += diff / len(x)
            x = np.clip(x, lower, upper)
            continue
        adjust = diff / cnt
        x[free] += adjust
        x = np.clip(x, lower, upper)
    return x

def _c5_value(h):
    """Compute the overlap constant C5 for a discretised step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # Determine problem size
    n = int(len(initial_h_values)) if 'initial_h_values' in globals() else 80

    # Initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n, 0.5, dtype=float)   # average 0.5 gives correct integral

    # Project to feasible region (bounds + exact integral)
    target_sum = n / 2.0                     # because ∫h = 1 ⇒ sum(h)*dx = 1 ⇒ sum = n/2
    h = _project_to_bounds_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # Simple simulated‑annealing / hill‑climbing loop
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # Temperature schedule (linear decay to a small floor)
        temperature = max(0.001, 1.0 - elapsed / budget_s)

        # Propose a move: transfer a small amount from i to j
        i, j = random.sample(range(n), 2)
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        # Random step up to half of the feasible transfer
        delta = random.random() * max_transfer * 0.5

        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # No need to re‑project: sum is preserved and bounds respected by construction
        cand_c5 = _c5_value(h_new)

        # Accept if improvement or probabilistically according to temperature
        if cand_c5 < best_c5 or random.random() < math.exp((best_c5 - cand_c5) / temperature):
            best_h = h_new
            best_c5 = cand_c5

        # Occasionally perform a random perturbation to escape plateaus
        if random.random() < 0.001:
            perturb = np.random.uniform(-0.05, 0.05, size=n)
            h_pert = best_h + perturb
            h_pert = _project_to_bounds_sum(h_pert, target_sum)
            pert_c5 = _c5_value(h_pert)
            if pert_c5 < best_c5:
                best_h = h_pert
                best_c5 = pert_c5

    # Final safeguard: ensure feasibility before returning
    best_h = _project_to_bounds_sum(best_h, target_sum)

    return best_h.tolist(), float(best_c5), n
