import time

def _compute_c5(h):
    """Return the overlap constant C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) over all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_bounds_and_sum(h, target_sum):

    h = np.clip(h, 0.0, 1.0)

            h *= target_sum / current
