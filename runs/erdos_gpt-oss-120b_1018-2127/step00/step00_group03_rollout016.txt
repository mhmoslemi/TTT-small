import time
import numpy as np
import cvxpy as cp

def _dx(n):
    return 2.0 / n

def _compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.shape[0]
    dx = _dx(n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_feasible(v, target_sum):
    """Project vector v onto {0<=x<=1, sum(x)=target_sum} via a small QP."""
    n = v.shape[0]
    x = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(x - v))
    constraints = [x >= 0, x <= 1, cp.sum(x) == target_sum]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=200)
    # In the rare case solver fails, fall back to clipping and scaling.
    if x.value is None:
        proj = np.clip(v, 0.0, 1.0)
        excess = proj.sum() - target_sum
        if abs(excess) > 1e-9:
            proj -= excess / n
            proj = np.clip(proj, 0.0, 1.0)
        return proj
    return np.asarray(x.value).reshape(-1)

def _worst_lag_gradient(h):
    """Return (lag, gradient) for the lag attaining the maximal overlap."""
    n = h.shape[0]
    dx = _dx(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)                     # index in full correlation array
    lag = idx - (n - 1)                       # convert to shift (positive means h shifted right)
    grad = np.zeros_like(h)

    if lag >= 0:
        i0, i1 = 0, n - lag
        grad[i0:i1] += 1.0 - h[i0 + lag:i1 + lag]   # d/d h[i]
        grad[i0 + lag:i1 + lag] += -h[i0:i1]        # d/d h[i+lag]
    else:
        L = -lag
        i0, i1 = L, n
        grad[i0:i1] += 1.0 - h[i0 - L:i1 - L]
        grad[i0 - L:i1 - L] += -h[i0:i1]

    # scale gradient by dx (same factor as objective)
    grad *= dx
    return lag, grad

    """
    np.random.seed(seed)
    start_time = time.time()

    n = h.shape[0]
    target_sum = n / 2.0

    # Ensure feasibility of the starting point.
    h = _project_feasible(h, target_sum)

    best_h = h.copy()
    best_c5 = _compute_c5(h)

    # Optimization hyper‑parameters
    max_iter = 5000
    base_step = 0.4               # initial step magnitude
