import time
import numpy as np

def _project_to_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the target sum (capped simplex projection)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(15):
        s = v.sum()
        if abs(s - target) <= 1e-9:
            break
        diff = target - s
        free = (v > 1e-12) & (v < 1.0 - 1e-12)
        if free.any():
            add = diff / free.sum()
            v[free] += add
        else:
            # all variables at a bound – scale the whole vector
            if s > 0:
                v *= target / s
            else:
                v[:] = target / len(v)
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    s = v.sum()
    if abs(s - target) > 1e-6 and s > 0:
        v *= target / s
    return np.clip(v, 0.0, 1.0)


def _eval_c5(h: np.ndarray, dx: float) -> float:
    """Return the C5 value for a given step vector."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _upsample(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear up‑sample a step vector to a new resolution."""
    old_n = len(old)
    if old_n == new_n:
        return old.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    return np.interp(new_x, old_x, old)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Stochastic search for a step function h on [0,2] with integral 1 that
    minimises the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best step values found (length n_points, each in [0,1]).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points used.
    """
    rnd = np.random.RandomState(seed)

    base_n = int(kwargs.get('n_points',
                len(globals().get('initial_h_values', [])) or 80))
    # stay under 1000 samples for speed

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum(h)*dx = 1

    # ------------------------------------------------------------------

    if 'initial_h_values' in globals():
        h0 = _upsample(np.asarray(initial_h_values, dtype=float), n_points)

        h0 = np.full(n_points, target_sum / n_points)

    h0 = _project_to_box_simplex(h0, target_sum)
    best_h = h0
    best_score = _eval_c5(best_h, dx)

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    start_time = time.time()
    max_iters = 200000  # safe upper bound, will break on time
    iter_count = 0
    # temperature schedule (range of noise)
    def noise_scale(frac):
        # start with larger moves, shrink to ~0.001
        return 0.5 * (0.5 * frac + 0.001)

                best_h, best_score = rand_vec, rand_score

    # Return best found solution
    # ------------------------------------------------------------------
