import time
import numpy as np

def _adjust_sum(vec, target):
    """Project vec to [0,1] box and enforce sum == target (simple iterative)."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(15):
        excess = v.sum() - target
        if abs(excess) < 1e-12:
            break
        # indices that can be moved in the needed direction
        if excess > 0:
            idx = np.where(v > 1e-12)[0]
        else:
            idx = np.where(v < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            break
        delta = excess / len(idx)
        v[idx] -= delta
        v = np.clip(v, 0.0, 1.0)
    return v

def _score(h):
    """Compute C5 for a step function sampled in h."""
    n = h.size
    dx = 2.0 / n
    # full convolution of h with (1-h) gives length 2n-1 array
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) trying to minimise the Erdős C5 constant."""
    start = time.time()
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target_sum = n / 2.0
        h_best = _adjust_sum(h0.copy(), target_sum)
    else:
        n = 95                     # fallback size if no parent provided
        target_sum = n / 2.0
        rand = np.random.rand(n)
        h_best = _adjust_sum(rand, target_sum)

    best_score = _score(h_best)

    # parameters for stochastic search
    max_iter = 200000
    sigma_start = 0.2
    sigma_end = 0.01
    report_interval = 2.0      # seconds between optional progress checks

    last_report = start
    iter_cnt = 0

    while True:
        now = time.time()
        if now - start > budget_s * 0.98:
            break
        if iter_cnt >= max_iter:
            break

        # linearly anneal sigma
        sigma = sigma_start + (sigma_end - sigma_start) * (iter_cnt / max_iter)

        cand = h_best + np.random.normal(scale=sigma, size=n)
        cand = _adjust_sum(cand, target_sum)

        cand_score = _score(cand)

        if cand_score < best_score:
            h_best = cand
            best_score = cand_score

        iter_cnt += 1

        # occasional cheap random restart to avoid stagnation
        if iter_cnt % 5000 == 0:
            rand = np.random.rand(n)
            rand = _adjust_sum(rand, target_sum)
            rand_score = _score(rand)
            if rand_score < best_score:
                h_best = rand
                best_score = rand_score

        # optional minimal reporting (no prints, just keep time checks)
        if now - last_report > report_interval:
            last_report = now

    return (h_best, best_score, n)
