import time
import numpy as np

def _project_to_sum_and_bounds(v, target, low=0.0, high=1.0, max_iter=100):
    """
    Projects vector v onto the box [low, high] while enforcing sum(v) = target.
    Simple iterative method: clip, then adjust uniformly, repeat.
    """
    v = np.clip(v, low, high)
    for _ in range(max_iter):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        v += diff / len(v)
        v = np.clip(v, low, high)
    return v

def _c5_score(h, dx):
    """
    Compute the C5 value for a discrete step function h.
    """
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimises a step function h on [0,2] (discretised) to minimise the
    Erdős overlap constant C5.

    Returns:
        (h_values, c5_bound, n_points)
    """
    rnd = np.random.default_rng(seed)

    # ------------------- parameters -------------------
    n_points = 120                     # < 1k for speed
    dx = 2.0 / n_points
    target_sum = n_points / 2.0        # because ∫h = 1  => sum(h)*dx = 1

    # ------------------- initial solution -------------------
    if 'initial_h_values' in globals():
        # interpolate the parent solution to our grid
        parent = np.asarray(initial_h_values, dtype=float)
        parent_n = parent.size
        x_parent = np.linspace(0, 2, parent_n, endpoint=False)
        x_new = np.linspace(0, 2, n_points, endpoint=False)
        h = np.interp(x_new, x_parent, parent)
        h = _project_to_sum_and_bounds(h, target_sum)
    else:
        # uniform start (feasible because 0.5 * n_points = target_sum)
        h = np.full(n_points, 0.5, dtype=float)

    best_h = h.copy()
    best_score = _c5_score(best_h, dx)

    # ------------------- optimisation loop -------------------
    start_time = time.time()
    deadline = start_time + budget_s * 0.95   # safety margin
    iteration = 0
    sigma = 0.4                               # initial perturbation scale
    decay = 0.9995

    while time.time() < deadline:
        iteration += 1
        sigma *= decay

        # propose a new candidate
        proposal = best_h + rnd.normal(scale=sigma, size=n_points)
        proposal = _project_to_sum_and_bounds(proposal, target_sum)

        score = _c5_score(proposal, dx)

        if score < best_score:
            best_h = proposal
            best_score = score

        # occasional large jump to escape local minima
        if rnd.random() < 0.01:
            large = best_h + rnd.normal(scale=1.0, size=n_points)
            large = _project_to_sum_and_bounds(large, target_sum)
            large_score = _c5_score(large, dx)
            if large_score < best_score:
                best_h = large
                best_score = large_score

    # final evaluation (ensuring feasibility)
    best_h = _project_to_sum_and_bounds(best_h, target_sum)
    final_score = _c5_score(best_h, dx)

    return best_h, final_score, n_points
