import time
import math
import numpy as np

def _compute_c5(h):
    """Return C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h):
    """Project h onto the feasible set: 0≤h≤1 and sum(h)=n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        h[:] = target / n
        return h
    h = h * (target / s)
    # A few iterations of clipping + renormalizing to handle boundary drifts
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        h = h * (target / s)
    return h

def _perturb(h, sigma):
    """Add zero‑mean Gaussian noise (scaled by sigma) and repair."""
    h_new = h + np.random.randn(h.size) * sigma
    return _repair(h_new)

def _upscale(h, new_n):
    """Interpolate h to a higher resolution keeping periodicity."""
    old_n = h.shape[0]
    # points are on [0,2) with step 2/old_n
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return _repair(h_new)
