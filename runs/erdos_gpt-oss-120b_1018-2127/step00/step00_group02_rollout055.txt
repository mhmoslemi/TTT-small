import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return the discrete C5 value for step function h on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h), full mode length 2n-1
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def hill_climb(initial_h, time_limit, max_iter=50000):
    """Simple stochastic hill climbing keeping the sum constraint exact."""
    h = initial_h.astype(float).copy()
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    # step size schedule
    step = 0.2
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # gradually shrink step
        step *= 0.9999
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 2 * step  # signed change
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # enforce box constraints
        if not (0.0 <= h_new[i] <= 1.0 and 0.0 <= h_new[j] <= 1.0):
            continue
        # sum unchanged by construction, but numerical drift possible
        if abs(h_new.sum() - target_sum) > 1e-9:
            continue
        val = compute_c5(h_new)
        if val < best_val - 1e-8:
            best_val = val
            best_h = h_new.copy()
            h = h_new  # move to new point
    return best_h, best_val
