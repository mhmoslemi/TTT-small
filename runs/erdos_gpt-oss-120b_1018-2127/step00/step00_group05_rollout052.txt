import numpy as np
import time

# Helper: compute C5 for a given h
def compute_c5(h, dx):
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

# Helper: enforce 0≤h≤1 and integral constraint sum(h)=target
def repair_h(h, target, max_iter=20):
    h = np.asarray(h, dtype=np.float64)
    for _ in range(max_iter):
        # box constraints
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # scale to satisfy integral; avoid division by zero
        factor = target / s if s != 0 else 0.0
        h *= factor
    # final box clipping
    np.clip(h, 0.0, 1.0, out=h)
    # tiny correction to hit the target more closely
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly (will stay inside bounds because diff is tiny)
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points

    # Initial feasible point
