import time
import math
import random
import numpy as np

# Helper to compute the Erdős C5 value for a discrete step function h
def _compute_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    # full convolution of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


# Bring a vector back into the feasible set:
#   0 ≤ h_i ≤ 1  and  sum(h) = n/2  (so that ∫ h = 1)
def _repair(h: np.ndarray) -> np.ndarray:
    n = h.shape[0]
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    target = n / 2.0
    if total == 0.0:
        h.fill(0.5)
        total = h.sum()
    # Scale to hit the integral constraint
    h *= target / total
    # Small clipping may break the sum again; fix it once more
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if abs(total - target) > 1e-12:
        # A cheap uniform correction; it stays inside [0,1] because the deviation is tiny
        h += (target - total) / n
        h = np.clip(h, 0.0, 1.0)
        # final re‑scale if needed
        total = h.sum()
        if abs(total - target) > 1e-12:
            h *= target / total
    return h


# One simulated‑annealing run at a fixed resolution
def _anneal(h_start: np.ndarray, time_deadline: float) -> tuple:
    n = h_start.shape[0]
    best_h = _repair(h_start.copy())
    best_score = _compute_c5(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    # annealing schedule parameters
    T = 1.0 + math.sqrt(n)          # initial temperature
    T_end = 1e-4
    cooling = 0.995
    perturb_scale = 0.25           # max change per coordinate (relative to 1)

    # main loop runs until time runs out
    while time.time() < time_deadline:
        # propose a new point
        proposal = cur_h + (np.random.rand(n) - 0.5) * perturb_scale
        proposal = _repair(proposal)

        proposal_score = _compute_c5(proposal)

        # accept if better, otherwise with Metropolis probability
        if proposal_score < cur_score:
            cur_h, cur_score = proposal, proposal_score
            if proposal_score < best_score:
                best_h, best_score = proposal.copy(), proposal_score
        else:
            delta = proposal_score - cur_score
            if random.random() < math.exp(-delta / max(T, 1e-12)):
                cur_h, cur_score = proposal, proposal_score

        # cool down
        T = max(T * cooling, T_end)
        if T <= T_end and (time.time() + 0.01) >= time_deadline:
            break

    return best_h, best_score
