import numpy as np
import time
import math

def _project_box_sum(y: np.ndarray, target: float) -> np.ndarray:
    """Project y onto the feasible set {0≤x≤1, sum(x)=target}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_score(h: np.ndarray) -> float:
    """Compute C₅ = max_k ∫ h(x)(1‑h(x+k)) dx for a discretised step function."""
    n = h.size
    dx = 2.0 / n
    # correlation with zero‑padding, mode='full'
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points) where
    - h_values : list of floats, the step function values on a uniform grid of [0,2]
    - c5_bound : the achieved C₅ value (the lower the better)
    - n_points : number of discretisation points used
    """
    start_time = time.time()
    rnd = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # 1. Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        # fallback: uniform function with exact integral 1
        n_fallback = 70
        h0 = np.full(n_fallback, 0.5, dtype=np.float64)

    n = h0.size
    target_sum = n / 2.0                     # because ∫ h = 1 ⇒ Σ h * dx = 1 ⇒ Σ h = n/2
    h_curr = _project_box_sum(h0, target_sum)
    best_h = h_curr.copy()
    best_score = _c5_score(best_h)
    curr_score = best_score

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing style local search
    # ------------------------------------------------------------------
    max_iters = 10_000_000                     # generous upper bound; loop stops by time
    temp_start, temp_end = 0.12, 1e-5
    sigma_start = 0.25

    it = 0
    while True:
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # annealing schedule (linear in iteration count)
        progress = it / max_iters
        temp = temp_start * (1.0 - progress) + temp_end * progress
        sigma = sigma_start * (temp / temp_start)

        # propose a perturbation on a small random subset
        k = max(1, int(0.07 * n))                # ~7 % of coordinates each move
        idx = rnd.choice(n, size=k, replace=False)
        proposal = h_curr.copy()
        proposal[idx] += rnd.normal(scale=sigma, size=k)
        proposal = _project_box_sum(proposal, target_sum)

        new_score = _c5_score(proposal)

        # acceptance rule
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            # Metropolis criterion
            if rnd.rand() < math.exp(-(new_score - curr_score) / max(temp, 1e-12)):
                accept = True

        if accept:
            h_curr = proposal
            curr_score = new_score

        if new_score < best_score:
            best_score = new_score
            best_h = proposal.copy()

        # occasional random restart to avoid deep local minima
        if it % (n * 50) == 0 and it > 0:
            # perturb the best solution a bit and re‑project
            rnd_restart = best_h + rnd.normal(scale=0.1, size=n)
            h_curr = _project_box_sum(rnd_restart, target_sum)
            curr_score = _c5_score(h_curr)

        it += 1

    # ------------------------------------------------------------------
    # 3. Return result
    # ------------------------------------------------------------------
    return best_h.tolist(), best_score, n
