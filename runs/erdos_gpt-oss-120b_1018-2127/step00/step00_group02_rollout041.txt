import time
import numpy as np
import scipy.optimize

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraints(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # water‑filling style correction
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        nfree = int(free.sum())
        if nfree == 0:
            # distribute uniformly, then re‑clip
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - h.sum()
            continue
        adjust = diff / nfree
        h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
    # tiny residual fix
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h

def initial_solution():
    """Return the starting vector, using the provided global if present."""
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        n = 60
        h0 = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)], dtype=float)
    target_sum = h0.size / 2.0
    return project_to_constraints(h0, target_sum)

def upscale(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid."""
    old_n = h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # start from the parent solution and optionally increase resolution
    base_h = initial_solution()
    n_target = max(45, base_h.size)          # ensure at least the parent size
    if n_target < 90:                        # try a finer grid if time permits
        n_target = 90
        base_h = upscale(base_h, n_target)

    n = base_h.size
    target_sum = n / 2.0

    # initial best
    best_h = project_to_constraints(base_h.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    # parameters for stochastic search
    iter_count = 0
    no_improve = 0
    sigma0 = 0.15

    while time.time() - start_time < budget_s * 0.95:
        # adaptive step size
        sigma = sigma0 * (0.995 ** (iter_count // 200))
        cand = best_h + rng.normal(scale=sigma, size=n)
        cand = project_to_constraints(cand, target_sum)
        cand_c5 = compute_c5(cand)

        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5
            no_improve = 0
        else:
            no_improve += 1

        # occasional local refinement with SLSQP
        if iter_count % 800 == 0 and iter_count > 0:
            def obj(x):
                return compute_c5(x)

            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)
            bounds = [(0.0, 1.0)] * n
            res = scipy.optimize.minimize(
                obj,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 150, 'ftol': 1e-9, 'disp': False}
            )
            if res.success:
                refined_c5 = compute_c5(res.x)
                if refined_c5 < best_c5:
                    best_h, best_c5 = res.x, refined_c5
                    no_improve = 0

        # random restart if stagnation persists
        if no_improve > 3000:
            rnd = rng.uniform(0.0, 1.0, size=n)
            best_h = project_to_constraints(rnd, target_sum)
            best_c5 = compute_c5(best_h)
            no_improve = 0

        iter_count += 1

    # final safety projection (ensures exact sum)
    best_h = project_to_constraints(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
