import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

# Helper: project onto [low,high] box while enforcing a given sum.
def _project_box_sum(v, low, high, target, tol=1e-9, max_iter=100):
    v = np.clip(v, low, high)
    for _ in range(max_iter):
        diff = target - v.sum()
        if abs(diff) <= tol:
            break
        free = (v > low + tol) & (v < high - tol)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds, distribute remaining diff proportionally
            v += diff / len(v)
            v = np.clip(v, low, high)
            break
        inc = diff / n_free
        v[free] += inc
        v = np.clip(v, low, high)
    # final correction to fix any tiny residual error
    residual = target - v.sum()
    if abs(residual) > tol:
        # adjust the first free variable if any, else spread uniformly
        idx = np.where(v > low + tol)[0]
        if idx.size:
            v[idx[0]] += residual
            v = np.clip(v, low, high)
    return v

def _c5_of_h(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def _make_objective(dx, target_sum):
    def obj(x):
        h = _project_box_sum(x, 0.0, 1.0, target_sum)
        return _c5_of_h(h, dx)
    return obj

def _make_constraints(target_sum):
    return ({'type': 'eq',
             'fun': lambda x: np.sum(x) - target_sum},)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine grid size

    dx = 2.0 / n

    # Objective for DE (works on the raw vector, projection inside)
    obj = _make_objective(dx, target_sum)

    bounds = [(0.0, 1.0)] * n

    # DE parameters respecting time budget
    max_iter = 200
    popsize = 10

    # Callback to respect budget
    def de_callback(xk, convergence):
        if time.time() - start_time > budget_s * 0.70:
            return True  # stop DE early
        return False

    # Run Differential Evolution
    result_de = differential_evolution(
        obj,
        bounds,
        strategy='best1bin',
        maxiter=max_iter,
        popsize=popsize,
        tol=1e-7,
        updating='deferred',
        polish=False,
        disp=False,
        seed=rng,
        callback=de_callback
    )

    h_best = _project_box_sum(result_de.x, 0.0, 1.0, target_sum)
