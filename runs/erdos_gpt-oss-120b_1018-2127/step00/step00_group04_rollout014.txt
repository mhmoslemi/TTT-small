import time
import numpy as np

def _c5_of_h(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx using discrete correlation."""
    # np.correlate with mode='full' gives sums over overlapping indices.
    # Multiply by dx to obtain integral approximation.
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize the
    Erdős overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points) :
        h_values – list of function values on the grid,
        c5_bound – achieved upper bound on C5,
        n_points – grid size used.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------
    # Grid size and initial vector
    # ------------------------------------------------------------
    try:
        # pre‑imported by the harness, may be absent
        n_points = len(initial_h_values)          # noqa: F821
        h = np.array(initial_h_values, dtype=float)  # noqa: F821
    except Exception:
        # fallback: modest resolution under 1 k points
        n_points = 200
        h = np.full(n_points, 0.5, dtype=float)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0      # because ∫ h = 1  ⇒  sum(h)*dx = 1

    # ------------------------------------------------------------
    # Project initial h onto feasibility region (bounds + integral)
    # ------------------------------------------------------------
    def _project(vec):
        # clip to [0,1]
        np.clip(vec, 0.0, 1.0, out=vec)
        # adjust to meet exact sum (simple affine scaling, then tiny correction)
        cur = vec.sum()
        if cur != 0:
            vec *= target_sum / cur
        np.clip(vec, 0.0, 1.0, out=vec)
        cur = vec.sum()
        diff = target_sum - cur
        if abs(diff) > 1e-12:
            vec += diff / n_points
            np.clip(vec, 0.0, 1.0, out=vec)
        return vec

    h = _project(h)

    best_h = h.copy()
    best_c5 = _c5_of_h(best_h, dx)

    # ------------------------------------------------------------
    # Simple stochastic descent with occasional global perturbation
    # ------------------------------------------------------------
    it = 0
    while time.time() - start_time < 0.95 * budget_s:
        # --- elementary transfer move (preserves sum) -----------------
        i, j = np.random.randint(0, n_points, size=2)
        if i == j:
            continue

        # propose a small transfer delta from i to j
        delta = (np.random.rand() - 0.5) * 0.08   # range ≈ [-0.04,0.04]
        # respect box constraints
        delta = max(delta, -best_h[i])               # cannot go below 0 at i
        delta = min(delta, 1.0 - best_h[j])          # cannot exceed 1 at j
        if abs(delta) < 1e-9:
            continue

        h_candidate = best_h.copy()
        h_candidate[i] -= delta
        h_candidate[j] += delta

        c5 = _c5_of_h(h_candidate, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate

        # --- periodic broader exploration ----------------------------
        it += 1
        if it % 2000 == 0:
            # random Gaussian shake of the whole vector
            shake = np.random.randn(n_points) * 0.02
            h_tmp = best_h + shake
            h_tmp = _project(h_tmp)
            c5 = _c5_of_h(h_tmp, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_tmp

    # ------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------
    return (best_h.tolist(), float(best_c5), n_points)
