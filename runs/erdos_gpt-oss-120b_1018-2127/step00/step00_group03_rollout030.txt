import time, random, math
import numpy as np

def _project_feasible(h, target_sum):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        h[:] = target_sum / h.size
        return h
    # Scale initially
    h *= target_sum / s
    h = np.clip(h, 0.0, 1.0)
    # Iteratively fix any remaining deficit/excess
    for _ in range(5):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if not free.any():
            # all at bounds – distribute uniformly (may violate bounds a bit)
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            break
        adjust = diff / free.sum()
        h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_score(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx using discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / h.size
    return corr.max() * dx
