def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    diff = target - current
    if abs(diff) < 1e-12:
        return h
    # distribute diff equally among non-bound elements
    free = np.where((h > 1e-12) & (h < 1-1e-12))[0]
    if free.size == 0:
        # all at bounds, fallback: adjust all uniformly then clip
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # final adjust by scaling to meet sum
        current = h.sum()
        if current != 0:
            h *= target / current
        h = np.clip(h, 0.0, 1.0)
        return h
    adjust = diff / free.size
    h[free] += adjust
    # after adjustment, some may go out of bounds, need iterate
    return project(h, target)  # recursion limited

import time
import numpy as np

def _objective(h):
    """Maximum overlap integral for a given step vector."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)


def _project(h, target_sum, eps=1e-12):
    """Project onto [0,1] box and enforce exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < eps:
        return h

    # indices that are not at a bound
    free = np.where((h > eps) & (h < 1.0 - eps))[0]
    if free.size == 0:
        # fallback: shift uniformly then clip + rescale
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if cur > eps:
            h *= target_sum / cur
        return np.clip(h, 0.0, 1.0)

    adjust = diff / free.size
    h[free] += adjust
    # recurse a few times – will converge quickly
    return _project(h, target_sum, eps)


def _interpolate(prev_h, n_new):
    """Linear interpolation of a previous solution to a finer grid."""
    n_old = prev_h.shape[0]
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, prev_h)


def _improve(initial_h, time_budget):
    """Local random-walk improvement under a time budget."""
    n = initial_h.shape[0]
    target = n / 2.0
    best_h = _project(initial_h.copy(), target)
    best_val = _objective(best_h)

    start = time.time()
    while time.time() - start < time_budget:
        # scale shrinks as we approach a good value
        scale = 0.04 * (best_val + 1.0) / np.sqrt(n)
        delta = np.random.randn(n) * scale
        cand = _project(best_h + delta, target)
        val = _objective(cand)
        if val + 1e-12 < best_val:      # accept only improvements
            best_h, best_val = cand, val
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step vector, its C5 bound, and its length."""
    np.random.seed(seed)
    start_total = time.time()

    # ------------------------------------------------------------------
    # Determine starting solution.

    try:
        init = initial_h_values.astype(float)

    # ------------------------------------------------------------------
    # Multi‑grid schedule (coarse → fine).
    # ------------------------------------------------------------------
    grid_sizes = [n0, n0 * 2, n0 * 4]          # 95 → 190 → 380
    best_overall_h = None
    best_overall_val = np.inf
    best_n = None

    # allocate time roughly proportional to problem size
    remaining = budget_s
    for i, n in enumerate(grid_sizes):
        now = time.time()
        elapsed = now - start_total
        if elapsed >= budget_s:
            break
        remaining = budget_s - elapsed
        # give a fraction of remaining time (larger grids get less)
        frac = 0.6 if i == 0 else (0.3 if i == 1 else 0.1)
        grid_time = min(remaining * frac, remaining)

        if i == 0:

    if best_overall_h is None:
        best_overall_h = init
        best_overall_val = _objective(init)
        best_n = n0

    return best_overall_h, float(best_overall_val), int(best_n)
