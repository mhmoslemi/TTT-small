import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine resolution
    n_points = kwargs.get('n_points')
    if n_points is None:
        try:
            n0 = len(initial_h_values)  # type: ignore[name-defined]
        except Exception:
            n0 = 70
        n_points = n0
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ------------------------------------------------------------------ #
    # Repair a vector to satisfy 0≤h_i≤1 and sum(h)=target_sum
    def repair(vec):
        h = np.clip(vec, 0.0, 1.0)
        for _ in range(12):
            s = h.sum()
            if abs(s - target_sum) < 1e-9:
                break
            if s == 0.0:
                h = np.full_like(h, target_sum / len(h))
                break
            factor = target_sum / s
            h = np.clip(h * factor, 0.0, 1.0)
        # Tiny residual fix
        diff = target_sum - h.sum()
        if abs(diff) > 1e-10:
            idx = rng.integers(0, len(h))
            h[idx] = min(max(h[idx] + diff, 0.0), 1.0)
            # one more scaling pass if needed
            for _ in range(5):
                s = h.sum()
                if abs(s - target_sum) < 1e-9:
                    break
                h = np.clip(h * (target_sum / s), 0.0, 1.0)
        return h

    # ------------------------------------------------------------------ #
    # Objective: max correlation * dx
    def evaluate(vec):
        h = repair(vec)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx, h

    # ------------------------------------------------------------------ #
    # Build initial population
    population = []

    # use the provided construction if available
    try:
        base = np.asarray(initial_h_values, dtype=float)  # type: ignore[name-defined]
        if base.size != n_points:
            # interpolate to desired size
            x_old = np.linspace(0.0, 2.0, base.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
            base = np.interp(x_new, x_old, base)
        population.append(repair(base))
    except Exception:
        pass

    while len(population) < 15:
        population.append(repair(rng.random(n_points)))

    # ------------------------------------------------------------------ #
    # Evaluate initial pop and keep the best
    best_h = None
    best_score = math.inf
    for h in population:
        sc, _ = evaluate(h)
        if sc < best_score:
            best_score, best_h = sc, h

    # ------------------------------------------------------------------ #
    t_start = time.time()
    sigma = 0.25          # initial mutation scale
    min_sigma = 0.008     # stop decreasing after this
    decay = 0.995         # geometric decay per iteration
    keep_size = len(population)

    while time.time() - t_start < budget_s - 5.0:
        sigma = max(min_sigma, sigma * decay)

        # generate offspring
        offspring = []
        for h in population:
            mutant = h + rng.normal(0.0, sigma, n_points)
            offspring.append(repair(mutant))
            offspring.append(h)                     # keep parent as candidate

        # evaluate all candidates
        scores = []
        for cand in offspring:
            sc, _ = evaluate(cand)
            scores.append(sc)
            if sc < best_score:
                best_score, best_h = sc, cand

        # select the top keep_size candidates for next generation
        idx = np.argsort(scores)[:keep_size]
        population = [offspring[i] for i in idx]

    return best_h, best_score, n_points
