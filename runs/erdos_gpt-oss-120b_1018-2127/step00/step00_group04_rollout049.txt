import time
import numpy as np

def compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def project_binary_sum(vec, target_sum):
    """Return a binary vector (0/1) of same length as vec with exactly target_sum ones.
    Keeps ones where vec is largest."""
    n = vec.size
    # argsort descending
    order = np.argsort(-vec)
    out = np.zeros(n, dtype=float)
    out[order[:target_sum]] = 1.0
    return out

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a binary step function h on [0,2] with integral 1
    (i.e. exactly half the samples are 1) to minimize the
    maximum overlap integral C5.

    Returns (h_values, c5_bound, n_points)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        # Ensure binary and correct sum
        target_sum = n // 2
        if not np.all(np.isin(init, [0.0, 1.0])) or int(init.sum()) != target_sum:
            # project onto feasible binary set preserving larger entries
            h = project_binary_sum(init, target_sum)
        else:
            h = init.copy()
    else:
        # fallback: use n = 100 (even) and random binary with half ones
        n = 100
        target_sum = n // 2
        h = np.zeros(n, dtype=float)
        h[:target_sum] = 1.0
        rng.shuffle(h)

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # Pre‑compute indices of zeros and ones for fast swaps
    ones_idx = np.flatnonzero(best_h == 1.0)
    zeros_idx = np.flatnonzero(best_h == 0.0)

    # ------------------------------------------------------------------
    # Simple stochastic local search (swap two positions)
    # ------------------------------------------------------------------
    # Parameters controlling the search intensity
    max_no_improve = 20000          # stop after many failed swaps
    no_improve_cnt = 0
    iteration = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.5:      # keep a small safety margin
            break

        # pick random swap (1 -> 0, 0 -> 1)
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        # perform swap
        best_h[i], best_h[j] = 0.0, 1.0

        # evaluate
        cur_c5 = compute_c5(best_h, dx)

        if cur_c5 < best_c5:
            # improvement found: update bookkeeping
            best_c5 = cur_c5
            # refresh index lists (only two positions changed)
            ones_idx = np.flatnonzero(best_h == 1.0)
            zeros_idx = np.flatnonzero(best_h == 0.0)
            no_improve_cnt = 0
        else:
            # revert swap
            best_h[i], best_h[j] = 1.0, 0.0
            no_improve_cnt += 1

        iteration += 1

        # occasional random restart to escape possible stagnation
        if no_improve_cnt >= max_no_improve:
            # generate a fresh random binary vector with same sum
            candidate = np.zeros(n, dtype=float)
            candidate[:target_sum] = 1.0
            rng.shuffle(candidate)

                best_c5 = cand_c5

            else:
                # keep current best but continue searching
                pass
            no_improve_cnt = 0

    # Return result
