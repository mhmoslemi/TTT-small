import time
import numpy as np
from scipy.optimize import minimize

# Helper: compute C5 bound for a given step function h
def compute_score(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# Helper: check feasibility (within tolerance)
def is_feasible(h):
    n = h.size
    if np.any(h < -1e-9) or np.any(h > 1 + 1e-9):
        return False
    if not np.isclose(h.sum(), n / 2.0, atol=1e-6):
        return False
    return True

# Transfer a solution from old_n points to new_n points via linear interpolation
def transfer_solution(old_h, new_n):
    old_n = old_h.size
    # original grid [0,2]
    old_grid = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_grid = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_grid, old_grid, old_h)
    # bring into bounds and adjust sum
    new_h = np.clip(new_h, 0.0, 1.0)
    target_sum = new_n / 2.0
    cur_sum = new_h.sum()
    if cur_sum == 0:
        # fallback to uniform half
        new_h[:] = 0.5
        return new_h
    scale = target_sum / cur_sum
    new_h *= scale
    # after scaling may violate bounds; simple projection loop
    for _ in range(5):
        over = new_h > 1.0
        under = new_h < 0.0
        if not (over.any() or under.any()):
            break
        # cap and redistribute excess/deficit uniformly among free variables
        excess = new_h[over] - 1.0
        deficit = -new_h[under]
        new_h[over] = 1.0
        new_h[under] = 0.0
        free = (~over) & (~under)
        total_free = free.sum()
        if total_free == 0:
            break
        delta = (excess.sum() - deficit.sum()) / total_free
        new_h[free] += delta
    # final clip
    new_h = np.clip(new_h, 0.0, 1.0)
    # final renormalize to exact sum (tiny adjustment on free entries)
    cur_sum = new_h.sum()
    if not np.isclose(cur_sum, target_sum, atol=1e-8):
        diff = target_sum - cur_sum
        free = (new_h > 0.0) & (new_h < 1.0)
        if free.any():
            new_h[free] += diff / free.sum()
            new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

# Generate a random feasible vector of given size
def random_feasible(n, rng):
    # Dirichlet ensures positivity; then scale to target sum
    alpha = np.ones(n)  # uniform Dirichlet
    vec = rng.dirichlet(alpha)
    vec *= (n / 2.0)  # target sum
    # clip any occasional >1 and renormalize
    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    target = n / 2.0
    if cur != 0:
        vec *= target / cur
    return vec

# Perform SLSQP optimization for a given starting point
def optimize_h(start_h, rng):
    n = start_h.size
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq',
             'fun': lambda x: x.sum() - n / 2.0},)

    # SLSQP may benefit from a small random perturbation to escape flat regions
    perturbed = start_h + 1e-4 * rng.standard_normal(n)
    perturbed = np.clip(perturbed, 0.0, 1.0)
    # re‑adjust sum exactly
    perturbed = transfer_solution(perturbed, n)

    res = minimize(compute_score,
                   perturbed,
                   method='SLSQP',
                   bounds=bounds,
                   constraints=cons,
                   options={'maxiter': 600, 'ftol': 1e-9, 'disp': False})
    return res

def run(seed=42, budget_s=1100, **kwargs):
