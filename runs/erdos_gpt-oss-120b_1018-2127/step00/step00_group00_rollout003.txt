import time
import numpy as np
from scipy.optimize import minimize

def compute_score(h, n):
    """Maximum overlap integral for step function h on [0,2] with n samples."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    return np.max(corr) * dx

def make_feasible(h, n, atol=1e-8):
    """Project h onto the feasible set: 0≤h≤1 and sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) <= atol:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds; distribute proportionally
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            delta = diff / cnt
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
    # final tiny correction by scaling (won't violate bounds appreciably)
    s = h.sum()
    if s != 0:
        h *= (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(old_h, new_n):
    """Linear interpolation of old_h (length m) to length new_n."""
    m = len(old_h)
    # sample positions at cell centers
    old_x = np.linspace(0, 2, m, endpoint=False) + 1.0 / m
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0 / new_n
    new_h = np.interp(new_x, old_x, old_h)
    return make_feasible(new_h, new_n)

def optimize_h(start_h, n, seed):
    """Run SLSQP from start_h, respecting bounds and integral constraint."""
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - n / 2.0}
    opts = {"maxiter": 400, "ftol": 1e-8, "disp": False}
    rng = np.random.default_rng(seed)

    jitter = rng.uniform(-0.01, 0.01, size=n)
    x0 = make_feasible(start_h + jitter, n)
    res = minimize(compute_score, x0, args=(n,), method="SLSQP",
                   bounds=bounds, constraints=cons, options=opts)
    if res.success:

    return None, None
