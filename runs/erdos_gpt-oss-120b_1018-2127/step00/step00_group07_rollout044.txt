import time, math, random
import numpy as np

def _c5_of_h(h):
    """Return the C5 value for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _repair(h, target_sum):
    """Clip to [0,1] and renormalize so that sum equals target_sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        # avoid division by zero – fallback to uniform fill
        h[:] = target_sum / h.size
    else:
        h = h * (target_sum / s)
        # a second clipping may be needed; iterate once
        h = np.clip(h, 0.0, 1.0)
        # final renorm (tiny drift allowed)
        h = h * (target_sum / h.sum())
    return h

def _perturb(h, scale):
    """Random perturbation that preserves the sum in expectation."""
    n = h.shape[0]
    # add zero‑mean noise
    delta = (np.random.rand(n) - 0.5) * scale
    h_new = h + delta
    # repair to meet constraints
    h_new = _repair(h_new, n / 2.0)
    return h_new

def _anneal(start_h, budget, max_iter=20000):
    """Simple simulated‑annealing refinement within a time budget."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    best_h = start_h.copy()
    best_score = _c5_of_h(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    # temperature schedule
    T0 = 0.5 * best_score
    Tmin = 1e-6
    T = T0
    decay = 0.9995

    start = time.time()
    it = 0
    while it < max_iter and (time.time() - start) < budget:

    """Upscale by integer factor using nearest‑neighbour repeat."""
    return np.repeat(h, factor)
