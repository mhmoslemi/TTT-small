import numpy as np
import time
from scipy.optimize import minimize

def _evaluate_c5(h, dx):
    """Compute the C5 value for a given h vector."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _project_to_feasible(h, target_sum):
    """
    Project an arbitrary vector onto the box [0,1] and the affine
    constraint sum(h) = target_sum using a simple iterative clipping.
    """
    h = np.clip(h, 0.0, 1.0)
    # Adjust to meet the sum constraint by adding a constant offset,
    # then re‑clip; iterate a few times for stability.
    for _ in range(5):
        current_sum = np.sum(h)
        delta = (target_sum - current_sum) / len(h)
        h += delta
        h = np.clip(h, 0.0, 1.0)
    # Final correction to enforce the sum exactly (may violate bounds minimally)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute the remaining diff uniformly on non‑boundary entries
        mask = (h > 0.0) & (h < 1.0)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h
