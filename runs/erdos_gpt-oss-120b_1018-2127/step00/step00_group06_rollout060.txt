import numpy as np
import time
import random

def _objective(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and enforce exact sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) > 1e-12:
        h += (target_sum - cur) / h.size
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) > 1e-12:
            # move diff to a non‑boundary element if possible
            free = np.where((h > 0.0) & (h < 1.0))[0]
            if free.size == 0:
                idx = np.argmax(h)  # fallback
            else:
                idx = free[0]
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _interpolate(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a periodic vector onto a finer grid."""
    old_n = old_h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, old_h)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 to minimize
    C5 = max_k ∫ h(x)(1 – h(x+k)) dx.

    Returns
    -------
    h_values : np.ndarray
        Optimized samples of h.
    c5_bound : float
        Upper bound achieved for C5.
    n_points : int
        Number of sample points (length of h_values).
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # ------- initial construction -------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # provided by harness
    except NameError:
        # fallback: half ones, half zeros shuffled
        n0 = 100
        init_h = np.zeros(n0, dtype=float)
        init_h[: n0 // 2] = 1.0
        np.random.shuffle(init_h)

    # ensure feasibility
    n0 = init_h.size
    target0 = n0 / 2.0
    init_h = _repair(init_h, target0)

    best_h = init_h.copy()
    best_score = _objective(best_h)
    best_n = n0

    # grid progression (doubling up to ~800 points)
    grid_sizes = [n0, n0 * 2, n0 * 4]
    for n in grid_sizes:
        # stop if budget exhausted
        if time.time() - start_time > budget_s * 0.95:
            break

        # interpolate to current grid size
        if n != init_h.size:
            h = _interpolate(init_h, n)
        else:
            h = init_h.copy()

        # enforce constraints on the interpolated vector
        target = float(n) / 2.0
        h = _repair(h, target)

        # local best for this grid
        local_best_h = h.copy()
        local_best_score = _objective(local_best_h)

        # simulated‑annealing style search
        T = 1.0
        it = 0
        while time.time() - start_time < budget_s * 0.95:
            it += 1
            # perturbation magnitude scaling with grid size
            scale = 0.02 * (n ** -0.2)

            h_new = _repair(h_new, target)

                best_h = local_best_h.copy()

        # use the refined solution as seed for the next finer grid
        init_h = local_best_h.copy()

    # final feasibility check (just in case)
    best_h = _repair(best_h, float(best_n) / 2.0)

    return best_h, best_score, best_n
