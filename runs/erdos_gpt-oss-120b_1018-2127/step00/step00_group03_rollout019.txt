import numpy as np
import time
import math
import random

# Helper: project a vector onto the box [0,1] and the linear equality sum = target_sum
def _project_feasible(v, target_sum, max_iter=20):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = v.sum()
        diff = cur_sum - target_sum
        if abs(diff) < 1e-12:
            break
        # shift all entries uniformly, then re‑clip
        v -= diff / len(v)
        v = np.clip(v, 0.0, 1.0)
    return v

# Scoring: compute the Erdős C5 value for a discrete step function
def _score(h):
    # use the pre‑imported evaluator if it exists
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        # fallback: direct correlation
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx

# Build an initial feasible vector

        h0 = _project_feasible(h0, n / 2)
