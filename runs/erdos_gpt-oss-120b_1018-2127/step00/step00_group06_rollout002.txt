import time
import numpy as np

def evaluate_h(h):
    """Return C5 bound for sample vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def feasible_random(n, rng):
    """Create a random feasible h with values in [0,1] and sum = n/2."""
    # start with n/2 ones and rest zeros, then shuffle and jitter a bit
    ones = int(np.floor(n / 2))
    remainder = n / 2.0 - ones
    h = np.zeros(n, dtype=np.float64)
    h[:ones] = 1.0
    if remainder > 1e-12:
        h[ones] = remainder  # fractional piece
    rng.shuffle(h)
    # add small random noise while keeping feasibility
    for _ in range(3):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a tiny amount epsilon from j to i
        eps = min(0.02, h[j])
        eps = min(eps, 1.0 - h[i])
        delta = eps * rng.random()
        h[i] += delta
        h[j] -= delta
    return h

def perturb(h, rng, max_swaps=1):
    """Perform a small feasible perturbation keeping the sum exactly constant."""
    n = h.shape[0]
    h_new = h.copy()
    for _ in range(max_swaps):
        # pick i with room to increase, j with room to decrease
        inc_candidates = np.where(h_new < 1.0 - 1e-12)[0]
        dec_candidates = np.where(h_new > 1e-12)[0]
        if inc_candidates.size == 0 or dec_candidates.size == 0:
            break
        i = rng.choice(inc_candidates)
        j = rng.choice(dec_candidates)
        # maximum transferable amount
        max_transfer = min(1.0 - h_new[i], h_new[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random() * 0.5  # limit step size
        h_new[i] += delta
        h_new[j] -= delta
    return h_new
