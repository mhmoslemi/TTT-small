import time
import numpy as np

def _eval_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a binary step function h on [0,2] to minimise the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimised vector of values (0 or 1) sampled on a uniform grid.
    c5_bound : float
        The achieved upper bound for C5.
    n_points : int
        Number of discretisation points (length of h_values).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose an even number of points so that exactly half can be 1.
    # 48 points give a reasonable trade‑off between speed and resolution.
    n = 48
    half = n // 2
    dx = 2.0 / n

    # ----- initialise -------------------------------------------------------
    # Start from a random balanced binary vector.
    h_best = np.zeros(n, dtype=float)
    h_best[:half] = 1.0
    rng.shuffle(h_best)

    # If a parent construction is supplied and matches the size, reuse it.
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size == n:
            # Round to 0/1 while keeping the total sum correct.
            rounded = np.rint(init).astype(float)
            if rounded.sum() == half:
                h_best = rounded.copy()

    c5_best = _eval_c5(h_best)

    # ----- simulated annealing ---------------------------------------------
    # Parameters (tuned to stay well within the time budget).
    max_iter = 250_000
    temp_start = 1.0
    temp_end = 1e-3

    # Pre‑compute index masks for speed.
    ones_mask = lambda h: np.where(h == 1.0)[0]
    zeros_mask = lambda h: np.where(h == 0.0)[0]

    for it in range(max_iter):
        # Stop early if the budget is almost exhausted.
        if time.time() - start_time > budget_s * 0.95:
            break

        # Linear cooling schedule.
        t = temp_start * (1 - it / max_iter) + temp_end * (it / max_iter)

        # Choose a random 1‑position and a random 0‑position to swap.
        idx_one = ones_mask(h_best)
        idx_zero = zeros_mask(h_best)
        if idx_one.size == 0 or idx_zero.size == 0:
            continue
        i = rng.choice(idx_one)
        j = rng.choice(idx_zero)

        # Propose a swap.
        h_new = h_best.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0

        c5_new = _eval_c5(h_new)
        delta = c5_new - c5_best

        # Metropolis acceptance.
        if delta < 0.0 or rng.random() < np.exp(-delta / t):
            h_best = h_new
            c5_best = c5_new

        # Occasionally keep track of the current best (already stored).

    # Return the best found solution.
    return (h_best, c5_best, n)
