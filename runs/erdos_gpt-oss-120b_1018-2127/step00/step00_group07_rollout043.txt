import time
import numpy as np
from scipy.optimize import minimize

# Helper: compute the maximal overlap C5 for a discretized step function h
def compute_c5(h):
    n = h.size
    # Linear convolution via FFT with zero padding (length 2n)
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    conv = np.fft.irfft(H * G, N)[: 2 * n - 1]
    # Multiply by dx = 2 / n
    return conv.max() * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 80
        # Simple feasible start: first n/2 entries 1, rest 0
        h0 = np.concatenate([np.ones(n_points // 2), np.zeros(n_points - n_points // 2)])
    # Ensure exact integral constraint
    target_sum = n_points / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    current_sum = h0.sum()
    if abs(current_sum - target_sum) > 1e-12:
        # Scale uniformly then project to bounds while fixing sum
        h0 = h0 * (target_sum / current_sum)
        # Simple projection loop
        for _ in range(10):
            excess = h0.sum() - target_sum
            if abs(excess) < 1e-12:
                break
            # move excess equally from entries >0 to those <1
            up = h0 < 1.0
            down = h0 > 0.0
            if excess > 0:
                # need to decrease sum
                delta = excess / down.sum()
                h0[down] = np.maximum(0.0, h0[down] - delta)
            else:
                delta = -excess / up.sum()
                h0[up] = np.minimum(1.0, h0[up] + delta)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------
    # Simulated annealing / stochastic local search
    # ------------------------------------------------------------
    max_iter = 10 ** 7  # safety cap
    temperature = 0.1
    decay = 0.99995

    i = 0
    while time.time() - start_time < budget_s * 0.95 and i < max_iter:
        i += 1

        # propose a move: transfer delta between two indices
        idx_i, idx_j = rng.integers(0, n_points, size=2)
        if idx_i == idx_j:
            continue

        # maximum amount we can move without violating bounds
        max_minus = best_h[idx_i]            # can remove at most this
        max_plus = 1.0 - best_h[idx_j]       # can add at most this
        max_transfer = min(max_minus, max_plus)

        if max_transfer <= 0:
            continue

        # draw a transfer amount
        delta = rng.uniform(-max_transfer, max_transfer)
        # we interpret delta as amount taken from i and given to j
        # (positive delta → i loses, j gains)
        if delta < 0:
            # swap direction
            delta = -delta
            idx_i, idx_j = idx_j, idx_i

        # Apply move
        h_new = best_h.copy()
        h_new[idx_i] -= delta
        h_new[idx_j] += delta

        # compute new C5
        new_c5 = compute_c5(h_new)

        # acceptance test
        improve = new_c5 < best_c5 - 1e-12
        if improve or rng.random() < np.exp(-(new_c5 - best_c5) / temperature):
            best_h = h_new
            best_c5 = new_c5

        # temperature schedule
        temperature *= decay

        # occasional random shake to escape plateaus
        if i % 50000 == 0:
            # small random perturbation respecting sum
            perturb = rng.normal(scale=0.02, size=n_points)
            perturb -= perturb.mean() * (2.0 / n_points)  # adjust to keep sum approx
            h_trial = np.clip(best_h + perturb, 0.0, 1.0)
            # renormalize exactly to target sum
            diff = h_trial.sum() - target_sum
            if abs(diff) > 1e-9:
                # correct uniformly on interior points
                mask = (h_trial > 0.0) & (h_trial < 1.0)
                if mask.any():
                    correction = diff / mask.sum()
                    h_trial[mask] -= correction
                    h_trial = np.clip(h_trial, 0.0, 1.0)
            trial_c5 = compute_c5(h_trial)
            if trial_c5 < best_c5:
                best_h, best_c5 = h_trial, trial_c5

    # Final evaluation (in case evaluate_erdos_solution is preferred)
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), int(n_points)
