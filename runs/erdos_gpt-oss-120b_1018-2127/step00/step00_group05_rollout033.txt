import time
import numpy as np

def _project_onto_capsimplex(v, target_sum):
    """Project vector v onto the set {0<=x_i<=1, sum x_i = target_sum}."""
    # Simple iterative clipping and equal redistribution.
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        if abs(s - target_sum) < 1e-12:
            break
        # redistribute the deficit/excess uniformly
        v += (target_sum - s) / v.size
        v = np.clip(v, 0.0, 1.0)
    return v

def _evaluate(h, dx):
    """Return max overlap integral C5 for a candidate h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    np.random.seed(seed)
