import time
import numpy as np

# Helper to enforce feasibility: values in [0,1] and correct integral.
def _project_to_feasible(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(5):
        s = v.sum()
        if s == 0:
            v[:] = target_sum / v.size
            break
        scale = target_sum / s
        v *= scale
        v = np.clip(v, 0.0, 1.0)
    # final tiny adjustment if needed
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        free = (v > 0.0) & (v < 1.0)
        if free.any():
            v[free] += diff / free.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

# Objective: max overlap integral for a given discrete h.
def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    h_values – numpy array on the grid [0,2] (step function values)
    c5_bound – the achieved upper bound C_5
    n_points – number of discretisation points used
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine grid size
    if "n_points" in kwargs:
        n_points = int(kwargs["n_points"])
    else:
        # Use the length of the provided construction if available, else a modest default.
        try:
            n_points = len(initial_h_values)  # type: ignore[name-defined]
        except Exception:
            n_points = 80

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum(h)*dx = 1

    # Initialise h
    try:
        base = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        # Upsample / downsample to required size by simple linear interpolation
        if base.size != n_points:
            x_old = np.linspace(0, 2, base.size, endpoint=False)
            x_new = np.linspace(0, 2, n_points, endpoint=False)
            base = np.interp(x_new, x_old, base)
    except Exception:
        # Fallback: start from a uniform half‑filled vector
        base = np.concatenate([
            np.ones(n_points // 2),
            np.zeros(n_points - n_points // 2)
        ]).astype(float)

    h_current = _project_to_feasible(base.copy(), target_sum)
    best_h = h_current.copy()
    best_score = _c5_of(best_h)

    # Simple stochastic hill‑climb with decreasing step size
    step_initial = 0.3
    step_final = 0.01
    max_iters = 500000  # generous upper bound; loop will exit by time budget
    iter_count = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.5:
            break
        if iter_count >= max_iters:
            break
        # Linear schedule for step magnitude
        t = elapsed / budget_s
        step = step_initial * (1 - t) + step_final * t

        # Propose a new candidate by perturbing a random subset of coordinates
        cand = best_h.copy()
        mask = np.random.rand(n_points) < 0.15  # perturb ~15% of entries
        perturb = np.random.uniform(-step, step, size=n_points)
        cand[mask] += perturb[mask]

        cand = _project_to_feasible(cand, target_sum)
        score = _c5_of(cand)

        if score < best_score:
            best_h = cand
            best_score = score

        # Occasionally inject a random restart to escape stagnation
        if iter_count % 2000 == 0 and iter_count > 0:

            rand_vec = _project_to_feasible(rand_vec, target_sum)
            rand_score = _c5_of(rand_vec)
            if rand_score < best_score:
                best_h = rand_vec
                best_score = rand_score

        iter_count += 1

    # Final safeguard: ensure feasibility one last time
    best_h = _project_to_feasible(best_h, target_sum)
    best_score = _c5_of(best_h)

    return best_h, float(best_score), n_points
