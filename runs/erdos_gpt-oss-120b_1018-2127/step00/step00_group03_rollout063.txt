import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 overlap value for a step function sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def projected_move(h: np.ndarray, i: int, j: int, eps: float) -> tuple:
    """
    Propose a move that adds eps to h[i] and subtracts eps from h[j].
    Returns (new_h, valid) where valid is False if bounds are violated.
    """
    hi, hj = h[i], h[j]
    hi_new = hi + eps
    hj_new = hj - eps
    if hi_new < 0.0 or hi_new > 1.0 or hj_new < 0.0 or hj_new > 1.0:
        return h, False
    new_h = h.copy()
    new_h[i] = hi_new
    new_h[j] = hj_new
    return new_h, True
