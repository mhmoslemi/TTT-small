import time, math
import numpy as np

def _project_to_constraints(v, target_sum):
    """Project vector v onto [0,1] box with exact sum = target_sum."""
    v = np.clip(v, 0.0, 1.0)
    # water-filling style
    excess = target_sum - v.sum()
    # tolerance
    eps = 1e-12
    while abs(excess) > eps:
        free = (v > 0.0 + eps) & (v < 1.0 - eps)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust further, break
            break
        delta = excess / n_free
        # apply delta and re‑clip
        v[free] += delta
        np.clip(v, 0.0, 1.0, out=v)
        excess = target_sum - v.sum()
    return v

def _c5_score(h, dx):
    """Compute C5 = max_k ∫ h(x)(1−h(x+k))dx via discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ----- initial vector -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float32)
        n = h0.size
        target = n / 2.0          # because ∫h = 1  and dx = 2/n
        h_curr = _project_to_constraints(h0, target)
    else:
        n = 70                     # fallback size
        h_curr = np.zeros(n, dtype=np.float32)
        h_curr[: n // 2] = 1.0
        rng.shuffle(h_curr)

    dx = 2.0 / n
    target_sum = n / 2.0

    # ----- bookkeeping ----------------------------------------------------
    cur_score = _c5_score(h_curr, dx)
    best_h = h_curr.copy()
    best_score = cur_score

    start_time = time.time()
    # simple simulated‑annealing parameters
    max_time = budget_s * 0.95   # leave margin
    while time.time() - start_time < max_time:
        # propose moving ε from i to j (keeps sum invariant)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        dec_max = h_curr[i]           # how much we can take from i
        inc_max = 1.0 - h_curr[j]     # how much we can add to j
        max_eps = min(dec_max, inc_max)
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps

        h_new = h_curr.copy()
        h_new[i] -= eps
        h_new[j] += eps

        new_score = _c5_score(h_new, dx)

        # acceptance
        if new_score < cur_score:
            accept = True
        else:
            # temperature decreases linearly with elapsed fraction
            elapsed = time.time() - start_time
            frac = elapsed / max_time
            temp = 0.01 * (1.0 - frac) + 1e-4
            prob = math.exp(-(new_score - cur_score) / temp)
            accept = prob > rng.random()

        if accept:
            h_curr = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score

        # occasional short local refinement: small random perturbation and projection
        if rng.random() < 0.01:
            perturb = rng.normal(scale=0.02, size=n).astype(np.float32)
            h_try = np.clip(best_h + perturb, 0.0, 1.0)
            h_try = _project_to_constraints(h_try, target_sum)
            try_score = _c5_score(h_try, dx)
            if try_score < best_score:
                best_h = h_try
                best_score = try_score
                h_curr = h_try
                cur_score = try_score

    return (best_h, best_score, n)
