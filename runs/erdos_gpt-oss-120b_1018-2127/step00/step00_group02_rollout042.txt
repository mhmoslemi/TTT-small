import time
import numpy as np
from scipy.optimize import minimize

# evaluate_erdos_solution and initial_h_values are provided by the harness


def compute_c5(h):
    """Return the overlap constant C5 for a sampled step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def random_amplitude_solution(rng, n, max_amp=0.45, n_terms=3, tries=2000):
    """
    Generate candidate step functions of the form
        h(x) = 0.5 + sum_{k=1}^{n_terms} a_k * sin(k*pi*x)
    while respecting 0≤h≤1.  Return the best one found.
    """
    x = np.linspace(0, 2, n, endpoint=False)
    best_h = None
    best_c = np.inf
    for _ in range(tries):
        # draw amplitudes
        a = rng.uniform(-max_amp / n_terms, max_amp / n_terms, size=n_terms)
        h = 0.5 + sum(a[k] * np.sin((k + 1) * np.pi * x) for k in range(n_terms))
        if np.any(h < 0) or np.any(h > 1):
            continue
        # integral constraint is automatically satisfied (average of sines = 0)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def hill_climb(h0, rng, time_limit):
    """
    Simple stochastic hill‑climbing that conserves the total integral.
    Moves a small amount of mass from one index to another.
    """
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_c = compute_c5(h)
    start = time.time()
    it = 0
    # annealing schedule parameters
    max_step_initial = 0.05
    while time.time() - start < time_limit:
        it += 1
        # gradually shrink step size
        max_step = max_step_initial * (1.0 - (time.time() - start) / time_limit)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving mass from i to j
        delta = rng.uniform(0, max_step)
        delta = min(delta, h[i])                 # cannot go negative at i
        delta = min(delta, 1.0 - h[j])           # cannot exceed 1 at j
        if delta <= 0:
            continue
        h[i] -= delta
        h[j] += delta
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            # keep the move
        else:
            # revert with Metropolis acceptance
            prob = np.exp(-(c - best_c) / (0.001 + 0.5 * max_step))
            if rng.random() < prob:
                best_c = c  # accept worse move
            else:
                h[i] += delta
                h[j] -= delta
    return h, best_c
