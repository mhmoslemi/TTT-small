import time
import numpy as np

def compute_c5(h):
    """Return the discrete approximation of C5 for a step vector h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def repair_constraints(vec, target_sum):
    """Project vec onto the box [0,1] and enforce sum(vec)=target_sum (within tolerance)."""
    vec = np.clip(vec, 0.0, 1.0)
    n = vec.shape[0]
    # simple iterative water‑filling
    for _ in range(8):
        diff = target_sum - vec.sum()
        if abs(diff) < 1e-12:
            break
        # distribute uniformly, then re‑clip
        vec += diff / n
        vec = np.clip(vec, 0.0, 1.0)
    # final fine‑tune on non‑saturated entries
    diff = target_sum - vec.sum()
    if abs(diff) > 1e-10:
        free = np.where((vec > 0.0) & (vec < 1.0))[0]
        if free.size == 0:
            # fallback: adjust everywhere (will re‑clip)
            vec += diff / n
            vec = np.clip(vec, 0.0, 1.0)
        else:
            inc = diff / free.size
            vec[free] += inc
            vec = np.clip(vec, 0.0, 1.0)
    return vec


def random_walk(base, scale, rng):
    """Generate a perturbed candidate from base."""
    return base + rng.normal(scale=scale, size=base.shape)


def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    start_time = time.time()

    # schedule parameters (tuned for speed)
