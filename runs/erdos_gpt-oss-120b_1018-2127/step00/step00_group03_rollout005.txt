import time
import numpy as np

def _binary_from_continuous(x, target_sum):
    """Return a binary vector (0/1) of same length with exactly target_sum ones
    by taking the largest entries of x."""
    idx = np.argpartition(-x, target_sum - 1)[:target_sum]
    b = np.zeros_like(x, dtype=np.float64)
    b[idx] = 1.0
    return b

def _evaluate(h):
    """Wrapper for the pre‑imported evaluator."""
    try:
        # evaluate_erdos_solution is provided in the harness
        return evaluate_erdos_solution(h)
    except NameError:
        # Fallback: compute max correlation directly
        n = h.size
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        return np.max(corr)

def _initial_binary():
    """Produce an initial feasible binary vector."""
    # Use the supplied construction if present, otherwise fabricate one.
    if 'initial_h_values' in globals():
        x = np.asarray(initial_h_values, dtype=np.float64)
        n = x.size
    else:
        n = 70
        x = np.linspace(0, 1, n)  # dummy; will be overwritten by binarisation
    target = n // 2
    return _binary_from_continuous(x, target)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretised) to minimise the
    Erdős C5 overlap constant.

    Returns
    -------
    h_values : np.ndarray
        The best binary vector found (float values 0.0 or 1.0).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Length of the discretisation.
    """
    rnd = np.random.RandomState(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    h = _initial_binary()
    n = h.size
    target_ones = n // 2
    best_h = h.copy()
    best_score = _evaluate(best_h)

    # Pre‑compute index groups for fast swaps
    ones_idx = np.flatnonzero(best_h == 1.0)
    zeros_idx = np.flatnonzero(best_h == 0.0)

    # ------------------------------------------------------------------
    # Main stochastic hill‑climbing loop
    # ------------------------------------------------------------------
    no_improve = 0
    max_no_improve = 2000          # restart after many stagnant steps
    while time.time() - start_time < budget_s - 0.1:
        # pick a random 1 and a random 0 to swap
        i = rnd.choice(ones_idx)
        j = rnd.choice(zeros_idx)

        # propose new candidate by swapping
        cand = best_h.copy()
        cand[i], cand[j] = 0.0, 1.0

        score = _evaluate(cand)

        if score < best_score - 1e-12:          # strict improvement
            best_h = cand
            best_score = score
            # update index sets
            ones_idx = np.flatnonzero(best_h == 1.0)
            zeros_idx = np.flatnonzero(best_h == 0.0)
            no_improve = 0
        else:
            no_improve += 1

        # occasional random restart to escape plateaus
        if no_improve >= max_no_improve:
            # re‑initialise from a fresh random binary vector
            cand = np.zeros(n, dtype=np.float64)
            cand[rnd.choice(n, target_ones, replace=False)] = 1.0
            cand_score = _evaluate(cand)
            if cand_score < best_score:
                best_h = cand
                best_score = cand_score
                ones_idx = np.flatnonzero(best_h == 1.0)
                zeros_idx = np.flatnonzero(best_h == 0.0)
            no_improve = 0

    # Ensure the returned vector respects the exact sum constraint (numerical safety)
    current_sum = best_h.sum()
    if current_sum != target_ones:

            best_h[flip] = 1.0

            flip = rnd.choice(one_positions, -diff, replace=False)
            best_h[flip] = 0.0
        best_score = _evaluate(best_h)

    return best_h.astype(np.float64), float(best_score), int(n)
