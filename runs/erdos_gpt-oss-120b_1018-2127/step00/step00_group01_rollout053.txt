import time
import numpy as np
import scipy.optimize as opt
import cvxpy as cp

def _dx(n):
    return 2.0 / n

def _compute_c5(h):
    """Compute the overlap integral C5 for a discrete step function."""
    n = len(h)
    dx = _dx(n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_feasible(x, target_sum):
    """
    Orthogonal projection of x onto the box [0,1] with an exact
    sum constraint sum(v)=target_sum using cvxpy.
    """
    n = len(x)
    v = cp.Variable(n)
    prob = cp.Problem(
        cp.Minimize(cp.sum_squares(v - x)),
        [v >= 0, v <= 1, cp.sum(v) == target_sum],
    )
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if v.value is None:
        # fallback: simple clipping and renormalisation
        y = np.clip(x, 0.0, 1.0)
        s = y.sum()
        if s == 0:
            y[:] = target_sum / n
        else:
            y *= target_sum / s
        y = np.clip(y, 0.0, 1.0)
        return y
    return np.asarray(v.value).ravel()

def _local_refine(start, target_sum, time_limit):
    """SLSQP refinement enforcing the constraints exactly."""
    n = len(start)

    def obj(v):
        return _compute_c5(v)

    cons = ({
        "type": "eq",
        "fun": lambda v: np.sum(v) - target_sum,
    },)

    bounds = [(0.0, 1.0)] * n
    opts = {"maxiter": 300, "ftol": 1e-12}
    res = opt.minimize(
        obj,
        start,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    # If SLSQP fails to satisfy constraints, fall back to projection
    if not res.success or abs(np.sum(res.x) - target_sum) > 1e-6:
        return _project_feasible(start, target_sum)
    return np.clip(res.x, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimizes
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a modest grid size (balance quality vs speed)
    n_points = 200
    target_sum = n_points / 2.0  # because ∑h * dx = 1  →  ∑h = n/2

    # Initialise from the provided construction if available, otherwise random.
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # Interpolate to the target grid size
        x_old = np.linspace(0.0, 2.0, len(init), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_old, init)
        h0 = _project_feasible(h0, target_sum)
    else:
        h0 = rng.random(n_points)
        h0 = _project_feasible(h0, target_sum)

    # ---------- Differential Evolution (global search) ----------
    bounds = [(0.0, 1.0)] * n_points
    penalty_factor = 1e6

    def de_obj(v):
        # penalty for sum deviation
        penalty = penalty_factor * abs(v.sum() - target_sum)
        return _compute_c5(v) + penalty

    # Callback to respect time budget
    def de_callback(xk, convergence):
        if time.time() - start_time > budget_s * 0.65:
