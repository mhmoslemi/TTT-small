import time
import numpy as np
from scipy import optimize as opt

def _compute_c5(h: np.ndarray) -> float:
    """Return the overlap value C5 for a given step function h."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _random_feasible(N: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^N with sum N/2 (approximately)."""
    target = N / 2.0
    v = rng.random(N)          # uniform in [0,1]
    for _ in range(12):        # simple projection loop
        s = v.sum()
        diff = target - s
        v += diff / N
        np.clip(v, 0.0, 1.0, out=v)
    # final tiny correction by uniform shift if still off
    s = v.sum()
    if abs(s - target) > 1e-8:
        v += (target - s) / N
        np.clip(v, 0.0, 1.0, out=v)
    return v


def _optimize_start(start_vec: np.ndarray, best):
    """Run a constrained local optimisation from start_vec."""
    N = start_vec.size
    target = N / 2.0

    # equality constraint: sum == target
    cons = ({
        "type": "eq",
        "fun": lambda x: np.sum(x) - target,
    },)

    bounds = [(0.0, 1.0)] * N

    res = opt.minimize(
        _compute_c5,
        start_vec,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 250, "ftol": 1e-9, "disp": False},
    )
    if not res.success:
        return

    h = np.clip(res.x, 0.0, 1.0)

    # tiny post‑projection to enforce the sum exactly
    cur = h.sum()
    if abs(cur - target) > 1e-8:
        for _ in range(5):
            h += (target - cur) / N
            np.clip(h, 0.0, 1.0, out=h)
            cur = h.sum()
            if abs(cur - target) <= 1e-9:
                break

    c5 = _compute_c5(h)
    if c5 < best["c5"]:
        best["c5"] = c5
        best["h"] = h.copy()


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points) with a step function on [0,2]
    that (approximately) minimizes the Erdős overlap constant C₅.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Acquire an initial construction if the harness supplied one.
    try:
        init = np.asarray(initial_h_values, dtype=float)
        N = init.size
    except NameError:
        N = 45
        init = np.full(N, 0.5)  # sum = N/2

    # Container for the best solution found so far.
    best = {"c5": np.inf, "h": None}

    # First optimisation from the provided (or default) start.
    _optimize_start(init, best)

    # Budget‑aware random restarts.
    while time.time() - start_time < 0.95 * budget_s:
        rand_start = _random_feasible(N, rng)
        _optimize_start(rand_start, best)

    # Fallback in the unlikely event that nothing succeeded.
    if best["h"] is None:
        best["h"] = init.copy()
        best["c5"] = _compute_c5(best["h"])

    return (best["h"], float(best["c5"]), N)
