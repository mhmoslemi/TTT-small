import time
import numpy as np
import random

def _max_overlap_count(h: np.ndarray) -> int:
    """Return the maximum overlap count = max_k sum h[i]*(1-h[i+k])"""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    return int(corr.max())

def _evaluate(h: np.ndarray) -> float:
    """Integral bound C5 for a given binary sample vector."""
    n = h.size
    return _max_overlap_count(h) * (2.0 / n)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a binary step function h on [0,2] (represented by samples)
    that minimizes the Erdős C5 bound.

    Returns:
        h_values (list of float) – the best binary sample vector found,
        c5_bound (float)       – the achieved C5 bound,
        n_points (int)         – number of samples used.
    """
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------
    # Choose sample size (even) – larger gives finer resolution.
    # 400 points is a good trade‑off between quality and speed.
    # ------------------------------------------------------------
    n = 400
    half_ones = n // 2

    # ------------------------------------------------------------
    # Initialise h using the provided parent construction if present.
    # ------------------------------------------------------------
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        n0 = parent.size
        repeats = n // n0
        remainder = n % n0
        h = np.repeat(parent, repeats)
        if remainder:
            h = np.concatenate([h, parent[:remainder]])
        # Binarise (values >0.5 become 1, else 0)
        h = (h > 0.5).astype(int)
        # Adjust to exact half_ones if needed
        cur = int(h.sum())
        if cur > half_ones:
            # turn some ones to zero
            idx = np.flatnonzero(h)[: cur - half_ones]
            h[idx] = 0
        elif cur < half_ones:
            idx = np.flatnonzero(1 - h)[: half_ones - cur]
            h[idx] = 1
    else:
        # Random balanced binary vector
        h = np.zeros(n, dtype=int)
        h[:half_ones] = 1
        np.random.shuffle(h)

    best_h = h.copy()
    best_score = _evaluate(best_h)

    # ------------------------------------------------------------
    # Local search: random swaps of a 1 with a 0.
    # ------------------------------------------------------------
    # Parameters controlling the search
    max_no_improve = 5000          # restart after this many rejected swaps
    restart_interval = 20000      # forced random restart interval
    iteration = 0
    no_improve = 0
    last_restart = 0

    # Pre‑compute index lists
    ones_idx = set(np.flatnonzero(h))
    zeros_idx = set(np.flatnonzero(1 - h))

    while time.time() - start_time < budget_s * 0.95:
        iteration += 1

        # Random pick a 1 and a 0 to swap
        i_one = random.choice(tuple(ones_idx))
        i_zero = random.choice(tuple(zeros_idx))

        # Perform tentative swap
        h[i_one], h[i_zero] = 0, 1

        # Evaluate
        cur_score = _evaluate(h)

        if cur_score <= best_score:
            # Accept swap
            ones_idx.discard(i_one)
            zeros_idx.discard(i_zero)
            ones_idx.add(i_zero)
            zeros_idx.add(i_one)

            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
                # Early exit if we beat target bound
                if best_score <= 0.3808:
                    break
            no_improve = 0
        else:
            # Reject – revert swap
            h[i_one], h[i_zero] = 1, 0
            no_improve += 1

        # Periodic random restart to escape stagnation
        if no_improve >= max_no_improve or iteration - last_restart >= restart_interval:
            # Keep the best found so far as starting point
            h = best_h.copy()
            ones_idx = set(np.flatnonzero(h))
            zeros_idx = set(np.flatnonzero(1 - h))
            no_improve = 0
            last_restart = iteration

    # ------------------------------------------------------------
    # Return results
    # ------------------------------------------------------------
    return (best_h.astype(float).tolist(), float(best_score), n)
