import time
import numpy as np

def _make_feasible(h, target_sum, max_iter=10):
    """Clip to [0,1] and adjust to hit target_sum by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        diff = target_sum - cur
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            # all at bounds, distribute equally (should rarely happen)
            free = np.ones_like(h, dtype=bool)
        delta = diff / free.sum()
        # limit delta to avoid leaving bounds
        delta = np.where(delta > 0,
                         np.minimum(delta, 1.0 - h),
                         np.maximum(delta, -h))
        h[free] += delta[free]
    # final clip for safety
    return np.clip(h, 0.0, 1.0)

def _random_feasible(n, target_sum, rng):
    """Generate a random vector in [0,1]^n with sum == target_sum."""
    h = rng.random(n)
    h = _make_feasible(h * (target_sum / h.sum()), target_sum)
    return h

def _objective(h, dx):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx via correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _perturb(h, eps, rng):
    """Transfer a small amount between two coordinates while keeping sum."""
    n = h.size
    a, b = rng.choice(n, 2, replace=False)
    ha, hb = h[a], h[b]
    # permissible delta so that both stay in [0,1]
    max_pos = min(eps, 1.0 - ha, hb)
    max_neg = -min(eps, ha, 1.0 - hb)
    delta = rng.uniform(max_neg, max_pos)
    h_new = h.copy()
    h_new[a] = ha + delta
    h_new[b] = hb - delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # decide number of points
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        # ensure feasibility
        target = n / 2.0
        h_best = _make_feasible(init, target)
    else:
        n = 100  # fallback size
        target = n / 2.0
        h_best = _random_feasible(n, target, rng)

    dx = 2.0 / n
    best_score = _objective(h_best, dx)

    # parameters
    max_iters = 200000
    eps = 0.05          # typical perturbation magnitude
    prob_restart = 0.02

    for it in range(max_iters):
        # time guard
