def _repair(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and rescale to hit the target sum (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        # distribute uniformly
        return np.full_like(h, target_sum / h.size)
    # linear scaling

    # final clip (tiny floating errors)
    return np.clip(h, 0.0, 1.0)


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Compute the C5 bound for a step vector h."""
    # correlation of h with 1-h (full mode)
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())


def _random_neighbor(h: np.ndarray, step_std: float, rng: np.random.Generator) -> np.ndarray:

    perturb = rng.normal(loc=0.0, scale=step_std, size=h.shape)
    h_new = h + perturb
    return h_new
