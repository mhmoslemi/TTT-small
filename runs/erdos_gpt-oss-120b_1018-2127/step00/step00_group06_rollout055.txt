import time
import math
import numpy as np

def _objective(h):
    """Compute C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_onto_capped_simplex(v, target, upper=1.0, max_iter=60):
    """
    Euclidean projection of v onto the set {x | 0<=x_i<=upper, sum x_i = target}.
    Uses a bisection on the Lagrange multiplier.
    """
    lo, hi = -1e5, 1e5
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, upper)

def _random_walk(h, scale, rng):
    """Add isotropic Gaussian noise scaled by `scale`."""
    return h + scale * rng.standard_normal(h.shape)

def _interpolate(h, new_n):
    """Linear interpolation of a periodic vector h from its size to new_n."""
    old_n = h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with integral 1 to minimize
    C5 = max_k ∫ h(x) (1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    end_time = start_time + budget_s
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise first grid
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = h0.shape[0]
        if n0 % 2:
            # force even length for exact sum requirement
            h0 = np.append(h0, h0[-1])
            n0 += 1
    else:
        # simple symmetric start: half ones, half zeros
        n0 = 100
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 // 2)])

    # enforce constraints exactly for the starting vector
    target_sum = n0 / 2.0
    h0 = _project_onto_capped_simplex(h0, target_sum)

    best_h = h0.copy()
    best_score = _objective(best_h)

    # ------------------------------------------------------------------
    # 2. Grid progression
    # ------------------------------------------------------------------
    grid_sizes = [n0, n0 * 2, n0 * 4, n0 * 8]
    # cap the maximum size to keep runtime reasonable
    max_size = 500
    grid_sizes = [min(g, max_size) for g in grid_sizes]
    grid_sizes = sorted(set(grid_sizes))

    # allocate a modest fraction of the budget to each grid (more to coarse)
    time_per_grid = {}
    remaining = budget_s
    weight_sum = sum(1.0 / (i + 1) for i in range(len(grid_sizes)))
    for i, n in enumerate(grid_sizes):
        w = 1.0 / (i + 1)
        alloc = budget_s * (w / weight_sum)
        time_per_grid[n] = min(alloc, remaining)
        remaining -= time_per_grid[n]

    for n in grid_sizes:
        if time.time() >= end_time:
            break

        # interpolate best solution so far to current resolution
        h_cur = _interpolate(best_h, n)
        target_sum = n / 2.0
        h_cur = _project_onto_capped_simplex(h_cur, target_sum)

                best_score = new_score

            # temperature schedule

    n_best = best_h.shape[0]
    best_h = _project_onto_capped_simplex(best_h, n_best / 2.0)
