import numpy as np, time, math, random

def compute_c5(h):
    """Return max overlap integral approximation for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx

def project_bounds_sum(h, target_sum):
    """Clip to [0,1] and enforce exact sum via water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    delta = target_sum - cur
    for _ in range(20):
        if abs(delta) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        nfree = free.sum()
        if nfree == 0:
            h += delta / len(h)
            h = np.clip(h, 0.0, 1.0)
            delta = target_sum - h.sum()
            break
        add = delta / nfree
        h[free] += add
        h = np.clip(h, 0.0, 1.0)
        delta = target_sum - h.sum()
    # tiny residual correction
    if abs(h.sum() - target_sum) > 1e-10:
        h += (target_sum - h.sum()) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def random_walk(h, sigma):
    """Add Gaussian noise with scale sigma."""
    return h + np.random.normal(scale=sigma, size=h.shape)

def interpolate_h(h_old, n_new):
    """Linear interpolation of a periodic vector onto a finer grid."""
    n_old = h_old.shape[0]
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def optimize_h(h_start, time_limit):
    """Simple simulated‑annealing style refinement."""
    n = h_start.shape[0]
    target_sum = n / 2.0
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    T = 1.0            # temperature
    sigma = 0.02       # noise magnitude
    it = 0
    while time.time() < time_limit:
        it += 1
        # geometric cooling
        T = max(1e-4, T * 0.9995)
        sigma = max(0.001, sigma * 0.9995)
        h_cand = random_walk(h_cur, sigma)
        h_cand = project_bounds_sum(h_cand, target_sum)
        cand_score = compute_c5(h_cand)
        # deterministic improvement
        if cand_score < best_score:
            best_score = cand_score
            best_h = h_cand.copy()
        # metropolis acceptance
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / T):
            h_cur = h_cand
            cur_score = cand_score
        # safety break if too many iterations without improvement
        if it % 5000 == 0 and best_score < 0.381:
            # good enough, stop early
            break
    return best_h, best_score
