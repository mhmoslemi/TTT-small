import time
import numpy as np

def _compute_c5(h):
    """Return C5 value for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_bounds_and_sum(h, target_sum):
    """
    Clip to [0,1] and then adjust to satisfy the exact sum constraint.
    Simple iterative redistribution on non‑boundary entries.
    """
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # redistribute diff over free variables (those not at a bound)
    free = np.where((h > 0.0) & (h < 1.0))[0]
    # if no free variables, fall back to a tiny random perturbation
    if free.size == 0:
        free = np.arange(h.size)
    while abs(diff) > 1e-12:
        # amount to add per free index
        delta = diff / free.size
        # limit delta so we stay inside bounds
        for i in free:
            max_inc = 1.0 - h[i]
            max_dec = -h[i]
            add = np.clip(delta, max_dec, max_inc)
            h[i] += add
            diff -= add
        # recompute free set
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
    return h
