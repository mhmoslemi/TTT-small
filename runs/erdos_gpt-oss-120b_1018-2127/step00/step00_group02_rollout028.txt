import time
import numpy as np
import random

def _evaluate(h):
    """Return C5 for a given step function array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _make_feasible(h, target_sum):
    """Scale, clip and adjust h so that 0≤h≤1 and sum(h)=target_sum."""
    cur = h.sum()
    if cur != 0:
        h = h * (target_sum / cur)
    np.clip(h, 0.0, 1.0, out=h)
    # iterative redistribution of remaining error
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        interior = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if interior.size == 0:
            break
        delta = diff / interior.size
        h[interior] += delta
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] that lowers the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # -------------------------------------------------------------
    # parameters
    n_points = 120                     # modest size → fast evaluation
    target_sum = n_points / 2.0        # integral constraint  ∫ h = 1
    # -------------------------------------------------------------

    # ---------- initial construction ----------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        m = init.size
        # interpolate to our discretisation
        x_src = np.linspace(0.0, 2.0, m, endpoint=False)
        x_tgt = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_tgt, x_src, init)
    else:
        h = np.full(n_points, 0.5, dtype=float)

    h = _make_feasible(h, target_sum)
    best_h = h.copy()
    best_val = _evaluate(best_h)

    start_time = time.time()

    # --------- annealing schedule ----------
    T_start = 0.12
    T_end = 5e-4
    max_iters = 250000

    for it in range(max_iters):
        if time.time() - start_time > budget_s * 0.95:
            break

        # temperature decay (exponential)
        t = T_start * (T_end / T_start) ** (it / max_iters)

        # propose a feasible move: transfer eps from j to i
        i, j = np.random.choice(n_points, 2, replace=False)
        eps = np.random.uniform(-t, t)

        if 0.0 <= h[i] + eps <= 1.0 and 0.0 <= h[j] - eps <= 1.0:
            h_new = h.copy()
            h_new[i] += eps
            h_new[j] -= eps

            val = _evaluate(h_new)
            delta = val - best_val

            # Metropolis acceptance
            if val < best_val or np.random.rand() < np.exp(-delta / max(t, 1e-12)):
                h = h_new
                if val < best_val:
                    best_val = val
                    best_h = h_new.copy()

    # final feasibility check (should already hold)
    best_h = _make_feasible(best_h, target_sum)

    return (best_h, best_val, n_points)
