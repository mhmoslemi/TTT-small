import time
import numpy as np

def objective(h, dx):
    """Maximum overlap integral for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def repair_constraints(h, target=None):
    """Project h onto the feasible box [0,1] with the exact integral constraint."""
    n = h.shape[0]
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # Simple iterative projection onto the hyperplane sum(h)=target
    for _ in range(5):
        cur = h.sum()
        diff = target - cur
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

    # One final redistribution on the interior points if needed
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            adjust = (target - cur) / free.sum()
            h[free] += adjust
            h = np.clip(h, 0.0, 1.0)
    return h

def random_walk(h, scale):
    """Add Gaussian perturbation with given scale."""
    return h + np.random.randn(h.shape[0]) * scale
