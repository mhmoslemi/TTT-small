import time

import numpy as np

# Helper: compute C5 from a discrete h using the specification
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

# Helper: enforce 0<=h<=1 and exact sum constraint (sum = n/2)
def project_to_feasible(h, target_sum):
    # clip to bounds first
    h = np.clip(h, 0.0, 1.0)
    # iterative redistribution for the sum constraint
    eps = 1e-12
    max_iter = 20
    for _ in range(max_iter):
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) <= eps:
            break
        free = np.where((h > eps) & (h < 1.0 - eps))[0]
        if free.size == 0:
            # all at bounds, cannot adjust further; distribute to all with tiny slack
            free = np.arange(h.size)
        if diff > 0:
            # room to increase
            room = np.min(1.0 - h[free])
            inc = min(diff / free.size, room)
            h[free] += inc
        else:
            # need to decrease
            room = np.min(h[free])
            dec = min(-diff / free.size, room)
            h[free] -= dec
    # final clip (numeric safety)
    h = np.clip(h, 0.0, 1.0)
    return h

# Helper: generate an initial feasible vector of length n
