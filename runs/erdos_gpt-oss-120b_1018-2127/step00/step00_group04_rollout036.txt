import time
import numpy as np

def compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def propose_move(h, max_step):
    """Return a new h after transferring a random delta between two indices,
    preserving the total sum exactly and staying within [0,1]."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    # feasible delta range so both stay in [0,1]
    lo = max(-max_step, -hi, hj - 1.0)
    hi_delta = min(max_step, 1.0 - hi, 1.0 - hj)
    if lo > hi_delta:
        return None
    delta = np.random.uniform(lo, hi_delta)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine number of sampling points.

        h = np.array(initial_h_values, dtype=float)
    except Exception:
        n = 200                     # moderate resolution – fast enough for many iterations

    # Enforce exact integral constraint sum(h) = n/2
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h *= target_sum / cur_sum

    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, dx)

    # Annealing schedule parameters
    max_step = 0.2                # initial amplitude of transfers
    temp = 0.05                   # initial acceptance temperature
    iteration = 0
    # Main optimization loop
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        # Propose a move preserving sum
        new_h = propose_move(best_h, max_step)
        if new_h is None:
            continue

        c_val = compute_c5(new_h, dx)

        # Accept if better, or with Metropolis probability
        if c_val < best_c or np.random.rand() < np.exp((best_c - c_val) / max(temp, 1e-12)):
            best_h = new_h
            best_c = c_val

        # Schedule adjustments every few hundred iterations
        if iteration % 500 == 0:
            # cool down temperature and step size
            temp *= 0. ninety # reduce temperature
            max_step *= 0. ninety # shrink move magnitude
            # ensure they don't become too tiny
            temp = max(temp, 1e-5)
            max_step = max(max_step, 1e-4)

    # Final safety projection onto constraints (clip then renormalize sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # adjust uniformly while respecting bounds
        diff = target_sum - cur_sum
