import time
import numpy as np

def _evaluate(h):
    """Return max overlap integral C5 for discretized h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _random_transfer(h, step_scale):
    """Propose a new h by transferring a small amount between two coordinates."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # limits so that both entries stay in [0,1]
    max_up = min(1.0 - h[i], h[j])          # how much can be moved from j to i
    max_down = min(h[i], 1.0 - h[j])        # how much can be moved from i to j
    if max_up + max_down == 0.0:
        return None, None, None
    # choose delta within allowed interval, scaled by step_scale
    delta = (np.random.rand() * (max_up + max_down) - max_down) * step_scale
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h, i, j

def _optimize(initial_h, budget_s, seed):
    """Hill‑climbing / simulated‑annealing optimisation within given time."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s * 0.95   # keep a safety margin
    h = initial_h.copy()
    n = h.size
    best_h = h.copy()
    best_val = _evaluate(h)

    # annealing parameters
    temp = 0.02
    step_scale = 0.5

    it = 0
    while time.time() < deadline:
        cand, _, _ = _random_transfer(h, step_scale)
        if cand is None:
            continue
        cand_val = _evaluate(cand)

        # accept if improvement or with Metropolis probability
        if cand_val < best_val:
            best_val = cand_val
            best_h = cand.copy()
            h = cand
        else:
            if np.random.rand() < np.exp(-(cand_val - best_val) / max(temp, 1e-12)):
                h = cand

        # schedule updates
        it += 1
        if it % 500 == 0:
            temp *= 0.95
            step_scale *= 0.97
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) after optimisation."""
    # Determine starting point
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fall back: uniform half‑density (integral 1)
        n0 = 200
        h0 = np.full(n0, 0.5)

    # ensure feasibility (sum = n/2 and bounds)
    n0 = h0.size
    target_sum = n0 / 2.0
    current_sum = h0.sum()
    if not np.isclose(current_sum, target_sum):
        # simple scaling to fix the sum while staying in [0,1]
        h0 = np.clip(h0, 0.0, 1.0)
        excess = h0.sum() - target_sum
        # distribute excess uniformly
        if excess > 0:
            while excess > 1e-12:
                idx = np.where(h0 > 0)[0]
                if idx.size == 0:
                    break
                delta = min(excess / idx.size, 0.01)
                h0[idx] -= delta
                excess -= delta * idx.size
        else:
            while excess < -1e-12:
                idx = np.where(h0 < 1)[0]
                if idx.size == 0:
                    break
                delta = min(-excess / idx.size, 0.01)
                h0[idx] += delta
                excess += delta * idx.size

    # Phase 1: optimise at original resolution
    remaining = budget_s
    t0 = time.time()
    best_h, best_val = _optimize(h0, remaining, seed)
    elapsed = time.time() - t0
    remaining -= elapsed

    # Phase 2: optionally upscale to double resolution and refine
    if remaining > 5.0:
        # double resolution by simple repetition and scaling
        h_up = np.repeat(best_h, 2) * 0.5
        best_h2, best_val2 = _optimize(h_up, remaining, seed + 1)
        if best_val2 < best_val:
            best_h, best_val = best_h2, best_val2

    return best_h, best_val, best_h.size
