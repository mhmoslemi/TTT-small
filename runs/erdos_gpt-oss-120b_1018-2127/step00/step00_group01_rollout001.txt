import time
from scipy.optimize import minimize, differential_evolution

def _c5_value(h):
    """Compute C5 for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _slsqp_optimize(h0, target_sum, time_limit):
    """Run a quick SLSQP optimization respecting bounds and equality."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)

    # SLSQP may take many iterations; enforce a time guard via callback
    start = time.time()
    def callback(xk, *args, **kwargs):
        return (time.time() - start) > time_limit

    res = minimize(
        fun=_c5_value,
        x0=h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False},
        callback=callback,
    )
    # Project onto equality if numerical drift occurred
    x = res.x
    s = np.sum(x)
    if not np.isclose(s, target_sum, atol=1e-8):
        x = x * (target_sum / s)
        x = np.clip(x, 0.0, 1.0)
    return x, res.fun

def _de_optimize(h0, target_sum, time_limit):
