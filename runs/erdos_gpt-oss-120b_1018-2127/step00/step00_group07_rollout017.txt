import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return the max overlap integral for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def random_step(n, rng):
    """Generate a random feasible binary step function of length n."""
    ones = n // 2
    h = np.zeros(n, dtype=float)
    h[:ones] = 1.0
    rng.shuffle(h)
    return h


def hill_climb(initial_h, budget_s, rng):
    """Simple hill‑climbing / simulated‑annealing optimizer."""
    h = initial_h.copy()
    n = h.size
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()
    it = 0
    while True:
        now = time.time()
        if now - start > budget_s * 0.95:
            break

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        cap_i = h[i]                     # max removable from i
        cap_j = 1.0 - h[j]               # max addable to j
        max_delta = min(cap_i, cap_j)
        if max_delta <= 1e-12:
            continue

        # move a random fraction (up to half) of the possible mass
        delta = max_delta * rng.random() * 0.5
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        c5 = compute_c5(new_h)

        # accept if improvement
        if c5 < best_c5 - 1e-12:
            h = new_h
            best_h = new_h
            best_c5 = c5
        else:
            # simulated‑annealing acceptance
            elapsed = now - start
            temperature = max(0.001, 1.0 - elapsed / budget_s)
            prob = np.exp(-(c5 - best_c5) / temperature)
            if rng.random() < prob:
                h = new_h

        it += 1

        # occasional random restart to escape deep local minima
        if it % 5000 == 0:
            h = random_step(n, rng)

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns:
        h_values : numpy.ndarray of shape (n_points,)
        c5_bound : float, the achieved upper bound on C5
        n_points : int, number of discretisation points
    """
    rng = np.random.default_rng(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.size
        # Ensure feasibility (tiny numerical drift possible)
        h0 = np.clip(h0, 0.0, 1.0)
        excess = h0.sum() - n / 2.0
        if abs(excess) > 1e-9:
            # simple correction by scaling the deviation uniformly
            h0 -= excess / n
            h0 = np.clip(h0, 0.0, 1.0)
    else:
        n = 80
        h0 = random_step(n, rng)

    # Run the optimizer within the time budget
    best_h, best_c5 = hill_climb(h0, budget_s, rng)

    # Final safety projection: enforce constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    total = best_h.sum()
    if abs(total - n / 2.0) > 1e-9:
        # redistribute the tiny error uniformly
        best_h += (n / 2.0 - total) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, float(best_c5), n
