import numpy as np
import time
import math

def _compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _feasible_projection(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto [0,1] with the given sum (simple iterative clipping)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(5):
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    # final tiny adjustment if needed
    cur = v.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # distribute remaining diff uniformly to non‑boundary entries
        free = (v > 0.0) & (v < 1.0)
        if free.any():
            v[free] += diff / free.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _local_search(h0: np.ndarray, time_budget: float, start_time: float):
    """Simple pair‑exchange stochastic descent respecting constraints."""
    n = h0.shape[0]
    best_h = h0.copy()
    best_val = _compute_c5(best_h)
    max_step = 0.07  # magnitude of mass moved between two sites
    end_time = start_time + time_budget

    while time.time() < end_time:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * max_step  # can be +/-
        new_i = best_h[i] - delta
        new_j = best_h[j] + delta
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue
        cand = best_h.copy()
        cand[i] = new_i
        cand[j] = new_j
        val = _compute_c5(cand)
        if val < best_val:
            best_h = cand
            best_val = val
    return best_h, best_val

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step vector, its C5 bound and its length."""
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_val = math.inf
    best_n = None

    # Resolutions to explore (45 is the size of the provided construction)
    resolutions = [45, 60, 90, 120]

    # How many resolutions remain when we start a new one
    remaining = len(resolutions)

    for n in resolutions:
        # stop if overall budget exhausted
        if time.time() - start > budget_s * 0.98:
            break

        # allocate a share of the remaining time to this resolution
        elapsed = time.time() - start
        time_left = budget_s - elapsed
        sub_budget = time_left / remaining
        remaining -= 1

        # --- initialise a feasible vector of length n -----------------
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, copy=True)
        elif n == 45 and 'initial_h_values' in globals():
            # down/up‑sample the provided 45‑point vector to the desired size
            x_src = np.linspace(0.0, 2.0, len(initial_h_values), endpoint=False)
            x_tgt = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_tgt, x_src, initial_h_values)
        else:
            # uniform feasible start
            h0 = np.full(n, 0.5)

        # enforce exact constraints
        target_sum = n / 2.0
        h0 = _feasible_projection(h0, target_sum)

        # --- local stochastic optimisation ----------------------------
        h_opt, val_opt = _local_search(h0, sub_budget, start)

        if val_opt < best_val:
            best_h, best_val, best_n = h_opt, val_opt, n

    # fallback if nothing succeeded
    if best_h is None:
        n = 45
        best_h = np.full(n, 0.5)
        best_val = _compute_c5(best_h)
        best_n = n

    # final safety projection (should already be feasible)
    best_h = _feasible_projection(best_h, best_n / 2.0)

    return best_h, best_val, best_n
