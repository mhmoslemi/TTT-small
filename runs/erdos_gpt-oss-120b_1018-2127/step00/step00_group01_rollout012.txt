import numpy as np
import time
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def evaluate_c5(h):
    """Return the C5 value for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def project_bounds(h):
    """Clip into [0,1]."""
    return np.clip(h, 0.0, 1.0)


def project_sum(h, target_sum, max_iter=20, tol=1e-8):
    """
    Adjust h (already within [0,1]) so that its sum equals target_sum.
    Simple iterative redistribution among non‑boundary entries.
    """
    for _ in range(max_iter):
        current = h.sum()
        diff = target_sum - current
        if abs(diff) < tol:
            break
        # indices that are not at a bound
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # all at bounds – distribute uniformly, then re‑clip
            h += diff / h.size
            h = project_bounds(h)
            continue
        delta = diff / free.size
        h[free] += delta
        h = project_bounds(h)
    return h


def make_feasible(h):
    """Project h onto the feasible set."""
    h = project_bounds(h)
    n = h.shape[0]
    target = n / 2.0           # because integral = 1, dx = 2/n ⇒ sum = n/2
    h = project_sum(h, target)
    return h


def initial_vector(n):
    """Construct an initial feasible vector of length n."""
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        if base.shape[0] == n:
            return make_feasible(base.copy())
        # interpolate if needed
        x_old = np.linspace(0, 2, base.shape[0], endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        interp = np.interp(x_new, x_old, base)
        return make_feasible(interp)
    # fallback: uniform 0.5 (already feasible)
    return np.full(n, 0.5, dtype=float)


def upscale(h, new_n):
    """Increase resolution of h to length new_n using linear interpolation."""
    old_n = h.shape[0]
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new)


def local_optimize(h, max_iter=200):
    """
    Refine h with a short SLSQP run. The objective is the C5 value.
    """
    n = h.shape[0]
    target_sum = n / 2.0

    def obj(v):
        return evaluate_c5(v)

    cons = ({'type': 'eq',
             'fun': lambda v: v.sum() - target_sum},)

    bounds = [(0.0, 1.0)] * n
    # Use a few iterations to keep it cheap
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter,
                                              'ftol': 1e-9,
                                              'disp': False})
    if not res.success:
        # fall back to the input vector (already feasible)
        return h
    return make_feasible(res.x)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points).

    """
