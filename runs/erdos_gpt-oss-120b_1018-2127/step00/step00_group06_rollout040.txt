import time
import numpy as np
import cvxpy as cp

def compute_c5(h):
    """Return the discrete C5 value for a step function h on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_constraints(v, target):
    """Euclidean projection of v onto {0<=x<=1, sum(x)=target} via a small QP."""
    n = v.shape[0]
    x = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(x - v))
    cons = [x >= 0, x <= 1, cp.sum(x) == target]
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
        if x.value is None:
            raise cp.SolverError
        return np.asarray(x.value).astype(float)
    except Exception:
        # Fallback: simple water-filling
        x = np.clip(v, 0.0, 1.0)
        cur = x.sum()
        if abs(cur - target) < 1e-12:
            return x
        # iterative rescaling with clipping
        for _ in range(20):
            if cur == 0:
                break
            scale = target / cur
            x = np.clip(x * scale, 0.0, 1.0)
            cur = x.sum()
        # final adjustment by adding/subtracting tiny amounts
        diff = target - cur
        if abs(diff) > 1e-9:
            free = np.where((x > 0) & (x < 1))[0]
            if free.size:
                add = diff / free.size
                x[free] += add
                x = np.clip(x, 0.0, 1.0)
        return x

def random_feasible(n, rng):
    """Generate a random feasible h (sum = n/2, 0<=h<=1)."""
    target = n / 2.0
    # start with uniform Dirichlet then clip
    v = rng.rand(n)
    v = project_to_constraints(v, target)
    return v

def upscale(h, new_n):
    """Linearly interpolate h (defined on 0..2) to a new grid with new_n points."""
    old_n = len(h)
    xs_old = np.linspace(0, 2, old_n, endpoint=False) + 1.0/old_n
    xs_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    return np.interp(xs_new, xs_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return (h_best, c5_best, n_points) where h_best is a ndarray of step values.
    """
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # parameters
    grid_sequence = [95, 190, 380, 760]          # increasingly fine grids
    max_iters_per_grid = {95: 3000, 190: 2000, 380: 1500, 760: 1000}
    step_initial = 0.2
    step_final = 0.005

    best_h = None
    best_score = np.inf
    best_n = None

    if 'initial_h_values' in globals():

            best_score = compute_c5(best_h)
    # If no initial, create a random feasible one on the coarsest grid
    if best_h is None:
        best_n = grid_sequence[0]
        best_h = random_feasible(best_n, rng)
        best_score = compute_c5(best_h)

    # Main multi-resolution loop
    for n in grid_sequence:
        # respect remaining budget
        if time.time() - start_time > budget_s * 0.95:

        if n != best_n:
            best_h = upscale(best_h, n)
            best_n = n

            best_score = compute_c5(best_h)
