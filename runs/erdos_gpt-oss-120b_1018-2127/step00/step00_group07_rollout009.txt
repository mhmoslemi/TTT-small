import time
import random
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # Load or create initial h
    # ----------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        # fallback: 80 points random but respecting the integral constraint
        n = 80
        h0 = np.random.rand(n)
        h0 = h0 / h0.sum() * (n / 2.0)
    n = h0.shape[0]
    dx = 2.0 / n

    # ----------------------------------------------------------------------
    # Helper: ensure values in [0,1] and exact sum = n/2
    # ----------------------------------------------------------------------
    def project_sum(arr):
        target = n / 2.0
        # clip first
        arr = np.clip(arr, 0.0, 1.0)
        total = arr.sum()
        if total == 0:
            # put mass uniformly
            arr[:] = target / n
            return arr
        # scale to hit target, then re‑clip
        arr *= target / total
        arr = np.clip(arr, 0.0, 1.0)
        # fix any remaining deficit / excess
        deficit = target - arr.sum()
        if abs(deficit) > 1e-12:
            # distribute deficit to indices not at a bound
            free = np.where((arr > 0.0) & (arr < 1.0))[0]
            if free.size == 0:
                # all at bounds, just adjust first element (will stay in bounds)
                arr[0] = np.clip(arr[0] + deficit, 0.0, 1.0)
            else:
                arr[np.random.choice(free)] += deficit
        return arr

    h = project_sum(h0.copy())

    # ----------------------------------------------------------------------
    # Compute C5 for a discrete h
    # ----------------------------------------------------------------------
    def compute_c5(arr):
        corr = np.correlate(arr, 1.0 - arr, mode='full')
        return np.max(corr) * dx

    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    # ----------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop
    # ----------------------------------------------------------------------
    start_time = time.time()
    end_time = start_time + budget_s - 5.0  # small safety margin

    t0 = 0.5          # initial temperature
    tf = 1e-4         # final temperature

    # Main iteration
    while time.time() < end_time:
        i, j = random.sample(range(n), 2)
        # decide direction of mass transfer
        if random.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        eps_max = min(h[src], 1.0 - h[dst])
        if eps_max <= 1e-12:
            continue

        # limited step size (up to half of the feasible amount)
        eps = random.random() * eps_max * 0.5
        h[src] -= eps
        h[dst] += eps

        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5

        # linear temperature schedule
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 5.0)
        temp = t0 * (1.0 - frac) + tf * frac

        # accept if improvement or by Metropolis criterion
        if delta < 0.0 or random.random() < np.exp(-delta / temp):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert move
            h[src] += eps
            h[dst] -= eps

        # occasional restart if stagnation (optional, cheap)
        if random.random() < 1e-4:
            # reset to a random feasible point
            h = np.random.rand(n)
            h = project_sum(h)
            current_c5 = compute_c5(h)

    # Final projection and evaluation
    best_h = project_sum(best_h)
    best_c5 = compute_c5(best_h)

    return (best_h, float(best_c5), n)
