import time
import numpy as np

def _project_to_constraints(h, target_sum, max_iter=10):
    """Clip h to [0,1] and adjust to meet the total sum target."""
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        if np.isclose(cur_sum, target_sum, atol=1e-12):
            break
        # distribute the difference uniformly over all entries
        diff = target_sum - cur_sum
        h += diff / h.size
    # final clip to guarantee bounds
    return np.clip(h, 0.0, 1.0)

def _c5_value(h, dx):
    """Compute C5 = max_k ∫ h(x) (1-h(x+k)) dx via discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] (discretized) that minimizes the
    overlap integral C5. Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        # fallback: 80 points, half ones half zeros
        n = 80
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
    dx = 2.0 / n
    target_sum = n / 2.0  # because ∑h * dx = 1  → ∑h = n/2

    # Ensure initial feasibility
    h = _project_to_constraints(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_value(best_h, dx)

    # Simple simulated‑annealing / hill‑climbing loop
    # Parameters controlling exploration
    max_iter = 200000
    sigma = 0.2               # initial perturbation scale
    sigma_decay = 0.9995      # decay factor per iteration
    min_sigma = 1e-4

    for it in range(max_iter):
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break

        # Propose a new candidate
        proposal = best_h + sigma * np.random.randn(n)
        proposal = _project_to_constraints(proposal, target_sum)

        c5_prop = _c5_value(proposal, dx)

        # Accept if better, else occasional Metropolis acceptance
        if c5_prop < best_c5:
            best_h, best_c5 = proposal, c5_prop
        else:
            # probability to accept worse move, decreasing with sigma
            accept_prob = np.exp(-(c5_prop - best_c5) / max(sigma, 1e-6))
            if np.random.rand() < accept_prob:
                best_h, best_c5 = proposal, c5_prop

        # Decay perturbation magnitude
        sigma = max(min_sigma, sigma * sigma_decay)

        # Early exit if target achieved
        if best_c5 < 0.3808:
            break

    # Final safety projection (should already be satisfied)
    best_h = _project_to_constraints(best_h, target_sum)
    best_c5 = _c5_value(best_h, dx)

    return best_h, best_c5, n
