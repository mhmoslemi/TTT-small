import numpy as np
import time
import math
import random

def _compute_c5(h, dx):
    """Maximum overlap integral for a given step function h."""
    # correlation of h with (1-h); mode='full' gives all shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _random_delta(h_i, h_j, max_step=0.05):
    """Generate a feasible delta to modify h_i up and h_j down while staying in [0,1]."""
    # feasible range limited by bounds
    lo = max(-max_step, -h_i)          # cannot go below 0
    hi = min(max_step, 1.0 - h_i)      # cannot exceed 1
    # also respect the opposite move on j
    lo = max(lo, h_j - 1.0)            # j will decrease, must stay >=0
    hi = min(hi, h_j)                  # j will decrease, must stay <=1
    if lo >= hi:
        return None
    return random.uniform(lo, hi)
