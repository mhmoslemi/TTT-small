import time
import numpy as np

def _c5_score(h):
    """Return the overlap integral C5 for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_to_feasible(h, target_sum):
    """
    Simple iterative projection onto the box [0,1] and the affine sum constraint.
    """
    for _ in range(5):
        np.clip(h, 0.0, 1.0, out=h)
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        h += diff / h.size
    np.clip(h, 0.0, 1.0, out=h)  # final clip
    # a last correction if the sum is still off due to clipping
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # distribute the residual uniformly to non‑saturated entries
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimises C5."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # initialise
    # ----------------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=float)  # pre‑imported if present
    except NameError:
        # fallback: simple binary construction (exactly half ones)
        n_default = 70
        h0 = np.zeros(n_default, dtype=float)
        h0[: n_default // 2] = 1.0
    n = h0.shape[0]
    target_sum = n / 2.0                     # because ∫_0^2 h = 1 and dx = 2/n
    h_current = _project_to_feasible(h0.copy(), target_sum)
    best_h = h_current.copy()
    best_score = _c5_score(best_h)

    # ----------------------------------------------------------------------
    # optimisation loop (simple stochastic hill‑climbing with occasional smoothing)
    # ----------------------------------------------------------------------
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        h_new = h_current.copy()

        # random move type
        if rng.random() < 0.7:
            # ---- mass transfer between two positions (preserves sum) ----
            i, j = rng.choice(n, size=2, replace=False)
            # maximum transferable mass while staying in [0,1]
            max_transfer = min(h_new[i], 1.0 - h_new[j])
            if max_transfer > 1e-12:
                delta = rng.random() * max_transfer * 0.5
                h_new[i] -= delta
                h_new[j] += delta
        else:
            # ---- smoothing move (low‑pass filter) ----
            kernel = np.array([0.25, 0.5, 0.25], dtype=float)
            h_smooth = np.convolve(h_new, kernel, mode="same")
            h_new = _project_to_feasible(h_smooth, target_sum)

        # evaluate
        new_score = _c5_score(h_new)

        # acceptance rule (simulated annealing style)
        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score

        # temperature schedule (linear cooling)
        temperature = max(0.001, 1.0 - (time.time() - start_time) / (budget_s * 0.9))

    best_h = _project_to_feasible(best_h, target_sum)
