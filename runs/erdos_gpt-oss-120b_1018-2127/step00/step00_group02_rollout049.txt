import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Discrete C5 value for step function h on [0,2] with uniform grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def penalized_objective(h, target_sum, penalty):
    """C5 plus quadratic penalty for violating the integral constraint."""
    c5 = compute_c5(h)
    diff = h.sum() - target_sum
    return c5 + penalty * diff * diff

def repair_solution(h, target_sum):
    """Force h into [0,1] and adjust its sum to target_sum (approximate)."""
    h = np.clip(h, 0.0, 1.0)
    # simple iterative redistribution of the residual sum error
    for _ in range(5):
        current = h.sum()
        if abs(current - target_sum) < 1e-12:
            break
        delta = (target_sum - current) / h.size
        h += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction by scaling interior values
    interior = (h > 0.0) & (h < 1.0)
    if interior.any():
        scale = (target_sum - h[~interior].sum()) / h[interior].sum()
        h[interior] *= scale
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) after a bounded search."""

        n = 60
        x0_candidates = []

    target_sum = n / 2.0            # because integral must be 1 (dx = 2/n)
    bounds = [(0.0, 1.0)] * n
    penalty = 1e3

    best_h = None
    best_c5 = np.inf

    # add a few random starts
    for _ in range(5):
        rand = rnd.rand(n)
        rand = rand / rand.sum() * target_sum
        x0_candidates.append(rand)

    # main optimisation loop within time budget
    while time.time() - start_time < budget_s * 0.9:
        for x0 in x0_candidates:
            # local optimisation
            res = minimize(
