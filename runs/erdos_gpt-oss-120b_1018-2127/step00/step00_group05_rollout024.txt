import time, math, random
import numpy as np

def _project_to_box_and_sum(vec, target):
    """Project vec onto [0,1] box while keeping sum = target (water‑filling)."""
    x = np.clip(vec, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) < 1e-12:
        return x
    lower, upper = 0.0, 1.0
    for _ in range(15):
        diff = target - s
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            break
        adj = diff / cnt
        x[free] += adj
        x = np.clip(x, lower, upper)
        s = x.sum()
        if abs(s - target) < 1e-12:
            break
    # If still off due to rounding, distribute the tiny residual uniformly
    if abs(s - target) > 1e-12:
        resid = target - s
        x += resid / len(x)
        x = np.clip(x, lower, upper)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """
