import numpy as np
import time
import math
import random

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def _project_feasible(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce the integral constraint sum(h)=n/2."""
    n = len(h)
    target = n / 2.0
    # First clip
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        # degenerate, fill uniformly
        h[:] = target / n
        return h
    # Scale to match the target sum, iterating a couple of times to stay in bounds
    for _ in range(3):
        h *= (target / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-9:
            break
    # Final tiny correction if needed
    if s != target:
        diff = target - s
        # distribute the tiny difference uniformly (will stay in bounds because diff is tiny)
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h on [0,2] that minimizes the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        The best step function found (values in [0,1]).
    c5_bound : float
        The corresponding C5 upper bound.
    n_points : int
        Number of discretisation points used.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. initialise problem size and starting vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
        h = _project_feasible(h)
    else:
        n = 80
        h = np.zeros(n, dtype=float)
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    best_h = h.copy()
    best_score = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. simulated‑annealing style local search
    # ------------------------------------------------------------------
    temperature = 1.0
    temp_decay = 0.9995
    min_temp = 1e-6
    step_scale = 0.05          # maximal mass moved in one elementary move

    # pre‑compute index sets for speed
    def _feasible_move_indices(arr):
        return np.where(arr > 1e-12)[0], np.where(arr < 1.0 - 1e-12)[0]

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # pick source (decrease) and target (increase) indices
        src_idx, tgt_idx = _feasible_move_indices(best_h)
        if len(src_idx) == 0 or len(tgt_idx) == 0:
            break

        i = np.random.choice(src_idx)
        j = np.random.choice(tgt_idx)

        max_delta = min(step_scale, best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()

        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta

        # no need for a full projection – the move preserves the sum exactly
        # but clipping may be required for numerical safety
        cand = np.clip(cand, 0.0, 1.0)

        cand_score = _compute_c5(cand)

        if cand_score < best_score:
