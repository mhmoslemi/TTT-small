import numpy as np
import time
import random

def _compute_c5(h):
    """Return C5 for binary step function h (numpy 1-D array)."""
    N = h.size
    dx = 2.0 / N
    # correlation of h with 1-h, mode='full' includes zero padding
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def _initial_binary(N, seed):
    """Create a balanced binary vector of length N using a deterministic seed."""
    rnd = np.random.RandomState(seed)
    h = np.zeros(N, dtype=int)
    ones = rnd.choice(N, N // 2, replace=False)
    h[ones] = 1
    return h

def _upgrade_initial(initial, N):
    """Upsample an arbitrary initial vector to length N and binarize."""
    M = initial.shape[0]
    if M == N:
        h = (initial >= 0.5).astype(int)
    else:
        x_src = np.linspace(0, 2, M, endpoint=False)
        x_tgt = np.linspace(0, 2, N, endpoint=False)
        interp = np.interp(x_tgt, x_src, initial)
        h = (interp >= 0.5).astype(int)
    # enforce exact balance
    diff = h.sum() - N // 2
    if diff > 0:
        # too many ones -> turn some to zero
        idx = np.flatnonzero(h == 1)
        np.random.shuffle(idx)
        h[idx[:diff]] = 0
    elif diff < 0:
        # too few ones -> turn some zeros to one
        idx = np.flatnonzero(h == 0)
        np.random.shuffle(idx)
        h[idx[: -diff]] = 1
    return h
