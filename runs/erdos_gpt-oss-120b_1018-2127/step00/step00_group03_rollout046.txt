import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for a given step function h."""
    # correlation of h with (1-h)
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(conv)

def _make_initial_binary(n, rng, parent=None):
    """Create a feasible binary vector of length n with sum n/2."""
    if parent is not None:
        # Project parent to binary while preserving total mass
        # Clip, round, then adjust count
        h = np.clip(parent, 0.0, 1.0)
        h_bin = (h >= 0.5).astype(float)
        cur = int(h_bin.sum())
        target = n // 2
        if cur > target:
            # turn some ones to zeros
            idx = np.where(h_bin == 1)[0]
            rng.shuffle(idx)
            h_bin[idx[: cur - target]] = 0.0
        elif cur < target:
            idx = np.where(h_bin == 0)[0]
            rng.shuffle(idx)
            h_bin[idx[: target - cur]] = 1.0
        return h_bin
    # pure random binary with exact half ones
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    rng.shuffle(h)
    return h

def _simulated_annealing(h0, dx, time_limit):
    """Simple SA on binary vectors, returns best (h,score) found within time_limit."""
    rng = np.random.default_rng()
    h = h0.copy()
    best_h = h.copy()
    best_score = _compute_c5(h, dx)

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # propose swap of two opposite entries
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0  # swap

        cand_score = _compute_c5(h, dx)

        # SA acceptance
        T = max(0.001, 0.5 * (1 - (time.time() - start) / time_limit))
        accept = cand_score < best_score or rng.random() < np.exp((best_score - cand_score) / T)
        if accept:
            if cand_score < best_score:
                best_score = cand_score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) after searching for a good construction."""
    rng = np.random.default_rng(seed)
    total_start = time.time()

    # Determine which grid sizes to try; keep them modest.
    candidate_ns = [70, 100, 140, 200]
    best_overall_h = None
    best_overall_score = np.inf
    best_n = None

    # Allocate time roughly equally, but ensure we always have a fallback.
    per_n_budget = max(5.0, budget_s / len(candidate_ns))

    for n in candidate_ns:
        elapsed = time.time() - total_start
        if elapsed + per_n_budget > budget_s:
            break

        dx = 2.0 / n

        # Prepare initial vector
        parent = None
        if 'initial_h_values' in globals():
            parent_arr = np.asarray(initial_h_values, dtype=float)
            if parent_arr.size == n:
                parent = parent_arr
        h0 = _make_initial_binary(n, rng, parent)

        # Run SA
        time_allowed = min(per_n_budget, budget_s - (time.time() - total_start))
        h_opt, score = _simulated_annealing(h0, dx, time_allowed)

        if score < best_overall_score:
            best_overall_score = score
            best_overall_h = h_opt
            best_n = n

        # early exit if we already beat target
