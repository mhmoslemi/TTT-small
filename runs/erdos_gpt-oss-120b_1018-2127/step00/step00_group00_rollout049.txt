import time
import numpy as np
from scipy.optimize import minimize

def _compute_score(h: np.ndarray, n: int) -> float:
    """Maximum overlap integral for a given discretization."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _make_feasible(h: np.ndarray, n: int) -> np.ndarray:
    """Clip to [0,1] and enforce the integral constraint Σh = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _transfer(old_h: np.ndarray, old_n: int, new_n: int) -> np.ndarray:
    """Linear interpolation from an old grid to a new one."""
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    return _make_feasible(new_h, new_n)


def _optimize(initial_h: np.ndarray, n: int, seed: int) -> tuple:
    """Run a single SLSQP optimisation, returning (success, h, score)."""
    np.random.seed(seed)

    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = {"type": "eq", "fun": lambda x: x.sum() - n / 2.0}

    def obj(x):
        return _compute_score(x, n)

    res = minimize(
        obj,
        initial_h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 500, "ftol": 1e-6, "disp": False},
    )
    if not res.success:
        return False, None, None
    h_opt = _make_feasible(res.x, n)
    return True, h_opt, _compute_score(h_opt, n)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points).
    h_values is a NumPy array satisfying the problem constraints,
    c5_bound is the achieved upper bound for C₅, and n_points is its length.
    """
    start = time.time()
    n_candidates = [30, 45, 60, 90, 180]

    # base construction: use provided initial_h_values if present, otherwise uniform 0.5
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
        base_n = len(base_h)
    else:
        base_n = 45
        base_h = np.full(base_n, 0.5, dtype=float)  # integral = 1 automatically

    best_h = base_h
    best_n = base_n
    best_score = _compute_score(best_h, best_n)

    # iterate over resolutions, respecting the time budget
    for n in n_candidates:
        if time.time() - start > budget_s:
            break

        # obtain an initial point for this resolution
        if n == base_n:
            h0 = best_h.copy()
        else:
            h0 = _transfer(best_h, best_n, n)

        # quick feasibility check
        if not (0.0 <= h0).all() or not (h0 <= 1.0).all():
            h0 = _make_feasible(h0, n)

        # run optimisation
        success, h_opt, score_opt = _optimize(h0, n, seed + n)
        if success and score_opt < best_score:
            best_h, best_n, best_score = h_opt, n, score_opt

        # keep the best solution found so far for transferring to larger n
        # (even if this level did not improve, it may be a better starting point)
        if best_n != n:
            # transfer current best to this n for the next iteration
            pass  # nothing needed; best_h, best_n already store the best

    # final sanity: ensure the returned vector obeys constraints
    best_h = _make_feasible(best_h, best_n)
    final_score = _compute_score(best_h, best_n)

    return best_h, final_score, best_n
