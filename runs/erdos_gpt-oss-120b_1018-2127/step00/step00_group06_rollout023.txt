import time, math, random
import numpy as np

def _binary_from_continuous(v, ones):
    """Return a binary (0/1) array with exactly `ones` ones, preserving largest entries."""
    idx = np.argsort(v)[::-1]          # descending
    out = np.zeros_like(v, dtype=float)
    out[idx[:ones]] = 1.0
    return out

def _initial_binary(n, seed):
    """Create an initial binary vector of length n with n/2 ones."""
    rng = np.random.default_rng(seed)
    v = rng.random(n)
    return _binary_from_continuous(v, n // 2)

def _prepare_initial(n, seed):
    """If a global `initial_h_values` exists, interpolate it; otherwise random."""
    try:
        init = initial_h_values  # provided by harness
        n0 = init.shape[0]
        # Interpolate onto new grid
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        interpolated = np.interp(x, x0, init)
        # Binarize while keeping exactly n/2 ones
        return _binary_from_continuous(interpolated, n // 2)
    except NameError:
        return _initial_binary(n, seed)

def _c5_value(h):
    """Return max overlap value (without dx scaling)."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] with low C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a modest grid size; 200 gives dx=0.01, fine enough and fast.
    n = 200
    dx = 2.0 / n

    # Initial feasible binary vector
    h = _prepare_initial(n, seed)
    best_h = h.copy()
    best_val = _c5_value(h) * dx

    # Pre-compute indices of ones and zeros for quick swaps
    ones_idx = np.flatnonzero(h == 1.0)
    zeros_idx = np.flatnonzero(h == 0.0)

    # Simulated annealing parameters
    max_iters = 200000
    temp_start = 0.5
    temp_end = 1e-4
    decay = (temp_end / temp_start) ** (1.0 / max_iters)

    temp = temp_start
    it = 0
    while it < max_iters and (time.time() - start_time) < budget_s * 0.95:
        # propose a swap between a 1 and a 0
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        # perform swap
        h[i], h[j] = 0.0, 1.0

        # evaluate
        cur_val = _c5_value(h) * dx

        # acceptance test
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        delta = cur_val - best_val
        if delta <= 0 or rng.random() < math.exp(-delta / temp):
            # keep change, update index lists
            ones_idx = np.flatnonzero(h == 1.0)
            zeros_idx = np.flatnonzero(h == 0.0)
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0

        # temperature schedule
        temp *= decay
        it += 1

        # occasional random restart if stagnating
        if it % 50000 == 0 and best_val > 0.38:
            # keep best, re‑randomize rest
            h = best_h.copy()
            ones_idx = np.flatnonzero(h == 1.0)
            zeros_idx = np.flatnonzero(h == 0.0)

    # Final evaluation to ensure bound is accurate
    final_c5 = _c5_value(best_h) * dx
    return best_h, float(final_c5), int(n)
