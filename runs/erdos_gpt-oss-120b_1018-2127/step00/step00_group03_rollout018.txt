import numpy as np
import time
import random
import math

def _score(h):
    """Return C5 for discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _init_h():
    """Create a feasible start vector, using initial_h_values if present."""
    # try to use the provided construction
    try:
        h0 = np.array(initial_h_values, dtype=np.float32)
    except NameError:
        # fallback: simple half‑filled block (sum = n/2)
        n = 70
        h0 = np.zeros(n, dtype=np.float32)
        h0[: n // 2] = 1.0
        return h0

    n = h0.shape[0]
    target = n / 2.0
    cur = h0.sum()
    if abs(cur - target) > 1e-12:                      # enforce exact integral
        h0 = h0 * (target / cur)
        np.clip(h0, 0.0, 1.0, out=h0)
        diff = target - h0.sum()
        # distribute the tiny residual uniformly over non‑boundary entries
        free = np.where((h0 > 0.0) & (h0 < 1.0))[0]
        if free.size:
            h0[free[0]] += diff
    np.clip(h0, 0.0, 1.0, out=h0)
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with a low C5 bound."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    h = _init_h()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _score(best_h)

    # annealing parameters
    max_iter = 30000
    temp_start = 0.2
    temp_end = 1e-4

    for it in range(max_iter):
        # respect the wall‑clock budget
        if time.time() - start_time > budget_s * 0.95:
            break

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # maximal admissible transfer from i to j
        max_move = min(best_h[i], 1.0 - best_h[j])
        if max_move <= 0.0:
            continue
