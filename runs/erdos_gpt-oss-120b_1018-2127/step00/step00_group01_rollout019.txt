import time
import numpy as np
import cvxpy as cp

def _evaluate(h, dx):
    """Return max overlap C5 for a given h."""
    conv = np.correlate(h, 1 - h, mode="full") * dx
    return float(conv.max())

def _project(y, target_sum):
    """Euclidean projection of y onto {0<=x<=1, sum x = target_sum}."""
    n = y.shape[0]
    x = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(x - y)),
                      [x >= 0, x <= 1, cp.sum(x) == target_sum])
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    return np.array(x.value).reshape(-1)

def _gradient(h, lag):
    """Sub‑gradient of the overlap corresponding to given lag."""
    n = h.shape[0]
    g = np.zeros(n)
    if lag >= 0:
        L = lag
        g[:n - L] += 1 - h[L:]
        g[L:]     -= h[:n - L]
    else:
        L = -lag
        g[L:]     += 1 - h[:n - L]
        g[:n - L] -= h[L:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the C5 bound."""
    rnd = np.random.RandomState(seed)
    start = time.time()

    # Determine problem size
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        h = init.copy()
    except NameError:
        n = 200                     # fallback size (still cheap)
        h = rnd.rand(n)

    target_sum = n / 2.0
    dx = 2.0 / n

    # Ensure feasibility of the starting point
    h = _project(h, target_sum)

    best_h = h.copy()
    best_c = _evaluate(best_h, dx)

    iter_count = 0
    while time.time() - start < budget_s * 0.95:   # leave margin
        # compute current objective and dominant lag
        conv = np.correlate(h, 1 - h, mode="full") * dx
        lag = int(conv.argmax()) - (n - 1)

        # gradient step
        g = _gradient(h, lag)
        lr = 0.5 / (1 + iter_count * 0.01)   # slowly decreasing step
        y = h - lr * g

        # project back to feasible set
        h_new = _project(y, target_sum)

        # evaluate
        c_new = _evaluate(h_new, dx)

        # keep improvement
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

        # optionally accept non‑improving move (simulated annealing style)
        # here we simply replace if not worse to encourage movement
        if c_new <= _evaluate(h, dx):
            h = h_new

        iter_count += 1

        # occasional random restart to escape local minima
        if iter_count % 250 == 0 and time.time() - start < budget_s * 0.9:
            h = _project(rnd.rand(n), target_sum)

    return best_h.tolist(), best_c, n
