def _project_feasible(vec, n):
    target = n / 2.0
    # initial clipping
    v = np.clip(vec, 0.0, 1.0)
    # iterative water‑filling / scaling
    for _ in range(15):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-10:
            break
        # decide which coordinates may be moved
        if diff > 0:          # need to decrease total mass
            free = v > 0.0
        else:                 # need to increase total mass
            free = v < 1.0
        if not np.any(free):
            break
        delta = diff / free.sum()
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
