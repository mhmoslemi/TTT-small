import numpy as np
import time
from math import pi

def _corr_max(h, dx):
    """Maximum of discrete correlation of h with (1-h) using FFT."""
    n = h.size
    # FFT based circular convolution, then shift to linear correlation range
    fh = np.fft.fft(h)
    f1 = np.fft.fft(1.0 - h)
    conv = np.fft.ifft(fh * np.conj(f1)).real
    # Linear correlation is obtained by taking appropriate slice
    # np.correlate with mode='full' returns length 2n-1, centered at n-1
    full = np.concatenate((conv[-(n-1):], conv[:n]))
    return np.max(full) * dx

def _evaluate_candidate(h, dx):
    """Return C5 value for candidate h."""
    # Clip to maintain feasibility (tiny numerical errors)
    h = np.clip(h, 0.0, 1.0)
    return _corr_max(h, dx)
