import time
import numpy as np

def _project_to_constraints(h, target_sum):
    """Clip h to [0,1] and adjust to satisfy sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    # Simple iterative redistribution
    for _ in range(15):
        if abs(total - target_sum) < 1e-12:
            break
        diff = target_sum - total
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # fallback: rescale whole vector then re‑clip
            if total != 0:
                h *= target_sum / total
            h = np.clip(h, 0.0, 1.0)
            total = h.sum()
            continue
        delta = diff / cnt
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
    # final tiny correction by scaling the free part if needed
    if abs(total - target_sum) > 1e-9:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] *= target_sum / total
            h = np.clip(h, 0.0, 1.0)
    return h

def _c5_score(h):
    """Return the discrete C5 value for step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def _perturb(h, rng, sigma, frac=0.15):
    """Return a perturbed copy of h respecting constraints."""
    n = h.size
    h_new = h.copy()
    k = max(1, int(frac * n))
    idx = rng.choice(n, size=k, replace=False)
    noise = rng.normal(loc=0.0, scale=sigma, size=k)
    h_new[idx] += noise
    target = n / 2.0
    h_new = _project_to_constraints(h_new, target)
    return h_new

def _initial_vector(n, rng):
    """Create a random feasible vector of length n."""
    h = rng.random(n)
    target = n / 2.0
    return _project_to_constraints(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.

    Returns
    -------
    h_values : numpy.ndarray
        The best step‑function values found.
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of discretisation points used.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # 1.  Choose base resolution and initialise the population
    # ------------------------------------------------------------------ #
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        # ensure feasibility (might be already exact)
        h0 = _project_to_constraints(h0, n0 / 2.0)
    else:
        n0 = 100  # fallback size (even)
        h0 = _initial_vector(n0, rng)

    best_h = h0
    best_score = _c5_score(best_h)

    # ------------------------------------------------------------------ #
    # 2.  Core stochastic local search loop
    # ------------------------------------------------------------------ #
    max_iters = 200_000  # safety cap
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95 or iteration >= max_iters:
            break
        # annealing‑style sigma decay
        sigma = 0.25 * (1.0 - elapsed / budget_s) + 0.01
        cand = _perturb(best_h, rng, sigma)
        cand_score = _c5_score(cand)
        if cand_score < best_score:
            best_h, best_score = cand, cand_score

                restart_score = _c5_score(restart)

        iteration += 1

    # ------------------------------------------------------------------ #
    # 3.  Optional refinement on a finer grid (double resolution)
    # ------------------------------------------------------------------ #
    if time.time() - start_time < budget_s * 0.85:
        # upsample by linear interpolation
