import time
import numpy as np

def _evaluate_c5(h, dx):
    """Return max correlation C5 for a given step vector h."""
    # correlation of h with (1-h) over all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _random_binary(n, rng):
    """Generate a binary vector of length n with exactly n//2 ones."""
    ones = n // 2
    h = np.zeros(n, dtype=np.float64)
    h[:ones] = 1.0
    rng.shuffle(h)
    return h
