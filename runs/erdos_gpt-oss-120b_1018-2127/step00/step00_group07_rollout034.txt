import numpy as np
import time
from math import inf

# Helper to compute the Erdős C5 bound for a discrete step function
def _c5_of_h(h):
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' yields length 2n-1
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

# Project a vector onto the affine set {h : sum(h)=target, 0<=h<=1}
def _project_onto_constraints(h, target_sum):
    # clip to [0,1] first
    h = np.clip(h, 0.0, 1.0)
    # adjust to meet the sum using a simple water‑filling algorithm
    excess = h.sum() - target_sum
    if abs(excess) < 1e-12:
        return h
    # If we need to add mass, raise the smallest entries first; if we need
    # to remove mass, lower the largest entries first.
    idx = np.argsort(h) if excess > 0 else np.argsort(-h)
    for i in idx:
        if excess > 0:
            give = min(h[i], excess)
            h[i] -= give
            excess -= give
        else:
            take = min(1.0 - h[i], -excess)
            h[i] += take
            excess += take
        if abs(excess) < 1e-12:
            break
    return h
