import numpy as np
import time

def _repair(h, target_sum, max_iter=20, tol=1e-10):
    """Project h onto [0,1] box and affine constraint sum(h)=target_sum."""
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < tol:
            break
        # distribute the deficit/excess uniformly
        h += diff / h.size
    # final clip and a last tiny scaling if needed
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur != 0 and abs(cur - target_sum) > tol:
        h *= target_sum / cur
    return h

def _c5_value(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h at the discretisation points.
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of discretisation points used.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    init = globals().get('initial_h_values', None)
    if init is not None:
        n_points = int(init.shape[0])
        h_current = np.array(init, dtype=float)
    else:
        n_points = 200
        h_current = np.full(n_points, 0.5)  # placeholder, will be repaired

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because ∑h = 1/dx = n/2

    # Ensure feasibility of the starting point
    h_current = _repair(h_current, target_sum)

    best_h = h_current.copy()
    best_c5 = _c5_value(best_h, dx)

    # Simple stochastic local search with occasional random restarts
    while time.time() - start_time < budget_s * 0.95:
        # propose a perturbation
        h_trial = best_h.copy()
        # number of coordinates to modify (≈2% of points, at least 1)
        k = max(1, n_points // 50)
        idx = np.random.choice(n_points, k, replace=False)
        # add gaussian noise, scale shrinking with time
        elapsed = time.time() - start_time
        remaining = max(budget_s - elapsed, 1e-6)
        scale = 0.2 * (remaining / budget_s)  # cool down
        h_trial[idx] += np.random.normal(scale=scale, size=k)

        h_trial = _repair(h_trial, target_sum)
        c5_trial = _c5_value(h_trial, dx)

        # accept if improvement
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial

        # occasional random restart to escape deep local minima
        if np.random.rand() < 0.001:
            restart = np.random.rand(n_points)
            restart = _repair(restart, target_sum)
            c5_restart = _c5_value(restart, dx)
            if c5_restart < best_c5:
                best_c5 = c5_restart
                best_h = restart

    # Final safety repair before returning
    best_h = _repair(best_h, target_sum)
    best_c5 = _c5_value(best_h, dx)

    return best_h, float(best_c5), n_points
