import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def _slsqp_optimize(initial_h, dx, budget, start_time):
    """Attempt a quick SLSQP optimisation; stop early if time runs out."""
    n = len(initial_h)

    def obj(x):
        return _compute_c5(x, dx)

    cons = ({
        "type": "eq",
        "fun": lambda x: np.sum(x) - n / 2.0,
    })
    bounds = [(0.0, 1.0)] * n

    opts = {"maxiter": 500, "ftol": 1e-9}
    res = minimize(
        obj,
        initial_h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    # If optimisation did not finish due to time, fall back to initial
    if not res.success:
        return initial_h, _compute_c5(initial_h, dx)
    return res.x, res.fun

def _random_swap_hill_climb(h0, dx, budget, start_time, rng):
    """Simple hill‑climbing that preserves the integral by swapping mass."""
    n = len(h0)
    best_h = h0.copy()
    best_val = _compute_c5(best_h, dx)

    # parameters for move size adaptation
    base_step = 0.05  # typical fraction of a point that may be moved
    while time.time() - start_time < budget * 0.9:
        i, j = rng.choice(n, size=2, replace=False)

        # maximum transferable amount from i to j while staying in [0,1]
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue

        # random delta, scaled to a fraction of max_delta
        delta = (rng.random() - 0.5) * 2.0 * max_delta * base_step
        delta = np.clip(delta, -max_delta, max_delta)

        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        # tiny numeric drift may appear; re‑project very small violations
        np.clip(new_h, 0.0, 1.0, out=new_h)

        new_val = _compute_c5(new_h, dx)
        if new_val < best_val:
            best_h, best_val = new_h, new_val

    return best_h, best_val
