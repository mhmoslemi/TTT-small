import time, math, random
import numpy as np

def _project_to_constraints(h, target_sum):
    """Clip to [0,1] and adjust to exact sum using simple water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(15):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        idx = np.where(free)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        # respect bounds while adding delta
        add = np.clip(delta, -h[idx], 1.0 - h[idx])
        h[idx] += add
    return np.clip(h, 0.0, 1.0)


def _compute_c5(h, dx):
    """Maximum overlap integral for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) respecting the Erdős C5 problem."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # ----- grid size -------------------------------------------------------
    base_n = 60                     # preferred small size (<1k)
    try:
        init = np.asarray(initial_h_values, dtype=float)
        init_n = init.size
    except Exception:
        init = None
        init_n = 0

    n = max(base_n, init_n)
    if n % 2:
        n += 1                     # keep n even for convenience
    dx = 2.0 / n
    target_sum = n / 2.0

    # ----- initial construction --------------------------------------------
    if init is not None and init_n > 0:
        # linear interpolation from the parent grid to the new grid
        x_old = np.linspace(0.0, 2.0, init_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n,   endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        h = np.full(n, 0.5)        # uniform 0.5 satisfies all constraints

    h = _project_to_constraints(h, target_sum)

    # ----- optimisation ----------------------------------------------------
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)
    cur_h = h.copy()

    max_time = budget_s * 0.95      # keep a safety margin
