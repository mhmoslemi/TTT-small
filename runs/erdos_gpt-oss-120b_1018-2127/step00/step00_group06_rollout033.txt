import time
import numpy as np

def _project_to_constraints(y: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0<=x<=1, sum x = target} using a simple bisection on the Lagrange multiplier."""
    low = -np.max(y) - 1.0
    high = np.max(y) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(y - (low + high) / 2.0, 0.0, 1.0)


def _c5_score(h: np.ndarray) -> float:
    """Discrete C5 score for a step function sampled on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the sample size."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # base sampling size – use the provided construction if present
    init_h = globals().get("initial_h_values")
    base_n = int(init_h.shape[0]) if isinstance(init_h, np.ndarray) else 95

    # candidate grid sizes (coarse to fine)
    grid_sizes = [base_n, base_n * 2, base_n * 4]

    best_h = None
    best_score = np.inf
    best_n = base_n

    for n in grid_sizes:
        # ----- initialise a feasible start vector -----
        if n == base_n and isinstance(init_h, np.ndarray):
            h0 = init_h.astype(float).copy()
        else:
            # interpolate the known construction or create a random one
            if isinstance(init_h, np.ndarray):
                x_old = np.linspace(0, 2, base_n, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h0 = np.interp(x_new, x_old, init_h)
            else:
                h0 = rng.uniform(0.0, 1.0, n)
            h0 = _project_to_constraints(h0, n / 2.0)

        # ----- local stochastic search -----
        h_curr = h0.copy()
        score_curr = _c5_score(h_curr)

        if score_curr < best_score:
            best_h, best_score, best_n = h_curr.copy(), score_curr, n

        # simple simulated‑annealing style hill climbing
        sigma = 0.2
        iteration = 0
        max_iter = 50000  # generous upper bound; actual loop stops on time

        while time.time() - start_time < budget_s and iteration < max_iter:
            # propose a new candidate
            cand = h_curr + sigma * rng.standard_normal(n)
            cand = _project_to_constraints(cand, n / 2.0)

            score_cand = _c5_score(cand)

            # accept if better or with a small Metropolis probability
            if score_cand < score_curr:
                h_curr, score_curr = cand, score_cand
                if score_cand < best_score:
                    best_h, best_score, best_n = cand.copy(), score_cand, n
            else:
                # temperature schedule (slowly cooling)
                temp = max(1e-4, 0.01 * np.exp(-0.0001 * iteration))
                if rng.random() < np.exp(-(score_cand - score_curr) / temp):
                    h_curr, score_curr = cand, score_cand

            # gradually shrink perturbation magnitude
            sigma *= 0.9997
            iteration += 1

            # early exit if we have already beaten the target
            if best_score < 0.3808:
                break

        if time.time() - start_time >= budget_s or best_score < 0.3808:
            break

    # final safety projection (should already be feasible)
    best_h = _project_to_constraints(best_h, best_n / 2.0)

    return (best_h, best_score, best_n)
