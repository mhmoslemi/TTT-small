import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# helper: compute C5 for discrete step function h on [0,2]
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

# helper: resample h to a different number of points, keep constraints
def upscale_h(old_h, new_n):
    old_n = old_h.shape[0]
    # map old points to [0,2)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, old_h)
    # enforce integral constraint sum = new_n/2
    target = new_n / 2.0
    total = h_new.sum()
    if total != 0:
        h_new = h_new * (target / total)
    # clip to [0,1] and repair the sum if needed
    h_new = np.clip(h_new, 0.0, 1.0)
    deficit = target - h_new.sum()
    if abs(deficit) > 1e-12:
        free = (h_new > 0.0) & (h_new < 1.0)
        if free.any():
            h_new[free] += deficit / free.sum()
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

# -------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # start from the supplied construction
    try:

        h_current = h_current.astype(float)
    best_h = h_current.copy()
    best_c5 = compute_c5(best_h)

    # resolutions to explore (coarse → fine)
    for n in (45, 60, 90, 120):
        # stop early if budget is almost exhausted
        if time.time() - start > budget_s * 0.95:
            break

        # generate initial guess for this resolution
        if n == best_h.shape[0]:

        else:

        # equality constraint: sum h = n/2

            "type": "eq",
            "fun": lambda h, nn=n: np.sum(h) - nn / 2.0,
        })
        bounds = [(0.0, 1.0)] * n

        # local optimisation (SLSQP works with simple bounds + eq)
        res = minimize(
            compute_c5,
