import time
import numpy as np

def _repair(h, target):
    """Clip to [0,1] and enforce sum == target (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # scale then re‑clip
        scale = target / s if s != 0 else 0.0
        h = np.clip(h * scale, 0.0, 1.0)
    # final tiny adjustment if still off (distribute to first element)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # adjust a single entry while staying feasible
        i = 0
        new_val = np.clip(h[i] + diff, 0.0, 1.0)
        h[i] = new_val
    return h

def _c5_value(h, dx):
    """Compute the Erdős C5 value for a discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the C5 overlap bound.

    Returns
    -------
    h_values : np.ndarray
        The best-found values of h on the discretised grid.
    c5_bound : float
        The corresponding C5 value.
    n_points : int
        Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 100  # fallback size
        # start from a uniform block of width 1 (integral 1)
        h0 = np.zeros(n_points, dtype=float)
        block_len = n_points // 2
        h0[:block_len] = 1.0

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum(h)*dx = 1

    # Ensure feasibility of the initial vector
    best_h = _repair(h0.copy(), target_sum)
    best_c5 = _c5_value(best_h, dx)

    # Simple stochastic search with annealing‑style step decay
    step_scale = 0.2  # initial perturbation magnitude
    min_step = 1e-4
    decay = 0.995

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:  # leave a little margin
            break

        # propose a new candidate
        proposal = best_h + rng.normal(scale=step_scale, size=n_points)
        proposal = _repair(proposal, target_sum)

        # evaluate
        cand_c5 = _c5_value(proposal, dx)

        # accept if better
        if cand_c5 < best_c5:
            best_h = proposal
            best_c5 = cand_c5

        # update step size
        step_scale = max(step_scale * decay, min_step)

    return best_h, best_c5, n_points
