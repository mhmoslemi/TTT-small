import time
import numpy as np

def _full_correlation_max(h):
    """Maximum correlation of h with (1-h) over all integer shifts."""
    n = h.shape[0]
    dx = 2.0 / n
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    H = np.fft.fft(h, size)
    G = np.fft.fft(1.0 - h, size)
    corr = np.fft.ifft(H * np.conj(G)).real
    # extract the full‑length (2n‑1) result centered at lag 0
    corr = np.concatenate((corr[-(n - 1):], corr[:n]))
    return np.max(corr) * dx

def _initial_binary(n, seed):
    """Create a binary vector of length n with exactly n/2 ones."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function h (as a list of values), its C5 bound, and n_points.
    The algorithm performs a simple simulated‑annealing on a binary vector
    with exactly n/2 ones, because this automatically satisfies the integral
    constraint ∫₀² h = 1.
    """
    start_time = time.time()
    # Choose number of sample points (must be even)
    n = 200
    dx = 2.0 / n

    # Use provided initial_h_values if present; otherwise start from random binary.
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        if init.shape[0] != n:
            # resample to target size using simple nearest‑neighbor copy
            idx = (np.linspace(0, init.shape[0] - 1, n)).astype(int)
            h_curr = init[idx].copy()
        else:
            h_curr = init.copy()
        # project to binary while keeping total sum n/2
        # keep the largest n/2 entries as 1, others 0
        thresh = np.partition(h_curr, -n // 2)[-n // 2]
        h_curr = (h_curr > thresh).astype(np.float64)
        # adjust in case of ties
        while h_curr.sum() != n // 2:
            diff = int(n // 2 - h_curr.sum())
            if diff > 0:
                zero_idx = np.where(h_curr == 0)[0]
                choose = zero_idx[:diff]
                h_curr[choose] = 1.0
            else:
                one_idx = np.where(h_curr == 1)[0]
                choose = one_idx[: -diff]
                h_curr[choose] = 0.0
    except NameError:
        # No initial_h_values supplied
        h_curr = _initial_binary(n, seed)

    best_h = h_curr.copy()
    best_c5 = _full_correlation_max(best_h)

    # simulated annealing parameters
    rng = np.random.default_rng(seed + 1)
    temp = 0.02
    cooling = 0.99995
    iter_count = 0
    report_interval = 500

    # Pre‑compute index lists for ones and zeros to speed swaps
    while time.time() - start_time < budget_s:
        # pick a random 1‑position and a random 0‑position
        ones = np.flatnonzero(h_curr == 1.0)
        zeros = np.flatnonzero(h_curr == 0.0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        h_curr[i], h_curr[j] = 0.0, 1.0
        new_c5 = _full_correlation_max(h_curr)

        # Metropolis acceptance
        accept = False
        if new_c5 < best_c5:
            accept = True
        else:
            delta = best_c5 - new_c5
            if rng.random() < np.exp(delta / temp):
                accept = True

        if accept:
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_curr.copy()
        else:
            # revert swap
            h_curr[i], h_curr[j] = 1.0, 0.0

        # cooling schedule
        temp *= cooling
        iter_count += 1

        # occasional early exit if time is almost up
        if iter_count % report_interval == 0:
            if time.time() - start_time > budget_s * 0.95:
                break

    # Return as a plain Python list for compatibility
    return (best_h.tolist(), float(best_c5), int(n))
