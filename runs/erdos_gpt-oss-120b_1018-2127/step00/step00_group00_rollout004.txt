import time
import random
import numpy as np

def _project(v, target):
    """Project vector v onto [0,1] box with sum = target using a scalar shift."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _c5_score(h, dx):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx via discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _make_random_feasible(n, rng):
    """Generate a random vector in [0,1]^n whose sum equals n/2."""
    v = rng.random(n)  # uniform in [0,1)
    target = n * 0.5
    return _project(v, target)

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h (numpy array), its C5 bound, and n_points."""
    start = time.time()
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # Determine problem size and initial vector
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    except NameError:
        n = 45
        init_h = None

    dx = 2.0 / n
    target_sum = n * 0.5

    # Starting point
    if init_h is not None:
        h = _project(init_h, target_sum)
    else:
        h = _make_random_feasible(n, rng)

    best_h = h.copy()
    best_score = _c5_score(best_h, dx)

    # Simple stochastic local search
    iteration = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break
        iteration += 1

        # Occasionally restart from a fresh random feasible point
        if iteration % 200 == 0:
            cand = _make_random_feasible(n, rng)
        else:
            cand = best_h.copy()
            # Perturb a few coordinates
            k = rng.integers(1, 5)          # number of positions to change
            idx = rng.choice(n, size=k, replace=False)
            delta = rng.normal(scale=0.15, size=k)
            cand[idx] += delta
            # No need to clip before projection; projection handles it
        cand = _project(cand, target_sum)

        score = _c5_score(cand, dx)

        if score < best_score:
            best_score, best_h = score, cand.copy()
            # optional early exit if we surpass target bound
            if best_score <= 0.3808:
                break

    # Return results
    return best_h, best_score, n
