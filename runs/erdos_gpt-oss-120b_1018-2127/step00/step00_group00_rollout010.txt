import time
import numpy as np
from scipy.optimize import minimize

# The harness provides these globals
# evaluate_erdos_solution(h) -> float
# initial_h_values -> ndarray (may be None)

def _c5_from_h(h):
    """Compute C5 for a given discrete h."""
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_bounds_and_sum(h):
    """Project a vector onto [0,1] box and exact-sum constraint."""
    N = h.size
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust to satisfy sum = N/2 using a simple affine shift and re‑clip
    target = N / 2.0
    diff = target - h.sum()
    # distribute diff uniformly; then re‑clip and iterate a few times
    for _ in range(5):
        h += diff / N
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

def _optimize_slsqp(h0, time_limit):
    """Run a short SLSQP optimisation respecting bounds and sum."""
    N = h0.size
    bounds = [(0.0, 1.0) for _ in range(N)]
    cons = {"type": "eq", "fun": lambda x: x.sum() - N / 2.0}

    opts = {"maxiter": 250, "ftol": 1e-12}
    start = time.time()
    res = minimize(_c5_from_h, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options=opts)
    # enforce time budget
    if time.time() - start > time_limit:
        return None, None
    if not res.success:
        return None, None
    h_opt = _project_bounds_and_sum(res.x)
    return h_opt, _c5_from_h(h_opt)

def _random_local_search(h_best, score_best, deadline):
    """Simple stochastic neighbour search using swaps that keep the sum."""
    N = h_best.size
    h_cur = h_best.copy()
    score_cur = score_best
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick two distinct indices
        i, j = rng.integers(0, N, size=2, endpoint=False)
        if i == j:
            continue
        # propose moving epsilon from i to j
        eps = rng.uniform(-0.05, 0.05)
        # ensure after move values stay in [0,1]
        if not (0.0 <= h_cur[i] - eps <= 1.0 and 0.0 <= h_cur[j] + eps <= 1.0):
            continue
        h_new = h_cur.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # evaluate
        sc = _c5_from_h(h_new)
        if sc < score_cur:
            h_cur, score_cur = h_new, sc
            if sc < score_best:
                h_best, score_best = h_new.copy(), sc
    return h_best, score_best
