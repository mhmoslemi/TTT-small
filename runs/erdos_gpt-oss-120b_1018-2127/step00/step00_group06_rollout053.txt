import time
import numpy as np

def _project_bounds_sum(v, target):
    """Project vector v onto the box [0,1] and the hyperplane sum(v)=target."""
    # Binary search for lambda such that sum(clip(v - lam,0,1)) == target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _objective(h, dx):
    """Maximum overlap integral for step function h."""
    # correlate h with (1-h); mode='full' gives all lags
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _initialize_feasible(n, rng):
    """Create a random feasible vector of length n."""
    v = rng.random(n)  # uniform in [0,1)
    target = n * 0.5
    return _project_bounds_sum(v, target)

def run(seed=42, budget_s=1100, **kwargs):

    Returns:

    """

    # Annealing parameters
