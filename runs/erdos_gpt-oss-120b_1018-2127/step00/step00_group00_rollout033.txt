import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------
def compute_score(h):
    """
    Compute the C5 overlap value for a step function sampled in h.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def is_feasible(h):
    """Check box constraints and integral constraint."""
    n = h.shape[0]
    if np.any(h < -1e-9) or np.any(h > 1 + 1e-9):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-6


def make_initial(old_h, new_n):
    """
    Interpolate an existing sample vector to a new resolution.
    """
    old_n = old_h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    return new_h


def eq_sum_constraint(x, n):
    """Equality constraint enforcing sum(x) == n/2."""
    return np.sum(x) - n / 2.0


def minimize_h(x0, n, maxiter=800, ftol=1e-6):
    """
    Run SLSQP to minimise compute_score under box and linear constraints.
    """
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{
        "type": "eq",
        "fun": eq_sum_constraint,
        "args": (n,)
    }]

    opts = {"maxiter": maxiter, "ftol": ftol, "disp": False}
    res = minimize(
        compute_score,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    return res

# ----------------------------------------------------------------------
