import time, math, random
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _project_feasible(h, n):
    """Project onto [0,1] with sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-9:
            break
        free = np.where((h > 1e-9) & (h < 1 - 1e-9))[0]
        if free.size == 0:
            free = np.arange(len(h))
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def _transfer(old_h, old_n, new_n):
    """Interpolate old_h (size old_n) onto a new grid of size new_n."""
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, old_h, left=0.0, right=0.0)
    if h_new.sum() != 0:
        h_new *= (new_n / 2.0) / h_new.sum()
    h_new = _project_feasible(h_new, new_n)
    return h_new
