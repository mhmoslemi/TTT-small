import time
import numpy as np
from math import exp

def compute_c5(h, dx):
    """Return max overlap integral C5 for given h array."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def random_binary_vector(n, ones):
    """Generate binary vector of length n with exactly `ones` entries equal to 1."""
    vec = np.zeros(n, dtype=float)
    vec[:ones] = 1.0
    np.random.shuffle(vec)
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] discretized to n_points samples
    to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimized values in [0,1] (binary 0/1) of length n_points.
    c5_bound : float
        Achieved C5 upper bound.
    n_points : int
        Number of discretization points (grid size).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Choose a modest grid size (faster evaluation, still expressive)
    n_points = 200                 # must be even so that sum = n/2 is integer
    ones_required = n_points // 2
    dx = 2.0 / n_points

    # Initialise h: use provided initial_h_values if sensible, otherwise random binary
    try:
        init = np.asarray(initial_h_values, dtype=float)
        if init.shape[0] == n_points and np.all((0 <= init) & (init <= 1)):
            # project to binary while keeping total mass
            cur = np.copy(init)
            # round to nearest binary
            cur = np.rint(cur)
            cur_sum = cur.sum()
            diff = ones_required - cur_sum
            if diff > 0:
                zero_idx = np.where(cur == 0)[0]
                if len(zero_idx) >= diff:
                    cur[np.random.choice(zero_idx, diff, replace=False)] = 1
            elif diff < 0:
                one_idx = np.where(cur == 1)[0]
                if len(one_idx) >= -diff:
                    cur[np.random.choice(one_idx, -diff, replace=False)] = 0
            h = cur
        else:
            raise ValueError
    except Exception:
        h = random_binary_vector(n_points, ones_required)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # Simulated annealing / greedy hill‑climb
    temperature0 = 0.02
    temperature_final = 0.0005
    max_iters = 200000  # generous upper bound; loop will break on time
    it = 0

    # Pre‑compute indices of 1s and 0s for fast swaps
    ones_idx = np.flatnonzero(h == 1)
    zeros_idx = np.flatnonzero(h == 0)

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        if it >= max_iters:
            break
        it += 1

        # Linear cooling schedule
        t = temperature0 + (temperature_final - temperature0) * (it / max_iters)

        # Propose a swap between a 1 and a 0
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        # Perform swap
        h[i], h[j] = 0.0, 1.0

        # Evaluate
        new_c5 = compute_c5(h, dx)

        # Acceptance rule
        accept = False
        if new_c5 < best_c5:
            accept = True
        else:
            if np.random.rand() < exp(-(new_c5 - best_c5) / t):
                accept = True

        if accept:
            # update index lists
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0

    # Final evaluation to ensure consistency
    final_c5 = compute_c5(best_h, dx)

    return best_h, float(final_c5), n_points
