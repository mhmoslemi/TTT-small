import numpy as np
import time
from scipy.optimize import minimize

# the harness provides these globals if available
# initial_h_values : np.ndarray of shape (n_initial,)
# evaluate_erdos_solution : function to compute C5 (unused here)

def _compute_c5(h):
    """Discrete approximation of C5 for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair_solution(h):
    """Clip to [0,1] and enforce the integral constraint exactly."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # distribute the deficit/surplus uniformly, then clip again
        delta = (target_sum - cur_sum) / n
        h += delta
        h = np.clip(h, 0.0, 1.0)
        # final tiny correction on first element to hit the exact sum
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            h[0] += target_sum - cur_sum
            h[0] = min(max(h[0], 0.0), 1.0)
    return h

def _optimize_h(initial_h, time_limit):
    """Run a few SLSQP attempts starting from (perturbed) initial_h."""
    n = initial_h.size
    bounds = [(0.0, 1.0)] * n
    eq_cons = {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}

    best_h = _repair_solution(initial_h.copy())
    best_val = _compute_c5(best_h)

    start = time.time()
    attempts = 0
    while attempts < 8 and time.time() - start < time_limit:
        attempts += 1
        # add small random noise to avoid stagnation
        pert = np.random.default_rng(attempts).normal(scale=0.05, size=n)
        x0 = best_h + pert
        x0 = _repair_solution(x0)

        res = minimize(
            lambda x: _compute_c5(x),
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[eq_cons],
            options={'maxiter': 800, 'ftol': 1e-9, 'disp': False}
        )
        if not res.success:
            continue
        h_candidate = _repair_solution(res.x)
        val = _compute_c5(h_candidate)
        if val < best_val - 1e-6:
            best_h, best_val = h_candidate, val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the allocated budget."""
    np.random.seed(seed)
    start_total = time.time()

    # Determine the initial discretisation size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: uniform 0.5 on a modest grid
        h0 = np.full(45, 0.5, dtype=float)

    # Ensure the initial vector satisfies the constraints
    h0 = _repair_solution(h0)

    # Candidate grid sizes (start with current, then larger)
    candidate_ns = [h0.size, 60, 90]
    best_overall_h = h0
    best_overall_val = _compute_c5(h0)

    for n_target in candidate_ns:
        # time budget slice for this grid size
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining <= 5.0:
            break

        # interpolate previous best to the new grid if size changes
        if n_target != best_overall_h.size:
            x_old = np.linspace(0.0, 2.0, best_overall_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
            h_init = np.interp(x_new, x_old, best_overall_h)
        else:
            h_init = best_overall_h.copy()

        # enforce constraints on the interpolated vector
        h_init = _repair_solution(h_init)

        # allocate a slice of the remaining time (at most half)
        slice_time = min(remaining * 0.5, 300.0)

        h_opt, val_opt = _optimize_h(h_init, slice_time)

        if val_opt < best_overall_val - 1e-6:
            best_overall_h, best_overall_val = h_opt, val_opt

        # early exit if we already beat the target bound
        if best_overall_val <= 0.3808:
            break

    # final repair to guarantee exact constraints
    best_overall_h = _repair_solution(best_overall_h)
    final_val = _compute_c5(best_overall_h)

    return best_overall_h, float(final_val), int(best_overall_h.size)
