import time, random, numpy as np

def _c5_score(h, dx):
    """Maximum overlap integral for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _project_to_bounds_and_sum(h, target_sum):
    """Clamp to [0,1] and adjust to hit exact sum (tiny tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # distribute diff to entries with slack
    slack_up = 1.0 - h
    slack_down = h
    if diff > 0:
        # need to increase some entries
        idx = np.where(slack_up > 1e-12)[0]
        if idx.size == 0:
            return h
        inc = diff / idx.size
        inc = np.minimum(inc, slack_up[idx])
        h[idx] += inc
    else:
        # need to decrease some entries
        idx = np.where(slack_down > 1e-12)[0]
        if idx.size == 0:
            return h
        dec = (-diff) / idx.size
        dec = np.minimum(dec, slack_down[idx])
        h[idx] -= dec
    # final tiny correction to exact sum
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # add residual to a random free slot
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            h[random.choice(free.tolist())] += residual
        else:
            # fallback: adjust first element within bounds
            h[0] = np.clip(h[0] + residual, 0.0, 1.0)
    return h

def _propose_move(h, max_step=0.1):
    """Return a new h after a random two‑index transfer respecting bounds."""
    n = h.size
    i, j = random.sample(range(n), 2)
    # delta added to i, subtracted from j
    low = max(-max_step, -h[i], h[j] - 1.0)
    high = min(max_step, 1.0 - h[i], h[j])
    if low >= high:
        return None  # no feasible move
    delta = random.uniform(low, high)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new
