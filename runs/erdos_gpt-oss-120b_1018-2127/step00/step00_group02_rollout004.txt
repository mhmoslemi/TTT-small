import time
import numpy as np

def compute_c5(h):
    """Return the discrete C5 value for step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def hill_climb(initial_h, time_limit, start_time):

    best_h = h.copy()

    target_sum = n / 2.0
    rng = np.random.default_rng()

    while time.time() - start_time < time_limit:
