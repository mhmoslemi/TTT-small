import time
import numpy as np

def compute_c5(h):
    """Return the maximal overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_to_feasible(h, target_sum):
    """
    Clip to [0,1] and adjust to hit exactly target_sum.
    Simple water‑filling style iterative projection.
    """
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # iterative correction
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        # indices that can move in the needed direction
        if diff > 0:
            free = np.where(h < 1.0 - 1e-14)[0]
        else:
            free = np.where(h > 1e-14)[0]
        if free.size == 0:
            break
        # distribute equally
        add = diff / free.size
        for i in free:
            if diff > 0:
                room = 1.0 - h[i]
                inc = min(add, room)
            else:
                room = -h[i]
                inc = max(add, room)
            h[i] += inc
            diff -= inc
        # recompute diff for safety
        diff = target_sum - h.sum()
    # final tiny correction by adding to first free element if needed
    if abs(diff) > 1e-12:
        i = np.argmin(h) if diff > 0 else np.argmax(h)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def random_perturb(h, rng, step_std):
    """Create a new candidate by a small random move that keeps the sum."""
    n = h.shape[0]
    h_new = h.copy()
    i = rng.integers(n)
    delta = rng.normal(0.0, step_std)
    h_new[i] += delta
    # clip the perturbed entry
    h_new[i] = np.clip(h_new[i], 0.0, 1.0)
    # enforce the sum constraint by projecting
    target = n / 2.0
    h_new = project_to_feasible(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic search for a step function h on [0,2] that minimizes
    C_5 = max_k ∫ h(x)(1-h(x+k)) dx.

    """
