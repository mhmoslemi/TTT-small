import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

# evaluate_erdos_solution is pre‑imported; we may use it for final verification
# initial_h_values may be pre‑imported; use if present

def _compute_c5(h, dx):
    """Return the overlap constant C5 for a discrete step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_feasible(vec, target_sum):
    """Project a vector onto the box [0,1] while forcing the sum to target_sum."""
    # First clip to [0,1]
    v = np.clip(vec, 0.0, 1.0)
    # Simple iterative scaling to meet the sum (works because target_sum ≤ len(v))
    cur = v.sum()
    if cur == 0:
        # fill uniformly
        v[:] = target_sum / len(v)
        return v
    scale = target_sum / cur
    v *= scale
    # After scaling some entries may leave the box; fix by a short projection loop
    for _ in range(5):
        over = v > 1.0
        under = v < 0.0
        if not (over.any() or under.any()):
            break
        excess = v[over] - 1.0
        deficit = -v[under]
        # redistribute excess to deficit proportionally
        total_excess = excess.sum()
        total_deficit = deficit.sum()
        if total_excess > 0 and total_deficit > 0:
            delta = min(total_excess, total_deficit)
            v[over] -= excess * (delta / total_excess)
            v[under] += deficit * (delta / total_deficit)
        v = np.clip(v, 0.0, 1.0)
    # final renormalisation if slight drift
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        # distribute uniformly while keeping bounds
        add = diff / len(v)
        v += add
        v = np.clip(v, 0.0, 1.0)
    return v

def _initial_feasible(n_points, target_sum):
    """Create a simple feasible start: block of ones then a fractional entry."""
    h = np.zeros(n_points)
    full = int(target_sum)
    h[:full] = 1.0
    if full < n_points:
        h[full] = target_sum - full
    return h

def _interpolate_initial(target_n):
    """Interpolate the global variable initial_h_values to target_n points."""
    try:
        src = initial_h_values  # may raise NameError
    except NameError:
        return None
    src = np.asarray(src, dtype=float)
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    # grid on [0,2)
    src_x = np.linspace(0.0, 2.0, src_n, endpoint=False)
    tgt_x = np.linspace(0.0, 2.0, target_n, endpoint=False)
    interp = np.interp(tgt_x, src_x, src, period=2.0)
    # enforce feasibility
    target_sum = target_n / 2.0
    interp = _project_feasible(interp, target_sum)
    return interp

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 minimizing the Erdős C₅."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a modest resolution; increase if time permits
    n_points = 94                     # even number gives integer target sum
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ------------------------------------------------------------------
    # 1. Build an initial feasible population member
    # ------------------------------------------------------------------
    h0 = _interpolate_initial(n_points)
    if h0 is None:
        h0 = _initial_feasible(n_points, target_sum)

    # ------------------------------------------------------------------
    # 2. Global search with Differential Evolution (penalised objective)
    # ------------------------------------------------------------------
    penalty_coef = 1e6

    def penalised_obj(vec):
        # vec is a candidate h (may violate sum)
        c5 = _compute_c5(vec, dx)
        pen = penalty_coef * abs(vec.sum() - target_sum)
        return c5 + pen

    bounds = [(0.0, 1.0)] * n_points
    # Allocate time roughly 2/3 to DE
    time_for_de = (budget_s - (time.time() - start_time)) * 0.66
    max_iter = max(5, int(time_for_de / 0.05))   # very rough conversion
    # Ensure a sane upper limit
    max_iter = min(max_iter, 300)

    # custom init population – start with h0 and random feasible vectors
    popsize = max(10, n_points // 5)
    init_pop = [h0]
    while len(init_pop) < popsize:
        rand = rng.random(n_points)
        rand = _project_feasible(rand, target_sum)
        init_pop.append(rand)

    result_de = differential_evolution(
        penalised_obj,
        bounds,
        strategy="best1exp",
        maxiter=max_iter,
        popsize=popsize,
        init=init_pop,
        polish=False,
        seed=rng,
        atol=0,
        updating="deferred",
        workers=1,
    )
    best_vec = result_de.x
    # Project to exact feasibility before local refinement
    best_vec = _project_feasible(best_vec, target_sum)

    # ------------------------------------------------------------------
    # 3. Local refinement with SLSQP (equality constraint on the sum)
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed

    # Final evaluation

    return (best_vec, float(final_c5), int(n_points))
