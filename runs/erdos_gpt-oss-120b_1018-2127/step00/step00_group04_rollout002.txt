import numpy as np
from scipy.optimize import differential_evolution

def _project_feasible(v, target_sum, eps=1e-12, max_iter=20):
    """Clamp v to [0,1] and adjust to meet exact sum = target_sum."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) < eps:
            break
        free = (v > eps) & (v < 1.0 - eps)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    cur = v.sum()
    if abs(cur - target_sum) > eps:
        scale = target_sum / cur if cur > 0 else 0.0

    return v

# ----------------------------------------------------------------------
def _c5_value(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for discretised h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

# ----------------------------------------------------------------------
def _objective(x, dx, target_sum):
    """Objective for the optimizer: returns C5 for the feasible projection of x."""
    h = _project_feasible(x, target_sum)
    return _c5_value(h, dx)
