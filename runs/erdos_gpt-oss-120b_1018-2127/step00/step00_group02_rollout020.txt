import time
import numpy as np
import random

def compute_c5(h):
    """Return C5 for a given step function sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # full linear (non‑circular) correlation
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_constraints(h):
    """Clip to [0,1] and then enforce the exact sum n/2 by a simple redistribution."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if np.isclose(cur, target, atol=1e-12):
        return h
    # redistribute the deficit/excess uniformly over non‑saturated entries
    mask = (h > 0.0) & (h < 1.0)
    free = np.where(mask)[0]
    # if no free variables (unlikely) fall back to uniform 0.5
    if free.size == 0:
        return np.full(n, 0.5)
    diff = target - cur
    delta = diff / free.size
    h[free] += delta
    # a second clipping pass (rarely needed)
    h = np.clip(h, 0.0, 1.0)
    # final correction by adjusting the first free entry to hit the sum exactly
    correction = target - h.sum()
    if free.size:
        h[free[0]] += correction
    return np.clip(h, 0.0, 1.0)

def upscale(h_old, n_new):
    """Interpolate an existing vector to a larger grid (periodic extension)."""
    n_old = h_old.shape[0]
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_to_constraints(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # decide grid size
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values, dtype=float)
        n0 = base_h.shape[0]
        # try a finer grid if original is small
        n_points = max(80, n0 * 2)
        h = upscale(base_h, n_points)
    else:
        # fallback: even grid with uniform 0.5
        n_points = 80
        h = np.full(n_points, 0.5, dtype=float)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # annealing schedule
    temp_start = 0.02
    temp_end = 0.0005
    max_iters = 200000  # safety cap
    it = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98 or it >= max_iters:
            break
        # linear temperature decay
        t = temp_start + (temp_end - temp_start) * (elapsed / budget_s)

        # propose a move: transfer eps from j to i
        eps = np.random.uniform(0.0, 0.02)
        i, j = random.sample(range(n_points), 2)
        if h[i] + eps > 1.0 or h[j] - eps < 0.0:
            # reject infeasible elementary move
            it += 1
            continue
        h_new = h.copy()
        h_new[i] += eps
        h_new[j] -= eps
        # still respects sum, only need box clipping check
        # (already ensured by eps bound)
        new_score = compute_c5(h_new)

        # acceptance rule
        if new_score < best_score:
            best_h = h_new.copy()
            best_score = new_score
            h = h_new
        else:
            # probabilistic uphill move
            if np.exp(-(new_score - best_score) / (t + 1e-12)) > np.random.rand():
                h = h_new

        it += 1

    # final projection (safety)
    best_h = project_to_constraints(best_h)

    return best_h.tolist(), float(best_score), n_points
