import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Exact overlap bound for a step function sampled on a grid."""
    # correlation of h with (1-h); mode='full' gives lags from -(n-1) to n-1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def upscale(h, new_n):
    """Linear interpolation of an existing grid to a finer grid."""
    old_n = h.shape[0]
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # renormalise to satisfy the integral constraint exactly
    dx_new = 2.0 / new_n
    h_new *= (1.0 / (np.sum(h_new) * dx_new))
    return h_new

def enforce_constraints(h, dx, max_iter=10):
    """Project h onto the feasible set: 0≤h≤1 and sum(h)*dx = 1."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = np.sum(h) * dx
        diff = 1.0 - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # no free variable – distribute proportionally (small perturbation)
            h = np.clip(h + diff / h.size, 0.0, 1.0)
            continue
        delta = diff / (n_free * dx)
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def random_feasible(n, dx, rng):
    """Generate a random feasible vector (uniform Dirichlet then clip)."""
    # start from Dirichlet
    v = rng.random(n)
    v = np.clip(v, 0.0, 1.0)
    v = enforce_constraints(v, dx)
    return v

def optimize_on_grid(init_h, dx, rng, time_limit):
    """Run a few SLSQP attempts starting from init_h and random perturbations."""
    n = init_h.shape[0]
    bounds = [(0.0, 1.0) for _ in range(n)]
    eq_cons = {
        'type': 'eq',
        'fun': lambda h: np.sum(h) * dx - 1.0
    }

    best_h = init_h.copy()
    best_score = compute_c5(best_h, dx)

    start = time.time()
    attempts = 0
    while time.time() - start < time_limit and attempts < 8:
        attempts += 1
        # perturbation for diversity
        if attempts == 1:
            x0 = init_h.copy()
        else:
            noise = rng.normal(scale=0.05, size=n)
            x0 = init_h + noise
            x0 = np.clip(x0, 0.0, 1.0)
            x0 = enforce_constraints(x0, dx)

        res = minimize(
            fun=lambda h: compute_c5(h, dx),
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[eq_cons],
            options={'ftol': 1e-9, 'maxiter': 250, 'disp': False}
        )
        if not res.success:
            continue
        h_candidate = enforce_constraints(res.x, dx)
        score = compute_c5(h_candidate, dx)
        if score < best_score:
            best_score = score
            best_h = h_candidate.copy()
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    best_overall_h = None
    best_overall_score = np.inf
    best_n = None

    # Determine whether an initial construction exists
    try:
        init_parent = initial_h_values  # provided by harness

    except NameError:
        init_parent = None

    # Candidate grid sizes (keep them modest for speed)
    candidate_ns = [45, 60, 80]

    # Allocate budget roughly evenly among grid sizes
    time_per_n = budget_s / len(candidate_ns)

    for n in candidate_ns:
        dx = 2.0 / n
        # Build an initial feasible vector for this grid
        if have_parent:
            h0 = upscale(init_parent, n)
        else:
            h0 = random_feasible(n, dx, rng)

        # Give each grid size a time slice
        remaining = budget_s - (time.time() - start_time)
        if remaining <= 0:

        slice_time = min(time_per_n, remaining)

        h_opt, score = optimize_on_grid(h0, dx, rng, slice_time)

        if score < best_overall_score:
            best_overall_score = score
            best_overall_h = h_opt
            best_n = n

        # Early exit if we already beat the target bound comfortably

    return best_overall_h, float(best_overall_score), int(best_n)
