import time
import numpy as np
import random

def _evaluate_overlap(h):
    """Return C5 for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all lags
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h on [0,2] to minimize the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimized values on a uniform grid.
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        # fallback: uniform 0.5 on a modest grid
        n = 120
        h = np.full(n, 0.5, dtype=float)

    n = h.size
    dx = 2.0 / n

    # enforce exact integral constraint (sum = n/2)
    target_sum = n / 2.0
    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-12:
        h *= target_sum / current_sum

    # ------------------------------------------------------------------
    # optimisation parameters
    max_step = 0.03          # maximal mass moved in one proposal
    temp_start = 0.02        # initial temperature for simulated annealing
    temp_end = 1e-4          # final temperature
    eval_fn = _evaluate_overlap

    best_h = h.copy()
    best_score = eval_fn(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    it = 0
    while time.time() < deadline:
        it += 1
        # linear temperature schedule
        t = (time.time() - start_time) / budget_s
        temp = temp_start * (1 - t) + temp_end * t

        # propose moving a small amount of mass from i to j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # ensure move respects bounds
        max_delta = min(max_step, cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_score = eval_fn(new_h)

        delta_score = new_score - cur_score
        accept = False
        if delta_score < 0:
            accept = True
        else:
            # probabilistic acceptance of worse move
            if temp > 0 and np.exp(-delta_score / temp) > np.random.rand():
                accept = True

        if accept:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score

        # occasional random restart if stagnation detected
        if it % 50000 == 0:
            # keep best so far but add small jitter
            jitter = np.random.uniform(-max_step, max_step, size=n)
            jitter = np.clip(jitter, -best_h, 1.0 - best_h)
            cur_h = np.clip(best_h + jitter, 0.0, 1.0)
            # re‑normalise to preserve exact sum
            cur_h *= target_sum / cur_h.sum()
            cur_score = eval_fn(cur_h)

    return (best_h, float(best_score), n)
