import time
import numpy as np
from scipy.optimize import minimize

def compute_score(h):
    """Maximum overlap integral for step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def resample_h(old_h, n_new):
    """Linear resampling of old_h (defined on [0,2)) to length n_new."""
    n_old = len(old_h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, old_h)

def feasible_initial(n):
    """Fallback feasible start: constant 0.5 (integral 1)."""
    return np.full(n, 0.5, dtype=float)

def minimize_score(h0, n, max_iter=500):
    """Run SLSQP to reduce the max overlap, respecting bounds and integral."""
    bounds = [(0.0, 1.0)] * n
    constraints = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    res = minimize(
        compute_score,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
    )
    return res

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function minimizing C5."""
    np.random.seed(seed)
    start = time.time()
    # resolutions to explore (keep arrays small)
    n_list = [30, 45, 60, 90, 180]

    best_h = None
    best_score = np.inf

    # starting point for n=45 if provided
    base_h = None
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, copy=True)
        # ensure feasibility of provided values (tiny numeric slack allowed)
        if not np.isclose(np.sum(base_h), len(base_h) / 2.0, atol=1e-6):
            # project onto the integral constraint by uniform scaling
            base_h *= (len(base_h) / 2.0) / np.sum(base_h)
        base_h = np.clip(base_h, 0.0, 1.0)
        best_h = base_h
        best_score = compute_score(base_h)

    for n in n_list:
        if time.time() - start > budget_s:
            break

        # obtain a feasible starting vector for this resolution
        if best_h is not None:
            h0 = resample_h(best_h, n)
        elif n == 45 and base_h is not None:
            h0 = base_h.copy()
        else:
            h0 = feasible_initial(n)

        # enforce the integral constraint by simple scaling (SLSQP will correct bounds)
        scale = (n / 2.0) / np.sum(h0)
        h0 = np.clip(h0 * scale, 0.0, 1.0)

        # local optimisation
        res = minimize_score(h0, n, max_iter=800)

        if not res.success:
            # try a quick random restart if optimisation failed
            rand = np.random.rand(n)
            rand *= (n / 2.0) / np.sum(rand)
            res = minimize_score(rand, n, max_iter=400)

        if res.success:
            cand = res.x
            cand_score = compute_score(cand)
            if cand_score < best_score - 1e-9:
                best_score = cand_score
                best_h = cand.copy()

        # occasional random perturbation to escape local minima
        if (time.time() - start) < budget_s and np.random.rand() < 0.15:
