import numpy as np
import time
import math
import random

def _compute_c5(h):
    """Return C5 for discrete step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _project_bounds_and_sum(h, target_sum):
    """Project h to [0,1] and enforce exact sum = target_sum (simple scaling loop)."""
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative scaling to meet sum while staying in bounds
    for _ in range(10):
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        scale = target_sum / cur if cur != 0 else 0.0
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    # Final tiny correction to hit sum exactly (adds/subtracts equally to non‑boundary entries)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size:
            add = diff / free.size
            h[free] += add
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    rnd = random.Random(seed)
    np_random = np.random.default_rng(seed)

    # Determine problem size
    try:
        n0 = len(initial_h_values)  # provided by harness if available
        h_current = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 60
        h_current = np.full(n0, 0.5, dtype=float)

    n = n0
    target_sum = n / 2.0

    # Ensure feasibility
    h_current = _project_bounds_and_sum(h_current, target_sum)

    best_h = h_current.copy()
    best_c5 = _compute_c5(best_h)

    start_time = time.time()
    end_time = start_time + budget_s * 0.95  # keep a safety margin

    # Annealing schedule parameters
    temp_start = 0.05
    temp_end = 1e-5
    max_iter = 2000000  # large ceiling, will stop by time

    for it in range(1, max_iter + 1):
        now = time.time()
        if now >= end_time:
            break
        # linear cooling
        t = temp_start + (temp_end - temp_start) * (it / max_iter)

        # propose a transfer between two random indices
        i, j = rnd.sample(range(n), 2)
        hi, hj = h_current[i], h_current[j]

        # maximum transferable amount without violating bounds
        max_transfer = min(hi, 1.0 - hj)
        if max_transfer <= 0:
            continue
        eps = rnd.random() * max_transfer * 0.6  # step size up to 60% of allowed

        h_new = h_current.copy()
        h_new[i] = hi - eps
        h_new[j] = hj + eps

        # small random jitter on a third index to increase mixing
        k = rnd.randrange(n)
        hk = h_new[k]
        delta = (rnd.random() - 0.5) * 0.02  # +-0.01
        new_val = hk + delta
        if 0.0 <= new_val <= 1.0:
            h_new[k] = new_val

        # project only if numerical drift occurs
        if abs(h_new.sum() - target_sum) > 1e-9:
            h_new = _project_bounds_and_sum(h_new, target_sum)
