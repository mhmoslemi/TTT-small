import time
import numpy as np

def _compute_c5_fft(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral using FFT for speed."""
    n = h.size
    m = 2 * n  # zero‑padding length for full correlation
    # FFT of h and reversed (1-h)
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(np.flip(1.0 - h), m)
    corr = np.fft.irfft(H * G, m)[: 2 * n - 1]  # full mode
    return float(corr.max() * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimise the
    Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimised values of h on the grid.
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of sample points used.
    """
    rng = np.random.default_rng(seed)

    # Determine grid size. Use the size of the provided parent if present,
    # otherwise a modest default.
    try:
        # initial_h_values is supplied by the harness (may be None)
        parent = np.asarray(initial_h_values, dtype=np.float32)
        n = parent.size
        h = parent.copy()
    except Exception:
        n = 70                     # default grid size (fits the “<1k” guideline)
        h = np.full(n, 0.5, dtype=np.float32)  # uniform feasible start

    # Enforce feasibility exactly (sum = n/2, bounds [0,1])
    target_sum = n / 2.0
    # Clip then correct the sum by a uniform shift; repeat once if needed.
    h = np.clip(h, 0.0, 1.0)
    shift = (target_sum - h.sum()) / n
    h += shift
    # Minor correction for any rounding error
    excess = h.sum() - target_sum
    if abs(excess) > 1e-12:
        # redistribute the tiny excess to the first element within bounds
        i = 0
        while i < n and (h[i] - excess < 0 or h[i] - excess > 1):
            i += 1
        if i < n:
            h[i] -= excess

    dx = 2.0 / float(n)

    best_h = h.copy()
    best_score = _compute_c5_fft(best_h, dx)

    start_time = time.time()
    max_time = budget_s * 0.95  # keep a safety margin

    # Parameters for the stochastic local search
    max_no_improve = 5000
    no_improve = 0

    while time.time() - start_time < max_time:
        # Propose a simple mass‑transfer move that keeps the total sum constant
        i, j = rng.choice(n, size=2, replace=False)
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 1e-12:

        eps = rng.random() * max_transfer
        new_h = best_h.copy()
        new_h[i] -= eps
