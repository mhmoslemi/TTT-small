def repair(h, target):
    h = np.clip(h, 0.0, 1.0)
    S = h.sum()
    if S == 0:
        # set uniform
        h = np.full_like(h, target / len(h))
        return h
    factor = target / S
    h = h * factor
    # clip again
    h = np.clip(h, 0.0, 1.0)
    # adjust residual
    residual = target - h.sum()
    if abs(residual) > 1e-8:
        # distribute residual to entries not at bound

        # add equally

        h[free] += delta
        # clip again
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import random
import math
import numpy as np

def _compute_c5(h):
    """Return the C5 value for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def _repair(h, target):
    """Project h onto the box [0,1] with the exact sum constraint = target."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        h[:] = target / h.size
        return h
    # simple scaling + residual fix
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-9:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            free = np.arange(h.size)
        delta = residual / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # --- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 80
        h = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    h = _repair(h, target_sum)
    best_h = h.copy()
    best_score = _compute_c5(best_h)

    # --- annealing schedule -------------------------------------------------
    T0 = 1.0
    cooling = 0.9995
    iteration = 0
    max_iter = 10_000_000  # safety cap; loop will break on time

    # probability of a full re‑initialisation (helps escape local minima)
    restart_prob = 0.001

    while True:
        # time check
        if time.time() - start_time > budget_s * 0.99:
            break
        if iteration >= max_iter:
            break
        iteration += 1

        # temperature
        T = max(T0 * (cooling ** iteration), 1e-8)

        # occasional full restart
        if random.random() < restart_prob:
            h_candidate = np.full(n, 0.5, dtype=float)
        else:
            # perturb a random subset of entries
            h_candidate = best_h.copy()
            k = max(1, int(0.05 * n))                     # 5 % of coordinates
            idx = np.random.choice(n, k, replace=False)
            # step size scales with temperature
            step = 0.2 * T
            h_candidate[idx] += np.random.randn(k) * step

        # enforce feasibility
        h_candidate = _repair(h_candidate, target_sum)

        # evaluate
        score = _compute_c5(h_candidate)

        # acceptance (simulated annealing)
        if score < best_score:
            best_h, best_score = h_candidate, score
        else:
            # metropolis criterion
            delta = best_score - score
            if math.exp(delta / T) > random.random():
                best_h, best_score = h_candidate, score

    # final post‑processing: ensure exact sum (tiny residuals)
    best_h = _repair(best_h, target_sum)
    best_score = _compute_c5(best_h)

    return best_h, best_score, n
