import time
import numpy as np

def _evaluate(h, dx):
    """Return max overlap C5 for a given h."""
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(conv)


def _project_to_bounds_and_sum(h, target_sum, tol=1e-9, max_iter=100):
    """
    Project vector h onto the feasible set:
        0 <= h_i <= 1
        sum(h) = target_sum
    Uses a simple iterative redistribution of excess/deficit.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) < tol:
            break
        # indices that can move in the needed direction
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final clip (numerical safety)
    h = np.clip(h, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimized sample values of h (length n_points).
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of discretisation points used.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine number of points: use the size of initial_h_values if present,
    # otherwise fall back to a modest size (e.g., 200) which balances speed and
    # resolution.
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.size
    except NameError:
        n = 200
        h0 = rng.random(n)

    dx = 2.0 / n
    target_sum = n / 2.0  # because integral of h must be 1

    # Ensure feasible start
    h_cur = _project_to_bounds_and_sum(h0, target_sum)
    cur_score = _evaluate(h_cur, dx)

    best_h = h_cur.copy()
    best_score = cur_score

    # Simulated‑annealing / stochastic‑hill‑climbing parameters
    max_iters = 10_000_000  # safety cap
    iter_cnt = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.5:  # keep a little slack
            break
        if iter_cnt >= max_iters:
            break

        # Annealing schedule
        progress = iter_cnt / max_iters
        temp = max(1e-4, (1.0 - progress) * 0.02)      # temperature
        step = (1.0 - progress) * 0.15                # noise magnitude

        # Propose a new candidate
        proposal = h_cur + step * rng.standard_normal(n)
        proposal = _project_to_bounds_and_sum(proposal, target_sum)

        proposal_score = _evaluate(proposal, dx)

        # Acceptance rule
        if proposal_score < cur_score:
            accept = True
        else:
            delta = proposal_score - cur_score
            accept = rng.random() < np.exp(-delta / temp)

        if accept:
            h_cur = proposal
            cur_score = proposal_score
            if cur_score < best_score:
                best_h = h_cur.copy()
                best_score = cur_score

        iter_cnt += 1

    # Final evaluation (in case of rounding drift)
    best_h = _project_to_bounds_and_sum(best_h, target_sum)
    best_score = _evaluate(best_h, dx)

    return best_h, float(best_score), n
