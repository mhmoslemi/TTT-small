import time
import numpy as np

def compute_c5(h):
    """Compute the overlap integral C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] that minimizes the Erdős C5 overlap bound.

    Returns
    -------
    h_values : np.ndarray
        The best step function found (values in [0,1]).
    c5_bound : float
        The corresponding C5 value.
    n_points : int
        Number of discretisation points (length of h_values).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise h ----------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        # fallback: uniform 0.5 on a modest grid
        n0 = 100
        h = np.full(n0, 0.5, dtype=np.float64)

    n = h.shape[0]

    # enforce the integral constraint sum(h) = n/2  (average 0.5)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if cur_sum != 0:
        h *= target_sum / cur_sum
    # clip to [0,1] after scaling (rarely needed) and re‑project to exact sum
    h = np.clip(h, 0.0, 1.0)
    excess = h.sum() - target_sum
    if abs(excess) > 1e-12:
        # distribute the excess uniformly (maintains bounds because excess is tiny)
        h -= excess / n
        h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # ----- simple simulated‑annealing style search -------------------------
    # parameters
    max_iter = 200000          # upper bound, actual loop ends by time
    delta_scale = 0.02         # maximal magnitude of a transfer
    temp0 = 0.01               # initial temperature for occasional uphill moves
    cooling = 0.9999

    temperature = temp0
    it = 0
    while time.time() - start < budget_s - 0.5 and it < max_iter:
        i, j = rng.choice(n, size=2, replace=False)
        delta = (rng.random() - 0.5) * delta_scale  # in [-delta_scale/2, delta_scale/2]

        hi, hj = h[i], h[j]
        hi_new = hi - delta
        hj_new = hj + delta

        # keep within [0,1]; reject otherwise
        if not (0.0 <= hi_new <= 1.0 and 0.0 <= hj_new <= 1.0):
            it += 1
            continue

        # apply provisional move
        h[i] = hi_new
        h[j] = hj_new

        new_score = compute_c5(h)

        # accept if improvement, else maybe accept uphill with probability
        accept = False
        if new_score < best_score:
            accept = True
        else:
            prob = np.exp(-(new_score - best_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert move
            h[i] = hi
            h[j] = hj

        # temperature schedule
        temperature *= cooling
        it += 1

    # final evaluation (use the pre‑imported helper if present for consistency)
    try:
        # evaluate_erdos_solution may provide a more precise bound
        from __main__ import evaluate_erdos_solution  # may be pre‑imported
        final_score = float(evaluate_erdos_solution(best_h))
        # safety: ensure it is not worse than our direct computation
        final_score = min(final_score, best_score)
    except Exception:
        final_score = best_score

    return best_h, final_score, best_h.shape[0]
