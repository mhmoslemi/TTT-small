import numpy as np
import time

def _project_feasible(x, target_sum):
    """Project vector x onto [0,1]^n with given sum using a simple bisection."""
    lo = -np.max(x)               # make all entries >=0
    hi = 1.0 - np.min(x)           # make all entries <=1
    for _ in range(60):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def _c5_value(h):
    """Return C5 for a discrete step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
