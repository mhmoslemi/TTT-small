import time
import numpy as np
from scipy.optimize import minimize

def compute_score(h, n):
    """Maximum overlap integral for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def constraint_sum(x, n):
    """Equality constraint: integral equals 1 (sum = n/2)."""
    return np.sum(x) - n / 2.0

def resample_solution(old_h, old_n, new_n):
    """Linear interpolation of a solution from old_n points to new_n points."""
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)

def make_random_feasible(n, rng):
    """Generate a random feasible vector (bounds [0,1], sum = n/2)."""
    h = rng.random(n)
    # Scale to the required sum
    h = h * (n / 2.0) / np.sum(h)
    # Project onto box and adjust a few times to respect the sum
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        diff = (n / 2.0) - np.sum(h)
        h += diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def optimize_h(initial_h, n, rng):
    """Run SLSQP to improve a candidate step function."""
    bounds = [(0.0, 1.0)] * n
    constraints = {'type': 'eq', 'fun': constraint_sum, 'args': (n,)}
    opt = minimize(
        compute_score,
        initial_h,
        args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False},
    )
    return opt

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Resolutions to explore (small to moderate)
    resolutions = [30, 45, 60, 90, 180]

    best_h = None
    best_score = np.inf
    best_n = None

    # Keep the latest feasible solution to seed the next resolution
    last_h = None
    last_n = None

    for n in resolutions:
        if time.time() - start_time > budget_s:
            break

        # Build an initial guess
        if n == 45 and 'initial_h_values' in globals():
            h0 = np.asarray(initial_h_values, dtype=float)
        elif last_h is not None:
            h0 = resample_solution(last_h, last_n, n)
        else:
            h0 = make_random_feasible(n, rng)

        # Ensure the guess respects bounds (SLSQP will enforce)
        h0 = np.clip(h0, 0.0, 1.0)

        # Quick evaluation of the current guess
        cur_score = compute_score(h0, n)

        # Optimize
        result = optimize_h(h0, n, rng)

        # Use optimizer output regardless of success flag (it respects constraints)
        h_opt = np.clip(result.x, 0.0, 1.0)
        opt_score = compute_score(h_opt, n)

        # Record the best solution found so far
        if opt_score < best_score:
            best_score = opt_score
            best_h = h_opt.copy()
            best_n = n

        # Prepare for the next resolution
        last_h = h_opt
        last_n = n

    # Fallback: if no solution was found (unlikely), use a simple uniform split
    if best_h is None:
        n = 45
        best_h = np.full(n, 0.5)
        best_score = compute_score(best_h, n)
        best_n = n

    return (best_h.tolist(), float(best_score), int(best_n))
