import time
import numpy as np

def _compute_c5(h):
    """Maximum overlap integral for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives 2n-1 lags
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _feasible_random(rng, n):
    """Generate a random feasible vector of length n with sum n/2 and values in [0,1]."""
    # start with Dirichlet samples scaled to sum n/2
    vals = rng.random(n)
    vals = vals / vals.sum() * (n / 2.0)
    # clip and redistribute any excess/deficit iteratively (few steps are enough)
    for _ in range(5):
        excess = vals - 1.0
        excess_mask = excess > 0
        deficit = -vals
        deficit_mask = deficit > 0
        if not excess_mask.any() and not deficit_mask.any():
            break
        # move excess to deficit proportionally
        total_excess = excess[excess_mask].sum()
        total_deficit = -deficit[deficit_mask].sum()
        move = min(total_excess, total_deficit)
        if move == 0:
            break
        # distribute move uniformly among eligible indices
        if excess_mask.any():
            vals[excess_mask] -= move * (excess[excess_mask] / total_excess)
        if deficit_mask.any():
            vals[deficit_mask] += move * (-deficit[deficit_mask] / total_deficit)
    # final clipping just in case
    vals = np.clip(vals, 0.0, 1.0)
    # final adjustment to exact sum
    diff = (n / 2.0) - vals.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly over non‑boundary entries
        mask = (vals > 0) & (vals < 1)
        if mask.any():
            vals[mask] += diff / mask.sum()
            vals = np.clip(vals, 0.0, 1.0)
    return vals

    """
    Optimize a step function h on [0,2] to minimise the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray

        Corresponding C5 value.
    n_points : int
        Number of discretisation points.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Obtain initial vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # default length 70, uniform area 1 → average 0.5
        n_default = 70
        h = np.full(n_default, 0.5, dtype=float)

    n = h.shape[0]
    # Ensure feasibility (sum = n/2, bounds [0,1])
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    target_sum = n / 2.0
    if abs(current_sum - target_sum) > 1e-12:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0.0, 1.0)
        # a tiny redistribution to hit the exact sum
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            mask = (h > 0) & (h < 1)
            if mask.any():
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # Parameters for the stochastic local search
    max_delta = 0.25          # maximum amount transferred between two positions
    inner_trials = 80        # proposals per outer iteration
    no_improve_limit = 2000  # restart after this many failed outer loops
    no_improve_cnt = 0

    while time.time() - start_time < budget_s - 0.5:
        improved = False
        for _ in range(inner_trials):
            i, j = rng.integers(0, n, size=2)
            if i == j:
