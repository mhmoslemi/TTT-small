import time, math, random
import numpy as np

def compute_c5(h):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx for the discrete step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_to_bounds_sum(v, target):
    """Project vector v onto [0,1] with prescribed sum = target."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(20):
        excess = x.sum() - target
        if abs(excess) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        if not np.any(free):
            # all variables are at a bound; distribute uniformly and clip again
            x -= excess / len(x)
            x = np.clip(x, 0.0, 1.0)
            break
        delta = excess / np.sum(free)
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
    h_values is a NumPy array of length n_points satisfying the constraints,
    and c5_bound is the achieved C5 value.
    """
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        n = int(initial_h_values.shape[0])
        h = np.array(initial_h_values, dtype=float)
        # enforce feasibility
        h = np.clip(h, 0.0, 1.0)
        target_sum = n / 2.0
        h = project_to_bounds_sum(h, target_sum)
    else:
        n = 200                     # fallback size (fast to evaluate)
        target_sum = n / 2.0
        # start with a binary vector (half ones, half zeros) when possible
        if n % 2 == 0:
            h = np.zeros(n, dtype=float)
            h[:n // 2] = 1.0
            np.random.shuffle(h)
        else:
            # odd n: random numbers then project
            h = np.random.rand(n)
            h = project_to_bounds_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    # indices of non‑boundary entries for swaps (binary case)
    def update_indices(arr):
        ones = np.where(arr > 0.5)[0]
        zeros = np.where(arr <= 0.5)[0]
        return ones, zeros
    ones_idx, zeros_idx = update_indices(current_h)

    # Simulated‑annealing loop
    while elapsed() < budget_s * 0.95:
        # Occasionally restart from a fresh random feasible point
        if random.random() < 0.001:
            rnd = np.random.rand(n)
            current_h = project_to_bounds_sum(rnd, target_sum)
            current_c5 = compute_c5(current_h)
            ones_idx, zeros_idx = update_indices(current_h)

        # Propose a swap (preserves sum automatically for binary vectors)
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            # fall back to a small random perturbation respecting bounds
            i = random.randrange(n)
            delta = (random.random() - 0.5) * 0.1  # up/down move
            cand = current_h.copy()
            cand[i] = np.clip(cand[i] + delta, 0.0, 1.0)
