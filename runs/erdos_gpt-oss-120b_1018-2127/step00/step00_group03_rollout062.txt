def _make_binary(vec, target_sum):
    """Return a binary (0/1) vector with exactly target_sum ones, close to vec."""
    # start by thresholding
    bin_vec = (vec > 0.5).astype(np.float64)
    cur = int(bin_vec.sum())
    if cur > target_sum:
        # turn some ones to zero
        ones = np.where(bin_vec == 1)[0]
        drop = np.random.choice(ones, cur - target_sum, replace=False)
        bin_vec[drop] = 0.0
    elif cur < target_sum:
        zeros = np.where(bin_vec == 0)[0]
        add = np.random.choice(zeros, target_sum - cur, replace=False)
        bin_vec[add] = 1.0
    return bin_vec

def _c5_score(h, dx):
    """Compute the C5 value for a discrete step function h."""
    # correlation h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _anneal_once(h, dx, budget, start_time):
    """Run a simulated‑annealing loop for a single resolution."""
    N = h.size
    target_sum = N // 2
    best_h = h.copy()
    best_score = _c5_score(best_h, dx)

    # temperature schedule
    temp = 1.0
    it = 0
    while time.time() - start_time < budget:
        # pick a random 1 and a random 0 to swap
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        score = _c5_score(h, dx)

        delta = score - best_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-delta / temp) if temp > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

        # cool down
        temp *= 0.9999
        it += 1

        # occasional random restart if stuck

            # re‑initialize preserving total sum
            h[:] = 0.0
            idx = np.random.choice(N, target_sum, replace=False)
            h[idx] = 1.0

    return best_h, best_score
