def project_vector(x, total, lower=0.0, upper=1.0):
    x = np.clip(x, lower, upper)
    # adjust sum
    for _ in range(10):
        s = x.sum()
        diff = s - total
        if abs(diff) < 1e-8:
            break
        if diff > 0:
            # need to reduce
            mask = x > lower
            if not np.any(mask):
                break
            reducible = (x[mask] - lower).sum()
            if reducible == 0:
                break
            delta = diff * (x[mask] - lower) / reducible
            x[mask] -= delta
        else:
            # need to increase
            mask = x < upper
            if not np.any(mask):
                break
            incable = (upper - x[mask]).sum()
            delta = diff * (upper - x[mask]) / incable  # diff negative
            x[mask] -= delta  # subtract negative = add
        x = np.clip(x, lower, upper)
    return x

import time, math, random
import numpy as np

def _project_vector(x, total, lower=0.0, upper=1.0, max_iter=20, tol=1e-10):
    """Project x onto the box [lower,upper] with sum == total."""
    x = np.clip(x, lower, upper)
    for _ in range(max_iter):
        s = x.sum()
        diff = s - total
        if abs(diff) <= tol:
            break
        if diff > 0:  # need to decrease sum
            mask = x > lower
            if not np.any(mask):
                break
            reducible = (x[mask] - lower).sum()
            if reducible == 0:
                break
            delta = diff * (x[mask] - lower) / reducible
            x[mask] -= delta
        else:        # need to increase sum
            mask = x < upper
            if not np.any(mask):
                break
            incable = (upper - x[mask]).sum()
            if incable == 0:
                break
            delta = diff * (upper - x[mask]) / incable  # diff negative
            x[mask] -= delta
        x = np.clip(x, lower, upper)
    return x

def _c5_value(h):
    """Compute C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _upsample(arr, new_n):
    """Linear interpolation of arr (defined on [0,2)) to new_n points."""
    old_n = arr.shape[0]
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    return np.interp(x_new, x_old, arr)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with a low C5 bound."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # ----- initialise grid and starting vector -----------------
    if 'initial_h_values' in globals():
        init = initial_h_values.astype(float)
        n0 = init.size
        n = max(200, (n0 // 2) * 2)          # even, at least 200
        if n % 2:
            n += 1
        h = _upsample(init, n)
    else:
        n = 200
        h = np.full(n, 0.5, dtype=float)

    total = n / 2.0
    h = _project_vector(h, total)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ----- optimisation loop ---------------------------------
    max_iter = 200000
    temp0 = 0.05

    for it in range(max_iter):
        if time.time() - start > budget_s - 0.5:
            break

        # propose a perturbation
        h_new = best_h.copy()
        # number of coordinates to touch, grows smaller as n grows
        k = max(1, np.random.randint(1, max(2, n // 20)))
        idx = np.random.choice(n, size=k, replace=False)
        delta = np.random.normal(scale=0.2, size=k)
        h_new[idx] += delta
        h_new = _project_vector(h_new, total)

        new_c5 = _c5_value(h_new)

        # simulated‑annealing acceptance
        temperature = temp0 * (1.0 - it / max_iter)
        if new_c5 < best_c5 or np.random.rand() < math.exp((best_c5 - new_c5) / max(temperature, 1e-12)):
            best_h, best_c5 = h_new, new_c5

    return best_h, float(best_c5), int(n)
