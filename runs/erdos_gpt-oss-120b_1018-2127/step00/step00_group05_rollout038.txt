import time
import numpy as np
from scipy.optimize import minimize

def _corr_max(h, dx):
    """Maximum overlap integral for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _repair(h):
    """Project h onto the feasible set: 0≤h≤1, sum(h)=len(h)/2."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # Simple scaling then bound‑respecting redistribution
    s = h.sum()
    if s > 0:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

    # Final redistribution to hit the target exactly
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        # walk through entries and adjust while staying within bounds
        for i in range(n):
            if diff == 0.0:
                break
            if diff > 0.0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0.0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

def _objective(x, dx, target):

    h = np.clip(x, 0.0, 1.0)
