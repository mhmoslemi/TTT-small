import numpy as np
import time
import random
from math import exp

# The harness provides `initial_h_values` if a parent solution exists.
# It also provides `evaluate_erdos_solution` for optional verification,
# but we will rely on our own fast evaluation.

def _eval_c5(h, dx):
    """Return the C5 value for a given step vector h."""
    # correlation of h with (1-h); mode='full' yields length 2*n-1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)


def _project_binary(h):
    """Project a continuous vector onto the nearest binary vector with
    exactly sum(h) = n/2 (i.e., n/2 ones). Returns a binary array."""
    n = h.size
    target_ones = n // 2
    # argsort descending, take top target_ones as 1
    idx = np.argsort(-h, kind='stable')
    bin_h = np.zeros_like(h, dtype=float)
    bin_h[idx[:target_ones]] = 1.0
    return bin_h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized vector (values in [0,1]).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Length of the discretization.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    time_limit = budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1. Determine problem size and initial vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.size
        # enforce even length (required for integral constraint with binary)
        if n % 2 == 1:
            h0 = h0[:-1]
            n -= 1
    else:
        # default modest size, even
        n = 80
        h0 = np.full(n, 0.5, dtype=float)

    dx = 2.0 / n

    # Ensure feasibility: sum(h)*dx == 1  => sum(h) == n/2
    # We'll work with binary vectors; project initial to feasible binary.
    target_sum = n // 2
    # If h0 already feasible (within tolerance), keep it, else project.
    if not np.isclose(h0.sum(), target_sum):
        h_bin = _project_binary(h0)
    else:
        # clip to [0,1] then round to nearest binary preserving sum
        h_bin = np.clip(h0, 0.0, 1.0)
        # adjust if rounding changed sum
        diff = int(target_sum - h_bin.sum())
        if diff > 0:
            # need more ones: set smallest zero entries to 1
            zero_idx = np.where(h_bin < 0.5)[0]
            chosen = np.random.choice(zero_idx, diff, replace=False)
            h_bin[chosen] = 1.0
        elif diff < 0:
            # too many ones: turn some ones to zero
            one_idx = np.where(h_bin > 0.5)[0]
            chosen = np.random.choice(one_idx, -diff, replace=False)
            h_bin[chosen] = 0.0
        h_bin = h_bin.astype(float)

    best_h = h_bin.copy()
    best_c5 = _eval_c5(best_h, dx)

    # ------------------------------------------------------------------
    # 2. Simulated annealing on binary space (swap two positions)
    # ------------------------------------------------------------------
    # Parameters (tuned for speed & quality)
    max_iters = 300000           # absolute cap
    temp_start = 0.1
    temp_end = 1e-4
    # linear temperature schedule
    temp_decay = (temp_end / temp_start) ** (1.0 / max_iters)

    temperature = temp_start
    h_cur = best_h.copy()
    cur_c5 = best_c5

    # Pre‑compute indices of zeros and ones for quick swaps
    zeros = np.flatnonzero(h_cur == 0.0)
    ones = np.flatnonzero(h_cur == 1.0)

    it = 0
    while it < max_iters and (time.time() - start_time) < time_limit:
        it += 1

        # pick a random zero and a random one to swap
        if zeros.size == 0 or ones.size == 0:
            break
        i = np.random.choice(zeros)
        j = np.random.choice(ones)

        # propose swap
        h_cur[i], h_cur[j] = 1.0, 0.0
        new_c5 = _eval_c5(h_cur, dx)

        delta = new_c5 - cur_c5
        accept = False
        if delta < 0:
            accept = True
        else:
            if random.random() < exp(-delta / temperature):
                accept = True

        if accept:
            cur_c5 = new_c5
            # update best if improved
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
            # update zero/one index sets
            zeros = np.flatnonzero(h_cur == 0.0)
            ones = np.flatnonzero(h_cur == 1.0)
        else:
            # revert swap
            h_cur[i], h_cur[j] = 0.0, 1.0

        # temperature schedule
        temperature *= temp_decay

    # ------------------------------------------------------------------
    # 3. Optional local continuous refinement (projected gradient)
    # ------------------------------------------------------------------
    # Start from the best binary vector; allow small real adjustments
    # while keeping the integral constraint.
    h_cont = best_h.copy()
    lr = 0.02
    for _ in range(2000):
        # gradient of max correlation is approximated by the shift achieving the max
        conv = np.correlate(h_cont, 1.0 - h_cont, mode='full')
        max_idx = np.argmax(conv)
        # shift offset relative to original array
        shift = max_idx - (n - 1)   # integer shift in samples
        # derivative w.r.t. each element: for position p, contribution from overlap with shifted position
        # d/dh[p] = (1 - h[p+shift]) - h[p-shift]  (with appropriate bounds)
        grad = np.zeros_like(h_cont)
        for p in range(n):
            a = p + shift
            b = p - shift
            val = 0.0
            if 0 <= a < n:
                val += 1.0 - h_cont[a]
            if 0 <= b < n:
                val -= h_cont[b]
            grad[p] = val
        # gradient step
        h_cont += lr * grad
        # project to [0,1]
        np.clip(h_cont, 0.0, 1.0, out=h_cont)
        # enforce integral: scale to achieve sum = target_sum
        cur_sum = h_cont.sum()
        if cur_sum == 0:
            h_cont[:] = 0.5
            cur_sum = h_cont.sum()
        scale = target_sum / cur_sum
        h_cont *= scale
        np.clip(h_cont, 0.0, 1.0, out=h_cont)

        # evaluate
        new_c5 = _eval_c5(h_cont, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_cont.copy()
        # early stop if improvement stalls
        if abs(new_c5 - best_c5) < 1e-8:
            break

    # ------------------------------------------------------------------
    # 4. Return result
    # ------------------------------------------------------------------
    return best_h, float(best_c5), int(n)
