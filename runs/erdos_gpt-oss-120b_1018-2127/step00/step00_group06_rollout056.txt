import time
import numpy as np

# Helper: objective C5 for a discretized h
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# Project onto [0,1] and enforce sum(h) = n/2
def repair(h):
    n = h.size
    target = n / 2.0
    # Clip to bounds
    np.clip(h, 0.0, 1.0, out=h)
    # Iterative water‑filling to meet the sum constraint
    for _ in range(10):
        s = h.sum()
        delta = target - s
        if abs(delta) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:                     # everything at a bound
            h += delta / n
        else:
            h[free] += delta / cnt
        np.clip(h, 0.0, 1.0, out=h)
    # Final tiny adjustment if needed
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        np.clip(h, 0.0, 1.0, out=h)
        # enforce exact sum by distributing residual uniformly
        residual = target - h.sum()
        h += residual / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

# Linear interpolation from old grid to new grid size
def interpolate(h, new_n):
    old_n = h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

# Produce a random neighbor
def random_walk(h, scale, rng):
    cand = h + rng.normal(scale=scale, size=h.shape)
    return cand

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Grid schedule (starting from the size of any provided parent)
    grid_sizes = [95, 190, 380, 760]

    # Initialise with supplied parent if it matches the first grid size
    if 'initial_h_values' in globals():

            best_h = np.full(grid_sizes[0], 0.5, dtype=float)
    else:
        best_h = np.full(grid_sizes[0], 0.5, dtype=float)

    best_h = repair(best_h)
    best_score = compute_c5(best_h)

    # Main multigrid optimisation
    for idx, n in enumerate(grid_sizes):
