import time, numpy as np, math, random, sys
from scipy import interpolate

# The harness supplies evaluate_erdos_solution and possibly initial_h_values.

def _compute_c5(h, dx):
    """Return max overlap C5 for a given step vector h."""
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return conv.max()

def _perturb(h, max_step=0.02, rng=None):
    """Randomly move a small amount of mass between two positions while keeping bounds."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h  # nothing to do
    # amount we can take from i and give to j
    available = h[i]
    capacity = 1.0 - h[j]
    delta = max_step * rng.random()
    delta = min(delta, available, capacity)
    if delta <= 0:
        return h
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def _local_search(start_h, dx, budget_s, target=None):
    """Simple stochastic local search for a low C5."""
    rng = np.random.default_rng(12345)
    best_h = start_h.copy()
    best_c5 = _compute_c5(best_h, dx)
    start_time = time.time()
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        # occasional full random shuffle to avoid stagnation
        if it % 5000 == 0 and it > 0:
            perm = rng.permutation(best_h.size)
            cand = best_h[perm]
        else:
            cand = _perturb(best_h, max_step=0.02, rng=rng)
        cand_c5 = _compute_c5(cand, dx)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5
            # early exit if target achieved
            if target is not None and best_c5 <= target:
                break
        it += 1
    return best_h, best_c5
