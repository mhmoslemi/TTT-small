import time
import numpy as np
from scipy.optimize import minimize

# Helper: compute the C5 bound for a sample vector h
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# Helper: build an initial vector of length n with the required integral
def construct_simple_initial(n, rng):
    # exactly n/2 entries equal 1, rest 0, then random shuffle and small jitter
    half = n // 2
    h = np.zeros(n)
    h[:half] = 1.0
    rng.shuffle(h)
    # add tiny uniform noise then clip to keep within [0,1] and re‑project sum
    noise = rng.uniform(-0.02, 0.02, size=n)
    h = np.clip(h + noise, 0.0, 1.0)
    # project to exact sum n/2 by adding a constant offset then clipping
    target = n / 2.0
    diff = target - h.sum()
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    # final correction if needed (very rare)
    while abs(h.sum() - target) > 1e-9:
        diff = target - h.sum()
        adjust = diff / n
        h += adjust
        h = np.clip(h, 0.0, 1.0)
    return h

# Helper: upscale a coarse solution to a finer grid using linear interpolation
def upscale_solution(h_coarse, n_fine):
    n_coarse = h_coarse.size
    x_coarse = np.linspace(0.0, 2.0, n_coarse, endpoint=False)
    x_fine = np.linspace(0.0, 2.0, n_fine, endpoint=False)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    # enforce bounds and integral exactly
    h_fine = np.clip(h_fine, 0.0, 1.0)
    target = n_fine / 2.0
    diff = target - h_fine.sum()
    h_fine += diff / n_fine
    h_fine = np.clip(h_fine, 0.0, 1.0)
    return h_fine

# Main entry point required by the harness
def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a target resolution (not too large to stay within time budget)

    n_target = min(200, max(90, n0 * 2))
    h_current = upscale_solution(base_h, n_target)

    # Build a list of starting points
    starts = [h_current]
    for _ in range(4):
        starts.append(construct_simple_initial(n_target, rng))

    # Common constraints for SLSQP
    n = n_target
    target_sum = n / 2.0
    cons = ({
        "type": "eq",
        "fun": lambda x: x.sum() - target_sum,
    },)

    bounds = [(0.0, 1.0) for _ in range(n)]

    best_h = None
    best_c5 = np.inf

    # Optimization loop respecting the time budget
    for idx, h0 in enumerate(starts):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
