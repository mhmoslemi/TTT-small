import time
import numpy as np
from scipy.optimize import differential_evolution

# Helper to compute the C5 value for a discrete h
def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h), zero‑padded (mode='full')
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)


def _project_to_constraints(x, target_sum):
    """Clip to [0,1] and enforce the exact sum by a simple redistribution."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return x
    # distribute the remaining mass to a random index that can still move
    rng = np.random.default_rng()
    for _ in range(100):
        idx = rng.integers(len(x))
        # compute how much we could add/remove without violating bounds
        if diff > 0:
            inc = min(diff, 1.0 - x[idx])
            if inc > 0:
                x[idx] += inc
                diff -= inc
        else:
            dec = min(-diff, x[idx])
            if dec > 0:
                x[idx] -= dec
                diff += dec
        if abs(diff) < 1e-12:
            break
    # final tiny correction (might slightly violate bounds but negligible)
    if abs(diff) > 1e-12:
        x += diff / len(x)
        x = np.clip(x, 0.0, 1.0)
    return x

        h0[::2] = 1.0  # half ones, half zeros
    target_sum = n / 2.0          # because ∫h = 1  =>  sum(h)*dx = 1  => sum(h)=n/2

    # make sure the initial vector respects the constraints

    def de_obj(y):

        return _c5_value(y)

    # time allocation: ~2/3 for DE, rest for local hill‑climb
    de_budget = budget_s * 0.6

    bounds = [(0.0, 1.0)] * n
    result = differential_evolution(
        de_obj,
        bounds,

        maxiter=de_maxiter,
        popsize=15,
        tol=1e-6,
        polish=False,
        updating='deferred',
        workers=1,

    )
    cand = _project_to_constraints(result.x, target_sum)
