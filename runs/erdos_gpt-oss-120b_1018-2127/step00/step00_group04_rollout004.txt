import time
import numpy as np

def _project_onto_constraints(v, target_sum):
    """Clip to [0,1] and adjust to meet exact sum target (simple iterative scaling)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass, distribute proportionally to remaining capacity
            cap = 1.0 - v
            mask = cap > 1e-12
        else:
            # need to remove mass, distribute proportionally to current values
            cap = v
            mask = cap > 1e-12
        total_cap = cap[mask].sum()
        if total_cap == 0:
            break
        delta = diff * (cap[mask] / total_cap)
        v[mask] += delta
        v = np.clip(v, 0.0, 1.0)
    # final correction for any tiny residual
    residual = target_sum - v.sum()
    if abs(residual) > 1e-12:
        # distribute residual uniformly on free variables
        free = (v > 0) & (v < 1)
        cnt = free.sum()
        if cnt > 0:
            v[free] += residual / cnt
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx using full convolution."""
    conv = np.convolve(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    return np.max(conv) * dx

    that minimizes the Erdős C5 overlap constant.

    Returns:

    """

    target_sum = n / 2.0
    best_h = _project_onto_constraints(init.copy(), target_sum)
    best_val = _c5_value(best_h)

    # Parameters for simulated annealing
    max_iter = 200000          # generous upper bound; loop will stop by time
    temp0 = 0.02               # initial temperature
    temp_end = 1e-5
    sigma0 = 0.05              # initial perturbation scale
    sigma_end = 0.001

    for it in range(max_iter):

        if time.time() - start_time > budget_s * 0.98:
            break

        # cooling schedules (linear in log‑scale)
        frac = it / max_iter
        temp = temp0 * (temp_end / temp0) ** frac
        sigma = sigma0 * (sigma_end / sigma0) ** frac

        # propose neighbor
        cand = best_h + sigma * rng.standard_normal(n)
        cand = _project_onto_constraints(cand, target_sum)

        cand_val = _c5_value(cand)

        # acceptance
        if cand_val < best_val or rng.random() < np.exp(-(cand_val - best_val) / max(temp, 1e-12)):
