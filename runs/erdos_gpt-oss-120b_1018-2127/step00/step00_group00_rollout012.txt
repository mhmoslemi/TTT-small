import time
import numpy as np

# evaluate_erdos_solution and initial_h_values are provided by the harness


def _project_to_feasible(vec, target):
    """
    Project vec onto the box [0,1] and the hyperplane sum(vec)=target.
    Simple water-filling style iterative adjustment.
    """
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(20):
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0
            if not free.any():
                break
            cap = (1.0 - v[free]).sum()
            inc = min(diff, cap)
            v[free] += inc / free.sum()
        else:
            free = v > 0.0
            if not free.any():
                break
            cap = v[free].sum()
            dec = min(-diff, cap)
            v[free] -= dec / free.sum()
    return np.clip(v, 0.0, 1.0)


def _c5_of(h):
    """
    Wrapper that uses the harness' evaluator if available,
    otherwise falls back to a direct correlation computation.
    """
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        # direct computation: max_k \int h(x)(1-h(x+k))dx
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        return float(corr.max())


def run(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic projected gradient / simulated‑annealing search for a step
    function h on [0,2] (discretised) that gives a low C5 value.

    Returns
    -------
    h_values : np.ndarray
        The best-found vector (length n_points) with entries in [0,1].
    c5_bound : float
        The corresponding C5 value.
    n_points : int
        Length of the vector (discretisation size).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    end_time = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise problem size and a feasible starting point
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: uniform half‑filled function
        n_fallback = 60
        h0 = np.zeros(n_fallback)
        half = n_fallback // 2
        h0[:half] = 1.0
        h0 = h0.astype(float)

    n = len(h0)
    target_sum = n / 2.0
    h_current = _project_to_feasible(h0, target_sum)
    best_h = h_current.copy()
    best_c5 = _c5_of(best_h)

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing loop
    # ------------------------------------------------------------------
    # annealing schedule
    temp = 0.12
    decay = 0.9997
    # perturbation scale
    sigma = 0.08

    # pre‑compute a mask of indices to perturb (10 % of entries)
    perturb_frac = 0.12
    n_perturb = max(1, int(perturb_frac * n))

    while time.time() < end_time:
        # propose a new candidate
        idx = rng.choice(n, size=n_perturb, replace=False)
        proposal = h_current.copy()
        proposal[idx] += rng.normal(loc=0.0, scale=sigma, size=n_perturb)
        proposal = _project_to_feasible(proposal, target_sum)

        cand_c5 = _c5_of(proposal)

        # acceptance rule
        if cand_c5 < best_c5:
            best_h, best_c5 = proposal.copy(), cand_c5
        if cand_c5 < _c5_of(h_current) or rng.random() < np.exp((_c5_of(h_current) - cand_c5) / max(temp, 1e-12)):
            h_current = proposal

        # schedule update
        temp *= decay
        # occasional reseeding of sigma to keep exploration alive
        if rng.random() < 0.01:
            sigma *= 0.9
            sigma = max(sigma, 0.01)

    return best_h, float(best_c5), int(n)
