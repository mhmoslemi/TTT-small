def _max_overlap(h, dx):
    """Compute C5 = max_x ∫ h(x)(1-h(x+τ)) dx using FFT."""
    N = h.size
    # Full convolution length is 2*N-1; use FFT length 2*N for convenience
    f = np.fft.rfft(h, n=2 * N)
    g = np.fft.rfft(1.0 - h, n=2 * N)
    conv = np.fft.irfft(f * g, n=2 * N)[: 2 * N - 1]
