import time
import numpy as np
import random

def compute_c5(h):
    """Return the overlap bound C5 for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    """Project h onto the simplex with box [0,1] and given total sum."""
    h = np.clip(h, 0.0, 1.0)
    # Simple water-filling: distribute the deficit/surplus uniformly
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            # all at bounds – distribute to all then re‑clip
            h += diff / h.size
        else:
            cnt = free.sum()
            h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final tiny adjustment on the whole vector if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def improve_h(h, budget_s, start_temp=0.2, end_temp=0.01):
    """Simple simulated‑annealing style local search."""
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(best_h)
    start = time.time()
    it = 0
    while True:
        it += 1
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break
        # temperature schedule linear in iterations
        t = max(end_temp, start_temp * (1 - elapsed / budget_s))
        # propose a move: modify a random window
        idx = random.randrange(n)
        delta = random.gauss(0.0, t)
        h_new = best_h.copy()
        h_new[idx] += delta
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum(h_new, target_sum)
        val = compute_c5(h_new)
        if val < best_val:
            best_h, best_val = h_new, val
        # occasional random restart to avoid stagnation
        if it % 2000 == 0:
            rand = np.random.rand(n)
            rand = project_sum(rand, target_sum)
            best_h = rand
            best_val = compute_c5(best_h)
    return best_h, best_val

def upscale(h, new_n):
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # enforce constraints after interpolation
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum(h_new, new_n / 2.0)
    return h_new

    np.random.seed(seed)

        h0 = np.full(n0, 0.5)               # uniform satisfies integral

    total_budget = budget_s
    start_time = time.time()

    # Stage list: try 45 → 60 → 90 points if time permits
    stages = [h0.size, 60, 90]
    best_h, best_val = h0, compute_c5(h0)
