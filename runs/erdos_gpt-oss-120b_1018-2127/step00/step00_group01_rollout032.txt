import time
import numpy as np
import scipy.optimize as opt

def _integral_constraint_sum(x, target):
    return np.sum(x) - target

def _evaluate_c5(h):
    """Return C5 for a discrete step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _objective(x, target, penalty_factor):
    """Objective for DE: C5 plus a big penalty for violating the integral."""
    s = np.sum(x)
    penalty = penalty_factor * abs(s - target)
    return _evaluate_c5(x) + penalty

def _de_optimize(n, target, max_time, rng):
    """Run a time‑bounded differential evolution."""
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    penalty = 1e6

    start = time.time()
    best_vec = None
    best_val = np.inf

    def callback(xk, convergence):
        nonlocal best_vec, best_val, start
        cur = _evaluate_c5(xk)
        if cur < best_val:
            best_val = cur
            best_vec = xk.copy()
        if time.time() - start > max_time:
            return True  # stop DE
        return False

    opt.differential_evolution(
        _objective,
        bounds,
        args=(target, penalty),
        seed=int(rng.integers(2**32 - 1)),
        maxiter=200,
        popsize=15,
        polish=False,
        callback=callback,
        updating='deferred',
        workers=1,
    )
    # If DE finished early without callback stop, result.x is also feasible
    if best_vec is None:
        # fallback to a uniform feasible vector
        best_vec = np.full(n, target / n)
    return best_vec

def _local_refine(x0, target, max_time):
    """Refine with SLSQP under equality and box constraints."""
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    constraints = {"type": "eq", "fun": _integral_constraint_sum, "args": (target,)}

    start = time.time()
    opts = {"maxiter": 300, "ftol": 1e-12}

    res = opt.minimize(
        _evaluate_c5,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[constraints],
        options=opts,
    )
    # If time exceeded, fall back to the input vector
    if time.time() - start > max_time:
        return x0
    return np.clip(res.x, 0.0, 1.0)

def _project_to_feasible(v, target):
    """Simple projection onto the simplex with box [0,1] and prescribed sum."""
    v = np.clip(v, 0.0, 1.0)
    excess = np.sum(v) - target
    if abs(excess) < 1e-12:
        return v
    # distribute excess uniformly and re‑clip iteratively (few iterations suffice)
    for _ in range(5):
        adjust = excess / v.size
        v -= adjust
        v = np.clip(v, 0.0, 1.0)
        excess = np.sum(v) - target
        if abs(excess) < 1e-12:
            break
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] minimizing the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Choose a modest grid size (fast evaluation, enough flexibility)
    n_points = 120  # < 1k as required
    target_sum = n_points / 2.0  # because ∫ h = 1 and dx = 2/n

    # Build an initial vector, using the supplied construction if present
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h_initial = np.interp(x_new, x_old, h0)
    else:
        h_initial = np.full(n_points, 0.5)
        h_initial += 0.2 * (rng.random(n_points) - 0.5)
        h_initial = np.clip(h_initial, 0.0, 1.0)

    # Ensure feasibility of the starting vector
    h_initial = _project_to_feasible(h_initial, target_sum)

    # Allocate time: ~70 % for global search, rest for local refinement
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed

    best_vec = _de_optimize(n_points, target_sum, de_time, rng)

    refined = _project_to_feasible(refined, target_sum)
