import time
import numpy as np
from math import log, exp

def compute_c5(h):
    """Return the C5 value for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    # linear (non‑circular) correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Clip to [0,1] and adjust the sum to target_sum by redistribution."""
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    diff = target_sum - current
    # simple redistribution: add equal share to non‑boundary entries
    mask = (h > 0.0) & (h < 1.0)
    if mask.any():
        share = diff / mask.sum()
        h[mask] += share
        h = np.clip(h, 0.0, 1.0)
    # if still off, perform a final scaling (rare)
    current = h.sum()
    if current != 0:
        h *= target_sum / current
    return np.clip(h, 0.0, 1.0)

        target = n / 2.0

    target_sum = n / 2.0
