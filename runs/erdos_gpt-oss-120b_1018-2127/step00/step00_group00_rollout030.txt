import time
import numpy as np
from scipy.optimize import minimize

# optional cvxpy for projection onto feasibility set
try:
    import cvxpy as cp
    _has_cvxpy = True
except Exception:
    _has_cvxpy = False


def _compute_score(h, n):
    """Maximum overlap integral for step vector h of length n."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    scale = 2.0 / n
    return np.max(corr) * scale


def _project_feasible(v, n):
    """Project vector v onto {0<=x<=1, sum(x)=n/2}."""
    if _has_cvxpy:
        x = cp.Variable(len(v))
        constraints = [x >= 0, x <= 1, cp.sum(x) == n / 2]
        prob = cp.Problem(cp.Minimize(cp.sum_squares(x - v)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
        except Exception:
            prob.solve(solver=cp.ECOS, warm_start=False, max_iters=500)
        if x.value is not None:
            return np.array(x.value).flatten()
    # fallback simple heuristic
    w = np.clip(v, 0.0, 1.0)
    total = np.sum(w)
    if total == 0:
        # put all mass into first entry
        w[0] = n / 2.0
        return w
    w *= (n / 2.0) / total
    w = np.clip(w, 0.0, 1.0)
    diff = n / 2.0 - np.sum(w)
    if abs(diff) > 1e-12:
        # distribute remaining mass greedily
        for i in range(len(w)):
            inc = min(diff, 1.0 - w[i])
            w[i] += inc
            diff -= inc
            if abs(diff) < 1e-12:
                break
    return w


def _resample_h(old_h, new_n):
    """Linear interpolation of old_h (defined on [0,2]) to length new_n."""
    old_n = len(old_h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h, left=old_h[0], right=old_h[-1])
    return _project_feasible(new_h, new_n)


def _make_initial_h(n):
    """Generate a simple feasible step vector of length n."""
    h = np.zeros(n)
    half = int(np.round(n / 2))
    h[:half] = 1.0
    # adjust exact sum if rounding issue
    current = np.sum(h)
    target = n / 2.0
    diff = target - current
    if abs(diff) > 1e-12:
        # add remaining mass to first element (still ≤1)
        addition = min(diff, 1.0 - h[0])
        h[0] += addition
    return h


def _optimize_h(h0, n, seed):
    """Run SLSQP from starting point h0."""
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    options = {"maxiter": 500, "ftol": 1e-7, "disp": False}
    res = minimize(
        lambda x: _compute_score(x, n),
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options=options,
    )
    return res


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within time budget."""
    start = time.time()
    np.random.seed(seed)

    # Determine available initial vector
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
        base_n = len(base_h)
    else:
        base_n = 45
        base_h = _make_initial_h(base_n)

    # Candidate resolutions (keep modest for speed)
    n_list = [30, 45, 60, 90, 180]

    best_h = None
    best_score = np.inf
    best_n = None

    for n in n_list:
        if time.time() - start > budget_s:
            break

        # prepare feasible starting point
        if n == base_n:
            h0 = _project_feasible(base_h.copy(), n)
        else:
            h0 = _resample_h(base_h, n)

        # quick feasibility check
        if not (np.all(h0 >= -1e-9) and np.all(h0 <= 1 + 1e-9) and abs(np.sum(h0) - n / 2.0) < 1e-6):
            h0 = _project_feasible(h0, n)

        # evaluate unoptimized score
        score0 = _compute_score(h0, n)
        if score0 < best_score:
            best_score, best_h, best_n = score0, h0.copy(), n

        # run local optimizer
        res = _optimize_h(h0, n, seed)
        if res.success:
            h_opt = _project_feasible(res.x, n)  # safety projection
            score_opt = _compute_score(h_opt, n)
            if score_opt < best_score:
                best_score, best_h, best_n = score_opt, h_opt.copy(), n

        # keep the best found so far as the base for higher resolutions
        base_h = best_h.copy()
        base_n = best_n

    # final safety: ensure return vector respects constraints
    if best_h is None:
        best_h = _make_initial_h(45)
        best_score = _compute_score(best_h, 45)
        best_n = 45

    # Clip final vector to [0,1] and renormalize exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    total = np.sum(best_h)
    if abs(total - best_n / 2.0) > 1e-10:
        best_h = _project_feasible(best_h, best_n)

    return best_h, best_score, best_n
