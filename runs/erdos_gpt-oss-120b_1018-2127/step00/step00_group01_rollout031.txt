import time
import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

def _compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _project_simple(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the feasible set:
        0 <= h[i] <= 1  and  sum(h) = n/2
    Uses an iterative clipping / redistribution scheme.
    """
    n = v.size
    target = n / 2.0
    h = np.clip(v, 0.0, 1.0)

    for _ in range(12):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            # all at bounds – distribute to any index that can move
            i = np.argmin(h) if diff > 0 else np.argmax(h)
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            break
        delta = diff / nfree
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)

    # final tiny adjustment if needed
    diff = target - h.sum()
    if abs(diff) > 1e-8:
        i = np.argmin(h) if diff > 0 else np.argmax(h)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)

    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] with integral 1
    that minimizes the Erdős overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points) :

        c5_bound – the achieved C5 upper bound (float)
        n_points – number of discrete samples used (int)
    """
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 47
        h = np.full(n, 0.5)  # satisfies sum = n/2 exactly

    h = _project_simple(h)  # ensure feasibility
    best_h = h.copy()
    best_score = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # Simple stochastic local search (90 % of budget)
    # ------------------------------------------------------------------
    sigma = 0.2
    target_sum = n / 2.0
    while time.time() - start_time < budget_s * 0.9:
        # perturb current best
        cand = best_h + sigma * np.random.randn(n)
        cand = _project_simple(cand)

        # cheap correction for any residual sum error
        diff = target_sum - cand.sum()
        if abs(diff) > 1e-8:

            if free.any():

        score = _compute_c5(cand)

        if score < best_score:

    return best_h, float(best_score), int(n)
