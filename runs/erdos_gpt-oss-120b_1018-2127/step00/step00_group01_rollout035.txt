import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def _compute_c5(h):
    """Return max overlap integral for discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full convolution of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _feasible_vector(n, rng, init_vals=None):
    """Create a feasible vector of length n (values in [0,1] and sum = n/2)."""
    target_sum = n / 2.0
    if init_vals is not None:
        # interpolate existing values to new size
        x_old = np.arange(len(init_vals))
        x_new = np.linspace(0, len(init_vals) - 1, n)
        vec = np.interp(x_new, x_old, init_vals)
        # Clip then renormalise to meet the exact sum
        vec = np.clip(vec, 0.0, 1.0)
        cur = vec.sum()
        if cur > 0:
            vec *= target_sum / cur
        else:
            vec[:] = target_sum / n
        return vec
    # random start then project onto the simplex with box constraints
    vec = rng.random(n)
    vec = np.clip(vec, 0.0, 1.0)
    cur = vec.sum()
    if cur > 0:
        vec *= target_sum / cur
    else:
        vec[:] = target_sum / n
    return vec

def _objective_factory(penalty):
    """Factory that returns a penalised objective for DE."""
    def obj(x):
        # penalty on the integral constraint
        n = x.shape[0]
        sum_err = np.abs(x.sum() - n / 2.0)
        return _compute_c5(x) + penalty * sum_err
    return obj

def _local_refine(x0, time_left):
    """Perform a short SLSQP refinement respecting the exact constraints."""
    n = x0.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target}
    opts = {"maxiter": 150, "ftol": 1e-9}
    # limit wall‑clock time (SLSQP has no built‑in timeout, so we abort manually)
    start = time.time()
    res = minimize(_compute_c5, x0, method="SLSQP", bounds=bounds,
                   constraints=cons, options=opts)
    if not res.success and time.time() - start > time_left:
        # give up early, return the best we have
        return x0
    # ensure feasibility after numerical drift
    y = np.clip(res.x, 0.0, 1.0)
    y *= target / y.sum()
    return y
