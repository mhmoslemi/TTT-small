import time
import numpy as np

def _compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation h * (1-h) shifted
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _repair(h, target, max_iter=10):
    """Project h onto [0,1] and enforce sum == target."""
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # scale towards target, then clip again
        scale = target / s if s != 0 else 0.0
        h *= scale
    # final clip to be safe
    h = np.clip(h, 0.0, 1.0)
    # tiny adjustment to hit the target exactly (redistribute uniformly)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff over non‑saturated entries
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            add = diff / free.sum()
            h[free] += add
        else:
            # all at bounds – fallback to uniform scaling
            h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 and 0≤h≤1
    to minimise C5 = max_k ∫ h(x)(1-h(x+k)) dx.

    Returns (h_values, c5_bound, n_points)
    """
    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 100
        h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(float)

    target_sum = n / 2.0  # because dx = 2/n and integral = 1
    h = _repair(h, target_sum)

    best_h = h.copy()
    best_score = _compute_c5(best_h)

    # Parameters for the stochastic search
    max_iter = 200000  # safety ceiling; loop will stop earlier by time
    step_scale = 0.05  # typical transfer amount
    temperature_start = 0.02
    temperature_end = 0.0001

    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # linear cooling schedule
        t_frac = it / max_iter
        temperature = temperature_start * (1 - t_frac) + temperature_end * t_frac

        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        # propose a transfer delta respecting bounds
        max_delta = min(step_scale, best_h[i], 1.0 - best_h[j])
        if max_delta <= 0:
            continue
        delta = (np.random.rand() * 2 - 1) * max_delta  # signed delta
        # construct candidate
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta

        # repair just in case of numerical drift
        cand = _repair(cand, target_sum, max_iter=2)

        cand_score = _compute_c5(cand)

        # accept if better or with Metropolis probability
        if cand_score < best_score or np.random.rand() < np.exp((best_score - cand_score) / temperature):
            best_h = cand
            best_score = cand_score

    # final repair and scoring
    best_h = _repair(best_h, target_sum)
    final_score = _compute_c5(best_h)

    return best_h, final_score, n
