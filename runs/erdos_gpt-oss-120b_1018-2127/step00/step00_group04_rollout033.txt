import numpy as np
import time

def _project_to_sum_and_box(v, target_sum, tol=1e-9, max_iter=100):
    """Project vector v onto the hyperplane sum(x)=target_sum with box [0,1].

    Returns x = clip(v + λ, 0, 1) where λ is chosen so that sum(x)=target_sum.
    """
    lo, hi = -2.0, 2.0  # wide enough bounds for λ
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def _compute_c5(h):
    """Compute the C5 value for a discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _random_feasible(n, rng):
    """Generate a random feasible vector of length n (values in [0,1], sum=n/2)."""
    v = rng.random(n)
    target = n / 2.0
    return _project_to_sum_and_box(v, target)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s - 0.5  # small safety margin

    # Choose resolution
    n = 200  # can be adjusted; stays well below 1000
    target_sum = n / 2.0
    dx = 2.0 / n

    # Initialise from provided parent if available
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        m = parent.shape[0]
        # interpolate parent onto our grid
        x_parent = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_parent, parent)
        h = _project_to_sum_and_box(h, target_sum)
    else:
        # uniform start satisfies constraints
        h = np.full(n, 0.5)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    iteration = 0
    while time.time() - start_time < time_limit:
        # Temperature schedule (linear decay)
        elapsed = time.time() - start_time
        t_frac = 1.0 - elapsed / time_limit
        temperature = max(1e-4, t_frac)

        # Propose a new candidate
        cand = best_h.copy()
        # perturb a random subset (10% of points)
        subset = rng.choice(n, size=max(1, n // 10), replace=False)
        sigma = 0.1 * temperature
        cand[subset] += rng.normal(scale=sigma, size=subset.shape[0])

        # Project back to feasible set
        cand = _project_to_sum_and_box(cand, target_sum)

        cand_c5 = _compute_c5(cand)

        # Acceptance rule
        if cand_c5 < best_c5 or rng.random() < np.exp((best_c5 - cand_c5) / temperature):
            best_h = cand
            best_c5 = cand_c5

            rand_c5 = _compute_c5(rand_h)
            if rand_c5 < best_c5:

        iteration += 1
