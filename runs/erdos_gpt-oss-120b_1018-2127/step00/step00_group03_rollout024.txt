import time
import numpy as np
import random

def _compute_c5(h, dx):
    """Return the C5 value for a given step function array h."""
    # full correlation gives length 2n-1, lags -n+1..n-1
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _initial_binary(n):
    """Create a binary feasible vector with exactly n/2 ones."""
    ones = np.zeros(n, dtype=np.float32)
    ones[: n // 2] = 1.0
    np.random.shuffle(ones)
    return ones

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized values in [0,1] of length n_points.
    c5_bound : float
        Achieved C5 upper bound.
    n_points : int
        Number of discretisation points used.
    """
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------
    # Starting point
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, copy=True, dtype=np.float32)
        n = h.size
        # enforce feasibility (clip + rescale sum)
        np.clip(h, 0.0, 1.0, out=h)
        current_sum = h.sum()
        target_sum = n / 2.0
        if abs(current_sum - target_sum) > 1e-8:
            # shift uniformly to satisfy the integral, then clip again
            h += (target_sum - current_sum) / n
            np.clip(h, 0.0, 1.0, out=h)
        # If still not feasible (rare), fall back to binary construction
        if not (0.0 <= h.min() and h.max() <= 1.0 and abs(h.sum() - target_sum) < 1e-6):
            h = _initial_binary(n)
    else:
        n = 70  # default resolution
        h = _initial_binary(n)

    dx = 2.0 / n
    best_h = h.copy()
    best_score = _compute_c5(best_h, dx)

    # ----------------------------------------------------------------------
    # Helper structures for binary swaps (maintain sets of indices)
    # ----------------------------------------------------------------------
    def _indices_of(val):
        return np.where(h == val)[0]

    # Start with binary representation if the current h is not binary
    if not np.all(np.isin(h, [0.0, 1.0])):
        # project to nearest binary while keeping the sum
        # take the largest n/2 entries as 1, rest 0
        order = np.argsort(-h)  # descending
        binary = np.zeros_like(h)
        binary[order[: n // 2]] = 1.0
        h = binary.astype(np.float32)

    ones_idx = set(np.where(h == 1.0)[0].tolist())
    zeros_idx = set(np.where(h == 0.0)[0].tolist())

    # ----------------------------------------------------------------------
    # Simulated annealing / hill climbing
    # ----------------------------------------------------------------------
    iteration = 0
    # temperature schedule parameters
    T0 = 0.02
    T_end = 0.0001
    max_iter = 2000000  # safety upper bound

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.2:  # leave a margin
            break
        if iteration >= max_iter:
            break

        # linear cooling
        T = T0 + (T_end - T0) * (iteration / max_iter)

        # propose a swap between a 1 and a 0
        i = random.sample(ones_idx, 1)[0]
        j = random.sample(zeros_idx, 1)[0]

        # perform swap
        h[i] = 0.0
        h[j] = 1.0

        new_score = _compute_c5(h, dx)

        accept = False
        if new_score < best_score:
            accept = True
            best_score = new_score
            best_h = h.copy()
        else:
            # Metropolis acceptance
            if T > 0 and random.random() < np.exp(-(new_score - best_score) / T):
                accept = True

        if accept:
            # update index sets
            ones_idx.remove(i)
            zeros_idx.remove(j)
            ones_idx.add(j)
            zeros_idx.add(i)
            # keep current h as is
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

        iteration += 1

    # ----------------------------------------------------------------------
    # Final evaluation and return
    # ----------------------------------------------------------------------
    final_c5 = _compute_c5(best_h, dx)

    return best_h, final_c5, n
