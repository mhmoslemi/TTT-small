import time
import numpy as np

# Helper: compute the Erdős C5 bound for a discrete h
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

# Project a vector onto [0,1] with a prescribed sum (water‑filling)
def project_box_sum(v, target_sum):
    lo, hi = 0.0, 1.0
    v = np.clip(v, lo, hi)
    # Simple iterative water‑filling
    for _ in range(10):
        cur_sum = v.sum()
        diff = cur_sum - target_sum
        if abs(diff) < 1e-12:
            break
        # Identify indices that can move in the needed direction
        if diff > 0:           # need to decrease total
            movable = v > lo + 1e-12
            if not np.any(movable):
                break
            delta = diff / movable.sum()
            v[movable] -= delta
            v = np.clip(v, lo, hi)
        else:                  # need to increase total
            movable = v < hi - 1e-12
            if not np.any(movable):
                break
            delta = diff / movable.sum()   # diff is negative
            v[movable] -= delta
            v = np.clip(v, lo, hi)
    # Final correction (tiny) by scaling the free part
    cur_sum = v.sum()
    if abs(cur_sum - target_sum) > 1e-10:
        free = (v > lo + 1e-12) & (v < hi - 1e-12)
        if np.any(free):
            v[free] += (target_sum - cur_sum) / free.sum()
            v = np.clip(v, lo, hi)
    return v

# Generate a random feasible h (values in [0,1], sum = n/2)
def random_feasible_h(n, rng):
    raw = rng.random(n)
    target = n / 2.0
    h = project_box_sum(raw, target)
    return h

    target_sum = n / 2.0

    # Parameters for the stochastic local search
    max_iter = 200000
    step_scale = 0.05   # typical magnitude of a move (in absolute h units)
    temperature = 0.01

    iter_count = 0
    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:
            break
        if iter_count >= max_iter:

        iter_count += 1

        # Pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Propose a transfer delta from i to j
        delta = (rng.random() - 0.5) * step_scale
        if not (0.0 <= best_h[i] + delta <= 1.0 and 0.0 <= best_h[j] - delta <= 1.0):
            continue

        # Apply tentative move
        h_trial = best_h.copy()
        h_trial[i] += delta
        h_trial[j] -= delta

        # Enforce exact sum (tiny drift may appear because of rounding)
        current_sum = h_trial.sum()
        if abs(current_sum - target_sum) > 1e-9:
            # redistribute the tiny error uniformly over non‑boundary entries
            free = (h_trial > 0.0 + 1e-12) & (h_trial < 1.0 - 1e-12)
