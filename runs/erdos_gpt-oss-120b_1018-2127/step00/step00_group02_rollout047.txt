import time
import numpy as np
from scipy import optimize as _opt

def _compute_c5(h):
    """Discrete approximation of C5 for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h, target_sum):
    """Project h back into [0,1] and adjust to hit the target integral."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur != target_sum:
        # simple additive correction, then re‑clip
        h += (target_sum - cur) / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 to minimize
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.

    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # --- initialise ---------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        # fallback: 60 uniform samples, each = 0.5 gives integral 1
        n0 = 60
        h0 = np.full(n0, 0.5, dtype=float)

    n = h0.size
    target_sum = n / 2.0                # because dx = 2/n and integral = 1
    h0 = _repair(h0, target_sum)        # guarantee feasibility

    # --- optimisation helpers -----------------------------------------------
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": lambda h: h.sum() - target_sum}]
    obj = lambda h: _compute_c5(h)

    best_h = h0.copy()
    best_c5 = obj(best_h)

    # limit for inner optimisation (iterations, function evaluations)
    opt_options = {"maxiter": 300, "ftol": 1e-9, "disp": False}

    # --- search loop --------------------------------------------------------
    while time.time() - start_time < 0.9 * budget_s:
        # perturb current best solution
        trial = best_h + 0.08 * np.random.randn(n)
        trial = _repair(trial, target_sum)

        res = _opt.minimize(
            obj,
            trial,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options=opt_options,
        )

        if not res.success:
            # try a completely random start if SLSQP fails repeatedly
            trial = np.clip(np.random.rand(n), 0.0, 1.0)
            trial = _repair(trial, target_sum)
            res = _opt.minimize(
                obj,
                trial,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options=opt_options,
            )
            if not res.success:
                continue

        cand_h = _repair(res.x, target_sum)
        cand_c5 = obj(cand_h)
