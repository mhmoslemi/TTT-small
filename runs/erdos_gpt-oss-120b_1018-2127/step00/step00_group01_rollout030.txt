import time, random, math
import numpy as np

def _c5_from_h(h: np.ndarray) -> float:
    """Discrete approximation of C5 for step size 2/n."""
    # correlation without scaling, then multiply by dx = 2/n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / h.size))

def _evaluate(h: np.ndarray) -> float:
    """Use the provided evaluator if available, else fallback."""
    if 'evaluate_erdos_solution' in globals():
        # the helper is expected to return the bound directly
        return float(evaluate_erdos_solution(h))
    return _c5_from_h(h)

def _feasible_pair_move(h: np.ndarray, i: int, j: int, delta: float) -> np.ndarray:
    """Return a copy of h after moving delta from i to j (preserving sum)."""
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def _random_feasible_h(n: int, rng) -> np.ndarray:
    """Generate a feasible vector with sum = n/2 and values in [0,1]."""
    # start with exactly half 1's and half 0's, then shuffle
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[:half] = 1.0
    rng.shuffle(h)
    return h
