import numpy as np
import time

def _project_onto_sum_and_bounds(v, target, max_iter=50, tol=1e-9):
    """
    Euclidean projection of vector v onto the set
        { x | 0 <= x_i <= 1, sum(x) = target }.
    Uses bisection on the dual variable tau.
    """
    lo, hi = -1.0, 2.0  # wide enough bounds for tau
    for _ in range(max_iter):
        tau = (lo + hi) / 2.0
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _correlation_max(h):
    """Return max correlation value and the shift index achieving it."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    max_val = corr[idx]
    # shift k corresponds to idx - (len(h) - 1)
    k = idx - (len(h) - 1)
    return max_val, k

def _gradient_for_shift(h, k):
    """
    Gradient of correlation at shift k with respect to h.
    Handles zero padding outside the array.
    """
    n = len(h)
    grad = np.empty_like(h)
    # term (1 - h_{j+k})
    term1 = np.ones_like(h)
    if k >= 0:
        term1[: n - k] -= h[k:]
    else:
        term1[-k:] -= h[: n + k]
    # term h_{j-k}
    term2 = np.zeros_like(h)
    if k >= 0:
        term2[k:] = h[: n - k]
    else:
        term2[: n + k] = h[-k:]
    grad = term1 - term2
    return grad

def _compute_c5(h):
    """Exact C5 value for current h (integral version)."""
    dx = 2.0 / len(h)
    max_corr, _ = _correlation_max(h)
    return max_corr * dx
