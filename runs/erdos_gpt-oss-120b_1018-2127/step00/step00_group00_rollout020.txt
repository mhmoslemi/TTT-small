import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Compute C5 for a given step vector h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _transfer(old_h, old_n, new_n):
    """Linear interpolation of old_h (length old_n) onto a grid of size new_n."""
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # Clip to [0,1] – SLSQP will later enforce bounds more strictly
    return np.clip(new_h, 0.0, 1.0)

def _random_feasible(n, rng):
    """Generate a random vector in [0,1] with sum = n/2."""
    # start from uniform[0,1], then scale to meet sum, finally clip and renormalize
    x = rng.random(n)
    current = x.sum()
    target = n / 2.0
    if current == 0:
        x[:] = target / n
    else:
        x *= target / current
    # ensure bounds after scaling
    x = np.clip(x, 0.0, 1.0)
    # final correction to meet exact sum (simple linear adjustment)
    diff = target - x.sum()

            x[free] += adj
            x = np.clip(x, 0.0, 1.0)
    return x
