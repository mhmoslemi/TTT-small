import time
import numpy as np

def _objective(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _repair(h, target_sum, eps=1e-12, max_iter=10):
    """Project h onto [0,1] and enforce sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) <= eps:
            break
        # distribute diff uniformly
        h += diff / h.size
        # re‑clip
        h = np.clip(h, 0.0, 1.0)
    # final correction of any tiny residual via scaling of a subset
    s = h.sum()
    if abs(s - target_sum) > eps:
        # adjust the first element that can move without violating bounds
        for i in range(h.size):
            avail = (1.0 - h[i]) if s < target_sum else h[i]
            step = min(abs(target_sum - s), avail)
            if s < target_sum:
                h[i] += step
            else:
                h[i] -= step
            s = h.sum()
            if abs(s - target_sum) <= eps:
                break
    return h

def _random_walk(h, scale, rng):
    """Return a perturbed copy of h."""
    return h + rng.uniform(-scale, scale, size=h.shape)

def _interpolate(h, new_n):
    """Linear interpolation of a periodic vector h onto new_n points."""
    old_n = h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)
