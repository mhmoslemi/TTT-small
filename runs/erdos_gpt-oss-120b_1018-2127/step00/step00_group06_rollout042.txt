import time
import math
import numpy as np

def _objective(h):
    """Maximum overlap integral for a discretized h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * (2.0 / n)

def _repair_constraints(h):
    """Project h onto [0,1]^n with exact sum = n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # Simple iterative redistribution
    for _ in range(5):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

    # Small residual handling
    residual = target - h.sum()
    if abs(residual) > 1e-9:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:
            h[free] += residual / free.size
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback: uniform scaling then clipping
            scale = target / max(h.sum(), 1e-12)
            h = np.clip(h * scale, 0.0, 1.0)

    return h

def _interpolate(old_h, new_n):
    """Linear interpolation of a step function from len(old_h) to new_n points."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, old_h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points) where:
      h_values – numpy array of length n_points with values in [0,1],
      c5_bound – the achieved upper bound C₅,
      n_points – the number of discretisation points used.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial solution (grid size 95)
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != 95:
            # resample to 95 points if the parent has a different length
            init = _interpolate(init, 95)
    else:
        # simple deterministic starter: half ones, half zeros, shuffled
        init = np.zeros(95, dtype=float)
        init[:47] = 1.0
        rng.shuffle(init)

    cur_h = _repair_constraints(init)
    best_h = cur_h.copy()
    best_score = _objective(best_h)
    best_n = cur_h.shape[0]

    # ------------------------------------------------------------------
    # Multi‑grid simulated‑annealing‑like search
    # ------------------------------------------------------------------
    grid_sizes = [95, 190, 380, 760]               # increasing resolutions
    for n in grid_sizes:
        # time guard
        if time.time() - start_time > 0.95 * budget_s:
            break

        # upscale / downscale to current grid size
        if cur_h.shape[0] != n:
            cur_h = _interpolate(cur_h, n)
            cur_h = _repair_constraints(cur_h)

        # search parameters (scale grows modestly with n)
        scale = 0.02 * (n ** 0.2)
        # allocate iterations roughly proportional to remaining budget
        elapsed = time.time() - start_time
        remaining = max(budget_s - elapsed, 0.0)
        iters = int(min(200_000, remaining * 800 / (n / 100)))
        iters = max(iters, 2000)

        for _ in range(iters):
            if time.time() - start_time > 0.99 * budget_s:
                break

            # random perturbation
            perturb = rng.normal(scale=scale, size=n)
            cand = cur_h + perturb
            cand = _repair_constraints(cand)

            cand_score = _objective(cand)

            # deterministic acceptance of improvements
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
                best_n = n

            if cand_score < _objective(cur_h):
                cur_h = cand

        # keep the best found at this resolution as the seed for the next grid
        cur_h = best_h.copy()

    # final sanity check (ensure feasible)
    best_h = _repair_constraints(best_h)
    best_score = _objective(best_h)

    return best_h, float(best_score), int(best_n)
