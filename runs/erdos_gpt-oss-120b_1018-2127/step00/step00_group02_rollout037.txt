import time
import numpy as np
import scipy.optimize as opt
import cvxpy as cp

def _project_initial(initial, n):
    """Linearly interpolate initial array (any length) to length n."""
    m = len(initial)
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, initial)

def _random_feasible(n, rng):
    """Return a random vector in [0,1]^n with sum = n/2 using cvxpy."""
    h = cp.Variable(n)
    obj = cp.Maximize(rng.random())
    cons = [h >= 0, h <= 1, cp.sum(h) == n / 2]
    prob = cp.Problem(cp.Minimize(cp.sum(cp.multiply(rng.random(n), h))), cons)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=200)
    if h.value is None:
        # fallback: simple projection
        v = rng.uniform(0, 1, n)
        excess = np.sum(v) - n / 2
        if excess > 0:
            v -= excess / n
        v = np.clip(v, 0, 1)
        # adjust to exact sum
        diff = n / 2 - np.sum(v)
        while abs(diff) > 1e-8:
            idx = rng.choice(np.where(v > 0)[0])
            step = min(diff, 1 - v[idx]) if diff > 0 else max(diff, -v[idx])
            v[idx] += step
            diff = n / 2 - np.sum(v)
        return v
    return np.clip(h.value, 0, 1)

def _c5_value(h):
    """Compute the C5 value for a step function h."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return conv.max()

def _objective(x, n):
    """Scalar objective for optimizer (C5)."""
    return _c5_value(x)

def _make_constraints(n):
    dx = 2.0 / n
    cons = [
        {"type": "eq", "fun": lambda x: np.sum(x) - n / 2},
    ]
    return cons

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = None

    # try a few grid sizes, preferring smaller for speed
    for n in (60, 80, 100):
        if time.time() - start > budget_s - 5:
            break
        # initial guess
        if 'initial_h_values' in globals():
            h0 = _project_initial(initial_h_values, n)
        else:
            h0 = _random_feasible(n, rng)

        # bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = _make_constraints(n)

        # SLSQP optimization
        try:
            res = opt.minimize(
                fun=_objective,
                x0=h0,
                args=(n,),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 300, "ftol": 1e-9, "disp": False},
            )
        except Exception:
            res = None

        if res is not None and res.success:
            h_opt = np.clip(res.x, 0.0, 1.0)
            # enforce exact sum (tiny numerical drift)
            total = np.sum(h_opt)
            if total != 0:
                h_opt *= (n / 2) / total
                h_opt = np.clip(h_opt, 0.0, 1.0)
            c5_val = _c5_value(h_opt)
            if c5_val < best_c5:
                best_c5, best_h, best_n = c5_val, h_opt, n

        # occasional random restart if time permits
        for _ in range(3):
            if time.time() - start > budget_s - 5:
                break
            h_rand = _random_feasible(n, rng)
            try:

                    fun=_objective,
                    x0=h_rand,
                    args=(n,),
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": 200, "ftol": 1e-8, "disp": False},
                )
            except Exception:
                continue
            if res.success:
                h_opt = np.clip(res.x, 0.0, 1.0)
                total = np.sum(h_opt)
                if total != 0:
                    h_opt *= (n / 2) / total
                    h_opt = np.clip(h_opt, 0.0, 1.0)
                c5_val = _c5_value(h_opt)
                if c5_val < best_c5:
                    best_c5, best_h, best_n = c5_val, h_opt, n

    # final fallback: if nothing succeeded, use projected initial or uniform

            n = len(initial_h_values)
