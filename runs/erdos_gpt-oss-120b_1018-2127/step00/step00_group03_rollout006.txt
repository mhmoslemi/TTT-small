import numpy as np
import time
import random

def compute_c5(h, dx):
    # correlation as used in the evaluator
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def simulated_anneal(initial_h, dx, max_time, seed):
    random.seed(seed)
    np.random.seed(seed)

    n = len(initial_h)
    h = initial_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)

    # positions of ones and zeros for quick swaps
    ones = np.flatnonzero(h > 0.5).tolist()
    zeros = np.flatnonzero(h < 0.5).tolist()

    start = time.time()
    # annealing schedule parameters
    T0 = 1.0
    Tf = 1e-4
    # approximate total iterations (will be limited by time)
    est_iters = 200000

    for it in range(1, est_iters + 1):
        now = time.time()
        if now - start > max_time:
            break
        # linear cooling
        T = T0 * (1 - it / est_iters) + Tf * (it / est_iters)

        # propose a swap between a 1 and a 0
        if not ones or not zeros:
            break
        i = random.choice(ones)
        j = random.choice(zeros)

        # perform swap
        h[i], h[j] = 0.0, 1.0
        # update index lists
        ones.remove(i)
        zeros.append(i)
        zeros.remove(j)
        ones.append(j)

        new_val = compute_c5(h, dx)

        delta = new_val - best_val
        if new_val < best_val:
            best_val = new_val
            best_h = h.copy()
        # Metropolis acceptance
        if delta < 0 or random.random() < np.exp(-delta / max(T, 1e-12)):
            # keep change
            pass
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            ones.remove(j)

    return best_h, best_val
