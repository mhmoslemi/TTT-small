import time
import numpy as np
import scipy.optimize as opt

def _compute_c5(vec):
    n = vec.size
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx

def _project(vec, target_sum):
    vec = np.clip(vec, 0.0, 1.0)
    diff = vec.sum() - target_sum
    if abs(diff) > 1e-12:
        vec -= diff / vec.size
    return np.clip(vec, 0.0, 1.0)
