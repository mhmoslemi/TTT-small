import time
import numpy as np

def compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def random_binary_h(n, rng):
    """Create a binary vector of length n with exactly n/2 ones (or floor/ceil)."""
    ones = n // 2
    zeros = n - ones
    h = np.concatenate([np.ones(ones), np.zeros(zeros)])
    rng.shuffle(h)
    return h.astype(float)

def hill_climb_binary(h0, dx, time_limit, rng):
    """Hill‑climbing with pair swaps on a binary vector."""
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)

    start = time.time()
    iter_cnt = 0
    # temperature schedule (optional)
    while time.time() - start < time_limit:
        # pick a random 1 and a random 0 to swap
        ones_idx = np.flatnonzero(h > 0.5)
        zeros_idx = np.flatnonzero(h < 0.5)
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        # perform swap
        h[i], h[j] = h[j], h[i]

        val = compute_c5(h, dx)

        # accept if improvement, else revert
        if val < best_val:
            best_val = val
            best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

        iter_cnt += 1

        # occasional restart to escape local minima
        if iter_cnt % 5000 == 0:
            h = best_h.copy()
            # random perturbation: perform a few random swaps regardless of improvement
            for _ in range(10):
                i = rng.integers(0, len(h))
                j = rng.integers(0, len(h))
                if h[i] != h[j]:
                    h[i], h[j] = h[j], h[i]

    return best_h, best_val
