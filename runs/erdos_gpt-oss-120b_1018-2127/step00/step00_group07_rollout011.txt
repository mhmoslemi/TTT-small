import numpy as np
import time
import math
from scipy.optimize import minimize

# helper to evaluate C5 for a given h vector
def compute_c5(h):
    # evaluate_erdos_solution is provided in the harness
    return evaluate_erdos_solution(h)

def project_bounds_and_sum(h, target_sum):
    # Clip to [0,1]
    np.clip(h, 0.0, 1.0, out=h)
    # Scale to meet exact sum while staying in bounds
    current_sum = h.sum()
    if abs(current_sum - target_sum) < 1e-12:
        return h
    # Simple linear scaling with redistribution of deficit/excess
    deficit = target_sum - current_sum
    # Distribute proportionally to free space
    free_up = (1.0 - h).sum()
    free_down = h.sum()
    if deficit > 0:
        # need to raise some entries, limited by free_up
        raise_amount = min(deficit, free_up)
        if free_up > 0:
            h += raise_amount * (1.0 - h) / free_up
    else:
        # need to lower some entries, limited by free_down
        lower_amount = min(-deficit, free_down)
        if free_down > 0:
            h -= lower_amount * h / free_down
    # final clip to ensure numeric safety
    np.clip(h, 0.0, 1.0, out=h)
    return h

def random_two_swap(h, step):
    """Propose a new h by moving a random amount between two indices."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # decide direction randomly
    if np.random.rand() < 0.5:
        i, j = j, i  # we will increase i, decrease j
    max_inc = 1.0 - h[i]
    max_dec = h[j]
    max_eps = min(max_inc, max_dec, step)
    if max_eps <= 0:
        return None  # cannot move
    eps = np.random.rand() * max_eps
    new_h = h.copy()
    new_h[i] += eps
    new_h[j] -= eps
    return new_h

def local_optimize(h, target_sum, max_iter=200):
    """Refine using scipy's L-BFGS-B with linear equality and bound constraints."""
    n = h.size
    bounds = [(0.0, 1.0) for _ in range(n)]

    def obj(x):
        return compute_c5(x)

    # equality constraint: sum(x) = target_sum
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum},)

    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-8})
    if res.success:
        return res.x, res.fun
    return h, compute_c5(h)
