import numpy as np
import time
from scipy.optimize import minimize

def _c5_of(h):
    """Compute the C5 overlap value for a step function sample array."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _run_optim(h0, time_limit):
    """Run a bounded SLSQP optimisation from a starting point within time_limit seconds."""
    n = h0.size
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq',
             'fun': lambda x: np.sum(x) - target_sum},)

    best_h = h0.copy()
    best_val = _c5_of(best_h)

    # optimiser parameters
    opts = {'maxiter': 200, 'ftol': 1e-9, 'disp': False}
    start = time.time()
    while time.time() - start < time_limit:
        # small random perturbation to escape possible stagnation
        perturb = np.random.uniform(-0.05, 0.05, size=n)
        x0 = np.clip(best_h + perturb, 0.0, 1.0)
        # re‑project to the affine sum constraint by a simple shift
        shift = (target_sum - x0.sum()) / n
        x0 += shift
        x0 = np.clip(x0, 0.0, 1.0)
        # final fix of the sum by distributing the residual uniformly
        residual = target_sum - x0.sum()
        if abs(residual) > 1e-12:
            # distribute residual to entries not at a bound
            free = (x0 > 0.0) & (x0 < 1.0)
            if free.any():
                x0[free] += residual / free.sum()
                x0 = np.clip(x0, 0.0, 1.0)
            else:
                # fallback: scale whole vector
                x0 *= target_sum / x0.sum()
        res = minimize(_c5_of, x0, method='SLSQP',
                       bounds=bounds, constraints=cons, options=opts)
        if not res.success:
            # if optimizer failed, just try another random start
            continue
        val = _c5_of(res.x)
        if val < best_val:
            best_val = val
            best_h = res.x.copy()
    return best_h, best_val

def _local_search(best_h, best_val, time_limit):
    """Simple stochastic local improvement keeping feasibility."""
    n = best_h.size
    target_sum = n / 2.0
    h = best_h.copy()
    val = best_val
    start = time.time()
    while time.time() - start < time_limit:
        i, j = np.random.choice(n, 2, replace=False)
        # propose moving a small amount from i to j (or opposite)
        delta = np.random.uniform(-0.02, 0.02)
        hi, hj = h[i], h[j]
        hi_new = np.clip(hi + delta, 0.0, 1.0)
        delta_actual = hi_new - hi
        hj_new = np.clip(hj - delta_actual, 0.0, 1.0)
        # adjust delta to keep the sum exact
        delta_actual = hi_new - hi
        hj_new = hj - delta_actual
        if not (0.0 <= hj_new <= 1.0):
            continue
        h_candidate = h.copy()
        h_candidate[i] = hi_new
        h_candidate[j] = hj_new
        # enforce exact sum (tiny numerical drift)
        drift = target_sum - h_candidate.sum()
        if abs(drift) > 1e-10:
            # distribute drift uniformly over non‑boundary entries
            free = (h_candidate > 0.0) & (h_candidate < 1.0)
            if free.any():
                h_candidate[free] += drift / free.sum()
                h_candidate = np.clip(h_candidate, 0.0, 1.0)
            else:
                continue
        cand_val = _c5_of(h_candidate)
        if cand_val < val:
            h = h_candidate
            val = cand_val
    return h, val
