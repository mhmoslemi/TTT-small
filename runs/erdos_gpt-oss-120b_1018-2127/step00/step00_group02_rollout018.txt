import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

# Helper to compute the C5 value for a discretized step function h
def compute_c5(h):
    """Return max overlap integral for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlate h with (1-h) using zero padding (mode='full')
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    return overlaps.max()


def upsample(old_h, new_n):
    """Linear interpolation of old_h (defined on [0,2)) to new_n points."""
    old_n = old_h.size
    # grid points (excluding the right endpoint to keep periodicity consistent)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)


def _objective(h):
    """Objective for optimizer: the C5 bound."""
    return compute_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h on [0,2] to lower the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized values on a uniform grid; shape (n_points,).

        The achieved C5 bound (max overlap integral).
    n_points : int
        Grid size used for the returned h.

    # Prepare an initial guess: use the provided construction if available.

    best_h = None
    best_c5 = np.inf
    best_n = None

    for n in candidate_ns:

        if time.time() - start_time > budget_s * 0.95:
            break

            np.random.shuffle(h0)

        # Enforce the sum constraint exactly (integral = 1 ⇒ sum = n/2)

        cur_sum = h0.sum()
        target_sum = n / 2.0
