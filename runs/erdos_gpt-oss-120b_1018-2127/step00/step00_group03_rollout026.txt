import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Stochastic local search for a step function h on [0,2] minimizing
    C_5 = max_k ∫ h(x)(1-h(x+k)) dx .
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    else:
        n = 70
        h = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    dx = 2.0 / n

    # ----------------------------------------------------------------------
    # Projection onto feasibility region
    def project(vec):
        vec = np.clip(vec, 0.0, 1.0)
        for _ in range(100):
            s = vec.sum()
            if abs(s - target_sum) < 1e-12:
                break
            diff = target_sum - s
            free = (vec > 0.0) & (vec < 1.0)
            cnt = free.sum()
            if cnt == 0:
                # fallback: scale whole vector then re‑clip
                if s == 0:
                    break
                vec = np.clip(vec * (target_sum / s), 0.0, 1.0)
                continue
            delta = diff / cnt
            vec[free] += delta
            vec = np.clip(vec, 0.0, 1.0)
        return vec

    h = project(h)

    # ----------------------------------------------------------------------
    # Objective evaluation
    def compute_c5(vec):
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # ----------------------------------------------------------------------
    # Search parameters
    max_iter = 200000
    iter_cnt = 0
    base_temp = 0.2

    while time.time() - start_time < budget_s * 0.95:
        iter_cnt += 1
        # simple linear cooling
        temperature = base_temp * (1.0 - iter_cnt / max_iter)

        # propose a new candidate by perturbing a few entries
        cand = best_h.copy()
        k = np.random.randint(1, 4)                     # 1–3 coordinates
        idx = np.random.choice(n, size=k, replace=False)
        cand[idx] += np.random.normal(scale=temperature, size=k)
        cand = project(cand)

        cand_score = compute_c5(cand)

        # accept if improvement or Metropolis criterion
        if cand_score < best_score or np.random.rand() < np.exp(-(cand_score - best_score) / (temperature + 1e-12)):
            best_h = cand
            best_score = cand_score

        # occasional random restart to escape stagnation
        if iter_cnt % 5000 == 0:
            rnd = np.random.rand(n)
            rnd = project(rnd)
            rnd_score = compute_c5(rnd)
            if rnd_score < best_score:
                best_h = rnd
                best_score = rnd_score

    final_c5 = compute_c5(best_h)
    return best_h, float(final_c5), n
