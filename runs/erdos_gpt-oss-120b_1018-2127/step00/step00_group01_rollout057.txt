import time
import numpy as np
from scipy import optimize as opt

def _c5_of_h(h):
    """Compute C5 for a discrete step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation h with (1-h); mode='full' gives length 2n-1
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _penalized_obj(h, target_sum, penalty):
    """Objective with equality penalty for the integral constraint."""
    # box constraints are handled by DE bounds, still clip for safety
    h = np.clip(h, 0.0, 1.0)
    c5 = _c5_of_h(h)
    viol = np.abs(h.sum() - target_sum)
    return c5 + penalty * viol

        lambda x: _c5_of_h(x),

        options=opts,
