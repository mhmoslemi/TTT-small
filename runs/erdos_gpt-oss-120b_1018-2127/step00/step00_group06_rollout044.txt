import time
import numpy as np

def _objective(h, dx):
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_constraints(h, target_sum):
    # clip to [0,1] and enforce sum == target_sum via simple redistribution
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        free_cnt = free.sum()
        if free_cnt == 0:
            # all at bounds, distribute uniformly (may violate bounds slightly)
            h += diff / len(h)
        else:
            h[free] += diff / free_cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _random_walk(h, scale, rng):
    perturb = rng.normal(loc=0.0, scale=scale, size=h.shape)
    return h + perturb

def _interpolate(prev_h, new_n):
    old_n = len(prev_h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, prev_h)
