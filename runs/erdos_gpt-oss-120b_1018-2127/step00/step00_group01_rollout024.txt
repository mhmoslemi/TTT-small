import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    # correlation of h with (1-h), full mode gives shifts from -(n-1) .. (n-1)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def _initial_h(n, rng):
    """Create a feasible starting vector of length n."""
    # start from a simple alternating pattern then add small noise
    h = np.zeros(n, dtype=float)
    half = n // 2
    # place exactly half the mass as 1's (feasible for integral)
    h[:half] = 1.0
    rng.shuffle(h)
    # add tiny uniform noise and re‑project to keep sum exact
    noise = rng.uniform(-0.02, 0.02, size=n)
    h = np.clip(h + noise, 0.0, 1.0)
    # enforce exact sum = n/2 by scaling the deviation uniformly
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the difference over indices that are not at bounds
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # fallback: adjust first element safely
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        else:
            adj = diff / free.size
            h[free] = np.clip(h[free] + adj, 0.0, 1.0)
            # final tiny correction
            h += (target - h.sum()) / n
    return h


def _perturb(h, rng):
    """Return a new feasible h by moving a small amount from one index to another."""
    n = h.size
    # choose donor (decrease) and receiver (increase) respecting bounds
    donors = np.where(h > 1e-12)[0]
    receivers = np.where(h < 1 - 1e-12)[0]
    if donors.size == 0 or receivers.size == 0:
        return None
    i = rng.choice(donors)
    j = rng.choice(receivers)
    # maximal admissible transfer
    max_delta = min(0.05, h[i], 1.0 - h[j])
    if max_delta <= 0:
        return None
    delta = rng.uniform(0, max_delta)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
