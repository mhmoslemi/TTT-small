import numpy as np
import time
import random

def _compute_c(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _repair_sum(h):
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to fix in one shot
    for i in range(n):
        if 0.0 <= h[i] + diff <= 1.0:
            h[i] += diff
            return h
    # otherwise distribute greedily
    remaining = diff
    for i in range(n):
        if remaining == 0.0:
            break
        if remaining > 0.0:
            cap = 1.0 - h[i]
            add = min(cap, remaining)
            h[i] += add
            remaining -= add
        else:
            cap = -h[i]
            add = max(cap, remaining)
            h[i] += add
            remaining -= add
    return h

def _upscale(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + (1.0 / old_n)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + (1.0 / new_n)
    new_h = np.interp(new_x, old_x, h)
    new_h = _repair_sum(new_h)
    return new_h

def _hill_climb(start_h, time_limit):
    n = len(start_h)
    target_sum = n / 2.0
    best_h = start_h.copy()
    best_c = _compute_c(best_h)
    start_time = time.time()
    # schedule step size: start large, shrink to min_step
    max_step = 0.15
    min_step = 0.001
    while True:
        now = time.time()
        if now - start_time >= time_limit:
            break
        elapsed = now - start_time
        frac = elapsed / time_limit
        step = max_step * (1.0 - frac) + min_step * frac
        i, j = random.sample(range(n), 2)
        # feasible delta range respecting bounds
        delta_low = max(-step, -best_h[i], best_h[j] - 1.0)
        delta_high = min(step, 1.0 - best_h[i], best_h[j])
        if delta_low >= delta_high:
            continue
        delta = random.uniform(delta_low, delta_high)
        trial = best_h.copy()
        trial[i] += delta
        trial[j] -= delta
        # tiny numerical repair
        trial = _repair_sum(trial)
        c = _compute_c(trial)
        if c < best_c:
            best_h, best_c = trial, c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_total = time.time()

    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 45
        h = np.full(n0, 0.5, dtype=float)  # satisfies integral automatically
    h = _repair_sum(h)
    best_h, best_c = h.copy(), _compute_c(h)

    # ------------------------------------------------------------------
    # Schedule: three resolutions (45 → 90 → 180) with time slices
    # ------------------------------------------------------------------
    total_budget = budget_s
    schedule = [
        (len(best_h), 0.30),          # keep current size for 30% time
        (len(best_h) * 2, 0.35),     # double size for 35% time
        (len(best_h) * 4, 0.30),     # quadruple size for 30% time
    ]

    for new_n, fraction in schedule:
        remaining = total_budget - (time.time() - start_total)
        if remaining <= 0:
            break
        phase_time = total_budget * fraction
        if phase_time > remaining:
            phase_time = remaining

        if new_n != len(best_h):
            best_h = _upscale(best_h, new_n)
            best_h = _repair_sum(best_h)

        best_h, best_c = _hill_climb(best_h, phase_time)

        # early exit if almost out of budget
        if time.time() - start_total > total_budget * 0.98:
            break

    # Final evaluation
    final_c = _compute_c(best_h)
    return best_h, final_c, len(best_h)
