import numpy as np
import random
import time

def _repair(h, target_sum, max_iter=10):
    """Clip to [0,1] and adjust to meet the target sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        if s == 0:
            # avoid division by zero – spread uniformly
            h[:] = target_sum / len(h)
            break
        scale = target_sum / s
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    # final normalization if still off due to clipping
    s = h.sum()
    if s != target_sum:
        diff = target_sum - s
        # distribute diff proportionally to free capacity
        free_up = 1.0 - h
        free_down = h
        if diff > 0:
            cap = free_up.sum()
            if cap > 0:
                h += diff * (free_up / cap)
        else:
            cap = free_down.sum()
            if cap > 0:
                h += diff * (free_down / cap)  # diff negative
        h = np.clip(h, 0.0, 1.0)
    return h

def _evaluate(h, dx):
    """Return C5 for a given h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _mutate(parent, target_sum, frac_resample=0.2, frac_noise=0.3, sigma=0.04):
    """Create a mutated child from parent."""
    n = len(parent)
    child = parent.copy()

    if random.random() < 0.5:  # strategy A: resample
        k = max(1, int(frac_resample * n))
        idx = np.random.choice(n, k, replace=False)
        child[idx] = np.random.rand(k)
    else:  # strategy B: add noise
        mask = np.random.rand(n) < frac_noise
        noise = sigma * np.random.randn(mask.sum())
        child[mask] += noise

    child = _repair(child, target_sum)
    return child
