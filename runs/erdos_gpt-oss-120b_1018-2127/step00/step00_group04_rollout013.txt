def _compute_c5(h, dx):
    """Return C5 for a discrete step function h on [0,2] with spacing dx."""
    # correlation of h with (1-h). mode='full' gives length 2*n-1, offsets -n+1..n-1
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def _project_bounds_and_mass(h, dx):
    """Clip to [0,1] and re‑scale to satisfy sum(h)*dx == 1."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum() * dx
    if total == 0:
        # distribute uniformly
        h[:] = 0.5
        total = h.sum() * dx
    # scale uniformly to hit the exact mass
    factor = 1.0 / total
    h *= factor
    # after scaling some values may leave bounds, clip again and redistribute the error
    # simple iterative correction (few passes are enough for modest n)
    for _ in range(3):
        excess_low = -np.minimum(h, 0.0).sum()
        excess_high = np.maximum(h - 1.0, 0.0).sum()
        if excess_low == 0 and excess_high == 0:

        if excess_low > 0:
            mask = h < 0
            h[mask] = 0.0

        # adjust remaining mass
        mass = h.sum() * dx
        if mass != 0:
