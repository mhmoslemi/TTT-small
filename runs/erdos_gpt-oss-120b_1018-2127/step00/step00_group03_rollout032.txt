import time
import numpy as np

# evaluate_erdos_solution is provided in the execution environment
# initial_h_values may also be provided.

def _project_pair_move(h, i, j, eps):
    """Move eps from j to i while keeping bounds [0,1]."""
    h[i] += eps
    h[j] -= eps
    return h

def _neighbor(rng, h, step):
    n = h.size
    i, j = rng.choice(n, 2, replace=False)
    # propose a signed epsilon
    eps = (rng.random() * 2.0 - 1.0) * step
    # enforce bounds for both coordinates
    lo = max(-h[i], h[j] - 1.0)          # eps must be >= -h[i] and >= h[j]-1
    hi = min(1.0 - h[i], h[j])          # eps must be <= 1-h[i] and <= h[j]
    if lo > hi:      # no feasible move, choose another pair
        return None, None, None
    eps = np.clip(eps, lo, hi)
    if abs(eps) < 1e-12:
        return None, None, None
    h_new = h.copy()
    _project_pair_move(h_new, i, j, eps)
    return h_new, i, j

    # Ensure feasibility (clip + mean 0.5)
    h0 = np.clip(h0, 0.0, 1.0)
    current_sum = h0.sum()
    target_sum = n * 0.5
    if abs(current_sum - target_sum) > 1e-12:
        # Redistribute the difference uniformly, then clip again
        diff = target_sum - current_sum
        h0 += diff / n
