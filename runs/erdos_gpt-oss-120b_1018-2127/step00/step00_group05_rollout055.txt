import time
import numpy as np

def _project_feasible(v, target, max_iter=20, tol=1e-9):
    """Project vector v onto {0<=h<=1, sum(h)=target} by simple iterative water‑filling."""
    h = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        # indices that can move in the needed direction
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        n = mask.sum()
        if n == 0:
            break
        delta = diff / n
        # apply delta but keep bounds
        h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if still off (distribute uniformly)
    s = h.sum()
    if abs(s - target) > tol:
        correction = (target - s) / len(h)
        h += correction
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h, dx):
    """Return the C5 value for a feasible step vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
