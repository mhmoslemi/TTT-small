import time
import numpy as np

def _dx(n):
    return 2.0 / n

def _c5_score(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * _dx(n)

def _resample(old_h, n_new):
    n_old = old_h.shape[0]
    # positions on [0,2)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    # enforce integral exactly 1
    new_h = new_h * (n_new / (2.0 * np.sum(new_h)))  # scale to sum = n_new/2
    # clip to [0,1] after scaling; if clipping changes sum, re‑scale once more
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = new_h * (n_new / (2.0 * np.sum(new_h)))
    return new_h

def _local_search(h_init, budget, start_time):
    h = h_init.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = _c5_score(h)
    # parameters
    max_iter = 20000
    temp0 = 0.1
    np.random.seed(0)
    for it in range(max_iter):
        if time.time() - start_time > budget:
            break
        # temperature schedule
        temp = temp0 * (1.0 - it / max_iter)
        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = np.random.uniform(-0.05, 0.05) * temp / temp0  # scale with temp
        hi, hj = h[i] + delta, h[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h_candidate = h.copy()
        h_candidate[i] = hi
        h_candidate[j] = hj
        cand_score = _c5_score(h_candidate)
        if cand_score < best_score - 1e-12 or np.random.rand() < np.exp((best_score - cand_score) / (temp + 1e-12)):
            h = h_candidate
            if cand_score < best_score:
                best_score = cand_score
                best_h = h_candidate.copy()
    return best_h, best_score
