import numpy as np
import time

def _project(vec, target):
    """Clip to [0,1] and enforce sum == target (iterative)."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(5):
        s = v.sum()
        if s == 0.0:
            v[:] = target / len(v)
            break
        factor = target / s
        v = np.clip(v * factor, 0.0, 1.0)
    # small residual correction
    diff = target - v.sum()
    if abs(diff) > 1e-10:
        free = (v > 0.0) & (v < 1.0)
        if free.any():
            v[free] += diff / free.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    """Compute C5 for discrete step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _hill_climb(start_h, max_iter, deadline, best_state):
    """Simple stochastic hill‑climbing with occasional uphill moves."""
    h = start_h.copy()
    n = len(h)
    target = n / 2.0
    cur = _c5_value(h)

    if cur < best_state[1]:
        best_state[:] = [h.copy(), cur]

    for it in range(max_iter):
        if time.time() > deadline:
            break
        # cooling schedule for step size
        sigma = 0.12 * (1.0 - it / max_iter)
        cand = h + np.random.normal(scale=sigma, size=n)
        cand = _project(cand, target)
        cand_val = _c5_value(cand)

        # Metropolis acceptance (helps escape shallow local minima)
        if cand_val < cur or np.random.rand() < np.exp((cur - cand_val) / (sigma + 1e-12)):
            h, cur = cand, cand_val
            if cur < best_state[1]:
                best_state[:] = [h.copy(), cur]

    return h
