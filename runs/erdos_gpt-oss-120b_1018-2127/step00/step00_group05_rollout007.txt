import time
import random
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def _repair(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to exactly target_sum (iterative)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3):
        s = h.sum()
        if s == 0:
            h[:] = target_sum / h.size
            break
        scale = target_sum / s
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
    # final normalization to hit target exactly (should already be close)
    s = h.sum()
    if s != 0:
        h = h * (target_sum / s)
    return h


def _evaluate(h: np.ndarray, dx: float) -> float:
    """C5 = dx * max correlation of h with (1-h)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(corr.max())


def _mutate_resample(parent: np.ndarray, target_sum: float, frac: float = 0.2) -> np.ndarray:
    """Replace a fraction of entries by fresh uniform samples."""
    n = parent.size
    child = parent.copy()
    k = max(1, int(frac * n))
    idx = np.random.choice(n, size=k, replace=False)
    child[idx] = np.random.rand(k)
    return _repair(child, target_sum)


def _mutate_perturb(parent: np.ndarray, target_sum: float,
                    frac: float = 0.3, amp: float = 0.015) -> np.ndarray:
    """Add small random noise to a fraction of entries."""
    n = parent.size
    child = parent.copy()
    mask = np.random.rand(n) < frac
    noise = (np.random.rand(mask.sum()) * 2.0 - 1.0) * amp
    child[mask] += noise
    return _repair(child, target_sum)


def _initialize_population(base_h: np.ndarray, popsize: int, target_sum: float) -> list:
    """Create an initial population from the base vector and random variations."""
    pop = [_repair(base_h.copy(), target_sum)]
    while len(pop) < popsize:
        rnd = np.random.rand(base_h.size)
        pop.append(_repair(rnd, target_sum))
    return pop

    dx = 2.0 / n_points

    # ------------------------------------------------------------------

    POP_SIZE = 12
