import time
import numpy as np
import random

# pre‑imported in the harness:
# evaluate_erdos_solution(h) -> float   (the C5 value)
# initial_h_values                       (optional starting vector)

def _project_to_box_and_sum(v, target, max_iter=50):
    """Projection of v onto [0,1]^n with sum == target."""
    # find tau such that sum(clip(v - tau,0,1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        tau = (lo + hi) * 0.5
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _random_move(h, step):
    """Return a new candidate by moving mass between two random indices."""
    n = h.shape[0]
    i, j = random.sample(range(n), 2)
    delta = random.uniform(-step, step)
    # propose transfer from i to j
    new = h.copy()
    new[i] += delta
    new[j] -= delta
    return new
