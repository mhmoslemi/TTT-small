def repair(h):
    # clip
    h = np.clip(h, 0.0, 1.0)
    target = len(h)/2.0
    current = h.sum()
    diff = target - current
    # Distribute diff proportionally to capacity remaining
    # If diff>0, need increase some values without exceeding 1.
    # If diff<0, need decrease some values without below 0.
    # We'll iteratively adjust.
    # Choose indices sorted by distance to bound.
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # capacities
        caps = 1.0 - h
        total_cap = caps.sum()
        if total_cap < diff - 1e-12:
            # cannot fix, set to max
            h = np.clip(h + diff/len(h), 0,1)
            # final clip
        else:
            # proportionally allocate
            h += caps * (diff / total_cap)
    else:
        caps = h
        total_cap = caps.sum()
        if total_cap < -diff - 1e-12:
            h = np.clip(h + diff/len(h), 0,1)
        else:
            h -= caps * ((-diff) / total_cap)
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # after adjustment sum may still be slightly off due to clipping; do simple correction by adding/subtracting small epsilon uniformly
    current = h.sum()
    diff = target - current
    if abs(diff) > 1e-9:
        h += diff/len(h)
        h = np.clip(h,0,1)
    return h

import time, math
import numpy as np

def _evaluate(h):
    """Return max overlap C5 for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h):
    """Project vector onto [0,1] box and onto hyperplane sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        total = cap.sum()
        if total > 0:
            h += cap * (diff / total)
    else:
        cap = h
        total = cap.sum()
        if total > 0:
            h -= cap * ((-diff) / total)
    h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _interpolate(src, new_n):
    """Linear interpolation of a periodic vector src onto new_n points."""
    old_n = src.size
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, src)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 minimizing C5."""
    np.random.seed(seed)
    start = time.time()
    def time_left():
        return budget_s - (time.time() - start)

    # ----- initial construction -----
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        # fallback: random feasible vector of length 100
        n0 = 100
        h0 = np.random.rand(n0)
        h0 = _repair(h0)
    else:
        n0 = h0.size
        h0 = _repair(h0)

    best_h = h0.copy()
    best_score = _evaluate(best_h)

    # ----- multigrid schedule -----
    levels = [n0]
    max_n = 760
    while levels[-1] < max_n:
        nxt = min(max_n, levels[-1] * 2)
        if nxt == levels[-1]:
            break
        levels.append(nxt)

    prev_best = h0.copy()

    for n in levels:
        if time_left() < 2.0:
            break

        # interpolate to current grid size if needed
        if n != prev_best.size:
            cur_h = _interpolate(prev_best, n)
        else:
            cur_h = prev_best.copy()
        cur_h = _repair(cur_h)
        cur_score = _evaluate(cur_h)

        # simulated‑annealing parameters
        T = 1.0
        sigma = 0.07  # initial perturbation magnitude
        # allocate about half of the remaining time to this level
        level_deadline = time.time() + time_left() * 0.5

        while time.time() < level_deadline:
            # propose a new point
            proposal = cur_h + np.random.randn(n) * sigma
            proposal = _repair(proposal)
            prop_score = _evaluate(proposal)

            # acceptance test
            if prop_score < cur_score or np.random.rand() < math.exp(-(prop_score - cur_score) / max(T, 1e-12)):

                cur_score = prop_score

            # keep global best
            if prop_score < best_score:
                best_score = prop_score
                best_h = proposal.copy()

            # cooling / step size decay
            T *= 0.9997
            sigma *= 0.9999

        prev_best = cur_h

    # final sanity check (ensure feasibility)
    best_h = _repair(best_h)
    best_score = _evaluate(best_h)

    return best_h, best_score, best_h.size
