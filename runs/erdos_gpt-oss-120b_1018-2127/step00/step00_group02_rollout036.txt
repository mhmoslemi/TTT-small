import numpy as np
import time
from math import inf

def _evaluate_c5(h):
    """Return C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _proj_cap_simplex(v, target):
    """
    Euclidean projection of vector v onto the set:
        0 <= x_i <= 1  and  sum x_i = target
    Uses a binary search for the Lagrange multiplier.
    """
    lo, hi = -1e5, 1e5
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def _optimize(n, init_h, time_budget, seed):
    """
    Heuristic optimizer for a fixed resolution n.
    Returns best vector and its C5 value found within time_budget seconds.
    """
    rng = np.random.default_rng(seed)
    target_sum = n / 2.0
    # start from a feasible copy of init_h
    h_best = _proj_cap_simplex(np.array(init_h, copy=True), target_sum)
    best_val = _evaluate_c5(h_best)

    start = time.time()
    it = 0
    sigma = 0.3  # initial mutation scale
    while time.time() - start < time_budget:
        it += 1
        # exponential decay of sigma
        sigma = 0.3 * (0.99 ** (it // 10))
        cand = h_best + sigma * rng.standard_normal(n)
        cand = _proj_cap_simplex(cand, target_sum)
        val = _evaluate_c5(cand)
        if val < best_val:
            h_best, best_val = cand, val
        # occasional random restart to avoid stagnation
        if rng.random() < 0.005:
            rnd = rng.random(n)
            rnd = _proj_cap_simplex(rnd, target_sum)
            rnd_val = _evaluate_c5(rnd)
            if rnd_val < best_val:
                h_best, best_val = rnd, rnd_val
    return h_best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
    else:
        n0 = 45
        # simple feasible start: first n/2 entries 1, rest 0
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)])
    # allocate half of budget to this stage
    budget_stage1 = budget_s * 0.35
    h_best, best_val = _optimize(n0, h0, budget_stage1, seed + 1)

    # ------------------------------------------------------------------
    # 2. Upscale to higher resolution for finer control
    # ------------------------------------------------------------------
    # Choose a higher n that is a multiple of n0 for easy interpolation
    n1 = n0 * 2  # 90 when n0=45
    x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n1, endpoint=False)
