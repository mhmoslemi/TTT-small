import numpy as np, time, math, random, sys
from scipy.optimize import dual_annealing

# The harness provides evaluate_erdos_solution and possibly initial_h_values.
# We must not copy any large data literals.

def _project_onto_capped_simplex(v, target_sum, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    """Project vector v onto the set {x | lower <= x_i <= upper, sum x_i = target_sum}."""
    lo = lower - np.max(v)  # ensure low enough lambda
    hi = upper - np.min(v)  # ensure high enough lambda
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, lower, upper)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2.0, lower, upper)

def _c5_value(h, dx):
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx via discrete correlation."""
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(conv.max())

def _initial_grid(initial, n_points):
    """Interpolate a possibly shorter initial vector onto a length-n_points grid."""
    if initial is None:
        # simple balanced start: first half 1, rest 0
        h = np.zeros(n_points)
        h[:n_points // 2] = 1.0
        return h
    m = len(initial)
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
    h = np.interp(x_new, x_old, initial)
    # enforce sum constraint exactly by projection
    target = n_points / 2.0
    h = _project_onto_capped_simplex(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # Choose a modest grid size (fast eval, enough resolution)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Obtain initial vector if provided by the harness
    init_vec = globals().get('initial_h_values', None)
    h_cur = _initial_grid(init_vec, n_points)
    best_h = h_cur.copy()
    best_c5 = _c5_value(best_h, dx)

    # Simple stochastic hill‑climbing with simulated‑annealing temperature schedule
    temperature = 0.05
    temp_decay = 0.9995
    iteration = 0
    max_iterations = 200000  # safety cap

    while True:
        now = time.time()
        if now - start_time > budget_s * 0.98:
            break
        if iteration >= max_iterations:
            break
        iteration += 1

        # Propose a perturbation: modify a small subset of entries
        proposal = best_h.copy()
        # Number of coordinates to tweak, decreasing with time
        n_change = max(1, int(0.02 * n_points * (1 - iteration / max_iterations)))
        idx = np.random.choice(n_points, size=n_change, replace=False)
        noise = np.random.normal(scale=0.2, size=n_change)
        proposal[idx] += noise
        # Project back to feasible set
        proposal = _project_onto_capped_simplex(proposal, target_sum)

        cand_c5 = _c5_value(proposal, dx)

        # Accept if better, else probabilistically
        if cand_c5 < best_c5 or random.random() < math.exp(-(cand_c5 - best_c5) / temperature):
            best_h, best_c5 = proposal, cand_c5
