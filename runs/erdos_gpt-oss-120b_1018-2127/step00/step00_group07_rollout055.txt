import time, math, random
import numpy as np

# Helper: compute C5 for a given step vector h
def _c5_of(h):
    # use provided evaluator if available, else fallback
    if 'evaluate_erdos_solution' in globals():
        try:
            return evaluate_erdos_solution(h)
        except Exception:
            pass
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# Helper: enforce 0≤h≤1 and the integral constraint sum(h)=n/2
def _repair(h):
    n = h.size
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        h[:] = 0.0
        h[:n // 2] = 1.0
        return h
    # scale to hit the target sum, then clip again
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # a tiny iterative refinement to get the sum exact (few iterations)
    for _ in range(3):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        # distribute diff proportionally to non‑boundary elements
        mask = (h > 0.0) & (h < 1.0)
        if not np.any(mask):
            # fallback: adjust a random free index
            idx = random.randrange(n)
            h[idx] = min(max(h[idx] + diff, 0.0), 1.0)
            continue
        adjust = diff / mask.sum()
        h[mask] += adjust
        h = np.clip(h, 0.0, 1.0)
    return h


# Helper: produce a higher‑resolution copy by linear interpolation
def _upsample(h, new_n):
    old_n = h.size
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _repair(h_new)

    total_time = budget_s
    # allocate a fraction for the first resolution, rest for up‑sampling steps
    first_phase = 0.55 * total_time
    second_phase = 0.35 * total_time
    final_phase = 0.10 * total_time

    # ----------- phase 1 : work on the original resolution ---------------
    def _anneal(h, time_limit, init_temp=5.0, cooling=0.995):
        nonlocal best_h, best_score
