def project_feasible(x, n):
    x = np.clip(x, 0.0, 1.0)
    target = n/2.0
    for _ in range(10):
        diff = target - np.sum(x)
        if abs(diff) < 1e-8:
            break
        free = (x > 0.0 + 1e-12) & (x < 1.0 - 1e-12)
        count = np.count_nonzero(free)
        if count == 0:
            # distribute to all with slack
            slack_low = x - 0.0
            slack_high = 1.0 - x
            total_slack = np.sum(slack_low + slack_high)
            if total_slack == 0:
                break
            # proportionally adjust
            # Simple: add diff/n uniformly then clip again
            delta = diff / n
            x += delta
        else:
            delta = diff / count
            x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final adjustment: if still off due to rounding, adjust by adding to first element within bounds
    diff = target - np.sum(x)
    if abs(diff) > 1e-6:
        for i in range(n):
            if diff > 0 and x[i] < 1.0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0 and x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
            if abs(diff) < 1e-8:
                break
    return x

import time
import numpy as np
from scipy.optimize import minimize

def compute_score(h):
    """Maximum overlap integral for step vector h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def project_feasible(x, n):
    """Clip to [0,1] and enforce sum(x)=n/2."""
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    for _ in range(15):
        diff = target - x.sum()
        if abs(diff) < 1e-9:
            break
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            delta = diff / n
            x += delta
        else:
            delta = diff / cnt
            x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        for i in range(n):
            if diff > 0 and x[i] < 1.0:
                inc = min(diff, 1.0 - x[i])
                x[i] += inc
                diff -= inc
            elif diff < 0 and x[i] > 0.0:
                dec = min(-diff, x[i])
                x[i] -= dec
                diff += dec
            if abs(diff) < 1e-9:
                break
    return x

def upscale(h, n_new):
    """Linear interpolation from old grid to a finer grid."""
    n_old = h.size
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def try_optimize(h0, n, best):
    """Run SLSQP from h0, update best tuple (score, h, n) if improved."""
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0},)

    res = minimize(
        compute_score,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    )
    if not res.success:
        return best
    h_opt = project_feasible(res.x, n)
    score = compute_score(h_opt)
    if score < best[0]:
        return (score, h_opt, n)
    return best

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) minimizing the overlap constant."""
    np.random.seed(seed)
    start = time.time()
    best = (np.inf, None, None)  # (score, h, n)

    # ----- base resolution (n = 45) ------------------------------------
    n_base = 45
    try:
        h_start = np.array(initial_h_values, dtype=float)
        if h_start.size != n_base:
            h_start = upscale(h_start, n_base)
    except NameError:
        h_start = np.random.rand(n_base)
    h_start = project_feasible(h_start, n_base)

    best = try_optimize(h_start, n_base, best)

    # ----- random restarts at base resolution -------------------------
    while time.time() - start < budget_s * 0.45:
        if best[1] is not None and np.random.rand() < 0.7:
            perturb = best[1] + 0.1 * np.random.randn(n_base)
        else:
            perturb = np.random.rand(n_base)
        perturb = project_feasible(perturb, n_base)
        best = try_optimize(perturb, n_base, best)

    # ----- move to finer grid (n = 90) if time permits -----------------
    if time.time() - start < budget_s * 0.85:
        n_fine = 90
        if best[1] is not None:
            h_fine = upscale(best[1], n_fine)
        else:
            h_fine = upscale(h_start, n_fine)
        h_fine = project_feasible(h_fine, n_fine)
        best = try_optimize(h_fine, n_fine, best)

        # few extra restarts on fine grid
        while time.time() - start < budget_s - 5:
            rand = np.random.rand(n_fine)
            rand = project_feasible(rand, n_fine)
            best = try_optimize(rand, n_fine, best)

    # ----- fallback if nothing succeeded --------------------------------
    if best[1] is None:
        best = (compute_score(h_start), h_start, n_base)

    # Ensure returned vector respects constraints (tiny numerical drift)
    h_ret = project_feasible(best[1], best[2])
    c5_bound = float(compute_score(h_ret))
    return (h_ret, c5_bound, int(best[2]))
