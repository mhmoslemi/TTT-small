import numpy as np
import time

def _objective(h, dx):
    """Maximum overlap integral for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_bounds_sum(v, target_sum, max_iter=60):
    """
    Euclidean projection of vector v onto the cube [0,1]^n intersected with
    the hyperplane sum(x)=target_sum.  This reduces to finding a shift λ such
    that sum(clip(v+λ,0,1)) = target_sum.
    """
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _interpolate(old_h, new_n):
    """Linear interpolation of a periodic vector onto a finer grid."""
    old_n = len(old_h)
    # domain points, exclude the endpoint to keep periodicity
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)

def _optimize_grid(h_start, max_time):
    """Simple simulated‑annealing style optimization on a fixed grid."""
    n = len(h_start)
    dx = 2.0 / n
    target_sum = n / 2.0

    # initial state
    h_cur = _project_bounds_sum(h_start.copy(), target_sum)
    cur_score = _objective(h_cur, dx)

    best_h = h_cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    T = 1.0
    start = time.time()
    it = 0
    # scale of perturbation – modest to keep feasibility after projection
    perturb_scale = 0.02

    while time.time() - start < max_time:
        it += 1
        # generate candidate
        cand = h_cur + rng.normal(scale=perturb_scale, size=n)
        cand = _project_bounds_sum(cand, target_sum)

        cand_score = _objective(cand, dx)

        # accept if better, otherwise with Metropolis probability
        if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / max(T, 1e-12)):
            h_cur, cur_score = cand, cand_score

        if cand_score < best_score:
            best_h, best_score = cand.copy(), cand_score

        # cooling
        T *= 0.9995

    return best_h, best_score
