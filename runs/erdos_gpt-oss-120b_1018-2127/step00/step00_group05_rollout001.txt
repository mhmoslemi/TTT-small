import numpy as np, time, math

def _evaluate(h):
    # pre‑imported helper evaluates the max overlap C5
    return evaluate_erdos_solution(h)

def _repair(h, target_sum):
    # Clip to [0,1] then adjust linearly to meet exact sum while staying in bounds
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        return np.full_like(h, target_sum / h.size)
    # simple scaling, may violate bounds; iterate once more
    h = np.clip(h * (target_sum / cur), 0.0, 1.0)
    cur = h.sum()
    if not math.isclose(cur, target_sum, rel_tol=1e-9, abs_tol=1e-9):
        diff = target_sum - cur
        # distribute diff uniformly on interior points that can absorb it
        mask_up = (diff > 0) & (h < 1.0)
        mask_dn = (diff < 0) & (h > 0.0)
        mask = mask_up if diff > 0 else mask_dn
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    target_sum = n / 2.0          # because dx = 2/n, integral = sum*h*dx = 1

    best_val = _evaluate(best_h)

    # Parameters
    step_scale = 0.02
    max_no_improve = 2000
    no_improve = 0

    while time.time() - start < budget_s * 0.95:
        # propose a transfer of mass from i to j
        i, j = np.random.choice(n, 2, replace=False)
        if h_cur[i] <= 0 or h_cur[j] >= 1:
            continue
        max_delta = min(step_scale, h_cur[i], 1.0 - h_cur[j])
        if max_delta <= 0:

        delta = max_delta * np.random.rand()
        h_new = h_cur.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # repair just in case numeric drift
        h_new = _repair(h_new, target_sum)

        val = _evaluate(h_new)
        if val < best_val - 1e-9:
            best_val, best_h = val, h_new.copy()
            h_cur = h_new
            no_improve = 0
        else:
            # occasional uphill move (simulated annealing style)

                h_cur = h_new
            no_improve += 1

            # random restart keeping feasibility
            h_cur = _repair(np.random.rand(n), target_sum)
            no_improve = 0
