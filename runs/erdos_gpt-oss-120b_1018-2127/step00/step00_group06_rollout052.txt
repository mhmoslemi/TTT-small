import time
import numpy as np

def _project_bounds_sum(v, lo, hi, target):
    """Project onto box [lo,hi] with equality sum=target via bisection on lambda."""
    # clip initial to box
    # binary search lambda so that sum(clip(v - lam, lo, hi)) = target
    lam_low, lam_high = -1e6, 1e6
    for _ in range(60):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    return np.clip(v - lam, lo, hi)

def _objective(h):
    """Compute max overlap integral approximation for step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)

def _interpolate(prev, n_new):
    """Linear interpolation of a periodic vector from size len(prev) to n_new."""
    x_old = np.linspace(0.0, 2.0, len(prev), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, prev)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()

    base_n = 95
    sizes = [base_n, base_n * 2, base_n * 4]          # 95,190,380
    best_h = None
    best_score = np.inf
    best_n = None

    for n in sizes:
        if time.time() - start > budget_s:
            break

        # initialise h for this grid
        if best_h is None:
            if "initial_h_values" in globals() and len(initial_h_values) == n:
                h = np.asarray(initial_h_values, dtype=float).copy()
            else:
                h = np.full(n, 0.5, dtype=float)      # uniform feasible start
        else:
            h = _interpolate(best_h, n)

        target = n / 2.0
        h = _project_bounds_sum(h, 0.0, 1.0, target)

        # local search parameters
        scale = 0.05
        temp = 1.0
        decay = 0.9995
        max_iters = int(3e5)

        for it in range(max_iters):
            if time.time() - start > budget_s:
                break

            # propose a random perturbation and project back to feasible set
            proposal = _project_bounds_sum(h + rng.normal(scale=scale, size=n),
                                           0.0, 1.0, target)

            score_prop = _objective(proposal)

            # keep global best
            if score_prop < best_score:
                best_score = float(score_prop)
                best_h = proposal.copy()
                best_n = n

            # accept if better or with Metropolis probability
            cur_score = _objective(h)
            if score_prop < cur_score or rng.random() < np.exp(-(score_prop - cur_score) / temp):
                h = proposal

            # anneal temperature and perturbation magnitude
            temp *= decay
            if it % 1000 == 0:
                scale *= 0.99

        # after finishing this grid keep the best found so far for next interpolation
        if best_h is not None:

            # (it will be interpolated in the next iteration)

    # fallback if nothing was found
