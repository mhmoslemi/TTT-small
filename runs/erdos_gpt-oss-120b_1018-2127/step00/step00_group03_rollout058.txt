import time
import numpy as np

def compute_c5(h, dx):
    """Return the C5 value for step function h given grid spacing dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_bounds_sum(v, target, max_iter=20, eps=1e-12):
    """
    Euclidean projection of vector v onto the set {x | 0<=x_i<=1, sum x_i = target}.
    Simple iterative water-filling.
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) <= eps:
            break
        if diff > 0:
            mask = x > 0.0  # can decrease these
        else:
            mask = x < 1.0  # can increase these
        if not np.any(mask):
            break
        adjust = diff / mask.sum()
        x[mask] -= adjust
        x = np.clip(x, 0.0, 1.0)
    return x

def improve_h(h, target_sum, dx, time_limit):
    """
    Simple stochastic hill‑climbing / simulated‑annealing loop.
    Returns the best vector found within the allocated time.
    """
    start = time.time()
    best_h = h.copy()
    best_score = compute_c5(best_h, dx)

    n = len(h)
    # annealing schedule
    init_temp = 0.02
    min_temp = 1e-5
    max_iters = 200000

    for it in range(max_iters):
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # temperature linear decay
        temp = init_temp * (1 - it / max_iters)
        temp = max(temp, min_temp)

        # propose a random perturbation that respects sum after projection
        noise = np.random.randn(n) * 0.05
        h_prop = h + noise
        h_prop = project_to_bounds_sum(h_prop, target_sum)

        score = compute_c5(h_prop, dx)

        # acceptance rule
        if score < best_score - 1e-9:
            best_h, best_score = h_prop, score
            h = h_prop
        else:
            # possibly accept worse move
            if np.random.rand() < np.exp(-(score - best_score) / temp):
                h = h_prop
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function to lower the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # -------------------------------------------------
    # initialise h

        init_h = np.asarray(initial_h_values, dtype=float)
    except NameError:
        # fallback: simple alternating 0/1 pattern of length 70
        n_fallback = 70
        init_h = np.tile([0.0, 1.0], n_fallback // 2)
    # optionally upsample to give more flexibility
    if len(init_h) <= 100:
        h = np.repeat(init_h, 2)
    else:
        h = init_h.copy()

    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0

    # make sure the starting point respects constraints
    h = project_to_bounds_sum(h, target_sum)

    # -------------------------------------------------
    # run the optimizer within a fraction of the budget
    remaining = budget_s - (time.time() - start_time)

    # -------------------------------------------------
    # final evaluation
    final_score = compute_c5(best_h, dx)

    # ensure we respect the integral exactly (project once more)
    best_h = project_to_bounds_sum(best_h, target_sum)
