import time
import numpy as np

def _objective(h, dx):
    """Maximum overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair_constraints(h, n):
    """Clip to [0,1] and enforce exact integral sum(h) = n/2."""
    h = np.clip(h, 0.0, 1.0)
    desired = n * 0.5
    for _ in range(10):
        s = h.sum()
        diff = desired - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = int(free.sum())
        if cnt == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / cnt
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny adjustment if needed
    s = h.sum()
    diff = desired - s
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _random_walk(h, scale, rng):
    """Add Gaussian noise with given scale."""
    return h + rng.normal(scale=scale, size=h.shape)

def _interpolate(h, new_n):
    """Linear interpolation of periodic sample h onto new_n points."""
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def _optimize_grid(h_start, n, time_allow, rng):
    """Simple hill‑climbing with decreasing step size."""
    h = _interpolate(h_start, n)
    h = _repair_constraints(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = _objective(best_h, dx)
    start = time.time()
    it = 0
    base_scale = 0.15  # initial noise magnitude
    while time.time() - start < time_allow:
        it += 1
        scale = base_scale * (0.995 ** it)
        cand = _random_walk(best_h, scale, rng)
        cand = _repair_constraints(cand, n)
        score = _objective(cand, dx)
        if score < best_score:
            best_score = score
            best_h = cand
    return best_h, best_score
