import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete approximation of the Erdős overlap constant."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    c5_bound : float
        The achieved C5 value.
    n_points : int

    target_sum = n / 2.0

    # Clip to box constraints and enforce the integral exactly
    h = np.clip(h, 0.0, 1.0)

    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Distribute the deficit/surplus uniformly, then correct edge cases
        h += (target_sum - cur_sum) / n
        h = np.clip(h, 0.0, 1.0)
        # final fix: adjust a single coordinate if needed

                if abs(cur_sum - target_sum) < 1e-12:
