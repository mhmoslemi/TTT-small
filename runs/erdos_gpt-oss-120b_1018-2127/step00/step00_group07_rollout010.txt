import time
import numpy as np
import random
import math

def _evaluate(h, dx):
    # max_k integral = max correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points)
    h_values : 1‑D numpy array with values in [0,1] and sum = n_points/2
    c5_bound : the achieved max overlap value
    n_points : number of discretisation points used
    """
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine discretisation size and initial h
    # ------------------------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
    except NameError:
        # no parent construction supplied
        n = 200
        init = None

    if n % 2 == 1:                     # enforce even length for exact half‑integral sum
        n += 1
        if init is not None:
            init = np.append(init, 0.0)

    target_sum = n // 2                 # because dx = 2/n, integral =1  <=> sum = n/2

    if init is None:
        # random binary start
        h_current = np.zeros(n, dtype=float)
        h_current[:target_sum] = 1.0
        np.random.shuffle(h_current)
    else:
        # project provided vector onto feasible binary set (preserve information as much as possible)
        h_proj = np.clip(init, 0.0, 1.0)
        # pick the largest target_sum entries to be 1, rest 0
        order = np.argsort(-h_proj)               # descending
        h_current = np.zeros(n, dtype=float)
        h_current[order[:target_sum]] = 1.0

    dx = 2.0 / n

    # pre‑compute index lists for fast swaps
    ones = list(np.flatnonzero(h_current == 1.0))
    zeros = list(np.flatnonzero(h_current == 0.0))

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    best_h = h_current.copy()
    best_val = _evaluate(best_h, dx)

    cur_h = h_current.copy()
    cur_val = best_val

    T = 1.0
    decay = 0.9995
    min_T = 1e-6

    # limit iterations loosely; stop earlier when time is up
    max_iters = 10_000_000

    it = 0
    while time.time() - start_time < budget_s * 0.95 and it < max_iters:
        it += 1
        # propose a swap between a 1 and a 0
        i_one = random.choice(ones)
        i_zero = random.choice(zeros)

        # perform swap
        cur_h[i_one] = 0.0
        cur_h[i_zero] = 1.0

        # update helper lists
        ones_idx = ones.index(i_one)
        zeros_idx = zeros.index(i_zero)
        ones[ones_idx] = i_zero
        zeros[zeros_idx] = i_one

        new_val = _evaluate(cur_h, dx)

        accept = False
        if new_val <= cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / T)
            if random.random() < prob:
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                # early exit if we beat target bound
                if best_val < 0.3808:
                    break
        else:
            # revert swap
            cur_h[i_one] = 1.0
            cur_h[i_zero] = 0.0
            ones[ones_idx] = i_one
            zeros[zeros_idx] = i_zero

        # temperature schedule
        T = max(T * decay, min_T)

    # final safety: ensure exact sum (numeric rounding may have occurred)
    current_sum = best_h.sum()
    diff = int(round(target_sum - current_sum))
    if diff > 0:
        # need to turn some zeros into ones
        zero_idxs = np.flatnonzero(best_h == 0.0)
        sel = np.random.choice(zero_idxs, size=diff, replace=False)
        best_h[sel] = 1.0
    elif diff < 0:
        one_idxs = np.flatnonzero(best_h == 1.0)
        sel = np.random.choice(one_idxs, size=-diff, replace=False)
        best_h[sel] = 0.0

    # final evaluation
    final_val = _evaluate(best_h, dx)

    return best_h, float(final_val), n
