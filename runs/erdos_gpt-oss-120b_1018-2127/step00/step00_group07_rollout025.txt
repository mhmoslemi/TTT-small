import time
import numpy as np
from scipy.optimize import minimize

# Helper: project onto capped simplex {x in [lb, ub]^n, sum(x)=target}
def _project_capped_simplex(v, target, lb=0.0, ub=1.0, max_iter=60):
    """Binary search for λ such that sum(clamp(v-λ,lb,ub)) = target."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, lb, ub)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, lb, ub)

def _c5_value(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _objective(x, target_sum):
    """Objective for SLSQP: returns C5; constraints handled externally."""
    # x already respects bounds and sum via projection in the calling code
    return _c5_value(x)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 120
        h0 = np.full(n_points, 0.5)  # fallback uniform
    target_sum = n_points / 2.0  # because dx = 2/n, integral 1 => sum = n/2

    # Ensure feasibility of initial vector
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_capped_simplex(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _c5_value(best_h)

    # Simple local search parameters
    step_scale = 0.2  # initial stddev of perturbations
    decay = 0.9995    # exponential decay per iteration
    iter_cnt = 0
    max_iter = 500000  # safety cap

    while time.time() - start_time < budget_s - 0.2 and iter_cnt < max_iter:
        iter_cnt += 1
        # Random perturbation
        proposal = best_h + rng.normal(scale=step_scale, size=n_points)
        proposal = _project_capped_simplex(proposal, target_sum)
        cand_c5 = _c5_value(proposal)

        if cand_c5 < best_c5:
            best_h, best_c5 = proposal, cand_c5

        # Gradually shrink step size
        step_scale *= decay
        # Occasionally re‑seed with a fresh random feasible point
        if iter_cnt % 30000 == 0:

            rand = _project_capped_simplex(rand, target_sum)
            rand_c5 = _c5_value(rand)
