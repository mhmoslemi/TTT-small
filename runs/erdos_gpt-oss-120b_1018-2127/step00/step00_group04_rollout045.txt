import time
import numpy as np
from scipy import optimize as _opt

def _compute_c5(h, dx):
    """Maximum overlap integral for a discretized step function."""
    # full correlation includes zero padding automatically
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _project_to_feasible(x, n):
    """Clip to [0,1] and enforce the exact sum constraint sum(x)=n/2."""
    x = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    cur = np.sum(x)
    if np.isclose(cur, target):
        return x
    # Simple redistribution of the surplus/deficit over free variables
    free = np.where((x > 0.0) & (x < 1.0))[0]
    if free.size == 0:
        # all at bounds, fallback to scaling uniformly then re‑clip
        x = x * (target / cur)
        return np.clip(x, 0.0, 1.0)
    # Distribute difference equally over free vars
    diff = target - cur
    adjust = diff / free.size
    x[free] += adjust
    # after adjustment may violate bounds; recurse once
    return _project_to_feasible(x, n)

def _optimize_h(start_h, dx, n, time_limit):
    """Run a single SLSQP optimisation, respecting the time limit."""
    cons = ({
        "type": "eq",
        "fun": lambda h: np.sum(h) - n / 2.0,
    })
    bounds = [(0.0, 1.0)] * n

    best_h = None
    best_val = np.inf

    def objective(h):
        return _compute_c5(h, dx)

    # SLSQP may stop early; we enforce iteration limit via time check
    options = {"maxiter": 2000, "ftol": 1e-9, "disp": False}

    res = _opt.minimize(
        objective,
        start_h,
        method="SLSQP",
        jac=False,
        bounds=bounds,
        constraints=cons,
        options=options,
    )
    if res.success:
        h_opt = _project_to_feasible(res.x, n)
        val = _compute_c5(h_opt, dx)
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()
    else:
        # fallback to the feasible projection of the result
        h_feas = _project_to_feasible(res.x, n)
        val = _compute_c5(h_feas, dx)
        if val < best_val:
            best_val = val
            best_h = h_feas.copy()
    return best_h, best_val
