import time
import numpy as np

def _repair(h, target_sum):
    """
    Clip h to [0,1] and scale to meet the target sum.
    Simple iterative scheme (max 5 iterations).
    """
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s == 0:
            # uniform fill to meet target
            h[:] = target_sum / len(h)
            break
        scale = target_sum / s
        h *= scale
    # final clip to guarantee bounds
    h = np.clip(h, 0.0, 1.0)
    # tiny correction for any remaining sum error
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly over non‑saturated entries
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
        else:
            # fallback uniform correction
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_score(h, dx):
    """Compute C5 = max_x ∫ h(t)(1‑h(t+x)) dt using discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
