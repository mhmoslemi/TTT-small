import time
import numpy as np

def _repair(h, target):
    """Clip to [0,1] and scale to meet the sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        h[:] = target / h.size
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    if s2 == 0.0:
        h[:] = target / h.size
    else:
        h = h * (target / s2)
    return h

def _evaluate(h, dx):
    """Return the C5 value for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _mutate(parent, target, rng):
    """Create a mutated child from parent."""
    n = parent.size
    child = parent.copy()
    # choose strategy
    if rng.random() < 0.5:          # Strategy A: random resampling
        frac = 0.2
        k = max(1, int(frac * n))
        idx = rng.choice(n, k, replace=False)
        child[idx] = rng.random(k)
    else:                           # Strategy B: small perturbation
        mask = rng.random(n) < 0.3
        delta = 0.03
        noise = delta * (rng.random(mask.sum()) * 2.0 - 1.0)
        child[mask] += noise
    child = _repair(child, target)
    return child

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        1‑D array of values in [0,1] with sum(h)*dx == 1.
    c5_bound : float
        Achieved upper bound for C5.
    n_points : int
        Number of discretisation points (length of h_values).
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 70
        h0 = rng.random(n)

    target_sum = n / 2.0               # because integral 1 => sum = n/2
    dx = 2.0 / n

    # Initial feasible point
    h_best = _repair(h0, target_sum)
    best_score = _evaluate(h_best, dx)

    start = time.time()
    max_time = budget_s * 0.95          # safety margin

    # Parameters controlling the search
    max_no_improve = 2000               # iterations without improvement before a restart
    it_no_improve = 0
    iter_count = 0

    while time.time() - start < max_time:
        # generate a child
        child = _mutate(h_best, target_sum, rng)
        score = _evaluate(child, dx)

        if score < best_score:
            h_best = child
            best_score = score
            it_no_improve = 0
        else:
            it_no_improve += 1

        # occasional random restart to escape stagnation
        if it_no_improve >= max_no_improve:
            # new random feasible vector
            rand_vec = rng.random(n)
            child = _repair(rand_vec, target_sum)
            child_score = _evaluate(child, dx)
            if child_score < best_score:
                h_best = child
                best_score = child_score
                it_no_improve = 0
            else:
                # keep the current best but reset counter
                it_no_improve = 0

        iter_count += 1

    return h_best, float(best_score), n
