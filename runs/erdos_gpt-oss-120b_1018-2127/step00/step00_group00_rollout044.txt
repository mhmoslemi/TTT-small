import time, math
import numpy as np
from scipy.optimize import minimize

def _compute_score(h):
    """Return C5 for step vector h on [0,2] with dx = 2/len(h)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _feasible(h, n, atol=1e-6):
    """Quick feasibility test."""
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return False
    return abs(h.sum() - n / 2) <= atol

def _resample(old_h, old_n, new_n):
    """Linear interpolation from old_n points to new_n points on [0,2)."""
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)

def _prepare_initial(old_h, old_n, new_n):
    """Create an initial guess for size new_n from old_h (size old_n)."""
    h = _resample(old_h, old_n, new_n)
    # scale to correct total mass (sum = new_n/2)
    target = new_n / 2.0
    cur = h.sum()
    if cur > 0:
        h = h * (target / cur)
    # clip to [0,1]; optimizer will enforce the equality constraint.
    return np.clip(h, 0.0, 1.0)

def _optimize(h0, n, time_left):
    """Run a single SLSQP optimisation, respecting a time budget."""
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    opts = {"maxiter": 500, "ftol": 1e-6, "disp": False}
    # Early abort if not enough time for a reasonable run.
    if time_left < 0.5:
        return None
    res = minimize(_compute_score, h0, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options=opts)
    return res
