import numpy as np
import time
import math

def _compute_c5(h, dx):
    """Return max_k integral h(x)*(1-h(x+k)) dx using full correlation."""
    # correlation length = 2*n-1, step = dx
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _make_initial_h(seed, n):
    """Create a feasible start: half 1's, half 0's, shuffled."""
    rng = np.random.default_rng(seed)
    h = np.empty(n, dtype=np.float32)
    half = n // 2
    h[:half] = 1.0
    h[half:] = 0.0
    rng.shuffle(h)
    return h
