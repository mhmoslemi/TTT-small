import time
import numpy as np

def _project_h(candidate, target):
    """Project onto box [0,1] and sum == target."""
    h = np.clip(candidate, 0.0, 1.0)
    for _ in range(15):
        s = h.sum()
        if abs(s - target) < 1e-9:
            break
        delta = target - s
        free = (h > 0.0) & (h < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # scale all values (may violate bounds, will be re‑clipped)
            h = np.clip(h * (target / s), 0.0, 1.0)
            continue
        adjust = delta / n_free
        h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
    # final safety scale if still off
    s = h.sum()
    if abs(s - target) > 1e-6:
        h = np.clip(h * (target / s), 0.0, 1.0)
    return h

def _c5_of_h(h):
    """Return C5 for a given discretised h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _initial_h(n_points, target):
    """Build a feasible starting vector when no parent is given."""
    # simple alternating blocks of 1 and 0 of equal total area
    half = n_points // 2
    h = np.zeros(n_points)
    # fill first half with 1, then adjust to meet exact sum
    h[:half] = 1.0
    h = _project_h(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and sample count."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem dimension
    if 'initial_h_values' in globals() and isinstance(initial_h_values, np.ndarray):
        h0 = np.array(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 140  # modest size, good trade‑off
        h0 = None

    target_sum = n_points / 2.0  # because sum(h)*dx = 1, dx = 2/n

    # Initialise feasible vector
    if h0 is not None:
        cur_h = _project_h(h0, target_sum)
    else:
        cur_h = _initial_h(n_points, target_sum)

    best_h = cur_h.copy()
    best_c5 = _c5_of_h(best_h)

    # Annealing schedule parameters
    max_iter = 50000
    temp_start = 0.5
    temp_end = 1e-4

    for it in range(1, max_iter + 1):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # temperature decay
        t = temp_start * (temp_end / temp_start) ** (it / max_iter)

        # propose perturbation
        proposal = cur_h + rng.normal(scale=0.2 * t, size=n_points)
        proposal = _project_h(proposal, target_sum)

        cand_c5 = _c5_of_h(proposal)

        # Metropolis accept
        if cand_c5 < best_c5:
            best_h, best_c5 = proposal.copy(), cand_c5
        if cand_c5 < _c5_of_h(cur_h) or rng.random() < np.exp((_c5_of_h(cur_h) - cand_c5) / (t + 1e-12)):
            cur_h = proposal

    # Final safety projection (should already be feasible)
    best_h = _project_h(best_h, target_sum)
    final_c5 = _c5_of_h(best_h)

    return best_h, float(final_c5), n_points
