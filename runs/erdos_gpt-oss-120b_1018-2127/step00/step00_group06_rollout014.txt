import numpy as np, time, math

def _project_onto_bounds_and_sum(x, target_sum, eps=1e-12):
    """Euclidean projection of vector x onto [0,1]^n with given sum."""
    n = x.size
    # initial clip
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if abs(s - target_sum) <= eps:
        return x
    # iterative water-filling
    lower = np.zeros(n, dtype=bool)   # at 0 bound
    upper = np.ones(n, dtype=bool)    # at 1 bound
    while True:
        free = ~(lower | upper)
        if not np.any(free):
            # cannot adjust any more; force projection by scaling residual uniformly
            x += (target_sum - x.sum()) / n
            x = np.clip(x, 0.0, 1.0)
            break
        delta = (target_sum - x.sum()) / free.sum()
        x[free] += delta
        # re‑apply bounds
        new_upper = x > 1.0 - eps
        new_lower = x < eps
        if not new_upper.any() and not new_lower.any():
            break
        upper = upper | new_upper
        lower = lower | new_lower
        x = np.clip(x, 0.0, 1.0)
        if abs(x.sum() - target_sum) <= eps:
            break
    return x

def _c5_score(h):
    """Return C5 for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _propose_move(h):
    """Return a new h by moving a random amount between two indices."""
    n = h.size
    a, b = np.random.choice(n, 2, replace=False)
    # determine allowable transfer from a to b while staying in [0,1]
    max_transfer = min(h[a], 1.0 - h[b])
    min_transfer = -min(h[b], 1.0 - h[a])
    if max_transfer <= 0 and min_transfer >= 0:
        return h.copy()  # nothing can be moved
    delta = np.random.uniform(min_transfer, max_transfer)
    h_new = h.copy()
    h_new[a] -= delta
    h_new[b] += delta
    return h_new

def _anneal(initial_h, time_allow):

    start = time.time()
    h = initial_h.copy()
    best_h = h.copy()
