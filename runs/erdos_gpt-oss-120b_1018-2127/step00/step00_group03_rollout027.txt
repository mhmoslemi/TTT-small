import numpy as np, time, random, math

def compute_c5(h):
    """Maximum overlap integral for a discretized step function."""
    N = h.size
    dx = 2.0 / N
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def initial_binary(N, rng):
    """Create a binary start with exactly N/2 ones."""
    h = np.zeros(N, dtype=float)
    h[: N // 2] = 1.0
    rng.shuffle(h)
    return h

def project_to_binary(h):
    """Project arbitrary values to a binary vector preserving sum N/2."""
    N = h.size
    target = N // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(N, dtype=float)
    out[idx[:target]] = 1.0
    return out

def sinusoidal_candidate(N, a, phi):
    """Generate h = 0.5 + a*cos(pi*(x+phi)), clipped to [0,1]."""
    dx = 2.0 / N
    x = np.arange(N) * dx + dx / 2.0
    h = 0.5 + a * np.cos(math.pi * (x + phi))
    np.clip(h, 0.0, 1.0, out=h)
    # enforce integral exactly 1 by adjusting a tiny offset if needed
    cur = h.sum() * dx
    if not math.isclose(cur, 1.0, rel_tol=1e-9, abs_tol=1e-9):
        # simple linear correction (preserves bounds because correction is tiny)
        corr = (1.0 - cur) / (dx * N)
        h += corr
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Determine problem size
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():

        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_to_binary(h0)
    else:
        N = 200                     # modest size, fast evaluations
        h0 = initial_binary(N, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Quick sinusoidal grid search (very cheap)
    # ------------------------------------------------------------------
    a_grid = np.linspace(-0.48, 0.48, 49)   # stay safely inside [‑0.5,0.5]
    phi_grid = np.linspace(0.0, 2.0, 45, endpoint=False)
    for a in a_grid:
        for phi in phi_grid:
            h = sinusoidal_candidate(N, a, phi)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
    # ------------------------------------------------------------------
    # 3. Stochastic local search on continuous variables
    # ------------------------------------------------------------------
    # Parameters
    max_iter = 200000
    step_initial = 0.04
    step_final = 0.004

    h = best_h.copy()
    step = step_initial

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        # anneal step size linearly
        frac = elapsed / (budget_s * 0.95)
        step = step_initial * (1 - frac) + step_final * frac

        i, j = rng.choice(N, 2, replace=False)
        eps = (rng.rand() * 2.0 - 1.0) * step
        hi, hj = h[i], h[j]

    return (best_h.tolist(), float(best_c), int(N))
