import time
import numpy as np

def _project_bounds_sum(v, target_sum, tol=1e-9, max_iter=100):
    """
    Project vector v onto the intersection of box [0,1]^n and the affine hyperplane sum=v_target.
    Uses a simple water-filling iterative scheme.
    """
    n = v.shape[0]
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        # indices that can still move (not at bound)
        free = np.where((v > 0.0 + tol) & (v < 1.0 - tol))[0]
        if free.size == 0:
            # cannot move any variable; distribute proportionally with saturation
            # This situation occurs only due to numerical issues; break.
            break
        delta = diff / free.size
        # apply delta and enforce bounds
        v[free] += delta
        # handle any overshoot
        over = v[free] > 1.0
        under = v[free] < 0.0
        if np.any(over):
            indices = free[over]
            diff -= (1.0 - (v[indices] - delta)).sum()
            v[indices] = 1.0
        if np.any(under):
            indices = free[under]
            diff -= (0.0 - (v[indices] - delta)).sum()
            v[indices] = 0.0
    # final clip
    return np.clip(v, 0.0, 1.0)

def _evaluate_c5(h, dx):
    """
    Evaluate C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation.
    """
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _initial_vector():
    """
    Obtain an initial feasible vector.
    If the global `initial_h_values` exists, use it; otherwise generate a uniform start.
    """
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        target = n / 2.0
        init = _project_bounds_sum(init, target)
        return init
    except NameError:
        # fallback: 42 points uniform with sum = n/2
        n = 42
        target = n / 2.0
        h = np.full(n, target / n)
        return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # obtain problem size
    h0 = _initial_vector()
    n = h0.size
    dx = 2.0 / n
    target_sum = n / 2.0

    # best solution tracking
    best_h = h0.copy()
    best_val = _evaluate_c5(best_h, dx)

    # Simulated annealing parameters
    temp_start = 0.2
    temp_end = 1e-4
    max_iters = 200000  # upper bound, will be cut by time
    iter_per_temp = 2000
    cur_h = best_h.copy()
    cur_val = best_val

    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.5:  # leave a tiny cushion
            break
        # temperature schedule (linear in log-space)
        frac = min(1.0, it / max_iters)
        temp = temp_start * (temp_end / temp_start) ** frac

        # propose move
        step_scale = 0.15  # magnitude of perturbation
        proposal = cur_h + step_scale * np.random.randn(n)
        proposal = _project_bounds_sum(proposal, target_sum)

        # evaluate
        prop_val = _evaluate_c5(proposal, dx)

        # acceptance
        if prop_val < cur_val or np.random.rand() < np.exp(-(prop_val - cur_val) / max(temp, 1e-12)):
            cur_h, cur_val = proposal, prop_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val

        # occasional random restart to avoid stagnation
        if it % (iter_per_temp * 5) == 0 and it > 0:
            # keep the best found, perturb it strongly
            cur_h = best_h + 0.4 * np.random.randn(n)
            cur_h = _project_bounds_sum(cur_h, target_sum)
            cur_val = _evaluate_c5(cur_h, dx)

        it += 1

    # final safeguard: ensure best_h respects constraints (project once more)
    best_h = _project_bounds_sum(best_h, target_sum)
    best_val = _evaluate_c5(best_h, dx)

    return best_h, best_val, n
