import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_score(h):
    """Return C5 for given step values h (numpy array)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def feasible_projection(x):
    """Project a vector onto the feasible set: 0<=x<=1 and sum(x)=n/2."""
    n = x.size
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Adjust sum by a simple scaling, then re‑clip if needed
    target = n / 2.0
    for _ in range(5):
        s = x.sum()
        if abs(s - target) < 1e-9:
            break
        scale = target / s if s != 0 else 0.0
        x *= scale
        x = np.clip(x, 0.0, 1.0)
    return x

def random_feasible(n, rng):
    """Generate a random feasible h vector of length n."""
    target = n / 2.0
    while True:
        # Dirichlet-like generation ensures positivity
        x = rng.random(n)
        x = feasible_projection(x)
        if np.all(x >= 0) and np.all(x <= 1) and abs(x.sum() - target) < 1e-9:
            return x

def transfer_solution(old_h, new_n):
    """Resample old_h (length m) to length new_n using linear interpolation."""
    m = old_h.size
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # Enforce integral constraint after interpolation
    new_h = feasible_projection(new_h)
    return new_h

def optimize_h(initial_h, rng, maxiter=800, ftol=1e-6):
    """Run SLSQP from a given starting point."""
    n = initial_h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - n / 2.0}
    result = minimize(
        compute_score,
        initial_h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": maxiter, "ftol": ftol, "disp": False},
    )
    if result.success:
        x = feasible_projection(result.x)
        return x, compute_score(x)
    # fallback: return None
    return None, None
