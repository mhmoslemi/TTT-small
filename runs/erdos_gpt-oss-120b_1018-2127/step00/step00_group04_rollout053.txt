import time
import numpy as np

def _evaluate_c5(h):
    """Return max overlap integral for step function h sampled on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def _random_move(h, eps):
    """Propose a new h by moving eps from one index to another, preserving sum."""
    n = h.size
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None
    # ensure feasibility
    if h[i] - eps < 0.0 or h[j] + eps > 1.0:
        return None
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new
