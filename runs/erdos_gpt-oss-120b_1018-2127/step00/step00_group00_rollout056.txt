import time
import numpy as np
from scipy.optimize import minimize

def compute_score(h):
    """Maximum overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    """Clip to [0,1] and enforce exact integral n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s == 0:
        # uniform fill as fallback
        return np.full(n, target / n)
    # Simple proportional scaling, then clip again if needed
    h = h * (target / s)
    # Second clipping pass (rare)
    h = np.clip(h, 0.0, 1.0)
    # Adjust leftover mass iteratively (few iterations suffice)
    for _ in range(3):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # distribute diff uniformly
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def transfer_solution(old_h, old_n, new_n):
    """Resample old_h (defined on old_n points) to length new_n, then renormalize."""
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    new_h = project_feasible(new_h, new_n)
    return new_h

def optimize_h(h0, max_iter=600, ftol=1e-8):
    """Run SLSQP from initial point h0."""
    n = h0.shape[0]
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    res = minimize(
        compute_score,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": ftol, "disp": False},
    )
    if res.success:
        return project_feasible(res.x, n)
    # fallback to original if optimization fails
    return h0

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] with integral 1 that minimizes C5."""
    start = time.time()
    np.random.seed(seed)

    # candidate discretizations (prefer small, still expressive)
    n_list = [30, 45, 60, 90, 180]
    best_h = None
    best_score = np.inf
    best_n = None

    # Determine whether an initial vector is supplied by the harness
    have_initial = "initial_h_values" in globals()
    init_h = None
    init_n = None
    if have_initial:
        init_h = np.asarray(initial_h_values, dtype=float)
        init_n = init_h.shape[0]

    for n in n_list:
        # stop if budget exhausted
        if time.time() - start > budget_s:
            break

        # obtain a feasible starting point for this resolution
        if have_initial and init_n == n:
            h0 = init_h.copy()
        elif have_initial:
            h0 = transfer_solution(init_h, init_n, n)
        else:
            # uniform start (sum = n/2, each entry = 0.5)
            h0 = np.full(n, 0.5, dtype=float)

        # sanity check feasibility
        h0 = project_feasible(h0, n)

        # evaluate current score
        cur_score = compute_score(h0)

        # attempt local optimization
        h_opt = optimize_h(h0)

        opt_score = compute_score(h_opt)
