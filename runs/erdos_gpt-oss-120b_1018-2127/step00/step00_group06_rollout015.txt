import time
import numpy as np
import random

def evaluate_c5(h):
    """Return max_k ∫ h(x)(1-h(x+k)) dx for discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_bounds_transfer(h, i, j, delta):
    """Transfer delta from i to j while keeping 0≤h≤1."""
    # assume delta>0, h[i]>=delta, h[j]<=1-delta
    h[i] -= delta
    h[j] += delta
    return h

def anneal_step(h, cur_c5, temperature, step_size):
    """Propose a feasible transfer move and possibly accept it."""
    n = h.shape[0]
    # pick source index with some mass and target index with room
    src_candidates = np.where(h > 1e-12)[0]
    tgt_candidates = np.where(h < 1 - 1e-12)[0]
    if src_candidates.size == 0 or tgt_candidates.size == 0:
        return h, cur_c5, False
    i = int(random.choice(src_candidates))
    j = int(random.choice(tgt_candidates))
    if i == j:
        return h, cur_c5, False
    max_delta = min(step_size, h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h, cur_c5, False
    delta = max_delta * random.random()
    h_new = h.copy()
    project_bounds_transfer(h_new, i, j, delta)
    new_c5 = evaluate_c5(h_new)
    if new_c5 < cur_c5 or random.random() < np.exp(-(new_c5 - cur_c5) / temperature):
        return h_new, new_c5, True
    return h, cur_c5, False
