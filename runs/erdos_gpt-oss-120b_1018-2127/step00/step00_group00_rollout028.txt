import time
import numpy as np

# helpers are pre‑imported by the harness:
# - evaluate_erdos_solution(h) -> float
# - initial_h_values (may or may not exist)

def _project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the box [0,1]^n intersect {sum = target_sum}
    using a simple bisection on the Lagrange multiplier.
    """
    # initial bounds for lambda
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _initial_vector(n: int) -> np.ndarray:
    """
    Produce a feasible start vector of length n.
    If the global `initial_h_values` exists with that length, use it.
    Otherwise create a simple half‑filled pattern.
    """
    iv = globals().get("initial_h_values", None)
    if iv is not None and len(iv) == n:
        return np.array(iv, dtype=float).copy()
    # simple construction: first n/2 entries = 1, rest = 0
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[:half] = 1.0
    # adjust sum exactly (might be off by 0.5 when n odd)
    target = n / 2.0
    h = _project_to_constraints(h, target)
    return h

    Returns (h_values, c5_bound, n_points).
    """

    best_h = _initial_vector(n)
    try:
        best_score = evaluate_erdos_solution(best_h)
    except Exception:
        best_score = float("inf")

    # parameters for the stochastic search
    max_iters = 200000
    sigma_start = 0.3
    sigma_end = 0.01

    it = 0
    while it < max_iters:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.94:   # reserve a bit of headroom

        # linearly decay the perturbation magnitude

        # propose a new candidate

        cand = _project_to_constraints(cand, target_sum)

            best_h = _project_to_constraints(best_h + sigma * rng.standard_normal(n), target_sum)

    best_h = _project_to_constraints(best_h, target_sum)
