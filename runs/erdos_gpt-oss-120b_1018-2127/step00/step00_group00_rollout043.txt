def adjust_sum(h, target):
    h = np.clip(h,0,1)
    total = h.sum()
    if abs(total - target) < 1e-8:
        return h
    # iterative
    free = np.ones_like(h, dtype=bool)
    while True:
        total = h[free].sum()
        if total == 0:
            break

        # apply scaling to free entries
        h[free] = np.clip(h[free] * scale, 0, 1)

import time
import numpy as np
from scipy import optimize as opt

def compute_score(h):
    """Maximum overlap integral for a given step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def adjust_sum(h, target):
    """Clip to [0,1] and enforce sum == target by iterative scaling."""
    h = np.clip(h, 0.0, 1.0)
    # iterative proportional scaling on the free (non‑boundary) entries
    free = np.ones_like(h, dtype=bool)
    for _ in range(50):
        total = h[free].sum()
        if total == 0:
            break
        scale = target / total
        h[free] *= scale
        h = np.clip(h, 0.0, 1.0)
        new_free = (h > 0.0) & (h < 1.0)
        if np.array_equal(new_free, free):
            break
        free = new_free
    # tiny residual correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.where((h > 0.0) & (h < 1.0))[0]
        if idx.size:
            i = idx[0]
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def interpolate_h(old_h, n_new):
    """Linear interpolation of an old step vector onto a new grid."""
    n_old = len(old_h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    new_h = adjust_sum(new_h, n_new / 2.0)
    return new_h

def hill_climb(h0, n, time_limit, start_time):
    """Simple stochastic hill‑climbing on the feasible set."""
    best_h = np.copy(h0)
    best_score = compute_score(best_h)
    step = 0.2  # maximal mass transferred per move
    while time.time() - start_time < time_limit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() * 2.0 - 1.0) * step
        h_candidate = best_h.copy()
        h_candidate[i] += delta
        h_candidate[j] -= delta
        # quick feasibility check before projection
        if (h_candidate[i] < -1e-9) or (h_candidate[i] > 1.0 + 1e-9) \
           or (h_candidate[j] < -1e-9) or (h_candidate[j] > 1.0 + 1e-9):
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h_candidate = adjust_sum(h_candidate, n / 2.0)
        score = compute_score(h_candidate)
        if score < best_score:
            best_h, best_score = h_candidate, score
        step *= 0.9999  # slowly anneal
    return best_h, best_score

def slsqp_optimize(h_start):
    """Refine a feasible vector using SLSQP on the max‑overlap objective."""
    n = len(h_start)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    res = opt.minimize(compute_score, h_start, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": 500, "ftol": 1e-9})
    if res.success:
        h_opt = adjust_sum(res.x, target)
        return h_opt, compute_score(h_opt)
    return h_start, compute_score(h_start)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_score = np.inf
    best_n = None

    # possible grid sizes (small enough for fast evaluation)
    n_list = [30, 45, 60, 90, 180]

    # obtain the parent construction if present
    try:
        init_h = np.asarray(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    for n in n_list:
        if time.time() - start >= budget_s:
            break

        # time slice for this resolution (half of remaining time)
        remaining = budget_s - (time.time() - start)
        sub_budget = remaining * 0.5

        # initialise a feasible vector for this n
        if init_h is not None and len(init_h) == n:
            h0 = np.copy(init_h)
        elif init_h is not None:
            h0 = interpolate_h(init_h, n)
        else:
            h0 = np.random.rand(n)
            h0 = adjust_sum(h0, n / 2.0)

        h0 = adjust_sum(np.clip(h0, 0.0, 1.0), n / 2.0)

        # stochastic local search
        h_best, score_best = hill_climb(h0, n, sub_budget, start)

        # optional refined SLSQP step if time permits
        if time.time() - start < budget_s:
            h_ref, score_ref = slsqp_optimize(h_best)
            if score_ref < score_best:
                h_best, score_best = h_ref, score_ref

        if score_best < best_score:
            best_h, best_score, best_n = h_best, score_best, n

        # early exit if we beat the target bound
        if best_score <= 0.3808:
            break

    # fallback in the unlikely case nothing was found
    if best_h is None:
        best_n = 45
        best_h = np.full(best_n, 0.5)
        best_score = compute_score(best_h)

    return best_h, best_score, best_n
