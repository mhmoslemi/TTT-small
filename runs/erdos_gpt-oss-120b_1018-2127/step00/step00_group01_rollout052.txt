import time
import numpy as np
import scipy.optimize as opt
import cvxpy as cp

def _c5_from_h(h):
    """Compute C5 for a discrete step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _objective(h, target_sum, penalty=1e6):
    """Objective for DE: C5 plus quadratic penalty for the integral constraint."""
    c5 = _c5_from_h(h)
    pen = penalty * (np.sum(h) - target_sum) ** 2
    return c5 + pen

def _project_to_feasible(v, target_sum):
    """Project vector v onto the set {0<=x<=1, sum x = target_sum}."""
    n = v.size
    x = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(x - v)),
                      [cp.sum(x) == target_sum,
                       x >= 0,
                       x <= 1])
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    return np.asarray(x.value).reshape(-1)

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
    start = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    try:
        init_h = np.asarray(initial_h_values, dtype=float)
        n = init_h.size
    except Exception:
        n = 100                     # fallback size if no parent is supplied
        init_h = np.full(n, 0.5)    # uniform feasible start

    target_sum = n / 2.0            # because integral of h must be 1

    # ------------------------------------------------------------------
    # Differential Evolution (global search)
    # ------------------------------------------------------------------
    bounds = [(0.0, 1.0)] * n
    max_iter = 200                 # generous but will be cut by time budget
    deadline = start + budget_s * 0.65   # reserve ~35% for final polishing

    def de_callback(xk, convergence):
        return time.time() > deadline

    result = opt.differential_evolution(
        _objective,
        bounds,
        args=(target_sum,),
        strategy='best1exp',
        maxiter=max_iter,
        popsize=15,
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=False,
        init='latinhypercube',
        callback=de_callback,
        disp=False,
    )
    best = result.x

    # ------------------------------------------------------------------
    # Feasibility projection
    # ------------------------------------------------------------------
    best = _project_to_feasible(best, target_sum)

    # ------------------------------------------------------------------
    # Local polishing (small gradient-free refinement)
    # ------------------------------------------------------------------
    polish_deadline = start + budget_s * 0.95

    def polish_callback(xk):
        return time.time() > polish_deadline

    res_local = opt.minimize(
        lambda x: _c5_from_h(x),
        best,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxfun': 5000, 'ftol': 1e-12},
        callback=polish_callback,
    )
    if res_local.success:
        best = _project_to_feasible(res_local.x, target_sum)

    # ------------------------------------------------------------------
    # Final evaluation
    # ------------------------------------------------------------------
    c5_bound = _c5_from_h(best)

    # Use the provided helper for final reporting if it exists
    try:
        c5_bound = evaluate_erdos_solution(best)
    except Exception:
        pass

    return best, float(c5_bound), n
