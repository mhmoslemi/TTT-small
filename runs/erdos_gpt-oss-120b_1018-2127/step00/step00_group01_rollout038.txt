import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with values in {0,1} and
    ∫ h = 1 (i.e., exactly half of the samples are 1).
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ---- problem size -------------------------------------------------
    if 'initial_h_values' in globals():
        n0 = len(initial_h_values)
        n = max(100, n0 * 2)          # try a finer discretization
        n = min(n, 300)               # keep work manageable
    else:
        n = 200
    dx = 2.0 / n

    # ---- helpers -------------------------------------------------------
    def initial_binary():
        """Create a binary start vector with exactly n/2 ones."""
        if 'initial_h_values' in globals():
            iv = np.array(initial_h_values, dtype=float)
            # linear interpolation to new grid
            x_old = np.linspace(0, 2, len(iv), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, iv)
        else:
            h = np.zeros(n, dtype=float)
            h[: n // 2] = 1.0
            np.random.shuffle(h)

        # force exact half‑ones
        h_bin = (h > 0.5).astype(float)
        cur = int(h_bin.sum())
        target = n // 2
        order = np.arange(n)
        np.random.shuffle(order)
        for idx in order:
            if cur > target and h_bin[idx] == 1:
                h_bin[idx] = 0
                cur -= 1
            elif cur < target and h_bin[idx] == 0:
                h_bin[idx] = 1
                cur += 1
            if cur == target:
                break
        return h_bin

    def c5_value(h_vec):
        """Compute the C₅ bound for a binary vector."""
        conv = np.convolve(h_vec, 1.0 - h_vec, mode='full') * dx
        return np.max(conv)

    # ---- initial solution -----------------------------------------------
    h = initial_binary()
    best_h = h.copy()
    best_val = c5_value(h)

    # ---- local search ----------------------------------------------------
    start = time.time()
    it = 0
    while time.time() - start < budget_s * 0.95:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # trial swap
        h[i], h[j] = 0.0, 1.0
        val = c5_value(h)

        if val < best_val:
            best_val = val
            best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0

        # occasional random restart to avoid stagnation
        if it % 5000 == 0:
            h = initial_binary()

    return (best_h.tolist(), float(best_val), n)
