import time
import numpy as np

def _evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _project_feasible(vec, target_sum):
    v = np.clip(vec, 0.0, 1.0)
    n = v.shape[0]
    for _ in range(10):
        s = v.sum()
        if abs(s - target_sum) < 1e-12:
            break
        v += (target_sum - s) / n
        v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target_sum) > 1e-8:
        free = (v > 1e-12) & (v < 1.0 - 1e-12)
        idx = np.where(free)[0]
        if idx.size:
            v[idx] += (target_sum - s) / idx.size
            v = np.clip(v, 0.0, 1.0)
    return v

def _resample(base, new_n):
    old_n = base.shape[0]
    if old_n == new_n:
        return base.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False) + (1.0 / old_n)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + (1.0 / new_n)
    interp = np.interp(x_new, x_old, base)
    return _project_feasible(interp, new_n / 2.0)

def _optimize(start_h, time_allow):
    n = start_h.shape[0]
    target = n / 2.0
    best_h = start_h.copy()
    best_score = _evaluate(best_h)
    rng = np.random.default_rng()
    start = time.time()
    it = 0
    while time.time() - start < time_allow:
        step = 0.12 * (0.995 ** it)
        cand = best_h + step * rng.standard_normal(n)
        cand = _project_feasible(cand, target)
        sc = _evaluate(cand)
        if sc < best_score:
            best_h, best_score = cand, sc
        it += 1
    return best_h, best_score
