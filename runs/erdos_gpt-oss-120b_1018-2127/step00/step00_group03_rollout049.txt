import time
import numpy as np

def _c5_score(h, dx):
    """Compute the overlap bound C5 for a step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _evaluate(h):
    """Wrapper that uses the pre‑imported evaluator if present."""
    try:
        # evaluate_erdos_solution should accept a 1‑D numpy array
        return evaluate_erdos_solution(h)
    except Exception:
        # fallback to direct computation
        dx = 2.0 / h.size
        return _c5_score(h, dx)

def _project_sum_and_bounds(h, target_sum):
    """Clamp to [0,1] and then adjust to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    diff = target_sum - current
    if abs(diff) < 1e-12:
        return h
    # Distribute the deficit/surplus uniformly over entries that are not at a bound
    free = ( (diff > 0) & (h < 1.0) ) | ( (diff < 0) & (h > 0.0) )
    free_idx = np.where(free)[0]
    if free_idx.size == 0:
        # nothing free – just adjust the first element within bounds
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        return h
    # simple proportional fill
    add = diff / free_idx.size
    h[free_idx] += add
    # one more clamp pass to stay in [0,1]
    h = np.clip(h, 0.0, 1.0)
    # small residual correction via the first free index
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        idx = free_idx[0]
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h
