import time
import numpy as np

def _compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _initial_h(n, rng):
    """Create a feasible start: exactly n/2 total mass."""
    # binary start then smooth a little
    h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
    rng.shuffle(h)
    # add tiny jitter while keeping sum fixed via pair swaps
    jitter = 0.02
    for _ in range(n):
        i, j = rng.choice(n, 2, replace=False)
        eps = jitter * rng.uniform(-1, 1)
        low = -min(h[i], 1 - h[j])
        high = min(1 - h[i], h[j])
        if low < high:
            eps = np.clip(eps, low, high)
            h[i] += eps
            h[j] -= eps
    return h

def _pair_move(h, rng, step_scale):
    """Propose a new h by moving mass between two coordinates."""
    n = h.shape[0]
    i, j = rng.choice(n, 2, replace=False)
    low = -min(h[i], 1 - h[j])
    high = min(1 - h[i], h[j])
    if low >= high:
        return None  # infeasible move
    # draw epsilon within allowed interval, scaled
    eps = step_scale * rng.uniform(low, high)
    new_h = h.copy()
    new_h[i] += eps
    new_h[j] -= eps
    return new_h

def _optimize(h0, budget, rng):
    """Simulated‑annealing search starting from h0 for at most budget seconds."""
    start = time.time()
    n = h0.shape[0]
    best_h = h0.copy()
    best_score = _compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score

    # annealing parameters
    T0 = 1.0
    T = T0
    cooling = 0.995
    step_scale = 0.1  # initial move magnitude

    # estimate iteration budget (~2 ms per eval for n≈320)
    target_iters = int(1e5)
    it = 0
    while time.time() - start < budget and it < target_iters:
        it += 1
        # slowly shrink step scale with temperature
        step = step_scale * (T / T0 + 0.1)
        cand = _pair_move(cur_h, rng, step)
        if cand is None:
            continue
        cand_score = _compute_c5(cand)

        if cand_score < cur_score:
            cur_h, cur_score = cand, cand_score
            if cand_score < best_score:
                best_h, best_score = cand.copy(), cand_score
        else:
            delta = cand_score - cur_score
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                cur_h, cur_score = cand, cand_score

        T *= cooling
        if T < 1e-5:
            T = T0  # reheating occasionally
    return best_h, best_score

def _upscale(h, new_n):
    """Upscale vector h to length new_n by linear interpolation."""
    n = h.shape[0]
    if new_n == n:
        return h.copy()
    # map old indices to new positions
    old_x = np.linspace(0, 2, n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)
