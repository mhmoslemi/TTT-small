import time
import numpy as np

def _repair(h, target_sum):
    """Clip to [0,1] and scale to hit target_sum (iteratively)."""
    for _ in range(10):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s == 0:
            # fallback to uniform fill
            h = np.full_like(h, target_sum / h.size)
            break
        if abs(s - target_sum) < 1e-12:
            break
        h *= target_sum / s
    # final safety clip
    h = np.clip(h, 0.0, 1.0)
    # tiny rescale if off by round‑off
    s = h.sum()
    if s != 0:
        h *= target_sum / s
    return h

def _evaluate(h):
    """Return C5 for the given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * corr.max()

def _mutate(parent, rng, target_sum):
    """Create a child from parent using one of two simple mutations."""
    child = parent.copy()
    n = child.size
    if rng.random() < 0.5:               # Strategy A: resample 20% of entries
        k = max(1, n // 5)
        idx = rng.choice(n, size=k, replace=False)
        child[idx] = rng.random(k)
    else:                                 # Strategy B: small perturbations
        mask = rng.random(n) < 0.3
        noise = (rng.random(mask.sum()) * 2.0 - 1.0) * 0.015
        child[mask] += noise
    child = _repair(child, target_sum)
    return child
