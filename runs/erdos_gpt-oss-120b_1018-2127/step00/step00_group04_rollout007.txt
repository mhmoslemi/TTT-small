import time
import numpy as np

def compute_c5(h):
    """Return the overlap constant for a discretized step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum_to_target(h, target_sum):
    """Scale h linearly to hit target_sum while keeping bounds [0,1]."""
    cur = h.sum()
    if cur == 0:
        # distribute uniformly
        h[:] = target_sum / h.size
        return h
    scale = target_sum / cur
    h *= scale
    # clip and rebalance any overflow/underflow
    excess = h - 1.0
    excess[excess < 0] = 0.0
    deficit = -h
    deficit[deficit < 0] = 0.0
    # simple redistribution loop (few iterations suffice)
    for _ in range(5):
        excess_sum = excess.sum()
        deficit_sum = deficit.sum()
        if excess_sum == 0 and deficit_sum == 0:
            break
        # move excess down to fill deficits
        if excess_sum > 0 and deficit_sum > 0:
            delta = min(excess_sum, deficit_sum) / h.size
            h -= deficit * (delta / deficit_sum + 1e-12)
            h += excess * (delta / excess_sum + 1e-12)
        h = np.clip(h, 0.0, 1.0)
        excess = np.maximum(h - 1.0, 0.0)
        deficit = np.maximum(-h, 0.0)
    # final scaling to hit target exactly (tiny numerical error tolerated)
    h *= (target_sum / h.sum())
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points) where
    h_values is a numpy array of the step function on a uniform grid,
    c5_bound is the achieved max overlap, and n_points is the grid size.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine grid size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h = h0.copy()
    else:
        n = 200  # default modest size
        h = np.full(n, 0.5)  # placeholder, will be projected

    target_sum = n / 2.0
    h = project_sum_to_target(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Parameters for the stochastic search
    max_iter = 10_000_000  # effectively unlimited, bounded by time
    eps_initial = 0.02      # typical magnitude of a transfer between two positions
    eps_decay = 0.9995
    eps = eps_initial

    it = 0
    while True:
        it += 1
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        delta = (rng.random() - 0.5) * 2.0 * eps  # in [-eps, eps]

        # feasibility test before constructing new vector
        hi = h[i] + delta
        hj = h[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue

        h_new = h.copy()
        h_new[i] = hi
        h_new[j] = hj

        # sum is preserved analytically; no need to re‑project
        new_c5 = compute_c5(h_new)

        # accept if improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
        else:
            # occasional uphill moves (simulated annealing style)
            # temperature proportional to remaining time
            remaining = max(budget_s - (time.time() - start_time), 0.0)
            temp = remaining / budget_s
            if temp > 0 and rng.random() < np.exp(-(new_c5 - best_c5) / (temp * 0.01)):
                h = h_new

        # gradually shrink move size

            eps = eps_initial

    best_h = project_sum_to_target(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, float(best_c5), n
