def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        # set uniform
        h = np.full_like(h, target_sum / len(h))
        return h
    # scale to meet target_sum
    h = h * (target_sum / total)
    # after scaling some may exceed 1; clip again and re-balance
    # iterate a few times
    for _ in range(3):
        overflow = h - 1.0
        overflow[overflow < 0] = 0
        deficit = 1.0 - h
        deficit[deficit < 0] = 0
        excess = overflow.sum()
        if excess == 0:
            break
        # redistribute excess proportionally to deficits
        total_deficit = deficit.sum()
        if total_deficit == 0:
            # can't redistribute, just clip and break
            h = np.clip(h, 0, 1)
            break
        # reduce those >1 to 1, and give back excess to others proportionally
        h = np.clip(h, 0, 1)
        # distribute excess uniformly
        h += excess * (deficit / total_deficit)
    # final clip and scale sum exactly
    h = np.clip(h, 0, 1)
    # final scaling to exact sum using small adjustment on free variables
    total = h.sum()
    if total != 0:
        h = h * (target_sum / total)
        h = np.clip(h, 0, 1)
    # ensure sum close to target
    return h

import time, math, random
import numpy as np

def _repair(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        h[:] = target_sum / len(h)
        return h
    h = h * (target_sum / total)
    # iterate a few times to respect bounds after scaling
    for _ in range(3):
        overflow = np.maximum(h - 1.0, 0.0).sum()
        if overflow == 0:
            break
        deficit = np.maximum(1.0 - h, 0.0)
        dsum = deficit.sum()
        if dsum == 0:
            break
        h = np.clip(h, 0.0, 1.0)
        h += overflow * (deficit / dsum)
    h = np.clip(h, 0.0, 1.0)
    # final tiny scaling to hit target sum exactly
    total = h.sum()
    if total > 0:
        h = h * (target_sum / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_score(h: np.ndarray) -> float:
    """Compute the discrete approximation of C5 for a step vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function giving a low upper bound for the Erdős C₅ constant.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,)
        c5_bound : float, the achieved max overlap integral
        n_points : int, number of samples (length of h_values)
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -----
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = len(h0)
        target = n / 2.0
        cur = _repair(h0, target)
    else:
        n = 80
        target = n / 2.0
        cur = _repair(rng.random(n), target)

    best_h = cur.copy()
    best_score = _c5_score(best_h)

    # ----- simulated‑annealing parameters -----
    T = 1.0                     # start temperature
    min_T = 1e-6
    cooling = 0.9995
    iter_cnt = 0

    # main optimisation loop
    while time.time() - start_time < budget_s * 0.95:
        # propose a perturbation
        prop = cur.copy()
        # modify a small random subset (≈5% of entries)
        k = max(1, int(0.05 * n))
        idx = rng.choice(n, size=k, replace=False)
        delta = (rng.random(k) - 0.5) * 2.0 * T   # scale with temperature
        prop[idx] += delta
        prop = _repair(prop, target)

        prop_score = _c5_score(prop)
        cur_score = _c5_score(cur)

        # keep global best
        if prop_score < best_score:
            best_score = prop_score
            best_h = prop.copy()

        # accept / reject (Metropolis rule)
        if prop_score < cur_score:
            cur = prop
        else:
            diff = prop_score - cur_score
            prob = math.exp(-diff / max(T, 1e-12))
            if rng.random() < prob:
                cur = prop

        # temperature schedule
        T = max(min_T, T * cooling)
        iter_cnt += 1

        # occasional full restart to avoid stagnation
        if iter_cnt % 2000 == 0:
            cur = _repair(rng.random(n), target)
            T = 1.0

    # final result
    return (best_h, best_score, n)
