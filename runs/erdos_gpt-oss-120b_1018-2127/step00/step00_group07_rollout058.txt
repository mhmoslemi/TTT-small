import numpy as np
import time
import math
import random

def _compute_c5(h, dx):
    """Return max overlap integral for step function h with spacing dx."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _anneal(h, dx, time_deadline, rng):
    """Simulated annealing on a fixed‑size vector h."""
    n = h.size
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = _compute_c5(best_h, dx)

    cur_h = h.copy()
    cur_score = best_score

    # temperature schedule
    T = 1.0
    cooling = 0.9995

    while time.time() < time_deadline:
        # propose a mass‑transfer move that preserves sum and bounds
        i, j = rng.sample(range(n), 2)
        if rng.random() < 0.5:
            # move from i to j
            delta_max = min(cur_h[i], 1.0 - cur_h[j])
            if delta_max == 0.0:
                continue
            delta = rng.random() * delta_max
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # move from j to i
            delta_max = min(cur_h[j], 1.0 - cur_h[i])
            if delta_max == 0.0:
                continue
            delta = rng.random() * delta_max
            new_h = cur_h.copy()
            new_h[j] -= delta
            new_h[i] += delta

        new_score = _compute_c5(new_h, dx)

        if new_score < cur_score:
            cur_h, cur_score = new_h, new_score
