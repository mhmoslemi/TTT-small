import time
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

# helper: compute C5 for a discrete h
def compute_c5(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

# helper: project a vector onto [0,1] with a fixed sum using cvxpy
def project_feasible(x, target_sum):
    n = x.shape[0]
    h = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(h - x)),
                      [cp.sum(h) == target_sum,
                       h >= 0,
                       h <= 1])
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    if h.value is None:
        # fallback: simple clipping and scaling
        h_proj = np.clip(x, 0, 1)
        cur = h_proj.sum()
        if cur != 0:
            h_proj *= target_sum / cur
        h_proj = np.clip(h_proj, 0, 1)
    else:
        h_proj = np.asarray(h.value).flatten()
    return h_proj

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine number of points
    if 'initial_h_values' in globals():
        n_points = int(len(initial_h_values))
        h_current = np.asarray(initial_h_values, dtype=float).copy()
        # ensure feasibility
        target_sum = n_points / 2.0
        h_current = project_feasible(h_current, target_sum)
    else:
        n_points = 120  # modest size for speed
        target_sum = n_points / 2.0
        # random start then project
        rand_vec = np.random.rand(n_points)
        h_current = project_feasible(rand_vec, target_sum)

    dx = 2.0 / n_points
    best_h = h_current.copy()
    best_c5 = compute_c5(best_h, dx)

    # annealing parameters
    max_time = budget_s * 0.95
    iteration = 0
    temperature0 = 0.02
    while time.time() - start_time < max_time:
        iteration += 1
        # gradually cool
        t_frac = (time.time() - start_time) / max_time
        temperature = temperature0 * (1 - t_frac)

        # propose a move: transfer mass between two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        hi, hj = best_h[i], best_h[j]
        delta_max = min(hi, 1.0 - hj)
        if delta_max <= 0:
            continue
        # random fraction of allowed transfer
        delta = np.random.rand() * delta_max * 0.5
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # compute C5
        new_c5 = compute_c5(h_new, dx)

        # accept if better or with Metropolis criterion
        if new_c5 < best_c5:
            best_h, best_c5 = h_new, new_c5
        else:
            prob = np.exp(-(new_c5 - best_c5) / (temperature + 1e-12))
            if np.random.rand() < prob:
                best_h, best_c5 = h_new, new_c5

        # occasional random restart to avoid stagnation
        if iteration % 5000 == 0:
            rand_vec = np.random.rand(n_points)
            h_restart = project_feasible(rand_vec, target_sum)
            restart_c5 = compute_c5(h_restart, dx)
            if restart_c5 < best_c5:
                best_h, best_c5 = h_restart, restart_c5

    # final clean‑up: ensure exact feasibility
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)

    return best_h, float(best_c5), n_points
