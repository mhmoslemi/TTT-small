import time
import numpy as np

def _binary_from_continuous(vec):
    """Convert any vector to a binary 0/1 vector with exactly half ones."""
    n = vec.size
    # ensure even
    if n % 2:
        n -= 1
        vec = vec[:n]
    half = n // 2
    # indices of largest half become 1
    order = np.argsort(vec)[::-1]
    binary = np.zeros(n, dtype=np.float64)
    binary[order[:half]] = 1.0
    return binary

def _initial_h(n_points):
    """Create a random binary start respecting the integral constraint."""
    half = n_points // 2
    h = np.zeros(n_points, dtype=np.float64)
    h[:half] = 1.0
    np.random.shuffle(h)
    return h

def _score(h, dx):
    """Return C5 value for a candidate h."""
    # correlation length 2n-1, max over all lags
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    Returns:

        c5_bound (float): corresponding C5 value.
        n_points (int): number of discretization points.
    """

    start_time = time.time()

            n -= 1
        # enforce exact integral by binarising (keeps feasibility)
        h = _binary_from_continuous(h0)
    else:
        # default modest size, even
        n = 200
        h = _initial_h(n)

    dx = 2.0 / n
    best_h = h.copy()
    best_score = _score(best_h, dx)

    # Pre‑compute index sets for quick swaps
    def update_index_sets(arr):

        return ones, zeros

    ones_idx, zeros_idx = update_index_sets(best_h)

    # Simple hill‑climbing with random swaps

        # Propose a swap

            ones_idx, zeros_idx = update_index_sets(best_h)
