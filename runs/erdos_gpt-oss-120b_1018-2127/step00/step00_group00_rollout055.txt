import numpy as np
from scipy.optimize import minimize

def compute_score(h):
    """Maximum overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / h.size
    return np.max(corr) * dx

def is_feasible(h, n, atol=1e-6):
    """Check box constraints and integral constraint."""
    if np.any(h < -atol) or np.any(h > 1 + atol):
        return False
    return abs(h.sum() - n / 2) <= atol

def transfer_solution(old_h, n_new):
    """Linearly interpolate old_h (defined on its own uniform grid) to length n_new."""
    n_old = old_h.size
    # old grid points (excluding the endpoint 2)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
