import time
import math
import random
import numpy as np

def _compute_c5(h):
    """Exact C5 evaluation for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_feasible(v, target_sum):
    """
    Project vector v onto the box [0,1]^n while enforcing sum(v)=target_sum.
    Simple iterative clipping + uniform correction; sufficient for small n.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-10:
            break
        v += diff / v.size
        v = np.clip(v, 0.0, 1.0)
    # final small correction on free variables
    s = v.sum()
    diff = target_sum - s
    if abs(diff) > 1e-8:
        free = (v > 1e-12) & (v < 1.0 - 1e-12)
        if free.any():
            v[free] += diff / free.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _upscale(h, new_n):
    """Linear interpolation of a step vector onto a finer grid."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def _optimize_h(h_start, n, time_limit):
    """
    Simple simulated‑annealing style search respecting the constraints.
    Returns the best vector found and its C5 value.
    """
    target_sum = n / 2.0
    best = _project_feasible(h_start, target_sum)
    best_score = _compute_c5(best)

    cur = best.copy()
    cur_score = best_score

    start = time.time()
    it = 0
    # annealing schedule parameters
    temp0, temp1 = 0.15, 0.001

    while time.time() - start < time_limit:
        it += 1
        # linear temperature decay
        t = (time.time() - start) / time_limit
        temp = temp0 * (temp1 / temp0) ** t

        # propose a Gaussian perturbation
        proposal = cur + np.random.randn(n) * (0.05 * temp)
        proposal = _project_feasible(proposal, target_sum)

        prop_score = _compute_c5(proposal)

        # accept if better or with Metropolis probability
        if prop_score < cur_score or random.random() < math.exp((cur_score - prop_score) / max(temp, 1e-8)):
            cur, cur_score = proposal, prop_score

        if prop_score < best_score:
            best, best_score = proposal, prop_score

    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
    h_values is a 1‑D NumPy array satisfying the problem constraints,
    c5_bound is the achieved maximal overlap, and n_points is its length.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Acquire an initial construction if supplied by the harness.
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float, copy=True)
        n0 = init_h.size
    except NameError:
        # Fallback: uniform 0.5 satisfying the integral constraint.
        n0 = 60
        init_h = np.full(n0, 0.5, dtype=float)

    # Ensure the initial vector respects the constraints.
    init_h = _project_feasible(init_h, n0 / 2.0)

    # ------------------------------------------------------------------
    # Decide a small schedule of resolutions to explore.
    # ------------------------------------------------------------------
    resolutions = [n0]
    if n0 * 2 <= 180:
        resolutions.append(n0 * 2)
    if n0 * 4 <= 180:
        resolutions.append(n0 * 4)

    # Allocate time roughly equally among the chosen resolutions.
    total_res = len(resolutions)
    per_res_budget = budget_s / total_res * 0.95  # keep a safety margin

    best_overall = None
    best_score = float('inf')
    best_n = None

    for n in resolutions:
        # time check before starting a new resolution
        if time.time() - time.process_time() > budget_s:
            break

        # upscale or downscale the starting vector to current resolution
        if n == n0:
            h_start = init_h.copy()
        else:
            h_start = _upscale(init_h, n)

        h_opt, score = _optimize_h(h_start, n, per_res_budget)

        if score < best_score:
            best_overall = h_opt
            best_score = score
            best_n = n

    # ------------------------------------------------------------------
    # Fallback safety: if nothing succeeded, return a uniform construction.
    # ------------------------------------------------------------------
    if best_overall is None:
        best_n = n0
        best_overall = np.full(best_n, 0.5, dtype=float)
        best_score = _compute_c5(best_overall)

    return best_overall, best_score, int(best_n)
