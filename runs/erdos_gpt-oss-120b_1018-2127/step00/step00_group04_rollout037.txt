import time
import math
import random
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1‑h(x+k)) dx for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    # convolution via FFT (full length = 2n‑1)
    m = 2 * n - 1
    H = np.fft.rfft(h, m)
    G = np.fft.rfft((1.0 - h)[::-1], m)
    conv = np.fft.irfft(H * G, m)
    return float(np.max(conv) * dx)


def _binary_initial(n: int) -> np.ndarray:
    """Create a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h


def _upgrade_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Upscale a possibly non‑binary initial array to length n and force binary form."""
    # simple linear interpolation, then binarise to keep the integral exact
    x_src = np.linspace(0.0, 2.0, init.size, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n, endpoint=False)
    interp = np.interp(x_dst, x_src, init)
    # turn into a binary vector with the right number of ones
    ones_needed = n // 2
    flat = np.argsort(-interp)  # descending order
    h = np.zeros(n, dtype=float)
    h[flat[:ones_needed]] = 1.0
    return h


def _local_search(h: np.ndarray, budget: float) -> tuple:
    """Hill‑climb / simulated‑annealing on a binary vector."""
    start = time.time()
    n = h.size
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    # annealing schedule parameters
    T0 = 0.2
    Tf = 1e-4
    max_iter = 20000

    for it in range(max_iter):
        if time.time() - start > budget:
            break
        t = it / max_iter
        T = T0 * (Tf / T0) ** t

        # propose a swap of a 1 and a 0
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]  # swap

        cur_val = _compute_c5(h)
        delta = cur_val - best_val
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-12)):
            # accept
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # reject – revert swap
            h[i], h[j] = h[j], h[i]

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function for the Erdős C₅ problem.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values – list of floats in [0,1] (length n_points)
        c5_bound – the achieved C₅ upper bound (float)
        n_points – number of discretisation points (int)
    """
    random.seed(seed)
    np.random.seed(seed)

    # choose a modest grid size (fast evaluation, enough resolution)
    n_points = 200
    total_budget = float(budget_s)

    # initialise h
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        h0 = _upgrade_initial(init_arr, n_points)
    else:
        h0 = _binary_initial(n_points)

    # allocate time: a first phase for a quick improvement, then repeat restarts
    start_time = time.time()
    best_h, best_c5 = _local_search(h0.copy(), total_budget * 0.3)

    # continue with fresh random starts while time permits
    while time.time() - start_time < total_budget * 0.95:
        remaining = total_budget - (time.time() - start_time)
        # each extra run gets a short budget proportional to remaining time
        h_start = _binary_initial(n_points)
        h_candidate, c5_candidate = _local_search(h_start, remaining * 0.2)
        if c5_candidate < best_c5:
            best_c5, best_h = c5_candidate, h_candidate

    return (best_h.tolist(), float(best_c5), n_points)
