import time, math
import numpy as np

def compute_c5(h):
    """Return the overlap bound C5 for a discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_onto_constraints(x, target_sum, tol=1e-9, max_iter=200):
    """
    Project vector x onto the box [0,1]^n and the affine hyperplane sum(x)=target_sum.
    Simple water‑filling style iteration.
    """
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = x.sum()
        diff = target_sum - cur
        if abs(diff) < tol:
            break
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size == 0:               # everything stuck at a bound
            free = np.arange(x.size)
        adj = diff / free.size
        x[free] += adj
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction by scaling the free part if needed
    cur = x.sum()
    if abs(cur - target_sum) > tol:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            x[free] *= target_sum / cur
            x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step‑function h on [0,2] with ∫h = 1 to minimise
    C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ---- initialise ---------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=float)   # provided by harness
    except NameError:
        n0 = 70
        h0 = np.zeros(n0, dtype=float)
        h0[: n0 // 2] = 1.0                                 # simple block function

    n = h0.shape[0]
    target = n / 2.0
    h = project_onto_constraints(h0.copy(), target)
    cur_score = compute_c5(h)

    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ---- optimisation loop -------------------------------------------------
    while time_left() > 1.0:               # keep a small safety margin
        # linear cooling of a pseudo‑temperature
        progress = (time.time() - start) / budget_s
        temp = 0.5 * (1.0 - progress) + 1e-4

        # propose a random perturbation affecting ~10 % of entries
        prop = h.copy()
        k = max(1, int(0.1 * n))
        idx = rng.choice(n, size=k, replace=False)
        prop[idx] += rng.normal(scale=temp, size=k)

        prop = project_onto_constraints(prop, target)
        prop_score = compute_c5(prop)

        # keep the global best
        if prop_score < best_score:
            best_score = prop_score
            best_h = prop.copy()

        # Metropolis acceptance
        if prop_score < cur_score or rng.random() < math.exp(-(prop_score - cur_score) / temp):
            h = prop
            cur_score = prop_score

        # occasional up‑sampling to a finer grid (once, when enough time)
        if n < 200 and time_left() < budget_s * 0.4 and progress > 0.6:
            # double the resolution using nearest‑neighbour repeat
            h = np.repeat(best_h, 2)
            n = h.shape[0]
            target = n / 2.0
            h = project_onto_constraints(h, target)
            cur_score = compute_c5(h)
            best_h = h.copy()
            best_score = cur_score

    return best_h, best_score, n
