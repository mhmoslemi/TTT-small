def repair(h, target):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if abs(total - target) < 1e-9:
        return h
    # adjust to meet target via water-filling
    # We'll use simple iterative approach:
    diff = target - total
    # allocate diff uniformly to non-boundary entries
    for _ in range(10):
        free = (h > 0) & (h < 1)
        count = free.sum()
        if count == 0:
            break
        delta = diff / count
        # apply delta, but ensure stay within bounds
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
        new_total = h.sum()
        diff = target - new_total
        if abs(diff) < 1e-9:
            break
    # final scaling if still off
    if abs(h.sum() - target) > 1e-6:
        # simple scaling, then clip again
        factor = target / h.sum()
        h = np.clip(h * factor, 0.0, 1.0)
        # may need final adjustment

            interior = (h > 0) & (h < 1)

                h[interior] += diff / interior.sum()
                np.clip(h, 0.0, 1.0, out=h)
    return h

import time
import random
import math
import numpy as np

def _evaluate(h):
    """Return C5 value for discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h, target):
    """Project h onto [0,1]^n with sum == target."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if abs(total - target) < 1e-12:
        return h
    diff = target - total
    # iterative water‑filling
    for _ in range(15):
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] += delta
        np.clip(h, 0.0, 1.0, out=h)
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
    # final tiny scaling if needed
    if abs(h.sum() - target) > 1e-10:
        factor = target / h.sum()
        h = np.clip(h * factor, 0.0, 1.0)
        # distribute any residual uniformly among interior points
        diff = target - h.sum()
        interior = (h > 0.0) & (h < 1.0)
        if interior.any():
            h[interior] += diff / interior.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _simulated_annealing(start_h, target, time_limit):
    """Run a cheap SA within the given time budget, return best (h,score)."""
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_score = _evaluate(best_h)
    cur_h = best_h.copy()
    cur_score = best_score

    T = 1.0               # initial temperature
    cooling = 0.995
    step_scale = 0.2      # max perturbation magnitude
    start = time.time()
    it = 0
    while True:
        now = time.time()
        if now - start >= time_limit:
            break
        # propose
        perturb = (np.random.rand(n) - 0.5) * step_scale
        cand = cur_h + perturb
        cand = _repair(cand, target)
        cand_score = _evaluate(cand)

        # SA acceptance
        if cand_score < cur_score or random.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_h, cur_score = cand, cand_score
            if cand_score < best_score:
                best_h, best_score = cand.copy(), cand_score

        # temperature update
        T *= cooling
        it += 1
        # occasional restart to avoid stagnation
        if it % 5000 == 0:
            cur_h = best_h.copy()
            cur_score = best_score
            T = 1.0
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    init = globals().get("initial_h_values", None)
    if init is not None:
        h0 = np.array(init, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        # simple balanced binary start
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

    target_sum = n / 2.0
    h0 = _repair(h0, target_sum)

    # Allocate time slices
    total_budget = budget_s
    # 70% for main SA, 30% for a quick second start if time remains
    main_budget = total_budget * 0.7
    remaining = total_budget - main_budget

    best_h, best_score = _simulated_annealing(h0, target_sum, main_budget)

    # Optional second restart with a random start if time permits
    if remaining > 5.0:
        # random interior start (continuous values)
        rand_h = np.random.rand(n)
        rand_h = _repair(rand_h, target_sum)
        h2, s2 = _simulated_annealing(rand_h, target_sum, remaining)
        if s2 < best_score:
            best_h, best_score = h2, s2

    # Ensure final vector respects constraints exactly
    best_h = _repair(best_h, target_sum)

    # Return plain Python list for compatibility
    return best_h.tolist(), float(best_score), int(n)
