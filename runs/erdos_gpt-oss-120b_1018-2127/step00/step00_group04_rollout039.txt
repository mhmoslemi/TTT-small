import time
import numpy as np

def compute_c5(h, dx):
    """Return max overlap integral for step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_to_feasible(h, target_sum, max_iter=20, tol=1e-9):
    """
    Project h onto the box [0,1] and enforce exact sum == target_sum.
    Uses simple iterative clipping and uniform correction.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        # Distribute the difference uniformly, then re‑clip
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # If still off by a tiny amount, perform a deterministic tidy‑up
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > tol:
        # Walk through entries adding/subtracting as needed without breaking bounds
        for i in range(len(h)):
            if diff == 0:
                break
            if diff > 0 and h[i] < 1.0:
                inc = min(diff, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            elif diff < 0 and h[i] > 0.0:
                dec = min(-diff, h[i])
                h[i] -= dec
                diff += dec
    return h

def initial_from_parent(n_points):
    """
    Upsample the globally available `initial_h_values` (if defined) to n_points
    using linear interpolation. If the variable does not exist, fall back to a
    random feasible vector.
    """
    global initial_h_values
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        n0 = parent.size
        x_parent = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_parent, parent)
    except Exception:
        # random feasible: half ones, half zeros shuffled
        h = np.zeros(n_points, dtype=float)
        h[:n_points // 2] = 1.0
        np.random.shuffle(h)
    # enforce exact integral
    target = n_points / 2.0
    h = project_to_feasible(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] sampled at n_points points to minimise
    the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimised vector of length n_points, values in [0,1] and sum == n_points/2.
    c5_bound : float
        Corresponding C5 value (max overlap integral).
    n_points : int
        Number of discrete samples used.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Choose a moderate resolution; use 200 points (dx = 0.01)
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ---- initial solution -------------------------------------------------
    h_curr = initial_from_parent(n_points)
    best_h = h_curr.copy()
    best_c5 = compute_c5(best_h, dx)

    # ---- simulated‑annealing parameters ----------------------------------
    temp_start = 0.5
    temp_end = 1e-4
    iteration = 0
    max_iter = 200000  # safeguard; loop will also stop on time

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        if iteration >= max_iter:
            break

        # linear cooling schedule
        frac = iteration / max_iter
        temp = temp_start * (1 - frac) + temp_end * frac

        # propose a new candidate by Gaussian perturbation
        sigma = 0.05 * (1 - frac) + 0.005 * frac  # shrink step size
        proposal = h_curr + sigma * np.random.randn(n_points)
        proposal = project_to_feasible(proposal, target_sum)

        c5_prop = compute_c5(proposal, dx)

        # Metropolis acceptance
        if c5_prop < best_c5:
            best_c5 = c5_prop
            best_h = proposal.copy()
        delta = c5_prop - compute_c5(h_curr, dx)
        if delta <= 0 or np.exp(-delta / max(temp, 1e-12)) > np.random.rand():
            h_curr = proposal

        iteration += 1

    # final tiny refinement: attempt deterministic local improvements
    # Flip tiny amounts between two coordinates while preserving feasibility.
    for _ in range(500):
        i, j = np.random.choice(n_points, 2, replace=False)
        step = 0.01
        # move step from i to j
        if best_h[i] >= step and best_h[j] + step <= 1.0:
            cand = best_h.copy()
            cand[i] -= step
            cand[j] += step
            c5_cand = compute_c5(cand, dx)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = cand

    # Ensure final feasibility exactly
    best_h = project_to_feasible(best_h, target_sum)

    return best_h, best_c5, n_points
