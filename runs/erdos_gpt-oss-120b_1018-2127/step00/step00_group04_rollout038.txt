import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for a step function h."""
    # correlation h with (1-h)
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(conv))

def _initial_binary(n, seed):
    """Create a binary vector of length n with exactly n/2 ones."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def _use_initial_h(initial, n_target):
    """Convert provided initial_h_values to a binary vector of length n_target."""
    rng = np.random.default_rng(0)
    init = np.asarray(initial, dtype=float)
    n0 = init.size
    # If length matches target, simply binarise and adjust count.
    if n0 == n_target:
        h = (init > 0.5).astype(float)
        cur = int(h.sum())
        need = n_target // 2 - cur
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if need > 0:
            flip = rng.choice(zeros, need, replace=False)
            h[flip] = 1.0
        elif need < 0:
            flip = rng.choice(ones, -need, replace=False)
            h[flip] = 0.0
        return h
    # Otherwise up‑sample / down‑sample by repeating pattern.
    repeats = int(np.ceil(n_target / n0))
    h_big = np.tile(init, repeats)[:n_target]
    h = (h_big > 0.5).astype(float)
    cur = int(h.sum())
    need = n_target // 2 - cur
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    if need > 0:
        flip = rng.choice(zeros, need, replace=False)
        h[flip] = 1.0
    elif need < 0:
        flip = rng.choice(ones, -need, replace=False)
        h[flip] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a binary step function h on [0,2] to minimise the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Optimised vector (values 0.0 or 1.0) sampled uniformly on [0,2].
    c5_bound : float
        The achieved max overlap integral.
    n_points : int
        Number of discretisation points used.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a moderate grid size (fast enough, yet expressive)
    n_points = 200               # must be even so that sum = n/2 is integer
    dx = 2.0 / n_points

    # Initialise h either from the provided construction or from scratch
    try:
        # `initial_h_values` is injected by the harness if present
        init = initial_h_values  # pylint: disable=undefined-variable
        h = _use_initial_h(init, n_points)
    except NameError:
        h = _initial_binary(n_points, seed)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    # Simulated‑annealing parameters
    temperature = 0.02
    decay = 0.9995
    max_iter = 200000          # safety cap
    iter_count = 0

    # Pre‑compute index pools for swaps
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95 or iter_count >= max_iter:
            break
        iter_count += 1

        # Choose a random 1‑index and a random 0‑index to swap
        ones_idx = np.flatnonzero(h == 1.0)
        zeros_idx = np.flatnonzero(h == 0.0)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break  # should never happen with correct initialization
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        # Perform swap
        h[i], h[j] = 0.0, 1.0

        new_c5 = _compute_c5(h, dx)

        # Acceptance test
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Metropolis criterion
            delta = new_c5 - best_c5
            if rng.random() < np.exp(-delta / temperature):
                # Accept worse state (helps escape local minima)
                best_c5 = new_c5
                best_h = h.copy()
            else:
                # Revert swap
                h[i], h[j] = 1.0, 0.0

        # Cool down
        temperature *= decay
        if temperature < 1e-7:
            temperature = 1e-7

    # Final evaluation (ensuring feasibility)
    final_c5 = _compute_c5(best_h, dx)

    # Return as required tuple
    return best_h, final_c5, n_points
