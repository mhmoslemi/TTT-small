import numpy as np
import time
import random

def _project_capped_simplex(x, target):
    """Project onto {0<=v_i<=1, sum(v)=target} using bisection on lambda."""
    # Clip initial bounds
    lo, hi = -1.0, 1.0
    for _ in range(40):
        lam = (lo + hi) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h, dx):
    """Compute the overlap bound C5 for a discretised step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _initial_vector(n):
    """Create a random feasible vector of length n."""
    x = np.random.rand(n)
    return _project_capped_simplex(x, n / 2.0)


def _optimize(h, dx, time_allow):
    """Simple simulated‑annealing style local search within a time budget."""
    start = time.time()
    n = h.size
    best = h.copy()
    best_val = _c5_value(best, dx)
    cur = best.copy()
    cur_val = best_val
    # Temperature schedule parameters
    T0 = 0.2
    while time.time() - start < time_allow:
        # linear cooling
        elapsed = time.time() - start
        frac = elapsed / time_allow
        T = T0 * (1.0 - frac)
        # propose a change on a random coordinate
        idx = np.random.randint(n)
        delta = (np.random.rand() * 2.0 - 1.0) * T
        cand = cur.copy()
        cand[idx] += delta
        cand = _project_capped_simplex(cand, n / 2.0)
        cand_val = _c5_value(cand, dx)
        # acceptance rule
        if cand_val < cur_val or np.random.rand() < np.exp((cur_val - cand_val) / (T + 1e-12)):
            cur, cur_val = cand, cand_val
            if cur_val < best_val:
                best, best_val = cur.copy(), cur_val
        # occasional global shake to avoid stagnation
        if random.random() < 0.001:
            cur = _project_capped_simplex(np.random.rand(n), n / 2.0)
            cur_val = _c5_value(cur, dx)
    return best, best_val
