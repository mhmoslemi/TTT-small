import time
import numpy as np
import random

# Helper: project a vector onto the box [0,1] with a prescribed sum T = n/2
def _project_bounds_sum(v, target):
    x = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        s = x.sum()
        if abs(s - target) <= 1e-12:
            break
        if s > target:
            adjustable = x > 0.0
        else:
            adjustable = x < 1.0
        if not np.any(adjustable):
            break
        delta = (s - target) / adjustable.sum()
        # move opposite to the excess direction
        x[adjustable] -= np.sign(s - target) * delta
        x = np.clip(x, 0.0, 1.0)
    # final small correction to hit the target exactly (if possible)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute the remaining diff uniformly over non‑saturated entries
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

# Evaluation of C5 for a discrete h
def _c5_value(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    t_start = time.time()

    # Determine resolution
    if 'n_points' in kwargs:
        n_points = int(kwargs['n_points'])
    else:
        # use the size of the provided initial vector if present, else 70
        try:
            n_points = len(initial_h_values)
        except NameError:
            n_points = 70

    target_sum = n_points / 2.0   # because integral must be 1 and interval length is 2

    # Build an initial feasible vector
    try:
        base = np.asarray(initial_h_values, dtype=float)
        if base.size != n_points:
            # linear interpolation to required size
            xs = np.linspace(0, 2, base.size, endpoint=False)
            xt = np.linspace(0, 2, n_points, endpoint=False)
            base = np.interp(xt, xs, base)
    except NameError:
        base = np.random.rand(n_points)

    h_current = _project_bounds_sum(base, target_sum)
    best_h = h_current.copy()
    best_val = _c5_value(best_h)

    # Parameters for the stochastic search
    max_no_improve = 2000
    no_improve = 0
    temperature0 = 0.05
    temperature_decay = 0.9995

    while True:
        elapsed = time.time() - t_start
        if elapsed >= budget_s - 0.5:
            break

        # temperature schedule
        temp = temperature0 * (temperature_decay ** (elapsed / 0.1))

        # propose a neighbour
        noise = np.random.randn(n_points) * temp
        h_candidate = h_current + noise
        h_candidate = _project_bounds_sum(h_candidate, target_sum)

        val = _c5_value(h_candidate)

        # accept if better or with Metropolis probability
        if val < best_val:
            best_val, best_h = val, h_candidate.copy()
            no_improve = 0
        else:
            no_improve += 1

        # simple hill‑climb acceptance to keep moving
        if val < _c5_value(h_current) or random.random() < np.exp(-(val - _c5_value(h_current)) / (temp + 1e-12)):
            h_current = h_candidate

        if no_improve > max_no_improve:
            # random restart
            h_current = _project_bounds_sum(np.random.rand(n_points), target_sum)
            no_improve = 0

    return best_h, best_val, n_points
