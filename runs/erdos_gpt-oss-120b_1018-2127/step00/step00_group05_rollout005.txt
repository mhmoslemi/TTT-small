import time
import numpy as np

def _repair(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the target sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        s = h.sum()
        if s == 0:
            break
        if abs(s - target_sum) <= 1e-12:
            break
        factor = target_sum / s
        h = np.clip(h * factor, 0.0, 1.0)
    # final tiny scaling if needed
    s = h.sum()
    if s != 0 and abs(s - target_sum) > 1e-12:
        h *= target_sum / s
        h = np.clip(h, 0.0, 1.0)
    return h


def _evaluate(h: np.ndarray, dx: float) -> float:
    """Return C5 for the step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _mutate(parent: np.ndarray, rng: np.random.Generator,
            target_sum: float, strategy: str) -> np.ndarray:
    """Create a mutated child from parent using the specified strategy."""
    n = parent.size
    child = parent.copy()

    if strategy == "A":  # resample ~20% of entries uniformly in [0,1]
        k = max(1, n // 5)
        idx = rng.choice(n, size=k, replace=False)
        child[idx] = rng.random(k)
    else:               # strategy "B": small perturbation on ~30% entries
        mask = rng.random(n) < 0.30
        noise = (rng.random(mask.sum()) * 2.0 - 1.0) * 0.015
        child[mask] += noise
        child = np.clip(child, 0.0, 1.0)

    child = _repair(child, target_sum)
    return child


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h on [0,2] that minimises the overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : 1‑D NumPy array with values in [0,1] and integral 1.
        c5_bound : the obtained C5 value (smaller is better).
        n_points : length of the discretisation (len(h_values)).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise problem size -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: 200 equally spaced points, start from a simple triangular shape
        n_fallback = 200
        x = np.linspace(0, 2, n_fallback, endpoint=False)
        h0 = np.where(x < 1, 2 * x / 2, 2 * (2 - x) / 2)  # tent, integral = 1
    n = h0.size
    target_sum = n / 2.0          # because ∫h = 1 and dx = 2/n
    dx = 2.0 / n

    # ----- build initial population ------------------------------------------------
    POP_SIZE = 12
    pop = []
    # ensure first individual respects constraints
    pop.append(_repair(h0.copy(), target_sum))
    while len(pop) < POP_SIZE:
        rand = rng.random(n)
        pop.append(_repair(rand, target_sum))

    scores = [_evaluate(ind, dx) for ind in pop]
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    # ----- evolutionary loop -------------------------------------------------------
    ITER_BATCH = 40   # children generated per loop
    while time.time() - start_time < budget_s * 0.95:  # leave a margin
        # generate children
        children = []
        for _ in range(ITER_BATCH):
            # pick a parent proportionally to fitness (better = higher chance)
            weights = np.reciprocal(np.array(scores) + 1e-12)
            parent = pop[rng.choice(POP_SIZE, p=weights / weights.sum())]
            strat = "A" if rng.random() < 0.6 else "B"
            child = _mutate(parent, rng, target_sum, strat)
            children.append(child)

        # evaluate children and possibly replace worst individuals
        for child in children:
            child_score = _evaluate(child, dx)
            # if child beats current worst, replace it
            worst_idx = int(np.argmax(scores))
            if child_score < scores[worst_idx]:
                pop[worst_idx] = child
                scores[worst_idx] = child_score

                # update global best
                if child_score < best_score:
                    best_score = child_score
                    best_h = child.copy()

        # early exit if excellent bound reached
        if best_score <= 0.3808:
            break

    # final safety repair (in case of tiny drift)
    best_h = _repair(best_h, target_sum)

    return best_h, float(best_score), n
