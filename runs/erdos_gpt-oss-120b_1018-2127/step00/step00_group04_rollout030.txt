import time
import numpy as np

def compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)

def make_binary_vector(n, ones_target, seed=None):
    """Create a binary vector of length n with exactly ones_target ones."""
    rng = np.random.default_rng(seed)
    vec = np.zeros(n, dtype=float)
    idx = rng.choice(n, size=ones_target, replace=False)
    vec[idx] = 1.0
    return vec

def project_to_feasible(vec):
    """Clip to [0,1] and adjust to have exactly sum = n/2."""
    n = vec.shape[0]
    target = n // 2
    # Clip
    vec = np.clip(vec, 0.0, 1.0)
    # Round to nearest binary while preserving total count
    # start from current rounded version
    rounded = (vec >= 0.5).astype(float)
    cur = int(rounded.sum())
    if cur == target:
        return rounded
    # Need to add or remove ones
    diff = target - cur
    if diff > 0:
        # turn diff zeros with largest fractional part into ones
        fractions = vec - rounded
        zero_idx = np.where(rounded == 0)[0]
        order = np.argsort(-fractions[zero_idx])  # descending
        pick = zero_idx[order[:diff]]
        rounded[pick] = 1.0
    else:
        # turn -diff ones with smallest fractional part into zeros
        fractions = vec - rounded
        one_idx = np.where(rounded == 1)[0]
        order = np.argsort(fractions[one_idx])  # ascending
        pick = one_idx[order[: -diff]]
        rounded[pick] = 0.0
    return rounded

def hill_climb(initial_h, time_limit):
    """Simple hill‑climbing with random swaps."""
    n = initial_h.shape[0]
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    h = best_h.copy()
    cur_c5 = best_c5
    rng = np.random.default_rng()
    start = time.time()
    while time.time() - start < time_limit:
        # pick a random 1 and a random 0 to swap
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # reject move
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5
