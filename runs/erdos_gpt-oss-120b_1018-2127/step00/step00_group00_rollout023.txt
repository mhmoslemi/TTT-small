    # cvxpy variable
    x = cvxpy.Variable(n)
    prob = cvxpy.Problem(cvxpy.Minimize(cvxpy.sum_squares(x - h)),

    prob.solve(solver=cvxpy.ECOS, warm_start=True, verbose=False, max_iters=5000)

import time
import numpy as np
import cvxpy as cp

def _project_feasible(vec, target_sum):
    """Project onto 0≤x≤1 and sum(x)=target_sum via a small QP."""
    n = vec.shape[0]
    x = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(x - vec)),
                      [x >= 0, x <= 1, cp.sum(x) == target_sum])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
    except Exception:  # pragma: no cover
        prob.status = cp.INFEASIBLE
    if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return np.array(x.value).flatten()
    # fallback: simple scaling and clipping
    proj = np.clip(vec, 0.0, 1.0)
    s = proj.sum()
    if s == 0:
        proj[:] = target_sum / n
    else:
        proj = proj * (target_sum / s)
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def _interpolate_h(old_h, new_n):
    """Resample a step function defined on [0,2) to a new resolution."""
    old_n = old_h.shape[0]
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(xs_new, xs_old, old_h)

def _c5_score(h):
    """Exact evaluation using the pre‑imported helper."""
    # evaluate_erdos_solution is supplied by the harness
    return evaluate_erdos_solution(h)

def _optimize_one(start_h, n, time_allow):
    """Simple hill‑climbing with Gaussian proposals."""
    target = n / 2.0
    best_h = _project_feasible(start_h, target)
    best_score = _c5_score(best_h)
    end_time = time.time() + time_allow
    sigma = 0.12
    it = 0
    while time.time() < end_time:
        cand = best_h + sigma * np.random.randn(n)
        cand = _project_feasible(cand, target)
        sc = _c5_score(cand)
        if sc < best_score:
            best_score, best_h = sc, cand
        it += 1
        if it % 250 == 0:
            sigma *= 0.9  # gradually tighten
    return best_h, best_score
