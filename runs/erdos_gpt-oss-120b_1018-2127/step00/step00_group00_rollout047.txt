import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def compute_score(h):
    """Return C5 value for step vector h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def is_feasible(h, atol=1e-6):
    """Check bounds and integral constraint."""
    n = h.size
    if np.any(h < -atol) or np.any(h > 1.0 + atol):
        return False
    return abs(h.sum() - n / 2.0) <= atol

def project_feasible(h):
    """Simple projection: clip to [0,1] then rescale to meet sum=n/2."""
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    cur = h.sum()
    if cur == 0:
        # distribute uniformly
        h[:] = target / n
    else:
        h *= target / cur
        # after rescaling may violate bounds slightly; clip again
        h = np.clip(h, 0.0, 1.0)
        # final small adjustment to match sum (could be tiny residual)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute diff uniformly while staying in bounds
            free = (h > 0) & (h < 1)
            if free.any():
                h[free] += diff / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def transfer_solution(old_h, new_n):
    """Resample old_h (length m) to length new_n via linear interpolation."""
    m = old_h.size
    if m == new_n:
        return old_h.copy()
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    return project_feasible(new_h)

def minimize_score(start_h, maxiter=400):
    n = start_h.size
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - n / 2.0}
    res = minimize(
        compute_score,
        start_h,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
    )
    if res.success:
        return project_feasible(res.x)
    return None
