import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Return the maximal overlap integral for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _feasible_projection(v, target_sum):
    """Project v onto [0,1] box and adjust to hit the required sum."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if np.isclose(cur, target_sum):
        return v
    # simple redistribution of deficit/excess over interior points
    diff = target_sum - cur
    free = (v > 0.0) & (v < 1.0)
    if free.any():
        amount = diff / free.sum()
        v[free] += amount
        v = np.clip(v, 0.0, 1.0)
        # second pass if still off (rare)
        cur = v.sum()
        if not np.isclose(cur, target_sum):
            v *= target_sum / cur
    else:
        # all at bounds, just scale globally and re‑clip
        v *= target_sum / cur
        v = np.clip(v, 0.0, 1.0)
    return v

def _interpolate_vector(old_h, new_n):
