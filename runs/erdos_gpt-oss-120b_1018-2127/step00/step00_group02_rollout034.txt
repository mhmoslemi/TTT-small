import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return the C5 value for a given step function sampled in h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def projection_feasible(h, target_sum):
    """Project h back onto the feasible set: 0<=h<=1 and sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # simple uniform correction, repeated a few times for safety
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target_sum - cur
    # final tiny fix on a single entry if needed
    if abs(diff) > 1e-12:
        idx = np.argmax((h > 0) & (h < 1)) if diff > 0 else np.argmin(h)
        delta = np.clip(diff, -h[idx], 1.0 - h[idx])
        h[idx] += delta
    return h

def random_feasible_h(n, target_sum):
    """Generate a random feasible h (mostly binary, with a 0.5 if n is odd)."""
    ones = n // 2
    zeros = n // 2
    h = np.zeros(n)
    idx = np.arange(n)
    np.random.shuffle(idx)
    h[idx[:ones]] = 1.0
    if n % 2:
        h[idx[ones]] = 0.5
    # guarantee exact sum (tiny numeric adjustments)
    return projection_feasible(h, target_sum)

def optimize_from_start(x0, target_sum, max_iter, time_limit, start_time):
    """Run SLSQP from a given start, respecting the global time budget."""
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    cons = ({
        "type": "eq",
        "fun": lambda h: h.sum() - target_sum,
    },)

    def obj(h):
        return compute_c5(h)

    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = minimize(
        obj,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    # project back to feasible set in case of tiny violations
    h_opt = projection_feasible(res.x, target_sum)
    c5_val = compute_c5(h_opt)
    return h_opt, c5_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the allotted time."""
    np.random.seed(seed)
    start = time.time()
    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 45
        h0 = np.full(n, 0.5)

    target_sum = n / 2.0  # because integral over [0,2] must be 1

    best_h = projection_feasible(h0, target_sum)
    best_c5 = compute_c5(best_h)

    # First, improve starting from the provided (or uniform) vector
    if time.time() - start < budget_s * 0.9:
        h_opt, c5_val = optimize_from_start(
            best_h, target_sum, max_iter=1500, time_limit=budget_s, start_time=start
        )
        if c5_val < best_c5:
            best_c5, best_h = c5_val, h_opt

    # Random restarts while time permits
    while time.time() - start < budget_s * 0.95:
        # generate a fresh feasible start
        x0 = random_feasible_h(n, target_sum)
        h_opt, c5_val = optimize_from_start(
            x0, target_sum, max_iter=800, time_limit=budget_s, start_time=start
        )
        if c5_val < best_c5:
            best_c5, best_h = c5_val, h_opt

    # Final sanity projection
    best_h = projection_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
