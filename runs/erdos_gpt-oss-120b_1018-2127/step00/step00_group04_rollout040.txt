import time
import numpy as np

def _project_box_simplex(y, target_sum, max_iter=60):
    """Project y onto {x : 0<=x_i<=1, sum x_i = target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return x

def _c5_value(h):
    """Compute C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.shape[0]
        # enforce sum constraint exactly
        target_sum = n / 2.0
        h = _project_box_simplex(h, target_sum)
    else:
        n = 200                       # fallback size (still <1k)
        target_sum = n / 2.0
        h = np.full(n, 0.5, dtype=float)
        h = _project_box_simplex(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    start_time = time.time()
    max_time = budget_s * 0.95  # keep a safety margin

    # parameters for the stochastic gradient descent
    step_size = 0.5
    step_decay = 0.995
    min_step = 1e-4

    while time.time() - start_time < max_time:
        # full correlation to locate the worst shift
        conv_full = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(conv_full))
        shift = idx_max - (n - 1)   # integer shift corresponding to translation

        # compute gradient of the worst overlap term
        idx = np.arange(n)
        if shift >= 0:
            term1 = np.where(idx + shift < n, 1.0 - h[idx + shift], 0.0)
            term2 = np.where(idx - shift >= 0, h[idx - shift], 0.0)
        else:
            s = -shift
            term1 = np.where(idx - s >= 0, 1.0 - h[idx - s], 0.0)
            term2 = np.where(idx + s < n, h[idx + s], 0.0)
        grad = term1 - term2               # gradient of the worst overlap

        # gradient descent step (move opposite to gradient to reduce the max)
        h_new = h - step_size * grad
        # project back onto feasible set
        h_new = _project_box_simplex(h_new, target_sum)

        # occasional random perturbation to escape shallow minima
        if rng.random() < 0.02:
            h_new += rng.normal(scale=0.02, size=n)
            h_new = _project_box_simplex(h_new, target_sum)

        h = h_new

        # evaluate
        cur_c5 = _c5_value(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # adapt step size
        step_size = max(min_step, step_size * step_decay)

    # final check with the provided evaluator (if it exists)
    if 'evaluate_erdos_solution' in globals():
        try:
            eval_val = evaluate_erdos_solution(best_h)
            # if the evaluator returns a better bound, use it
            if isinstance(eval_val, (float, int)) and eval_val < best_c5:
                best_c5 = float(eval_val)
        except Exception:
            pass

    return best_h, best_c5, best_h.shape[0]
