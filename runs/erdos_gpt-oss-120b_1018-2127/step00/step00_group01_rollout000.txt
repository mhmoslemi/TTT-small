import numpy as np

def _evaluate(h):
    """Return the C5 value for a given h vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _project_to_bounds(x, lo=0.0, hi=1.0):
    """Clamp array to [lo,hi] and return mask of changed entries."""
    mask = (x < lo) | (x > hi)
    np.clip(x, lo, hi, out=x)
    return mask

def _local_search(initial, time_limit, target=None):

    h = initial.copy()
    n = h.size
    best_h = h.copy()
