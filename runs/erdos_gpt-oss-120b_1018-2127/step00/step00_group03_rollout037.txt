import time
import numpy as np

def compute_c5(h):
    """Return the overlap constant C5 for a step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def random_feasible(n, target_sum=None, max_tries=1000):
    """Generate a random vector in [0,1]^n whose sum equals target_sum (default n/2)."""
    if target_sum is None:
        target_sum = n / 2.0
    for _ in range(max_tries):
        x = np.random.rand(n)
        x = x / x.sum() * target_sum
        if np.all(x <= 1.0):
            return x
    # fallback – simple projection (rare)
    x = np.clip(x, 0.0, 1.0)
    excess = x.sum() - target_sum
    if excess > 0:
        # subtract uniformly from entries that are >0
        mask = x > 0
        x[mask] -= excess / mask.sum()
        x = np.clip(x, 0.0, 1.0)
    elif excess < 0:
        mask = x < 1
        x[mask] -= excess / mask.sum()
        x = np.clip(x, 0.0, 1.0)
    return x


def pattern_block(n):
    """Vector with first n/2 entries =1, rest =0 (feasible when n even)."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    h[:half] = 1.0
    return h


def pattern_alternating(n):
    """Alternating 1,0 pattern; sum = n/2 for even n."""
    h = np.zeros(n, dtype=float)
    h[::2] = 1.0
    return h


def ensure_feasible(h):
    """Clip to [0,1] and rescale to exact sum n/2 while staying inside box."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == target:
        return h
    # simple affine scaling, then re‑clip (rarely violates bounds)
    if s > 0:
        h = h * (target / s)
    else:
        h = np.full_like(h, target / n)
    h = np.clip(h, 0.0, 1.0)
    # final fix of sum by adding/subtracting tiny amount from a random index
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.random.randint(0, n)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h minimising C5."""
    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = ensure_feasible(h0)
    else:
        n = 70
        h0 = random_feasible(n)

    best_h = h0
    best_score = compute_c5(best_h)

    # Evaluate a few deterministic patterns
    for pat in (pattern_block, pattern_alternating):
        if n % 2 == 0:
            h_pat = pat(n)
            sc = compute_c5(h_pat)
            if sc < best_score:
                best_h, best_score = h_pat, sc

    # Main stochastic optimisation loop
    while time.time() - start_time < budget_s * 0.9:
        # With some probability, restart from a fresh random point
        if np.random.rand() < 0.15:
            cand = random_feasible(n)
        else:
            cand = best_h.copy()
            # Perform a small number of random transfers that respect bounds
            moves = np.random.randint(1, 4)
            for _ in range(moves):
                i, j = np.random.choice(n, 2, replace=False)
                # Direction: transfer from j to i or opposite
                if np.random.rand() < 0.5:
                    max_delta = min(1.0 - cand[i], cand[j])
                    if max_delta <= 0:
                        continue
                    delta = np.random.rand() * max_delta
                    cand[i] += delta
                    cand[j] -= delta
                else:
                    max_delta = min(cand[i], 1.0 - cand[j])
                    if max_delta <= 0:
                        continue
                    delta = np.random.rand() * max_delta
                    cand[i] -= delta
                    cand[j] += delta

        # Ensure feasibility (should already hold, but guard against numerical drift)
        cand = ensure_feasible(cand)
        score = compute_c5(cand)

        if score < best_score:
            best_h, best_score = cand, score

    return best_h, best_score, n
