import time
import numpy as np

def _c5_of_h(h, dx):
    """Return max overlap C5 for a given h vector."""
    # correlation of h with (1-h), zero‑padded, multiplied by dx
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _random_initial(rng, n):
    """Create a random feasible vector with average 0.5 (sum = n/2)."""
    # start with uniform 0.5 then add zero‑mean noise and project
    h = np.full(n, 0.5)
    # small random perturbation
    noise = rng.uniform(-0.2, 0.2, size=n)
    h += noise
    # clip to [0,1]
    np.clip(h, 0.0, 1.0, out=h)
    # adjust to exact sum using water‑filling
    target = n / 2.0
    excess = h.sum() - target
    if abs(excess) > 1e-12:
        # distribute excess uniformly while respecting bounds
        while abs(excess) > 1e-12:
            free = (h > 0) & (h < 1)
            if not np.any(free):
                break

            adj = np.where(free, delta, 0.0)
            h += adj
            np.clip(h, 0.0, 1.0, out=h)
            excess = h.sum() - target
    return h
