import numpy as np
import time
import math
import random

def _compute_c5(h):
    """Return C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair(h):
    """Clip to [0,1] and renormalise so sum(h)=n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        h[:] = target / n
        return h
    # iterative clip‑renorm to respect bounds after scaling
    for _ in range(5):
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    return h

def _initial_vector(n, rng):
    """Create a feasible random start (or use provided parent)."""
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.shape[0] == n:
            return _repair(parent.copy())
        # otherwise upscale/downscale linearly
        x_parent = np.linspace(0, 2, parent.shape[0], endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_parent, parent, period=2.0)
        return _repair(h)
    # no parent -> random start
    h = rng.random(n)
    return _repair(h)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] that minimises the Erdős C5 bound.

    Returns:
        h_values (np.ndarray): the best function values found (length n_points)
        c5_bound (float): the corresponding C5 value
        n_points (int): number of discretisation points used
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Primary resolution – start with 80 (the size of the supplied parent)
    n = 80
    best_h = _initial_vector(n, rng)
    best_score = _compute_c5(best_h)

    # Simulated annealing parameters
    T = best_score * 5.0                     # initial temperature
    T_min = 1e-6
    cooling = 0.9995
    sigma = 0.2                               # perturbation scale
    iter_count = 0
    restart_interval = 20000                  # steps before a random restart

    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:
            break

        # propose perturbation
        perturb = (rng.random(n) - 0.5) * sigma
        cand = best_h + perturb
        cand = _repair(cand)

        cand_score = _compute_c5(cand)

        # acceptance test
        if cand_score < best_score:
            best_h, best_score = cand, cand_score
        else:
            delta = cand_score - best_score
            if T > 0.0 and rng.random() < math.exp(-delta / T):
                best_h = cand
                best_score = cand_score

        # temperature schedule
        T = max(T * cooling, T_min)

        # slowly shrink perturbation magnitude
        sigma = max(sigma * 0.9999, 1e-4)

        iter_count += 1

        # occasional random restart to escape stagnation
        if iter_count % restart_interval == 0:
            # keep the current best as a seed for a new random start
            best_h = _repair(best_h + (rng.random(n) - 0.5) * 0.1)
            best_score = _compute_c5(best_h)
            T = best_score * 5.0
            sigma = 0.2

    # If time permits, try a higher resolution based on the best 80‑point pattern
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 30.0:      # need a few seconds to upscale and optimise

            cand = h2 + perturb
            cand = _repair(cand)
            cand_score = _compute_c5(cand)

            if cand_score < best_score:

                h2 = cand

            iter2 += 1
