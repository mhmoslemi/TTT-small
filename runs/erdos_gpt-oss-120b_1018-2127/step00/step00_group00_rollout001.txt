import numpy as np
import random

def _project_to_sum_and_bounds(vec, target):
    """Clip to [0,1] and adjust to meet the exact sum `target`."""
    vec = np.clip(vec, 0.0, 1.0)
    for _ in range(20):
        cur = vec.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        free = (vec > 0.0) & (vec < 1.0)
        if not free.any():
            # all at bounds – distribute uniformly (will stay clipped)
            vec += diff / vec.size
            vec = np.clip(vec, 0.0, 1.0)
            continue
        adj = diff / free.sum()
        vec[free] += adj
        vec = np.clip(vec, 0.0, 1.0)
    return vec
