import time
import numpy as np
from scipy.optimize import minimize

# The evaluation helper is provided in the harness.
# evaluate_erdos_solution(h) -> float (the C5 value)

def _project_feasible(v, target_sum):
    """Clip to [0,1] and adjust to meet the required sum."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(15):
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (v > 1e-12) & (v < 1 - 1e-12)
        if not free.any():
            # all at bounds – distribute uniformly (will be re‑clipped)
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
            continue
        adjust = diff / free.sum()
        v[free] += adjust
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_objective(h, dx):
    """Return the overlap value C5 for a candidate step function."""
    # correlation length is 2*n-1; max over all shifts
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Make sure the starting vector is feasible

    # Initial evaluation
