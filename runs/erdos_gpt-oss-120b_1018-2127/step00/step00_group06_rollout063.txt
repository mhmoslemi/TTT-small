import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the discrete approximation of C5 for vector h."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * (2.0 / n))

def random_pair_swap(h: np.ndarray, delta_max: float) -> np.ndarray:
    """Return a copy of h where a random amount delta is transferred
    from one entry to another while keeping the sum unchanged and
    staying inside [0,1]."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    # feasible delta range so that both stay in [0,1]
    max_pos = min(delta_max, 1.0 - hi, hj)          # add to i, remove from j
    max_neg = -min(delta_max, hi, 1.0 - hj)        # subtract from i, add to j
    if max_pos < max_neg:  # no feasible move, return unchanged copy
        return h.copy()
    delta = np.random.uniform(max_neg, max_pos)

    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

def project_to_constraints(h: np.ndarray, desired_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust linearly to hit the exact sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = desired_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute the difference uniformly; may violate bounds slightly,
    # so iterate a few times
    for _ in range(5):
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        diff = desired_sum - h.sum()
        if abs(diff) < 1e-12:
            break
    return h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        # fallback: uniform 0.5 vector of length 100 (even)
        n0 = 100
        h0 = np.full(n0, 0.5, dtype=float)

    # ensure even length and correct integral
    if h0.size % 2:
        h0 = np.append(h0, 0.0)
    n0 = h0.size
    desired_sum = n0 / 2.0
    h0 = project_to_constraints(h0, desired_sum)

    best_h = h0.copy()
    best_score = compute_c5(best_h)
    best_n = n0

    current_h = best_h.copy()
    current_n = n0

    # ------------------------------------------------------------------
    # Multi‑grid stochastic improvement
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        # decide next grid size
        next_n = current_n * 2
        if next_n > 1200:      # upper bound to keep runtime reasonable
            break

        # linear interpolation to new grid
        x_old = np.linspace(0.0, 2.0, current_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, next_n, endpoint=False)
        h_interp = np.interp(x_new, x_old, current_h)

        # enforce constraints on interpolated vector
        desired_sum = next_n / 2.0
        h_interp = project_to_constraints(h_interp, desired_sum)

        # local stochastic search on this grid
        cur_h = h_interp

        current_h = cur_h
        current_n = next_n

        if best_score <= 0.3808:
