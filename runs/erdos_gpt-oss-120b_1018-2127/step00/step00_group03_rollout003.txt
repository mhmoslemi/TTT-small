def compute_c5(h):
    """Exact C5 for a given step function h sampled on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives length 2n-1, lags -n+1..n-1
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_feasible(x, target_sum):
    """
    Project a vector onto the feasible set:
    0 <= x_i <= 1 and sum(x) = target_sum.
    Uses successive clipping and equal redistribution on the free set.
    """
    x = np.clip(x, 0.0, 1.0)
    # Simple iterative redistribution
    for _ in range(10):
        cur_sum = x.sum()

        if abs(deficit) < 1e-12:
            break
        # indices that are not on a bound

        if free.size == 0:
            # all variables are at bounds; distribute deficit uniformly and clip again
            delta = deficit / x.size
            x += delta

            continue

        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
