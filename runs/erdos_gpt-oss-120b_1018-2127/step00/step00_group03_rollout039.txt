import time
import numpy as np
import random

def _enforce_feasibility(h, target_sum):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        # put all mass on first coordinate
        h[0] = target_sum
        return np.clip(h, 0.0, 1.0)

    # Scale uniformly to hit the target sum
    h = h * (target_sum / s)
    # Small post‑scale violations are fixed by a single redistribution pass
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        # indices that are not at a bound can absorb the remaining mass
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0:
            # no free slot – push the first entry (it will stay in bounds)
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        else:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
    return h

def _c5_score(h):
    """Return the overlap constant C5 for a discretised step function."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretised) to minimise C5.
    Returns (list_of_h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------
    # 1. Determine problem size and initialise h
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 70                     # default size when no parent is supplied
        h0 = np.full(n, 0.5)       # uniform start satisfies the integral constraint

    target_sum = n / 2.0           # because ∫ h = 1  and  Δx = 2/n
    h_best = _enforce_feasibility(h0, target_sum)
    best_score = _c5_score(h_best)

    # ----------------------------------------------------------------------
    # 2. Simple stochastic hill‑climbing with occasional random restarts
    # ----------------------------------------------------------------------
    no_improve = 0
    iter_count = 0
    max_iter_without_improve = 2000

    while time.time() - start_time < budget_s * 0.95:
        # propose a local move: transfer a small amount between two points
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = np.random.uniform(-0.05, 0.05)   # amount moved from i to j
        h_candidate = h_best.copy()
        h_candidate[i] = np.clip(h_candidate[i] - delta, 0.0, 1.0)
        h_candidate[j] = np.clip(h_candidate[j] + delta, 0.0, 1.0)

        # re‑enforce the exact integral constraint
        h_candidate = _enforce_feasibility(h_candidate, target_sum)

        cand_score = _c5_score(h_candidate)

        if cand_score < best_score - 1e-9:
            h_best = h_candidate
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1

        # Random restart if stuck for many iterations
        if no_improve >= max_iter_without_improve:
            # generate a fresh random feasible vector
            raw = np.random.rand(n)
            raw = _enforce_feasibility(raw, target_sum)
            h_candidate = raw
            cand_score = _c5_score(h_candidate)
            if cand_score < best_score:
                h_best = h_candidate
                best_score = cand_score
            no_improve = 0

    # ----------------------------------------------------------------------
    # 3. Return the best discovered construction
    # ----------------------------------------------------------------------
    return h_best.tolist(), float(best_score), n
