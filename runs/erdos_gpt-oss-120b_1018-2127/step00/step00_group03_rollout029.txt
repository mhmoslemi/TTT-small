import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with discretization n points
    to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h on the grid.
    c5_bound : float
        Estimated C5 value (max overlap).
    n_points : int
        Number of discretization points used.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialise h -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: random feasible vector of length 70
        n = 70
        target = n / 2.0
        h0 = rng.random(n)
        h0 = h0 / h0.sum() * target
        h0 = np.clip(h0, 0.0, 1.0)
        # exact sum correction
        diff = target - h0.sum()
        while abs(diff) > 1e-12:
            idx = rng.integers(n)
            add = np.clip(diff, -h0[idx], 1.0 - h0[idx])
            h0[idx] += add
            diff = target - h0.sum()

    n = h0.size
    target_sum = n / 2.0
    # ensure feasibility of the supplied vector
    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # scale and then fix residual
        h0 = h0 * (target_sum / cur_sum)
        h0 = np.clip(h0, 0.0, 1.0)
        diff = target_sum - h0.sum()
        while abs(diff) > 1e-12:
            idx = rng.integers(n)
            add = np.clip(diff, -h0[idx], 1.0 - h0[idx])
            h0[idx] += add
            diff = target_sum - h0.sum()

    dx = 2.0 / n

    def c5_score(vec):
        """Compute max overlap C5 for a given vector."""
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return np.max(corr) * dx

    # ----- optimisation loop --------------------------------------------
    cur_h = h0.copy()
    cur_score = c5_score(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    max_iters = 5_000_000  # safety upper bound
    for it in range(max_iters):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # propose a mass transfer between two positions
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = c5_score(new_h)

        # accept if not worse (simple hill‑climb)
        if new_score <= cur_score:
            cur_h, cur_score = new_h, new_score
            if new_score < best_score:
                best_h, best_score = new_h.copy(), new_score

        # occasional random perturbation to escape plateaus
        if it % 10000 == 0 and rng.random() < 0.01:
            pert = rng.normal(scale=0.04, size=n)
            cand = cur_h + pert
            cand = np.clip(cand, 0.0, 1.0)
            # enforce exact sum
            diff = target_sum - cand.sum()
            cand += diff / n
            cand = np.clip(cand, 0.0, 1.0)
            diff = target_sum - cand.sum()
            if abs(diff) > 1e-12:
                cand[0] = np.clip(cand[0] + diff, 0.0, 1.0)
            cand_score = c5_score(cand)
            if cand_score <= cur_score:
                cur_h, cur_score = cand, cand_score
                if cand_score < best_score:
                    best_h, best_score = cand.copy(), cand_score

    # final evaluation via provided helper if present

        final_score = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_score = float(best_score)

    return best_h, final_score, n
