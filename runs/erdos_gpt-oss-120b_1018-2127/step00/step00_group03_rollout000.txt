import time
import numpy as np
import cvxpy as cp
import random

# Helper: project arbitrary vector onto the feasible set
def project_feasible(v, target_sum):
    n = v.shape[0]
    x = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(x - v))
    constraints = [cp.sum(x) == target_sum,
                   x >= 0,
                   x <= 1]
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    return x.value

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_feasible(n, target_sum, rng):
    v = rng.random(n)  # uniform [0,1)
    return project_feasible(v, target_sum)

def perturb(h, rng, max_delta=0.05):
    """Move a small amount of mass from one index to another while keeping bounds."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    delta = max_delta * rng.random()
    # adjust delta to respect bounds
    delta = min(delta, h[i])               # cannot remove more than present
    delta = min(delta, 1.0 - h[j])         # cannot exceed upper bound at destination
    if delta <= 0:
        return h.copy()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h
