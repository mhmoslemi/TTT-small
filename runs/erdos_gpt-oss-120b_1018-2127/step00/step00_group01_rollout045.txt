def _project_feasible(x, target_sum):

    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if np.isclose(s, target_sum):
        return x
    # Simple iterative redistribution
    diff = target_sum - s
    # Indices that can move in the needed direction
    if diff > 0:
        idx = np.where(x < 1.0)[0]
        capacity = 1.0 - x[idx]
    else:
        idx = np.where(x > 0.0)[0]
        capacity = x[idx]
    if capacity.size == 0:
        return x
    proportion = capacity / capacity.sum()
    delta = proportion * diff
    x[idx] += delta
    # Clip again to eliminate tiny numerical overshoots
    return np.clip(x, 0.0, 1.0)
