def repair_h(h):
    n = len(h)
    target = n/2.0
    h = np.clip(h, 0.0, 1.0)
    if h.sum() == 0:
        h[:] = target / n
        return h
    for _ in range(3):
        s = target / h.sum()
        h = np.clip(h * s, 0.0, 1.0)

import time
import math
import random
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _repair_h(h: np.ndarray) -> np.ndarray:
    """Project onto [0,1] and enforce sum(h)=n/2."""
    n = h.shape[0]
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)

    if h.sum() == 0.0:
        return np.full(n, target / n)

    # simple scaling loop – usually converges in a few passes
    for _ in range(3):
        scale = target / h.sum()
        h = np.clip(h * scale, 0.0, 1.0)

    # tiny residual correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            h[:] = target / n
    return h

def _simulated_anneal(initial_h: np.ndarray, time_limit: float):
    """Local search with a basic simulated‑annealing schedule."""
    start = time.time()
    h = _repair_h(initial_h.copy())
    cur_score = _compute_c5(h)
    best_h, best_score = h.copy(), cur_score

    n = h.shape[0]
    T = 0.02                # initial temperature
    T_min = 1e-6
    decay = 0.9995

    while time.time() - start < time_limit:
        #--- perturbation -------------------------------------------------
        cand = h.copy()
        k = max(1, n // 20)                     # ~5 % of entries
        idx = np.random.choice(n, k, replace=False)
        cand[idx] += np.random.randn(k) * 0.1   # Gaussian step
        cand = _repair_h(cand)
        cand_score = _compute_c5(cand)

        #--- record best --------------------------------------------------
        if cand_score < best_score:
            best_score, best_h = cand_score, cand.copy()

        #--- acceptance ----------------------------------------------------
        if cand_score < cur_score:
            h, cur_score = cand, cand_score
        else:
            delta = cand_score - cur_score
            if T > 0.0 and random.random() < math.exp(-delta / T):
                h, cur_score = cand, cand_score

        #--- temperature decay --------------------------------------------
        T = max(T * decay, T_min)

        #--- early stop if we already beat the target --------------------
        if best_score <= 0.3808:
            break

    return best_h, best_score

def _upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid, followed by projection."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _repair_h(h_new)
