import time
import numpy as np

def _objective(h, dx):
    """Return max overlap C5 for a given h and grid spacing dx."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _feasible_step(h, i, j, delta):
    """
    Propose a feasible update: h[i] -= delta, h[j] += delta.
    Returns a new copy if the move respects bounds, else None.
    """
    if delta <= 0:
        return None
    hi, hj = h[i], h[j]
    if hi - delta < 0.0 or hj + delta > 1.0:
        return None
    new_h = h.copy()
    new_h[i] = hi - delta
    new_h[j] = hj + delta
    return new_h

            diff = target_sum - cur_sum

    except NameError:

    dx = 2.0 / n
    best_h = h.copy()
    best_val = _objective(best_h, dx)

    # ----------------------------------------------------------------------

    # Parameters
    max_iters = 200000          # safety cap

    while True:

        if now - start_time > budget_s:

        # propose a delta proportional to the smaller slack at the two positions
        slack_i = h[i]          # how much we can decrease

        max_delta = min(slack_i, slack_j)

        delta = np.random.rand() * max_delta * 0.2
