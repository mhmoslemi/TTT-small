import time
import numpy as np
from scipy.optimize import minimize

def _dx(n):
    return 2.0 / n

def _compute_score(h):
    """Maximum overlap integral for step vector h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * _dx(n)

def _is_feasible(h, n, atol=1e-6):
    if h.min() < -1e-9 or h.max() > 1 + 1e-9:
        return False
    return abs(h.sum() - n / 2) <= atol

def _project_feasible(h, n):
    """Simple projection onto box [0,1] and sum=n/2 via scaling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    cur = h.sum()
    # scale uniformly, then clip again if needed
    if cur > 0:
        scale = target / cur
        h *= scale
    else:
        h[:] = target / n
    # final clip & renorm (few iterations)
    for _ in range(3):
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - target) < 1e-10:
            break
        diff = target - cur
        # distribute diff uniformly respecting bounds
        free = (h > 0) & (h < 1)
        if free.any():
            add = diff / free.sum()
            h[free] += add
        else:
            # all at bounds, fallback to uniform fill
            h[:] = target / n
    return h

def _transfer(old_h, old_n, new_n):
    """Linear interpolation of a step function from old_n to new_n points."""
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    return _project_feasible(new_h, new_n)

def _optimize(initial_h, n, rng, max_iter=800):
    """Run SLSQP to minimise the max overlap."""
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}
    opts = {'maxiter': max_iter, 'ftol': 1e-8, 'disp': False}

    best = {'x': None, 'fun': np.inf, 'success': False}
    # primary run from given initial
    for start in [initial_h]:
        res = minimize(_compute_score, start, method='SLSQP',
                       bounds=bounds, constraints=cons, options=opts)
        if res.success and res.fun < best['fun']:
            best.update({'x': res.x, 'fun': res.fun, 'success': True})
    # a few random restarts
    for _ in range(2):
        rand = rng.dirichlet(np.ones(n)) * (n / 2.0)
        rand = _project_feasible(rand, n)
        res = minimize(_compute_score, rand, method='SLSQP',
                       bounds=bounds, constraints=cons, options=opts)
        if res.success and res.fun < best['fun']:
            best.update({'x': res.x, 'fun': res.fun, 'success': True})
