import time
import numpy as np

def _c5_of(h):
    """Return C5 for a discrete step function h (values in [0,1])."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), (1.0 - h).astype(float), mode="full")
    return dx * np.max(corr)


def _random_binary(n, rng):
    """Binary vector of length n with exactly n/2 ones (n assumed even)."""
    ones = n // 2
    h = np.zeros(n, dtype=float)
    h[:ones] = 1.0
    rng.shuffle(h)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that minimizes
    the Erdős overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points) : tuple
        h_values – numpy array of length n_points with values in [0,1]
        c5_bound – the achieved C5 value (float)
        n_points – the discretisation size used (int)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s * 0.95  # keep a safety margin

    # Primary resolution – a modest size that still allows many iterations.
    n = 200  # even number, gives granularity 0.01 in the integral
    h = _random_binary(n, rng)
    best_h = h.copy()
    best_score = _c5_of(best_h)

    # If a parent construction is supplied, evaluate it and possibly adopt it.
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.ndim == 1 and np.all((parent >= 0) & (parent <= 1)):
            # Project parent onto the required sum if needed (tiny numerical drift)
            target_sum = n / 2
            cur_sum = parent.sum()
            if not np.isclose(cur_sum, target_sum, atol=1e-8):
                # Rescale while staying in bounds – simple linear shift and clip.
                delta = (target_sum - cur_sum) / n
                parent = np.clip(parent + delta, 0.0, 1.0)
                # Finally enforce exact sum by adjusting the first element.
                parent[0] += target_sum - parent.sum()
            parent_score = _c5_of(parent)
            if parent_score < best_score:
                best_h = parent.copy()
                best_score = parent_score
                n = parent.size

    # Main stochastic hill‑climbing loop.
    # We repeatedly propose swaps of a 1↔0 pair (keeping the total sum fixed).
    # If the swap improves the maximal overlap we keep it, otherwise we reject.
    # Occasionally we reshuffle globally to escape plateaus.
    swap_attempts = 0
    while time.time() - start_time < time_limit:
        i, j = rng.integers(0, n, size=2)
        if best_h[i] == best_h[j]:
            continue  # need opposite values to preserve the sum constraint

        # Perform the swap
        best_h[i], best_h[j] = best_h[j], best_h[i]
        new_score = _c5_of(best_h)

        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert the swap
            best_h[i], best_h[j] = best_h[j], best_h[i]

        swap_attempts += 1

        # Every few thousand attempts, randomise the ordering to avoid stagnation.
        if swap_attempts % 5000 == 0:
            rng.shuffle(best_h)

        if best_score <= 0.3808:
