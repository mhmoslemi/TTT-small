def project_feasible(y, S):
    x = np.clip(y, 0.0, 1.0)
    # compute sum
    total = x.sum()
    if abs(total - S) < 1e-9:
        return x
    # water-filling
    free = np.arange(len(y))
    while True:
        k = len(free)
        if k == 0:
            break
        delta = (S - x.sum()) / k
        x[free] += delta
        # check bounds violations
        over = x > 1.0
        under = x < 0.0
        violated = np.where(over | under)[0]
        if len(violated) == 0:
            break
        # set violated to bound and remove from free
        for idx in violated:
            if x[idx] > 1.0:
                x[idx] = 1.0
            else:
                x[idx] = 0.0
        free = np.setdiff1d(free, violated, assume_unique=True)
    # final clip
    return np.clip(x, 0.0, 1.0)

import time
import random
import numpy as np

# ------------------------------------------------------------
# Helper: project onto the box simplex {0<=x<=1, sum x = S}
def project_feasible(y, S):
    x = np.clip(y, 0.0, 1.0)          # start with box projection
    total = x.sum()
    if abs(total - S) < 1e-12:
        return x
    # water‑filling on the free (non‑bound) indices
    free = np.arange(len(y))
    while True:
        k = len(free)
        if k == 0:
            break
        delta = (S - x.sum()) / k
        x[free] += delta
        # detect newly violated bounds
        over = x > 1.0
        under = x < 0.0
        viol = np.where(over | under)[0]
        if viol.size == 0:
            break
        # fix them and remove from free set
        for idx in viol:
            x[idx] = 1.0 if over[idx] else 0.0
        free = np.setdiff1d(free, viol, assume_unique=True)
    return np.clip(x, 0.0, 1.0)

def compute_gradient(h):
    n = len(h)
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argmax(corr)                 # index of the worst shift
    lag = idx - (n - 1)                    # actual integer shift (may be negative)

    grad = np.full_like(h, dx)            # derivative of the constant sum part

    # subtract contribution of the quadratic term for this lag
    if lag >= 0:
        grad[: n - lag] -= dx * h[lag:]
        grad[lag:]     -= dx * h[: n - lag]
    else:
        l = -lag
        grad[l:]       -= dx * h[: n - l]
        grad[: n - l]  -= dx * h[l:]

    return grad
