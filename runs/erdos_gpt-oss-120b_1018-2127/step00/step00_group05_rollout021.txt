import time
import numpy as np

def _project_bounds_sum(x: np.ndarray, target: float, max_iter: int = 20) -> np.ndarray:
    """
    Project vector x onto the box [0,1]^n and enforce sum(x) = target.
    Simple iterative method: clip, then redistribute the sum difference
    among non‑boundary entries.
    """
    for _ in range(max_iter):
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:      # cannot adjust further
            break
        x[free] += diff / cnt
    return np.clip(x, 0.0, 1.0)

def _random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with the prescribed sum."""
    x = rng.random(n)
    x = _project_bounds_sum(x, target)
    return x

def _c5_score(h: np.ndarray, dx: float) -> float:
    """Compute the Erdős C5 value for a discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the C5 overlap bound.

    Returns
    -------
    (h_values, c5_bound, n_points) :
        h_values – numpy array of shape (n_points,) with values in [0,1];
        c5_bound – the achieved maximum overlap integral;
        n_points – number of discretisation points.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n_points = init.size
    except NameError:
        # No parent construction supplied – use a modest default size
        n_points = 120
        init = None

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because ∫ h = 1  ⇒  sum(h)*dx = 1

    # Initialise best solution
    if init is not None:
        best_h = _project_bounds_sum(init.copy(), target_sum)
    else:
        best_h = _random_feasible(n_points, target_sum, rng)

    best_c5 = _c5_score(best_h, dx)

    # Simulated‑annealing‑style search
    start_time = time.time()
    max_time = budget_s * 0.95  # safety margin
    temperature = 0.1
    decay = 0.9995
    sigma = 0.03  # perturbation scale

    # Helper for a single proposal
    def propose(current):
        proposal = current + rng.normal(scale=sigma, size=n_points)
        proposal = _project_bounds_sum(proposal, target_sum)
        return proposal

    iteration = 0
    while True:
        # time check
        if time.time() - start_time > max_time:
            break

        cand_h = propose(best_h)
        cand_c5 = _c5_score(cand_h, dx)

        # accept if better or by Metropolis criterion
        if cand_c5 < best_c5 or rng.random() < np.exp((best_c5 - cand_c5) / max(temperature, 1e-12)):
            best_h, best_c5 = cand_h, cand_c5

        # cool down

        iteration += 1
