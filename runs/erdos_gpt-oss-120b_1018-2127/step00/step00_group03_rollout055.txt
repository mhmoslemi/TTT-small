import time
import numpy as np

def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    # np.correlate with mode='full' gives sums over overlapping indices
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] with values in [0,1] and total integral 1.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # Build initial feasible vector
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float32).copy()
        n = h.size
        # fix tiny numerical drift to satisfy exact sum constraint
        target_sum = n / 2.0
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) > 1e-8:
            diff = target_sum - cur_sum
            # distribute diff uniformly while respecting bounds
            for idx in np.argsort(h):
                if diff == 0:
                    break
                if diff > 0:
                    add = min(diff, 1.0 - h[idx])
                else:
                    add = max(diff, -h[idx])
                h[idx] += add
                diff -= add
    else:
        # Default: n = 70, half ones, half zeros, shuffled
        n = 70
        h = np.zeros(n, dtype=np.float32)
        h[: n // 2] = 1.0
        np.random.shuffle(h)

    dx = 2.0 / n

    # ----------------------------------------------------------------------
    # Prepare for binary‑swap simulated annealing
    # ----------------------------------------------------------------------
    # Ensure binary feasibility (values exactly 0 or 1) – any fractional entries
    # are rounded while preserving the sum.
    if not np.all(np.isin(h, [0.0, 1.0])):
        # simple projection: make the largest n/2 entries 1, rest 0
        order = np.argsort(h)[::-1]
        h[:] = 0.0
        h[order[: n // 2]] = 1.0

    # Lists of indices with 1 and 0 for O(1) sampling
    ones = np.flatnonzero(h == 1.0).tolist()
    zeros = np.flatnonzero(h == 0.0).tolist()

    cur_c5 = _compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5

    # Simulated‑annealing schedule
    max_iters = 5_000_000  # hard cap; loop will break on time
    for it in range(max_iters):
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break

        # temperature decreases linearly to near zero
        t = max(0.001, 0.1 * (1.0 - it / max_iters))

        # propose swapping a 1 and a 0
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # perform swap
        h[i] = 0.0
        h[j] = 1.0

        new_c5 = _compute_c5(h, dx)
        delta = new_c5 - cur_c5

        accept = delta < 0 or np.random.rand() < np.exp(-delta / t)
        if accept:
            # commit swap in index lists
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0

    # ----------------------------------------------------------------------
    # Return result
    # ----------------------------------------------------------------------
    return (best_h.tolist(), best_c5, best_h.size)
