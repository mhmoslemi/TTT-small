import time
import math
import numpy as np
from scipy import optimize  # imported to satisfy allowed libraries

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x) (1 - h(x+k)) dx approximated on the discrete grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h = 1 that minimizes
    C₅ = max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # 1. Initialise h --------------------------------------------------- #
    # ------------------------------------------------------------------ #
    try:
        # use the parent construction if present
        h = np.asarray(initial_h_values, dtype=float).copy()
    except NameError:
        # fallback: simple half‑ones / half‑zeros with the correct integral
        n_points = 200
        h = np.zeros(n_points, dtype=float)
        h[: n_points // 2] = 1.0

    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0  # because integral = sum * dx = 1  → sum = 1/dx = n/2

    # enforce constraints exactly
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-12:
        h *= target_sum / current_sum

    # ------------------------------------------------------------------ #
    # 2. Simulated‑annealing loop --------------------------------------- #
    # ------------------------------------------------------------------ #
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # annealing schedule
    temp = 0.02
    temp_decay = 0.99995

    # move magnitude (max absolute transfer between two bins)
    max_transfer = 0.08

    iter_count = 0
    while time.time() - start_time < budget_s * 0.94:
        # propose a transfer of mass between two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # feasible transfer range respecting bounds
        max_from_i = min(max_transfer, cur_h[i])          # can take at most this from i
        max_to_j   = min(max_transfer, 1.0 - cur_h[j])    # can add at most this to j
        max_pos = min(max_from_i, max_to_j)

        max_from_j = min(max_transfer, cur_h[j])
        max_to_i   = min(max_transfer, 1.0 - cur_h[i])
        max_neg = min(max_from_j, max_to_i)

        # draw a signed delta
        if max_pos == 0 and max_neg == 0:
            continue
        if max_pos > 0 and max_neg > 0:
            delta = (rng.random() * (max_pos + max_neg)) - max_neg
        elif max_pos > 0:
            delta = rng.random() * max_pos
        else:
            delta = -rng.random() * max_neg

        # apply transfer
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        # numerical safety: clip (should be unnecessary but guards rounding)
        new_h = np.clip(new_h, 0.0, 1.0)

        # evaluate
        new_c5 = _compute_c5(new_h)

        # acceptance test
        improve = new_c5 < cur_c5
        if improve or rng.random() < math.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h, new_c5

                cur_h, cur_c5 = perm_h, perm_c5

    # ------------------------------------------------------------------ #
    # 3. Return result --------------------------------------------------- #
    # ------------------------------------------------------------------ #
    return best_h, float(best_c5), n
