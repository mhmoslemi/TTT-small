import time
import numpy as np

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def make_initial(n):
    """Create an initial h of length n.
    If a parent vector `initial_h_values` exists, interpolate it;
    otherwise use a simple half‑filled pattern."""
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        m = parent.size
        # map parent samples onto new grid via linear interpolation
        xs_parent = np.linspace(0.0, 2.0, m, endpoint=False)
        xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(xs_new, xs_parent, parent)
        # enforce exact integral constraint by a small uniform shift
        total = h.sum()
        target = n / 2.0
        h += (target - total) / n
        np.clip(h, 0.0, 1.0, out=h)
        # re‑balance if clipping broke the sum
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # distribute the remaining mass uniformly on interior points
            mask = (h > 0.0) & (h < 1.0)
            if mask.any():
                h[mask] += diff / mask.sum()
                np.clip(h, 0.0, 1.0, out=h)
    except NameError:
        # no parent supplied – start from a simple 0/1 pattern with half mass
        h = np.zeros(n, dtype=float)
        half = n // 2
        h[:half] = 1.0
        # random shuffle to avoid symmetry
        np.random.default_rng(0).shuffle(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # choose a moderate resolution – larger n gives smoother functions but costs more
    n = 240  # 240 samples → dx≈0.00833, fast enough for many evaluations
    h = make_initial(n)

    # ensure the integral constraint exactly
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h += (target_sum - cur_sum) / n
        np.clip(h, 0.0, 1.0, out=h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # simple stochastic hill‑climb with occasional uphill moves (simulated annealing)
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.98:  # leave a tiny margin
            break
        iteration += 1

        # propose a transfer between two points
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if rng.random() < 0.5:
            # move mass from i to j
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 1e-12:
                continue
            delta = rng.random() * max_transfer
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # move mass from j to i
            max_transfer = min(h[j], 1.0 - h[i])
            if max_transfer <= 1e-12:
                continue
            delta = rng.random() * max_transfer
            h_new = h.copy()
            h_new[j] -= delta
            h_new[i] += delta

        # evaluate
        c5_new = compute_c5(h_new)

        # acceptance rule
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # temperature schedule: slowly decay acceptance probability
            temperature = max(0.001, 0.05 * (1.0 - elapsed / budget_s))
            if rng.random() < np.exp((best_c5 - c5_new) / temperature):
                h = h_new

    # final sanity clamp and renormalize (tiny numerical drift possible)
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        best_h += (target_sum - cur_sum) / n
        np.clip(best_h, 0.0, 1.0, out=best_h)

    return best_h.tolist(), float(best_c5), n
