import numpy as np, time, math, random

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair(h):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        # fallback to uniform distribution
        h[:] = 0.5
        total = h.sum()
    h *= (h.shape[0] / 2.0) / total
    # one more clip in case scaling pushed a tiny amount out of bounds
    return np.clip(h, 0.0, 1.0)

def _resample(h_old, n_new):
    n_old = h_old.shape[0]
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return _repair(h_new)

def _anneal(h0, time_limit):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_score = _compute_c5(h)
    score = best_score
    n = h.shape[0]
    T = best_score * 0.5 + 1e-3
    step_scale = 0.2
    it = 0
    while True:
        now = time.time()
        if now - start > time_limit:
            break
        # generate perturbation
        step = step_scale * (T / (best_score + 1e-8))
        perturb = (np.random.rand(n) - 0.5) * step
        h_candidate = _repair(h + perturb)
        cand_score = _compute_c5(h_candidate)
        # acceptance
        if cand_score < score or random.random() < math.exp(-(cand_score - score) / (T + 1e-12)):
            h = h_candidate
            score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_h = h_candidate.copy()
        # cooling
        T *= 0.9995
        it += 1
        # occasional restart to avoid stagnation
        if it % 5000 == 0 and random.random() < 0.2:
            h = _repair(np.random.rand(n))
            score = _compute_c5(h)
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):

    best_overall_score = float('inf')
