import time, math, numpy as np
from scipy.optimize import minimize

# helpers --------------------------------------------------------------
def _dx(n):
    return 2.0 / n

def _c5_of(h):
    n = h.size
    dx = _dx(n)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def _project_feasible(x, target_sum):
    """Project x onto [0,1]^n with sum == target_sum (simple water‑filling)."""
    n = x.size
    # start by clipping
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    if abs(cur - target_sum) < 1e-12:
        return x
    # indices that can still move
    free = np.ones(n, dtype=bool)
    while True:
        deficit = target_sum - cur
        if abs(deficit) < 1e-12:
            break
        movable = free.sum()
        if movable == 0:
            break
        delta = deficit / movable
        # apply delta but respect bounds
        add = np.where(free, delta, 0.0)
        x_new = x + add
        # detect violations
        over = x_new > 1.0
        under = x_new < 0.0
        if over.any():
            # clamp those to 1 and mark as fixed
            idx = np.where(over)[0]
            x[idx] = 1.0
            free[idx] = False
        elif under.any():
            idx = np.where(under)[0]
            x[idx] = 0.0
            free[idx] = False
        else:
            x = x_new
        cur = x.sum()
    return x

def _random_feasible(n, rng):
    """Generate a random vector in [0,1] with sum == n/2."""
    target = n / 2.0
    # start with uniform draws then project

    return _project_feasible(x, target)

def _optimize_once(h0, rng, maxiter=200):
