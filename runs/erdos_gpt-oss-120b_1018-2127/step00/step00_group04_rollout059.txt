import time
import numpy as np

def _evaluate(h, dx):
    """Return max overlap integral for sampled step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_pair_delta(h, i, j, step):
    """Compute a feasible delta to add to h[i] and subtract from h[j]."""
    lo = max(-h[i], h[j] - 1.0, -step)
    hi = min(1.0 - h[i], h[j], step)
    if lo >= hi:
        return None
    return np.random.uniform(lo, hi)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a low‑C₅ step function.

    Returns
    -------
    h_values : np.ndarray
        The sampled function values on a uniform grid of length n_points.
    c5_bound : float
        The achieved upper bound C₅ = max_k ∫ h(x)(1‑h(x+k))dx .
    n_points : int
        Number of uniform samples in [0,2].
    """
    start_time = time.time()
    np.random.seed(seed)

    # ----- problem size -------------------------------------------------
    # use a modest resolution (fast evaluation) but enough to beat the record
    n_points = 300                # can be changed; kept ≤1000 for speed
    dx = 2.0 / n_points
    desired_sum = n_points / 2.0  # because ∫ h = 1

    # ----- initial feasible vector (binary, exact sum) -------------------
    h = np.zeros(n_points, dtype=np.float64)
    ones_idx = np.random.choice(n_points, int(desired_sum), replace=False)
    h[ones_idx] = 1.0

    best_h = h.copy()
    best_val = _evaluate(best_h, dx)

    # ----- main stochastic hill‑climbing loop ---------------------------
    step = 0.2                     # max change per update
    it = 0
    while True:
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break

        # occasionally reduce step size to refine
        if it % 5000 == 0 and it > 0:
            step = max(0.01, step * 0.9)

        # propose a move: pick two distinct indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = _random_pair_delta(h, i, j, step)
        if delta is None:
            it += 1
            continue

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # evaluate
        val = _evaluate(h_new, dx)

        # accept if improvement
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            h = h_new
        else:
            # occasional uphill move (simulated‑annealing style)
            prob = np.exp(-(val - best_val) / (0.001 + 0.01 * (it / 10000.0)))
            if np.random.rand() < prob:
                h = h_new

        it += 1

    # final polishing: small random swaps without worsening
    for _ in range(2000):
        if time.time() - start_time > budget_s * 0.98:
            break
        i, j = np.random.choice(n_points, 2, replace=False)
        if h[i] == h[j]:
            continue
        # swap values
        h_swapped = h.copy()
        h_swapped[i], h_swapped[j] = h_swapped[j], h_swapped[i]
        val = _evaluate(h_swapped, dx)
        if val < best_val:
            best_val = val
            best_h = h_swapped.copy()
            h = h_swapped

    return best_h, best_val, n_points
