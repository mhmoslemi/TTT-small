import time
import numpy as np

def _evaluate(h, dx):
    # correlation of h with (1-h), full mode, then max * dx
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _random_binary(n):
    arr = np.zeros(n, dtype=np.float64)
    arr[: n // 2] = 1.0
    np.random.shuffle(arr)
    return arr

def _hill_climb(h, dx, cur_score):
    n = h.shape[0]
    improved = True
    while improved:
        improved = False
        # try random swaps; stop after a full pass without improvement
        for _ in range(n):
            i, j = np.random.randint(n, size=2)
            if h[i] == h[j]:
                continue
            # swap
            h[i], h[j] = h[j], h[i]
            new_score = _evaluate(h, dx)
            if new_score < cur_score:
                cur_score = new_score
                improved = True
                # keep the swap and break to restart scanning
                break
            else:
                # revert
                h[i], h[j] = h[j], h[i]
    return h, cur_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points)
    where h_values is a numpy array on a uniform grid over [0,2],
    c5_bound is the achieved upper bound for C_5, and n_points is the
    size of the discretisation.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Choose a modest grid size (even) that balances resolution and speed.
    n_points = 200  # can be changed; must be even for exact integral constraint

    dx = 2.0 / n_points

    # Initialise: use provided initial_h_values if it matches our grid,
    # otherwise start from a random feasible binary vector.
    try:
        init = initial_h_values  # supplied by harness if present
        if init.shape[0] == n_points and np.isclose(np.sum(init) * dx, 1.0):
            h_curr = init.astype(np.float64).copy()
        else:
            raise ValueError
    except Exception:
        h_curr = _random_binary(n_points)

    cur_score = _evaluate(h_curr, dx)
    best_h = h_curr.copy()
    best_score = cur_score

    # Simulated annealing parameters
    temp_initial = 1.0
    temp_final = 1e-4

    # Main SA loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        # linear temperature schedule
        t = 1.0 - elapsed / (budget_s * 0.95)
        temperature = max(temp_final, temp_initial * t)

        # propose a swap of a 1 and a 0 (preserves sum)
        i, j = np.random.randint(n_points, size=2)
        if h_curr[i] == h_curr[j]:
            continue
        h_new = h_curr.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_score = _evaluate(h_new, dx)

        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            h_curr = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score

    # Final deterministic polishing with hill climbing
    best_h, best_score = _hill_climb(best_h, dx, best_score)

    return best_h, best_score, n_points
