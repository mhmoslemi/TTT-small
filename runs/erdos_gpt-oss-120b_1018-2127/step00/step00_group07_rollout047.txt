def project_h(h):
    # clip
    h = np.clip(h, 0.0, 1.0)
    target = len(h)/2.0
    # iterative water-filling
    while True:
        s = h.sum()
        if abs(s - target) < 1e-9:
            break
        diff = target - s
        free = (h > 0) & (h < 1)
        if not free.any():
            # distribute to all equally
            h += diff/len(h)
            h = np.clip(h,0,1)
            break
        # adjust free entries uniformly
        adjust = diff / free.sum()
        h[free] += adjust
        # clip again
        h = np.clip(h,0,1)
    return h

import time
import math
import random
import numpy as np

def _project_h(vec):
    """Project vector onto [0,1] bounds with sum == len(vec)/2."""
    n = vec.shape[0]
    target = n / 2.0
    v = np.clip(vec, 0.0, 1.0)
    # water-filling style projection
    for _ in range(10):  # enough for convergence in practice
        s = v.sum()
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        free = (v > 0.0) & (v < 1.0)
        if not free.any():
            # all at bounds, spread uniformly and re‑clip
            v += diff / n
            v = np.clip(v, 0.0, 1.0)
            break
        adjust = diff / free.sum()
        v[free] += adjust
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_score(h):
    """Return the C5 value for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C5 bound.

    Returns
    -------
    h_values : list of float
        Optimised samples of h.
    c5_bound : float
        Achieved C5 value.
    n_points : int
        Number of discretisation points.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------------ #
    # Initialise problem size and starting vector
    # ------------------------------------------------------------------ #
    if 'initial_h_values' in globals():
        h_cur = np.array(initial_h_values, dtype=float)
        n = h_cur.shape[0]
    else:
        n = 200                       # fallback size (still <1000)
        h_cur = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
    h_cur = _project_h(h_cur)

    best_h = h_cur.copy()
    best_score = _c5_score(best_h)

    # ------------------------------------------------------------------ #
    # Simulated‑annealing parameters (tuned for the given time budget)
    # ------------------------------------------------------------------ #
    T0 = 4.0                         # initial temperature
    T = T0
    cooling = 0.996
    perturb_scale = 0.07            # stddev of Gaussian perturbation
    iter_cnt = 0
    no_improve = 0
    restart_interval = 5000        # force a random restart if stuck

    # ------------------------------------------------------------------ #
    # Main optimisation loop
    # ------------------------------------------------------------------ #
    while time.time() - start_time < budget_s * 0.93:
        # propose a new candidate
        proposal = best_h + np.random.randn(n) * perturb_scale
        proposal = _project_h(proposal)

        score_prop = _c5_score(proposal)

        # acceptance rule
        if score_prop < best_score:
            best_h, best_score = proposal, score_prop
            no_improve = 0
        else:
            delta = best_score - score_prop
            if random.random() < math.exp(delta / max(T, 1e-12)):
                best_h, best_score = proposal, score_prop
                no_improve = 0
            else:
                no_improve += 1

        # temperature update
        T *= cooling
        iter_cnt += 1

        # occasional restart to escape deep plateaus
        if no_improve >= restart_interval:
            # generate a fresh feasible vector (random half‑filled)
            rand_vec = np.random.rand(n)
            rand_vec = _project_h(rand_vec)
            rand_score = _c5_score(rand_vec)
            if rand_score < best_score:
                best_h, best_score = rand_vec, rand_score
                no_improve = 0
            else:
                # keep the current best but reset temperature
                T = T0
                no_improve = 0

    # ------------------------------------------------------------------ #
    # Return results
    # ------------------------------------------------------------------ #
    return (best_h.tolist(), float(best_score), int(n))
