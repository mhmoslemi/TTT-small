import time
import numpy as np
import random
from scipy.optimize import minimize

def _compute_c5(h, dx):
    """Return max overlap integral for discretized h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _project_feasible(h, target_sum):
    """Clip to [0,1] and enforce sum == target_sum by simple redistribution."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    # simple scaling while staying in bounds
    if s != 0:
        h *= target_sum / s
    else:
        # fallback: put mass in first entries
        h[:int(np.round(target_sum))] = 1.0
    # iterative refinement to respect bounds
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target_sum
        if abs(diff) < 1e-12:
            break
        h -= diff / len(h)
    # final tiny fix
    diff = h.sum() - target_sum
    if abs(diff) > 1e-12:
        for i in range(len(h)):
            if 0.0 <= h[i] - diff <= 1.0:
                h[i] -= diff
                break
    return h

    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    # ------------------------------------------------------------

    eps = 0.04            # typical amount transferred between two coordinates
