import time
import numpy as np

def _project_capped_simplex(v, target_sum, max_iter=100, tol=1e-9):
    """Project vector v onto the set {0<=x_i<=1, sum x_i = target_sum}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if abs(s - target_sum) <= tol:
            return w
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _evaluate(h):
    """Return C5 value for a candidate h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation length = 2*n-1, center at lag n-1 (zero shift)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _neighbor(h, step_scale, rng):
    """Generate a nearby feasible vector."""
    n = h.shape[0]
    i = rng.integers(0, n)
    delta = rng.normal(scale=step_scale)
    new = h.copy()
    new[i] = np.clip(new[i] + delta, 0.0, 1.0)
    # re‑project to satisfy the sum constraint
    target = n / 2.0
    return _project_capped_simplex(new, target)

    n_points : int

    """
    start_time = time.time()

    # Ensure feasibility of the starting point

    # Simulated‑annealing parameters
