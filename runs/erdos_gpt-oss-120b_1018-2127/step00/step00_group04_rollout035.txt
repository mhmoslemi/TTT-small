import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper to evaluate the max overlap C5 for a given h vector (dx factor included)
def max_overlap(h):
    """Return the C5 value for step function defined by h (array of samples)."""
    N = h.size
    dx = 2.0 / N
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)
