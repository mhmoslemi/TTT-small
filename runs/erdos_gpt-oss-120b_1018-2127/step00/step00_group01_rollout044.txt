import numpy as np
import time

def _compute_c5(h):
    """Compute the C5 value for a discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
