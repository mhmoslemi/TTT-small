import time
import numpy as np

def _project_to_box_simplex(v, target, max_iter=10):
    """Project v onto the box [0,1] with sum equal to target (L2 optimal)."""
    w = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = w.sum()
        if abs(s - target) < 1e-12:
            break
        diff = s - target
        free = (w > 0.0) & (w < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # all variables at a bound; distribute diff uniformly then clip
            w -= diff / len(w)
            w = np.clip(w, 0.0, 1.0)
            continue
        w[free] -= diff / n_free
        w = np.clip(w, 0.0, 1.0)
    return w

def _c5_of_h(h, dx):
    """Compute the overlap constant C5 for a given step vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with integral 1 to minimise the
    Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimised vector of length n_points with values in [0,1] and sum=n_points/2.
    c5_bound : float
        The corresponding C5 value.
    n_points : int
        Number of discretisation points used.
    """
    np.random.seed(seed)

    # ---- parameters ---------------------------------------------------------
    n_points = int(kwargs.get('n_points', 100))          # default resolution
    dx = 2.0 / n_points
    target_sum = n_points / 2.0                           # integral constraint

    # ---- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # linear interpolation to desired resolution
        x_old = np.linspace(0.0, 2.0, len(init), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_old, init)
    else:
        h0 = np.full(n_points, target_sum / n_points)

    h0 = _project_to_box_simplex(h0, target_sum)

    best_h = h0.copy()
    best_score = _c5_of_h(best_h, dx)

    # ---- optimisation loop --------------------------------------------------
    t_start = time.time()
    it = 0
    max_iter = 200000
    sigma = 0.12                      # initial mutation scale

    while it < max_iter and (time.time() - t_start) < (budget_s - 0.5):
        # propose a neighbour
        cand = best_h + np.random.normal(scale=sigma, size=n_points)
        cand = _project_to_box_simplex(cand, target_sum)

        score = _c5_of_h(cand, dx)

        if score < best_score:
            best_h, best_score = cand, score
            sigma = max(0.01, sigma * 0.98)   # tighten step size on success
        else:
            # occasionally explore with a larger step
            if it % 5000 == 0:
                sigma = min(0.25, sigma * 1.07)

        # random restart if stuck for long
        if it % 30000 == 0 and it > 0:
            restart = np.random.rand(n_points)
            restart = _project_to_box_simplex(restart, target_sum)
            restart_score = _c5_of_h(restart, dx)
            if restart_score < best_score:
                best_h, best_score = restart, restart_score

        it += 1

    return best_h, best_score, n_points
