import time
import numpy as np

def evaluate_c5(h):
    """Return the C5 value for a step function sampled as array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def hill_climb_binary(h, max_iter, time_deadline):
    """Improve a binary (0/1) vector h by swapping 0/1 entries."""
    n = h.shape[0]
    ones = int(round(n / 2))
    # ensure feasibility
    if h.sum() != ones:
        # project to correct number of ones
        idx = np.argsort(h)[::-1]
        h[:] = 0.0
        h[idx[:ones]] = 1.0

    best = h.copy()
    best_score = evaluate_c5(best)

    rng = np.random.default_rng(12345)

    for _ in range(max_iter):
        if time.time() >= time_deadline:
            break
        # pick a random 1 and a random 0 to swap
        ones_idx = np.where(best == 1)[0]
        zeros_idx = np.where(best == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i1 = rng.choice(ones_idx)
        i0 = rng.choice(zeros_idx)

        cand = best.copy()
        cand[i1] = 0.0
        cand[i0] = 1.0
        cand_score = evaluate_c5(cand)

        if cand_score < best_score - 1e-9:
            best = cand
            best_score = cand_score

    return best, best_score
