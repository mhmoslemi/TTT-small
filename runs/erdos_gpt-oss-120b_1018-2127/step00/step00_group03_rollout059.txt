import time
import numpy as np

def _compute_c5(h):
    """Return the overlap constant C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives all shifts
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _initial_binary(n, rng):
    """Generate a binary vector of length n with exactly n/2 ones."""
    half = n // 2
    arr = np.zeros(n, dtype=np.float64)
    arr[:half] = 1.0
    rng.shuffle(arr)
    return arr

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] that minimizes the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best step‑function values found (length n, values in [0,1]).
    c5_bound : float
        The achieved C5 value for h_values.
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s * 0.96   # small safety margin

    # ------------------------------------------------------------------ #
    #   Initialise problem size and starting vector
    # ------------------------------------------------------------------ #
    try:
        # pre‑imported by the harness
        parent = np.asarray(initial_h_values, dtype=np.float64)
        n = parent.shape[0]
        # make sure it satisfies the linear constraints (clip & renormalise)
        parent = np.clip(parent, 0.0, 1.0)
        target_sum = n / 2.0
        cur_sum = parent.sum()
        if not np.isclose(cur_sum, target_sum, atol=1e-9):
            # scale to the right total and re‑clip
            parent *= target_sum / cur_sum
            parent = np.clip(parent, 0.0, 1.0)
            # final correction of any residual error by adding to a random index
            residual = target_sum - parent.sum()
            if abs(residual) > 1e-12:
                idx = rng.integers(0, n)
                parent[idx] = np.clip(parent[idx] + residual, 0.0, 1.0)
        current_h = parent.copy()
    except NameError:
        # no parent supplied – build a fresh binary start
        n = 120                     # modest size, fast evaluation
        current_h = _initial_binary(n, rng)

    best_h = current_h.copy()
    best_score = _compute_c5(best_h)

    # ------------------------------------------------------------------ #
    #   Core stochastic optimisation (hill‑climbing + occasional restarts)
    # ------------------------------------------------------------------ #
    it = 0
    last_improve = start_time
    max_iter_no_improve = 30000          # trigger a restart after this many tries
    while time.time() - start_time < time_limit:
        it += 1

        # propose a swap of a small amount of mass between two indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        hi, hj = current_h[i], current_h[j]

        # maximum delta that keeps both entries in [0,1]
        max_pos = min(hi, 1.0 - hj)      # move from i to j
        max_neg = min(1.0 - hi, hj)      # move from j to i
        if max_pos + max_neg == 0.0:
            continue
        delta = rng.uniform(-max_neg, max_pos)

        new_h = current_h.copy()
        new_h[i] = hi - delta
        new_h[j] = hj + delta

        new_score = _compute_c5(new_h)

        # accept if we improve (greedy)
        if new_score < best_score:
            best_h = new_h.copy()
            best_score = new_score
            last_improve = time.time()
            # also keep this as current for further moves
            current_h = new_h
        else:
            # otherwise keep the move only if it does not worsen too much
            # simple simulated‑annealing acceptance
            temp = 0.001 * (1.0 + (time.time() - start_time) / budget_s)
            if rng.random() < np.exp(-(new_score - best_score) / temp):
                current_h = new_h

        # restart heuristics
        if it % max_iter_no_improve == 0 or (time.time() - last_improve) > 30.0:
            # random binary start preserving the sum constraint
            current_h = _initial_binary(n, rng)
            # optionally keep a small fraction of best values (elitism)
            if rng.random() < 0.2:
                mask = rng.random(n) < 0.1
                current_h[mask] = best_h[mask]

    # ------------------------------------------------------------------ #
    #   Return result
    # ------------------------------------------------------------------ #
    return best_h, best_score, int(n)
