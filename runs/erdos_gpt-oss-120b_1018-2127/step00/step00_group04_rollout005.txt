import time
import math
import numpy as np

def _objective(h, dx):
    """Return max overlap integral for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_feasible(h, target_sum):
    """Clamp h to [0,1] and adjust to hit the exact sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # distribute the remaining diff greedily
    if abs(diff) < 1e-12:
        return h
    # indices that can still move in the needed direction
    if diff > 0:
        candidates = np.where(h < 1.0)[0]
    else:
        candidates = np.where(h > 0.0)[0]
    if candidates.size == 0:
        return h
    # spread diff uniformly over candidates (may need several passes)
    while abs(diff) > 1e-12 and candidates.size > 0:
        inc = diff / candidates.size
        for idx in candidates:
            avail = 1.0 - h[idx] if diff > 0 else h[idx]
            delta = np.clip(inc, -avail, avail)
            h[idx] += delta
            diff -= delta
        # recompute candidates
        if diff > 0:
            candidates = np.where(h < 1.0)[0]
        else:
            candidates = np.where(h > 0.0)[0]
    return h

def _propose(h, step, target_sum):
    """Generate a neighbor by moving a small amount between two indices."""
    n = h.size
    new = h.copy()
    # perform a few pairwise transfers
    for _ in range(3):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        amt = (np.random.rand() * 2.0 - 1.0) * step  # [-step, step]
        if not (0.0 <= new[i] + amt <= 1.0 and 0.0 <= new[j] - amt <= 1.0):
            continue
        new[i] += amt
        new[j] -= amt
    # fix any sum drift
    cur = new.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # try a random index that can absorb the whole diff
        for _ in range(10):
            idx = np.random.randint(0, n)
            if 0.0 <= new[idx] + diff <= 1.0:
                new[idx] += diff
                diff = 0.0
                break
        if abs(diff) > 1e-12:
            # fallback to projection
            new = _project_to_feasible(new, target_sum)
    return new

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.

    Returns (h_values, c5_bound, n_points)
    """
    start_time = time.time()
    np.random.seed(seed)

    # Determine discretisation size
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n_points = init.size
        # ensure it satisfies the integral constraint; if not, project
        target_sum = n_points / 2.0
        init = _project_to_feasible(init, target_sum)
        best_h = init.copy()
    else:
        # fallback: 80 points gives decent resolution while staying small
        n_points = 80
        target_sum = n_points / 2.0
        best_h = np.full(n_points, 0.5, dtype=float)  # sum = target_sum

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_score = _objective(best_h, dx)

    # Simulated‑annealing / random‑search parameters
    step = 0.07                     # maximal transfer amount
    temp = 0.002                    # initial temperature for SA
    decay = 0.9995                  # geometric cooling per iteration
    max_iters = 10_000_000         # safety cap

    it = 0
    while time.time() - start_time < budget_s * 0.99 and it < max_iters:

    best_h = _project_to_feasible(best_h, target_sum)
    best_score = _objective(best_h, dx)
