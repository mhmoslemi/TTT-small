import numpy as np
import time
import math

# Helper: evaluate C5 for a given vector h
def evaluate_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    # convolution h with (1-h) in full mode
    conv = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(conv)

# Project a vector to the feasible set: 0<=h<=1 and sum(h)=n/2
def project_feasible(h, target_sum, max_iter=10):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        scale = target_sum / cur if cur > 0 else 0.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
    # Final correction for any tiny residual using a simple redistribution
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual to indices that are not at a boundary
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # fallback: adjust the smallest or largest element
            idx = np.argmin(h) if residual > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
        else:
            inc = residual / free.size
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Desired number of samples (keep modest for speed)
    n_points = kwargs.get("n_points", 200)
    target_sum = n_points / 2.0

    # Initialise h
    init = globals().get("initial_h_values", None)
    if init is not None:
        init = np.asarray(init, dtype=float)
        if init.size != n_points:
            # Upsample / downsample via linear interpolation
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
            h = np.interp(x_new, x_old, init)
        else:
            h = init.copy()
    else:
        # Random start satisfying bounds, then project
        h = np.random.rand(n_points)
    h = project_feasible(h, target_sum)

    best_h = h.copy()
    best_val = evaluate_h(best_h)

    # Parameters for stochastic local search
    max_iters = 200000
    report_interval = 1000
    epsilon_scale = 0.05  # typical step size

    it = 0
    while it < max_iters and (time.time() - start_time) < budget_s:
        # Propose a small redistribution between two random indices
        i, j = np.random.randint(0, n_points, size=2)
        if i == j:
            continue
        eps = (np.random.rand() - 0.5) * 2.0 * epsilon_scale  # in [-ε, ε]
        # ensure move stays within [0,1]
        if eps > 0:
            eps = min(eps, 1.0 - h[i], h[j])  # move from i to j

            eps = max(eps, -h[i], -(1.0 - h[j]))  # move from j to i
        if eps == 0.0:
            continue

        h_new = h.copy()
        h_new[i] += eps
        h_new[j] -= eps
