import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Return the overlap max for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def _feasible_start(n, base=None):
    """Create a feasible start vector (sum = n/2, 0<=h<=1)."""
    target = n / 2.0
    if base is not None:
        h = np.array(base, dtype=float)
        # force exact sum by scaling; clipping later handled by optimizer bounds
        cur = h.sum()
        if cur != 0:
            h *= target / cur
        h = np.clip(h, 0.0, 1.0)
        # if scaling broke the sum, adjust uniformly on free entries
        deficit = target - h.sum()
        if abs(deficit) > 1e-12:
            free = (h > 0) & (h < 1)
            if free.any():
                h[free] += deficit / free.sum()
            else:
                # fallback uniform vector
                h = np.full(n, 0.5)
    else:
        # uniform feasible start
        h = np.full(n, 0.5)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimise the
    Erdős overlap constant C5.

    Returns (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    start_time = time.time()
    # number of sampling points (use the length of the supplied parent if any)
    try:
        n = int(len(initial_h_values))
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        n = 80
        parent_h = None

    # bounds and equality constraint
    bounds = [(0.0, 1.0) for _ in range(n)]
    eq_cons = {"type": "eq", "fun": lambda h: np.sum(h) - n / 2.0}

    # keep track of best solution
    best_h = None
    best_c5 = np.inf

    # initial deterministic run using the parent or uniform vector
    init_vec = _feasible_start(n, parent_h)
    res = minimize(
        _compute_c5,
        init_vec,
        method="SLSQP",
        bounds=bounds,
        constraints=[eq_cons],
        options={"maxiter": 250, "ftol": 1e-9, "disp": False},
    )
    if res.success:
        cand_c5 = _compute_c5(res.x)
        if cand_c5 < best_c5:
