import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

# helper to compute the Erdős C5 bound for a discrete h
def _c5_of_h(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# objective for DE: C5 plus heavy penalty for violating the integral constraint
def _de_obj(y, target_sum, penalty, dx):
    h = np.clip(y, 0.0, 1.0)
    s = h.sum()
    pen = penalty * abs(s - target_sum)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5 + pen

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)

    # decide resolution; start from the provided construction if possible
    base_h = globals().get("initial_h_values", None)
    if base_h is not None:
        n0 = int(base_h.size)
    else:
        n0 = 47  # fallback size
    # try a modestly larger grid for more flexibility but still cheap
    n_points = max(64, int(2 ** np.ceil(np.log2(n0))))  # power of two >=64
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # obtain an initial guess on the new grid via linear interpolation
    if base_h is not None and base_h.size != n_points:
        x_old = np.linspace(0, 2, base_h.size, endpoint=False)
        x_new = np.linspace(0, 2, n_points, endpoint=False)
        h0 = np.interp(x_new, x_old, base_h)
    else:
        h0 = np.full(n_points, 0.5)  # simple feasible start

    # ensure feasibility of the start (project onto sum constraint)
    h0 = np.clip(h0, 0.0, 1.0)
    excess = h0.sum() - target_sum
    if abs(excess) > 1e-12:
        # distribute the correction uniformly, then re‑clip
        h0 -= excess / n_points
        h0 = np.clip(h0, 0.0, 1.0)
        # final tiny adjustment on first entry to hit the sum exactly
        diff = target_sum - h0.sum()
        h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)

    bounds = [(0.0, 1.0)] * n_points
    penalty = 1e4  # large enough to enforce the integral

    # allocate most of the budget to the global DE search
    remaining = budget_s - (time.time() - start)
    de_max_iter = max(1, int(remaining * 0.6 / (n_points * 0.001)))  # heuristic
    result = differential_evolution(
        func=_de_obj,
        bounds=bounds,
        args=(target_sum, penalty, dx),
        strategy="best1exp",
        maxiter=de_max_iter,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=rng,
        polish=False,
        updating="deferred",
    )

    # local refinement with equality constraint (exact integral)
    def local_obj(h):
        return _c5_of_h(h)

    eq_cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    loc_res = minimize(
        fun=local_obj,
        x0=np.clip(result.x, 0.0, 1.0),
        method="SLSQP",
        bounds=bounds,
        constraints=[eq_cons],
        options={"maxiter": 200, "ftol": 1e-9},
    )

    h_opt = np.clip(loc_res.x, 0.0, 1.0)

    # final tiny post‑processing to guarantee the sum exactly
    cur_sum = h_opt.sum()

    # safety fallback: if time ran out earlier return the best we have
    elapsed = time.time() - start
    if elapsed > budget_s:

        pass

    return h_opt, float(c5_bound), int(n_points)
