import time
import numpy as np

# The harness provides:
# - evaluate_erdos_solution(h): returns the C5 value for a discrete h vector
# - initial_h_values (optional): a starting vector (numpy array)

def _ensure_sum_constraint(h, target_sum):
    """Small corrective step to enforce exact sum while staying in [0,1]."""
    current = h.sum()
    diff = target_sum - current
    if abs(diff) < 1e-12:
        return h
    # distribute the difference to the element with most slack
    if diff > 0:
        # need to increase some entries
        slack = 1.0 - h
        idx = np.argmax(slack)
        inc = min(diff, slack[idx])
        h[idx] += inc
    else:
        # need to decrease some entries
        slack = h
        idx = np.argmax(slack)
        dec = min(-diff, slack[idx])
        h[idx] -= dec
    return h

def _perturb(h, rng):
    """Return a new candidate by moving a random amount between two indices."""
    n = h.size
    i, j = rng.choice(n, 2, replace=False)

    # amount moved from j to i (positive delta) keeps the total sum constant
    max_inc_i = 1.0 - h[i]          # how much i can increase
    max_dec_j = h[j]                # how much j can decrease
    max_pos = min(max_inc_i, max_dec_j)

    max_inc_j = 1.0 - h[j]          # how much j can increase (if we move opposite)
    max_dec_i = h[i]                # how much i can decrease
    max_neg = min(max_inc_j, max_dec_i)

    if max_pos + max_neg == 0:
        return h.copy()  # nothing to do

    delta = rng.uniform(-max_neg, max_pos)
    nh = h.copy()
    nh[i] += delta
    nh[j] -= delta
    return nh
