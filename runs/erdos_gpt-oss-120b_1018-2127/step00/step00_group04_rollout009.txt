import time
import numpy as np

def _evaluate_c5(h, dx):
    # Full correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def _project_initial(h, target_sum):
    # Ensure sum exactly target by scaling small correction uniformly
    cur_sum = np.sum(h)
    if abs(cur_sum - target_sum) < 1e-12:
        return h
    # Adjust proportionally and clip
    h = h * (target_sum / cur_sum)
    h = np.clip(h, 0.0, 1.0)
    # Re‑balance any residual error by adding/subtracting from first non‑boundary entry
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size == 0:
            idx = np.arange(len(h))
        i = idx[0]
        h[i] = np.clip(h[i] + residual, 0.0, 1.0)
    return h
