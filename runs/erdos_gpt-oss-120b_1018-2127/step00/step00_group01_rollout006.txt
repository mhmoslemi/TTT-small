import numpy as np
import time
from scipy import optimize

def _c5_value(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_onto_constraints(h, target_sum):
    """Project vector onto [0,1] box with exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    eps = 1e-12
    max_iter = 1000
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < eps:
            break
        free = (h > 0.0 + eps) & (h < 1.0 - eps)
        nfree = free.sum()
        if nfree == 0:
            # all at bounds, distribute proportionally (should not happen)
            break
        delta = diff / nfree
        h[free] += delta
        # re‑clip any that crossed bounds
        h = np.clip(h, 0.0, 1.0)
    # final fix for any tiny sum error
    s = h.sum()
    if abs(s - target_sum) > 1e-9:
        # adjust uniformly on non‑boundary entries
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += (target_sum - s) / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _objective_penalized(x, dx, target_sum, penalty):
    """Objective for DE: C5 plus penalty for sum deviation."""
    c5 = _c5_value(x, dx)
    sum_err = x.sum() - target_sum
    return c5 + penalty * (sum_err ** 2)

def _objective_simple(x, dx):
    """Objective for local refinement: just C5."""
    return _c5_value(x, dx)
