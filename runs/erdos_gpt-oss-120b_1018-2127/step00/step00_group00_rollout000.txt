import numpy as np
import time
import random

# helper: evaluate C5 for a given step function h on [0,2] with n points
def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# project a vector onto the feasible set {0<=h_i<=1, sum h_i = n/2}
def _project_feasible(h):
    n = h.shape[0]
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    # simple correction by adding the same offset and re‑clipping
    if abs(s - target) > 1e-12:
        offset = (target - s) / n
        h = np.clip(h + offset, 0.0, 1.0)
        # a second pass if clipping changed the sum
        s2 = h.sum()
        if abs(s2 - target) > 1e-12:
            # distribute the remaining discrepancy on a random index that can absorb it
            diff = target - s2
            idx = np.argmin(h) if diff > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

# one‑step random feasible perturbation (mass transfer between two coordinates)
def _perturb(h, rng):
    n = h.shape[0]
    i, j = rng.sample(range(n), 2)
    # amount that can be moved from i to j without breaking bounds
    max_from_i = h[i]
    max_to_j = 1.0 - h[j]
    max_transfer = min(max_from_i, max_to_j)
    if max_transfer <= 0:
        return h  # no feasible move
    delta = rng.random() * max_transfer
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def _hill_climb(start_h, time_limit, rng):
    best_h = start_h.copy()
    best_val = _c5_value(best_h)
    start = time.time()
    while time.time() - start < time_limit:
        cand = _perturb(best_h, rng)
        cand = _project_feasible(cand)
        val = _c5_value(cand)
        if val < best_val - 1e-12:
            best_h, best_val = cand, val
    return best_h, best_val
