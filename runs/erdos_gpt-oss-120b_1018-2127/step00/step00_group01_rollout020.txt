import time, math, random
import numpy as np

def _evaluate_c5(h, n):
    """Return the max overlap integral for a binary step function."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # scale by dx = 2/n
    return np.max(corr) * (2.0 / n)

def _feasible_binary(n, seed_array=None):
    """Create a binary vector of length n with exactly n/2 ones."""
    half = n // 2
    if seed_array is not None:
        # keep the strongest half of the seeded values
        order = np.argsort(-seed_array)
        h = np.zeros(n, dtype=float)
        h[order[:half]] = 1.0
        return h
    h = np.zeros(n, dtype=float)
    h[:half] = 1.0
    np.random.shuffle(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    # choose a modest grid size (fast enough for many iterations)
    n = 200  # must be even; integral constraint ⇒ sum = n/2

    # ------------------------------------------------------------------
    # Initialise h
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # linear interpolation to the new grid
        x_old = np.linspace(0.0, 2.0, len(init), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(x_new, x_old, init)
        h = _feasible_binary(n, seed_array=interp)
    else:
        h = _feasible_binary(n)

    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h, n)

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    temp0 = 0.2               # initial temperature
    cooling = 0.995           # exponential cooling factor
    iter_count = 0
    max_iters_without_improve = 2000
    iters_no_improve = 0

    # pre‑compute index lists for speed
    ones_idx = np.flatnonzero(best_h == 1)
    zeros_idx = np.flatnonzero(best_h == 0)

    while time.time() - start < budget_s * 0.95:
        # propose a swap that keeps the sum constraint
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        h_new = best_h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0

        c5_new = _evaluate_c5(h_new, n)
        delta = c5_new - best_c5
        temp = temp0 * (cooling ** iter_count)

        if delta < 0 or np.random.rand() < math.exp(-delta / (temp + 1e-12)):
            best_h = h_new
            best_c5 = c5_new
            # update index caches
            ones_idx = np.flatnonzero(best_h == 1)
            zeros_idx = np.flatnonzero(best_h == 0)
            iters_no_improve = 0
        else:
            iters_no_improve += 1

        # early stop if we beat the target
        if best_c5 <= 0.3808:
            break

        iter_count += 1
        # occasional restart if stuck
        if iters_no_improve >= max_iters_without_improve:
            # random re‑initialisation preserving the sum
            best_h = _feasible_binary(n)
            best_c5 = _evaluate_c5(best_h, n)
            ones_idx = np.flatnonzero(best_h == 1)
            zeros_idx = np.flatnonzero(best_h == 0)

                break

    return (best_h.tolist(), float(best_c5), n)
