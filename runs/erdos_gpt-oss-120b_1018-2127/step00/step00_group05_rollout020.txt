import time, numpy as np

def _evaluate(h):
    """Compute C5 = max overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _repair(h, target):
    """Clip to [0,1] and enforce sum == target (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        return np.full_like(h, target / h.shape[0])
    # scale to hit target, then re‑clip a few times
    h = h * (target / s)
    for _ in range(5):
        over = h > 1.0
        if not over.any():
            break
        excess = h[over] - 1.0
        h[over] = 1.0
        under = h < 1.0
        if under.any():
            h[under] += excess.sum() / under.sum()
    h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
        else:
            h += diff / h.shape[0]
        h = np.clip(h, 0.0, 1.0)
    return h

def _mutate(h):
    """Transfer a random amount of mass between two positions."""
    n = h.shape[0]
    i, j = np.random.randint(n, size=2)
    if i == j:
        return h.copy()
    # amount we can move without breaking bounds
    max_take = h[i]
    max_give = 1.0 - h[j]
    eps = np.random.rand() * 0.5  # up to 0.5
    delta = min(eps, max_take, max_give)
    if delta <= 0:
        return h.copy()
    new = h.copy()
    new[i] -= delta
    new[j] += delta
    return new

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with a low C5 value."""
    start_time = time.time()
    np.random.seed(seed)

    # Determine number of points.
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = kwargs.get('n_points', 100)
        # start from a simple binary pattern (half ones)
        h0 = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
        np.random.shuffle(h0)

    target_sum = n / 2.0
    h_best = _repair(h0, target_sum)
    best_val = _evaluate(h_best)

    # Also keep a pure binary random start as backup
    binary_start = np.concatenate([np.ones(int(target_sum)), np.zeros(n - int(target_sum))])
    np.random.shuffle(binary_start)
    h_bin = _repair(binary_start, target_sum)
    val_bin = _evaluate(h_bin)
    if val_bin < best_val:
        h_best, best_val = h_bin, val_bin

    # Main stochastic search
    while time.time() - start_time < budget_s * 0.95:
        # produce a candidate
        cand = _mutate(h_best)
        cand = _repair(cand, target_sum)
        cand_val = _evaluate(cand)

        if cand_val < best_val:
            h_best, best_val = cand, cand_val
            continue

        # occasional random restart to escape plateaus
        if np.random.rand() < 0.01:
            rnd = np.random.rand(n)
            rnd = _repair(rnd, target_sum)
            rnd_val = _evaluate(rnd)
            if rnd_val < best_val:
                h_best, best_val = rnd, rnd_val

    return h_best, float(best_val), n
