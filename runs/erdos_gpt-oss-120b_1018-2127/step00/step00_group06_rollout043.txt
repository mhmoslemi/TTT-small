def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # distribute uniformly
        h = np.full_like(h, target_sum / len(h))
        return h
    # Scale
    h = h * (target_sum / cur)
    # Clip again
    h = np.clip(h, 0.0, 1.0)
    # Adjust residual
    residual = target_sum - h.sum()
    # Distribute to indices where h not at bounds
    free = np.where((h > 1e-12) & (h < 1-1e-12))[0]
    if len(free) > 0:
        h[free] += residual / len(free)
    else:
        # distribute to all
        h += residual / len(h)
    h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np, time, math

def objective(h, dx):
    """Maximum overlap integral approximation."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair(h, target_sum):
    """Project h onto [0,1] and adjust its sum to target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        return np.full_like(h, target_sum / h.size)
    # scale to meet sum, then clip again
    h = h * (target_sum / cur)
    h = np.clip(h, 0.0, 1.0)
    # residual redistribution
    resid = target_sum - h.sum()
    if abs(resid) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0:
            free = np.arange(h.size)
        h[free] += resid / free.size
        h = np.clip(h, 0.0, 1.0)
    return h

def random_walk(h, scale):
    """Add Gaussian perturbation of given scale."""
    return h + np.random.randn(h.size) * scale
