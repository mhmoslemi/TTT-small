import time
import numpy as np

def _c5_score(h, dx):
    """Compute C5 for a discretized step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _feasible_random(n, rng):
    """Generate a random feasible h: 0<=h<=1, sum(h)=n/2."""
    target = n / 2.0
    h = np.zeros(n, dtype=float)
    remaining = target
    while remaining > 1e-12:
        idx = rng.integers(0, n)
        # add a random share of the remaining amount, respecting the upper bound 1
        add = min(1.0 - h[idx], remaining * rng.random())
        h[idx] += add
        remaining = target - h.sum()
    return h

def run(seed=42, budget_s=1100, **kwargs):

    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.

        c5_bound : float, the achieved C5 bound
        n_points : int, the discretization size used
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        # ensure feasibility (clip then project minimally)
        h0 = np.clip(h0, 0.0, 1.0)
        target_sum = n / 2.0
        diff = target_sum - h0.sum()
        if abs(diff) > 1e-9:
            # distribute the difference uniformly and re‑clip
            h0 += diff / n
            h0 = np.clip(h0, 0.0, 1.0)
            # final simple fix if still off
            if abs(h0.sum() - target_sum) > 1e-6:
                # scale to exact sum while staying in [0,1]
                scale = target_sum / h0.sum()
                h0 = np.clip(h0 * scale, 0.0, 1.0)
        h = h0.copy()
    else:
        # modest size for speed, but >45 to allow improvement
        n = 120
        h = _feasible_random(n, rng)

    dx = 2.0 / n

    best_h = h.copy()
    best_score = _c5_score(best_h, dx)

    start_time = time.time()
    # simple hill‑climbing / simulated‑annealing loop
    while True:

        if elapsed >= budget_s * 0.95:
