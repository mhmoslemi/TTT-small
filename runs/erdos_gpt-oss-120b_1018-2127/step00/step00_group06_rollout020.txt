import time, math
import numpy as np

def _evaluate(h: np.ndarray) -> float:
    """Return the C5 value for a discretized h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _project_to_constraints(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clamp to [0,1] and enforce exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) <= 1e-12:
        return h
    # distribute the difference uniformly, then re‑clamp a few times
    diff = target_sum - cur
    h += diff / h.size
    for _ in range(5):
        over = h > 1.0
        under = h < 0.0
        if not over.any() and not under.any():
            break
        excess = (h[over] - 1.0).sum()
        deficit = (-h[under]).sum()
        total = excess + deficit
        if total == 0:
            break
        h[over] = 1.0
        h[under] = 0.0
        remaining = target_sum - h.sum()
        h += remaining / h.size
    # final tiny adjustment on a non‑boundary element
    cur = h.sum()
    if abs(cur - target_sum) > 1e-9:
        diff = target_sum - cur
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            free = np.arange(h.size)
        i = free[0]
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] minimizing
        C5 = max_k ∫ h(x)(1‑h(x+k)) dx
    using a simple simulated‑annealing scheme.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_score = math.inf
    best_n = None

    # candidate grid sizes (even numbers give exact integral)
    for n in (80, 96, 120, 144):
        if time.time() - start > budget_s * 0.95:
            break
        target_sum = n / 2.0

        # initialise h
        try:
            # interpolate provided parent solution
            init = initial_h_values  # supplied by the harness if present
            x_old = np.linspace(0.0, 2.0, len(init), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, init)
        except NameError:
            # fall back to a random half‑filled binary vector
            h = np.zeros(n, dtype=float)
            ones = rng.choice(n, n // 2, replace=False)
            h[ones] = 1.0

        h = _project_to_constraints(h, target_sum)
        score = _evaluate(h)

        if score < best_score:
            best_h, best_score, best_n = h.copy(), score, n

        # Simulated annealing parameters
        temp = 0.12
        decay = 0.9996
        max_iters = 250000
        it = 0

        while time.time() - start < budget_s * 0.99 and it < max_iters:
            it += 1
            i, j = rng.choice(n, 2, replace=False)
            new_val = rng.random()
            delta = new_val - h[i]
            new_j = h[j] - delta
            if not (0.0 <= new_val <= 1.0 and 0.0 <= new_j <= 1.0):

            h_new[j] = new_j
