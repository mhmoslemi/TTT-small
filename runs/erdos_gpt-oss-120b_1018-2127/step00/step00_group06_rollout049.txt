import time, math, random
import numpy as np

def _repair(h, target_sum):
    """Clip to [0,1] and enforce exact sum=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # indices that can move without hitting bounds
    mask = (h > 0) & (h < 1)
    if mask.any():
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    else:
        # all at bounds – distribute uniformly then clip again
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-9:
        h += (target_sum - cur) / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _objective(h, dx):
    """Maximum overlap integral approximation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C₅ problem."""
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------
    # 1. Initialise base grid
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # default modest size; even length guarantees target sum integer
        n0 = 200
        h0 = np.full(n0, 0.5, dtype=float)   # already satisfies sum=n/2
    n = h0.size
    target_sum = n / 2.0
    dx = 2.0 / n

    # ----------------------------------------------------------------------
    # 2. Basic repair & objective wrappers for current grid
    # ----------------------------------------------------------------------
    best_h = _repair(h0.copy(), target_sum)
    best_score = _objective(best_h, dx)

    # ----------------------------------------------------------------------
    # 3. Simulated‑annealing‑style search
    # ----------------------------------------------------------------------
    T = 1.0                 # temperature
    scale = 0.02            # std‑dev of Gaussian perturbation
    it = 0
    # allocate a fraction of the budget for this single‑grid optimisation
    while time.time() - start_time < budget_s * 0.95:
        # 3a. Propose neighbour
        h_new = best_h + np.random.randn(n) * scale
        h_new = _repair(h_new, target_sum)

        # 3b. Evaluate
        new_score = _objective(h_new, dx)

        # 3c. Acceptance rule
        if new_score < best_score or random.random() < math.exp((best_score - new_score) / (T + 1e-12)):
            best_h = h_new
            best_score = new_score

        # 3d. Cooling schedule
        it += 1
        if it % 2000 == 0:
            T *= 0.90
            scale *= 0.95
            # keep temperature from hitting zero
            if T < 1e-6:
                T = 1e-6

    # ----------------------------------------------------------------------
    # 4. Return the best discovered solution
    # ----------------------------------------------------------------------
    return (best_h.tolist(), float(best_score), n)
