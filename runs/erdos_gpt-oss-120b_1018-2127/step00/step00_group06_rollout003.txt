import time
import numpy as np

def _project_swap(h, i, j, delta):
    """Move delta from i to j while keeping bounds [0,1]."""
    # limit delta so that both stay within bounds
    max_reduce_i = h[i]  # can reduce at most this
    max_increase_j = 1.0 - h[j]  # can increase at most this
    delta = np.clip(delta, -max_increase_j, max_reduce_i)
    # also ensure we don't increase i beyond 1 or reduce j below 0
    max_increase_i = 1.0 - h[i]
    max_reduce_j = h[j]
    # if delta is negative, we are moving from j to i
    if delta < 0:
        delta = np.clip(delta, -max_reduce_j, max_increase_i)
    else:
        delta = np.clip(delta, -max_reduce_j, max_increase_i)  # keep same bounds
    h[i] -= delta
    h[j] += delta
    return h

def _c5_score(h):
    """Compute the C5 overlap bound for a step function h."""
    n = h.shape[0]
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _initial_feasible(n):
    """Return a simple feasible vector of length n (sum = n/2)."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[:half] = 1.0
    # if n is odd, distribute the remaining 0.5 uniformly
    if n % 2:
        h[half] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_best : ndarray
        The best step values found (length n_points).
    best_score : float
        Corresponding C5 bound.
    n_points : int
        Number of discretization points used.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():
        h_curr = np.array(initial_h_values, dtype=float).copy()
    else:
        # fallback size 95 (as mentioned in the statement)
        n_points = 95
        h_curr = _initial_feasible(n_points)

    n = h_curr.shape[0]
    # Ensure feasibility: clip and then adjust sum exactly
    h_curr = np.clip(h_curr, 0.0, 1.0)
    total = h_curr.sum()
    target = n / 2.0
    if abs(total - target) > 1e-12:
        # simple scaling then clipping
        scale = target / total if total != 0 else 0.0
        h_curr *= scale
        h_curr = np.clip(h_curr, 0.0, 1.0)
        # final correction by adding/subtracting small epsilon to first non‑boundary entries
        diff = target - h_curr.sum()
        idx = np.where((h_curr > 0) & (h_curr < 1))[0]
        if idx.size:
            h_curr[idx[0]] += diff

    best_h = h_curr.copy()
    best_score = _c5_score(best_h)

    # Parameters for stochastic search
    step_initial = 0.2          # initial magnitude of delta transfers
    step_min = 1e-4
    decay = 0.995               # exponential decay per iteration
    max_iter = 10_000_000       # safety upper bound
    iter_count = 0

    while True:
        now = time.time()
        if now - start_time > budget_s - 0.5:
            break
        if iter_count >= max_iter:
            break
        iter_count += 1

        step = max(step_min, step_initial * (decay ** iter_count))
