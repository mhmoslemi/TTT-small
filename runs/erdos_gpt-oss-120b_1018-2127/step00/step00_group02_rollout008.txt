import numpy as np, time, math
from scipy.signal import correlate

def compute_c5(h):
    """Return C5 for a discrete step function h on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def hill_climb(initial, budget, rng):
    """Simple hill‑climbing under sum and box constraints."""
    n = len(initial)
    best = initial.copy()
    best_val = compute_c5(best)
    # parameters
    max_step = 0.07          # maximal amount moved in one swap
    attempts = 0
    while True:
        # time check inside loop
        if time.time() - start_time > budget * 0.95:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = (rng.random() - 0.5) * 2 * max_step
        hi, hj = best[i], best[j]
        # propose move: hi -> hi+eps, hj -> hj-eps
        hi_new = hi + eps
        hj_new = hj - eps
        if not (0.0 <= hi_new <= 1.0 and 0.0 <= hj_new <= 1.0):
            continue
        # apply tentative change
        cand = best.copy()
        cand[i] = hi_new
        cand[j] = hj_new
        cand_val = compute_c5(cand)
        attempts += 1
        if cand_val < best_val - 1e-8:
            best = cand
            best_val = cand_val
    return best, best_val
