import time
import numpy as np

def _objective(h, dx):
    """Return max overlap integral C5 for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_bounds_and_sum(v, target):
    """
    Project vector v onto the box [0,1] and enforce sum(v)=target.
    Simple iterative water-filling scheme.
    """
    v = np.clip(v, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) < 1e-12:
        return v
    # iterative adjustment on free (non‑boundary) coordinates
    for _ in range(20):
        if abs(total - target) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = (target - total) / n_free
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
        total = v.sum()
    # final tiny correction if needed
    diff = target - total
    if abs(diff) > 1e-12:
        # distribute the remaining diff uniformly and clip again
        v += diff / len(v)
        v = np.clip(v, 0.0, 1.0)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimises
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.

    Returns (h_values, c5_bound, n_points)
    where h_values is a NumPy 1‑D array of length n_points.
    """
    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 50                     # fallback size if no parent supplied
        h0 = np.full(n, 0.5)       # uniform start satisfies the integral constraint

    dx = 2.0 / n
    target_sum = n / 2.0           # because ∑h * dx = 1

    # Ensure initial vector is feasible
    h0 = _project_to_bounds_and_sum(h0, target_sum)

    best_h = h0.copy()
    best_val = _objective(best_h, dx)

    # Simulated‑annealing‑style parameters
    max_iter = 200000
    init_sigma = 0.3
    cooling = 0.9995
    sigma = init_sigma

    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s * 0.98:
            break

        # propose a new candidate
        cand = best_h + sigma * np.random.randn(n)
        cand = _project_to_bounds_and_sum(cand, target_sum)

        cand_val = _objective(cand, dx)

        # accept if better
        if cand_val < best_val:
            best_h = cand
            best_val = cand_val

        # gradually cool
        sigma *= cooling

        # occasional random restart to avoid stagnation
        if it % 5000 == 0 and it > 0:
            rand = np.random.rand(n)
            rand = _project_to_bounds_and_sum(rand, target_sum)
            rand_val = _objective(rand, dx)
            if rand_val < best_val:
                best_h = rand
                best_val = rand_val

    return best_h, float(best_val), n
