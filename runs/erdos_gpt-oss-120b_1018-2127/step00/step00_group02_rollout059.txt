import time
import math
import numpy as np
from scipy import optimize as opt

def _compute_c5(h):
    """Return the discrete C5 value for step values h on [0,2]."""
    dx = 2.0 / len(h)
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_onto_capped_simplex(v, target_sum):
    """Project vector v onto [0,1]^n with given total sum (capped simplex)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        diff = target_sum - v.sum()
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds – distribute uniformly (small violation tolerated)
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
            break
        delta = diff / cnt
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    return v
