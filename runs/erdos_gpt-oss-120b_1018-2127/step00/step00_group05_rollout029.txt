import time
import numpy as np

def _project_box_sum(v, target, lower=0.0, upper=1.0, max_iter=20):
    """
    Project vector v onto the intersection of the box [lower,upper]^n
    and the affine hyperplane sum(x)=target.
    Simple iterative redistribution.
    """
    w = np.clip(v, lower, upper)
    for _ in range(max_iter):
        s = w.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # indices that can move in the needed direction
        if diff > 0:
            mask = w < upper
        else:
            mask = w > lower
        if not np.any(mask):
            break
        # distribute equally among free coordinates
        delta = diff / mask.sum()
        w[mask] += delta
        # re‑clip in case we overshoot
        w = np.clip(w, lower, upper)
    # final correction if tiny numerical error remains
    s = w.sum()
    if abs(s - target) > 1e-10:
        w += (target - s) / len(w)
        w = np.clip(w, lower, upper)
    return w

def _evaluate_c5(h, dx):
    """Return C5 for a candidate step vector."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with integral 1 to minimise the
    Erdős overlap constant C5.

    Returns:
        h_values : np.ndarray of shape (n_points,)
        c5_bound : float, the attained C5 value
        n_points : int, length of the returned vector
    """
    rng = np.random.default_rng(seed)
    t_start = time.time()

    # ------------------------------------------------------------
    # 1. Choose discretisation size (must stay modest)
    # ------------------------------------------------------------
    base_n = globals().get('initial_h_values', None)
    if base_n is not None:
        base_n = len(initial_h_values)
    else:
        base_n = 70
    # target size – a compromise between expressive power and speed
    n_points = int(min(120, max(70, base_n * 1.5)))
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because integral h dx =1  -> sum(h)=n/2

    # ------------------------------------------------------------
    # 2. Initialise a feasible vector
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        # linear interpolation to new grid
        old_x = np.linspace(0, 2, len(initial_h_values), endpoint=False)
        new_x = np.linspace(0, 2, n_points, endpoint=False)
        h0 = np.interp(new_x, old_x, initial_h_values)
    else:
        h0 = rng.random(n_points)

    h0 = _project_box_sum(h0, target_sum)
    best_h = h0.copy()
    best_score = _evaluate_c5(best_h, dx)

    # ------------------------------------------------------------
    # 3. Simple stochastic local search with decreasing step size
    # ------------------------------------------------------------
    # parameters
    max_no_improve = 2000
    no_improve = 0
    step_scale = 0.2  # initial noise amplitude
    decay = 0.9995    # per iteration decay

    while True:
        elapsed = time.time() - t_start
        if elapsed > budget_s - 0.5:  # safety margin
            break

        # generate candidate
        noise = rng.normal(scale=step_scale, size=n_points)
        cand = best_h + noise
        cand = _project_box_sum(cand, target_sum)

        score = _evaluate_c5(cand, dx)

        # accept if better
        if score < best_score:

            no_improve = 0

        # keep step size from vanishing completely
