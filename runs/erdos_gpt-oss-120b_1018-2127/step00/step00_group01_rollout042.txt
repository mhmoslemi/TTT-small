import time
import numpy as np
import random

def compute_c5(h, dx):
    """Return the C5 value for a given h and grid spacing dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * float(np.max(corr))

def project_to_sum_and_bounds(x, target_sum, max_iter=5):
    """
    Project vector x onto the feasible set:
        0 <= x_i <= 1  and  sum(x) = target_sum
    Simple alternating projection (box then hyperplane) suffices for our use.
    """
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    for _ in range(max_iter):
        s = x.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x

def initial_feasible(n):
    """Create a random feasible vector of length n."""
    x = np.random.rand(n)
    target = n / 2.0
    return project_to_sum_and_bounds(x, target)

def improve_h(h, dx, target_sum, budget, start_time,
              step_init=0.1, patience=2000, check_interval=5000):
    """
    Simple stochastic search improving h while time allows.
    Returns the best h found and its C5 value.
    """
    n = h.size
    best_h = h.copy()
    best_val = compute_c5(best_h, dx)

    step = step_init
    no_improve = 0
    it = 0
    while time.time() - start_time < budget:
        cand = best_h + step * np.random.randn(n)
        cand = project_to_sum_and_bounds(cand, target_sum)
        val = compute_c5(cand, dx)
        if val < best_val - 1e-12:
            best_h, best_val = cand, val
            no_improve = 0
        else:
            no_improve += 1

        it += 1
        if no_improve > patience:
            step *= 0.7
            no_improve = 0
        if it % check_interval == 0:
            # occasional random restart
            rnd = initial_feasible(n)
            rnd_val = compute_c5(rnd, dx)
            if rnd_val < best_val:
                best_h, best_val = rnd, rnd_val
    return best_h, best_val

def final_adjust(h, target_sum):
    """Make the sum exact while keeping bounds."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # distribute diff over non‑boundary entries
    mask = (h > 1e-9) & (h < 1.0 - 1e-9)
    idx = np.where(mask)[0]
    if idx.size == 0:
        # fall back: adjust the first entry (keeps feasibility)
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    else:
        inc = diff / idx.size
        h[idx] = np.clip(h[idx] + inc, 0.0, 1.0)
    # a second clipping may be needed
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that
    minimizes the overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    total_budget = budget_s

    # Determine initial size
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    except Exception:
        n = 47
        init_h = None

    target_sum = n / 2.0
    dx = 2.0 / n

    if init_h is not None:
        h0 = project_to_sum_and_bounds(init_h, target_sum)
    else:
        h0 = initial_feasible(n)

    # Primary search on the base grid
    remaining = total_budget * 0.55
    best_h, best_val = improve_h(h0, dx, target_sum,
                                 remaining, start,
                                 step_init=0.12)

    # Optional upscale phases
    while True:
        elapsed = time.time() - start
        if elapsed >= total_budget * 0.95:
            break
        # decide whether to double the grid
        n_new = n * 2
        if n_new > 2000:  # keep size reasonable
            break
        # upscale current best
        best_h = np.repeat(best_h, 2)
        n = n_new
        target_sum = n / 2.0
        dx = 2.0 / n
        # small local refinement on the finer grid
        remaining = total_budget * 0.35 - (elapsed)
        if remaining <= 0:
            break
        best_h, best_val = improve_h(best_h, dx, target_sum,
                                     remaining, start,
                                     step_init=0.05)
        # after refinement, continue loop to possibly double again

    # Final feasibility adjustment
    best_h = final_adjust(best_h, n / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / n)

    return (best_h, final_c5, n)
