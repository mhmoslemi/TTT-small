import numpy as np
import time
import math
from numpy import correlate
from numpy.random import default_rng

def compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    conv = correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def make_binary_with_sum(h, target_sum):
    """Convert float array h to binary {0,1} keeping exactly target_sum ones."""
    n = h.size
    # initial rounding
    binary = (h >= 0.5).astype(float)
    cur = int(binary.sum())
    if cur == target_sum:
        return binary
    # compute fractional distances to change
    diff = h - binary  # +ve where we rounded down, -ve where rounded up
    if cur < target_sum:
        # need to flip some zeros to ones: choose largest positive diff
        zeros_idx = np.where(binary == 0)[0]
        order = np.argsort(-diff[zeros_idx])  # descending diff
        to_flip = zeros_idx[order[: target_sum - cur]]
        binary[to_flip] = 1.0
    else:
        # need to flip some ones to zeros: choose most negative diff
        ones_idx = np.where(binary == 1)[0]
        order = np.argsort(diff[ones_idx])  # ascending diff (most negative first)
        to_flip = ones_idx[order[: cur - target_sum]]
        binary[to_flip] = 0.0
    return binary

def initial_binary(n):
    """Simple alternating binary pattern with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[::2] = 1.0
    # adjust if parity mismatch
    if h.sum() != n // 2:
        # flip the last element if needed
        diff = int(h.sum() - n // 2)
        if diff > 0:
            idx = np.where(h == 1)[0][-diff:]
            h[idx] = 0.0
        elif diff < 0:
            idx = np.where(h == 0)[0][: -diff]
            h[idx] = 1.0
    return h

    """Optimize a step function h on [0,2] to lower the Erdős C5 bound.

    Returns:
        h_values (np.ndarray): optimized binary values on the grid.
        c5_bound (float): achieved max overlap integral.
        n_points (int): number of discretization points.
    """
    rng = default_rng(seed)
    start_time = time.time()

    # Determine initial grid size and starting vector

        init_h = np.array(initial_h_values, copy=False)
        n = init_h.size

        h = make_binary_with_sum(init_h, target_ones)
    except NameError:
        # No parent construction provided – choose modest size
        n = 100  # even

        h = initial_binary(n)

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # Pre‑compute index lists for fast swap sampling
    ones_idx = np.where(best_h == 1)[0]
    zeros_idx = np.where(best_h == 0)[0]

    # Simulated‑annealing parameters
    max_iter = 10_000_000  # safety cap
    temp_start = 0.02
    temp_end = 1e-5
    # linear cooling schedule based on iteration count
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.98:  # leave margin for final steps
            break
        # temperature decreasing
        t = temp_start + (temp_end - temp_start) * (it / max_iter)

        # propose a random swap between a 1 and a 0
