import time
import numpy as np

def _project_to_bounds_and_sum(v, target_sum, max_iter=20, eps=1e-9):
    """Project vector v onto [0,1] box with exact sum constraint."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target_sum - x.sum()
        if abs(diff) < eps:
            break
        # indices that can still move in the needed direction
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        delta = diff / cnt
        # apply delta and re‑clip
        x[mask] += delta
        np.clip(x, 0.0, 1.0, out=x)
    return x

def _evaluate(h, dx):
    """Return max overlap value for h."""
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return conv.max(), conv.argmax() - (len(h) - 1)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # grid size – moderate for speed while still expressive
    n_points = 200  # even, gives dx = 0.01
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum(h)*dx == 1

    # initialise from provided parent if present, else random
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        # interpolate/extrapolate to current n_points
        x_parent = np.linspace(0, 2, len(parent), endpoint=False)
        x_new = np.linspace(0, 2, n_points, endpoint=False)
        h = np.interp(x_new, x_parent, parent)
    else:
        h = np.random.rand(n_points)

    # enforce constraints
    h = _project_to_bounds_and_sum(h, target_sum)

    # keep best solution seen
    best_h = h.copy()
    best_val, _ = _evaluate(best_h, dx)

    # optimisation parameters
    lr = 0.5
    iter_count = 0
    max_iters = 50000  # safety cap

    while time.time() - start_time < budget_s * 0.98 and iter_count < max_iters:
        iter_count += 1

        # evaluate objective and dominant lag
        val, lag = _evaluate(h, dx)

        # update best
        if val < best_val:
            best_val = val
            best_h = h.copy()

        # compute sub‑gradient of the active lag term
        grad = np.zeros_like(h)
        if lag >= 0:
            # h[i] interacts with (1-h)[i+lag]
            grad[:n_points - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n_points - lag]
        else:
            # lag negative: shift opposite direction
            pos = -lag
            grad[pos:] += (1.0 - h[:n_points - pos])
            grad[:n_points - pos] -= h[pos:]

        grad *= dx

        # gradient step with simple backtracking line search
        step = h - lr * grad
        step = _project_to_bounds_and_sum(step, target_sum)

        # if no improvement, shrink learning rate
        new_val, _ = _evaluate(step, dx)
        if new_val < val:
            h = step
            lr = min(lr * 1.02, 1.0)  # gently increase if successful
        else:
            lr *= 0.7  # reduce step size on failure

        # occasional random perturbation to escape shallow minima
        if iter_count % 2000 == 0:
            noise = np.random.normal(scale=0.05, size=n_points)
            h = _project_to_bounds_and_sum(h + noise, target_sum)

    return best_h, float(best_val), int(n_points)
