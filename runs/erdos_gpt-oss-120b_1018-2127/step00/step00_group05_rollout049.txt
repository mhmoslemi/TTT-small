import time
import numpy as np

def _project(vec, target_sum, max_iter=3):
    """Clip to [0,1] and scale to meet target sum (integral constraint)."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        if s == 0.0:
            v.fill(target_sum / len(v))
            break
        factor = target_sum / s
        v = np.clip(v * factor, 0.0, 1.0)
    # final tiny adjustment to hit target sum exactly (within eps)
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        # try to fix by moving a non‑saturated entry
        free = np.where((v > 0.0) & (v < 1.0))[0]
        if free.size == 0:
            # all saturated – distribute uniformly (will stay within bounds)
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
        else:
            i = free[0]
            v[i] = np.clip(v[i] + diff, 0.0, 1.0)
    return v

def _evaluate(h, dx):
    """Return max overlap integral C5 for a given step vector."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _upsample(src, n_target):
    """Resize src (assumed equispaced on [0,2]) to length n_target by linear interp."""
    x_src = np.linspace(0.0, 2.0, len(src), endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_tgt, x_src, src)
