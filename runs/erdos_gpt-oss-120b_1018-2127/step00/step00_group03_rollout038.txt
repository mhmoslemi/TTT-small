import numpy as np
import time
import random
import math

def compute_c5(h):
    """Return max overlap integral for step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_sum(h, target):
    """Adjust h in-place to have sum == target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to add mass
        while diff > 1e-12:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            incr = diff / idx.size
            h[idx] += incr
            over = np.where(h > 1.0)[0]
            if over.size:
                excess = h[over] - 1.0
                diff = excess.sum()
                h[over] = 1.0
                continue
            diff = target - h.sum()
    else:
        # need to remove mass
        while diff < -1e-12:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            decr = -diff / idx.size
            h[idx] -= decr
            under = np.where(h < 0.0)[0]
            if under.size:
                excess = -h[under]
                diff = -excess.sum()
                h[under] = 0.0
                continue
            diff = target - h.sum()
    np.clip(h, 0.0, 1.0, out=h)
    return h

def get_initial(n):
    """Create an initial h of length n, using the global initial_h_values if present."""
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        m = len(base)
        # interpolate onto n points (periodic interval [0,2))
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, base)
        h = np.clip(h, 0.0, 1.0)
        h = project_sum(h, n / 2.0)
    else:
        # simple half‑ones, half‑zeros pattern
        h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
    return h

def propose_move(h, max_step=0.1):
    """Return a new h after transferring a random epsilon from i to j (preserving sum)."""
    n = len(h)
    i = random.randrange(n)
    j = random.randrange(n)
    if i == j:
        return None
    eps = max_step * random.random()
    eps = min(eps, h[i])          # cannot go below 0
    eps = min(eps, 1.0 - h[j])    # cannot exceed 1
    if eps <= 1e-12:
        return None
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # Choose a moderate resolution; larger n gives finer control but costs more time.
    n = 140
    h = get_initial(n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()
    # Simple hill‑climbing / simulated‑annealing loop.
    while time.time() - start < budget_s * 0.95:
        h_candidate = propose_move(h, max_step=0.08)
        if h_candidate is None:
            continue
        c5_candidate = compute_c5(h_candidate)

        # Greedy acceptance
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            h = h_candidate
        else:
            # Occasional uphill move (annealing)
            elapsed = time.time() - start
            temperature = max(0.001, 1.0 - elapsed / budget_s)
            prob = math.exp((best_c5 - c5_candidate) / temperature)
            if random.random() < prob:
                h = h_candidate

    # Ensure the returned vector respects the exact integral constraint.
    best_h = project_sum(np.asarray(best_h, dtype=float), n / 2.0)

    return (best_h.tolist(), float(best_c5), n)
