import time
import numpy as np

# Helper: keep total sum fixed while staying in [0,1]
def random_transfer(rng, h):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # maximum amount we can move from i to j (positive epsilon)
    pos_max = min(h[i], 1.0 - h[j])
    # maximum amount we can move from j to i (negative epsilon)
    neg_max = -min(h[j], 1.0 - h[i])
    eps = rng.random() * (pos_max - neg_max) + neg_max
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def ensure_constraints(h):
    # Clip and then adjust any residual sum error by water‑filling
    h = np.clip(h, 0.0, 1.0)
    S = h.size / 2.0
    tol = 1e-12
    it = 0
    while abs(h.sum() - S) > tol and it < 1000:
        diff = S - h.sum()
        free = (h > 0) & (h < 1)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds; distribute diff uniformly and re‑clip
            h += diff / h.size
        else:
            delta = diff / cnt
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
        it += 1
    return h

def run(seed=42, budget_s=1100, **kwargs):

            h = np.interp(x_new, x_old, h0)
        else:

        target_n = 70
        h = np.zeros(target_n, dtype=np.float64)
