import time
import numpy as np

def _ensure_initial_h():
    """Obtain initial h and its size."""
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n = h0.size
        # enforce feasibility just in case
        target_sum = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        current = h0.sum()
        if abs(current - target_sum) > 1e-12:
            # simple scaling to meet sum, then re‑clip
            h0 = h0 * (target_sum / current)
            h0 = np.clip(h0, 0.0, 1.0)
            # final correction by adjusting first element
            diff = target_sum - h0.sum()
            h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)
        return h0, n
    except NameError:
        # no parent supplied – use uniform 0.5 (satisfies sum constraint)
        n = 70
        h0 = np.full(n, 0.5, dtype=np.float64)
        return h0, n

def _c5_value(h, dx):
    """Compute the overlap constant C5 for a given h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rnd = np.random.RandomState(seed)
    h, n = _ensure_initial_h()
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = h.copy()
    best_score = _c5_value(best_h, dx)

    start = time.time()
    it = 0

    # Simulated‑annealing schedule
    T0 = 0.02
    T_min = 1e-5
    max_iters = 200000  # safety cap
    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.97 or it >= max_iters:
            break
        it += 1

        # linear temperature decrease
        T = max(T_min, T0 * (1 - elapsed / budget_s))

        # propose a transfer between two random positions
        i, j = rnd.choice(n, size=2, replace=False)
        # maximal feasible transfer respecting box constraints
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        eps = rnd.rand() * max_transfer * 0.5  # limit step size
        if eps <= 0:
            continue

        # apply perturbation
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps

        # evaluate
        cand_score = _c5_value(cand, dx)

        # acceptance rule
        if cand_score < best_score:
            best_h, best_score = cand, cand_score
        else:
            # Metropolis acceptance
            if rnd.rand() < np.exp(-(cand_score - best_score) / T):
                best_h, best_score = cand, cand_score

        # occasional random shake to diversify
        if it % 5000 == 0:
            shake_idx = rnd.choice(n, size=5, replace=False)
            shake = rnd.rand(5) * 0.2 - 0.1  # [-0.1,0.1]
            cand = best_h.copy()
            cand[shake_idx] += shake
            cand = np.clip(cand, 0.0, 1.0)
            # renormalize to keep total sum correct
            diff = target_sum - cand.sum()
            cand[0] = np.clip(cand[0] + diff, 0.0, 1.0)
            cand_score = _c5_value(cand, dx)
            if cand_score < best_score:
                best_h, best_score = cand, cand_score

    # final safeguard: ensure feasibility of the returned vector
    best_h = np.clip(best_h, 0.0, 1.0)
    current = best_h.sum()

        best_h *= (target_sum / current)
        best_h = np.clip(best_h, 0.0, 1.0)
        # final tiny adjustment
        diff = target_sum - best_h.sum()
        best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    final_c5 = _c5_value(best_h, dx)
    return best_h.tolist(), float(final_c5), int(n)
