import time
import numpy as np

def _project_feasible(h, target_sum):
    """Clip to [0,1] and adjust to hit the target sum (iterative simple projection)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        cur = h.sum()
        if cur == 0:
            h[:] = target_sum / h.size
            break
        scale = target_sum / cur
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    # final correction for any remaining tiny error
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute the remaining diff uniformly respecting bounds
        for _ in range(10):
            add = diff / h.size
            h += add
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - h.sum()
            if abs(diff) < 1e-12:
                break
    return h

def _c5_value(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function yielding a low C5 bound."""
    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():

    else:

    n = h.size
    target_sum = n / 2.0  # because integral over [0,2] must be 1, dx=2/n

    h = _project_feasible(h, target_sum)

    best_h = h.copy()
    best_score = _c5_value(best_h)

    # Simulated annealing parameters
    max_iter = 200000
    iter_count = 0
    temperature0 = 0.05
    temperature = temperature0
    step_scale0 = 0.2  # max delta change for a single perturbation
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.5:  # leave a small safety margin
            break
        if iter_count >= max_iter:
            break

        # cooling schedules
        progress = iter_count / max_iter
        temperature = temperature0 * (1.0 - progress) + 1e-4
        step_scale = step_scale0 * (1.0 - progress) + 1e-4

        # Propose perturbation: pick two indices, transfer delta
        i, j = np.random.randint(0, n, size=2)
        while j == i:

        delta = (np.random.rand() - 0.5) * 2.0 * step_scale  # in [-step, step]

        cand = best_h.copy()

        cand[j] = np.clip(cand[j] - opp_delta, 0.0, 1.0)
