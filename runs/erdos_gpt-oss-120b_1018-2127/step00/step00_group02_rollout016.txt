import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(h, target_sum):
    """Project h onto [0,1] box and sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds, distribute uniformly then re‑clip
            h += diff / h.size
        else:
            h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _interpolate_initial(init_h, n_points):
    """Linear interpolation of the parent vector to a new length."""
    old_n = init_h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_points, endpoint=False)
    return np.interp(new_x, old_x, init_h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h=1, 0≤h≤1,
    minimizing C5 = max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rnd = np.random.RandomState(seed)
    start_time = time.time()

    # problem size – compromise between flexibility and speed
    n_points = 90
    target_sum = n_points / 2.0          # because ∫h = 1 ⇒ sum(h)*dx = 1, dx = 2/n
    # initialise from parent if present, otherwise uniform half‑ones
    try:
        init_h = _interpolate_initial(initial_h_values, n_points)
    except NameError:
        init_h = np.full(n_points, 0.5)

    # ensure feasibility
    init_h = _project_to_constraints(init_h, target_sum)

    best_h = init_h.copy()
    best_val = _compute_c5(best_h)

    # ---- local continuous optimisation (SLSQP) -----------------------------
    bounds = [(0.0, 1.0)] * n_points
    eq_cons = {"type": "eq", "fun": lambda x: x.sum() - target_sum}
    opts = {"maxiter": 200, "ftol": 1e-9, "disp": False}
    res = minimize(_compute_c5, best_h, method="SLSQP",
                   bounds=bounds, constraints=[eq_cons], options=opts)
    if res.success and res.fun < best_val:
        best_h = _project_to_constraints(res.x, target_sum)
        best_val = _compute_c5(best_h)

    # ---- stochastic local search -----------------------------------------
    # adapt step size over time
    max_iter = 20000
    sigma0 = 0.07
    iter_done = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:

        remaining = max_iter - iter_done

        # geometric decay of sigma

    res = minimize(_compute_c5, best_h, method="SLSQP",

    best_h = _project_to_constraints(best_h, target_sum)
    best_val = _compute_c5(best_h)
