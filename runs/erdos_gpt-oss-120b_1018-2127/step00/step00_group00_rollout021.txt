import time
import numpy as np

def _evaluate_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete samples."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _initial_h(n):
    """Create a feasible initial vector of length n."""
    # uniform 0.5 satisfies the sum constraint exactly
    return np.full(n, 0.5, dtype=np.float64)
