import time
import math

def _c5_from_h(h):
    """Compute C5 for a discretized step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def _perturb(h, rng, max_step):
    """Return a new h by moving mass delta between two positions."""
    n = h.size
    # choose donor (with >0) and receiver (with <1)
    donors = np.where(h > 1e-12)[0]
    receivers = np.where(h < 1 - 1e-12)[0]
    if donors.size == 0 or receivers.size == 0:
        return h.copy()

    j = rng.choice(receivers)
    # maximum transferable mass
    max_delta = min(max_step, h[i], 1.0 - h[j])

        return h.copy()

    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
