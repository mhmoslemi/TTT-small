import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _make_initial(n):
    """Create a feasible uniform start when no parent is supplied."""
    h = np.full(n, 0.5)          # average value 0.5 gives integral 1
    # ensure exact sum n/2
    current = h.sum()
    target = n / 2.0
    h *= target / current
    return h

def _propose_move(h, max_step=0.1):
    """Generate a new feasible h by moving mass between two positions."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # how much can we move from i to j without violating bounds
    allowable = min(h[i], 1.0 - h[j], max_step)
    if allowable <= 0:
        return h  # no feasible move; return unchanged
    eps = np.random.rand() * allowable
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize
    C5 = max_k ∫ h(x)(1-h(x+k))dx.

    Returns
    -------
    h_values : np.ndarray
        Optimized values on the uniform grid.
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretization points.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    try:
        n = len(initial_h_values)          # provided by the harness if any
        h = np.array(initial_h_values, dtype=float).copy()
        # Slight adjustment to enforce exact integral (sum == n/2)
        target_sum = n / 2.0
        cur_sum = h.sum()
        if not np.isclose(cur_sum, target_sum, atol=1e-9):
            h *= target_sum / cur_sum
        # clip to [0,1] after scaling
        h = np.clip(h, 0.0, 1.0)
    except NameError:
        # No parent construction – start from a uniform feasible vector
        n = 45
        h = _make_initial(n)

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    # Simulated‑annealing style loop
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        # Temperature proportional to remaining time
        temperature = max(0.001, (budget_s - elapsed) / budget_s)

        # Propose a feasible neighbour
        h_candidate = _propose_move(h, max_step=0.15)

        # Evaluate
        cand_c5 = _compute_c5(h_candidate, dx)
        delta = cand_c5 - best_c5

        # Accept if better or by Metropolis criterion
        if cand_c5 < best_c5 or np.random.rand() < np.exp(-delta / (temperature + 1e-12)):
            h = h_candidate
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = h_candidate.copy()

        # Occasionally re‑inject the best solution to avoid drift
        if np.random.rand() < 0.01:
            h = best_h.copy()

        # Simple diversification every 2000 iterations
        iteration += 1
        if iteration % 2000 == 0:
            # random shuffle of mass while preserving sum
            perm = np.random.permutation(n)
            h = best_h[perm].copy()

    # Final sanity clamp
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _compute_c5(best_h, dx)
    return best_h, final_c5, n
