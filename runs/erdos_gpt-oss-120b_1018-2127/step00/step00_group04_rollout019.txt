import time
import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

def _c5_of_h(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _objective(h, dx):
    """Objective for optimizer: C5 (to be minimised)."""
    return _c5_of_h(h, dx)

def _run_once(initial_h, dx, budget, rng):
    """Run a single SLSQP optimisation starting from `initial_h`."""
    n = initial_h.size
    # bounds 0..1
    bounds = Bounds(0.0, 1.0)
