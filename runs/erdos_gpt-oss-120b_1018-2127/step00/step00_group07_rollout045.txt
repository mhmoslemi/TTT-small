import time, math, random
import numpy as np

def _compute_c5(h):
    """Return C5 bound for step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _repair_sum_and_bounds(h, target_sum):
    """Clip to [0,1] and adjust to meet exact sum (within floating error)."""
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative projection onto the simplex with box constraints
    for _ in range(3):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds, distribute proportionally
            h = np.clip(h + diff / len(h), 0.0, 1.0)
            continue
        delta = diff / cnt
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final scaling to eliminate tiny residual error
    cur = h.sum()
    if cur != 0:
        h *= target_sum / cur
    h = np.clip(h, 0.0, 1.0)
    return h

def _initial_vector(n):
