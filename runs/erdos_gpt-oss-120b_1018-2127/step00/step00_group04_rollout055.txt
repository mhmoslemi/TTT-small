import numpy as np
import time
import math
import random

# evaluate_erdos_solution and initial_h_values are provided by the harness

def compute_c5(h):
    """Return the overlap bound C5 for a step function h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_onto_feasible(v):
    """
    Project vector v onto the box [0,1] and the affine constraint sum(v)=len(v)/2.
    Simple iterative clipping: first clip to [0,1], then shift uniformly to meet the sum,
    repeating a few times.
    """
    n = v.size
    target_sum = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    for _ in range(5):
        cur_sum = v.sum()
        diff = target_sum - cur_sum
        # distribute diff over entries that are not at a bound
        free = (v > 0.0) & (v < 1.0)
        if free.any():
            adjust = diff / free.sum()
            v[free] += adjust
            v = np.clip(v, 0.0, 1.0)
        else:
            # all at bounds, just scale globally (should not happen often)
            v = v * (target_sum / cur_sum) if cur_sum != 0 else v
        # final clip
        v = np.clip(v, 0.0, 1.0)
    # final correction to enforce exact sum (tiny numeric error)
    cur_sum = v.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # adjust one free element if possible
        idx = np.where((v > 0.0) & (v < 1.0))[0]
        if idx.size:
            v[idx[0]] += target_sum - cur_sum
            v = np.clip(v, 0.0, 1.0)
    return v

def random_feasible(n, rng):
    """Generate a random feasible h with sum = n/2 and values in [0,1]."""
    # start with n/2 ones and rest zeros, then shuffle and add small noise

    h += rng.uniform(-0.1, 0.1, size=n)

    return h

def hill_climb(initial_h, rng, time_limit, start_time):
    """Simple hill‑climbing with pairwise mass transfers."""
    h = initial_h.copy()

def upscale_initial(initial, target_n):
    """Upscale initial_h_values to target_n points via linear interpolation."""
    src_n = initial.size
    x_src = np.linspace(0.0, 2.0, src_n, endpoint=False) + (1.0 / src_n)

    h_up = np.interp(x_tgt, x_src, initial)
    h_up = project_onto_feasible(h_up)
    return h_up
