import numpy as np

from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1

    fb = np.fft.rfft(b, size)

def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)

    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_A(f):
    """Build constraint matrix A for the LP: A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M, g >= 0."""
    n = len(f)
    A = build_constraint_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 15})

        return np.maximum(result.x, 0.0)
    return None

        w_full[:len(w)] = w
        w_prefix = np.zeros(2 * n)
        w_prefix[1:] = np.cumsum(w_full)
        
        for j in range(n):
            # i ranges from 0 to n-1, k = i+j ranges from j to j+n-1
            k_start = j
            k_end = min(j + n, 2 * n - 1)  # inclusive
            if k_end <= k_start:
                continue
            # Second term: a[j] * sum_{k=k_start}^{k_end-1} w[k]
            w_sum = w_prefix[k_end] - w_prefix[k_start]
            # First term: sum_{i=0}^{n-1} a[i] * w[i+j] = sum_{k=j}^{j+n-1} a[k-j] * w[k]
            # This is a cross-correlation
            first_term = np.dot(a, w_full[k_start:k_start+n])
            g_a[j] = first_term + a[j] * w_sum

        # Normalize gradient
        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)

        # Adam update
        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (g_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        lr = base_lr / np.sqrt(1.0 + 0.005 * step_count)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Check score
        t_shifted = theta - np.max(theta)
        a_check = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
        score = local_score(a_check)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()

        if verbose and step_count % 100 == 0:
            print(f"    Grad step={step_count}, score={score:.8f}, "
                  f"best={best_score:.8f}, tau={tau:.6f}")

    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)


def sa_polish(f_init, time_limit, rng):
    """Simulated annealing with adaptive moves for final polishing."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0a = time.monotonic()
    temperature = 0.005
    cooling = 0.9999
    iter_count = 0

    while time.monotonic() - t0a < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.4:
            # Targeted peak reduction
            conv = fft_convolve(f, f)
            # Get top peaks
            n_peaks = min(3, 2 * n - 1)
            top_indices = np.argsort(conv)[-n_peaks:]
            for peak_idx in top_indices:
                peak_idx = int(peak_idx)
                contribs = np.zeros(n)
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = f[i] * f[j]
                # Find top contributors
                n_top = max(2, n // 15)
                top_idx = np.argsort(contribs)[-n_top:]
                # Reduce top contributors
                candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=len(top_idx)))
                # Increase non-contributors
                other_idx = np.setdiff1d(np.arange(n), top_idx)
                if len(other_idx) > 0:
                    n_inc = min(4, len(other_idx))
                    inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                    candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=len(inc_idx)))
        elif r < 0.65:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.8:
            # Two block perturbations
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 10))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.9:
            # Smooth global perturbation
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Sparse perturbation (set some to 0 or increase)
            n_mod = max(2, n // 20)
            idx = rng.choice(n, size=n_mod, replace=False)
            candidate[idx] *= np.exp(rng.normal(0.0, 0.05, size=n_mod))

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif np.log(rng.random()) < -(new_score - current_score) / max(temperature, 1e-12):
            f = candidate
            current_score = new_score

        temperature *= cooling
        if temperature < 0.0001:
            temperature = 0.001  # Reheat

    return best_f, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng, n_tones=8):
    """Multi-tone initialization with random frequencies."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.03, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Gaussian random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_oscillating_init(n, rng):
    """Oscillating pattern initialization."""
    t = np.arange(n, dtype=np.float64)
    freq = rng.uniform(0.02, 0.1)
    a = 1.0 + 0.15 * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def interpolate_sequence(seq, n):
    """Interpolate a sequence to length n."""
    if len(seq) == n:
        return seq.copy()
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP-based iterative optimization with diverse initializations.
    Phase 2: Gradient-based polishing with annealed smoothing.
    Phase 3: Simulated annealing for fine-tuning.
    Phase 4: Final LP refinement.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            if len(provided_seq) > 0 and np.sum(provided_seq) > 0.01:
                s_prov = local_score(provided_seq)
                print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.8f}")
                if s_prov < best_score:
                    best_score = s_prov
                    best_seq = provided_seq.copy()
        except Exception as e:
            print(f"Error with provided sequence: {e}")

    n_primary = len(provided_seq) if provided_seq is not None else 256

    # === Phase 1: LP-based optimization on multiple candidates ===
    print("\n" + "=" * 60)
    print(f"Phase 1: LP-based optimization (primary n={n_primary})")
    print("=" * 60)

    # Define search configurations
    search_configs = [
        # (n, init_type, time_fraction)
        (n_primary, 'provided', 0.25),
        (n_primary, 'chirp1', 0.12),
        (n_primary, 'chirp2', 0.10),
        (n_primary, 'multitone', 0.10),
        (n_primary, 'random', 0.08),
        (n_primary, 'oscillating', 0.08),
        (n_primary, 'chirp3', 0.07),
        (256, 'multitone2', 0.05),
        (128, 'chirp1', 0.05),
        (384, 'multitone', 0.05),
        (n_primary, 'random2', 0.05),
    ]

    total_frac = sum(c[2] for c in search_configs)
    available_time = deadline - time.monotonic()
    candidates_to_refine = []  # Store (f, score) for later refinement

    for idx, (n, init_name, frac) in enumerate(search_configs):
        if time.monotonic() > deadline - 60:
            break

        time_alloc = max(20.0, frac / total_frac * available_time * 0.9)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 15)
        if time_alloc < 15:
            break

        # Generate initialization
        if init_name == 'provided' and provided_seq is not None:
            a0 = make_provided_init(provided_seq, n, rng)
        elif init_name == 'chirp1':
            a0 = make_chirp_init(n, 0.3 + 0.2 * rng.random(), rng)
        elif init_name == 'chirp2':
            a0 = make_chirp_init(n, 0.5 + 0.3 * rng.random(), rng)
        elif init_name == 'chirp3':
            a0 = make_chirp_init(n, 0.1 + 0.1 * rng.random(), rng)
        elif init_name == 'multitone':
            a0 = make_multitone_init(n, rng, n_tones=6)
        elif init_name == 'multitone2':
            a0 = make_multitone_init(n, rng, n_tones=10)
        elif init_name == 'random':
            a0 = make_random_init(n, rng, spread=0.2 + 0.2 * rng.random())
        elif init_name == 'random2':
            a0 = make_random_init(n, rng, spread=0.5 + 0.3 * rng.random())
        elif init_name == 'oscillating':
            a0 = make_oscillating_init(n, rng)
        else:
            a0 = make_random_init(n, rng, 0.3)

        print(f"  [{idx+1}/{len(search_configs)}] n={n}, init='{init_name}', alloc={time_alloc:.0f}s")
        t_start = time.monotonic()

        f_best, score = lp_iterate(a0, time_alloc, verbose=False)
        elapsed = time.monotonic() - t_start
        print(f"    LP result: score={score:.8f}, time={elapsed:.1f}s")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"    *** New best: {best_score:.8f} (n={n}, init={init_name}) ***")

        # Store for potential later refinement
        if score < best_score + 0.05:
            candidates_to_refine.append((f_best.copy(), score, n, init_name))

    # === Phase 2: Gradient polish on best candidates ===
    print("\n" + "=" * 60)
    print("Phase 2: Gradient-based polishing")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        # Polish the best sequence
        if remaining > 60:
            grad_time = min(300.0, remaining * 0.4)
            print(f"  Polishing best (n={len(best_seq)}), time={grad_time:.0f}s")
            a_grad, score_grad = gradient_polish(best_seq, grad_time, verbose=False)
            print(f"  Gradient polish result: {score_grad:.8f}")
            if score_grad < best_score:
                best_score = score_grad
                best_seq = a_grad.copy()
                print(f"  *** New best: {best_score:.8f} ***")

        # Also try polishing the second-best if time permits
        remaining = deadline - time.monotonic()
        for f_cand, sc_cand, n_cand, name in candidates_to_refine[:2]:
            if remaining < 30:
                break
            if np.allclose(f_cand, best_seq):
                continue
            grad_time = min(150.0, remaining * 0.3)
            if grad_time < 20:
                break
            print(f"  Polishing candidate (n={n_cand}, init={name}, score={sc_cand:.6f}), time={grad_time:.0f}s")
            a_grad2, score_grad2 = gradient_polish(f_cand, grad_time, verbose=False)
            if score_grad2 < best_score:
                best_score = score_grad2
                best_seq = a_grad2.copy()
                print(f"  *** New best: {best_score:.8f} ***")
            remaining = deadline - time.monotonic()

    # === Phase 3: SA polish ===
    print("\n" + "=" * 60)
    print("Phase 3: Simulated annealing polish")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(250.0, remaining * 0.5)
        if sa_time > 20:
            print(f"  SA polish, n={len(best_seq)}, time={sa_time:.0f}s")
            a_sa, score_sa = sa_polish(best_seq, sa_time, rng)
            print(f"  SA result: {score_sa:.8f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.8f} ***")

    # === Phase 4: Final LP refinement ===
    print("\n" + "=" * 60)
    print("Phase 4: Final LP refinement")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        final_lp_time = min(200.0, remaining * 0.8)
        if final_lp_time > 10:
            print(f"  Final LP, n={len(best_seq)}, time={final_lp_time:.0f}s")
            a_final, score_final = lp_iterate(best_seq, final_lp_time, verbose=False)
            print(f"  Final LP result: {score_final:.8f}")
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.8f} ***")

    # === Phase 5: Quick SA after final LP ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_final_sa, score_final_sa = sa_polish(best_seq, remaining * 0.7, rng)
            if score_final_sa < best_score:
                best_score = score_final_sa
                best_seq = a_final_sa.copy()
                print(f"  Final SA improvement: {best_score:.8f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.8f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256


def make_provided_init(provided_seq, n, rng):
    """Create initialization from provided sequence, possibly with perturbation."""
    a = interpolate_sequence(provided_seq.astype(np.float64), n)
    a = np.clip(a, 0.01, None)
    # Add small perturbation
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
