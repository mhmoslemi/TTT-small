"""
Improved optimization for minimizing peak auto-convolution of non-negative sequences.
Combines LP-based direction finding, softmax-gradient descent, and simulated annealing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_conv(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute score = 2*n*max(conv(a,a)) / sum(a)^2"""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_conv(a, a)
    return 2.0 * n * c.max() / (s * s)


def normalize(a, target_sum):
    """Normalize a to have the given sum, maintaining non-negativity."""
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) * (target_sum / len(a))
    return a * (target_sum / s)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    # For k in 0..2n-2, j in 0..n-1: A[k,j] = f[k-j] if 0 <= k-j < n else 0
    # This is a block-Toeplitz-like matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j)
        hi = min(2 * n - 1, j + n)
        A[lo:hi, j] = f[:hi - lo]
    return A


def solve_lp(f, M, timeout=20):
    """Solve LP: max sum(g) s.t. f*g <= M (componentwise), g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    if res.success:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_iterate(a, time_limit, rng, S):
    """LP-based iterative optimization."""
    n = len(a)
    f = np.maximum(a, 0.0).copy()
    f = normalize(f, S)
    cur_score = fast_score(f)
    best_f = f.copy()
    best_score = cur_score
    t0 = time.monotonic()
    t = 0.1  # mixing parameter
    stall = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_conv(f, f)
        M = c.max()
        
        # Solve LP for direction
        g = solve_lp(f, M, timeout=15)
        iters += 1
        
        if g is None:
            stall += 10
            t *= 0.5
            if stall > 50:
                # Perturb
                f = f * np.exp(rng.normal(0, 0.05, n))
                f = np.maximum(f, 0)
                f = normalize(f, S)
                cur_score = fast_score(f)
                stall = 0
                t = 0.1
            continue
        
        g = normalize(g, S)
        
        # Line search
        improved = False
        for trial in [t, t * 0.5, t * 0.25, t * 0.1, 0.05, 0.02]:
            f_new = (1 - trial) * f + trial * g
            f_new = np.maximum(f_new, 0)
            s = fast_score(f_new)

                f = f_new
                cur_score = s

                stall = 0
                t = 0.15
        
        if iters % 500 == 0:
            print(f"    LP iter {iters}: score={cur_score:.6f}, best={best_score:.6f}, "
                  f"t={t:.4f}, stall={stall}, time={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def softmax_gradient(f, temperature):
    """Compute gradient of softmax approximation to max(conv(f,f))."""
    n = len(f)
    c = fft_conv(f, f)
    # Softmax weights
    c_max = c.max()
    w = np.exp((c - c_max) / temperature)
    w /= w.sum()
    
    # Gradient: g[m] = 2 * sum_k w[k] * f[k-m] for valid k-m
    # This is 2 * cross_correlation(w, f)
    # Cross-correlation: (w ~~ f)[m] = sum_k w[k] * f[k-m]
    # = (w * f_reversed)[n-1+m-n+1] ... let me just compute directly or via FFT
    
    # Using FFT: cross_correlation(w, f)[m] = ifft(fft(w) * conj(fft(f)))[m]
    # But we need the "full" cross-correlation, not circular
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    W = np.fft.fft(w, size)
    F = np.fft.fft(f, size)
    # w[k] * f[k-m] summed over k = ifft(W * conj(F)) but with appropriate shift
    # Actually, ifft(W * F[::-1]) gives something different
    # Let me think again...
    
    # sum_k w[k] * f[k-m] where f is defined on 0..n-1 and w on 0..2n-2
    # This is the convolution of w with a "reflected and shifted" f
    # = (w * f_ref)[m + n - 1] where f_ref[j] = f[n-1-j]
    
    f_ref = f[::-1]
    size2 = 1
    while size2 < len(w) + n - 1:
        size2 <<= 1
    conv = np.fft.irfft(np.fft.rfft(w, size2) * np.fft.rfft(f_ref, size2), size2)
    # conv[j] = sum_k w[k] * f_ref[j-k] = sum_k w[k] * f[n-1-(j-k)] = sum_k w[k] * f[k-j+n-1]
    # We want g[m] = sum_k w[k] * f[k-m]
    # So we need k-m = n-1-(j-k) => j = n-1+m
    # So conv[n-1+m] = g[m]/2 (times the factor of 2)
    
    g = 2.0 * conv[n - 1:n - 1 + n]
    return g


def softmax_descent(a, time_limit, rng, S):
    """Softmax gradient descent with adaptive temperature."""
    n = len(a)
    f = np.maximum(a, 0.0).copy()
    f = normalize(f, S)
    cur_score = fast_score(f)
    best_f = f.copy()
    best_score = cur_score
    t0 = time.monotonic()
    
    temperature = 0.5  # starts high (smooth), decreases
    lr = 0.5
    momentum = 0.9
    vel = np.zeros(n)
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        g = softmax_gradient(f, temperature)
        
        # Gradient step with momentum
        vel = momentum * vel - lr * g
        f_new = f + vel
        f_new = np.maximum(f_new, 0)
        f_new = normalize(f_new, S)
        
        new_score = fast_score(f_new)
        
        if new_score < cur_score:
            f = f_new
            cur_score = new_score
            lr = min(lr * 1.05, 2.0)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
        else:
            lr *= 0.7
            vel *= 0.5
        
        # Cool down temperature
        temperature *= 0.999
        temperature = max(temperature, 0.01)
        
        # Periodic perturbation
        if iters % 200 == 199:
            f = f * np.exp(rng.normal(0, 0.03, n))
            f = np.maximum(f, 0)
            f = normalize(f, S)
            cur_score = fast_score(f)
        
        iters += 1
        if iters % 1000 == 0:
            print(f"    SG iter {iters}: score={cur_score:.6f}, best={best_score:.6f}, "
                  f"T={temperature:.4f}, lr={lr:.4f}, time={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def local_search(a, time_limit, rng, S):

    n = len(a)
    f = np.maximum(a, 0.0).copy()
    f = normalize(f, S)
    cur_score = fast_score(f)

    cooling = 0.9998
    iters = 0

        candidate = f.copy()
        r = rng.random()
        
        if r < 0.35:
            # Peak-targeted: reduce elements contributing to peak
            c = fft_conv(f, f)
            peak_idx = int(np.argmax(c))
            # Find top contributors to c[peak_idx]
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):

                if 0 <= j < n:

            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.015, len(top_idx)))
            # Increase some other elements to maintain sum
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 10:
                n_inc = min(15, len(other) // 3)
                inc_idx = rng.choice(other, n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, n_inc))
        elif r < 0.55:
            # Block perturbation
            width = int(rng.integers(2, max(3, n // 4)))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.025, width))
        elif r < 0.70:

                width = int(rng.integers(1, max(2, n // 20)))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.02, width))

                candidate[idx[1]] = val
        else:

        candidate = np.maximum(candidate, 0)
        candidate = normalize(candidate, S)
        new_score = fast_score(candidate)
        
        if new_score < cur_score:
            f = candidate
            cur_score = new_score

                best_f = f.copy()
        elif rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            f = candidate
            cur_score = new_score

        iters += 1
        
        if iters % 5000 == 0:
            print(f"    LS iter {iters}: best={best_score:.6f}, temp={temp:.6f}, "

    return best_f, best_score


def make_bell_init(n, S, rng):
    """Create a bell-shaped initial sequence."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a += 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return normalize(a, S)


def make_chirp_init(n, S, rng, alpha=0.5):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.05)
    return normalize(a, S)


def make_random_init(n, S, rng):
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.maximum(a, 0.05)
    return normalize(a, S)

"""
Multi-phase optimization for minimizing peak auto-convolution of non-negative sequences.
Uses LP-based direction finding with dual information, softmax gradient descent,
and simulated annealing local search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_conv(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def fast_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_conv(a, a)
    return 2.0 * n * c.max() / (s * s)


def normalize(a, S):
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) * (S / len(a))
    return a * (S / s)


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(n, 2 * n - 1 - j)
        A[j:j + end, j] = f[:end]
    return A


def solve_lp_direction(f, M, timeout=10):
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-12:
            # Extract dual info if available
            duals = None
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                duals = res.ineqlin.slack  # or .marginals
            return g, duals
        return g, None
    return None, None


def lp_phase(a_init, time_limit, rng, S, verbose=True):
    n = len(a_init)
    f = np.maximum(a_init, 0.0).copy()
    f = normalize(f, S)
    cur = fast_score(f)
    best = f.copy()
    best_s = cur
    t0 = time.monotonic()
    alpha = 0.12
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_conv(f, f)
        M = c.max()
        
        g, duals = solve_lp_direction(f, M, timeout=10)
        it += 1
        
        if g is None:
            stall += 5
            alpha *= 0.8
            if stall > 40:
                f = f * np.exp(rng.normal(0, 0.06, n))
                f = np.maximum(f, 0)
                f = normalize(f, S)
                cur = fast_score(f)
                stall = 0
                alpha = 0.12
            continue
        
        g = normalize(g, S)

        improved = False
        candidates_t = [alpha, alpha*0.7, alpha*0.5, alpha*0.3, 0.05, 0.02, 0.01]

            if s_trial < cur:

            print(f"    SA[{it}]: best={best_s:.6f}, temp={temp:.6f}, t={time.monotonic()-t0:.0f}s")
    
    return best, best_s


def make_bell(n, S, rng):
    t = np.arange(n, dtype=float)
    a = np.exp(-2.0 * ((t - n/2) / (n/3))**2)
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    return normalize(a, S)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP iteration, softmax gradient, simulated annealing.
    Minimizes 2n*max(conv(a,a))/sum(a)^2 for non-negative a.
    """
    tg = time.monotonic()
    deadline = tg + max(0, budget_s - 25)
    rng = np.random.default_rng(seed)
    n = 512
    S = np.sqrt(2.0 * n)
    
    best_s = np.inf
    best_seq = None
    
    # Init
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if len(a0) == n and a0.sum() > 0.01:
            a0 = normalize(a0, S)
            s0 = fast_score(a0)
            print(f"Start: n={n}, score={s0:.6f}")
            best_s = s0
            best_seq = a0.copy()
    
    if best_seq is None:
        best_seq = make_bell(n, S, rng)
        best_s = fast_score(best_seq)
        print(f"Generated: score={best_s:.6f}")
    
    # Phase 1: LP
    print("\n--- Phase 1: LP ---")
    t1 = min(800.0, max(60.0, deadline - time.monotonic() - 500))
    b, s = lp_phase(best_seq, t1, rng, S, verbose=True)
    print(f"  P1: {s:.6f}")
    if s < best_s:
        best_s, best_seq = s, b.copy()
    
    # Phase 2: Softmax gradient
    if time.monotonic() < deadline - 250:
        print(f"\n--- Phase 2: Softmax ---")
        t2 = min(350.0, max(20.0, deadline - time.monotonic() - 250))
        b, s = softmax_phase(best_seq, t2, rng, S)
        print(f"  P2: {s:.6f}")
        if s < best_s:
            best_s, best_seq = s, b.copy()
    
    # Phase 3: Annealing
    if time.monotonic() < deadline - 80:
        print(f"\n--- Phase 3: Anneal ---")
        t3 = min(250.0, max(15.0, deadline - time.monotonic() - 80))
        b, s = anneal_phase(best_seq, t3, rng, S)
        print(f"  P3: {s:.6f}")
        if s < best_s:
            best_s, best_seq = s, b.copy()
    
    # Phase 4: Final LP
    if time.monotonic() < deadline - 15:
        print(f"\n--- Phase 4: Final LP ---")
        t4 = max(5.0, deadline - time.monotonic() - 15)
        if t4 > 30:
            b, s = lp_phase(best_seq, t4, rng, S, verbose=False)
            print(f"  P4: {s:.6f}")
            if s < best_s:
                best_s, best_seq = s, b.copy()
    
    # Final
    flist = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    fs = evaluate_sequence(flist)
    print(f"\nFINAL: {fs:.6f}, n={len(flist)}")
    return flist
