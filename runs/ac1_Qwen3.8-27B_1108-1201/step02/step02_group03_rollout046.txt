Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses mirror descent (multiplicative updates), soft-max annealing, LP refinement,
and Nesterov-accelerated subgradients with adaptive learning rates.

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target):
    """Scale a to have the given sum."""
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def soft_max_gradient(a, tau):
    """
    Compute gradient of the soft-max of conv(a,a).
    Returns gradient of the same length as a.
    """
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # Compute via convolution
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad


def hard_max_gradient(a):
    """Compute hard max gradient (subgradient at the peak)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def mirror_descent_step(a, grad, lr, noise=0.0):
    """
    Multiplicative (mirror) descent step.
    a_new[i] = a[i] * exp(-lr * grad[i] / a[i]) (approximately)
    Or more practically: a_new[i] = a[i] - lr * grad[i] (additive but with exp projection)
    
    Actually, for the entropy mirror map, the update is:
    a_new[i] propional to a[i] * exp(-lr * grad[i])
    """
    n = len(a)
    # Use log-space update for better handling of non-negativity
    # a_new = a * exp(-lr * grad) then renormalize
    # But this can underflow, so use: a_new = a * exp(-lr * grad / max(a, eps))
    
    # Simpler approach: additive with clipping
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    return a_new


def mirror_descent_step2(a, grad, lr):
    """
    True mirror descent with entropy mirror map.
    a_new[i] = a[i] * exp(-lr * grad[i]) / sum(a[j] * exp(-lr * grad[j]))
    """
    n = len(a)
    log_a = np.where(a > 1e-300, np.log(a), -300.0)
    # a_new[i] ∝ a[i] * exp(-lr * grad[i])
    # = exp(log(a[i]) - lr * grad[i])
    log_new = log_a - lr * grad
    # Shift for numerical stability
    log_new -= log_new.max()
    new = np.exp(log_new)
    S = new.sum()
    if S < 1e-300:
        return a.copy()
    return new / S

One important detail: the LP formulation. Let me double-check.

Given current a (length n), conv(a,a) = c (length 2n-1).
We want to find δ (length n) to minimize max_k (c[k] + 2*sum_i a[k-i]*δ[i])

subject to:
- a[i] + t*δ[i] >= 0 for all i (for step size t)
- sum(δ) = 0 (to maintain normalization)

The LP minimizes M s.t.:
- c[k] + 2*sum_i a[k-i]*δ[i] <= M for all k
- δ[i] >= -a[i] (for t=1; for general t, δ[i] >= -a[i]/t)
- sum(δ) = 0
- M >= 0

The optimal M is the minimum achievable (linearized) peak. The direction δ tells us how to move.

Then we take a step a_new = a + t*δ and check the actual objective.

For the step size t, we should ensure a_new >= 0. Since δ[i] >= -a[i] from the LP, a + δ >= 0 is guaranteed for t <= 1. For t > 1, we need to be more careful, but let's just use t <= 1.

Actually, we can also use t > 1 by scaling: if we use step size t, the constraint becomes δ[i] >= -a[i]/t. So we can reformulate the LP with a parameter t. But for simplicity, let's just use t = 1 and if the improvement is small, we can try multiple steps.

"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses LP-based linearized peak minimization (sequential quadratic programming via LP),
soft-max gradient descent with annealing, asymmetric initialization strategies,
and adaptive step-size line search. Key insight: asymmetric sequences can have
lower peak autocorrelation than symmetric ones.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target):
    """Scale a to have the given sum."""
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def soft_max_gradient(a, tau):
    """Compute gradient of soft-max of conv(a,a) w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n].copy()
    return grad


def hard_max_gradient(a):
    """Compute subgradient at the peak of conv(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def lp_step(a, M_init=None):
    """
    Solve LP to find direction delta that minimizes linearized peak autocorrelation.
    
    Minimize M
    s.t. c[k] + 2*sum_i a[k-i]*delta[i] <= M  for all k
         delta[i] >= -a[i]  (ensures a + delta >= 0)
         sum(delta) = 0
         M >= 0
    
    Returns (delta, M_opt) or (None, None) on failure.
    """
    n = len(a)
    c = np.convolve(a, a)
    n_lags = 2 * n - 1
    
    # Objective: minimize M (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Build constraint matrix
    # Type 1: n_lags constraints for the peak bound
    # Type 2: n constraints for non-negativity (delta[i] >= -a[i])
    n_constraints = n_lags + n
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    # Type 1: c[k] + 2*sum_i a[k-i]*delta[i] - M <= 0
    # => 2*sum_i a[k-i]*delta[i] - M <= -c[k]
    for k in range(n_lags):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = 2.0 * a[j]
        A_ub[k, n] = -1.0
        b_ub[k] = -c[k]
    
    # Type 2: -delta[i] <= a[i]
    for i in range(n):
        A_ub[n_lags + i, i] = -1.0
        b_ub[n_lags + i] = a[i]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(None, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x[:n].copy(), float(res.x[n])
    except Exception:
        pass
    
    return None, None

def line_search(a, delta, max_steps=8):
    """
    Line search: find best t in {1, 0.5, 0.25, ...} such that obj(a + t*delta) improves.
    Returns (best_t, best_a, best_score).
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    base_score = compute_obj(a)
    best_t = 0.0
    best_a = a.copy()
    best_s = base_score
    
    for step in range(max_steps):
        t = 1.0 / (2.0 ** step)
        a_cand = a + t * delta
        if np.any(a_cand < -1e-10):
            continue
        a_cand = np.maximum(a_cand, 0.0)
        a_cand = normalize(a_cand, S_t)
        s = compute_obj(a_cand)
        if s < best_s:
            best_s = s
            best_t = t
            best_a = a_cand
    
    return best_t, best_a, best_s


def gradient_step(a, lr, momentum_buf, decay, tau=None):
    """One gradient descent step with momentum."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    
    if tau is None:
        grad = hard_max_gradient(a)
    else:
        grad = soft_max_gradient(a, tau)
    
    grad -= grad.mean()
    momentum_buf[:] = decay * momentum_buf - lr * grad
    a_new = a + momentum_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new


def make_asymmetric_starts(n, rng):
    """Generate diverse asymmetric starting sequences."""
    starts = []
    i = np.arange(n, dtype=np.float64)
    
    # Linear ramp
    starts.append(i / n + 0.1)
    # Exponential decay
    starts.append(np.exp(-3.0 * i / n) + 0.05)
    # Reverse exponential
    starts.append(np.exp(-3.0 * (n - 1 - i) / n) + 0.05)
    # Quadratic
    starts.append((i / n) ** 2 + 0.1)
    # Sine asymmetric
    starts.append(0.5 + 0.5 * np.sin(np.pi * i / n) + 0.1)
    # Two-bump asymmetric
    starts.append(np.exp(-((i - n * 0.3) / (n * 0.15)) ** 2) + 0.3 * np.exp(-((i - n * 0.8) / (n * 0.1)) ** 2) + 0.05)
    # Random
    starts.append(rng.random(n) * 0.5 + 0.2)
    # Sparse random
    a = np.zeros(n)
    mask = rng.random(n) < 0.4
    a[mask] = rng.random(mask.sum()) * 0.8 + 0.2
    starts.append(a)
    # Step function
    starts.append(np.where(i < n // 3, 1.0, 0.3) + 0.1)
    # Triangle
    center = n * 0.4
    starts.append(np.abs(1.0 - 2.0 * np.abs(i - center) / n) + 0.1)
    
    return starts

                if r == 3:
                    a0 = np.maximum(best_seq * (1.0 + rng.normal(0, 0.3, best_n)), 0.0)
            
            t_alloc = min(time_per_restart, deadline - time.monotonic() - 30.0)
            if t_alloc < 20:
                break
            
            print(f"  Restart {r}:")
            seq, score = optimize_from_start(a0, time.monotonic() + t_alloc, use_lp=True, max_lp_iters=300)
            
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"    *** New best: {best_score:.6f}")
    
    # Phase C: Try a different length if time remains
    print("\n=== Phase C: Alternative lengths ===")
    phase_c_deadline = deadline - 30.0
    
    alt_lengths = [256, 448, 640]
    for n in alt_lengths:
        if time.monotonic() > phase_c_deadline:
            break
        if n == len(best_seq) if best_seq is not None else False:
            continue
        
        # Quick check with a couple of starts
        starts = make_asymmetric_starts(n, rng)
        if has_prior and len(gs) >= n:
            starts.insert(0, gs[:n].copy())
        
        for idx in range(min(2, len(starts))):
            if time.monotonic() > phase_c_deadline - 10:
                break
            t_alloc = min(60.0, deadline - time.monotonic() - 30.0)
            if t_alloc < 20:
                break
            
            print(f"  Alt n={n}, start {idx}:")
            seq, score = optimize_from_start(starts[idx], time.monotonic() + t_alloc, use_lp=True, max_lp_iters=100)
            
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"    *** New best: {best_score:.6f} at n={n}")
    
    # Phase D: Final intensive polish
    print("\n=== Phase D: Final polish ===")
    
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 10
        print(f"  Polishing best (score={best_score:.6f}, n={len(best_seq)}) with {remaining:.0f}s")
        
        seq, score = optimize_from_start(best_seq, deadline - 10.0, use_lp=True, max_lp_iters=500)
        
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"    *** Final best: {best_score:.6f}")
    
    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]
    
    # Fallback
    print("Falling back to ramp sequence.")
    n = 384
    a = np.arange(n, dtype=np.float64) / n + 0.1
    return [float(x) for x in a]
