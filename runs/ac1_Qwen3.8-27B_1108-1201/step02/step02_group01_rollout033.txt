import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum (default sqrt(2n))."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.ones(len(a)) * S / len(a)
    return a * S / s


def soft_max_objective_and_grad(a, beta):
    """
    Compute the log-sum-exp objective and its gradient.
    Returns (objective_value, gradient).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    # Numerically stable log-sum-exp
    exp_c = np.exp(beta * (c - c_max))
    total = np.sum(exp_c)
    obj = c_max + np.log(total) / beta
    weights = exp_c / total

    # Gradient: d/da_j [log sum_k exp(beta*c_k)] = beta * sum_k weights[k] * dc_k/da_j
    # dc_k/da_j = 2*a[k-j] if 0<=k-j<n, else 0
    # So grad[j] = 2*beta * sum_{k: 0<=k-j<n} weights[k] * a[k-j]
    grad = np.zeros(n)
    for j in range(n):
        k_min = j
        k_max = min(j + n - 1, 2 * n - 2)
        if k_min <= k_max:
            # sum_{k=k_min}^{k_max} weights[k] * a[k-j]
            l_range = np.arange(k_min - j, k_max - j + 1)
            grad[j] = 2.0 * beta * np.dot(weights[k_min:k_max + 1], a[l_range])

    return obj, grad


def nesterov_smoothed_descent(a, time_limit, beta_start=5.0, beta_end=200.0,

    lr = 0.1  # Initial step size (in normalized units)
    y = a.copy()  # Nesterov extrapolation point
    stall = 0
    max_stall = 50

    step = 0
    n_total_steps = 10000  # Will be limited by time

        # Linearly anneal beta
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start * (beta_end / beta_start) ** progress

        # Evaluate at extrapolation point
        obj_y, grad_y = soft_max_objective_and_grad(y, beta)

        # Nesterov update

        y = np.maximum(y, 0.0)
        y = normalize_seq(y, S)

def build_lp_matrix_sparse(a):
    """Build convolution constraint matrix for LP (dense for moderate n)."""
    n = len(a)
    # A has shape (2n-1, n), A[k,j] = a[k-j] if valid else 0
    # Use the structure: row k has non-zeros at columns max(0,k-n+1)..min(k,n-1)
    # For n up to ~1000, dense is fine
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_min:k - j_max + 1][::-1]
    return A


def lp_solve(a, M):
    """
    Solve LP: maximize sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    """
    n = len(a)
    if n > 2000:
        return None  # Too large for dense LP
    A = build_lp_matrix_sparse(a)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 10})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_refine(a, time_limit, verbose=True):
    """
    Improved LP-based iterative refinement with adaptive M and line search.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)

    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))

        if time.monotonic() - t0 > time_limit:
            break
        if n > 1500:
            break  # LP too slow for very large n

        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        g = lp_solve(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        g_norm = normalize_seq(g, S)

        # Line search for optimal blend

        best_conv_t = M

def water_filling_step(a, n_iterations=50, verbose=False):
    """
    Water-filling heuristic: repeatedly reduce elements that contribute most
    to the peak and redistribute to elements with low autocorrelation contribution.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))

    for it in range(n_iterations):
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        # Find elements that contribute most to the peak
        # conv[peak_idx] = sum_i a[i] * a[peak_idx - i]
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        # Reduce top contributors
        top_k = max(2, n // 20)
        top_idx = np.argsort(contribs)[-top_k:]

        # Find elements with low contribution (good to increase)
        low_idx = np.argsort(contribs)[:n // 4]

        # Move some mass from top to low
        for _ in range(3):
            candidate = a.copy()
            # Reduce top contributors by a small fraction
            reduction_total = 0.0
            for idx in top_idx:
                frac = 0.05 + 0.1 * np.random.random()
                amt = candidate[idx] * frac
                candidate[idx] -= amt
                reduction_total += amt

            # Distribute to low contributors
            n_inc = len(low_idx)
            if n_inc > 0:
                per_inc = reduction_total / n_inc
                for idx in low_idx:
                    candidate[idx] += per_inc

            candidate = np.maximum(candidate, 0.0)
            candidate = normalize_seq(candidate, S)
            cmax = np.max(fft_convolve(candidate, candidate))

            if cmax < best_conv_max:
                best_conv_max = cmax
                best_a = candidate.copy()
                a = candidate

    return best_a

        # U-shaped: high at edges, low in middle
        center = n / 2
        width = n * (0.3 + 0.3 * rng.random())
        a = 1.0 + 0.5 * np.cosh((t - center) / width)
        a = a / np.max(a)
    elif init_type == 'bimodal':
        c1 = n * (0.2 + 0.1 * rng.random())
        c2 = n * (0.7 - 0.1 * rng.random())
        sigma = n * 0.08
        a = np.sqrt(np.exp(-0.5 * ((t - c1) / sigma) ** 2) +
                     np.exp(-0.5 * ((t - c2) / sigma) ** 2) + 0.1)
    elif init_type == 'random_smooth':
        a = 1.0 + 0.3 * rng.standard_normal(n)
        # Smooth it
        kernel_size = max(3, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    elif init_type == 'multi_cos':
        a = np.ones(n)
        for _ in range(3 + int(rng.integers(0, 4))):
            k = 1 + int(rng.integers(0, n // 8))
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t / n + phase)
    else:
        a = np.ones(n)

    a = np.clip(a, 0.01, None)
    return a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining smoothed subgradient descent,
    LP refinement, coordinate descent, and water-filling.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            print(f"  *** New best: score={s:.6f}, n={len(a_clip)} ***")
            return True
        return False

    # Get provided sequence if available
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)

    # === Phase 1: Smoothed Nesterov descent on provided sequence ===
    print("\n=== Phase 1: Smoothed Nesterov descent ===")
    if provided_seq is not None:
        n = len(provided_seq)
        t_limit = min(300.0, (deadline - time.monotonic()) * 0.25)
        print(f"  Nesterov on provided (n={n}), time={t_limit:.0f}s")
        result = nesterov_smoothed_descent(provided_seq, t_limit,
                                           beta_start=5.0, beta_end=150.0,
                                           verbose=True, rng=rng)
        update_best(result)

    # === Phase 2: LP refinement ===
    print("\n=== Phase 2: LP refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(200.0, (deadline - time.monotonic()) * 0.2)
        if n <= 1000:
            print(f"  LP refine (n={n}), time={t_limit:.0f}s")
            result = lp_refine(best_seq, t_limit, verbose=True)
            update_best(result)
        else:
            print(f"  Skipping LP (n={n} too large)")

    # === Phase 3: Water filling + Nesterov on best ===
    print("\n=== Phase 3: Water filling + Nesterov ===")
    if best_seq is not None:
        n = len(best_seq)
        # Water filling
        result = water_filling_step(best_seq, n_iterations=100)
        update_best(result)

        # More Nesterov with higher beta range
        t_limit = min(200.0, (deadline - time.monotonic()) * 0.15)
        result = nesterov_smoothed_descent(best_seq, t_limit,
                                           beta_start=20.0, beta_end=500.0,
                                           verbose=True, rng=rng)
        update_best(result)

    # === Phase 4: Try different lengths from provided ===
    print("\n=== Phase 4: Multi-length exploration ===")
    if provided_seq is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 100:
            n_options = [384, 512, 640, 768, 896, 1024]
            # Filter to reasonable ones
            n_options = [x for x in n_options if x >= 128]
            t_per = min(40.0, remaining / len(n_options))

            for n_try in n_options[:3]:
                if time.monotonic() > deadline - 80:
                    break
                a0 = interpolate_sequence(provided_seq, n_try)
                a0 = np.clip(a0, 0.01, None)
                result = nesterov_smoothed_descent(a0, t_per,
                                                   beta_start=5.0, beta_end=100.0,
                                                   verbose=False, rng=rng)
                if local_score(result) < best_score:
                    update_best(result)
                    # Follow up with LP if n is small enough
                    if n_try <= 800 and time.monotonic() < deadline - 60:
                        result2 = lp_refine(result, min(30.0, (deadline - time.monotonic()) - 60),
                                            verbose=False)
                        update_best(result2)

    # === Phase 5: Diverse initializations ===
    print("\n=== Phase 5: Diverse starts ===")
    remaining = deadline - time.monotonic() - 60
    if remaining > 30:
        n_base = len(best_seq) if best_seq is not None else 512
        n_options = [n_base, max(256, n_base // 2), min(1024, n_base * 2)]
        init_types = ['cosine', 'u_shape', 'multi_cos', 'bimodal', 'random_smooth']

        t_per = max(15.0, remaining / (len(n_options) * len(init_types)))
        attempts = 0
        for n_try in n_options[:2]:
            for init_type in init_types[:3]:
                if time.monotonic() > deadline - 50:
                    break
                a0 = make_initialization(n_try, init_type, rng)
                result = nesterov_smoothed_descent(a0, min(t_per, 30.0),
                                                   beta_start=3.0, beta_end=80.0,
                                                   verbose=False, rng=rng)
                if local_score(result) < best_score:
                    update_best(result)
                attempts += 1

    # === Phase 6: Coordinate descent polish ===
    print("\n=== Phase 6: Coordinate descent polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 20:
            n = len(best_seq)
            if n <= 2000:
                t_limit = min(remaining, 120.0)
                print(f"  CD polish (n={n}), time={t_limit:.0f}s")
                result = coordinate_descent_polish(best_seq, t_limit, verbose=True)
                update_best(result)

    # === Phase 7: Final LP polish ===
    print("\n=== Phase 7: Final LP ===")
    if best_seq is not None and len(best_seq) <= 800:
        remaining = deadline - time.monotonic() - 20
        if remaining > 20:
            result = lp_refine(best_seq, remaining, verbose=True)
            update_best(result)

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, "
              f"local_score={best_score:.6f} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
