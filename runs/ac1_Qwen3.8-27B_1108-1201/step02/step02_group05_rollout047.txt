"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Subgradient descent with momentum and adaptive learning rate
2. Projected gradient (onto {a>=0, sum=1})
3. Multi-peak subgradient for smoother updates
4. Periodic random perturbations to escape local minima
5. LP-based refinement for final polish
6. Multiple sequence lengths and initializations
"""

import numpy as np
import time
import scipy.optimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_norm(a):
    """Objective with sum(a) assumed to be 1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

def full_gradient_at_peak(a, k):
    """Compute gradient of conv(a,a)[k] with respect to a."""
    n = len(a)
    idx1 = k - np.arange(n)
    idx2 = np.arange(n) - k
    grad = np.zeros(n)
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    grad[v1] += a[idx1[v1]]
    grad[v2] += a[idx2[v2]]
    return grad

def multi_peak_gradient(a, top_k=10, temperature=0.0):
    """Compute weighted subgradient from top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    
    if temperature > 0:
        # Soft attention over all peaks
        shifted = conv - conv.max()
        weights = np.exp(shifted / temperature)
        weights /= weights.sum()
        # Only use top-k for efficiency
        top_idx = np.argsort(conv)[::-1][:top_k]
        w_top = weights[top_idx]
        w_top /= w_top.sum()
        grad = np.zeros(n)
        for k, w in zip(top_idx, w_top):
            grad += w * full_gradient_at_peak(a, k)
        return grad
    else:
        # Hard: just average top-k
        top_idx = np.argsort(conv)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            grad += full_gradient_at_peak(a, k)
        return grad / top_k

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_optimize(a0, time_limit, top_k=10, lr_init=0.01, momentum=0.9):
    """Subgradient descent with momentum and adaptive lr."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_a = a.copy()
    best_obj = obj_norm(a)
    v = np.zeros(n)  # momentum
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    patience = 200
    
    while time.monotonic() - t0 < time_limit:
        grad = multi_peak_gradient(a, top_k=top_k)
        # Center the gradient (zero-sum direction)
        grad_centered = grad - grad.mean()
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad_centered
        
        # Trial step
        a_trial = a - lr * v
        a_trial = project_simplex(a_trial)
        
        obj_trial = obj_norm(a_trial)
        if obj_trial < best_obj - 1e-12:
            a = a_trial
            best_obj = obj_trial
            best_a = a.copy()
            no_improve = 0
            # Slightly increase lr if making good progress
            lr = min(lr * 1.05, 0.1)
        elif obj_trial < obj_norm(a) + 1e-10:
            # Accept if not too worse (annealing-like)
            a = a_trial
            no_improve += 1
            lr *= 0.98
        else:
            no_improve += 1
            lr *= 0.95
        
        if no_improve >= patience:
            # Random perturbation to escape
            noise = np.abs(np.random.randn(n)) * 0.05
            a_perturbed = a + noise
            a_perturbed = np.maximum(a_perturbed, 0)
            a_perturbed = a_perturbed / a_perturbed.sum()
            if obj_norm(a_perturbed) < best_obj:
                a = a_perturbed
                best_obj = obj_norm(a)
                best_a = a.copy()
            no_improve = 0
            lr = lr_init * 0.5
        
    return best_a, best_obj

def local_exchange_search(a, time_limit, max_rounds=500):
    """Local search by swapping mass between indices."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    
    for _ in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Compute contribution of each pair to conv[k_star]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        # Find high-contributing and low-contributing indices
        max_c = contrib.max()
        if max_c < 1e-15:
            break
        
        # Sort by contribution
        order = np.argsort(contrib)
        safe_indices = order[:20]  # low contribution
        danger_indices = order[-20:]  # high contribution
        
        found = False
        for di in danger_indices:
            if a[di] < 1e-8:
                continue
            for si in safe_indices:
                if si == di:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amount = a[di] * frac
                    if amount < 1e-10:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-12:
                        best_obj = o
                        a = a_trial
                        found = True
                        stuck = 0
                        break
            if found:
                break

def solve_lp(f, M, time_limit=30):
    """LP: find g >= 0 maximizing sum(g) s.t. conv(f,g)[k] <= M for all k."""
    n = len(f)
    size = 2 * n - 1
    
    # Build constraint matrix efficiently
    # conv(f,g)[k] = sum_j f[j] * g[k-j] <= M
    # For each k, this is a linear constraint on g
    
    # For n=512, this is a 1023 x 512 LP - might be slow
    # Use FFT-based check instead for large n
    
    if n > 256:
        return None  # Skip LP for large n
    
    A = np.zeros((size, n))
    for j in range(n):
        end = min(size, j + n)
        A[j:end, j] = f[:end - j]
    
    c = -np.ones(n)
    b = np.full(size, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', 
                                       options={'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Subgradient descent with Nesterov momentum and adaptive learning rate
2. Multi-peak subgradient (soft attention over hot convolution positions)
3. LP-based refinement for smaller sequences
4. Local exchange search for final polish
5. Multiple sequence lengths (512, 1024) and diverse initializations
6. Periodic random perturbations to escape local minima

import numpy as np
import time
import scipy.optimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gradient_at_peak(a, k):
    """Exact gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    idx1 = k - np.arange(n)
    idx2 = np.arange(n) - k
    grad = np.zeros(n)
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    grad[v1] += a[idx1[v1]]
    grad[v2] += a[idx2[v2]]
    return grad


def multi_peak_subgradient(a, top_k=20, temperature=0.0):
    """Weighted subgradient from top-k peaks with optional soft weighting."""
    n = len(a)
    conv = fast_convolve(a)
    if temperature > 0:
        shifted = conv - conv.max()
        weights = np.exp(shifted / temperature)
        top_idx = np.argsort(conv)[::-1][:top_k]
        w_top = weights[top_idx].copy()
        w_sum = w_top.sum()
        if w_sum < 1e-15:
            w_top = np.ones(top_k) / top_k
        else:
            w_top /= w_sum
    else:
        top_idx = np.argsort(conv)[::-1][:top_k]
        w_top = np.ones(top_k) / top_k

    grad = np.zeros(n)
    for k, w in zip(top_idx, w_top):
        if w > 1e-15:
            grad += w * gradient_at_peak(a, k)
    return grad


def subgradient_optimize(a0, time_limit, top_k=15, lr_init=0.02, momentum=0.9,
                         patience=150, temp_base=0.0):
    """Subgradient descent with momentum, adaptive lr, and periodic perturbation."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_obj = obj_norm(a)
    v = np.zeros(n)
    lr = lr_init
    beta = momentum
    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        # Adaptive temperature for soft subgradient
        temp = temp_base * (1.0 + 0.1 * np.sin(iter_count * 0.01)) if temp_base > 0 else 0.0

        grad = multi_peak_subgradient(a, top_k=top_k, temperature=temp)
        grad_c = grad - grad.mean()

        # Momentum
        v = beta * v + (1.0 - beta) * grad_c

        # Step
        a_trial = a - lr * v
        a_trial = project_simplex(a_trial)

        obj_trial = obj_norm(a_trial)
        current_obj = obj_norm(a)

        if obj_trial < best_obj - 1e-13:
            a = a_trial
            best_obj = obj_trial
            best_a = a.copy()
            no_improve = 0
            # Slightly increase lr on success
            lr = min(lr * 1.02, 0.15)
        elif obj_trial < current_obj + 0.005 * best_obj:
            # Accept small regressions (simulated annealing flavor)
            a = a_trial
            no_improve += 1
            lr *= 0.99
        else:
            no_improve += 1
            lr *= 0.95

        # Periodic perturbation
        if no_improve >= patience:
            # Multiple perturbation strategies
            strategies = [
                lambda: a + np.abs(np.random.randn(n)) * 0.03,
                lambda: a * (1.0 + 0.1 * np.random.randn(n)),
                lambda: a + 0.02 * np.random.randn(n),
            ]
            accepted = False
            for _ in range(5):
                strat = strategies[np.random.randint(len(strategies))]
                a_p = np.maximum(strat(), 0)
                sp = a_p.sum()
                if sp > 1e-12:
                    a_p /= sp
                    if obj_norm(a_p) < best_obj:
                        a = a_p
                        best_obj = obj_norm(a)
                        best_a = a.copy()
                        accepted = True
                        break
            no_improve = 0
            lr = lr_init * 0.7
            if not accepted:
                # Gradual increase in perturbation
                lr = lr_init * (0.5 + 0.3 * (iter_count % 10) / 10.0)

        # Safety: ensure we're not stuck with zero
        if a.sum() < 1e-10:
            a = np.ones(n) / n

    return best_a, best_obj


def solve_lp_direction(f, M, time_limit=20):
    """LP: find g >= 0 maximizing sum(g) s.t. conv(f,g)[k] <= M."""
    n = len(f)
    size = 2 * n - 1
    if n > 768:
        return None

    A = np.zeros((size, n))
    for j in range(n):
        end = min(size, j + n)
        A[j:end, j] = f[:end - j]

    c = -np.ones(n)
    b = np.full(size, M)
    bounds = [(0.0, None)] * n

    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                     method='highs', options={'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None


def lp_optimize(f0, time_limit, top_k=10):
    """LP-based iterative optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones_like(f) / len(f)
    else:
        f /= s

    best_f = f.copy()
    best_obj = obj_norm(f)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(f, M, time_limit=min(15, time_limit * 0.1))
        if g is None:
            # Fallback: use subgradient direction
            grad = multi_peak_subgradient(f, top_k=top_k)
            grad_c = grad - grad.mean()
            g = project_simplex(f - 0.01 * grad_c)

        if np.allclose(g, f, atol=1e-12):
            no_improve += 5
        else:
            # Line search
            best_t = 0.0
            best_t_obj = obj_norm(f)
            for t in [0.1, 0.25, 0.5, 0.75, 0.9]:
                a_t = (1.0 - t) * f + t * g
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st < 1e-12:
                    continue
                a_t /= st
                o_t = obj_norm(a_t)
                if o_t < best_t_obj:
                    best_t = t
                    best_t_obj = o_t
                    best_g = a_t.copy()

            if best_t > 0 and best_t_obj < best_obj - 1e-12:
                f = best_g
                best_obj = best_t_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve >= 50:
            break

    return best_f, best_obj


def local_exchange_search(a, time_limit, max_rounds=2000):
    """Local search: move mass from high-contributing to low-contributing indices."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0

    for _ in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        if contrib.max() < 1e-15:
            break

        order = np.argsort(contrib)
        safe = order[:25]
        danger = order[-25:]

        found = False
        for di in danger:
            if found:
                break
            if a[di] < 1e-9:
                continue
            for si in safe:
                if found:
                    break
                if si == di:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.35, 0.5]:
                    amount = a[di] * frac
                    if amount < 1e-12:
                        continue
                    a_tr = a.copy()
                    a_tr[di] -= amount
                    a_tr[si] += amount
                    a_tr = np.maximum(a_tr, 0)
                    ts = a_tr.sum()
                    if ts < 1e-12:
                        continue
                    a_tr /= ts
                    o = obj_norm(a_tr)
                    if o < best_obj - 1e-12:
                        best_obj = o
                        a = a_tr
                        found = True
                        stuck = 0
                        break
        if not found:
            stuck += 1
            if stuck > 200:
                break

    return a, best_obj


def make_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.35, 0.5, 0.7]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-12:
            inits[f'gauss_{sf}'] = g / s

    # Chirp
    chirp = 0.5 + 0.5 * np.cos(2 * np.pi * x * x / (2.0 * n))
    chirp = np.maximum(chirp, 0.01)
    inits['chirp'] = chirp / chirp.sum()

    # Hann
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    else:
        hann = np.ones(n)
    hann = np.maximum(hann, 0)
    sh = hann.sum()
    if sh > 1e-12:
        inits['hann'] = hann / sh

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Binary-ish
    b = np.where(rng.random(n) > 0.5, 0.8, 0.2)
    inits['binary'] = b / b.sum()

    # Edges
    edges = np.zeros(n)
    m = max(1, n // 3)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    # Parent
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0)
        s = pc.sum()
        if s > 1e-10:
            inits['parent'] = pc / s

    return inits


def get_parent(n):
    """Get parent sequence resized to length n, normalized."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry: multi-phase optimization combining subgradient descent,
    LP refinement, and local search. Tries lengths 512 and 1024.
    """
    rng = np.random.default_rng(seed)
    total = float(budget_s)
    deadline = time.monotonic() + total - 30.0

    best_sequence = None
    best_score = float('inf')

    lengths = [512, 1024]
    parent_512 = get_parent(512)
    parent_1024 = get_parent(1024)

    # Phase 1: Subgradient from diverse starts (55% budget)
    phase1_end = time.monotonic() + (total - 30.0) * 0.55
    print("Phase 1: Subgradient optimization from diverse starts...")

    candidates = []
    for n in lengths:
        par = parent_512 if n == 512 else parent_1024
        inits = make_initializations(n, rng, par)
        for name, a0 in inits.items():
            candidates.append((n, name, a0))

    # Prioritize parent, then shorter n
    def sort_key(item):
        n, name, _ = item
        if name == 'parent':
            return (0, n)
        if name in ('hann', 'gauss_0.35', 'gauss_0.5'):
            return (1, n)
        return (2, n)

    candidates.sort(key=sort_key)

    n_cands = len(candidates)
    phase1_time = max(1.0, phase1_end - time.monotonic())
    time_per_cand = max(8.0, phase1_time / n_cands)

    results = []
    for i, (n, name, a0) in enumerate(candidates):
        remaining = deadline - time.monotonic()
        if remaining < 90:
            break
        t_alloc = min(time_per_cand, 80.0)
        t_alloc = min(t_alloc, remaining - 90.0)
        if t_alloc < 5.0:
            break

        # Use different top_k depending on n
        tk = 20 if n <= 512 else 30
        a_opt, obj_opt = subgradient_optimize(a0, time_limit=t_alloc, top_k=tk,
                                              lr_init=0.02, momentum=0.9)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Subgrad: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 1 best: {best_score:.6f}")

    # Phase 2: LP refinement + deeper subgradient (30% budget)
    phase2_end = min(time.monotonic() + (total - 30.0) * 0.30, deadline - 90)
    print("Phase 2: LP refinement and deeper optimization...")

    results.sort(key=lambda x: x[0])
    top_cands = results[:8]

    for score, n, name, a0 in top_cands:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        t_alloc = min(remaining * 0.15, 50.0)
        if t_alloc < 5.0:
            break

        if n <= 600:
            # LP-based refinement
            a_lp, obj_lp = lp_optimize(a0, time_limit=t_alloc, top_k=15)
            score_lp = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_sequence = a_lp.copy()
                print(f"  LP: n={n}, {name}, score={score_lp:.6f}")
        else:
            # Deeper subgradient for larger n
            a_sg, obj_sg = subgradient_optimize(a0, time_limit=t_alloc, top_k=30,
                                                lr_init=0.01, momentum=0.93)
            score_sg = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(score_sg) and score_sg < best_score:
                best_score = score_sg
                best_sequence = a_sg.copy()
                print(f"  Deep SG: n={n}, {name}, score={score_sg:.6f}")

    # Phase 3: Intensive polish (remaining time)
    print("Phase 3: Intensive polish on best candidate...")
    if best_sequence is not None:
        polish_end = deadline - 15.0
        polish_iter = 0
        while time.monotonic() < polish_end:
            polish_iter += 1
            remaining = polish_end - time.monotonic()
            if remaining < 15:
                break

            n_best = len(best_sequence)

            # Alternate methods
            method = polish_iter % 3

            if method == 0:
                # Subgradient with fine parameters
                t_alloc = min(remaining * 0.35, 60.0)
                a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc,
                                                      top_k=25, lr_init=0.008,
                                                      momentum=0.95, patience=100)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish SG: score={score:.6f}")

            elif method == 1:
                # Local exchange
                t_alloc = min(remaining * 0.3, 45.0)
                a_ls, obj_ls = local_exchange_search(best_sequence, time_limit=t_alloc)
                score = float(evaluate_sequence(a_ls.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_ls.copy()
                    print(f"  Polish LS: score={score:.6f}")

            else:
                # LP (only for n <= 600)
                if n_best <= 600:
                    t_alloc = min(remaining * 0.3, 45.0)
                    a_lp, obj_lp = lp_optimize(best_sequence, time_limit=t_alloc, top_k=15)
                    score = float(evaluate_sequence(a_lp.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_lp.copy()
                        print(f"  Polish LP: score={score:.6f}")
                else:
                    # Additional subgradient with different params
                    t_alloc = min(remaining * 0.3, 45.0)
                    a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc,
                                                          top_k=40, lr_init=0.005,
                                                          momentum=0.92, patience=80,
                                                          temp_base=0.01)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Polish SG2: score={score:.6f}")

    # Final validation
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
