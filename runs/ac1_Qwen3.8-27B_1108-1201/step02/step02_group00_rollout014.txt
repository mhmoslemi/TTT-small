"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth log-sum-exp objective
with annealed temperature, Adam optimization with simplex projection,
and see-saw QP refinement.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_and_grad(conv, tau):
    """Compute smooth max of conv and its gradient w.r.t. conv elements."""
    # For numerical stability
    c_max = conv.max()
    exp_c = np.exp((conv - c_max) / tau)
    log_sum = np.log(exp_c.sum()) + c_max / tau
    smooth_max = tau * log_sum
    # Gradient w.r.t. conv elements
    weights = exp_c / exp_c.sum()
    return smooth_max, weights

def compute_smooth_obj_and_grad(a, tau):
    """Compute smooth objective and gradient for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    smooth_max, weights = smooth_max_and_grad(conv, tau)
    obj = 2.0 * n * smooth_max
    # Gradient: d(obj)/d(a_i) = 2n * sum_k weights_k * a_{k-i} (when 0 <= k-i < n)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-15:
            continue
        # For lag k, the contribution to a_i is a_{k-i} for i in [max(0,k-n+1), min(n-1,k)]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            grad[i] += 2.0 * n * weights[k] * a[k - i]
    return obj, grad

# Vectorized version
def compute_smooth_obj_and_grad_v(a, tau, top_k=50):
    """Vectorized smooth objective and gradient."""
    n = len(a)
    conv = fast_convolve(a)
    # Use top-k for approximation if conv is very long
    if top_k < len(conv):
        idx = np.argpartition(conv, -top_k)[-top_k:]
        conv_k = conv[idx].copy()
        # Mask
        mask = np.zeros(len(conv))
        mask[idx] = 1
    else:
        conv_k = conv
        idx = np.arange(len(conv))
        mask = np.ones(len(conv))
    
    c_max = conv_k.max()
    exp_c = np.exp((conv_k - c_max) / tau)
    log_sum = np.log(exp_c.sum()) + c_max / tau
    smooth_max = tau * log_sum
    obj = 2.0 * n * smooth_max
    
    weights_k = exp_c / exp_c.sum()
    
    # Gradient using the top-k weights
    grad = np.zeros(n)
    for j in range(len(idx)):
        w = weights_k[j]
        if w < 1e-15:
            continue
        k = int(idx[j])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1)
        grad[i_range] += 2.0 * n * w * a[k - i_range]
    
    return obj, grad

def adam_simplex(a, obj_grad_fn, time_limit, lr=0.01, beta1=0.9, beta2=0.999, 
                 eps=1e-8, tau=None, top_k=50, print_interval=500):
    """Adam optimizer with simplex projection."""
    n = len(a)
    a = project_simplex(a.copy())
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    best_obj = float('inf')
    best_a = a.copy()
    obj, grad = obj_grad_fn(a)
    best_obj = obj
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_new = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))
        a = a_new
        obj, grad = obj_grad_fn(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        step += 1
        if step % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam step {step}, obj={obj:.8f}, best={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

def optimize_annealed(a0, time_limit, tau_start=0.1, tau_end=0.001, 
                      n_restarts=3, lr=0.05, top_k=50, print_info=True):
    """Anneal temperature from tau_start to tau_end with multiple restarts."""
    n = len(a0)
    a = project_simplex(a0.copy())
    best_obj = float('inf')
    best_a = a.copy()
    t0 = time.monotonic()
    
    # Anneal temperature geometrically
    n_steps_anneal = 20
    taus = np.geomspace(tau_start, tau_end, n_steps_anneal)
    
    for tau in taus:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        # Allocate time: more time at smaller tau
        time_for_tau = remaining / (n_steps_anneal - int(np.log(tau/tau_start)/np.log(tau_end/tau_start)))
        time_for_tau = min(time_for_tau, remaining * 0.3)
        
        def obj_grad_fn(a, tau=tau):
            return compute_smooth_obj_and_grad_v(a, tau, top_k=top_k)
        
        a_opt, obj_opt = adam_simplex(a, obj_grad_fn, time_limit=time_for_tau, lr=lr, top_k=top_k)
        
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
        
        a = a_opt
        if print_info:
            print(f"  tau={tau:.6f}, obj={obj_opt:.8f}, best={best_obj:.8f}, elapsed={time.monotonic()-t0:.1f}s")
    
    # Final evaluation with hard max
    conv = fast_convolve(best_a)
    hard_obj = 2.0 * n * conv.max()
    
    return best_a, hard_obj

def seesaw_optimize(a0, time_limit, n_inner=50, top_k=20):
    """
    See-saw optimization:
    - Fix p (weights over lags), solve QP: min a^T Q a s.t. a>=0, sum(a)=1
    - Update p: put mass on argmax of conv(a,a)
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    best_obj = float('inf')
    best_a = a.copy()
    t0 = time.monotonic()
    
    outer_iter = 0
    while time.monotonic() - t0 < time_limit:
        # Update p: compute conv and get weights
        conv = fast_convolve(a)
        # Smooth weights
        c_max = conv.max()
        exp_c = np.exp(-(conv - c_max) / 0.01)  # temperature for p
        p = exp_c / exp_c.sum()
        
        # Solve inner QP: min a^T Q a s.t. a >= 0, sum(a) = 1
        # Q_{ij} = p_{i+j}
        # Use projected gradient on the QP
        a_qp = a.copy()
        for _ in range(n_inner):
            # Gradient of a^T Q a = 2 * Q @ a
            # Q @ a: (Q @ a)_i = sum_j p_{i+j} * a_j
            Qa = np.zeros(n)
            for k in range(1, min(2*n-1, top_k*2)):  # Only use top peaks
                if p[k] < 1e-10:
                    continue
                # (Q@ a)_i gets contribution p[k] * a[j] where i+j=k, i.e., j=k-i
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    Qa[i] += p[k] * a_qp[k - i]
            
            grad_qp = 2 * Qa
            # Projected gradient step
            lr_qp = 0.01
            a_qp = project_simplex(a_qp - lr_qp * grad_qp)
        
        conv_qp = fast_convolve(a_qp)
        obj_qp = 2.0 * n * conv_qp.max()
        
        if obj_qp < best_obj:
            best_obj = obj_qp
            best_a = a_qp.copy()
        
        # Blend
        a = project_simplex(0.5 * a + 0.5 * a_qp)
        outer_iter += 1
        
        if outer_iter % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SeeSaw iter {outer_iter}, obj={obj_qp:.8f}, best={best_obj:.8f}, elapsed={elapsed:.1f}s")

    smax, weights = smooth_max_weights(conv, tau)
    obj = 2.0 * n * smax

    grad = np.zeros(n)
    sig_thresh = weights.max() * 1e-5
    sig_idx = np.where(weights > sig_thresh)[0]
    for k_int in sig_idx:
        k = int(k_int)
        w = weights[k_int]
        if w < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        grad[i_vals] += 2.0 * n * w * a[k - i_vals]

    return obj, grad


def softmax_from_z(z):
    """Convert z to simplex via softmax."""
    z = z - z.max()
    exp_z = np.exp(z)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lbfgs_obj_grad_z(z, tau):
    """
    Objective and gradient w.r.t. z (log-parameterization of simplex).
    a = softmax(z), objective = 2n * smooth_max(conv(a,a)).
    d(obj)/dz_i = a_i * (grad_i - sum_j grad_j * a_j)
    """
    a = softmax_from_z(z)
    n = len(a)
    obj, grad_a = simplex_obj_grad(a, tau)
    # Gradient w.r.t. z
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    return obj, grad_z


def opt_lbfgs_z(z0, tau, time_limit, maxiter=300):
    """Run L-BFGS on z-parameterization."""
    t0 = time.monotonic()
    state = {'stop': False}

    def cb(xk):
        if time.monotonic() - t0 > time_limit:
            state['stop'] = True
            return True
        return None

    n = len(z0)
    bounds = [(-20.0, 20.0)] * n
    res = minimize(lbfgs_obj_grad_z, z0, method='L-BFGS-B',
                   jac=True, bounds=bounds, callback=cb,
                   options={'maxiter': maxiter, 'ftol': 1e-10,
                            'gtol': 1e-8, 'maxls': 40})
    a_opt = softmax_from_z(res.x)
    conv = fast_convolve(a_opt)
    hard_obj = 2.0 * len(a_opt) * conv.max()
    return a_opt, hard_obj


def opt_mirror_descent(a0, tau, time_limit, eta_init=0.3, eta_decay=0.9995):
    """Mirror descent (multiplicative weights) on smooth objective."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 1e-10).astype(np.float64))
    eta = eta_init
    t0 = time.monotonic()
    best_obj = float('inf')
    best_a = a.copy()
    step = 0

    while time.monotonic() - t0 < time_limit:
        obj, grad = simplex_obj_grad(a, tau)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Multiplicative update
        log_ratio = -eta * grad
        log_ratio -= log_ratio.max()
        a_new = a * np.exp(log_ratio)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = np.maximum(a_new, 1e-12)
        a = a / a.sum()

        eta = max(0.01, eta * eta_decay)
        step += 1

    conv = fast_convolve(best_a)
    return best_a, 2.0 * n * conv.max()


def anneal_optimize(a0, time_limit, tau_schedule=None, use_lbfgs=True,
                    use_mirror=True, lbfgs_frac=0.5):
    """
    Anneal temperature and alternate between L-BFGS and mirror descent.
    Returns best (a_normalized, hard_objective).
    """
    n = len(a0)
    if tau_schedule is None:
        tau_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    best_obj = float('inf')
    best_a = a.copy()
    t0 = time.monotonic()

    for i, tau in enumerate(tau_schedule):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break
        # Allocate more time to smaller tau
        frac = 0.7 + 0.3 * (i / len(tau_schedule))
        time_this = remaining * frac / (len(tau_schedule) - i)
        time_this = min(time_this, remaining - 2.0)
        if time_this < 2.0:
            time_this = remaining - 1.0

        # Mirror descent phase
        if use_mirror and time_this > 3.0:
            t_md = time_this * (1.0 - lbfgs_frac) if use_lbfgs else time_this
            a_md, obj_md = opt_mirror_descent(a, tau, time_limit=t_md,
                                              eta_init=0.2, eta_decay=0.9995)
            conv_md = fast_convolve(a_md)
            hard_md = 2.0 * n * conv_md.max()
            if hard_md < best_obj:
                best_obj = hard_md
                best_a = a_md.copy()
            a = a_md

        # L-BFGS phase
        if use_lbfgs and (time.monotonic() - t0) < time_limit - 3.0:
            t_lb = time_limit - (time.monotonic() - t0) - 2.0
            if t_lb > 3.0:
                z0 = np.log(np.maximum(a, 1e-15))
                z0 -= z0.mean()
                a_lb, obj_lb = opt_lbfgs_z(z0, tau, time_limit=t_lb, maxiter=200)
                conv_lb = fast_convolve(a_lb)
                hard_lb = 2.0 * n * conv_lb.max()
                if hard_lb < best_obj:
                    best_obj = hard_lb
                    best_a = a_lb.copy()
                a = a_lb

        elapsed = time.monotonic() - t0
        print(f"    tau={tau:.6f}, best={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()

    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    for sf in [0.1, 0.2, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.1f}'] = g

    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    bump = np.exp(-x / (0.2 * n))
    bump /= bump.sum()
    inits['bump_l'] = bump
    inits['bump_r'] = bump[::-1].copy()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits['parent'] = p
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pp_s = pp.sum()
                if pp_s > 1e-10:
                    inits['parent_ds'] = pp / pp_s
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pp_s = pp.sum()
                if pp_s > 1e-10:
                    inits['parent_up'] = pp / pp_s

    return inits


def peak_targeted_refine(a0, time_limit, n_peaks=5):
    """
    Targeted refinement: identify peak lags and try to reduce them
    via localized mirror descent with concentrated weights.
    """
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0).astype(np.float64))
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for _range in range(10):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(a)
        # Get top peak lags
        peak_lags = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_vals = conv[peak_lags]
        
        # Create targeted weights concentrated on peaks
        tau_target = (peak_vals.max() - peak_vals.min() + 1e-10) * 0.1
        if tau_target < 0.001:
            tau_target = 0.001
        
        # Run short mirror descent with small eta targeting peaks
        eta = 0.05
        steps = 200
        a_work = a.copy()
        
        for _st in range(steps):
            if time.monotonic() - t0 > time_limit:
                break
            # Focused gradient: only top peaks
            grad = np.zeros(n)
            for k_int in peak_lags:
                k = int(k_int)
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                i_vals = np.arange(i_start, i_end + 1)
                grad[i_vals] += 2.0 * n * a_work[k - i_vals]
            
            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad = grad / gnorm

            log_ratio = -eta * grad
            log_ratio -= log_ratio.max()
            a_new = a_work * np.exp(log_ratio)
            s = a_new.sum()
            if s < 1e-15:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / s
            a_work = np.maximum(a_new, 1e-12)
            a_work = a_work / a_work.sum()

        obj_work = 2.0 * n * fast_convolve(a_work).max()
        if obj_work < best_obj:
            best_obj = obj_work
            best_a = a_work.copy()
            a = a_work
        else:
            # Decay peak targets (recompute)
            a = project_simplex(0.9 * a + 0.1 * a_work)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization using smooth log-sum-exp
    objective with L-BFGS and mirror descent, annealed temperature,
    and peak-targeted refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent available: length={len(p)}, score targets parent")
    except Exception:
        pass

    # Phase 1: Optimize parent sequence with annealed smooth objective
    print("=== Phase 1: Parent Optimization ===")
    if parent is not None:
        n_parent = len(parent)
        t_alloc = min(400.0, deadline - time.monotonic() - 100.0)
        if t_alloc > 30:
            a_opt, obj_opt = anneal_optimize(
                parent, time_limit=t_alloc,
                tau_schedule=[0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001],
                use_lbfgs=True, use_mirror=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Parent optimized: score={score:.6f}")

    # Phase 2: Scan other lengths and initializations
    print("=== Phase 2: Length/Init Scan ===")
    lengths_to_try = [128, 192, 256, 320, 384, 512, 768, 1024]
    # Remove parent length to avoid redundancy
    if parent is not None:
        lengths_to_try = [l for l in lengths_to_try if l != len(parent)] or [len(parent)]
    
    scan_deadline = deadline - 400.0
    results = []

    for n in lengths_to_try:
        if time.monotonic() > scan_deadline:
            break
        inits = generate_initializations(n, parent, rng)
        
        # Prioritize initializations
        priority = ['parent', 'parent_ds', 'parent_up', 'uniform', 'ramp_inc',
                    'gauss_0.2', 'bimodal', 'random', 'triangle']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:5]  # Limit inits per length

        time_per_init = max(15.0, min(60.0, 300.0 / (len(lengths_to_try) * 4)))

        for name in init_names:
            if time.monotonic() > scan_deadline:
                break
            t_alloc = min(time_per_init, scan_deadline - time.monotonic() - 20)
            if t_alloc < 10:
                break

            a0 = inits[name]
            # Quick optimization
            a_opt, obj_opt = anneal_optimize(
                a0, time_limit=t_alloc,
                tau_schedule=[0.05, 0.02, 0.01, 0.005],
                use_lbfgs=True, use_mirror=True)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, score={score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 2 done. Best so far: {best_score:.6f}")

    # Phase 3: Deep optimization on top configurations
    print("=== Phase 3: Deep Optimization ===")
    top_configs = []
    seen_n = set()
    for score, n, name, a_opt in results:
        if n not in seen_n:
            top_configs.append((n, name, a_opt))
            seen_n.add(n)
        if len(top_configs) >= 5:
            break

    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    deep_deadline = deadline - 150.0
    remaining = deep_deadline - time.monotonic()
    n_configs = max(1, len(top_configs))
    time_per_deep = max(20.0, remaining / n_configs)

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deep_deadline:
            break
        t_alloc = min(time_per_deep, deep_deadline - time.monotonic() - 20)
        if t_alloc < 10:
            break

        # Deep anneal
        a_opt, obj_opt = anneal_optimize(
            a0, time_limit=t_alloc,
            tau_schedule=[0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0003],
            use_lbfgs=True, use_mirror=True, lbfgs_frac=0.4)

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep: n={n}, {name}, score={score:.6f}")

    # Phase 4: Peak-targeted refinement
    print("=== Phase 4: Peak-Targeted Refinement ===")
    if best_sequence is not None:
        refine_deadline = deadline - 60.0
        remaining = refine_deadline - time.monotonic()
        if remaining > 20:
            a_opt, obj_opt = peak_targeted_refine(
                best_sequence, time_limit=remaining * 0.6, n_peaks=5)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Peak refine: score={score:.6f}")

    # Phase 5: Final L-BFGS polish with very small temperature
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 10:
            # Multiple polish attempts
            for tau_final in [0.001, 0.0005, 0.0002]:
                if remaining < 5:
                    break
                t_alloc = remaining / 2.0
                z0 = np.log(np.maximum(best_sequence, 1e-15))
                z0 -= z0.mean()
                a_opt, _ = opt_lbfgs_z(z0, tau_final, time_limit=t_alloc, maxiter=300)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish tau={tau_final}: score={score:.6f}")
                remaining = deadline - time.monotonic() - 10.0

        # Mirror descent final
        if remaining > 10:
            a_opt, _ = opt_mirror_descent(best_sequence, 0.001,
                                          time_limit=remaining, eta_init=0.05,
                                          eta_decay=0.9999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  MD polish: score={score:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
