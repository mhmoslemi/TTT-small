    n = len(a)

    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_conv_matrix(f):
    """Build matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

**Approach: Direct optimization via projected subgradient method**

Since the problem is non-convex, we can't use a standard subgradient method directly. But we can use a "proximal" method:

minimize f(a) + (μ/2)||a - a_prev||^2
subject to a >= 0, sum(a) = 1

where f(a) = max_k (a*a)_k.

The proximal term makes the subproblem well-defined. The subgradient of f at a is 2*H_k a / ||H_k a||_0 (where k is the argmax, and H_k is the matrix for the kth convolution term). Actually, the subgradient of (a*a)_k at a is 2*H_k a (where H_k is the symmetric matrix representing the quadratic form).

So the proximal subgradient method would be:
1. Compute c = conv(a, a)
2. Find k* = argmax(c)
3. Compute g = 2 * H_{k*} a (the gradient of the active constraint)
4. Solve: a_new = argmin (1/2)||a - a_prev||^2 + (η/μ) * g^T a s.t. a >= 0, sum(a) = 1
   This is a simple QP that can be solved in closed form!

The closed form solution:
a_new[j] = max(0, a_prev[j] - (η/μ) * g[j] + λ)
where λ is chosen so that sum(a_new) = 1.

This means the subgradient of f(a) = max_k (a*a)_k at point a is:

Now, for the proximal subgradient method:
minimize f(a) + (μ/2)||a - a_prev||^2 s.t. a >= 0, sum(a) = 1

The solution is:
a_new = proj_{a>=0, sum=1}(a_prev - (η/μ) * g)

where proj is the projection onto the simplex {a >= 0, sum(a) = 1}.

The projection onto the simplex has a closed form:
1. Compute u = a_prev - (η/μ) * g
2. Sort u in decreasing order
3. Find the threshold τ such that sum(max(u_i - τ, 0)) = 1
4. a_new = max(u - τ, 0)

This is O(n log n) per step (due to the sort), which is fast.

"""
Hybrid optimization for minimizing peak autocorrelation of non-negative step sequences.
Uses LP-based direction finding with adaptive step sizes, proximal subgradient method
with simplex projection, spectral shaping, and multi-start across lengths.
Key improvements: adaptive line search, top-k peak subgradients, structured restarts,
and a fast projection-based subgradient phase.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    a_p = np.zeros(size)
    b_p = np.zeros(size)
    a_p[:n] = a
    b_p[:m] = b
    return irfft(rfft(a_p) * rfft(b_p), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_A(f):
    """Build matrix A where A @ b = conv(f, b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, timeout=5.0):
    """Solve: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, top_k=5):
    """Compute averaged subgradient of max_k (a*a)_k using top-k peaks."""
    n = len(a)
    c = fft_convolve(a, a)
    top_idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for ki in top_idx:
        ki = int(ki)
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        for m in range(lo, hi + 1):
            g[m] += 2.0 * a[ki - m]
    g /= float(len(top_idx))
    return g, c


def make_hann(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bartlett(n, rng):
    t = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
    a = 0.2 + 0.8 * w
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_rectangular(n, rng):
    a = np.ones(n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_cosine_mod(n, rng):
    """Sequence modulated by a slow cosine to create structure."""
    t = np.arange(n, dtype=np.float64)
    window = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
    freq = rng.uniform(0.5, 3.0)
    a = (0.5 + 0.5 * window) * (0.7 + 0.3 * np.cos(2.0 * np.pi * freq * t / n))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def optimize_lp_adaptive(f_init, time_limit, verbose=False):
    """LP-based optimization with adaptive step size and periodic restarts."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f / S  # Normalize sum to 1
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    step = 0.08
    min_step = 0.01
    max_step = 0.3

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))

        b = solve_lp(f, M, timeout=min(5.0, max(1.0, time_limit - (time.monotonic() - t0))))
        if b is None:
            break

        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b / Sb  # Normalize b to sum to 1

        # Adaptive line search: try multiple step sizes
        best_s = np.inf
        best_t = step
        candidates = [step, step * 0.5, step * 1.5, step * 0.25]
        for t_try in candidates:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.0, None)
            Sc = float(np.sum(f_cand))
            if Sc < 1e-10:
                continue
            f_cand = f_cand / Sc
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, None)
        Sc = float(np.sum(f))
        if Sc < 1e-10:
            break
        f = f / Sc
        it += 1

        if best_s < best_score - 1e-12:
            best_score = best_s
            best_f = f.copy()
            stall = 0
            # Increase step if improving
            step = min(max_step, step * 1.05)
        else:
            stall += 1
            # Decrease step if stalling
            step = max(min_step, step * 0.95)

        # Periodic restart to escape local minima
        if stall >= 50:
            rng_esc = np.random.default_rng(1000 + it * 31)
            # Structured restart: add a smooth perturbation
            t = np.arange(n, dtype=np.float64)
            pert = rng.uniform(0.05, 0.15) * np.cos(2.0 * np.pi * rng.uniform(1, 5) * t / n)
            f = f + pert
            f = np.clip(f, 0.001, None)
            f = f / np.sum(f)
            stall = 0
            step = 0.08

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [LP {elapsed:6.1f}s] it={it:4d} score={best_s:.6f} best={best_score:.6f} stall={stall}")

    return best_f, best_score


def optimize_subgrad(f_init, time_limit, top_k=5, verbose=False):
    """Proximal subgradient descent with simplex projection."""
    n = len(f_init)
    a = f_init.astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.85
    eta = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        g, c = compute_subgradient(a, top_k=top_k)
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn

        momentum = beta * momentum + (1.0 - beta) * g
        mn = np.linalg.norm(momentum)
        if mn > 1e-15:
            momentum = momentum / mn

        # Proximal step with simplex projection
        step_size = eta / np.sqrt(max(it, 1))
        a_new = a - step_size * momentum
        a_new = proj_simplex(a_new)

        s = score_fast(a_new)
        if s < best_score:
            best_score = s
            best_a = a_new.copy()
            eta = min(eta * 1.01, 2.0)
        else:
            eta = max(eta * 0.995, 0.001)

        a = a_new
        it += 1

        if it % 500 == 0:
            # Occasional perturbation
            rng_p = np.random.default_rng(it)
            a += 0.01 * rng_p.standard_normal(n)
            a = np.clip(a, 0.001, None)
            a = a / np.sum(a)

            print(f"    [SG {time.monotonic()-t0:6.1f}s] it={it} cur={s_cur:.6f} best={best_score:.6f} eta={eta:.4f}")

    return best_a, best_score


def local_search(f_init, time_limit, verbose=False):
    """Structured local search focusing on peak-contributing elements."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = f / np.sum(f)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    temp = 0.001
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        candidate = f.copy()
        c = fft_convolve(f, f)
        peak_idx = int(np.argmax(c))

        r = rng.random()
        if r < 0.5:
            # Peak-targeted: reduce contributors to the peak, increase others
            contribs = np.zeros(n)
            lo = max(0, peak_idx - (n - 1))
            hi = min(n - 1, peak_idx)
            for m in range(lo, hi + 1):
                contribs[m] = f[m] * f[peak_idx - m]
            total = contribs.sum()
            if total > 1e-15:
                contribs /= total
            # Reduce top contributors
            n_reduce = max(3, n // 15)
            top_mask = np.zeros(n, dtype=bool)
            top_mask[np.argsort(contribs)[-n_reduce:]] = True
            candidate[top_mask] *= np.exp(rng.normal(-0.05, 0.03, size=n_reduce))
            # Increase some others
            other_idx = np.where(~top_mask)[0]
            if len(other_idx) > 0:
                n_inc = max(3, n // 20)
                chosen = rng.choice(other_idx, size=min(n_inc, len(other_idx)), replace=False)
                candidate[chosen] *= np.exp(rng.normal(0.05, 0.03, size=len(chosen)))
        elif r < 0.7:
            # Smooth perturbation
            width = max(2, n // 10)
            start = int(rng.integers(0, max(1, n - width)))
            t = np.linspace(-1, 1, width)
            kernel = np.exp(-t**2 * 3)
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04) * kernel)
        elif r < 0.85:
            # Global smooth perturbation
            t = np.arange(n, dtype=np.float64)
            freq = rng.uniform(0.5, 3.0)
            candidate *= np.exp(rng.normal(0, 0.02) * np.cos(2 * np.pi * freq * t / n))
        else:
            # Random coordinate update
            n_coords = max(5, n // 10)
            idxs = rng.choice(n, size=n_coords, replace=False)
            candidate[idxs] *= np.exp(rng.normal(0, 0.05, size=n_coords))

        candidate = np.clip(candidate, 0.001, None)
        Sc = np.sum(candidate)
        candidate = candidate / Sc
        new_score = score_fast(candidate)

        cur_score = score_fast(f)
        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()
            f = candidate
        elif rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-15)):
            f = candidate

        temp *= 0.9997
        it += 1

        if verbose and it % 500 == 0:
            print(f"    [LS {time.monotonic()-t0:6.1f}s] it={it:5d} best={best_score:.6f} temp={temp:.6f}")

    return best_f, best_score


def spectral_refine(f_init, time_limit, verbose=False):
    """Try to flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(f_init)
    a = f_init.astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    it = 0

    # Target: flat power spectrum |A(k)|^2
    # We want to minimize the variance of |A(k)|^2 (normalized)
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        size = 1
        while size < 2 * n - 1:
            size <<= 1
        A = rfft(a, size)
        power = np.abs(A)**2
        # Normalize power
        pn = power / (np.sum(power) + 1e-30)
        # Gradient of variance of power w.r.t. a
        # d(power[k])/da[m] = 2 * Re(conj(A[k]) * dA[k]/da[m])
        # dA[k]/da[m] = exp(-2*pi*i*k*m/size)
        # This is expensive to compute directly, so use a simpler approach:
        # Perturb a in the direction that reduces power variance

        # Simple heuristic: compute the "imbalanced" frequencies and
        # modify a to reduce the power at those frequencies
        mean_p = np.mean(pn)
        # Frequencies with excess power
        excess = np.maximum(pn - mean_p, 0)
        if np.sum(excess) < 1e-10:
            break

        # Compute a "correction" in the frequency domain and transform back
        # Reduce excess power by subtracting a correction
        correction_freq = excess.copy()
        correction_freq[0] = 0  # Don't modify DC (preserves sum)
        # Transform correction to time domain
        corr_time = irfft(correction_freq, size)[:n]

        # Subtract correction (reduces elements that contribute to high-power freqs)
        alpha = 0.01
        a_new = a - alpha * corr_time
        a_new = np.clip(a_new, 0.001, None)
        Sc = np.sum(a_new)
        if Sc < 1e-10:
            break
        a_new = a_new / Sc

        s = score_fast(a_new)
        if s < best_score:
            best_score = s
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new2 = a - alpha * 0.1 * corr_time
            a_new2 = np.clip(a_new2, 0.001, None)
            a_new2 = a_new2 / np.sum(a_new2)
            s2 = score_fast(a_new2)
            if s2 < best_score:
                best_score = s2
                best_a = a_new2.copy()
                a = a_new2
            else:
                # No improvement, add small noise
                rng_s = np.random.default_rng(it)
                a += 0.005 * rng_s.standard_normal(n)
                a = np.clip(a, 0.001, None)
                a = a / np.sum(a)

        it += 1
        if verbose and it % 100 == 0:
            print(f"    [SPEC {time.monotonic()-t0:6.1f}s] it={it} best={best_score:.6f}")

    return best_a, best_score

        # Very final subgradient polish
        if time.monotonic() < deadline - 10:
            t_final_sg = min(20.0, deadline - time.monotonic() - 5.0)
            if t_final_sg > 5:
                f_fsg, s_fsg = optimize_subgrad(best_seq, t_final_sg, top_k=3, verbose=False)
                if s_fsg < best_score:
                    best_score = s_fsg
                    best_seq = f_fsg.copy()
                    print(f"  *** FINAL BEST (SG): {best_score:.6f} ***")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify with evaluate_sequence
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    print("WARNING: No valid sequence found, using fallback.")
    return [1.0] * 256
