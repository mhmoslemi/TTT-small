3. Update: a_new = project(a - lr * d)
4. Repeat

b) Parameterization: a[i] = exp(x[i]) / sum(exp(x[j])) (softmax parameterization)

Let me also think about the simplex projection more carefully. Given a vector v, the projection onto {a >= 0, sum(a) = 1} is:
a[i] = max(v[i] - tau, 0) where tau is chosen so that sum(a) = 1.

To find tau: sort v in decreasing order. For each possible support set S = {i: v[i] > tau}, we have:
sum_{i in S} (v[i] - tau) = 1
|S| * tau = sum_{i in S} v[i] - 1
tau = (sum_{i in S} v[i] - 1) / |S|

We need tau < min_{i in S} v[i] and tau >= max_{i not in S} v[i] (or -inf if S is empty).

This can be done in O(n log n) time.

import numpy as np
import scipy.optimize
import time

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    # Use the bisection method or sorting method
    # Sorting method:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    rho_star = np.where(cond)[0][-1] + 1
    theta = (cssv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - theta, 0)

def obj_and_grad_smoothed(a, beta):
    """Compute smoothed objective and gradient for sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Softmax weights
    conv_shifted = conv - conv.max()
    beta_conv = beta * conv_shifted
    beta_conv -= beta_conv.max()  # additional shift for numerical stability
    w = np.exp(beta_conv)
    w /= w.sum()
    
    # Objective: 2n * (conv.max() + log(sum(exp(beta*(conv - conv.max())))) / beta)
    log_sum_exp = np.log(np.sum(np.exp(beta * conv_shifted)))
    smooth_max = conv.max() + log_sum_exp / beta
    obj = 2.0 * n * smooth_max
    
    # Gradient: 2n * (1/beta) * sum_k w_k * 2*a[k-i] (if valid)
    # = 4n/beta * cross_correlate(w, a)
    corr = np.correlate(w, a, mode='full')
    grad = (4.0 * n / beta) * corr[:n]
    
    return obj, grad

def adam_simplex(a, beta, lr, n_steps, time_limit=None, t0=None):
    """Adam optimizer with simplex projection and adaptive beta."""
    n = len(a)
    a = a.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    if t0 is None:
        t0 = time.monotonic()
    
    for step in range(n_steps):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        obj, grad = obj_and_grad_smoothed(a, beta)
        
        t = step + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)
        
        # Check objective with actual max (not smoothed)
        actual_obj = 2.0 * n * fast_convolve(a_new).max()
        
        if actual_obj < best_obj:
            best_obj = actual_obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def active_set_lp(a, M, top_k=20):
    """Solve LP with only top_k active constraints."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Find top_k positions
    top_indices = np.argsort(conv)[::-1][:top_k]
    
    # Build constraint matrix (only for top_k constraints)
    A_ub = np.zeros((top_k, n))
    for idx, k in enumerate(top_indices):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[idx, i] = a[j]  # d(conv[k])/da[i] / 2 = a[k-i]
    
    # Wait, the constraint is conv(f, g)[k] <= M where f is fixed
    # conv(f, g)[k] = sum_i f[i] * g[k-i] (if 0 <= k-i < n)
    # So the constraint matrix has A_ub[k, i] = f[k-i] (if 0 <= k-i < n)
    
    A_ub = np.zeros((top_k, n))
    for idx, k in enumerate(top_indices):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[idx, i] = a[j]
    
    c = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                                      method='highs', options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    
    return a.copy()

def local_search_peak(a, time_limit, max_iters=10000):

import numpy as np
import scipy.optimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_val(a):
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_star = idx[-1] + 1
    theta = (cssv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - theta, 0)

def grad_of_smoothed_max(a, beta):
    """
    Gradient of LSE(beta * conv(a,a)) / beta with respect to a.
    Assumes a is on the simplex (sum = 1, a >= 0).
    """
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    shifted = conv - conv.max()
    bconv = beta * shifted
    bconv -= bconv.max()
    w = np.exp(bconv)
    w /= w.sum()
    
    # grad[i] = (2/beta) * sum_k w[k] * a[k-i] * 1[0<=k-i<n]
    # = (2/beta) * cross_correlate(w, a)[i]
    corr = np.correlate(w, a, mode='full')
    return (2.0 / beta) * corr[:n], conv.max() + np.log(np.sum(np.exp(beta * shifted))) / beta

def nesterov_simplex(a0, time_limit, beta_init=20, beta_end=200, lr_init=0.01):
    """Nesterov-accelerated projected gradient with adaptive beta."""
    n = len(a0)
    x = a0.copy()
    y = a0.copy()
    best_obj = 2.0 * n * fast_convolve(x).max()
    best_x = x.copy()
    
    t0 = time.monotonic()
    step = 0
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Adaptive beta: increase over time
        progress = min(1.0, step / max(1, int(time_limit * 10)))
        beta = beta_init + (beta_end - beta_init) * progress
        
        # Gradient at y (Nesterov lookahead)
        grad_y, _ = grad_of_smoothed_max(y, beta)
        
        # Nesterov update
        t_k = (1 + np.sqrt(1 + 4 * (step - 1)**2)) / 2
        alpha = (step - 1) / t_k if step > 1 else 0
        
        x_new = simplex_project(y - lr * grad_y)
        y = x_new + alpha * (x_new - x)
        y = simplex_project(y)
        x = x_new
        
        # Evaluate
        obj = 2.0 * n * fast_convolve(x).max()
        if obj < best_obj:
            best_obj = obj
            best_x = x.copy()
        
        # Adaptive learning rate: reduce if not improving
        if step % 50 == 0:
            lr *= 0.95
    
    return best_x, best_obj

def adam_optimize(a0, time_limit, beta=50):
    """Adam optimizer with simplex projection."""
    n = len(a0)
    x = a0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.005
    
    best_obj = 2.0 * n * fast_convolve(x).max()
    best_x = x.copy()
    t0 = time.monotonic()
    step = 0
    stale = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Adaptive beta
        current_beta = min(500, beta + step * 0.1)
        
        grad, _ = grad_of_smoothed_max(x, current_beta)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        
        x_new = simplex_project(x - lr * m_hat / (np.sqrt(v_hat) + eps))
        
        obj = 2.0 * n * fast_convolve(x_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_x = x_new.copy()
            stale = 0
        else:
            stale += 1
        
        x = x_new
        
        if stale > 200:
            # Reduce learning rate and reset
            lr *= 0.5
            stale = 0
            if lr < 1e-8:
                break
    
    return best_x, best_obj

def get_parent(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            x_orig = np.linspace(0, 1, p.size)
            x_new = np.linspace(0, 1, n)
            pc = np.interp(x_new, x_orig, p)
        pc = np.maximum(pc, 0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None
