"""
Hybrid optimizer combining LP-guided alternating optimization, log-sum-exp
smoothed mirror descent, and coordinate-wise local search to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses FFT-based gradient computation and adaptive step sizes.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_smoothed_gradient(f, w, n):
    """Compute gradient g[i] = sum_k w[k]*f[k-i] using FFT.
    w has length 2n-1 (softmax weights over autocorrelation)."""
    w_rev = w[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    F = np.fft.rfft(f, fft_size)
    Wr = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * Wr, fft_size)
    g = conv[n - 1:2 * n - 1][::-1]
    return g


def make_lp_structure(n):
    """Precompute sparse structure for cross-convolution LP."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=20):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, verbose=False):
    """LP-based alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_lp_structure(n)

    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        lp_timeout = min(max(2.0, remaining * 0.1), 30)
        g = solve_lp(f, M, rows, cols, n, timeout=lp_timeout)
        if g is None:
            patience += 5
            if patience > 80:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [t, 0.8 * t, 0.6 * t, 0.4 * t, 0.25 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.9
            if patience > 100:
                break

        if verbose and iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def smoothed_mirror_descent(f0, time_limit, tau0=5.0, tau_min=0.005, eta0=0.5, verbose=False):
    """Mirror descent on probability simplex with log-sum-exp smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    tau = tau0
    eta = eta0
    patience = 0
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Compute softmax weights over autocorrelation
        c_max = conv.max()
        scaled = (conv - c_max) / tau
        weights = np.exp(scaled)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(conv))] = 1.0
        else:
            w = weights / w_sum

        # Compute gradient
        g = compute_smoothed_gradient(f, w, n)

        # Mirror descent step: f_new[i] = f[i] * exp(-eta * g[i])
        # Scale eta by tau to get reasonable step sizes
        eta_eff = eta * tau / max(c_max, 1e-10)
        logits = -eta_eff * g / max(g.max(), 1e-15)
        logits -= logits.max()
        exp_logits = np.exp(logits)
        f_new = f * exp_logits
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-30:
            f_new = np.ones(n) / n
        else:
            f_new /= s

        conv = fast_convolve(f_new)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            patience = 0
            f = f_new
            eta = min(eta * 1.02, 5.0)
        else:
            patience += 1
            if patience % 30 == 0:
                eta *= 0.75
                if eta < 0.01:
                    eta = eta0
                    patience = 0
            if patience > 800:
                break

        # Anneal tau slowly
        tau = max(tau * 0.997, tau_min)

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD iter {iter_count}, obj={best_obj:.8f}, tau={tau:.4f}, eta={eta:.4f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def local_search(f0, time_limit, n_trials=5000, verbose=False):
    """Coordinate-wise local search: try adjusting individual elements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    accepted = 0

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        i = rng.integers(0, n)
        deltas = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.005, -0.005]
        rng.shuffle(deltas)

        for d in deltas:
            f_try = f.copy()
            f_try[i] = max(0.0, f_try[i] + d * f[i])
            s = f_try.sum()
            if s < 1e-12:
                continue
            f_try /= s
            conv_try = fast_convolve(f_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_f = f_try.copy()
                f = f_try
                accepted += 1
                break

        if verbose and trial % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LS trial {trial}, obj={best_obj:.8f}, accepted={accepted}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.08, 0.12, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    # Anti-correlating patterns
    alt = np.ones(n)
    alt[1::2] = 0.3
    inits['alt'] = alt / alt.sum()

    alt2 = np.ones(n)
    alt2[1::2] = 0.1
    inits['alt2'] = alt2 / alt2.sum()

    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    # Triangle
    tri = np.maximum(1.0 - 2 * np.abs(x - 0.5 * (n - 1)) / (n - 1), 0.01)
    inits['tri'] = tri / tri.sum()

    # Random
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr

    # Power law
    r = np.arange(1, n + 1, dtype=np.float64)
    pl = r ** (-1.5)
    inits['power'] = pl / pl.sum()

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()
        elif parent.size > 0:
            pp = np.interp(np.linspace(0, 1, n, endpoint=False),
                           np.linspace(0, 1, parent.size, endpoint=False), parent)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()

    return inits
