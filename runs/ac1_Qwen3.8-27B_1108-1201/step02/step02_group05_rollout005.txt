    """Compute soft-max of auto-convolution."""
    conv = fast_convolve(a)
    # Shift for numerical stability
    conv_shifted = conv - conv.max()
    exp_conv = np.exp(T * conv_shifted)
    log_sum_exp = conv.max() + np.log(exp_conv.sum()) / T
    return log_sum_exp

    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def spectral_iterations(a0, n_iter=200, alpha=0.5, time_limit=30.0):
    """Gerchberg-Saxton style spectral iterations."""
    a = a0.copy()
    a = a / a.sum()
    n = len(a)
    L = 2 * n - 1  # zero-pad length
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        # Forward transform
        A = np.fft.rfft(a, L)
        # Flat magnitude
        mag_target = np.sqrt(L) / np.sqrt(L)  # = 1, since sum(a)=1, mean of |A|^2 is L*(sum a)^2/L^2 = 1/L... 
        # Actually: sum_k |A[k]|^2 = L * sum_n a[n]^2 (by Parseval for the zero-padded version)
        # Hmm, let me just use the mean of |A|^2 as the target
        mag_sq = np.abs(A)**2
        target_mag = np.sqrt(mag_sq.mean())
        # Update amplitude, keep phase
        phase = A / (np.abs(A) + 1e-30)
        A_new = target_mag * phase
        # Inverse transform
        a_padded = np.fft.irfft(A_new, L)
        # Project
        a_new = np.maximum(a_padded[:n], 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s
        # Relaxation
        a = (1 - alpha) * a + alpha * a_new
        a = np.maximum(a, 0.0)
        a /= a.sum()
        
        obj = obj_norm(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def adam_optimize(a0, time_limit=60.0, T=50.0, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer for soft-max objective."""
    a = a0.copy()
    a = a / a.sum()
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Compute gradient
        conv = fast_convolve(a)
        conv_shifted = conv - conv.max()
        exp_conv = np.exp(T * conv_shifted)
        p = exp_conv / exp_conv.sum()
        
        # grad[m] = 2 * sum_{j=0}^{n-1} p[m+j] * a[j]
        size = len(p) + n - 1
        fft_size = 1 << (size - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        A_rev = np.fft.rfft(a[::-1], fft_size)
        corr = np.fft.irfft(P * A_rev, fft_size)[:n]
        grad = 2.0 * corr
        
        # Project gradient (remove mean to maintain sum constraint approximately)
        grad = grad - np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s
        
        obj = obj_norm(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Reduce learning rate if stuck
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
            if lr < 1e-8:
                break
        if no_improve > 500:
            break
    
    return best_a, best_obj


def solve_lp(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)[k] <= M for all k."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()
