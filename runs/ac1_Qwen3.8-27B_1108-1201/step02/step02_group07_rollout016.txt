"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2
over non-negative sequences. Uses LP-based direction finding (min-peak
formulation), adaptive line search, Fourier parameter optimization,
subgradient refinement, and coordinate descent.

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def solve_min_peak_lp(f, timeout=5.0):
    """Solve: min t s.t. conv(f, x) <= t, sum(x)=1, x>=0.
    Returns (x, t) or (None, None)."""
    n = len(f)
    # Build A: A[k, j] = f[k-j] if 0 <= k-j < n
    # A has shape (2n-1, n)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    
    # Variables: [x(0), ..., x(n-1), t]
    # Minimize: t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints: A@x - t <= 0 for each row
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(x) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def optimize_lp(f_init, time_limit, verbose=True):
    """LP-based optimization with min-peak formulation."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = f / f.sum()  # normalize to sum=1
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break
        
        x, t_val = solve_min_peak_lp(f, timeout=5.0)
        if x is None:
            break
        
        x = np.clip(x, 0, None)
        S = x.sum()
        if S < 1e-10:
            break
        x = x / S  # ensure sum = 1
        
        # Line search: try different step sizes
        best_t = 0.1
        best_s = np.inf
        for t_try in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_cand = (1 - t_try) * f + t_try * x
            f_cand = np.clip(f_cand, 0, None)
            s = f_cand.sum()
            if s < 1e-10:
                continue
            s_score = 2.0 * n * fft_convolve(f_cand, f_cand).max() / (s * s)
            if s_score < best_s:
                best_s = s_score
                best_t = t_try
        
        f = (1 - best_t) * f + best_t * x
        f = np.clip(f, 0, None)
        s = f.sum()
        if s < 0.01:
            break
        f = f / s  # renormalize
        
        it += 1
        if best_s < best_score - 1e-8:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= 50:

            rng = np.random.default_rng(42 + it * 7919)
            f = f + 0.01 * rng.standard_normal(n)
            f = np.clip(f, 0.001, None)
            f = f / f.sum()
            stall = 0

def fourier_optimize(n, K, time_limit):

            return 1e10
        return score_fast(a)
    
    # Initial parameters
    best_score = np.inf

def subgradient_polish(f_init, time_limit, verbose=True):
    """Subgradient descent with momentum."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.01
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        c = fft_convolve(f, f)
        M = c.max()
        # Active set: top K lags
        K = min(3, len(c))
        top_idx = np.argsort(c)[-K:]
        
        # Subgradient of max(c) w.r.t. f
        g = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    g[i] += 2.0 * f[j]
        g /= K
        
        # Gradient of 2*n*M/S^2
        # = 2*n * (2*M*S - 2*M*S) / S^3 ... 
        # Actually: d(2nM/S^2)/df[i] = 2n * (dM/df[i] * S^2 - M * 2S) / S^4
        # = 2n * (dM/df[i] * S - 2M) / S^3
        S = f.sum()
        grad = 2.0 * n * (g * S - 2.0 * M) / (S * S * S)
        
        momentum = beta * momentum + (1 - beta) * grad
        f = f - lr * momentum
        f = np.clip(f, 0, None)
        S = f.sum()
        if S < 0.01:
            break
        f = f / S
        
        it += 1
        if it % 200 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"    SG it={it:4d} score={s:.6f} best={best_score:.6f} t={time.monotonic()-t0:.1f}s")
    
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score

def coordinate_descent(f_init, time_limit, verbose=True):
    """Coordinate descent: optimize one element at a time."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    def try_change(f, i, d):
        """Try changing f[i] by d, return new score."""
        f_new = f.copy()
        f_new[i] = max(0, f[i] + d)
        s = f_new.sum()
        if s < 0.01:
            return np.inf
        c = fft_convolve(f_new, f_new)
        return 2.0 * n * c.max() / (s * s)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        # Pick an element to modify
        if it < n:
            i = it % n
        else:
            i = rng.integers(0, n)
        
        # Try a range of step sizes
        base_step = f[i] * 0.1 if f[i] > 0.01 else 0.01
        improvements = False
        for sign in [1, -1]:
            for mult in [0.1, 0.3, 0.5, 1.0, 2.0]:
                d = sign * mult * base_step
                new_score = try_change(f, i, d)
                if new_score < best_score - 1e-10:
                    f[i] = max(0, f[i] + d)
                    best_score = new_score
                    best_f = f.copy()
                    improvements = True
                    break
            if improvements:
                break
        
        it += 1
        if it % 500 == 0:
            if verbose:
                print(f"    CD it={it:5d} best={best_score:.6f} t={time.monotonic()-t0:.1f}s")
        
        if it % n == 0 and it > 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
    
    return best_f, best_score

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over
non-negative sequences. Combines LP-based direction finding (min-peak
formulation), adaptive line search, Fourier parameter optimization,
subgradient refinement, and coordinate descent with analytical updates.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * c.max() / (S * S))


def solve_min_peak_lp(f, timeout=5.0):
    """Solve: min t s.t. conv(f, x) <= t, sum(x)=1, x>=0. Returns (x, t) or (None, None)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_max_sum_lp(f, M, timeout=5.0):
    """Solve: max sum(b) s.t. conv(f, b) <= M, b >= 0. Returns b or None."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(f_init, time_limit, verbose=True):
    """LP-based optimization using min-peak formulation with adaptive steps."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-10:
        f = np.ones(n)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        x, t_val = solve_min_peak_lp(f, timeout=5.0)
        if x is None:
            # Fallback to max-sum formulation
            c = fft_convolve(f, f)
            M = c.max()
            b = solve_max_sum_lp(f, M, timeout=5.0)
            if b is None:
                break
            x = b
            Sx = x.sum()
            if Sx < 1e-10:
                break
            x = x / Sx
        else:
            x = np.clip(x, 0, None)
            Sx = x.sum()
            if Sx < 1e-10:
                break
            x = x / Sx

        # Line search
        best_t = 0.1
        best_s = np.inf
        for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]:
            f_cand = (1.0 - t_try) * f + t_try * x
            f_cand = np.clip(f_cand, 0, None)
            S_cand = f_cand.sum()
            if S_cand < 0.01:
                continue
            s_cand = 2.0 * n * fft_convolve(f_cand, f_cand).max() / (S_cand * S_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = (1.0 - best_t) * f + best_t * x
        f = np.clip(f, 0, None)
        S = f.sum()
        if S < 0.01:
            break
        f = f / S

        it += 1
        if best_s < best_score - 1e-9:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if stall >= 40:
            rng_esc = np.random.default_rng(42 + it * 7919)
            f = f + 0.02 * rng_esc.standard_normal(n)
            f = np.clip(f, 0.001, None)
            f = f / f.sum()
            stall = 0

        if verbose and it % 30 == 0:
            print(f"    LP it={it:4d} step={best_t:.2f} score={best_s:.6f} "
                  f"best={best_score:.6f} elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def fourier_optimize(n, K, time_limit):
    """Optimize Fourier series: a[i] = c0 + sum_{k=1}^{K} c_k * cos(2*pi*k*i/n)."""
    t = np.arange(n)
    cosines = [np.cos(2.0 * np.pi * k * t / n) for k in range(1, K + 1)]

    def score_from_params(params):
        c0 = params[0]
        coeffs = params[1:]
        a = np.full(n, c0)
        for k in range(K):
            a += coeffs[k] * cosines[k]
        a = np.clip(a, 0, None)
        if a.sum() < 0.01:
            return 1e10
        c = fft_convolve(a, a)
        S = a.sum()
        return float(2.0 * n * c.max() / (S * S))

    best_score = np.inf
    best_params = None
    rng = np.random.default_rng(123)
    t_start = time.monotonic()

    for trial in range(30):
        if time.monotonic() - t_start > time_limit:
            break
        c0 = 1.0 + 0.5 * rng.random()
        coeffs = 0.2 * rng.standard_normal(K)
        params = np.concatenate([[c0], coeffs])

        res = minimize(score_from_params, params, method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-7, 'fatol': 1e-9,
                                'adaptive': True})
        s = res.fun
        if s < best_score:
            best_score = s
            best_params = res.x.copy()
            print(f"    Fourier trial {trial}: score={s:.6f}")

    if best_params is None:
        return None, np.inf

    c0 = best_params[0]
    coeffs = best_params[1:]
    a = np.full(n, c0)
    for k in range(K):
        a += coeffs[k] * cosines[k]
    a = np.clip(a, 0, None)
    if a.sum() < 0.01:
        return None, np.inf
    return a, best_score


def subgradient_polish(f_init, time_limit, verbose=True):
    """Subgradient descent with momentum targeting top autocorrelation peaks."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.005
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(f, f)
        M = float(c.max())
        S = float(f.sum())

        K = min(5, len(c))
        top_idx = np.argsort(c)[-K:]

        g = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                g[j] += 2.0 * f[ki - j]
        g /= float(K)

        # Gradient of 2*n*M/S^2: = 2*n*(g*S - 2*M) / S^3
        grad = 2.0 * n * (g * S - 2.0 * M) / (S * S * S)

        momentum = beta * momentum + (1.0 - beta) * grad
        f = f - lr * momentum
        f = np.clip(f, 0.0, None)
        S = f.sum()
        if S < 0.01:
            break
        f = f / S

        it += 1
        if it % 500 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            # Adaptive learning rate
            if s > best_score * 1.001:
                lr *= 0.9
            if verbose:
                print(f"    SG it={it:4d} score={s:.6f} best={best_score:.6f} "
                      f"lr={lr:.5f} t={time.monotonic()-t0:.1f}s")

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def coordinate_descent(f_init, time_limit, verbose=True):
    """Coordinate descent with analytical convolution update."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = f / f.sum()
    S = float(f.sum())
    c = fft_convolve(f, f)
    best_f = f.copy()
    best_score = float(2.0 * n * c.max() / (S * S))
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0
    pass_stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        # Pick element cyclically with some randomization
        if it < 2 * n:
            i = it % n
        else:
            i = int(rng.integers(0, n))

        # Compute influence of changing f[i]
        # c'[k] = c[k] + d * 2*f[k-i] for k where 0<=k-i<n and k!=2i
        # c'[2i] = c[2i] + d * 2*f[i] + d^2
        # S' = S + d
        best_d = 0.0
        best_local = best_score

        base = max(f[i] * 0.05, 0.005)
        for sign in [-1, 1]:
            for mult in [0.2, 0.5, 1.0, 2.0, 3.0]:
                d = sign * mult * base
                new_val = f[i] + d
                if new_val < 0 or new_val > 1000:
                    continue
                S_new = S + d
                if S_new < 0.01:
                    continue

                # Compute new max of c
                new_c_max = c.max()
                # Update affected entries
                for k in range(n):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2 * i:
                            new_c_max = max(new_c_max, c[k] + 2.0 * d * f[i] + d * d)
                        else:
                            new_c_max = max(new_c_max, c[k] + 2.0 * d * f[j])

                # Also need to check that non-affected entries don't exceed
                # (they don't change, so c.max() is already the max of those)
                if new_c_max < 0:
                    new_c_max = 0

                new_score = 2.0 * n * new_c_max / (S_new * S_new)
                if new_score < best_local:
                    best_local = new_score
                    best_d = d

        if best_d != 0.0:
            f[i] = max(0.0, f[i] + best_d)
            S = float(f.sum())
            c = fft_convolve(f, f)
            current_score = float(2.0 * n * c.max() / (S * S))
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()
                pass_stall = 0
            else:
                pass_stall += 1
        else:
            pass_stall += 1

        it += 1
        if it % n == 0:
            if pass_stall >= 5:
                # Perturb to escape
                f += 0.005 * rng.standard_normal(n)
                f = np.clip(f, 0.001, None)
                f = f / f.sum()
                S = float(f.sum())
                c = fft_convolve(f, f)
                pass_stall = 0
            if verbose and it % (10 * n) == 0:
                print(f"    CD it={it:5d} best={best_score:.6f} t={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def make_hann_init(n, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bartlett_init(n, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    t = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
    a = 0.4 + 0.6 * w
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_windowed(n, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    t = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
    a = (0.3 + 0.7 * rng.random(n)) * w
    return np.clip(a, 0.01, None)


def make_chirp_init(n, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    t = np.arange(n, dtype=np.float64)
    window = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
    a = 0.4 + 0.6 * window + 0.1 * np.cos(2.0 * np.pi * t * t / (2.0 * n))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
