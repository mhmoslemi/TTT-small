import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_lse_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*b)) where b=conv(a,a).
    Returns (grad, b_array)."""
    n = len(a)
    b = fast_convolve(a)
    b_scaled = beta * b
    b_scaled -= b_scaled.max()
    p = np.exp(b_scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
    else:
        p /= p_sum
    
    # grad_i = sum_k p_k * a_{k-i} (cross-correlation of p and a)
    grad = np.zeros(n)
    for i in range(n):
        end = min(n, 2 * n - 1 - i)
        if end > 0:
            grad[i] = np.dot(p[i:i + end], a[:end])
    return grad, b


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1} using binary search."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_optimize(a0, time_limit=30.0, beta_start=1.0, beta_end=200.0, n_iter=5000):
    """FISTA with LSE annealing on the probability simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    x = a.copy()
    z = a.copy()
    t_mom = 1.0
    best_obj = 2.0 * n * fast_convolve(x).max()
    best_a = x.copy()
    t0 = time.monotonic()
    iter_count = 0
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        iter_count += 1
        
        # Anneal beta (geometric schedule)
        progress = it / max(1, n_iter - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient at z
        grad, _ = compute_lse_grad(z, beta)
        
        # FISTA step
        step = 1.0 / max(beta, 0.3)
        new_x = simplex_project(z - step * grad)
        
        # Momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        z = new_x + (t_mom - 1.0) / t_new * (new_x - z)

        s = z.sum()
        if s < 1e-12:
            z = np.ones(n) / n
        else:
            z /= s
        t_mom = t_new
        x = new_x
        
        # Evaluate
        b_x = fast_convolve(x)
        obj_x = 2.0 * n * b_x.max()
        if obj_x < best_obj:
            best_obj = obj_x
            best_a = x.copy()

def make_lp_structure(n):
    """Precompute sparse structure for cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_fast(f, M, n, rows, cols, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit=30.0):
    """LP-based alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    rows, cols = make_lp_structure(n)
    t0 = time.monotonic()
    t = 1.0
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        g = solve_lp_fast(f, M, n, rows, cols)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        
        if np.max(np.abs(g - f)) < 1e-14:
            break
        
        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.85
            if patience > 30:
                break
    
    return best_f, best_obj

def mirror_polish(f0, time_limit=20.0, t_param=50.0, eta0=0.3):
    """Multiplicative weights / mirror descent polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    eta = eta0
    patience = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Compute subgradient direction
        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[int(conv.argmax())] = 1.0
        else:
            w /= w_sum
        
        # Subgradient: sg_i = sum_k w_k * a_{k-i}
        sg = np.zeros(n)
        for i in range(n):
            end = min(n, 2 * n - 1 - i)
            if end > 0:
                sg[i] = np.dot(w[i:i + end], f[:end])
        
        # Center the gradient
        sg_c = sg - sg.mean()
        
        # Mirror descent update
        lf = -eta * sg_c
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s
        
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 30 == 0:
                eta *= 0.7
                if eta < 0.01:
                    t_param = min(t_param * 1.5, 300.0)
                    eta = 0.2
            if patience > 300:
                break
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with log-sum-exp annealing,
partial-LP refinement (active constraint subset), multiplicative-weights polishing,
and multi-start random restarts. Exploits the quadratic (non-convex) structure with
diverse initializations and adaptive annealing schedules.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def cross_corr_pa(p, a):
    """Compute grad_i = sum_k p[k] * a[k-i] (valid range only).
    p has length 2n-1, a has length n. Result has length n."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        end = min(n, 2 * n - 1 - i)
        if end > 0:
            grad[i] = np.dot(p[i:i + end], a[:end])
    return grad


def compute_lse_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*b_k)) where b=conv(a,a).
    Returns (grad, b_array)."""
    n = len(a)
    b = fast_convolve(a)
    b_scaled = beta * b
    b_scaled -= b_scaled.max()
    p = np.exp(b_scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
    else:
        p /= p_sum
    grad = cross_corr_pa(p, a)
    return grad, b


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    cond = mu * np.arange(1, n + 1) > (cv - 1)
    rho_indices = np.nonzero(cond)[0]
    if len(rho_indices) == 0:
        return np.ones(n) / n
    rho = rho_indices[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_optimize(a0, time_limit=30.0, beta_start=1.0, beta_end=200.0, n_iter=5000):
    """FISTA with LSE annealing on the probability simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    x = a.copy()
    z = a.copy()
    t_mom = 1.0
    b_init = fast_convolve(x)
    best_obj = 2.0 * n * b_init.max()
    best_a = x.copy()
    t0 = time.monotonic()
    last_improve = 0

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Geometric annealing
        progress = min(1.0, it / max(1, n_iter - 1))
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute gradient at z (FISTA evaluates grad at extrapolated point)
        grad, _ = compute_lse_grad(z, beta)

        # Step size: L <= beta * ||a||^2 <= beta, use step = 0.9/beta for safety
        step = 0.9 / max(beta, 0.3)

        # FISTA proximal step
        new_x = simplex_project(z - step * grad)

        # Momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        z = new_x + (t_mom - 1.0) / t_new * (new_x - z)
        z = np.maximum(z, 0.0)
        s = z.sum()
        if s < 1e-12:
            z = np.ones(n) / n
        else:
            z /= s
        t_mom = t_new
        x = new_x

        # Evaluate
        b_x = fast_convolve(x)
        obj_x = 2.0 * n * b_x.max()
        if obj_x < best_obj - 1e-14:
            best_obj = obj_x
            best_a = x.copy()
            last_improve = it

        # Staleness check
        if it - last_improve > 500 and beta > 50:
            break

    return best_a, best_obj


def solve_lp_full(f, M, n, rows, cols, timeout=8):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    res = optimize.linprog(c, A_ub=A, b_ub=b_vec, bounds=(0, None), method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def solve_lp_partial(f, M, n, active_k, timeout=5):
    """Solve LP with only active constraints (dense, fast for small active set)."""
    A = np.zeros((len(active_k), n))
    for row, k in enumerate(active_k):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_end > i_start:
            A[row, i_start:i_end] = f[k - (i_end - 1): k - i_start + 1][::-1]

    c = -np.ones(n)
    b_vec = np.full(len(active_k), M)
    res = optimize.linprog(c, A_ub=A, b_ub=b_vec, bounds=(0, None), method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def make_lp_structure(n):
    """Precompute row/col indices for cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def lp_optimize(f0, time_limit=30.0, use_partial=True, n_active=80):
    """LP-based alternating optimization with backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_lp_structure(n)
    t0 = time.monotonic()
    t = 1.0
    patience = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        it_count += 1

        if use_partial and n > 128:
            # Use only top n_active lags
            conv = fast_convolve(f)
            M = conv.max()
            active_k = np.argsort(conv)[::-1][:n_active]
            g = solve_lp_partial(f, M, n, active_k, timeout=5)
        else:
            conv = fast_convolve(f)
            M = conv.max()
            g = solve_lp_full(f, M, n, rows, cols, timeout=8)

        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [t, 0.8 * t, 0.6 * t, 0.4 * t, 0.2 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.08)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.8
            if patience > 40:
                break

        if it_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {it_count}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def mirror_polish(f0, time_limit=20.0, t_param=50.0, eta0=0.3):
    """Multiplicative weights / mirror descent polish on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[int(conv.argmax())] = 1.0
        else:
            w /= w_sum

        sg = cross_corr_pa(w, f)
        sg_c = sg - sg.mean()

        lf = -eta * sg_c
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s

        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 30 == 0:
                eta *= 0.7
                if eta < 0.01:
                    t_param = min(t_param * 1.4, 400.0)
                    eta = 0.25
            if patience > 350:
                break

    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits['g' + str(int(sf * 100))] = g

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    cb = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr

    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.01)
    inits['tri'] = tri / tri.sum()

    bm = np.exp(-0.5 * ((x - 0.25 * n) / (0.12 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.12 * n)) ** 2)
    inits['bim'] = bm / bm.sum()

    # Power-law
    pl = np.power(x + 1, 0.5)
    inits['plaw'] = pl / pl.sum()

    # Log-normal in frequency
    lf = np.exp(-0.5 * (np.log(x + 1) - np.log(n) / 2.0) ** 2 * 4.0)
    inits['logn'] = lf / lf.sum()

    return inits

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # FISTA step (short burst, high beta for local refinement)
        t_fista = min(remaining * 0.5, 15.0)
        f, obj = fista_optimize(f, time_limit=t_fista, beta_start=5.0, beta_end=300.0)
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()

        # LP step (for global correction)
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        t_lp = min(remaining * 0.4, 12.0)
        f2, obj2 = lp_optimize(f, time_limit=t_lp, use_partial=True, n_active=60)
        if obj2 < best_obj:
            best_obj = obj2
            best_f = f2.copy()
        else:
            # No improvement from LP, check if we should stop
            if obj2 >= best_obj - 1e-10:
                # Try one more FISTA round
                remaining = time_limit - (time.monotonic() - t0)
                if remaining > 3:
                    f3, obj3 = fista_optimize(best_f, time_limit=min(remaining, 10.0),
                                              beta_start=10.0, beta_end=500.0)
                    if obj3 < best_obj:
                        best_obj = obj3
                        best_f = f3.copy()
                    else:
                        break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry: multi-phase optimization with FISTA, partial-LP, and mirror polish."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    all_results = []

    def record(a, source):
        nonlocal best_seq, best_score
        lst = [float(max(0.0, min(1000.0, x))) for x in a]
        score = float(evaluate_sequence(lst))
        if np.isfinite(score):
            all_results.append((score, len(a), source, a.copy()))
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  *** NEW BEST: {score:.8f} (n={len(a)}, {source})")
        return score

    # === Phase 1: Quick FISTA scan ===
    print("=== Phase 1: FISTA scan ===")
    lengths = [256, 512, 1024]
    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        inits = gen_inits(n, rng)
        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()

        n_total = len(inits)
        remaining = deadline - time.monotonic() - 300
        t_per = max(4.0, remaining / max(1, n_total * max(1, len(lengths) - lengths.index(n))))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 250:
                break
            ta = min(t_per, deadline - time.monotonic() - 250)
            if ta < 3:
                break
            f_opt, obj_opt = fista_optimize(a0, time_limit=ta, beta_start=1.0, beta_end=80.0)
            record(f_opt, f"P1.{name}.n{n}")

    print(f"Phase 1 done. Best: {best_score:.8f}")
    all_results.sort(key=lambda x: x[0])

    # === Phase 2: Deep optimization (alternating FISTA + LP) on top candidates ===
    print("=== Phase 2: Deep optimization ===")
    remaining = deadline - time.monotonic() - 200
    if remaining > 50 and all_results:
        top_k = min(5, len(all_results))
        t_each = remaining / top_k
        for idx in range(top_k):
            if time.monotonic() > deadline - 150:
                break
            sc, n, src, f = all_results[idx]
            ta = min(t_each * 0.9, deadline - time.monotonic() - 150)
            if ta < 10:
                break
            print(f"  Deep opt [{idx}] n={n} src={src} score={sc:.6f}, time={ta:.0f}s")
            f2, obj2 = alternating_optimize(f, time_limit=ta)
            record(f2, f"P2.{idx}.n{n}")

    all_results.sort(key=lambda x: x[0])

    # === Phase 3: Intensive LP on the best ===
    print("=== Phase 3: Intensive LP polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_lp = min(remaining * 0.6, 120.0)
            print(f"  LP polish, time={t_lp:.0f}s")
            f3, obj3 = lp_optimize(best_seq, time_limit=t_lp, use_partial=True, n_active=100)
            record(f3, "P3.LP")

    # === Phase 4: Mirror descent polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            print("=== Phase 4: Mirror polish ===")
            for tp, eta in [(30.0, 0.25), (80.0, 0.12), (150.0, 0.06), (300.0, 0.03)]:
                if time.monotonic() > deadline - 30:
                    break
                ta = min(remaining * 0.3, 30.0)
                f4, _ = mirror_polish(best_seq, time_limit=ta, t_param=tp, eta0=eta)
                record(f4, f"P4.md.tp{int(tp)}")
                remaining = deadline - time.monotonic() - 30

    # === Phase 5: Random restarts ===
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 15:
            print("=== Phase 5: Random restarts ===")
            n_best = len(best_seq)
            n_restarts = min(int(remaining / 12), 15)
            for r in range(n_restarts):
                if time.monotonic() > deadline - 20:
                    break
                # Various perturbation strategies
                strategy = r % 3
                if strategy == 0:
                    noise = rng.normal(0, 0.08, size=n_best)
                    perturbed = best_seq + noise
                elif strategy == 1:
                    # Swap two random blocks
                    i = rng.integers(0, n_best - 10)
                    j = rng.integers(0, n_best - 10)
                    L = rng.integers(5, min(30, n_best // 2))
                    perturbed = best_seq.copy()
                    perturbed[i:i+L], perturbed[j:j+L] = perturbed[j:j+L].copy(), perturbed[i:i+L].copy()
                else:
                    # Multiplicative perturbation
                    perturbed = best_seq * np.exp(rng.normal(0, 0.15, size=n_best))

                perturbed = np.maximum(perturbed, 0.0)
                s = perturbed.sum()
                if s < 1e-12:
                    continue
                perturbed /= s

                ta = min(8.0, deadline - time.monotonic() - 20)
                if ta < 3:
                    break
                f_r, obj_r = fista_optimize(perturbed, time_limit=ta,
                                            beta_start=5.0, beta_end=200.0)
                record(f_r, f"P5.r{r}")

    # === Phase 6: Final fine-tune ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        print("=== Phase 6: Final fine-tune ===")
        tr = deadline - time.monotonic() - 5
        f6, _ = fista_optimize(best_seq, time_limit=min(tr * 0.5, 20.0),
                               beta_start=20.0, beta_end=500.0)
        record(f6, "P6.final")
        if time.monotonic() < deadline - 3:
            f7, _ = mirror_polish(best_seq, time_limit=min(deadline - time.monotonic() - 3, 10.0),
                                  t_param=200.0, eta0=0.05)
            record(f7, "P6.md")

    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Ensure valid output
    final = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = float(evaluate_sequence(final))
    print(f"\nFINAL: n={len(final)}, score={final_score:.8f}")
    return final
