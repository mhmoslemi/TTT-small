"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-guided optimization
with projected gradient polish, adaptive mirror descent, and diverse initializations.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def fast_cross_conv(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]


def make_lp_matrix(n):
    """Precompute the sparse matrix A for constraint (f*g)[k] <= M.
    A is (2n-1) x n, A[k, i] = f[k-i] if 0 <= k-i <= n-1."""
    # We'll build the structure (row, col) and fill values later
    rows = []
    cols = []
    for i in range(n):
        for k in range(max(0, i), min(2*n-1, i+n)):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_cross(f, M, rows, cols, n):
    """Solve max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    vals = np.empty(len(rows))
    # For entry (k, i), the value is f[k-i]
    # rows[k_idx] = k, cols[k_idx] = i
    # f[k-i] where k = rows[idx], i = cols[idx]
    idx = rows - cols  # = k - i
    vals = f[idx]
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lp_optimize(f0, time_limit, rows=None, cols=None):
    """LP-based alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    if rows is None:
        rows, cols = make_lp_matrix(n)
    
    t0 = time.monotonic()
    t = 0.5
    patience = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = solve_lp_cross(f, M, rows, cols, n)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        
        if np.max(np.abs(g - f)) < 1e-14:
            break
        
        # Backtracking line search
        improved = False
        for t_try in [t, 0.7*t, 0.5*t, 0.3*t, 0.1*t, 0.05*t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new < 1e-12:
                continue
            f_new /= s_new
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.1)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.85
            if patience > 80:
                break
    
    return best_f, best_obj

def mirror_polish(f0, time_limit, t_param=50.0, eta0=0.3):
    """Mirror descent polish with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    
    conv = fast_convolve(f)
    max_conv = conv.max()
    best_obj = 2.0 * n * max_conv
    best_f = f.copy()
    
    eta = eta0
    patience = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        if max_conv < 1e-15:
            break
        
        # Weight the positions by their proximity to the peak
        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient-like signal
        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end+1] += w[k] * f[k - j_end:k - j_start + 1][::-1]
        
        # Center the signal
        sg_c = sg - sg.mean()
        
        # Mirror descent update
        lf = -eta * sg_c
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s
        
        conv = fast_convolve(f)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 30 == 0:
                eta *= 0.7
                if eta < 0.01:
                    t_param = min(t_param * 1.3, 500.0)
                    eta = 0.15
            if patience > 300:
                break
    
    return best_f, best_obj

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = parent.copy()
            print(f"Parent score: {sc:.6f}, n={len(parent)}")
    
    lengths = [256, 512, 1024]
    all_results = []
    
    # Precompute LP matrices
    lp_matrices = {}
    for n in lengths:
        lp_matrices[n] = make_lp_matrix(n)
    
    # Phase 1: LP scan
    print("Phase 1: LP scan")
    for n in lengths:
        if time.monotonic() > deadline - 120:
            break
        
        inits = gen_inits(n, rng)
        
        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()
            elif parent.size > 0:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()
        
        rows, cols = lp_matrices[n]
        n_cfgs = len(inits)
        remaining = deadline - time.monotonic() - 120
        time_per = max(10.0, remaining / max(1, n_cfgs * len(lengths)))
        
        for name, a0 in inits.items():
            if time.monotonic() > deadline - 60:
                break
            ta = min(time_per, deadline - time.monotonic() - 60)
            if ta < 5:
                break
            
            f_opt, obj_opt = lp_optimize(a0, time_limit=ta, rows=rows, cols=cols)
            score = float(evaluate_sequence(f_opt.tolist()))
            
            if np.isfinite(score):
                all_results.append((score, n, name, f_opt))
                if score < best_score:
                    best_score = score
                    best_seq = f_opt.copy()
                    print(f"  n={n:4d} {name:8s} score={score:.6f}")
    
    print(f"Phase 1 done. Best: {best_score:.6f}")
    
    # Phase 2: Deep optimization
    all_results.sort(key=lambda x: x[0])
    top = all_results[:6]
    
    remaining = deadline - time.monotonic() - 30
    if remaining > 60 and top:
        print(f"Phase 2: Deep optimization, {remaining:.0f}s remaining")
        t_each = remaining / len(top)
        
        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > deadline - 20:
                break
            ta = min(t_each, deadline - time.monotonic() - 20)
            if ta < 8:
                break
            
            rows, cols = lp_matrices[n]
            
            # LP continuation
            f2, _ = lp_optimize(f, time_limit=ta*0.4, rows=rows, cols=cols)
            sc2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = f2.copy()
                print(f"  LP [{idx}] n={n} {name} score={sc2:.6f}")
            
            # Gradient polish
            if time.monotonic() < deadline - 15:
                tr = min(ta*0.25, deadline - time.monotonic() - 15)
                f3, _ = gradient_polish(f2, time_limit=tr)
                sc3 = float(evaluate_sequence(f3.tolist()))
                if np.isfinite(sc3) and sc3 < best_score:
                    best_score = sc3
                    best_seq = f3.copy()
                    print(f"  Grad [{idx}] n={n} {name} score={sc3:.6f}")
            
            # Mirror polish
            if time.monotonic() < deadline - 10:
                tr = min(ta*0.2, deadline - time.monotonic() - 10)
                f4, _ = mirror_polish(f3 if 'f3' in dir() else f2, time_limit=tr, t_param=40.0)
                sc4 = float(evaluate_sequence(f4.tolist()))
                if np.isfinite(sc4) and sc4 < best_score:
                    best_score = sc4
