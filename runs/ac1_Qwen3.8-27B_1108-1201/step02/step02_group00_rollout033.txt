import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_peak_reduction(f, M):
    """
    Solve the LP: maximize sum(g) subject to f*g <= M (elementwise), g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build the constraint matrix A such that A @ g <= M_vector
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a banded matrix
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    
    # Actually, can we vectorize this?
    # A[k, j] = f[k-j] for max(0,k-n+1) <= j <= min(n-1,k)
    # This is like a Toeplitz matrix (banded)
    
    # For efficiency with large n, let's use a sparse matrix
    # But for n=256, dense is fine
    
    b = np.full(num_constraints, M)
    
    # Maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    
    # g >= 0 is handled by bounds
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: return f (no change)
        return f.copy()

def optimize_lp(f0, time_limit, t_init=0.1, min_t=0.001, epsilon=1e-8):
    """
    LP-based iterative peak reduction.
    """
    f = np.maximum(f0, epsilon).astype(np.float64)
    s = f.sum()
    f = f / s  # normalize to sum=1
    n = len(f)
    
    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    
    t = t_init
    stall_count = 0
    max_stall = 20
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        cur_obj = 2.0 * n * M
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_f = f.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            break
        
        # Solve LP
        g = lp_peak_reduction(f, M)
        
        # Check if g is useful
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        # Rescale g to have same sum as f
        g = g * (s / g_sum)
        
        # Update
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-10:
            break
        f = f_new / f_new_sum
        
        t = max(min_t, t * 0.95)
    
    return best_f, best_obj

"""
Hybrid LP-based and subgradient optimization for minimizing 
2*n*max(conv(a,a))/(sum(a))^2.
Main innovation: LP-based peak reduction that solves a linear program 
at each iteration to find the best direction for reducing the 
convolution peak, combined with subgradient polishing.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """
    Build the LP constraint matrix for f*g <= M.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    Returns dense matrix (fine for n <= 4096).
    """
    n = len(f)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = f[k - j_end : k - j_start : -1]
    return A


def solve_lp_reduction(f, M):
    """
    Solve: max sum(g) s.t. f*g <= M, g >= 0.
    Returns (g_opt, has_value) where has_value indicates sum(g_opt) > sum(f).
    """
    n = len(f)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    
    if res.success:
        g = res.x
        return g, g.sum() > f.sum() + 1e-10
    else:
        return f.copy(), False


def optimize_lp(f0, time_limit, t_init=0.3, min_t=0.01, verbose=False):
    """
    LP-based iterative peak reduction.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    t0 = time.monotonic()
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    
    t = t_init
    stall = 0
    max_stall = 50
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()
        cur_obj = 2.0 * n * M
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > max_stall:
            break
        
        # Solve LP
        g, has_value = solve_lp_reduction(f, M)
        
        if not has_value:
            # LP doesn't help, try a different approach
            # Perturb and continue
            rng = np.random.default_rng(iter_count)
            f = project_simplex(f + rng.standard_normal(n) * 0.01)
            t = max(min_t, t * 0.8)
            continue
        
        g = g / g.sum()  # normalize to sum=1
        
        # Update
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            break
        f = f_new / f_new_sum
        
        # Decay step size
        t = max(min_t, t * 0.9)
        
        if verbose and iter_count % 100 == 0:
            print(f"  LP iter {iter_count}: obj={cur_obj:.6f}, t={t:.4f}, stall={stall}")
    
    return best_f, best_obj


def subgradient_multi_peak(a, conv, top_k, temperature):
    """Compute weighted subgradient over top-k convolution peaks."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
    
    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = w[i] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    
    return sg


def optimize_subgrad(a0, time_limit, lr_init=0.02, momentum=0.9, top_k=30,
                     temp_init=0.0003, decay=0.9998, perturb_every=2000,
                     perturb_scale=0.003, batch=50):
    """
    Projected subgradient descent with momentum.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    max_no_improve = perturb_every
    momentum_cur = momentum
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            
            sg = subgradient_multi_peak(a, conv, top_k, temp)
            v = momentum_cur * v - lr * sg
            a_new = project_simplex(a + v)
            a = a_new
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve >= max_no_improve:
            rng = np.random.default_rng()
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.3)
            lr = max(lr_init * 0.3, lr * 0.5)
        
        lr *= decay
        temp *= 1.003
        momentum_cur = min(0.98, momentum_cur + 0.0002)
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    ramp_inc = np.arange(1, n + 1, dtype=np.float64)

    inits['ramp_inc'] = ramp_inc
    
    inits['ramp_dec'] = ramp_inc[::-1].copy()
    
    tri = np.minimum(x, n - 1 - x) + 1.0

    inits['triangle'] = tri
    
    for sf in [0.15, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g
    
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    bump = np.exp(-x / (0.3 * n))
    bump /= bump.sum()
    inits['bump_left'] = bump
    inits['bump_right'] = bump[::-1].copy()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),

            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:

        if len(parent) > n:

            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:

    return inits

"""
Hybrid LP-based and subgradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Main innovation: LP-based peak reduction solving a linear program at each iteration to find
directions that provably reduce the convolution peak, combined with adaptive subgradient
polishing and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """
    Build the LP constraint matrix A where (A@g)[k] = (f*g)[k].
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(f)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = f[k - j_start:k - j_end - 1:-1]
    return A


def solve_lp_reduction(f, M, timeout=3.0):
    """
    Solve: max sum(g) s.t. f*g <= M (elementwise), g >= 0.
    Returns (g_opt, sum_g).
    """
    n = len(f)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and np.all(np.isfinite(res.x)):
            return res.x, float(res.x.sum())
    except Exception:
        pass
    return f.copy(), float(f.sum())


def optimize_lp(f0, time_limit, t_init=0.3, min_t=0.005, epsilon=1e-10):
    """
    LP-based iterative peak reduction.
    At each step, solve LP to find direction that reduces the convolution peak.
    """
    f = np.maximum(f0.astype(np.float64), epsilon)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    t0 = time.monotonic()
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t = t_init
    stall = 0
    max_stall = 40
    iter_count = 0
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()
        cur_obj = 2.0 * n * M

        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_f = f.copy()
            stall = 0
            stuck_count = 0
        else:
            stall += 1

        if stall > max_stall:
            break

        # Solve LP for direction
        g, sum_g = solve_lp_reduction(f, M, timeout=2.0)

        if sum_g <= 1.0 + 1e-8:
            # LP doesn't improve - stuck
            stuck_count += 1
            if stuck_count > 5:
                break
            # Perturb to escape
            rng = np.random.default_rng(iter_count * 1000 + 7)
            f = project_simplex(f + rng.standard_normal(n) * 0.005)
            t = max(min_t, t * 0.7)
            continue

        # Normalize g to sum=1
        g = g / sum_g
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Verify this direction actually helps
        f_test = (1 - min(t, 0.1)) * f + min(t, 0.1) * g
        f_test = np.maximum(f_test, 0.0)
        f_test /= f_test.sum()
        test_obj = 2.0 * n * fast_convolve(f_test).max()

        if test_obj < cur_obj + 1e-10:
            # Direction helps, take the step
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            f_new_sum = f_new.sum()
            if f_new_sum < 1e-15:
                break
            f = f_new / f_new_sum
            t = max(min_t, t * 0.92)
        else:
            # Direction doesn't help much, use smaller step
            t = max(min_t, t * 0.5)
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            f = f_new

    return best_f, best_obj


def subgradient_multi_peak(a, conv, top_k, temperature):
    """Compute weighted subgradient over top-k convolution peaks."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = w[i] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]

    return sg


def optimize_subgrad(a0, time_limit, lr_init=0.015, momentum=0.92, top_k=30,
                     temp_init=0.0002, decay=0.9998, perturb_every=1500,
                     perturb_scale=0.003, batch=80):
    """
    Projected subgradient descent with heavy-ball momentum and adaptive scheduling.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    max_no_improve = perturb_every
    momentum_cur = momentum
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            iter_count += 1

            sg = subgradient_multi_peak(a, conv, top_k, temp)
            v = momentum_cur * v - lr * sg
            a_new = project_simplex(a + v)
            a = a_new
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()

            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iter_count)
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.3)
            lr = max(lr_init * 0.25, lr * 0.5)
            perturb_scale *= 0.8

        lr *= decay
        temp *= 1.004
        momentum_cur = min(0.99, momentum_cur + 0.0001)

    return best_a, best_obj


def optimize_mirror(a0, time_limit, t_param=60.0, eta_init=0.4):
    """
    Multiplicative mirror descent (exponential weights).
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    s = a.sum()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    t_cur = t_param
    no_improve = 0
    max_no_improve = 80

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = t_cur * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(peak_vals) if hasattr(w, 'peak_vals') else np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Subgradient from weighted peaks
        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active = np.where(w > thresh)[0]
        for k_int in active:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[k_int] * 2.0 * a[k - idx]

        sg_centered = sg - sg.mean()
        log_ratio = -eta * sg_centered
        log_ratio -= log_ratio.max()
        a = a * np.exp(log_ratio)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            no_improve = 0
            eta *= 0.75
            t_cur *= 1.4
            if eta < 0.03:
                eta = 0.35
                t_cur = t_param
                max_no_improve = 50

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    ramp_inc /= ramp_inc.sum()
    inits['ramp_inc'] = ramp_inc
    inits['ramp_dec'] = ramp_inc[::-1].copy()

    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    for sf in [0.15, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_{}'.format('{:.2f}'.format(sf))] = g

    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    bump = np.exp(-x / (0.3 * n))
    bump /= bump.sum()
    inits['bump_left'] = bump
    inits['bump_right'] = bump[::-1].copy()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)

    # Cosine-based
    cos_init = np.cos(np.pi * x / n) ** 2 + 0.1
    cos_init /= cos_init.sum()
    inits['cosine'] = cos_init

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Hybrid LP-based and subgradient search to minimize
    2n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Multi-length LP-based search (novel direction-finding via linear programming)
    2. Subgradient and mirror descent polishing
    3. Final micro-optimization with very small steps
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent available: length={}, sum={:.2f}".format(len(p), p.sum()))
    except Exception:
        pass

    # Phase 1: Multi-length search with LP + subgradient
    print("=== Phase 1: Multi-length Search ===")

    # (length, num_inits, methods)
    search_configs = [
        (128, 3, ['lp', 'subgrad']),
        (256, 4, ['lp', 'subgrad']),
        (512, 3, ['lp', 'subgrad']),
        (1024, 2, ['lp']),
    ]

    phase1_budget = 650.0
    total_slots = sum(c[1] for c in search_configs)
    time_per_slot = phase1_budget / total_slots

    for n, num_inits, methods in search_configs:
        if time.monotonic() > deadline - 50:
            break
        inits = generate_initializations(n, parent, rng)
        init_priority = ['parent', 'parent_ds', 'parent_up', 'ramp_inc',
                         'gauss_0.30', 'random', 'cosine', 'bimodal']
        init_names = [nm for nm in init_priority if nm in inits][:num_inits]
        if not init_names:
            init_names = list(inits.keys())[:num_inits]

        for name in init_names:
            if time.monotonic() > deadline - 50:
                break
            t_alloc = min(time_per_slot, deadline - time.monotonic() - 40)
            if t_alloc < 5:
                break

            a0 = inits[name].copy()

            for method in methods:
                if time.monotonic() > deadline - 45:
                    break
                t_single = t_alloc / len(methods)
                if t_single < 3:
                    continue

                if method == 'lp':
                    a_opt, _ = optimize_lp(a0, time_limit=t_single, t_init=0.3, min_t=0.005)
                elif method == 'subgrad':
                    a_opt, _ = optimize_subgrad(a0, time_limit=t_single,
                                                lr_init=0.02, momentum=0.9,
                                                top_k=max(15, n // 25), temp_init=0.0003)
                elif method == 'mirror':
                    a_opt, _ = optimize_mirror(a0, time_limit=t_single,
                                               t_param=50.0, eta_init=0.4)

                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print("  Phase1: n={}, {}, {}, score={:.6f}".format(n, name, method, score))

    print("Phase 1 best: {:.6f}".format(best_score))

    # Phase 2: Deep alternating optimization on best
    print("=== Phase 2: Deep Optimization ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 20
        num_rounds = 6
        t_per_round = remaining / num_rounds

        for i in range(num_rounds):
            if time.monotonic() > deadline - 10:
                break
            t_cur = min(t_per_round, deadline - time.monotonic() - 10)
            if t_cur < 5:
                break

            if i % 3 == 0:
                # LP-based
                a_opt, _ = optimize_lp(best_sequence, time_limit=t_cur,
                                       t_init=max(0.05, 0.2 * 0.7**i), min_t=0.002)
            elif i % 3 == 1:
                # Subgradient
                a_opt, _ = optimize_subgrad(best_sequence, time_limit=t_cur,
                                            lr_init=0.01 * (0.85 ** i),
                                            momentum=0.93 + 0.005 * i,
                                            top_k=max(20, n_best // 12),
                                            temp_init=0.0001 * (0.8 ** i))
            else:
                # Mirror descent
                a_opt, _ = optimize_mirror(best_sequence, time_limit=t_cur,
                                           t_param=50.0 + 30.0 * i,
                                           eta_init=max(0.1, 0.4 * 0.75**i))

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print("  Phase2 round {}: score={:.6f}".format(i, score))

    # Phase 3: Final fine-tuning
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 5
        n_final = len(best_sequence)

        # Very fine subgradient passes
        fine_configs = [
            {'lr_init': 0.003, 'momentum': 0.98, 'top_k': n_final, 'temp_init': 5e-6},
            {'lr_init': 0.002, 'momentum': 0.99, 'top_k': max(100, n_final // 2), 'temp_init': 3e-6},
            {'lr_init': 0.005, 'momentum': 0.97, 'top_k': n_final, 'temp_init': 8e-6},
        ]

        t_per_fine = remaining / 4
        for cfg in fine_configs:
            if time.monotonic() > deadline - 5:
                break
            t_cur = min(t_per_fine, deadline - time.monotonic() - 5)
            if t_cur < 3:
                break
            a_opt, _ = optimize_subgrad(best_sequence, time_limit=t_cur, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print("  Phase3 fine: score={:.6f}".format(best_score))

        # One final LP pass
        if time.monotonic() < deadline - 10:
            t_cur = min(30, deadline - time.monotonic() - 5)
            if t_cur > 5:
                a_opt, _ = optimize_lp(best_sequence, time_limit=t_cur,
                                       t_init=0.05, min_t=0.001)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print("  Phase3 LP: score={:.6f}".format(best_score))

        # Random perturb + re-optimize
        if time.monotonic() < deadline - 10:
            for _ in range(2):
                if time.monotonic() > deadline - 5:
                    break
                t_cur = min(15, deadline - time.monotonic() - 5)
                if t_cur < 3:
                    break
                noise = rng.standard_normal(n_final) * 0.001
                a_pert = project_simplex(best_sequence + noise)
                a_opt, _ = optimize_subgrad(a_pert, time_limit=t_cur,
                                            lr_init=0.003, momentum=0.98,
                                            top_k=n_final, temp_init=5e-6)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print("  Phase3 perturb: score={:.6f}".format(best_score))

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL SCORE: {:.6f} (n={})".format(best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
