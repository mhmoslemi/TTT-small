def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def score_normalized(f):
    """Compute 2*n*max(conv(f,f)) assuming sum(f)=1."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()

def make_lp_matrix(n):
    """Create sparse matrix A where A[k,i] = 1 if i is valid for position k in cross-conv."""
    # For cross-convolution f*g, position k has terms f[i]*g[k-i] for valid i
    # A[k, i] = f[k-i] when 0 <= k-i <= n-1
    # But we want the matrix structure (without f values)
    rows = []
    cols = []
    for i in range(n):
        for j in range(n):
            rows.append(i + j)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

    c = -np.ones(n)
    b = np.full(2*n-1, M)

def multiplicative_update(f, t_param=50.0, eta=0.3, n_steps=10):

    f = f / f.sum()

        max_conv = conv.max()
        if max_conv < 1e-15:
            break

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided optimization with
momentum, adaptive line search, stochastic escape, and mirror descent polishing.
Key improvements over previous approach: momentum-based LP direction combination,
adaptive M relaxation, golden-section line search, targeted coordinate descent,
and multi-scale mirror descent polishing.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def normalize(f):
    """Clip to non-negative and normalize sum to 1."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def objective(f):
    """Compute 2*n*max(conv(f,f)) assuming sum(f)=1."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def make_lp_indices(n):
    """Precompute row/col indices for cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        rows_list.extend(range(i, i + n))
        cols_list.extend([i] * n)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=25):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout,
                                    'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def golden_line_search(f, g, f_obj, max_evals=12):
    """Golden section search for best t in (1-t)*f + t*g."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)

    def eval_t(t):
        fn = (1.0 - t) * f + t * g
        fn = normalize(fn)
        return fast_convolve(fn).max()

    fc = eval_t(c)
    fd = eval_t(d)
    for _ in range(max(0, max_evals // 2 - 1)):
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = eval_t(c)
        else:
            a, c, fc = c, d, fc
            d = a + gr * (b - a)
            fd = eval_t(d)
        if b - a < 1e-7:
            break
    t_opt = (a + b) / 2.0
    f_opt = normalize((1.0 - t_opt) * f + t * g)
    obj_opt = fast_convolve(f_opt).max()
    if obj_opt < f_obj:
        return f_opt, obj_opt
    return f, f_obj


def lp_optimize(f0, time_limit, rows, cols, use_momentum=False, m0=0.8):
    """LP-guided optimization with adaptive M and optional momentum."""
    f = normalize(f0.astype(np.float64).copy())
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    t_step = 0.8
    patience = 0
    iter_count = 0
    velocity = np.zeros(n) if use_momentum else None

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break

        # Adaptive M relaxation: larger early, tighter later
        progress = min(1.0, elapsed / (time_limit * 0.5))
        m_relax = 1.0 + 0.08 * (1.0 - progress)
        M_lp = M * m_relax

        g = solve_lp(f, M_lp, rows, cols, n, timeout=min(30, max(5, remaining // 4)))
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        if use_momentum and velocity is not None:
            velocity = m0 * velocity + (1.0 - m0) * (g - f)
            f_cand = normalize(f + velocity)
            conv_cand = fast_convolve(f_cand)
            if conv_cand.max() < M * 1.01:
                f = f_cand
                M = conv_cand.max()
                best_obj = 2.0 * n * M
                best_f = f.copy()
                patience = 0
            else:
                velocity *= 0.5
                patience += 1
        else:
            # Standard line search (try golden section, fall back to fixed grid)
            improved = False
            f_new, obj_new = golden_line_search(f, g, M, max_evals=10)
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = obj_new
                best_obj = 2.0 * n * M
                best_f = f.copy()
                patience = 0
                t_step = min(1.0, t_step * 1.08)
                improved = True

            if not improved:
                # Try a few fixed steps
                for t_try in [t_step, 0.5 * t_step, 0.25 * t_step, 0.1 * t_step]:
                    f_try = normalize((1.0 - t_try) * f + t_try * g)
                    conv_try = fast_convolve(f_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-14:
                        f = f_try
                        M = conv_try.max()
                        best_obj = obj_try
                        best_f = f.copy()
                        patience = 0
                        improved = True
                        break
                if not improved:
                    patience += 1
                    t_step *= 0.85

        # Stochastic escape every 60 iterations
        if patience > 60 and iter_count % 3 == 0:
            rng = np.random.default_rng(iter_count * 7919)
            noise_scale = 0.003 * (1.0 + patience / 60.0)
            f_pert = f + rng.normal(0, noise_scale, n)
            f_pert = normalize(f_pert)
            conv_pert = fast_convolve(f_pert)
            if conv_pert.max() < M * 1.03:
                f = f_pert
                M = conv_pert.max()
                patience = 0
                t_step = 0.5
            elif patience > 150:
                break

        if iter_count % 40 == 0:
            print(f"      LP iter={iter_count}, obj={best_obj:.8f}, "
                  f"patience={patience}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def mirror_descent_polish(f0, time_limit, t_params=None, eta0=0.25):
    """Multi-scale mirror descent (multiplicative weights) polishing."""
    if t_params is None:
        t_params = [20, 50, 100, 200, 400, 800]
    f = normalize(f0.astype(np.float64).copy())
    n = len(f)
    best_obj = objective(f)
    best_f = f.copy()

    t0 = time.monotonic()
    n_params = len(t_params)

    for pi, tp in enumerate(t_params):
        if time.monotonic() - t0 > time_limit:
            break
        tp_budget = min(40.0, time_limit / n_params)
        t_start = time.monotonic()
        eta = eta0
        stall = 0

        while time.monotonic() - t_start < tp_budget:
            conv = fast_convolve(f)
            max_cv = conv.max()
            if max_cv < 1e-15:
                break

            scaled = tp * conv / max_cv
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-12:
                w = np.zeros(len(conv))
                w[int(np.argmax(conv))] = 1.0
            else:
                w = w / w_sum

            # Sparse gradient accumulation
            sg = np.zeros(n)
            thresh = w.max() * 0.003
            active = np.where(w > thresh)[0]
            for k in active:
                k = int(k)
                j_lo = max(0, k - n + 1)
                j_hi = min(n - 1, k)
                segment = f[k - j_hi:k - j_lo + 1][::-1]
                sg[j_lo:j_hi + 1] += w[k] * segment

            sg_c = sg - sg.mean()
            lf = -eta * sg_c
            lf -= lf.max()
            f = f * np.exp(lf)
            f = normalize(f)

            obj = objective(f)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                stall = 0
            else:
                stall += 1
                if stall > 40:
                    eta *= 0.7
                    if eta < 0.03:
                        break

    return best_f, best_obj


def targeted_cd(f, n_steps=30):
    """Coordinate descent targeting the worst convolution position."""
    f = f.copy()
    f = normalize(f)
    n = len(f)
    best_obj = objective(f)

    for _ in range(n_steps):
        conv = fast_convolve(f)
        k_max = int(np.argmax(conv))

        # Gradient of c_{k_max} w.r.t. f[i] = f[k_max - i] when valid
        grad = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = f[j]

        # Reduce high-gradient coordinates, increase low-gradient ones
        top_mask = grad > np.percentile(grad, 90)
        bot_mask = grad < np.percentile(grad, 10)

        if not np.any(top_mask) or not np.any(bot_mask):
            break

        mass = 0.05 * f[top_mask].sum()
        f[top_mask] -= mass / top_mask.sum() * grad[top_mask] / (grad[top_mask].max() + 1e-15)
        f[bot_mask] += mass / bot_mask.sum()
        f = normalize(f)

        obj = objective(f)
        if obj < best_obj - 1e-13:
            best_obj = obj
        else:
            f = normalize(f)  # revert (approximate)
            break

    return f, best_obj


def gen_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.08, 0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n / 2.0) / (sf * n)) ** 2)
        inits[f'g{int(sf * 100)}'] = normalize(g)

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    w = max(1, n // 4)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    for freq in [1, 2, 3, 5]:
        sp = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n))
        inits[f'cos{freq}'] = normalize(sp)

    for period in [3, 5, 7, 13]:
        pat = np.zeros(n)
        pat[::period] = 1.0
        if pat.sum() >= n // 5:
            inits[f'per{period}'] = normalize(pat)

    alt = np.ones(n)
    alt[::2] = 2.5
    inits['alt'] = normalize(alt)

    # Random Dirichlet with different concentrations
    for conc in [0.5, 2.0, 10.0]:
        rd = rng.dirichlet(np.ones(n) * conc)
        inits[f'dir{int(conc)}'] = rd

    tri = np.maximum(1.0 - np.abs(x - n / 2.0) / (n / 2.0), 0.03)
    inits['tri'] = normalize(tri)

    # W-shape
    wsh = np.zeros(n)
    wsh[:n // 6] = 1.0
    wsh[n // 3:2 * n // 3] = 0.3
    wsh[-n // 6:] = 1.0
    inits['wsh'] = normalize(wsh)

    return inits
