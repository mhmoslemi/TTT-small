"""
Optimizes non-negative step function to minimize normalized peak autocorrelation.
Uses sparse LP-based iterative refinement with multiple starting points,
adaptive blending, and coordinate-descent final polish.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.sparse import lil_matrix
from scipy.optimize import linprog
import cvxpy as cp

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))

def normalize_seq(a, S=None):
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s

def build_sparse_conv_matrix(a):
    """Build sparse matrix A where (A@g)[k] = sum_j a[k-j]*g[j] (convolution)."""
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_sparse(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0 using sparse matrix."""
    n = len(a)
    A = build_sparse_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)

def solve_lp_cvxpy(a, M):
    """Solve LP using cvxpy with SCS or ECOS."""
    n = len(a)
    A = build_sparse_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    # A is (2n-1) x n, we need A @ g <= M
    # But cvxpy uses A @ x <= b format
    # Actually for sparse, we can use the constraint directly
    conv_constraint = A @ g <= M
    constraints = [conv_constraint]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS)
    except:
        prob.solve(solver=cp.SCS)
    if g.value is not None:
        return np.maximum(g.value, 0)
    return np.zeros(n)

def lp_refine_fast(a, time_limit, verbose=True):
    """Fast LP-based iterative refinement with sparse matrices."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    
    # Precompute the matrix structure (it changes as a changes, so we rebuild each time)
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        if time.monotonic() - t0 > time_limit:
            break
        
        g = solve_lp_sparse(a, M)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        c = normalize_seq(g)
        
        # Line search for best blend
        best_t = 0.0
        best_new_conv = M
        for t_try in np.linspace(0.5, 0.02, 25):
            b = (1 - t_try) * a + t_try * c
            b = np.clip(b, 0.0, None)
            b = normalize_seq(b, S)
            new_conv = np.max(fft_convolve(b, b))
            if new_conv < best_new_conv:
                best_new_conv = new_conv
                best_t = t_try
        
        if best_t < 0.02:
            # Try even smaller steps
            for t_try in np.linspace(0.05, 0.005, 10):
                b = (1 - t_try) * a + t_try * c

                b = normalize_seq(b, S)
                new_conv = np.max(fft_convolve(b, b))
                if new_conv < best_new_conv:
                    best_new_conv = new_conv
                    best_t = t_try
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * c
        a = np.clip(a, 0.0, None)
        a = normalize_seq(a, S)
        
        new_conv_max = np.max(fft_convolve(a, a))
        if new_conv_max < best_conv_max:
            best_conv_max = new_conv_max
            best_a = a.copy()
        
        iteration += 1
        if verbose and iteration % 50 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    LP iter {iteration}: conv_max={best_conv_max:.6f}, score={score:.6f}")
        
        if best_new_conv >= best_conv_max * (1 - 1e-10):
            break
    
    return best_a

def peak_shaving(a, time_limit, verbose=False):
    """
    Local search that specifically targets reducing the current peak lag.
    Uses a combination of coordinate descent and block updates.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]
        
        if current_peak < best_peak:
            best_peak = current_peak
            best_a = a.copy()
        
        # Compute contribution of each pair to the peak
        # conv[peak_idx] = sum_i a[i] * a[peak_idx - i]
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        # Strategy: reduce elements with high contribution, increase others
        top_k = max(5, n // 10)
        top_idx = np.argsort(contribs)[-top_k:]
        
        # Try multiple random perturbations
        for trial in range(20):
            candidate = a.copy()
            
            # Reduce top contributors
            reduce_count = max(2, top_k // 2)
            reduce_idx = rng.choice(top_idx, size=reduce_count, replace=False)
            for idx in reduce_idx:
                factor = np.exp(rng.uniform(-0.3, 0.05))
                candidate[idx] *= factor
            
            # Increase some other elements
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                inc_count = max(2, n // 25)
                inc_idx = rng.choice(other, size=inc_count, replace=False)
                for idx in inc_idx:
                    factor = np.exp(rng.uniform(0.0, 0.3))
                    candidate[idx] *= factor
            
            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize_seq(candidate, S)
            
            new_peak = np.max(fft_convolve(candidate, candidate))
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = candidate.copy()
                a = candidate
            elif new_peak < current_peak * 1.02 and rng.random() < 0.2:
                a = candidate
        
        # Also try block operations
        if rng.random() < 0.3:
            width = max(2, n // 20)
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.05, size=width))
            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize_seq(candidate, S)
            new_peak = np.max(fft_convolve(candidate, candidate))
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = candidate.copy()
                a = candidate
    
    return best_a

def multiplicative_weight_descent(a, time_limit, verbose=False):
    """
    Multiplicative weight descent for minimizing the peak autocorrelation.
    Uses the subgradient at the peak lag with geometric updates.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    lr = 0.02  # Learning rate for multiplicative updates
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        
        # Subgradient of conv(a,a)[peak_idx] w.r.t. a_j
        # = a[peak_idx - j] + a[j - peak_idx] (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            i1 = peak_idx - j
            if 0 <= i1 < n:
                grad[j] += a[i1]
            i2 = j - peak_idx
            if 0 <= i2 < n:
                grad[j] += a[i2]
        
        # Also consider the constraint sum(a) = S
        # Projected multiplicative update
        # a_j <- a_j * exp(-lr * grad_j) * exp(lr * mean(grad))  [to preserve sum approximately]
        grad_centered = grad - np.mean(grad)
        a = a * np.exp(-lr * grad_centered)
        a = np.clip(a, 1e-6, None)
        a = normalize_seq(a, S)
        
        new_peak = np.max(fft_convolve(a, a))
        if new_peak < best_peak - 1e-12:
            best_peak = new_peak
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                lr *= 0.7
                if lr < 1e-5:
                    break
        
        if verbose and stall % 200 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    MWD: peak={best_peak:.6f}, score={score:.6f}, lr={lr:.6f}")
    
    return best_a

def make_initialization(n, init_type, rng):

    elif init_type == 'multi_tone':

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def build_sparse_conv_matrix(a):
    """Build sparse convolution matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    n = len(a)

        if res.status == 0:
            return res.x
    except Exception:
        pass
    return np.zeros(n)

        if verbose and iteration % 30 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    LP iter {iteration}: conv_max={best_conv_max:.6f}, score={score:.6f}, t={best_t:.3f}")

        if best_new_conv >= best_conv_max * (1 - 1e-9):
            break

    return best_a


def mwd_polish(a, time_limit, verbose=False):
    """Multiplicative weight descent to reduce peak autocorrelation."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))

    lr = 0.03
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))

        # Subgradient at peak
        grad = np.zeros(n)
        for j in range(n):
            i1 = peak_idx - j
            if 0 <= i1 < n:
                grad[j] += a[i1]
            i2 = j - peak_idx
            if 0 <= i2 < n:
                grad[j] += a[i2]

        # Center gradient to preserve sum
        grad_centered = grad - np.mean(grad)
        a = a * np.exp(-lr * grad_centered)
        a = np.clip(a, 1e-8, None)
        a = normalize_seq(a, S)

        new_peak = np.max(fft_convolve(a, a))
        if new_peak < best_peak - 1e-10:
            best_peak = new_peak
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.7
                if lr < 5e-5:
                    break

        if verbose and stall % 200 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    MWD: peak={best_peak:.6f}, score={score:.6f}, lr={lr:.6f}")

    return best_a


def aggressive_polish(a, time_limit, verbose=False):
    """Aggressive local search targeting the peak lag with diverse perturbations."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]

        if current_peak < best_peak:
            best_peak = current_peak
            best_a = a.copy()

        # Compute contributions to peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_k = max(5, n // 8)
        top_idx = np.argsort(contribs)[-top_k:]

        # Multiple perturbation strategies
        for strategy in range(15):
            candidate = a.copy()

            if strategy < 6:
                # Reduce top contributors, boost others
                n_red = max(2, top_k // 3)
                red_idx = rng.choice(top_idx, size=n_red, replace=False)
                for idx in red_idx:
                    candidate[idx] *= np.exp(rng.uniform(-0.4, -0.05))

                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 0:
                    n_inc = max(3, n // 15)
                    inc_idx = rng.choice(other, size=min(n_inc, len(other)), replace=False)
                    for idx in inc_idx:
                        candidate[idx] *= np.exp(rng.uniform(0.05, 0.4))
            elif strategy < 10:
                # Block perturbation
                width = max(2, n // 25)
                start = int(rng.integers(0, max(1, n - width)))
                candidate[start:start + width] *= np.exp(rng.normal(0, 0.08, size=width))
            elif strategy < 13:
                # Swap-like: reduce high, increase low
                high_mask = contribs > np.percentile(contribs, 70)
                low_mask = contribs < np.percentile(contribs, 30)
                n_swap = max(3, n // 20)
                high_idx = np.where(high_mask)[0]
                low_idx = np.where(low_mask)[0]
                if len(high_idx) > 0 and len(low_idx) > 0:
                    h_pick = rng.choice(high_idx, size=min(n_swap, len(high_idx)), replace=False)
                    l_pick = rng.choice(low_idx, size=min(n_swap, len(low_idx)), replace=False)
                    for idx in h_pick:
                        candidate[idx] *= np.exp(rng.uniform(-0.3, -0.05))
                    for idx in l_pick:
                        candidate[idx] *= np.exp(rng.uniform(0.05, 0.3))
            else:
                # Small random perturbation
                candidate *= np.exp(rng.normal(0, 0.03, n))

            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize_seq(candidate, S)

            new_peak = np.max(fft_convolve(candidate, candidate))
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = candidate.copy()
                a = candidate
            elif new_peak < current_peak * 1.03 and rng.random() < 0.15:
                a = candidate

        if verbose:
            elapsed = time.monotonic() - t0
            if int(elapsed * 10) % 50 == 0:
                score = 2.0 * n * best_peak / (S * S)
                print(f"    Polish t={elapsed:.0f}s: score={score:.6f}")

    return best_a


def make_init(n, init_type, rng):
    """Create diverse initial sequences."""
    t = np.arange(n, dtype=np.float64)
    x = t / max(1, n - 1)  # normalized to [0,1]

    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'u_shape':
        # Large at edges, small in middle
        depth = 0.3 + 0.4 * rng.random()
        a = 1.0 - depth * np.cos(np.pi * x) ** 2
    elif init_type == 'u_shape2':
        depth = 0.2 + 0.5 * rng.random()
        a = 1.0 - depth * np.sin(np.pi * x)
    elif init_type == 'w_shape':
        a = 1.0 - 0.3 * np.sin(2 * np.pi * x) ** 2
    elif init_type == 'chirp':
        alpha = 0.2 + 0.4 * rng.random()
        rate = 0.5 + 1.0 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * rate * (2 * x - 1) ** 2)
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 4))
        amp = 0.15 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * x + phase)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for _ in range(2 + int(rng.integers(0, 3))):
            f = int(rng.integers(1, n // 4 + 1))
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'random':
        a = 1.0 + 0.25 * rng.standard_normal(n)
    elif init_type == 'edge_heavy':
        # Very large at edges
        sigma = 0.1 + 0.1 * rng.random()
        a = 1.0 + 0.8 * np.exp(-0.5 * (x / sigma) ** 2) + 0.8 * np.exp(-0.5 * ((1 - x) / sigma) ** 2)
    else:
        a = np.ones(n)

    a = np.clip(a, 0.01, None)
    return a


def interpolate_seq(a, n_new):
    """Linear interpolation to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Sparse LP refinement from provided sequence and diverse inits
    2. MWD polish
    3. Aggressive local search
    4. Final LP + polish cycle
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)

    # Phase 1: LP on provided sequence (most important)
    print("\n=== Phase 1: LP on provided ===")
    if provided_seq is not None:
        n = len(provided_seq)
        t_limit = min(400.0, (deadline - time.monotonic()) * 0.22)
        print(f"  n={n}, limit={t_limit:.0f}s")
        result = lp_refine(provided_seq, t_limit, verbose=True)
        if update_best(result):
            print(f"  *** LP improved: {best_score:.6f} ***")

    # Phase 2: Try other lengths from provided
    print("\n=== Phase 2: Multi-length LP ===")
    if provided_seq is not None:
        for n_try in [512, 576, 640, 768, 896, 1024]:
            if time.monotonic() > deadline - 400:
                break
            if n_try == len(provided_seq):
                continue
            a0 = interpolate_seq(provided_seq, n_try)
            a0 = np.clip(a0, 0.01, None)
            t_per = min(50.0, (deadline - time.monotonic()) * 0.06)
            result = lp_refine(a0, t_per, verbose=False)
            if update_best(result):
                print(f"  *** n={n_try}: {best_score:.6f} ***")

    # Phase 3: Structured inits with LP
    print("\n=== Phase 3: Structured inits ===")
    init_configs = [
        (640, 'u_shape'), (640, 'edge_heavy'), (512, 'u_shape2'),
        (768, 'w_shape'), (640, 'chirp'), (512, 'edge_heavy'),
    ]
    for n_try, init_type in init_configs:
        if time.monotonic() > deadline - 300:
            break
        a0 = make_init(n_try, init_type, rng)
        # Quick LP (short time)
        t_per = min(25.0, (deadline - time.monotonic()) * 0.05)
        result = lp_refine(a0, t_per, verbose=False)
        if update_best(result):
            print(f"  *** n={n_try}, {init_type}: {best_score:.6f} ***")

    # Phase 4: MWD on best
    print("\n=== Phase 4: MWD polish ===")
    if best_seq is not None:
        t_limit = min(100.0, (deadline - time.monotonic()) * 0.12)
        print(f"  limit={t_limit:.0f}s")
        result = mwd_polish(best_seq, t_limit, verbose=True)
        if update_best(result):
            print(f"  *** MWD: {best_score:.6f} ***")

    # Phase 5: Aggressive local search
    print("\n=== Phase 5: Aggressive polish ===")
    if best_seq is not None:
        t_limit = min(200.0, (deadline - time.monotonic()) * 0.2)
        print(f"  limit={t_limit:.0f}s")
        result = aggressive_polish(best_seq, t_limit, verbose=True)
        if update_best(result):
            print(f"  *** Polish: {best_score:.6f} ***")

    # Phase 6: Final LP on best
    print("\n=== Phase 6: Final LP ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        if remaining > 30:
            t_limit = min(remaining, 200.0)
            print(f"  limit={t_limit:.0f}s")
            result = lp_refine(best_seq, t_limit, verbose=True)
            if update_best(result):
                print(f"  *** Final LP: {best_score:.6f} ***")

    # Phase 7: Final aggressive polish
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15
        if remaining > 15:
            print(f"\n=== Phase 7: Final polish ({remaining:.0f}s) ===")
            result = aggressive_polish(best_seq, remaining, verbose=False)
            update_best(result)

    # Final
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)}, local={best_score:.6f} ===")
        return final_list

    return [1.0] * 256
