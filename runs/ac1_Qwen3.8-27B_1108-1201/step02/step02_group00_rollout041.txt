"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp gradient descent
with adaptive temperature, momentum, and multi-phase optimization.
Key innovations: smooth surrogate for max via log-sum-exp, adaptive beta
scheduling, Euclidean simplex projection, and hybrid gradient/subgradient
switching for fine-tuning.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def logsumexp_gradient(a, conv, beta):
    """
    Compute the gradient of log(sum(exp(beta * conv))) w.r.t. a.
    grad_j = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
    where w[k] = softmax(beta * conv[k])
    """
    n = len(a)
    scaled = beta * conv

def subgradient_top_peaks(a, conv, top_k):
    """Compute subgradient using only top-k peaks of convolution."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
    
    # softmax over peaks
    scaled = peak_vals
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        wi = w[i] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    
    return sg


def optimize_lse(a0, time_limit, beta_init=5.0, beta_final=500.0,

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    ramp_inc /= ramp_inc.sum()
    inits['ramp_inc'] = ramp_inc
    inits['ramp_dec'] = ramp_inc[::-1].copy()
    
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri
    
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.1f}'] = g
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Edge-concentrated
    edge_conc = np.exp(-x / (0.1 * n)) + np.exp(-(n-1-x) / (0.1 * n))
    edge_conc /= edge_conc.sum()
    inits['edge'] = edge_conc
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Power law
    power = 1.0 / (1.0 + x)
    power /= power.sum()
    inits['power'] = power
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits
