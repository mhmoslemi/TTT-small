"""
Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Uses Adam-based gradient descent in log-softmax
parameterization with soft-max annealing, LP-based refinement, and simulated
annealing for final polishing. Key improvements over previous approach:
- More aggressive and sustained gradient descent with adaptive learning rates
- Multiple diverse initializations including frequency-domain constructions
- LP refinement interleaved with gradient steps for precision
- Better time management and longer optimization runs
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_gradient_soft(a, tau):
    """Compute soft-peak objective and its gradient w.r.t. a (unconstrained)."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    soft_peak_val = max_c + tau * np.log(sum_exp)

    # Gradient: g[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    # B2[i] = sum_j w[i-j] * a[j] = conv(w, a)[i] for i in [0, n-1]
    # B1[i] = sum_j w[i+j] * a[j]
    
    w_rev = w[::-1].copy()
    B2_full = fft_convolve(w, a)  # length (2n-1)+n-1 = 3n-2
    B2 = B2_full[:n]
    
    B1_full = fft_convolve(w_rev, a)  # length 3n-2
    # B1[i] = B1_full[2n-2-i] for i in [0, n-1]
    B1 = B1_full[n-1:2*n-1][::-1]
    
    g = B1 + B2
    return soft_peak_val, g, conv


def adam_gradient_descent(a_init, time_limit, seed=0, verbose=False):
    """Adam-based gradient descent in log-softmax parameterization with soft-max annealing."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    
    a_safe = np.clip(a_init, 1e-10, None)
    theta = np.log(a_safe) - np.log(np.sum(a_safe))
    
    best_theta = theta.copy()
    best_score = local_score(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step_count = 0
    
    t0 = time.monotonic()
    tau_start = 8.0 / n
    tau_end = 0.1 / n
    base_lr = 2.0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        
        # Anneal tau: start smooth, end sharp
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        
        # Compute a from theta
        t_shifted = theta - np.max(theta)
        exp_t = np.exp(t_shifted)
        a = exp_t / np.sum(exp_t) * S
        
        # Compute gradient
        _, g_a, _ = compute_gradient_soft(a, tau)
        
        # Chain rule through softmax: g_theta[i] = a[i] * (g_a[i] - dot(g_a,a)/S)
        inner = np.dot(g_a, a) / S
        g_theta = a * (g_a - inner)
        
        # Adam update
        step_count += 1
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** step_count)
        v_hat = v / (1.0 - beta2 ** step_count)
        
        # Adaptive learning rate
        lr = base_lr / (1.0 + 0.005 * step_count) ** 0.5
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        t_shifted = theta - np.max(theta)
        a_check = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
        score = local_score(a_check)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
        
        # Periodic random restart of momentum to escape local minima
        if step_count % 2000 == 0:
            m = 0.5 * m
        
        if verbose and step_count % 200 == 0:
            print(f"        GD step={step_count}, tau={tau:.5f}, score={score:.6f}, best={best_score:.6f}")
    
    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)


def build_constraint_A(f):
    """Build constraint matrix for LP: A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. A@conjugate(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_constraint_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})
    if result.success:
        return result.x
    return None


def lp_refine(f_init, time_limit, verbose=False):
    """LP-based iterative refinement."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = 0.25
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit - 5:
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        g = solve_lp(f, M, time_limit=5)
        if g is None:
            if stagnant > 10:
                break
            stagnant += 1
            continue
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g_rescaled = g * (sum_f / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (sum_f / sum_new)
        
        score_new = local_score(f_new)
        
        if score_new < best_score - 1e-14:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
        else:
            f = f_new
            stagnant += 1
            if stagnant > 5:
                t = max(0.05, t * 0.8)
                stagnant = 0
            if stagnant > 20:
                # Perturb to escape
                f = f + 0.01 * np.random.default_rng(int(time.monotonic()*1000)).standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                t = 0.15
                stagnant = 0
    
    return best_f, best_score


def sa_polish(f_init, time_limit, rng):
    """Simulated annealing for final polishing."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.004
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.3:
            # Peak-targeted reduction
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(8, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 5))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.75:
            # Dual block
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 10))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.9:
            # Smooth global perturbation
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Small random kick
            idx = rng.integers(0, n)
            candidate[idx] *= np.exp(rng.normal(0.0, 0.1))
        
        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best_f, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(8):
        k = int(rng.integers(1, max(2, n // 2)))
        amp = rng.uniform(0.03, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_gaussian_bump_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - n//2) / (n * 0.25)) ** 2)
    a += 0.3 * np.ones(n)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_banded_init(n, rng, band_width=None):
    if band_width is None:
        band_width = n // 4
    a = np.zeros(n)
    center = n // 2
    start = max(0, center - band_width // 2)
    end = min(n, start + band_width)
    a[start:end] = 1.0 + 0.2 * np.sin(2 * np.pi * np.arange(end - start) / (end - start))
    a += 0.02
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def interpolate_to_length(seq, n):
    if len(seq) == n:
        return seq.copy()
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Adam gradient descent on multiple initializations (60% time)
    Phase 2: LP refinement on best candidates (20% time)
    Phase 3: SA polishing (10% time)
    Phase 4: Final LP + verification (10% time)
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.001, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()
    
    available_time = deadline - time.monotonic()
    print(f"Available time: {available_time:.0f}s")
    
    # === Phase 1: Gradient descent on multiple initializations ===
    print("\n" + "=" * 60)
    print("Phase 1: Adam gradient descent")
    print("=" * 60)
    
    n_base = len(provided_seq) if provided_seq is not None else 256
    
    # Define search configurations: (n, init_type, time_fraction)
    configs = [
        (n_base, 'provided', 0.30),
        (n_base, 'chirp', 0.15),
        (n_base, 'multitone', 0.12),
        (512, 'provided_interp', 0.12),
        (n_base, 'gauss', 0.10),
        (n_base, 'banded', 0.10),
        (384, 'chirp', 0.08),
        (n_base, 'random', 0.03),
    ]
    
    total_frac = sum(c[2] for c in configs)
    p1_time = available_time * 0.58
    
    for n, init_type, frac in configs:
        if time.monotonic() > deadline - 60:
            break
        
        time_alloc = max(10.0, frac / total_frac * p1_time)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 30)
        if time_alloc < 8:
            break
        
        # Generate initialization
        if init_type == 'provided' and provided_seq is not None and n == len(provided_seq):
            a0 = provided_seq.copy()
        elif init_type == 'provided_interp' and provided_seq is not None:
            a0 = interpolate_to_length(provided_seq, n)
        elif init_type == 'chirp':
            alpha = 0.3 + 0.3 * rng.random()
            a0 = make_chirp_init(n, alpha, rng)
        elif init_type == 'multitone':
            a0 = make_multitone_init(n, rng)
        elif init_type == 'gauss':
            a0 = make_gaussian_bump_init(n, rng)
        elif init_type == 'banded':
            a0 = make_banded_init(n, rng)
        else:
            a0 = 1.0 + 0.3 * rng.standard_normal(n)
            a0 = np.clip(a0, 0.01, None)
        
        print(f"\n  n={n}, init='{init_type}', alloc={time_alloc:.0f}s")
        t_start = time.monotonic()
        s_init = local_score(a0)
        print(f"    Initial score: {s_init:.6f}")
        
        a_best, score = adam_gradient_descent(a0, time_alloc, seed=int(rng.integers(0, 10000)))
        elapsed = time.monotonic() - t_start
        print(f"    GD done: score={score:.6f}, time={elapsed:.1f}s")
        
        if score < best_score:
            best_score = score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} (n={n}, init={init_type}) ***")
    
    # === Phase 2: LP refinement on best candidate ===
    print("\n" + "=" * 60)
    print("Phase 2: LP refinement")
    print("=" * 60)
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        lp_time = min(remaining * 0.25, 300.0)
        if lp_time > 20:
            print(f"  LP refining n={len(best_seq)}, time={lp_time:.0f}s")
            t_start = time.monotonic()
            a_lp, score_lp = lp_refine(best_seq, lp_time)
            elapsed = time.monotonic() - t_start
            print(f"  LP result: {score_lp:.6f} (time={elapsed:.0f}s)")
            if score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # === Phase 3: Additional GD from LP result ===
    print("\n" + "=" * 60)
    print("Phase 3: Additional gradient descent from LP result")
    print("=" * 60)
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        gd2_time = min(remaining * 0.15, 150.0)
        if gd2_time > 15:
            print(f"  GD from best, n={len(best_seq)}, time={gd2_time:.0f}s")
            a_gd2, score_gd2 = adam_gradient_descent(best_seq, gd2_time, seed=int(rng.integers(0, 10000)))
            print(f"  GD2 result: {score_gd2:.6f}")
            if score_gd2 < best_score:
                best_score = score_gd2
                best_seq = a_gd2.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # === Phase 4: SA polishing ===
    print("\n" + "=" * 60)
    print("Phase 4: Simulated annealing polish")
    print("=" * 60)
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(remaining * 0.2, 200.0)
        if sa_time > 15:
            print(f"  SA polish, n={len(best_seq)}, time={sa_time:.0f}s")
            a_sa, score_sa = sa_polish(best_seq, sa_time, rng)
            print(f"  SA result: {score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # === Phase 5: Final LP refinement ===
    print("\n" + "=" * 60)
    print("Phase 5: Final LP refinement")
    print("=" * 60)
    
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        final_lp_time = min(remaining * 0.8, 100.0)
        if final_lp_time > 10:
            print(f"  Final LP, n={len(best_seq)}, time={final_lp_time:.0f}s")
            a_final, score_final = lp_refine(best_seq, final_lp_time)
            print(f"  Final LP result: {score_final:.6f}")
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
