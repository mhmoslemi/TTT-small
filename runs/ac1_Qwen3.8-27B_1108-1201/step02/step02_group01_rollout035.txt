```python
"""
Softmax-smoothed gradient descent with FFT-based gradient computation for
minimizing peak autocorrelation of non-negative step functions. Uses
exponential parameterization, beta annealing, and LP-based refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_to_sum(a, S):
    """Normalize sequence to have sum S."""
    s = np.sum(a)
    if s < 1e-12:
        return np.full_like(a, S / len(a))
    return a * S / s


def project_onto_simplex(a, S):
    """Project a onto {x >= 0, sum(x) = S} using bisection."""
    lo = a.min() - 1.0
    hi = a.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.maximum(a - mid, 0.0)
        s = projected.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    projected = np.maximum(a - (lo + hi) / 2.0, 0.0)
    s = projected.sum()
    if s > 1e-15:
        projected = projected * S / s
    return projected


def compute_softmax_gradient(a, p):
    """
    Compute gradient of smoothed max convolution w.r.t. a.
    grad[m] = 2 * sum_j a[j] * p[j+m]
    Computed via cross-correlation with FFT.
    """
    n = len(a)
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, p)
   grad = 2.0 * cross[n - 1: 2 * n - 1]
    return grad


def softmax_gradient_descent(a_init, time_limit, beta_start=5.0, beta_end=5000.0,
                              lr=0.5, use_adam=True, verbose=True):
    """
    Softmax-smoothed gradient descent to minimize max(conv(a,a)).
    Uses exponential parameterization for non-negativity.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(np.maximum(a_init, 1e-6), S)
    t0 = time.monotonic()

    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    best_improve_iter = 0
    iter_count = 0
    no_improve_count = 0

    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    adam_step = 0

    # Learning rate schedule
    lr_base = lr
    patience = 2000

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Annealing: beta increases over time
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)

        # Compute convolution
        conv = fft_convolve(a, a)
        conv_max = conv.max()

        # Compute softmax weights (with numerical stability)
        conv_shifted = conv - conv_max
        conv_shifted = np.clip(conv_shifted, -200.0 / max(beta, 1.0), 0.0)
        exp_vals = np.exp(beta * conv_shifted)
        p = exp_vals / exp_vals.sum()

        # Compute gradient
        grad_a = compute_softmax_gradient(a, p)

        # Convert to log-space gradient:
        # dF/db[j] = a[j] * (grad_a[j] - sum(grad_a * a) / S)
        weighted_mean = np.dot(grad_a, a) / S
        grad_b = a * (grad_a - weighted_mean)

        # Adam update in log-space
        adam_step += 1
        m_state = beta1 * m_state + (1 - beta1) * grad_b
        v_state = beta2 * v_state + (1 - beta2) * grad_b ** 2
        m_hat = m_state / (1 - beta1 ** adam_step)
        v_hat = v_state / (1 - beta2 ** adam_step)

        # Adaptive learning rate
        lr_current = lr_base / (1.0 + 0.05 * adam_step)
        b_update = lr_current * m_hat / (np.sqrt(v_hat) + eps)

        # Update in log-space
        # a = S * exp(b) / sum(exp(b)) → we work directly with log(a/S)
        # New b = b - lr * grad_b
        # New a = S * exp(new_b) / sum(exp(new_b))
        log_a = np.log(a / S + 1e-30)
        log_a = log_a - b_update
        # Stability
        log_a = log_a - log_a.max()
        a = S * np.exp(log_a) / np.exp(log_a).sum()

        # Safety: ensure non-negative and correct sum
        a = np.maximum(a, 1e-15)
        a = normalize_to_sum(a, S)

        iter_count += 1

        # Check improvement
        if conv_max < best_conv_max - 1e-10:
            best_conv_max = conv_max
            best_a = a.copy()
            best_improve_iter = iter_count
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Restart with perturbation if stalled
        if no_improve_count > patience:
            perturbation_scale = 0.1 * (no_improve_count / patience)
            perturbation_scale = min(perturbation_scale, 0.5)
            a = a * np.exp(np.random.default_rng(iter_count).normal(0, perturbation_scale, n))
            a = normalize_to_sum(np.maximum(a, 1e-10), S)
            no_improve_count = 0

        if verbose and iter_count % 1000 == 0:
            score = 2.0 * n / (S * S) * best_conv_max
            print(f"      SGD iter {iter_count}: beta={beta:.1f}, score={score:.6f}, "
                  f"conv_max={best_conv_max:.6f}")

    final_score = 2.0 * n * best_conv_max / (S * S)
    return best_a, final_score


def lp_refine(a, time_limit, verbose=False):
    """
    LP-based iterative refinement (improved version).
    Solves: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0
    Then blends.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))

    for iteration in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        # Build LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0
        # conv(a,g)[k] = sum_i a[i] * g[k-i] (with bounds)
        # This is A @ g <= M where A[k,j] = a[k-j] if valid
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            A[k, j_min:j_max + 1] = a[k - (j_min:j_max + 1)]

        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, time_limit - (time.monotonic() - t0))})
        except Exception:
            break

        if res.status != 0:
            break

        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        g_norm = normalize_to_sum(g, S)

        # Find best blend
        best_t = 0.0
        best_new_max = M
        for t_try in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            b = (1 - t_try) * a + t_try * g_norm
            b = np.maximum(b, 0)
            b = normalize_to_sum(b, S)
            new_max = np.max(fft_convolve(b, b))
            if new_max < best_new_max:
                best_new_max = new_max
                best_t = t_try

        if best_t == 0.0:
            break

        a = (1 - best_t) * a + best_t * g_norm
        a = np.maximum(a, 0)
        a = normalize_to_sum(a, S)

        new_conv_max = np.max(fft_convolve(a, a))
        if new_conv_max < best_conv_max - 1e-12:
            best_conv_max = new_conv_max
            best_a = a.copy()

        if best_t < 0.005:
            break

    final_score = 2.0 * n * best_conv_max / (S * S)
    return best_a, final_score


def targeted_perturbation(a, time_limit, verbose=False):
    """
    Targeted local search: perturb elements that contribute to the peak.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(np.maximum(a, 1e-10), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(42)
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]

        # Compute contributions to the peak
        # conv[peak_idx] = sum_i a[i] * a[peak_idx - i]
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        # Strategy: reduce top contributors, increase others
        top_k = max(5, n // 8)
        top_idx = np.argsort(contribs)[-top_k:]

        for attempt in range(10):
            candidate = a.copy()
            # Reduce top contributors
            reduce_frac = rng.uniform(0.02, 0.15)
            n_reduce = rng.integers(max(2, top_k // 4), top_k)
            chosen_reduce = rng.choice(top_idx, size=n_reduce, replace=False)
            for idx in chosen_reduce:
                candidate[idx] *= (1.0 - reduce_frac * rng.uniform(0.5, 1.5))

            # Increase some non-top elements to maintain sum
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = rng.integers(max(3, n // 25), n // 10)
                n_inc = min(n_inc, len(other_idx))
                chosen_inc = rng.choice(other_idx, size=n_inc, replace=False)
                for idx in chosen_inc:
                    candidate[idx] *= (1.0 + reduce_frac * rng.uniform(0.5, 2.0))

            candidate = np.maximum(candidate, 0)
            candidate = normalize_to_sum(candidate, S)

            new_conv_max = np.max(fft_convolve(candidate, candidate))
            if new_conv_max < best_conv_max:
                best_conv_max = new_conv_max
                best_a = candidate.copy()
                a = candidate
            elif new_conv_max < current_peak * 1.02 and rng.random() < 0.2:
                a = candidate

        # Random block perturbation
        if rng.random() < 0.3:
            width = rng.integers(max(2, n // 50), n // 10)
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.05, width))
            candidate = np.maximum(candidate, 0)
            candidate = normalize_to_sum(candidate, S)
            new_conv_max = np.max(fft_convolve(candidate, candidate))
            if new_conv_max < best_conv_max:
                best_conv_max = new_conv_max
                best_a = candidate.copy()
                a = candidate

        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"      Targeted iter {iter_count} ({elapsed:.0f}s): score={score:.6f}")

    final_score = 2.0 * n * best_conv_max / (S * S)
    return best_a, final_score


def make_initialization(n, init_type, rng):
    """Create various initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'gaussian_bump':
        center = n // 2
        sigma = n * rng.uniform(0.2, 0.4)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'anti_symmetric':
        # High in first half, low in second half (reduces center peak)
        a = np.ones(n)
        transition = n // 8
        for i in range(n):
            if i < n // 2 - transition:
                a[i] = 1.5
            elif i < n // 2 + transition:
                frac = (i - (n // 2 - transition)) / (2 * transition)
                a[i] = 1.5 - frac * 1.0
            else:
                a[i] = 0.5
    elif init_type == 'chirp':
        alpha = rng.uniform(0.3, 0.7)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * alpha * t * t / (n * n))
    elif init_type == 'sine':
        f = rng.integers(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining softmax gradient descent,
    LP refinement, and targeted local search.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            print(f"    *** NEW BEST: score={s:.6f}, n={len(a_clip)} ***")
            return True
        return False

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)

    # === Phase 1: Softmax gradient descent at multiple lengths ===
    print("\n=== Phase 1: Softmax Gradient Descent ===")

    # Priority lengths
    lengths_priority = [640, 512, 768, 896, 1024, 576, 704]

    for n in lengths_priority:
        if time.monotonic() > deadline - 400:
            break

        # Time allocation
        n_remaining = len(lengths_priority) - lengths_priority.index(n)
        t_budget = min(150.0, (deadline - time.monotonic() - 300.0) / n_remaining)
        if t_budget < 20:
            t_budget = 20.0

        # Multiple initializations
        inits = []
        if provided_seq is not None:
            inits.append(('interp', interpolate_sequence(provided_seq, n)))
        inits.append(('gaussian', make_initialization(n, 'gaussian_bump', rng)))
        inits.append(('anti_sym', make_initialization(n, 'anti_symmetric', rng)))

        for init_name, a0 in inits:
            if time.monotonic() > deadline - 350:
                break
            a0 = np.maximum(a0, 0.01)
            result, score = softmax_gradient_descent(
                a0, t_budget / len(inits),
                beta_start=5.0, beta_end=3000.0,
                lr=0.5, verbose=False
            )
            if score < best_score:
                update_best(result)
                print(f"  n={n}, init={init_name}: score={score:.6f}")

    print(f"  Phase 1 best: {best_score:.6f}")

    # === Phase 2: Intensified softmax gradient on best ===
    print("\n=== Phase 2: Intensive Softmax Refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(200.0, (deadline - time.monotonic()) * 0.35)
        print(f"  n={n}, time_limit={t_limit:.0f}s")

        # High beta, long run
        result, score = softmax_gradient_descent(
            best_seq, t_limit,
            beta_start=50.0, beta_end=20000.0,
            lr=0.3, verbose=True
        )
        update_best(result)
        print(f"  Phase 2 best: {best_score:.6f}")

    # === Phase 3: LP refinement ===
    print("\n=== Phase 3: LP Refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(120.0, (deadline - time.monotonic()) * 0.3)
        if t_limit > 20:
            print(f"  LP refine n={n}, time_limit={t_limit:.0f}s")
            result, score = lp_refine(best_seq, t_limit, verbose=False)
            update_best(result)
            print(f"  Phase 3 best: {best_score:.6f}")

    # === Phase 4: Targeted perturbation ===
    print("\n=== Phase 4: Targeted Local Search ===")
    if best_seq is not None:
        t_limit = min(120.0, (deadline - time.monotonic()) * 0.4)
        if t_limit > 20:
            print(f"  Targeted search, time_limit={t_limit:.0f}s")
            result, score = targeted_perturbation(best_seq, t_limit, verbose=True)
            update_best(result)

    # === Phase 5: Final softmax polish ===
    print("\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15
        if remaining > 30:
            result, score = softmax_gradient_descent(
                best_seq, remaining,
                beta_start=200.0, beta_end=50000.0,
                lr=0.2, verbose=False
            )
            update_best(result)

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local_best={best_score:.6f} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
```
