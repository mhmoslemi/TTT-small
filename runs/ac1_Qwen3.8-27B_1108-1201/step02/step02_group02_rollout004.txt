Coordinate descent optimization for minimizing peak autocorrelation of
non-negative step functions. Uses exact 1D optimization per coordinate with
golden section search, combined with LP-based direction finding and
simulated annealing for global exploration.

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def fast_score(a):
    """Compute the evaluation score for a sequence (fast)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def compute_conv(a):
    """Compute convolution and score components."""
    c = fft_convolve(a, a)
    S = np.sum(a)
    n = len(a)
    score = 2.0 * n * np.max(c) / (S * S)
    return c, S, score

def lp_direction(f, M, relax=1.0):
    """Solve LP to find best direction with bounded cross-correlation."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M * relax)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def coord_descent_sweep(f, c, S, order, n_golden=30):
    """One pass of coordinate descent over all coordinates in given order."""
    n = len(f)
    gr = (np.sqrt(5) - 1) / 2
    
    for i in order:
        # Build d vector: d[k] = f[k-i] for valid k (0 <= k-i < n)
        # Valid k range: max(0,i) to min(i+n-1, 2n-2)
        d = np.zeros(2*n-1)
        k_start = i
        k_end = min(i + n - 1, 2*n - 2)
        d[k_start:k_end+1] = f[0:k_end - k_start + 1]
        
        # The quadratic correction at k = 2i
        has_quad = (2*i < 2*n - 1)
        quad_pos = 2 * i
        
        # Golden section search for optimal delta
        lo = -f[i] + 1e-10
        hi = max(f[i] * 2.0, 0.1) + 0.5
        
        a_g = lo
        b_g = hi
        c1 = b_g - gr * (b_g - a_g)
        c2 = a_g + gr * (b_g - a_g)
        
        def obj(delta):
            # c_new = c + 2*delta*d + delta^2 * (1 if k==2i else 0)
            new_max = np.max(c + 2.0 * delta * d)
            if has_quad:
                q_val = c[quad_pos] + 2.0 * delta * d[quad_pos] + delta * delta
                if q_val > new_max:
                    new_max = q_val
            new_S = S + delta
            if new_S < 0.01:
                return 1e10
            return new_max / (new_S * new_S)
        
        f1 = obj(c1)
        f2 = obj(c2)
        
        for _ in range(n_golden):
            if f1 < f2:
                b_g = c2
                c2 = c1
                f2 = f1
                c1 = b_g - gr * (b_g - a_g)
                f1 = obj(c1)
            else:
                a_g = c1
                c1 = c2
                f1 = f2
                c2 = a_g + gr * (b_g - a_g)
                f2 = obj(c2)
        
        optimal_d = (a_g + b_g) / 2.0

            # Update c
            c += 2.0 * optimal_d * d
            if has_quad:
                c[quad_pos] += optimal_d * optimal_d
            S += optimal_d
    
    return f, c, S

def resample_sequence(seq, new_n):

    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)

"""
Coordinate descent with exact 1D subproblem solving for minimizing peak
autocorrelation of non-negative step functions. Combines fast per-coordinate
golden section optimization with LP-based direction finding, random restarts,
and adaptive perturbation to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute the evaluation score for a sequence (fast local version)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)


def compute_conv(a):
    """Compute convolution, sum, and score."""
    c = fft_convolve(a, a)
    S = np.sum(a)
    n = len(a)
    score = 2.0 * n * np.max(c) / (S * S)
    return c, S, score


def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix A where A[k,i] = f[k-i] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, relax=1.0):
    """Solve LP to find best direction with cross-correlation bounded by M*relax."""
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def coord_descent_sweep(f, c, S, order, n_golden=22):
    """One pass of coordinate descent over all coordinates in given order."""
    n = len(f)
    gr = (np.sqrt(5) - 1) / 2
    improved = False

    for i in order:
        fi = f[i]
        if fi < 1e-12:
            lo = 1e-10
            hi = 0.5
        else:
            lo = -fi + 1e-10
            hi = fi * 2.0 + 0.1

        if lo >= hi:
            continue

        # Build d vector: d[k] = f[k-i] for valid k (0 <= k-i < n)
        d = np.zeros(2 * n - 1)
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        d[k_start:k_end + 1] = f[0:k_end - k_start + 1]

        has_quad = (2 * i < 2 * n - 1)
        quad_pos = 2 * i

        # Current objective at delta=0
        current_max = np.max(c)
        current_obj = current_max / (S * S)

        # Golden section search
        a_g = lo
        b_g = hi
        c1 = b_g - gr * (b_g - a_g)
        c2 = a_g + gr * (b_g - a_g)

        # Evaluate at c1
        new_max_c1 = np.max(c + 2.0 * c1 * d)
        if has_quad:
            qv = c[quad_pos] + 2.0 * c1 * d[quad_pos] + c1 * c1
            if qv > new_max_c1:
                new_max_c1 = qv
        S_c1 = S + c1
        f1 = new_max_c1 / (S_c1 * S_c1) if S_c1 > 0.01 else 1e10

        # Evaluate at c2
        new_max_c2 = np.max(c + 2.0 * c2 * d)
        if has_quad:
            qv = c[quad_pos] + 2.0 * c2 * d[quad_pos] + c2 * c2
            if qv > new_max_c2:
                new_max_c2 = qv
        S_c2 = S + c2
        f2 = new_max_c2 / (S_c2 * S_c2) if S_c2 > 0.01 else 1e10

        for _ in range(n_golden):
            if f1 < f2:
                b_g = c2
                c2 = c1
                f2 = f1
                c1 = b_g - gr * (b_g - a_g)
                new_max = np.max(c + 2.0 * c1 * d)
                if has_quad:
                    qv = c[quad_pos] + 2.0 * c1 * d[quad_pos] + c1 * c1
                    if qv > new_max:
                        new_max = qv
                S_new = S + c1
                f1 = new_max / (S_new * S_new) if S_new > 0.01 else 1e10
            else:
                a_g = c1
                c1 = c2
                f1 = f2
                c2 = a_g + gr * (b_g - a_g)
                new_max = np.max(c + 2.0 * c2 * d)
                if has_quad:
                    qv = c[quad_pos] + 2.0 * c2 * d[quad_pos] + c2 * c2
                    if qv > new_max:
                        new_max = qv
                S_new = S + c2
                f2 = new_max / (S_new * S_new) if S_new > 0.01 else 1e10

        optimal_d = (a_g + b_g) / 2.0

        # Evaluate final objective
        new_max_final = np.max(c + 2.0 * optimal_d * d)
        if has_quad:
            qv = c[quad_pos] + 2.0 * optimal_d * d[quad_pos] + optimal_d * optimal_d
            if qv > new_max_final:
                new_max_final = qv
        S_final = S + optimal_d
        final_obj = new_max_final / (S_final * S_final) if S_final > 0.01 else 1e10

        # Only accept if it improves
        if final_obj < current_obj - 1e-14 and abs(optimal_d) > 1e-12:
            f[i] = max(0.0, f[i] + optimal_d)
            c += 2.0 * optimal_d * d
            if has_quad:
                c[quad_pos] += optimal_d * optimal_d
            S += optimal_d
            improved = True

    return f, c, S, improved


def make_chirp_init(n, alpha, rng):
    """Create chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Create multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 8):
        amp = 0.2 / k
        freq = rng.uniform(0.3, 2.5) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Create random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_gradient_init(n, rng):
    """Create a gradient/ramp initialization (non-palindromic)."""
    t = np.arange(n, dtype=np.float64)
    a = 0.5 + 0.5 * t / n  # ramp
    a += 0.2 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_sequence(seq, new_n):
    """Resample sequence to new length via linear interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def optimize_from_start(f0, time_limit, rng, verbose=False):
    """Run coordinate descent optimization from a starting sequence."""
    n = len(f0)
    f = f0.astype(np.float64).copy()

    S = np.sum(f)
    if S < 0.01:
        f = np.ones(n) * 0.5
        S = np.sum(f)

    c, S, cur_score = compute_conv(f)
    best_f = f.copy()
    best_score = cur_score
    t0 = time.monotonic()
    sweep_count = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        f, c, S, did_improve = coord_descent_sweep(f, c, S, order, n_golden=22)
        sweep_count += 1

        # Recompute from scratch periodically to avoid numerical drift
        if sweep_count % 200 == 0:
            c_re = fft_convolve(f, f)
            S_re = np.sum(f)
            c = c_re
            S = S_re

        new_score = 2.0 * n * np.max(c) / (S * S)

        if did_improve and new_score < cur_score - 1e-10:
            cur_score = new_score
            stall_count = 0
        else:
            stall_count += 1

        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        # LP kick when stalled
        if stall_count > 100 and stall_count % 150 == 0:
            M = np.max(c)
            g = lp_direction(f, M, relax=1.02)
            if g is not None and np.sum(g) > 0.01:
                for t_blend in [0.15, 0.25, 0.4, 0.6]:
                    candidate = (1.0 - t_blend) * f + t_blend * g
                    candidate = np.clip(candidate, 0, None)
                    cs = fast_score(candidate)
                    if cs < cur_score:
                        f = candidate
                        c = fft_convolve(f, f)
                        S = np.sum(f)
                        cur_score = cs
                        stall_count = 0
                        if cur_score < best_score:
                            best_score = cur_score
                            best_f = f.copy()
                        break

        # Perturbation for escape
        if stall_count > 400 and stall_count % 200 == 0:
            noise_strength = min(0.1, 0.02 + 0.01 * (stall_count // 200))
            candidate = f * np.exp(rng.normal(0, noise_strength, n))
            candidate = np.clip(candidate, 0, None)
            cs = fast_score(candidate)
            if cs < best_score * 1.02:
                f = candidate
                c = fft_convolve(f, f)
                S = np.sum(f)
                cur_score = cs
                stall_count = 0

        # Aggressive restart if very stalled
        if stall_count > 1500 and stall_count % 500 == 0:
            # Try to reduce peak contributions
            c_peak = np.max(c)
            peak_idx = int(np.argmax(c))

            contribs = np.zeros(n)
            for j in range(n):

                if 0 <= k < n:

            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            # Reduce peaks

            candidate[top_idx] *= 0.7
            cs = fast_score(candidate)
            if cs < cur_score:
                f = candidate
                c = fft_convolve(f, f)

                    k = peak_idx - j
                    if 0 <= k < n:
                        contribs[j] = current[j] * current[k]
                top_k = max(5, n // 6)
                top_idx = np.argsort(contribs)[-top_k:]
                for idx in top_idx:
                    candidate[idx] *= np.exp(rng.normal(-0.02, 0.01))
                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 5:
                    n_inc = min(10, len(other) // 3)
                    inc_idx = rng.choice(other, size=n_inc, replace=False)
                    candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
            elif r < 0.55:
                # Block perturbation
                width = max(1, int(rng.integers(1, max(2, n // 5))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
            elif r < 0.70:
                # Pair redistribution
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    delta = rng.uniform(-0.1, 0.1) * current[i]
                    candidate[i] = max(0, current[i] - delta)
                    candidate[j] = max(0, current[j] + delta)
            elif r < 0.85:
                # Small global
                candidate = current * np.exp(rng.normal(0.0, 0.005, size=n))
            else:
                # Trilinear
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                k = int(rng.integers(0, n))
                if len(set([i, j, k])) == 3:
                    delta = rng.uniform(-0.08, 0.08) * current[i]
                    candidate[i] = max(0, current[i] - 2 * delta)
                    candidate[j] = max(0, candidate[j] + delta)
                    candidate[k] = max(0, candidate[k] + delta)

            candidate = np.clip(candidate, 0.0, 1000.0)
            new_score = fast_score(candidate)

            if new_score < cur_score:
                current = candidate
                cur_score = new_score
                c = fft_convolve(current, current)
                S = np.sum(current)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = current.copy()
            elif rng.random() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
                current = candidate
                cur_score = new_score
                c = fft_convolve(current, current)
                S = np.sum(current)

            temperature *= cooling
            polish_sweeps += 1

        print(f"  Polish done ({polish_sweeps} steps), best={best_score:.6f}")

    # Phase 5: Final CD pass if time remains
    if time.monotonic() < deadline - 15 and best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            print(f"\n=== Phase 5: Final CD ({remaining:.0f}s) ===")
            f_best, s_best = optimize_from_start(best_seq, remaining * 0.7, rng, verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = f_best.copy()
                print(f"  After final CD: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
