    return 2.0 * n * c.max() / (s * s)

    valid = (diff >= 0) & (diff < n)
    safe = np.where(valid, diff, 0)
    return np.where(valid, a[safe], 0.0)

def mult_grad_step(a, peak_idx, lr):
    """Vectorized multiplicative gradient step targeting peak lag."""
    n = len(a)
    # conv[peak_idx] = sum_i a[i] * a[peak_idx - i] (with bounds)
    # d/d(a[i]) = a[peak_idx - i] + a[i - peak_idx] (with bounds)
    
    # For the first term: a[peak_idx - i] for i where 0 <= peak_idx - i < n
    # i.e., peak_idx - n + 1 <= i <= peak_idx
    i_min = max(0, peak_idx - n + 1)
    i_max = min(peak_idx, n - 1)
    
    grad = np.zeros(n)
    # First term: a[peak_idx - i] for i in [i_min, i_max]
    idx = peak_idx - np.arange(i_min, i_max + 1)
    grad[i_min:i_max+1] += a[idx]
    
    # Second term: a[i - peak_idx] for i where 0 <= i - peak_idx < n
    # i.e., peak_idx <= i <= peak_idx + n - 1, but also i < n
    i_min2 = max(peak_idx, 0)
    i_max2 = min(peak_idx + n - 1, n - 1)
    if i_min2 <= i_max2:
        idx2 = np.arange(i_min2, i_max2 + 1) - peak_idx
        grad[i_min2:i_max2+1] += a[idx2]
    
    # Multiplicative update
    max_grad = grad.max()
    if max_grad > 0:
        a_new = a * np.exp(-lr * grad / max_grad)
    else:
        a_new = a.copy()
    return np.maximum(a_new, 0)

"""
Hybrid LP + Multiplicative Gradient optimization for minimizing peak autocorrelation
of non-negative step functions. Combines LP direction finding with fast multiplicative
gradient descent, peak tracking, and adaptive annealing for escaping local optima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize_seq(a, target=None):
    """Normalize sequence to have given sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.full(len(a), target / len(a))
    return a * (target / s)


def build_lp_matrix(a):
    """Build LP constraint matrix A where A@G = conv(a, G). Vectorized."""
    n = len(a)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    diff = k - j
    valid = (diff >= 0) & (diff < n)
    safe = np.where(valid, diff, 0)
    return np.where(valid, a[safe], 0.0)


def solve_lp(a, M, warm_start=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    if warm_start is not None:
        from scipy.optimize import linprog as lp
        res = lp(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                 bounds=bounds, method='highs', x0=warm_start,
                 options={'presolve': True})
    else:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    if res.status == 0:
        return res.x
    return np.zeros(n)


def mult_grad_subgrad(a, peak_idx):
    """Compute subgradient of conv[peak_idx] w.r.t. a. Vectorized O(n)."""
    n = len(a)
    grad = np.zeros(n)
    # First term: a[peak_idx - i] for valid i
    i_min = max(0, peak_idx - n + 1)
    i_max = min(peak_idx, n - 1)
    if i_min <= i_max:
        idx = peak_idx - np.arange(i_min, i_max + 1)
        grad[i_min:i_max + 1] += a[idx]
    # Second term: a[i - peak_idx] for valid i
    i_min2 = max(peak_idx, 0)
    i_max2 = min(peak_idx + n - 1, n - 1)
    if i_min2 <= i_max2:
        idx2 = np.arange(i_min2, i_max2 + 1) - peak_idx
        grad[i_min2:i_max2 + 1] += a[idx2]
    return grad


def mult_grad_step(a, peak_idx, lr):
    """One multiplicative gradient step."""
    grad = mult_grad_subgrad(a, peak_idx)
    max_g = grad.max()
    if max_g > 1e-15:
        a_new = a * np.exp(-lr * grad / max_g)
    else:
        a_new = a.copy()
    return np.maximum(a_new, 0.0)


def mult_grad_step_multi_peak(a, peak_indices, weights, lr):
    """Multiplicative gradient step targeting multiple peaks."""
    n = len(a)
    grad_sum = np.zeros(n)
    for idx, w in zip(peak_indices, weights):
        grad_sum += w * mult_grad_subgrad(a, idx)
    max_g = grad_sum.max()
    if max_g > 1e-15:
        a_new = a * np.exp(-lr * grad_sum / max_g)
    else:
        a_new = a.copy()
    return np.maximum(a_new, 0.0)

                best_t = m1
        else:
            a0 = m1
            if v2 < best_val:

        g_norm = normalize_seq(g, S)
        best_t, best_val = ternary_blend(a, g_norm, S, iters=10)
        if best_val < M * (1.0 - 1e-10) and best_t > 1e-6:
            a = (1 - best_t) * a + best_t * g_norm
            a = np.maximum(a, 0.0)
            a = normalize_seq(a, S)
            new_max = fft_convolve(a, a).max()
            if new_max < best_max - 1e-13:
                best_max = new_max
                best_a = a.copy()
        else:
            if i > 5:
                break

    return best_a


def interpolate_to_length(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def make_structured_init(n, kind, rng):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if kind == 'chirp':
        alpha = 0.3 + 0.4 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.7))
    elif kind == 'sine':
        f = 1 + int(rng.integers(0, 5))
        amp = 0.15 + 0.35 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif kind == 'multi_sine':
        a = np.ones(n)
        for _ in range(3 + int(rng.integers(0, 3))):
            f = int(rng.integers(1, n // 3))
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif kind == 'bimodal':
        center = n // 2
        sigma = n * (0.1 + 0.15 * rng.random())
        a = 0.5 + 0.8 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif kind == 'ramp_mod':
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 4) * t / n)
        a *= (1.0 + 0.2 * t / n)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing peak autocorrelation.
    Phase 1: Quick LP on provided sequence
    Phase 2: Deep optimization on provided length
    Phase 3: Multi-length exploration
    Phase 4: Final deep optimization on best
    Phase 5: Polish
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def try_update(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        try_update(provided_seq)

    if provided_seq is None:
        provided_seq = np.ones(640)
        print("No provided sequence, using default n=640")

    n_prov = len(provided_seq)

    # Phase 1: Quick LP refinement on provided sequence
    print("\n=== Phase 1: Quick LP on provided ===")
    t1_start = time.monotonic()
    t1_limit = min(60.0, (deadline - t1_start) * 0.1)
    result = quick_lp_refine(provided_seq, t1_limit, n_lp=30)
    improved = try_update(result)
    print(f"  Phase 1 done: score={best_score:.6f} {'(improved)' if improved else '(no improvement)'}")

    # Phase 2: Deep optimization on provided length
    print("\n=== Phase 2: Deep optimization ===")
    t2_start = time.monotonic()
    remaining = deadline - t2_start
    t2_limit = min(500.0, remaining * 0.35)
    print(f"  n={n_prov}, time_limit={t2_limit:.0f}s")
    result = deep_optimize(provided_seq, t2_limit, verbose=True, tag=f"n={n_prov}")
    improved = try_update(result)
    print(f"  Phase 2 done: score={best_score:.6f} {'(improved)' if improved else '(no improvement)'}")

    # Phase 3: Multi-length exploration
    print("\n=== Phase 3: Multi-length exploration ===")
    t3_start = time.monotonic()
    remaining = deadline - t3_start
    t3_limit = min(300.0, remaining * 0.2)
    per_length = max(30.0, t3_limit / 5)

    other_lengths = [n_prov - 64, n_prov + 64, 512, 768, 896]
    # Remove duplicates and the provided length
    other_lengths = [x for x in sorted(set(other_lengths)) if x != n_prov and x >= 128 and x <= 1200]

    for n_try in other_lengths[:4]:
        if time.monotonic() > deadline - 200:
            break
        t_per = min(per_length, deadline - time.monotonic() - 150)
        if t_per < 10:
            break
        a0 = interpolate_to_length(best_seq if best_seq is not None else provided_seq, n_try)
        a0 = np.clip(a0, 0.01, None)
        print(f"  Trying n={n_try}...")
        # Quick LP first
        r1 = quick_lp_refine(a0, min(t_per * 0.4, 20.0), n_lp=15)
        # Then brief deep optimization
        t_deep = min(t_per * 0.6, deadline - time.monotonic() - 150)
        if t_deep > 10:
            r2 = deep_optimize(r1, t_deep, verbose=False, tag=f"n={n_try}")
        else:
            r2 = r1
        if try_update(r2):
            print(f"  *** n={n_try}: NEW BEST score={best_score:.6f} ***")
        else:
            print(f"  n={n_try}: score={local_score(r2):.6f} (not better)")

    print(f"  Phase 3 done: best score={best_score:.6f}")

    # Phase 4: Final deep optimization on the best found
    print("\n=== Phase 4: Final deep optimization ===")
    t4_start = time.monotonic()
    remaining = deadline - t4_start - 30
    if remaining > 50 and best_seq is not None:
        n_best = len(best_seq)
        t4_limit = min(remaining, 600.0)
        print(f"  n={n_best}, time_limit={t4_limit:.0f}s")
        result = deep_optimize(best_seq, t4_limit, verbose=True, tag="final")
        try_update(result)
        print(f"  Phase 4 done: score={best_score:.6f}")

    # Phase 5: Polish - try small perturbations around best
    print("\n=== Phase 5: Final polish ===")
    t5_start = time.monotonic()
    remaining = deadline - t5_start - 5
    if remaining > 20 and best_seq is not None:
        n_best = len(best_seq)
        S = np.sqrt(2.0 * n_best)
        a = normalize_seq(best_seq.copy(), S)

        # Try several small random perturbations with quick LP
        for attempt in range(min(20, int(remaining / 5))):
            if time.monotonic() > deadline - 5:
                break
            # Random perturbation
            a_pert = a * np.exp(0.02 * rng.standard_normal(n_best))
            a_pert = np.maximum(a_pert, 0)
            a_pert = normalize_seq(a_pert, S)
            # Quick LP refinement
            r = quick_lp_refine(a_pert, 3.0, n_lp=10)
            try_update(r)

        # Also try from structured inits of same length
        for kind in ['chirp', 'multi_sine', 'bimodal']:
            if time.monotonic() > deadline - 10:
                break
            a_struct = make_structured_init(n_best, kind, rng)
            # Blend with best
            a_blend = 0.5 * normalize_seq(best_seq, S) + 0.5 * normalize_seq(a_struct, S)
            a_blend = np.maximum(a_blend, 0)
            a_blend = normalize_seq(a_blend, S)
            t_left = min(20.0, deadline - time.monotonic() - 5)
            if t_left > 5:
                r = deep_optimize(a_blend, t_left, verbose=False, tag="blend")
                try_update(r)

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local_score={best_score:.6f} ===")

        # Cross-check
        if abs(final_score - best_score) > 0.01:
            print(f"  WARNING: score mismatch! eval={final_score:.6f} vs local={best_score:.6f}")

        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
