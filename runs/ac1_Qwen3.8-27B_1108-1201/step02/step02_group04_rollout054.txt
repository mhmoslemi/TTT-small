def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max(), conv

    f = f0.copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    patience = 0

    total_time = time_limit

        progress = (time.monotonic() - t0) / total_time
        beta = beta_start + (beta_end - beta_start) * progress

        C = conv  # already computed
        C_shifted = C - C.max()
        w = np.exp(beta * C_shifted)
        w_sum = w.sum()

            w[int(np.argmax(C))] = 1.0
        else:
            w /= w_sum

        # This is 2 * cross_correlation(w, f)
        # cross_correlation(w, f)[m] = sum_k w[k] * f[k-m]
        # = convolve(w, f_flipped)[m + n - 1]...

        # grad[m] = 2 * sum_{k:m}^{m+n-1} w[k] * f[k-m]
        # This is the cross-correlation of w and f at lag m.
        # Using FFT: cross_correlation = ifft(fft(w) * conj(fft(f)))
        # But w has length 2n-1 and f has length n, so we need to be careful.

        # cross_correlation(w, f_pad)[m] = sum_k w[k] * f_pad[k-m]
        # = sum_k w[k] * f[k-m] (since f_pad[k-m] = f[k-m] only when 0 <= k-m < n)
        
        # So: grad[m] = 2 * cross_correlation(w, f_pad)[m]

        update = -eta * grad

        if s < 1e-30:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated mirror descent with
log-sum-exp smoothing, multi-scale optimization, LP fallback, and mutation-based
exploration. The key innovation is using the log-sum-exp gradient (a smooth
approximation of the max subgradient) computed efficiently via FFT, enabling
very fast iterations at large sequence lengths where LP-based methods become
impractical.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_gradient(f, w):
    """Compute the smoothed gradient grad[m] = 2 * sum_j w[m+j] * f[j].
    Uses FFT for O(n log n) computation.
    grad[m] = 2 * conv(w_rev, f)[2n-2-m] where w_rev = w[::-1].
    """
    n = len(f)
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W_rev = np.fft.rfft(w[::-1], fft_size)
    F = np.fft.rfft(f, fft_size)
    h = np.fft.irfft(W_rev * F, fft_size)
    # grad[m] = 2 * h[2n-2-m] for m=0..n-1
    # i.e., grad = 2 * h[n-1 : 2n-1][::-1]
    grad = 2.0 * h[n - 1:2 * n - 1][::-1]
    return grad


def smoothed_mirror_descent(f0, time_limit, beta_start=5.0, beta_end=500.0,
                            eta_start=0.4, momentum_coeff=0.7):
    """Mirror descent with log-sum-exp smoothing and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-15:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    eta = eta_start
    t0 = time.monotonic()
    patience = 0
    mom = np.zeros(n)
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start + (beta_end - beta_start) * progress

        # Compute softmax weights over convolution values
        C = conv
        C_shifted = C - C.max()
        # Clip for numerical stability
        C_shifted = np.clip(C_shifted, -100.0 / max(beta, 1.0), 0.0)
        w = np.exp(beta * C_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(C))
            w[int(np.argmax(C))] = 1.0
        else:
            w /= w_sum

        # Compute smoothed gradient
        grad = compute_gradient(f, w)
        # Center to preserve sum
        grad -= grad.mean()

        # Momentum
        mom = momentum_coeff * mom + (1.0 - momentum_coeff) * grad
        mom -= mom.mean()

        # Multiplicative (mirror) update
        update = -eta * mom
        update -= update.max()
        f_new = f * np.exp(update)
        f_new = np.maximum(f_new, 1e-30)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f
        else:
            f_new /= s

        conv = fast_convolve(f_new)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            patience = 0
            # Slightly increase step size on improvement
            eta = min(eta * 1.02, eta_start)
        else:
            patience += 1
            if patience % 30 == 0:
                eta *= 0.85
                if eta < 0.003:
                    break

        if patience > 300:
            break

    return best_f, best_obj


def smooth_with_adaptive_beta(f0, time_limit):
    """Run mirror descent with multiple beta passes (coarse to fine)."""
    f = f0.copy()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    # Multiple passes with increasing beta
    beta_schedule = [3, 10, 30, 100, 300, 1000]
    num_passes = len(beta_schedule)
    time_per_pass = time_limit / num_passes

    for i, beta in enumerate(beta_schedule):
        if time.monotonic() - t0 > time_limit * 0.9:
            break
        remaining = time_limit - (time.monotonic() - t0)
        t_pass = min(time_per_pass, remaining)
        if t_pass < 2:
            break

        # For lower beta, use larger eta; for higher beta, smaller eta
        eta = 0.5 / (1.0 + 0.3 * i)
        b_start = beta
        b_end = min(beta * 3, 2000.0)

        f_new, obj_new = smoothed_mirror_descent(
            best_f, t_pass, beta_start=b_start, beta_end=b_end, eta_start=eta)

        sc = 2.0 * n * fast_convolve(f_new).max()
        if sc < best_obj - 1e-12:
            best_obj = sc
            best_f = f_new.copy()

    return best_f, best_obj


def make_lp_structure(n):
    """Precompute row/col indices for the cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_cross(f, M, rows, cols, n, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, max_iters=200):
    """LP-based alternating optimization with backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-15:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_lp_structure(n)

    t0 = time.monotonic()
    t = 0.8
    patience = 0

    while time.monotonic() - t0 < time_limit and patience < 40:
        g = solve_lp_cross(f, M, rows, cols, n, timeout=min(8, max(2, time_limit / 10)))
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [t, 0.6 * t, 0.4 * t, 0.2 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.85

    return best_f, best_obj


def gen_inits(n, rng, parent=None):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    for sf in [0.1, 0.2, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{int(sf*100)}'] = g / g.sum()

    # Bi-modal
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    # Dirichlet (spiky)
    inits['dir_spiky'] = rng.dirichlet(np.ones(n) * 0.1)

    # Dirichlet (moderate)
    inits['dir_mod'] = rng.dirichlet(np.ones(n) * 0.5)

    # Random
    r = rng.random(n)
    inits['rand'] = r / r.sum()

    # Triangle
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.01)
    inits['tri'] = tri / tri.sum()

    if parent is not None:
        if parent.size >= n:
            p = np.maximum(parent[:n].copy(), 0.0)
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            p = np.maximum(p, 0.0)
        if p.sum() > 1e-10:
            inits['parent'] = p / p.sum()

    return inits


def upsample(f, factor=2):
    """Upsample a sequence by the given factor using linear interpolation."""
    n = len(f)
    n_new = n * factor
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, f)
