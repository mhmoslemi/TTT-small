import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute 2n * max(conv) / sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (S * S)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_step(a, top_k=50, temperature=0.001, lr=0.02, momentum_buf=None):
    """One subgradient step with momentum."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()

    # Get top-k peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    # Softmax weights
    scaled = peak_vals / max(temperature, 1e-15)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    # Subgradient of max(conv) w.r.t. a
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx_arr = np.arange(j_start, j_end + 1)
        sg[idx_arr] += w[idx] * 2.0 * a[k - idx_arr]

    # Full subgradient of the objective 2n*max(conv)/sum^2 (with sum=1):
    # d/d(a_i) [2n * M] = 2n * sg[i]  (since sum=1, the /sum^2 term's derivative
    # gives -4n*M*sg[i]... actually let's just use the gradient of 2n*max(conv)
    # subject to sum=1, which is the projection of 2n*sg onto the tangent space)
    grad = 2.0 * n * sg - 2.0 * n * sg.mean()  # project onto sum=0 hyperplane

    if momentum_buf is None:
        momentum_buf = np.zeros(n)

    v = 0.9 * momentum_buf - lr * grad
    a_new = project_simplex(a + v)

    return a_new, v, np.sqrt((conv**2).sum())


def optimize_hybrid(a0, time_limit, seed=42):
    """
    Hybrid optimizer: alternates between coordinate descent sweeps and
    subgradient steps, with periodic perturbation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n)
    else:
        a = a / S
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    momentum_buf = np.zeros(n)
    no_improve_count = 0
    cd_sweeps = 10
    sg_steps = 20
    lr = 0.03
    temp = 0.0005

    while time.monotonic() - t0 < time_limit:
        # Phase A: Coordinate descent
        for _ in range(cd_sweeps):
            if time.monotonic() - t0 > time_limit:
                break
            a, obj = coordinate_optimize(a, max_sweeps=1, random_order=True,
                                         seed=int(rng.integers(2**31)))
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1

        # Phase B: Subgradient steps
        for _ in range(sg_steps):
            if time.monotonic() - t0 > time_limit:
                break
            a, momentum_buf, _ = subgradient_step(a, top_k=max(20, n // 10),
                                                   temperature=temp, lr=lr,
                                                   momentum_buf=momentum_buf)
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1

        # Adaptive schedule
        if no_improve_count > 50:
            # Perturb and restart
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            momentum_buf *= 0.3
            lr = max(0.005, lr * 0.7)
            temp = max(0.0001, temp * 0.5)
            no_improve_count = 0
        elif no_improve_count > 20:
            temp *= 1.1
            lr *= 0.9

    return best_a, best_obj

    ramp /= ramp.sum()
    inits['ramp'] = ramp
    inits['ramp_rev'] = ramp[::-1].copy()

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    # Gaussians
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.1f}'] = g

    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.2)
    inits['random2'] = rng.dirichlet(np.ones(n) * 5.0)

    # From parent
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            pp = pp / ps
            if len(pp) == n:
                inits['parent'] = pp
            elif len(pp) > n:
                # Downsample
                pp_ds = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(pp)), pp)
                pp_ds = np.maximum(pp_ds, 0)
                pp_ds /= pp_ds.sum()
                inits['parent_ds'] = pp_ds
            else:
                # Upsample
                pp_up = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(pp)), pp)
                pp_up = np.maximum(pp_up, 0)
                pp_up /= pp_up.sum()
                inits['parent_up'] = pp_up

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search using exact coordinate descent and
    hybrid subgradient+CD optimization to minimize 2n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: length={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Phase 1: Quick scan with coordinate descent at various lengths
    print("=== Phase 1: Coordinate Descent Scan ===")
    lengths = [128, 256, 512]
    scan_results = []
    phase1_deadline = time.monotonic() + 400.0

    for n in lengths:
        if time.monotonic() > phase1_deadline - 60:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        # Priority order
        priority = ['parent', 'parent_ds', 'parent_up', 'uniform', 'random',
                    'gauss_0.3', 'triangle', 'bimodal']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:5]

        for name in init_names:
            if time.monotonic() > phase1_deadline - 30:
                break
            t_remaining = phase1_deadline - time.monotonic() - 20
            if t_remaining < 10:
                break
            t_alloc = min(t_remaining, 80.0)

            a0 = inits[name]
            a_opt, obj_opt = coordinate_optimize(a0, max_sweeps=80, random_order=True,
                                                  seed=int(rng.integers(2**31)))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  CD: n={n}, {name}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Hybrid optimization on top results
    print("=== Phase 2: Hybrid Optimization ===")
    top_configs = []
    seen_set = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen_set:
            top_configs.append((n, name, a_opt))
            seen_set.add(key)
        if len(top_configs) >= 6:
            break

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = deadline - time.monotonic() - 120.0
    time_per = max(20.0, remaining / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 80:
            break
        t_alloc = min(time_per, deadline - time.monotonic() - 60)
        if t_alloc < 10:
            break

        a_opt, obj_opt = optimize_hybrid(a0, time_limit=t_alloc,
                                          seed=int(rng.integers(2**31)))
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Hybrid: n={n}, {name}, score={score:.6f}")

    # Phase 3: Final polishing with pure coordinate descent (many sweeps)
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # Multiple rounds of coordinate descent with different random seeds
        for round_num in range(5):
            if remaining < 10:
                break
            t_round = remaining / (5 - round_num)
            a_opt, obj_opt = coordinate_optimize(best_sequence, max_sweeps=200,
                                                  random_order=True,
                                                  seed=int(rng.integers(2**31)))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish {round_num}: score={score:.6f}")
            remaining = deadline - time.monotonic() - 20.0

        # Perturbation + coordinate descent
        for _ in range(4):
            if remaining < 10:
                break
            n_best = len(best_sequence)
            noise_scale = 0.005 / (1 + _ * 0.5)
            noise = rng.standard_normal(n_best) * noise_scale
            a_pert = project_simplex(best_sequence + noise)
            t_per = min(remaining / 3, 60.0)
            a_opt, obj_opt = coordinate_optimize(a_pert, max_sweeps=100,
                                                  random_order=True,
                                                  seed=int(rng.integers(2**31)))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Perturb+CD: score={score:.6f}")
            remaining = deadline - time.monotonic() - 20.0

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
