def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        # b_i appears in constraints k = i, i+1, ..., i+n-1
        for k in range(i, i+n):
            A_ub[k, i] = a[k-i]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: b_i >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    b = result.x
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum

def optimize_smoothed(a0, time_limit, beta_init=50.0, beta_end=200.0, lr_init=0.02,
                       momentum=0.9, perturb_every=500, perturb_scale=0.003):
    """
    Gradient descent with Nesterov momentum on the smoothed objective.
    a0 should be on the simplex (sum=1, non-negative).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)  # momentum
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    total_iters = 0
    
    # We'll anneal beta from beta_init to beta_end
    # Use a schedule based on fraction of time elapsed
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        beta = beta_init + (beta_end - beta_init) * frac
        
        # Nesterov step
        y = a + momentum * v
        conv_y = fast_convolve(y)
        grad = smoothed_gradient(y, conv_y, beta)
        
        # Project gradient descent
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        a = a_new
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic perturbation
        if no_improve >= perturb_every:
            rng = np.random.default_rng()
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)
            lr = max(0.001, lr * 0.7)
        
        # Diminishing learning rate
        lr = max(0.0005, lr_init * (1 - frac * 0.8))
    
    return best_a, best_obj

def optimize_multistage(a0, time_limit):
    """
    Multi-stage optimization combining smoothed gradient, LP direction,
    and coordinate descent.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    # Stage 1: Smoothed gradient descent (60% of time)
    t1 = time_limit * 0.5
    a, obj1 = optimize_smoothed(a, t1, beta_init=30, beta_end=150,
                                  lr_init=0.015, momentum=0.92)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a.copy()
    
    # Stage 2: LP-based refinement (25% of time)
    remaining = time_limit - (time.monotonic() - t0)
    t2 = remaining * 0.4
    t_lp = t0 + time_limit - t2  # start time for LP phase
    # Actually let me just use time directly
    t_lp_start = time.monotonic()
    t_lp_budget = time_limit * 0.3
    best_a_lp = best_a.copy()
    
    while time.monotonic() - t_lp_start < t_lp_budget:
        conv_cur = fast_convolve(best_a_lp)
        M = conv_cur.max()
        
        # Solve LP to find direction
        b = lp_direction(best_a_lp, conv_cur, M, n)
        if b is None:
            break
        
        # Try different step sizes
        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_try = (1 - t) * best_a_lp + t * b
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()
            if obj_try < best_obj - 1e-8:
                best_obj = obj_try
                best_a = a_try.copy()
                best_a_lp = a_try.copy()
                improved = True
                break
            elif obj_try < M:
                # Accept even if not better than global best
                best_a_lp = a_try.copy()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
        
        if not improved:
            # Try smaller steps
            for t in [0.05, 0.1, 0.15]:
                a_try = (1 - t) * best_a_lp + t * b
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = conv_try.max()
                if obj_try < M - 1e-10:
                    best_a_lp = a_try.copy()
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                    break
    
    # Stage 3: Fine-tuned gradient descent (remaining time)
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 5:
        a, obj3 = optimize_smoothed(best_a, remaining, beta_init=100, beta_end=500,
                                      lr_init=0.005, momentum=0.95,
                                      perturb_every=300, perturb_scale=0.001)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a.copy()
    
    return best_a, best_obj

def coordinate_descent_polish(a0, time_limit):
    """
    Coordinate descent: try adjusting pairs of coordinates to reduce max convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        # Pick a random pair (i, j) and try to transfer mass
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try transferring small amount from i to j (or vice versa)
        for direction in [1, -1]:
            for eps in [0.001, 0.005, 0.01, 0.02]:
                a_try = a.copy()
                if direction == 1:
                    # Move mass from i to j
                    amt = min(a[i], eps)
                    if amt < 1e-10:
                        continue
                    a_try[i] -= amt
                    a_try[j] += amt
                else:
                    amt = min(a[j], eps)
                    if amt < 1e-10:
                        continue
                    a_try[j] -= amt
                    a_try[i] += amt
                
                conv_try = fast_convolve(a_try)
                obj_try = conv_try.max()
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break
        
        iters += 1
        if iters % 1000 == 0:
            print(f"  CD iter {iters}: obj={best_obj:.6f}")
    
    return best_a, best_obj

def generate_candidates(n, parent, rng):
    """Generate diverse initializations."""
    candidates = {}
    x = np.arange(n, dtype=np.float64)
    
    candidates['uniform'] = np.ones(n) / n
    
    # Ramp
    ramp = np.arange(1, n+1, dtype=np.float64)
    ramp /= ramp.sum()
    candidates['ramp'] = ramp
    
    # Triangle
    tri = np.minimum(x + 0.5, n - 0.5 - x)
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    candidates['triangle'] = tri
    
    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
    g /= g.sum()
    candidates['gauss_center'] = g
    
    # Gaussian (off-center)
    g2 = np.exp(-0.5 * ((x - n/3) / (n/5))**2)
    g2 /= g2.sum()
    candidates['gauss_off'] = g2
    
    # Bimodal
    bi = np.zeros(n)
    bi[:n//4] = 1.0
    bi[3*n//4:] = 1.0
    bi /= bi.sum()
    candidates['bimodal'] = bi
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2*np.pi*x/n))
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    candidates['cosine'] = cos_w
    
    # Random
    candidates['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    candidates['random2'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                candidates['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                candidates['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                candidates['parent_ds'] = pp / ps
    
    return candidates

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed gradient descent
with Nesterov momentum, LP-based direction finding (from SOTA), peak-shaving
coordinate descent, and multi-phase diverse search across lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, conv, beta):
    """
    Gradient of logsumexp(beta*conv(a,a))/beta w.r.t. a.
    Uses FFT-based cross-correlation for efficiency.
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    w = conv * p  # w_k = c_k * softmax_k

    # grad_i = 2 * sum_{m=0}^{n-1} a[m] * w[m+i]
    # This is cross-correlation (a, w) at lag i
    # Need FFT size >= n + (2n-1) - 1 = 3n - 2
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)

    return 2.0 * cross[:n]


def lp_direction(a, M, n):
    """
    Solve LP: max sum(b_j) s.t. b_j >= 0, (a*b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)

    # Build constraint matrix: A_ub[k, i] = a[k-i] if 0 <= k-i < n
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None

    return b / b_sum


def optimize_smoothed(a0, time_limit, beta_init=30.0, beta_end=200.0,
                      lr_init=0.015, momentum=0.92, perturb_every=400,
                      perturb_scale=0.003, seed=0):
    """
    Nesterov accelerated projected gradient descent on smoothed objective.
    Input: a0 on simplex (sum=1, non-negative).
    Returns (best_a, best_max_conv).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac

        # Nesterov lookahead
        y = a + momentum * v
        conv_y = fast_convolve(y)
        grad = smoothed_gradient(y, conv_y, beta)

        # Momentum update
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic perturbation to escape local minima
        if no_improve >= perturb_every:
            noise_scale = perturb_scale * (1.0 + 0.5 * np.sin(total_iters * 0.01))
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.25)
            lr = max(0.0005, lr * 0.65)

        # Adaptive learning rate
        lr = max(0.0003, lr_init * (1.0 - 0.7 * frac))

    return best_a, best_obj


def optimize_lp_phase(a0, time_limit, seed=42):
    """
    LP-based direction finding phase.
    Repeatedly solve LP to find directions that reduce max convolution.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    current_a = a.copy()

    t0 = time.monotonic()
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(current_a)
        M = conv_cur.max()

        # Try slightly relaxed M to ensure feasibility
        for M_factor in [1.0, 1.001, 1.005, 1.01]:
            M_try = M * M_factor
            b = lp_direction(current_a, M_try, n)
            if b is not None and b.sum() > 1e-10:
                break
        else:
            b = None

        if b is None:
            # Try perturbing and solving
            noise = rng.standard_normal(n) * 0.01
            current_a = project_simplex(current_a + noise)
            stuck_count += 1
            if stuck_count > 10:
                break
            continue

        stuck_count = 0

        # Try different step sizes
        improved = False
        steps = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
        for t in steps:
            a_try = (1 - t) * current_a + t * b
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()

            if obj_try < best_obj - 1e-10:
                best_obj = obj_try
                best_a = a_try.copy()
                current_a = a_try.copy()
                improved = True
                break
            elif obj_try < M * 0.999:
                current_a = a_try.copy()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                break

        if not improved:
            # Smaller steps
            for t in [0.02, 0.03, 0.05]:
                a_try = (1 - t) * current_a + t * b
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = conv_try.max()
                if obj_try < M * 0.9999:
                    current_a = a_try.copy()
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                    break

    return best_a, best_obj


def peak_shaving(a0, time_limit, seed=0):
    """
    Targeted peak reduction: identify convolution peaks and adjust
    specific element pairs to reduce them.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        max_val = conv_cur.max()

        # Find top peaks
        top_k = min(5, len(conv_cur))
        peak_indices = np.argpartition(conv_cur, -top_k)[-top_k:]
        peak_indices = peak_indices[conv_cur[peak_indices] > max_val * 0.95]

        if len(peak_indices) == 0:
            break

        # For each peak, try to reduce it by moving mass
        improved = False
        for k in peak_indices:
            k = int(k)
            # Elements contributing to conv[k]: pairs (i, k-i) for valid i
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)

            # Find the pair with maximum contribution
            contributions = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
            if len(contributions) == 0:
                continue

            max_contrib_idx = np.argmax(contributions)
            i_max = i_start + max_contrib_idx

            # Try reducing a[i_max] and adding to a random other position
            for _ in range(3):
                j = rng.integers(0, n)
                if j == i_max or abs(j - i_max) < 2:
                    continue

                # Try transferring small amount
                for frac in [0.05, 0.1, 0.2]:
                    amt = a[i_max] * frac
                    if amt < 1e-8:
                        break
                    a_try = a.copy()
                    a_try[i_max] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-10:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break

            if improved:
                break

        iters += 1

        # Also try random pair adjustments
        if not improved and iters % 500 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                for frac in [0.02, 0.05, 0.1]:
                    amt = a[i] * frac
                    if amt < 1e-8:
                        break
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-10:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()

    # Triangle (inverted - peak in middle)
    tri = 1.0 / (1.0 + 0.5 * ((x - n / 2) / (n / 4)) ** 2)
    tri /= tri.sum()
    inits['bell'] = tri

    # Gaussians at different widths
    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g

    # Bimodal (concentrated at edges)
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Raised cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w

    # Off-center Gaussian
    g_off = np.exp(-0.5 * ((x - 0.3 * n) / (0.2 * n)) ** 2)
    g_off /= g_off.sum()
    inits['gauss_off'] = g_off

    # Random
    inits['rand_dirichlet'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand_dirichlet2'] = rng.dirichlet(np.ones(n) * 2.0)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits

    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 1.5)))

    remaining = deadline - time.monotonic() - 80.0
    n_configs = len(top_configs)
    per_config = max(15.0, remaining / n_configs)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 50:
            break
        t_alloc = min(per_config, 200.0)
        if t_alloc < 10:
            break

        print(f"  Config {i}: n={n}", flush=True)

        # Sub-phase A: Smoothed gradient (50% of allocation)
        t_a = t_alloc * 0.45
        a_a, obj_a = optimize_smoothed(
            a0, t_a, beta_init=30, beta_end=200,
            lr_init=0.01, momentum=0.92,
            perturb_every=300, perturb_scale=0.004,
            seed=rng.integers(0, 100000))

        score_a = float(evaluate_sequence(a_a.tolist()))
        if np.isfinite(score_a) and score_a < best_score:
            best_score = score_a
            best_sequence = a_a.copy()
            print(f"    Grad: {score_a:.6f}", flush=True)

        # Sub-phase B: LP direction (30% of allocation)
        t_b = t_alloc * 0.30
        if t_b > 5:
            a_b, obj_b = optimize_lp_phase(a_a, t_b, seed=rng.integers(0, 100000))
            score_b = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"    LP: {score_b:.6f}", flush=True)
            a_a = a_b  # continue from LP result

        # Sub-phase C: Refined gradient (25% of allocation)
        t_c = t_alloc * 0.25
        if t_c > 5:
            a_c, obj_c = optimize_smoothed(
                a_a, t_c, beta_init=100, beta_end=500,
                lr_init=0.004, momentum=0.95,
                perturb_every=200, perturb_scale=0.002,
                seed=rng.integers(0, 100000))
            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"    Refine: {score_c:.6f}", flush=True)

    # Phase 3: Polish with peak shaving and fine gradient
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0

        # Peak shaving
        t_ps = remaining * 0.35
        if t_ps > 10:
            a_ps, obj_ps = peak_shaving(best_sequence, t_ps, seed=777)
            score_ps = float(evaluate_sequence(a_ps.tolist()))
            if np.isfinite(score_ps) and score_ps < best_score:
                best_score = score_ps
                best_sequence = a_ps.copy()
                print(f"  PeakShave: {best_score:.6f}", flush=True)

        # Fine gradient with high beta
        remaining = deadline - time.monotonic() - 10.0
        t_fg = remaining * 0.35
        if t_fg > 5:
            a_fg, obj_fg = optimize_smoothed(
                best_sequence, t_fg, beta_init=200, beta_end=1000,
                lr_init=0.002, momentum=0.97,
                perturb_every=150, perturb_scale=0.001,
                seed=1234)
            score_fg = float(evaluate_sequence(a_fg.tolist()))
            if np.isfinite(score_fg) and score_fg < best_score:
                best_score = score_fg
                best_sequence = a_fg.copy()
                print(f"  FineGrad: {best_score:.6f}", flush=True)

        # Final peak shaving
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            a_fs, obj_fs = peak_shaving(best_sequence, remaining, seed=999)
            score_fs = float(evaluate_sequence(a_fs.tolist()))
            if np.isfinite(score_fs) and score_fs < best_score:
                best_score = score_fs
                best_sequence = a_fs.copy()
                print(f"  FinalShave: {best_score:.6f}", flush=True)

        # One more LP attempt if time remains
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 10:
            a_lp2, obj_lp2 = optimize_lp_phase(best_sequence, remaining * 0.7, seed=555)
            score_lp2 = float(evaluate_sequence(a_lp2.tolist()))
            if np.isfinite(score_lp2) and score_lp2 < best_score:
                best_score = score_lp2
                best_sequence = a_lp2.copy()
                print(f"  LP2: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
