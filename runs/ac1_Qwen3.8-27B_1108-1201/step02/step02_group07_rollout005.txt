import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def subgradient_descent(f_init, time_limit, K=5, lr_init=0.01, momentum=0.9):
    """Subgradient descent with top-K peaks and momentum."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f * (float(n) / S)  # normalize so sum = n
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    mom = np.zeros(n)
    lr = lr_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        c = fft_convolve(f, f)
        # Get top K peaks
        top_idx = np.argsort(c)[-K:]
        
        # Compute subgradient: average gradient of top K peaks
        g = np.zeros(n)
        for k in top_idx:
            ki = int(k)
            # dC(ki)/da[j] = 2*f[ki-j] if 0 <= ki-j < n
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                g[j] += 2.0 * f[ki - j]
        g /= float(K)
        
        # Center the gradient (to maintain sum approximately)
        g -= np.mean(g)
        
        # Momentum update
        mom = momentum * mom + (1.0 - momentum) * g
        f = f - lr * mom
        f = np.clip(f, 0.0, 1000.0)
        f = f * (float(n) / float(np.sum(f)))  # renormalize
        
        it += 1
        if it % 100 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
        if it % 1000 == 0:
            lr *= 0.95  # slowly decay learning rate
    
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score
