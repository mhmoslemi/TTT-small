"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based peak reduction with blocking sets, FFT-accelerated gradients,
and multi-length multi-start strategy.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    if target is None:
        target = math.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def autocorr(a):
    return np.convolve(a, a)

def build_lp_matrix(f, lags):
    """Build LP constraint matrix for specified lags only."""
    n = len(f)
    A = np.zeros((len(lags), n))
    for idx, m in enumerate(lags):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[idx, j] = f[m - j]
    return A

def lp_solve(f, M, lags=None):
    """Solve LP: max sum(g) s.t. rows of A * g <= M, g >= 0."""
    n = len(f)
    if lags is None:
        lags = list(range(2 * n - 1))
    
    c_obj = -np.ones(n)
    A = build_lp_matrix(f, lags)
    b = np.full(len(lags), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def get_blocking_set(c, M, threshold=0.7):
    """Get lags where autocorrelation is above threshold * M."""
    return list(np.where(c >= threshold * M)[0].tolist())

def soft_grad(a, tau):
    """Soft-max gradient using convolution."""
    n = len(a)
    c = autocorr(a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad

def hard_grad(a):
    n = len(a)
    c = autocorr(a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad

def grad_step(a, lr, momentum, decay, tau=None):
    n = len(a)
    S_t = math.sqrt(2.0 * n)
    a = normalize(a, S_t)
    if tau is None:
        g = hard_grad(a)
    else:
        g = soft_grad(a, tau)
    g -= g.mean()
    momentum[:] = decay * momentum - lr * g
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new, momentum

def lp_refine_iteration(f, M, blocking_frac=0.8, timeout=None):
    """One LP refinement iteration with blocking set."""
    n = len(f)
    c = autocorr(f)
    M_actual = float(c.max())
    
    # Get blocking set: lags where c >= blocking_frac * M_actual
    thresh = blocking_frac * M_actual
    lags = list(np.where(c >= thresh)[0].tolist())
    
    if len(lags) > 3 * n // 2:
        # Too many constraints, use top half
        lags = list(np.where(c >= 0.9 * M_actual)[0].tolist())
    if len(lags) > n:
        # Still too many, just use top n
        sorted_lags = sorted(range(2*n-1), key=lambda k: -c[k])
        lags = sorted_lags[:n]
    
    g = lp_solve(f, M_actual, lags)
    if g is None:
        return None, 0
    
    g_sum = float(g.sum())
    if g_sum < 1e-6:
        return None, 0
    
    return g, g_sum

def optimize_lp(a, deadline, max_iters=200):
    """Main LP-based optimization loop."""
    n = len(a)
    S_t = math.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    
    momentum = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        c = autocorr(a)
        M = float(c.max())
        
        # LP step
        g, g_sum = lp_refine_iteration(a, M)
        
        if g is not None and g_sum > S_t * (1 + 1e-6):
            # Normalize g
            g_norm = g * (S_t / g_sum)
            
            # Line search for mixing ratio
            improved = False
            for t_blend in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1]:
                a_new = (1.0 - t_blend) * a + t_blend * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-10:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break
            
            if not improved:
                # Try a small gradient step to escape
                a, momentum = grad_step(a, 0.1, momentum, 0.9, tau=5.0)
                no_improve += 1
        else:

            a, momentum = grad_step(a, 0.15, momentum, 0.9, tau=3.0)
            s = compute_obj(a)
            if s < best_score - 1e-10:
                best = a.copy()
                best_score = s
                no_improve = 0

                no_improve += 1

            a = np.maximum(a, 0)
            a = normalize(a, S_t)

            if it % 10 == 0:  # Check if we still made progress

                    a = best.copy()

def grad_descent_init(a, deadline, max_iters=5000):
    """Quick gradient descent initialization."""
    n = len(a)
    S_t = math.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    momentum = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        tau = 3.0 + 3.0 * (it / 1000.0)
        lr = 0.2 * (0.999 ** it)
        lr = max(lr, 0.01)
        a, momentum = grad_step(a, lr, momentum, 0.9, tau=tau)
        if it % 100 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-8:
                best = a.copy()
                best_score = s
    
    return best, best_score

def coordinate_polish(a, deadline, max_rounds=5):
    """Coordinate descent polishing."""
    n = len(a)
    S_t = math.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    patience = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            c = autocorr(a)
            k_star = int(np.argmax(c))
            j = k_star - i
            candidates = [0.0, orig * 0.5, orig * 1.5]
            if 0 <= j < n:
                candidates.append(orig * 0.7)
                candidates.append(orig * 1.1)
            
            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a_norm = normalize(a, S_t)
                s = compute_obj(a_norm)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
                    # Keep a_norm for next iteration
                    a = a_norm
                a[i] = orig  # restore for next candidate
            
            if best_val != orig:
                a[i] = best_val
                a = normalize(a, S_t)
                best_score = best_s
                improved = True
        
        if not improved:
            patience += 1
            if patience >= 2:
                break
        else:
            patience = 0
    
    return a, best_score

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive LP-based peak reduction with blocking sets, FFT-free fast convolutions,
soft-gradient escape, line-search mixing, and multi-length multi-start strategy.
Key improvements over previous: tighter blocking sets, adaptive thresholds,
small-step LP with line search, and aggressive time allocation to LP phase.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = math.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def autocorr(a):
    """Compute full autocorrelation (linear convolution)."""
    return np.convolve(a, a)


def build_lp_matrix(f, lags):
    """Build LP constraint matrix for specified lags."""
    n = len(f)
    A = np.zeros((len(lags), n))
    for idx, m in enumerate(lags):
        lo = max(0, m - (n - 1))
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[idx, j] = f[m - j]
    return A


def lp_solve(f, M, lags):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(f)
    c_obj = -np.ones(n)
    A = build_lp_matrix(f, lags)
    b = np.full(len(lags), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def get_blocking_set(c, M, alpha):
    """Get lags where autocorrelation >= alpha * M."""
    mask = c >= alpha * M
    return list(np.where(mask)[0].tolist())


def soft_grad(a, tau):
    """Compute soft-max gradient of max autocorrelation."""
    n = len(a)
    c = autocorr(a)
    c_max = c.max()
    c_shifted = c - c_max
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def hard_grad(a):
    """Compute hard-max gradient of peak autocorrelation."""
    n = len(a)
    c = autocorr(a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def grad_step(a, lr, momentum, decay, tau=None):
    """One gradient step with momentum, projecting onto sum and non-negativity."""
    n = len(a)
    S_t = math.sqrt(2.0 * n)
    a = normalize(a, S_t)
    if tau is None:
        g = hard_grad(a)
    else:
        g = soft_grad(a, tau)
    g -= g.mean()
    momentum[:] = decay * momentum - lr * g
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new, momentum

            print(f"  After final LP2: score={best_score:.6f}")

    if best_seq is not None:
        final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = compute_obj(np.array(final_seq))
        print(f"\n{'='*50}")
        print(f"FINAL RESULT: n={len(final_seq)}, score={final_score:.6f}")
        print(f"{'='*50}")
        return final_seq

    # Fallback: uniform sequence
    print("Falling back to uniform sequence of length 384.")
    return [1.0] * 384
