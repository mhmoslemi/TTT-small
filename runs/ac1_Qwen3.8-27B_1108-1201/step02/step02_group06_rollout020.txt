"""
Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Uses efficient FFT-based gradient computation
with annealed softmax smoothing, Adam optimizer, and multi-start strategy
across different sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_smoothed_gradient(a, tau):
    """
    Compute the gradient of log(sum(exp(c/tau))) with respect to a,
    where c = conv(a, a) is the linear autocorrelation.
    
    Returns:
        g: gradient vector of length n
        w: softmax weights of length 2n-1
        c: the autocorrelation
    """
    n = len(a)
    c = fft_convolve(a, a)  # length 2n-1
    
    # Compute softmax weights (numerically stable)
    c_max = np.max(c)
    c_shifted = (c - c_max) / tau
    exp_c = np.exp(c_shifted)
    w = exp_c / np.sum(exp_c)
    
    # Compute cross-correlation: cross[j] = sum_i a[i] * w[i+j] for j in [0, n-1]
    # Using FFT: cross = ifft(fft(w, N) * conj(fft(a, N)))[0:n]
    N = 1
    while N < 3 * n:
        N <<= 1
    W_ft = np.fft.rfft(w, N)
    A_ft = np.fft.rfft(a, N)
    cross_full = np.fft.irfft(W_ft * np.conj(A_ft), N)
    cross = cross_full[:n]
    
    # Gradient: g[j] = 2 * cross[j]
    g = 2.0 * cross
    
    return g, w, c


def adam_update(theta, grad, m, v, step, lr, beta1, beta2, eps):
    """Adam optimizer update step."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** step)
    v_hat = v / (1.0 - beta2 ** step)
    theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
    return theta, m, v


def optimize_gradient(a_init, n, time_budget, rng, tau_start=None, tau_end=None, lr_start=1.0):
    """
    Optimize using Adam gradient descent on the smoothed peak objective.
    
    Parameters:
        a_init: initial sequence (will be normalized)
        n: target length
        time_budget: seconds to spend
        rng: numpy random generator
        tau_start: initial temperature (default: 5/n)
        tau_end: final temperature (default: 0.05/n)
        lr_start: initial learning rate
    
    Returns:
        best_a: best sequence found
        best_score: best score
    """
    if tau_start is None:
        tau_start = 5.0 / n
    if tau_end is None:
        tau_end = 0.05 / n
    
    S = np.sqrt(2.0 * n)  # Fixed sum for normalization
    
    # Initialize theta from a_init
    a = np.clip(a_init, 1e-10, None)
    a = a * (S / np.sum(a))
    theta = np.log(a + 1e-30)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    stagnant = 0
    patience = 500
    
    while time.monotonic() - t0 < time_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_budget)
        
        # Anneal temperature (use a smooth schedule)
        # Use exponential decay for most of the time, then linear
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Learning rate with cosine annealing
        lr = lr_start * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr = max(lr, lr_start * 0.01)
        
        # Compute gradient
        g, w, c = compute_smoothed_gradient(a, tau)
        
        # Project gradient: grad_theta = a * (g - dot(g,a)/S)
        mean_g = np.dot(g, a) / S
        grad_theta = a * (g - mean_g)
        
        # Adam update
        step += 1
        theta, m, v = adam_update(theta, grad_theta, m, v, step, lr, beta1, beta2, eps)
        
        # Convert back to a
        theta_shifted = theta - np.max(theta)
        a_new = np.exp(theta_shifted)
        a_new = a_new / np.sum(a_new) * S
        a_new = np.clip(a_new, 1e-10, 1000.0)
        a_new = a_new * (S / np.sum(a_new))
        
        a = a_new
        
        # Track best
        score = local_score(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # If stuck, add perturbation
        if stagnant > patience:
            pert = rng.standard_normal(n) * 0.1
            theta = np.log(np.clip(a + pert, 1e-10, None))
            m = np.zeros(n)
            v = np.zeros(n)
            step = 0
            stagnant = 0
            patience = max(100, patience // 2)
    
    return best_a, best_score


def coordinate_descent(a_init, time_budget, rng, max_iter_per_coord=50):
    """
    Coordinate descent polish. For each coordinate, try to improve the score
    by adjusting that coordinate while keeping others fixed.
    """
    n = len(a_init)
    a = np.clip(a_init, 0.01, None).copy()
    S = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        improved = False
        # Process coordinates in random order
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 >= time_budget:
                break
            
            # Try to optimize a[j]
            # The contribution of a[j] to c[k] is: a[j]*a[k-j] for valid k-j
            # We want to minimize max(c) by adjusting a[j]
            
            # Current c
            c = fft_convolve(a, a)
            M = np.max(c)
            
            # The peak is at k* = argmax(c)
            # Changing a[j] to x changes:
            # c[k] by: (x - a[j]) * (a[k-j] + a[k-j]) if 0<=k-j<n
            # Wait, more precisely:
            # c[k] = sum_i a[i]*a[k-i]
            # If we change a[j] to x:
            # c[k] changes by: (x - a[j]) * (a[k-j] if 0<=k-j<n and k-j!=j) 
            #                    + (x - a[j]) * (a[k-j] if 0<=k-j<n and k-j!=j)
            #                    + (x^2 - a[j]^2) if k=2j
            # Actually: c[k] = ... + a[j]*a[k-j] + a[k-j]*a[j] + ...
            # If k != 2j: c[k] changes by 2*(x-a[j])*a[k-j] (if 0<=k-j<n)
            # If k = 2j: c[k] changes by (x^2 - a[j]^2) + 2*(x-a[j])*a[k-j] = (x-a[j])*(x+a[j]) + 0
            #   Wait: c[2j] includes the term a[j]*a[j] = a[j]^2
            #   After change: x^2
            #   Change: x^2 - a[j]^2 = (x-a[j])(x+a[j])
            #   Plus other terms involving a[j]: none (since a[j] only appears as a[j]*a[2j-j] = a[j]*a[j])
            #   So total change in c[2j] = x^2 - a[j]^2
            
            # For k != 2j (and 0<=k-j<n): change in c[k] = 2*(x-a[j])*a[k-j]
            # For k where k-j = j (i.e., k=2j): change = x^2 - a[j]^2
            # For k where 0 > k-j or k-j >= n: no change
            
            # This is a 1D problem: minimize max_k (c[k] + delta_k(x)) where
            # delta_k(x) = (x-a[j]) * 2*a[k-j] (if 0<=k-j<n, k!=2j)
            #             (x^2 - a[j]^2) (if k=2j)
            #             0 (otherwise)
            
            # This is non-convex in general (because of the max), but we can
            # use a simple approach: try a few candidate values of x and pick the best.
            
            # Simple approach: try x = a[j] * factor for various factors
            base = a[j]
            factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5,
                       0.3, 0.1, 0.6, 0.9, 1.1, 1.4, 1.7, 2.0]
            
            best_x = base
            best_local_M = M
            
            for f in factors:
                x = base * f
                if x < 0:
                    x = 0.0
                if x > 1000:
                    continue
                    
                a_trial = a.copy()
                a_trial[j] = x
                # Recompute only the affected c values
                # c[k] changes for k where 0<=k-j<n: k in [j, j+n-1]
                # c[2j] changes by x^2 - a[j]^2 (but this is included in the general formula
                # if we use: delta_c[k] = (x - a[j]) * (a[k-j] + a[k-j]) for k!=2j
                # and delta_c[2j] = x^2 - a[j]^2
                # But since the "2" factor and the special case are tricky, let's just
                # recompute c for the affected indices.
                
                # Actually, for efficiency, let's just recompute the full c
                # (it's fast enough with FFT for moderate n)
                c_trial = fft_convolve(a_trial, a_trial)
                M_trial = np.max(c_trial)
                
                if M_trial < best_local_M:
                    best_local_M = M_trial
                    best_x = x
            
            if best_x != base:
                a[j] = best_x
                improved = True
                # Check if overall score improved
                new_score = local_score(a)
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_score


def make_legendre_init(n, rng):
    """Create initialization based on Legendre/quadratic residue sequence."""
    # Find the smallest prime >= n
    p = n
    while True:
        is_prime = True
        for i in range(2, int(np.sqrt(p)) + 1):
            if p % i == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 1
    
    # Legendre symbol: 1 if x is QR mod p, -1 if not, 0 if x=0
    legendre = np.ones(p + 1)
    for x in range(1, p):
        legendre[x] = np.power(x, (p - 1) // 2, p)
        if legendre[x] == p - 1:
            legendre[x] = -1
    
    # Map to non-negative: a[i] = 0.5 + 0.5 * legendre[i] (gives 0 or 1)
    # Or smoother: a[i] = 1 + 0.5 * legendre[i % p] (gives 0.5 or 1.5)
    a = np.ones(n)
    for i in range(n):
        a[i] = 1.0 + 0.5 * legendre[i % p]
    
    # Add small noise and interpolate
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_chirp_init(n, alpha, beta, rng):
    """Chirp (quadratic phase) based initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * beta * t * t / (n + 1))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_multitone_init(n, rng, n_tones=5):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_smooth_random_init(n, rng, sigma=0.3):
    """Smooth random initialization (low-pass filtered)."""
    a = np.ones(n) + sigma * rng.standard_normal(n)
    # Simple moving average filter
    kernel_size = max(3, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    return np.clip(a, 0.05, None)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def sa_final_polish(a_init, time_budget, rng):
    """Final simulated annealing polish with small perturbations."""
    a = np.clip(a_init, 0.01, None).copy()
    n = len(a)
    S = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    current_score = best_score
    
    t0 = time.monotonic()
    temperature = 0.001
    cooling = 0.9999
    iter_count = 0
    
    while time.monotonic() - t0 < time_budget:
        # Generate candidate
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.5:
            # Perturb top contributors to peak
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(2, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.02, 0.008, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(4, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.008, size=n_inc))
        elif r < 0.8:
            # Local perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.004, size=n))
        
        candidate = np.clip(candidate, 0.01, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (S / cand_sum)
        
        new_score = local_score(candidate)
        iter_count += 1
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            a = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best_a, best_score


def lp_optimize_fast(f_init, time_budget, t_init=0.3):
    """LP-based optimization (faster version with smaller time limit per LP solve)."""
    from scipy.optimize import linprog
    
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    rng_local = np.random.default_rng(99999)
    
    while time.monotonic() - t0 < time_budget:
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        # Build constraint matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = f
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        remaining = time_budget - (time.monotonic() - t0)
        lp_time = min(15.0, remaining * 0.2)
        if lp_time < 2:
            break
        
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': lp_time})
        
        if not result.success:
            break
        
        g = result.x
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g_rescaled = g * (sum_f / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (sum_f / sum_new)
        
        score_new = local_score(f_new)
        
        if score_new < best_score - 1e-12:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
            t = min(0.8, t * 1.05)
        else:
            f = f_new
            stagnant += 1
            if stagnant > 10:
                t *= 0.7
                stagnant = 0
            if stagnant > 40:
                f = f + 0.02 * rng_local.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                t = 0.2
                stagnant = 0
    
    return best_f, best_score
