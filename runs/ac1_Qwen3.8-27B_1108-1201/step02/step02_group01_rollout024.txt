"""
Optimized peak reduction for non-negative step functions using
subgradient descent, LP-based refinement with constraint reduction,
spectral shaping, and multi-length search to minimize
2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.ones_like(a) * S / len(a)
    return a * S / s


def subgradient_optimize(a, time_limit, verbose=False):
    """
    Fast subgradient-based peak reduction.
    Tracks multiple peaks and uses weighted subgradient descent
    with adaptive step size and occasional perturbations.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    lr = 1.0 / n
    rng = np.random.default_rng(777)
    stall = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        if step % 50 == 0:
            conv = fft_convolve(a, a)
            current_peak = np.max(conv)
            mean_conv = np.mean(conv)
            excess = np.maximum(conv - mean_conv, 0)
            
            # Track top peaks
            num_peaks = max(3, min(10, n // 50))
            peak_thresh = np.mean(conv) + 0.5 * np.std(conv)
            peak_mask = conv > peak_thresh
            if np.sum(peak_mask) < 2:
                top_idx = np.argsort(conv)[-num_peaks:]
                peak_mask = np.zeros(len(conv), dtype=bool)
                peak_mask[top_idx] = True
            
            # Compute weighted subgradient
            grad = np.zeros(n)
            for m in np.where(peak_mask)[0]:
                w = excess[m]
                for j in range(n):
                    jm = m - j
                    if 0 <= jm < n:
                        grad[j] += w * a[jm]
                    jpm = j - m
                    if 0 <= jpm < n and jpm != jm:
                        grad[j] += w * a[jpm]
            
            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-10:
                grad = grad / gnorm
            
            # Step
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_seq(a_new, S)
            
            new_peak = np.max(fft_convolve(a_new, a_new))
            
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = a_new.copy()
                a = a_new
                stall = 0
                lr = min(lr * 1.02, 3.0 / n)
            elif new_peak < current_peak:
                a = a_new
                stall += 1
            else:
                stall += 1
                lr *= 0.95
            
            # Periodic perturbation to escape local minima
            if stall > 50:
                pert = rng.normal(0, 0.01, n)
                a = a + pert * a
                a = np.maximum(a, 0.0)
                a = normalize_seq(a, S)
                stall = 0
                lr = 0.5 / n
            
            if verbose and step % 500 == 0:
                score = 2.0 * n * best_peak / (S * S)
                print(f"    SG step {step}: peak={best_peak:.6f}, score={score:.6f}, lr={lr:.5f}")
        step += 1
    
    return best_a


def build_lp_matrix_reduced(a, conv_a, M, frac=0.85):
    """
    Build reduced LP constraint matrix using only near-active constraints.
    Only includes rows where conv_a[m] > frac * M.
    """
    n = len(a)
    threshold = frac * M
    active_rows = np.where(conv_a > threshold)[0]
    
    if len(active_rows) == 0:
        active_rows = np.array([np.argmax(conv_a)])
    
    A = np.zeros((len(active_rows), n))
    for idx, m in enumerate(active_rows):
        j_min = max(0, m - n + 1)
        j_max = min(m, n - 1)
        for j in range(j_min, j_max + 1):
            A[idx, j] = a[m - j]
    return A


def lp_refine_fast(a, time_limit, verbose=False):
    """
    LP-based iterative refinement with constraint reduction.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        # Build reduced LP
        A = build_lp_matrix_reduced(a, conv_a, M, frac=0.80)
        c_obj = -np.ones(n)
        b_ub = np.full(A.shape[0], M)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'maxiter': 5000})
        except Exception:
            break
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        g = normalize_seq(g, S)
        
        # Line search for best blend parameter
        best_t = 0.0
        best_conv_t = M
        for t_try in np.concatenate([np.linspace(0.05, 0.5, 10), np.linspace(0.5, 0.9, 5)]):
            b = (1 - t_try) * a + t_try * g
            b = np.maximum(b, 0.0)
            b = normalize_seq(b, S)
            b_peak = np.max(fft_convolve(b, b))
            if b_peak < best_conv_t:
                best_conv_t = b_peak
                best_t = t_try
        
        if best_t < 0.01:
            break
        
        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = normalize_seq(a, S)
        
        new_peak = np.max(fft_convolve(a, a))
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a.copy()
        
        if verbose and iteration % 10 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    LP iter {iteration}: peak={best_peak:.6f}, score={score:.6f}, t={best_t:.3f}")
        
        if best_t < 0.02:
            break
        iteration += 1
    
    return best_a


def spectral_refine(a, time_limit, verbose=False):
    """
    Spectral domain refinement: iteratively flatten the squared spectrum.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    
    # Use FFT length that is a power of 2 >= 2n-1
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    lr = 0.1
    stall = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        F = np.fft.rfft(a, L)
        G = F * F  # This gives FFT of circular conv = linear conv (since L >= 2n-1)
        c = np.fft.irfft(G, L)[:2 * n - 1]
        
        current_peak = np.max(c)
        mean_c = np.mean(c)
        
        # Find peaks
        peak_thresh = mean_c + 0.3 * (current_peak - mean_c)
        peak_positions = np.where(c > peak_thresh)[0]
        
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(c)])
        
        # Build the "error signal" in time domain
        error = np.zeros(L)
        for p in peak_positions:
            error[p] = c[p] - mean_c
        
        # Transform error to frequency
        error_F = np.fft.rfft(error, L)
        
        # Update F: we want to reduce G = F*F at positions that cause peaks
        # Newton-like: F_new = F - lr * error_F / (2*F)
        F_real = F.real.copy()
        F_imag = F.imag.copy()
        
        G_real = G.real
        G_imag = G.imag
        G_mag = np.abs(G)
        
        # Scale factor to avoid division by zero
        scale = np.maximum(G_mag, np.max(G_mag) * 0.01)
        
        # Complex division: error_F / G
        div_real = (error_F.real * G_real + error_F.imag * G_imag) / scale**2
        div_imag = (error_F.imag * G_real - error_F.real * G_imag) / scale**2
        
        F_new = F - lr * 0.5 * (div_real + 1j * div_imag)
        
        # Preserve Hermitian symmetry (rfft already handles this)
        a_new = np.fft.irfft(F_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)
        
        new_peak = np.max(fft_convolve(a_new, a_new))
        
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
            stall = 0
        elif new_peak < current_peak:
            a = a_new
            stall += 1
        else:
            stall += 1
            lr *= 0.9
        
        if stall > 100:
            lr = 0.05
            stall = 0
        
        step += 1
        if verbose and step % 200 == 0:
            score = 2.0 * n * best_peak / (S * S)
            print(f"    Spec step {step}: peak={best_peak:.6f}, score={score:.6f}, lr={lr:.4f}")
    
    return best_a


def coordinate_search(a, time_limit, verbose=False):
    """
    Coordinate-wise perturbation search.
    Tries small random perturbations to groups of coordinates.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        
        # Find elements contributing to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        # Strategy: reduce top contributors, boost others
        top_k = max(2, n // 20)
        top_idx = np.argsort(contribs)[-top_k:]
        other_idx = np.setdiff1d(np.arange(n), top_idx)
        
        candidate = a.copy()
        
        # Reduce top contributors
        for idx in top_idx:
            candidate[idx] *= np.exp(rng.normal(-0.03, 0.01))
        
        # Boost some others
        if len(other_idx) > 0:
            n_boost = max(2, n // 30)
            boost_idx = rng.choice(other_idx, size=n_boost, replace=False)
            candidate[boost_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_boost))
        
        candidate = np.maximum(candidate, 0.0)
        candidate = normalize_seq(candidate, S)
        
        new_peak = np.max(fft_convolve(candidate, candidate))
        
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = candidate.copy()
            a = candidate
        elif new_peak < np.max(conv) * 1.01 and rng.random() < 0.3:
            a = candidate
    
    return best_a


def make_structured_init(n, rng):
    """Create a structured initial sequence designed to have flat convolution."""
    t = np.arange(n, dtype=np.float64)
    init_type = rng.integers(0, 6)
    
    if init_type == 0:
        # Constant with small noise
        a = 1.0 + 0.1 * rng.standard_normal(n)
    elif init_type == 1:
        # Cosine taper (reduces edge effects)
        a = 0.5 * (1 + np.cos(np.pi * t / (n - 1)))
        a = a + 0.05 * rng.standard_normal(n)
    elif init_type == 2:
        # Chirp
        alpha = 0.2 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(2 * np.pi * alpha * t**2 / (2 * n))
    elif init_type == 3:
        # Multi-tone
        a = np.ones(n)
        for _ in range(3):
            f = rng.integers(1, n // 4)
            amp = rng.uniform(0.05, 0.15)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 4:
        # Raised cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * t / n))
        a = np.maximum(a, 0.01)
    else:
        # Log-normal
        a = np.exp(rng.normal(0, 0.2, n))
    
    return np.maximum(a, 0.01)


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old + 1)[:-1]
    x_new = np.linspace(0, 1, n_new + 1)[:-1]
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining subgradient, LP, spectral,
    and coordinate search methods across multiple sequence lengths.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    def try_update(a):
        nonlocal best_score, best_seq
        if a is None:
            return False
        a_clip = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False
    
    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        try_update(provided_seq)
    
    # Define sequence lengths to try
    lengths = [384, 512, 640, 768]
    
    # For each length, try the multi-phase approach
    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        
        # Get initial sequence for this length
        if provided_seq is not None:
            a0 = interpolate_sequence(provided_seq, n)
            a0 = np.clip(a0, 0.01, None)
        else:
            a0 = make_structured_init(n, rng)
        
        before_score = local_score(a0)
        
        # Phase 1: Fast subgradient (20% of per-length budget)

        t1 = 0.25 * per_length_budget
        
        result = subgradient_optimize(a0, t1, verbose=False)
        try_update(result)

        improved = try_update(result)

        # Phase 4: Coordinate search (15% of budget)

        t4 = 0.15 * per_length_budget
        result = coordinate_search(result, t4, verbose=False)

        if remaining > 30:
