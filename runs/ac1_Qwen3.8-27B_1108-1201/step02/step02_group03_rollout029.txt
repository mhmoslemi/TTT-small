"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses a linear program at each step to find the optimal direction, combined with
adaptive line search, gradient polishing, and multi-restart strategy.
Primary focus on n=384 with exploration of neighboring lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to a target sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def build_toeplitz(f):
    """Build matrix A where A @ g gives convolve(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    return A


def lp_max_sum(f, M, time_limit=20):
    """LP: max sum(g) s.t. T(f)g <= M, g >= 0. Returns g or None."""
    n = len(f)
    c_obj = -np.ones(n)
    A = build_toeplitz(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception as e:
        print(f"    LP error: {e}")
    return None


def lp_min_peak(f, S_target, time_limit=20):
    """LP: min M' s.t. T(f)g <= M'*1, sum(g) = S_target, g >= 0. Returns (g, M') or (None, None)."""
    n = len(f)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A = build_toeplitz(f)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S_target])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presize': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None


def soft_max_gradient(a, tau):
    """Gradient of (1/tau)*log(sum(exp(tau*conv(a,a)))) w.r.t. a. Returns (grad, weights)."""
    n = len(a)
    c = np.convolve(a, a)
    c_shift = c - c.max()
    w = np.exp(tau * c_shift)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad, w


def line_search_blend(a, g_norm, S_target, candidates=None):
    """Find best blend factor t minimizing objective of (1-t)*a + t*g_norm."""
    if candidates is None:
        candidates = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]
    best_t = 0.0
    best_s = compute_obj(a)
    for t in candidates:
        cand = (1.0 - t) * a + t * g_norm
        cand = np.maximum(cand, 0.0)
        cand = normalize(cand, S_target)
        s = compute_obj(cand)
        if s < best_s - 1e-12:
            best_s = s
            best_t = t
    return best_t, best_s

        a_new = np.maximum(a_new, 0.0)

            best_score = score

        else:
            a = a_new
            no_improve += 1

            mom = np.zeros(n)
            lr = min(lr * 1.1, 0.5)
            no_improve = 0
            if lr > 0.4:
                break
        
        tau *= 1.002
        if tau > tau_end:

def sparsify_and_retry(a, S_target, deadline, max_rounds=5):
    """Try zeroing out small entries and re-optimizing."""

    best_score = compute_obj(a)
    best_a = a.copy()

            s = compute_obj(a_sp)
