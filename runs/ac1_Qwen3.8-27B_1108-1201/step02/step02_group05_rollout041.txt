**Key insight for local search**: The peak of the convolution is at some position k*. To reduce `conv[k*]`, we need to reduce `sum_i a[i]*a[k*-i]`. The most effective way is to:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of mirror descent,
LP-based refinement, and targeted local search. Starts from the parent
sequence and explores multiple initializations with adaptive time allocation.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def normalize(a):
    """Normalize non-negative vector to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgrad(a, k):
    """Subgradient of conv(a,a)[k] w.r.t. a. Result: grad[i] = 2*a[k-i] if valid."""
    n = len(a)
    grad = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    if i_lo <= i_hi:
        # grad[i] = 2*a[k-i] for i in [i_lo, i_hi]
        # k-i ranges from k-i_lo down to k-i_hi
        grad[i_lo:i_hi+1] = 2.0 * a[k-i_hi:k-i_lo+1][::-1]
    return grad


def mirror_descent(a, time_limit, eta0=0.5, decay=0.999, multi_peak=True, peak_frac=0.95):
    """Mirror descent optimization. Returns (best_a, best_obj)."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        if multi_peak:
            threshold = peak_frac * M
            active = np.where(conv >= threshold)[0]
        else:
            active = np.array([np.argmax(conv)])
        
        # Average subgradient over active peaks
        grad = np.zeros(n)
        for k in active:
            grad += compute_subgrad(a, k)
        grad /= len(active)
        
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - eta * grad
        # Rescale to prevent underflow
        log_a = log_a - log_a.max() + 1.0  # Shift so max is ~1
        a_new = np.exp(log_a)
        a_new = normalize(a_new)
        
        obj_new = obj_value(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        eta *= decay
        
        if no_improve > 200:
            # Try with smaller eta
            eta = eta0 * 0.5
            no_improve = 0
        
        if eta < 1e-6:
            break
    
    return best_a, best_obj


def solve_lp_direction(f, M):
    """LP: find g >= 0 maximizing sum(g) s.t. conv(f,g)[k] <= M for all k.
    Returns normalized g or None if infeasible."""
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A_ub where A_ub @ g <= M for each k
    # (A_ub[k, j]) = f[k-j] if 0 <= k-j < n, else 0
    # But this is dense in general. For n=512, it's 1023x512 which is manageable.
    
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A[k, i_lo:i_hi+1] = f[k-i_hi:k-i_lo+1][::-1]
    
    c = -np.ones(n)  # Minimize -sum(g) = maximize sum(g)
    b = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                                     options={'presolve': True})
        if res.success and res.fun > -1e-10:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None


def lp_step_with_line_search(a, M, max_trials=8):
    """Try LP direction with geometric line search. Returns (new_a, new_obj) or None."""
    g = solve_lp_direction(a, M)
    if g is None:
        return None
    
    a = normalize(a.copy())
    best_obj = obj_value(a)
    best_a = a.copy()
    
    t = 0.5
    for _ in range(max_trials):
        a_trial = normalize((1.0 - t) * a + t * g)
        obj_trial = obj_value(a_trial)
        if obj_trial < best_obj - 1e-12:
            best_obj = obj_trial
            best_a = a_trial
            # Try to go further
            t = min(1.0, t * 1.5)
        else:
            t *= 0.5
            if t < 0.01:
                break
    
    return best_a, best_obj


def lp_optimize_fast(a0, time_limit):
    """Fast LP-based optimization with simple line search."""
    a = normalize(a0.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = obj_value(a)

                best_a = a_new.copy()

                stall = 0
            else:

        else:

        if stall > 50:

def local_search_improved(a, time_limit):
    """Improved local search: batch mass redistribution from peak contributors."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()
    stall = 0

        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)

        # Find top contributors and safe targets
        sorted_idx = np.argsort(-contrib)
        top_contributors = sorted_idx[:20]

            stall += 1
            if stall > 200:
                break
        else:
            stall = 0
    
    return best_a, best_obj


def get_parent(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            p = p[:n]
        else:
            # Upsample via interpolation
            x_old = np.linspace(0, 1, p.size)
            x_new = np.linspace(0, 1, n)
            p = np.interp(x_new, x_old, p)
        p = np.maximum(p, 0.0)
        s = p.sum()
        if s < 1e-10:
            return None
        return p / s
    except Exception:
        return None


def make_init(n, rng, init_type='parent'):

    if init_type == 'parent':
        p = get_parent(n)
        if p is not None:
            return p
        return np.ones(n) / n
    elif init_type == 'uniform':

```python
def proj_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

The projection of `v` onto the simplex `{x : x >= 0, sum(x) = 1}` is:
`proj(v) = max(v - theta, 0)` where `theta` is chosen so that `sum(max(v - theta, 0)) = 1`.

The algorithm:

3. `theta = (sum_{j=0}^{rho} u[j] - 1) / (rho+1)`
4. `proj(v) = max(v - theta, 0)`

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def normalize(a):
    """Normalize non-negative vector to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * (np.arange(1, n + 1)) > (cssv - 1.0)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, k):

def solve_lp_direction(f, M):
    """LP: find g >= 0 maximizing sum(g) s.t. conv(f,g)[k] <= M for all k."""
    n = len(f)
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A[k, i_lo:i_hi + 1] = f[k - i_hi:k - i_lo + 1][::-1]
    c = -np.ones(n)
    b = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                     method='highs', options={'presolve': True})
        if res.success and res.fun > -1e-10:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None
