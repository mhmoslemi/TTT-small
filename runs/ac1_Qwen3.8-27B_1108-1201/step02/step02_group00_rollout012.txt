import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def soft_max(conv, t):
    """Compute soft max of convolution values."""
    m = conv.max()
    shifted = t * (conv - m)
    log_sum = m * t + np.log(np.exp(shifted).sum())
    return log_sum / t

def soft_max_weights(conv, t):
    """Compute softmax weights for the smooth max."""
    m = conv.max()
    shifted = t * (conv - m)
    w = np.exp(shifted)
    w /= w.sum()
    return w

def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def adam_smooth_max(a, t_init=100, t_final=10000, n_steps=5000, 
                     lr=0.001, beta1=0.9, beta2=0.999, time_limit=100):
    """Optimize using Adam with smooth max objective on the simplex."""
    n = len(a)
    a = a.copy() / a.sum()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    
    t0 = time.monotonic()
    best_obj = float('inf')
    best_a = a.copy()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        progress = step / n_steps
        t = t_init * (t_final / t_init) ** progress
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        if step % 100 == 0:
            print(f"  Step {step}: obj={cur_obj:.6f}, t={t:.1f}")
        
        # Compute gradient of smooth max
        w = soft_max_weights(conv, t)
        
        # Gradient: grad[j] = sum_k w[k] * 2 * a[k-j] * I(0<=k-j<n)
        grad = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-10:
                continue
            wk = 2.0 * w[k]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += wk * a[k - idx]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(step + 1))
        v_hat = v / (1 - beta2**(step + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = project_simplex(a)
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating linear programming,
smooth gradient descent, and coordinate descent polishing.
The key innovation is the alternating LP approach: given fixed a, solve an LP
to find the optimal b that minimizes max_k conv(a,b)_k, then blend b into a.
This exploits the convexity of the subproblem while handling the non-convexity
of the overall bilinear problem through alternating minimization.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def convolve_xy(x, y):
    """Linear cross-convolution via FFT."""
    n = len(x)
    m = len(y)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, fft_size)
    Y = np.fft.rfft(y, fft_size)
    return np.fft.irfft(X * Y, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_for_b(a):
    """
    Given fixed a (sum=1, a>=0), find b (sum=1, b>=0)
    that minimizes max_k conv(a,b)_k.
    
    Returns (b, optimal_t).
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix C where C[k, i] = a_i if 0 <= k-i < n
    C = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            C[k, i] = a[k - i]
    
    # LP: minimize t
    # s.t. C @ b - t <= 0  (num_constraints inequalities)
    #      sum(b) = 1      (1 equality)
    #      b >= 0
    # Variables: [b_0, ..., b_{n-1}, t]
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((num_constraints, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        b = result.x[:n]
        b = np.maximum(b, 0)
        s = b.sum()
        if s > 1e-15:
            b = b / s
        t = result.x[n]
        return b, t
    else:
        return a.copy(), 2.0 * n  # fallback
    
def alternating_lp(a0, max_iters=1000, alpha_init=0.5, alpha_min=0.1,
                   time_limit=60.0):
    """
    Alternating LP: iterate between solving LP for b given a, and updating a.
    """
    n = len(a0)
    a = a0.copy() / a0.sum()
    a = np.maximum(a, 0)
    
    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    alpha = alpha_init
    
    prev_obj = best_obj
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: given a, find optimal b
        b, t_opt = solve_lp_for_b(a)
        
        # Blend
        a_new = (1 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s
        
        # Check objective
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a_new.copy()
        
        a = a_new
        
        # Decrease alpha over time
        progress = min(1.0, it / max_iters)
        alpha = alpha_init - (alpha_init - alpha_min) * progress
        
        # Convergence check
        if abs(cur_obj - prev_obj) < 1e-7:
            # Try with smaller alpha to squeeze out more
            alpha = max(0.01, alpha * 0.5)
        
        prev_obj = cur_obj
    
    return best_a, best_obj

def smooth_max_gradient_descent(a0, time_limit=60.0, t_init=50.0, t_max=5000.0,
                                 lr=0.005):
    """
    Gradient descent on the simplex with smooth max (log-sum-exp) objective.
    """
    n = len(a0)
    a = a0.copy() / a0.sum()
    a = np.maximum(a, 0)
    
    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        # Anneal temperature
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        t = t_init * (t_max / t_init) ** progress
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Compute smooth max gradient
        m_conv = conv.max()
        shifted = t * (conv - m_conv)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum
        
        # Gradient: grad[j] = sum_k w[k] * 2 * a[k-j] * I(0<=k-j<n)
        grad = np.zeros(n)
        for k in range(len(w)):
            wk = 2.0 * w[k]
            if wk < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            i_arr = np.arange(i_start, i_end)
            grad[i_arr] += wk * a[k - i_arr]
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = project_simplex(a)
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit=30.0, max_rounds=100):
    """
    Coordinate descent: try transferring mass between pairs of coordinates.
    """
    n = len(a0)
    a = a0.copy() / a0.sum()
    a = np.maximum(a, 0)
    
    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < max_rounds:
        # Pick random pair
        j = rng.integers(n)
        k = rng.integers(n)
        if j == k:
            continue
        
        # Try transferring mass from j to k (and k to j)
        best_delta = 0.0
        best_new_obj = best_obj
        
        # Bisection for optimal delta
        # If we transfer delta from j to k:
        #   a_j' = a_j - delta, a_k' = a_k + delta
        #   delta in [0, a_j]
        
        if a[j] < 1e-15:
            no_improve += 1
            continue
        
        lo, hi = 0.0, a[j]
        # We want to minimize max_k conv(a')_k
        # Use ternary search
        for _ in range(20):
            mid1 = lo + (hi - lo) / 3
            mid2 = hi - (hi - lo) / 3
            
            a_test1 = a.copy()
            a_test1[j] -= mid1
            a_test1[k] += mid1
            obj1 = fast_convolve(a_test1).max()
            
            a_test2 = a.copy()
            a_test2[j] -= mid2
            a_test2[k] += mid2
            obj2 = fast_convolve(a_test2).max()
            
            if obj1 < obj2:
                hi = mid2
            else:
                lo = mid1
        
        delta = (lo + hi) / 2
        a_test = a.copy()
        a_test[j] -= delta
        a_test[k] += delta
        a_test = np.maximum(a_test, 0)
        s = a_test.sum()
        if s > 1e-15:
            a_test = a_test / s
        
        obj_test = 2.0 * n * fast_convolve(a_test).max()
        
        if obj_test < best_obj - 1e-12:
            a = a_test
            best_obj = obj_test
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating linear programming
with gradual relaxation, smooth gradient descent polishing, and coordinate descent.
The alternating LP exploits the convexity of the cross-convolution subproblem
while the gradient methods handle the non-convex bilevel structure.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_cross_conv_matrix(a):
    """Build matrix C where (C @ b)_k = conv(a,b)_k."""
    n = len(a)
    num_k = 2 * n - 1
    C = np.zeros((num_k, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, num_k)
        C[k_start:k_end, i] = a[:k_end - k_start]
    return C

def solve_lp_cross(a, method='highs'):
    """Solve LP: minimize t s.t. conv(a,b)_k <= t, sum(b)=1, b>=0."""
    n = len(a)
    C = build_cross_conv_matrix(a)
    num_k = 2 * n - 1
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.hstack([C, -np.ones((num_k, 1))])
    b_ub = np.zeros(num_k)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method=method)
    
    if res.status == 0:
        b = np.maximum(res.x[:n], 0)
        s = b.sum()
        if s > 1e-15:
            b /= s
        return b, float(res.x[n])
    return a.copy(), None

def alt_lp_optimize(a0, time_limit=60.0, alpha_init=0.5, alpha_min=0.05,
                    max_iters=5000):
    """Alternating LP optimization with blending."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    t0 = time.monotonic()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        alpha = alpha_init * (alpha_min / alpha_init) ** (it / max_iters)
        if it > max_iters * 0.8:
            alpha = alpha_min
        
        b, t_opt = solve_lp_cross(a)
        
        a_new = (1 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 500:
            # Reset to best and reduce alpha
            a = best_a.copy()
            alpha_init = max(0.01, alpha_init * 0.5)
            no_improve = 0
    
    return best_a, best_obj

def smooth_grad_optimize(a0, time_limit=60.0, t_init=20.0, t_max=2000.0,
                         lr=0.01):
    """Smooth max gradient descent with Adam on the simplex."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    t0 = time.monotonic()

    best_a = a.copy()
    
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    b1, b2 = 0.85, 0.999
    step = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        t = t_init * (t_max / t_init) ** progress
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improbe = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 2000:
            a = best_a + np.random.standard_normal(n) * 0.005
            a = project_simplex(a)
            m_adam = np.zeros(n)
            v_adam = np.zeros(n)
            step = 0
            no_improve = 0
        
        # Softmax weights
        m_conv = conv.max()
        if m_conv < 1e-15:
            break
        shifted = t * (conv - m_conv)
        shifted -= shifted.max()
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        # Gradient
        grad = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-10:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            idx = np.arange(i_start, i_end)
            grad[idx] += 2.0 * w[k] * a[k - idx]
        
        # Adam
        step += 1
        m_adam = b1 * m_adam + (1 - b1) * grad
        v_adam = b2 * v_adam + (1 - b2) * grad ** 2
        m_hat = m_adam / (1 - b1 ** step)
        v_hat = v_adam / (1 - b2 ** step)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = project_simplex(a)
    
    return best_a, best_obj

def coord_descent_optimize(a0, time_limit=30.0, seed=42):

    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        j = int(rng.integers(n))
        k = int(rng.integers(n))
        if j == k or a[j] < 1e-10:
            no_improve += 1
            continue
        
        # Ternary search for optimal transfer amount
        lo, hi = 0.0, a[j] * 0.99
        for _ in range(15):
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            
            a1 = a.copy()
            a1[j] -= m1
            a1[k] += m1
            obj1 = fast_convolve(a1).max()
            
            a2 = a.copy()
            a2[j] -= m2
            a2[k] += m2
            obj2 = fast_convolve(a2).max()
            
            if obj1 < obj2:
                hi = m2
            else:
                lo = m1
        
        delta = (lo + hi) / 2
        if delta < 1e-12:
            no_improve += 1
            continue
        
        a_new = a.copy()
        a_new[j] -= delta
        a_new[k] += delta
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
        
        obj_new = 2.0 * n * fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-12:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_a, best_obj
