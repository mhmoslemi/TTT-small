def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smoothed_obj_grad(a, T):
    """Compute smoothed objective and its gradient."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return 1e10, np.zeros(n)
    
    b = fast_convolve(a)  # length 2n-1
    
    # Softmax weights (numerically stable)
    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    w = exp_b / exp_b.sum()
    
    # Smoothed max
    m_T = T * (np.log(exp_b.sum()) + b_max / T)
    
    # Objective
    obj = 2.0 * n * m_T / (S * S)
    
    # Gradient of m_T w.r.t. a
    # dm_T/da_i = 2 * sum_{k=i}^{i+n-1} w[k] * a[k-i]
    # (k-i ranges from 0 to n-1)
    dm = np.zeros(n)
    for i in range(n):
        k_max = min(i + n, len(w))
        dm[i] = 2.0 * np.dot(w[i:k_max], a[:k_max-i])
    
    # Full gradient
    # dg/da_i = 2n * (dm_T/da_i / S - 2*m_T / S^2)
    grad = 2.0 * n * (dm / S - 2.0 * m_T / (S * S))
    
    return obj, grad

def lbfgsb_optimize(a0, T, time_limit):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    
    start_time = time.monotonic()
    
    def objective_and_grad(x):
        x = np.maximum(x, 0.0)  # ensure non-negativity (L-BFGS-B should handle this, but just in case)
        obj, grad = compute_smoothed_obj_grad(x, T)
        return obj, grad
    
    # Use L-BFGS-B
    result = minimize(
        objective_and_grad,
        a,
        method='L-BFGS-B',
        jac=True,  # we provide the gradient
        bounds=[(0.0, None)] * n,
        options={'maxiter': 10000, 'maxfun': 50000, 'ftol': 1e-12, 'gtol': 1e-8,
                 'disp': False}
    )
    
    a_opt = np.maximum(result.x, 0.0)
    
    # If time remains, try continued optimization
    elapsed = time.monotonic() - start_time
    if elapsed < time_limit * 0.8 and result.success:
        # Try again with slightly different initial point
        noise = np.random.default_rng(int(start_time * 1000) % 10000).standard_normal(n) * 0.01
        a_pert = np.maximum(a_opt + noise, 0.0)
        result2 = minimize(
            objective_and_grad,
            a_pert,
            method='L-BFGS-B',
            jac=True,
            bounds=[(0.0, None)] * n,

            new_ai = max(0.0, res.x)
            a[i] = new_ai
            if abs(new_ai - a_i_old) > 1e-12:
                improved_any = True
        
        cur_obj = evaluate_sequence(a.tolist())
        if np.isfinite(cur_obj) and cur_obj < best_obj:
            best_obj = cur_obj
        
        if not improved_any:
            break
    
    return a


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Parent resampled
    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            if p.sum() > 1e-10:
                inits.append(('parent', p / p.sum()))
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits.append(('parent_rs', pp / pp.sum()))
    
    # Random
    inits.append(('random', rng.dirichlet(np.ones(n) * 0.5)))
    inits.append(('random_sparse', rng.dirichlet(np.ones(n) * 0.1)))
    
    # Sinusoidal
    for freq in [1, 2, 3, 5]:
        s = np.abs(np.cos(2 * np.pi * freq * x / n)) + 0.05
        inits.append((f'sin_{freq}', s / s.sum()))
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits.append(('bimodal', bi / bi.sum()))
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(('ramp', ramp / ramp.sum()))
    
    # Gaussian
    for sf in [0.2, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits.append((f'gauss_{sf}', g / g.sum()))
    
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: L-BFGS-B on smoothed objective with temperature
    annealing, multi-length exploration, and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass
    
    best_score = float('inf')
    best_a = None
    
    # === Phase 1: Quick Scan (400s) ===
    print("=== Phase 1: Quick Multi-Length Scan ===")
    n_parent = len(parent) if parent is not None else 256
    
    lengths = sorted(set([
        n_parent,
        max(64, n_parent - 64),
        n_parent + 64,
        max(64, n_parent - 128),
        n_parent + 128,
        max(64, n_parent // 2),
    ]))
    lengths = [n for n in lengths if 64 <= n <= 1024]
    
    phase1_end = time.monotonic() + 400.0
    all_results = []
    
    for n in lengths:
        if time.monotonic() > phase1_end - 60:
            break
        inits = generate_inits(n, parent, rng)
        # Limit to 5 best initializations
        init_names = ['parent', 'parent_rs', 'uniform', 'random', 'sin_2', 'sin_1']
        inits = [(nm, a0) for nm, a0 in inits if nm in init_names]
        if len(inits) == 0:
            inits = generate_inits(n, parent, rng)[:4]
        inits = inits[:5]
        
        time_per = max(15.0, (phase1_end - time.monotonic() - 30) /
                       (len(inits) * max(1, len(lengths) - lengths.index(n))))
        
        for init_name, a0 in inits:
            if time.monotonic() > phase1_end - 30:
                break
            t_alloc = min(time_per, phase1_end - time.monotonic() - 20)
            if t_alloc < 8:
                break
            
            T = 0.005
            a_opt = lbfgsb_optimize(a0, T, t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                all_results.append((score, n, init_name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"  Phase1: n={n}, {init_name}, score={score:.6f}")
    
    all_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")
    
    # === Phase 2: Deep Temperature Annealing (1000s) ===
    print("=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + 1000.0
    
    # Take top 4 candidates
    top_candidates = []
    seen_keys = set()
    for score, n, name, a_opt in all_results:
        key = (n, name)
        if key not in seen_keys:
            top_candidates.append(a_opt.copy())
            seen_keys.add(key)
        if len(top_candidates) >= 4:
            break
    
    if best_a is not None and best_a not in top_candidates:
        top_candidates.append(best_a.copy())
    
    temps = [0.004, 0.002, 0.001, 0.0005, 0.0003]
    t_per_step = (phase2_end - time.monotonic() - 60) / (len(top_candidates) * len(temps))
    t_per_step = max(8.0, t_per_step)
    
    for a_init in top_candidates:
        a_cur = a_init.copy()
        n_cur = len(a_cur)
        
        for T in temps:
            if time.monotonic() > phase2_end - 30:
                break
            t_alloc = min(t_per_step, phase2_end - time.monotonic() - 20)
            if t_alloc < 5:
                break
            
            a_opt = lbfgsb_optimize(a_cur, T, t_alloc)
            a_cur = a_opt
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"  Phase2: T={T}, n={n_cur}, score={score:.6f}")
    
    print(f"Phase 2 best: {best_score:.6f}")
    
    # === Phase 3: Polish (remaining time) ===
    print("=== Phase 3: Polish ===")
    phase3_end = deadline
    
    if best_a is not None:
        remaining = phase3_end - time.monotonic() - 10
        n_best = len(best_a)
        
        # 3a: Very low temperature L-BFGS-B
        if remaining > 30:
            t_alloc = min(120.0, remaining * 0.4)
            a_opt = lbfgsb_optimize(best_a, 0.0002, t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  Polish L-BFGS: score={score:.6f}")
        
        # 3b: Coordinate descent
        if remaining > 30:
            t_alloc = min(120.0, remaining * 0.4)
            a_opt = coordinate_descent_polish(best_a, t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"  CoordDesc: score={score:.6f}")
        
        # 3c: Multi-start perturbation + re-optimize
        if remaining > 30:
            for attempt in range(5):
                if time.monotonic() > phase3_end - 15:
                    break
                t_alloc = min(40.0, (phase3_end - time.monotonic() - 10) / (5 - attempt))
                if t_alloc < 5:
                    break
                
                scale = max(0.001, 0.005 * best_a.max())
                noise = rng.standard_normal(n_best) * scale
                a_pert = np.maximum(best_a + noise, 0.0)
                
                T_small = 0.0005
                a_opt = lbfgsb_optimize(a_pert, T_small, t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"  Perturb {attempt}: score={score:.6f}")
        
        # 3d: Try slightly different lengths (insert/zeros)
        if remaining > 50 and best_a is not None:
            for delta_n in [16, -16, 32]:
                if time.monotonic() > phase3_end - 20:
                    break
                new_n = n_best + delta_n
                if new_n < 64 or new_n > 1024:
                    continue
                a_resamp = np.interp(np.linspace(0, 1, new_n),
                                     np.linspace(0, 1, n_best), best_a)
                a_resamp = np.maximum(a_resamp, 0)
                if a_resamp.sum() < 1e-10:
                    continue
                
                t_alloc = min(30.0, (phase3_end - time.monotonic() - 10) / 2)
                if t_alloc < 5:
                    continue
                a_opt = lbfgsb_optimize(a_resamp, 0.001, t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_a = a_opt.copy()
                    print(f"  Resample n={new_n}: score={score:.6f}")
    
    # Final verification
    if best_a is None:
        best_a = np.ones(128) / 128
    
    final_score = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final_score):
        best_a = np.ones(128) / 128
        final_score = float(evaluate_sequence(best_a.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_a)}")
    return [float(x) for x in best_a]
