Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

import scipy.optimize as optimize

import time

def lp_step(f, M, n, timeout=10):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M."""
    rows_list = []
    cols_list = []
    vals_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
            vals_list.append(f[j])
    
    A = sparse.csc_matrix((vals_list, (rows_list, cols_list)), shape=(2*n-1, n))
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def smoothed_mirror_descent(a, time_limit, t0=50.0, t1=5.0, eta0=0.5, patience_limit=200):
    """Mirror descent with log-sum-exp smoothing and temperature annealing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    a /= a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = eta0
    patience = 0
    t0_time = time.monotonic()
    
    while time.monotonic() - t0_time < time_limit:
        # Anneal temperature
        frac = min(1.0, (time.monotonic() - t0_time) / time_limit)
        t = t0 * (t1 / t0) ** frac
        
        conv = fast_convolve(a)
        conv_max = conv.max()
        scaled = t * (conv - conv_max)
        exp_scaled = np.exp(scaled)
        p = exp_scaled / exp_scaled.sum()
        
        # Compute gradient: g[i] = 2 * sum_k p[k] * a[k-i]
        # Use sparse p (only significant entries)
        thresh = p.max() * 1e-4
        active = np.where(p > thresh)[0]
        
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad[j_start:j_end+1] += 2.0 * p[k] * a[k-j_end:k-j_start+1][::-1]
        
        # Center the gradient (projection onto simplex tangent space)
        grad -= grad.mean()
        
        # Mirror descent update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= s
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
            a = a_new
            eta = min(eta * 1.01, 2.0)
        else:
            patience += 1
            eta *= 0.95
            if patience > patience_limit:
                break
            if patience % 50 == 0:
                # Restart with perturbation
                a = best_a * (1 + 0.1 * np.random.randn(n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 0:
                    a /= s
    
    return best_a, best_obj


def lp_optimize(a0, time_limit, n_lp_iters=50):

    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a /= a.sum()
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    t = 1.0
    patience = 0
    
    # Precompute LP structure
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    
    while time.monotonic() - t0 < time_limit:

        b = np.full(2*n-1, M)

                               options={'disp': False, 'time_limit': 5})
        
        if not res.success:
            break
        
        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        
        if np.max(np.abs(g - a)) < 1e-13:
            break
        
        improved = False
        for t_try in [t, 0.7*t, 0.5*t, 0.3*t, 0.1*t]:
            a_new = (1.0 - t_try) * a + t_try * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-12:

            if patience > 40:
                break

def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.05, 0.1, 0.15, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5*n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g
    
    r = np.arange(1, n+1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    w = max(1, n // 4)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    
    # Cosine window
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()
    
    # Random Dirichlet
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr
    
    # Triangle
    tri = np.maximum(1.0 - np.abs(x - 0.5*n) / (0.5*n), 0.01)
    inits['tri'] = tri / tri.sum()
    
    # Two peaks
    two = np.exp(-0.5*((x-0.25*n)/(0.05*n))**2) + np.exp(-0.5*((x-0.75*n)/(0.05*n))**2)
    inits['two'] = two / two.sum()
    
    # Sqrt ramp
    sr = np.sqrt(np.arange(1, n+1, dtype=np.float64))
    inits['sqrt'] = sr / sr.sum()
    
    return inits

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-guided alternating optimization with multi-M aggression,
smoothed log-sum-exp gradient descent with temperature annealing,
and multi-scale exploration across sequence lengths.

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_lp_indices(n):
    """Precompute row/col index structure for LP constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        rows_list.extend(range(i, i + n))
        cols_list.extend([i] * n)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_once(a, M, rows, cols, n, timeout=8):
    """Solve single LP: max sum(g) s.t. g>=0, (a*g)[k]<=M."""
    vals = np.tile(a, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(a0, time_limit, aggressive=False):
    """LP alternating optimization with optional multi-M aggression."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    t = 1.0
    patience = 0
    rows, cols = build_lp_indices(n)
    m_factors = [1.0, 0.95, 0.9, 0.85, 0.8] if aggressive else [1.0]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for mf in m_factors:
            M_try = M * mf
            if M_try < 1e-12:
                continue

            g = solve_lp_once(a, M_try, rows, cols, n)
            if g is None:
                continue

            g_sum = g.sum()
            if g_sum < 1e-12:
                continue
            g /= g_sum

            if np.max(np.abs(g - a)) < 1e-13:
                continue

            for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t]:
                a_new = (1.0 - t_try) * a + t_try * g
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-12:
                    continue
                a_new /= s_new
                conv_new = fast_convolve(a_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < best_obj - 1e-13:
                    a = a_new
                    M = conv_new.max()
                    best_obj = obj_new
                    best_a = a.copy()
                    patience = 0
                    t = min(1.0, t * 1.05)
                    improved = True
                    break

            if improved:
                break

        if not improved:
            patience += 1
            t *= 0.85
            if patience > 35:
                break

    return best_a, best_obj


def smoothed_gradient_step(a, t, top_k=None):
    """Compute smoothed gradient of log-sum-exp objective. Returns (grad, smoothed_val)."""
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    scaled = t * (conv - conv_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    smoothed_val = conv_max + np.log(exp_scaled.sum()) / t

    if top_k is not None and top_k < len(conv):
        idx = np.argsort(p)[::-1][:top_k]
        p_restricted = np.zeros_like(p)
        p_restricted[idx] = p[idx]
        p_restricted /= p_restricted.sum()
        p = p_restricted

    thresh = p.max() * 1e-4
    active = np.where(p > thresh)[0]

    grad = np.zeros(n)
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        grad[j_start:j_end + 1] += 2.0 * p[k] * a[k - j_end:k - j_start + 1][::-1]

    return grad, smoothed_val


def mirror_descent_smoothed(a0, time_limit, t_init=80.0, t_final=3.0, eta_init=0.4):
    """Mirror descent with log-sum-exp smoothing and temperature annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta_init
    patience = 0
    t0_time = time.monotonic()

    while time.monotonic() - t0_time < time_limit:
        frac = min(1.0, (time.monotonic() - t0_time) / time_limit)
        t = t_init * (t_final / t_init) ** frac

        grad, _ = smoothed_gradient_step(a, t)
        grad -= grad.mean()

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= s_new

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
            a = a_new
            eta = min(eta * 1.02, 1.5)
        else:
            patience += 1
            eta *= 0.95
            if patience > 150:
                # Perturb and restart
                rng_local = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                a = best_a * (1.0 + 0.05 * rng_local.standard_normal(n))
                a = np.maximum(a, 0.0)
                s_p = a.sum()
                if s_p > 1e-10:
                    a /= s_p
                else:
                    a = best_a.copy()
                patience = 0
                eta = eta_init
            elif patience > 300:
                break

    return best_a, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.08, 0.12, 0.18, 0.25]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    w = max(1, n // 5)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    rr = rng.dirichlet(np.ones(n) * 0.4)
    inits['rnd'] = rr

    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.02)
    inits['tri'] = tri / tri.sum()

    two = np.exp(-0.5 * ((x - 0.25 * n) / (0.06 * n)) ** 2) + \
          np.exp(-0.5 * ((x - 0.75 * n) / (0.06 * n)) ** 2)
    inits['two'] = two / two.sum()

    sr = np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    inits['sqrt'] = sr / sr.sum()

    # Oscillating
    osc = 0.5 + 0.5 * np.cos(4 * np.pi * x / n)
    osc = np.maximum(osc, 0.01)
    inits['osc'] = osc / osc.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry: hybrid LP + smoothed gradient + multi-scale search.
    Phases: (1) LP scan, (2) deep LP on top results, (3) gradient polish, (4) final refinement.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(0.0, float(budget_s) - 20.0)
    deadline = time.monotonic() + total_budget

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    if parent is not None:
        sc = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(sc):
            best_score = sc
            best_seq = parent.copy()
            print(f"Parent loaded: n={len(parent)}, score={sc:.6f}")

    all_results = []

    # === Phase 1: LP scan across lengths ===
    print(f"Phase 1: LP scan (budget {total_budget:.0f}s)")
    phase1_deadline = deadline - total_budget * 0.55

    lengths = [256, 384, 512, 768]

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break

        inits = gen_inits(n, rng)

        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size),
                               np.maximum(parent, 0))
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()

        n_cfgs = len(inits)
        remaining = phase1_deadline - time.monotonic()
        time_per = max(6.0, remaining / max(1, n_cfgs))

        for name, a0 in inits.items():
            if time.monotonic() > phase1_deadline:
                break
            ta = min(time_per, phase1_deadline - time.monotonic())
            if ta < 4:
                break

            f_opt, _ = lp_optimize(a0, time_limit=ta)
            sc = float(evaluate_sequence(f_opt.tolist()))

            if np.isfinite(sc):
                all_results.append((sc, n, name, f_opt))
                if sc < best_score:
                    best_score = sc
                    best_seq = f_opt.copy()
                    print(f"  [P1] n={n:4d} {name:5s} score={sc:.6f}")

    print(f"Phase 1 done. Best so far: {best_score:.6f}")

    # === Phase 2: Deep LP on top results ===
    all_results.sort(key=lambda x: x[0])
    top = all_results[:6]

    phase2_deadline = deadline - total_budget * 0.20
    remaining2 = phase2_deadline - time.monotonic()

    if remaining2 > 30 and top:
        print(f"Phase 2: Deep LP on top {len(top)} ({remaining2:.0f}s)")
        t_each = remaining2 / len(top)

        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > phase2_deadline:
                break
            ta = min(t_each, phase2_deadline - time.monotonic())
            if ta < 8:
                break

            # Aggressive multi-M LP
            f2, _ = lp_optimize(f, time_limit=ta * 0.6, aggressive=True)
            sc2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = f2.copy()
                print(f"  [P2a] n={n} {name} aggressive LP: {sc2:.6f}")
            elif np.isfinite(sc2):
                sc2 = sc  # keep original if no improvement

            f3, _ = lp_optimize(f2, time_limit=ta * 0.4)
            sc3 = float(evaluate_sequence(f3.tolist()))
            if np.isfinite(sc3) and sc3 < best_score:
                best_score = sc3
                best_seq = f3.copy()
                print(f"  [P2b] n={n} {name} standard LP: {sc3:.6f}")

    print(f"Phase 2 done. Best so far: {best_score:.6f}")

    # === Phase 3: Smoothed gradient polish ===
    phase3_deadline = deadline - total_budget * 0.05
    remaining3 = phase3_deadline - time.monotonic()

    if remaining3 > 10 and best_seq is not None:
        print(f"Phase 3: Gradient polish ({remaining3:.0f}s)")
        n_polish = min(3, len(top)) if top else 1
        t_each3 = remaining3 / n_polish

        candidates_to_polish = []
        if best_seq is not None:
            candidates_to_polish.append(best_seq)
        for sc, n, name, f in top[:n_polish]:
            if f is not best_seq:
                candidates_to_polish.append(f)

        for i, f in enumerate(candidates_to_polish[:n_polish]):
            if time.monotonic() > phase3_deadline:
                break
            ta = min(t_each3, phase3_deadline - time.monotonic())
            if ta < 5:
                break

            f_g, _ = mirror_descent_smoothed(f, time_limit=ta, t_init=60.0, t_final=2.0, eta_init=0.35)
            sc_g = float(evaluate_sequence(f_g.tolist()))
            if np.isfinite(sc_g) and sc_g < best_score:
                best_score = sc_g
                best_seq = f_g.copy()
                print(f"  [P3-{i}] Gradient: {sc_g:.6f}")

            # Follow with LP
            if time.monotonic() < phase3_deadline - 5:
                tr = min(20.0, phase3_deadline - time.monotonic() - 5)
                f_lp, _ = lp_optimize(f_g, time_limit=tr, aggressive=True)
                sc_lp = float(evaluate_sequence(f_lp.tolist()))
                if np.isfinite(sc_lp) and sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = f_lp.copy()
                    print(f"  [P3-{i}] Post-grad LP: {sc_lp:.6f}")

    print(f"Phase 3 done. Best so far: {best_score:.6f}")

    # === Phase 4: Final refinement ===
    remaining4 = deadline - time.monotonic()
    if remaining4 > 10 and best_seq is not None:
        print(f"Phase 4: Final refinement ({remaining4:.0f}s)")

        # Multiple rounds of LP + gradient
        for round_i in range(4):
            if time.monotonic() > deadline:
                break
            tr = min(remaining4 / (4 - round_i), deadline - time.monotonic() - 3)
            if tr < 5:
                break

            if round_i % 2 == 0:
                f_r, _ = lp_optimize(best_seq, time_limit=tr * 0.7, aggressive=True)
                f_r, _ = lp_optimize(f_r, time_limit=tr * 0.3)
            else:
                f_r, _ = mirror_descent_smoothed(best_seq, time_limit=tr * 0.5,
                                                 t_init=40.0, t_final=1.5, eta_init=0.25)
                f_r, _ = lp_optimize(f_r, time_limit=tr * 0.4, aggressive=True)

            sc_r = float(evaluate_sequence(f_r.tolist()))
            if np.isfinite(sc_r) and sc_r < best_score:
                best_score = sc_r
                best_seq = f_r.copy()
                print(f"  [P4-{round_i}] {sc_r:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.8f}")
    return [float(x) for x in best_seq]
