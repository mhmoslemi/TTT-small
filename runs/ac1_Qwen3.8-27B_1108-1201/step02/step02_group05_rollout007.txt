def lp_direction(f, M, n_constraints=None):
    """Find direction g >= 0 maximizing sum(g) subject to conv(f,g) <= M."""
    n = len(f)
    # conv(f, g)[k] = sum_i f[i] * g[k-i] for valid i
    # This is linear in g!
    # conv(f,g)[k] = sum_{i: 0<=k-i<n} f[i] * g[k-i]
    #               = sum_{j: 0<=j<n, 0<=k-j<=n-1} f[k-j] * g[j]
    # Wait, let me be more careful.
    # conv(f,g)[k] = sum_{i} f[i] * g[k-i] where 0<=i<n and 0<=k-i<n
    #              = sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} f[i] * g[k-i]
    
    # For the constraint matrix, row k, column j:
    # coefficient of g[j] in conv(f,g)[k] is f[k-j] if 0 <= k-j < n and 0 <= j < n, else 0
    
    num_constraints = 2 * n - 1
    if n_constraints is not None:
        # Only use the most binding constraints
        A = np.zeros((n_constraints, n))
        for k in range(n_constraints):  # k is the constraint index (which lag to include)
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    A[k, j] = f[i]
    else:
        A = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    A[k, j] = f[i]
    
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()

def annealing_refine(a0, time_limit):
    """Simulated annealing to escape local minima."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    n = len(a)
    best_a = a.copy()
    best_obj = obj_norm(a)
    current_obj = best_obj
    
    t0 = time.monotonic()
    T_init = 0.01
    T_final = 1e-6
    cooling = 0.999
    
    elapsed = time.monotonic() - t0
    total_time = time_limit
    T = T_init * (T_final / T_init) ** (elapsed / total_time) if total_time > 0 else T_final
    
    iterations = 0
    while time.monotonic() - t0 < time_limit:
        # Random perturbation: pick two random indices, swap some mass
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = np.random.uniform(-0.05, 0.05) * a[i]
        a_trial = a.copy()
        a_trial[i] -= delta
        a_trial[j] += delta
        a_trial = np.maximum(a_trial, 0.0)
        ts = a_trial.sum()
        if ts < 1e-12:
            continue
        a_trial /= ts
        obj_trial = obj_norm(a_trial)
        
        iterations += 1
        # Update temperature
        T = T_init * (T_final / T_init) ** ((time.monotonic() - t0) / total_time)
        
        if obj_trial < current_obj:
            a = a_trial
            current_obj = obj_trial
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
        else:
            # Accept with probability exp(-delta/T)
            delta = obj_trial - current_obj
            if delta / T < 50:  # avoid overflow
                if np.random.uniform() < np.exp(-delta / T):
                    a = a_trial
                    current_obj = obj_trial
    
    return best_a, best_obj

Combines subgradient descent, LP-based direction finding, peak shaving,
simulated annealing, coarse-to-fine scaling, and structured asymmetric

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective: 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if v.sum() < 1e-15:
        return np.full(n, target_sum / n)
    # Binary search for threshold tau such that sum(max(v - tau, 0)) = target_sum
    lo = v.min() - target_sum
    hi = v.max()
    for _ in range(80):
        tau = (lo + hi) / 2.0
        x = np.maximum(v - tau, 0.0)
        s = x.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    x = np.maximum(v - hi, 0.0)
    s = x.sum()
    if s > 1e-12:
        x *= target_sum / s
    else:
        x = np.full(n, target_sum / n)
    return x

def build_lp_matrix(f):
    """Build the constraint matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)

        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A

            break

        # Select constraints: top peaks only (for speed on large n)
        num_constraints = 2 * n - 1
        if top_k_frac < 1.0:
            num_to_use = max(n, int(num_constraints * top_k_frac))
            peak_order = np.argsort(conv)[::-1]
            selected_k = peak_order[:num_to_use]
        else:
            selected_k = np.arange(num_constraints)

        # Build constraint matrix
        A = np.zeros((len(selected_k), n), dtype=np.float64)
        for idx, k in enumerate(selected_k):
            k = int(k)
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A[idx, j] = f[k - j]

        b = np.full(len(selected_k), M)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n

        try:
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if not res.success:
                no_improve += 1
                if no_improve >= 10:
                    break
                continue
        except Exception:
            break

        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-12:
            break
        g /= gs

        # Line search: find best t in [0, 1]
        # Try a few values
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_trial = (1.0 - t) * f + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ts = a_trial.sum()
            if ts < 1e-12:
                continue
            a_trial /= ts
            o = obj_norm(a_trial)
            if o < best_t_obj:
                best_t_obj = o
                best_t = t

        if best_t > 0 and best_t_obj < best_obj - 1e-10:
            f_new = (1.0 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0.0)
            f_new /= f_new.sum()
            f = f_new
            best_obj = best_t_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            break

    return best_f, best_obj


def peak_shaving(a0, time_limit):
    """
    Targeted local search: identify top peaks of convolution and
    redistribute mass from high-contributing indices to low-contributing ones.
    Considers impact on ALL top peaks simultaneously.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    n = len(a)
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 800

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        # Top 8 peaks
        top_k = np.argsort(conv)[-8:]

        # Precompute contributions for all top peaks
        contrib_matrix = np.zeros((len(top_k), n))
        for idx, k in enumerate(top_k):
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contrib_matrix[idx, i] = a[i] * a[j]

        # Danger: high total contribution across all peaks
        total_contrib = contrib_matrix.sum(axis=0)
        # Safe: low total contribution
        danger_threshold = np.percentile(total_contrib, 80)
        safe_threshold = np.percentile(total_contrib, 20)
        danger = np.where(total_contrib > danger_threshold)[0]
        safe = np.where(total_contrib < safe_threshold)[0]

        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue

        found = False
        for di in danger[:20]:
            if found:
                break
            for si in safe[:20]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-11:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break

        if not found:
            stuck += 1
        else:
            stuck = 0

    return a, best_obj


def annealing_refine(a0, time_limit, rng):
    """Simulated annealing to escape local minima."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    n = len(a)
    best_a = a.copy()
    best_obj = obj_norm(a)
    current_obj = best_obj

    t0 = time.monotonic()
    total_time = max(1.0, time_limit)
    T_init = 0.005
    T_final = 1e-7

    while time.monotonic() - t0 < time_limit:
        # Temperature schedule
        frac = (time.monotonic() - t0) / total_time
        T = T_init * (T_final / T_init) ** min(1.0, frac)

        # Random perturbation
        op = rng.integers(0, 3)
        a_trial = a.copy()
        if op == 0:
            # Swap mass between two indices
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                delta = rng.uniform(-0.1, 0.1) * a[i]
                a_trial[i] -= delta
                a_trial[j] += delta
        elif op == 1:
            # Scale a random subset
            mask = rng.random(n) < 0.3
            if mask.sum() > 0:
                a_trial[mask] *= rng.uniform(0.5, 1.5)
        else:
            # Gaussian perturbation
            a_trial += rng.normal(0, 0.01, n)

        a_trial = np.maximum(a_trial, 0.0)
        ts = a_trial.sum()
        if ts < 1e-12:
            continue
        a_trial /= ts
        obj_trial = obj_norm(a_trial)

        if obj_trial < current_obj:
            a = a_trial
            current_obj = obj_trial
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
        else:
            delta = obj_trial - current_obj
            if delta / max(T, 1e-10) < 50:
                if rng.random() < np.exp(-delta / max(T, 1e-10)):
                    a = a_trial
                    current_obj = obj_trial

    return best_a, best_obj


def coarse_to_fine(parent, target_n, time_per_level, rng):
    """Coarse-to-fine: optimize at small n, upsample, repeat."""
    if parent is None:
        a = np.ones(16) / 16
    else:
        # Downsample parent to 16
        if parent.size >= 16:
            idx = np.linspace(0, parent.size - 1, 16).astype(int)
            a = parent[idx].copy()
        else:
            a = parent.copy()
            while len(a) < 16:
                a = np.concatenate([a, a])[:16]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(16) / 16
        else:
            a /= s

    levels = []
    n = 16
    while n < target_n:
        n = min(n * 2, target_n)
        levels.append(n)

    for level_n in levels:
        if len(a) < level_n:
            # Upsample via interpolation
            a_up = np.interp(np.linspace(0, 1, level_n),
                             np.linspace(0, 1, len(a)), a)
            a_up = np.maximum(a_up, 0)
            s = a_up.sum()
            if s > 0:
                a = a_up / s
            else:
                a = np.ones(level_n) / level_n
        elif len(a) > level_n:
            a = a[:level_n].copy()
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0:
                a /= s

        # Optimize at this level
        t0 = time.monotonic()
        a, _ = lp_optimize(a, time_limit=min(time_per_level, 3.0 * level_n / level_n))
        a, _ = peak_shaving(a, time_limit=min(time_per_level, 2.0))

    # Final resize to target_n
    if len(a) != target_n:
        a = np.interp(np.linspace(0, 1, target_n),
                      np.linspace(0, 1, len(a)), a)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a /= s

    return a


def make_initializations(n, rng, parent):
    """Generate diverse initializations for sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    # Asymmetric Gaussians
    for center in [0.25, 0.35, 0.45, 0.55, 0.65, 0.75]:
        for width in [0.15, 0.25, 0.35]:
            g = np.exp(-0.5 * ((x - center * n) / (width * n)) ** 2)
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-10:
                inits[f'gauss_{center}_{width}'] = g / s

    # Dual bumps (asymmetric)
    for c1, c2, w1, w2 in [(0.25, 0.75, 0.1, 0.15), (0.2, 0.6, 0.1, 0.2),
                            (0.3, 0.8, 0.15, 0.1), (0.15, 0.55, 0.08, 0.2)]:
        g = np.exp(-0.5 * ((x - c1 * n) / (w1 * n)) ** 2) + \
            np.exp(-0.5 * ((x - c2 * n) / (w2 * n)) ** 2)
        g = np.maximum(g, 0)
        s = g.sum()
        if s > 1e-10:
            inits[f'dual_{c1}_{c2}'] = g / s

    # Edge concentrated
    m = max(1, n // 4)
    edges = np.zeros(n)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    # Asymmetric edges (more mass on one side)
    edges_asym = np.zeros(n)
    edges_asym[:n // 3] = 2.0
    edges_asym[-n // 6:] = 1.0
    inits['edges_asym'] = edges_asym / edges_asym.sum()

    # One-sided
    one = np.zeros(n)
    one[:n // 2 + 1] = 1.0
    inits['one_sided'] = one / one.sum()

    # Cosine window (asymmetric by shifting)
    for shift in [0.3, 0.5, 0.7]:
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x / n - shift)))
        w = np.maximum(w, 0)
        s = w.sum()
        if s > 1e-10:
            inits[f'cos_shift_{shift}'] = w / s

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Parent (resized)
    if parent is not None:
        if parent.size >= n:
            pc = parent[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        pc = np.maximum(pc, 0)
        s = pc.sum()
        if s > 1e-10:
            inits['parent'] = pc / s

    # Perturbed parent
    if parent is not None:
        if parent.size >= n:
            pc = parent[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        pc = np.maximum(pc, 0)
        pc += rng.normal(0, 0.02, n)
        pc = np.maximum(pc, 0)
        s = pc.sum()
        if s > 1e-10:
            inits['parent_perturbed'] = pc / s

    return inits
