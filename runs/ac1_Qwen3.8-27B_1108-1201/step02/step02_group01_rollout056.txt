```python
"""
Hybrid optimization combining LP-based refinement, QP lag minimization,
smooth gradient methods, and random restarts for minimizing the normalized
peak autocorrelation of non-negative step functions.

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def build_lp_matrix(a):
    """Build the convolution constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve_multi(a, M, top_k=1):
    """
    Solve LP: maximize sum(g) subject to conv(a,g)[k] <= M for top_k lags, g >= 0.
    """
    n = len(a)
    A = build_lp_matrix(a)
    conv_a = fft_convolve(a, a)
    
    # Select top-k lags
    top_indices = np.argsort(conv_a)[-top_k:]
    
    A_sub = A[top_indices, :]
    b_sub = np.full(len(top_indices), M)
    
    c = -np.ones(n)
    res = linprog(c, A_ub=A_sub, b_ub=b_sub,
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)


def lp_refine_improved(a, time_limit, verbose=False, top_k=3):
    """

    best_conv_max = np.max(fft_convolve(a, a))
    no_improve = 0
    
    for iteration in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        g = lp_solve_multi(a, M, top_k=top_k)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        c = normalize_seq(g, S)
        
        # Adaptive line search: try multiple t values
        best_t = 0.0
        best_new_conv = M
        t_candidates = np.linspace(0.01, 0.5, 20)
        for t_try in t_candidates:
            b = (1 - t_try) * a + t_try * c
            b = np.clip(b, 0.0, None)
            b = normalize_seq(b, S)
            new_conv_max = np.max(fft_convolve(b, b))
            if new_conv_max < best_new_conv:
```

Let me also think about the QP more carefully.

For lag k*, minimize:
f(a) = sum_{i=0}^{n-1-k*} a[i] * a[i+k*]
subject to: sum(a) = S, a >= 0

The gradient: grad[i] = a[i+k*] (if i+k* < n) + a[i-k*] (if i-k* >= 0)
The Hessian: H[i,j] = 1 if j = i+k* or j = i-k* (for valid indices), 0 otherwise.

This Hessian is a "bidiagonal" matrix with bandwidth k*. It's PSD (it's the Gram matrix of a set of vectors).

So this is a convex QP! We can solve it with:
- Projected gradient descent
- scipy.optimize.minimize with L-BFGS-B
- Or a specialized solver for banded QPs

For the projected gradient:
1. Compute gradient: grad[i] = a[i+k*] + a[i-k*] (with bounds)
2. Step: a_new = a - lr * grad
3. Project: a_new = max(a_new, 0), then normalize to sum = S

This is simple and fast! Let me implement this.

```
for each iteration:
    compute all autocorrelation values

    do several projected gradient steps to minimize lag k*
```

This is like a coordinate descent over lags. It should work well in practice.

```python
"""
Hybrid optimization: LP-based refinement + alternating QP lag minimization + 
smooth gradient + random restarts for minimizing normalized peak autocorrelation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def build_lp_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve(a, M, top_k=1):
    n = len(a)
    A = build_lp_matrix(a)
    conv_a = fft_convolve(a, a)
    top_indices = np.argsort(conv_a)[-top_k:]
    A_sub = A[top_indices, :]
    b_sub = np.full(len(top_indices), M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_sub, b_ub=b_sub,
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)


def lp_refine(a, time_limit, verbose=False, top_k=1):
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    no_improve = 0
    
    for iteration in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        g = lp_solve(a, M, top_k=top_k)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        c = normalize_seq(g, S)
        
        best_t = 0.0
        best_new_conv = M
        for t_try in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            b = (1 - t_try) * a + t_try * c
            b = np.clip(b, 0.0, None)
            b = normalize_seq(b, S)
            new_cm = np.max(fft_convolve(b, b))
            if new_cm < best_new_conv:
                best_new_conv = new_cm
                best_t = t_try
        
        if best_t < 0.005:
            no_improve += 1
            if no_improve > 30:
                break
            # Try smaller top_k
            top_k = max(1, top_k - 1)
            continue
        
        a = (1 - best_t) * a + best_t * c
        a = np.clip(a, 0.0, None)
        a = normalize_seq(a, S)
        
        new_cm = np.max(fft_convolve(a, a))
        if new_cm < best_conv_max - 1e-12:
            best_conv_max = new_cm
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and iteration % 500 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    LP iter {iteration}: conv_max={best_conv_max:.6f}, score={score:.6f}")
    
    return best_a


def alm_minimize(a, time_limit, verbose=False, max_steps=500):
    """Alternating Lag Minimization: alternately minimize the peak lag using projected gradient."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    
    lr = 1.0 / (2.0 * n)  # Initial learning rate
    step = 0
    
    while time.monotonic() - t0 < time_limit and step < max_steps:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        
        # Convert peak_idx to lag k*
        # conv has length 2n-1, indexed 0 to 2n-2
        # Lag k corresponds to index k + (n-1) in the symmetric representation
        # But in our 0-indexed conv, lag 0 is at index n-1
        # conv[k] for k in 0..2n-2 represents the (k-(n-1))-th lag
        lag_k = peak_idx - (n - 1)
        
        # Compute gradient for this lag
        # d conv[peak_idx] / d a[j] = a[j - lag_k] (if 0 <= j - lag_k < n) 
        #                         + a[peak_idx - j] (if 0 <= peak_idx - j < n)
        # Actually, let me use the formula:
        # conv(a,a)[k] = sum_i a[i] * a[k-i] for i where both indices valid
        # d/d a[j] = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n)
        # But for j = k-j (i.e., 2j = k), it's just a[k-j] (not doubled)
        
        k = peak_idx
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            i1 = k - j
            if 0 <= i1 < n:
                g += a[i1]
            i2 = j - k
            if 0 <= i2 < n:
                g += a[i2]
            grad[j] = g
        
        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)
        
        new_cm = np.max(fft_convolve(a_new, a_new))
        if new_cm < best_conv_max:
            best_conv_max = new_cm
            best_a = a_new.copy()
            a = a_new
        elif new_cm < np.max(conv) * 1.01:
            a = a_new
        else:
            lr *= 0.9
        
        # Periodically adjust learning rate
        if step % 50 == 0:
            lr = np.clip(lr * 1.05, 1.0/(10*n), 1.0/n)
        
        step += 1
        
        if verbose and step % 100 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    ALM step {step}: score={score:.6f}, lr={lr:.5f}")
    
    return best_a
```

```python
"""
Hybrid optimization combining LP-based refinement, Alternating Lag Minimization (ALM),
smooth gradient methods, and random restarts for minimizing the normalized peak
autocorrelation of non-negative step functions.

Key innovations over previous approach:
- ALM: Directly minimizes the peak lag using projected gradient with the simple
  structure of the autocorrelation Hessian (reversal matrix), with momentum.
- Top-k LP constraints to encourage flatter autocorrelation.
- Adaptive line search and multi-phase optimization.
- Random restarts with re-optimization to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def build_lp_matrix(a):
    """Build the convolution constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve(a, M, top_k=1):
    """Solve LP: maximize sum(g) subject to top-k autocorrelation constraints, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    conv_a = fft_convolve(a, a)
    top_indices = np.argsort(conv_a)[-top_k:]
    A_sub = A[top_indices, :]
    b_sub = np.full(len(top_indices), M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_sub, b_ub=b_sub,
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)


def lp_refine(a, time_limit, verbose=False, top_k=1):
    """LP-based iterative refinement with top-k constraint and adaptive line search."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    no_improve = 0

    for iteration in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        g = lp_solve(a, M, top_k=top_k)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        c = normalize_seq(g, S)

        best_t = 0.0
        best_new_conv = M
        for t_try in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            b = (1 - t_try) * a + t_try * c
            b = np.clip(b, 0.0, None)
            b = normalize_seq(b, S)
            new_cm = np.max(fft_convolve(b, b))
            if new_cm < best_new_conv:
                best_new_conv = new_cm
                best_t = t_try

        if best_t < 0.005:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        a = (1 - best_t) * a + best_t * c
        a = np.clip(a, 0.0, None)
        a = normalize_seq(a, S)

        new_cm = np.max(fft_convolve(a, a))
        if new_cm < best_conv_max - 1e-12:
            best_conv_max = new_cm
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and iteration % 500 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    LP iter {iteration}: conv_max={best_conv_max:.6f}, score={score:.6f}")

    return best_a


def compute_lag_gradient(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a. Result: grad[j] = 2*a[k-j] if valid."""
    n = len(a)
    grad = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    if j_min <= j_max:
        j_range = np.arange(j_min, j_max + 1)
        grad[j_range] = 2.0 * a[k - j_range]
    return grad


def alm_minimize(a, time_limit, verbose=False, use_momentum=True):
    """
    Alternating Lag Minimization: alternately minimize the peak autocorrelation lag
    using projected gradient descent with momentum.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))

    lr = 0.4
    beta = 0.7
    v = np.zeros(n)
    step = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))

        grad = compute_lag_gradient(a, peak_idx)

        if use_momentum:
            v = beta * v + (1 - beta) * grad
            a_new = a - lr * v
        else:
            a_new = a - lr * grad

        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)

        new_cm = np.max(fft_convolve(a_new, a_new))
        if new_cm < best_conv_max:
            best_conv_max = new_cm
            best_a = a_new.copy()
            a = a_new
            stall = 0
        elif new_cm < conv[peak_idx] * 1.001:
            a = a_new
            stall += 1
        else:
            stall += 1
            lr = max(lr * 0.95, 0.05)

        # Adaptive learning rate
        if step % 100 == 0:
            lr = np.clip(lr * 1.02, 0.05, 0.6)
            beta = np.clip(beta + 0.001, 0.5, 0.9)

        if stall > 200:
            lr *= 0.7
            stall = 0

        step += 1

        if verbose and step % 500 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    ALM step {step}: score={score:.6f}, lr={lr:.4f}")

    return best_a


def smooth_gradient_optimize(a, time_limit, verbose=False, tau=0.05):
    """
    Gradient descent on the smoothed objective logsumexp(conv/tau).
    Uses FFT-based gradient computation.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = local_score(a)

    lr = 0.3
    step = 0
    v = np.zeros(n)
    beta = 0.8

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)

        # Compute softmax weights
        max_conv = np.max(conv)
        exp_val = np.exp((conv - max_conv) / tau)
        w = exp_val / np.sum(exp_val)

        # Compute gradient using cross-correlation
        # grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
        # This is a cross-correlation of w and a
        size = 1
        while size < 2 * n - 1 + n:
            size <<= 1
        W = np.fft.rfft(w, size)
        A = np.fft.rfft(a, size)
        corr = np.fft.irfft(W * np.conj(A), size)

        # The cross-correlation: corr[j] = sum_k w[k] * conj(a[k-j])
        # For real signals: corr[j] = sum_k w[k] * a[k-j]
        # We need: grad[j] = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
        # Let m = k-j: sum_m w[m+j] * a[m] = cross-corr at shift j
        # In our FFT convention: ifft(fft(w) * conj(fft(a)))[j] = sum_k w[k] * a[k-j]
        # So grad = 2 * corr[0:n] (approximately, with some boundary effects)

        grad = 2.0 * corr[:n]

        # Momentum step
        v = beta * v + (1 - beta) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)

        new_score = local_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new

        step += 1

        if verbose and step % 200 == 0:
            print(f"    SmoothGrad step {step}: score={best_score:.6f}, lr={lr:.4f}")

        # Adaptive
        if step % 100 == 0:
            lr = np.clip(lr * 1.01, 0.05, 0.5)

    return best_a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def random_perturb(a, sigma, rng):
    """Perturb a sequence with multiplicative noise."""
    perturbed = a * np.exp(rng.normal(0, sigma, len(a)))
    perturbed = np.clip(perturbed, 0.001, None)
    return normalize_seq(perturbed, np.sqrt(2.0 * len(perturbed)))


def block_perturb(a, rng, n_blocks=5, sigma=0.1):
    """Perturb random blocks of the sequence."""
    n = len(a)
    perturbed = a.copy()
    for _ in range(n_blocks):
        width = max(1, n // (n_blocks * 3))
        start = int(rng.integers(0, max(1, n - width)))
        perturbed[start:start + width] *= np.exp(rng.normal(0, sigma, size=width))
    perturbed = np.clip(perturbed, 0.001, None)
    return normalize_seq(perturbed, np.sqrt(2.0 * n))

            for _ in range(5):
                if time.monotonic() > deadline - 5:
                    break
                perturbed = random_perturb(best_seq, 0.02, rng)
                result = lp_refine(perturbed, min(remaining * 0.3, 30), top_k=1)
                s = local_score(result)
                if s < best_score:
                    best_score = s
                    best_seq = result.copy()
                    print(f"  Light perturbation improvement: score={best_score:.6f}")

    # Final verification and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local_score={best_score:.6f} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
```
