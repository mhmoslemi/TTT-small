    grad_max = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            idx = np.arange(lo, hi + 1)
            grad_max[idx] += 2.0 * a[k - idx]
    grad_max /= len(peak_indices)

    # Gradient of score = 2n * max_c / s^2
    # d(score)/da[i] = 2n * (grad_max[i] * s - 2*max_c*a[i]) / s^3
    score_grad = 2.0 * n * (grad_max * s - 2.0 * max_c * a) / (s ** 3)
    return score, score_grad


def adam_optimize(a0, time_limit, rng, lr_init=0.05, verbose=False):
    """Adam optimizer with adaptive learning rate."""
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n)
        s = float(n)
    a = a / s * np.sqrt(2.0 * n)

    score, _ = score_and_subgrad(a)
    best_a = a.copy()
    best_score = score

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        sc, grad = score_and_subgrad(a)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.clip(a_new, 0.0, 1000.0)

        # Renormalize to maintain scale
        s_new = a_new.sum()
        if s_new < 0.01:
            a_new = a.copy()
        else:
            a_new = a_new / s_new * s

        new_score = fast_score(a_new)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        score = new_score

        # Adaptive LR
        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
        elif t % 500 == 0 and lr < 0.5:
            lr = min(lr * 1.05, 0.3)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t}, score={score:.6f}, best={best_score:.6f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def nesterov_subgrad(a0, time_limit, rng, verbose=False):
    """Nesterov accelerated subgradient method."""
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n)
    a = a / a.sum() * np.sqrt(2.0 * n)

    y = a.copy()
    beta = 1.0
    lr = 0.02
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        sc, grad = score_and_subgrad(y)
        a_new = np.clip(y - lr * grad, 0.0, 1000.0)
        s_new = a_new.sum()
        if s_new > 0.01:
            a_new = a_new / s_new * s

        beta_new = (1 + np.sqrt(1 + 4 * beta * beta)) / 2
        y = a_new + (beta / beta_new - 1) * (a_new - a)
        y = np.clip(y, 0.0, 1000.0)
        sy = y.sum()
        if sy > 0.01:
            y = y / sy * s

        a = a_new
        beta = beta_new
        lr = max(0.001, lr * 0.9999)

        new_score = fast_score(a)
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSG t={t}, score={new_score:.6f}, best={best_score:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def simulated_annealing(a0, time_limit, rng, verbose=False):
    """Simulated annealing for final polish."""
    a = a0.astype(np.float64).copy()
    score = fast_score(a)
    best_a = a.copy()
    best_score = score
    n = len(a)
    t0 = time.monotonic()
    temp = 0.001
    cooling = 0.9998
    iter_count = 0

    while time.monotonic() - t0 < time_limit and temp > 1e-10:
        iter_count += 1
        a_new = a.copy()
        r = rng.random()

        if r < 0.35:
            # Global Gaussian
            a_new = a * np.exp(rng.normal(0, 0.012, n))
        elif r < 0.60:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            a_new[start:start + width] *= np.exp(rng.normal(0, 0.025, width))
        elif r < 0.80:
            # Targeted peak reduction
            conv = fft_autocorr(a)
            peak_idx = int(np.argmax(conv))
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            idx_range = np.arange(lo, hi + 1)
            contribs = a[idx_range] * a[peak_idx - idx_range]
            top_k = max(3, len(contribs) // 10)
            top_local = np.argsort(contribs)[-top_k:]
            top_global = idx_range[top_local]
            a_new[top_global] *= np.exp(rng.normal(-0.02, 0.008, len(top_global)))
            other = np.setdiff1d(np.arange(n), top_global)
            if len(other) > 3:
                n_inc = min(8, len(other) // 3)
                inc_idx = rng.choice(other, n_inc, replace=False)
                a_new[inc_idx] *= np.exp(rng.normal(0.015, 0.008, n_inc))
        elif r < 0.90:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.12, 0.12) * max(a[i], 0.01)
                a_new[i] = max(0.0, a[i] - delta)
                a_new[j] = max(0.0, a[j] + delta)
        else:
            # Multi-point
            for _ in range(4):
                i = int(rng.integers(0, n))
                a_new[i] *= np.exp(rng.normal(0, 0.01))

        a_new = np.clip(a_new, 0.0, 1000.0)
        new_score = fast_score(a_new)

        if new_score < score or rng.random() < np.exp(-(new_score - score) / max(temp, 1e-12)):
            a = a_new
            score = new_score
            if score < best_score:
                best_score = score
                best_a = a.copy()

        temp *= cooling

        if verbose and iter_count % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA iter={iter_count}, score={score:.6f}, best={best_score:.6f}, temp={temp:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, rng, verbose=False):
    """Cyclic coordinate descent with line search."""
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    score = fast_score(a)
    best_a = a.copy()
    best_score = score
    t0 = time.monotonic()
    passed = 0

    while time.monotonic() - t0 < time_limit:
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            # Try several step sizes
            for frac in [0.5, 0.1, 0.01, -0.01, -0.1]:
                a_new = a.copy()
                a_new[i] = max(0.0, a[i] * (1.0 + frac))
                new_score = fast_score(a_new)
                if new_score < score:
                    a = a_new
                    score = new_score
                    break

        passed += 1
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and passed % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={passed}, score={score:.6f}, best={best_score:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def slsqp_optimize(a0, max_iter=200):
    """SLSQP optimization for final polish."""
    a0 = np.clip(a0, 0.0, 1000.0)
    n = len(a0)

    def objective(x):
        return fast_score(x)

    def objective_grad(x):
        _, g = score_and_subgrad(x)
        return g

    bounds = [(0.0, 1000.0)] * n
    result = minimize(objective, a0, method='SLSQP', jac=objective_grad,
                      bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12})

    if result.x is not None:
        best_a = np.clip(result.x, 0.0, 1000.0)
        best_score = fast_score(best_a)
        return best_a, best_score
    return a0, fast_score(a0)


def make_chirp(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone(n, rng):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 8):
        amp = 0.2 / k
        freq = rng.uniform(0.3, 3.0) * k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_smooth_bump(n, rng, center_frac=0.5, width_frac=0.3):
    """Gaussian bump initialization."""
    t = np.arange(n, dtype=np.float64) / n
    center = center_frac + rng.uniform(-0.1, 0.1)
    width = width_frac + rng.uniform(-0.05, 0.05)
    a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length via interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimizer: Adam, Nesterov subgradient, SA, coordinate descent, SLSQP.
    Starts from multiple diverse initializations and refines the best.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            if len(start_seq) > 0:
                s = fast_score(start_seq)
                print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
                if s < best_score:
                    best_score = s
                    best_seq = start_seq.copy()
        except Exception as e:
            print(f"Failed to load provided sequence: {e}")

    # Check uniform baseline
    for n_test in [128, 256, 384, 512, 768, 1024]:
        s = fast_score(np.ones(n_test))
        if s < best_score:
            best_score = s
            best_seq = np.ones(n_test)
    print(f"Baseline (uniform): score={best_score:.6f}")

    # Build initialization pool
    inits = []
    if best_seq is not None and len(best_seq) <= 2048:
        inits.append(('provided', best_seq.copy()))

    n_main = start_seq.shape[0] if start_seq is not None else 512

    # Diverse initializations at the main length
    inits.append(('chirp1', make_chirp(n_main, 0.4, rng)))
    inits.append(('chirp2', make_chirp(n_main, 0.7, rng)))
    inits.append(('multitone1', make_multitone(n_main, rng)))
    inits.append(('multitone2', make_multitone(n_main, rng)))
    inits.append(('bump1', make_smooth_bump(n_main, rng)))
    inits.append(('bump2', make_smooth_bump(n_main, rng, center_frac=0.3)))
    inits.append(('random1', np.clip(1.0 + 0.4 * rng.standard_normal(n_main), 0.05, None)))
    inits.append(('random2', np.clip(1.0 + 0.6 * rng.standard_normal(n_main), 0.05, None)))

    # Different lengths
    for n_alt in [256, 384, 768]:
        if n_alt != n_main:
            inits.append((f'chirp_n{n_alt}', make_chirp(n_alt, 0.5, rng)))
            inits.append((f'random_n{n_alt}', np.clip(1.0 + 0.3 * rng.standard_normal(n_alt), 0.05, None)))

    print(f"Total initializations: {len(inits)}")

    # Phase 1: Quick Adam on all inits (short time each)
    print("\n=== Phase 1: Quick Adam exploration ===")
    t1_total = time_budget * 0.35
    t_per_init = t1_total / len(inits)

    for label, a_init in inits:
        if time.monotonic() > deadline - 300:
            break
        t_alloc = max(5.0, min(t_per_init, deadline - time.monotonic() - 300))
        a_best, s_best = adam_optimize(a_init, t_alloc, rng, lr_init=0.03)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  {label}: score={s_best:.6f} *** NEW BEST ***")
        else:
            print(f"  {label}: score={s_best:.6f}")

    # Phase 2: Focused Adam on top candidates (longer time)
    print(f"\n=== Phase 2: Focused Adam (best so far: {best_score:.6f}) ===")
    t2_total = time_budget * 0.25
    t_per_focus = max(30.0, t2_total / 3)

    # Resample best to a few lengths and optimize
    focus_inits = [best_seq.copy()]
    for n_res in [n_main, n_main // 2, n_main * 2]:
        if 32 <= n_res <= 4096 and n_res != n_main:
            focus_inits.append(resample_sequence(best_seq, n_res))

    for fi, a_f in enumerate(focus_inits):
        if time.monotonic() > deadline - 200:
            break
        t_alloc = max(10.0, min(t_per_focus, deadline - time.monotonic() - 200))
        a_best, s_best = adam_optimize(a_f, t_alloc, rng, lr_init=0.02)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  Focus {fi} (n={len(a_short if 'a_short' in dir() else a_best)}): score={s_best:.6f} *** NEW BEST ***")
        else:
            print(f"  Focus {fi}: score={s_best:.6f}")

    # Phase 3: Nesterov subgradient on best
    print(f"\n=== Phase 3: Nesterov subgradient (best: {best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        t3 = min(200.0, max(30.0, deadline - time.monotonic() - 150))
        a_best, s_best = nesterov_subgrad(best_seq, t3, rng, verbose=True)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  After NSG: score={s_best:.6f} *** NEW BEST ***")
        else:
            print(f"  After NSG: score={s_best:.6f} (no improvement)")

    # Phase 4: Simulated annealing
    print(f"\n=== Phase 4: Simulated Annealing (best: {best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        t4 = min(300.0, max(30.0, deadline - time.monotonic() - 100))
        a_best, s_best = simulated_annealing(best_seq, t4, rng, verbose=True)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  After SA: score={s_best:.6f} *** NEW BEST ***")
        else:
            print(f"  After SA: score={s_best:.6f} (no improvement)")

    # Phase 5: SLSQP polish
    print(f"\n=== Phase 5: SLSQP polish (best: {best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 20 and len(best_seq) <= 1024:
        try:
            a_best, s_best = slsqp_optimize(best_seq, max_iter=300)
            if s_best < best_score:
                best_score = s_best
                best_seq = a_best.copy()
                print(f"  After SLSQP: score={s_best:.6f} *** NEW BEST ***")
            else:
                print(f"  After SLSQP: score={s_best:.6f}")
        except Exception as e:
            print(f"  SLSQP error: {e}")

    # Phase 6: Final Adam micro-polish
    print(f"\n=== Phase 6: Final Adam polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        t6 = max(5.0, deadline - time.monotonic() - 8)
        a_best, s_best = adam_optimize(best_seq, t6, rng, lr_init=0.01)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  Final: score={s_best:.6f} *** NEW BEST ***")
        else:
            print(f"  Final: score={s_best:.6f}")

    # Output
    elapsed = time.monotonic() - t_global
    print(f"\n=== SUMMARY: elapsed={elapsed:.1f}s ===")
    print(f"Best score: {best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Verify with the actual evaluate function
        try:
            verified = evaluate_sequence(final_list)
            print(f"Verified score: {verified:.6f}")
            if verified != best_score:
                print(f"  (local={best_score:.6f}, verified={verified:.6f})")
        except Exception as e:
            print(f"Verification error: {e}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
