"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.
Uses analytical gradient (soft-max) with Adam, multi-scale initialization,
coordinate descent refinement, and diverse starting points to beat previous best.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def softmax(theta):
    """Numerically stable softmax."""
    shifted = theta - np.max(theta)
    e = np.exp(shifted)
    return e / np.sum(e)


def compute_grad(a, tau):
    """
    Compute soft-max score and gradient w.r.t. theta (where a = softmax(theta)).
    Returns (soft_score, grad_theta, exact_score).
    """
    n = len(a)
    conv = fft_convolve(a, a)
    shifted = (conv - np.max(conv)) / tau
    exp_conv = np.exp(shifted)
    p = exp_conv / np.sum(exp_conv)
    soft_max = np.dot(p, conv)
    soft_score = 2.0 * n * soft_max

    # g[i] = sum_j a[j] * p[i+j] = cross-correlation of p and a (valid mode)
    g = np.correlate(p, a, mode='valid')

    inner = np.dot(g, a)  # should equal soft_max
    grad_theta = 4.0 * n * a * (g - inner)

    exact_score = 2.0 * n * np.max(conv)
    return soft_score, grad_theta, exact_score, p, conv


def adam_optimize(theta_init, time_limit, tau_start, tau_end, base_lr=1.0,
                  verbose=False, seed=0):
    """Adam optimizer with annealed soft-max tau."""
    n = len(theta_init)
    theta = theta_init.astype(np.float64).copy()
    theta = theta - np.max(theta)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0

    best_exact_score = np.inf
    best_theta = theta.copy()
    patience_counter = 0
    PATIENCE = 3000

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)

        # Cosine annealing for tau
        tau = tau_end + 0.5 * (tau_start - tau_end) * (1 + np.cos(np.pi * elapsed_frac))

        a = softmax(theta)

        soft_score, grad_theta, exact_score, p, conv = compute_grad(a, tau)

        # Gradient clipping
        grad_norm = np.linalg.norm(grad_theta)
        if grad_norm > 100:
            grad_theta = grad_theta * 100.0 / grad_norm

        # Adam update
        lr = base_lr / (1.0 + 0.05 * t * 0.01)
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * (grad_theta ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Track best by exact score
        if exact_score < best_exact_score - 1e-10:
            best_exact_score = exact_score
            best_theta = theta.copy()
            patience_counter = 0
        else:
            patience_counter += 1

        # Restart if stuck
        if patience_counter > PATIENCE:
            theta = best_theta + 0.1 * rng.standard_normal(n)
            theta = theta - np.max(theta)
            m = np.zeros(n)
            v = np.zeros(n)
            patience_counter = 0

        if verbose and t % 1000 == 0:
            print(f"    Adam t={t}, tau={tau:.5f}, exact={exact_score:.6f}, "
                  f"best={best_exact_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")

    a_best = softmax(best_theta)
    return a_best, best_exact_score


def coordinate_descent(a_init, time_limit, seed=0, verbose=False):
    """Coordinate descent: randomly pick coordinates and try small perturbations."""
    n = len(a_init)
    a = a_init.copy()
    sum_a = np.sum(a)
    best_score = local_score(a)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    step_size_start = 0.05
    step_size_end = 0.005
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        step_size = step_size_start * (step_size_end / step_size_start) ** elapsed_frac

        # Pick a random coordinate
        i = rng.integers(0, n)

        # Try several perturbations
        best_delta = 0.0
        best_local_score = local_score(a)
        old_val = a[i]

        # Try multiplicative changes
        for mult in [1.0 + step_size, 1.0 - step_size,
                     1.0 + 2 * step_size, 1.0 - 2 * step_size,
                     1.0 + 0.5 * step_size, 1.0 - 0.5 * step_size]:
            a[i] = old_val * mult
            a[i] = max(0.0, a[i])
            s = np.sum(a)
            if s < 0.01:
                a[i] = old_val
                continue
            sc = local_score(a)
            if sc < best_local_score:
                best_local_score = sc
                best_delta = a[i] - old_val
            a[i] = old_val

        if best_delta != 0:
            a[i] = old_val + best_delta
            if best_local_score < best_score:
                best_score = best_local_score
                best_a = a.copy()

        iterations += 1
        if verbose and iterations % 2000 == 0:
            print(f"    CD iter={iterations}, score={best_score:.6f}, "
                  f"elapsed={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def sa_refine(a_init, time_limit, seed=0, verbose=False):
    """Simulated annealing refinement with targeted peak reduction."""
    n = len(a_init)
    a = a_init.copy()
    sum_a = np.sum(a)
    best_score = local_score(a)
    best_a = a.copy()
    current_score = best_score
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    iterations = 0

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.4:
            # Target the peak: reduce elements that contribute to the peak lag
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find which elements contribute to the peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(2, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))
            # Boost some non-peak elements
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_boost = min(4, len(other))
                boost_idx = rng.choice(other, size=n_boost, replace=False)
                candidate[boost_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_boost))
        elif r < 0.7:
            # Local window perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.85:
            # Two-window perturbation
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 10))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.005, size=n))

        candidate = np.clip(candidate, 0.0, None)
        csum = np.sum(candidate)
        if csum < 1e-10:
            continue
        candidate = candidate * (sum_a / csum)

        new_score = local_score(candidate)
        iterations += 1

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            a = candidate
            current_score = new_score

        temperature *= cooling

    return best_a, best_score


def make_chirp(n, alpha, rng, beta=0.0):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n + beta)
    if beta != 0:
        a = 1.0 + alpha * np.cos(np.pi * t * t / n + beta)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone(n, rng, n_tones=8):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 2)))
        amp = rng.uniform(0.03, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.015 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_gaussian_bumps(n, rng, n_bumps=6):
    """Gaussian bump initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_bumps):
        center = rng.uniform(0, n)
        width = rng.uniform(n / 10, n / 3)
        amp = rng.uniform(-0.3, 0.3)
        a += amp * np.exp(-(t - center) ** 2 / (2 * width ** 2))
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_interpolated(provided, n, rng):
    """Interpolate provided sequence to new length."""
    if len(provided) == n:
        return provided.astype(np.float64).copy()
    x_old = np.linspace(0, 1, len(provided))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, provided)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Adam gradient descent on multiple (n, init) combos
    Phase 2: Coordinate descent on best candidates
    Phase 3: SA refinement
    Phase 4: Final coordinate descent
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    # === Phase 1: Adam optimization ===
    print("\n" + "=" * 60)
    print("Phase 1: Adam gradient optimization")
    print("=" * 60)

    # Configurations: (n, init_type, time_fraction)
    n_prov = len(provided_seq) if provided_seq is not None else 256
    search_configs = [
        (n_prov, 'provided', 0.20),
        (n_prov, 'chirp1', 0.15),
        (n_prov, 'multitone', 0.10),
        (n_prov, 'chirp2', 0.10),
        (n_prov, 'bumps', 0.10),
        (512, 'chirp1', 0.10),
        (512, 'multitone', 0.08),
        (384, 'chirp1', 0.08),
        (n_prov, 'random', 0.05),
        (512, 'bumps', 0.04),
    ]

    total_frac = sum(c[2] for c in search_configs)
    available_time = deadline - time.monotonic()

    for n, init_name, frac in search_configs:
        if time.monotonic() > deadline - 60:
            break

        time_alloc = max(20.0, frac / total_frac * available_time * 0.88)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 30)
        if time_alloc < 15:
            break

        # Generate initialization
        if init_name == 'provided' and provided_seq is not None:
            a0 = make_interpolated(provided_seq, n, rng)
        elif init_name == 'chirp1':
            alpha = 0.3 + 0.3 * rng.random()
            a0 = make_chirp(n, alpha, rng, 0.0)
        elif init_name == 'chirp2':
            alpha = 0.2 + 0.5 * rng.random()
            a0 = make_chirp(n, alpha, rng, rng.uniform(0, 2 * np.pi))
        elif init_name == 'multitone':
            a0 = make_multitone(n, rng, n_tones=6 + int(rng.integers(0, 6)))
        elif init_name == 'bumps':
            a0 = make_gaussian_bumps(n, rng, n_bumps=4 + int(rng.integers(0, 5)))
        else:
            a0 = make_random(n, rng, 0.2 + 0.4 * rng.random())

        # Convert to log-space for Adam
        a0 = np.clip(a0, 0.01, None)
        a0 = a0 / np.sum(a0)
        theta0 = np.log(a0 + 1e-10)

        print(f"\n  n={n}, init='{init_name}', alloc={time_alloc:.0f}s")
        t_start = time.monotonic()

        # Tau schedule: start soft, end hard
        tau_start = 5.0 / n
        tau_end = 0.1 / n
        base_lr = 0.5 + 0.5 * rng.random()

        f_best, score = adam_optimize(theta0, time_alloc, tau_start, tau_end,
                                       base_lr=base_lr, verbose=False,
                                       seed=seed + n + hash(init_name) % 1000)
        elapsed = time.monotonic() - t_start
        print(f"  Adam done: score={score:.6f}, time={elapsed:.1f}s")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}, init={init_name}) ***")

    # === Phase 2: Coordinate descent on best ===
    print("\n" + "=" * 60)
    print("Phase 2: Coordinate descent refinement")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        cd_time = min(200.0, remaining * 0.3)
        if cd_time > 20:
            print(f"  CD on n={len(best_seq)}, time={cd_time:.0f}s")
            a_cd, score_cd = coordinate_descent(best_seq, cd_time, seed=seed + 100)
            print(f"  CD result: {score_cd:.6f}")
            if score_cd < best_score:
                best_score = score_cd
                best_seq = a_cd.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 3: SA refinement ===
    print("\n" + "=" * 60)
    print("Phase 3: SA refinement")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(250.0, remaining * 0.35)
        if sa_time > 20:
            print(f"  SA on n={len(best_seq)}, time={sa_time:.0f}s")
            a_sa, score_sa = sa_refine(best_seq, sa_time, seed=seed + 200)
            print(f"  SA result: {score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Final quick Adam + CD ===
    print("\n" + "=" * 60)
    print("Phase 4: Final refinement")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        final_time = min(150.0, remaining * 0.7)
        if final_time > 30:
            # Quick Adam on the best
            n = len(best_seq)
            a_norm = best_seq / np.sum(best_seq)
            theta0 = np.log(np.clip(a_norm, 1e-10, None))
            tau_start = 2.0 / n
            tau_end = 0.05 / n

            a_final, score_final = adam_optimize(theta0, final_time * 0.6,
                                                  tau_start, tau_end,
                                                  base_lr=0.3, seed=seed + 300)
            print(f"  Final Adam: {score_final:.6f}")
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()

            # Quick CD
            if time.monotonic() < deadline - 15:
                cd_time2 = min(50.0, deadline - time.monotonic() - 10)
                if cd_time2 > 10:
                    a_cd2, score_cd2 = coordinate_descent(best_seq, cd_time2, seed=seed + 400)
                    print(f"  Final CD: {score_cd2:.6f}")
                    if score_cd2 < best_score:
                        best_score = score_cd2
                        best_seq = a_cd2.copy()

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
