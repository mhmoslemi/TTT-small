from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

4. Project `a_new` onto simplex

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with 
log-sum-exp smoothed gradients, QP subproblem refinement, and multi-phase search.
Uses full-softmax subgradients (all convolution peaks with softmax weights)
for better gradient information than top-k selection.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_grad_and_obj(a, conv, temperature):
    """
    Compute smooth gradient (log-sum-exp) and smooth objective value.
    Returns (gradient, smooth_obj).
    """
    n = len(a)
    scaled = temperature * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    else:
        w /= w_sum
    # Smooth objective: (n/temperature) * (log(sum(exp(temp*conv))) - temp*max(conv))
    # Actually: (1/temp)*log(sum(exp(temp*conv))) = max(conv) + (1/temp)*log(sum(exp(temp*(conv-max(conv)))))
    smooth_obj = (1.0 / temperature) * (np.log(w_sum) + temperature * conv.max())
    # Wait, let me redo:
    # (1/t)*log(sum(exp(t*x_k))) = max(x) + (1/t)*log(sum(exp(t*(x_k-max(x)))))
    # = max(x) + (1/t)*log(sum(w_k))  [since w_k = exp(t*(x_k-max(x)))/sum(exp(t*(x_k-max(x))))]
    # Actually w_sum = sum(exp(t*(x_k - max(x))))
    # So (1/t)*log(sum(exp(t*x_k))) = max(x) + (1/t)*log(w_sum)
    
    # Recompute properly
    scaled2 = temperature * conv - temperature * conv.max()
    w2 = np.exp(scaled2)
    w2_sum = w2.sum()
    smooth_obj = conv.max() + (1.0 / temperature) * np.log(w2_sum)
    
    # Gradient: (1/w2_sum) * sum_k exp(t*(b[k]-max(b))) * 2 * shift(a, k)
    # = sum_k w2[k]/w2_sum * 2 * shift(a, k)
    w_norm = w2 / w2_sum
    g = np.zeros(n)
    for k in range(len(conv)):
        if w_norm[k] < 1e-15:
            continue
        wk = 2.0 * w_norm[k]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        g[idx] += wk * a[k - idx]
    
    return g, smooth_obj

def adam_simplex(a0, time_limit, temp_init=50.0, temp_growth=1.01,
                 lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                 perturb_every=500, perturb_scale=0.01, batch_size=20):
    """
    Adam optimizer on the probability simplex with log-sum-exp smoothed gradients.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    obj = 2.0 * n * conv.max()
    best_obj = obj
    best_a = a.copy()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    temp = temp_init
    no_improve = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break
            
            t += 1
            g, _ = smooth_grad_and_obj(a, conv, temp)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g * g
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_simplex(a_new)
            
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adapt temperature (increase to include more peaks)
        temp *= temp_growth
        # Adapt learning rate (decrease slowly)
        lr = max(0.001, lr * 0.9995)
        
        # Perturbation to escape local minima
        if no_improve >= perturb_every:
            rng = np.random.default_rng()
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            m *= 0.5
            v *= 0.5
            no_improve = 0
            perturb_scale *= 0.7
            perturb_every = int(perturb_every * 1.3)
    
    return best_a, best_obj

def qp_refine(a, w, iters=200, lr=0.01):
    """
    Solve the QP: minimize a^T H a subject to a >= 0, sum(a) = 1
    where H[i,j] = w[i+j]. Uses projected gradient.
    """
    n = len(a)
    a_q = a.copy()
    
    for _ in range(iters):
        # Gradient of a^T H a = 2 * H * a
        # (H * a)[i] = sum_j H[i,j] * a[j] = sum_j w[i+j] * a[j]
        grad = np.zeros(n)
        for i in range(n):
            j_start = 0
            j_end = min(n - 1, 2 * n - 2 - i)
            if j_start <= j_end:
                grad[i] = 2.0 * np.dot(w[i:i + j_end + 1], a[j_start:j_end + 1])
        
        a_q = project_simplex(a_q - lr * grad)
    
    return a_q

def qp_refine_fft(a, conv, temperature, iters=100):
    """
    QP refinement using the current convolution as weights.
    minimizes sum_k softmax(t*conv_k) * b_k(a)
    """
    n = len(a)
    scaled = temperature * conv - temperature * conv.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    a_q = a.copy()
    lr = 0.01 / (w.max() * n)  # Adaptive learning rate
    
    for _ in range(iters):
        # Compute gradient: 2 * C_w * a where C_w[i,j] = w[i+j]
        # (C_w * a)[i] = sum_j w[i+j] * a[j]
        grad = np.zeros(n)
        for i in range(n):
            j_max = min(n - 1, 2 * n - 2 - i)
            if j_max >= 0:
                grad[i] = 2.0 * np.dot(w[i:i + j_max + 1], a[:j_max + 1])
        
        a_q = project_simplex(a_q - lr * grad)
    
    return a_q

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n, dtype=np.float64)
    
    # Asymmetric: concentrated at left
    a = np.exp(-x / (0.2 * n))
    a /= a.sum()
    inits['left_concentrated'] = a
    
    # Asymmetric: concentrated at right
    a = np.exp(-(n - 1 - x) / (0.2 * n))
    a /= a.sum()
    inits['right_concentrated'] = a
    
    # Biased toward edges
    a = np.ones(n)
    edge_w = 1.0 + 2.0 * np.minimum(x, n - 1 - x) / (n / 2)
    a = edge_w
    a /= a.sum()
    inits['edge_biased'] = a
    
    # Sine wave
    a = np.abs(np.sin(np.pi * x / n)) + 0.1
    a /= a.sum()
    inits['sine'] = a
    
    # From parent
    if parent is not None:
        if len(parent) == n:
            p = parent.copy()
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        elif len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_up'] = p / ps
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Random with structure (blocks)
    n_blocks = rng.integers(3, 10)
    block_sizes = rng.dirichlet(np.ones(n_blocks) * 2) * n
    block_heights = rng.dirichlet(np.ones(n_blocks) * 2)
    a_block = np.zeros(n)
    pos = 0
    for i in range(n_blocks):
        start = int(round(pos))
        end = int(round(pos + block_sizes[i]))
        if end > start:
            a_block[start:end] = block_heights[i]
        pos += block_sizes[i]
    a_block = np.maximum(a_block, 0)
    a_s = a_block.sum()
    if a_s > 1e-10:
        inits['blocks'] = a_block / a_s
    
    return inits

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with log-sum-exp
smoothed full-softmax gradients, QP subproblem refinement with precomputed
Hankel matrix, multi-phase search with diverse initializations, and adaptive
temperature scheduling. Key innovations: (1) full-softmax gradient over ALL
convolution peaks (not just top-k) for richer subgradient information,
(2) exact QP subproblem via precomputed Hankel matrix for large structured steps,
(3) asymmetric initializations that exploit the freedom in sequence shape.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, conv, temperature):
    """
    Compute smooth gradient using log-sum-exp over ALL convolution entries.
    Returns gradient vector of the smoothed objective 2n * (1/t)*log(sum(exp(t*b))).
    """
    n = len(a)
    scaled = temperature * conv - temperature * conv.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback: only the max
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
        w_sum = 1.0
    w_norm = w / w_sum

    # Gradient: sum_k w_norm[k] * 2 * shift(a, k)
    g = np.zeros(n)
    threshold = w_norm.max() * 1e-10
    active = np.where(w_norm > threshold)[0]
    for k_int in active:
        k = int(k_int)
        wk = 2.0 * w_norm[k_int]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        g[idx] += wk * a[k - idx]
    return g


def compute_objective(n, conv):
    """Compute 2*n*max(conv)."""
    return 2.0 * n * conv.max()


def adam_simplex(a0, time_limit, temp_init=50.0, temp_growth=1.02,
                 lr=0.015, beta1=0.9, beta2=0.999, eps=1e-8,
                 perturb_every=400, perturb_scale=0.015, batch_size=15,
                 lr_decay=0.9998):
    """
    Adam optimizer on the probability simplex with log-sum-exp smoothed gradients.
    Returns (best_a_normalized, best_objective).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    obj = compute_objective(n, conv)
    best_obj = obj
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    no_improve = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break

            t += 1
            g = smooth_grad(a, conv, temp)

            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * g * g
            m_hat = m / (1.0 - beta1**t)
            v_hat = v / (1.0 - beta2**t)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_simplex(a_new)

            conv = fast_convolve(a)
            cur_obj = compute_objective(n, conv)

            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        temp *= temp_growth
        lr *= lr_decay

        if no_improve >= perturb_every:
            rng = np.random.default_rng()
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            m *= 0.3
            v *= 0.3
            no_improve = 0
            perturb_scale *= 0.65
            perturb_every = int(perturb_every * 1.4)

    return best_a, best_obj
