"""
Multi-method optimization for minimum peak autocorrelation of non-negative sequences.
Combines log-sum-exp gradient descent with adaptive beta annealing, LP blending refinement,
coordinate descent, and simulated annealing to minimize 2n*max(conv(a,a))/sum(a)^2.
Key innovation: uses smooth log-sum-exp surrogate for the max objective enabling
efficient gradient-based optimization, combined with LP for escaping local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum, maintaining non-negativity."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def project_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(n, S / n)
    return a * (S / s)


def logsumexp_gd(a, time_limit, beta_start=0.5, beta_end=20.0, verbose=True):
    """
    Log-sum-exp gradient descent with adaptive beta annealing.
    Minimizes the smooth surrogate of the peak autocorrelation.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_max = np.max(fft_convolve(a, a))
    
    total_iters = max(100, int(time_limit * 50))
    lr = 1.0 / (n * beta_end)
    
    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal beta
        progress = min(1.0, it / (0.8 * total_iters))
        beta = beta_start + (beta_end - beta_start) * progress
        lr = 0.5 / (n * beta + 1.0)
        
        # Compute convolution and softmax weights
        R = fft_convolve(a, a)
        R_max = np.max(R)
        exp_R = np.exp(beta * (R - R_max))
        w = exp_R / np.sum(exp_R)
        
        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
        grad = 2.0 * np.correlate(w, a, mode='valid')
        
        # Gradient step
        a_new = a - lr * grad
        
        # Project onto {x >= 0, sum = S}
        a_new = project_simplex(a_new, S)
        
        # Evaluate
        new_R = fft_convolve(a_new, a_new)
        new_max = np.max(new_R)
        
        if new_max < best_max:
            best_max = new_max
            best_a = a_new.copy()
            a = a_new
        else:

                a_try = a - lr * shrink * grad
                a_try = project_simplex(a_try, S)
                try_max = np.max(fft_convolve(a_try, a_try))
                if try_max < np.max(fft_convolve(a, a)):
                    a = a_try
                    if try_max < best_max:
                        best_max = try_max
                        best_a = a.copy()
                    break
        
        if verbose and it % 2000 == 0:
            score = 2.0 * n * best_max / (S * S)
            print(f"      LSE-GD iter {it}: beta={beta:.2f}, score={score:.6f}, max_R={best_max:.6f}")
    
    return best_a


def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]
    return A


def lp_refine(a, time_limit, verbose=True):
    """
    LP-based iterative refinement: maximize sum(g) s.t. conv(a,g) <= M, g >= 0.
    Blend with current a to reduce peak.
    """

    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S).copy()

    A = None

            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

            A = build_conv_matrix(a)
        else:
            # Rebuild since a changed
            A = build_conv_matrix(a)
        
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        g_norm = normalize_seq(g, S)
        
        # Find best blend parameter via bisection
        lo, hi = 0.0, 1.0
        best_t = 0.0
        best_val = M
        for _ in range(12):
            mid = (lo + hi) / 2.0
            b = (1.0 - mid) * a + mid * g_norm

            b = normalize_seq(b, S)
            b_max = np.max(fft_convolve(b, b))
            if b_max <= M + 1e-10:

                best_val = b_max
                lo = mid
            else:
                hi = mid
        
        if best_t < 1e-6:
            break
        
        a = (1.0 - best_t) * a + best_t * g_norm

def coordinate_descent(a, time_limit, verbose=True):

        """Try changing a[i] by delta, return new max R."""
        # R[k] changes by 2*delta*a[k-i] + delta^2 for valid k-i

        hi = min(2 * n - 1, i + n - 1)
        if lo > hi:
            return np.max(fft_convolve(a, a))
        
        R = fft_convolve(a, a)
        # Quick check of affected lags
        affected = R[lo:hi+1] + 2.0 * delta * a[hi-i:lo+i+1][::-1] + delta * delta
        # Recompute properly
        a_new = a.copy()
        a_new[i] = max(0.0, a[i] + delta)
        new_R = fft_convolve(a_new, a_new)
        return np.max(new_R)
    
    sweeps = 0
    while time.monotonic() - t0 < time_limit:
        # Random permutation of coordinates
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            R = fft_convolve(a, a)
            current_max = np.max(R)
            
            # Try reducing a[i] if it contributes to the peak
            peak_idx = int(np.argmax(R))
            # Contribution of a[i] to R[peak_idx]: a[peak_idx - i] + a[i - peak_idx] (if valid)
            contrib = 0.0
            if 0 <= peak_idx - i < n:
                contrib += a[peak_idx - i]
            if 0 <= i - peak_idx < n and i - peak_idx != peak_idx - i:
                contrib += a[i - peak_idx]

                # Try reducing

                    a_try[i] = new_val
                    a_try = normalize_seq(a_try, S)
                    new_max = np.max(fft_convolve(a_try, a_try))
                    if new_max < best_max * 0.999:

                        best_a = a_try.copy()
                        a = a_try
                        break
                    elif new_max < current_max:
                        a = normalize_seq(a_try, S)
                        break

                        delta = frac * S / n
                        a_try = a.copy()
                        a_try[j] += delta
                        a_try = normalize_seq(a_try, S)
                        new_max = np.max(fft_convolve(a_try, a_try))
                        if new_max < best_max:
                            best_max = new_max
                            best_a = a_try.copy()
                            a = a_try
                            break
        
        sweeps += 1

def simulated_annealing(a, time_limit, verbose=True):
    """
    Simulated annealing with adaptive perturbation size.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = local_score(a)
    current_score = best_score
    rng = np.random.default_rng(777)
    
    T = 0.1
    T_min = 0.001
    cool_rate = 0.999
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        # Generate perturbation
        perturb_type = rng.random()
        a_try = a.copy()
        
        if perturb_type < 0.3:
            # Multiply a random block
            width = max(1, int(n * rng.uniform(0.01, 0.1)))
            start = int(rng.integers(0, max(1, n - width)))
            factor = np.exp(rng.normal(0, 0.1))
            a_try[start:start+width] *= factor
        elif perturb_type < 0.6:
            # Gaussian perturbation on random subset
            n_pert = max(1, int(n * rng.uniform(0.01, 0.1)))
            idx = rng.choice(n, size=n_pert, replace=False)
            a_try[idx] *= np.exp(rng.normal(0, 0.05, size=n_pert))
        elif perturb_type < 0.8:
            # Swap mass between two positions
            i, j = rng.choice(n, size=2, replace=False)
            transfer = rng.uniform(0.01, 0.1) * S / n
            a_try[i] -= transfer
            a_try[j] += transfer
        else:
            # Large perturbation
            a_try *= np.exp(rng.normal(0, 0.02, size=n))
        
        a_try = np.maximum(a_try, 0.0)
        a_try = normalize_seq(a_try, S)
        
        new_score = local_score(a_try)
        
        # Accept/reject
        if new_score < current_score or rng.random() < np.exp(-(new_score - current_score) / T):
            a = a_try
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                if verbose:
                    elapsed = time.monotonic() - t0
                    if it % 1000 == 0:
                        print(f"      SA iter {it} (t={elapsed:.0f}s): best={best_score:.6f}")
        
        T = max(T_min, T * cool_rate)
        it += 1
        
        # Reheat occasionally
        if it % 5000 == 0:
            T = max(T, 0.05)
    
    return best_a

    # Blend with original
    best_a = a.copy()
    best_score = local_score(a)
    
    for t_blend in [0.3, 0.5, 0.7, 1.0]:
        a_blend = (1 - t_blend) * a + t_blend * a_up
        a_blend = normalize_seq(a_blend, S)
        s = local_score(a_blend)
        if s < best_score:
            best_score = s
            best_a = a_blend
    
    # Final GD on the result
    remaining = time_limit - (time.monotonic() - t0) - 5
    if remaining > 10:
        best_a = logsumexp_gd(best_a, remaining, verbose=False)
    
    return best_a


def make_chirp_initialization(n, rng):
    """Create a chirp-like initialization with random parameters."""
    t = np.arange(n, dtype=np.float64)
    n_cycles = rng.uniform(1, 4)
    alpha = rng.uniform(0.5, 1.5)
    phase = rng.uniform(0, 2 * np.pi)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * alpha * (t / n) ** 2 * n_cycles + phase)
    # Add some asymmetry
    asym = rng.uniform(-0.3, 0.3)
    a += asym * (t / n - 0.5)
    return np.clip(a, 0.05, None)


def random_initialization(n, rng):
    """Random positive initialization."""
    a = rng.uniform(0.5, 1.5, size=n)
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LSE-GD, LP refinement, coordinate descent,
    and simulated annealing. Tries multiple sequence lengths and initializations.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False
    
    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)
    
    # Phase 1: Log-sum-exp gradient descent on provided sequence
    print("\n=== Phase 1: LSE-GD on provided sequence ===")
    if provided_seq is not None:
        n = len(provided_seq)
        t_limit = min(300.0, (deadline - time.monotonic()) * 0.25)
        print(f"  LSE-GD n={n}, time_limit={t_limit:.0f}s")
        result = logsumexp_gd(provided_seq, t_limit, beta_start=0.5, beta_end=25.0, verbose=True)
        if update_best(result):
            print(f"  New best after LSE-GD: {best_score:.6f}")
    
    # Phase 2: LP refinement
    print("\n=== Phase 2: LP refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(200.0, (deadline - time.monotonic()) * 0.2)
        print(f"  LP refine n={n}, time_limit={t_limit:.0f}s")
        result = lp_refine(best_seq, t_limit, verbose=True)
        if update_best(result):
            print(f"  New best after LP: {best_score:.6f}")
    
    # Phase 3: Coordinate descent
    print("\n=== Phase 3: Coordinate descent ===")
    if best_seq is not None:
        t_limit = min(200.0, (deadline - time.monotonic()) * 0.2)
        print(f"  CD, time_limit={t_limit:.0f}s")
        result = coordinate_descent(best_seq, t_limit, verbose=True)
        if update_best(result):
            print(f"  New best after CD: {best_score:.6f}")
    
    # Phase 4: Second round of LSE-GD (with different parameters)
    print("\n=== Phase 4: LSE-GD round 2 ===")
    if best_seq is not None:
        t_limit = min(200.0, (deadline - time.monotonic()) * 0.2)
        print(f"  LSE-GD round 2, time_limit={t_limit:.0f}s")
        result = logsumexp_gd(best_seq, t_limit, beta_start=2.0, beta_end=50.0, verbose=True)
        if update_best(result):
            print(f"  New best after LSE-GD2: {best_score:.6f}")
    
    # Phase 5: Simulated annealing
    print("\n=== Phase 5: Simulated annealing ===")
    if best_seq is not None:
        t_limit = min(150.0, (deadline - time.monotonic()) * 0.2)
        print(f"  SA, time_limit={t_limit:.0f}s")
        result = simulated_annealing(best_seq, t_limit, verbose=True)
        if update_best(result):
            print(f"  New best after SA: {best_score:.6f}")
    
    # Phase 6: Try other lengths from interpolated provided sequence
    print("\n=== Phase 6: Other lengths ===")
    if provided_seq is not None:
        for n_try in [384, 512, 768]:
            if time.monotonic() > deadline - 100:
                break
            x_old = np.linspace(0, 1, len(provided_seq))
            x_new = np.linspace(0, 1, n_try)
            a0 = np.interp(x_new, x_old, provided_seq)
            a0 = np.clip(a0, 0.01, None)
            
            t_per = min(60.0, (deadline - time.monotonic()) * 0.3)
            if t_per < 20:
                break
            
            result = logsumexp_gd(a0, t_per, beta_start=1.0, beta_end=20.0, verbose=False)
            if update_best(result):
                print(f"  *** n={n_try}: new best score={best_score:.6f} ***")
    
    # Phase 7: Try structured initializations
    print("\n=== Phase 7: Structured initializations ===")
    for n in [512, 640, 768]:
        if time.monotonic() > deadline - 100:
            break
        init = make_chirp_initialization(n, rng)
        t_per = min(40.0, (deadline - time.monotonic()) * 0.2)
        if t_per < 15:
            break
        result = logsumexp_gd(init, t_per, beta_start=0.5, beta_end=20.0, verbose=False)
        if update_best(result):
            print(f"  *** chirp n={n}: new best score={best_score:.6f} ***")
    
    # Phase 8: Final coordinate descent polish
    print("\n=== Phase 8: Final CD polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        if remaining > 30:
            t_limit = min(remaining, 120.0)
            print(f"  Final CD, time_limit={t_limit:.0f}s")
            result = coordinate_descent(best_seq, t_limit, verbose=True)
            if update_best(result):
                print(f"  Final best: {best_score:.6f}")
    
    # Phase 9: Final LSE-GD polish with high beta
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            t_limit = min(remaining, 60.0)
            print(f"\n=== Phase 9: Final LSE-GD polish ===")
            result = logsumexp_gd(best_seq, t_limit, beta_start=5.0, beta_end=100.0, verbose=True)
            if update_best(result):
                print(f"  After final polish: {best_score:.6f}")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    
    return [1.0] * 640
