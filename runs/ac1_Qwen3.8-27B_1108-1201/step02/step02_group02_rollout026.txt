```python
import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def project_simplex(v, t):
    """Project v onto {x >= 0, sum(x) = t}."""
    n = len(v)
```

```python
"""
Smoothed proximal gradient optimization for minimizing peak self-convolution
of non-negative step functions. Uses log-sum-exp smoothing, Nesterov momentum,
projected gradient steps, periodic LP guidance, and simulated annealing polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score (fast)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def project_simplex(v, t):
    """Project v onto {x >= 0, sum(x) = t}."""
    n = len(v)
    u, idx = np.sort(v)[::-1], np.argsort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - t)
    if not np.any(cond):
        return np.full(n, t / n)
    rho_val = np.max(np.where(cond)[0])
    theta = (cssv[rho_val] - t) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, c, beta):
    """
    Compute gradient of lse(beta * c) with respect to a.
    c = a * a (convolution, length 2n-1).
    Returns gradient of length n.
    """
    n = len(a)
    # Softmax weights
    c_max = np.max(c)
    log_w = beta * (c - c_max)
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w /= np.sum(w)
    
    # Gradient: grad_j = lse(beta*c) * 2 * sum_k w[k] * a[k-j]
    # The sum is a cross-correlation
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(W * np.conj(A), size)[:n]
    
    lse_val = c_max + np.log(np.sum(np.exp(beta * (c - c_max))))
    return 2.0 * lse_val * cross


def compute_exact_subgradient(a, c):
    """Compute subgradient of max_k c_k at point a."""
    n = len(a)
    max_c = np.max(c)
    # Find peaks (within tolerance)
    tol = 1e-8
    peaks = np.where(c >= max_c - tol * max_c)[0]
    
    # g_j = 2 * sum_{k in peaks} a[k-j] for valid k-j
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    w = np.zeros(2 * n - 1)
    w[peaks] = 1.0 / len(peaks)
    
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(W * np.conj(A), size)[:n]
    return 2.0 * cross


def solve_lp_direction(f, M):
    """Solve LP to find best descent direction."""
    n = len(f)
    # Build constraint matrix: (f * g)_k <= M for all k
    # (f * g)_k = sum_j f[k-j] * g[j]
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return np.ones(n) / n

    a = a_init.astype(np.float64).copy()
    score = local_score(a)
    best_a = a.copy()
    best_score = score
    n = len(a)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        cand = a.copy()
        r = rng.random()
        
        if r < 0.4:
            # Perturb a few random elements
            k = rng.integers(3, 20)
            idx = rng.choice(n, size=k, replace=False)

        elif r < 0.6:
            # Block perturbation
            width = rng.integers(2, max(3, n // 8))
            start = rng.integers(0, n - width)
            cand[start:start+width] *= np.exp(rng.normal(0, 0.02, width))
        elif r < 0.8:
            # Pair swap
            i, j = rng.integers(0, n, 2)
            if i != j:
                delta = rng.uniform(-0.05, 0.05) * a[i]
                cand[i] = max(0, a[i] - delta)
                cand[j] = max(0, a[j] + delta)
        else:
            # Global small perturbation
            cand = a * np.exp(rng.normal(0, 0.005, n))
        
        cand = project_simplex(cand, 1.0)
        new_score = local_score(cand)

            a = cand
            score = new_score
            if score < best_score:
                best_score = score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - score) / max(temp, 1e-10)):
            a = cand
            score = new_score
        
        temp *= cooling
    
    return best_a, best_score
```

The convolution c[k] = sum_i a[i] * a[k-i] (with a[m] = 0 for m < 0 or m >= n).

The derivative of c[k] with respect to a[j]:
- c[k] = sum_i a[i] * a[k-i]
- dc[k]/da[j] = a[k-j] (from the term where i=j) + a[j] (from the term where k-i=j, i.e., i=k-j)
