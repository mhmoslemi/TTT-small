import numpy as np
import time
import scipy.fft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_normalized(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def spectral_flatten(a0, n_iters=2000, target_quantile=0.7, time_limit=60):
    """
    Spectral flattening: iteratively flatten the magnitude spectrum
    while maintaining non-negativity and support constraints.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
        s = 1.0
    else:
        a /= s
    
    N = 2 * n - 1

    best_a = a.copy()

        if time.monotonic() - t0 > time_limit:
            break
        
        # Zero-pad and FFT
        a_padded = np.zeros(N)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        
        # Get magnitude and phase
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: flatten non-DC magnitudes
        # Use a soft clipping approach
        non_dc_mag = np.concatenate([mag[1:], mag[:0]])  # exclude DC (index 0)

        # But let's just work with all bins except 0
        
        # Compute target as a quantile of non-DC magnitudes
        # We want to bring peaks down towards the median
        target = np.quantile(non_dc_mag, target_quantile)
        
        # Soft clipping: reduce magnitudes above target
        # Use a smooth function to avoid discontinuities
        alpha = 2.0  # sharpness of clipping
        new_mag = mag.copy()
        for k in range(1, N):  # skip DC

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using annealed mirror descent
with spectral initialization and diverse restarts.
Key improvements over previous approach:
1. Simulated annealing to escape local minima
2. Multi-scale temperature schedule (coarse-to-fine)

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def norm_obj(a):
    """Objective value assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def annealed_mirror_descent(a0, time_limit, t_start=5.0, t_end=500.0,
                             eta_start=1.0, eta_end=0.01,
                             perturb_prob_start=0.3, perturb_prob_end=0.0,
                             perturb_magnitude=0.1,
                             rng=None):
    """
    Mirror descent on the simplex with simulated annealing.
    
    Parameters:
    - t_start, t_end: temperature for softmax (coarse to fine)
    - eta_start, eta_end: step size
    - perturb_prob_start, perturb_prob_end: probability of random perturbation
    """

        rng = np.random.default_rng()
    
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    total_time = time_limit
    patience = 0
    max_patience = 200
    
    while time.monotonic() - t0 < total_time:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_time)
        
        # Interpolate parameters
        t_param = t_start + (t_end - t_start) * progress
        eta = eta_start + (eta_end - eta_start) * progress
        perturb_prob = perturb_prob_start + (perturb_prob_end - perturb_prob_start) * progress
        
        # Compute autocorrelation
        c = fast_convolve(a)
        c_max = c.max()

        scaled = t_param * (c - c_max) / c_max if c_max > 1e-15 else np.zeros_like(c)
        scaled -= scaled.max()  # for numerical stability
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(2*n-1)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum

        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]
        
        # Mirror descent update
        sg_c = sg - sg.mean()
        log_f = -eta * sg_c
        log_f -= log_f.max()  # numerical stability
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s

        if rng.random() < perturb_prob:
            # Add multiplicative noise
            noise = rng.standard_normal(n) * perturb_magnitude
            noise -= noise.mean()  # keep mean approximately 0
            a = a * np.exp(noise)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:

            else:
                a /= s
        
        # Check objective
        c = fast_convolve(a)
        obj = 2.0 * n * c.max()
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
    
    return best_a, best_obj

            j_end = min(n - 1, k)
            sg[j_start:j_end+1] += w[k] * a[k-j_end : k-j_start+1][::-1]
        
        # Update
        sg_c = sg - sg.mean()
        lf = -eta * sg_c
        lf -= lf.max()
        a = a * np.exp(lf)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        
        c_new = fast_convolve(a)
        obj = 2.0 * n * c_new.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > 300:
                break
    
    return best_a, best_obj

def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.1, 0.2, 0.35]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n))**2)
        inits[f'g{int(sf*100)}'] = g / g.sum()
    
    inits['ramp'] = np.arange(1, n+1, dtype=float)
    inits['ramp'] /= inits['ramp'].sum()
    
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    
    cb = 0.5 * (1 + np.cos(2*np.pi*x/n))
    inits['cos'] = cb / cb.sum()
    
    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.5)
    
    return inits

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Corrected mirror descent (with proper factor-of-2 gradient)
- Multi-scale temperature schedule (coarse-to-fine)
- Iterated local search with diverse perturbations
- Solution averaging to exploit cross-term reduction
- Momentum for smoother convergence
- Diverse initializations and multiple sequence lengths

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def mirror_descent(a0, time_limit, t_start=10.0, t_end=300.0,
                   eta_start=0.5, eta_end=0.01,
                   momentum=0.5, rng=None):
    """
    Mirror descent on probability simplex with softmax-smoothed max objective.
    Uses corrected gradient (factor of 2) and momentum.
    """
    a = normalize(a0.astype(np.float64))
    n = len(a)
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    patience = 0
    max_patience = 500
    vel = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))

        # Interpolate parameters
        t_param = t_start + (t_end - t_start) * progress
        eta = eta_start + (eta_end - eta_start) * progress

        # Compute autocorrelation and softmax weights
        c = fast_convolve(a)
        c_max = c.max()
        if c_max < 1e-15:
            break

        scaled = t_param * (c - c_max) / c_max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum

        # Compute subgradient with correct factor of 2
        sg = np.zeros(n)
        thresh = w.max() * 0.002
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end + 1] += 2.0 * w[k] * a[k - j_end:k - j_start + 1][::-1]

        # Momentum
        vel = momentum * vel + (1.0 - momentum) * sg
        sg_use = vel - vel.mean()

        # Mirror descent update (multiplicative on simplex)
        lf = -eta * sg_use
        lf -= lf.max()
        a = a * np.exp(lf)
        a = normalize(a)

        # Evaluate
        c_new = fast_convolve(a)
        obj = 2.0 * n * c_new.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

    return best_a, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    inits['uni'] = np.ones(n) / n

    for sf in [0.08, 0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n)) ** 2)
        inits[f'g{int(sf * 100)}'] = g / g.sum()

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    tri = np.maximum(1.0 - np.abs(x - mid) / (0.5 * n), 0.05)
    inits['tri'] = tri / tri.sum()

    dp = np.zeros(n)
    q = n // 3
    dp[q - 1:q + 1] = 1.0
    dp[2 * q - 1:2 * q + 1] = 1.0
    inits['dpeak'] = dp / dp.sum()

    return inits
