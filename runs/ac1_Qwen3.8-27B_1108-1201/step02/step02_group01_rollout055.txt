Multi-phase optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Uses LP-based refinement with warm starts,
subgradient descent, and targeted local search.
"""

import numpy as np
import time
from scipy.optimize import linprog

# Pre-allocated buffers for FFT
def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))

def normalize(a, target_sum=None):
    """Normalize to target sum (default sqrt(2n))."""
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s

def build_lp_matrix(a):
    """Build A matrix where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A

def lp_solve(a, M, x0=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    if x0 is not None:
        # Check feasibility of x0
        if np.all(A @ x0 <= M + 1e-6) and np.all(x0 >= -1e-9):
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'dual_infeasible_tolerance': 1e-9})
            # Try to use x0 as starting point - HiGHS doesn't directly support this
            # but we can warm-start via the primal solution
            if res.status == 0:
                return res.x
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)

    n = len(a)

    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv = np.max(fft_convolve(a, a))
    
    for it in range(2000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        M = float(np.max(conv_a))
        
        g = lp_solve(a, M)
        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break
        
        g_norm = g * S / g_sum
        
        # Line search for best blend

"""
Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Combines LP-based refinement with warm starts,
fast subgradient descent exploiting the shifted-structure of the gradient,
simulated-annealing kicks to escape local minima, and targeted local search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * float(np.max(conv)) / (s * s))


def normalize(a, target_sum=None):
    """Normalize sequence to target sum."""
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    s = float(np.sum(a))
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * (target_sum / s)


def build_lp_matrix(a):
    """Build A matrix where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns optimal g."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 10, 'mip_rel_gap': 0.01})
    if res.status == 0:
        return np.maximum(res.x, 0.0)
    return np.zeros(n)


def lp_refine(a, time_limit, verbose=False):
    """LP-based iterative refinement with adaptive line search and kicks."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv = float(np.max(fft_convolve(a, a)))
    rng = np.random.default_rng(777)

    stall_count = 0
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fft_convolve(a, a)
        M = float(np.max(conv_a))

        g = lp_solve(a, M)
        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break

        g_norm = g * (S / g_sum)

        # Adaptive line search
        best_t = 0.0
        best_new_conv = M
        t_grid = [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
        for t_try in t_grid:
            b = (1.0 - t_try) * a + t_try * g_norm
            b = np.clip(b, 0.0, None)
            b = normalize(b, S)
            bc = float(np.max(fft_convolve(b, b)))
            if bc < best_new_conv:
                best_new_conv = bc
                best_t = t_try

        if best_t == 0.0:
            stall_count += 1
            if stall_count > 5:
                # Kick: add noise to escape
                a = a + rng.normal(0, 0.01 * np.mean(a), n)
                a = np.clip(a, 0.0, None)
                a = normalize(a, S)
                stall_count = 0
            continue

        a = (1.0 - best_t) * a + best_t * g_norm
        a = np.clip(a, 0.0, None)
        a = normalize(a, S)

        new_conv = float(np.max(fft_convolve(a, a)))
        if new_conv < best_conv - 1e-14:
            best_conv = new_conv
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if verbose and it % 50 == 0:
            score = 2.0 * n * best_conv / (S * S)
            print(f"    LP it={it}: conv={best_conv:.8f}, score={score:.6f}, t={best_t:.4f}")

        if best_t < 0.005 and stall_count > 10:
            break

    return best_a


def subgradient_optimize(a, time_limit, verbose=False):
    """
    Fast subgradient descent exploiting the structure:
    grad[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    Uses adaptive step size and handles shifting peaks.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv = float(np.max(fft_convolve(a, a)))

    lr = 0.5 / n
    step = 0
    stall = 0
    peak_history = []

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        peak_val = float(conv[peak_idx])

        # Compute gradient: grad[i] = 2*a[peak_idx - i] if in range
        grad = np.zeros(n)
        ki_start = max(0, peak_idx - n + 1)
        ki_end = min(peak_idx, n - 1)
        for i in range(n):
            ki = peak_idx - i
            if 0 <= ki < n:
                grad[i] = 2.0 * a[ki]

        # Adaptive step size
        if len(peak_history) > 30:
            recent = peak_history[-30:]
            unique_peaks = len(set(recent))
            if unique_peaks <= 3:
                lr = max(0.01 / n, lr * 0.85)  # oscillating, reduce
            elif unique_peaks >= 15:
                lr = min(2.0 / n, lr * 1.05)   # exploring, increase
        peak_history.append(peak_idx)
        if len(peak_history) > 100:
            peak_history.pop(0)

        # Gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S)

        new_conv = float(np.max(fft_convolve(a_new, a_new)))
        if new_conv < best_conv:
            best_conv = new_conv
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.9
                if stall > 200:
                    # Add small noise to escape
                    a_new = a_new + np.random.default_rng(stall).normal(0, 0.005 * np.mean(a_new), n)
                    a_new = np.clip(a_new, 0, None)
                    a_new = normalize(a_new, S)
                    stall = 0

        a = a_new
        step += 1

        if verbose and step % 200 == 0:
            score = 2.0 * n * best_conv / (S * S)
            print(f"    Subgrad step={step}: score={score:.6f}, lr={lr:.5f}, peak={peak_idx}")

    return best_a


def local_search(a, time_limit, verbose=False):
    """Targeted local search: peak-aware coordinate and block updates."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv = float(np.max(fft_convolve(a, a)))
    rng = np.random.default_rng(42)

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        peak_val = float(conv[peak_idx])

        # Compute contributions to peak
        contribs = np.zeros(n)
        for j in range(n):
            ki = peak_idx - j
            if 0 <= ki < n:
                contribs[j] = a[j] * a[ki]

        # Strategy 1: Reduce top contributors, increase non-contributors
        top_k = max(5, n // 8)
        top_idx = np.argsort(contribs)[-top_k:]
        other_idx = np.setdiff1d(np.arange(n), top_idx)

        for trial in range(4):
            candidate = a.copy()
            # Reduce top contributors with varying strength
            for idx in top_idx:
                candidate[idx] *= np.exp(rng.normal(-0.04, 0.015))
            # Increase others
            if len(other_idx) > 0:
                n_inc = max(3, len(other_idx) // 8)
                inc_idx = rng.choice(other_idx, size=min(n_inc, len(other_idx)), replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=len(inc_idx)))
            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize(candidate, S)
            new_conv = float(np.max(fft_convolve(candidate, candidate)))
            if new_conv < best_conv:
                best_conv = new_conv
                best_a = candidate.copy()
                a = candidate
            elif new_conv < peak_val * 1.03 and rng.random() < 0.15:
                a = candidate

        # Strategy 2: Coordinate descent on single elements
        if rng.random() < 0.4:
            idx = int(rng.integers(0, n))
            for delta in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]:
                candidate = a.copy()
                candidate[idx] = max(0.0, a[idx] * (1.0 + delta))
                candidate = normalize(candidate, S)
                new_conv = float(np.max(fft_convolve(candidate, candidate)))
                if new_conv < best_conv:
                    best_conv = new_conv
                    best_a = candidate.copy()
                    a = candidate

        # Strategy 3: Block perturbation
        if rng.random() < 0.3:
            width = max(2, n // 15)
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize(candidate, S)
            new_conv = float(np.max(fft_convolve(candidate, candidate)))
            if new_conv < best_conv:
                best_conv = new_conv
                best_a = candidate.copy()
                a = candidate

        iteration += 1
        if verbose and iteration % 30 == 0:
            score = 2.0 * n * best_conv / (S * S)
            print(f"    Local it={iteration}: score={score:.6f}")

    return best_a


def simulate_annealing_kick(a, time_limit, verbose=False):
    """Simulated annealing to escape local minima, then LP polish."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv = float(np.max(fft_convolve(a, a)))
    rng = np.random.default_rng(123)

    T = 0.1  # initial temperature
    step = 0
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_conv = float(np.max(conv))

        # Generate perturbation
        candidate = a.copy()
        # Pick a few random positions and perturb
        n_perturb = max(3, n // 20)
        idx = rng.choice(n, size=n_perturb, replace=False)
        candidate[idx] *= np.exp(rng.normal(0.0, 0.1, size=n_perturb))
        candidate = np.clip(candidate, 0.0, None)
        candidate = normalize(candidate, S)

        new_conv = float(np.max(fft_convolve(candidate, candidate)))
        delta = new_conv - current_conv

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
            a = candidate
            if new_conv < best_conv:
                best_conv = new_conv
                best_a = a.copy()

        T *= 0.995
        step += 1
        if verbose and step % 500 == 0:
            score = 2.0 * n * best_conv / (S * S)
            print(f"    SA step={step}: T={T:.4f}, score={score:.6f}")

    return best_a
