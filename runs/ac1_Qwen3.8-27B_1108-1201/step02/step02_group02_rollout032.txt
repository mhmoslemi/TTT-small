import numpy as np

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_subgradient(f):
    """Compute subgradient of the peak autocorrelation (convolution peak)."""
    n = len(f)
    c = fft_convolve(f, f)
    max_c = np.max(c)
    peak_mask = c >= max_c * (1.0 - 1e-6)
    peak_indices = np.where(peak_mask)[0]
    
    grad = np.zeros(n)
    for k in peak_indices:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += f[idx]
    grad /= len(peak_indices)
    return grad, max_c

    n = len(f)
    if use_all_constraints:
        A = np.zeros((2 * n - 1, n))

        c_conv = fft_convolve(f, f)

        A = np.zeros((len(peak_indices), n))

            jmax = min(k, n - 1)

                A[ii, j] = f[k - j]
        b_ub = np.full(len(peak_indices), M)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

    return np.ones(n) / n

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):

                best_f = f.copy()
        else:
            lr = max(0.005, lr * 0.88)
            stall += 1
            if stall % 8 == 0:
                # Small perturbation
                f = f * np.exp(rng.normal(0, 0.015, n))
                f = np.clip(f, 0, None)
                f = f * (S / np.sum(f))
                current_score = local_score(f)
            if stall > 80:
                f = f * np.exp(rng.normal(0, 0.03, n))
                f = np.clip(f, 0, None)
                f = f * (S / np.sum(f))
                current_score = local_score(f)
                lr = 0.03
                stall = 20

    return best_f, best_score


def momentum_subgradient(f0, time_limit, rng):
    """Subgradient descent with momentum on the peak."""
    n = len(f0)
    S = np.sqrt(2.0 * n)
    f = f0.astype(np.float64).copy()
    f = f * (S / np.sum(f))
    v = np.zeros(n)  # momentum
    beta = 0.7
    lr = 0.05
    current_score = local_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(f, f)
        max_c = np.max(c)
        threshold = max_c * 0.97
        peak_indices = np.where(c >= threshold)[0]

        # Subgradient
        grad = np.zeros(n)
        for k in peak_indices:
            jmin = max(0, k - n + 1)
            jmax = min(k, n - 1)
            if jmax >= jmin:
                grad[jmin:jmax + 1] += f[k - jmin:k - jmax + 1][::-1]
        grad /= len(peak_indices)

        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm

        # Momentum update
        v = beta * v + (1 - beta) * grad
        f_new = f - lr * v
        f_new = np.clip(f_new, 0, None)
        sf = np.sum(f_new)
        if sf < 0.01:
            break
        f_new = f_new * (S / sf)

        new_score = local_score(f_new)
        if new_score < current_score - 1e-9:
            f = f_new
            current_score = new_score
            lr = min(lr * 1.02, 0.15)
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()
        else:
            lr = max(0.01, lr * 0.9)
            if lr < 0.015:
                f = f * np.exp(rng.normal(0, 0.02, n))
                f = np.clip(f, 0, None)
                f = f * (S / np.sum(f))
                current_score = local_score(f)
                v *= 0.3
                lr = 0.05

    return best_f, best_score


def simulated_annealing(a_init, time_limit, rng):
    """Convolution-aware simulated annealing with diverse move types."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    current = a_init.astype(np.float64).copy()
    current = np.clip(current, 0, None)
    current = current * (S / np.sum(current))
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.0018
    cooling = 0.9994
    iter_count = 0

    while time.monotonic() - t0 < time_limit and temperature > 1e-11:
        candidate = current.copy()
        r = rng.random()
        iter_count += 1

        if r < 0.25:
            # Peak-targeted redistribution
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            jmin = max(0, peak_idx - n + 1)
            jmax = min(peak_idx, n - 1)
            contribs = current[jmin:jmax + 1] * current[peak_idx - jmin:peak_idx - jmax + 1][::-1]
            total_contrib = np.sum(contribs)
            if total_contrib > 1e-10:
                # Normalize contributions to get "importance"
                importance = contribs / total_contrib
                # Reduce top contributors
                n_reduce = max(4, n // 10)
                top_local = np.argsort(importance)[-n_reduce:]
                top_global = jmin + top_local
                red_amt = rng.uniform(0.02, 0.06)
                candidate[top_global] *= (1 - red_amt)
                # Increase random others
                others = np.setdiff1d(np.arange(n), top_global)
                if len(others) > 3:
                    n_inc = min(12, len(others) // 4)
                    inc_idx = rng.choice(others, size=n_inc, replace=False)
                    inc_amt = red_amt * n_reduce / n_inc
                    candidate[inc_idx] *= (1 + inc_amt)
        elif r < 0.40:
            # Pair swap (mass transfer)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amt = rng.uniform(0, 0.15) * current[i]
                candidate[i] = max(0.0, current[i] - amt)
                candidate[j] = current[j] + amt
        elif r < 0.50:
            # Asymmetry shift: blend with reversed version
            blend = rng.uniform(0.05, 0.25)
            candidate = (1 - blend) * current + blend * current[::-1]
        elif r < 0.60:
            # Block perturbation
            n_blocks = int(rng.integers(2, 8))
            block_size = n // n_blocks
            for b in range(n_blocks):
                start = b * block_size
                end = min(start + block_size, n)
                candidate[start:end] *= np.exp(rng.normal(0, 0.02, end - start))
        elif r < 0.72:
            # Trilinear: 3 reduce, 3 increase
            idx = rng.choice(n, size=6, replace=False)
            amts = rng.uniform(0.02, 0.07, size=3)
            for i in range(3):
                candidate[idx[i]] *= (1 - amts[i])
                candidate[idx[i + 3]] *= (1 + amts[i])
        elif r < 0.85:
            # Frequency-domain perturbation
            F = np.fft.rfft(current)
            phase_pert = rng.normal(0, 0.05, len(F))
            F_pert = F * np.exp(1j * phase_pert)
            candidate = np.fft.irfft(F_pert, n)
            candidate = np.clip(candidate, 0, None)
        else:
            # Small global noise
            candidate = current * np.exp(rng.normal(0, 0.008, n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        sc = np.sum(candidate)
        if sc < 0.01:
            temperature *= cooling
            continue
        candidate = candidate * (S / sc)

        new_score = local_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                temperature = max(temperature * 1.03, 0.001)
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling

    return best, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_asymmetric_init(n, rng, strength=0.3):
    """Strongly asymmetric initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + strength * (2 * t / n - 1)
    a += 0.2 * np.cos(2 * np.pi * rng.uniform(0.5, 2.0) * t / n)
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_exponential_init(n, rng, rate=1.0):
    """Exponentially decaying initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * t / n)
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_block_init(n, rng, n_blocks=5):
    """Block-structured initialization."""
    block_size = n // n_blocks
    a = np.zeros(n)
    for i in range(n_blocks):
        start = i * block_size
        end = min(start + block_size, n)
        a[start:end] = rng.uniform(0.3, 1.5)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def diversity_search(start_seq, time_limit, rng):
    """Try diverse initializations and optimize each briefly."""
    n_base = len(start_seq)
    best = start_seq.copy()
    best_score = local_score(start_seq)
    t0 = time.monotonic()

    inits = []
    # Chirp with various parameters
    for alpha in [0.3, 0.5, 0.7, 1.0]:
        inits.append(make_chirp_init(n_base, alpha, rng))
    # Asymmetric
    for s in [0.2, 0.4, 0.6]:
        inits.append(make_asymmetric_init(n_base, rng, strength=s))
    # Exponential
    for r in [0.5, 1.0, 2.0, 3.0]:
        inits.append(make_exponential_init(n_base, rng, rate=r))
    # Blocks
    inits.append(make_block_init(n_base, rng, n_blocks=3))
    inits.append(make_block_init(n_base, rng, n_blocks=7))
    # Random
    inits.append(make_random_init(n_base, rng, spread=0.5))
    inits.append(make_random_init(n_base, rng, spread=1.0))
    # Resampled best
    inits.append(resample_sequence(start_seq, n_base) + 0.05 * rng.standard_normal(n_base))

    n_inits = len(inits)
    time_per_init = max(5.0, (time_limit - 10) / n_inits)

    for i, init in enumerate(inits):
        if time.monotonic() - t0 > time_limit - 20:
            break
        t_remaining = time_limit - (time.monotonic() - t0) - 10
        t_this = min(time_per_init, t_remaining)
        if t_this < 3:
            break

        # Quick LP + short SA
        f_res, s_res = lp_iterate(init, t_this * 0.6, rng)
        if s_res < best_score:
            best_score = s_res
            best = f_res.copy()
            print(f"  diversity init {i}: score={s_res:.6f} (NEW BEST)")

        # Short SA if time allows
        t_sa = t_this * 0.4
        if t_sa > 5:
            f_sa, s_sa = simulated_annealing(f_res, t_sa, rng)
            if s_sa < best_score:
                best_score = s_sa
                best = f_sa.copy()
                print(f"  diversity init {i} SA: score={s_sa:.6f} (NEW BEST)")

    return best, best_score


def try_different_length(seq, target_n, time_limit, rng):
    """Try optimizing at a different length."""
    new_seq = resample_sequence(seq, target_n)
    f_res, s_res = lp_iterate(new_seq, time_limit * 0.6, rng)
    if time_limit * 0.4 > 5:
        f_sa, s_sa = simulated_annealing(f_res, time_limit * 0.4, rng)
        s_res = min(s_res, s_sa)
        if s_sa < s_res:
            f_res = f_sa
    return f_res, s_res


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function combining LP iteration, redistribution descent,
    momentum subgradient, and convolution-aware simulated annealing.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0:
            s = local_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    if start_seq is None:
        # Fallback initialization
        n = 512
        best_seq = make_chirp_init(n, 0.5, rng)
        best_score = local_score(best_seq)
        print(f"Generated fallback: n={n}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP iteration on best sequence (primary optimization) ===
    print(f"\n=== Phase 1: LP iteration ===")
    if time.monotonic() < deadline - 500:
        t1 = min(350.0, deadline - time.monotonic() - 500)
        print(f"  Time budget: {t1:.0f}s")
        f_best, s_best = lp_iterate(best_seq, t1, rng, verbose=True)
        print(f"  LP result: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 2: Redistribution descent (novel method) ===
    print(f"\n=== Phase 2: Redistribution descent ===")
    if time.monotonic() < deadline - 400:
        t2 = min(150.0, deadline - time.monotonic() - 400)
        print(f"  Time budget: {t2:.0f}s")
        f_best, s_best = redistribution_descent(best_seq, t2, rng)
        print(f"  Redistribution result: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 3: Momentum subgradient ===
    print(f"\n=== Phase 3: Momentum subgradient ===")
    if time.monotonic() < deadline - 300:
        t3 = min(100.0, deadline - time.monotonic() - 300)
        f_best, s_best = momentum_subgradient(best_seq, t3, rng)
        print(f"  Momentum subgrad result: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: SA polish ===
    print(f"\n=== Phase 4: Simulated annealing ===")
    if time.monotonic() < deadline - 250:
        t4 = min(180.0, deadline - time.monotonic() - 250)
        print(f"  Time budget: {t4:.0f}s, starting score={best_score:.6f}")
        polished, polish_score = simulated_annealing(best_seq, t4, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  *** After SA: {polish_score:.6f} (NEW BEST) ***")
        else:
            print(f"  After SA: {polish_score:.6f} (no improvement)")

    # === Phase 5: Diversity search ===
    print(f"\n=== Phase 5: Diversity search ===")
    if time.monotonic() < deadline - 200:
        t5 = min(200.0, deadline - time.monotonic() - 200)
        print(f"  Time budget: {t5:.0f}s")
        div_best, div_score = diversity_search(best_seq, t5, rng)
        if div_score < best_score:
            best_score = div_score
            best_seq = div_best.copy()
            print(f"  *** Diversity found: {div_score:.6f} (NEW BEST) ***")
        else:
            print(f"  Diversity best: {div_score:.6f}")

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Different lengths ===")
    if time.monotonic() < deadline - 100:
        t6_total = min(120.0, deadline - time.monotonic() - 100)
        alt_lengths = [n_base * 3 // 4, n_base * 3 // 2, n_base * 2]
        alt_lengths = [l for l in alt_lengths if 64 <= l <= 2048 and l != n_base]
        if alt_lengths:
            t_per_len = t6_total / len(alt_lengths)
            for alt_n in alt_lengths:
                if time.monotonic() > deadline - 60:
                    break
                print(f"  Trying n={alt_n}...")
                f_alt, s_alt = try_different_length(best_seq, alt_n, t_per_len, rng)
                print(f"  n={alt_n}: score={s_alt:.6f}")
                if s_alt < best_score:
                    best_score = s_alt
                    best_seq = f_alt.copy()
                    print(f"  *** New best at n={alt_n}: {best_score:.6f} ***")

    # === Phase 7: Final refinement ===
    print(f"\n=== Phase 7: Final refinement ===")
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 20
        # LP
        t_lp = remaining * 0.35
        f_best, s_best = lp_iterate(best_seq, t_lp, rng)
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()

        # Redistribution
        t_rd = remaining * 0.25
        f_best, s_best = redistribution_descent(best_seq, t_rd, rng)
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()

        # SA
        t_sa = remaining * 0.35
        polished, polish_score = simulated_annealing(best_seq, t_sa, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()

        print(f"  Final refinement best: {best_score:.6f}")

    # === Final verification ===
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        elapsed = time.monotonic() - t_global
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)}, "
              f"elapsed={elapsed:.1f}s ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
