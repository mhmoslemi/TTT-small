```python
"""
Optimization of non-negative sequences to minimize peak linear autocorrelation.
Uses subgradient descent with Nesterov momentum, LP-based direction finding,
and coordinate descent for polishing. Multi-start across sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Bisection on lambda
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - (lo + hi) / 2.0, 0.0)


def subgradient_descent(p_init, time_limit, verbose=True, seed=42):
    """
    Subgradient descent on the simplex to minimize max_k conv(p,p)[k].
    p is normalized to sum=1. Score = 2n * max_k conv(p,p)[k].
    """
    n = len(p_init)
    p = p_init.astype(np.float64).copy()
    p = np.clip(p, 0.0, None)
    S = p.sum()
    if S < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / S
    
    best_p = p.copy()
    best_val = float(np.max(fft_convolve(p, p)))
    best_score = 2.0 * n * best_val
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Nesterov parameters
    beta = 0.9
    momentum = np.zeros(n)
    lr = 0.01
    top_k = 5
    stall = 0
    max_stall = 50
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        c = fft_convolve(p, p)
        val = float(np.max(c))
        
        # Find top-k active constraints
        idx = np.argsort(c)[-top_k:]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in idx:
            k = int(k)
            # d conv[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            for j in range(n):
                term = 0.0
                if 0 <= k - j < n:
                    term += p[k - j]
                if 0 <= j - k < n:
                    term += p[j - k]
                g[j] += term
        g /= float(top_k)
        
        # Update with momentum
        momentum = beta * momentum + (1.0 - beta) * g
        p_new = p - lr * momentum
        p_new = project_simplex(p_new)
        
        # Evaluate
        c_new = fft_convolve(p_new, p_new)
        val_new = float(np.max(c_new))
        score_new = 2.0 * n * val_new
        
        if score_new < best_score:
            best_score = score_new
            best_p = p_new.copy()
            best_val = val_new
            stall = 0
            p = p_new
        else:
            p = p_new  # still move, but track stall
            stall += 1
        
        it += 1
        
        # Adaptive learning rate
        if stall > 0 and stall % 10 == 0:
            lr = max(0.0001, lr * 0.95)
        
        # Periodic perturbation to escape local minima
        if stall >= max_stall:
            pert = rng.standard_normal(n) * 0.05
            p = project_simplex(p + pert)
            lr = 0.01
            stall = 0
            max_stall = max(20, max_stall // 2)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [SGD {elapsed:6.1f}s] it={it:5d} val={val_new:.6f} best={best_score:.6f} lr={lr:.6f}")
    
    return best_p, best_score
```

```python
"""
Optimization of non-negative sequences to minimize peak linear autocorrelation.
Strategy: subgradient descent with Nesterov momentum on the probability simplex,
LP-based direction finding for refinement, and coordinate descent for final polish.
Multi-start across different sequence lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} via bisection."""
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.maximum(v - lam, 0.0)


def compute_subgrad(p, top_indices):
    """Compute subgradient of max_{k in top_indices} conv(p,p)[k] w.r.t. p."""
    n = len(p)
    g = np.zeros(n)
    for l in top_indices:
        l = int(l)
        if l < n:
            g[:l+1] += p[l::-1]
        else:
            g[l-n+1:] += p[n-1 : l-n : -1]
    return 2.0 * g / float(len(top_indices))


def sgd_optimize(p_init, time_limit, verbose=False, seed=42, top_k=5):
    """Subgradient descent on simplex to minimize 2n*max(conv(p,p))."""

    else:
        p = p / S

    lr = 0.05

    max_stall = 100

        g = compute_subgrad(y, top_idx)

def build_A(f):

    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
```

```python
"""
Optimization of non-negative sequences to minimize peak linear autocorrelation.
Uses softmax-parameterized subgradient descent with Nesterov momentum,
LP-based direction finding, and multi-start across lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def softmax(w):
    """Stable softmax."""
    w_shift = w - w.max()
    exp_w = np.exp(w_shift)
    return exp_w / exp_w.sum()


def compute_subgrad(p, top_indices):
    """Compute subgradient of max_{k in top_indices} conv(p,p)[k] w.r.t. p.
    d conv[k]/dp[j] = 2*p[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(p)
    g = np.zeros(n)
    for l in top_indices:
        l = int(l)
        if l < n:
            g[:l + 1] += p[l::-1]
        else:
            g[l - n + 1:] += p[n - 1 : l - n : -1]
    return 2.0 * g / float(len(top_indices))


def softmax_sgd(w_init, time_limit, verbose=False, seed=42, top_k=7):
    """Subgradient descent using softmax parameterization.
    Minimizes 2n * max_k conv(p,p)[k] where p = softmax(w).
    """
    n = len(w_init)
    w = w_init.astype(np.float64).copy()
    p = softmax(w)

    best_p = p.copy()
    best_score = 2.0 * n * float(np.max(fft_convolve(p, p)))

    t0 = time.monotonic()
    lr = 0.3
    momentum = np.zeros(n)
    beta = 0.85
    stall = 0
    it = 0
    rng = np.random.default_rng(seed)
    best_lr = lr
    max_stall = 150

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.3:
            break

        c = fft_convolve(p, p)
        top_idx = np.argsort(c)[-top_k:]
        g_p = compute_subgrad(p, top_idx)

        # Chain rule through softmax: g_w[j] = g_p[j]*p[j] - p[j]*sum(g_p*p)
        gp_dot = float(np.dot(g_p, p))
        g_w = p * (g_p - gp_dot)

        # Nesterov momentum
        momentum = beta * momentum + g_w
        w = w - lr * momentum
        p = softmax(w)

        # Evaluate
        c = fft_convolve(p, p)
        score = 2.0 * n * float(np.max(c))

        if score < best_score:
            best_score = score
            best_p = p.copy()
            stall = 0
        else:
            stall += 1

        it += 1

        # Adaptive learning rate
        if stall > 20:
            lr = best_lr * (0.998 ** (stall - 20))
            if lr < 0.01:
                lr = 0.01

        # Perturbation to escape local minima
        if stall >= max_stall:
            w += rng.standard_normal(n) * 0.3
            p = softmax(w)
            momentum = np.zeros(n)
            lr = best_lr
            stall = 0
            max_stall = min(300, max_stall + 100)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [SGD {elapsed:6.1f}s] it={it:6d} score={score:.6f} best={best_score:.6f} lr={lr:.4f}")

    return best_p, best_score


def build_A(f):
    """Build matrix A where A@b = conv(f,b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refine(f_init, time_limit, verbose=False):
    """LP-based refinement."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-15:
        f = np.ones(n) / n
        S = 1.0
    f = f / S  # normalize to sum = 1

    best_f = f.copy()
    best_score = 2.0 * n * float(np.max(fft_convolve(f, f)))
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 8.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))

        A = build_A(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': 8.0, 'presolve': True})
            if not res.success:
                break
            b = res.x
        except Exception:
            break

        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b / Sb

        # Line search
        best_t = 0.1
        best_s = np.inf
        for t_try in [0.05, 0.1, 0.15, 0.2, 0.3]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.0, None)
            Sc = f_cand.sum()
            if Sc < 1e-10:
                continue
            f_cand = f_cand / Sc
            s_cand = 2.0 * n * float(np.max(fft_convolve(f_cand, f_cand)))
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, None)
        S_new = f.sum()
        if S_new < 1e-10:
            break
        f = f / S_new
        it += 1

        if best_s < best_score - 1e-12:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if verbose and it % 5 == 0:
            print(f"    [LP {time.monotonic()-t0:6.1f}s] it={it} score={best_s:.6f} best={best_score:.6f}")

        if stall >= 15:
            break

    return best_f, best_score


def grad_polish(f_init, time_limit, verbose=False):
    """Additional gradient polish using subgradient with Bregman-like steps."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = f.sum()
    f = f / S

    best_f = f.copy()
    best_score = 2.0 * n * float(np.max(fft_convolve(f, f)))
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    lr = 0.02
    momentum = np.zeros(n)
    beta = 0.9
    stall = 0
    it = 0
    top_k = 5
    max_stall = 80

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break

        p = f
        c = fft_convolve(p, p)
        top_idx = np.argsort(c)[-top_k:]
        g = compute_subgrad(p, top_idx)
        # Center the gradient (removes component in the all-ones direction)
        g = g - np.mean(g)

        momentum = beta * momentum + g
        f_new = f - lr * momentum
        f_new = np.clip(f_new, 0.0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            break
        f_new = f_new / S_new

        c_new = fft_convolve(f_new, f_new)
        score = 2.0 * n * float(np.max(c_new))

        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            f = f_new
            stall += 1

        it += 1

        if stall > 10:
            lr = max(0.001, lr * 0.997)

        if stall >= max_stall:
            pert = rng.standard_normal(n) * 0.05
            f = np.clip(f + pert, 0.0, None)
            S_new = f.sum()
            if S_new < 1e-10:
                break
            f = f / S_new
            momentum = np.zeros(n)
            lr = 0.02
            stall = 0
            max_stall = min(200, max_stall + 50)

        if verbose and it % 500 == 0:
            print(f"    [Polish {time.monotonic()-t0:6.1f}s] it={it:5d} score={score:.6f} best={best_score:.6f}")

    return best_f, best_score


def make_init(n, init_type, rng):
    """Generate initial weight vector (for softmax parameterization)."""
    if init_type == 'uniform':
        return np.zeros(n)
    elif init_type == 'random':
        return rng.standard_normal(n) * 0.5
    elif init_type == 'edge_heavy':
        t = np.arange(n, dtype=np.float64)
        return 0.5 * t / n  # log-linear increasing
    elif init_type == 'v_shape':
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        return -0.5 * np.abs(t - center) / n  # V-shaped (high at edges)
    elif init_type == 'hann_neg':
        t = np.arange(n, dtype=np.float64)
        return -0.3 * np.cos(2.0 * np.pi * t / max(1, n - 1))
    elif init_type == 'chirp':
        t = np.arange(n, dtype=np.float64)
        return 0.2 * np.cos(2.0 * np.pi * t * t / (2.0 * n))
    elif init_type == 'random_sparse':
        w = np.zeros(n)
        n_nonzero = max(10, n // 4)
        idx = rng.choice(n, size=n_nonzero, replace=False)
        w[idx] = rng.standard_normal(n_nonzero)
        return w
    else:
        return rng.standard_normal(n) * 0.3


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: softmax SGD + LP refinement + gradient polish."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_p = None
    best_n = 0

    # Check for provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.asarray(height_sequence_1, dtype=np.float64)
        provided = np.clip(provided, 0.0, 1000.0)
        S_prov = provided.sum()
        if S_prov > 0.01:
            provided_norm = provided / S_prov
            s_prov = 2.0 * len(provided) * float(np.max(fft_convolve(provided_norm, provided_norm)))
            print(f"Provided: n={len(provided)}, score={s_prov:.6f}")
            if s_prov < best_score:
                best_score = s_prov
                best_p = provided_norm.copy()
                best_n = len(provided)

    # Phase 1: Softmax SGD multi-start
    print("\n=== Phase 1: Softmax SGD ===")

    # Define tasks: (n, init_type)
    tasks = []
    if provided is not None:
        n_prov = len(provided)
        tasks.append((n_prov, 'provided'))
        tasks.append((n_prov, 'random'))
        tasks.append((n_prov, 'edge_heavy'))

    for n in [256, 384, 512, 768, 1024]:
        tasks.append((n, 'uniform'))
        tasks.append((n, 'random'))
        tasks.append((n, 'v_shape'))
        tasks.append((n, 'edge_heavy'))
        tasks.append((n, 'hann_neg'))
        tasks.append((n, 'chirp'))
        tasks.append((n, 'random_sparse'))

    # Time allocation: prioritize smaller n (faster iterations) and diverse inits
    # Give more time to n=256,384,512 and less to 768,1024
    time_weights = []
    for n, init in tasks:
        base = 1.0 / (n / 128.0)
        if init == 'uniform':
            base *= 0.7
        elif init in ('random', 'random_sparse'):
            base *= 1.2
        time_weights.append(base)
    total_w = sum(time_weights)

    results = []

    for idx, (n, init_type) in enumerate(tasks):
        if time.monotonic() > deadline - 400:
            print(f"  Stopping SGD phase (time for LP)")
            break

        # Compute time allocation
        remaining = deadline - time.monotonic() - 400
        if remaining <= 0:
            break
        fractional_time = sum(time_weights[idx:]) / sum(time_weights[idx:])
        t_alloc = remaining * (time_weights[idx] / sum(time_weights[idx:]))
        t_alloc = max(10.0, min(t_alloc, 200.0))

        # Generate initialization
        if init_type == 'provided' and provided is not None and len(provided) == n:
            # Convert to log-space for softmax
            p_prov = provided / provided.sum()
            w0 = np.log(p_prov + 1e-10)
        else:
            w0 = make_init(n, init_type, rng)

        verbose = (t_alloc > 60)
        p_best, s_best = softmax_sgd(w0, t_alloc, verbose=verbose,
                                      seed=seed + idx * 31, top_k=7)
        results.append((s_best, p_best, n, init_type))

        if s_best < best_score:
            best_score = s_best
            best_p = p_best.copy()
            best_n = n
            print(f"  *** n={n} init={init_type:12s} score={s_best:.6f} (NEW BEST)")
        else:
            print(f"  n={n} init={init_type:12s} score={s_best:.6f}")

    # Phase 2: LP refinement on best sequence
    if best_p is not None and time.monotonic() < deadline - 100:
        print(f"\n=== Phase 2: LP Refinement (n={best_n}) ===")
        lp_time = min(400.0, deadline - time.monotonic() - 100)
        if lp_time > 20:
            f_lp, s_lp = lp_refine(best_p, lp_time, verbose=True)
            if s_lp < best_score:
                best_score = s_lp
                best_p = f_lp.copy()
                print(f"  LP improved: {s_lp:.6f}")
            else:
                print(f"  LP no improvement: {s_lp:.6f}")

    # Phase 3: Gradient polish
    if best_p is not None and time.monotonic() < deadline - 50:
        print(f"\n=== Phase 3: Gradient Polish ===")
        pol_time = min(150.0, deadline - time.monotonic() - 50)
        if pol_time > 20:
            f_pol, s_pol = grad_polish(best_p, pol_time, verbose=True)
            if s_pol < best_score:
                best_score = s_pol
                best_p = f_pol.copy()
                print(f"  Polish improved: {s_pol:.6f}")
            else:
                print(f"  Polish no improvement: {s_pol:.6f}")

    # Phase 4: Final short SGD restart
    if best_p is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Phase 4: Final SGD Restart ===")
        final_time = min(60.0, deadline - time.monotonic() - 15)
        if final_time > 10:
            n = len(best_p)
            w0 = np.log(best_p + 1e-10)
            p_fin, s_fin = softmax_sgd(w0, final_time, verbose=False,
                                        seed=seed + 9999, top_k=10)
            if s_fin < best_score:
                best_score = s_fin
                best_p = p_fin.copy()
                print(f"  Final SGD improved: {s_fin:.6f}")

    # Finalize
    if best_p is not None:
        # Scale back: the sequence can be any positive scaling
        # Use sum = sqrt(2n) for the evaluate_sequence call
        n = len(best_p)
        S = best_p.sum()
        final_seq = best_p * (np.sqrt(2.0 * n) / S)
        final_list = [float(max(0.0, min(1000.0, x))) for x in final_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    # Fallback
    return [1.0] * 256
```
