"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast multiplicative weight descent
with temperature annealing, diverse initializations, multi-scale search, and
LP-based final polish. Explores sequence lengths up to 2048.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_hard(a):
    """Objective: 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def compute_soft_grad(a, tau):
    """Compute log-sum-exp soft objective and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return 0.0, np.zeros(n)
    exp_c = np.exp((conv - max_c) / tau)
    sum_exp = exp_c.sum()
    if sum_exp < 1e-300:
        return 2.0 * n * max_c, np.zeros(n)
    p = exp_c / sum_exp
    soft_max = max_c + tau * np.log(sum_exp)
    obj = 2.0 * n * soft_max
    
    # Gradient: (4n/tau) * sum_j a[j] * p[j+m] for m = 0,...,n-1
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(A * np.conj(P), fft_size)[:n]
    grad = (4.0 * n / tau) * cross
    return obj, grad


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0.0).copy()
    s = v.sum()
    if s < 1e-15:
        return np.full(len(v), target_sum / len(v))
    return target_sum * v / s


def multiplicative_weight_descent(a0, time_limit, seed=0):
    """
    Fast multiplicative weight descent with adaptive step size and
    temperature annealing. Returns (best_a, best_obj).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= a.sum()
    
    n = len(a)
    best_a = a.copy()
    best_obj = obj_hard(a)
    t0 = time.monotonic()

    conv0 = fast_convolve(a)
    max0 = conv0.max()
    min0 = conv0.min()
    spread = max(1e-8, max0 - min0)

    max_no_improve = 3000

    max_restarts = 30
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        obj_soft, grad = compute_soft_grad(a, tau)
        
        if not np.all(np.isfinite(grad)) or not np.all(np.isfinite(a)):
            break
        
        # Adaptive step size
        max_grad = np.abs(grad).max()
        if max_grad < 1e-15:
            tau *= 0.5
            continue
        eta = 0.5 / (max_grad + 1e-12)
        eta = min(eta, 2.0 / n)

            a = a_new
            no_improve = 0

def solve_lp_fast(f, M, max_iter=5000):
    """
    Fast LP: find g >= 0, sum(g)=1 maximizing... 
    Actually, let's solve: max sum(g) s.t. conv(f,g)[k] <= M, g >= 0.
    Then normalize g.
    Use scipy linprog for small n, or a simple iterative method.
    """
    n = len(f)
    if n > 2048:
        # Too large for LP, use a simpler approach
        return f.copy()
    
    # Build constraint matrix
    # conv(f, g)[k] = sum_j f[j] * g[k-j] <= M
    # For k = 0, ..., 2n-2
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, j + n):
            if k < 2 * n - 1:
                A[k, j] = f[k - j]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_refine(f0, time_limit):

    f = np.maximum(f0.astype(np.float64), 0.0).copy()

    if s < 1e-12:

        M = conv.max()
        
        g = solve_lp_fast(f, M)
        
        if np.allclose(g, f, atol=1e-10):
            no_improve += 1
            if no_improve > 20:
                break
        else:
            # Line search
            best_t = 0
            best_obj_t = obj_hard(f)
            for t in [0.1, 0.25, 0.5, 0.75, 1.0]:
                a_try = (1 - t) * f + t * g
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-12:
                    continue
                a_try /= s_try
                obj_try = obj_hard(a_try)
                if obj_try < best_obj_t:
                    best_obj_t = obj_try
                    best_t = t
            
            if best_t > 0 and best_obj_t < best_obj - 1e-12:
                f = (1 - best_t) * f + best_t * g
                f = np.maximum(f, 0)
                f /= f.sum()
                best_obj = best_obj_t
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
    
    return best_f, best_obj

        order = rng.permutation(n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast multiplicative weight descent
with temperature annealing, diverse structured initializations, multi-scale search
(n up to 2048), and LP-based final polish. Combines subgradient ideas with
preconditioned updates for fast convergence.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_hard(a):
    """Objective: 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def compute_soft_obj_grad(a, tau):
    """Compute log-sum-exp soft objective and gradient w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return 0.0, np.zeros(n)
    shifted = (conv - max_c) / tau
    shifted = np.maximum(shifted, -500)
    exp_c = np.exp(shifted)
    sum_exp = exp_c.sum()
    if sum_exp < 1e-300:
        return 2.0 * n * max_c, np.zeros(n)
    p = exp_c / sum_exp
    soft_max = max_c + tau * np.log(sum_exp)
    obj = 2.0 * n * soft_max

    # Gradient: d_obj/d_a[m] = (4n/tau) * sum_j a[j] * p[j+m]
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(A * np.conj(P), fft_size)[:n]
    grad = (4.0 * n / tau) * cross
    return obj, grad


def multiplicative_weight_descend(a0, time_limit, seed=0):
    """
    Fast multiplicative weight descent with adaptive step size,
    preconditioning, and temperature annealing.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    n = len(a)
    best_a = a.copy()
    best_obj = obj_hard(a)
    t0 = time.monotonic()

    conv0 = fast_convolve(a)
    spread0 = max(1e-10, conv0.max() - conv0.min())
    tau = 0.15 * spread0

    no_improve = 0
    max_no_improve = 4000
    n_restarts = 0
    max_restarts = 40

    while time.monotonic() - t0 < time_limit:
        obj_soft, grad = compute_soft_obj_grad(a, tau)

        if not np.all(np.isfinite(grad)) or not np.all(np.isfinite(a)):
            break

        max_grad = np.abs(grad).max()
        if max_grad < 1e-15:
            tau *= 0.5
            continue

        # Adaptive step size with preconditioning
        # Use sqrt preconditioning: effective step is larger for small a[m]
        precond = 1.0 / (np.sqrt(a + 1e-12) + 1e-6)
        effective_grad = grad * precond
        max_eff = np.abs(effective_grad).max()
        if max_eff < 1e-15:
            break
        eta = 0.8 / (max_eff + 1e-12)
        eta = min(eta, 5.0 / n)

        # Multiplicative update
        log_w = -eta * effective_grad
        log_w = np.clip(log_w, -40, 40)
        a_new = a * np.exp(log_w)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s_new

        obj_new = obj_hard(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        elif obj_new <= best_obj:
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Temperature annealing
        tau *= 0.9992
        tau = max(tau, 1e-5 * fast_convolve(a).max())

        # Restart mechanism
        if no_improve >= max_no_improve:
            n_restarts += 1
            if n_restarts > max_restarts:
                break
            no_improve = 0
            noise = rng.dirichlet(np.ones(n) * 0.3)
            mix = 0.05 + 0.15 * min(1.0, n_restarts / 20.0)
            a = (1.0 - mix) * best_a + mix * noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
            conv_a = fast_convolve(a)
            tau = 0.2 * max(1e-10, conv_a.max() - conv_a.min())

    return best_a, best_obj


def solve_lp_direction(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)[k] <= M for all k."""
    n = len(f)
    if n > 4096:
        return f.copy()

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                     method='highs', options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_refine(f0, time_limit):
    """LP-based iterative refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones_like(f) / len(f)
    else:
        f /= s

    best_f = f.copy()
    best_obj = obj_hard(f)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(f, M)

        if np.allclose(g, f, atol=1e-10):
            no_improve += 1
            if no_improve > 30:
                break
        else:
            best_t = 0.0
            best_obj_t = obj_hard(f)
            for t in [0.05, 0.1, 0.25, 0.5, 0.75, 1.0]:
                a_try = (1.0 - t) * f + t * g
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-12:
                    continue
                a_try /= s_try
                obj_try = obj_hard(a_try)
                if obj_try < best_obj_t - 1e-12:
                    best_obj_t = obj_try
                    best_t = t

            if best_t > 0:
                f = (1.0 - best_t) * f + best_t * g
                f = np.maximum(f, 0.0)
                f /= f.sum()
                best_obj = best_obj_t
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

    return best_f, best_obj


def targeted_local_search(a0, time_limit, seed=0):
    """
    Targeted local search: identify peak-contributing coordinates and

    """

        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        # Compute contribution of each pair to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        # Identify danger (high contrib) and safe (low contrib) coordinates
        thr_high = 0.5 * max_c
        thr_low = 0.1 * max_c
        danger = np.where(contrib > thr_high)[0]
        safe = np.where(contrib < thr_low)[0]

        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue

        found = False
        # Try moving mass from danger to safe coordinates
        for di in danger[:20]:
            if found:
                break
            if a[di] < 1e-12:
                continue
            for si in safe[:20]:
                if found:
                    break
                if di == si:
                    continue

                    if amount < 1e-15:

                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount

                    ts = a_trial.sum()
                    if ts < 1e-12:

                    a_trial /= ts
                    o = obj_hard(a_trial)
                    if o < best_obj - 1e-10:

                        a = a_trial.copy()
                        best_a = a.copy()
                        found = True
                        break

        else:
            stuck = 0

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Bimodal Gaussians with different parameters
    for s in [0.3, 0.5, 0.8, 1.2, 2.0]:
        dm = np.minimum(x, n - 1 - x)
        g = np.exp(-s * dm / n)
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs > 1e-10:
            inits[f'bimodal_{s}'] = g / gs

    # Power-law edges
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        dm = np.minimum(x, n - 1 - x) + 1.0
        pw = dm ** (-p)
        pw = np.maximum(pw, 0.0)
        ps = pw.sum()
        if ps > 1e-10:
            inits[f'power_{p}'] = pw / ps

    # Cosine raised to power
    if n > 1:
        for p in [0.5, 1.0, 2.0, 4.0]:
            c = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
            c = np.maximum(c, 0.0) ** p
            cs = c.sum()
            if cs > 1e-10:
                inits[f'cos_{p}'] = c / cs

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Random Dirichlet
    inits['rand_dir'] = rng.dirichlet(np.ones(n) * 0.2)

    # Block-based random
    nblocks = max(2, n // 64)
    block_size = n // nblocks
    block_heights = rng.dirichlet(np.ones(nblocks) * 0.5)
    blocks = np.zeros(n)
    for bi in range(nblocks):
        start = bi * block_size
        end = min(n, start + block_size)
        blocks[start:end] = block_heights[bi]
    bs = blocks.sum()
    if bs > 1e-10:
        inits['blocks'] = blocks / bs

    # Sinc-like
    xc = x - 0.5 * (n - 1)
    sinc = np.sinc(xc / (n * 0.1))
    sinc = np.maximum(sinc, 0.0)
    ss = sinc.sum()
    if ss > 1e-10:
        inits['sinc'] = sinc / ss

    # Parent resampled
    if parent is not None:

    return inits

            return None
        if n == 512:
            return parent_512.copy()
        x_new = np.linspace(0, 1, n)
        x_old = np.linspace(0, 1, 512)
        pc = np.interp(x_new, x_old, parent_512)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s

    # === Phase 1: Fast scan (50% of budget) ===
    phase1_end = time.monotonic() + total_budget * 0.50
    print(f"Phase 1: Fast multiplicative weight descent scan...")

    all_configs = []
    for n in lengths:
        p_n = get_parent_for_n(n)
        inits = generate_initializations(n, rng, parent=p_n)
        for name, a0 in inits.items():
            all_configs.append((n, name, a0))

    # Prioritize: parent first, then shorter n, then structured inits
    def config_priority(item):
        n, name, _ = item
        if name == 'parent':
            return (0, n)
        elif name.startswith('bimodal') or name.startswith('power'):
            return (1, n)
        elif name in ['uniform', 'triangle', 'cos_1.0']:
            return (2, n)
        else:
            return (3, n)

    all_configs.sort(key=config_priority)

    n_configs = len(all_configs)
    phase1_time = phase1_end - time.monotonic()
    time_per_config = max(2.0, phase1_time / max(1, n_configs))

    scan_results = []
    for i, (n, name, a0) in enumerate(all_configs):
        remaining = deadline - time.monotonic() - 60.0
        if remaining < 30:
            break
        t_alloc = min(time_per_config, remaining * 0.8)
        t_alloc = max(1.0, t_alloc)

        a_opt, obj_opt = multiplicative_weight_descend(a0, time_limit=t_alloc, seed=seed + i)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  n={n}, {name}: score={score:.6f} (obj_norm={obj_opt:.6f})")

    print(f"  Phase 1 done. Best so far: {best_score:.6f}")

    # === Phase 2: LP refinement on top candidates (30% of budget) ===
    phase2_end = time.monotonic() + total_budget * 0.30
    print(f"Phase 2: LP refinement on top candidates...")

    scan_results.sort(key=lambda x: x[0])
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, name, a_opt))
            seen.add(key)
        if len(top_configs) >= 10:
            break

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining2 = phase2_end - time.monotonic()
    if remaining2 < 10:
        remaining2 = 10
    time_per_lp = max(5.0, remaining2 / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        remaining = deadline - time.monotonic() - 60.0
        if remaining < 20:
            break
        t_alloc = min(time_per_lp, remaining * 0.8)
        t_alloc = max(3.0, t_alloc)

        # LP refinement
        a_lp, obj_lp = lp_refine(a0, time_limit=t_alloc)
        score_lp = float(evaluate_sequence(a_lp.tolist()))
        if np.isfinite(score_lp) and score_lp < best_score:
            best_score = score_lp
            best_sequence = a_lp.copy()
            print(f"  LP n={n}, {name}: score={score_lp:.6f}")

        # Targeted local search on LP result
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 5:
            ls_time = min(20.0, remaining * 0.3)
            a_ls, obj_ls = targeted_local_search(a_lp, time_limit=ls_time, seed=seed + 100 + i)
            score_ls = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(score_ls) and score_ls < best_score:
                best_score = score_ls
                best_sequence = a_ls.copy()
                print(f"  LS n={n}, {name}: score={score_ls:.6f}")

    print(f"  Phase 2 done. Best so far: {best_score:.6f}")

    # === Phase 3: Intensive polish on best (remaining time) ===
    print(f"Phase 3: Intensive polish...")
    if best_sequence is not None:
        polish_deadline = deadline - 10.0
        polish_iter = 0

        while time.monotonic() < polish_deadline:
            remaining = polish_deadline - time.monotonic()
            if remaining < 10:
                break

            # Alternate between MWD, LP, and local search
            strategy = polish_iter % 3

            if strategy == 0:
                # MWD with smaller restarts
                t_alloc = min(remaining * 0.4, 120.0)
                a_opt, obj_opt = multiplicative_weight_descend(
                    best_sequence, time_limit=t_alloc, seed=seed + 500 + polish_iter)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score - 1e-8:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish MWD {polish_iter}: score={score:.6f}")

            elif strategy == 1:
                # LP refinement
                t_alloc = min(remaining * 0.3, 90.0)
                a_opt, obj_opt = lp_refine(best_sequence, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score - 1e-8:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish LP {polish_iter}: score={score:.6f}")

            else:
                # Targeted local search
                t_alloc = min(remaining * 0.3, 60.0)
                a_opt, obj_opt = targeted_local_search(
                    best_sequence, time_limit=t_alloc, seed=seed + 900 + polish_iter)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score - 1e-8:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish LS {polish_iter}: score={score:.6f}")

            polish_iter += 1

    # Final validation
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Ensure all values are non-negative
    best_sequence = np.maximum(best_sequence, 0.0)
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"FINAL: score={best_score:.6f}, length={len(best_sequence)}")
    return [float(x) for x in best_sequence]
