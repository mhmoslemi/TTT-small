```python
def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def gerchberg_saxton(a0, time_limit, relax=0.5, n_iter_max=100000):
    """Spectral flattening via Gerchberg-Saxton type iteration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    N = 2 * n - 1
    fft_size = 1 << (N - 1).bit_length()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit and iter_count < n_iter_max:
        # Forward FFT
        P = np.fft.rfft(a, fft_size)
        
        # Target magnitude
        mag = np.abs(P)
        # Target: flat at 1/sqrt(fft_size) but keep DC at 1
        target = np.ones(len(P)) / np.sqrt(fft_size)
        target[0] = 1.0  # DC component

            scale = target / np.maximum(mag, 1e-15)
            P_new = P * (1 - relax + relax * scale)
        else:
            P_new = P
        
        # Inverse FFT
        a_new = np.fft.irfft(P_new, fft_size)[:n]
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-15:
            # Fallback: add small uniform
            a_new = a_new + 1e-6
            s_new = a_new.sum()
        a_new = a_new / s_new
        
        # Check objective
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_new.copy()
        
        a = a_new
        iter_count += 1
    
    return best_a, best_obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Gerchberg-Saxton spectral flattening (novel main algorithm)
2. Smooth-max (logsumexp) gradient descent with simplex projection
3. Subgradient descent on the hard max for final polish
4. Multi-length, multi-initialization search with the parent sequence

The key insight is that minimizing the peak autocorrelation is equivalent to
flattening the power spectrum |P(k)|^2. The GS algorithm iteratively enforces
flat spectrum in Fourier domain and non-negativity in time domain.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_from_conv(n, conv):
    """Compute objective from convolution."""
    return 2.0 * n * conv.max()


def gs_spectral_flatten(a0, time_limit, relax=0.5, target_scale=1.0,
                        use_relaxation=True, max_iters=500000):
    """
    Gerchberg-Saxton spectral flattening algorithm.
    Alternates between flattening the power spectrum and enforcing
    non-negativity + support constraints.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    N = 2 * n - 1
    fft_size = 1 << (N - 1).bit_length()

    conv = fast_convolve(a)
    best_obj = objective_from_conv(n, conv)
    best_a = a.copy()

    target_mag = target_scale / np.sqrt(fft_size)

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    best_no_improve = 5000

    while time.monotonic() - t0 < time_limit and it < max_iters:
        # Forward FFT
        P = np.fft.rfft(a, fft_size)
        mag = np.abs(P)

        # Build target magnitude spectrum
        target = np.full(len(P), target_mag, dtype=np.float64)
        target[0] = 1.0  # DC = sum(a) = 1

        # Update: keep phase, set magnitude to target (with relaxation)
        safe_mag = np.maximum(mag, 1e-15)
        scale = target / safe_mag

        if use_relaxation:
            P_new = P * (1.0 - relax + relax * scale)
        else:
            P_new = P * scale

        # Inverse FFT
        a_full = np.fft.irfft(P_new, fft_size)

        # Truncate to support and enforce non-negativity
        a_new = np.maximum(a_full[:n], 0.0)

        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 1e-8
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
        a_new = a_new / s_new

        # Evaluate
        conv = fast_convolve(a_new)
        cur_obj = objective_from_conv(n, conv)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1

        # Adaptive relaxation: increase if oscillating
        if no_improve > best_no_improve:
            relax = max(0.1, relax * 0.8)
            no_improve = 0
            best_no_improve = int(best_no_improve * 1.5)

    return best_a, best_obj


def smooth_max_descent(a0, time_limit, beta_init=50.0, lr_init=0.05,
                       momentum=0.9, beta_growth=1.001, batch=100):
    """
    Gradient descent on the logsumexp approximation of max(conv).
    f_beta(a) = (1/beta) * log(sum_k exp(beta * c_k))
    Gradient: (1/beta) * sum_k p_k * 2 * a_{k-i}
    where p_k = softmax(beta * c_k)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = objective_from_conv(n, conv)
    best_a = a.copy()

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    max_no_improve = 2000

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute softmax weights over convolution
            bc = beta * conv
            bc_max = bc.max()
            exp_bc = np.exp(bc - bc_max)
            p = exp_bc / exp_bc.sum()

            # Compute gradient: grad_i = (2/beta) * sum_k p_k * a[k-i]
            grad = np.zeros(n)
            # For each lag k, the contribution to grad[i] is 2*p[k]*a[k-i]
            # This is a weighted "cross-correlation" of p with a
            for k in range(len(conv)):
                if p[k] < 1e-10:
                    continue
                # grad[i] += 2 * p[k] * a[k-i] for valid k-i
                wi = 2.0 * p[k]
                # k-i in [0, n-1] means i in [k-n+1, k]
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                if i_start <= i_end:
                    # a[k-i] for i in [i_start, i_end] is a[k - i_start : k - i_end + 1] reversed
                    idx = np.arange(i_start, i_end + 1)
                    grad[idx] += wi * a[k - idx]

            # Gradient step with momentum
            v = momentum * v - lr * grad
            a_new = project_simplex(a + v)
            a = a_new

            conv = fast_convolve(a)
            cur_obj = objective_from_conv(n, conv)

            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Schedule
        beta *= beta_growth
        lr *= 0.9995

        if no_improve >= max_no_improve:
            # Perturb and reset
            rng = np.random.default_rng(no_improve)
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v *= 0.3
            lr = lr_init * 0.5
            beta = beta_init
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.3)

    return best_a, best_obj

        a = a / s
    n = len(a)

    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    max_no_improve = 3000

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break

            # Multi-peak subgradient
            k = min(top_k, len(conv))
            peaks = np.argpartition(conv, -k)[-k:]
            pv = conv[peaks]
            scaled = pv / max(temp, 1e-15)
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(k)
                w[np.argmax(pv)] = 1.0
            else:
                w /= w_sum

            sg = np.zeros(n)
            for i in range(k):
                if w[i] < 1e-12:
                    continue
                ki = int(peaks[i])
                wi = w[i] * 2.0
                j_start = max(0, ki - n + 1)
                j_end = min(n - 1, ki)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += wi * a[ki - idx]

            v = momentum * v - lr * sg
            a_new = project_simplex(a + v)
            a = a_new
            conv = fast_convolve(a)
            cur_obj = objective_from_conv(n, conv)

            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        lr *= 0.9997
        if no_improve >= max_no_improve:
            rng = np.random.default_rng()
            noise = rng.standard_normal(n) * 0.004
            a = project_simplex(a + noise)
            v *= 0.4
            lr = max(lr_init * 0.3, lr * 0.5)
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.4)

    return best_a, best_obj


def generate_initializations(n, parent=None, seed=42):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    # Chirp-based (rectified)
    for freq in [0.5, 1.0, 2.0, 4.0]:
        chirp = 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * x * x / (2 * n))
        chirp = np.maximum(chirp, 0.0)
        if chirp.sum() > 1e-10:
            inits[f'chirp_{freq}'] = chirp / chirp.sum()

    # Alternating patterns
    alt = np.ones(n)
    alt[1::2] = 0.3
    inits['alternating'] = alt / alt.sum()

    alt2 = np.ones(n)
    alt2[2::4] = 0.2
    alt2[3::4] = 0.5
    inits['alt4'] = alt2 / alt2.sum()

    # Gaussian
    for sf in [0.2, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.1f}'] = g

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(2, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Random sparse
    sp = np.zeros(n)
    n_nonzero = max(4, n // 4)
    idx = rng.choice(n, size=n_nonzero, replace=False)
    sp[idx] = rng.dirichlet(np.ones(n_nonzero))
    inits['sparse_random'] = sp

    # Parent-based
    if parent is not None and len(parent) > 0:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        if p.ndim == 1 and p.size > 0 and p.sum() > 1e-10:
            if p.size == n:
                inits['parent'] = p / p.sum()
            elif p.size >= n:
                # Interpolate down
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, p.size), p)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent_interp'] = pp / pp.sum()
            else:
                # Interpolate up
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, p.size), p)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent_up'] = pp / pp.sum()
            # Add noise to parent
            if 'parent_interp' in inits or 'parent' in inits or 'parent_up' in inits:
                key = [k for k in ['parent', 'parent_interp', 'parent_up'] if k in inits][0]
                noised = inits[key] + rng.standard_normal(n) * 0.02
                noised = np.maximum(noised, 0)
                if noised.sum() > 1e-10:
                    inits['parent_noised'] = noised / noised.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining GS spectral flattening,
    smooth-max gradient descent, and subgradient methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: length={len(p)}")
    except Exception:
        pass

    def update_best(a_opt, score):
        nonlocal best_sequence, best_score
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  NEW BEST: score={score:.6f}, n={len(a_opt)}")

    # === Phase 1: GS Spectral Flattening ===
    print("=== Phase 1: GS Spectral Flattening ===")
    t0 = time.monotonic()
    phase1_time = 500.0

    lengths = [128, 256, 512, 1024]

    for n in lengths:
        remaining = deadline - time.monotonic() - 300
        if remaining < 30:
            break
        t_phase = min(120.0, remaining / (len(lengths) + 1))

        inits = generate_initializations(n, parent, seed=seed + n)
        init_names = list(inits.keys())

        # Priority order
        priority = ['parent', 'parent_interp', 'parent_up', 'parent_noised',
                    'chirp_0.5', 'chirp_1.0', 'uniform', 'random', 'gauss_0.4']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in inits if nm not in priority]
        init_names = init_names[:5]

        time_per_init = t_phase / len(init_names)

        for name in init_names:
            if time.monotonic() > deadline - 300:
                break
            t_alloc = min(time_per_init, deadline - time.monotonic() - 290)
            if t_alloc < 3:
                break

            a0 = inits[name]

            # Try GS with different relaxation parameters
            for relax, t_scale in [(0.5, 1.0), (0.3, 1.0), (0.7, 0.8)]:
                t_single = t_alloc / 3.0
                if t_single < 2:
                    break
                a_opt, obj_opt = gs_spectral_flatten(
                    a0, time_limit=t_single, relax=relax, target_scale=t_scale)
                score = float(evaluate_sequence(a_opt.tolist()))
                update_best(a_opt, score)

    # === Phase 2: Smooth-max gradient descent ===
    print(f"=== Phase 2: Smooth-max Descent (best so far: {best_score:.6f}) ===")
    t1 = time.monotonic()
    phase2_time = 400.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        # Use best and also try at other lengths
        candidates = [('best', best_sequence.copy(), n_best)]

        # Add a few more at the best length
        inits_b = generate_initializations(n_best, parent, seed=seed + 100)
        for name in ['chirp_0.5', 'random', 'uniform']:
            if name in inits_b:
                candidates.append((name, inits_b[name], n_best))

        # Also try parent length
        if parent is not None and len(parent) != n_best:
            n_p = len(parent)
            inits_p = generate_initializations(n_p, parent, seed=seed + 200)
            if 'parent' in inits_p:
                candidates.append(('parent_orig', inits_p['parent'], n_p))

        time_per_cand = phase2_time / max(1, len(candidates))

        for name, a0, n in candidates:
            if time.monotonic() > deadline - 200:
                break
            t_alloc = min(time_per_cand, deadline - time.monotonic() - 190)
            if t_alloc < 5:
                break

            # Smooth max descent
            a_opt, obj_opt = smooth_max_descent(
                a0, time_limit=t_alloc * 0.6,
                beta_init=80.0, lr_init=0.04, momentum=0.92)
            score = float(evaluate_sequence(a_opt.tolist()))
            update_best(a_opt, score)

            # Then subgradient polish
            if time.monotonic() < deadline - 200:
                a_opt2, _ = subgradient_descend(
                    a_opt, time_limit=t_alloc * 0.4,
                    lr_init=0.01, momentum=0.9, top_k=max(10, n // 20))
                score2 = float(evaluate_sequence(a_opt2.tolist()))
                update_best(a_opt2, score2)
    else:
        # Fallback: try a few from scratch
        for n in [256, 512]:
            a0 = np.ones(n) / n
            a_opt, _ = smooth_max_descent(a0, time_limit=30, beta_init=50, lr_init=0.05)
            score = float(evaluate_sequence(a_opt.tolist()))
            update_best(a_opt, score)

    # === Phase 3: Intensive GS on best ===
    print(f"=== Phase 3: Intensive GS Polish (best: {best_score:.6f}) ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = min(400.0, deadline - time.monotonic() - 100)
        if remaining > 20:
            # Multiple GS runs with different parameters from best
            for params in [
                {'relax': 0.4, 'target_scale': 1.0},
                {'relax': 0.6, 'target_scale': 0.9},
                {'relax': 0.3, 'target_scale': 1.1},
                {'relax': 0.5, 'target_scale': 0.95},
            ]:
                if time.monotonic() > deadline - 100:
                    break
                t_alloc = remaining / 4.0
                if t_alloc < 5:
                    break
                a_opt, _ = gs_spectral_flatten(
                    best_sequence, time_limit=t_alloc, **params)
                score = float(evaluate_sequence(a_opt.tolist()))
                update_best(a_opt, score)

    # === Phase 4: Final polish with subgradient ===
    print(f"=== Phase 4: Final Polish (best: {best_score:.6f}) ===")
    if best_sequence is not None:
        remaining = min(300.0, deadline - time.monotonic() - 20)
        if remaining > 10:
            n_best = len(best_sequence)
            # Several subgradient runs with different hyperparams
            for cfg in [
                {'lr_init': 0.008, 'momentum': 0.93, 'top_k': n_best // 15},
                {'lr_init': 0.015, 'momentum': 0.90, 'top_k': n_best // 10},
                {'lr_init': 0.005, 'momentum': 0.95, 'top_k': n_best // 25},
                {'lr_init': 0.01, 'momentum': 0.92, 'top_k': n_best // 8},
            ]:
                if time.monotonic() > deadline - 15:
                    break
                t_alloc = remaining / 4.0
                if t_alloc < 5:
                    break
                a_opt, _ = subgradient_descend(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                update_best(a_opt, score)

            # Random perturb + re-optimize
            if time.monotonic() < deadline - 20:
                t_remaining = deadline - time.monotonic() - 15
                n_tries = min(5, int(t_remaining / 15))
                for i in range(n_tries):
                    noise = rng.standard_normal(n_best) * 0.005
                    a_pert = project_simplex(best_sequence + noise)
                    a_opt, _ = smooth_max_descent(
                        a_pert, time_limit=t_remaining / (n_tries - i),
                        beta_init=100, lr_init=0.03)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    update_best(a_opt, score)

    # === Phase 5: Try larger sequences if time permits ===
    print(f"=== Phase 5: Larger sequences (best: {best_score:.6f}) ===")
    if time.monotonic() < deadline - 60:
        for n_try in [2048, 4096]:
            if time.monotonic() > deadline - 50:
                break
            t_alloc = min(60.0, deadline - time.monotonic() - 40)
            if t_alloc < 10:
                break

            # Initialize from best (upsampled) or uniform
            if best_sequence is not None:
                a0 = np.interp(np.linspace(0, 1, n_try),
                               np.linspace(0, 1, len(best_sequence)), best_sequence)
                a0 = np.maximum(a0, 0)
                a0 = a0 / a0.sum()
            else:
                a0 = np.ones(n_try) / n_try

            a_opt, _ = gs_spectral_flatten(a0, time_limit=t_alloc * 0.5, relax=0.5)
            score = float(evaluate_sequence(a_opt.tolist()))
            update_best(a_opt, score)

            if time.monotonic() < deadline - 40:
                a_opt2, _ = subgradient_descend(
                    a_opt, time_limit=t_alloc * 0.5,
                    lr_init=0.01, momentum=0.9, top_k=n_try // 20)
                score2 = float(evaluate_sequence(a_opt2.tolist()))
                update_best(a_opt2, score2)

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
```
