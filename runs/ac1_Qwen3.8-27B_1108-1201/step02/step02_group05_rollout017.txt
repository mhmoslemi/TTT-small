def lse_grad_descent(a0, time_limit, tau0=0.1, tau_min=0.01):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    # Adam optimizer state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.5
    t = tau0
    
    t0 = time.monotonic()
    step_count = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        # LogSumExp with temperature
        c_max = c.max()
        log_sum_exp = c_max + np.log(np.sum(np.exp((c - c_max) / t)))
        # Weights (softmax)
        w = np.exp((c - c_max) / t)
        w /= w.sum()
        
        # Gradient: 2 * sum_k w[k] * a[k-j]
        grad = np.zeros(n)
        for j in range(n):
            s = 0.0
            for k in range(max(0, j), min(2*n-1, j+n)):
                idx = k - j
                if 0 <= idx < n:
                    s += w[k] * a[idx]
            grad[j] = 2.0 * s
        
        # Adam update
        step_count += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**step_count)
        v_hat = v / (1 - beta2**step_count)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project: clip to [0, inf) and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new /= s
        
        # Check improvement
        obj = 2.0 * n * fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        
        # Decay learning rate and temperature
        # Check every 100 steps
        if step_count % 100 == 0:
            elapsed = time.monotonic() - t0
            progress = elapsed / time_limit
            lr = 0.5 * (1 - progress) + 0.01
            t = max(tau_min, tau0 * (1 - progress))
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using logsumexp-smoothed gradient
descent with Adam optimizer, multi-temperature annealing, multiple
initializations (parent, U-shaped, spectral), and local search polishing.
Key innovation: uses correlation-based gradient computation for efficient
updates on the simplex, with Nesterov momentum and adaptive restarts.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Objective with sum(a)=1: 2n*max(conv(a,a))."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones_like(a) / len(a)
    return a / s


def smooth_gradient(a, tau):
    """
    Compute gradient of 2n*logsumexp(conv(a,a)/tau) w.r.t. a.
    Returns gradient projected onto the simplex tangent space.
    """
    n = len(a)
    c = fast_convolve(a)
    # Softmax weights with numerical stability
    c_max = c.max()
    exponents = (c - c_max) / tau
    # Clip to avoid overflow
    exponents = np.clip(exponents, -500, 500)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad[j] = 4n/tau * sum_k w[k] * a[k-j]
    # This is 4n/tau * correlate(w, a, 'valid')
    raw_grad = np.correlate(w, a, mode='valid')
    grad = (4.0 * n / tau) * raw_grad
    # Project onto simplex tangent space (remove mean)
    grad = grad - grad.mean()
    return grad


def grad_descent_adam(a0, time_limit, tau_max=0.5, tau_min=0.02, lr_init=1.0):
    """
    Logsumexp-smoothed gradient descent with Adam on the simplex.
    Uses multi-temperature annealing.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_obj = obj_value(a)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.95, 1e-8
    step_count = 0

    t0 = time.monotonic()
    no_improve_count = 0
    max_no_improve = 500
    restart_count = 0
    max_restarts = 20

    while time.monotonic() - t0 < time_limit:
        if no_improve_count >= max_no_improve:
            if restart_count >= max_restarts:
                break
            # Restart with noise
            rng = np.random.default_rng(time.monotonic() * 1000)
            noise = rng.random(n) * 0.1
            a = a + noise
            a = project_simplex(a)
            best_a = a.copy()
            best_obj = obj_value(a)
            no_improve_count = 0
            restart_count += 1
            m = np.zeros(n)
            v = np.zeros(n)
            step_count = 0
            continue

        # Temperature based on progress
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        tau = tau_min + (tau_max - tau_min) * (1.0 - progress) ** 2

        # Learning rate decay
        lr = lr_init * (1.0 - 0.5 * progress)

        # Compute gradient
        grad = smooth_gradient(a, tau)

        # Clip gradient magnitude
        grad_norm = np.linalg(grad_norm_dummy(grad))
        if grad_norm > 100.0:
            grad = grad * (100.0 / grad_norm)

        # Adam update
        step_count += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step_count)
        v_hat = v / (1.0 - beta2 ** step_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        # Evaluate true objective
        obj = obj_value(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        a = a_new

    return best_a, best_obj


def grad_norm_dummy(x):
    """Helper to compute norm (avoiding shadowing)."""
    return np.sqrt(np.sum(x * x))

                        if ts < 1e-12:

                        o = obj_value(a_trial)

def make_u_shaped(n, dip_width=0.3):
    """Create a U-shaped sequence (high at edges, low in middle)."""
    x = np.linspace(0, 1, n)
    # Distance from center, normalized
    dist = np.abs(x - 0.5) * 2.0  # 0 at center, 1 at edges
    a = 1.0 + 2.0 * (dist / dip_width) ** 2
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a

def make_spectral_flat(n):
    """
    Create sequence by inverse FFT of a flat magnitude spectrum.
    Aim for flat autocorrelation.
    """
    # Try to inverse design: want flat autocorrelation
    # autocorr spectrum = |FFT(a)|^4 (for the full autocorrelation)
    # For flat autocorrelation, want |FFT(a)|^2 = constant
    # So FFT(a) should have constant magnitude
    rng = np.random.default_rng(123)
    # Random phases with constant magnitude
    fft_size = 2 * n
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    # Make it Hermitian for real signal
    A = np.zeros(fft_size, dtype=np.complex128)
    A[0] = 1.0
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        A[fft_size - k] = np.exp(-1j * phases[k])
    # IFFT
    a_full = np.fft.irfft(A, fft_size)
    # Take first n elements, clip negatives
    a = np.maximum(a_full[:n], 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a /= s
    return a

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: score={score:.6f}, n={len(a)}")

    # Generate initializations
    lengths = [256, 512, 1024]
    all_candidates = []

    print("Generating initializations...")
    for n in lengths:
        init_name = "uniform"
        a = np.ones(n) / n
        all_candidates.append((n, init_name, a.copy()))

        # U-shaped
        a = make_u_shaped(n, dip_width=0.2)
        all_candidates.append((n, "u_shaped_02", a.copy()))
        a = make_u_shaped(n, dip_width=0.35)
        all_candidates.append((n, "u_shaped_035", a.copy()))

        # Cosine
        for f in [2.0, 4.0, 6.0]:
            a = make_cosine_pattern(n, freq=f)
            all_candidates.append((n, f"cos_{f}", a.copy()))

        # Spectral flat
        for trial in range(3):
            a = make_spectral_flat(n)
            # Add randomness by mutating
            a_mut = a + rng.normal(0, 0.05, n)
            a_mut = np.maximum(a_mut, 0)
            a_mut /= a_mut.sum()
            all_candidates.append((n, f"spectral_{trial}", a_mut))

        # Random Dirichlet
        for alpha in [0.2, 0.5, 1.0]:
            a = rng.dirichlet(np.ones(n) * alpha)
            all_candidates.append((n, f"dirichlet_{alpha}", a.copy()))

        # Parent
        p = get_parent(n)
        if p is not None:
            all_candidates.append((n, "parent", p.copy()))
            # Slightly perturbed parent
            p_pert = p + rng.normal(0, 0.02, n)
            p_pert = np.maximum(p_pert, 0)
            p_pert /= p_pert.sum()
            all_candidates.append((n, "parent_pert", p_pert.copy()))

    print(f"Total candidates: {len(all_candidates)}")

    # Phase 1: Quick gradient descent scan (35% of time)
    phase1_end = time.monotonic() + (float(budget_s) - 30.0) * 0.35
    print(f"Phase 1: Quick scan ({phase1_end - time.monotonic():.0f}s budget)")

    # Sort: parent first, then shorter n
    def sort_key(item):
        n, name, _ = item
        if name.startswith("parent"):
            return (0, n)
        if name.startswith("u_shaped"):
            return (1, n)
        return (2, n)

    all_candidates.sort(key=sort_key)

    scan_results = []
    for i, (n, name, a0) in enumerate(all_candidates):
        remaining = deadline - time.monotonic()
        if remaining < 30 or time.monotonic() > phase1_end:
            break
        t_alloc = min(15.0, max(3.0, (phase1_end - time.monotonic()) / max(1, len(all_candidates) - i)))
        t_alloc = min(t_alloc, deadline - time.monotonic() - 30.0)
        if t_alloc < 2.0:
            break

        a_opt, obj_opt = grad_descent_adam(a0, time_limit=t_alloc, tau_max=0.3, tau_min=0.05, lr_init=0.8)
        try_update(a_opt)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))

    print(f"Phase 1 complete. Best so far: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates (35% of time)
    phase2_end = time.monotonic() + (float(budget_s) - 30.0) * 0.35
    print(f"Phase 2: Deep optimization ({phase2_end - time.monotonic():.0f}s budget)")

    scan_results.sort(key=lambda x: x[0])
    top_configs = []
    seen_keys = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen_keys:
            top_configs.append((n, name, a_opt))
            seen_keys.add(key)
        if len(top_configs) >= 10:
            break

    # Always include current best
    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    # Deduplicate
    unique_configs = []
    seen = set()
    for n, name, a in top_configs:
        key = (n, name)
        if key not in seen:
            unique_configs.append((n, name, a))
            seen.add(key)
    top_configs = unique_configs

    t_per_deep = max(20.0, (phase2_end - time.monotonic()) / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        remaining = deadline - time.monotonic()
        if remaining < 60 or time.monotonic() > phase2_end:
            break
        t_alloc = min(t_per_deep, max(10.0, phase2_end - time.monotonic() - 5.0))
        t_alloc = min(t_alloc, deadline - time.monotonic() - 60.0)
        if t_alloc < 8.0:
            break

        # Run gradient descent with more aggressive annealing
        a_opt, obj_opt = grad_descent_adam(a0, time_limit=t_alloc, tau_max=0.4, tau_min=0.01, lr_init=1.0)
        try_update(a_opt)

        # Quick local search
        remaining = deadline - time.monotonic()
        if remaining > 30:
            ls_time = min(15.0, remaining * 0.3)
            a_ls, obj_ls = local_search(a_opt, time_limit=ls_time, rng=rng)
            try_update(a_ls)

    print(f"Phase 2 complete. Best so far: {best_score:.6f}")

    # Phase 3: Intensive polish (30% of time)
    print(f"Phase 3: Intensive polish ({deadline - time.monotonic():.0f}s budget)")

    if best_sequence is not None:
        polish_end = deadline - 15.0
        polish_loop = 0
        no_polish_improve = 0

        while time.monotonic() < polish_end and no_polish_improve < 10:
            polish_loop += 1
            score_before = best_score
            remaining = polish_end - time.monotonic()
            if remaining < 15:
                break

            # Alternate between gradient descent and local search
            if polish_loop % 2 == 1:
                # Gradient descent with very low temperature
                gs_time = min(remaining * 0.4, 120.0)
                a_opt, _ = grad_descent_adam(best_sequence, time_limit=gs_time,
                                             tau_max=0.1, tau_min=0.005, lr_init=0.5)
                try_update(a_opt)
            else:
                # Local search
                ls_time = min(remaining * 0.4, 60.0)
                a_ls, _ = local_search(best_sequence, time_limit=ls_time, rng=rng)
                try_update(a_ls)

            if best_score < score_before - 1e-8:
                no_polish_improve = 0
            else:
                no_polish_improve += 1

    # Phase 4: Final local search with more time
    if best_sequence is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"Phase 4: Final local search ({remaining:.0f}s)")
            a_final, _ = local_search(best_sequence, time_limit=min(remaining - 10, 120), rng=rng)
            try_update(a_final)

    # Final check
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
