"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max subgradient descent with simulated annealing, LP-based refinement,
and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_sum(a, target):
    """Normalize sequence to have a specific sum."""
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def soft_max_gradient(a, tau):
    """
    Compute gradient using soft-max weighting over all convolution peaks.
    grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    where w[k] = exp(tau * c[k]) / sum(exp(tau * c[j]))
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # Use FFT-based cross-correlation
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad, c


def hard_max_gradient(a):
    """Compute gradient using hard max (single peak)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, c


def subgradient_multi_peak(a, threshold=0.95):
    """
    Compute subgradient using all peaks above threshold * max.
    This gives a more robust gradient direction.
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    threshold_val = threshold * c_max
    
    # Weight peaks above threshold by their value
    w = np.zeros_like(c)
    mask = c >= threshold_val
    w[mask] = c[mask]
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad, c


def lp_solve(f, M, n):
    """
    Solve LP: max sum(g) s.t. conv(f, g) <= M (elementwise), g >= 0.

    """

    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    A = np.zeros((2 * n - 1, n))

        pass
    return None

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max subgradient with multi-peak awareness, simulated annealing,
LP-based refinement, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * float(c.max()) / (S * S))


def normalize_sum(a, target):
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def gradient_soft(a, tau):
    """Soft-max weighted gradient."""
    n = len(a)
    c = np.convolve(a, a)
    w = np.exp(tau * (c - c.max()))
    w /= w.sum()
    a_rev = a[::-1]
    corr = np.convolve(w, a_rev, mode='full')
    return 2.0 * corr[n-1:n-1+n], c


def gradient_multi(a, top_k=5):
    """Gradient averaging over top-k peaks weighted by proximity to max."""
    n = len(a)
    c = np.convolve(a, a)
    c_sorted = np.sort(c)[::-1]
    c_max = c_sorted[0]
    
    # Weight peaks by exp(-tau * (c_max - c[k]))
    tau = 20.0
    w = np.zeros_like(c)
    for k in range(len(c)):
        w[k] = np.exp(-tau * (c_max - c[k]))
    w /= w.sum()
    
    a_rev = a[::-1]
    corr = np.convolve(w, a_rev, mode='full')
    return 2.0 * corr[n-1:n-1+n], c


def lp_refine(f, M, n, time_limit=20):
    """LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    if n > 500:
        return None
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, min(j + n, 2 * n - 1)):
            A[m, j] = f[m - j]
    
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max subgradient descent with simulated annealing, LP-based refinement,
multi-peak awareness, and multi-length exploration. Key improvements over prior
approach: better annealing schedule, tighter LP integration, peak-targeting
coordinate descent, and diverse initialization from height_sequence_1.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * float(c.max()) / (S * S))


def normalize_sum(a, target):
    """Normalize sequence to have a specific sum."""
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def gradient_soft(a, tau):
    """Soft-max weighted gradient of max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a)
    w = np.exp(tau * (c - c.max()))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    corr = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * corr[n - 1:n - 1 + n]
    return grad, c


def gradient_hard(a):
    """Hard max gradient (single peak)."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, c


def lp_refine(f, M, n, time_limit=25):
    """
    Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    if n > 512:
        return None
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo_m = j
        hi_m = min(j + n, 2 * n - 1)
        for m in range(lo_m, hi_m):
            A[m, j] = f[m - j]
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def optimize_one(a0, deadline, verbose=True):
    """Full optimization pipeline from a single starting point."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    S_t = np.sqrt(2.0 * n)
    a = normalize_sum(a, S_t)
    score = compute_obj(a)
    best = a.copy()
    best_score = score

    if verbose:
        print(f"    n={n}, init_score={score:.6f}")

    # === Phase 1: Soft-max gradient descent with simulated annealing ===
    mom = np.zeros(n)
    mom_decay = 0.9
    it = 0
    T = 0.3
    T_decay = 0.9993
    base_lr = 0.35
    no_improve = 0

    total_time = deadline - time.monotonic()
    phase1_end = time.monotonic() + max(20.0, total_time * 0.5)

    while time.monotonic() < phase1_end and it < 50000:
        it += 1
        T *= T_decay
        lr = base_lr * max(0.008, 0.9996 ** it)

        # Progressive tau: start soft, end sharp
        tau = 3.0 + 17.0 * min(1.0, it / 5000.0)
        grad, c = gradient_soft(a, tau)
        grad -= grad.mean()

        mom = mom_decay * mom - lr * grad
        a_new = np.maximum(a + mom, 0.0)
        a_new = normalize_sum(a_new, S_t)

        s_new = compute_obj(a_new)
        if s_new < score - 1e-12:
            a = a_new
            score = s_new
            if s_new < best_score - 1e-12:
                best_score = s_new
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Simulated annealing: accept worse solutions
            if T > 0.005 and (s_new - score) < 3.0 * T:
                if np.random.random() < np.exp(-(s_new - score) / T):
                    a = a_new
                    score = s_new
            # Periodic restart from best if stuck
            if no_improve > 300:
                a = best.copy() + np.random.normal(0, 0.03 * S_t / n, n)
                a = np.maximum(a, 0.0)
                a = normalize_sum(a, S_t)
                score = compute_obj(a)
                mom *= 0.3
                no_improve = 0
            if no_improve > 2000:
                break

    if verbose:
        print(f"    Phase 1 done: {it} iters, score={best_score:.6f}")

    # === Phase 2: LP-based refinement ===
    if n <= 420 and time.monotonic() < deadline - 25:
        a = best.copy()
        S_t = np.sqrt(2.0 * n)
        a = normalize_sum(a, S_t)
        best_score = compute_obj(a)

        total_remaining = deadline - time.monotonic()
        lp_end = time.monotonic() + min(120.0, total_remaining * 0.35)
        it_lp = 0
        stuck = 0

        while time.monotonic() < lp_end and it_lp < 40:
            it_lp += 1
            c = np.convolve(a, a)
            M = float(c.max())
            g = lp_refine(a, M, n, time_limit=20)
            if g is None:
                break
            g_sum = g.sum()
            if g_sum < 1e-6:
                break
            g_norm = g * (S_t / g_sum)

            improved = False
            for t in [0.4, 0.6, 0.3, 0.5, 0.7, 0.2, 0.8, 0.1]:
                a_new = (1.0 - t) * a + t * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize_sum(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-12:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break

            if improved:
                stuck = 0
            else:
                stuck += 1
                if stuck > 4:
                    break
                # Gradient nudge to escape
                grad, _ = gradient_soft(a, 8.0)
                grad -= grad.mean()
                a = np.maximum(a - 0.04 * grad, 0.0)
                a = normalize_sum(a, S_t)
                s_check = compute_obj(a)
                if s_check < best_score:
                    best = a.copy()
                    best_score = s_check
                    stuck = 0

        if verbose:
            print(f"    Phase 2 (LP): {it_lp} iters, score={best_score:.6f}")

    # === Phase 3: Peak-targeting coordinate descent ===
    if n <= 420 and time.monotonic() < deadline - 8:
        a = best.copy()
        best_score_cd = best_score
        coord_deadline = deadline - 5

        for rnd in range(4):
            if time.monotonic() > coord_deadline:
                break
            c = np.convolve(a, a)
            c_max = c.max()
            # Find all significant peaks
            peak_threshold = 0.95 * c_max
            peak_lags = np.where(c >= peak_threshold)[0]

            improved = False
            for lag in peak_lags:
                if time.monotonic() > coord_deadline:
                    break
                # For this peak lag, find the contributing pairs
                for i in range(n):
                    if time.monotonic() > coord_deadline:
                        break
                    j = int(lag) - i
                    if not (0 <= j < n):
                        continue
                    # Try reducing a[i] (which contributes a[i]*a[j] to this peak)
                    orig = a[i]
                    if orig < 1e-8:
                        continue
                    candidates = [orig * 0.5, orig * 0.7, orig * 0.3, 0.0]
                    bs = best_score_cd
                    bv = orig
                    for v in candidates:
                        a[i] = v
                        a_tmp = normalize_sum(a, S_t)
                        s = compute_obj(a_tmp)
                        if s < bs - 1e-12:
                            bs = s
                            bv = v
                    a[i] = orig
                    if bv != orig:
                        a[i] = bv
                        a = normalize_sum(a, S_t)
                        best_score_cd = bs
                        improved = True
                        if best_score_cd < best_score:
                            best_score = best_score_cd
                            best = a.copy()

            if not improved:
                break

        if verbose:
            print(f"    Phase 3 (coord): score={best_score:.6f}")

    return best, best_score


def make_starts(rng, n):
    """Generate diverse starting sequences of length n."""
    starts = []
    i = np.arange(n, dtype=np.float64)

    # From height_sequence_1 (highest priority)

                starts.append(gs[:n][::-1].copy())
            else:
                idx = np.linspace(0, len(gs) - 1, n).astype(int)
                starts.append(gs[idx].copy())
            # Perturbed versions
            base = gs[:min(n, len(gs))]
            if len(base) < n:
                base = np.concatenate([base, np.zeros(n - len(base))])
            starts.append(base + rng.normal(0, 0.03 * base.mean(), n))
            starts.append(base + rng.normal(0, 0.08 * base.mean(), n))
            starts.append(base * (1 + rng.normal(0, 0.1, n)))
    except (NameError, AttributeError, ValueError):
        pass

    # Structured starts
    starts.append(np.ones(n))
    starts.append(rng.random(n) * 0.5 + 0.25)
    starts.append(0.5 + 0.4 * np.cos(2 * np.pi * i * 3.0 / n))
    starts.append(0.5 + 0.4 * np.cos(2 * np.pi * i * 7.0 / n))
    starts.append(0.5 + 0.3 * np.cos(2 * np.pi * i * 5.0 / n) + 0.2 * np.cos(2 * np.pi * i * 11.0 / n))
    starts.append(np.abs(np.sin(2 * np.pi * i * 2.0 / n)) + 0.05)
    starts.append(np.exp(-0.5 * ((i - n * 0.25) / (n * 0.15)) ** 2) + 0.7 * np.exp(-0.5 * ((i - n * 0.75) / (n * 0.15)) ** 2))
    starts.append(0.8 * np.exp(-0.5 * ((i - n * 0.3) / (n * 0.12)) ** 2) + 1.0 * np.exp(-0.5 * ((i - n * 0.8) / (n * 0.12)) ** 2))
    starts.append(1.0 - np.abs(i - n / 2) / (n / 2) + 0.05)
    starts.append(np.abs(i - n / 2) / (n / 2) + 0.05)
    starts.append(0.5 + 0.3 * np.sin(np.pi * i * i / (n * 2.0)))

    # Clean and validate
    clean = []
    seen = set()
    for s in starts:
        s = np.maximum(np.asarray(s, dtype=np.float64), 0.0)
        if s.sum() > 1e-6 and len(s) == n and np.all(np.isfinite(s)):
            key = (len(s), round(float(s.sum()), 2), round(float(s[0]), 2))
            if key not in seen:
                seen.add(key)
                clean.append(s)
    return clean
