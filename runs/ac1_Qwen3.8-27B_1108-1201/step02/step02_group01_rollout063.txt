"""
Hybrid optimization for minimizing normalized peak autocorrelation of non-negative
step functions. Combines fast projected subgradient descent (using the structural
property that the gradient of conv[k] w.r.t. a[i] is 2*a[k-i]), LP-based blending,
and adaptive local search. Multiple sequence lengths and initializations are explored.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the score 2*n*max(conv)/sum^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return float(2.0 * n * np.max(conv) / (s * s))

def compute_conv_max(a):
    """Compute max of conv(a,a) efficiently."""
    conv = fft_convolve(a)
    return np.max(conv)

def normalize(a, S=None):
    """Normalize to target sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)

def subgradient_of_peak(a, k, n):
    """
    Compute subgradient of conv(a,a)[k] w.r.t. a.
    grad[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    grad = np.zeros(n)
    # For each i, k-i must be in [0, n-1]
    # i ranges from max(0, k-(n-1)) to min(k, n-1)
    i_min = max(0, k - n + 1)
    i_max = min(k, n - 1)
    for i in range(i_min, i_max + 1):
        grad[i] = 2.0 * a[k - i]
    return grad

def subgradient_of_peak_fast(a, k, n):
    """Vectorized subgradient computation."""
    grad = np.zeros(n)
    i_min = max(0, k - n + 1)
    i_max = min(k, n - 1)
    if i_min <= i_max:
        indices = np.arange(i_min, i_max + 1)
        grad[indices] = 2.0 * a[k - indices]
    return grad

def projected_subgradient_descent(a, time_limit, verbose=False):
    """
    Fast projected subgradient descent with adaptive step size.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv = compute_conv_max(a)
    
    lr = 2.0 / n  # Initial learning rate
    momentum = 0.9
    v = np.zeros(n)
    step = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]
        
        # Compute subgradient
        grad = subgradient_of_peak_fast(a, peak_idx, n)
        
        # Momentum
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S)
        
        new_conv = compute_conv_max(a_new)
        
        if new_conv < best_conv:
            best_conv = new_conv
            best_a = a_new.copy()
            a = a_new
            stall = 0
        elif new_conv < peak_val:
            a = a_new
            stall += 1
        else:
            lr *= 0.9
            stall += 1
        
        if stall > 100:
            lr *= 0.5
            stall = 0
            v = np.zeros(n)
        
        step += 1
        if verbose and step % 500 == 0:
            score = 2.0 * n * best_conv / (S * S)
            print(f"  SGD step {step}: score={score:.6f}, lr={lr:.5f}")
    
    return best_a

def smoothed_subgradient(a, top_k=3):
    """
    Compute a smoothed subgradient using the top peak lags.
    Returns (gradient, peak_val) where gradient is weighted average of subgradients.
    """
    n = len(a)
    conv = fft_convolve(a)
    # Get top k lag indices
    top_indices = np.argsort(conv)[-top_k:]
    # Weight by how close they are to the max
    top_vals = conv[top_indices]
    weights = (top_vals - conv.min() + 1) ** 2  # Emphasize peaks
    weights = weights / weights.sum()
    
    # Average subgradient
    grad = np.zeros(n)
    for idx, k in enumerate(top_indices):
        grad += weights[idx] * subgradient_of_peak_fast(a, int(k), n)
    
    return grad, conv[top_indices[0]]  # Return gradient and max value

def lp_refine_fast(a, time_limit, max_iters=200, verbose=False):
    """
    LP-based refinement with fast feasibility check.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv = compute_conv_max(a)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fft_convolve(a)
        M = np.max(conv)
        
        # Build LP matrix
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_min <= i_max:
                js = np.arange(i_min, i_max + 1)
                A[k, js] = a[k - js]
        
        # Solve LP
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30})
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        g_norm = normalize(g, S)
        
        # Find best blend parameter
        best_t = 0.0
        best_new_conv = M
        for t in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05, 0.03, 0.02, 0.01]:
            a_try = (1 - t) * a + t * g_norm
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize(a_try, S)
            new_conv = compute_conv_max(a_try)
            if new_conv < best_new_conv:
                best_new_conv = new_conv
                best_t = t
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * g_norm
        a = np.maximum(a, 0.0)
        a = normalize(a, S)
        
        new_conv = compute_conv_max(a)
        if new_conv < best_conv:
            best_conv = new_conv
            best_a = a.copy()
        
        if verbose and it % 20 == 0:
            score = 2.0 * n * best_conv / (S * S)
            print(f"  LP it {it}: conv={best_conv:.4f}, score={score:.6f}, t={best_t:.3f}")
        
        if best_t < 0.01:
            break
    
    return best_a

def coordinate_descent(a, time_limit, verbose=False):
    """
    Coordinate descent: optimize each element one at a time.
    For element i, find the value that minimizes the peak convolution.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv = compute_conv_max(a)
    
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 50:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            
            # Current value
            orig = a[i]
            
            # Try a range of values
            # The optimal value balances contributions to various lags
            # Try geometric spacing around current value
            candidates = [orig]
            for delta in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = orig * delta
                if new_val >= 0:
                    candidates.append(new_val)
            candidates.append(0.0)
            candidates.append(orig * 2)
            
            best_local = orig
            best_local_conv = compute_conv_max(a)
            
            for cv in candidates:
                a[i] = cv
                a_clip = np.clip(a, 0, 1000)
                s = np.sum(a_clip)
                if s < 0.01:
                    a[i] = orig
                    continue
                conv_max = compute_conv_max(a_clip)
                # Compare using normalized metric
                score = 2.0 * n * conv_max / (s * s)
                if score < 2.0 * n * best_local_conv / (np.sum(best_a) ** 2) * 1.001:
                    best_local = cv
                    best_local_conv = conv_max
            
            if best_local != orig:
                a[i] = best_local
                improved = True
        
        a = np.maximum(a, 0.0)
        a = normalize(a, S)
        new_conv = compute_conv_max(a)
        if new_conv < best_conv:
            best_conv = new_conv
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if not improved:
                stall += 5
    
    return best_a

def make_initialization(n, init_type, rng):
    """Create various initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'linear_up':
        a = np.linspace(0.1, 2.0, n)
    elif init_type == 'linear_down':
        a = np.linspace(2.0, 0.1, n)
    elif init_type == 'quadratic':
        a = 1.0 + 0.5 * (t / n - 0.5) ** 2 * 4
    elif init_type == 'chirp':
        alpha = 0.5 + 0.5 * rng.random()
        a = 1.0 + alpha * np.abs(np.cos(np.pi * t * t / (n * 0.3)))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 5))
        amp = 0.3 + 0.4 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'sawtooth':
        a = 1.0 + np.linspace(-0.5, 0.5, n)
    elif init_type == 'bump':
        center = n * rng.uniform(0.3, 0.7)
        sigma = n * 0.2
        a = 1.0 + np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'random_smooth':
        a = 1.0 + 0.5 * np.cumsum(rng.standard_normal(n))
        a = a - a.min() + 0.1
    elif init_type == 'bimodal':
        c1 = n * 0.25
        c2 = n * 0.75
        sigma = n * 0.1
        a = 1.0 + np.exp(-0.5 * ((t - c1) / sigma) ** 2) + np.exp(-0.5 * ((t - c2) / sigma) ** 2)
    else:
        a = np.ones(n)
    a = np.clip(a, 0.01, None)
    return a

def interpolate_seq(a, n_new):
    """Interpolate sequence to new length."""
    x = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x, a)

"""
Hybrid optimization for minimizing normalized peak autocorrelation of non-negative
step functions. Combines fast projected subgradient descent (exploiting the structural
property that d(conv[k])/d(a[i]) = 2*a[k-i]), LP-based blending, peak-targeted
perturbation, and multi-length exploration to improve upon previous results.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b=None):
    """Compute linear convolution of 1-D arrays using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def conv_max(a):
    """Compute max of conv(a,a)."""
    return np.max(fft_convolve(a))


def normalize(a, S=None):
    """Normalize sequence to target sum S (default sqrt(2n))."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def subgrad(a, k, n):
    """
    Subgradient of conv(a,a)[k] w.r.t. a.
    grad[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    grad = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(k, n - 1)
    if i_lo <= i_hi:
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def topk_subgrad(a, top_k=3):
    """Weighted subgradient using top-K peak lags."""
    n = len(a)
    conv = fft_convolve(a)
    idx = np.argsort(conv)[-top_k:]
    vals = conv[idx]
    # Exponential weighting emphasizing the highest peaks
    weights = np.exp((vals - vals[-1]) * 5.0)
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, idx):
        grad += w * subgrad(a, int(k), n)
    return grad, vals[-1], conv


def sgd_refine(a, time_limit, top_k=2, verbose=False):
    """
    Fast projected subgradient descent with momentum and adaptive step size.
    Uses top-K peak subgradient for smoother convergence.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_cm = conv_max(a)

    lr = 1.0 / n
    beta = 0.7
    v = np.zeros(n)
    step = 0
    stall = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        grad, peak_val, conv = topk_subgrad(a, top_k)

        # Momentum step
        v = beta * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            lr *= 0.5
            stall += 1
            continue
        a_new = a_new * (S / s_new)

        new_cm = conv_max(a_new)

        if new_cm < best_cm:
            best_cm = new_cm
            best_a = a_new.copy()
            a = a_new
            stall = 0
        elif new_cm < peak_val * 1.001:
            a = a_new
            stall += 1
        else:
            lr *= 0.92
            stall += 1
            v *= 0.5

        if stall > 80:
            lr = max(lr * 0.5, 0.01 / n)
            v = np.zeros(n)
            stall = 0
            # Try increasing top_k to smooth
            top_k = min(top_k + 1, 5)

        if stall == 0:
            best_lr = lr

        step += 1
        if verbose and step % 200 == 0:
            score = 2.0 * n * best_cm / (S * S)
            print(f"    SGD[{step}]: score={score:.6f}, lr={lr:.5f}, top_k={top_k}")

    return best_a


def lp_refine(a, time_limit, verbose=False, max_iters=100):
    """
    LP-based iterative refinement.
    Solves: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Blends g with current a to reduce the peak.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_cm = conv_max(a)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 15:
            break

        conv = fft_convolve(a)
        M = np.max(conv)

        # Build LP constraint matrix
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                js = np.arange(i_lo, i_hi + 1)
                A[k, js] = a[k - js]

        # Solve LP with time limit
        c = -np.ones(n)
        time_left = time_limit - (time.monotonic() - t0) - 10
        if time_left < 5:
            break
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max(5, time_left)})

        if res.status != 0:
            if verbose:
                print(f"    LP[{it}]: status={res.status}, breaking")
            break

        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        g_norm = normalize(g, S)

        # Find best blend parameter via bisection on t
        # We want conv((1-t)*a + t*g, (1-t)*a + t*g) peak to be minimized
        lo, hi = 0.0, 0.8
        best_t = 0.0
        best_new_cm = M

        for _ in range(12):
            mid = (lo + hi) / 2
            a_try = (1 - mid) * a + mid * g_norm
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize(a_try, S)
            cm = conv_max(a_try)
            if cm < best_new_cm:
                best_new_cm = cm
                best_t = mid
                lo = mid
            else:
                hi = mid

        if best_t < 0.005:
            break

        a = (1 - best_t) * a + best_t * g_norm
        a = np.maximum(a, 0.0)
        a = normalize(a, S)

        new_cm = conv_max(a)
        improved = new_cm < best_cm
        if improved:
            best_cm = new_cm
            best_a = a.copy()

        if verbose and (it % 10 == 0 or improved):
            score = 2.0 * n * best_cm / (S * S)
            print(f"    LP[{it}]: score={score:.6f}, t={best_t:.3f}, M={M:.4f}")

        if not improved and best_t < 0.02:
            break

    return best_a


def targeted_perturb(a, time_limit, verbose=False):
    """
    Targeted local search: identify top contributors to peak lags and
    perturb them to reduce the peak.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_cm = conv_max(a)
    rng = np.random.default_rng(999)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 500:
        conv = fft_convolve(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        # Compute contributions to the peak
        i_lo = max(0, peak_idx - n + 1)
        i_hi = min(peak_idx, n - 1)
        idx = np.arange(i_lo, i_hi + 1)
        contribs = a[idx] * a[peak_idx - idx]

        # Get top contributors
        top_frac = max(0.1, 0.3)
        n_top = max(3, int(n * top_frac))
        local_top = np.argsort(contribs)[-n_top:]
        top_global = idx[local_top]

        # Strategy: reduce top contributors, increase others
        for _ in range(10):
            candidate = a.copy()
            # Reduce top contributors
            n_reduce = max(2, n_top // 2)
            reduce_idx = rng.choice(top_global, size=n_reduce, replace=False)
            for i in reduce_idx:
                candidate[i] *= np.exp(rng.normal(-0.08, 0.03))

            # Increase some others
            others = np.setdiff1d(np.arange(n), top_global)
            if len(others) > 0:
                n_inc = max(3, n // 15)
                inc_idx = rng.choice(others, size=min(n_inc, len(others)), replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.02, size=len(inc_idx)))

            candidate = np.maximum(candidate, 0.0)
            candidate = normalize(candidate, S)
            new_cm = conv_max(candidate)

            if new_cm < best_cm:
                best_cm = new_cm
                best_a = candidate.copy()
                a = candidate
                stall = 0
            elif new_cm < peak_val * 1.02 and rng.random() < 0.2:
                a = candidate
                stall += 1
            else:
                stall += 1

    return best_a


def make_initialization(n, init_type, rng):
    """Create structured initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'linear_up':
        a = np.linspace(0.2, 3.0, n)
    elif init_type == 'linear_down':
        a = np.linspace(3.0, 0.2, n)
    elif init_type == 'chirp':
        alpha = 0.5 + 0.5 * rng.random()
        a = 1.0 + alpha * np.abs(np.sin(np.pi * t * t / (n * 0.4)))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 6))
        amp = 0.3 + 0.5 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'bimodal':
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        sigma = n * 0.12
        a = 1.0 + np.exp(-0.5 * ((t - c1) / sigma) ** 2) + np.exp(-0.5 * ((t - c2) / sigma) ** 2)
    elif init_type == 'quadratic':
        a = 0.5 + 2.0 * (t / n) ** 2
    elif init_type == 'random_smooth':
        a = 1.0 + 0.8 * np.cumsum(rng.standard_normal(n))
        a = (a - a.min()) * 2 + 0.1
    elif init_type == 'sawtooth':
        period = max(2, n // 3)
        a = 1.0 + 0.5 * (2 * np.abs((t % period) / period - 1) - 1) ** 2
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)


def interpolate_seq(a, n_new):
    """Interpolate sequence to new length."""
    x = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid optimization:
    1. Fast projected subgradient with momentum (top-K peak subgradient)
    2. LP-based iterative refinement with bisection on blend parameter
    3. Targeted perturbation of peak contributors
    4. Multi-length exploration via interpolation
    5. Final polish rounds
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def try_update(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = np.sum(a_clip)
        if s < 0.01:
            return False
        cm = conv_max(a_clip)
        score = float(2.0 * len(a_clip) * cm / (s * s))
        if score < best_score:
            best_score = score
            best_seq = a_clip.copy()
            return True
        return False

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided: n={len(provided_seq)}, score={compute_score(provided_seq):.6f}")
        try_update(provided_seq)

    if best_seq is not None:
        print(f"Initial best: {best_score:.6f}")

    # === Phase 1: Fast SGD ===
    print("\n=== Phase 1: Subgradient descent ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = 100.0
        print(f"  n={n}, limit={t_limit}s")
        result = sgd_refine(best_seq, t_limit, top_k=2, verbose=True)
        improved = try_update(result)
        print(f"  After SGD: score={best_score:.6f} {'*** NEW BEST ***' if improved else ''}")

    # === Phase 2: LP refinement ===
    print("\n=== Phase 2: LP refinement ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_limit = min(200.0, remaining * 0.25)
        print(f"  n={len(best_seq)}, limit={t_limit:.0f}s")
        result = lp_refine(best_seq, t_limit, verbose=True, max_iters=60)
        improved = try_update(result)
        print(f"  After LP: score={best_score:.6f} {'*** NEW BEST ***' if improved else ''}")

    # === Phase 3: Alternate SGD + LP ===
    print("\n=== Phase 3: Alternating optimization ===")
    if best_seq is not None:
        for rnd in range(4):
            remaining = deadline - time.monotonic()
            if remaining < 100:
                break

            # SGD round
            t_sgd = min(60.0, remaining * 0.3)
            a = sgd_refine(best_seq, t_sgd, top_k=3, verbose=False)
            try_update(a)

            # LP round
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            t_lp = min(100.0, remaining * 0.4)
            a = lp_refine(best_seq, t_lp, verbose=False, max_iters=30)
            improved = try_update(a)
            if improved:
                print(f"  Round {rnd}: NEW BEST score={best_score:.6f}")

    # === Phase 4: Targeted perturbation ===
    print("\n=== Phase 4: Targeted perturbation ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t_limit = min(60.0, remaining * 0.3)
            print(f"  limit={t_limit:.0f}s")
            result = targeted_perturb(best_seq, t_limit, verbose=False)
            improved = try_update(result)
            print(f"  After perturbation: score={best_score:.6f} {'*** NEW BEST ***' if improved else ''}")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-length exploration ===")
    if best_seq is not None:
        for n_try in [512, 768, 1024, 1280, 1600]:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            a0 = interpolate_seq(best_seq, n_try)
            a0 = np.clip(a0, 0.01, None)
            # Quick SGD then LP
            a1 = sgd_refine(a0, min(40.0, remaining * 0.3), top_k=2, verbose=False)
            try_update(a1)
            remaining = deadline - time.monotonic()
            if remaining < 40:
                continue
            a2 = lp_refine(a1, min(60.0, remaining * 0.5), verbose=False, max_iters=20)
            improved = try_update(a2)
            if improved:
                print(f"  *** n={n_try}: NEW BEST score={best_score:.6f} ***")

    # === Phase 6: Final combined round ===
    print("\n=== Phase 6: Final optimization ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 80:
            # SGD
            t_sgd = min(40.0, remaining * 0.3)
            a = sgd_refine(best_seq, t_sgd, top_k=4, verbose=False)
            try_update(a)

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 40:
                t_lp = min(remaining - 20, 120.0)
                a = lp_refine(best_seq, t_lp, verbose=True, max_iters=40)
                improved = try_update(a)
                if improved:
                    print(f"  Final LP: NEW BEST score={best_score:.6f}")

    # === Phase 7: Try structured initializations at best length ===
    print("\n=== Phase 7: Structured inits ===")
    if best_seq is not None:
        n = len(best_seq)
        remaining = deadline - time.monotonic()
        init_types = ['chirp', 'sine', 'bimodal', 'quadratic', 'random_smooth']
        for itype in init_types:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a0 = make_initialization(n, itype, rng)
            a1 = sgd_refine(a0, min(25.0, remaining * 0.4), top_k=2, verbose=False)
            improved = try_update(a1)
            if improved:
                print(f"  *** init={itype}: NEW BEST score={best_score:.6f} ***")

    # === Final verification ===
    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")
    print(f"Time used: {time.monotonic() - t_global:.1f}s / {budget_s}s")

    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        val_score = evaluate_sequence(final_list)
        print(f"evaluate_sequence confirms: {val_score:.6f}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
