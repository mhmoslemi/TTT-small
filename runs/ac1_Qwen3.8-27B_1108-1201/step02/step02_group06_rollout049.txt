    return None

"""
Hybrid optimization for minimizing normalized peak autocorrelation of non-negative step functions.
Uses LP-based directional search with adaptive targets and line search, combined with
Nesterov-accelerated annealed-softmax gradient polishing and simulated annealing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_A(f):
    """Build constraint matrix A for the LP. A[k,j] such that (A@g)[k] = (f*g)[k]."""
    n = len(f)
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid k-j
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    # But we want (A@g)[k] = sum_j A[k,j] * g[j] = (f*g)[k]
    # (f*g)[k] = sum_j g[j] * f[k-j]
    # So A[k,j] = f[k-j] if 0 <= k-j < n else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (f*g)[k] includes term g[j] * f[k-j] for k-j in [0, n-1]
        # i.e., k in [j, j+n-1]
        A[j:j+n, j] = f
    return A


def solve_lp(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_constraint_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})
    if result.success:
        return result.x
    return None


def line_search(f, g_rescaled, sum_f, max_iters=40):
    """Find optimal t that minimizes score of (1-t)*f + t*g_rescaled."""
    best_t = 0.0
    best_score = local_score(f)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lb, a_ub = 0.01, 0.7
    for _ in range(max_iters):
        m1 = a_ub - gr * (a_ub - a_lb)
        m2 = a_lb + gr * (a_ub - a_lb)
        f_m1 = (1.0 - m1) * f + m1 * g_rescaled
        f_m2 = (1.0 - m2) * f + m2 * g_rescaled
        s1 = local_score(f_m1)
        s2 = local_score(f_m2)
        if s1 < s2:
            a_ub = m2
        else:
            a_lb = m1
        if a_ub - a_lb < 0.005:
            break
    t = (a_lb + a_ub) / 2.0
    f_t = (1.0 - t) * f + t * g_rescaled
    f_t = np.clip(f_t, 0.0, None)
    s_t = local_score(f_t)
    if s_t < best_score:
        best_t = t
        best_score = s_t
    return best_t, best_score


def lp_optimize(f_init, time_limit, verbose=False, seed=42):
    """LP-based iterative optimization with adaptive target M and line search."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    stagnant = 0
    iteration = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        # Adaptive epsilon: increase when stagnant
        base_eps = 0.0005
        eps = base_eps + 0.003 * min(stagnant / 30.0, 1.0)
        M_target = M * (1.0 - eps)
        
        # Try with tighter constraint first
        lp_time = min(20.0, remaining * 0.3)
        g = solve_lp(f, M_target, time_limit=lp_time)
        if g is None or np.sum(g) < 1e-10:
            g = solve_lp(f, M, time_limit=lp_time)
        
        if g is None or np.sum(g) < 1e-10:
            # Try to escape
            if stagnant > 5:
                f = f + 0.05 * rng.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                stagnant = 0
            continue
        
        sum_g = np.sum(g)
        g_rescaled = g * (sum_f / sum_g)
        
        # Line search
        t_opt, score_t = line_search(f, g_rescaled, sum_f, max_iters=30)
        
        if t_opt > 0.01:
            f_new = (1.0 - t_opt) * f + t_opt * g_rescaled
            f_new = np.clip(f_new, 0.0, None)
            sum_new = np.sum(f_new)
            if sum_new < 1e-10:
                continue
            f_new = f_new * (sum_f / sum_new)
            score_new = local_score(f_new)
            f = f_new
            iteration += 1
            
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
        
        # Escalating perturbations when stuck
        if stagnant > 8:
            strength = 0.02 * (1.0 + stagnant / 10.0)
            f = f + strength * rng.standard_normal(n)
            f = np.clip(f, 0.0, None)
            f = f * (sum_f / np.sum(f))
            stagnant = 0
            best_score = min(best_score, local_score(f))
            if local_score(f) < best_score:
                best_score = local_score(f)
                best_f = f.copy()
        
        if verbose and iteration % 30 == 0:
            print(f"    LP iter={iteration}, score={local_score(f):.6f}, "
                  f"best={best_score:.6f}, stagnant={stagnant}, elapsed={elapsed:.1f}s")
    
    return best_f, best_score


def gradient_polish(f_init, time_limit, verbose=False, seed=42):
    """Nesterov-accelerated gradient descent on annealed softmax parameterization."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f_safe = np.clip(f_init, 1e-8, None)
    theta = np.log(f_safe / np.sum(f_safe))
    
    # Nesterov momentum
    v = np.zeros(n)
    beta = 0.9
    
    t0 = time.monotonic()
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(f_init)
    tau_start = 6.0 / n
    tau_end = 0.5 / n
    base_lr = 0.5
    
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        elapsed_frac = min(1.0, elapsed / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        # Compute a from theta
        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S
        
        # Compute gradient of smoothed max
        conv = fft_convolve(a, a)
        shifted_conv = (conv - np.max(conv)) / tau
        exp_conv = np.exp(shifted_conv)
        w = exp_conv / np.sum(exp_conv)
        
        # Gradient: dL/da = 2 * (w cross-correlate a)
        # (w ⋆ a)[j] = (w_rev * a)[2n-2-j]
        w_rev = w[::-1]
        full_corr = fft_convolve(w_rev, a)
        grad_a = 2.0 * full_corr[n-1:2*n-1][::-1]
        
        # Convert to theta gradient
        inner = np.dot(a, grad_a) / S
        g_theta = a * (grad_a - inner)
        
        # Nesterov step
        v = beta * v + (1.0 - beta) * g_theta
        lr = base_lr / (1.0 + 0.05 * step_count ** 0.5)
        theta = theta - lr * v
        
        step_count += 1
        
        # Evaluate
        t_shifted = theta - np.max(theta)
        e_new = np.exp(t_shifted)
        a_check = e_new / np.sum(e_new) * S
        score = local_score(a_check)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
        
        # Periodic reset of momentum to avoid oscillation
        if step_count % 500 == 0:
            v = 0.5 * v
        
        if verbose and step_count % 100 == 0:
            print(f"    Grad step={step_count}, score={score:.6f}, "
                  f"best={best_score:.6f}, tau={tau:.5f}")
    
    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)

    f = f_init.astype(np.float64).copy()
