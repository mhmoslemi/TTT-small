"""
Hybrid optimization for minimizing 2*n*max(auto_conv)/(sum)^2.
Combines: (1) subgradient descent with Polyak averaging, (2) LP-guided 
alternating optimization with warm-starting, (3) spectral shaping via
multiplicative updates. Multiple initializations and adaptive time allocation.

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def objective_normalized(f):
    """Compute 2*n*max_conv for a normalized (sum=1) non-negative vector."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max(), conv


def subgradient_step(f, conv, t_param, eta, k_weight=None):
    """One step of mirror descent on the simplex.
    Uses the 'temperature' t_param to control concentration on peak positions."""
    n = len(f)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return f.copy()
    
    # Build weight distribution over convolution positions
    scaled = t_param * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-12:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    
    # Compute subgradient: sg[i] = sum_k w[k] * f[k-i] (where valid)
    sg = np.zeros(n)
    # Use FFT-based approach for speed
    # sg[i] = sum_k w[k] * f[k-i] for valid k-i in [0, n-1]
    # This is cross-correlation of w and f
    # Actually: sg[i] = sum_{k} w[k] * f[k-i] where 0<=k-i<=n-1
    # = sum_j f[j] * w[i+j] where 0<=i+j<=2n-2
    # This is the cross-correlation!
    
    # Use FFT for cross-correlation
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    # cross-correlation: ifft(conj(W) * F) gives sum_k w[k-j] f[j]... 
    # Actually let me just do it directly for clarity and use the fact that
    # for n=512, the direct computation is O(n * support(w)) which is fast if w is sparse
    
    thresh = w.max() * 0.001
    active = np.where(w > thresh)[0]
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # sg[j] += w[k] * f[k-j] for j in [j_start, j_end]
        # f[k-j] for j in [j_start, j_end] is f[k-j_end] to f[k-j_start]
        sg[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]
    
    # Center the gradient (projection onto simplex tangent space)
    sg_c = sg - sg.mean()
    
    # Multiplicative update
    lf = -eta * sg_c
    lf -= lf.max()  # numerical stability
    f_new = f * np.exp(lf)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-12:
        return f.copy()
    f_new /= s
    return f_new


def subgradient_optimize(f0, time_limit, t_start=50.0, t_end=500.0, 
                          eta_start=0.5, eta_end=0.01, patience_max=300):
    """Subgradient descent with Polyak averaging and adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    # Polyak averaging
    avg_f = f.copy()
    total_weight = 1.0
    start_iter = 100  # Start averaging after some burn-in
    
    t0 = time.monotonic()
    iter_count = 0
    patience = 0
    eta = eta_start
    
    while time.monotonic() - t0 < time_limit:

        # Adaptive parameters

        t_param = t_start * (t_end / t_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** (progress ** 0.5)
        
        f_new = subgradient_step(f, conv, t_param, eta)
        f = f_new
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()
        
        # Polyak averaging

            avg_f = (avg_f * (total_weight - 1) + f) / total_weight

            patience = 0

            if avg_obj < best_obj - 1e-10:

                best_f = avg_f.copy()

def make_lp_sparse(n):
    """Precompute sparse structure for cross-convolution constraints."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_fast(f, M, rows, cols, n, x0=None, timeout=10):
    """Solve the LP: max sum(g) s.t. g>=0, (f*g)[k]<=M."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    options = {'disp': False, 'time_limit': timeout}
    if x0 is not None:
        options['presolve'] = 'off'  # Hmm, might not help
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options=options, x0=x0)
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, lp_timeout=10):
    """LP-guided alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    rows, cols = make_lp_sparse(n)
    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0
    x0 = None
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = solve_lp_fast(f, M, rows, cols, n, x0=x0, timeout=lp_timeout)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        x0 = g
        
        if np.max(np.abs(g - f)) < 1e-13:
            break
        
        improved = False
        for t_try in [t, 0.5*t, 0.25*t, 0.1*t, 0.05*t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.15)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.7
            if patience > 80:
                break
    
    return best_f, best_obj


def gen_inits(n, rng):

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def normalize_vec(f):
    """Normalize to probability vector (sum=1, non-negative)."""
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-12:
        return np.ones(len(f)) / len(f)
    return f / s


def make_lp_sparse(n):
    """Precompute sparse structure for cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_fast(f, M, rows, cols, n, x0=None, timeout=8):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'disp': False, 'time_limit': timeout}
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options=options, x0=x0)
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

        if iter_count > burn_in:
            total_weight += 1.0
            avg_f = (avg_f * (total_weight - 1.0) + f) / total_weight

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1

        if iter_count % 200 == 0:
            avg_obj = 2.0 * n * fast_convolve(avg_f).max()
            if avg_obj < best_obj - 5e-5:
                best_obj = avg_obj
                best_f = avg_f.copy()

        if patience > patience_max:
            break

    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    # U-shape (bimodal at edges) - good for flattening auto-conv
    u = 0.5 * (1.0 + np.cos(np.pi * x / max(n - 1, 1)))
    u = np.maximum(u, 0.05)
    inits['u_shape'] = u / u.sum()

    # Strong U-shape
    norm_x = x / max(x[-1], 1)
    u2 = norm_x ** 6 + (1.0 - norm_x) ** 6
    u2 = np.maximum(u2, 0.02)
    inits['u_strong'] = u2 / u2.sum()

    # Multiple U-shapes (4 bumps)
    u4 = np.zeros(n)
    for c in [0.125, 0.375, 0.625, 0.875]:
        u4 += np.exp(-0.5 * ((x - c * n) / (0.06 * n)) ** 2)
    u4 = np.maximum(u4, 0.01)
    inits['u_four'] = u4 / u4.sum()

    # Bimodal at quartiles
    bi = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    bi = np.maximum(bi, 0.01)
    inits['bimodal'] = bi / bi.sum()

    # Inverted parabola
    q = 1.0 - ((x - 0.5 * n) / (0.5 * n)) ** 2
    q = np.maximum(q, 0.05)
    inits['quad'] = q / q.sum()

    # Random
    rr = rng.dirichlet(np.ones(n) * 0.3)
    inits['rnd'] = rr

    # Alternating
    alt = np.ones(n)
    alt[::2] = 3.0
    inits['alt2'] = alt / alt.sum()

    # Period-4 pattern
    p4 = np.ones(n)
    p4[::4] = 5.0
    inits['p4'] = p4 / p4.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry: Hybrid LP + subgradient optimization with adaptive phases."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={p.size}, sum={p.sum():.4f}")
    except Exception:
        pass

    # Phase 1: Quick subgradient scan (~90s)
    print("Phase 1: Quick subgradient scan")
    results = []
    n_primary = 512
    inits = gen_inits(n_primary, rng)

    if parent is not None:
        if len(parent) == n_primary:
            pp = np.maximum(parent, 0.0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n_primary),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()

    t_p1 = min(90.0, (deadline - time.monotonic()) * 0.2)
    time_per = max(4.0, t_p1 / len(inits))

    for name, a0 in inits.items():
        if time.monotonic() > deadline - 90:
            break
        ta = min(time_per, deadline - time.monotonic() - 90)
        if ta < 3:
            break
        f_opt, _ = subgradient_optimize(a0, time_limit=ta)
        score = float(evaluate_sequence(f_opt.tolist()))
        if np.isfinite(score):
            results.append((score, n_primary, name, f_opt))
            if score < best_score:
                best_score = score
                best_seq = f_opt.copy()
                print(f"  SubGrad {name}: {score:.6f}")

    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: LP optimization on top candidates (~1000s)
    print("Phase 2: LP-guided optimization")
    results.sort(key=lambda x: x[0])
    top_k = min(4, len(results))
    t_p2 = min(1000.0, (deadline - time.monotonic()) * 0.55)
    time_per_lp = t_p2 / max(1, top_k)

    for idx in range(top_k):
        if time.monotonic() > deadline - 180:
            break
        sc, n, name, f = results[idx]
        ta = min(time_per_lp, deadline - time.monotonic() - 180)
        if ta < 15:
            break

        print(f"  LP[{idx}] {name} (start={sc:.6f}, t={ta:.0f}s)")
        f_lp, _ = lp_optimize(f, time_limit=ta, lp_timeout=10,
                              use_momentum=True, over_relax=True)
        score = float(evaluate_sequence(f_lp.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_seq = f_lp.copy()
                print(f"  LP[{idx}] {name}: {score:.6f} *")
            results[idx] = (score, n, name, f_lp)

    print(f"Phase 2 best: {best_score:.6f}")

    # Phase 3: Subgradient polish on best (~250s)
    print("Phase 3: Subgradient polishing")
    if best_seq is not None:
        t_p3 = min(250.0, deadline - time.monotonic() - 60)
        configs = [
            (40.0, 400.0, 0.25, 0.01),
            (80.0, 600.0, 0.15, 0.008),
            (150.0, 1000.0, 0.08, 0.005),
        ]
        if t_p3 > 30:
            t_each = t_p3 / len(configs)
            for ts, te, es, ee in configs:
                if time.monotonic() > deadline - 30:
                    break
                tr = min(t_each, deadline - time.monotonic() - 30)
                if tr < 5:
                    break
                f_pol, _ = subgradient_optimize(best_seq, time_limit=tr,
                                                t_start=ts, t_end=te,
                                                eta_start=es, eta_end=ee)
                score = float(evaluate_sequence(f_pol.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = f_pol.copy()
                    print(f"  Polish t=[{ts},{te}]: {score:.6f}")

    print(f"Phase 3 best: {best_score:.6f}")

    # Phase 4: Try n=1024 (~200s)
    print("Phase 4: Alternative length n=1024")
    if time.monotonic() < deadline - 60:
        n_alt = 1024
        t_p4 = min(200.0, deadline - time.monotonic() - 30)
        if best_seq is not None:
            alt_init = np.interp(np.linspace(0, 1, n_alt),
                                 np.linspace(0, 1, len(best_seq)), best_seq)
            alt_init = np.maximum(alt_init, 0.0)
            if alt_init.sum() > 1e-10:
                alt_init /= alt_init.sum()
        else:
            alt_init = np.ones(n_alt) / n_alt

        f_alt, _ = lp_optimize(alt_init, time_limit=min(t_p4 * 0.7, deadline - time.monotonic() - 30),
                               lp_timeout=12)
        score = float(evaluate_sequence(f_alt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = f_alt.copy()
            print(f"  n=1024 LP: {score:.6f}")

        if time.monotonic() < deadline - 20:
            f_alt2, _ = subgradient_optimize(f_alt, time_limit=min(40.0, deadline - time.monotonic() - 20),
                                             t_start=50.0, t_end=500.0,
                                             eta_start=0.2, eta_end=0.01)
            score2 = float(evaluate_sequence(f_alt2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_seq = f_alt2.copy()
                print(f"  n=1024 polish: {score2:.6f}")

    # Phase 5: Final micro-polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        print("Phase 5: Final micro-polish")
        tr = deadline - time.monotonic() - 5
        f_fin, _ = subgradient_optimize(best_seq, time_limit=tr,
                                        t_start=300.0, t_end=2000.0,
                                        eta_start=0.04, eta_end=0.003,
                                        patience_max=400)
        score = float(evaluate_sequence(f_fin.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = f_fin.copy()
            print(f"  Final: {score:.6f}")

    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final validation
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(final_score) or final_score > best_score + 1e-6:
        best_seq = np.ones(512)
        final_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={final_score:.6f}")
    return [float(x) for x in best_seq]
