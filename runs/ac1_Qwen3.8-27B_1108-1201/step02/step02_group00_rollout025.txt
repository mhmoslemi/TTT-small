Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe iterations
with multi-scale initialization, adaptive line search, and subgradient polishing.
The key innovation is using a linear programming oracle to find the Frank-Wolfe
direction, which provides a globally informed step at each iteration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Pre-build the LP constraint matrix for sequence of length n."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = np.arange(n)  # placeholder, will be filled with f values
    return A


def lp_direction(f, M):
    """
    Solve LP: max sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g vector.
    """
    n = len(f)
    c = -np.ones(n)
    
    # Build A_ub: A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

LP-based Frank-Wolfe for minimizing max auto-convolution ratio.
Uses linear programming oracle for globally-informed search directions,
combined with adaptive line search and subgradient polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        g_sum = g.sum()
        if g_sum < 1e-15:

        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:

        # Line search

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe iterations
with adaptive constraint relaxation, bisection line search, multi-scale exploration,
and subgradient polishing. The key innovation is using a linear programming oracle
to find globally-informed search directions at each iteration, combined with
adaptive M-expansion to escape stagnation.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None if infeasible/failed."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

                                  lr=0.003, mom=0.94, top_k=n_best // 8, temp=0.00005)
            try_seq(a_opt, "sg_p2")

    # === Phase 3: Final fine-tuning ===
    print(f"\n=== Phase 3: Final Polish ===")
    print(f"  Current best: {best_score:.6f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine SG
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 80),
                                  lr=0.001, mom=0.96, top_k=n_best // 4, temp=0.00002)
            try_seq(a_opt, "sg_fine")

        # Another FW-LP on the polished result
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "fwl_final")

        # Tiny perturbation + quick SG
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            for _ in range(3):
                rem = deadline - time.monotonic() - 8
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = sg_polish(f_pert, time_limit=min(rem, 20),
                                      lr=0.001, mom=0.95, top_k=n_best // 4, temp=0.00001)
                try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
