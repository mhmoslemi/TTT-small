import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def make_lp_structure(n):
    """Precompute row/col indices for the cross-convolution constraint matrix.
    
    Args:
        n: Length of sequence.
    
    Returns:
        (row indices, col indices): sparse matrix indices
    """
    # We need to enforce 2*n-1 constraints for the convolution
    rows = []
    cols = []
    # For each output index of the convolution (0 to 2*n-2)
    for k in range(2 * n - 1):
        # For each element in the first sequence (index i)
        for i in range(n):
            if k - i < n:
                rows.append(k)
                cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def evaluate_sequence(seq):
    """Evaluates a sequence using the provided objective function.
    
    Args:
        seq: Sequence of coefficients.
        
    Returns:
        float: Objective value or inf if invalid.
    """
    try:
        if not isinstance(seq, (list, tuple)):
            raise ValueError
        n = len(seq)
        if n == 0:
            raise ValueError
        for x in seq:
            if not isinstance(x, (int, float)) or isinstance(x, bool):
                raise ValueError("Elements must be integers or floats, and not booleans")
            if np.isnan(x) or np.isinf(x):
                raise ValueError("NaN or infinity not allowed")
        # Convert to floats for consistency with the evaluation function
        sequence = np.array(seq, dtype=np.float64)
        sequence = np.maximum(sequence, 0.0)  # Remove negatives
        sequence = np.minimum(sequence, 1000.0)  # Capping at 1000
        sequence_list = sequence.tolist()
        # Recalculate n because of potential padding and clipping
        n = len(sequence_list)
        if np.sum(sequence) < 0.01:
            return np.inf
        conv_sequence = fast_convolve(sequence_list)
        max_b = np.max(conv_sequence)
        sum_a = np.sum(sequence)
        return 2 * n * max_b / (sum_a ** 2)
    except Exception as e:
        return np.inf

def solve_lp(f, M, rows, cols, n, timeout=15):
    """Solves a linear program to find the maximum sum sequence \(g\) such that \(\\|f \\circ g\\|_\\infty \\leq M\\).
    
    Args:
        f: Initial sequence (as a list)
        M: Maximum allowable convolution value
        rows, cols: Indices from make_lp_structure
        n: Length of sequence
        timeout: Maximum time in seconds
        
    Returns:
        list or None: Optimized sequence >=0 or None/inf if failed
    """
    try:
        # Convert to float ndarray
        f_vals = np.asarray(f, dtype=np.float64)
        # Apply defensive clipping (convert to the internal evaluator's format)
        f_vals = np.maximum(f_vals, 0.0)
        f_vals = np.minimum(f_vals, 1000.0)
        # Build constraint matrix
        total_count = 2 * n - 1
        vals = np.tile(f_vals, n)
        A = sparse.csc_matrix((vals, (rows, cols)), shape=(total_count, n))
        
        # Objective vector (-ones) because we maximize sum(g), but CVXOPT will give minimizer via -c
        c = -np.ones(n, dtype=np.float64)
        
        # We want to satisfy -sum(g) as minimum, so solve min(-sum(g)) subject to constraints.
        b = np.full(total_count, M, dtype=np.float64)
        
        # Bounds: non-negative only, no upper bound beyond cap (handled in evaluate by clipping)
        bounds = [(0, None)] * n
        
        res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', 
                               options={'disp': False, 'time_limit': timeout})
        if res.success:
            g = res.x.clip(min=0)
            if np.sum(g) < 1e-12:
                return None  # Not normalized candidate
            return g.tolist()
        else:
            return None                    
    except Exception as e:
        print(f"LP failed: {e}")
        return None

        return f, obj_current
