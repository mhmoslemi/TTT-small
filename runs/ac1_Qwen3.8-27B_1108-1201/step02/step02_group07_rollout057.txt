def solve_lp_min_peak(f, S, timeout=5.0):
    """Solve: min t s.t. conv(f,g)_k <= t for all k, sum(g) = S, g >= 0."""
    n = len(f)
    m = 2 * n - 1  # length of convolution
    
    # A matrix: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((m, n))
    for j in range(n):
        for k in range(j, j + n):
            A[k, j] = f[k - j]
    
    # Variables: g_0, ..., g_{n-1}, t
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A @ g - t <= 0 for each k
    # i.e., A @ g + [-1, 0, ..., 0] @ t <= 0
    A_ub = np.hstack([A, -np.eye(m)])
    b_ub = np.zeros(m)
    
    # Equality constraint: sum(g) = S
    A_eq = np.hstack([np.ones(n), [[0]]])
    b_eq = [S]
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': timeout})
    
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def smooth_score(f, tau=10.0):
    """Smooth approximation of the score using log-sum-exp."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return np.inf, np.zeros(n)
    
    c = fft_convolve(f, f)
    # Log-sum-exp for numerical stability
    c_max = np.max(c)
    log_sum = c_max + np.log(np.sum(np.exp(tau * (c - c_max))))
    smooth_max = (1.0 / tau) * log_sum
    score = 2.0 * n * smooth_max / (S * S)
    
    # Gradient
    weights = np.exp(tau * (c - c_max))
    weights /= np.sum(weights)
    g = np.zeros(n)
    for k in range(len(c)):
        if weights[k] > 1e-15:
            # d(c[k])/d(f[j]) = f[k-j] if 0 <= k-j < n
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                g[j] += 2.0 * f[k - j] * weights[k]
    g /= tau
    # d(score)/d(f_j) = (2n * c'_k / S^2) - (2n * max_c * 2*S / S^4) * 1 ... 
    # Actually score = 2n * smooth_max / S^2
    # d(score)/d(f_j) = 2n * (d smooth_max/d f_j) / S^2 - 2n * smooth_max * 2 * (dS/d f_j) / S^3
    # = 2n * g_j / S^2 - 4n * smooth_max / S^3
    grad = 2.0 * n * g / (S * S) - 4.0 * n * smooth_max / (S * S * S)
    
    return score, grad

Multi-strategy optimization for minimizing peak aperiodic convolution
of non-negative sequences. Combines LP direction-finding, smooth-max
gradient descent, and local search with diverse initializations.

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_max_and_grad(c, tau):
    """Compute smooth max and gradient weights using log-sum-exp."""
    c_max = np.max(c)
    shifted = tau * (c - c_max)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = c_max + (np.log(np.sum(exp_shifted)) - np.log(tau))
    # Actually: smooth_max = (1/tau) * log(sum(exp(tau*c)))
    # = (1/tau) * (tau*c_max + log(sum(exp(tau*(c-c_max)))))
    # = c_max + (1/tau)*log(sum(exp(tau*(c-c_max))))
    smooth_max = c_max + (1.0/tau) * np.log(np.sum(exp_shifted))
    weights = exp_shifted / np.sum(exp_shifted)
    return smooth_max, weights

def solve_lp_min_peak(f, S, timeout=3.0):
    """Solve: min M' s.t. conv(f,g)[k] <= M' for all k, sum(g)=S, g>=0."""
    n = len(f)
    m = 2 * n - 1
    
    # A matrix: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(m - 1, j + n - 1)
        A[lo:hi+1, j] = f[:hi-lo+1]
    
    # Variables: [g_0, ..., g_{n-1}, M']
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M'
    
    # Inequality: A @ g - M' <= 0 for each k
    A_ub = np.hstack([A, -np.eye(m)])
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None


def solve_lp_max_sum(f, M, timeout=3.0):
    """Solve: max sum(b) s.t. conv(f,b)[k] <= M for all k, b>=0."""
    n = len(f)
    m = 2 * n - 1
    
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(m - 1, j + n - 1)
        A[lo:hi+1, j] = f[:hi-lo+1]
    
    c_obj = -np.ones(n)  # minimize -sum(b)
    A_ub = A
    b_ub = np.full(m, M)
    
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def make_exp_init(n, alpha, rng):
    """Exponential ramp initialization."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = np.exp(alpha * t)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_poly_init(n, p, rng):
    """Polynomial ramp: (j/n)^p."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = t**p + 0.1
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bell_init(n, center, width, rng):
    """Gaussian bell centered at 'center' fraction of n."""
    t = np.arange(n, dtype=np.float64)
    mu = center * (n - 1)
    a = np.exp(-0.5 * ((t - mu) / width) ** 2)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_hann_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def optimize_lp(f_init, time_limit, t_step=0.1, use_min_peak=True, verbose=True):
    """LP-based iterative optimizer."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()

    stall = 0

            break
        
        c = fft_convolve(f, f)

        else:
            g = solve_lp_max_sum(f, M, timeout=2.0)
            if g is None:

            direction = g
            direction = direction * (S / max(1e-15, np.sum(direction)))

        best_t = t_step
        best_s = np.inf
        for t_try in [t_step, 0.5*t_step, 0.2*t_step, 2.0*t_step]:
            f_cand = (1.0 - t_try) * f + t_try * direction
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try
        
        f = (1.0 - best_t) * f + best_t * direction
        f = np.clip(f, 0.01, 1000.0)
        f = f / np.sum(f)  # re-normalize
        it += 1
        
        if best_s < best_score - 1e-12:

def gradient_optimize(f_init, time_limit, verbose=True):
    """Optimize using smooth-max gradient with L-BFGS-B and adaptive tau."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S0 = float(np.sum(f))
    f = f / S0  # normalize
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    
    # Adaptive tau schedule
    tau_schedule = [5, 10, 20, 50, 100, 200, 500]
    tau_idx = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break
        
        tau = tau_schedule[min(tau_idx, len(tau_schedule) - 1)]
        remaining = time_limit - (time.monotonic() - t0)
        phase_time = min(remaining * 0.3, 60.0)
        
        # Define objective for L-BFGS-B
        def obj(x):
            x = np.clip(x, 0.0, 1000.0)
            s, g = smooth_score_and_grad(x, tau)
            return s, g
        
        # Run L-BFGS-B
        try:
            res = minimize(obj, f, method='L-BFGS-B', jac=True,
                          options={'maxiter': 200, 'maxfun': 500,
                                   'maxls': 50, 'ftol': 1e-12, 'gtol': 1e-8})
            if res.x is not None:
                f_new = np.clip(res.x, 0.01, 1000.0)
                f_new = f_new / np.sum(f_new)
                s_new = score_fast(f_new)
                if s_new < best_score:
                    best_score = s_new
                    best_f = f_new.copy()
                f = f_new
        except Exception as e:
            if verbose:
                print(f"    L-BFGS-B error: {e}")
        
        if verbose and tau_idx % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [grad {elapsed:6.1f}s] tau={tau} score={score_fast(f):.6f} best={best_score:.6f}")
        
        tau_idx += 1
        if tau_idx >= len(tau_schedule):
            break
    
    return best_f, best_score


def local_search(f_init, time_limit, verbose=True):
    """Perturbation-based local search."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    temp = 0.001
    cooling = 0.9999
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break
        
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.3:
            # Perturb near peaks
            c = fft_convolve(f, f)
            peak_idx = int(np.argmax(c))
            # Reduce contributions to the peak
            contribs = np.zeros(n)
            for j in range(n):
                idx = peak_idx - j
                if 0 <= idx < n:
                    contribs[j] = f[j] * f[idx]
            top_k = max(2, n // 10)
            top_mask = contribs >= np.sort(contribs)[-top_k]
            candidate[top_mask] *= np.exp(rng.normal(-0.05, 0.03, size=top_mask.sum()))
            # Slightly increase others
            other = ~top_mask
            if other.sum() > 0:
                idxs = np.where(other)[0]
                chosen = rng.choice(idxs, size=min(5, len(idxs)), replace=False)
                candidate[chosen] *= np.exp(rng.normal(0.05, 0.03, size=len(chosen)))
        elif r < 0.6:
            # Random smooth perturbation
            kernel_size = max(1, n // 16)
            noise = rng.normal(0, 0.02, size=n)
            # Smooth the noise
            kernel = np.ones(kernel_size) / kernel_size
            noise = np.convolve(noise, kernel, mode='same')
            candidate *= np.exp(noise)
        elif r < 0.8:
            # Shift weight from one region to another
            half = n // 2
            amount = rng.uniform(0.01, 0.05)
            source = rng.choice([0, half])
            target = half if source == 0 else 0
            region_size = max(1, n // 8)
            candidate[source:source+region_size] *= (1 - amount)
            candidate[target:target+region_size] *= (1 + amount)
        else:
            # Global smooth change
            t = np.arange(n) / max(1, n-1)
            shift = rng.uniform(-0.1, 0.1)
            candidate *= np.exp(shift * (t - 0.5) * 2)
        
        candidate = np.clip(candidate, 0.01, 1000.0)
        candidate = candidate / np.sum(candidate)
        new_score = score_fast(candidate)
        
        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()
            f = candidate
        elif rng.random() < np.exp(-(new_score - score_fast(f)) / max(temp, 1e-15)):
            f = candidate
        
        temp *= cooling
        it += 1
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [LS {elapsed:6.1f}s] it={it} score={score_fast(f):.6f} best={best_score:.6f}")
    
    return best_f, best_score
