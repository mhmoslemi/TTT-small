"""
Multi-peak subgradient optimization with momentum and simulated annealing
for minimizing the normalized peak self-convolution of non-negative step functions.
Uses FFT-based convolution, adaptive step sizes, multiple sequence lengths,
and periodic restarts to escape local minima.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def subgradient_single_peak(a, k):
    """
    Subgradient of conv(a,a)[k] w.r.t. a, computed in O(n).
    d(conv[k])/d(a[i]) = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    g = np.zeros(n)
    i_min = max(0, k - n + 1)
    i_max = min(k, n - 1)
    if i_min <= i_max:
        j_min = k - i_max
        j_max = k - i_min
        g[i_min:i_max + 1] = 2.0 * a[j_min:j_max + 1][::-1]
    return g


def normalize_to(a, S):
    """Normalize sequence to have sum S."""
    s = np.sum(a)
    if s < 1e-14:
        return np.full_like(a, S / len(a))
    return a * S / s


def make_initialization(n, init_type, rng):
    """Create various initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.3 + 0.4 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.3))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 5))
        amp = 0.2 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif init_type == 'bimodal':
        center = n // 2
        sigma = n * (0.1 + 0.15 * rng.random())
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'flat_spectrum':
        # Generate random phases for the DFT, take abs of IFFT
        nfft = 1
        while nfft < 2 * n:
            nfft <<= 1
        F = np.zeros(nfft // 2 + 1, dtype=np.complex128)
        F[0] = n * 0.5  # DC component to keep signal mostly positive
        for k in range(1, nfft // 2 + 1):
            F[k] = np.exp(1j * rng.uniform(0, 2 * np.pi))
        sig = np.fft.irfft(F, nfft)[:n]
        a = np.abs(sig)
    else:
        a = np.ones(n)
    a = np.clip(a, 0.01, None)
    return a


def optimize_multipeak(a, time_limit, K=8, verbose=False):
    """
    Multi-peak subgradient optimization with momentum and adaptive step size.
    Key: uses top K peaks of the convolution to build a combined subgradient,
    with momentum to accelerate convergence and annealing to escape local minima.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = a.copy().astype(np.float64)
    a = normalize_to(a, S)

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)

    lr = 0.5 / np.sqrt(n)
    beta = 0.8
    v = np.zeros(n)
    stall = 0
    iteration = 0
    rng = np.random.default_rng(42)
    temperature = 0.05
    historical_peaks = []

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        max_conv = np.max(conv)

        # Find top K peaks
        top_idx = np.argsort(conv)[-K:]

        # Also include some historical peaks for stability
        if len(historical_peaks) > 0:
            all_peaks = list(set(list(top_idx) + historical_peaks[-3:]))
        else:
            all_peaks = list(top_idx)

        # Update historical peaks
        historical_peaks.append(int(top_idx[0]))
        if len(historical_peaks) > 20:
            historical_peaks.pop(0)

        # Compute combined subgradient
        g = np.zeros(n)
        for k in all_peaks:
            g += subgradient_single_peak(a, k)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            lr *= 0.9
            continue
        g = g / g_norm

        # Momentum
        v = beta * v + (1.0 - beta) * g

        # Gradient step with projection
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to(a_new, S)

        # Evaluate
        new_score = compute_score(a_new)
        current_score = 2.0 * n * max_conv / (S ** 2)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slightly increase lr
            lr = min(lr * 1.02, 2.0 / np.sqrt(n))
            beta = min(beta + 0.002, 0.9)
        elif new_score < current_score * 1.01:
            a = a_new
            stall += 1
        else:
            stall += 1
            lr *= 0.97

        # Annealing: accept worse moves with some probability
        if new_score > current_score and rng.random() < np.exp(-(new_score - current_score) / temperature):
            a = a_new

        # Adaptive temperature
        temperature *= 0.9995
        temperature = max(temperature, 0.001)

        # Stall handling: perturbation + parameter reset
        if stall > 80:
            # Random perturbation to escape
            perturb_scale = lr * 0.5
            a += rng.normal(0, perturb_scale, n)
            a = np.maximum(a, 0.0)
            a = normalize_to(a, S)
            lr *= 0.7
            beta = max(0.5, beta - 0.1)
            stall = 0
            temperature = max(temperature, 0.02)

        # Very long stall: bigger restart
        if stall > 200:
            # Blend with a random sequence
            a_rand = make_initialization(n, 'random', rng)
            a_rand = normalize_to(a_rand, S)
            a = 0.7 * best_a + 0.3 * a_rand
            a = np.maximum(a, 0.0)
            a = normalize_to(a, S)
            lr = 0.5 / np.sqrt(n)
            beta = 0.8
            v = np.zeros(n)
            stall = 0
            temperature = 0.05

        iteration += 1

        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad {iteration} ({elapsed:.0f}s): best={best_score:.6f}, lr={lr:.5f}")

    return best_a


def local_search_refine(a, time_limit, verbose=False):

    Local search: targeted perturbations focusing on peak contributors.
    Uses coordinate-style updates on the top contributors to the peak.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = a.copy().astype(np.float64)
    a = normalize_to(a, S)

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(77)

    iteration = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_pos = int(np.argmax(conv))
        peak_val = conv[peak_pos]

        # Find top contributors to the peak
        # conv[peak_pos] = sum_i a[i]*a[peak_pos-i]
        contribs = np.zeros(n)
        i_min = max(0, peak_pos - n + 1)
        i_max = min(peak_pos, n - 1)
        for i in range(i_min, i_max + 1):
            j = peak_pos - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        # Strategy 1: Reduce top contributors, increase low ones
        top_k = max(5, n // 10)
        top_idx = np.argsort(contribs)[-top_k:]
        bottom_idx = np.argsort(contribs)[:top_k]

        candidate = a.copy()
        # Reduce top contributors
        reduce_factors = np.exp(rng.normal(-0.08, 0.04, size=len(top_idx)))
        candidate[top_idx] *= reduce_factors
        # Increase bottom contributors
        inc_factors = np.exp(rng.normal(0.08, 0.04, size=len(bottom_idx)))
        candidate[bottom_idx] *= inc_factors
        candidate = np.maximum(candidate, 0.0)
        candidate = normalize_to(candidate, S)

        new_score = compute_score(candidate)
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
            a = candidate
            stall = 0
        elif new_score < peak_val * 2 * n / (S ** 2) * 1.05:
            if rng.random() < 0.3:
                a = candidate
            stall += 1
        else:
            stall += 1

        # Strategy 2: Block perturbation
        if iteration % 3 == 1:
            block_size = max(2, n // 40)
            start = int(rng.integers(0, n - block_size))
            candidate2 = a.copy()
            candidate2[start:start + block_size] *= np.exp(rng.normal(0, 0.05, size=block_size))
            candidate2 = np.maximum(candidate2, 0.0)
            candidate2 = normalize_to(candidate2, S)
            s2 = compute_score(candidate2)
            if s2 < best_score:
                best_score = s2
                best_a = candidate2.copy()
                a = candidate2
                stall = 0

        # Strategy 3: Mirror swap
        if iteration % 5 == 3:
            candidate3 = a.copy()
            swap_size = max(1, n // 50)
            pos1 = int(rng.integers(0, n - swap_size))
            pos2 = int(rng.integers(0, n - swap_size))
            if pos1 != pos2:
                candidate3[pos1:pos1 + swap_size], candidate3[pos2:pos2 + swap_size] = \
                    candidate3[pos2:pos2 + swap_size].copy(), candidate3[pos1:pos1 + swap_size].copy()
                candidate3 = normalize_to(candidate3, S)
                s3 = compute_score(candidate3)
                if s3 < best_score:
                    best_score = s3
                    best_a = candidate3.copy()
                    a = candidate3
                    stall = 0

        # Stall handling
        if stall > 100:
            a = 0.8 * best_a + 0.2 * make_initialization(n, 'random', rng)
            a = np.maximum(a, 0.0)
            a = normalize_to(a, S)
            stall = 0

        iteration += 1
        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Local {iteration} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a


def optimize_combined(a, time_limit):
    """
    Combined optimization: alternate between subgradient and local search.
    """
    n = len(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)

    # Phase A: Subgradient (60% of time)
    t_sub = time_limit * 0.6
    result = optimize_multipeak(a, t_sub, K=10, verbose=True)
    s = compute_score(result)
    if s < best_score:
        best_score = s
        best_a = result

    # Phase B: Local search (30% of time)
    remaining = time_limit - (time.monotonic() - t0)
    t_local = remaining * 0.5
    if t_local > 5:
        result = local_search_refine(best_a, t_local, verbose=True)
        s = compute_score(result)
        if s < best_score:
            best_score = s
            best_a = result

    # Phase C: Final subgradient polish (10% of time)
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 5:
        result = optimize_multipeak(best_a, remaining, K=15, verbose=False)
        s = compute_score(result)
        if s < best_score:
            best_score = s
            best_a = result

    return best_a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing the normalized peak self-convolution.
    Uses multi-peak subgradient with momentum, local search, and multiple
    sequence lengths/initializations.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def try_update(a):
        nonlocal best_score, best_seq
        a = np.clip(a, 0.0, 1000.0)
        s = compute_score(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()
            return True
        return False

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        print(f"Provided: n={len(provided_seq)}, score={compute_score(provided_seq):.6f}")
        try_update(provided_seq)

    # === Phase 1: Optimize the provided sequence (most promising) ===
    if provided_seq is not None:
        n = len(provided_seq)
        t_alloc = min(budget_s * 0.40, deadline - time.monotonic())
        print(f"\n=== Phase 1: Combined optimization (n={n}, t={t_alloc:.0f}s) ===")
        result = optimize_combined(provided_seq, t_alloc)
        try_update(result)
        print(f"  Phase 1 best: {best_score:.6f}")

    # === Phase 2: Try other lengths from provided (interpolated) ===
    if provided_seq is not None and time.monotonic() < deadline - 120:
        print(f"\n=== Phase 2: Other lengths ===")
        for n_try in [512, 768, 1024]:
            if time.monotonic() > deadline - 60:
                break
            t_per = min(40.0, (deadline - time.monotonic()) * 0.25)
            if t_per < 10:
                break
            a0 = interpolate_sequence(provided_seq, n_try)
            a0 = np.clip(a0, 0.01, None)
            print(f"  n={n_try}, t={t_per:.0f}s")
            result = optimize_combined(a0, t_per)
            if try_update(result):
                print(f"  *** n={n_try}: new best {best_score:.6f} ***")
        print(f"  Phase 2 best: {best_score:.6f}")

    # === Phase 3: Different initializations at promising lengths ===
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 3: New initializations ===")
        init_configs = [
            (640, 'chirp'), (640, 'flat_spectrum'), (640, 'multi_tone'),
            (512, 'sine'), (768, 'bimodal'), (512, 'flat_spectrum'),
        ]
        for n, init_type in init_configs:
            if time.monotonic() > deadline - 30:
                break
            t_per = min(25.0, (deadline - time.monotonic()) * 0.3)
            if t_per < 8:
                break
            a0 = make_initialization(n, init_type, rng)
            print(f"  n={n}, init={init_type}, t={t_per:.0f}s")
            result = optimize_combined(a0, t_per)
            if try_update(result):
                print(f"  *** n={n}, {init_type}: new best {best_score:.6f} ***")
        print(f"  Phase 3 best: {best_score:.6f}")

    # === Phase 4: Final intensive polish on the best ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Phase 4: Final polish ===")
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            # Alternate between subgrad and local search
            n = len(best_seq)
            t_each = remaining / 3.0
            result = optimize_multipeak(best_seq, t_each, K=12, verbose=True)
            try_update(result)
            result = local_search_refine(best_seq, t_each, verbose=False)
            try_update(result)
            result = optimize_multipeak(best_seq, t_each, K=20, verbose=False)
            try_update(result)
            print(f"  Phase 4 best: {best_score:.6f}")

    # Final
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)}, local={best_score:.6f} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
