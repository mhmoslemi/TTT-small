"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth minimax optimization
with log-sum-exp continuation, FISTA acceleration, and subgradient polish.
Key innovations: FFT-based gradient computation for the smooth objective,
adaptive tau continuation, diverse structured initializations (chirp, window,
exponential), and a final exact subgradient polish phase.
"""

import numpy as np
import time


def fft_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fft_cross_corr(p, a):
    """Compute cross-correlation g[i] = sum_j p[i+j]*a[j] for i=0..len(a)-1."""
    n = len(a)
    len_p = len(p)
    need = len_p + n - 1
    fft_size = 1 << (need - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    xcorr = np.fft.irfft(P * np.conj(A), fft_size)
    return xcorr[:n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_and_grad(a, tau):
    """Compute F_tau(a) = (1/tau)*log(sum exp(tau*conv)) and its gradient."""
    conv = fft_convolve(a)
    max_conv = conv.max()
    shifted = tau * (conv - max_conv)
    shifted -= shifted.max()
    p = np.exp(shifted)
    p_sum = p.sum()
    F = max_conv + (shifted.max() + np.log(p_sum)) / tau
    p /= p_sum
    # Gradient: 2 * xcorr(p, a)
    grad = 2.0 * fft_cross_corr(p, a)
    return F, grad, conv, p


def fista_smooth(a0, tau, time_limit, lr_init=0.5, max_iter=5000):
    """FISTA for smooth objective on the simplex. Returns (best_a, best_F)."""
    n = len(a0)
    a = a0.copy()
    a_prev = a.copy()
    v = np.zeros(n)
    t_k = 1.0
    lr = lr_init
    
    F, grad, conv, p = smooth_obj_and_grad(a, tau)
    best_F = F
    best_a = a.copy()
    t0 = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        y = a + ((t_k - 1) / (t_k + 1)) * (a - a_prev)
        y = np.maximum(y, 0)
        s = y.sum()
        if s > 1e-15:
            y = y / s
        
        F_y, grad_y, _, _ = smooth_obj_and_grad(y, tau)
        
        # Backtracking on step size
        lr_cur = lr
        for _bt in range(20):
            a_new = project_simplex(y - lr_cur * grad_y)
            F_new, _, _, _ = smooth_obj_and_grad(a_new, tau)
            if F_new <= F_y + np.dot(grad_y, a_new - y) + (1.0 / (2.0 * lr_cur)) * np.sum((a_new - y) ** 2) + 1e-10:
                break
            lr_cur *= 0.5
        lr = lr_cur
        
        a_prev = a
        a = a_new
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        
        if F_new < best_F:
            best_F = F_new
            best_a = a.copy()
    
    return best_a, best_F, conv


def subgradient_polish(a0, time_limit, lr_init=0.02, momentum=0.9, top_k=50,
                       temp_init=0.001, decay=0.9998):
    """Subgradient descent with momentum on the exact max objective."""
    n = len(a0)
    a = project_simplex(a0.copy())
    v = np.zeros(n)
    conv = fft_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        peak_vals = conv[peaks]
        scaled = peak_vals / max(temp, 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            wi = w[i] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fft_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 2000:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fft_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(lr * 0.5, 0.001)
        
        lr *= decay
        temp *= 1.003
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Exponential decay
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        e = np.exp(-alpha * x / n)
        inits[f'exp_{alpha}'] = e / e.sum()
    
    # Two-sided exponential (centered)
    for alpha in [1.0, 2.0, 4.0]:
        e = np.exp(-alpha * np.abs(x - (n - 1) / 2.0) / n)
        inits[f'biexp_{alpha}'] = e / e.sum()
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    rc = np.maximum(rc, 1e-10)
    inits['raised_cos'] = rc / rc.sum()
    
    # Hanning window
    h = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    h = np.maximum(h, 1e-10)
    inits['hanning'] = h / h.sum()
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    
    # Bimodal (edges heavy)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Chirp-like: quadratic phase made real via envelope
    chirp_env = np.exp(-0.5 * ((x - n / 2) / (n * 0.3)) ** 2)
    inits['chirp_env'] = chirp_env / chirp_env.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits


def continuation_optimize(a0, time_limit, tau_schedule=None):
    """Run log-sum-exp continuation with FISTA at each tau level."""
    if tau_schedule is None:
        tau_schedule = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 3000.0, 8000.0]
    
    n = len(a0)
    a = project_simplex(a0.copy())
    t0 = time.monotonic()
    
    for tau in tau_schedule:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        t_alloc = max(1.0, remaining * 0.7)
        a, F, conv = fista_smooth(a, tau, time_limit=t_alloc)
    
    return a, conv
