"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Projected subgradient descent with momentum for fast local optimization
2. LP-based direction finding for principled descent directions
3. Coordinate descent (paired mass transfers) for final polishing
4. Multiple initializations and restarts to escape local minima
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def proj_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][0]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def subgradient(a, k_star):
    """Compute subgradient of 2n*max(conv) at index k_star."""
    n = len(a)
    g = np.zeros(n)
    # g[i] = 4n * a[k_star - i] if 0 <= k_star - i < n
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        g[i] = 4.0 * n * a[k_star - i]
    return g


def subgradient_smooth(a, top_m=5):

    n = len(a)
    conv = fast_convolve(a)
    # Use top m peaks with softmax weighting
    vals = conv.copy()

    # Softmax-like weights
    temps = [1.0, 0.1, 0.01]

def pgd_optimize(a0, time_limit, lr_init=0.01, momentum=0.9):
    """Projected gradient descent with momentum and adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    best_a = a.copy()
    best_obj = objective(a)
    n = len(a)
    
    velocity = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Compute subgradient
        g = subgradient(a, k_star)
        
        # Momentum update
        velocity = momentum * velocity - lr * g
        a_new = proj_simplex(a + velocity)
        
        # Evaluate
        obj_new = objective(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Increase lr slightly
            lr = min(lr * 1.05, 0.5)
        else:
            stall += 1
            lr = lr * 0.5
            velocity = velocity * 0.5
            a = a  # Don't update
        
        if stall >= max_stall:
            break
    
    return best_a, best_obj

    # So g[m] appears in constraint k with coefficient f[k-m] (where m = k-j)
    
    # Build sparse matrix
    row_idx = []
    col_idx = []
    vals = []
    
    for k in range(2 * n - 1):
        # g[m] appears with coefficient f[k-m] where max(0, k-n+1) <= m <= min(n-1, k)
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            coeff = f[k - m]
            if coeff > 1e-15:
                row_idx.append(k)
                col_idx.append(m)
                vals.append(coeff)
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize(a0, time_limit, max_iters=1000):
    """LP-guided optimization with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    best_a = a.copy()
    best_obj = objective(a)
    n = len(a)
    t0 = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_lp_direction(a, M)
        
        if np.allclose(g, a, atol=1e-10):
            stall += 10
        else:
            # Line search
            best_t = 0
            best_t_obj = best_obj
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_trial = (1 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts < 1e-12:
                    continue
                a_trial /= ts
                obj_t = objective(a_trial)
                if obj_t < best_t_obj:
                    best_t_obj = obj_t
                    best_t = t
            
            if best_t > 0 and best_t_obj < best_obj - 1e-12:
                a = (1 - best_t) * a + best_t * g
                a = np.maximum(a, 0)
                a /= a.sum()
                best_a = a.copy()
                best_obj = best_t_obj
                stall = 0
            else:
                stall += 1
        
        if stall >= 50:
            break
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Paired mass transfer coordinate descent."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    n = len(a)
    best_a = a.copy()
    best_obj = objective(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Find elements that contribute most to the peak
        contrib = np.zeros(n)
        for i in range(n):
            m = k_star - i
            if 0 <= m < n:
                contrib[i] = a[i] * a[m]
        
        # Sort by contribution: high contributors should lose mass, low should gain
        order = np.argsort(contrib)
        givers = order[-20:]   # Top 20 contributors (highest)
        receivers = order[:20]  # Bottom 20 (lowest)
        
        found = False
        for gi in givers:
            if found:
                break
            for ri in receivers:
                if gi == ri:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amount = a[gi] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[gi] -= amount
                    a_trial[ri] += amount
                    a_trial = np.maximum(a_trial, 0)
                    a_trial /= a_trial.sum()
                    obj_t = objective(a_trial)
                    if obj_t < best_obj - 1e-10:
                        best_obj = obj_t
                        a = a_trial.copy()
                        best_a = a.copy()
                        found = True
                        break
                if found:
                    break
        
        if not found:
            stall += 1
            if stall > 100:
                break
        else:
            stall = 0
    
    return best_a, best_obj


def make_initializations(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian
    for sigma in [0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sigma * n)) ** 2)
        inits[f'gauss_{sigma}'] = g / g.sum()
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Fejér kernel (sinc^2 window)
    t = (x - 0.5 * (n - 1)) / (0.5 * (n - 1) + 1)
    fejer = np.sinc(t) ** 2  # sinc(x) = sin(pi*x)/(pi*x)
    fejer = np.maximum(fejer, 0)
    if fejer.sum() > 0:
        inits['fejer'] = fejer / fejer.sum()
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits['random'] = r
    
    # Edge-concentrated
    edge = np.zeros(n)
    m = n // 3
    edge[:m] = 1.0
    edge[-m:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s > 1e-10:
            inits['parent'] = pc / s
    except Exception:
        pass
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Diverse initializations (m-sequence/chirp inspired, structured, random, parent)
2. Mirror descent (exponential weights) for fast simplex-constrained optimization
3. LP-guided direction finding for principled descent
4. Coordinate descent (paired mass transfer) for final polishing
Combines FFT-based convolution, adaptive step sizes, and time-budget management.

import numpy as np
import scipy.optimize
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def proj_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[0]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_val(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def compute_gradient_at_peak(a, k_star):
    """Compute subgradient of 2n*max_k conv_k at index k_star.
    grad[i] = 4n * a[k_star - i] if 0 <= k_star - i < n, else 0."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        g[lo:hi + 1] = 4.0 * n * a[k_star - hi:k_star - lo + 1][::-1]
    return g


def compute_smooth_gradient(a, top_m=3):
    """Average subgradient over top-m convolution peaks."""
    n = len(a)
    conv = fast_convolve(a)
    top_idx = np.argsort(conv)[::-1][:top_m]
    g = np.zeros(n)
    for idx in top_idx:
        lo = max(0, idx - n + 1)
        hi = min(n - 1, idx)
        if lo <= hi:
            g[lo:hi + 1] += 4.0 * n * a[idx - hi:idx - lo + 1][::-1]
    g /= top_m
    return g


def mirror_descent_step(a, g, lr):
    """Exponential weights (mirror descent) step on the simplex."""
    # a_new[i] proportional to a[i] * exp(-lr * g[i])
    log_update = np.where(a > 1e-300, np.log(a), -300.0) - lr * g
    log_update -= log_update.max()  # numerical stability
    a_new = np.exp(log_update)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def mirror_descent_optimize(a0, time_limit, lr_init=0.005, top_m=3):
    """Mirror descent (exponential weights) optimization with adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = objective_val(a)
    n = len(a)
    t0 = time.monotonic()
    stall = 0
    max_stall = 150
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        g = compute_smooth_gradient(a, top_m=top_m)

        # Mirror descent update
        a_new = mirror_descent_step(a, g, lr)

        obj_new = objective_val(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slightly increase lr if making progress
            if stall == 0:
                lr = min(lr * 1.02, 0.5)
        else:
            stall += 1
            lr = lr * 0.7

        if stall >= max_stall:
            break

    return best_a, best_obj


def build_lp_matrix(f):
    """Build sparse constraint matrix for LP: (f*g)[k] <= M for all k."""
    n = len(f)
    row_idx = []
    col_idx = []
    vals = []
    for k in range(2 * n - 1):
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            coeff = f[k - m]
            if coeff > 1e-15:
                row_idx.append(k)
                col_idx.append(m)
                vals.append(coeff)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                     method='highs', options={'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize(a0, time_limit, max_iters=500):
    """LP-guided optimization with multi-point line search."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = objective_val(a)
    n = len(a)
    t0 = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(a, M)

        if np.allclose(g, a, atol=1e-10):
            stall += 5
        else:
            # Multi-point line search
            best_t = 0
            best_t_obj = best_obj
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                ts = a_trial.sum()
                if ts < 1e-12:
                    continue
                a_trial = a_trial / ts
                obj_t = objective_val(a_trial)
                if obj_t < best_t_obj:
                    best_t_obj = obj_t
                    best_t = t

            if best_t > 0 and best_t_obj < best_obj - 1e-12:
                a = (1.0 - best_t) * a + best_t * g
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_a = a.copy()
                best_obj = best_t_obj
                stall = 0
            else:
                stall += 1

        if stall >= 40:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_givers=15, n_receivers=15):
    """Paired mass transfer coordinate descent."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s

    n = len(a)
    best_a = a.copy()
    best_obj = objective_val(a)
    t0 = time.monotonic()
    stall = 0
    max_stall = 200

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            contrib[i] = a[i] * a[k_star - i]

        # Also consider secondary peaks
        # Find top 3 peaks and accumulate contributions
        top_peaks = np.argsort(conv)[::-1][:3]
        total_contrib = np.zeros(n)
        for pk in top_peaks:
            lo2 = max(0, pk - n + 1)
            hi2 = min(n - 1, pk)
            for i in range(lo2, hi2 + 1):
                total_contrib[i] += a[i] * a[pk - i] * (conv[pk] / conv.max())

        order = np.argsort(total_contrib)
        receivers = order[:n_receivers]
        givers = order[-n_givers:]

        found = False
        for gi in givers:
            if found:
                break
            if a[gi] < 1e-15:
                continue
            for ri in receivers:
                if found:
                    break
                if gi == ri:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
                    amount = a[gi] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[gi] -= amount
                    a_trial[ri] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    obj_t = objective_val(a_trial)
                    if obj_t < best_obj - 1e-11:
                        best_obj = obj_t
                        a = a_trial.copy()
                        best_a = a.copy()
                        found = True
                        break

        if not found:
            stall += 1
            if stall > max_stall:
                break
        else:
            stall = 0

    return best_a, best_obj


def generate_m_sequence(n):
    """Generate a maximum-length sequence using LFSR. Returns +/- 1 array of length n."""
    # Find smallest m with 2^m - 1 >= n
    m = 1
    while (1 << m) - 1 < n:
        m += 1
    # Use a known primitive polynomial for small m
    # For m=7 (n=127): x^7 + x^4 + 1
    # For m=8 (n=255): x^8 + x^6 + x^5 + x^4 + 1  (actually let's use a standard one)
    # m=9 (n=511): x^9 + x^5 + 1
    primitive_poly = {
        2: 0b11,      # x^2+x+1
        3: 0b1011,    # x^3+x+1
        4: 0b10011,   # x^4+x+1
        5: 0b100101,  # x^5+x^2+1
        6: 0b1010001, # x^6+x^5+1
        7: 0b10001001,# x^7+x^4+1
        8: 0b101101001, # x^8+x^6+x^5+x^4+1
        9: 0b1000100001, # x^9+x^5+1
        10: 0b11000010001, # x^10+x^7+x^3+x^2+1 (approximate)
    }
    if m not in primitive_poly:
        # Fallback: use chirp
        return np.exp(2j * np.pi * np.arange(n) ** 2 / n).real

    poly = primitive_poly[m]
    period = (1 << m) - 1

    # LFSR generation
    state = (1 << m) - 1  # Initialize to all ones
    seq = []
    for _ in range(period):
        seq.append(1 if (state & 1) else -1)
        # Feedback
        feedback = 1
        for bit in range(m - 1, -1, -1):
            if (poly >> bit) & 1:
                feedback ^= (state >> bit) & 1
        state = (state >> 1) | (feedback << (m - 1))

    seq = np.array(seq[:n])
    return seq


def make_initializations(n, rng):
    """Generate diverse initializations for sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussian bumps
    for sigma in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sigma * n)) ** 2)
        inits[f'gauss_{sigma}'] = g / g.sum()

    # Cosine window
    cos_w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    inits['cosine'] = cos_w / cos_w.sum()

    # Chirp-inspired: (1 + alpha * cos(2*pi*i^2/n))
    for alpha in [0.5, 0.8, 0.95]:
        chirp = 1.0 + alpha * np.cos(2.0 * np.pi * x ** 2 / n)
        chirp = np.maximum(chirp, 0.0)
        if chirp.sum() > 1e-10:
            inits[f'chirp_{alpha}'] = chirp / chirp.sum()

    # M-sequence inspired
    try:
        m_seq = generate_m_sequence(n)
        for alpha in [0.5, 0.8, 0.95]:
            a_chirp = 1.0 + alpha * m_seq
            a_chirp = np.maximum(a_chirp, 0.0)
            if a_chirp.sum() > 1e-10:
                inits[f'mseq_{alpha}'] = a_chirp / a_chirp.sum()
    except Exception:
        pass

    # Random Dirichlet
    for conc in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * conc)
        inits[f'rand_{conc}'] = r

    # Edge-concentrated
    edge = np.zeros(n)
    m = max(1, n // 4)
    edge[:m] = 1.0
    edge[-m:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Fejer kernel
    t = (x - 0.5 * (n - 1)) / (0.5 * n)
    fejer = np.sinc(t) ** 2
    fejer = np.maximum(fejer, 0.0)
    if fejer.sum() > 1e-10:
        inits['fejer'] = fejer / fejer.sum()

    # Parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            if p.size >= n:
                pc = p[:n].copy()
            else:
                pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
            pc = np.maximum(pc, 0.0)
            s = pc.sum()
            if s > 1e-10:
                inits['parent'] = pc / s
    except Exception:
        pass

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization combining mirror descent,
    LP-guided direction finding, and coordinate descent to minimize
    2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 25.0

    best_sequence = None
    best_score = float('inf')

    # Try primary length 512, and 1024 if time allows
    lengths_to_try = [512]
    if total_budget > 900:
        lengths_to_try.append(1024)

    for n in lengths_to_try:
        remaining = deadline - time.monotonic()
        if remaining < 90:
            break

        print(f"=== Starting optimization for n={n}, remaining={remaining:.0f}s ===")
        inits = make_initializations(n, rng)
        print(f"  {len(inits)} initializations: {list(inits.keys())}")

        candidates = []

        # Phase 1: Mirror descent on all initializations (30% of remaining)
        phase1_budget = min(remaining * 0.30, 300.0)
        per_init_time = max(3.0, phase1_budget / max(1, len(inits)))
        print(f"  Phase 1: Mirror descent ({phase1_budget:.0f}s total, {per_init_time:.1f}s each)")

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(per_init_time, deadline - time.monotonic() - 30)
            if t_alloc < 2:
                break
            a_opt, obj_opt = mirror_descent_optimize(a0, time_limit=t_alloc,
                                                     lr_init=0.005, top_m=3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                candidates.append((score, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"    MD {name}: {score:.6f}")

        candidates.sort(key=lambda x: x[0])
        if candidates:
            print(f"  Phase 1 best: {candidates[0][0]:.6f} ({candidates[0][1]})")

        # Phase 2: LP optimization on top candidates (40% of remaining)
        phase2_budget = min((deadline - time.monotonic()) * 0.40, 360.0)
        top_k = min(3, len(candidates))
        if top_k > 0 and phase2_budget > 30:
            per_lp_time = max(15.0, phase2_budget / top_k)
            print(f"  Phase 2: LP optimization on top {top_k} ({phase2_budget:.0f}s)")

            for i in range(top_k):
                if time.monotonic() > deadline - 30:
                    break
                score, name, a0 = candidates[i]
                t_alloc = min(per_lp_time, deadline - time.monotonic() - 30)
                if t_alloc < 5:
                    break

                a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc)
                score_lp = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score_lp):
                    candidates.append((score_lp, f'lp_{name}', a_opt.copy()))
                    if score_lp < best_score:
                        best_score = score_lp
                        best_sequence = a_opt.copy()
                        print(f"    LP {name}: {score_lp:.6f}")

            candidates.sort(key=lambda x: x[0])

        # Phase 3: Coordinate descent polish (remaining time minus buffer)
        remaining = deadline - time.monotonic() - 20
        if remaining > 20 and len(candidates) > 0:
            score, name, a0 = candidates[0]
            print(f"  Phase 3: CD polish on best ({remaining:.0f}s)")
            a_opt, obj_opt = coordinate_descent(a0, time_limit=remaining)
            score_cd = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_opt.copy()
                print(f"    CD {name}: {score_cd:.6f}")

            # Also polish the overall best
            if best_sequence is not None and time.monotonic() < deadline - 20:
                rem2 = deadline - time.monotonic() - 20
                if rem2 > 10:
                    a_opt2, _ = coordinate_descent(best_sequence, time_limit=rem2)
                    score_cd2 = float(evaluate_sequence(a_opt2.tolist()))
                    if np.isfinite(score_cd2) and score_cd2 < best_score:
                        best_score = score_cd2
                        best_sequence = a_opt2.copy()
                        print(f"    CD best: {score_cd2:.6f}")

    # Final: one more LP polish if time permits
    if best_sequence is not None and time.monotonic() < deadline - 30:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            print(f"  Final LP polish: {rem:.0f}s")
            a_opt, _ = lp_optimize(best_sequence, time_limit=rem)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"    Final LP: {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
