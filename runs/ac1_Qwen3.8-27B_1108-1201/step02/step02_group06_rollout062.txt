def fft_convolve(a, b):

def sa_optimize(a_init, time_limit, rng, temp_init=0.002, cooling=0.9999):

            other_idx = np.setdiff1d(np.arange(n), top_idx)

                n_inc = min(5, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.65:
            # Local block mutation

            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.8:
            # Global smooth perturbation
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))
        elif r < 0.9:

def make_init(n, type_name, rng):
    """Generate initial sequences of various types."""
    t = np.arange(n, dtype=np.float64)
    if type_name == 'constant':
        return np.ones(n)
    elif type_name == 'chirp':
        alpha = 0.2 + 0.5 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (2.0 * n))
        return np.clip(a, 0.01, None)
    elif type_name == 'sine':
        k = int(rng.integers(1, max(2, n // 4)))
        amp = 0.1 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * k * t / n + phase)
        return np.clip(a, 0.01, None)
    elif type_name == 'multitone':
        a = np.ones(n)
        for _ in range(4 + int(rng.integers(0, 4))):
            k = int(rng.integers(1, max(2, n // 3)))
            amp = rng.uniform(0.05, 0.25)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t / n + phase)
        return np.clip(a, 0.01, None)
    elif type_name == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif type_name == 'bump':
        center = n // 2
        width = n // (4 + int(rng.integers(0, 4)))
        a = 1.0 + rng.uniform(0.2, 0.8) * np.exp(-0.5 * ((t - center) / width) ** 2)
        return np.clip(a, 0.01, None)
    elif type_name == 'sawtooth':
        a = 1.0 + 0.3 * (2.0 * (t / n - np.floor(t / n + 0.5)))
        return np.clip(a, 0.01, None)
    else:
        return np.ones(n)

"""
Gradient-based optimization for minimizing normalized peak self-convolution
of non-negative step functions. Uses FFT-accelerated annealed softmax gradient
descent with adaptive step sizes, diverse initializations, and simulated
annealing polishing. Key improvement: O(n log n) gradient computation via FFT
enables thousands of iterations within the time budget.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_subgrad(a, w):
    """
    Compute subgrad[i] = sum_k w[k] * a[k-i] for valid indices.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    v = w[::-1]
    conv_av = fft_convolve(a, v)
    # subgrad[i] = conv_av[2n-2-i] for i=0,...,n-1
    return conv_av[n - 1:2 * n - 1][::-1]


def gradient_step(a, tau, step_size):
    """
    One gradient step for minimizing peak conv / sum^2 using annealed softmax.
    """
    n = len(a)
    conv = fft_convolve(a, a)
    S = np.sum(a)
    
    # Softmax weights over convolution lags
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / tau
    exp_conv = np.exp(shifted)
    w = exp_conv / np.sum(exp_conv)
    
    # Compute subgradient
    subgrad = compute_subgrad(a, w)
    
    # M_soft = weighted average of conv values
    M_soft = np.dot(w, conv)
    
    # Gradient direction: we want to decrease a where subgrad is high
    # (those positions contribute most to the peak)
    # direction[i] = M_soft/S - subgrad[i]
    # Positive direction means increase, negative means decrease
    direction = M_soft / S - subgrad
    
    # Normalize direction for stable steps
    dir_norm = np.sqrt(np.mean(direction ** 2))
    if dir_norm < 1e-15:
        return a.copy()
    direction = direction / dir_norm
    
    a_new = a + step_size * direction
    a_new = np.clip(a_new, 0.0, None)
    
    new_S = np.sum(a_new)
    if new_S < 1e-10:
        return a.copy()
    a_new = a_new * (S / new_S)
    
    return a_new


def gradient_optimize(a_init, time_limit, tau_init=0.3, tau_min=0.03,
                      step_init=0.3, verbose=False, seed=42):
    """
    Annealed gradient descent for minimizing the score.
    Adaptive step size with stagnation detection and restarts.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    # Normalize to sum = sqrt(2n)
    if S > 1e-10:
        a = a * (np.sqrt(2.0 * n) / S)
    
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    
    t0 = time.monotonic()
    tau = tau_init
    step = step_init
    stagnant = 0
    iter_count = 0
    rng_local = np.random.default_rng(seed)
    no_improve = 0
    best_no_improve_score = best_score
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)

            a = a_new
            current_score = new_score
            stagnant = 0
            no_improve = 0

                best_score = new_score
                best_a = a.copy()
                best_no_improve_score = best_score
        else:
            stagnant += 1
            no_improve += 1
            step *= 0.85
            
            if stagnant > 50:
                # Try a different tau
                tau = max(tau_min, tau * 0.5)
                stagnant = 0

                a = a + perturbation * rng_local.standard_normal(n)

def sa_optimize(a_init, time_limit, rng, temp_init=0.001, cooling=0.99995,
                reheat_threshold=2000, reheat_factor=2.0):
    """
    Simulated annealing for polishing with reheat mechanism.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = np.sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    
    t0 = time.monotonic()
    temperature = temp_init
    iter_count = 0
    iter_since_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.40:
            # Peak-targeted: reduce contributors to peak, boost others
            conv = fft_convolve(a, a)
            # Find top contributing positions
            peak_lags = np.argsort(conv)[-max(5, n // 6):]
            contribs = np.zeros(n)
            for pi in peak_lags[:max(3, n // 32)]:
                for i in range(n):
                    j = int(pi) - i
                    if 0 <= j < n:
                        contribs[i] += a[i] * a[j]
            top_k = max(5, n // 6)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.025, 0.01, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(max(3, n // 16), len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.025, 0.01, size=n_inc))
        elif r < 0.60:
            # Local block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.018, size=width))
        elif r < 0.75:
            # Global smooth perturbation
            candidate *= np.exp(rng.normal(0.0, 0.006, size=n))
        elif r < 0.85:
            # Paired mutation (preserve sum approximately)
            n_pairs = int(rng.integers(2, max(3, n // 32)))
            for _ in range(n_pairs):
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))
                delta = rng.normal(0, 0.015)
                candidate[i1] *= np.exp(delta)
                candidate[i2] *= np.exp(-delta)
        elif r < 0.95:
            # Small shift with noise
            shift = int(rng.integers(1, max(2, n // 25)))
            candidate = np.roll(a, shift)
            candidate *= np.exp(rng.normal(0.0, 0.003, size=n))
        else:
            # Larger random restart (rare)
            candidate = a * np.exp(rng.normal(0.0, 0.03, size=n))
        
        candidate = np.clip(candidate, 0.0, None)
        cand_S = np.sum(candidate)
        if cand_S < 1e-10:
            continue
        candidate = candidate * (S / cand_S)
        
        new_score = compute_score(candidate)
        iter_count += 1
        iter_since_improve += 1
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            iter_since_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
        else:
            diff = new_score - current_score
            if diff < 0.05:
                accept_prob = np.exp(-diff / max(temperature, 1e-15))
                if rng.random() < accept_prob:
                    a = candidate
                    current_score = new_score
        
        temperature *= cooling
        
        # Reheat if stuck
        if iter_since_improve > reheat_threshold:
            temperature = min(temp_init, temperature * reheat_factor)
            iter_since_improve = 0
    
    return best_a, best_score


def make_init(n, type_name, rng):
    """Generate initial sequences of various types."""
    t = np.arange(n, dtype=np.float64)
    if type_name == 'constant':
        return np.ones(n)
    elif type_name == 'chirp':
        alpha = 0.15 + 0.4 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (2.0 * n))
        return np.clip(a, 0.01, None)
    elif type_name == 'sine':
        k = int(rng.integers(1, max(2, n // 4)))
        amp = 0.1 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * k * t / n + phase)
        return np.clip(a, 0.01, None)
    elif type_name == 'multitone':
        a = np.ones(n)
        n_tones = 3 + int(rng.integers(0, 5))
        for _ in range(n_tones):
            k = int(rng.integers(1, max(2, n // 3)))
            amp = rng.uniform(0.05, 0.25)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t / n + phase)
        return np.clip(a, 0.01, None)
    elif type_name == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif type_name == 'bump':
        center = n // 2
        width = n // (4 + int(rng.integers(0, 4)))
        a = 1.0 + rng.uniform(0.2, 0.8) * np.exp(-0.5 * ((t - center) / width) ** 2)
        return np.clip(a, 0.01, None)
    elif type_name == 'sawtooth':
        a = 1.0 + 0.3 * (2.0 * (t / n - np.floor(t / n + 0.5)))
        return np.clip(a, 0.01, None)
    elif type_name == 'pseudo_random':
        # Golay-like or Barker-like inspired
        signs = rng.choice([-1, 1], size=n)
        a = 1.0 + 0.3 * signs
        # Smooth it
        kernel = np.ones(3) / 3.0
        a = np.convolve(a, kernel, mode='same')
        return np.clip(a, 0.01, None)
    elif type_name == 'walsh':
        # Walsh-Hadamard inspired (first few rows)
        a = np.ones(n)
        for level in range(min(4, int(np.log2(n)) + 1)):
            period = n // (2 ** level)
            if period < 2:
                break
            pattern = np.ones(period)
            pattern[period // 2:] = -1.0
            block = np.tile(pattern, n // period + 1)[:n]
            a = 1.0 + 0.15 / (level + 1) * block
        return np.clip(a, 0.01, None)
    else:
        return np.ones(n)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining FFT-accelerated gradient descent,
    diverse initializations, and simulated annealing to minimize the
    normalized peak self-convolution of non-negative step functions.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = compute_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()
    
    n_base = len(provided_seq) if provided_seq is not None else 256
    
    # === Phase 1: Quick gradient exploration on diverse initializations ===
    print("\n" + "=" * 60)
    print("Phase 1: Gradient descent exploration (diverse inits)")
    print("=" * 60)
    
    phase1_deadline = t_global + 500.0  # First ~500s for exploration
    
    # Configs: (n, init_type, time_alloc)
    exploration_configs = []
    
    # Most important: provided sequence
    if provided_seq is not None:
        exploration_configs.append((n_base, 'provided', 80))
    
    # Good initializations at base length
    exploration_configs.append((n_base, 'multitone', 50))
    exploration_configs.append((n_base, 'chirp', 50))
    exploration_configs.append((n_base, 'pseudo_random', 40))
    exploration_configs.append((n_base, 'sine', 40))
    exploration_configs.append((n_base, 'walsh', 40))
    exploration_configs.append((n_base, 'random', 30))
    exploration_configs.append((n_base, 'sawtooth', 25))
    exploration_configs.append((n_base, 'bump', 25))
    
    # Try other lengths
    for n_try in [128, 192, 384, 512]:
        if n_try != n_base:
            exploration_configs.append((n_try, 'multitone', 30))
            exploration_configs.append((n_try, 'chirp', 25))
    
    for n, init_name, alloc in exploration_configs:
        if time.monotonic() > phase1_deadline or time.monotonic() > deadline - 400:
            break
        
        actual_alloc = min(alloc, phase1_deadline - time.monotonic(), deadline - time.monotonic() - 350)
        if actual_alloc < 8:
            break
        
        if init_name == 'provided' and provided_seq is not None and n == n_base:
            a0 = provided_seq.copy()
        else:
            a0 = make_init(n, init_name, rng)
        
        print(f"\n  n={n}, init='{init_name}', alloc={actual_alloc:.0f}s")
        a_best, score = gradient_optimize(a0, actual_alloc,
                                           tau_init=0.3, tau_min=0.03,
                                           step_init=0.3, seed=seed + n + hash(init_name) % 1000)
        print(f"  Result: score={score:.6f}")
        
        if score < best_score:
            best_score = score
            best_seq = a_best.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}, init={init_name}) ***")
    
    # === Phase 2: Focused refinement on best result ===
    print("\n" + "=" * 60)
    print("Phase 2: Focused gradient refinement")
    print("=" * 60)
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        phase2_time = min(350, remaining * 0.45)
        if phase2_time > 20:
            n_cur = len(best_seq)
            print(f"  n={n_cur}, time={phase2_time:.0f}s")
            a2, score2 = gradient_optimize(best_seq, phase2_time,
                                            tau_init=0.1, tau_min=0.01,
                                            step_init=0.2, verbose=True,
                                            seed=seed + 1000)
            print(f"  Phase 2 result: {score2:.6f}")
            if score2 < best_score:
                best_score = score2
                best_seq = a2.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # === Phase 3: Simulated annealing polishing ===
    print("\n" + "=" * 60)
    print("Phase 3: Simulated annealing polish")
    print("=" * 60)
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        phase3_time = min(300, remaining * 0.55)
        if phase3_time > 15:
            n_cur = len(best_seq)
            print(f"  n={n_cur}, time={phase3_time:.0f}s")
            a3, score3 = sa_optimize(best_seq, phase3_time, rng,
                                      temp_init=0.0008, cooling=0.99997,
                                      reheat_threshold=3000, reheat_factor=2.0)
            print(f"  SA result: {score3:.6f}")
            if score3 < best_score:
                best_score = score3
                best_seq = a3.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # === Phase 4: Final gradient + SA micro-polish ===
    print("\n" + "=" * 60)
    print("Phase 4: Final polish")
    print("=" * 60)
    
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        
        # Quick gradient pass
        phase4a_time = min(120, remaining * 0.5)
        if phase4a_time > 10:
            a4, score4 = gradient_optimize(best_seq, phase4a_time,
                                            tau_init=0.05, tau_min=0.005,
                                            step_init=0.1, seed=seed + 2000)
            if score4 < best_score:
                best_score = score4
                best_seq = a4.copy()
                print(f"  After final gradient: {best_score:.6f}")
        
        # Micro SA
        if time.monotonic() < deadline - 10:
            sa_time = deadline - time.monotonic() - 5
            if sa_time > 5:
                a5, score5 = sa_optimize(best_seq, sa_time, rng,
                                          temp_init=0.0003, cooling=0.99999,
                                          reheat_threshold=5000, reheat_factor=1.5)
                if score5 < best_score:
                    best_score = score5
                    best_seq = a5.copy()
                    print(f"  After micro SA: {best_score:.6f}")
    
    # Final output
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"Total time: {time.monotonic() - t_global:.1f}s")
        print(f"{'=' * 60}")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
