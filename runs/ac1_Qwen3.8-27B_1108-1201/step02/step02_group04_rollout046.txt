    """Solve max sum(g) s.t. conv(f,g)_k <= M for top K peaks, g >= 0."""
    if n is None:
        n = len(f)

    # Get top K peak indices
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    # Build constraint matrix for only these K constraints
    rows_list = []
    cols_list = []
    vals_list = []
    for k in top_idx:

        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows_list.append(k)  # We'll re-index later
            cols_list.append(i)
            vals_list.append(f[k - i])
    
    # Re-index rows to 0..K-1
    k_to_idx = {int(k): idx for idx, k in enumerate(top_idx)}
    rows_arr = np.array([k_to_idx[r] for r in rows_list], dtype=np.int32)
    cols_arr = np.array(cols_list, dtype=np.int32)
    vals_arr = np.array(vals_list, dtype=np.float64)
    
    A = sparse.csc_matrix((vals_arr, (rows_arr, cols_arr)), shape=(top_k, n))

    bounds = [(0, None)] * n

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def compute_sg(a, w, n):
    """sg_i = 2*sum_j a_j * w_{i+j}, computed via FFT cross-correlation."""
    # cross_conv(a[::-1], w) gives conv where index n-1+i = sum_j a[j]*w[i+j]
    result = cross_conv(a[::-1], w)
    return 2.0 * result[n-1 : 2*n-1]

def mirror_step(a, T, eta, n):
    c = fast_autoconv(a)
    M = c.max()
    if M < 1e-15:
        return a, M
    scaled = c / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(2*n-1)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= ws
    sg = compute_sg(a, w, n)
    sg_c = sg - sg.mean()
    log_u = np.clip(-eta * sg_c, -4.0, 4.0)
    a_new = a * np.exp(log_u)
    a_new = np.maximum(a_new, 1e-30)
    a_new /= a_new.sum()
    return a_new, M

def solve_rp(f, M, K, n, timeout=3):
    """Reduced LP: max sum(g) s.t. top-K conv constraints."""
    c = fast_autoconv(f)
    top = np.argpartition(c, -K)[-K:]
    rows_l, cols_l, vals_l = [], [], []
    kmap = {}
    for idx, k in enumerate(top):
        k = int(k)
        kmap[k] = idx
        i0 = max(0, k - n + 1)
        i1 = min(n - 1, k)
        for i in range(i0, i1 + 1):
            rows_l.append(idx)
            cols_l.append(i)
            vals_l.append(f[k - i])
    A = sparse.csc_matrix((vals_l, (rows_l, cols_l)), shape=(K, n))
    res = optimize.linprog(-np.ones(n), A_ub=A, b_ub=np.full(K, M),
                           bounds=[(0, None)]*n, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def optimize_seq(a0, time_limit, n, seed=0):
    """Main optimization: mirror descent + periodic LP + perturbation."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    c = fast_autoconv(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    
    T0 = 0.05 * c.max()
    T = T0
    T_min = 0.002 * c.max()
    gamma = 0.995
    eta = 1.0
    t0 = time.monotonic()
    last_lp = 0
    last_perturb = 0
    patience = 0
    iter_i = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_i += 1
        # Mirror descent step
        a_new, M = mirror_step(a, T, eta, n)
        c_new = fast_autoconv(a_new)
        obj_new = 2.0 * n * c_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            eta = min(eta * 1.05, 5.0)
        else:
            patience += 1
            if patience > 20:
                eta *= 0.8
                if eta < 0.05:
                    # Try perturbation
                    noise = rng.normal(0, 0.02, n)
                    a_pert = np.maximum(a + noise, 0.0)
                    a_pert /= a_pert.sum()
                    c_pert = fast_autoconv(a_pert)
                    if 2.0*n*c_pert.max() < best_obj:
                        a = a_pert
                        obj_new = 2.0*n*c_pert.max()
                        patience = 0
                        eta = 0.5
                    else:
                        a = a_pert  # accept anyway for exploration
                        patience = 0
                        eta = 0.5
                T = max(T_min, T * 0.95)
        T = max(T_min, T * gamma)
        
        # Periodic LP
        if iter_i - last_lp >= 30 and time.monotonic() - t0 < time_limit - 5:
            M_cur = fast_autoconv(a).max()
            g = solve_rp(a, M_cur, K=24, n=n, timeout=2)
            if g is not None and g.sum() > 1e-12:
                g /= g.sum()
                alpha = 0.3
                a_mix = (1 - alpha) * a + alpha * g
                a_mix = np.maximum(a_mix, 0.0)
                a_mix /= a_mix.sum()
                c_mix = fast_autoconv(a_mix)
                if 2.0 * n * c_mix.max() < best_obj:
                    best_obj = 2.0 * n * c_mix.max()
                    best_a = a_mix.copy()
                    a = a_mix
            last_lp = iter_i
    
    return best_a, best_obj
