Actually wait, there's a subtle issue. When computing the change in c[k] due to changing a[i]:
- c[k] = sum_j a[j] * a[k-j]
- The terms involving a[i] are: a[i]*a[k-i] (when j=i) and a[k-i]*a[i] (when j=k-i).
- If i ≠ k-i: both terms are present, change = delta*a[k-i] + a[k-i]*delta = 2*delta*a[k-i]. ✓
- If i = k-i (k=2i): only one term a[i]*a[i], change = new^2 - old^2 = (new-old)(new+old). ✓

"""
Top-k gradient optimization with Nesterov momentum and fast coordinate descent
for minimizing peak auto-convolution ratio of non-negative sequences.
Key differences from LP approach: continuous gradient information enables many more
iterations, momentum accelerates convergence, adaptive k smooths the objective,
and incremental convolution updates make coordinate descent very fast.
"""

import numpy as np
import time


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute evaluation score quickly."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (s * s)


def topk_gradient(a, k):
    """Compute gradient of sum of top-k convolution values w.r.t. a."""
    n = len(a)
    c = fft_convolve(a)
    if k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for K in top_idx:
        K = int(K)
        i_lo = max(0, K - n + 1)
        i_hi = min(n - 1, K)
        if i_lo <= i_hi:
            seg = a[K - i_hi:K - i_lo + 1][::-1]
            grad[i_lo:i_hi + 1] += 2.0 * seg
    return grad


def normalize_sum(a, S):
    """Normalize array to have sum S, keeping non-negativity."""
    s = np.sum(a)
    if s < 0.01:
        return np.ones(len(a)) * (S / len(a))
    return a * (S / s)

    cur_score = fast_score(a)
    best_a = a.copy()
    best_score = cur_score

    lr = 0.015
