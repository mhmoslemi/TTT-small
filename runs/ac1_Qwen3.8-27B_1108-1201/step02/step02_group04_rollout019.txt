def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v

    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov's accelerated subgradient
method on the probability simplex with FFT-based auto-convolution, combined with
log-sum-exp smoothed gradient variants and LP-based polishing for fine refinement.
Key innovation: exploits the O(n log n) FFT structure for both auto-convolution
and cross-correlation gradients, enabling rapid convergence in the time budget.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n <= 1:
        return np.ones(max(n, 1)) / max(n, 1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_smooth_grad(a, tau, n):
    """Compute gradient of smoothed max auto-convolution using FFT cross-correlation."""
    c = fast_convolve(a)
    c_scaled = c / tau
    c_scaled -= c_scaled.max()
    w = np.exp(c_scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[c.argmax()] = 1.0
    else:
        w /= w_sum
    # grad_i = (4n/tau) * sum_k w[k] * a[k-i]
    # This is (w cross-correlate a)[i] for i in [0, n-1]
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    xcorr = np.fft.irfft(np.conj(W) * A, N)
    grad = (4.0 * n / tau) * xcorr[:n]
    return grad, c

    if init is None:

            y /= s

    L = 2.0

    while time.monotonic() - t0 < time_limit:

        c = fast_convolve(y)
        k_star = int(np.argmax(c))

    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    # Precompute sparse structure
    rows_list = []
    cols_list = []
    for i in range(n):
        i_start = max(0, 0)
        i_end = min(n - 1, 2 * n - 2 - i)
        for k in range(i, i + n):
            rows_list.append(k)
            cols_list.append(i)
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)

    t0 = time.monotonic()
    alpha = 1.0
    patience = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        vals = np.tile(f, n)
        A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = optimize.linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                                   method='highs', options={'disp': False, 'time_limit': 5})
        except Exception:
            break

        if not res.success:
            break

        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for a_try in [alpha, 0.7 * alpha, 0.5 * alpha, 0.3 * alpha, 0.1 * alpha]:
            f_new = (1.0 - a_try) * f + a_try * g

            conv_new = fast_convolve(f_new)

            if obj_new < best_obj - 1e-14:

                M = conv_new.max()

def sd_polish(f0, time_limit, verbose=True):

            f = np.ones(n) / n
        else:
            f /= s

        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 30 == 0:
                eta *= 0.75
                if eta < 0.01:
                    temp = min(temp * 2.0, 500.0)
                    eta = 0.25
            if patience > 300:
                break

    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations on the simplex."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.08, 0.12, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    w = max(1, n // 4)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    cb = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.3)

    # Concentrated at center
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.3 * n), 0.01)
    inits['tri'] = tri / tri.sum()

    # Two peaks
    two = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2) + \
          np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    inits['two'] = two / two.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry: Nesterov + SD + LP hybrid optimization with multiple restarts."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    n = 512

    if parent is not None and parent.size == n:
        parent_norm = np.maximum(parent, 0.0)
        parent_norm = parent_norm / parent_norm.sum()
    elif parent is not None:
        parent_norm = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        parent_norm = np.maximum(parent_norm, 0.0)
        parent_norm = parent_norm / parent_norm.sum()
    else:
        parent_norm = None

    print(f"Starting optimization with n={n}")

    # Phase 1: Nesterov from multiple inits
    inits = gen_inits(n, rng)
    if parent_norm is not None:
        inits['parent'] = parent_norm

    # Also add perturbed parent
    if parent_norm is not None:
        for _ in range(2):
            pert = parent_norm + 0.1 * rng.standard_normal(n)
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            inits[f'par_pert{_}'] = pert

    results = []
    n_inits = len(inits)
    time_per_init = min(25.0, (deadline - time.monotonic() - 120.0) / max(n_inits, 1))

    print(f"\nPhase 1: Nesterov from {n_inits} inits, ~{time_per_init:.1f}s each")

    for name, a0 in inits.items():
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic() - 100
        ta = min(time_per_init, remaining)
        if ta < 3:
            break

        # Use hybrid (smooth then hard) for parent, non-smooth for others
        if name == 'parent':
            f_opt, obj_opt = nesterov_smooth_nonsmooth(n, time_limit=ta, init=a0, verbose=False)
        else:
            f_opt, obj_opt = nesterov_nonsmooth(n, time_limit=ta, init=a0, verbose=False)

        score = float(evaluate_sequence(f_opt.tolist()))
        if np.isfinite(score):
            results.append((score, name, f_opt))
            if score < best_score:
                best_score = score
                best_seq = f_opt.copy()
                print(f"  NEW BEST: '{name}', score={score:.6f}")
            else:
                print(f"  '{name}': score={score:.6f}")

    print(f"\nPhase 1 done. Best: {best_score:.6f}")

    # Phase 2: SD polish + LP polish on top results
    results.sort(key=lambda x: x[0])
    top = results[:5]

    remaining = deadline - time.monotonic() - 30
    if remaining > 30 and top:
        print(f"\nPhase 2: Polish top {len(top)}, {remaining:.0f}s left")
        t_each = remaining / len(top)

        for idx, (sc, name, f) in enumerate(top):
            if time.monotonic() > deadline - 20:
                break
            ta = min(t_each, deadline - time.monotonic() - 20)
            if ta < 5:
                break

            # SD polish
            t_sd = ta * 0.4
            f_sd, _ = sd_polish(f, time_limit=t_sd, verbose=False)
            sc_sd = float(evaluate_sequence(f_sd.tolist()))
            if np.isfinite(sc_sd) and sc_sd < best_score:
                best_score = sc_sd
                best_seq = f_sd.copy()
                print(f"  SD [{idx}] '{name}': {sc_sd:.6f} (NEW BEST)")

            # LP polish
            t_lp = min(ta * 0.5, deadline - time.monotonic() - 15)
            if t_lp > 3:
                f_lp, _ = lp_polish(f_sd, time_limit=t_lp, verbose=False)
                sc_lp = float(evaluate_sequence(f_lp.tolist()))
                if np.isfinite(sc_lp) and sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = f_lp.copy()
                    print(f"  LP [{idx}] '{name}': {sc_lp:.6f} (NEW BEST)")

    # Phase 3: Deep optimization from overall best
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 15
        print(f"\nPhase 3: Deep optimization from best, {remaining:.0f}s")

        # Nesterov from current best
        t_nest = remaining * 0.4
        f_nest, _ = nesterov_nonsmooth(n, time_limit=t_nest, init=best_seq, verbose=False)
        sc_nest = float(evaluate_sequence(f_nest.tolist()))
        if np.isfinite(sc_nest) and sc_nest < best_score:
            best_score = sc_nest
            best_seq = f_nest.copy()
            print(f"  Deep Nesterov: {sc_nest:.6f} (NEW BEST)")

        # SD from best
        t_sd = min(remaining * 0.3, deadline - time.monotonic() - 10)
        if t_sd > 3:
            f_sd2, _ = sd_polish(best_seq, time_limit=t_sd, verbose=False)
            sc_sd2 = float(evaluate_sequence(f_sd2.tolist()))
            if np.isfinite(sc_sd2) and sc_sd2 < best_score:
                best_score = sc_sd2
                best_seq = f_sd2.copy()
                print(f"  Deep SD: {sc_sd2:.6f} (NEW BEST)")

        # Final LP
        t_lp = min(remaining * 0.3, deadline - time.monotonic() - 5)
        if t_lp > 3:
            f_lp2, _ = lp_polish(best_seq, time_limit=t_lp, verbose=False)
            sc_lp2 = float(evaluate_sequence(f_lp2.tolist()))
            if np.isfinite(sc_lp2) and sc_lp2 < best_score:
                best_score = sc_lp2
                best_seq = f_lp2.copy()
                print(f"  Final LP: {sc_lp2:.6f} (NEW BEST)")

        # One more round of SD + LP if time permits
        if time.monotonic() < deadline - 8:
            t_final = min(20.0, deadline - time.monotonic() - 5)
            f_f, _ = sd_polish(best_seq, time_limit=t_final * 0.5, verbose=False)
            sc_f = float(evaluate_sequence(f_f.tolist()))
            if np.isfinite(sc_f) and sc_f < best_score:
                best_score = sc_f
                best_seq = f_f.copy()
                print(f"  Final SD2: {sc_f:.6f} (NEW BEST)")
            if time.monotonic() < deadline - 4:
                t_f2 = min(15.0, deadline - time.monotonic() - 4)
                f_f2, _ = lp_polish(best_seq, time_limit=t_f2, verbose=False)
                sc_f2 = float(evaluate_sequence(f_f2.tolist()))
                if np.isfinite(sc_f2) and sc_f2 < best_score:
                    best_score = sc_f2
                    best_seq = f_f2.copy()
                    print(f"  Final LP2: {sc_f2:.6f} (NEW BEST)")

    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
