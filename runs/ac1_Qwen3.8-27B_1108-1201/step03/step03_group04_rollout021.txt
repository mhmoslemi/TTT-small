```python
    """Compute gradient of log-sum-exp objective.
    a: sequence of length n
    w: softmax weights of length 2n-1
    Returns gradient of length n.
    """
    n = len(a)

        s = 0.0
        for d in range(n):

            if 0 <= j - d < 2*n - 1:
```

```python
def optimize_sequence(a0, time_limit, tau_init, tau_final, lr_init, momentum=0.9):
    """Optimize sequence using gradient descent with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum
    lr = lr_init
    n_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute convolution and objective
        conv = fast_convolve(a)
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Adaptive tau
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        tau = tau_init * (tau_final / tau_init) ** progress
        
        # Compute softmax weights
        scaled = tau * conv
        scaled -= scaled.max()  # numerical stability
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute gradient
        grad = compute_grad_fast(a, w)
        
        # Update with momentum
        v = momentum * v - lr * grad
        a_new = a + v
        
        # Project onto simplex (sum = 1, a >= 0)
        a_new = project_simplex(a_new)
        
        a = a_new
        lr *= 0.9999  # slow decay
        n_iters += 1
    
    return best_a, best_obj
```

```python
def project_simplex(v):
    """Project v onto the probability simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
import numpy as np
import time

def fast_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, fft_size) * np.fft.rfft(b, fft_size), fft_size)[:size]

def compute_grad(a, w):
    """Compute gradient of sum_k w_k * conv_k w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    c1 = fast_convolve(a_rev, w)[n-1:n-1+n]
    c2 = fast_convolve(a, w)[:n]
    return c1 + c2

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_mask = u * np.arange(1, n+1) > (cssv - 1)
    if not rho_mask.any():
        return np.ones(n) / n
    rho = np.nonzero(rho_mask)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def gradient_optimize(a0, time_limit, tau_init=30, tau_final=300, 
                      lr=0.003, momentum=0.9, verbose=False, verbose_every=1000):
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_c = conv.max()
        obj = 2.0 * n * max_c
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-6))
        tau = tau_init * (tau_final / tau_init) ** progress
        
        # Log-sum-exp weights
        scaled = tau * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            break
        w /= ws
        
        grad = compute_grad(a, w)
        
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        
        it += 1
        if verbose and it % verbose_every == 0:
            print(f"    it={it}, obj={best_obj:.6f}, max_c={max_c:.6e}, tau={tau:.1f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj
```

```python
"""
Smooth gradient-based optimization for minimizing peak auto-convolution of
non-negative sequences. Uses FFT-accelerated gradients of a log-sum-exp
surrogate, adaptive temperature scheduling, simplex projection, multi-scale
initialization, diverse restarts, and momentum-based updates.
"""

import numpy as np
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_grad(a, w):
    """Compute gradient of sum_k w_k * conv_k(a,a) w.r.t. a.
    a: array of length n
    w: array of length 2n-1 (weights on convolution peaks)
    Returns: gradient array of length n
    """
    n = len(a)
    a_rev = a[::-1]
    c1 = fast_convolve(a_rev, w)[n - 1:n - 1 + n]
    c2 = fast_convolve(a, w)[:n]
    return c1 + c2


def project_simplex(v):
    """Project v onto the probability simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_mask = u * np.arange(1, n + 1) > (cssv - 1)
    if not rho_mask.any():
        return np.ones(n) / n
    rho = np.nonzero(rho_mask)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gradient_optimize(a0, time_limit, tau_init=30.0, tau_final=300.0,
                      lr=0.004, momentum=0.9, verbose=False, verbose_every=2000):
    """Optimize sequence using gradient descent with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    stall_count = 0
    stall_threshold = 10000

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_c = conv.max()
        obj = 2.0 * n * max_c

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Stall handling: perturb and reset momentum
        if stall_count > stall_threshold:
            rng_local = np.random.default_rng(hash((n, it)) & 0xFFFFFFFF)
            noise = rng_local.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v = np.zeros(n)
            stall_count = 0
            stall_threshold = int(stall_threshold * 1.5)

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-6))
        tau = tau_init * (tau_final / tau_init) ** progress

        # Log-sum-exp weights (softmax over peaks)
        scaled = tau * conv
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()

        # Gradient
        grad = compute_grad(a, w)

        # Momentum update
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        it += 1
        if verbose and it % verbose_every == 0:
            print(f"    it={it}, obj={best_obj:.6f}, max_c={max_c:.6e}, "
                  f"tau={tau:.1f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def gradient_optimize_topk(a0, time_limit, tau_init=50.0, tau_final=500.0,
                           lr=0.003, momentum=0.9, top_k=10,
                           verbose=False, verbose_every=2000):
    """Optimize using top-k peak weighting instead of full softmax."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_c = conv.max()
        obj = 2.0 * n * max_c

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-6))
        tau = tau_init * (tau_final / tau_init) ** progress

        # Top-k softmax: only weight the top-k peaks
        k = min(top_k, len(conv))
        top_indices = np.argpartition(conv, -k)[-k:]
        top_vals = conv[top_indices]
        scaled = tau * top_vals
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights /= weights.sum()

        # Full weight vector (zeros elsewhere)
        w = np.zeros(len(conv))
        w[top_indices] = weights

        grad = compute_grad(a, w)
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        it += 1
        if verbose and it % verbose_every == 0:
            print(f"    it={it}, obj={best_obj:.6f}, max_c={max_c:.6e}, "
                  f"tau={tau:.1f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Parent (if available)
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    # Edge-concentrated
    for frac in [0.2, 0.3, 0.4]:
        edge = max(1, int(n * frac))
        ec = np.zeros(n)
        ec[:edge] = 1.0
        ec[-edge:] = 1.0
        ec /= ec.sum()
        inits[f'edge_{frac}'] = ec

    # Triangle (center-heavy)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    # Inverted triangle (edge-heavy)
    itri = np.maximum(x, n - 1 - x).astype(float) + 1.0
    itri /= itri.sum()
    inits['inv_triangle'] = itri

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Gaussian edge (two peaks)
    ge = np.exp(-((x / (0.15 * n)) ** 2)) + np.exp(-(((n - 1) - x) / (0.15 * n)) ** 2)
    ge /= ge.sum()
    inits['gauss_edge'] = ge

    # Cosine perturbations of uniform
    for freq in [1, 2, 3, 4]:
        for eps in [0.2, 0.4]:
            c = 1.0 + eps * np.cos(2 * np.pi * freq * x / max(n - 1, 1))
            c = np.maximum(c, 0.0)
            s = c.sum()
            if s > 1e-10:
                inits[f'cos{freq}_e{eps}'] = c / s

    # Ramp (one-sided concentration)
    for direction in [1, -1]:
        r = np.arange(1, n + 1, dtype=np.float64)
        if direction == -1:
            r = r[::-1]
        r /= r.sum()
        inits[f'ramp_{direction}'] = r

    # Power-law
    for p in [0.5, 2.0, 3.0]:
        pw = np.minimum(x, n - 1 - x) ** p + 0.1
        pw /= pw.sum()
        inits[f'power_{p}'] = pw

    # Random
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['random3'] = rng.dirichlet(np.ones(n) * 5.0)

    # Sparse random
    sparse = rng.random(n)
    sparse[sparse < 0.6] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.random(n)
    sparse /= sparse.sum()
    inits['sparse'] = sparse

    return inits


def upsample(a, new_n):
    """Upsample sequence to new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s
```
