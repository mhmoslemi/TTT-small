Now, the approach in the previous code uses LP-based iterative refinement. The idea is:
1. Start with some initial sequence `a`.
2. Compute `M = max(conv(a,a))`.
3. Find the sequence `b` that maximizes `sum(b)` subject to `conv(a,b) <= M` element-wise and `b >= 0`.
4. Rescale `b` to have the same sum as `a`.
5. Update `a ← (1-t)*a + t*b`.

This is a form of projected subgradient / trust region method. The LP in step 3 finds the best "direction" to move in that keeps the convolution with the current `a` bounded.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase search:
LP-based iterative refinement, projected subgradient, and multi-scale exploration.
Key improvements over previous: adaptive LP step size, sparse LP for larger n,
power-law initializations, and multi-scale upscaling.

import numpy as np
import time
from scipy.optimize import linprog, minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix where (A @ b)[k] = conv(a,b)[k]."""
    from scipy.sparse import csr_matrix
    n = len(a)
    num_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(num_rows, n))
    return A

def lp_direction(a, M, method='max_sum'):
    """Find LP direction: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    from scipy.sparse import csr_matrix
    # Build sparse convolution matrix
    num_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(num_rows, n))
    
    if method == 'max_sum':
        c = -np.ones(n)
        b_ub = np.full(num_rows, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    else:
        # min M' s.t. sum(b)=1, b>=0, conv(a,b) <= M'
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        A_ub = np.zeros((num_rows, n + 1))
        A_ub[:, :n] = A.toarray()
        A_ub[:, n] = -1.0
        b_ub = np.zeros(num_rows)
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return np.maximum(res.x[:n], 0)
        return None
    
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def line_search(a, b, n, max_evals=20):
    """Find best t in [0,1] minimizing obj((1-t)*a + t*b)."""
    best_t = 0.0
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    # Check t=1
    obj_1 = 2.0 * n * fast_convolve(b).max()
    if obj_1 < best_obj:
        best_obj = obj_1
        best_t = 1.0
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = 2.0 * n * fast_convolve((1-c)*a + c*b).max()
    fd = 2.0 * n * fast_convolve((1-d)*a + d*b).max()
    
    for _ in range(max_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = 2.0 * n * fast_convolve((1-c)*a + c*b).max()
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = 2.0 * n * fast_convolve((1-d)*a + d*b).max()
    
        if fd < best_obj:
            best_obj = fd
            best_t = d
        if fc < best_obj:
            best_obj = fc
            best_t = c
    
    return best_t, best_obj

Multi-phase optimization for minimum peak auto-correlation of non-negative sequences.
Uses LP-based iterative refinement with line search, projected gradient polish,

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_objective(a):

    s = a.sum()

        return float('inf')

    return 2.0 * n * conv.max() / (s * s)

    A = np.zeros((2 * n - 1, n))

        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

    if res.status == 0:

    return None
