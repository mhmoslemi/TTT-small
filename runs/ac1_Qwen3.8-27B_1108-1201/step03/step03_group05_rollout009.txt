```python
import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, conv, beta):
    """
    Gradient of (1/beta)*logsumexp(beta*conv) w.r.t. a.
    Uses FFT cross-correlation.
    """
    n = len(a)
    scaled = beta * conv - beta * conv.max()  # numerical stability
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        # Degenerate: all mass on argmax
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = p / p_sum
    
    # Gradient: g[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    # = 2 * cross_correlation(p, a)[j]
    # cross_correlation(p, a)[j] = sum_m p[m+j] * a[m]
    
    fft_size = 1 << (len(conv) + n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(P * np.conj(A), fft_size)
    
    return 2.0 * cross[:n]


def topk_gradient(a, conv, K, sharpness=10.0):
    """
    Gradient using only the top K peaks with exponential weighting.
    More focused than full softmax.
    """
    n = len(a)
    m = len(conv)
    
    if K >= m:
        top_idx = np.argsort(conv)[::-1]
    else:
        top_idx = np.argpartition(conv, -K)[-K:]
    
    # Exponential weighting among top K
    top_vals = conv[top_idx]
    max_top = top_vals.max()
    weights = np.exp(sharpness * (top_vals - max_top))
    weights /= weights.sum()
    
    # Build sparse weight vector
    p = np.zeros(m)
    p[top_idx] = weights
    
    # Same cross-correlation as smoothed_gradient
    fft_size = 1 << (m + n - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(P * np.conj(A), fft_size)
    
    return 2.0 * cross[:n]
```
