def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    rho_final = np.max(rho[cond]) if np.any(cond) else 1
    theta = (cssv[rho_final - 1] - 1.0) / rho_final
    return np.maximum(v - theta, 0.0)

        conv = fast_convolve(a)
        sm = smooth_max_val(conv, beta)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-strategy optimization:
- FFT-based fast convolution
- Projected gradient descent with log-sum-exp smoothing
- L-BFGS-B with softmax reparametrization
- Subgradient method for non-smooth max
- Multi-start with diverse initializations and adaptive length selection
- Coordinate descent polishing
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_val(x, beta):
    """Log-sum-exp approximation of max(x)."""
    x_max = np.max(x)
    shifted = np.clip(beta * (x - x_max), -100, 100)
    s = np.exp(shifted)
    return x_max + np.log(s.sum()) / beta


def smooth_max_weights(x, beta):
    """Softmax weights corresponding to log-sum-exp of beta*x."""
    x_max = np.max(x)
    shifted = np.clip(beta * (x - x_max), -100, 100)
    s = np.exp(shifted)
    return s / s.sum()


def compute_conv_grad(a, w):
    """
    Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Uses the identity: d/da[j] conv(a,a)[k] = 2*a[k-j] (valid indices).
    Result: grad[j] = 2 * sum_k w[k]*a[k-j] = 2*np.correlate(w, a, 'valid')[j]
    """
    return 2.0 * np.correlate(w, a, mode='valid')


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho_final = rho[np.last_nonzero(cond)]
    theta = (cssv[rho_final - 1] - 1.0) / rho_final
    return np.maximum(v - theta, 0.0)


def objective_exact(a):
    """Compute 2*n*max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def pgd_optimize(a0, beta_start=50.0, beta_end=5.0, lr=0.1, n_iter=2000):
    """
    Projected gradient descent with annealed temperature.
    Works on simplex {a >= 0, sum(a)=1}.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_obj = objective_exact(a)
    best_a = a.copy()

    step_size = lr
    beta = beta_start
    beta_decay = (beta_end / beta_start) ** (1.0 / max(1, n_iter))

    for it in range(n_iter):
        conv = fast_convolve(a)
        sm = smooth_max_val(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        w = smooth_max_weights(conv, beta)
        grad = 2.0 * beta * compute_conv_grad(a, w)
        grad = grad - np.mean(grad)

        # Backtracking line search
        best_a_new = a
        best_obj_new = obj
        for ls in range(15):
            a_try = project_simplex(a - step_size * grad)
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * smooth_max_val(conv_try, beta)
            if obj_try <= obj + 1e-14:
                best_a_new = a_try
                best_obj_new = obj_try
                break
            step_size *= 0.5
        else:
            best_a_new = a
            best_obj_new = obj

        a = best_a_new
        step_size *= 1.01  # slightly increase after successful step
        step_size = min(step_size, 0.5)
        beta *= beta_decay

    return best_a, best_obj


def subgradient_optimize(a0, n_iter=3000, lr_init=0.05):
    """
    Subgradient method for the non-smooth max.
    Works on simplex {a >= 0, sum(a)=1}.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_obj = objective_exact(a)
    best_a = a.copy()

    lr = lr_init
    for it in range(n_iter):
        conv = fast_convolve(a)
        max_k = np.argmax(conv)
        obj = 2.0 * n * conv.max()

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Subgradient: pick the argmax coordinate
        w = np.zeros_like(conv)
        w[max_k] = 1.0

        # Average with a few argmax points for stability
        top_k = max(1, min(5, int(0.05 * len(conv))))
        idx = np.argsort(conv)[-top_k:]
        w_avg = np.zeros_like(conv)
        for i in idx:
            w_avg[i] = 1.0
        w_avg /= top_k

        grad = 2.0 * n * (compute_conv_grad(a, w_avg) - np.mean(compute_conv_grad(a, w_avg)))

        a = project_simplex(a - lr * grad)
        lr *= 0.9995

    return best_a, best_obj

def structured_init(n, rng):
    """Generate a structured (low-autocorrelation-inspired) initialization."""
    # Try to create a sequence with approximately flat spectrum
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = np.sqrt(n)
    for k in range(1, fft_size // 2 + 1):
        if k < fft_size - k:
            phase = rng.uniform(0, 2 * np.pi)
            A[k] = np.exp(1j * phase)
            A[fft_size - k] = np.exp(-1j * phase)
        else:
            A[k] = np.exp(1j * rng.uniform(0, 2 * np.pi))
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n

def downsample(parent, n):
    """Downsample a sequence to length n."""
    if len(parent) == n:
        return parent.copy()
    p = np.maximum(parent, 0)
    ps = p.sum()
    if ps < 1e-10:
        return np.ones(n) / n
    p = p / ps
    if len(parent) > n:
        result = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
    else:
        result = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-10:
        result /= s
    else:
        result = np.ones(n) / n
    return result
