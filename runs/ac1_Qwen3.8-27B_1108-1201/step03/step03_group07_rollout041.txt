def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]

def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max(x, beta):
    """Numerically stable log-sum-exp."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_s = np.exp(shifted)
    total = exp_s.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage optimization:
annealed smooth-max L-BFGS-B, peak-targeted local search, and random restarts.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_s / total
    sm = x_max + np.log(total) / beta
    return sm, w


def conv_grad(a, w):
    """Gradient of sum_k w[k]*conv[a,a][k] w.r.t. a via FFT."""
    n = len(a)
    N = 1 << (len(w) + n - 2 + 7).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    return np.real(ifft(np.conj(W) * A, N))[:n] * 2.0


def obj_fn(x, beta, n):
    """Objective for L-BFGS-B (softmax parameterization)."""
    xs = x - x.max()
    a = np.exp(xs)
    a /= a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def jac_fn(x, beta, n):
    """Jacobian for L-BFGS-B (softmax parameterization)."""
    xs = x - x.max()
    a = np.exp(xs)
    a /= a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    ga = conv_grad(a, w)
    lag = float(np.dot(a, ga))
    return a * (ga - lag)


def lbfgsb_opt(a0, beta, maxiter=300):
    """L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    a0c = np.maximum(a0, 1e-30)
    a0c /= a0c.sum()
    x0 = np.log(a0c)
    res = minimize(obj_fn, x0, jac=lambda x: jac_fn(x, beta, n),
                   method='L-BFGS-B',
                   options={'maxiter': maxiter, 'maxls': 40, 'ftol': 1e-18, 'gtol': 1e-12})
    xo = res.x
    a = np.exp(xo - xo.max())
    a /= a.sum()
    return a, res.fun

    s = a.sum()

    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using annealed smooth-max L-BFGS-B,
peak-targeted local search, diverse multi-length exploration, and random restarts.
Key innovations over prior work: geometric beta annealing, peak-aware perturbation,
and ultra-tight convergence tolerances for the final polish stage.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_weights(x, beta):
    """Smooth max (log-sum-exp) and softmax weights, numerically stable."""
    xm = x.max()
    sh = beta * (x - xm)
    sh = np.clip(sh, -100.0, 100.0)
    e = np.exp(sh)
    tot = e.sum()
    if tot < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return xm, w
    w = e / tot
    sm = xm + np.log(tot) / beta
    return sm, w


def conv_grad(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a via FFT cross-correlation."""
    n = len(a)
    N = 1 << (len(w) + n - 2 + 7).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    return np.real(ifft(np.conj(W) * A, N))[:n] * 2.0


def _obj(x, beta, n):
    """Smooth objective for scipy L-BFGS-B (log-softmax parametrization)."""
    xs = x - x.max()
    a = np.exp(xs)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_weights(conv, beta)
    return 2.0 * n * sm


def _jac(x, beta, n):
    """Jacobian for scipy L-BFGS-B (log-softmax parametrization)."""
    xs = x - x.max()
    a = np.exp(xs)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_weights(conv, beta)
    ga = conv_grad(a, w)
    lag = float(np.dot(a, ga))
    return a * (ga - lag)


def lbfgsb_opt(a0, beta, maxiter=300):
    """L-BFGS-B with log-softmax reparametrization."""
    n = len(a0)
    a0c = np.maximum(a0, 1e-30)
    a0c = a0c / a0c.sum()
    x0 = np.log(a0c)
    res = minimize(_obj, x0, jac=_jac, args=(beta, n),
                   method='L-BFGS-B',
                   options={'maxiter': maxiter, 'maxls': 50,
                            'ftol': 1e-18, 'gtol': 1e-13})
    xo = res.x
    a = np.exp(xo - xo.max())
    a = a / a.sum()
    return a, float(res.fun)


def anneal_lbfgsb(a0, betas, maxiter_each=200):
    """Multi-beta annealed L-BFGS-B. Returns final a and list of (beta, obj)."""
    a = a0.copy()
    history = []
    for beta in betas:
        a, obj = lbfgsb_opt(a, beta, maxiter=maxiter_each)
        history.append((beta, obj))
    return a, history


def peak_local_search(a, n_iters=300):
    """
    Peak-targeted local search: identify coordinates contributing most to the
    convolution peak and perform targeted mass redistribution with acceptance.
    """
    n = len(a)
    a = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * float(conv.max())
    best_a = a.copy()

    for it in range(n_iters):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        max_val = float(conv[max_k])
        lo = max(0, max_k - n + 1)
        hi = min(n - 1, max_k)

        # Contribution of each coordinate to the peak
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = max_k - i
            contrib[i] = a[i] * a[j]

        # Total contribution to top-5 peaks
        top_peaks = np.argsort(conv)[::-1][:5]
        total_contrib = np.zeros(n)
        for pk in top_peaks:
            plo = max(0, pk - n + 1)
            phi = min(n - 1, pk)
            for i in range(plo, phi + 1):
                total_contrib[i] += a[i] * a[pk - i]

        # Source: highest contribution to peak
        src = lo + int(np.argmax(contrib[lo:hi + 1]))
        if contrib[src] < 1e-20:
            break

        # Destination: lowest total contribution among candidates
        candidates = np.arange(n)
        mask = candidates != src
        masked_tc = total_contrib.copy()
        masked_tc[~mask] = np.inf
        dst = int(np.argmin(masked_tc))

        # Transfer amount: proportional to source value
        frac = 0.02 + 0.03 * np.random.random()
        transfer = a[src] * frac
        a[src] -= transfer
        a[dst] += transfer
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            continue
        a = a / s

        obj = 2.0 * n * float(fast_convolve(a).max())
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        else:
            # Revert
            a[src] += transfer
            a[dst] -= transfer
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []

    # From parent
    if parent is not None:

        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits.append(pp / ps)

    # Flat spectrum (random phase, multiple trials)
    for _ in range(3):
        ff = 1 << (n - 1).bit_length()
        A = np.zeros(ff, dtype=complex)
        A[0] = np.sqrt(n)
        ph = rng.uniform(0, 2 * np.pi, ff // 2)
        for k in range(1, ff // 2):
            A[k] = np.exp(1j * ph[k - 1])
            A[ff - k] = np.exp(-1j * ph[k - 1])
        if ff % 2 == 0:
            A[ff // 2] = np.exp(1j * ph[-1])
        av = np.real(ifft(A, ff))[:n]
        av = np.maximum(av, 0)
        sv = av.sum()
        if sv > 1e-10:
            inits.append(av / sv)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussians
    x_arr = np.arange(n, dtype=np.float64)
    for sf in [0.18, 0.28, 0.42]:
        g = np.exp(-0.5 * ((x_arr - 0.5 * (n - 1)) / (sf * n)) ** 2)
        inits.append(g / g.sum())

    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits.append(bi / bi.sum())

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    inits.append(ramp[::-1] / ramp.sum())

    # Random
    for _ in range(2):
        inits.append(rng.dirichlet(np.ones(n) * 0.4))

    # Sparse random
    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    return inits


def perturb_peak_targeted(a, conv, rng):
    """Perturb by reducing peak contributors and increasing non-contributors."""
    n = len(a)
    a_pert = a.copy()
    pk = int(np.argmax(conv))
    lo = max(0, pk - n + 1)
    hi = min(n - 1, pk)
    # Scale down peak contributors
    scale = 1.0 - 0.08 * rng.random()
    a_pert[lo:hi + 1] *= scale
    a_pert = np.maximum(a_pert, 0)
    a_pert = project_simplex(a_pert)
    return a_pert


def perturb_gaussian(a, rng, scale=None):
    """Gaussian perturbation with simplex projection."""
    n = len(a)
    if scale is None:
        scale = 0.003 + 0.007 * rng.random()
    noise = rng.standard_normal(n) * scale
    return project_simplex(a + noise)


def perturb_block_swap(a, rng):
    """Swap two random blocks of moderate size."""
    n = len(a)
    a_pert = a.copy()
    bs = max(4, n // 20)
    i1 = int(rng.integers(0, n - bs))

    a_pert[i1:i1 + bs] = a_pert[i2:i2 + bs]

def perturb_mass_transfer(a, rng, n_transfers=5):

                a_pert = perturb_gaussian(a_best, rng, scale=0.008)
            elif strat == 2:
                a_pert = perturb_peak_targeted(a_best, conv_best, rng)
            elif strat == 3:
                a_pert = perturb_block_swap(a_best, rng)
            else:
                a_pert = perturb_mass_transfer(a_best, rng)

            # Medium optimization
            a_pert, _ = lbfgsb_opt(a_pert, beta=30.0, maxiter=150)
            a_pert, _ = lbfgsb_opt(a_pert, beta=10.0, maxiter=200)
            a_pert, _ = lbfgsb_opt(a_pert, beta=5.0, maxiter=200)

            sc = score_and_update(a_pert)
            if sc <= best_score:
                a_best = a_pert
                conv_best = fast_convolve(a_best)

    print("Phase 3 done. Best: %.8f" % best_score)

    # ========== PHASE 4: Ultra-fine final polish ==========
    print("=== Phase 4: Final polish ===")

    if best_sequence is not None and time.monotonic() < deadline - 20:
        a_best = best_sequence / best_sequence.sum()

        # Very low beta for tight convergence
        betas_ultra = [3.0, 2.0, 1.5, 1.2, 1.0, 0.8]
        for beta in betas_ultra:
            if time.monotonic() > deadline - 10:
                break
            a_opt, _ = lbfgsb_opt(a_best, beta=beta, maxiter=300)
            sc = score_and_update(a_opt)
            if sc <= best_score:
                a_best = a_opt

        # Final peak local search
        if time.monotonic() < deadline - 15:
            a_lps, _ = peak_local_search(a_best, n_iters=300)
            score_and_update(a_lps)

    # ========== PHASE 5: Try alternative lengths if time permits ==========
    if best_sequence is not None and time.monotonic() < deadline - 150:
        n_cur = len(best_sequence)
        alt_lengths = sorted(set([n_cur * 2, n_cur // 2, 384, 512, 128]))
        alt_lengths = [l for l in alt_lengths if 64 <= l <= 2048 and l != n_cur]

        for n_alt in alt_lengths[:2]:
            if time.monotonic() > deadline - 30:
                break
            pp = np.interp(np.linspace(0, 1, n_alt),
                           np.linspace(0, 1, n_cur), best_sequence)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                a_try = pp / ps
                for beta in [40.0, 20.0, 10.0, 5.0]:
                    if time.monotonic() > deadline - 10:
                        break
                    a_try, _ = lbfgsb_opt(a_try, beta=beta, maxiter=200)
                score_and_update(a_try)

    # ========== Final verification ==========
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    print("\nFINAL: %.8f (n=%d, sum=%.4f)" % (best_score, len(best_sequence), best_sequence.sum()))

    return [float(x) for x in best_sequence]
