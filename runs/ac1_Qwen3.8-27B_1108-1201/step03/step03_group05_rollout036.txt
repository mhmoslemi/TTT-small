```python
def mirror_descent_step(a, grad, lr):
    """Multiplicative weight update (mirror descent with KL)."""
    log_a = np.log(a + 1e-300)
    log_a_new = log_a - lr * grad
    log_a_new -= log_a_new.max()  # numerical stability
    a_new = np.exp(log_a_new)
    a_new = a_new / a_new.sum()
    return a_new
```

```python
    """
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-15:
        return np.zeros(n)
    
    threshold = max_c * (1.0 - rel_tol)
    peak_indices = np.where(conv >= threshold)[0]
    
    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            indices = k - np.arange(m_start, m_end + 1)
            grad[m_start:m_end + 1] += 2.0 * a[indices]
    
    # Normalize so the subgradient has a consistent scale
    # The directional derivative in direction d is sum(grad * d)
    return grad


def mirror_step(a, grad, lr):
    """Multiplicative weight update (mirror descent with KL Bregman)."""
    log_ratio = -lr * grad
    log_ratio -= log_ratio.max()  # numerical stability
    weights = np.exp(log_ratio)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def lp_direction_active(a, M, active_indices, n, x0=None):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_indices, b >= 0.
    Uses active constraints only for speed.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)
    n_cons = len(active_indices)
    
    A_ub = np.zeros((n_cons, n))
    for row, k in enumerate(active_indices):
        k = int(k)
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            indices = k - np.arange(m_start, m_end + 1)
            A_ub[row, m_start:m_end + 1] = a[indices]
    
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    
    options = {'presolve': True, 'time_limit': 10}
    if x0 is not None:
        options['presolve'] = False
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options=options)
    except Exception:
        return None
    
    if not result.success or result.x is None:
        return None
    
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum


def lp_direction_full(a, M, n, x0=None):
    """Solve LP with all constraints (for final polishing)."""
    c_obj = -np.ones(n)
    
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    options = {'presolve': True, 'time_limit': 30}
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options=options)
    except Exception:
        return None
    
    if not result.success or result.x is None:
        return None
    
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum


def find_active_peaks(conv, rel_tol=0.05):
    """Find indices of peaks within rel_tol of the maximum."""
    max_c = conv.max()
    if max_c < 1e-15:
        return np.array([np.argmax(conv)], dtype=int)
    threshold = max_c * (1.0 - rel_tol)
    return np.where(conv >= threshold)[0]


def optimize_subgradient_mirrordesc(a0, time_limit, lr_init=0.05, lr_end=0.001,
                                     decay_rate=0.9995, seed=0, rel_tol=0.02):
    """
    Mirror descent with exact subgradients for the max-conv objective.
    Input: a0 on simplex (sum=1, non-negative).
    Returns (best_a, best_max_conv).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    iters = 0
    momentum_buf = np.zeros(n)
    momentum_beta = 0.7
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_c = conv.max()
        
        if max_c < best_obj - 1e-12:
            best_obj = max_c
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient
        active = find_active_peaks(conv, rel_tol)
        grad = exact_subgrad_maxconv(a, conv, 0.0)  # exact (all peaks at max)
        
        # Use active peaks subgradient for direction
        if len(active) > 1:
            grad_active = exact_subgrad_maxconv(a, conv, rel_tol)
            # Weight by peak values
            peak_vals = conv[active]
            weights = (peak_vals - peak_vals.min() + 1e-10)
            weights /= weights.sum()
            grad_weighted = np.zeros(n)
            for w, k in zip(weights, active):
                k = int(k)
                m_start = max(0, k - n + 1)
                m_end = min(n - 1, k)
                if m_start <= m_end:
                    indices = k - np.arange(m_start, m_end + 1)
                    grad_weighted[m_start:m_end + 1] += w * 2.0 * a[indices]
            grad = 0.5 * grad + 0.5 * grad_weighted
        
        # Momentum (exponential moving average of gradients)
        momentum_buf = momentum_beta * momentum_buf + (1 - momentum_beta) * grad
        grad_use = momentum_buf
        
        # Mirror descent step
        a_new = mirror_step(a, grad_use, lr)
        a = a_new
        
        iters += 1
        lr = max(lr_end, lr * decay_rate)
        
        # Perturbation to escape local minima
        if no_improve > 200:
            noise_scale = 0.02 * (1.0 + 0.5 * np.sin(iters * 0.05))
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a * np.exp(noise))
            momentum_buf *= 0.3
            no_improve = 0
            lr = max(lr_init * 0.3, lr)
        
        # Periodic larger perturbation
        if no_improve > 500:
            noise_scale = 0.05
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a * np.exp(noise))
            momentum_buf *= 0.1
            no_improve = 0
            lr = lr_init * 0.5
        
        # Check time
        if iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MG: iter={iters}, obj={best_obj:.8f}, lr={lr:.6f}, "
                  f"peaks={len(active)}, elapsed={elapsed:.1f}s", flush=True)
    
    return best_a, best_obj


def optimize_lp_based(a0, time_limit, seed=0, tol_progression=None):
    """
    LP-based direction finding with active constraints.
    Iteratively solves LPs to find directions that reduce the max peak.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    current_a = a.copy()
    
    t0 = time.monotonic()
    stuck_count = 0
    iters = 0
    last_lp_solution = None
    
    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(current_a)
        M = conv_cur.max()
        
        if M < best_obj - 1e-12:
            best_obj = M
            best_a = current_a.copy()
        
        # Find active peaks (progressively tighter)
        frac = (time.monotonic() - t0) / max(time_limit, 1e-6)
        if tol_progression is None:
            if frac < 0.3:
                rel_tol = 0.10
            elif frac < 0.6:
                rel_tol = 0.05
            else:
                rel_tol = 0.02
        else:
            rel_tol = tol_progression
        
        active = find_active_peaks(conv_cur, rel_tol)
        # Limit to top 20 peaks for LP efficiency
        if len(active) > 20:
            top_idx = np.argsort(conv_cur[active])[-20:]
            active = active[top_idx]
        
        # Solve LP with active constraints
        M_try = M * 1.001  # slight relaxation for feasibility
        b = lp_direction_active(current_a, M_try, active, n, x0=last_lp_solution)
        
        if b is None:
            # Try with relaxed M
            M_try = M * 1.01
            b = lp_direction_active(current_a, M_try, active, n, x0=last_lp_solution)
        
        if b is None:
            # Fallback: use subgradient direction
            grad = exact_subgrad_maxconv(current_a, conv_cur, 0.0)
            # Project to feasible direction
            b = project_simplex(current_a - 0.01 * grad / max(grad.max(), 1e-10))
        
        last_lp_solution = b
        
        # Line search: try multiple step sizes
        improved = False
        steps = [0.3, 0.5, 0.7, 1.0, 0.1, 0.2, 0.4, 0.6, 0.8]
        # Randomize order slightly
        rng.shuffle(steps)
        
        best_step_obj = M
        best_step_a = current_a.copy()
        
        for t in steps:
            a_try = (1 - t) * current_a + t * b
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()
            
            if obj_try < best_step_obj:
                best_step_obj = obj_try
                best_step_a = a_try.copy()
        
        if best_step_obj < best_obj - 1e-10:
            best_obj = best_step_obj
            best_a = best_step_a.copy()
            current_a = best_step_a.copy()
            stuck_count = 0
            improved = True
        elif best_step_obj < M * (1 - 0.0001):
            current_a = best_step_a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        iters += 1
        
        if iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP: iter={iters}, best={best_obj:.8f}, cur_M={M:.8f}, "
                  f"stuck={stuck_count}, tol={rel_tol:.3f}, elapsed={elapsed:.1f}s", flush=True)
        
        # If stuck, perturb and continue
        if stuck_count > 8:
            noise = rng.standard_normal(n) * 0.015
            current_a = project_simplex(current_a + noise)
            stuck_count = 0
            last_lp_solution = None
        
        # If very stuck, bigger restart
        if stuck_count > 15:
            noise = rng.standard_normal(n) * 0.04
            current_a = project_simplex(current_a + noise)
            stuck_count = 0
            last_lp_solution = None
            if elapsed > time_limit * 0.9:
                break
    
    return best_a, best_obj


def optimize_combined(a0, time_limit, seed=0):
    """
    Combined optimizer: alternates between mirror descent and LP steps.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = 0.05
    iters = 0
    no_improve = 0
    lp_buf = a.copy()
    last_lp = None
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        
        # Decide: LP or mirror descent
        use_lp = (iters % 5 == 0) and (frac < 0.85)
        
        if use_lp:
            conv_cur = fast_convolve(a)
            M = conv_cur.max()
            
            # Active peaks
            if frac < 0.3:
                rel_tol = 0.08
            elif frac < 0.6:
                rel_tol = 0.04
            else:
                rel_tol = 0.015
            
            active = find_active_peaks(conv_cur, rel_tol)
            if len(active) > 15:
                top_idx = np.argsort(conv_cur[active])[-15:]
                active = active[top_idx]
            
            b = lp_direction_active(a, M * 1.001, active, n, x0=last_lp)
            
            if b is not None:
                last_lp = b
                # Take a step toward b
                for t in [0.4, 0.6, 0.8, 0.3, 0.5, 0.7]:
                    a_try = (1 - t) * a + t * b
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < M:
                        a = a_try
                        break
            
            iters += 1
        else:
            # Mirror descent step
            conv_cur = fast_convolve(a)
            grad = exact_subgrad_maxconv(a, conv_cur, 0.0)
            
            # Adaptive learning rate
            lr_eff = lr * (1.0 - 0.5 * frac)
            a_new = mirror_step(a, grad, lr_eff)
            a = a_new
            lr = max(0.001, lr * 0.999)
            iters += 1
        
        # Check improvement
        conv = fast_convolve(a)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Perturbation
        if no_improve > 100:
            noise_scale = 0.02 * (1.0 + frac)
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a * np.exp(noise))
            no_improve = 0
            last_lp = None
        
        if no_improve > 300:
            noise_scale = 0.05
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a * np.exp(noise))
            no_improve = 0
            last_lp = None
            lr = 0.03
        
        if iters % 20 == 0:
            print(f"    Comb: iter={iters}, best={best_obj:.8f}, no_impr={no_improve}, "
                  f"elapsed={elapsed:.1f}s", flush=True)
    
    return best_a, best_obj


def peak_balancing(a0, time_limit, seed=0):
    """
    Direct peak balancing: identify top peaks and try to equalize them
    by moving mass between contributing positions.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_c = conv.max()
        
        if max_c < best_obj - 1e-12:
            best_obj = max_c
            best_a = a.copy()
        
        # Find top 3 peaks
        top_k = min(3, len(conv))
        peak_idx = np.argsort(conv)[-top_k:]
        peak_idx = peak_idx[conv[peak_idx] > max_c * 0.9]
        
        if len(peak_idx) == 0:
            break
        
        improved = False
        
        # For the highest peak, find the contributing pair with max product
        k_max = int(peak_idx[-1])
        m_start = max(0, k_max - n + 1)
        m_end = min(n - 1, k_max)
        
        contributions = np.array([a[i] * a[k_max - i] for i in range(m_start, m_end + 1)])
        if len(contributions) == 0:
            iters += 1
            continue
        
        i_max_local = np.argmax(contributions)
        i_max = m_start + i_max_local
        j_max = k_max - i_max
        
        # Try moving mass from i_max to a position that doesn't increase other peaks
        for _ in range(5):
            j_new = rng.integers(0, n)
            if j_new == i_max or abs(j_new - i_max) < 3:
                continue
            
            for frac in [0.03, 0.05, 0.08, 0.12]:
                amt = a[i_max] * frac
                if amt < 1e-9:
                    break
                a_try = a.copy()
                a_try[i_max] -= amt
                a_try[j_new] += amt
                conv_try = fast_convolve(a_try)
                if conv_try.max() < best_obj - 1e-10:
                    best_obj = conv_try.max()
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
                elif conv_try.max() < max_c * 0.999:
                    a = a_try.copy()
                    improved = True
                    break
        
        # Also try moving from j_max
        if not improved:
            for _ in range(5):
                j_new = rng.integers(0, n)
                if j_new == j_max or abs(j_new - j_max) < 3:
                    continue
                for frac in [0.03, 0.05, 0.08]:
                    amt = a[j_max] * frac
                    if amt < 1e-9:
                        break
                    a_try = a.copy()
                    a_try[j_max] -= amt
                    a_try[j_new] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-10:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
        
        iters += 1
        if iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS: iter={iters}, best={best_obj:.8f}, elapsed={elapsed:.1f}s", flush=True)
        
        # Random perturbation if stuck
        if iters % 100 == 0 and not improved:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                amt = a[i] * 0.05
                a[i] -= amt
                a[j] += amt
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Raised cosine (Hann-like)
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    cos_w = np.maximum(cos_w, 1e-8)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w
    
    # Flat-top window
    # a[k] = 1 - a1*cos(2*pi*k/(n-1)) + a2*cos(4*pi*k/(n-1)) - a3*cos(6*pi*k/(n-1))
    a1, a2, a3 = 0.21557895, 0.41663158, 0.277263158
    flat = (1 - a1 * np.cos(2 * np.pi * x / (n - 1)) +
            a2 * np.cos(4 * np.pi * x / (n - 1)) -
            a3 * np.cos(6 * np.pi * x / (n - 1)))
    flat = np.maximum(flat, 1e-8)
    flat /= flat.sum()
    inits['flattop'] = flat
    
    # Kaiser-Bessel-like (approximate with Gaussian modulated by cosine)
    sigma = 0.25 * n
    kaiser = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2) * 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    kaiser = np.maximum(kaiser, 1e-8)
    kaiser /= kaiser.sum()
    inits['kaiser'] = kaiser
    
    # Optimal-ish: try a^p shape
    for p in [0.5, 1.0, 2.0, 3.0]:
        base = np.minimum(x, n - 1 - x) + 0.1
        w = base ** p
        w /= w.sum()
        inits[f'pow_{p}'] = w
    
    # Bimodal
    edge = max(3, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Gaussian
    g = np.exp(-0.5 * ((x - n / 2) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 3.0)
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining mirror descent, LP direction finding,
    and peak balancing to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}, score={parent_score:.6f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent: {e}", flush=True)
    
    # Phase 1: Quick scan over lengths and initializations using mirror descent
    print("=== Phase 1: Quick Scan (Mirror Descent) ===", flush=True)
    lengths = [320, 384, 448, 512]
    
    scan_results = []
    phase1_budget = 300.0
    t_p1_start = time.monotonic()
    
    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        
        inits = generate_initializations(n, parent, rng)
        names = list(inits.keys())
        
        # Priority ordering
        priority = ['parent', 'parent_ds', 'parent_up', 'flattop', 'raised_cos',
                    'kaiser', 'pow_1.0', 'pow_2.0', 'uniform', 'rand1']
        names = [nm for nm in priority if nm in names] + \
                [nm for nm in names if nm not in priority]
```

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a combination of:
- Mirror descent (multiplicative weights) with exact subgradients at active peaks
- LP-based direction finding with active constraint sets

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_subgrad_maxconv(a, conv, rel_tol=0.0):
    """
    Compute exact subgradient of max_k c[k] at a.
    Includes all peaks within rel_tol fraction of the max.
    """
    n = len(a)

        return np.zeros(n)

    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            indices = k - np.arange(m_start, m_end + 1)
            grad[m_start:m_end + 1] += 2.0 * a[indices]
    
    return grad


def mirror_step(a, grad, lr):
    """Multiplicative weight update (mirror descent with KL Bregman)."""
    log_ratio = -lr * grad
    log_ratio -= log_ratio.max()
    weights = np.exp(log_ratio)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

        m_end = min(n - 1, k)
        if m_start <= m_end:
            indices = k - np.arange(m_start, m_end + 1)
            A_ub[row, m_start:m_end + 1] = a[indices]

    bounds = [(0, None)] * n
    
    options = {'presolve': True, 'time_limit': 15}

        return None

        return None
