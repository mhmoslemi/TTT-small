def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using adaptive top-K peak gradient
descent with Nesterov momentum, sparse LP direction finding, and pairwise
coordinate descent polish. Key innovations over prior approach: adaptive
peak weighting (replaces fixed-beta log-sum-exp), analytical pairwise
line search, and multi-scale initialization from smaller optimized sequences.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_peak_gradient(a, conv, K=30, temperature=0.15):
    """
    Gradient of soft top-K peak objective.
    Uses adaptive weighting focused on the most prominent peaks.
    """
    n = len(a)
    conv_max = conv.max()
    conv_mean = conv.mean()
    spread = conv_max - conv_mean
    if spread < 1e-15:
        return 2.0 * conv.sum() / (2 * n - 1) * np.ones(n)

    # Select peaks above adaptive threshold
    threshold = conv_mean + temperature * spread
    peak_mask = conv > threshold
    n_peaks = peak_mask.sum()

    # Ensure at least K peaks if possible
    if n_peaks < K:
        top_idx = np.argpartition(conv, -K)[-K:]
        peak_mask = np.zeros(len(conv), dtype=bool)
        peak_mask[top_idx] = True

    # Softmax weights on selected peaks (sharper for more focused gradient)
    selected = conv[peak_mask]
    sel_max = selected.max()
    sel_min = selected.min()
    sel_range = sel_max - sel_min
    if sel_range < 1e-15:
        w_sel = np.ones(len(selected)) / len(selected)
    else:
        # Adaptive sharpness: sharper when peaks are more separated
        sharpness = 5.0 + 10.0 * (n_peaks / len(conv))
        w_sel = np.exp((selected - sel_max) / (sel_range * 0.1 * sharpness))
        w_sel /= w_sel.sum()

    w = np.zeros(len(conv))
    w[peak_mask] = w_sel

    # Gradient via cross-correlation: grad_m = 2 * sum_k w[k] * a[k-m]
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)

    return 2.0 * cross[:n]


def nesterov_smoothed(a0, time_limit, K=30, temperature=0.15,
                       lr_init=0.012, momentum=0.92, seed=0,
                       perturb_every=350, perturb_scale=0.003):
    """
    Nesterov accelerated projected gradient descent with adaptive top-K peak gradient.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    cur_perturb = perturb_scale
    cur_every = perturb_every

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))

        # Adaptive K: start broad, narrow down
        K_cur = max(10, int(K * (1.0 - 0.5 * frac)))
        temp_cur = temperature * (1.0 + 0.5 * frac)

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a
        else:
            y = y / y_sum

        conv_y = fast_convolve(y)
        grad = compute_peak_gradient(y, conv_y, K_cur, temp_cur)

        # Adaptive learning rate: decrease over time
        lr = lr_init * (1.0 - 0.6 * frac)
        lr = max(lr, 0.0003)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        cur_obj = conv_new.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new

        # Periodic perturbation to escape local minima
        if no_improve >= cur_every:
            # Structured perturbation: move mass from high to low positions
            noise = rng.standard_normal(n) * cur_perturb
            # Bias perturbation toward "flattening" the sequence
            noise += 0.001 * (1.0 / n - a)
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            cur_every = int(cur_every * 1.3)
            cur_perturb = max(0.0005, cur_perturb * 0.7)

    return best_a, best_obj


def sparse_lp_direction(a, M, n, tolerance=0.02):
    """
    Solve LP with only binding (near-max) constraints for speed.
    max sum(b_j) s.t. b_j >= 0, (a*b)[k] <= M for k in binding set.
    """
    conv = fast_convolve(a)
    conv_max = conv.max()

    # Only include constraints within tolerance of max
    threshold = conv_max * (1.0 - tolerance)
    binding = np.where(conv > threshold)[0]

    if len(binding) == 0:
        binding = np.where(conv > conv_max * 0.9)[0]

    if len(binding) == 0:
        binding = np.array([np.argmax(conv)])

    c_obj = -np.ones(n)

    # Build sparse constraint matrix
    A_ub = np.zeros((len(binding), n))
    for idx, k in enumerate(binding):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j_range = np.arange(i_start, i_end + 1)
        A_ub[idx, j_range] = a[k - j_range]

    b_ub = np.full(len(binding), M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 15})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None

    return b / b_sum


def lp_phase(a0, time_limit, seed=42, n_iters_max=200):
    """LP-based direction finding with sparse constraints."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    current_a = a.copy()

    t0 = time.monotonic()
    stuck = 0
    iters = 0

    while time.monotonic() - t0 < time_limit and iters < n_iters_max:
        conv_cur = fast_convolve(current_a)
        M = conv_cur.max()

        improved = False
        for M_factor in [1.0, 1.002, 1.01, 1.02]:
            b = sparse_lp_direction(current_a, M * M_factor, n)
            if b is not None:
                break
        else:
            b = None

        if b is None:
            noise = rng.standard_normal(n) * 0.008
            current_a = project_simplex(current_a + noise)
            stuck += 1
            iters += 1
            if stuck > 8:
                break
            continue

        stuck = 0
        iters += 1

        # Line search along LP direction
        best_t = 0
        best_obj_try = M
        for t in [0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02]:
            a_try = (1 - t) * current_a + t * b
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()
            if obj_try < best_obj_try:
                best_obj_try = obj_try
                best_t = t

        if best_t > 0:
            current_a = (1 - best_t) * current_a + best_t * b
            current_a = np.maximum(current_a, 0)
            s = current_a.sum()
            if s > 1e-10:
                current_a = current_a / s
            if best_obj_try < best_obj - 1e-11:
                best_obj = best_obj_try
                best_a = current_a.copy()

    return best_a, best_obj


def pairwise_cd(a0, time_limit, seed=0):
    """
    Pairwise coordinate descent: transfer mass between pairs of positions
    with line search to reduce max convolution.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    sweep_count = 0

    while time.monotonic() - t0 < time_limit and no_improve < 50:
        conv_cur = fast_convolve(a)
        cur_max = conv_cur.max()

        # Identify top peaks to target
        n_peaks = min(10, 2 * n - 1)
        peak_idx = np.argpartition(conv_cur, -n_peaks)[-n_peaks:]
        peak_idx = peak_idx[conv_cur[peak_idx] > cur_max * 0.9]

        improved = False

        # For each peak, find the most contributing pair and try to reduce it
        for k in peak_idx:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end < i_start:
                continue

            # Find contributing pairs
            pairs_start = range(i_start, i_end + 1)
            contributions = np.array([a[i] * a[k - i] for i in pairs_start])
            if len(contributions) == 0:
                continue

            # Top contributing pairs
            top_pairs = np.argsort(contributions)[::-1][:3]
            for tp in top_pairs:
                i = i_start + int(tp)
                j = k - i
                if i == j:
                    # Diagonal: try moving mass to a random other position
                    for _ in range(2):
                        dest = rng.integers(0, n)
                        if dest == i:
                            continue
                        for frac in [0.05, 0.1, 0.15]:
                            amt = a[i] * frac
                            if amt < 1e-9:
                                break
                            a_try = a.copy()
                            a_try[i] -= amt
                            a_try[dest] += amt
                            conv_try = fast_convolve(a_try)
                            if conv_try.max() < best_obj - 1e-10:
                                best_obj = conv_try.max()
                                best_a = a_try.copy()
                                a = a_try.copy()
                                improved = True
                                break
                else:
                    # Off-diagonal: try reducing one of the pair members
                    for target in [i, j]:
                        if target < 0 or target >= n:
                            continue
                        for _ in range(2):
                            dest = rng.integers(0, n)
                            if dest == target:
                                continue
                            # Prefer destinations far from peak
                            for frac in [0.03, 0.05, 0.08, 0.12]:
                                amt = a[target] * frac
                                if amt < 1e-9:
                                    break
                                a_try = a.copy()
                                a_try[target] -= amt
                                a_try[dest] += amt
                                conv_try = fast_convolve(a_try)
                                if conv_try.max() < best_obj - 1e-10:
                                    best_obj = conv_try.max()
                                    best_a = a_try.copy()
                                    a = a_try.copy()
                                    improved = True
                                    break

        # Also try random pairwise transfers
        for _ in range(3):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.02, 0.04, 0.06, 0.1]:
                amt = a[i] * frac
                if amt < 1e-9:
                    break
                a_try = a.copy()
                a_try[i] -= amt
                a_try[j] += amt
                conv_try = fast_convolve(a_try)
                if conv_try.max() < best_obj - 1e-10:
                    best_obj = conv_try.max()
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            # Escalating perturbation
            if no_improve % 10 == 0:
                scale = 0.005 * (no_improve // 10 + 1)
                noise = rng.standard_normal(n) * scale
                a = project_simplex(a + noise)

        else:
            no_improve = 0

        sweep_count += 1

    return best_a, best_obj


def multi_scale_init(n, small_seq, rng):
    """Upscale a smaller sequence to length n via interpolation."""
    if len(small_seq) >= n:
        # Downsample
        idx = np.linspace(0, len(small_seq) - 1, n)
        result = np.interp(idx, np.arange(len(small_seq)), small_seq)
    else:
        # Upsample
        idx = np.linspace(0, len(small_seq) - 1, n)
        result = np.interp(idx, np.arange(len(small_seq)), small_seq)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-10:
        result /= s
    else:
        result = np.ones(n) / n
    return result


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    hann = np.maximum(hann, 1e-8)
    hann /= hann.sum()
    inits['hann'] = hann

    # Blackman
    a0, a1, a2 = 0.42, 0.5, 0.08
    blackman = a0 - a1 * np.cos(2 * np.pi * x / (n - 1)) + a2 * np.cos(4 * np.pi * x / (n - 1))
    blackman = np.maximum(blackman, 1e-8)
    blackman /= blackman.sum()
    inits['blackman'] = blackman

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc = np.maximum(rc, 1e-6)
    rc /= rc.sum()
    inits['raised_cos'] = rc

    # Gaussian (centered)
    for sf in [0.25, 0.35, 0.45, 0.55]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g

    # Bimodal
    bi = np.zeros(n)
    edge = max(3, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi = bi + 0.1  # small base to avoid exact zeros
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Triangular (peaked in middle)
    tri = 1.0 - 2.0 * np.abs(x - 0.5 * (n - 1)) / (n - 1)
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    inits['triangle'] = tri

    # Random Dirichlet
    inits['dirichlet_05'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['dirichlet_10'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['dirichlet_20'] = rng.dirichlet(np.ones(n) * 2.0)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
        # Perturbed parent
        if 'parent' in inits:
            p = inits['parent']
            p_pert = p + rng.standard_normal(n) * 0.02
            p_pert = np.maximum(p_pert, 0)
            ps = p_pert.sum()
            if ps > 1e-10:
                inits['parent_pert'] = p_pert / ps
        elif 'parent_ds' in inits:
            p = inits['parent_ds']
            p_pert = p + rng.standard_normal(n) * 0.02
            p_pert = np.maximum(p_pert, 0)
            ps = p_pert.sum()
            if ps > 1e-10:
                inits['parent_pert'] = p_pert / ps

    return inits

                break

            a0 = inits[name]
            a_opt, obj_opt = nesterov_smoothed(
                a0, t_alloc, K=25, temperature=0.15,
                lr_init=0.015, momentum=0.91, seed=rng.integers(0, 100000))

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 2 best: {best_score:.6f}", flush=True)

    # Phase 3: Deep optimization on top candidates
    print("=== Phase 3: Deep Optimization ===", flush=True)

    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt.copy()))
            seen.add(key)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))

    # Add a couple of random restarts at best length
    best_n = len(best_sequence) if best_sequence is not None else 384
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 0.6)))
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 1.5)))

    remaining = deadline - time.monotonic() - 200.0
    n_configs = len(top_configs)
    per_config = max(20.0, remaining / n_configs)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 150:
            break
        t_alloc = min(per_config, 250.0)
        if t_alloc < 10:
            break

        print(f"  Config {i}: n={n}, t_alloc={t_alloc:.0f}s", flush=True)

        # Sub-phase A: Nesterov gradient (40%)
        t_a = t_alloc * 0.40
        a_a, obj_a = nesterov_smoothed(
            a0, t_a, K=30, temperature=0.12,
            lr_init=0.012, momentum=0.92,
            perturb_every=300, perturb_scale=0.004,
            seed=rng.integers(0, 100000))
        score_a = float(evaluate_sequence(a_a.tolist()))
        if np.isfinite(score_a) and score_a < best_score:
            best_score = score_a
            best_sequence = a_a.copy()
            print(f"    Grad: {score_a:.6f}", flush=True)

        # Sub-phase B: LP direction (30%)
        t_b = t_alloc * 0.30
        if t_b > 8:
            a_b, obj_b = lp_phase(a_a, t_b, seed=rng.integers(0, 100000), n_iters_max=100)
            score_b = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"    LP: {score_b:.6f}", flush=True)
            a_a = a_b

        # Sub-phase C: Fine gradient (20%)
        t_c = t_alloc * 0.20
        if t_c > 5:
            a_c, obj_c = nesterov_smoothed(
                a_a, t_c, K=50, temperature=0.08,
                lr_init=0.005, momentum=0.94,
                perturb_every=200, perturb_scale=0.002,
                seed=rng.integers(0, 100000))
            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"    Fine: {score_c:.6f}", flush=True)
            a_a = a_c

        # Sub-phase D: Pairwise CD (10%)
        t_d = t_alloc * 0.10
        if t_d > 5:
            a_d, obj_d = pairwise_cd(a_a, t_d, seed=rng.integers(0, 100000))
            score_d = float(evaluate_sequence(a_d.tolist()))
            if np.isfinite(score_d) and score_d < best_score:
                best_score = score_d
                best_sequence = a_d.copy()
                print(f"    CD: {score_d:.6f}", flush=True)

    # Phase 4: Final polish on best sequence
    print("=== Phase 4: Final Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0

        # Extended pairwise CD
        t_cd = remaining * 0.4
        if t_cd > 10:
            a_cd, obj_cd = pairwise_cd(best_sequence, t_cd, seed=12345)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD polish: {best_score:.6f}", flush=True)

        # Very fine gradient
        remaining = deadline - time.monotonic() - 10.0
        t_fg = remaining * 0.35
        if t_fg > 5:
            a_fg, obj_fg = nesterov_smoothed(
                best_sequence, t_fg, K=50, temperature=0.05,
                lr_init=0.003, momentum=0.95,
                perturb_every=200, perturb_scale=0.001,
                seed=77777)
            score_fg = float(evaluate_sequence(a_fg.tolist()))
            if np.isfinite(score_fg) and score_fg < best_score:
                best_score = score_fg
                best_sequence = a_fg.copy()
                print(f"  FineGrad: {best_score:.6f}", flush=True)

        # Final LP attempt
        remaining = deadline - time.monotonic() - 10.0
        t_lp = remaining * 0.2
        if t_lp > 8:
            a_lp, obj_lp = lp_phase(best_sequence, t_lp, seed=99999, n_iters_max=50)
            score_lp = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_sequence = a_lp.copy()
                print(f"  FinalLP: {best_score:.6f}", flush=True)

        # Final tiny CD
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            a_fcd, obj_fcd = pairwise_cd(best_sequence, remaining, seed=54321)
            score_fcd = float(evaluate_sequence(a_fcd.tolist()))
            if np.isfinite(score_fcd) and score_fcd < best_score:
                best_score = score_fcd
                best_sequence = a_fcd.copy()
                print(f"  FinalCD: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
