Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
LP-based alternating minimization, smoothed gradient refinement,
spectral initialization, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix where (A @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b) <= M."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def lp_minimize_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b) <= M'."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    A_eq[0, n] = 0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None

def objective_normalized(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def lp_iterate(a, time_limit, method, step_init, max_iters, step_min, step_max):
    """LP-based alternating minimization."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        M = conv.max()
        if method == 'max_sum':
            b = lp_maximize_sum(a, M)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        else:
            b, _ = lp_minimize_peak(a)
            if b is None:
                break
            b = b / b.sum()
        if np.linalg.norm(b - a) < 1e-10:
            break
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(step_max, step * 1.03)
        else:
            no_improve += 1
            step = max(step_min, step * 0.93)
        a = a_new
        conv = conv_new
        if no_improve > 100:
            break
    return best_a, best_obj

def smooth_gradient_optimize(a, time_limit, lr, momentum, t_init, t_rate, max_iters):
    """Projected gradient on log-sum-exp smoothed objective."""
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    t = t_init
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        # Compute smooth gradient
        scaled = t * conv
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        p = exp_vals / exp_vals.sum()
        # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j]
        # This is 2 * conv(p_reversed, a) but let's compute directly
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-15:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                grad[j] += 2.0 * p[k] * a[k - j]
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        t = min(500.0, t * t_rate)
        if no_improve > 2000:
            break
    return best_a, best_obj

def generate_spectral_init(n, rng):
    """Generate a random spectral initialization."""
    # Random phases in frequency domain
    phases = rng.uniform(0, 2 * np.pi, n)
    # Flat spectrum
    spec = np.exp(1j * phases)
    a = np.real(np.fft.ifft(spec))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    
    for freq in [1, 2, 3]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0)
        a /= a.sum()
        inits[f'cos_{freq}'] = a
    
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    inits['spectral_1'] = generate_spectral_init(n, rng)
    inits['spectral_2'] = generate_spectral_init(n, rng)
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Power-law
    a = (1.0 + x / n) ** 1.5
    a /= a.sum()
    inits['power'] = a
    
    return inits

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build dense matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a):

def lp_iterate(a, time_limit, method, step_init, max_iters, step_min, step_max):
    """LP-based alternating minimization with adaptive step size."""
    a = np.maximum(a, 0.0)
    s = a.sum()

    no_improve = 0
    lp_time_total = 0.0

            if b_sum < 1e-15:

        else:  # 'linearized'

            if b is None:

            if b_sum < 1e-15:

def smooth_gradient_optimize(a, time_limit, lr, momentum, t_init, t_rate, max_iters):
    """Projected gradient on log-sum-exp smoothed objective with FFT-accelerated gradient."""
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    t = t_init
    no_improve = 0

    # Precompute FFT size for gradient computation
    conv_len = 2 * n - 1
    N_fft = 1 << (conv_len + n - 1 - 1).bit_length()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute smooth gradient via FFT
        scaled = t * conv
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        p = exp_vals / exp_vals.sum()

        # grad[j] = 2 * sum_k p[k] * a[k-j] = 2 * cross_correlation(p, a)[j]
        P = np.fft.rfft(p, N_fft)
        A = np.fft.rfft(a, N_fft)
        cross = np.fft.irfft(P * np.conj(A), N_fft)
        grad = 2.0 * cross[:n]

        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        t = min(500.0, t * t_rate)
        if no_improve > 3000:
            break

    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
LP-based alternating minimization with adaptive steps, FFT-accelerated smooth
gradient descent on log-sum-exp objective, spectral initializations,
multi-length exploration, and perturbation-based diversification.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build dense matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    A_eq[0, n] = 0.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None

    method: 'max_sum' or 'min_peak'
    """

def smooth_gradient_optimize(a, time_limit, lr, momentum, t_init, t_rate, max_iters):
    """Projected gradient on log-sum-exp smoothed objective with FFT-accelerated gradient."""
    a = np.maximum(a, 0.0)

            no_improve += 1
