import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_flatten_a(a, n_iter=100, blend_start=0.1, blend_end=0.5, seed=0):
    """Flatten the spectrum of a using iterative amplitude matching."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a / n
    a = a / s
    
    rng = np.random.default_rng(seed)
    
    for it in range(n_iter):
        A = np.fft.fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:

        a = a_new / s

def solve_lp_oracle(f, M, n_constraints=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    
    # Build constraint matrix
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    if n_constraints is None:
        n_constraints = 2 * n - 1
    
    A_ub = np.zeros((n_constraints, n))
    for j in range(n):
        # g[j] appears in (f*g)[k] for k = j, j+1, ..., j+n-1
        # (f*g)[k] includes f[k-j] * g[j]
        for k in range(j, min(j + n, n_constraints)):
            A_ub[k, j] = f[k - j]
    
    b_ub = np.full(n_constraints, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f)) / len(f)
    else:

        if g is None:
            no_improve += 1
            if no_improve > 20:
                # Perturb and restart
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisect
        if best_alpha > 0 and best_alpha < 1.0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(6):

def freq_subgradient_polish(f0, time_limit, lr=0.002, top_k=16, verbose=False):
    """Subgradient descent using frequency-domain peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    fft_size = 1 << (2 * n - 2).bit_length()

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        # Find top peaks in auto-convolution
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            # Weight by peak height (emphasize taller peaks)
            weights = (pv - conv.min() + 1e-10)
            weights /= weights.sum()

        # Subgradient in time domain
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * f[k - idx]
        sg *= 2.0  # derivative of (a*a)[k] w.r.t. a

        f = project_simplex(f - cur_lr * sg)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            best_obj = min(best_obj, conv.max())
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj

def coordinate_descent_polish(f0, time_limit, delta=0.005):
    """Coordinate descent: try adjusting each element slightly."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            # Try reducing a[i]
            if f[i] > delta:
                f_trial = f.copy()
                f_trial[i] -= delta
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st > 1e-15:
                    f_trial = f_trial / st
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-12:
                        f = f_trial
                        best_obj = val
                        improved = True
            # Try increasing a[i] (steal from the max element that's not i)
            max_idx = np.argmax(f)
            if max_idx != i and f[max_idx] > delta:
                f_trial = f.copy()
                f_trial[i] += delta
                f_trial[max_idx] -= delta
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st > 1e-15:
                    f_trial = f_trial / st
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-12:
                        f = f_trial
                        best_obj = val
                        improved = True

        sweeps += 1
        if not improved and sweeps > 2:
            # Try smaller steps
            delta *= 0.5
            if delta < 0.0001:
                break
        elif improved:
            delta = min(delta * 1.2, 0.01)

    return f, best_obj

def spectral_refinement(a, time_limit, n_iter=500, seed=42):
    """Spectral flattening with adaptive blending."""
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    best = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = np.fft.fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat spectrum
        S = mag ** 2
        S_mean = S.sum() / fft_size
        target_mag = np.sqrt(S_mean)
        
        # Adaptive blend: start gentle
        t = it / n_iter
        blend = 0.05 + 0.3 * t
        
        # Compress: new_mag = mag^(1-blend) * target_mag^blend (geometric mean)
        # This is more stable than arithmetic mean
        new_mag = mag ** (1 - blend) * (target_mag) ** blend
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best = a_new.copy()
    
    return best, best_obj

def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Sine wave
    for freq in [1, 2, 3]:
        s = np.abs(np.sin(2 * np.pi * freq * x / n)) + 0.1
        inits.append(s / s.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
    inits.append(g / g.sum())
    
    # Two peaks
    p1 = np.exp(-0.5 * ((x - n/4) / (n/8))**2)
    p2 = np.exp(-0.5 * ((x - 3*n/4) / (n/8))**2)
    inits.append((p1 + p2) / (p1 + p2).sum())
    
    # Alternating
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.3)
    inits.append(alt / alt.sum())
    
    # Quadratic
    q = 1.0 - 4.0 * ((x / (n-1) - 0.5))**2
    q = np.maximum(q, 0.1)
    inits.append(q / q.sum())
    
    return inits

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def l4_spectral_descent(a0, time_limit, n_iter=5000, lr_init=0.01, momentum_beta=0.9, lr_decay=0.9998, seed=0):

def spectral_amplitude_match(a0, time_limit, n_iter=500, blend_start=0.05, blend_end=0.3, seed=0):

def solve_lp_oracle(f, M):
    n = len(f)
    c = -np.ones(n)
    n_con = 2 * n - 1
    A_ub = np.zeros((n_con, n))
    for j in range(n):
        end = min(j + n, n_con)
        A_ub[j:end, j] = f[0:end - j]
    b_ub = np.full(n_con, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

def coordinate_descent(a0, time_limit, delta=0.004):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

    while time.monotonic() - t0 < time_limit:

            if time.monotonic() - t0 > time_limit:
                break
            if a[i] > delta:
                ft = a.copy()
                ft[i] -= delta
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st > 1e-15:
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_obj - 1e-13:
                        a = ft
                        best_obj = val
                        improved = True
            if max_idx != i and a[max_idx] > delta:
                ft = a.copy()
                ft[i] += delta
                ft[max_idx] -= delta
                ft = np.maximum(ft, 0)
                st = ft.sum()

        if not improved:
            delta *= 0.6
            if delta < 1e-5:
                break
        else:

def generate_inits(n, parent, rng):
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.4))
    x = np.arange(n, dtype=np.float64)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.4)
    inits.append(alt / alt.sum())
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.insert(0, px / ps)
    return inits

    print(f"=== Phase 3: Alternating (current best: {best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        strategies = ['l4', 'lp', 'spec', 'freq', 'l4', 'freq']
        for round_num, strat in enumerate(strategies):
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break

            if strat == 'l4':
                a_opt, _ = l4_spectral_descent(best_seq, time_limit=min(rem, 50),
                                               lr_init=0.003, momentum_beta=0.94, lr_decay=0.99995)
                _try(a_opt, f"alt_l4_{round_num}")
            elif strat == 'lp':
                a_opt, _ = lp_frank_wolfe(best_seq, time_limit=min(rem, 50))
                _try(a_opt, f"alt_lp_{round_num}")
            elif strat == 'spec':
                a_opt, _ = spectral_amplitude_match(best_seq, time_limit=min(rem, 40),
                                                    n_iter=300, blend_start=0.01, blend_end=0.15)
                _try(a_opt, f"alt_spec_{round_num}")
            elif strat == 'freq':
                a_opt, _ = freq_subgradient(best_seq, time_limit=min(rem, 40),
                                            lr_init=0.002, top_k=max(4, n_best // 20))
                _try(a_opt, f"alt_freq_{round_num}")

            # Perturbation escape
            if rem > 30:
                noise = rng.standard_normal(n_best) * 0.004
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = l4_spectral_descent(f_pert, time_limit=min(rem - 10, 25),
                                               lr_init=0.004, momentum_beta=0.93, lr_decay=0.9999)
                _try(a_opt, f"perturb_{round_num}")

    print(f"=== Phase 4: Polish (current best: {best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 50), delta=0.002)
            _try(a_opt, "coord")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = l4_spectral_descent(best_seq, time_limit=min(rem, 30),
                                           lr_init=0.001, momentum_beta=0.95, lr_decay=0.99998)
            _try(a_opt, "l4_final")

        rem = deadline - time.monotonic() - 8
        if rem > 8:
            a_opt, _ = freq_subgradient(best_seq, time_limit=min(rem, 20),
                                        lr_init=0.001, top_k=max(4, n_best // 32))
            _try(a_opt, "freq_final")

    if best_seq is None:
        n = 384
        best_seq = np.ones(n) / n
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"  Fallback used: {best_score:.6f}")

    # Final verification
    final_score = evaluate_sequence(best_seq.tolist())
    if isinstance(final_score, float) and np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
