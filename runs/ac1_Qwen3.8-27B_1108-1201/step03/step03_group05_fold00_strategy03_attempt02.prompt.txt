def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, conv, beta):
    """
    Gradient of logsumexp(beta*conv(a,a))/beta w.r.t. a.
    Uses FFT-based cross-correlation for efficiency.
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    w = conv * p  # w_k = c_k * softmax_k

    # grad_i = 2 * sum_{m=0}^{n-1} a[m] * w[m+i]
    # This is cross-correlation (a, w) at lag i
    # Need FFT size >= n + (2n-1) - 1 = 3n - 2
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)

    return 2.0 * cross[:n]


def lp_direction(a, M, n):
    """
    Solve LP: max sum(b_j) s.t. b_j >= 0, (a*b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)

    # Build constraint matrix: A_ub[k, i] = a[k-i] if 0 <= k-i < n
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None

    return b / b_sum


def optimize_smoothed(a0, time_limit, beta_init=30.0, beta_end=200.0,

                      perturb_scale=0.003, seed=0):

        grad = smoothed_gradient(y, conv_y, beta)

        # Momentum update
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic perturbation to escape local minima
        if no_improve >= perturb_every:

            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.25)
            lr = max(0.0005, lr * 0.65)

        # Adaptive learning rate
        lr = max(0.0003, lr_init * (1.0 - 0.7 * frac))

            a_try = np.maximum(a_try, 0.0)

            if s_try < 1e-10:
                continue
            a_try = a_try / s_try

            obj_try = conv_try.max()

            if obj_try < best_obj - 1e-10:
                best_obj = obj_try
                best_a = a_try.copy()
                current_a = a_try.copy()
                improved = True
                break
            elif obj_try < M * 0.999:
                current_a = a_try.copy()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                break

                    current_a = a_try.copy()
                    if obj_try < best_obj:

                        best_a = a_try.copy()

def peak_shaving(a0, time_limit, seed=0):
    """
    Targeted peak reduction: identify convolution peaks and adjust
    specific element pairs to reduce them.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        max_val = conv_cur.max()

        # Find top peaks
        top_k = min(5, len(conv_cur))
        peak_indices = np.argpartition(conv_cur, -top_k)[-top_k:]
        peak_indices = peak_indices[conv_cur[peak_indices] > max_val * 0.95]

        if len(peak_indices) == 0:
            break

        # For each peak, try to reduce it by moving mass
        improved = False
        for k in peak_indices:
            k = int(k)
            # Elements contributing to conv[k]: pairs (i, k-i) for valid i
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)

            # Find the pair with maximum contribution
            contributions = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
            if len(contributions) == 0:
                continue

            max_contrib_idx = np.argmax(contributions)
            i_max = i_start + max_contrib_idx

            # Try reducing a[i_max] and adding to a random other position
            for _ in range(3):
                j = rng.integers(0, n)
                if j == i_max or abs(j - i_max) < 2:
                    continue

                # Try transferring small amount
                for frac in [0.05, 0.1, 0.2]:
                    amt = a[i_max] * frac
                    if amt < 1e-8:
                        break
                    a_try = a.copy()
                    a_try[i_max] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-10:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break

            if improved:
                break

        iters += 1

        # Also try random pair adjustments
        if not improved and iters % 500 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                for frac in [0.02, 0.05, 0.1]:
                    amt = a[i] * frac
                    if amt < 1e-8:
                        break
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-10:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()

    # Triangle (inverted - peak in middle)
    tri = 1.0 / (1.0 + 0.5 * ((x - n / 2) / (n / 4)) ** 2)
    tri /= tri.sum()
    inits['bell'] = tri

    # Gaussians at different widths
    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g

    # Bimodal (concentrated at edges)
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Raised cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w

    # Off-center Gaussian
    g_off = np.exp(-0.5 * ((x - 0.3 * n) / (0.2 * n)) ** 2)
    g_off /= g_off.sum()
    inits['gauss_off'] = g_off

    # Random
    inits['rand_dirichlet'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand_dirichlet2'] = rng.dirichlet(np.ones(n) * 2.0)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits
