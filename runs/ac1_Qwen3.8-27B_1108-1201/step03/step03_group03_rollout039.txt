1. **Tighter LP constraints**: Use only near-peak constraints (adaptive top-K) for faster LP solves, enabling more iterations
2. **Line search on step size**: After finding LP direction, binary search for optimal blend
3. **Mirror descent**: Multiplicative weights update that naturally maintains non-negativity
4. **Better initialization**: Anti-bowl shapes (larger at edges) to reduce central autocorrelation
5. **Multi-length exploration**: Try different n values since the optimal trade-off varies with n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP directional optimization
with adaptive constraint selection, line search, mirror descent refinement,
and coordinate descent polishing. Explores multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute objective for normalized sequence (sum=1)."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def solve_lp_direction(f, M, n, top_k=100):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for top_k peak positions."""
    conv = fast_convolve(f)
    if top_k >= 2 * n - 1:
        peak_indices = np.arange(2 * n - 1)
    else:
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    
    A_ub = np.zeros((len(peak_indices), n))
    b_ub = np.zeros(len(peak_indices))
    for idx, k in enumerate(peak_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[idx, j] = f[k - j]
        b_ub[idx] = M
    
    c_vec = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 10})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_with_tightening(f, M, n, top_k=80, n_targets=3):
    """Try multiple target values, return best (g, target_M) pair."""
    best_g = None
    best_target = M
    best_sum = -1
    
    targets = [M * (1 - 0.005 * i) for i in range(n_targets)]
    targets = [max(t, 1e-10) for t in targets]
    
    for target in targets:
        g = solve_lp_direction(f, target, n, top_k)
        if g is not None:
            gs = g.sum()
            if gs > best_sum:
                best_sum = gs
                best_g = g
                best_target = target
            break
    
    if best_g is None:
        best_g = solve_lp_direction(f, M, n, top_k)
    
    return best_g, best_target


def line_search_blend(a, g, t_low, t_high, n_iterations=12):
    """Binary search for optimal blend parameter in [t_low, t_high]."""
    conv_a = fast_convolve(a)
    obj_a = conv_a.max()
    
    best_t = t_low
    best_obj = obj_a
    
    t = 0.5 * (t_low + t_high)
    for _ in range(n_iterations):
        a_try = (1.0 - t) * a + t * g
        a_try = project_simplex(a_try)
        conv_try = fast_convolve(a_try)
        obj_try = conv_try.max()
        
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
        
        if obj_try < obj_a:
            t_low = t
        else:
            t_high = t
        t = 0.5 * (t_low + t_high)
    
    return best_t, best_obj


def optimize_lp_fast(a0, time_limit, top_k_init=60, top_k_max=200,
                     blend_init=0.3, blend_min=0.01):
    """Fast LP-based optimization with adaptive top-K and line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    top_k = top_k_init
    prev_best = best_obj
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        # Adaptive top-K: increase if LP solution is too sparse
        g, target_M = solve_lp_with_tightening(a, M, n, top_k=top_k)
        if g is None:
            top_k = min(top_k_max, int(top_k * 1.5))
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check sparsity
        if g.max() > 0.3:
            uniform = np.ones(n) / n
            g = 0.7 * g + 0.3 * uniform
            g = g / g.sum()
        
        # Line search for optimal blend
        t_opt, obj_try = line_search_blend(a, g, 0.0, blend, n_iterations=10)
        
        if obj_try < best_obj:
            a = (1.0 - t_opt) * a + t_opt * g
            a = project_simplex(a)
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        # Adapt parameters
        if stall > 0:
            blend = max(blend_min, blend * 0.8)
            if stall % 5 == 0:
                top_k = min(top_k_max, int(top_k * 1.3))
        
        if blend < blend_min and stall > 30:
            # Perturbation to escape
            rng_local = np.random.default_rng(hash(int(a.sum() * 1e12)) % (2**31))
            a = project_simplex(a + rng_local.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            stall = 0
            blend = max(blend_min, 0.1)
        
        if time.monotonic() - t0 > time_limit - 5:
            break
    
    return best_a, best_obj


def mirror_descent(a0, time_limit, lr_init=0.01, decay=0.999, top_k=30):
    """Mirror descent (multiplicative weights) on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if conv.max() < 1e-15:
            break
        
        # Find top peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        
        # Subgradient: weighted sum of convolution gradients at peaks
        sg = np.zeros(n)
        w_sum = 0.0
        for idx in range(len(peaks)):
            k = int(peaks[idx])
            w = peak_vals[idx] ** 2  # emphasize larger peaks
            w_sum += w
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += w * a[k - j]
        
        if w_sum < 1e-15:
            break
        sg /= w_sum
        
        # Multiplicative update
        a_new = a * np.exp(-lr * sg)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15 or not np.isfinite(a_new_sum):
            a_new = project_simplex(a - lr * sg)
        else:
            a_new = a_new / a_new_sum
            a_new = np.maximum(a_new, 0.0)
            a_new_sum = a_new.sum()
            if a_new_sum > 1e-15:
                a_new /= a_new_sum
            else:
                a_new = project_simplex(a - lr * sg)
        
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            a = project_simplex(a + np.random.default_rng().standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            no_improve = 0
            lr = max(0.001, lr * 0.5)
        
        lr *= decay
    
    return best_a, best_obj


def subgradient_momentum(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=40):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if conv.max() < 1e-15:
            break
        
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        
        # Softmax weighting
        scaled = peak_vals / (0.05 * pmax + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 150:
            a = project_simplex(a + np.random.default_rng().standard_normal(n) * 0.008)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.003, lr * 0.75)
        
        lr *= 0.9995
    
    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=50, delta_frac=0.03):
    """Coordinate descent: try small mass transfers between coordinates."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    # Precompute which pairs to try based on current peaks
    for rnd in range(max_rounds):
        improved = False
        peaks = np.argpartition(conv, -10)[-10:]
        
        # For each peak, try reducing contributing pairs
        for k in peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            
            # Find top contributing j
            contributions = []
            for j in range(j_start, j_end + 1):
                contrib = a[j] * a[k - j]
                contributions.append((contrib, j))
            contributions.sort(reverse=True)
            
            for contrib, j in contributions[:5]:
                if contrib < 1e-10:
                    continue
                delta = a[j] * delta_frac
                if delta < 1e-8:
                    continue
                
                # Try moving mass from j to a "quiet" position
                a_test = a.copy()
                a_test[j] -= delta
                a_test = np.maximum(a_test, 0.0)
                
                # Find quiet position (low contribution to all peaks)
                quiet_score = np.zeros(n)
                for pk in peaks:
                    pk = int(pk)
                    j2_start = max(0, pk - n + 1)
                    j2_end = min(n - 1, pk)
                    for j2 in range(j2_start, j2_end + 1):
                        quiet_score[j2] += a_test[pk - j2]
                
                # Pick position with lowest quiet score that has some mass
                candidates = np.where(quiet_score < np.median(quiet_score))[0]
                if len(candidates) == 0:
                    candidates = np.arange(n)
                
                # Try a few candidates
                for target in candidates[:5]:
                    a_try = a_test.copy()
                    a_try[target] += delta
                    st = a_try.sum()
                    if st < 1e-15:
                        continue
                    a_try /= st
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-13:
                        a = a_try
                        best_obj = obj_try
                        conv = conv_try
                        improved = True
                        break
        
        if not improved:
            break
    
    return a


def gen_initializations(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Anti-bowl (larger at edges)
    alpha = 0.5
    anti_bowl = 1.0 + alpha * np.cos(np.pi * x / (n - 1))
    anti_bowl = np.maximum(anti_bowl, 0.01)
    inits['anti_bowl'] = anti_bowl / anti_bowl.sum()
    
    # Strong anti-bowl
    alpha = 0.9
    anti_bowl2 = 1.0 + alpha * np.cos(np.pi * x / (n - 1))
    anti_bowl2 = np.maximum(anti_bowl2, 0.01)
    inits['anti_bowl_strong'] = anti_bowl2 / anti_bowl2.sum()
    
    # Edge concentration
    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.2 * n) / (0.15 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * n) / (0.15 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_res = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(parent)), p)
                p_res = np.maximum(p_res, 0)
                prs = p_res.sum()
                if prs > 1e-10:
                    inits['parent_res'] = p_res / prs
                # Perturbed parent
                if len(parent) == n:
                    p_pert = p / ps + rng.standard_normal(n) * 0.02
                    p_pert = np.maximum(p_pert, 0)
                    pps = p_pert.sum()
                    if pps > 1e-10:
                        inits['parent_pert'] = p_pert / pps
    
    return inits


def full_optimize(a0, time_limit):
    """Run full optimization pipeline: LP + subgrad + mirror + CD."""
    n = len(a0)
    
    # Phase A: LP optimization (60% of time)
    t_lp = time_limit * 0.55
    a_opt, obj_lp = optimize_lp_fast(a0, t_lp, top_k_init=50, top_k_max=150,
                                      blend_init=0.25, blend_min=0.008)
    
    # Phase B: Subgradient refinement (15% of time)
    t_sg = time_limit * 0.15
    a_opt2, obj_sg = subgradient_momentum(a_opt, t_sg, lr_init=0.006,
                                           momentum=0.92, top_k=50)
    if obj_sg < obj_lp:
        a_opt, obj_lp = a_opt2, obj_sg
    
    # Phase C: Mirror descent (15% of time)
    t_md = time_limit * 0.15
    a_opt3, obj_md = mirror_descent(a_opt, t_md, lr_init=0.005,
                                     decay=0.998, top_k=40)
    if obj_md < obj_lp:
        a_opt, obj_lp = a_opt3, obj_md
    
    # Phase D: Coordinate descent polish (remaining time)
    t_cd = time_limit * 0.15
    rounds = max(20, int(t_cd / 0.5))
    a_opt = coordinate_descent_polish(a_opt, max_rounds=rounds)
    obj_cd = compute_obj(a_opt)
    if obj_cd < obj_lp:
        obj_lp = obj_cd
    
    return a_opt, obj_lp

                                               blend_init=0.15, blend_min=0.005)
            a_opt2, obj2 = subgradient_momentum(a_opt, t_cyc * 0.25,
                                                 lr_init=0.003, momentum=0.93,
                                                 top_k=60)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Cycle {cycle}: {score:.6f}")
            
            # Quick CD
            a_cd2 = coordinate_descent_polish(best_sequence, max_rounds=20)
            score_cd2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(score_cd2) and score_cd2 < best_score:
                best_score = score_cd2
                best_sequence = a_cd2.copy()
                print(f"  CD cycle {cycle}: {score_cd2:.6f}")
        
        # Mirror descent final pass
        if time.monotonic() < deadline - 15:
            t_md = min(30.0, deadline - time.monotonic() - 10)
            if t_md > 5:
                a_md, obj_md = mirror_descent(best_sequence, t_md, lr_init=0.003,
                                               decay=0.997, top_k=50)
                score_md = float(evaluate_sequence(a_md.tolist()))
                if np.isfinite(score_md) and score_md < best_score:
                    best_score = score_md
                    best_sequence = a_md.copy()
                    print(f"  Mirror: {score_md:.6f}")
        
        # Random restart perturbations
        if time.monotonic() < deadline - 15:
            n_b = len(best_sequence)
            for ps in [0.002, 0.005, 0.01]:
                if time.monotonic() > deadline - 12:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 10:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_opt, _ = optimize_lp_fast(a_p, 5.0, top_k_init=40,
                                                 top_k_max=80, blend_init=0.2)
                    a_opt2, obj2 = subgradient_momentum(a_opt, 3.0,
                                                         lr_init=0.004, momentum=0.9,
                                                         top_k=40)
                    if obj2 < compute_obj(a_opt):
                        a_opt = a_opt2
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Perturb({ps}): {score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
