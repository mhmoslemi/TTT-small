"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-scale L-BFGS-B optimization
with peak targeting, symmetric sequence exploitation, and adaptive beta annealing.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_conv(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a using FFT."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_func(a, beta):
    """Compute smooth objective and gradient at point a (normalized to sum=1)."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return 2.0 * n, np.zeros(n)
    a = a / s
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    obj = 2.0 * n * sm
    grad = compute_grad_conv(a, w)
    return obj, grad


def optimize_smooth(a0, beta, maxiter=300):
    """L-BFGS-B optimization with simplex constraint via softmax reparam."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-30)
    a0 = a0 / a0.sum()
    x0 = np.log(a0)
    
    def obj(x):
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm
    
    def jac(x):
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        grad_a = compute_grad_conv(a, w)
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange)
    
    result = minimize(obj, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def peak_shave(a, n_steps=20, strength=0.5):
    """Reduce the peak of the auto-convolution by redistributing mass."""
    a = a.copy()
    n = len(a)
    for _ in range(n_steps):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        target = conv[max_k]
        
        # Find contributing indices
        i_start = max(0, max_k - n + 1)
        i_end = min(n - 1, max_k)
        
        # Contribution of each a[i] to conv[max_k]
        contrib = np.zeros(n)
        for i in range(i_start, i_end + 1):
            j = max_k - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break
        
        # Reduce contributing elements, increase others
        reduce_mask = contrib > 0
        increase_mask = ~reduce_mask
        
        if not increase_mask.any():
            break
        
        # Amount to move
        amount = strength * (target - conv.mean()) * contrib / total_contrib
        amount = np.minimum(amount, a * 0.5)  # Don't reduce below half
        
        a[reduce_mask] -= amount[reduce_mask]
        a = np.maximum(a, 0)
        
        # Distribute to non-contributing elements
        excess = amount[reduce_mask].sum()
        if excess > 0 and increase_mask.sum() > 0:
            # Distribute proportional to current values
            weights = a[increase_mask]
            if weights.sum() < 1e-15:
                a[increase_mask] += excess / increase_mask.sum()
            else:
                a[increase_mask] += excess * weights / weights.sum()
        
        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a /= s
    
    return a

    fft_size = 1 << (n - 1).bit_length()
    best_a = None
    best_obj = float('inf')

            if k < fft_size - k:

        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()

            if obj < best_obj:

def random_init(n, rng, alpha=1.0):
    """Random Dirichlet initialization."""
    return rng.dirichlet(np.ones(n) * alpha)


def parent_init(n, parent, rng):
    """Initialize from parent sequence."""
    if parent is None:
        return np.ones(n) / n
    p = np.maximum(parent, 0)
    ps = p.sum()
    if ps < 1e-10:
        return np.ones(n) / n
    if len(p) == n:
        return p / ps
    # Interpolate
    pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
    pp = np.maximum(pp, 0)
    ps2 = pp.sum()
    if ps2 < 1e-10:
        return np.ones(n) / n
    return pp / ps2

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-scale L-BFGS-B optimization
with peak shaving, adaptive beta annealing, diverse initializations including
flat-spectrum designs, and multi-length exploration to beat the previous bound.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_conv(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a using FFT cross-correlation."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _lbfgs_obj(x, n, beta):
    """Top-level objective for scipy minimize."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _lbfgs_jac(x, n, beta):
    """Top-level Jacobian for scipy minimize."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_smooth(a0, beta, maxiter=300):
    """L-BFGS-B optimization with softmax reparametrization of simplex."""
    n = len(a0)
    a0 = np.maximum(a0, 1e-30)
    a0 = a0 / a0.sum()
    x0 = np.log(a0)

    result = minimize(_lbfgs_obj, x0, jac=_lbfgs_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 40})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def peak_shave(a, n_steps=20, strength=0.4):
    """Reduce the peak of the auto-convolution by targeted redistribution."""
    a = a.copy()
    n = len(a)
    for step in range(n_steps):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        target_val = conv[max_k]
        mean_val = conv.mean()

        if target_val <= mean_val * 1.01:
            break

        # Find contributing indices to lag max_k
        i_start = max(0, max_k - n + 1)
        i_end = min(n - 1, max_k)

        # Compute contribution of each a[i] to conv[max_k]
        contrib = np.zeros(n)
        for i in range(i_start, i_end + 1):
            j = max_k - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        reduce_mask = contrib > 0
        increase_mask = ~reduce_mask

        if not increase_mask.any():
            break

        # Amount to reduce (proportional to contribution)
        excess_to_remove = strength * (target_val - mean_val)
        proportions = contrib / total_contrib
        amount = excess_to_remove * proportions
        amount = np.minimum(amount, a * 0.4)  # Don't reduce more than 40%

        a[reduce_mask] -= amount[reduce_mask]
        a = np.maximum(a, 0)

        # Redistribute to non-contributing elements
        excess = amount[reduce_mask].sum()
        if excess > 1e-15 and increase_mask.sum() > 0:
            weights = a[increase_mask].astype(np.float64)
            if weights.sum() < 1e-15:
                a[increase_mask] += excess / increase_mask.sum()
            else:
                a[increase_mask] += excess * weights / weights.sum()

        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a /= s

    return a


def flat_spectrum_init(n, rng, n_trials=25):
    """Initialize with approximately flat power spectrum (minimize PAPR)."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')

    for _ in range(n_trials):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        half = fft_size // 2
        phases = rng.uniform(0, 2 * np.pi, half + 1)
        for k in range(1, half + 1):
            A[k] = np.exp(1j * phases[k])
            if k < fft_size - k:
                A[fft_size - k] = np.exp(-1j * phases[k])

        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)

        if s > 1e-10:

            obj = 2.0 * n * fast_convolve(a).max()

                best_a = a.copy()

            deep_candidates.append((sc, n, a))

    for dc_idx in range(len(deep_candidates)):
        if time.monotonic() > phase2_end - 30:
            break
        sc, n, a_cur = deep_candidates[dc_idx]
        print("  Deep optimizing n=%d (initial score=%.4f)" % (n, sc))

        # Multi-beta annealing with peak shaving
        remaining = phase2_end - time.monotonic() - 20
        n_betas = max(5, int(remaining / 15))
        n_betas = min(n_betas, 15)
        a_cur = multi_beta_optimize(a_cur, n_betas=n_betas)
        score = do_score(a_cur)
        deep_candidates[dc_idx] = (score, n, a_cur.copy())

    deep_candidates.sort(key=lambda x: x[0])
    print("Phase 2 complete. Best: %.6f" % best_score)

    # === PHASE 3: Random Restarts (150s) ===
    print("=== Phase 3: Random Restarts ===")
    phase3_end = deadline - 150.0

    if best_sequence is not None:
        best_n = len(best_sequence)
        n_restarts = 10
        time_per_restart = max(8.0, (phase3_end - time.monotonic() - 10) / n_restarts)

        for r in range(n_restarts):
            if time.monotonic() > phase3_end - 8:
                break
            t_alloc = min(time_per_restart, phase3_end - time.monotonic() - 5)
            if t_alloc < 5:
                break

            # Different perturbation strategies
            strategy = r % 4
            if strategy == 0:
                # Small Gaussian perturbation
                scale = 0.01 + 0.02 * rng.random()
                noise = rng.standard_normal(best_n) * scale
                a_pert = project_simplex(best_sequence / best_sequence.sum() + noise)
            elif strategy == 1:
                # Block swap
                a_pert = best_sequence / best_sequence.sum()
                block = max(2, best_n // 8)
                i = rng.integers(0, best_n - block)
                a_pert[i:i + block], a_pert[i + block:i + 2 * block] = \
                    a_pert[i + block:i + 2 * block].copy(), a_pert[i:i + block].copy()
                a_pert = np.maximum(a_pert, 0)
                s = a_pert.sum()
                a_pert = a_pert / s
            elif strategy == 2:
                # Exploit from different length
                alt_n = best_n if best_n != 384 else 512
                a_alt = np.interp(np.linspace(0, 1, best_n),
                                  np.linspace(0, 1, alt_n),
                                  np.ones(alt_n) / alt_n)
                blend = rng.uniform(0.3, 0.7)
                a_pert = project_simplex(blend * best_sequence / best_sequence.sum() +
                                         (1 - blend) * a_alt)
            else:
                # Frequency-domain perturbation
                a_pert = best_sequence / best_sequence.sum()
                fft_size = 1 << (best_n - 1).bit_length()
                A = fft(a_pert, fft_size)
                # Randomly adjust some frequency bins
                k_idx = rng.choice(range(1, fft_size // 2), size=max(1, fft_size // 16), replace=False)
                for k in k_idx:
                    A[k] *= (1 + 0.1 * rng.standard_normal())
                    A[fft_size - k] = np.conj(A[k])
                a_pert = np.real(ifft(A, fft_size))[:best_n]
                a_pert = project_simplex(a_pert)

            # Quick optimization
            maxiter = int(min(200, t_alloc * 15))
            a_opt = optimize_smooth(a_pert, beta=30, maxiter=maxiter)
            a_opt = peak_shave(a_opt, n_steps=8, strength=0.25)
            score = do_score(a_opt)

            # If promising, do deeper optimization
            if score < best_score + 0.05 and t_alloc > 25:
                a_opt = optimize_smooth(a_opt, beta=10, maxiter=int(min(250, t_alloc * 10)))
                a_opt = peak_shave(a_opt, n_steps=10, strength=0.2)
                do_score(a_opt)

    print("Phase 3 complete. Best: %.6f" % best_score)

    # === PHASE 4: Final Polish (150s) ===
    print("=== Phase 4: Final Polish ===")

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10

        # Very low beta for precise optimization
        low_betas = [2.0, 1.5, 1.2, 1.0, 0.8]
        for beta in low_betas:
            if remaining < 8:
                break
            t_single = min(remaining / (len(low_betas) + 1), 30.0)
            maxiter = int(min(500, max(50, t_single * 25)))
            a_opt = optimize_smooth(best_sequence / best_sequence.sum(), beta=beta, maxiter=maxiter)
            a_opt = peak_shave(a_opt, n_steps=15, strength=0.15)
            do_score(a_opt)
            remaining = deadline - time.monotonic() - 10

        # Try nearest different length
        if remaining > 50 and best_sequence is not None:
            bn = len(best_sequence)
            alt_lengths = [l for l in [bn * 2, bn // 2, 384, 512, 256]
                          if l != bn and 64 <= l <= 1024]
            if alt_lengths:
                alt_n = alt_lengths[0]
                a0 = np.interp(np.linspace(0, 1, alt_n),
                               np.linspace(0, 1, bn),
                               best_sequence)
                a0 = np.maximum(a0, 0)
                s = a0.sum()
                if s > 1e-10:
                    a0 /= s
                    t_alloc = min(remaining - 10, 40)
                    if t_alloc > 10:
                        a_opt = optimize_smooth(a0, beta=40, maxiter=int(min(300, t_alloc * 20)))
                        a_opt = peak_shave(a_opt, n_steps=10, strength=0.3)
                        do_score(a_opt)

        # Final peak shaving on best
        if remaining > 15:
            a_opt = peak_shave(best_sequence / best_sequence.sum(), n_steps=30, strength=0.1)
            do_score(a_opt)

            # Re-optimize from shaved point
            if remaining > 20:
                a_opt = optimize_smooth(a_opt, beta=5, maxiter=int(min(300, remaining * 15)))
                do_score(a_opt)

    # Ensure we have a valid result
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        print("Warning: using fallback uniform sequence")

    # Final cleanup
    best_sequence = np.maximum(best_sequence, 0.0)

    # Verify final score
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print("\nFINAL SCORE: %.6f (n=%d)" % (final_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
