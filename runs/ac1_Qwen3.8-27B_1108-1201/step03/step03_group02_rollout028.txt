"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Smooth optimization via log-sum-exp with L-BFGS on softmax-parameterized variables
2. LP-based refinement for precise peak reduction
3. Subgradient polishing with adaptive parameters
4. Multi-restart with diverse initializations
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear cross-convolution via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def smooth_objective_and_grad(x, t):
    """
    Compute smoothed objective and gradient for softmax-parameterized a.
    a = softmax(x), objective = 2*n*soft_max(conv(a,a), t)
    """
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)
    
    # Log-sum-exp smoothing
    scaled = t * conv
    scaledmax = scaled.max()
    log_sum_exp = scaledmax + np.log(np.exp(scaled - scaledmax).sum())
    obj = 2.0 * n * log_sum_exp / t
    
    # Gradient of soft_max wrt conv
    p = np.exp(scaled - scaledmax) / np.exp(scaled - scaledmax).sum()
    d_obj_d_conv = 2.0 * n * p
    
    # Gradient of conv wrt a: d(conv[k])/d(a[i]) = a[k-i] (if in range)
    # Then d(a[i])/d(x[j]) = a[i]*(delta_ij - a[j])
    # So d_obj/d(x[j]) = sum_i d_obj/d_conv * d_conv/d_a[i] * d_a[i]/d_x[j]
    # = sum_k d_obj_d_conv[k] * sum_i:valid a[k-i] * d_a[i]/d_x[j]
    # = sum_k d_obj_d_conv[k] * (sum_i:valid a[k-i] * a[i] * delta_ij - a[j] * sum_i:valid a[k-i] * a[i])
    # = sum_k d_obj_d_conv[k] * (a[j] * a[k-j] - a[j] * conv[k])  ... no this isn't right
    
    # Let me be more careful.
    # d(conv[k])/d(a[i]) = a[k-i] if 0 <= k-i < n, else 0
    # So d_obj/d(a[i]) = sum_k d_obj_d_conv[k] * a[k-i] (where k-i is valid)
    # This is the cross-correlation of d_obj_d_conv with a, which is conv(d_obj_d_conv, a) reversed...
    # Actually: d_obj/d(a[i]) = sum_{k: 0<=k-i<n} d_obj_d_conv[k] * a[k-i]
    # Let j = k-i, then k = i+j, so: = sum_{j=0}^{n-1} d_obj_d_conv[i+j] * a[j] (with bounds)
    # This is the cross-correlation of d_obj_d_conv with a, evaluated at lag i.
    
    d_obj_d_a = np.zeros(n)
    for i in range(n):
        # d_obj_d_a[i] = sum_j d_obj_d_conv[i+j] * a[j] for valid j and i+j

    sum_d_a_weighted = np.dot(d_obj_d_a, a)
    grad_x = a * (d_obj_d_a - sum_d_a_weighted)
    
    return obj, grad_x


def smooth_objective_and_grad_fast(x, t):
    """
    Faster version using FFT for the d_obj_d_a computation.
    """
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)
    
    scaled = t * conv
    scaledmax = scaled.max()
    log_sum_exp = scaledmax + np.log(np.exp(scaled - scaledmax).sum())
    obj = 2.0 * n * log_sum_exp / t
    
    p = np.exp(scaled - scaledmax) / np.exp(scaled - scaledmax).sum()
    d_obj_d_conv = 2.0 * n * p
    
    # d_obj_d_a[i] = cross-correlation of d_obj_d_conv with a at lag i
    # = sum_j d_obj_d_conv[i+j] * a[j] for valid indices
    # This is the "unreversed" cross-correlation, which can be computed as:
    # cross_corr(m, n)[i] = sum_j m[i+j] * n[j]
    # = (m * ifft(conj(fft(n))))... hmm, let me think.
    # Actually: sum_j m[i+j] * n[j] = (m correlated with reversed n) ...

    # cross_corr[i] = sum_j m[i+j] * n[j]
    # If we reverse n: n_rev[j] = n[len(n)-1-j]
    # Then sum_j m[i+j] * n[j] = sum_j m[i+j] * n_rev[len(n)-1-j]
    # This is the cross-correlation of m and n_rev at a specific lag...
    
    # Simpler: just compute it directly for n=512 (it's only 512*512/2 ≈ 131K operations)
    d_obj_d_a = np.zeros(n)
    for i in range(n):
        end = min(n, len(d_obj_d_conv) - i)
        if end > 0:
            d_obj_d_a[i] = np.dot(d_obj_d_conv[i:i+end], a[:end])
    
    sum_d_a_weighted = np.dot(d_obj_d_a, a)
    grad_x = a * (d_obj_d_a - sum_d_a_weighted)
    
    return obj, grad_x


def optimize_smooth(x0, time_limit, t_init=50.0, t_final=500.0, max_iters=2000):
    """
    Optimize using smooth objective with increasing temperature.
    Uses custom gradient descent with momentum.
    """
    x = x0.copy()
    n = len(x)
    t = t_init
    t0 = time.monotonic()
    
    best_a = softmax(x)
    best_obj = 2.0 * n * fast_convolve(best_a).max()
    
    v = np.zeros(n)
    lr = 0.5
    momentum = 0.9
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        obj, grad = smooth_objective_and_grad_fast(x, t)
        
        # Adaptive learning rate
        v = momentum * v - lr * grad
        x = x + v
        # Subtract mean to fix gauge (softmax is invariant to adding constant)
        x = x - x.mean()
        
        a = softmax(x)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            lr = min(2.0, lr * 1.02)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.9
            if no_improve > 200:
                # Increase temperature to smooth more
                t = min(t_final, t * 1.5)
                no_improve = 0
                lr = 0.3
        
        # Slowly increase temperature
        if it % 100 == 0 and t < t_final:
            t *= 1.3
    
    return best_a, best_obj


def build_conv_matrix_sparsity(a):
    """Build the convolution matrix in a memory-efficient way for medium n."""
    n = len(a)
    # For n up to ~512, the dense matrix is 1023 x 512 ≈ 524K entries, which is fine
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_minimize_peak(a, M_init=None):
    """
    Solve: min M s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M for all k.
    Returns (b, M) or (None, None) if infeasible.
    """
    n = len(a)
    A = build_conv_matrix_sparsity(a)
    num_rows = 2 * n - 1
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'maxiter': 5000})

def subgradient_refine(a0, time_limit, top_k=20, temp_init=0.0002,

    temp = temp_init

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            peak_vals = conv.copy()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
        
        # Weighted subgradient
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            wi = w[i] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]

        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        temp *= 1.002
        
        if no_improve > 1000:

            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach combining:
1. Smooth optimization via log-sum-exp on softmax-parameterized variables
2. LP-based iterative refinement for precise peak reduction

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear cross-convolution via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def logsumexp_fn(v):
    """Numerically stable log-sum-exp."""
    v_max = v.max()
    return v_max + np.log(np.exp(v - v_max).sum())

def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_minimize_peak(a):
    """Solve: min M s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'maxiter': 10000})
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'maxiter': 10000})
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def subgradient_refine(a0, time_limit, top_k=25, temp_init=0.0001,
                       lr_init=0.025, momentum_val=0.88):
    """Multi-peak subgradient with adaptive temperature and learning rate."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            peak_vals = conv.copy()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]

        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w = w / w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            wi = w[i] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]

        v = momentum_val * v - lr * sg
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9993
        temp *= 1.003

        if no_improve > 800:
            noise = rng.standard_normal(n) * 0.004
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            lr = max(lr_init * 0.3, lr * 0.4)

    return best_a, best_obj

def peak_shaving(a0, time_limit):
    """
    Direct peak shaving: identify the peak lag and reduce contributions
    by redistributing mass from high-product pairs to low-product ones.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    for it in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Find pairs (i, peak_k - i) that contribute to the peak
        j_start = max(0, peak_k - n + 1)
        j_end = min(n - 1, peak_k)
        idx = np.arange(j_start, j_end + 1)
        products = a[idx] * a[peak_k - idx]
        total_prod = products.sum()

        if total_prod < 1e-15:
            break

        # Identify top contributors
        top贡献 = np.argsort(products)[::-1][:max(3, n // 32)]
        # Redistribute: decrease top contributors, increase others
        decrease_amount = 0.01 * peak_val / max(total_prod, 1e-15)
        decrease_amount = min(decrease_amount, 0.05)

        a_new = a.copy()
        for idx_i in top贡献:
            i = int(idx[i])
            j = peak_k - i
            # Decrease the larger of a[i], a[j]
            if a[i] >= a[j]:
                reduction = min(a[i] * decrease_amount, a[i] * 0.1)
                a_new[i] -= reduction
                # Add to a random low-position
                low_positions = np.where(a_new < np.median(a_new))[0]
                if len(low_positions) > 0:
                    target = rng.choice(low_positions)
                    a_new[target] += reduction
            else:
                reduction = min(a[j] * decrease_amount, a[j] * 0.1)
                a_new[j] -= reduction
                low_positions = np.where(a_new < np.median(a_new))[0]
                if len(low_positions) > 0:
                    target = rng.choice(low_positions)
                    a_new[target] += reduction

        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            a = a_new  # Still accept for exploration

    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach combining:
1. Smooth optimization via log-sum-exp on softmax-parameterized variables

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def logsumexp_fn(v):
    """Numerically stable log-sum-exp."""
    v_max = v.max()
    return v_max + np.log(np.exp(v - v_max).sum())
