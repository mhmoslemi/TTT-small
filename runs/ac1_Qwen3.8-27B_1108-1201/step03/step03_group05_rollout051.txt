def gs_iteration(a, n_overall=1, n_inner=10):
    """
    One outer iteration of the GS algorithm.
    a: current sequence (non-negative, sum=1), length n.
    n_overall: over-relaxation parameter (0 < n_overall <= 1, 1 = standard GS).
    n_inner: number of inner frequency projections.
    """
    n = len(a)
    for _ in range(n_inner):
        # Frequency domain: flatten magnitude
        A = np.fft.fft(a)
        mag = np.abs(A)
        mag[mag < 1e-15] = 1e-15  # avoid division by zero
        # Target: flat magnitude

        a_new = np.fft.ifft(A_new).real
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
    
    # Over-relaxation
    return n_overall * a_new + (1 - n_overall) * a

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Spectral (Gerchberg-Saxton) iteration for flat power spectrum
2. Smoothed gradient descent with Nesterov momentum
3. LP-based direction finding
4. Peak shaving coordinate descent
5. Simulated annealing for diversity
6. Symmetric constraint exploitation

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gs_optimize(a0, time_limit, n_inner=5, blend=0.5, seed=0):
    """
    Gerchberg-Saxton style optimization for flat power spectrum.
    Input: a0 on simplex.
    Returns (best_a, best_max_conv).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    stagnant_threshold = 500

    while time.monotonic() - t0 < time_limit:
        # Frequency domain step: flatten magnitude
        A = np.fft.fft(a)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-15)
        
        # Compute target: blend between flat and current
        mean_mag = np.mean(mag)
        target_mag = blend * mean_mag + (1 - blend) * mag
        
        # Update phase
        A_new = target_mag * (A / mag_safe)
        
        # Time domain: enforce non-negativity
        a_new = np.fft.ifft(A_new).real
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Multiple inner iterations
        for _ in range(n_inner - 1):
            A = np.fft.fft(a_new)
            mag = np.abs(A)
            mag_safe = np.maximum(mag, 1e-15)
            mean_mag = np.mean(mag)
            target_mag = blend * mean_mag + (1 - blend) * mag
            A_new = target_mag * (A / mag_safe)
            a_new = np.fft.ifft(A_new).real
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
        
        a = a_new
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # If stagnant, perturb
        if no_improve >= stagnant_threshold:
            # Try different blend
            blend = 0.3 + 0.4 * rng.random()
            # Perturb
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            no_improve = 0
            stagnant_threshold = int(stagnant_threshold * 1.3)

    return best_a, best_obj


def smoothed_gradient(a, conv, beta):
    """Gradient of logsumexp(beta*conv)/beta w.r.t. a."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    w = conv * p

    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)

    return 2.0 * cross[:n]


def optimize_smoothed(a0, time_limit, beta_init=30.0, beta_end=200.0,
                      lr_init=0.015, momentum=0.92, perturb_every=400,
                      perturb_scale=0.003, seed=0):
    """Nesterov accelerated projected gradient descent on smoothed objective."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac

        y = a + momentum * v
        conv_y = fast_convolve(y)
        grad = smoothed_gradient(y, conv_y, beta)

        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= perturb_every:
            noise_scale = perturb_scale * (1.0 + 0.5 * np.sin(total_iters * 0.01))
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.25)
            lr = max(0.0005, lr * 0.65)

        lr = max(0.0003, lr_init * (1.0 - 0.7 * frac))

    return best_a, best_obj


def lp_direction(a, M, n):
    """Solve LP: max sum(b_j) s.t. b_j >= 0, (a*b)[k] <= M for all k."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None

    return b / b_sum

def peak_shaving(a0, time_limit, seed=0):
    """Targeted peak reduction via coordinate descent."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iters = 0
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        max_val = conv_cur.max()

        # Find top peaks
        top_k = min(10, len(conv_cur))
        peak_indices = np.argpartition(conv_cur, -top_k)[-top_k:]
        peak_indices = peak_indices[conv_cur[peak_indices] > max_val * 0.9]

        if len(peak_indices) == 0:
            break

        improved = False
        for k in peak_indices:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)

            contributions = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
            if len(contributions) == 0:
                continue

            # Find top contributing pairs
            top_contribs = np.argsort(contributions)[::-1][:3]
            for ci in top_contribs:
                i_max = i_start + ci
                
                # Try reducing a[i_max] and adding elsewhere
                for _ in range(5):
                    j = rng.integers(0, n)
                    if j == i_max or abs(j - i_max) < 2:
                        continue

                    for frac in [0.03, 0.05, 0.08, 0.1]:
                        amt = a[i_max] * frac
                        if amt < 1e-8:
                            break
                        a_try = a.copy()
                        a_try[i_max] -= amt
                        a_try[j] += amt
                        conv_try = fast_convolve(a_try)
                        if conv_try.max() < best_obj - 1e-10:
                            best_obj = conv_try.max()
                            best_a = a_try.copy()
                            a = a_try.copy()
                            improved = True
                            break

                if improved:
                    break
            if improved:
                break

        if not improved:
            stuck_count += 1
            # Random pair adjustment
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                for frac in [0.01, 0.02, 0.03, 0.05]:
                    amt = a[i] * frac
                    if amt < 1e-8:
                        break
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < max_val * 0.9999:
                        a = a_try.copy()
                        if conv_try.max() < best_obj:
                            best_obj = conv_try.max()
                            best_a = a_try.copy()
                        break
        
        iters += 1
        if stuck_count > 2000:
            break

    return best_a, best_obj


def simulated_annealing(a0, time_limit, seed=0):
    """
    Simulated annealing with adaptive perturbation size.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    cur_obj = best_obj

    t0 = time.monotonic()
    T = 0.01  # initial temperature
    T_min = 1e-6
    cool_rate = 0.9995

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        T_cur = max(T_min, T * cool_rate ** int(elapsed * 10))

        # Generate perturbation
        pert_scale = 0.02 * (1 + frac)  # increase perturbation size over time
        n_perturb = max(1, int(n * 0.05 * rng.random() + 1))
        indices = rng.choice(n, size=n_perturb, replace=False)
        deltas = rng.standard_normal(n_perturb) * pert_scale
        
        a_try = a.copy()
        a_try[indices] += deltas
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        conv_try = fast_convolve(a_try)
        obj_try = conv_try.max()

        delta = obj_try - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T_cur, 1e-10)):
            a = a_try
            cur_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a.copy()

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()

    tri = 1.0 / (1.0 + 0.5 * ((x - n / 2) / (n / 4)) ** 2)
    tri /= tri.sum()
    inits['bell'] = tri

    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g

    # Bimodal - concentrated at edges (key for our problem!)
    for edge_frac in [0.1, 0.15, 0.2, 0.3]:
        bi = np.zeros(n)
        edge = max(2, int(n * edge_frac))
        bi[:edge] = 1.0
        bi[-edge:] = 1.0
        bi /= bi.sum()
        inits[f'bimodal_{int(edge_frac*100)}'] = bi

    # Anti-triangular: high at edges, low in middle
    anti_tri = np.ones(n)
    anti_tri[n//4:3*n//4] = 0.3
    anti_tri /= anti_tri.sum()
    inits['anti_tri'] = anti_tri

    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w

    g_off = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2)
    g_off /= g_off.sum()
    inits['gauss_off'] = g_off

    inits['rand_dirichlet'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand_dirichlet2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand_dirichlet3'] = rng.dirichlet(np.ones(n) * 3.0)

    # Edge-concentrated Gaussian
    g_edge = np.exp(-0.5 * ((x / (0.1 * n)) ** 2)) + np.exp(-0.5 * (((x - n) / (0.1 * n)) ** 2))
    g_edge /= g_edge.sum()
    inits['edge_gauss'] = g_edge

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def symmetric_fold(a):
    """Fold a sequence to be symmetric: a[i] = (a[i] + a[n-1-i])/2, then renormalize."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym
