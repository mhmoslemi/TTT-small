"""
Hybrid gradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast projected subgradient with momentum for bulk optimization,
LP-based Frank-Wolfe for precise polishing, and multi-start with diverse
initializations across multiple sequence lengths.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Binary search for tau
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv[-1] - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_multi_peak(a, conv, top_k=10, temperature=0.05):
    """Compute weighted subgradient using top-k peaks of convolution."""
    n = len(a)
    conv_len = len(conv)
    
    # Select top-k peaks
    if top_k > conv_len:
        top_k = conv_len
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peak_indices]
    
    # Softmax weights
    peak_vals_scaled = peak_vals / max(temperature * peak_vals.max(), 1e-15)
    peak_vals_scaled -= peak_vals_scaled.max()
    weights = np.exp(peak_vals_scaled)
    weights /= weights.sum()
    
    # Compute subgradient
    grad = np.zeros(n)
    for i, k in enumerate(peak_indices):
        if weights[i] < 1e-12:
            continue
        k = int(k)
        # grad[j] += 2 * w * a[k-j] for valid j
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += 2.0 * weights[i] * a[k - idx]
    
    return grad

def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_step(a, M):
    """LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns normalized b."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 10.0})
    if res.success:
        b = np.maximum(res.x, 0.0)
        sb = b.sum()
        if sb > 1e-15:
            return b / sb
    return None


def fpw_optimize(a0, time_limit, verbose=False, verbose_every=50):
    """Frank-Wolfe with LP oracle."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit:
        b = lp_step(a, M)
        if b is None:
            break
        
        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            # Line search
            best_t = 0.0
            best_obj_local = 2.0 * n * fast_convolve(a).max()
            for t in [0.1, 0.25, 0.5, 0.75, 0.9]:
                candidate = (1 - t) * a + t * b
                obj = 2.0 * n * fast_convolve(candidate).max()
                if obj < best_obj_local:
                    best_obj_local = obj
                    best_t = t
            if best_t > 0:
                a = (1 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        if no_improve >= max_stall:
            rng_local = np.random.default_rng(it)
            noise = rng_local.standard_normal(n) * 0.002
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
        
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW it={it}, obj={best_obj:.6f}, M={M:.4e}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r
    
    # Triangle (peak at center)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri
    
    # V-shape (valley at center) - anti-peak
    vshape = np.maximum(np.minimum(x, n - 1 - x), 0.1)
    vshape = 1.0 / vshape
    vshape /= vshape.sum()
    inits['vshape'] = vshape
    
    # Edge concentrated
    edge_w = max(1, n // 6)
    ec = np.zeros(n)
    ec[:edge_w] = 1.0
    ec[-edge_w:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Blackman-like window (good autocorrelation)
    bl = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    bl = np.maximum(bl, 0)
    if bl.sum() > 1e-10:
        bl /= bl.sum()
        inits['blackman'] = bl
    
    # Random
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)
    
    # Sparse random
    sparse = rng.random(n)
    sparse[sparse < 0.3] = 0.0
    if sparse.sum() > 1e-10:
        sparse /= sparse.sum()
        inits['sparse'] = sparse
    
    # Alternating pattern
    alt = np.ones(n)
    alt[::2] = 2.0
    alt /= alt.sum()
    inits['alternating'] = alt
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        if len(pp) == n:
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp_interp = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(pp)),
                                  pp)
            ps = pp_interp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp_interp / ps
    
    return inits

"""
Hybrid gradient + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines fast projected subgradient (with momentum) for bulk optimization,
LP-based Frank-Wolfe for precise polishing, Fourier-inspired initializations,
and perturbation restarts across multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT, returns full-length result."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1. Returns (obj, conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1)
    cond = v_sorted * idx > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_multi_peak(a, conv, top_k=10, temperature=0.05):
    """Compute weighted subgradient using top-k peaks of convolution."""
    n = len(a)
    conv_len = len(conv)
    k = min(top_k, conv_len)
    peak_indices = np.argpartition(conv, -k)[-k:]
    peak_vals = conv[peak_indices]
    
    scale = max(temperature * peak_vals.max(), 1e-15)
    peak_vals_scaled = peak_vals / scale
    peak_vals_scaled -= peak_vals_scaled.max()
    weights = np.exp(peak_vals_scaled)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for i in range(len(peak_indices)):
        if weights[i] < 1e-12:
            continue
        k_idx = int(peak_indices[i])
        j_start = max(0, k_idx - n + 1)
        j_end = min(n - 1, k_idx)
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += 2.0 * weights[i] * a[k_idx - idx]
    
    return grad


def gradient_optimize(a0, time_limit, lr_init=0.02, momentum=0.9,
                      top_k=10, temperature=0.05, verbose=False, verbose_every=5000):
    """Optimize with projected subgradient + momentum and adaptive restarts."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v_mom = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 300
    stall_threshold = 1e-10
    
    while time.monotonic() - t0 < time_limit:
        grad = subgradient_multi_peak(a, conv, top_k=top_k, temperature=temperature)
        
        v_mom = momentum * v_mom - lr * grad
        a_new = project_simplex(a + v_mom)
        a = a_new
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - stall_threshold:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay learning rate
        lr *= 0.99995
        if lr < 1e-6:
            lr = 1e-6
        
        # Stall handling: restart with noise
        if no_improve >= max_stall:
            rng_local = np.random.default_rng(12345 + it)
            noise_scale = 0.001 * (1.0 + 0.5 * (no_improve // max_stall))
            noise = rng_local.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            v_mom = np.zeros(n)
            lr = lr_init * 0.3
            no_improve = 0
            max_stall = int(max_stall * 1.4)
        
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"        Grad it={it}, obj={best_obj:.6f}, lr={lr:.2e}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def build_conv_matrix_sparse(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]. Returns CSR."""
    n = len(a)
    nnz = n * n  # Each variable appears in exactly n equations
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    vals = np.empty(nnz, dtype=np.float64)
    
    pos = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        rows[pos:pos+count] = k
        cols[pos:pos+count] = np.arange(j_start, j_end + 1)
        vals[pos:pos+count] = a[k - np.arange(j_start, j_end + 1)]
        pos += count
    
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_frank_wolfe_step(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns normalized b or None."""
    n = len(a)
    A = build_conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': 15.0})
        if res.success:
            b = np.maximum(res.x, 0.0)
            sb = b.sum()
            if sb > 1e-15:
                return b / sb
    except Exception:
        pass
    return None


def fpw_lp_optimize(a0, time_limit, verbose=False, verbose_every=20):
    """Frank-Wolfe with exact LP oracle and line search."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 80
    
    while time.monotonic() - t0 < time_limit:
        b = lp_frank_wolfe_step(a, M)
        if b is None:
            break
        
        if np.allclose(a, b, atol=1e-12):
            no_improve += 1
        else:
            # Line search with golden-section-like approach
            best_t = 0.0
            best_obj_ls = 2.0 * n * fast_convolve(a).max()
            for t in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9, 0.98]:
                candidate = (1.0 - t) * a + t * b
                obj = 2.0 * n * fast_convolve(candidate).max()
                if obj < best_obj_ls:
                    best_obj_ls = obj
                    best_t = t
            
            if best_t > 0.01:
                a = (1.0 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        # Stall handling

            rng_local = np.random.default_rng(99999 + it)
            noise = rng_local.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
        
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"        LP it={it}, obj={best_obj:.6f}, M={M:.4e}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def make_fourier_init(n, rng):
    """Create initialization from a flat-spectrum (random phase) signal."""
    N = 1 << (2 * n + 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, N)
    spec = np.exp(1j * phases)
    spec[0] = n  # DC component to ensure positive sum
    a = np.fft.irfft(spec, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def make_optimal_binary_init(n, rng):
    """Create initialization inspired by optimal binary autocorrelation sequences."""
    # Use a m-sequence or LFSR-based pattern, then rectify
    # Simple approach: use a maximal-length sequence pattern
    pattern = np.zeros(n)
    period = (1 << min(20, int(np.log2(n)))) - 1
    if period > 0 and period < n:
        # Generate m-sequence
        m = int(np.log2(period + 1))
        state = 1
        taps = [m - 1, m - 2] if m >= 2 else [m - 1]
        for i in range(n):
            new_bit = 1
            for t in range(len(taps)):
                new_bit ^= (state >> taps[t]) & 1
            pattern[i] = (state >> (m - 1)) & 1
            state = ((state << 1) | new_bit) & period
    else:
        # Fallback: alternating pattern
        pattern = np.tile([1, 0], n // 2 + 1)[:n]
    
    a = np.maximum(pattern, 0).astype(np.float64)
    # Smooth it a bit
    if n > 10:
        kernel = np.ones(3) / 3.0
        a = np.convolve(a, kernel, mode='same')
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()
    
    # V-shape (valley at center - anti-peak)
    vshape = np.maximum(np.minimum(x, n - 1 - x) + 1.0, 1.0)
    vshape = 1.0 / vshape
    vshape /= vshape.sum()
    inits['vshape'] = vshape.copy()
    
    # Edge concentrated
    edge_w = max(2, n // 8)
    ec = np.zeros(n)
    ec[:edge_w] = 1.0
    ec[-edge_w:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()
    
    # Blackman window
    bl = 0.42 - 0.5 * np.cos(2 * np.pi * x / max(n - 1, 1)) + 0.08 * np.cos(4 * np.pi * x / max(n - 1, 1))
    bl = np.maximum(bl, 0.01)
    bl /= bl.sum()
    inits['blackman'] = bl.copy()
    
    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri.copy()
    
    # Fourier-inspired
    inits['fourier'] = make_fourier_init(n, rng)
    
    # Binary-sequence inspired
    inits['binary'] = make_optimal_binary_init(n, rng)
    
    # Random Dirichlet
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)
    
    # Sparse (30% zeros)
    sparse = rng.random(n)
    sparse[sparse < 0.3] = 0.0
    if sparse.sum() > 1e-10:
        sparse /= sparse.sum()
        inits['sparse'] = sparse.copy()
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 3.0
    alt /= alt.sum()
    inits['alternating'] = alt.copy()
    
    # Plateau: high in middle third
    plateau = np.zeros(n)
    start = n // 3
    end = 2 * n // 3
    plateau[start:end] = 1.0
    plateau /= plateau.sum()
    inits['plateau'] = plateau.copy()
    
    # Parent
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        if len(pp) == n:
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp_interp = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(pp)), pp)
            ps = pp_interp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp_interp / ps
    
    return inits

        obj2_final = 2.0 * n * fast_convolve(a2).max()

            # Interpolate to new length
            interp = np.interp(np.linspace(0, 1, t_n),
                               np.linspace(0, 1, n_best),
                               best_sequence)
            interp = project_simplex(interp)
            
            a_opt, obj_opt = gradient_optimize(interp, time_limit=t_alloc,
                                               lr_init=0.01, momentum=0.9,
                                               top_k=10, temperature=0.03)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                marker = " *" if score < best_score else ""
                print(f"  n={t_n}: score={score:.6f}{marker}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(384) / 384
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_sequence)}")
    print(f"{'='*50}")
    
    return [float(x) for x in best_sequence]
