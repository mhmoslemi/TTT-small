import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import fft, ifft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]

def smooth_max_and_weights(x, beta):
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w

def lp_refine_step(a, M, method='highs'):
    """
    Solve LP: maximize sum(g) subject to conv(a,g)_k <= M, g >= 0.
    Returns the optimal g.
    """
    n = len(a)
    # conv(a,g)_k = sum_{i} a[i] * g[k-i] for valid i, k-i
    # For k = 0: a[0]*g[0]
    # For k = 1: a[0]*g[1] + a[1]*g[0]
    # ...
    # The A matrix has shape (2n-1, n)
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    #            = a[k-j] if 0 <= k-j < n, else 0
    # Wait, conv(a,g)_k = sum_i a[i]*g[k-i]
    # For a fixed k, the terms are a[i]*g[k-i] for i where both i and k-i are in [0, n-1]
    # So i ranges from max(0, k-n+1) to min(k, n-1)
    # And g[k-i] is the variable, with index k-i
    # So A[k, j] where j = k-i, i.e., i = k-j
    # A[k, j] = a[k-j] if 0 <= k-j < n and 0 <= j < n, else 0
    
    # Build A matrix (sparse would be better but for n=256, dense is fine)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i  # index of g
            A[k, j] = a[i]
    
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method=method)
    
    if res.success:
        return res.x
    return None

def lp_iteration(a, n_lp_iters=50, blend_factor=0.3):
    """Iteratively apply LP refinement."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    for it in range(n_lp_iters):
        conv = fast_convolve(a)
        M = conv.max()
        
        g = lp_refine_step(a, M)
        if g is None or g.sum() < 1e-15:
            break
        
        g = g / g.sum()
        
        # Blend
        t = blend_factor
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        obj_new = 2.0 * n * fast_convolve(a_new).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

(Sinkhorn-like alternating projection) with multiple starting points,
adaptive scheduling, and smooth L-BFGS-B polishing.

import numpy as np

from scipy.fft import fft, ifft


def fast_convolve(a):

    n = len(a)

    A = fft(a, fft_size)

    """Build matrix A such that A @ g = conv(a, g) (linear)."""
    n = len(a)

    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)

    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -100, 100)

    Returns (g_opt, M_opt).

    c = np.zeros(n + 1)
    c[-1] = 1.0

    Each step finds g minimizing max(conv(a, g)) s.t. sum(g)=1, g>=0,

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    #
    # So: grad_obj[j] = 2*n * sum_k w[k] * 2*a[k-j] * 1[0<=k-j<n]
    # = 4*n * sum_k w[k] * a[k-j] * 1[0<=k-j<n]
    # = 4*n * (conv(w, a_rev))[j]  ... hmm let me think about this differently.
    #
    # sum_k w[k]*a[k-j] for valid k-j: this is a cross-correlation.
    # Let's define: h[j] = sum_k w[k]*a[k-j] where 0<=k-j<n
    # = sum_{m=0}^{n-1} w[m+j]*a[m] where m = k-j, so k = m+j
    # = sum_{m=0}^{n-1} w[m+j]*a[m]
    # This is the cross-correlation of w and a, evaluated at lag j.
    #
    # Using FFT: this is real(ifft(fft(w, N) * conj(fft(a, N))))[:n] ... 
    # Actually, cross-correlation: (w ⋆ a)[j] = sum_m w[m+j]*a[m] = sum_m w[m]*a[m-j]
    # = real(ifft(conj(fft(w)) * fft(a))) ... let me just compute it directly with FFT.
    
    N = 1 << (2 * n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    # h[j] = sum_k w[k]*a[k-j] for valid indices = cross-correlation
    # Using: ifft(conj(W) * A)[j] = sum_k w[k]*a[k-j] (circular, but N >= 2n-1 so linear)
    h = np.real(ifft(np.conj(W) * A, N))[:n]
    grad = 4.0 * n * h
    
    return obj, grad


def smooth_optimize(a0, beta=20.0, maxiter=300, deadline=None):
    """Optimize using smooth approximation with simple gradient descent + projection."""
    n = len(a0)
    a = np.maximum(a0, 1e-30).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    lr = 0.1
    for it in range(maxiter):
        if deadline is not None and time.monotonic() > deadline:
            break

        obj, grad = smooth_obj_jac(a, beta)

        # Projected gradient step
        a_new = project_simplex(a - lr * grad)
        obj_new = 2.0 * n * fast_convolve(a_new).max()

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        lr *= 0.999

        if best_obj < 1.005:
            break

    return best_a, best_obj


def smooth_optimize_lbfgs(a0, beta=20.0, maxiter=200, deadline=None):
    """Optimize using L-BFGS-B with log-parametrization."""
    n = len(a0)

    def _obj(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        if deadline is not None and time.monotonic() > deadline:
            return 1e10
        return 2.0 * n * sm

    def _jac(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)

        N = 1 << (2 * n - 1 + 63).bit_length()
        W = fft(w, N)
        A = fft(a, N)
        h = np.real(ifft(np.conj(W) * A, N))[:n]
        grad_a = 4.0 * n * h

        # Jacobian of reparametrization: a = exp(x - max(x)) / sum(exp(x - max(x)))
        # da_i/dx_j = a_i*(delta_ij - a_j)
        jac = grad_a[:, None] * a[None, :] - np.outer(grad_a * a, a)

        if deadline is not None and time.monotonic() > deadline:
            return np.zeros(n)
        return jac

    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)

    result = minimize(_obj, x0, jac=_jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 30})

    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum, clipped to non-negative."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k <= fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Geometric progression (inspired by n=2 optimal: 2:1 ratio)
    ratio = 1.1
    geo = ratio ** (np.arange(n) - n // 2)
    geo = np.maximum(geo, 0)
    geo /= geo.sum()
    inits['geo'] = geo

    # Power law
    for p in [0.5, 1.0, 2.0]:
        pw = (x + 1) ** p
        pw /= pw.sum()
        inits['power_%s' % str(p)] = pw

    # Bimodal (edges weighted more)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Flat spectrum
    inits['flat_spec'] = flat_spectrum_init(n, rng)

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                pp_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), pp)
                pp_interp = np.maximum(pp_interp, 0)
                ps2 = pp_interp.sum()
                if ps2 > 1e-10:
                    inits['parent_interp'] = pp_interp / ps2

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining LP Sinkhorn iteration,
    smooth optimization, and multi-start to minimize 2n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent available: length=%d, sum=%.2f" % (len(p), p.sum()))
    except Exception:
        pass

    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d (t=%.0fs)" % (score, len(a), time.monotonic() - t_start))
        return score if np.isfinite(score) else float('inf')

    # ============ Phase 1: Quick LP scan over lengths ============
    print("=== Phase 1: LP Sinkhorn Scan ===")
    lengths = [128, 192, 256, 320, 384, 512]
    phase1_deadline = t_start + 400.0

    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_interp', 'flat_spec', 'uniform', 'geo', 'bimodal', 'gauss', 'random']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        time_per_init = max(10.0, 350.0 / (len(lengths) * 3))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(time_per_init, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break

            a0 = inits[name]
            n_sinkhorn = max(50, int(t_alloc * 200))
            a_opt, obj = lp_sinkhorn_iteration(a0, n_iters=n_sinkhorn,
                                               deadline=phase1_deadline - 10)
            score = score_and_update(a_opt)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])
    print("Phase 1 done. Best: %.6f" % best_score)

    # ============ Phase 2: Deep LP optimization on best lengths ============
    print("=== Phase 2: Deep LP Optimization ===")

    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 256

    # Focus on lengths near the best
    target_lengths = sorted(set([l for l in [max(64, best_n // 2), best_n,
                                             min(512, best_n * 2)]
                                 if 64 <= l <= 512]))
    # Add a couple more for diversity
    for l in [192, 256, 384]:
        if l not in target_lengths and 64 <= l <= 512:
            target_lengths.append(l)
    target_lengths = sorted(set(target_lengths))[:4]

    phase2_deadline = deadline - 200.0

    for n in target_lengths:
        if time.monotonic() > phase2_deadline - 60:
            break

        inits = generate_initializations(n, parent, rng)

        # Add best sequence as initialization
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits['from_best'] = best_sequence / best_sequence.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_sequence)),
                               np.maximum(best_sequence, 0))
                pp = np.maximum(pp, 0)
                ps = pp.sum()
                if ps > 1e-10:
                    inits['from_best'] = pp / ps

        init_names = list(inits.keys())
        priority2 = ['from_best', 'parent', 'parent_interp', 'flat_spec', 'uniform', 'geo']
        init_names = [nm for nm in priority2 if nm in inits] + \
                     [nm for nm in init_names if nm not in priority2]
        init_names = init_names[:3]

        time_per_config = max(30.0, (phase2_deadline - time.monotonic()) / max(1, len(target_lengths) * 2))

        for name in init_names:
            if time.monotonic() > phase2_deadline - 30:
                break
            t_alloc = min(time_per_config, phase2_deadline - time.monotonic() - 20)
            if t_alloc < 10:
                break

            a0 = inits[name]

            # Long Sinkhorn iteration
            n_sinkhorn = max(200, int(t_alloc * 300))
            a_opt, obj = lp_sinkhorn_iteration(a0, n_iters=n_sinkhorn,
                                               deadline=phase2_deadline - 10)
            score = score_and_update(a_opt)

            # Follow up with blend iteration for fine-tuning
            if np.isfinite(score) and time.monotonic() < phase2_deadline - 10:
                t_blend = max(20, int((phase2_deadline - time.monotonic() - 10) * 100))
                a_blend, _ = lp_blend_iteration(a_opt, n_iters=t_blend, t=0.3,
                                                deadline=phase2_deadline - 10)
                score_and_update(a_blend)

    print("Phase 2 done. Best: %.6f" % best_score)

    # ============ Phase 3: Smooth optimization polishing ============
    print("=== Phase 3: Smooth Polishing ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        a_norm = best_sequence / best_sequence.sum()
        remaining = deadline - time.monotonic() - 30.0

        # Multi-beta smooth gradient descent
        for beta in [50.0, 20.0, 10.0, 5.0]:
            if remaining < 15:
                break
            t_single = min(120.0, remaining / 3)
            if t_single < 5:
                break
            n_iters = max(100, int(t_single * 50))
            a_opt, _ = smooth_optimize(a_norm, beta=beta, maxiter=n_iters,
                                       deadline=time.monotonic() + t_single)
            score = score_and_update(a_opt)
            a_norm = best_sequence / best_sequence.sum()
            remaining = deadline - time.monotonic() - 30.0

        # L-BFGS-B polish
        if remaining > 20:
            t_single = min(180.0, remaining / 2)
            maxiter = max(100, int(t_single * 10))
            a_opt, _ = smooth_optimize_lbfgs(a_norm, beta=10.0, maxiter=maxiter,
                                             deadline=time.monotonic() + t_single)
            score = score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 30.0

        # LP from the smooth-optimized point
        if remaining > 30:
            n_sinkhorn = max(100, int(remaining * 200))
            a_opt, _ = lp_sinkhorn_iteration(a_norm, n_iters=n_sinkhorn,
                                             deadline=deadline - 20)
            score = score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 20.0

        # Random perturbations + re-optimize
        if remaining > 40:
            for attempt in range(5):
                if remaining < 15:
                    break
                scale = 0.005 + 0.005 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(a_norm + noise)
                t_single = min(60.0, remaining / 2)
                n_sinkhorn = max(50, int(t_single * 200))
                a_opt, _ = lp_sinkhorn_iteration(a_pert, n_iters=n_sinkhorn,
                                                 deadline=time.monotonic() + t_single)
                score_and_update(a_opt)
                remaining = deadline - time.monotonic() - 20.0

        # Final LP + smooth combo
        if remaining > 20:
            a_norm = best_sequence / best_sequence.sum()
            n_sinkhorn = max(50, int(remaining * 150))
            a_opt, _ = lp_sinkhorn_iteration(a_norm, n_iters=n_sinkhorn,
                                             deadline=deadline - 15)
            score = score_and_update(a_opt)

    # ============ Phase 4: Try a different length if time permits ============
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 80:
            # Try doubling or halving the length
            alt_n = min(512, len(best_sequence) * 2) if len(best_sequence) < 512 else max(64, len(best_sequence) // 2)
            if alt_n != len(best_sequence):
                inits_alt = generate_initializations(alt_n, parent, rng)
                if best_sequence is not None:
                    pp = np.interp(np.linspace(0, 1, alt_n), np.linspace(0, 1, len(best_sequence)),
                                   np.maximum(best_sequence, 0))
                    pp = np.maximum(pp, 0)
                    ps = pp.sum()
                    if ps > 1e-10:
                        inits_alt['from_best'] = pp / ps
                a0 = inits_alt.get('from_best', inits_alt.get('flat_spec', inits_alt['uniform']))
                n_sinkhorn = max(100, int(remaining * 200))
                a_opt, _ = lp_sinkhorn_iteration(a0, n_iters=n_sinkhorn,
                                                 deadline=deadline - 10)
                score = score_and_update(a_opt)

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
