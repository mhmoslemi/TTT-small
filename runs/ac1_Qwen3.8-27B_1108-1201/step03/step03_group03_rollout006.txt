def optimize_subgrad(a0, time_limit, lr_init=0.01, momentum=0.9,
                     top_k=50, temp_scale=0.05):

    a = np.maximum(a0.astype(np.float64), 0.0)

    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Get top-k peaks
        if len(conv) > top_k:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break
        
        # Soft weighting of peaks
        scaled = peak_vals / (temp_scale * pmax + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])

            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_min_peak_lp(a, n, timeout=5.0):
    """
    Solve: min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0.
    Returns optimal g (sum=1) or None.
    """
    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[j]
        A_ub[k, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    b_ub = np.zeros(2 * n - 1)
    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': timeout})
        if result.success:
            g = np.maximum(result.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def solve_max_sum_lp(a, M, n, timeout=5.0):
    """
    Solve: max sum(g) s.t. g>=0, (a*g)_k <= M for all k.
    Returns optimal g (normalized to sum 1) or None.
    """
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': timeout})
        if result.success:
            g = np.maximum(result.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def optimize_lp(a_init, time_limit, blend_init=0.4, blend_min=0.005,
                halve_every=12, use_minpeak=True, lp_timeout=3.0):
    """LP-based optimization with adaptive blending."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        if use_minpeak:
            g = solve_min_peak_lp(a, n, timeout=lp_timeout)
        else:
            M = conv.max()
            g = solve_max_sum_lp(a, M, n, timeout=lp_timeout)

        if g is None:
            break

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        it_count += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend <= blend_min * 1.01:
                break

        if stall > halve_every * 5:
            seed = int(np.sum(np.abs(np.round(a[:20], 4)) * np.arange(1, 21))) % (2**31 - 1)
            rng_local = np.random.default_rng(seed + it_count)
            a = project_simplex(a + rng_local.standard_normal(n) * 0.03)
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
            stall = 0

    return best_a, best_obj


def optimize_subgrad(a0, time_limit, lr_init=0.012, momentum=0.9,
                     top_k=50, temp_scale=0.04):
    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if len(conv) > top_k:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        scaled = peak_vals / (temp_scale * pmax + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        it += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            rng_r = np.random.default_rng(it)
            a = project_simplex(a + rng_r.standard_normal(n) * 0.004)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.001, lr * 0.7)

        lr *= 0.9997

    return best_a, best_obj


def targeted_coord_descent(a_in, max_rounds=40, n_samples=200):
    """Targeted coordinate descent: move mass from hot to cold coordinates."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()

    for rnd in range(max_rounds):
        improved = False
        # Identify hot coordinates (contributing to peaks)
        peaks = np.argsort(conv)[::-1][:10]
        hot_set = set()
        for k in peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                if a[j] > a.mean() * 0.5:
                    hot_set.add(j)

        # Randomly sample coordinate pairs to try
        rng = np.random.default_rng(rnd * 7919 + 13)
        tried = set()
        for _ in range(n_samples):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or (i, j) in tried:
                continue
            tried.add((i, j))
            if a[i] < 1e-7:
                continue
            for frac in [0.15, 0.4, 0.7]:
                delta = a[i] * frac
                if delta < 1e-9:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-13:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
                    break

        if not improved:
            break

    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Edge-concentrated
    edge = np.zeros(n)
    e = max(1, n // 8)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    # Random (Dirichlet with low concentration)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # M-sequence inspired: use a pseudo-random binary pattern
    def m_seq_like(length, seed):
        sr = np.random.default_rng(seed)
        bits = sr.integers(0, 2, length)
        # Make it more "spread out" by adding small noise
        return bits.astype(np.float64) + sr.uniform(0.01, 0.05, length)

    inits['mseq1'] = m_seq_like(n, 123) / m_seq_like(n, 123).sum()
    inits['mseq2'] = m_seq_like(n, 456) / m_seq_like(n, 456).sum()

    # Periodic pattern
    per = np.tile([1.0, 0.3, 0.7, 0.3], n // 4 + 1)[:n]
    inits['periodic'] = per / per.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            elif len(parent) > n:
                p_res = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(parent)), p)
                inits['parent_ds'] = p_res / p_res.sum()
            else:
                p_res = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(parent)), p)
                inits['parent_up'] = p_res / p_res.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: min-peak LP + max-sum LP + subgradient + coordinate descent.
    Explores multiple lengths and initializations with annealing restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            if np.isfinite(par_score):
                best_score = par_score
                best_sequence = p.copy()
    except Exception:
        pass

    # Phase 1: Quick LP scan over lengths and initializations
    print("=== Phase 1: LP Scan ===")
    lengths = [384, 448, 512, 576, 640]
    if parent is not None and len(parent) not in lengths:
        lengths.append(len(parent))
    lengths = sorted(set(lengths))[:6]

    results = []
    phase1_time = min(350.0, (deadline - time.monotonic()) * 0.25)
    phase1_end = time.monotonic() + phase1_time

    all_inits_priority = ['parent', 'parent_ds', 'parent_up', 'uniform',
                          'bimodal', 'mseq1', 'random', 'gauss', 'edge',
                          'triangle', 'periodic', 'ramp', 'mseq2']

    for n in lengths:
        if time.monotonic() > phase1_end - 15:
            break
        inits = gen_inits(n, parent, rng)
        names = [nm for nm in all_inits_priority if nm in inits][:4]

        t_per = max(10.0, phase1_time / (len(lengths) * 4))

        for name in names:
            if time.monotonic() > phase1_end - 8:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 5)
            if t_alloc < 5:
                break

            a0 = inits[name]
            # Try both LP types
            a_opt, obj_opt = optimize_lp(a0, t_alloc * 0.6, blend_init=0.35,
                                         blend_min=0.01, use_minpeak=True,
                                         lp_timeout=2.0)
            a_opt2, obj2 = optimize_lp(a_opt, t_alloc * 0.4, blend_init=0.2,
                                       blend_min=0.005, use_minpeak=False,
                                       lp_timeout=2.0)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep LP optimization on top candidates
    print("=== Phase 2: Deep LP ===")
    top_list = []
    seen = set()
    for score, n, name, a_opt in results:
        key = n
        if key not in seen:
            top_list.append((n, a_opt))
            seen.add(key)
        if len(top_list) >= 4:
            break
    if best_sequence is not None and len(best_sequence) not in seen:
        top_list.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 350.0
    t_per = max(30.0, remaining / max(1, len(top_list) * 2))

    for n, a0 in top_list:
        if time.monotonic() > deadline - 320:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 280)
        if t_alloc < 10:
            break

        # Min-peak LP with varying blends
        for bi in [0.3, 0.15, 0.08]:
            if time.monotonic() > deadline - 280:
                break
            t_s = t_alloc / 3.0
            if t_s < 8:
                break
            a_opt, obj_opt = optimize_lp(a0, t_s, blend_init=bi,
                                         blend_min=0.003, use_minpeak=True,
                                         lp_timeout=3.0)
            # Follow up with max-sum LP
            a_opt2, obj2 = optimize_lp(a_opt, min(15.0, t_s * 0.3),
                                       blend_init=0.1, blend_min=0.003,
                                       use_minpeak=False, lp_timeout=3.0)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep n={n} bi={bi}: {score:.6f}")

    # Phase 3: Subgradient refinement
    print("=== Phase 3: Subgradient ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        t_sg = min(remaining * 0.4, 250.0)

        for lr_try in [0.008, 0.004, 0.002]:
            if time.monotonic() > deadline - 180:
                break
            t_s = t_sg / 3.0
            if t_s < 5:
                break
            a_opt, obj_opt = optimize_subgrad(best_sequence, t_s,
                                              lr_init=lr_try, momentum=0.92,
                                              top_k=60, temp_scale=0.03)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Subgrad lr={lr_try}: {score:.6f}")

    # Phase 4: Alternating LP + Subgrad cycles
    print("=== Phase 4: LP+SG Cycles ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 120.0
        n_cycles = 3
        t_cyc = remaining / n_cycles

        for cyc in range(n_cycles):
            if time.monotonic() > deadline - 100:
                break
            t_avail = min(t_cyc, deadline - time.monotonic() - 80)
            if t_avail < 10:
                break

            # LP step
            a_opt, _ = optimize_lp(best_sequence, t_avail * 0.5,
                                   blend_init=0.08, blend_min=0.002,
                                   use_minpeak=True, lp_timeout=3.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

            # Subgrad step
            a_opt2, _ = optimize_subgrad(best_sequence, t_avail * 0.4,
                                         lr_init=0.003, momentum=0.93,
                                         top_k=80, temp_scale=0.02)
            score2 = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_sequence = a_opt2.copy()

            print(f"  Cycle {cyc}: {best_score:.6f}")

    # Phase 5: Coordinate descent polish
    print("=== Phase 5: Coord Descent Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 40.0
        t_cd = min(remaining * 0.5, 120.0)

        if t_cd > 5:
            a_cd = targeted_coord_descent(best_sequence,
                                          max_rounds=int(t_cd / 2) + 10,
                                          n_samples=300)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD: {score_cd:.6f}")

    # Phase 6: Annealing restarts on best
    print("=== Phase 6: Annealing ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 30:
            n_b = len(best_sequence)
            perturbations = [0.005, 0.01, 0.02, 0.015]
            t_per_pert = remaining / (len(perturbations) + 1)

            for ps in perturbations:
                if time.monotonic() > deadline - 15:
                    break
                t_avail = min(t_per_pert, deadline - time.monotonic() - 12)
                if t_avail < 8:
                    break
                noise = rng.standard_normal(n_b) * ps
                a_p = project_simplex(best_sequence + noise)
                # Quick LP + subgrad
                a_opt, _ = optimize_lp(a_p, t_avail * 0.5,
                                       blend_init=0.12, blend_min=0.004,
                                       use_minpeak=True, lp_timeout=2.0)
                a_opt2, _ = optimize_subgrad(a_opt, min(5.0, t_avail * 0.3),
                                             lr_init=0.004, momentum=0.91,
                                             top_k=40, temp_scale=0.03)
                score = float(evaluate_sequence(a_opt2.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt2.copy()
                    print(f"  Anneal({ps}): {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
