def fast_convolve(a):

    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Edge (mass at edges)
    edge = np.zeros(n)
    e = max(1, n // 4)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Cosine (non-negative, wavy)
    cos = 1.0 + np.cos(2.0 * np.pi * x / n)
    cos = np.maximum(cos, 0.01)
    inits['cosine'] = cos / cos.sum()
    
    # Multi-frequency cosine
    multi = 1.0 + 0.5 * np.cos(2.0 * np.pi * x / n) + 0.3 * np.cos(4.0 * np.pi * x / n)
    multi = np.maximum(multi, 0.01)
    inits['multi_cos'] = multi / multi.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_resampled = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
                p_resampled = np.maximum(p_resampled, 0)
                ps2 = p_resampled.sum()
                if ps2 > 1e-10:
                    inits['parent_res'] = p_resampled / ps2
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast subgradient descent with momentum and multiplicative weights
- Simulated annealing for global exploration
- Efficient coordinate descent for local refinement
- Multi-length, multi-start search with adaptive time allocation

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, top_k=30):
    """Compute subgradient of max(conv(a,a)) w.r.t. a.
    Returns (subgradient, max_value, convolution).
    """
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    # Weight peaks by excess over median (focus on the worst peaks)
    peak_vals = conv[peaks]
    med = np.median(peak_vals)
    excess = peak_vals - med
    if excess.max() > 1e-15:
        w = excess / excess.sum()
    else:
        w = np.ones(len(peaks)) / len(peaks)
    
    sg = np.zeros(n)
    for idx, k in enumerate(peaks):
        k = int(k)
        wi = w[idx]
        # First term: sum over j of a[k-j] (valid when 0 <= k-j < n)
        j1_start = max(0, k - n + 1)
        j1_end = min(n - 1, k)
        if j1_start <= j1_end:
            sg[j1_start:j1_end + 1] += wi * a[k - j1_end:k - j1_start + 1][::-1]
        # Second term: sum over j of a[j-k] (valid when 0 <= j-k < n)
        j2_start = max(0, k)
        j2_end = min(n - 1, k + n - 1)
        if j2_start <= j2_end:
            sg[j2_start:j2_end + 1] += wi * a[j2_start - k:j2_end - k + 1]
    
    return sg, M, conv


def optimize_subgrad(a0, time_limit, lr_init=0.08, momentum=0.88,
                     top_k=30, cooling=0.9992, use_mw=False, mw_lr=0.05):
    """Subgradient descent with momentum (or multiplicative weights) on simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        sg, M, conv = compute_subgradient(a, top_k)
        
        if use_mw:
            # Multiplicative weights update
            a_new = a * np.exp(-mw_lr * sg)
            a_new = np.maximum(a_new, 1e-15)
            a_new /= a_new.sum()
        else:
            sg_norm = np.linalg.norm(sg)
            if sg_norm > 1e-15:
                sg = sg / sg_norm
            v = momentum * v - lr * sg
            a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            conv = conv_new
        else:
            no_improve += 1
            # Partial update to maintain exploration
            a = 0.7 * a + 0.3 * a_new
            a = project_simplex(a)
        
        if no_improve > 80:
            rng_local = np.random.default_rng((iters * 7919 + n) % 2**31)
            a = project_simplex(a + rng_local.standard_normal(n) * 0.015)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.01, lr * 0.6)
        
        lr *= cooling
        iters += 1
    
    return best_a, best_obj


def optimize_sa(a0, time_limit, top_k=20):
    """Simulated annealing with adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)

    total_count = 0

        s_new = a_new.sum()
        if s_new < 1e-15:

def coordinate_descent(a_in, time_limit=30.0):
    """Coordinate descent: move mass between pairs to reduce peak."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()

        return a
    a = a / s
    n = len(a)

    best_obj = 2.0 * n * conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    rounds = 0
    while time.monotonic() - t0 < time_limit and rounds < 200:
        improved = False
        active = np.where(a > 1e-5)[0]
        if len(active) == 0:
            break
        
        # Sample a subset of active positions for speed
        if len(active) > 50:
            active_sample = rng.choice(active, size=50, replace=False)
        else:
            active_sample = active
        
        for i in active_sample:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-6:
                continue
            
            # Try moving mass to random destinations
            n_try = min(15, n - 1)
            j_cands = rng.choice(n, size=n_try, replace=False)
            
            local_best_obj = best_obj
            local_best_j = -1
            local_best_frac = 0.0
            
            for j in j_cands:
                if j == i:
                    continue
                for frac in [0.03, 0.08]:
                    delta = a[i] * frac
                    if delta < 1e-9:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = np.maximum(a_test, 0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < local_best_obj - 1e-13:
                        local_best_obj = obj_test
                        local_best_j = int(j)
                        local_best_frac = frac
            
            if local_best_j >= 0:
                delta = a[i] * local_best_frac
                a[i] -= delta
                a[local_best_j] += delta
                a = np.maximum(a, 0)
                a /= a.sum()
                conv = fast_convolve(a)
                best_obj = 2.0 * n * conv.max()
                improved = True
        
        rounds += 1
        if not improved:
            break
    
    return a

    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    edge = np.zeros(n)
    e = max(1, n // 4)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Cosine-based (non-negative, smooth)
    for freq in [1, 2, 3]:
        c = 1.0 + 0.6 * np.cos(2.0 * np.pi * freq * x / n)
        c = np.maximum(c, 0.01)
        inits[f'cos{freq}'] = c / c.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    #_plateau: flat in middle, zero at edges
    plateau = np.ones(n)
    border = max(1, n // 8)
    fade = np.linspace(0, 1, border)
    plateau[:border] = fade
    plateau[-border:] = fade[::-1]
    inits['plateau'] = plateau / plateau.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_r = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(parent)), p)
                p_r = np.maximum(p_r, 0)
                ps2 = p_r.sum()
                if ps2 > 1e-10:
                    inits['parent_res'] = p_r / ps2
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: subgradient + SA + coordinate descent
    to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            best_sequence = p.copy()
            best_score = par_score
    except Exception:
        pass
    
    # Phase 1: Broad exploration
    print("=== Phase 1: Broad Exploration ===")
    lengths = [128, 192, 256, 384, 512, 768]
    
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.append(pn)
        lengths.sort()
    
    results = []
    phase1_time = min(400.0, (deadline - time.monotonic()) * 0.30)
    phase1_end = time.monotonic() + phase1_time
    
    for n in lengths:
        if time.monotonic() > phase1_end - 20:
            break
        inits = gen_inits(n, parent, rng)
        
        # Priority: parent first, then diverse structures
        prio = ['parent', 'parent_res', 'cos1', 'cos2', 'uniform',
                'plateau', 'triangle', 'bimodal', 'edge', 'random', 'cos3']
        names = [nm for nm in prio if nm in inits]
        
        # Limit to top 3-4 per length
        names = names[:4]
        t_per = max(12.0, phase1_time / (len(lengths) * 4))
        
        for name in names:
            if time.monotonic() > phase1_end - 8:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 5)
            if t_alloc < 8:
                break
            
            a0 = inits[name]
            a_opt, obj_opt = optimize_subgrad(a0, t_alloc,
                                               lr_init=0.06, momentum=0.87,
                                               top_k=30, cooling=0.999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")
    
    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")
    
    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    top_list = []
    seen = set()
    for score, n, name, a_opt in results:
        key = (n,)
        if key not in seen and len(top_list) < 5:
            top_list.append((n, a_opt, name))
            seen.add(key)
    
    if best_sequence is not None and len(best_sequence) > 0:
        bn = len(best_sequence)
        if bn not in [x[0] for x in top_list]:
            top_list.append((bn, best_sequence.copy(), 'best'))
    
    remaining = deadline - time.monotonic() - 250.0
    t_per = max(40.0, remaining / max(1, len(top_list)))
    
    for n, a0, name in top_list:
        if time.monotonic() > deadline - 230:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 200)
        if t_alloc < 20:
            break
        
        # Extended subgradient
        a_opt1, _ = optimize_subgrad(a0, t_alloc * 0.35,
                                      lr_init=0.04, momentum=0.91,
                                      top_k=50, cooling=0.9995)
        # SA for global moves
        a_opt2, _ = optimize_sa(a_opt1, t_alloc * 0.30)
        # Coordinate descent
        a_opt3 = coordinate_descent(a_opt2, time_limit=t_alloc * 0.25)
        
        score = float(evaluate_sequence(a_opt3.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt3.copy()
            print(f"  Deep n={n} {name}: {score:.6f}")
        
        # Also try multiplicative weights variant
        if time.monotonic() < deadline - 180:
            a_mw, _ = optimize_subgrad(a_opt1, t_alloc * 0.15,
                                        use_mw=True, mw_lr=0.03,
                                        lr_init=0.03, top_k=40)
            a_mw2 = coordinate_descent(a_mw, time_limit=t_alloc * 0.10)
            score_mw = float(evaluate_sequence(a_mw2.tolist()))
            if np.isfinite(score_mw) and score_mw < best_score:
                best_score = score_mw
                best_sequence = a_mw2.copy()
                print(f"  MW n={n} {name}: {score_mw:.6f}")
    
    # Phase 3: Iterative refinement
    print("=== Phase 3: Refinement ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 40.0
        n_best = len(best_sequence)
        
        for cycle in range(6):
            if time.monotonic() > deadline - 25:
                break
            t_cyc = min(remaining / (6 - cycle), 70.0)
            if t_cyc < 10:
                break
            
            # Subgradient with smaller LR
            a_opt, _ = optimize_subgrad(best_sequence, t_cyc * 0.35,
                                         lr_init=0.02, momentum=0.93,
                                         top_k=60, cooling=0.9997)
            # SA
            a_opt2, _ = optimize_sa(a_opt, t_cyc * 0.25)
            # CD
            a_opt3 = coordinate_descent(a_opt2, time_limit=t_cyc * 0.30)
            
            score = float(evaluate_sequence(a_opt3.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt3.copy()
                print(f"  Refine cycle {cycle}: {score:.6f}")
        
        # Random restarts
        if time.monotonic() < deadline - 20:
            for ps in [0.004, 0.008, 0.012]:
                if time.monotonic() > deadline - 12:
                    break
                for _ in range(4):
                    if time.monotonic() > deadline - 8:
                        break
                    noise = rng.standard_normal(n_best) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_opt, _ = optimize_subgrad(a_p, 4.0,
                                                 lr_init=0.03, momentum=0.9,
                                                 top_k=40, cooling=0.999)
                    a_opt2 = coordinate_descent(a_opt, time_limit=3.0)
                    score = float(evaluate_sequence(a_opt2.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt2.copy()
                        print(f"  Restart(ps={ps}): {score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
