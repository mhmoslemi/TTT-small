Hybrid LP + Subgradient + Simulated Annealing optimizer for minimizing

import numpy as np

from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def solve_lp(a, M, A_ub=None):
    n = len(a)
    if A_ub is None:
        A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0

def compute_obj(a, n):
    conv = fast_convolve(a)
    M = conv.max()
    return 2.0 * n * M, M, conv

def line_search(a, d, best_obj, n, max_evals=25):
    best_t = 0.0
    best_o = best_obj
    evals = 0
    candidates = [1.0, 0.8, 0.6, 0.45, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.02, 0.01]
    for t in candidates:
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        o, _, _ = compute_obj(a_try, n)
        evals += 1
        if o < best_o - 1e-14:
            best_o = o
            best_t = t
        if evals >= max_evals:
            break
    # Refine
    if 0.02 < best_t < 0.98:
        for _ in range(3):
            found = False
            for dt in [-0.1, 0.1]:
                t_new = best_t + dt
                if t_new <= 0 or t_new >= 1:
                    continue
                a_try = np.maximum(a + t_new * d, 0.0)
                s = a_try.sum()
                if s < 1e-15:

                a_try = a_try / s

def subgradient_polish(a, time_limit, lr=0.005, momentum=0.93, top_k=60, temp=0.0002):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    _, M, conv = compute_obj(a, n)
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    batch = 20
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks].copy()
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= ws
            
            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w[i] * 2.0 * a[k - idx]
            
            v = momentum * v - lr * sg
            a_new = np.maximum(a + v, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()
            
            _, M, conv = compute_obj(a, n)
            cur = 2.0 * n * M
            if cur < best_obj - 1e-13:
                best_obj = cur
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        lr *= 0.9985
        temp *= 1.003
        momentum = min(0.96, momentum + 0.0002)
        if no_improve >= 400:
            break
    
    return best_a, best_obj

def make_init(n, parent, rng, name):
    x = np.arange(n, dtype=np.float64)
    if name == 'uniform':
        return np.ones(n) / n
    elif name == 'random':
        return rng.dirichlet(np.ones(n) * 0.5)
    elif name == 'ramp':
        r = x + 1.0
        return r / r.sum()
    elif name == 'bimodal':
        b = np.zeros(n)
        e = max(1, n // 4)
        b[:e] = 1.0
        b[-e:] = 1.0
        return b / b.sum()
    elif name == 'gaussian':
        g = np.exp(-0.5 * ((x - 0.5*n) / (0.2*n))**2)
        return g / g.sum()
    elif name == 'sine':
        s = np.sin(np.pi * x / n) + 0.05
        return s / s.sum()

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M, A_ub=None):
    """Solve LP: maximize sum(g) s.t. a*g <= M (componentwise), g >= 0."""
    n = len(a)
    if A_ub is None:
        A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def compute_obj(a, n):
    """Compute objective and return (obj, M, conv)."""
    conv = fast_convolve(a)

    best_obj = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks].copy()
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * a[k - idx]

        v = momentum * v - lr * sg
        a_new = np.maximum(a + v, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        _, M, conv = compute_obj(a, n)
        cur = 2.0 * n * M
        if cur < best_obj - 1e-13:
            best_obj = cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        temp *= 1.002
        momentum = min(0.965, momentum + 0.00015)
        if no_improve >= 500:
            break

    return best_a, best_obj

    """Retrieve parent sequence from global if available."""
    try:

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
