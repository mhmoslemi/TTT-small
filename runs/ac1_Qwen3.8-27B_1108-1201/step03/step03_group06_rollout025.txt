"""
Hybrid optimization for minimum peak convolution sequences.
Uses LP direction finding, subgradient methods, CMA-ES on Fourier params,
and multi-scale search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(a):
    """Build constraint matrix for LP: (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub

def solve_lp_direction(a, M, top_k=None, conv=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M."""
    n = len(a)
    num_k = 2 * n - 1
    
    if top_k is not None and top_k < num_k and conv is not None:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        A_ub_full = build_lp_matrix(a)
        A_ub = A_ub_full[peaks]
        b_ub = np.full(len(peaks), M)
    else:
        A_ub = build_lp_matrix(a)
        b_ub = np.full(num_k, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def subgradient_step(a, eta):
    """Compute subgradient step for minimizing max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    k_star = np.argmax(conv)
    # grad[m] = 2*a[k_star - m] if 0 <= k_star - m < n, else 0
    grad = np.zeros(n)
    m_range = np.arange(n)
    valid = (k_star - m_range >= 0) & (k_star - m_range < n)
    grad[valid] = 2.0 * a[k_star - m_range[valid]]
    # Projected step
    return project_simplex(a - eta * grad), conv

        M = conv.max()

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        b = b / b_sum
        a_new = (1.0 - step) * a + step * b

def fourier_parametrize(c, n):
    """Convert Fourier coefficients to a non-negative sequence on simplex."""
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)  # DC term
    for j, cj in enumerate(c, start=1):
        a += cj * np.cos(2.0 * np.pi * j * (x - n // 2) / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def cmaes_fourier(n, m, time_limit, x0=None, seed=0):
    """CMA-ES on Fourier coefficients."""
    rng = np.random.default_rng(seed)
    dim = m
    if x0 is None:
        x0 = rng.normal(0, 0.1, dim)
    else:
        x0 = x0.copy()
    
    sigma = 0.1
    pop_size = max(4, int(4 + 3 * np.log(dim)))
    best_x = x0.copy()
    best_f = float('inf')
    
    # Mean and covariance
    mean = x0.copy()
    C = np.eye(dim)
    lambda_ = pop_size
    mu_ = lambda_ // 2
    
    t0 = time.monotonic()
    gen = 0
    
    while time.monotonic() - t0 < time_limit:
        # Generate population
        population = []
        for i in range(lambda_):
            z = rng.standard_normal(dim)
            # Simple: use diagonal C (identity) for simplicity
            x = mean + sigma * z
            population.append(x)

        fitnesses = []

            fitnesses.append(f)
        
        fitnesses = np.array(fitnesses)

        weights[0] = np.log(mu_ + 0.5) - np.log(mu_)

        weights = weights[::-1]

        new_mean = sum(weights[i] * population[order[i]] for i in range(mu_))
        
        # Update sigma (simple adaptation)
        sigma *= 0.999
        
        mean = new_mean

        if gen % 100 == 0:
            print(f"      CMA gen {gen}: best_f={best_f:.6f}, sigma={sigma:.4f}")
    
    a_best = fourier_parametrize(best_x, n)
    conv_best = fast_convolve(a_best)
