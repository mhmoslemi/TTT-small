import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    cv = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(cv, 0)  # numerical safety

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_max(conv, mu):
    """Log-sum-exp approximation of max."""
    m = conv.max()
    if m < 1e-30:
        return 0.0
    return m + mu * np.log(np.sum(np.exp((conv - m) / mu)))

def smoothed_max_grad(conv, mu):
    """Gradient of log-sum-exp w.r.t. conv (i.e., softmax weights)."""
    m = conv.max()
    if m < 1e-30:
        return np.ones(len(conv)) / len(conv)
    shifted = (conv - m) / mu
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w /= w.sum()
    return w

def lp_direction(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    from scipy.optimize import linprog
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. LP-based directional search with adaptive target M
2. Smoothed (log-sum-exp) gradient descent with momentum  
3. Multiplicative weights / exponential tilting
4. Local coordinate descent polishing
Explores multiple sequence lengths starting from the parent construction.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    cv = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(cv, 0.0)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(a, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp(a, M, n, lp_matrix=None):
    if lp_matrix is None:
        lp_matrix = build_lp_matrix(a, n)
    # But lp_matrix depends on a, so we need to rebuild each time
    A_ub = build_lp_matrix(a, n)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None

def compute_smoothed_gradient(a, conv, mu, n):
    """Compute gradient of mu*log(sum exp(conv[k]/mu)) w.r.t. a."""
    m = conv.max()
    if m < 1e-30:
        return np.zeros(n)
    shifted = (conv - m) / mu
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        wk = 2.0 * w[k]
        for j in range(j_start, j_end + 1):
            grad[j] += wk * a[k - j]
    return grad

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = initial_blend
    stall = 0
    
    while time.monotonic() - t0 < t_limit:

        # Try with slightly reduced M for more aggressive search

        g = solve_lp(a, M_target, n)

            if g is None:

        g_sum = g.sum()
        if g_sum < 1e-15:

        g = g / g_sum
        
        a_new = (1.0 - blend) * a + blend * g

def local_polish(a, t_limit, n):
    """Local coordinate descent polishing."""
    a = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = max(1, n // 300)
    
    while time.monotonic() - t0 < t_limit:
        improved = False
        for i in range(0, n, step):
            if time.monotonic() - t0 > t_limit - 5:
                break
            if a[i] < 1e-9:
                continue
            delta = a[i] * 0.02
            if delta < 1e-12:
                continue
            for j in range(0, n, step):
                if j == i:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-14:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
        if not improved:
            # Try smaller step
            step = max(1, step // 2)
            if step == 1:
                break
    
    return best_a, best_obj
