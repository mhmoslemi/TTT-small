    fft_size = 1 << (size-1).bit_length()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, p):
    """Compute gradient of smoothed objective.
    a: sequence (length n, sum=1)
    p: softmax probabilities over convolution peaks (length 2n-1)
    Returns gradient (length n).
    """
    n = len(a)
    # grad[i] = (2n/T) * (sum_m p[m+i]*a[m] + a[i]*sum_m p[m+i])
    # We'll return the inner part (without the 2n/T factor)
    
    # q[i] = sum_{m=0}^{n-1} p[m+i] * a[m]
    # s[i] = sum_{m=0}^{n-1} p[m+i]

    q = np.zeros(n)
    s = np.zeros(n)
    for i in range(n):
        ps = p[i:i+n]
        q[i] = np.dot(ps, a)
        s[i] = ps.sum()
    
    return q + a * s  # The 2n/T factor is applied externally

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""

# Precomputed index matrix for n=512 gradient computation
def _get_idx(n):
    return np.arange(n)[:, None] + np.arange(n)[None, :]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_inner_grad(a, p, idx):
    """Compute q[i] + a[i]*s[i] for the gradient of smoothed obj.
    q[i] = sum_m p[m+i]*a[m], s[i] = sum_m p[m+i]
    """
    n = len(a)
    M = p[idx]
    q = M @ a
    s = M.sum(axis=1)
    return q + a * s

def smooth_obj_and_grad(a, T, idx):
    """Compute smoothed objective, true objective, and gradient."""
    n = len(a)
    conv = fast_convolve(a)

    scaled = T * (conv - conv.max())
    max_scaled = scaled.max()
    exp_vals = np.exp(scaled - max_scaled)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp

    smooth_max = conv.max() + (max_scaled + np.log(sum_exp)) / T
    obj = 2.0 * n * smooth_max
    true_obj = 2.0 * n * conv.max()

    inner = compute_inner_grad(a, p, idx)
    grad = (2.0 * n / T) * inner

    return obj, true_obj, grad, conv

def fista_optimize(a0, idx, T_schedule, max_time, lr_init=0.02,
                   max_inertia=1.5, backtrack_factor=0.5):
    """FISTA (accelerated proximal gradient) with temperature schedule.
    T_schedule: list of (T, n_iters) tuples.
    """
    n = len(a0)
    a = a0.copy()
    a_prev = a.copy()
    t_fista = 1.0

    best_true_obj = float('inf')
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for T, n_iters in T_schedule:
        for _ in range(n_iters):
            if time.monotonic() - t0 > max_time:
                return best_a, best_true_obj

            # FISTA extrapolation
            mu = (t_fista - 1.0) / (t_fista + 1.0)
            a_extrap = a + mu * (a - a_prev)
            a_extrap = project_simplex(a_extrap)

            # Gradient step
            obj, true_obj, grad, conv = smooth_obj_and_grad(a_extrap, T, idx)

            a_new = project_simplex(a_extrap - lr * grad)

            # Check improvement
            if true_obj < best_true_obj - 1e-12:
                best_true_obj = true_obj
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Backtracking: if no improvement for a while, reduce lr
            if no_improve > 200:
                lr *= backtrack_factor
                no_improve = 0
            elif no_improve > 100 and lr > lr_init * 0.1:
                lr *= 0.8

            a_prev = a
            a = a_new
            t_fista += 1.0

            if t_fista > 50:
                t_fista = 1.0  # Reset momentum periodically

    return best_a, best_true_obj

def subgradient_peak_optimize(a0, idx, max_time, lr_init=0.005, momentum=0.9):
    """Optimize using exact subgradient of the max function.
    Focuses on the current peak position(s).
    """
    n = len(a0)
    a = project_simplex(a0.copy())
    v = np.zeros(n)
    best_true_obj = float('inf')
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init

    while time.monotonic() - t0 < max_time:
        conv = fast_convolve(a)
        true_obj = 2.0 * n * conv.max()

        if true_obj < best_true_obj - 1e-12:
            best_true_obj = true_obj
            best_a = a.copy()

        # Find peak positions (within tolerance)
        peak_val = conv.max()
        peak_mask = conv >= peak_val - 1e-10
        peak_indices = np.where(peak_mask)[0]
        n_peaks = len(peak_indices)

        # Compute subgradient: average over peak positions
        sg = np.zeros(n)
        for k in peak_indices:
            # d conv[k] / d a[i] = (a[k-i] + a[i]) * 1[0<=k-i<n]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                sg[i] += (a[k - i] + a[i])
        sg /= n_peaks

        # Momentum update
        v = momentum * v - lr * sg
        a = project_simplex(a + v)

        lr *= 0.9999

        if lr < 1e-6:
            break

    return best_a, best_true_obj

            j = rng.integers(0, n)

            continue

        # Try several transfer fractions
        for frac in [0.05, 0.1, 0.2, 0.02]:
            delta = a[i] * frac
            if delta < 1e-12:
                continue
            a[i] -= delta
            a[j] += delta

def block_rebalance(a0, max_time, block_size=8, max_iters=50000):
    """Try redistributing mass within small blocks to reduce peak."""
    n = len(a0)
    a = project_simplex(a0.copy())
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() - t0 > max_time:
            break

        peak_idx = np.argmax(conv)
        # Find blocks that contribute to the peak
        contrib_idx = []
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        for i in range(lo, hi + 1):
            contrib_idx.append(i)
            contrib_idx.append(peak_idx - i)
        contrib_idx = list(set(contrib_idx))

        if not contrib_idx:
            continue

        # Pick a random index near the peak contributors
        i = contrib_idx[rng.integers(0, len(contrib_idx))]
        block_start = max(0, i - block_size // 2)
        block_end = min(n, block_start + block_size)
        if block_end - block_start < 2:
            continue

        # Try random redistribution within the block
        block = a[block_start:block_end].copy()
        block_sum = block.sum()
        if block_sum < 1e-15:
            continue

        # Random Dirichlet perturbation
        new_block = rng.dirichlet(np.ones(block_end - block_start) * 2.0) * block_sum
        a[block_start:block_end] = new_block

        conv_new = fast_convolve(a)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
            conv = conv_new
        else:
            a[block_start:block_end] = block

    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            inits[f'cos_{freq}'] = a / s

    # Two-peak (W shape)
    for spread in [0.2, 0.3, 0.4]:
        a = (np.exp(-((x - n*0.25)/(n*spread))**2) +
             np.exp(-((x - n*0.75)/(n*spread))**2))
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            inits[f'twopeak_{spread}'] = a / s

    # Edge-concentrated
    a = np.exp(-x / (n * 0.3)) + np.exp(-(n - 1 - x) / (n * 0.3))
    a = np.maximum(a, 0)
    inits['edges'] = a / a.sum()

    # Center-concentrated (Gaussian)
    g = np.exp(-0.5 * ((x - n/2) / (n * 0.15)) ** 2)
    inits['center'] = g / g.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Alternating pattern
    a = np.ones(n)
    a[::2] = 1.8
    a[1::2] = 0.8
    inits['alternating'] = a / a.sum()

    return inits
