```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of multiplicative
subgradient descent (mirror descent on the simplex) and LP-based direction
finding with dual-variable-guided peak weighting. Explores multiple sequence
lengths and initializations, then refines the best candidates with alternating
methods and line search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, weights):
    """
    Compute sg[j] = sum_i a[i] * weights[j+i] for j = 0..n-1.
    This is the cross-correlation of weights (len 2n-1) with a (len n).
    """
    n = len(a)
    return np.correlate(weights, a, mode='valid')


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M, A_ub=None):
    """
    Solve LP: maximize sum(g) s.t. conv(a,g) <= M (componentwise), g >= 0.
    Returns (g, g_sum, dual_vars).
    """
    n = len(a)
    if A_ub is None:
        A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        dual = None
        if hasattr(result, 'ineqlin') and result.ineqlin is not None:
            dual = result.ineqlin.marginals
        if g_sum > 1e-10:
            return g, g_sum, dual
    return None, 0.0, None


def obj_value(a, n):
    """Compute objective for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def mirror_descent(a0, time_limit, eta_init=0.05, eta_decay=0.9995,
                   peak_power=2.0, barrier=1e-10, verbose=False):
    """
    Multiplicative subgradient descent (mirror descent on simplex).
    Uses peak-weighted subgradient with adaptive peak selection.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    max_no_improve = 200
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        # Compute peak weights
        conv_max = conv.max()
        if conv_max < 1e-15:
            break
        # Sharpen weights: only top peaks contribute significantly
        normalized = conv / conv_max
        weights = normalized ** peak_power
        # Add small barrier to avoid zero subgradient
        weights = weights + barrier

        # Compute subgradient
        sg = compute_subgradient(a, weights)

        # Multiplicative update
        a_new = a * np.exp(-eta * sg)
        # Ensure positivity
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        eta *= eta_decay
        peak_power = min(10.0, peak_power * 1.001)

        iteration += 1
        if verbose and iteration % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD iter {iteration}, obj={best_obj:.8f}, eta={eta:.6f}, t={elapsed:.1f}s")

        if no_improve >= max_no_improve:
            # Restart with perturbation
            rng = np.random.default_rng(iteration * 31 + 7)
            scale = 0.01 * min(no_improve / max_no_improve, 3.0)
            a = best_a * (1.0 + rng.standard_normal(n) * scale)
            a = np.maximum(a, 1e-30)
            a = a / a.sum()
            conv = fast_convolve(a)
            best_obj = 2.0 * n * conv.max()
            best_a = a.copy()
            no_improve = 0
            max_no_improve = min(500, max_no_improve + 50)
            peak_power = 2.0
            eta = eta_init * 0.5

    return best_a, best_obj


def lp_refine(a0, time_limit, verbose=False):
    """
    LP-based iterative optimization with line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 80

    while time.monotonic() - t0 < time_limit:
        A_ub = build_conv_matrix(a)
        M_lp = M * (1.0 + 5e-4)
        g, g_sum, dual = solve_lp_direction(a, M_lp, A_ub)

        if g is None or g_sum <= 1.0 + 1e-7:
            no_improve += 1
        else:
            d = g / g_sum - a
            norm_d = np.linalg.norm(d)

            if norm_d < 1e-12:
                no_improve += 1
            else:
                # Line search
                best_t = 0.0
                best_obj_ls = best_obj
                for t in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    obj_try, _ = obj_value(a_try, n)
                    if obj_try < best_obj_ls - 1e-14:
                        best_obj_ls = obj_try
                        best_t = t

                if best_t > 0 and best_obj_ls < best_obj - 1e-13:
                    a_new = a + best_t * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    conv = fast_convolve(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        iteration += 1

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")

        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 17 + 3)
            scale = 0.005 + 0.005 * (iteration // max_no_improve)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_no_improve = min(200, max_no_improve + 40)

            if iteration > 500:
                break

    return best_a, best_obj


def dual_guided_mirror_descent(a0, time_limit, dual_weights=None,
                               eta_init=0.03, verbose=False):
    """
    Mirror descent using LP dual variables as peak weights.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    iteration = 0

    # Use dual weights if provided, otherwise use peak-based weights
    if dual_weights is not None:
        weights = dual_weights.copy()
        if weights.max() < 1e-15:
            weights = conv / max(conv.max(), 1e-15)
    else:
        weights = conv / max(conv.max(), 1e-15)
        weights = weights ** 2

    while time.monotonic() - t0 < time_limit:
        sg = compute_subgradient(a, weights)

        # Multiplicative update
        a_new = a * np.exp(-eta * sg)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            # Update weights to current peaks
            normalized = conv / max(conv.max(), 1e-15)
            weights = normalized ** 2
        else:
            no_improve += 1
            # Slowly shift weights toward current peaks
            normalized = conv / max(conv.max(), 1e-15)
            weights = 0.95 * weights + 0.05 * (normalized ** 2)

        eta *= 0.9997
        iteration += 1

        if verbose and iteration % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      DMD iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")

            break

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * (x + 0.5) / n))
    inits['raised_cos'] = rc / rc.sum()

    # Bimodal (two bumps)
    b1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    b2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    bi = b1 + b2
    inits['bimodal'] = bi / bi.sum()

    # Trimodal
    b3 = np.exp(-0.5 * ((x - 0.5 * n) / (0.08 * n)) ** 2)
    tri_mod = b1 + b2 + b3
    inits['trimodal'] = tri_mod / tri_mod.sum()

    # Edge-concentrated
    edge = max(1, n // 3)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    inits['edge'] = ec / ec.sum()

    # Sine
    sn = np.sin(np.pi * (x + 0.5) / n) ** 2
    inits['sine'] = sn / sn.sum()

    # Power law
    pw = (x + 1.0) ** 0.3
    inits['power'] = pw / pw.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps

    return inits

                                            eta_init=0.06, eta_decay=0.9998,
                                            peak_power=2.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 2: LP refinement on top candidates
    print("=== Phase 2: LP Refinement ===")
    t_start2 = time.monotonic()

    # Deduplicate: keep best per (n, name)
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 5:
            break

    if best_sequence is not None:
        top_configs.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 200.0
    if remaining < 30:
        remaining = 60
    n_candidates = max(1, min(len(top_configs), 4))
    t_per_lp = max(20.0, remaining * 0.6 / n_candidates)

    for i in range(min(n_candidates, len(top_configs))):
        if time.monotonic() > deadline - 150:
            break
        n_cand, a0 = top_configs[i]
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 100)
        if t_alloc < 10:
            break

        a_opt, obj_opt = lp_refine(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP {i} (n={n_cand}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: Alternating optimization on best
    print("=== Phase 3: Alternating Optimize ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        if remaining > 30:
            a_opt, obj_opt = alternating_optimize(best_sequence, time_limit=remaining)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Alt: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Final mirror descent polish with tuned params
    print("=== Phase 4: Final Polish ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 15:
            # Try different eta and peak_power
            configs = [
                {'eta_init': 0.02, 'eta_decay': 0.9999, 'peak_power': 3.0},
                {'eta_init': 0.05, 'eta_decay': 0.9997, 'peak_power': 1.5},
            ]
            t_per = remaining / (len(configs) + 1)

            for ci, cfg in enumerate(configs):
                if time.monotonic() > deadline - 10:
                    break
                t_alloc = min(t_per, deadline - time.monotonic() - 8)
                if t_alloc < 5:
                    break
                a_opt, obj_opt = mirror_descent(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish {ci}: {score:.6f}")

            # Final short LP
            if time.monotonic() < deadline - 10:
                t_alloc = min(20.0, deadline - time.monotonic() - 8)
                if t_alloc > 5:
                    a_opt, obj_opt = lp_refine(best_sequence, time_limit=t_alloc)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Final LP: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Phase 5: Perturbation + quick re-optimization
    print("=== Phase 5: Perturb & Reoptimize ===")
    t_start5 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 20:
            n_best = len(best_sequence)
            n_perturb = max(1, int(remaining / 25))
            t_per = remaining / n_perturb

            for pi in range(n_perturb):
                if time.monotonic() > deadline - 5:
                    break
                if t_per < 5:
                    break

                scale = 0.005 + 0.003 * pi
                noise = rng.standard_normal(n_best) * scale
                a_pert = best_sequence + noise
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s

                t_alloc = min(t_per, deadline - time.monotonic() - 3)
                if t_alloc < 3:
                    break
                a_opt, obj_opt = mirror_descent(a_pert, time_limit=t_alloc,
                                                eta_init=0.04, eta_decay=0.9998,
                                                peak_power=2.5)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb {pi}: {score:.6f}")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
