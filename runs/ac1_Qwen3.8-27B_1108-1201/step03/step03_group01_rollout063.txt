    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def coordinate_sweep(a, n, order=None, iterations=1):
    """Perform coordinate descent sweeps."""
    if order is None:
        order = np.arange(n)
    
    for _ in range(iterations):
        for i in order:
            c = a.sum() - a[i]
            if c < 1e-15:
                continue
            
            # Precompute B (conv without a[i]'s contribution)
            # conv(a,a)_k = sum_j a_j * a_{k-j}
            # B_k = conv(a,a)_k - (contribution of a[i])
            
            # Contribution of a[i] to conv(a,a)_k:
            # For k in [i, i+n-1]:
            #   if k == 2i: a[i]^2
            #   else: 2*a[i]*a[k-i]
            # For other k: 0
            
            a_i = a[i]
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            
            # Compute full conv
            conv = fast_convolve(a)
            
            # Compute B
            B = conv.copy()
            for k in range(k_start, k_end):
                if k == 2 * i:
                    B[k] -= a_i * a_i
                else:
                    B[k] -= 2.0 * a_i * a[k - i]
            
            # Now minimize over t
            def obj_t(t):
                s = c + t
                if s < 1e-15:
                    return float('inf')
                # Max of (B_k + delta_k)
                max_val = B.max()  # baseline
                # Update affected entries
                for k in range(k_start, k_end):
                    if k == 2 * i:
                        val = B[k] + t * t
                    else:
                        val = B[k] + 2.0 * t * a[k - i]
                    if val > max_val:
                        max_val = val
                return 2.0 * n * max_val / (s * s)
            
            # Use bounded minimization
            # t should be in [0, some upper bound]
            # Upper bound: if t is very large, obj → 2n (from the quadratic term)
            # Current value is obj_t(a_i), so if that's < 2n, we don't need to go too far
            upper = max(10 * c, 1.0)
            
            result = minimize_scalar(obj_t, bounds=(0, upper), method='bounded',
                                     options={'xatol': 1e-14})
            a[i] = max(0, result.x)
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with 
incremental convolution updates, combined with LP-based direction finding
and adaptive restarts. The coordinate descent exploits the structure of 
the autocorrelation to perform exact 1D minimizations for each element.
"""

import numpy as np
from scipy.optimize import minimize_scalar
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def coord_descent(a, n, n_sweeps=1, random_order=False, rng=None, verbose=False):
    """
    Coordinate descent: for each element, find the optimal value
    while keeping others fixed. Uses the structure of autocorrelation.
    """
    a = a.copy()
    current_obj = 2.0 * n * fast_convolve(a).max() / (a.sum()**2)
    
    for sweep in range(n_sweeps):
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for i in order:
            c = a.sum() - a[i]
            if c < 1e-15:
                continue
            
            a_i = a[i]
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            if k_end <= k_start:
                continue
            ks = np.arange(k_start, k_end)
            
            # Compute B (conv without a[i]'s contribution) for affected entries
            conv = fast_convolve(a)
            B_aff = conv[ks] - 2.0 * a_i * a[ks - i]
            if 2 * i >= k_start and 2 * i < k_end:
                B_aff[2 * i - k_start] = conv[2 * i] - a_i * a_i
            
            # Max outside affected range
            total_len = 2 * n - 1
            if k_start > 0:
                max_out = conv[:k_start].max()
            else:
                max_out = 0.0
            if k_end < total_len:
                max_out = max(max_out, conv[k_end:].max())
            
            # Precompute a[ks - i] for fast evaluation
            a_slice = a[ks - i]
            has_quadratic = (2 * i >= k_start and 2 * i < k_end)
            quad_idx = 2 * i - k_start if has_quadratic else -1
            
            def obj_t(t):
                s = c + t
                if s < 1e-15:
                    return 1e30
                affected = B_aff + 2.0 * t * a_slice
                if has_quadratic:
                    affected[quad_idx] = B_aff[quad_idx] + t * t
                max_val = max(affected.max(), max_out)
                return 2.0 * n * max_val / (s * s)
            
            # Upper bound for t
            upper = max(2.0 * c, a_i * 3.0 + 0.1)
            
            result = minimize_scalar(obj_t, bounds=(0, upper), method='bounded',
                                     options={'xatol': 1e-10})
            new_val = max(0.0, result.x)
            
            if abs(new_val - a_i) > 1e-12:
                a[i] = new_val
        
        # Compute new objective
        new_obj = 2.0 * n * fast_convolve(a).max() / (a.sum()**2)
        if new_obj < current_obj - 1e-12:
            current_obj = new_obj
        if verbose and sweep % 10 == 0:
            print(f"  Coord sweep {sweep}: obj={current_obj:.8f}")
    
    return a, current_obj


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 100
    
    while time.monotonic() - t0 < time_limit:
        A_ub = build_conv_matrix(a)
        M_lp = M * 1.001
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_lp)
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            no_improve += 10
            iteration += 1
            if no_improve > max_no_improve:
                # Perturb
                rng = np.random.default_rng(iteration)
                a = best_a + rng.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    conv = fast_convolve(a)
                    M = conv.max()
                no_improve = 0
            continue
        
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1.0 + 1e-7:
                d = g / g_sum - a
                norm_d = np.linalg.norm(d)
                if norm_d > 1e-12:
                    # Line search
                    best_t = 0.0
                    best_local_obj = best_obj
                    for t in [0.5, 0.3, 0.2, 0.1, 0.8, 1.0, 0.05]:
                        a_try = a + t * d
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        obj_try = 2.0 * n * fast_convolve(a_try).max()
                        if obj_try < best_local_obj - 1e-13:
                            best_local_obj = obj_try
                            best_t = t
                    
                    if best_t > 0:
                        a_new = a + best_t * d
                        a_new = np.maximum(a_new, 0)
                        s_new = a_new.sum()
                        if s_new > 1e-15:
                            a = a_new / s_new
                            conv = fast_convolve(a)
                            M = conv.max()
                            best_obj = 2.0 * n * M
                            best_a = a.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        iteration += 1
        
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 7 + 13)
            scale = 0.003 + 0.002 * min(no_improve // max_no_improve, 5)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
                conv = fast_convolve(a)
                M = conv.max()
            no_improve = 0
            max_no_improve = min(300, 100 + 20 * (iteration // 100))
            
            if iteration > 5000:
                break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent
exploiting the structure of autocorrelation. Each coordinate update is
a 1D minimization of the max of affine/quadratic functions over a quadratic
denominator. Combined with LP refinement and adaptive restarts.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def coord_descent(a, n, n_sweeps=1, random_order=False, rng=None,
                  verbose=False, stall_check=True):
    """
    Coordinate descent: for each element i, find the optimal a[i] keeping
    others fixed. Uses the structure: conv(a,a)_k as function of a[i]=t
    is B_k + 2t*a[k-i] (or B_k + t^2 at k=2i).
    """
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    current_obj = 2.0 * n * fast_convolve(a).max() / (s * s)
    best_a = a.copy()
    best_obj = current_obj
    stall_count = 0
    max_stall = 30

    for sweep in range(n_sweeps):
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        sweep_improved = False

        for i in order:
            c = a.sum() - a[i]
            if c < 1e-15:
                continue

            a_i = a[i]
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            if k_end <= k_start:
                continue
            ks = np.arange(k_start, k_end)
            a_slice = a[ks - i]

            # Compute B for affected entries (conv without a[i]'s contribution)
            conv = fast_convolve(a)
            B_aff = conv[ks] - 2.0 * a_i * a_slice
            has_quad = (2 * i >= k_start and 2 * i < k_end)
            if has_quad:
                B_aff[2 * i - k_start] = conv[2 * i] - a_i * a_i

            # Max outside affected range
            total_len = 2 * n - 1
            if k_start > 0:
                max_out = conv[:k_start].max()
            else:
                max_out = 0.0
            if k_end < total_len:
                mo2 = conv[k_end:].max()
                if mo2 > max_out:
                    max_out = mo2

            quad_idx = 2 * i - k_start if has_quad else -1

            # 1D objective in t
            def obj_t(t):
                s_new = c + t
                if s_new < 1e-15:
                    return 1e30
                affected = B_aff + 2.0 * t * a_slice
                if has_quad:
                    affected[quad_idx] = B_aff[quad_idx] + t * t
                m = affected.max()
                if max_out > m:
                    m = max_out
                return 2.0 * n * m / (s_new * s_new)

            # Upper bound: t beyond this won't help (obj approaches 2n)
            upper = max(3.0 * c, a_i * 4.0 + 0.5)

            result = minimize_scalar(obj_t, bounds=(0.0, upper),
                                     method='bounded',
                                     options={'xatol': 1e-12})
            new_val = max(0.0, result.x)

            if abs(new_val - a_i) > 1e-14:
                a[i] = new_val
                sweep_improved = True

        new_s = a.sum()
        if new_s < 1e-15:
            a = best_a.copy()
            continue
        new_obj = 2.0 * n * fast_convolve(a).max() / (new_s * new_s)

        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_check and stall_count >= max_stall:
            break

        if verbose and (sweep + 1) % 10 == 0:
            print(f"    Coord sweep {sweep+1}: obj={best_obj:.8f}")

    return best_a, best_obj


def lp_step(a, n):
    """One LP-based direction finding step."""
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    if M < 1e-15:
        return a

    A_ub = build_conv_matrix(a)
    M_lp = M * 1.001
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_lp)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs')
    except Exception:
        return a

    if not result.success:
        return a

    g = result.x
    g_sum = g.sum()
    if g_sum <= 1.0 + 1e-7:
        return a

    d = g / g_sum - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a

    # Line search
    best_t = 0.0
    best_obj = 2.0 * n * M
    for t in [0.3, 0.5, 0.1, 0.8, 0.2, 0.6, 1.0, 0.05, 0.15]:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        obj_try = 2.0 * n * fast_convolve(a_try).max()
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t

    if best_t > 0:
        a_new = a + best_t * d
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            return a_new / s_new
    return a

def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['parent'] = parent.copy() if parent is not None else np.ones(n) / n

    inits['uniform'] = np.ones(n) / n

    # Sine window
    x = np.linspace(0, np.pi, n)
    inits['sine'] = (np.sin(x) + 0.01)
    inits['sine'] /= inits['sine'].sum()

    # Triangular
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1.0
    inits['tri'] = tri / tri.sum()

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Power law
    pw = (np.arange(n) + 1.0) ** 0.7
    inits['power'] = pw / pw.sum()

    return inits

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def coord_descent(a0, n, n_sweeps=1, random_order=False, rng=None,
                  verbose=False):
    """
    Exact coordinate descent. For each coordinate i, the objective as a
    function of t=a[i] is max of (affine funcs + one quadratic) / (c+t)^2.
    We minimize this in 1D using bounded scalar optimization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    n_total = 2 * n - 1

    best_obj = 2.0 * n * fast_convolve(a).max() / (a.sum()**2)
    best_a = a.copy()
    stall = 0
    max_stall = 50

    for sweep in range(n_sweeps):
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        improved = False

        for i in order:
            c = a.sum() - a[i]
            if c < 1e-15:
                continue

            a_i = a[i]
            k_start = i
            k_end = min(i + n, n_total)
            if k_end <= k_start:
                continue

            ks = np.arange(k_start, k_end)
            a_slice = a[ks - i]
            has_quad = (2 * i >= k_start and 2 * i < k_end)

            # Compute convolution once per coordinate (using FFT)
            conv = fast_convolve(a)

            # B = conv with a[i]'s contribution removed (for affected entries)
            B_aff = conv[ks] - 2.0 * a_i * a_slice
            if has_quad:
                B_aff[2 * i - k_start] = conv[2 * i] - a_i * a_i

            # Max of conv outside affected range
            max_out = 0.0
            if k_start > 0:
                max_out = conv[:k_start].max()
            if k_end < n_total:
                v = conv[k_end:].max()
                if v > max_out:
                    max_out = v

            quad_idx = (2 * i - k_start) if has_quad else -1

            # 1D objective
            nn = n
            cval = c
            Blocal = B_aff
            aloc = a_slice
            mout = max_out
            hq = has_quad
            qi = quad_idx

            def obj_t(t):
                s_new = cval + t
                if s_new < 1e-15:
                    return 1e30
                affected = Blocal + 2.0 * t * aloc
                if hq:
                    affected[qi] = Blocal[qi] + t * t
                m = affected.max()
                if mout > m:
                    m = mout
                return 2.0 * nn * m / (s_new * s_new)

            # Bounds: t in [0, upper]
            upper = max(3.0 * c, a_i * 5.0 + 1.0)

            res = minimize_scalar(obj_t, bounds=(0.0, upper),
                                  method='bounded', options={'xatol': 1e-13})
            new_val = max(0.0, res.x)

            if abs(new_val - a_i) > 1e-13:
                a[i] = new_val
                improved = True

        # Evaluate after full sweep
        s_new = a.sum()
        if s_new < 1e-15:
            a = best_a.copy()
            continue
        cur_obj = 2.0 * n * fast_convolve(a).max() / (s_new * s_new)

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= max_stall:
            break

        if verbose and (sweep + 1) % 20 == 0:
            print(f"      sweep {sweep+1}: obj={best_obj:.8f}")

    return best_a, best_obj


def lp_iterate(a0, time_limit):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 40
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        A_ub = build_conv_matrix(a)
        M_lp = M * 1.0005
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_lp)
        bounds = [(0.0, None)] * n

        try:
            result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                             method='highs')
        except Exception:
            no_improve += 5
            iteration += 1
            continue

        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1.0 + 1e-7:
                d = g / g_sum - a
                if np.linalg.norm(d) > 1e-12:
                    best_t = 0.0
                    best_local = best_obj
                    for t in [0.2, 0.4, 0.1, 0.6, 0.3, 0.8, 1.0, 0.05, 0.5]:
                        a_try = np.maximum(a + t * d, 0.0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        obj_try = 2.0 * n * fast_convolve(a_try).max()
                        if obj_try < best_local - 1e-14:
                            best_local = obj_try
                            best_t = t

                    if best_t > 0:
                        a_new = np.maximum(a + best_t * d, 0.0)
                        s_new = a_new.sum()
                        if s_new > 1e-15:
                            a = a_new / s_new
                            M = fast_convolve(a).max()
                            best_obj = 2.0 * n * M
                            best_a = a.copy()
                            no_improve = 0
                            continue

        no_improve += 1
        iteration += 1

        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 13 + 7)
            scale = 0.003 * (1 + 0.3 * min(no_improve // max_no_improve, 5))
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
                M = fast_convolve(a).max()
            no_improve = 0
            max_no_improve = min(150, 40 + 15 * (iteration // 50))
            if iteration > 3000:
                break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: coordinate descent (primary), LP refinement (secondary),
    multi-start exploration, and final polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    n = 512

    # Build initial from parent
    if parent is not None:
        if len(parent) == n:
            a_init = parent.copy()
        else:
            frac = np.linspace(0, 1, n)
            if len(parent) > n:
                a_init = np.interp(frac, np.linspace(0, 1, len(parent)), parent)
            else:
                a_init = np.interp(frac, np.linspace(0, 1, len(parent)), parent)
        a_init = np.maximum(a_init, 0.0)
        s = a_init.sum()
        if s > 1e-10:
            a_init = a_init / s
        else:
            a_init = np.ones(n) / n
    else:
        a_init = np.ones(n) / n

    score = float(evaluate_sequence(a_init.tolist()))
    if np.isfinite(score):
        best_score = score
        best_sequence = a_init.copy()
        print(f"Initial: {score:.6f}")

    # === Phase 1: Coordinate Descent (main workhorse) ===
    print("=== Phase 1: Coordinate Descent ===")
    t0 = time.monotonic()
    p1_deadline = t0 + min(1000.0, deadline - t0 - 250.0)

    a = a_init.copy()
    stall_count = 0
    last_best_p1 = best_score
    batch = 40

    while time.monotonic() < p1_deadline:
        a, obj = coord_descent(a, n, n_sweeps=batch, random_order=False, rng=rng)
        s = a.sum()
        if s > 1e-15:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score - 1e-12:
                best_score = score
                best_sequence = a.copy()
                stall_count = 0
                print(f"  P1 [{time.monotonic()-t0:.0f}s]: {score:.6f}")
            else:
                stall_count += 1
                if stall_count % 10 == 0:
                    # Perturb and continue
                    scale = 0.002 * min(stall_count // 5, 8)
                    a = best_sequence + rng.standard_normal(n) * scale
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s

    print(f"Phase 1 done ({time.monotonic()-t0:.0f}s). Best: {best_score:.6f}")

    # === Phase 2: LP refinement ===
    print("=== Phase 2: LP Refinement ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic() - 180.0
    if best_sequence is not None and remaining > 30:
        t_alloc = min(remaining * 0.5, 250.0)
        a_opt, _ = lp_iterate(best_sequence, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  P2 LP: {score:.6f}")

        # Coord descent from LP result
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 20:
            n_sw = max(20, int(remaining * 3))
            a_opt, _ = coord_descent(a_opt, n, n_sweeps=n_sw,
                                     random_order=False, rng=rng)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2 Coord: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t0:.0f}s). Best: {best_score:.6f}")

    # === Phase 3: Multi-start with different inits ===
    print("=== Phase 3: Multi-Start ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic() - 60.0

    if remaining > 30:
        # Generate diverse inits
        inits = []
        inits.append(('best', best_sequence.copy()))
        inits.append(('uniform', np.ones(n) / n))

        x = np.linspace(0, 1, n)
        sine_w = np.sin(np.pi * x) + 0.05
        inits.append(('sine', sine_w / sine_w.sum()))

        tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float) + 1
        inits.append(('tri', tri / tri.sum()))

        bi = np.zeros(n)
        bi[:n//3] = 1.0
        bi[-n//3:] = 1.0
        inits.append(('bimodal', bi / bi.sum()))

        inits.append(('rand', rng.dirichlet(np.ones(n) * 0.25)))

        t_per = max(8.0, remaining / len(inits))

        for name, a0 in inits:
            if time.monotonic() > deadline - 45:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 35)
            if t_alloc < 5:
                break
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s < 1e-10:
                continue
            a0 = a0 / s

            n_sw = max(10, int(t_alloc * 3))
            a_opt, _ = coord_descent(a0, n, n_sweeps=n_sw,
                                     random_order=False, rng=rng)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3 {name}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t0:.0f}s). Best: {best_score:.6f}")

    # === Phase 4: Final polish ===
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 8.0
        if remaining > 30:
            # LP
            t_lp = min(remaining * 0.35, 60.0)
            if t_lp > 10:
                a_opt, _ = lp_iterate(best_sequence, time_limit=t_lp)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P4 LP: {score:.6f}")

            # Coord
            remaining = deadline - time.monotonic() - 5.0
            if remaining > 5:
                n_sw = max(15, int(remaining * 4))
                a_opt, _ = coord_descent(best_sequence, n, n_sweeps=n_sw,
                                         random_order=False, rng=rng)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P4 Coord: {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(n) / n

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
