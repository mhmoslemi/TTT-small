"""
Multi-phase optimizer for minimum peak autocorrelation of non-negative sequences.
Combines L-BFGS-B with log-sum-exp smoothing (via softmax parameterization),
projected subgradient with adaptive step size, and LP-based direction refinement.
Key improvement: efficient gradient computation via cross-correlation and
adaptive temperature schedule for the smoothed surrogate.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective_raw(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for non-normalized a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_objective_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_obj_grad(z, t):
    """
    Compute smoothed objective F = (2n/t)*log(sum exp(t*b[k])) and gradient w.r.t. z.
    a = softmax(z).
    """
    n = len(z)
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()

    b = fast_convolve(a)

    t_b = t * b
    t_b_max = t_b.max()
    exp_t_b = np.exp(t_b - t_b_max)
    sum_exp = exp_t_b.sum()
    F = (2.0 * n / t) * (np.log(sum_exp) + t_b_max)

    p = exp_t_b / sum_exp
    h = np.correlate(p, a, mode='valid')
    S = np.dot(h, a)
    grad_z = 4.0 * n * t * a * (h - S)

    return F, grad_z


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_top_peaks(a, conv, top_k, temperature):
    """Weighted subgradient over top-k convolution peaks."""
    n = len(a)
    m = len(conv)
    if top_k >= m:
        peaks = np.arange(m)
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = 2.0 * w[i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M'."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def optimize_lp_iterative(a0, time_limit, method='max_sum', max_iters=500,
                          step_init=0.3, step_min=0.02):
    """LP-based iterative refinement."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    current_method = method

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()

        if current_method == 'max_sum':
            b = lp_maximize_sum(a, M)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        else:
            b, _ = lp_minimize_peak(a)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum

        if np.linalg.norm(b - a) < 1e-10:
            return best_a, best_obj

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(0.8, step * 1.05)
        else:
            no_improve += 1
            step = max(step_min, step * 0.92)

        a = a_new
        conv = conv_new

        if no_improve > 80:
            current_method = 'min_peak' if current_method == 'max_sum' else 'max_sum'
            step = step_init
            no_improve = 0

        if no_improve > 300:
            break

    return best_a, best_obj


def optimize_softmax_lbfgs(a0, time_limit, t_values=None):
    """Optimize using L-BFGS-B on softmax parameterization with log-sum-exp smoothing."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if t_values is None:
        t_values = [20.0, 50.0, 100.0, 200.0, 500.0]

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    z = np.log(np.maximum(a, 1e-15))
    t0 = time.monotonic()

    for t in t_values:
        if time.monotonic() - t0 > time_limit:
            break

        timeout = min(60.0, time_limit - (time.monotonic() - t0))
        if timeout < 5:
            break

        def func_g(z_vec):
            F, g = softmax_obj_grad(z_vec, t)
            return F, g

        try:
            res = minimize(func_g, z, method='L-BFGS-B',
                           jac=True, options={'maxiter': 200, 'maxfun': 5000,
                                              'gtol': 1e-6})
        except Exception:
            break

        z_opt = res.x
        z_max = z_opt.max()
        exp_z = np.exp(z_opt - z_max)
        a_new = exp_z / exp_z.sum()

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            z = z_opt

    return best_a, best_obj


def optimize_subgrad_v2(a0, time_limit, lr_init=0.04, momentum=0.85,
                        top_k=50, temp_init=0.001, perturb_every=2000,
                        perturb_scale=0.003):
    """Projected subgradient with momentum, adaptive step, and perturbation."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        sg = subgradient_top_peaks(a, conv, top_k, temp)
        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new

        lr *= 0.9998
        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            perturb_every = int(perturb_every * 1.5)
            lr = max(lr_init * 0.2, lr * 0.4)

    return best_a, best_obj


def optimize_annealed_subgrad(a0, time_limit, n_restarts=5):
    """Subgradient with simulated annealing schedule."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    current_temp = 0.01
    time_per_restart = time_limit / n_restarts

    for restart in range(n_restarts):
        restart_deadline = t0 + (restart + 1) * time_per_restart
        v = np.zeros(n)
        lr = 0.05
        no_improve = 0
        temp = current_temp

        while time.monotonic() < restart_deadline:
            conv = fast_convolve(a)
            M = conv.max()
            threshold = M - 0.02 * M
            active = conv > threshold
            if not active.any():
                active[np.argmax(conv)] = True

            sg = np.zeros(n)
            for k in range(len(conv)):
                if not active[k]:
                    continue
                wi = 2.0 * conv[k] / conv[active].sum()
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += wi * a[k - idx]

            v = 0.88 * v - lr * sg
            a = project_simplex(a + v)

            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()

            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            lr *= 0.9995
            temp *= 0.999

            if no_improve > 3000:
                break

        current_temp *= 0.5

    return best_a, best_obj


def try_multiple_lengths(parent, deadline, best_tracker):

    for n in unique_lengths:
        remain = deadline - time.monotonic() - 100

            if len(pp) >= n:
                pp = pp[:n]
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(pp)), pp)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits.append(('parent', pp / ps))

        x = np.arange(n, dtype=np.float64)
        for freq in [1, 2, 3, 5]:
            a_init = 1.0 + 0.7 * np.cos(2.0 * np.pi * freq * x / n + rng.random() * 2 * np.pi)
            a_init = np.maximum(a_init, 0)
            a_init /= a_init.sum()
            inits.append((f'cos{freq}', a_init))

        g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
        g /= g.sum()
        inits.append(('gauss', g))

        inits.append(('random', rng.dirichlet(np.ones(n) * 0.3)))

        time_per = max(10.0, min(120.0, remain / len(inits) / 2))

        for name, a0 in inits:
            remain = deadline - time.monotonic() - 80
            if remain < 10:
                break
            t_alloc = min(time_per, remain)
            if t_alloc < 5:
                break

            a_opt, _ = optimize_subgrad_v2(a0, time_limit=t_alloc,
                                           lr_init=0.03, momentum=0.87,
                                           top_k=max(20, n // 10),
                                           temp_init=0.001)
            best_tracker(a_opt)

            if time.monotonic() > deadline - 80:
                break

            if 64 <= n_try <= 2048:
                if n_try != n_par:
                    pp = np.interp(np.linspace(0, 1, n_try),
                                   np.linspace(0, 1, n_par), a_par)
                    pp = np.maximum(pp, 0)
                    ps = pp.sum()
                    if ps > 1e-10:
                        try_update(pp / ps)

    # Try various initializations at n=512
    n = 512
    x = np.arange(n, dtype=np.float64)
    inits_512 = []
    inits_512.append(np.ones(n) / n)
    if parent is not None:
        pp = np.maximum(parent, 0.0)
        if len(pp) >= n:
            pp = pp[:n]
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            inits_512.append(pp / pp.sum())
    for freq in [1, 2, 3]:
        a_i = 1.0 + 0.8 * np.cos(2.0 * np.pi * freq * x / n + rng.random() * 6.28)
        a_i = np.maximum(a_i, 0)
        inits_512.append(a_i / a_i.sum())
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits_512.append(g / g.sum())
    inits_512.append(rng.dirichlet(np.ones(n) * 0.3))
    inits_512.append(rng.dirichlet(np.ones(n) * 1.0))

    # Quick subgradient on each
    time_per = max(8.0, (phase1_end - time.monotonic()) / len(inits_512))
    for a0 in inits_512:
        if time.monotonic() > phase1_end:
            break
        t_alloc = min(time_per, phase1_end - time.monotonic())
        if t_alloc < 3:
            break
        a_opt, _ = optimize_subgrad_v2(a0, time_limit=t_alloc, lr_init=0.04,
                                       momentum=0.87, top_k=40, temp_init=0.001)
        try_update(a_opt)

    # Try other lengths
    for n_other in [256, 1024]:
        if time.monotonic() > phase1_end - 10:
            break
        a_init = np.ones(n_other) / n_other
        if parent is not None:
            pp = np.interp(np.linspace(0, 1, n_other),
                           np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                a_init = pp / pp.sum()
        t_alloc = min(40.0, phase1_end - time.monotonic())
        if t_alloc > 5:
            a_opt, _ = optimize_subgrad_v2(a_init, time_limit=t_alloc, lr_init=0.04,
                                           momentum=0.87, top_k=30, temp_init=0.001)
            try_update(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # Phase 2: L-BFGS-B with log-sum-exp smoothing (500s)
    print("=== Phase 2: L-BFGS-B with log-sum-exp ===")
    t2_start = time.monotonic()
    phase2_end = t2_start + min(500.0, budget_s * 0.28)

    if best_sequence is not None:
        a_best = best_sequence / best_sequence.sum()
        n_best = len(a_best)

        # Temperature schedule: start low, increase
        t_sched = [10.0, 30.0, 80.0, 200.0, 500.0, 1000.0]
        time_per_t = max(15.0, (phase2_end - time.monotonic()) / len(t_sched))

        for t_val in t_sched:
            if time.monotonic() > phase2_end - 10:
                break
            t_alloc = min(time_per_t, phase2_end - time.monotonic())
            if t_alloc < 5:
                break

            z = np.log(np.maximum(a_best, 1e-15))

            def make_obj(t_cur):
                def func(z_vec):
                    F, g = softmax_obj_grad(z_vec, t_cur)
                    return F, g
                return func

            func = make_obj(t_val)
            try:
                res = minimize(func, z, method='L-BFGS-B', jac=True,
                               options={'maxiter': 300, 'maxfun': 10000,
                                        'gtol': 1e-7})
                z_opt = res.x
                z_max = z_opt.max()
                exp_z = np.exp(z_opt - z_max)
                a_new = exp_z / exp_z.sum()
                try_update(a_new)
            except Exception as e:
                print(f"  L-BFGS-B error at t={t_val}: {e}")

        # Also try from a few perturbed starts
        for _ in range(3):
            if time.monotonic() > phase2_end - 20:
                break
            noise = rng.standard_normal(n_best) * 0.01
            a_pert = project_simplex(a_best + noise)
            t_val = 50.0
            z = np.log(np.maximum(a_pert, 1e-15))

            def make_obj2(t_cur):
                def func(z_vec):
                    F, g = softmax_obj_grad(z_vec, t_cur)
                    return F, g
                return func

            func = make_obj2(t_val)
            try:
                t_alloc = min(30.0, phase2_end - time.monotonic())
                if t_alloc < 5:
                    break
                res = minimize(func, z, method='L-BFGS-B', jac=True,
                               options={'maxiter': 200, 'maxfun': 5000, 'gtol': 1e-7})
                z_opt = res.x
                z_max = z_opt.max()
                exp_z = np.exp(z_opt - z_max)
                a_new = exp_z / exp_z.sum()
                try_update(a_new)
            except Exception:
                pass

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # Phase 3: LP-based refinement (400s)
    print("=== Phase 3: LP-based refinement ===")
    t3_start = time.monotonic()
    phase3_end = t3_start + min(400.0, budget_s * 0.22)

    if best_sequence is not None and len(best_sequence) <= 1024:
        a_best = best_sequence / best_sequence.sum()

        for step_init in [0.35, 0.5, 0.2, 0.15]:
            if time.monotonic() > phase3_end - 10:
                break
            for method in ['max_sum', 'min_peak']:
                if time.monotonic() > phase3_end - 10:
                    break
                t_alloc = min(60.0, phase3_end - time.monotonic())
                if t_alloc < 10:
                    break
                a_opt, _ = optimize_lp_iterative(a_best, time_limit=t_alloc,
                                                 method=method, max_iters=400,
                                                 step_init=step_init)
                try_update(a_opt)
                a_best = best_sequence / best_sequence.sum()

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # Phase 4: Annealed subgradient (200s)
    print("=== Phase 4: Annealed subgradient ===")
    t4_start = time.monotonic()
    phase4_end = t4_start + min(200.0, budget_s * 0.12)

    if best_sequence is not None:
        a_best = best_sequence / best_sequence.sum()
        n_best = len(a_best)
        t_alloc = min(150.0, phase4_end - time.monotonic())
        if t_alloc > 10:
            a_opt, _ = optimize_annealed_subgrad(a_best, time_limit=t_alloc, n_restarts=5)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # Phase 5: Final perturb-and-refine (remaining time)
    print("=== Phase 5: Final polish ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        while time.monotonic() < deadline - 10:
            a_best = best_sequence / best_sequence.sum()
            n_best = len(a_best)

            # Try small perturbations + LP refinement
            for scale in [0.001, 0.003, 0.005, 0.002, 0.008]:
                if time.monotonic() > deadline - 15:
                    break
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(a_best + noise)
                t_alloc = min(30.0, deadline - time.monotonic() - 10)
                if t_alloc < 5:
                    break

                if n_best <= 768:
                    a_opt, _ = optimize_lp_iterative(a_pert, time_limit=t_alloc,
                                                     method='min_peak',
                                                     max_iters=200, step_init=0.25)
                else:
                    a_opt, _ = optimize_subgrad_v2(a_pert, time_limit=t_alloc,
                                                   lr_init=0.02, momentum=0.9,
                                                   top_k=max(20, n_best // 15),
                                                   temp_init=0.0005)
                try_update(a_opt)

            # Try subgradient with different params
            t_alloc = min(30.0, deadline - time.monotonic() - 10)
            if t_alloc > 5:
                a_opt, _ = optimize_subgrad_v2(a_best, time_limit=t_alloc,
                                               lr_init=0.01, momentum=0.92,
                                               top_k=max(30, n_best // 8),
                                               temp_init=0.0002,
                                               perturb_every=1500,
                                               perturb_scale=0.001)
                try_update(a_opt)

            if time.monotonic() > deadline - 20:
                break

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
