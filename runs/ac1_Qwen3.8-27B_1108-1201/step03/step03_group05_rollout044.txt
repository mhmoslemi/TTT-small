This is a hard problem, but there are known constructions. For instance, "all-ones" sequences (Barker sequences) have low peak sidelobe, but they're for ±1 sequences, not non-negative.

For non-negative sequences on the simplex, I think the optimal solution has a "tent" or "bell" shape - concentrated in the middle with smooth falloff. The previous approaches found this.

Let me focus on making the optimization more effective:

**Key improvements I'll make:**
1. **Better FISTA implementation** with:
   - Proper backtracking line search
   - Restart rule (when ||y - a||^2 > ||a - a_prev||^2, reset momentum)
   - Adaptive beta for the log-sum-exp smoothing
   - Multiple restarts with different beta values

2. **2D coordinate descent** with:
   - Greedy pair selection (pick the pair that gives the largest decrease)
   - Efficient 1D minimization using bisection
   - Multiple sweeps

3. **Spectral equalization** as an alternative initialization:
   - Gerchberg-Saxton style
   - Enforce non-negativity and sum constraint at each step

4. **Simulated annealing / perturbation** to escape local minima:
   - Periodically add noise and re-optimize
   - Try different n values

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated projected gradient) with log-sum-exp smoothing
2. Greedy 2D coordinate descent for exact convex optimization
3. Spectral equalization (Gerchberg-Saxton) for diverse initializations
Combined in a multi-phase search with diverse starting points.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_gradient(a, conv, beta):
    """
    Smoothed gradient of logsumexp(beta*conv)/beta.
    Returns gradient vector of length n.
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    # grad = 2 * cross-correlation of a with (conv * p)
    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]

def fista_optimize(a0, time_limit, beta_init=50.0, beta_end=500.0,
                   lr_init=0.02, momentum_init=0.9, seed=0):
    """
    FISTA with log-sum-exp smoothing, backtracking, and restart rule.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    y = a.copy()
    t = 1.0
    conv_y = fast_convolve(y)
    best_obj = conv_y.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    lr = lr_init
    momentum = momentum_init

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac

        # Compute gradient at y
        grad = softmax_gradient(y, conv_y, beta)

        # Backtracking line search
        step = lr
        a_new = project_simplex(y - step * grad)
        conv_new = fast_convolve(a_new)
        
        # Check sufficient decrease
        for _ in range(20):
            if conv_new.max() <= conv_y.max() + 0.5 * step * step * grad @ grad:
                break
            step *= 0.5
            a_new = project_simplex(y - step * grad)
            conv_new = fast_convolve(a_new)

        # Update
        a_old = a.copy()
        a = a_new
        conv = conv_new
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Nesterov update for y
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y = a + ((t - 1) / t_new) * (a - a_old)
        # Clamp y to non-negative (soft constraint)
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum > 1e-10:
            y = y / y_sum  # Keep y on or near the simplex
        t = t_new

        conv_y = fast_convolve(y)

        # Restart rule: if y is far from optimal direction, reset
        if no_improve > 200:
            y = a.copy()
            t = 1.0
            momentum = momentum_init * (1.0 - 0.3 * frac)
            no_improve = 0
            # Add small perturbation
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            y = a.copy()
            conv_y = fast_convolve(y)
            conv = fast_convolve(a)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_a = a.copy()

        # Adaptive learning rate
        lr = max(0.002, lr_init * (1.0 - 0.8 * frac))

    return best_a, best_obj

def coordinate_descent(a0, time_limit, seed=0, max_sweeps=200):
    """
    Greedy 2D coordinate descent on the simplex.
    For each sweep, try to improve by moving mass from one coordinate to another.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        improved = False
        conv = fast_convolve(a)
        current_max = conv.max()

        # Get candidate source coordinates (large values)
        # and target coordinates (small values)
        sorted_idx = np.argsort(a)
        # Sources: top 20% largest
        n_src = max(1, n // 5)
        sources = sorted_idx[-n_src:]
        # Targets: bottom 20% smallest (but not zero)
        n_tgt = max(1, n // 5)
        targets = sorted_idx[:n_tgt]

        # For each source-target pair, find optimal transfer
        for m in sources:
            if a[m] < 1e-8:
                continue
            for j in targets:
                if j == m or a[j] > a[m] * 0.5:
                    continue

                # Minimize h(t) = max_k conv_k(a + t*(e_j - e_m)) for t in [0, a[m]]
                # Use bisection
                t_lo = 0.0
                t_hi = min(a[m] * 0.9, current_max * 0.1)  # Limit step size
                t_hi = min(t_hi, a[m])
                
                if t_hi < 1e-10:
                    continue

                # Bisection for optimal t
                t_opt = 0.0
                for _ in range(30):
                    t_mid = (t_lo + t_hi) / 2
                    a_try = a.copy()
                    a_try[m] -= t_mid
                    a_try[j] += t_mid
                    
                    # Compute directional derivative at t_mid
                    # d(conv_k)/dt = 2*(a[k-j] - a[k-m]) for valid indices
                    # (using the UPDATED a_try for accuracy, but for bisection
                    #  we use the original a for the derivative)
                    
                    # Active set: k where conv_k is close to max
                    conv_try = fast_convolve(a_try)
                    max_try = conv_try.max()
                    
                    if max_try >= current_max and max_try >= best_obj:
                        # This direction makes it worse or no better
                        t_hi = t_mid
                        continue
                    
                    # Check if we can go further
                    if max_try < current_max:
                        t_lo = t_mid
                        t_opt = t_mid
                    else:
                        t_hi = t_mid

                if t_opt > 1e-10:
                    a_new = a.copy()
                    a_new[m] -= t_opt
                    a_new[j] += t_opt
                    conv_new = fast_convolve(a_new)
                    if conv_new.max() < best_obj - 1e-15:
                        best_obj = conv_new.max()
                        best_a = a_new.copy()
                        a = a_new
                        improved = True
                        if not improved:
                            break

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 5:
                break
        else:
            no_improve_sweeps = 0

    return best_a, best_obj

def spectral_equalize(a0, time_limit, seed=0):
    """
    Gerchberg-Saxton style spectral equalization.
    Iteratively modify the magnitude spectrum to flatten the time-domain peak.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    fft_size = 1 << ((2 * n - 1) - 1).bit_length()
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # FFT
        A = np.fft.rfft(a, fft_size)
        phase = np.angle(A)
        mag = np.abs(A)

        # Current time-domain signal
        conv_cur = fast_convolve(a)
        peak = conv_cur.max()

        # Modify magnitude: reduce bins that contribute to peaks
        # Use a "peak shaving" in frequency domain
        # The contribution of bin k to the time-domain peak is related to
        # |A[k]|^2 * exp(i*phase). We want to reduce |A[k]| for bins
        # that constructively interfere at the peak location.

        peak_idx = np.argmax(conv_cur)
        
        # For each frequency bin, compute its contribution to the peak
        # conv[peak_idx] = sum_k |A[k]|^2 * exp(i * 2*pi*k*peak_idx/fft_size) / fft_size
        # (approximately, from IFFT)
        
        # Simpler approach: iteratively reduce the magnitude of the strongest bins
        # and increase the weakest (water-filling)
        
        target_peak = peak * 0.995  # Reduce peak by 0.5%
        
        # Water-filling: try to equalize |A|^2
        # But this might destroy the non-negativity of a
        # Instead, use a multiplicative update:
        # mag_new[k] = mag[k] * (1 - gamma * contrib[k]) where contrib[k] 
        # measures how much bin k contributes to the peak
        
        # Compute contribution of each bin to the peak
        # conv[peak_idx] = (1/fft_size) * sum_k |A[k]|^2 * exp(i*2*pi*k*peak_idx/fft_size)
        # The real part contribution is |A[k]|^2 * cos(2*pi*k*peak_idx/fft_size) / fft_size
        
        k_vals = np.arange(fft_size // 2 + 1)
        angles = 2 * np.pi * k_vals * peak_idx / fft_size
        contribs = mag ** 2 * np.cos(angles)
        total_contrib = contribs.sum() / fft_size
        
        if total_contrib < 1e-10:
            break
        
        # Reduce bins with positive contribution, increase those with negative
        gamma = 0.1
        contribs_norm = contribs / total_contrib
        mag_new = mag * (1 - gamma * contribs_norm)
        mag_new = np.maximum(mag_new, mag * 0.8)  # Don't reduce too much
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to simplex
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        if conv_new.max() < best_obj - 1e-12:
            best_obj = conv_new.max()
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj

def fast_2d_cd(a0, time_limit, seed=0):
    """
    Fast 2D coordinate descent with smarter pair selection.
    Uses gradient information to pick promising pairs.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 100:
        conv = fast_convolve(a)
        current_max = conv.max()
        peak_indices = np.where(conv >= current_max * 0.99)[0]

        # Compute gradient at the peak
        # grad[j] = 2 * sum_{k in peaks} a[k-j] / |peaks|
        grad = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            grad[i_start:i_end+1] += 2.0 * a[k - i_end:k - i_start + 1][::-1]
        grad /= len(peak_indices)

        # We want to decrease a where grad is high (reduces peak)
        # and increase a where grad is low
        # Move mass from high-grad positions to low-grad positions
        
        # Pick source: position with max a[j] * grad[j] (biggest reduction per unit mass)
        # Pick target: position with min grad[j] (least increase in peak per unit mass)
        
        source_score = a * grad  # Reduction in peak per unit mass moved out
        target_score = grad  # Increase in peak per unit mass moved in

        # Sort
        src_order = np.argsort(source_score)[::-1]  # Best sources first
        tgt_order = np.argsort(target_score)  # Best targets first

        improved = False
        n_try = min(10, n)
        for _ in range(n_try):
            m = src_order[_ % n]
            j = tgt_order[_ % n]
            if m == j or a[m] < 1e-8:
                continue

            # Binary search for optimal t
            t_lo, t_hi = 0.0, min(a[m] * 0.95, 0.1 / max(grad[m], 1e-10))
            t_hi = min(t_hi, a[m] * 0.5)
            
            best_t = 0.0
            best_f = best_obj
            
            for _ in range(25):
                t_mid = (t_lo + t_hi) / 2
                a_try = a.copy()
                a_try[m] -= t_mid
                a_try[j] += t_mid
                f_try = fast_convolve(a_try).max()
                if f_try < best_f:
                    best_f = f_try
                    best_t = t_mid
                    t_lo = t_mid
                else:
                    t_hi = t_mid

            if best_t > 1e-10 and best_f < best_obj - 1e-12:
                best_obj = best_f
                best_a = a.copy()
                best_a[m] -= best_t
                best_a[j] += best_t
                a = best_a.copy()
                improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Perturb to escape
            if no_improve % 20 == 0:
                noise = rng.standard_normal(n) * 0.001
                a = project_simplex(a + noise)

    return best_a, best_obj
