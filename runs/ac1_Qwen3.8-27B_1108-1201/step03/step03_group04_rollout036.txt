"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Uses Nesterov-accelerated subgradient descent with multi-peak weighting,
spectral shaping (Gerchberg-Saxton style), and adaptive restarts.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

def nesterov_subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9, 
                                    top_k=20, verbose=False):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    # Nesterov: y is the look-ahead point
    y = a.copy()
    v = np.zeros(n)  # velocity
    mu = momentum  # momentum parameter
    lr = lr_init
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    patience = 500
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        
        # Multi-peak weighting: exponential weighting on top peaks
        top_k_actual = min(top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -top_k_actual)[-top_k_actual:]
        peak_vals = conv_y[peaks]
        
        # Softmax weighting
        temp = max(peak_vals.max() * 0.1, 1e-15)
        weights = np.exp((peak_vals - peak_vals.max()) / temp)
        weights /= weights.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # g[j] += 2 * w_i * y[k - j] for j in [j_start, j_end]
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * weights[i] * y[k - idx]
        
        # Nesterov update
        v = mu * v - lr * g
        y_new = y + v
        
        # Project to simplex: y_new >= 0, sum(y_new) = 1
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = np.ones(n) / n
        else:
            y_new = y_new / s_new
        
        # Check improvement
        conv_new = fast_convolve(y_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = y_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        y = y_new
        a = y_new
        
        # Adaptive learning rate
        if no_improve > patience:
            lr *= 0.7
            no_improve = 0
            patience = int(patience * 1.5)
        
        lr *= 0.9999  # slow decay
        
        it += 1
        if verbose and it % 500 == 0:
            print(f"    it={it}, obj={best_obj:.6f}, lr={lr:.6f}")
    
    return best_a, best_obj

"""
Multi-phase optimization for minimum peak autocorrelation non-negative sequences.
Combines Nesterov-accelerated subgradient descent with multi-peak weighting,
Gerchberg-Saxton spectral shaping, multiplicative weights (mirror descent),
coordinate descent local search, and adaptive multi-start strategy.
Key improvements over LP-based Frank-Wolfe: faster iterations (O(n log n) vs O(n^2)),
better escape from local minima via spectral domain operations, and finer-grained
local refinement via coordinate descent.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_multi_peak_subgradient(y, top_k, soft):
    """Compute multi-peak weighted subgradient and convolution."""
    n = len(y)
    conv = fast_convolve(y)
    top_k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k_actual)[-top_k_actual:]
    peak_vals = conv[peaks]

    temp = max(peak_vals.max() * soft, 1e-15)
    weights = np.exp((peak_vals - peak_vals.max()) / temp)
    weights /= weights.sum()

    g = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        g[idx] += 2.0 * weights[i] * y[k - idx]

    return g, conv


def nesterov_optimize(a0, time_limit, top_k=20, soft=0.1,
                      lr_init=0.003, momentum=0.93, verbose=False, verbose_every=1000):
    """Nesterov-accelerated subgradient descent on non-negative simplex."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    obj = compute_obj(a)
    best_obj = obj
    best_a = a.copy()

    y = a.copy()
    v = np.zeros(n)
    mu = momentum
    lr = lr_init

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    patience = 400

    while time.monotonic() - t0 < time_limit:
        g, conv_y = compute_multi_peak_subgradient(y, top_k, soft)

        v = mu * v - lr * g
        y_new = y + v
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = np.ones(n) / n
        else:
            y_new = y_new / s_new

        obj_new = compute_obj(y_new)
        y = y_new

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            seed_val = (it * 7919 + n * 31) % (2**31)
            rng = np.random.default_rng(seed_val)
            noise_scale = min(0.005, 0.001 * (1 + no_improve // patience))
            noise = rng.standard_normal(n) * noise_scale
            y = np.maximum(y + noise, 0.0)
            y = y / y.sum()
            v *= 0.3
            no_improve = 0
            patience = int(patience * 1.3)
            lr = max(lr * 0.8, 1e-7)

        lr = max(lr * 0.9999, 1e-7)
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"      Nesterov it={it}, obj={best_obj:.6f}, lr={lr:.2e}, elapsed={elapsed:.1f}s")

    return best_a, best_obj

def gs_spectral_shape(a0, n_iters=300, blend=0.4):
    """Gerchberg-Saxton style spectral shaping to flatten autocorrelation."""
    n = len(a0)
    N = 1 << (2 * n - 1).bit_length()

    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for _ in range(n_iters):
        a_pad = np.zeros(N, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)

        total_energy = np.sum(np.abs(A) ** 2)
        dc_energy = np.abs(A[0]) ** 2
        non_dc_energy = max(total_energy - dc_energy, 0)
        n_non_dc = N - 1

        if non_dc_energy > 1e-15 and n_non_dc > 0:
            target_mag = np.sqrt(non_dc_energy / n_non_dc)
        else:
            target_mag = 1e-10

        phases = np.angle(A[1:])
        current_mags = np.abs(A[1:])
        new_mags = (1.0 - blend) * current_mags + blend * target_mag
        A[1:] = new_mags * np.exp(1j * phases)
        A[0] = a.sum()

        a_new = np.fft.ifft(A).real
        a_new = np.maximum(a_new[:n], 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def coordinate_descent_polish(a0, time_limit, n_sweeps_max=50):
    """Coordinate descent: try small adjustments to individual coordinates."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_obj = compute_obj(a)
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0
    deltas = [0.001, -0.001, 0.002, -0.002, 0.0005, -0.0005]

    while time.monotonic() - t0 < time_limit and sweep < n_sweeps_max:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            for d in deltas:
                trial = a.copy()
                trial[j] = max(0.0, trial[j] + d)
                ts = trial.sum()
                if ts < 1e-10:
                    continue
                trial = trial / ts
                obj = compute_obj(trial)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = trial.copy()
                    a = trial
                    improved = True
                    break
        sweep += 1
        if not improved:
            break

    return best_a

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    inits['ramp_rev'] = r[::-1] / r.sum()

    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    edge = max(1, n // 5)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    inits['edge'] = ec / ec.sum()

    sigma_vals = [0.2, 0.3, 0.5]
    for i, sig in enumerate(sigma_vals):
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sig * n)) ** 2)
        inits[f'gauss_{i}'] = g / g.sum()

    cos_env = 0.5 * (1 + np.cos(np.pi * x / max(n - 1, 1)))
    inits['cosine'] = cos_env / cos_env.sum()

    # Oscillatory with envelope
    for freq in [3.0, 7.0, 13.0]:
        osc = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * x / n))
        env = np.exp(-0.5 * ((x - n / 2) / (n / 3)) ** 2)
        combo = osc * env
        if combo.sum() > 1e-10:
            inits[f'osc_{int(freq)}'] = combo / combo.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    sp = rng.random(n)
    sp[sp < 0.6] = 0.0
    if sp.sum() > 1e-10:
        inits['sparse'] = sp / sp.sum()

    # Bi-modal
    bm = np.zeros(n)
    bm[:n // 3] = 1.0
    bm[2 * n // 3:] = 1.0
    if bm.sum() > 0:
        inits['bimodal'] = bm / bm.sum()

    if parent is not None:
        p = np.maximum(parent, 0.0)
        if len(p) == n:
            if p.sum() > 1e-10:
                inits['parent'] = p / p.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()

    return inits
