def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_sparse_conv_rows(a, k_indices):
    """Build sparse matrix rows for specified lag indices.
    Returns (data, row_idx, col_idx, shape) for constraints conv(a,b)[k] = (A @ b)[k].
    """
    n = len(a)
    data = []
    row_idx = []
    col_idx = []
    for idx, k in enumerate(k_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            data.append(a[k - j])
            row_idx.append(idx)
            col_idx.append(j)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(len(k_indices), n))
    return A

                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
