"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with
analytical 1-D subproblems, multi-start initialization, and adaptive
cycle scheduling. Unlike the LP-based approach, this method exploits
the structure of the auto-correlation to reduce each coordinate update
to a fast 1-D optimization, enabling thousands of cycles within the
time budget.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar


def auto_corr(a):
    """Full linear auto-correlation (same as convolve(a, a))."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    c = auto_corr(a)
    return 2.0 * n * c.max() / (s * s)


def coord_1d_minimize(i, a, bg, S, n):
    """
    Minimize f(x) = 2n * peak(x) / (x+S)^2 over x >= 0,
    where peak(x) = max_k (bg[k] + contrib(k,x)).
    Uses scipy minimize_scalar with bounds.
    """
    ai = a[i]
    
    # Precompute the linear coefficients: for each k where k-i is valid and k-i != i
    # contrib(k, x) = 2*x*a[k-i]
    # And the quadratic term at k=2i: contrib = x^2
    
    # For efficiency, precompute the "effective" background and slope for each k
    # peak(x) = max( bg[k] + 2*x*a[k-i] for valid k != 2i, bg[2i] + x^2, max bg[k] for invalid k )
    
    # We can write peak(x) as the max of:
    #   - A quadratic: bg[2i] + x^2
    #   - Several lines: bg[k] + 2*x*a[k-i]
    #   - A constant: max of bg[k] where k-i is out of range
    
    # For the 1-D search, we just evaluate f(x) directly.
    
    def f(x):
        # Compute peak at this x
        # Contribution at k=2i: bg[2i] + x^2
        ki = 2 * i
        peak = bg[ki] + x * x
        # Linear contributions
        # k ranges from max(0, i) to min(2n-2, i+n-1) where k-i is in [0, n-1]
        k_min = max(0, i)  # k-i >= 0 => k >= i
        k_max = min(2 * n - 2, i + n - 1)  # k-i <= n-1 => k <= i+n-1
        if k_min <= k_max:
            ks = np.arange(k_min, k_max + 1)
            js = ks - i  # = k-i, ranges over [0, n-1]
            vals = bg[ks] + 2.0 * x * a[js]
            # Exclude k=2i which we already handled
            mask = ks != ki
            if np.any(mask):
                lin_peak = vals[mask].max()
                peak = max(peak, lin_peak)
        # Constant part: max of bg[k] where k-i is out of [0, n-1]
        # k-i < 0 => k < i; k-i >= n => k >= i+n
        const_ks = np.arange(0, 2*n-1)
        mask_const = (const_ks < i) | (const_ks >= i + n)
        if np.any(mask_const):
            const_peak = bg[const_ks[mask_const]].max()
            peak = max(peak, const_peak)
        
        denom = (x + S) ** 2
        if denom < 1e-30:
            return np.inf
        return 2.0 * n * peak / denom
    
    # Search range
    x_lo = 0.0
    x_hi = max(4.0 * ai, 4.0 * S / n, 0.1)
    
    # Try current value and a few candidates
    candidates = [ai, 0.0, x_hi * 0.5, x_hi * 0.25, x_hi * 0.75]
    best_x = ai
    best_f = f(ai)
    for xc in candidates:
        if xc < x_lo or xc > x_hi:
            continue
        fc = f(xc)
        if fc < best_f:
            best_f = fc
            best_x = xc
    
    # Refine with scipy
    result = minimize_scalar(f, bounds=(x_lo, x_hi), method='bounded',
                             options={'xatol': 1e-10})
    if result.success and result.fun < best_f:
        best_x = result.x
        best_f = result.fun
    
    return best_x


def coordinate_descent(a0, n_cycles, verbose=False):
    """
    Coordinate descent: cycle through all coordinates, each time solving
    a 1-D subproblem.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = objective(a)
    
    no_improve = 0
    max_no_improve = 50
    
    for cycle in range(n_cycles):
        # Compute full auto-correlation
        conv_full = auto_corr(a)
        
        improved = False
        for i in range(n):
            ai = a[i]
            S = s - ai  # sum of other elements
            
            # Compute background: conv_full minus contribution of a[i]
            bg = conv_full.copy()
            ki = 2 * i
            if ki < 2 * n - 1:
                bg[ki] -= ai * ai
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            if k_min <= k_max:
                ks = np.arange(k_min, k_max + 1)
                js = ks - i
                mask = ks != ki
                if np.any(mask):
                    bg[ks[mask]] -= 2.0 * ai * a[js[mask]]
            
            # 1-D minimize
            x_new = coord_1d_minimize(i, a, bg, S, n)
            
            if abs(x_new - ai) > 1e-12:
                a[i] = x_new
                s += x_new - ai
                improved = True
        
        if not improved:
            break
        
        # Recompute s accurately
        s = a.sum()
        
        obj = objective(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and cycle % 10 == 0:
            print(f"    CD cycle {cycle}: obj={obj:.8f}, best={best_obj:.8f}")
        
        if no_improve >= max_no_improve:
            break
    
    return best_a, best_obj

And contrib_new[k] is the contribution of a_i = x to the SAME other elements:
- x^2 if k = 2i
- 2*x*a[k-i] if k-i valid and k != 2i
- 0 otherwise

So conv_new[k] = conv_old[k] - contrib_old[k] + contrib_new[k]

Let bg[k] = conv_old[k] - contrib_old[k] (the "background" with a_i = 0)

Then conv_new[k] = bg[k] + contrib_new[k]

- If k = 2i: conv_new[2i] = bg[2i] + x^2
- If k-i valid and k != 2i: conv_new[k] = bg[k] + 2*x*a[k-i]
- Otherwise: conv_new[k] = bg[k]

And the peak is max over all k.

The objective is f(x) = 2n * max_k (bg[k] + contrib_new[k]) / (s - ai + x)^2
