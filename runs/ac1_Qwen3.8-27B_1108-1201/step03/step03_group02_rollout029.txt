"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses coarse-to-fine LP refinement, multiplicative weight updates,
smoothed gradient descent with Adam, and multi-restart exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M'."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def objective_value(a):
    """Compute the objective (assuming sum(a) is normalized to 1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def multiplicative_update(a, conv, top_k=10, eta=0.1):
    """Multiplicative weight update to reduce peak contributions."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    a_new = a.copy()
    for k in peaks:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        contrib = a[k - j_end:k - j_start + 1][::-1]
        a_new[j_start:j_end + 1] *= np.exp(-eta * contrib / (a.max() + 1e-15))
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new


def optimize_lp_adaptive(a0, time_limit, method='max_sum', max_iters=1000,

    """LP-based iterative refinement with adaptive step size (backtracking)."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = objective_value(a)
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    current_method = method

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()

        if current_method == 'max_sum':
            b = lp_maximize_sum(a, M)
        else:
            b, _ = lp_minimize_peak(a)

        if b is None or b.sum() < 1e-15:
            break
        b = b / b.sum()

        if np.linalg.norm(b - a) < 1e-12:
            break

        # Backtracking line search
        best_step = step
        best_trial_obj = objective_value(a)
        for shrink in range(10):
            trial = (1.0 - step) * a + step * b
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                break
            trial = trial / ts
            trial_obj = objective_value(trial)
            if trial_obj < best_obj - 1e-12:
                best_step = step
                best_trial_obj = trial_obj
                break
            step *= 0.5

        if best_trial_obj >= best_obj - 1e-12:
            no_improve += 1
        else:
            a = project_simplex((1.0 - best_step) * a + best_step * b)
            conv = fast_convolve(a)
            cur_obj = objective_value(a)
            if cur_obj < best_obj - 1e-10:

                no_improve = 0

def softmax_subgradient(a, conv, top_k, temperature):
    """Compute subgradient using top-k peaks with temperature weighting."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = w[i] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg


def optimize_smoothed_grad(a0, time_limit, lr_init=0.01, momentum=0.9,
                           top_k=30, temp_init=0.001, beta1=0.9, beta2=0.999):
    """Adam-optimized smoothed subgradient descent."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = objective_value(a)
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 1
    temp = temp_init
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        sg = softmax_subgradient(a, conv, top_k, temp)
        sg = (sg - sg.mean()) / (np.std(sg) + 1e-15)

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + 1e-8))

        conv = fast_convolve(a)
        cur_obj = objective_value(a)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        t += 1
        if no_improve > 500:
            lr *= 0.5
            no_improve = 0
        if no_improve > 100:
            temp *= 2
        if no_improve > 3000:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def spectral_init(n, num_iters=50):
    """Generate initialization using power iteration on a related operator."""
    a = np.ones(n) / n
    for _ in range(num_iters):
        conv = fast_convolve(a)
        # Use the "pressure" from peaks
        peak_loc = np.argmax(conv)
        shadow = np.zeros(n)
        j_start = max(0, peak_loc - n + 1)
        j_end = min(n - 1, peak_loc)
        shadow[j_start:j_end + 1] = a[peak_loc - j_end:peak_loc - j_start + 1][::-1]
        # Reduce where shadow is large, increase elsewhere
        a_new = a * (1.0 - 0.3 * shadow / (shadow.max() + 1e-15))
        a_new = np.maximum(a_new, 1e-10)
        a_new /= a_new.sum()
        a = a_new
    return a


def upsample(a, new_n):
    """Upsample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_a = np.interp(x_new, x_old, a)
    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s > 1e-15:
        new_a /= s
    else:
        new_a = np.ones(new_n) / new_n
    return new_a


def diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:

                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    # Cosine-based
    for freq in [1, 2, 3, 4]:
        a = 1.0 + 0.7 * np.cos(2.0 * np.pi * freq * x / n + rng.uniform(0, np.pi))
        a = np.maximum(a, 0.01)
        a /= a.sum()
        inits[f'cos_{freq}'] = a

    # Gaussian bump
    center = n * rng.uniform(0.3, 0.7)
    width = n * rng.uniform(0.15, 0.35)
    g = np.exp(-0.5 * ((x - center) / width) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Spectral
    inits['spectral'] = spectral_init(n, num_iters=20)

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Asymmetric ramp
    ramp = np.linspace(0.1, 1.0, n)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    # Double hump
    c1, c2 = n * 0.25, n * 0.75
    w = n * 0.12
    d = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2)
    d /= d.sum()
    inits['double_hump'] = d

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: multi-phase search combining LP refinement,
    multiplicative updates, Adam-optimized gradients, and multi-restart.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            p_norm = p / p.sum()
            print(f"Parent: len={len(p)}, obj={objective_value(p_norm):.6f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        a_ok = np.clip(a, 0.0, 1000.0)
        if a_ok.sum() < 0.01:
            return best_score
        score = float(evaluate_sequence(a_ok.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_ok.copy()
            print(f"  New best: {score:.6f} (n={len(a_ok)})")
        return score

    # === Phase 1: Coarse-to-fine exploration ===
    print("=== Phase 1: Coarse-to-Fine ===")
    t1_start = time.monotonic()

    lengths_c2f = [32, 64, 128, 256, 512]
    current_best = None
    current_best_n = 32

    for target_n in lengths_c2f:
        remain = deadline - time.monotonic() - 600
        if remain < 30:
            break

        if current_best is not None and current_best_n != target_n:
            init_from_parent = upsample(current_best, target_n)
        elif parent is not None:
            init_from_parent = upsample(parent, target_n)
        else:
            init_from_parent = np.ones(target_n) / target_n

        inits = diverse_inits(target_n, rng, parent=init_from_parent if target_n == len(init_from_parent) else None)

        # Prioritize good inits
        priority = ['parent', 'uniform', 'spectral', 'gauss', 'cos_1', 'double_hump', 'random', 'ramp']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:3]

        time_per_init = max(10.0, remain / (len(lengths_c2f) * len(init_names)))

        for name in init_names:
            remain_single = deadline - time.monotonic() - 500
            if remain_single < 5:
                break
            t_alloc = min(time_per_init, remain_single)
            if t_alloc < 3:
                break

            a0 = inits[name]
            a_opt, _ = optimize_lp_adaptive(a0, time_limit=t_alloc, method='max_sum',
                                            step_init=0.35, max_iters=300)
            try_update(a_opt)

            if best_sequence is not None and len(best_sequence) == target_n:
                current_best = best_sequence
                current_best_n = target_n

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # === Phase 2: Intensive LP on best ===
    print("=== Phase 2: Intensive LP ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        phase2_deadline = deadline - 400
        n_best = len(best_sequence)
        a_best = best_sequence / best_sequence.sum()

        for step_init in [0.4, 0.25, 0.5, 0.15, 0.6]:
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(60.0, phase2_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            for method in ['max_sum', 'min_peak']:
                if time.monotonic() > phase2_deadline:
                    break
                t_single = t_alloc / 2.0
                if t_single < 3:
                    break
                a_opt, _ = optimize_lp_adaptive(a_best, time_limit=t_single,
                                                method=method, step_init=step_init,
                                                max_iters=500)
                try_update(a_opt)
                if best_sequence is not None:
                    a_best = best_sequence / best_sequence.sum()

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # === Phase 3: Multiplicative updates ===
    print("=== Phase 3: Multiplicative Updates ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = deadline - 300
        a_base = best_sequence / best_sequence.sum()

        for eta in [0.03, 0.08, 0.015, 0.12]:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(40.0, phase3_deadline - time.monotonic() - 5)
            if t_alloc < 3:
                break
            for top_k in [10, 30, 50]:
                if time.monotonic() > phase3_deadline:
                    break
                t_single = t_alloc / 3.0
                if t_single < 2:
                    break
                a_opt, _ = optimize_multiplicative(a_base, time_limit=t_single,
                                                   eta_init=eta, top_k=top_k)
                try_update(a_opt)
                if best_sequence is not None:
                    a_base = best_sequence / best_sequence.sum()

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # === Phase 4: Adam smoothed gradient ===
    print("=== Phase 4: Adam Gradient ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 200
        a_base = best_sequence / best_sequence.sum()
        n_best = len(a_base)

        for lr in [0.01, 0.005, 0.02, 0.003]:
            if time.monotonic() > phase4_deadline:
                break
            t_alloc = min(40.0, phase4_deadline - time.monotonic() - 5)
            if t_alloc < 3:
                break
            a_opt, _ = optimize_smoothed_grad(a_base, time_limit=t_alloc,
                                              lr_init=lr, top_k=max(20, n_best // 15),
                                              temp_init=0.0005)
            try_update(a_opt)
            if best_sequence is not None:
                a_base = best_sequence / best_sequence.sum()

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # === Phase 5: Multi-restart perturbation ===
    print("=== Phase 5: Multi-Restart ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        phase5_deadline = deadline - 30
        n_best = len(best_sequence)
        a_base = best_sequence / best_sequence.sum()

        restarts = 8
        for r in range(restarts):
            remain = phase5_deadline - time.monotonic()
            if remain < 10:
                break
            t_alloc = remain / (restarts - r)
            if t_alloc < 5:
                break

            scale = 0.002 * (1 + 0.5 * r)
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(a_base + noise)

            # Alternate between LP and multiplicative
            if r % 2 == 0:
                a_opt, _ = optimize_lp_adaptive(a_pert, time_limit=t_alloc,
                                                method='min_peak', step_init=0.25,
                                                max_iters=200)
            else:
                a_opt, _ = optimize_multiplicative(a_pert, time_limit=t_alloc * 0.7,
                                                   eta_init=0.05, top_k=30)
                # Follow up with LP
                t_remain = t_alloc - t_alloc * 0.7
                if t_remain > 3:
                    a_opt, _ = optimize_lp_adaptive(a_opt, time_limit=t_remain,
                                                    method='max_sum', step_init=0.3,
                                                    max_iters=100)
            try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # === Phase 6: Final polish ===
    print("=== Phase 6: Final Polish ===")
    t6_start = time.monotonic()

    if best_sequence is not None:
        phase6_deadline = deadline - 5
        a_base = best_sequence / best_sequence.sum()

        # Very fine LP
        t_alloc = min(30.0, phase6_deadline - time.monotonic())
        if t_alloc > 3:
            a_opt, _ = optimize_lp_adaptive(a_base, time_limit=t_alloc,
                                            method='max_sum', step_init=0.1,
                                            max_iters=300)
            try_update(a_opt)

        # Fine multiplicative
        if time.monotonic() < phase6_deadline:
            t_alloc = min(20.0, phase6_deadline - time.monotonic())
            if t_alloc > 3:
                a_opt, _ = optimize_multiplicative(a_base, time_limit=t_alloc,
                                                   eta_init=0.01, top_k=50)
                try_update(a_opt)

    print(f"Phase 6 done: best={best_score:.6f}, elapsed={time.monotonic()-t6_start:.0f}s")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        if final_score < best_score:
            best_score = final_score
    else:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
