import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def subgradient_step(a, conv, top_k=20, temperature=0.01):
    """Compute subgradient of max(conv) using multi-peak softmax."""
    n = len(a)
    # Get top peaks
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    
    # Softmax weights
    if temperature <= 0:
        # Only use the argmax
        max_idx = np.argmax(peak_vals)
        weights = np.zeros(top_k)
        weights[max_idx] = 1.0
    else:
        logits = (peak_vals - peak_vals.max()) / temperature
        weights = np.exp(logits)
        weights /= weights.sum()
    
    # Compute gradient: g[m] = 2 * sum_i w[i] * a[k_i - m] (with valid indices)
    g = np.zeros(n)
    for i in range(top_k):
        if weights[i] < 1e-12:
            continue
        k = int(peaks[i])
        # g[m] += 2 * w[i] * a[k - m] for valid m (0 <= k-m < n)
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        for m in range(m_start, m_end + 1):
            g[m] += 2.0 * weights[i] * a[k - m]
    
    return g

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_oracle(a, M):
    n = len(a)
    A_mat = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 3.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None
```

"""
Hybrid optimization combining Nesterov-accelerated subgradient descent,
LP-based Frank-Wolfe, and coordinate descent to minimize
2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def local_obj(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, conv, top_k=15, temperature=0.002):
    """Multi-peak subgradient of max(conv)."""
    n = len(a)
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    
    if temperature <= 1e-10:
        max_i = np.argmax(peak_vals)
        weights = np.zeros(top_k)
        weights[max_i] = 1.0
    else:
        logits = (peak_vals - peak_vals.max()) / temperature
        weights = np.exp(logits)
        ws = weights.sum()
        if ws < 1e-30:
            weights = np.zeros(top_k)
            weights[np.argmax(peak_vals)] = 1.0
        else:
            weights /= ws
    
    g = np.zeros(n)
    for i in range(top_k):
        w = weights[i]
        if w < 1e-12:
            continue
        k = int(peaks[i])
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        m_range = range(m_start, m_end + 1)
        a_idx = k - np.array(m_range)
        g[m_range] += 2.0 * w * a[a_idx]
    
    return g

    a = np.maximum(a0, 0.0)
    s = a.sum()

    beta_t = beta
    top_k = top_k_init
    temperature = temp_init
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:

        # Evaluate

            stall = 0
        else:

            a = simplex_project(a + noise)
            v = np.zeros(n)
            stall = 0
            lr = max(0.0005, lr * 2.0)
        
        it += 1
    
    return best_a, best_obj

def build_conv_matrix(a):
    """Build sparse matrix for conv(a, b) as A @ b."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_oracle(a, M, time_limit=3.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A_mat = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def lp_step(a, time_limit=5.0):
    """One LP-based Frank-Wolfe step."""
    n = len(a)
    a = a / a.sum()
    conv = fast_convolve(a)
    M = conv.max()
    
    b = lp_oracle(a, M, time_limit=time_limit)
    if b is None or b.sum() < 1e-15:
        return a, 2.0 * n * M
    
    b = b / b.sum()
    
    # Line search
    best_t = 0.0
    best_obj = 2.0 * n * M
    for t in np.linspace(0.1, 0.8, 8):
        candidate = (1 - t) * a + t * b
        c = fast_convolve(candidate)
        obj = 2.0 * n * c.max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    if best_t > 0:
        a = (1 - best_t) * a + best_t * b
    
    return a, 2.0 * n * fast_convolve(a).max()

def hybrid_optimize(a0, time_limit, verbose=False):
    """
    Hybrid: alternate between momentum subgradient and LP steps.
    """
    n = len(a0)

def coord_descent(a, n_rounds=50, n_trials=7):
    """Coordinate descent polish."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(12345)
    
    for rnd in range(n_rounds):
        # Pick coordinate: focus on high-mass positions
        probs = a / a.sum()
        j = rng.choice(n, p=probs)
        
        old_val = a[j]
        best_local = best_obj
        best_local_val = old_val
        
        for scale in [0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
            a[j] = old_val * scale
            s = a.sum()
            if s < 1e-10:
                a[j] = old_val
                continue
            c = fast_convolve(a)
            obj = 2.0 * n * c.max() / (s * s)
            if obj < best_local:
                best_local = obj
                best_local_val = old_val * scale
        
        a[j] = best_local_val
        if best_local < best_obj:
            best_obj = best_local
            best_a = a.copy()
    
    return best_a, best_obj

    # Parent interpolation
    if parent is not None:

        if ps > 1e-10 and len(p) == n:
            inits['parent'] = p / ps
    
    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    inits['ramp_rev'] = r[::-1] / r.sum()
    
    # Triangle

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    return inits
