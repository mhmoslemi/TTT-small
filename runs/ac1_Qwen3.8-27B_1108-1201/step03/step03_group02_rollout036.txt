def fft_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def subgradient_at_peaks(a, conv, top_k=10, temperature=0.001):
    """Compute subgradient using top-k peaks with softmax weighting."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    # Softmax weights
    peak_vals = conv[peaks]
    scaled = peak_vals / max(temperature, 1e-15)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Compute subgradient
    g = np.zeros(n)
    for i, k in enumerate(peaks):
        if w[i] < 1e-12:

        k = int(k)

    return g

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using convex optimization via
mirror descent (multiplicative weights) with FFT-accelerated subgradients,
combined with smoothed gradient polishing and multi-length exploration.
The key insight is that after normalizing sum(a)=1, the problem becomes
convex (minimize max of convex quadratic forms over the simplex), so
mirror descent converges to the global optimum.
"""

import numpy as np
import time


def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, conv, top_k=50, temperature=0.001):
    """
    Compute (approximate) subgradient of max_k conv_k(a) at point a.
    Uses top-k peaks with softmax weighting, computed via FFT cross-correlation.
    """
    n = len(a)
    size = 2 * n - 1

    if top_k >= size:
        peak_idx = np.arange(size)
        peak_vals = conv.copy()
    else:
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_idx]

    # Softmax weights over selected peaks
    if temperature > 1e-15:
        s = peak_vals / temperature
        s -= s.max()
        w = np.exp(s)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(peak_idx))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= wsum
    else:
        w = np.zeros(len(peak_idx))
        w[np.argmax(peak_vals)] = 1.0

    # Build full weight vector
    w_full = np.zeros(size)
    w_full[peak_idx] = w

    # Cross-correlation via FFT: g[m] = 2 * sum_k w_full[k] * a[k-m]
    corr_len = size + n - 1
    fft_size = 1 << (corr_len - 1).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)

    g = 2.0 * corr[:n]
    return g


def compute_full_softmax_subgradient(a, conv, tau):
    """Compute exact gradient of smoothed objective (1/tau)*log(sum exp(tau*conv_k))."""
    n = len(a)
    size = 2 * n - 1
    scaled = tau * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    psum = p.sum()
    if psum < 1e-30:

        p[np.argmax(conv)] = 1.0
    else:

    corr_len = size + n - 1
    fft_size = 1 << (corr_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size)

    g = 2.0 * corr[:n]
    return g


def mirror_descent_update(a, g, eta):
    """Multiplicative weights update on the simplex."""
    log_w = -eta * g
    log_w -= log_w.max()
    weights = a * np.exp(log_w)
    total = weights.sum()
    if total < 1e-300:
        return np.ones(len(a)) / len(a)
    return weights / total


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def optimize_mirror_descent(a0, time_limit, eta0=1.0, decay_rate=0.9995,

    """
    Mirror descent (multiplicative weights) for min_{a in simplex} max_k (a*a)_k.
    Uses FFT-accelerated subgradients with top-k peaks and adaptive step size.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fft_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        g = compute_subgradient(a, conv, top_k, temp)
        a_new = mirror_descent_update(a, g, eta)

        conv = fft_convolve(a_new)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        iter_count += 1

        # Adaptive step size
        eta *= decay_rate
        if no_improve > 200:
            eta = max(0.01, eta * 0.5)
            no_improve = 0

        # Adapt top_k
        if decay_topk and iter_count % 5000 == 0 and top_k > 5:
            top_k = max(5, top_k // 2)

        # Safety: check time more frequently for long runs
        if iter_count % 1000 == 0:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 0:
                break

    return best_a, best_obj


def optimize_smoothed_grad(a0, time_limit, tau_init=50.0, tau_growth=1.1,
                           eta_init=0.5, eta_decay=0.999):
    """
    Mirror descent with full softmax subgradient (smoothed problem).
    Gradually increases tau for sharper peak focusing.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fft_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    tau = tau_init
    eta = eta_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        g = compute_full_softmax_subgradient(a, conv, tau)
        a = mirror_descent_update(a, g, eta)

        conv = fft_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            tau = min(tau * tau_growth, 500.0)
        else:
            no_improve += 1

        eta *= eta_decay
        if no_improve > 300:
            eta = max(0.005, eta * 0.5)
            no_improve = 0
            tau = max(tau_init, tau * 0.8)

    return best_a, best_obj


def optimize_projected_subgrad(a0, time_limit, eta_init=0.3, eta_decay=0.9995,
                               top_k=20, temp=0.001):
    """Projected subgradient method (Euclidean projection) as alternative."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fft_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    v = np.zeros(n)
    momentum = 0.8

    while time.monotonic() - t0 < time_limit:
        g = compute_subgradient(a, conv, top_k, temp)
        v = momentum * v - eta * g
        a = project_simplex(a + v)

        conv = fft_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        eta *= eta_decay
        if no_improve > 500:
            v *= 0.1
            eta = max(0.01, eta_init * 0.3)
            no_improve = 0

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_direction(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M. Returns normalized b."""
    from scipy.optimize import linprog
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        b = np.maximum(res.x, 0)
        bsum = b.sum()
        if bsum > 1e-15:
            return b / bsum
    return None


def optimize_lp_blend(a0, time_limit, lp_iters=20, blend=0.4):
    """Alternate between LP direction and mirror descent."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fft_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = 0.5

    while time.monotonic() - t0 < time_limit:
        # LP step
        M = conv.max()
        b = lp_direction(a, M)
        if b is not None:
            a_new = (1 - blend) * a + blend * b
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss > 1e-15:
                a_new /= ss
                conv_new = fft_convolve(a_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
                conv = conv_new
        else:
            # Fallback to mirror descent
            g = compute_subgradient(a, conv, 20, 0.001)
            a = mirror_descent_update(a, g, eta)
            conv = fft_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()

        eta *= 0.999

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        if len(pp) != n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(pp)), pp)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
        as_ = a.sum()
        if as_ > 1e-10:
            inits[f'cos_{freq}'] = a / as_

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Bell-shaped
    bell = np.exp(-0.5 * (2 * x / n - 1) ** 2)
    inits['bell'] = bell / bell.sum()

    # Two peaks
    a2 = np.zeros(n)
    a2[n // 4] = 1.0
    a2[3 * n // 4] = 1.0
    from scipy.ndimage import gaussian_filter1d
    a2 = gaussian_filter1d(a2, sigma=n // 20)
    a2 = np.maximum(a2, 0)
    inits['two_peaks'] = a2 / a2.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase convex optimization using mirror descent
    with FFT-accelerated subgradients, smoothed gradient polishing, and
    LP-based direction finding.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return best_score
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)})")
        return score

    # ============ Phase 1: Quick exploration across lengths ============
    print("=== Phase 1: Quick Exploration ===")
    t1_start = time.monotonic()

    lengths_to_try = [512, 256, 1024, 768]
    if parent is not None:
        lp = len(parent)
        lengths_to_try = [lp] + [l for l in lengths_to_try if l != lp]

    for n in lengths_to_try:
        remain = deadline - time.monotonic() - 600
        if remain < 30:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'uniform', 'random', 'cos_1', 'gauss']
        init_names = [nm for nm in priority if nm in inits] + \
                    [nm for nm in init_names if nm not in priority]
        init_names = init_names[:3]

        time_per = max(10.0, remain / (len(lengths_to_try) * len(init_names)))

        for name in init_names:
            remain = deadline - time.monotonic() - 500
            if remain < 10:
                break
            t_alloc = min(time_per, remain)
            if t_alloc < 5:
                break

            a0 = inits[name]
            a_opt, _ = optimize_mirror_descent(a0, time_limit=t_alloc,
                                               eta0=1.0, decay_rate=0.999,
                                               top_k=30, temp=0.001)
            try_update(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # ============ Phase 2: Deep mirror descent on best ============
    print("=== Phase 2: Deep Mirror Descent ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        phase2_deadline = deadline - 300
        n_best = len(best_sequence)

        # Run multiple mirror descent variants with different parameters
        configs = [
            {'eta0': 1.5, 'decay_rate': 0.9995, 'top_k': 50, 'temp': 0.001},
            {'eta0': 0.8, 'decay_rate': 0.999, 'top_k': 20, 'temp': 0.0005},
            {'eta0': 2.0, 'decay_rate': 0.999, 'top_k': 10, 'temp': 0.0001},
            {'eta0': 0.5, 'decay_rate': 0.9995, 'top_k': 100, 'temp': 0.01},
        ]

        for cfg in configs:
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(120.0, phase2_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break
            print(f"  MD config: {cfg}")
            a_opt, _ = optimize_mirror_descent(best_sequence, time_limit=t_alloc, **cfg)
            try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # ============ Phase 3: Smoothed gradient polishing ============
    print("=== Phase 3: Smoothed Gradient ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = deadline - 200

        for tau_init in [30.0, 80.0, 20.0]:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(80.0, phase3_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break
            a_opt, _ = optimize_smoothed_grad(best_sequence, time_limit=t_alloc,
                                              tau_init=tau_init, eta_init=0.4,
                                              eta_decay=0.999)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # ============ Phase 4: Projected subgradient with momentum ============
    print("=== Phase 4: Projected Subgradient ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        phase4_deadline = deadline - 150
        for eta_init in [0.4, 0.2, 0.6]:
            if time.monotonic() > phase4_deadline:
                break
            t_alloc = min(60.0, phase4_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break
            a_opt, _ = optimize_projected_subgrad(best_sequence, time_limit=t_alloc,
                                                  eta_init=eta_init, top_k=30,
                                                  temp=0.001)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # ============ Phase 5: LP-based refinement ============
    print("=== Phase 5: LP Refinement ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        phase5_deadline = deadline - 100
        n_best = len(best_sequence)
        if n_best <= 256:
            t_alloc = min(60.0, phase5_deadline - time.monotonic() - 10)
            if t_alloc > 10:
                a_opt, _ = optimize_lp_blend(best_sequence, time_limit=t_alloc,
                                             lp_iters=20, blend=0.3)
                try_update(a_opt)
        elif n_best <= 128:
            t_alloc = min(60.0, phase5_deadline - time.monotonic() - 10)
            if t_alloc > 10:
                a_opt, _ = optimize_lp_blend(best_sequence, time_limit=t_alloc,
                                             lp_iters=30, blend=0.4)
                try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # ============ Phase 6: Final long mirror descent run ============
    print("=== Phase 6: Final Long Run ===")
    t6_start = time.monotonic()

    if best_sequence is not None:
        t_alloc = max(10.0, deadline - time.monotonic() - 20)
        print(f"  Allocating {t_alloc:.0f}s for final run")
        a_opt, _ = optimize_mirror_descent(best_sequence, time_limit=t_alloc,
                                           eta0=0.8, decay_rate=0.9997,
                                           top_k=30, temp=0.0005)
        try_update(a_opt)

    print(f"Phase 6 done: best={best_score:.6f}, elapsed={time.monotonic()-t6_start:.0f}s")

    # ============ Phase 7: Perturbation + re-optimize ============
    print("=== Phase 7: Perturbations ===")
    t7_start = time.monotonic()

    if best_sequence is not None:
        phase7_deadline = deadline - 10
        n_best = len(best_sequence)
        for scale in [0.003, 0.01, 0.001]:
            if time.monotonic() > phase7_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = max(5.0, phase7_deadline - time.monotonic() - 5)
            if t_alloc < 3:
                break
            a_opt, _ = optimize_mirror_descent(a_pert, time_limit=t_alloc,
                                               eta0=1.0, decay_rate=0.999,
                                               top_k=30, temp=0.001)
            try_update(a_opt)

    print(f"Phase 7 done: best={best_score:.6f}, elapsed={time.monotonic()-t7_start:.0f}s")

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
