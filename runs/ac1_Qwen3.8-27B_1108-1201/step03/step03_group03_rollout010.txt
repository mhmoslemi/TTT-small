def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:

    a = softmax(x), obj = (1/tau) * log(sum_k exp(tau * conv_k(a)))

def subgradient_polish(a0, time_limit, lr_init=0.008, momentum=0.9,
                       top_k=30, temp_scale=0.03):
    """FFT-accelerated subgradient descent with momentum on the simplex."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = 2.0 * n * fft_convolve(a).max()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        pmax = conv.max()
        if pmax < 1e-15:
            break
        
        # Top-k peaks with softmax weighting
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        peak_vals = conv[peaks]
        
        tc = temp_scale * peak_vals / (peak_vals.max() + 1e-30)
        w_peaks = np.exp(tc - tc.max())
        w_peaks /= w_peaks.sum()
        
        # Build full weight vector
        w_full = np.zeros(len(conv))
        for idx in range(len(peaks)):
            w_full[int(peaks[idx])] = w_peaks[idx]

        W = np.fft.rfft(w_full, fft_size)

        if s_new < 1e-15:

            v *= 0.3
            lr = max(0.001, lr * 0.7)
            no_improve = 0

def coordinate_descent(a_in, max_rounds=50, step_frac=0.05):
    """Coordinate descent: move mass between coordinates to reduce peak."""
    a = np.maximum(a_in, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fft_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        conv = fft_convolve(a)
        peak_k = int(np.argmax(conv))
        
        for i in range(n):
            if a[i] < 1e-8:
                continue
            delta = a[i] * step_frac
            if delta < 1e-10:
                continue
            
            # Try moving delta from i to each j
            # The peak is at k*, so we want to reduce a_{k*-i} contribution
            # Best targets: j such that j is NOT at a peak-contributing position
            # Simple: try all j but limit to a random subset for speed
            candidates = np.random.default_rng(rnd * 1000 + i).choice(n, size=min(20, n), replace=False)
            
            for j in candidates:
                if j == i:
                    continue
                a_test = a.copy()
                a_test[i] = a[i] - delta
                a_test[j] = a[j] + delta
                if a_test[i] < 0:
                    continue
                conv_test = fft_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-13:
                    a = a_test
                    best_obj = obj_test
                    improved = True
        
        if not improved:
            break
    
    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()
    
    # Log-uniform (more mass at edges)
    log_u = 1.0 / (x + 1.0)
    inits['log_uniform'] = log_u / log_u.sum()
    
    # Alternating
    alt = np.ones(n)
    alt[1::2] = 0.3
    inits['alternating'] = alt / alt.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                inits['parent_interp'] = np.interp(
                    np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
                inits['parent_interp'] = inits['parent_interp'] / inits['parent_interp'].sum()
    
    return inits
