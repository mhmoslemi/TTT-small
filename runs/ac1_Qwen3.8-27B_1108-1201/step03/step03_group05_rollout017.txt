    max_val = conv.max()
    active = np.where(conv >= max_val - epsilon)[0]
    if len(active) == 0:

    # Random phases with conjugate symmetry
    phases = rng.uniform(-np.pi, np.pi, fft_size)
    # Enforce conjugate symmetry for real output
    phi = np.zeros(fft_size)
    for k in range(fft_size // 2 + 1):
        phi[k] = phases[k]
        phi[fft_size - k] = -phases[k]

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using active-set subgradient descent
with FFT-based gradient computation, phase-retrieval inspired initialization,
adaptive epsilon for active set, backtracking line search, LP direction finding,
and multi-phase multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def active_set_gradient(a, conv, epsilon_frac=0.05):
    """
    Compute subgradient of max(conv) at point a using FFT.
    Active set: indices where conv >= max - epsilon_frac * max.
    Returns gradient vector of length n.
    """
    n = len(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return np.ones(n) / n * 2.0
    
    epsilon = epsilon_frac * max_val
    active = np.where(conv >= max_val - epsilon)[0]
    
    # Build indicator M of active set (length 2n-1)
    M = np.zeros(2 * n - 1)
    M[active] = 1.0
    M /= len(active)
    
    # grad[i] = 2 * sum_k M[k] * a[k-i] (cross-correlation)
    # = 2 * ifft(fft(M) * conj(fft(a)))[:n]
    fft_size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    Mf = np.fft.rfft(M, fft_size)
    cross = np.fft.irfft(Mf * np.conj(A), fft_size)
    grad = 2.0 * cross[:n]
    
    return grad


def logsumexp_gradient(a, conv, beta):
    """Gradient of logsumexp(beta*conv)/beta w.r.t. a via FFT."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    
    # w_k = softmax_k (this is the weight for each conv component)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(W * np.conj(A), fft_size)
    
    return 2.0 * cross[:n]


def optimize_active_gradient(a0, time_limit, epsilon_init=0.05, epsilon_end=0.3,
                             lr_init=0.02, momentum=0.9, seed=0,
                             use_lse=False, beta_init=50, beta_end=300):
    """
    Active-set subgradient descent with Nesterov momentum and adaptive parameters.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        
        # Adaptive epsilon: start tight, loosen over time
        epsilon = epsilon_init + (epsilon_end - epsilon_init) * frac
        
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y / y_sum
        
        conv_y = fast_convolve(y)
        
        # Compute gradient
        if use_lse:
            beta = beta_init + (beta_end - beta_init) * frac
            grad = logsumexp_gradient(y, conv_y, beta)
        else:
            grad = active_set_gradient(y, conv_y, epsilon)
        
        # Nesterov update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        total_iters += 1
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive learning rate and perturbation
        if no_improve >= 300:
            # Try perturbation
            noise_scale = 0.005 * (1.0 + 0.5 * np.sin(total_iters * 0.005))
            noise = rng.standard_normal(n) * noise_scale
            a_pert = project_simplex(a + noise)
            conv_pert = fast_convolve(a_pert)
            if conv_pert.max() < best_obj:
                a = a_pert
                best_obj = conv_pert.max()
                best_a = a.copy()
            else:
                a = a_pert
            v *= 0.3
            lr = max(0.002, lr * 0.6)
            no_improve = 0
        elif no_improve >= 150:
            lr = max(0.003, lr * 0.8)
        
        # Slow decay of learning rate
        lr = max(0.002, lr_init * (1.0 - 0.5 * frac))
        momentum = 0.9 + 0.05 * frac  # Increase momentum over time
    
    return best_a, best_obj


def optimize_lse(a0, time_limit, beta_init=30.0, beta_end=500.0,
                 lr_init=0.015, momentum=0.92, seed=0):
    """Log-sum-exp smoothed gradient descent with Nesterov momentum."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac
        
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y / y_sum
        
        conv_y = fast_convolve(y)
        grad = logsumexp_gradient(y, conv_y, beta)
        
        v = momentum * v - lr * grad
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 400:
            noise_scale = 0.004
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.001, lr * 0.5)
        
        lr = max(0.001, lr_init * (1.0 - 0.6 * frac))
    
    return best_a, best_obj


def lp_direction(a, M, n, time_limit_lp=20):
    """
    Solve LP: max sum(b_j) s.t. b_j >= 0, (a*b)[k] <= M for all k.
    Returns b normalized to sum=1, or None.
    """
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit_lp})
    except Exception:
        return None
    
    if not result.success or result.x is None:
        return None
    
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum


def optimize_lp(a0, time_limit, seed=42):
    """LP-based direction finding phase."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    current_a = a.copy()
    
    t0 = time.monotonic()
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(current_a)
        M = conv_cur.max()
        
        b = None
        for M_factor in [1.0, 1.002, 1.005, 1.01, 1.02]:
            M_try = M * M_factor
            b = lp_direction(current_a, M_try, n, time_limit_lp=15)
            if b is not None:
                break
        
        if b is None:
            noise = rng.standard_normal(n) * 0.01
            current_a = project_simplex(current_a + noise)
            stuck_count += 1
            if stuck_count > 8:
                break
            continue
        
        stuck_count = 0
        improved = False
        
        steps = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
        for t in steps:
            a_try = (1 - t) * current_a + t * b
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()
            
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                current_a = a_try.copy()
                improved = True
                break
            elif obj_try < M * 0.998:
                current_a = a_try.copy()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                improved = True
                break
        
        if not improved:
            for t in [0.02, 0.03]:
                a_try = (1 - t) * current_a + t * b
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = conv_try.max()
                if obj_try < M * 0.9999:
                    current_a = a_try.copy()
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()
                    break
    
    return best_a, best_obj


def peak_shaving(a0, time_limit, seed=0):
    """
    Targeted peak reduction: identify convolution peaks and adjust
    specific elements to reduce them using pair transfers.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        max_val = conv_cur.max()
        
        # Find top peaks
        top_k = min(10, len(conv_cur))
        peak_indices = np.argpartition(conv_cur, -top_k)[-top_k:]
        peak_indices = peak_indices[conv_cur[peak_indices] > max_val * 0.97]
        
        if len(peak_indices) == 0:
            break
        
        improved = False
        for k in peak_indices:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            
            if i_end < i_start:
                continue
            
            contributions = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
            if len(contributions) == 0:
                continue
            
            max_contrib_idx = np.argmax(contributions)
            i_max = i_start + max_contrib_idx
            
            # Try transferring mass from i_max to other positions
            # Prefer positions far from the peak
            for _ in range(5):
                j = rng.integers(0, n)
                if j == i_max:
                    continue
                # Check if this would help
                dist = abs((k - j) - (k - i_max))  # change in "other index"
                
                for frac in [0.1, 0.2, 0.3, 0.05]:
                    amt = a[i_max] * frac
                    if amt < 1e-10:
                        break
                    a_try = a.copy()
                    a_try[i_max] -= amt
                    a_try[j] += amt
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        break
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-12:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
            
            if improved:
                break
        
        iters += 1
        
        # Random pair adjustments when stuck
        if not improved and iters % 200 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-8:
                for frac in [0.05, 0.1, 0.2]:
                    amt = a[i] * frac
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj - 1e-12:
                        best_obj = conv_try.max()
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break
    
    return best_a, best_obj


def phase_retrieval_init(n, rng, n_iters=200):
    """
    Initialize using phase retrieval: alternate between flat spectrum
    and non-negativity constraints.
    """
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Random phases with conjugate symmetry
    phases = rng.uniform(-np.pi, np.pi, fft_size // 2 + 1)
    full_phases = np.zeros(fft_size)
    full_phases[:fft_size // 2 + 1] = phases
    for k in range(1, fft_size // 2 + 1):
        if k < fft_size - k:
            full_phases[fft_size - k] = -phases[k]
    
    a = np.ones(n) / n
    
    for it in range(n_iters):
        # FFT of current a (zero-padded)
        A = np.fft.rfft(a, fft_size)
        
        # Target: minimize variance of |A|^2 (make spectrum as flat as possible)
        # Actually, we want |A[k]|^4 to be flat, which means |A[k]|^2 flat
        # Use a "water-filling" approach: target magnitude
        mag = np.abs(A)
        if mag.sum() < 1e-15:
            break
        
        # Target: make mag as flat as possible
        target_mag = np.full(len(mag), mag.sum() / len(mag))
        # But keep A[0] = sum(a) = 1
        target_mag[0] = A[0].real
        
        # Update: new spectrum = target_mag * phase(A)
        phase_A = A / np.maximum(mag, 1e-15)
        A_new = target_mag * phase_A
        
        # IFFT
        a = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negativity
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
            break
        a = a / s
    
    return a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Phase retrieval
    inits['phase_ret'] = phase_retrieval_init(n, rng, n_iters=300)
    
    # Bell-shaped
    tri = 1.0 / (1.0 + 0.3 * ((x - n / 2) / (n / 5)) ** 2)
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    inits['bell'] = tri
    
    # Raised cosine with edge suppression
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0.001)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random Dirichlet
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # Edge-suppressed random
    a_rand = rng.dirichlet(np.ones(n) * 1.0)
    # Suppress edges
    for i in range(min(5, n // 4)):
        a_rand[i] *= 0.3
        a_rand[n - 1 - i] *= 0.3
    a_rand /= a_rand.sum()
    inits['edge_sup_rand'] = a_rand
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        elif len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
    
    return inits


def kkt_residual(a, conv, epsilon=0.1):
    """
    Compute KKT residual: at optimum, gradient should be constant on support.
    Returns the standard deviation of the gradient on the support.
    """
    n = len(a)
    grad = active_set_gradient(a, conv, epsilon)
    support = a > 1e-10
    if support.sum() < 2:
        return 0.0
    grad_support = grad[support]
    return grad_support.std()
