"""
Combined LP-Frank-Wolfe + Softmax L-BFGS + Nesterov Gradient optimizer
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous: golden-section line search in FW,
softmax-parameterized L-BFGS with annealing temperature for final squeeze,
Nesterov-accelerated projected gradient, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': 8.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def compute_local_obj(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def golden_section_line_search(a, b, n_trials=8):
    """Golden section line search for t in (0,1) minimizing obj((1-t)*a + t*b)."""
    n = len(a)
    best_t = 0.0
    best_obj = 2.0 * n * fast_convolve(a).max()
    # Initial scan
    ts = np.linspace(0.05, 0.95, n_trials)
    objs = np.zeros(n_trials)
    for i in range(n_trials):
        candidate = (1 - ts[i]) * a + ts[i] * b
        objs[i] = 2.0 * n * fast_convolve(candidate).max()
    idx = np.argmin(objs)
    if objs[idx] < best_obj:
        best_t = ts[idx]
        best_obj = objs[idx]
    # Refine around best
    if 0 < idx < n_trials - 1:
        lo = ts[max(0, idx - 2)]
        hi = ts[min(n_trials - 1, idx + 2)]
        golden = 0.618
        c1 = hi - golden * (hi - lo)
        c2 = lo + golden * (hi - lo)
        f1 = 2.0 * n * fast_convolve((1 - c1) * a + c1 * b).max()
        f2 = 2.0 * n * fast_convolve((1 - c2) * a + c2 * b).max()
        for _ in range(6):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - golden * (hi - lo)
                f1 = 2.0 * n * fast_convolve((1 - c1) * a + c1 * b).max()
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + golden * (hi - lo)
                f2 = 2.0 * n * fast_convolve((1 - c2) * a + c2 * b).max()
        t_best = (lo + hi) / 2.0
        f_best = 2.0 * n * fast_convolve((1 - t_best) * a + t_best * b).max()
        if f_best < best_obj:
            best_obj = f_best
            best_t = t_best
    return best_t, best_obj

def softmax_weights(conv_vals, temperature):
    """Compute softmax weights for smooth max."""
    shifted = conv_vals - conv_vals.max()
    w = np.exp(shifted / temperature)
    s = w.sum()
    if s < 1e-300:
        return np.full_like(conv_vals, 1.0 / len(conv_vals))
    return w / s


def smooth_max_val(conv_vals, temperature):
    """Smooth max value."""
    m = conv_vals.max()
    shifted = conv_vals - m
    return m + temperature * np.log(np.exp(shifted / temperature).sum())


def softmax_obj_and_grad(x, temperature, n):
    """
    Compute objective and gradient for softmax-parameterized problem.
    a = softmax(x), minimize 2n * smooth_max(conv(a,a), temperature).
    """
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    a = a / a_sum
    conv = fast_convolve(a)
    obj_val = 2.0 * n * smooth_max_val(conv, temperature)

    # Gradient computation
    w = softmax_weights(conv, temperature)
    # sg[j] = 2 * sum_m w[j+m] * a[m] for m in [0, n-1], j+m in [0, 2n-2]
    N = 1 << (3 * n).bit_length()
    w_pad = np.zeros(N)
    w_pad[:len(w)] = w
    a_pad = np.zeros(N)
    a_pad[:n] = a
    W_ft = np.fft.rfft(w_pad)
    A_ft = np.fft.rfft(a_pad)
    sg_direct = np.fft.irfft(W_ft * np.conj(A_ft), N)[:n]
    sg = 2.0 * sg_direct

    # Chain rule: d/dx_i = a[i] * (sg[i] - sum_j sg[j]*a[j])
    dot_sa = np.dot(sg, a)
    grad = a * (sg - dot_sa)
    grad = 2.0 * n * grad
    return obj_val, grad

                    best_obj = obj_new
                    best_a = a_new.copy()
                    x = res.x
        except Exception:
            pass

        # Also update x from the res even if not best (for warm start)
        try:
            if res.x is not None:
                x = res.x - res.x.max()
                a_check = np.exp(x)
                a_check /= a_check.sum()
                obj_check = 2.0 * n * fast_convolve(a_check).max()
                if obj_check < best_obj:
                    best_obj = obj_check
                    best_a = a_check.copy()
        except Exception:
            pass

        if verbose:
            print(f"  Softmax L-BFGS temp={temp:.5f}, obj={best_obj:.6f}")

    return best_a, best_obj


def nesterov_polish(a_init, time_limit, lr_init=0.004, momentum=0.93,
                    top_k=20, verbose=False):
    """Nesterov-accelerated projected subgradient on simplex."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    decline = 0.9997

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Multi-peak subgradient with adaptive temperature
        top_k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -top_k_actual)[-top_k_actual:]
        peak_vals = conv[peaks]
        temp = max(peak_vals.max() * 0.03, 1e-15)
        shifted = peak_vals - peak_vals.max()
        w = np.exp(shifted / temp)
        w /= w.sum()

        # Compute subgradient: sg[j] = 2 * sum_k w[k] * a[k-j] (valid range)
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            a_kj = np.array([a[k - j] if 0 <= (k - j) < n else 0.0 for j in idx])
            sg[idx] += 2.0 * w[i] * a_kj

        # Nesterov: evaluate gradient at lookahead point
        a_look = a + momentum * v
        a_look = np.maximum(a_look, 0.0)
        s_look = a_look.sum()
        if s_look > 1e-15:
            a_look = a_look / s_look
            conv_look = fast_convolve(a_look)
            peaks_look = np.argpartition(conv_look, -top_k_actual)[-top_k_actual:]
            pv = conv_look[peaks_look]
            temp_l = max(pv.max() * 0.03, 1e-15)
            sh = pv - pv.max()
            wl = np.exp(sh / temp_l)
            wl /= wl.sum()
            sg_l = np.zeros(n)
            for i in range(len(peaks_look)):
                if wl[i] < 1e-12:
                    continue
                k = int(peaks_look[i])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                a_kj = np.array([a_look[k - j] if 0 <= (k - j) < n else 0.0 for j in idx])
                sg_l[idx] += 2.0 * wl[i] * a_kj
            sg = 0.5 * (sg + sg_l)

        v = momentum * v - lr * sg
        a_new = a + v
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new
        a = a_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()

        lr *= decline
        it += 1
        if verbose and it % 2000 == 0:
            print(f"    Nesterov it={it}, obj={best_obj:.6f}, lr={lr:.6f}")

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    # Raised cosine
    x = np.arange(n, dtype=np.float64)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    rc = np.maximum(rc, 1e-10)
    rc /= rc.sum()
    inits['raised_cos'] = rc.copy()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.35 * n)) ** 2)
    g = np.maximum(g, 1e-10)
    g /= g.sum()
    inits['gauss'] = g.copy()

    # Edge-concentrated (W-shape)
    edge = max(1, n // 4)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    # Smooth the edges
    for i in range(min(3, edge)):
        ec[i] = 1.0 - i / (edge * 0.5)
        ec[-1 - i] = 1.0 - i / (edge * 0.5)
    ec = np.maximum(ec, 0.0)
    if ec.sum() > 1e-10:
        ec /= ec.sum()
        inits['edge'] = ec.copy()

    # Palindromic bump (smooth)
    pal = np.exp(-0.5 * ((x - n // 2) / (0.4 * n)) ** 2)
    pal = pal + pal[::-1]  # ensure palindromic
    pal = np.maximum(pal, 1e-10)
    pal /= pal.sum()
    inits['palindromic'] = pal.copy()

    # Two-bump (W-shaped)
    two_bump = np.zeros(n)
    center_gap = n // 4
    two_bump[:n // 3] = np.exp(-0.5 * ((np.arange(n // 3) - n // 6) / (n / 12)) ** 2)
                if n >= 12 else np.ones(max(1, n // 3))
                if n >= 12 else np.ones(max(1, n // 3))
    two_bump[2 * n // 3:] = np.exp(-0.5 * ((np.arange(n // 3) - n // 6) / (n / 12)) ** 2)
    two_bump = np.maximum(two_bump, 0.0)
    if two_bump.sum() > 1e-10:
        two_bump /= two_bump.sum()
        inits['two_bump'] = two_bump.copy()

    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # From parent
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                x_interp = np.linspace(0, 1, n)
                x_parent = np.linspace(0, 1, len(parent))
                pp_interp = np.interp(x_interp, x_parent, pp)
                pp_interp = np.maximum(pp_interp, 0.0)
                ps_interp = pp_interp.sum()
                if ps_interp > 1e-10:
                    inits['parent_interp'] = pp_interp / ps_interp

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: LP-FW scan -> Deep LP-FW -> Softmax L-BFGS -> Nesterov polish.
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}, score_est={2*len(p)*fast_convolve(p/ p.sum()).max():.6f}")
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 384

    # === Phase 1: LP-FW Scan across lengths and inits ===
    print("=== Phase 1: LP-FW Scan ===")
    # Focus around parent length, plus a few others
    lengths_to_try = []
    if parent is not None:
        base = n_parent
        for offset in [0, -16, 16, -32, 32, -48, 48, 64, -64]:
            nl = base + offset
            if 128 <= nl <= 1024 and nl not in lengths_to_try:
                lengths_to_try.append(nl)
    else:
        lengths_to_try = [256, 384, 512]
    # Cap at 6 lengths
    lengths_to_try = lengths_to_try[:6]

    print(f"  Lengths to try: {lengths_to_try}")

    phase1_deadline = deadline - 900  # Reserve 15 min for later phases
    results = []

    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline:
            break
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_interp', 'raised_cos', 'uniform',
                    'palindromic', 'gauss', 'edge', 'two_bump', 'random']
        init_names.sort(key=lambda x: priority.index(x) if x in priority else 99)

        n_inits_to_try = min(4, len(init_names))
        time_per_init = max(10.0, min(60.0, (phase1_deadline - time.monotonic()) / max(1, n_inits_to_try * max(1, len(lengths_to_try) - lengths_to_try.index(n)))))

        for name in init_names[:n_inits_to_try]:
            if time.monotonic() > phase1_deadline - 30:
                break
            t_alloc = min(time_per_init, phase1_deadline - time.monotonic() - 20)
            if t_alloc < 8:
                break

            a0 = inits[name]
            print(f"  n={n}, init={name}, t={t_alloc:.0f}s")
            a_opt, obj_opt = fpw_lp_optimize(a0, time_limit=t_alloc, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"    score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                results.append((score, n, name, a_opt.copy()))

    results.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.6f}")

    # === Phase 2: Deep LP-FW on best ===
    if best_sequence is not None:
        print("=== Phase 2: Deep LP-FW ===")
        remaining = deadline - time.monotonic() - 600
        t_deep = max(10.0, min(600.0, remaining * 0.6))
        if t_deep > 10:
            a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=t_deep,
                                             verbose=True, verbose_every=500)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 3: Try second-best from Phase 1 with LP-FW ===
    if len(results) > 1:
        remaining = deadline - time.monotonic() - 500
        if remaining > 30:
            print("=== Phase 3: Second-best LP-FW ===")
            _, _, _, a_sec = results[1]
            t_alloc = min(120.0, remaining * 0.4)
            a_opt, obj_opt = fpw_lp_optimize(a_sec, time_limit=t_alloc, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 3 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 4: Softmax L-BFGS refinement ===
    if best_sequence is not None:
        print("=== Phase 4: Softmax L-BFGS ===")
        remaining = deadline - time.monotonic() - 300
        t_smax = max(10.0, min(400.0, remaining * 0.5))
        if t_smax > 10:
            n_best = len(best_sequence)
            a_opt, obj_opt = softmax_lbfgs_refine(best_sequence, time_limit=t_smax,
                                                   temperature_start=0.05,
                                                   temperature_end=0.001,
                                                   verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 4 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 5: Nesterov polish ===
    if best_sequence is not None:
        print("=== Phase 5: Nesterov Polish ===")
        remaining = deadline - time.monotonic() - 200
        t_nest = max(10.0, min(300.0, remaining * 0.5))
        if t_nest > 10:
            a_opt, obj_opt = nesterov_polish(best_sequence, time_limit=t_nest,
                                             lr_init=0.003, momentum=0.94,
                                             top_k=15, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 5 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 6: Final LP-FW squeeze ===
    if best_sequence is not None:
        print("=== Phase 6: Final LP-FW Squeeze ===")
        remaining = deadline - time.monotonic() - 120
        t_final = max(5.0, remaining * 0.5)
        if t_final > 5:
            a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=t_final,
                                             verbose=True, verbose_every=300)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 6 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 7: Nesterov again on final ===
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            print("=== Phase 7: Final Nesterov ===")
            t_nest2 = min(60.0, remaining * 0.6)
            a_opt, obj_opt = nesterov_polish(best_sequence, time_limit=t_nest2,
                                             lr_init=0.001, momentum=0.95,
                                             top_k=10, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 7 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 8: Perturbation restarts ===
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 20:
            print("=== Phase 8: Perturbation Restarts ===")
            n_best = len(best_sequence)
            n_restarts = max(1, min(5, int(remaining // 20)))
            for i in range(n_restarts):
                if time.monotonic() > deadline - 30:
                    break
                scale = 0.0005 * (1 + i * 0.3)
                noise = rng.standard_normal(n_best) * scale
                a_pert = np.maximum(best_sequence + noise, 0.0)
                s = a_pert.sum()
                if s < 1e-15:
                    continue
                a_pert = a_pert / s
                t_alloc = min(15.0, deadline - time.monotonic() - 20)
                if t_alloc < 5:
                    break
                a_opt, obj_opt = fpw_lp_optimize(a_pert, time_limit=t_alloc, verbose=False)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Restart {i}: score={score:.6f} (improved!)")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
