"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using active-set LP based
successive linearization, combined with smooth gradient descent and
multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def compute_conv_grad(a, k_indices):
    """
    Compute gradients of conv[k] w.r.t. a for k in k_indices.
    Returns array of shape (len(k_indices), n) where row k is grad of conv[k].
    grad_conv[k][j] = 2*a[k-j] if 0<=k-j<n, else 0
    """
    n = len(a)
    grads = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        # grad[j] = 2*a[k-j] for j where 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            grads[idx, j] = 2.0 * a[k - j]
    return grads


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max(x, beta):
    """Smooth max via log-sum-exp."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    return x_max + np.log(np.exp(shifted).sum()) / beta


def smooth_max_grad(x, beta):
    """Gradient of smooth max."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w = w / w.sum()
    return w

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def compute_conv_grad(a, k_indices):
    """
    Compute gradients of conv[k] w.r.t. a for k in k_indices.
    grad_conv[k][j] = 2*a[k-j] if 0<=k-j<n, else 0
    Returns array of shape (len(k_indices), n).
    """
    n = len(a)
    grads = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            grads[idx, j_start:j_end + 1] = 2.0 * a[k - j_end:k - j_start:-1]
    return grads


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max(x, beta):
    """Smooth max via log-sum-exp."""
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -50, 50)
    return x_max + np.log(np.exp(shifted).sum()) / beta


def smooth_max_weights(x, beta):
    """Softmax weights for smooth max."""
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -50, 50)
    w = np.exp(shifted)
    total = w.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
    else:
        w /= total
    return w


def smooth_gradient_step(a, beta, lr, n_steps):

        w = smooth_max_weights(conv, beta)

        # grad_a[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
        # = 2 * (w cross-correlate a)[j]
        # Compute via FFT: ifft(FFT(w) * FFT(a_reversed))[n-1 : n-1+n]
        W = fft(w, fft_size)
        A = fft(a[::-1], fft_size)
        grad_raw = np.real(ifft(W * A, fft_size))
        grad_a = 2.0 * grad_raw[n - 1:n - 1 + n]

def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

    return rng.dirichlet(np.ones(n) * alpha)

    a /= a.sum()
    return a

                ps = pp.sum()
                if ps > 1e-10:
                    inits['best_up'] = pp / ps

        # Also add candidates of this length
        for sc, cn, name, a in candidates:
            if cn == n:
                inits['cand_%s' % name] = a / a.sum()

        init_names = list(inits.keys())
        priority2 = ['best', 'best_ds', 'best_up', 'parent', 'parent_ds',
                     'parent_up', 'flat_spec', 'cos_taper', 'uniform']
        priority2 = priority2 + ['cand_%s' % nm for nm in init_names if nm.startswith('cand_')]
        init_names = [nm for nm in priority2 if nm in inits] + \
                     [nm for nm in init_names if nm not in priority2]
        init_names = init_names[:4]

        for name in init_names:
            if time.monotonic() > phase2_deadline - 20:
                break
            a0 = inits[name]
            remaining = phase2_deadline - time.monotonic() - 15
            if remaining < 10:
                break

            # Deep LP optimization with adaptive parameters
            max_outer = int(min(500, max(50, remaining * 8)))
            a_opt, local_score = lp_optimize(a0, max_outer=max_outer, top_k=10,
                                             smooth_every=15, smooth_steps=10,
                                             smooth_beta=50.0, smooth_lr=0.003)
            score = score_and_update(a_opt)

            # If we got good, try even deeper with larger top_k
            if np.isfinite(score) and score < 1.51 and time.monotonic() < phase2_deadline - 30:
                a_opt2, local_score2 = lp_optimize(a_opt, max_outer=200, top_k=20,
                                                   smooth_every=10, smooth_steps=5,
                                                   smooth_beta=80.0, smooth_lr=0.002)
                score = score_and_update(a_opt2)

    print("Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: Fine-tuning with alternating LP and smooth gradient
    print("=== Phase 3: Fine-tuning ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 20.0

        if remaining > 10:
            a_cur = best_sequence / best_sequence.sum()

            # Alternating LP with increasing top_k
            for top_k in [5, 10, 15, 20, 30]:
                if remaining < 10:
                    break
                a_opt, _ = lp_optimize(a_cur, max_outer=100, top_k=top_k,
                                       smooth_every=8, smooth_steps=10,
                                       smooth_beta=60.0, smooth_lr=0.002)
                score = score_and_update(a_opt)
                if np.isfinite(score) and score < best_score:
                    a_cur = a_opt
                remaining = deadline - time.monotonic() - 20.0

            # Smooth gradient polishing with decreasing beta (annealing)
            for beta in [100.0, 50.0, 20.0, 10.0, 5.0]:
                if remaining < 8:
                    break
                a_opt, _ = smooth_gradient_step(best_sequence / best_sequence.sum(),
                                                beta=beta, lr=0.01,
                                                n_steps=int(min(200, remaining * 10)))
                score = score_and_update(a_opt)
                if np.isfinite(score) and score < best_score:
                    # Re-do LP from here
                    if remaining > 15:
                        a_opt2, _ = lp_optimize(a_opt, max_outer=80, top_k=15,
                                                smooth_every=0, smooth_steps=0)
                        score_and_update(a_opt2)
                remaining = deadline - time.monotonic() - 20.0

            # Random perturbations + re-optimize
            for _ in range(3):
                if remaining < 15:
                    break
                scale = 0.002 + 0.003 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence / best_sequence.sum() + noise)
                a_opt, _ = lp_optimize(a_pert, max_outer=100, top_k=10,
                                       smooth_every=10, smooth_steps=5,
                                       smooth_beta=40.0, smooth_lr=0.003)
                score = score_and_update(a_opt)
                remaining = deadline - time.monotonic() - 20.0

    # Try alternative lengths if time permits
    if best_sequence is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic() - 40.0
        alt_lengths = [l for l in [192, 256, 320, 384] if l != len(best_sequence)]
        for alt_n in alt_lengths[:1]:
            if remaining < 30:
                break
            inits_alt = generate_inits(alt_n, parent, rng)
            pp = np.interp(np.linspace(0, 1, alt_n),
                           np.linspace(0, 1, len(best_sequence)), best_sequence)
            pp = np.maximum(pp, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits_alt['from_best'] = pp / ps
            a0_alt = inits_alt.get('from_best', inits_alt.get('flat_spec', inits_alt['uniform']))
            a_opt, _ = lp_optimize(a0_alt, max_outer=150, top_k=10,
                                   smooth_every=10, smooth_steps=10,
                                   smooth_beta=50.0, smooth_lr=0.004)
            score_and_update(a_opt)

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
