(1-x^{2n-1})^{1/2} * (1-x)^{-1/2}

(1-x)^{-1/2} = sum_{k=0}^∞ binom(2k,k)/4^k * x^k (all positive coefficients)

(1-x^{2n-1})^{1/2} = sum_{k=0}^∞ binom(1/2,k) * (-x^{2n-1})^k = 1 - (1/2)x^{2n-1} - (1/8)x^{2(2n-1)} - ...

So P(x) ∝ (1 - (1/2)x^{2n-1} - ...) * (sum_{k=0}^∞ c_k x^k) where c_k > 0.

The coefficients of P(x) would be: c_k for k < 2n-1, and c_k - (1/2)*c_{k-(2n-1)} for k >= 2n-1.

For k = 2n-1: coefficient = c_{2n-1} - (1/2)*c_0. Since c_k = binom(2k,k)/4^k, c_0 = 1, c_{2n-1} = binom(4n-2, 2n-1)/4^{2n-1}.

For large n, c_k ~ 1/sqrt(pi*k) (by Stirling's approximation). So c_{2n-1} ~ 1/sqrt(pi*(2n-1)).

The coefficient at position 2n-1 would be approximately 1/sqrt(pi*2n) - 1/2, which is negative for n > ~1.

So the perfectly flat convolution is NOT achievable with non-negative p! The coefficients of P(x) would be negative for some positions.

This means the optimal solution has some structure where the convolution is "as flat as possible" given the non-negativity constraint. The flatness is limited by the requirement that p_i >= 0.

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a combination of:
- Log-sum-exp smoothed gradient descent with adaptive beta
- Direct subgradient on active peaks
- Sinusoidal and structured initializations
- Multi-length exploration and random restarts
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_max = n - 1
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else 0
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, conv, beta):
    """
    Gradient of logsumexp(beta*conv)/beta w.r.t. a.
    """
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def subgradient_max(a, conv, active_frac=0.1):
    """
    Subgradient of max_k conv(a,a)[k] using active set.
    
    Returns the subgradient vector.
    """
    n = len(a)
    max_val = conv.max()
    # Active set: peaks within (1 - active_frac) of max
    threshold = max_val * (1 - active_frac)
    active = np.where(conv >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Subgradient: (1/|A|) * sum_{k in A} grad c[k]
    # grad c[k][i] = 2*a[k-i] for valid i
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # grad[i] += 2 * a[k-i] for i in [i_start, i_end]
        # a[k-i] for i in [i_start, i_end] = a[k-i_start : k-i_end+1] = a[k-i_start : k-i_end+1]
        # k-i_start to k-i_end (reversed)
        idx = k - np.arange(i_start, i_end + 1)
        grad[i_start:i_end+1] += 2.0 * a[idx]
    
    grad /= len(active)
    return grad


def optimize_gradient(a0, time_limit, beta_init=50.0, beta_end=500.0,
                      lr_init=0.01, momentum=0.9, seed=0,
                      use_subgrad_switch=False, switch_frac=0.5):
    """
    Smoothed gradient descent with Nesterov momentum.
    Optionally switches to subgradient method after a fraction of time.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0
    perturb_every = 200

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        
        if use_subgrad_switch and frac > switch_frac:
            # Use subgradient
            conv = fast_convolve(a)
            grad = subgradient_max(a, conv, active_frac=0.05)
        else:
            beta = beta_init + (beta_end - beta_init) * frac
            # Nesterov lookahead
            y = a + momentum * v
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a
            conv_y = fast_convolve(y)
            grad = smoothed_gradient(y, conv_y, beta)

        # Update
        lr = lr_init * (1.0 - 0.6 * frac)
        lr = max(0.0005, lr)
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Perturbation
        if no_improve >= perturb_every:
            noise_scale = 0.005 * (0.5 + 0.5 * frac)
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            perturb_every = int(min(1000, perturb_every * 1.3))

    return best_a, best_obj


def coordinate_descent(a0, time_limit, seed=0):
    """
    Randomized coordinate descent: try moving mass between pairs of positions.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iters_per_check = max(10, 5000 // max(1, n // 100))

    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(iters_per_check):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Try moving frac of a[i] to a[j]
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = a[i] * frac
                if amt < 1e-10:
                    break
                a_try = a.copy()
                a_try[i] -= amt
                a_try[j] += amt
                conv_try = fast_convolve(a_try)
                obj_try = conv_try.max()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
        
        if not improved:
            # Try larger perturbations
            for _ in range(5):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                amt = rng.uniform(0.01, 0.1) * a[i]
                if amt < 1e-10:
                    continue
                a_try = a.copy()
                a_try[i] -= amt
                a_try[j] += amt
                conv_try = fast_convolve(a_try)
                if conv_try.max() < best_obj:
                    best_obj = conv_try.max()
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
        
        if not improved:
            # Give up on this sequence
            break

    return best_a, best_obj
