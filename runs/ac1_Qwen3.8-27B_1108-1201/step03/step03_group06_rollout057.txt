import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_A_ub(a):
    """Build the constraint matrix for LP: (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub

def solve_lp(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = build_A_ub(a)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(num_k, M)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def line_search(a, b, n, max_iter=50):
    """Find optimal step size t in [0, 1] to minimize objective along a->b."""
    best_t = 0.0
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        pt = (1 - mid) * a + mid * b
        obj = 2.0 * n * fast_convolve(pt).max()
        if obj < best_obj:

            best_t = mid
        # The function should be unimodal (convex along the line), so:
        # If obj at mid < obj at lo, search right
        obj_lo = 2.0 * n * fast_convolve((1-lo)*a + lo*b).max()
        if obj < obj_lo:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-6:

    return best_t, best_obj

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_A_ub(a):
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub

def solve_lp(a, M, active_indices=None):
    n = len(a)
    if active_indices is not None:
        A_ub = build_A_ub(a)[active_indices]
        b_ub = np.full(len(active_indices), M)
    else:
        A_ub = build_A_ub(a)
        b_ub = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

    best_obj = 2.0 * n * M

    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        active = np.where(conv >= threshold)[0]

            if b is None or b.sum() < 1e-15:

        b = b / b.sum()

        b_sum = b.sum()
        if b_sum < 1e-15:

        b = b / b_sum

            s_new = a_new.sum()
            if s_new < 1e-15:

"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_A_ub_full(a):
    """Build full constraint matrix for LP: (a*b)[k] <= M for all k=0..2n-2."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub


def solve_lp(a, M, active_indices=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for active k."""
    n = len(a)
    A_full = build_A_ub_full(a)
    if active_indices is not None:
        A_ub = A_full[active_indices]
    else:
        A_ub = A_full
    b_ub = np.full(A_ub.shape[0], M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def symmetrize(v):
    """Enforce v[i] = v[n-1-i] symmetry."""
    return (v + v[::-1]) / 2.0


def normalize_nonneg(v):
    """Clip to non-negative and normalize to sum=1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def lp_optimize_core(a0, time_limit, seed=0, verbosity=0):
    """
    Main LP-based optimizer with:
    - Adaptive active set (constraints near M)
    - Symmetrization after each LP solve
    - Multi-step-size line search
    - Stagnation detection with perturbation
    - Relaxation trick when stuck
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = normalize_nonneg(a)
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    iter_count = 0
    relaxation = 1.0
    step_candidates = [1.0, 0.85, 0.7, 0.55, 0.4, 0.25, 0.1]

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        obj_curr = 2.0 * n * M

        # Adaptive active set: use constraints where conv[k] >= 0.85 * M
        threshold = 0.85 * M
        active = np.where(conv >= threshold)[0]
        if len(active) < 5:
            active = np.argsort(conv)[-max(10, n // 4):]

        # Solve LP
        b = solve_lp(a, M * relaxation, active_indices=active)
        if b is None or b.sum() < 1e-15:
            # Try with full constraints
            b = solve_lp(a, M * relaxation)
            if b is None or b.sum() < 1e-15:
                # Try relaxed M
                b = solve_lp(a, M * 1.05)
                if b is None or b.sum() < 1e-15:
                    break

        b = normalize_nonneg(b)

        # Symmetrize the direction
        b = symmetrize(b)
        b = normalize_nonneg(b)

        # Multi-step-size search
        improved = False
        for t in step_candidates:
            a_new = (1.0 - t) * a + t * b
            a_new = normalize_nonneg(a_new)
            # Also try symmetrized version
            a_new_sym = symmetrize(a_new)
            a_new_sym = normalize_nonneg(a_new_sym)

            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()

            conv_new_sym = fast_convolve(a_new_sym)
            obj_new_sym = 2.0 * n * conv_new_sym.max()

            # Take the better of sym and non-sym
            if obj_new_sym < obj_new:
                obj_new = obj_new_sym
                a_new = a_new_sym

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

        if improved:
            no_improve = 0
            # Gradually tighten relaxation
            relaxation = max(1.0, relaxation * 0.99)
        else:
            no_improve += 1
            if no_improve == 10:
                # Try relaxation
                relaxation = 1.02
            elif no_improve == 25:
                relaxation = 1.05
                # Small perturbation
                noise = rng.standard_normal(n) * 0.003
                a = normalize_nonneg(a + noise)
            elif no_improve >= 50:
                # Bigger perturbation + frequency domain guidance
                conv = fast_convolve(a)
                M = conv.max()
                # Identify peaks to reduce
                peak_indices = np.argsort(conv)[-5:]
                # Perturb in direction that reduces peaks
                noise = rng.standard_normal(n) * 0.01
                a = normalize_nonneg(a + noise)
                a = normalize_nonneg(symmetrize(a))
                relaxation = 1.0
                no_improve = 0
                # Reset best from current if current is still good
                conv_check = fast_convolve(a)
                if 2.0 * n * conv_check.max() < best_obj * 1.01:
                    best_obj = 2.0 * n * conv_check.max()
                    best_a = a.copy()

        iter_count += 1
        if verbosity > 0 and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP[{iter_count}]: obj={best_obj:.8f}, relax={relaxation:.3f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def gradient_optimize(a0, time_limit, seed=0, beta=1000.0, lr=0.01, momentum=0.9):
    """Smooth gradient descent with momentum on the simplex."""
    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        # Softmax weights on the top peaks
        scaled = beta * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Compute gradient: grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
        # Use FFT-based cross-correlation
        fft_size = 1 << (2 * n - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        A = np.fft.rfft(a, fft_size)

        # conv(w, a) gives sum_j w[j] * a[k-j]
        wa_conv = np.fft.irfft(W * A, fft_size)[:2 * n - 1]

        # cross(w, a)[i] = sum_j w[j+i] * a[j]
        w_flipped = w[::-1]
        WF = np.fft.rfft(w_flipped, fft_size)
        cross = np.fft.irfft(WF * A, fft_size)
        # cross[i] for i in range(n) gives sum_j w_flipped[i-j] * a[j] = sum_j w[n-1-(i-j)] * a[j]
        # This is getting complicated, let me just compute gradient directly for the top peaks

        # Simpler: compute gradient from top few peaks
        top_peaks = np.argsort(conv)[-10:]
        grad = np.zeros(n)
        for k in top_peaks:
            wk = w[k]
            for i in range(n):
                j1 = k - i
                j2 = i - k
                if 0 <= j1 < n:
                    grad[i] += wk * a[j1]
                if 0 <= j2 < n and j2 != j1:
                    grad[i] += wk * a[j2]

        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        a_new = normalize_nonneg(symmetrize(a_new))

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 300:
                noise = rng.standard_normal(n) * 0.005
                a = normalize_nonneg(project_simplex(a + noise))
                a = normalize_nonneg(symmetrize(a))
                v = np.zeros(n)
                lr = lr_init
                no_improve = 0

        beta = min(5000.0, beta * 1.002)

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


lr_init = 0.01


def annealing_optimize(a0, time_limit, seed=0):
    """Simulated annealing with LP-guided and random moves."""
    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    T = 0.01
    T_min = 1e-6
    cooling = 0.9995

    t0 = time.monotonic()
    accept_count = 0
    total_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        curr_obj = 2.0 * n * M

        # Generate move: mix of LP direction and random
        if rng.random() < 0.6:
            # LP-guided move
            threshold = 0.9 * M
            active = np.where(conv >= threshold)[0]
            if len(active) < 3:
                active = np.argsort(conv)[-5:]
            b = solve_lp(a, M, active_indices=active)
            if b is not None and b.sum() > 1e-15:
                b = normalize_nonneg(symmetrize(b))
                t_step = rng.uniform(0.1, 0.5)
                a_new = normalize_nonneg((1 - t_step) * a + t_step * b)
            else:
                a_new = normalize_nonneg(a + rng.standard_normal(n) * 0.01)
        else:
            # Random move with various scales
            scale = rng.choice([0.005, 0.01, 0.02, 0.05])
            noise = rng.standard_normal(n) * scale
            a_new = normalize_nonneg(a + noise)
            a_new = normalize_nonneg(symmetrize(a_new))

        conv_new = fast_convolve(a_new)
        new_obj = 2.0 * n * conv_new.max()
        delta = new_obj - curr_obj

        total_count += 1
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            a = a_new
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a_new.copy()
            accept_count += 1

        T = max(T_min, T * cooling)

        if total_count % 500 == 0:
            elapsed = time.monotonic() - t0
            acc_rate = accept_count / max(1, total_count)
            print(f"    SA[{total_count}]: best={best_obj:.8f}, T={T:.6f}, acc={acc_rate:.3f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_bimodal(n, edge_fraction=0.3):
    """Generate a bimodal initialization (high at edges)."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # Bathtub shape: high at edges, low at center
    shape = 1.0 + np.cos(np.pi * x / (n - 1)) ** 2 if n > 1 else np.ones(1)
    # Or: high at edges using distance from center
    dist = np.abs(x - center) / center
    shape = 0.3 + 0.7 * dist ** 2
    shape = shape / shape.sum()
    return shape


def generate_cosine_bell(n):
    """Cosine bell: smooth, peaked in middle but with wide support."""
    x = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    shape = np.cos(np.pi * (x - (n - 1) / 2.0) / (n - 1)) ** 2
    shape = np.maximum(shape, 0.01)
    return shape / shape.sum()


def generate_tukey(n, r=0.5):
    """Tukey (tapered cosine) window."""
    x = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1)
    tukey = np.ones(n)
    edge = int(r * n / 2)
    if edge > 0:
        for i in range(edge):
            val = 0.5 * (1 + np.cos(np.pi * (2 * i / r / n - 1)))
            tukey[i] = val
            tukey[n - 1 - i] = val
    return tukey / tukey.sum()
