def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage approach:
1. LP-based direction finding with tight constraints
2. Smooth gradient (logsumexp) polishing with conjugate gradient

import numpy as np

from scipy.optimize import linprog

def fast_convolve(a):
    """Auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i+n, i] = a
    return C

def solve_lp_direction(a, M, epsilon=1e-12):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M+eps, g >= 0."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M + epsilon)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        g = np.maximum(result.x, 0)
        gs = g.sum()
        if gs > 1e-10:
            return g / gs
    return None

def line_search(a, d, best_obj, n, max_trials=15):
    """Geometric line search for step size."""
    best_t = 0.0
    best_val = best_obj
    # First try large steps
    for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s

        if val < best_val:

            best_t = t

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = T_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute smooth gradient
        conv = fast_convolve(a)
        scaled = conv / T
        smax = scaled.max()
        softmax = np.exp(scaled - smax)
        softmax /= softmax.sum()
        
        # Gradient
        grad = np.zeros(n)
        for k in range(2*n-1):
            if softmax[k] < 1e-14:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += softmax[k] * 2.0 * a[k - idx]
        
        if np.abs(grad).max() < 1e-12:
            break
        
        # CG steps (a few per outer iteration)
        d = -grad.copy()
        alpha = 1.0 / (n * T)  # initial step
        
        for cg_iter in range(10):
            if time.monotonic() - t0 > time_limit:
                break
            a_new = a + alpha * d
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = best_a.copy()
            else:
                a_new = a_new / sn
            
            conv_new = fast_convolve(a_new)
            cur_obj = 2.0 * n * conv_new.max()
            
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            a = a_new
            
            # New gradient
            conv = fast_convolve(a)
            scaled = conv / T
            smax = scaled.max()
            softmax = np.exp(scaled - smax)
            softmax /= softmax.sum()
            grad_new = np.zeros(n)
            for k in range(2*n-1):
                if softmax[k] < 1e-14:
                    continue
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                grad_new[idx] += softmax[k] * 2.0 * a[k - idx]
            
            # CG update
            r_old = grad
            r_new = grad_new
            beta = np.dot(r_new, r_new) / (np.dot(r_old, r_old) + 1e-30)
            d = -r_new + beta * d
            grad = r_new
            
            # Step size
            gnorm = np.abs(grad).max()
            if gnorm > 1e-12:
                alpha = 0.5 / (n * T * gnorm)
        
        # Decrease T
        T = max(T_end, T * 0.995)
        
        if no_improve >= 100:
            break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage approach:
1. LP-based direction finding with tight constraints for global search
2. Smooth gradient (logsumexp continuation) polishing for local refinement
3. Adaptive perturbations to escape local minima
Key improvements: tighter LP margins, better line search, continuation method,
and Nesterov-accelerated gradient steps on the smoothed surrogate.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M, epsilon=1e-12):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M+eps, g >= 0. Returns normalized g or None."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + epsilon)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def line_search(a, d, best_obj, n, max_trials=20):
    """Geometric line search with refinement."""
    best_t = 0.0
    best_val = best_obj
    candidates = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]
    for t in candidates:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv = fast_convolve(a_try)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_t = t
    # Refine around best
    if best_t > 0.01:
        for frac in [0.8, 1.2, 0.5, 1.5, 0.9, 1.1]:
            t_try = best_t * frac
            if t_try <= 0 or t_try > 2.0:
                continue
            a_try = a + t_try * d
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            conv = fast_convolve(a_try)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_t = t_try
    return best_t, best_val


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with adaptive margins and perturbation."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0
    max_stall = 40
    perturb_count = 0
    lp_fail = 0

    while time.monotonic() - t0 < time_limit:
        # Try LP with increasingly relaxed margin
        g = None
        for eps_factor in [0.0, 1e-10, 1e-8, 1e-6]:
            g = solve_lp_direction(a, M, epsilon=M * eps_factor)
            if g is not None:
                break

        if g is None:
            lp_fail += 1
            stall += 5
        else:
            lp_fail = 0
            d = g - a
            inf_norm = np.abs(d).max()
            if inf_norm < 5e-13:
                stall += 3
            else:
                t_best, obj_best = line_search(a, d, best_obj, n)
                if obj_best < best_obj - 1e-13:
                    a = normalize(a + t_best * d)
                    conv = fast_convolve(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    stall = 0
                    if verbose and perturb_count % 100 == 0:
                        print(f"    LP: obj={best_obj:.10f}, t={time.monotonic()-t0:.1f}s")
                else:
                    stall += 1

        # Handle stagnation
        if stall >= max_stall:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000 + perturb_count)
            scale = 0.003 + 0.002 * min(perturb_count, 15)
            a = normalize(best_a + rng.standard_normal(n) * scale)
            conv = fast_convolve(a)
            M = conv.max()
            stall = 0
            perturb_count += 1
            max_stall = min(150, 40 + 15 * perturb_count)
            if perturb_count > 60:
                break

        if lp_fail > 30:
            rng = np.random.default_rng(int(time.monotonic() * 7000) % 1000000)
            a = normalize(best_a + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            M = conv.max()
            lp_fail = 0
            stall = 0

    return best_a, best_obj


def smooth_gradient(a, T):
    """Compute gradient of T * logsumexp(conv(a,a)/T) at point a."""
    conv = fast_convolve(a)
    scaled = conv / T
    smax = scaled.max()
    softmax = np.exp(scaled - smax)
    softmax /= softmax.sum()
    n = len(a)
    grad = np.zeros(n)
    # Only process components with significant weight
    threshold = 1e-12
    mask = softmax > threshold
    for k in np.where(mask)[0]:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += softmax[k] * 2.0 * a[k - idx]
    return grad


def continuation_polish(a0, time_limit, verbose=False):
    """Continuation method: optimize smooth surrogate with decreasing T."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    T_schedule = [0.05, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001, 0.00005]
    no_improve_global = 0

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_phase = time_limit - (time.monotonic() - t0)
        if t_phase < 5:
            break

        # Projected gradient with Nesterov acceleration for this T
        y = a.copy()
        momentum_t = 1.0
        stall_local = 0
        max_stall_local = 50

        while time.monotonic() - t0 < time_limit - 3:
            if stall_local > max_stall_local:
                break

            # Step size
            lr = 0.3 / (n * T)
            lr = min(lr, 0.5)

            # Nesterov lookahead
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * momentum_t * momentum_t)) / 2.0
            beta = (momentum_t - 1.0) / t_new
            momentum_t = t_new

            grad = smooth_gradient(y, T)
            gnorm = np.abs(grad).max()
            if gnorm < 1e-14:
                break

            # Gradient step
            step = lr / max(gnorm, 1e-10) * grad
            y_new = y - step
            y_new = np.maximum(y_new, 0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = best_a.copy()
            else:
                y_new = y_new / sy

            # Inertia update
            a_new = y_new + beta * (y_new - a)
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new
            else:
                a_new = a_new / sa

            a = a_new
            y = y_new

            # Check improvement
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                stall_local = 0
                no_improve_global = 0
            else:
                stall_local += 1
                no_improve_global += 1

            if verbose and stall_local % 50 == 0:
                print(f"    CG T={T:.6f}: obj={best_obj:.10f}, stall={stall_local}")

        # If no improvement at this T, try a quick LP from current best
        if time.monotonic() - t0 < time_limit - 30:
            t_rem = time_limit - (time.monotonic() - t0) - 10
            if t_rem > 20:
                a_lp, obj_lp = lp_optimize(best_a, time_limit=min(t_rem, 60.0))
                if obj_lp < best_obj - 1e-12:
                    best_obj = obj_lp
                    best_a = a_lp.copy()
                    no_improve_global = 0

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # U-shaped: more mass at edges
    for alpha in [0.3, 0.5, 0.8, 1.2]:
        u = 1.0 + alpha * np.cos(np.pi * x / max(1, n - 1))
        u = np.maximum(u, 0.01)
        inits[f'u_{alpha}'] = u / u.sum()

    # Raised cosine window
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    rc = np.maximum(rc, 0.01)
    inits['raised_cos'] = rc / rc.sum()

    # Bimodal
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian at edges (two bumps)
    g1 = np.exp(-0.5 * (x / (n * 0.15)) ** 2)
    g2 = np.exp(-0.5 * ((x - n) / (n * 0.15)) ** 2)
    inits['edge_gauss'] = (g1 + g2 + 0.01) / (g1 + g2 + 0.01).sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) >= n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent_ds'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    inits['parent_up'] = pp / pp.sum()

    return inits
