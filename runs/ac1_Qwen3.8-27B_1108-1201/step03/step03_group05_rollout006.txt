Minimizes 2n*max(conv(a,a))/(sum(a))^2 using a Gerchberg-Saxton-like 
frequency-domain peak compression algorithm combined with smoothed
gradient descent and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_gradient(a, conv, beta):
    """Gradient of logsumexp(beta*conv)/beta w.r.t. a."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    
    w = conv * p
    
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    
    return 2.0 * cross[:n]

import numpy as np

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gs_compress(a, n_iter, gamma=0.7, seed=0):
    """GS-like peak compression via frequency domain manipulation."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    for it in range(n_iter):
        # Compute spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Get auto-conv from spectrum
        c = np.fft.irfft(P, fft_size)[:conv_len]
        
        # Peak compression: power compression
        c_max = c.max()
        if c_max < 1e-15:
            break
        c_norm = c / c_max  # normalize to [0, 1]
        c_comp = c_norm ** gamma  # compress: peaks reduced more
        c_mod = c_comp * c_max  # rescale back
        
        # Preserve sum
        c_sum_orig = c.sum()
        c_sum_mod = c_mod.sum()
        if c_sum_mod > 1e-15:
            c_mod = c_mod * (c_sum_orig / c_sum_mod)
        
        # Transform back to frequency
        P_new = np.fft.rfft(c_mod, fft_size)
        P_new = np.maximum(np.real(P_new), 0)
        
        # Update: scale amplitude, keep phase
        A_mag = np.abs(A)
        A_phase = np.ones_like(A)
        mask = A_mag > 1e-15
        A_phase[mask] = A[mask] / A_mag[mask]
        
        A_new = np.sqrt(P_new) * A_phase
        
        # Back to time
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
    
    return best_a, best_obj


def target_grad_descent(a0, time_limit, seed=0):
    """Gradient descent targeting peaks above mean."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = 0.008
    momentum = 0.9
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        target = conv.mean()  # = 1/(2n-1) approximately
        
        # Compute gradient of sum(max(0, conv - target)^2)
        excess = np.maximum(conv - target, 0)

        E = np.fft.rfft(excess, fft_size)

        cross = np.fft.irfft(E * np.conj(A), fft_size)
        grad = 2.0 * cross[:n]
        
        # Nesterov update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        if conv_new.max() < best_obj:
            a = a_new
            best_obj = conv_new.max()
            best_a = a.copy()
            no_improve = 0
        else:
            a = a + 0.5 * v  # half step
            a = project_simplex(a)
            no_improve += 1
        
        if no_improve > 300:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            lr *= 0.8
    
    return best_a, best_obj


def ls_grad_descent(a0, time_limit, beta_init=50.0, beta_end=400.0,
                    lr_init=0.01, momentum=0.9, seed=0):
    """Log-sum-exp smoothed gradient descent."""
    # (similar to previous but with different params)
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)

    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac

        conv_y = fast_convolve(y)
        scaled = beta * conv_y
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum
        w = conv_y * p
        fft_size = 1 << (3 * n - 1).bit_length()
        A = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        cross = np.fft.irfft(A * np.conj(W), fft_size)
        grad = 2.0 * cross[:n]
        
        v = momentum * v - lr_init * grad
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 300:
            noise = rng.standard_normal(n) * 0.004
            a = project_simplex(a + noise)
            v *= 0.4
            no_improve = 0
    
    return best_a, best_obj


def peak_shave(a0, time_limit, seed=0):
    """Coordinate descent peak shaving."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    iters = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        max_idx = np.argmax(conv)
        
        # Find elements contributing most to the peak
        k = max_idx
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        
        # Find the max contributing pair
        contributions = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
        if len(contributions) == 0:
            break
        max_ci = np.argmax(contributions)
        i_peak = i_start + max_ci
        
        # Try moving mass from i_peak to a random other location
        improved = False
        for attempt in range(5):
            j = rng.integers(0, n)
            if j == i_peak:
                continue
            
            for frac in [0.03, 0.05, 0.08, 0.12]:
                amt = a[i_peak] * frac
                if amt < 1e-10:
                    break
                a_try = a.copy()
                a_try[i_peak] -= amt
                a_try[j] += amt
                conv_try = fast_convolve(a_try)
                if conv_try.max() < best_obj - 1e-12:
                    best_obj = conv_try.max()
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stuck += 1
            if stuck > 200:
                # Random perturbation
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    amt = 0.02 * a[i]
                    a[i] -= amt
                    a[j] += amt
                    a = project_simplex(a)
                stuck = 0
                if stuck > 500:
                    break
        else:
            stuck = 0
        iters += 1
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc /= rc.sum()
    inits['cosine'] = rc
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5*n) / (0.3*n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 6)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random Dirichlet
    inits['dirichlet'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['dirichlet2'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # Parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p /= p.sum()
            inits['parent'] = p
        elif len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            p /= p.sum()
            inits['parent_ds'] = p
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            p /= p.sum()
            inits['parent_up'] = p
    
    return inits
