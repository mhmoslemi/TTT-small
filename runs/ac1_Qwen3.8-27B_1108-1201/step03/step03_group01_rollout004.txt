    """Compute smoothed objective and its gradient for p (sum=1)."""
    n = len(p)
    c = fast_convolve(p)  # length 2n-1
    # Smoothed max: (1/beta) * log(sum(exp(beta * c)))
    beta_c = beta * c
    beta_c -= beta_c.max()  # for numerical stability
    w = np.exp(beta_c)
    w_sum = w.sum()
    w /= w_sum
    obj = 2.0 * n * (c.max() + np.log(w_sum) / beta)  # log-sum-exp trick
    
    # Gradient: sum_k w_k * dc_k/dp_i
    # dc_k/dp_i = p[k-i] if 0 <= k-i < n
    # So grad_i = sum_k w_k * p[k-i] (where valid)
    # This is the convolution of w with p (reversed)!
    # Actually: grad_i = sum_{k: 0<=k-i<n} w_k * p[k-i]

    # grad_i = sum_j w_{i+j} * p_j for j=0..n-1, where w_{i+j} is 0 if i+j >= 2n-1
    
    # This is a cross-correlation of w and p
    # w has length 2n-1, p has length n
    # grad_i = sum_{j=0}^{n-1} w[i+j] * p[j]
    
    # We can compute this as: for each i, dot(w[i:i+n], p)
    # This is a valid matrix multiplication if we construct the right matrix
    # Or we can use FFT: it's the "anti-convolution" or correlation
    
    # Actually, let's think of it differently:
    # grad = correlation of w (padded) with p
    # In numpy: np.correlate(w, p, mode='valid') gives us exactly this!
    # np.correlate(w, p, 'valid')[i] = sum_j w[i+j] * p[j]

def adam_step(z, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new, m, v

Let me also add one more idea: **the "peak-shifting" approach**. After finding a good solution, identify the peak location(s) and try to "shift" mass away from the indices that contribute to the peak. This is a form of coordinate descent.

For the peak at position k, `c_k = sum_i a[i]*a[k-i]`. To reduce c_k, we can decrease a[i] for the i that maximize a[k-i] (or vice versa). This is like a local move.

Let me code this as a "local search" step:
1. Find the peak position k* 
2. For each i, compute the "contribution" of a[i] to c_k*: it's a[k*-i] (if valid)
3. Move a small amount from a[i] to a[j] where j minimizes the increase in other c_k's

This is essentially a form of coordinate descent / local search.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel multi-phase approach:
1. Smoothed gradient optimization (Adam + beta-annealing) with softmax parameterization
   - Uses softmax approximation of max for smooth gradients
   - Adam optimizer with learning rate warmup and decay
   - Beta annealing from small (mean-like) to large (peak-like)
2. LP-based direction finding for polishing (improved from state of the art)
3. Multi-start with diverse initializations
4. Coordinate descent local search
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def smooth_grad(z, beta):
    """Compute (objective, gradient w.r.t. z) for smoothed max of conv(softmax(z), softmax(z))."""
    p = softmax(z)
    n = len(p)
    c = fast_convolve(p)
    
    bc = beta * c
    bc_max = bc.max()
    w = np.exp(bc - bc_max)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[bc.argmax()] = 1.0
    else:
        w /= w_sum
    
    obj = 2.0 * n * (bc_max / beta + np.log(w_sum) / beta)
    
    # Gradient w.r.t. p
    grad_p = np.correlate(w, p, mode='valid')
    
    # Convert to gradient w.r.t. z (Fisher metric on simplex)
    grad_p_mean = np.dot(p, grad_p)
    grad_z = 2.0 * n * p * (grad_p - grad_p_mean)
    
    return obj, grad_z


def adam_smoothed_optimize(p0, time_limit, verbose=False):
    """
    Optimize using smoothed gradient with Adam and beta annealing.
    """
    n = len(p0)
    p = np.maximum(p0, 1e-10).astype(np.float64)
    p = p / p.sum()
    
    z = np.log(p)
    z = (z - z.mean())  # Center for numerical stability
    
    best_obj = 2.0 * n * fast_convolve(p).max()
    best_p = p.copy()
    
    t0 = time.monotonic()
    
    # Beta schedule: geometric from small to large
    # For n=512, we want beta*mean_diff >> 1 to focus on peaks
    # mean_diff ≈ (peak - mean) ≈ 0.5 * 1/(2n-1) ≈ 5e-4
    # So beta >> 2000 for good peak focusing
    n_stages = 15
    betas = np.geomspace(5, 50000, n_stages)
    
    total_iterations_budget = 3000
    iters_per_stage = total_iterations_budget // n_stages
    
    for si, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        
        # Learning rate: start moderate, decay
        lr_base = 0.5 / np.sqrt(n)
        
        n_iters = max(100, int(iters_per_stage * (1 + 2.0 * (n_stages - 1 - si) / max(1, n_stages - 1))))
        
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit - 3:
                break
            
            obj_s, grad_z = smooth_grad(z, beta)
            t += 1
            
            # LR schedule: warmup for first 10%, then linear decay
            frac = it / max(1, n_iters)
            if frac < 0.1:
                lr = lr_base * (frac / 0.1)
            else:
                lr = lr_base * (1.0 - 0.5 * (frac - 0.1) / 0.9)
            
            # Adam step
            m = 0.9 * m + 0.1 * grad_z
            v = 0.999 * v + 0.001 * grad_z * grad_z
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            
            # Clip z for stability
            z = np.clip(z, -30, 30)
            
            # Update p
            p_new = softmax(z)
            p_new = np.maximum(p_new, 1e-15)
            p_new = p_new / p_new.sum()
            z = np.log(p_new)  # Re-derive z from p for consistency
            
            # Check actual objective
            c = fast_convolve(p_new)
            cur_obj = 2.0 * n * c.max()
            
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_p = p_new.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Stage {si+1}/{n_stages}, beta={beta:.0f}, best_obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_p, best_obj


def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M, A_ub=None):
    n = len(a)
    if A_ub is None:
        A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def lp_optimize(a0, time_limit, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    _, conv = obj_and_conv(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lp_fail = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        A_ub = build_conv_matrix(a)
        M_lp = M * 1.0001
        g, g_sum = solve_lp_direction(a, M_lp, A_ub)

        else:

            if norm_d < 1e-14:

            else:
                # Line search
                best_t = 0.0
                best_t_obj = best_obj
                for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
                    a_try = np.maximum(a + t * d, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:

                    a_try = a_try / s_try
                    obj_try = 2.0 * n * fast_convolve(a_try).max()
                    if obj_try < best_t_obj - 1e-14:

                        best_t = t
                
                if best_t > 0 and best_t_obj < best_obj - 1e-14:

                    M = fast_convolve(a).max()
                    best_obj = 2.0 * n * M

            a = a / a.sum()

def obj_and_conv(a):
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max(), c


def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=50, temp=0.0002):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    _, conv = obj_and_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        for _ in range(50):
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(peak_vals)] = 1.0
            else:
                w /= w_sum
            
            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-12:
                    continue
                k = int(peaks[i])
                j_start = max(0, k - n + 1)
                j_end = min(n - 1, k)
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w[i] * 2.0 * a[k - idx]
            
            v = momentum * v - lr * sg
            a_new = np.maximum(a + v, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = best_a.copy()
            
            _, conv = obj_and_conv(a)
            cur_obj = 2.0 * n * conv.max()
            
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        lr *= 0.995
        temp *= 1.005
        
        if no_improve >= 200:
            break
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Local coordinate descent: for each position, find optimal value.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    _, conv = obj_and_conv(a)
    best_obj = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        # Random order of coordinates
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            # Try perturbing a[i]
            best_val = a[i]
            best_local_obj = best_obj
            
            # Binary search for optimal value of a[i]
            # The objective as a function of a[i] is convex (max of linear functions)
            # in a[i] (given other values fixed and normalized)
            # Actually it's not quite that simple due to normalization...
            
            # Simple approach: try a few values
            for frac in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                a_try = a.copy()
                a_try[i] = a[i] * frac
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                obj_try = 2.0 * n * fast_convolve(a_try).max()
                if obj_try < best_local_obj - 1e-14:
                    best_local_obj = obj_try
                    best_val = a_try[i]
            
            if best_val != a[i]:
                a[i] = best_val
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
        
        _, conv = obj_and_conv(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 5:
            break
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Asymmetric ramp
    ramp = (x + 1.0) ** 2
    inits['ramp2'] = ramp / ramp.sum()
    inits['ramp2_rev'] = inits['ramp2'][::-1].copy()
    
    # Exponential
    exp_a = np.exp(-x / (n * 0.3))
    inits['exp'] = exp_a / exp_a.sum()
    inits['exp_rev'] = inits['exp'][::-1].copy()
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Sine
    sine = (np.sin(np.pi * x / n) ** 2 + 0.05)
    inits['sine2'] = sine / sine.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps
    
    return inits
