```python
    """Compute smoothed objective and its gradient."""
    n = len(a)
    conv = np.convolve(a, a, mode='full')
    # Softmax approximation of max
    max_val = conv.max()
    shifted = T * (conv - max_val)
    exp_vals = np.exp(shifted)
    log_sum = np.log(exp_vals.sum())
    # smoothed max = max_val + log_sum / T
    smooth_max = max_val + np.log(exp_vals.sum()) / T
    
    # Gradient of smooth_max w.r.t. a_j
    # d/d a_j [max_val + log(sum exp(T*conv_k))/T]
    # = (1/T) * sum_k (T*conv_k' * exp(T*conv_k) / sum_l exp(T*conv_l))
    # = sum_k (conv_k' * softmax_weight_k)
    # where conv_k' = sum_j a_j * delta(k - i - j) = a[k-j] (for valid j)
    
    weights = exp_vals / exp_vals.sum()  # softmax weights
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-10:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            grad[j] += weights[k] * a[k - j]
    
    # Now the full objective is 2n * smooth_max / (sum(a))^2
    s = a.sum()
    obj = 2 * n * smooth_max / (s * s)
    
    # Gradient of full objective
    # d/d a_j [2n * smooth_max / s^2]
    # = 2n * (grad_j * s^2 - smooth_max * 2 * s) / s^4
    # = 2n * (grad_j / s - 2 * smooth_max / s^2)
    full_grad = 2 * n * (grad / s - 2 * smooth_max / (s * s))
    
    return obj, full_grad
```

```python
"""
Hybrid optimizer combining smoothed gradient descent (Adam with temperature annealing),
LP-based Frank-Wolfe directions, symmetrization, and multi-length search to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def smoothed_objective_and_grad(a, T):
    """
    Compute smoothed objective and gradient.
    Uses softmax approximation: max(x) ≈ max(x) + log(sum(exp(T*(x - max(x)))))/T
    Gradient computed via FFT-based cross-correlation.
    """
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    
    # Softmax weights
    max_conv = conv.max()
    shifted = T * (conv - max_conv)

    weights = exp_vals / exp_vals.sum()
    
    # Smoothed max
    smooth_max = max_conv + np.log(exp_vals.sum()) / T
    
    # Gradient of smooth_max w.r.t. a:
    # grad[j] = sum_k weights[k] * a[k-j] (convolution structure)
    # This is cross-correlation of weights and a

    A = np.fft.rfft(a, N)
    # corr[k] = sum_m weights[m+k] * a[m], we want grad[j] = sum_k weights[k]*a[k-j] = corr[j]
    grad = np.fft.irfft(W * np.conj(A), N)[:n]
    
    # Objective: 2n * smooth_max / s^2
    obj = 2.0 * n * smooth_max / (s * s)
    
    # Full gradient:
    # d/da_j [2n * smooth_max / s^2] = 2n * (grad_j / s - 2*smooth_max/(s^2))
    # Wait, let me re-derive.
    # f = 2n * S / s^2 where S = smooth_max, s = sum(a)
    # df/da_j = 2n * (dS/da_j * s^2 - S * 2s) / s^4
    #         = 2n * (dS/da_j / s - 2S/s^2)
    full_grad = 2.0 * n * (grad / s - 2.0 * smooth_max / (s * s))
    
    return obj, full_grad


def adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step with projection to non-negative simplex."""
    n = len(a)
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new = a_new / s
    return a_new, m, v


def symmetrize(a):
    """Symmetrize: (a + reverse(a)) / 2."""
    return (a + a[::-1]) / 2.0


def build_conv_matrix(a):
    """Build sparse matrix for LP oracle."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(a, M, time_limit=3.0):
    """LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns normalized b."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        b = np.maximum(res.x, 0.0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return None

    best_a = a.copy()

def generate_diverse_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Symmetric edge-concentrated
    decay = np.exp(-x / (n * 0.15))
    decay = decay + decay[::-1]
    inits['edge_sym'] = decay / decay.sum()
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - (n-1)/2) / (n * 0.25)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Triangle
    tri = np.minimum(x + 1, n - x)
    inits['triangle'] = tri / tri.sum()
    
    # Two-peak (W-shape)
    peak_pos = n // 4
    w = np.exp(-0.5 * ((x - peak_pos) / (n * 0.1)) ** 2) + \
        np.exp(-0.5 * ((x - (n - 1 - peak_pos)) / (n * 0.1)) ** 2)
    inits['two_peak'] = w / w.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sparse random
    sp = rng.random(n)
    sp[sp < 0.6] = 0.0
    if sp.sum() > 1e-10:
        inits['sparse'] = sp / sp.sum()
    
    # From parent
    if parent is not None:
        p = np.maximum(parent, 0.0)
        ps = p.sum()
        if ps > 1e-10 and len(p) == n:
            inits['parent'] = p / ps
        elif ps > 1e-10:
            # Interpolate
            p_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            inits['parent_interp'] = p_interp / p_interp.sum()
    
    return inits
```

```python
def compute_grad(weights, a):
    n = len(a)
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.fft(a, N)
    W = np.fft.fft(weights, N)
    corr = np.fft.ifft(np.conj(A) * W).real
    # grad[j] = 2 * corr[(-j) % N]
    indices = [(-j) % N for j in range(n)]
    grad = 2.0 * corr[indices]
    return grad
```
