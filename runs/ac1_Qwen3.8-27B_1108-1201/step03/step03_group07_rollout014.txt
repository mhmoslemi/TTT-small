Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Frequency-domain inspired initializations
2. Smooth gradient optimization (L-BFGS-B with softmax approximation)
3. Linear Programming Method (LPM) for guaranteed descent
4. Adaptive active set refinement

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    return np.real(ifft(A * A, N))[:size]

def smooth_max(x, beta):
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -50, 50)
    e = np.exp(shifted)
    total = e.sum()
    if total < 1e-30:
        return x_max, np.zeros_like(x)
    w = e / total
    sm = x_max + np.log(total) / beta
    return sm, w

def grad_conv(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 64).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(W * A, N))[:n] * 2.0
    return grad

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lpm_step(f, M, top_k=None):
    """
    One step of LPM: find g >= 0 maximizing sum(g) subject to (f*g)[k] <= M for all k.
    Only uses top_k constraints for efficiency.
    """
    n = len(f)
    conv = fast_convolve(f)
    
    if top_k is not None and top_k < 2*n-1:
        # Use only the top_k largest constraints
        k_active = np.argsort(conv)[-top_k]
        active_conv = conv[k_active]
    else:
        k_active = np.arange(2*n-1)
        active_conv = conv
    
    M_eff = M  # Use the same M for all active constraints
    
    # Build constraint matrix for active constraints only
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((len(k_active), n))
    for i, k in enumerate(k_active):
        # For constraint k: sum_j f[k-j]*g[j] <= M_eff
        # f[k-j] is valid when 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_ub[i, j] = f[k - j]
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=np.full(len(k_active), M_eff),
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

def freq_init(n, rng):
    """Initialize with flat power spectrum."""
    N = 1 << (n - 1).bit_length()
    A = np.zeros(N, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2)
    for k in range(1, N // 2 + 1):
        A[k] = np.exp(1j * phases[k-1])
        if k < N - k:
            A[N - k] = np.exp(-1j * phases[k-1])
    a = np.real(ifft(A, N))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

def optimize_smooth(a0, beta, maxiter=300):
    """Smooth gradient optimization."""
    n = len(a0)
    
    def obj(x):
        a = project_simplex(x)
        conv = fast_convolve(a)
        sm, _ = smooth_max(conv, beta)
        return 2.0 * n * sm
    
    def jac(x):
        a = project_simplex(x)
        conv = fast_convolve(a)
        sm, w = smooth_max(conv, beta)
        g = grad_conv(a, w)
        # Simplified gradient (ignoring projection Jacobian)
        return g
    
    result = minimize(obj, a0, jac=jac, method='L-BFGS-B',
                      bounds=[(0, None)]*n,
                      options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-12})
    
    a_opt = np.maximum(result.x, 0)
    a_opt /= a_opt.sum()
    return a_opt

def optimize_pg(a0, beta, lr=0.05, n_iter=200):
    """Projected gradient descent."""
    n = len(a0)
    a = a0.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max(conv, beta)
        obj = 2.0 * n * sm
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        g = grad_conv(a, w)
        
        # Backtracking line search
        t = lr
        for _ in range(10):
            a_new = project_simplex(a - t * g)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-12:

            t *= 0.5
        
        a = a_new
        lr *= 0.999
    
    return best_a, best_obj

from scipy.fft import fft, ifft

_best_sequence = None
_best_score = float('inf')
_t_start = 0.0
_deadline = 0.0


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    return np.real(ifft(A * A, N))[:size]


def smooth_max(x, beta):
    """Log-sum-exp approximation of max with weights."""
    x_max = x.max()
    shifted = np.clip(beta * (x - x_max), -60, 60)
    e = np.exp(shifted)
    total = e.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = e / total
    sm = x_max + np.log(total) / beta
    return sm, w


def grad_conv_weighted(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a using FFT."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(W * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def freq_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    N = 1 << (n - 1).bit_length()
    A = np.zeros(N, dtype=complex)
    A[0] = np.sqrt(n)
    phases = rng.uniform(0, 2 * np.pi, N // 2)
    for k in range(1, N // 2 + 1):
        A[k] = np.exp(1j * phases[k - 1])
        if k < N - k:
            A[N - k] = np.exp(-1j * phases[k - 1])
    a = np.real(ifft(A, N))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def freq_init_weighted(n, rng, alpha=0.5):
    """Weighted frequency init: |A[k]| = (1 + alpha*cos(k*pi/n))."""
    N = 1 << (n - 1).bit_length()
    A = np.zeros(N, dtype=complex)
    A[0] = n
    phases = rng.uniform(0, 2 * np.pi, N // 2)
    for k in range(1, N // 2 + 1):
        mag = 1.0 + alpha * np.cos(k * np.pi / n)
        mag = max(mag, 0.1)
        A[k] = mag * np.exp(1j * phases[k - 1])
        if k < N - k:
            A[N - k] = mag * np.exp(-1j * phases[k - 1])
    a = np.real(ifft(A, N))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def rand_dirichlet_init(n, rng, alpha=0.3):
    """Random Dirichlet initialization."""
    a = rng.dirichlet(np.ones(n) * alpha)
    return a


def optimize_lbfgsb(a0, beta, maxiter=300):
    """L-BFGS-B with simplex projection in the objective."""
    n = len(a0)

    def obj(x):
        a = np.maximum(x, 0.0)
        s = a.sum()
        if s < 1e-15:
            return 1e10
        a = a / s
        conv = fast_convolve(a)
        sm, _ = smooth_max(conv, beta)
        return 2.0 * n * sm

    def jac(x):
        a = np.maximum(x, 0.0)
        s = a.sum()
        if s < 1e-15:
            return np.zeros(n)
        a = a / s
        conv = fast_convolve(a)
        sm, w = smooth_max(conv, beta)
        g = grad_conv_weighted(a, w)
        # Adjust for normalization: d(a/s)/da_i = (1/s - a_i/s^2)
        factor = (1.0 / s - a / s**2)
        return g * factor

    result = minimize(obj, a0, jac=jac, method='L-BFGS-B',
                      bounds=[(0, None)] * n,
                      options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-14})
    a_opt = np.maximum(result.x, 0.0)
    s = a_opt.sum()
    if s > 1e-15:
        a_opt /= s
    else:
        a_opt = a0.copy()
    return a_opt


def optimize_pg(a0, beta, lr=0.06, n_iter=200):
    """Projected gradient descent on simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_obj = float('inf')
    best_a = a.copy()

    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if obj < 1e-12:
            break

        g = grad_conv_weighted(a, w)
        g = np.maximum(g, -10.0)  # Clip extreme gradients

        t = lr
        for _ in range(12):
            a_new = project_simplex(a - t * g)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj * (1 + 1e-10) + 1e-14:
                break
            t *= 0.5

        a = a_new
        lr *= 0.9992

    return best_a, best_obj


def lpm_solve(f, M, top_k=30):
    """
    Solve LPM LP: maximize sum(g) s.t. g>=0, (f*g)[k] <= M for top_k lags.
    Returns g or None.
    """
    n = len(f)
    conv_ff = fast_convolve(f)

    if top_k < 2 * n - 1:
        k_active = np.argsort(conv_ff)[-top_k]
    else:
        k_active = np.arange(2 * n - 1)

    # Build sparse-ish constraint matrix
    A_ub = np.zeros((len(k_active), n))
    for i, k in enumerate(k_active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_ub[i, j] = f[k - j]

    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(len(k_active), M)

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lpm_step_full(a, n_lpm_steps=5, max_top_k=40):
    """
    Perform multiple LPM steps on sequence a (normalized to sum=1).
    Returns improved sequence.
    """
    n = len(a)
    for step in range(n_lpm_steps):
        if time.monotonic() > _deadline:
            break

        conv = fast_convolve(a)
        M = conv.max()

        g = lpm_solve(a, M, top_k=min(max_top_k, 2 * n - 1))

        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        # Try mixing with decreasing t
        best_t = None
        best_new = None
        best_new_score = float('inf')

        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            conv_new = fast_convolve(a_new)
            new_max = conv_new.max()
            new_score = 2.0 * n * new_max
            if new_score < best_new_score:
                best_new_score = new_score
                best_t = t
                best_new = a_new.copy()

        if best_new is not None and best_new_score < 2.0 * n * M:
            a = best_new
        else:
            # Try smaller steps
            for t in [0.01, 0.02, 0.03]:
                a_new = (1 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
                conv_new = fast_convolve(a_new)
                if conv_new.max() < M:
                    a = a_new
                    break

    return a


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            s = p.sum()
            if s > 1e-10:
                inits['parent'] = p / s
        else:
            p = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            p = np.maximum(p, 0)
            s = p.sum()
            if s > 1e-10:
                inits['parent_interp'] = p / s

    inits['freq'] = freq_init(n, rng)
    inits['freq_w'] = freq_init_weighted(n, rng, alpha=0.7)
    inits['freq_w2'] = freq_init_weighted(n, rng, alpha=0.3)
    inits['dirichlet'] = rand_dirichlet_init(n, rng, 0.3)
    inits['dirichlet_sparse'] = rand_dirichlet_init(n, rng, 0.1)
    inits['dirichlet_conc'] = rand_dirichlet_init(n, rng, 2.0)

    # Triangular window
    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining smooth gradients,
    LPM guaranteed descent, and multi-length exploration.
    """
    global _best_sequence, _best_score, _t_start, _deadline
    _best_sequence = None
    _best_score = float('inf')
    _t_start = time.monotonic()
    _deadline = _t_start + budget_s - 30.0

    rng = np.random.default_rng(seed)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print("Parent loaded: n=%d, score pre-check..." % len(parent))
            eval_score(p)
    except Exception:
        pass

    print("Starting search. Budget: %ds. Deadline at t=%.0fs" % (budget_s, _deadline - _t_start))

    # =========================================================
    # Phase 1: Quick scan over lengths and initializations
    # =========================================================
    print("\n=== Phase 1: Quick Scan ===")
    lengths = [128, 192, 256, 320, 384, 512]
    phase1_end = _t_start + 200.0

    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_end - 20:
            break
        inits = generate_inits(n, parent, rng)

        # Priority order
        priority = ['parent', 'parent_interp', 'freq', 'freq_w', 'dirichlet',
                    'dirichlet_sparse', 'triangle', 'gauss', 'bimodal', 'uniform']
        ordered = [nm for nm in priority if nm in inits] + \
                  [nm for nm in inits if nm not in priority]

        # Limit inits per length
        n_inits_to_try = 4
        if n <= 256:
            n_inits_to_try = 5

        t_per_init = max(8.0, 160.0 / (len(lengths) * 5))

        for name in ordered[:n_inits_to_try]:
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(t_per_init, phase1_end - time.monotonic() - 5)
            if t_alloc < 3:
                break

            a0 = inits[name]
            t0 = time.monotonic()
            # Quick PG
            n_iters = int(min(200, t_alloc * 40))
            a_opt, obj = optimize_pg(a0, beta=40.0, lr=0.07, n_iter=n_iters)
            sc = eval_score(a_opt)
            if np.isfinite(sc):
                scan_results.append((sc, n, name, a_opt.copy()))
            elapsed = time.monotonic() - t0
            if sc < 1.6:
                print("  n=%d %s: %.6f (%.1fs)" % (n, name, sc, elapsed))

    scan_results.sort(key=lambda x: x[0])
    print("Phase 1 complete. Best: %.6f" % _best_score)

    # =========================================================
    # Phase 2: Deep L-BFGS-B optimization on top candidates
    # =========================================================
    print("\n=== Phase 2: Deep Optimization ===")
    phase2_end = _deadline - 400.0

    # Select top candidates (avoid duplicates at same length)
    selected = []
    seen_lengths = set()
    for sc, n, name, a in scan_results:
        if n not in seen_lengths:
            selected.append((sc, n, name, a))
            seen_lengths.add(n)
        if len(selected) >= 4:
            break

    # Always include the global best if not already there
    if _best_sequence is not None:
        n_best = len(_best_sequence)
        if n_best not in [s[1] for s in selected]:
            selected.append((_best_score, n_best, 'best', _best_sequence.copy()))

    for sc, n, name, a0 in selected:
        if time.monotonic() > phase2_end - 30:
            break

        print("  Optimizing n=%d (init: %s, score=%.6f)" % (n, name, sc))

        a_cur = a0.copy()
        # Multi-beta schedule
        betas = [50.0, 30.0, 15.0, 8.0, 5.0, 3.0, 2.0]
        for beta in betas:
            if time.monotonic() > phase2_end - 20:
                break
            remaining = phase2_end - time.monotonic() - 10
            if remaining < 5:
                break
            maxiter = int(min(400, max(50, remaining / len(betas) * 25)))
            if maxiter < 20:
                break
            a_opt = optimize_lbfgsb(a_cur, beta, maxiter=maxiter)
            eval_score(a_opt)
            a_cur = a_opt

        # LPM refinement after gradient optimization
        if time.monotonic() < phase2_end - 15:
            n_lpm = int(min(15, max(3, (phase2_end - time.monotonic() - 15) / 5)))
            a_lpm = lpm_step_full(a_cur, n_lpm_steps=n_lpm, max_top_k=30)
            eval_score(a_lpm)
            if a_lpm is not None:
                # Re-optimize from LPM result
                if time.monotonic() < phase2_end - 10:
                    a_opt = optimize_lbfgsb(a_lpm, beta=5.0, maxiter=150)
                    eval_score(a_opt)

    print("Phase 2 complete. Best: %.6f" % _best_score)

    # =========================================================
    # Phase 3: Focused LPM + gradient alternation on best
    # =========================================================
    print("\n=== Phase 3: LPM + Gradient Alternation ===")
    if _best_sequence is not None:
        phase3_end = _deadline - 150.0
        a = _best_sequence.copy()
        n = len(a)

        for it in range(30):
            if time.monotonic() > phase3_end:
                break

            # LPM step
            conv = fast_convolve(a)
            M = conv.max()
            g = lpm_solve(a, M, top_k=min(25, 2 * n - 1))

            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                best_mix = None
                best_mix_score = 2.0 * n * M
                for t in [0.03, 0.05, 0.08, 0.1, 0.15, 0.2]:
                    a_mix = (1 - t) * a + t * g
                    a_mix = np.maximum(a_mix, 0.0)
                    s = a_mix.sum()
                    if s < 1e-10:
                        continue
                    a_mix = a_mix / s
                    sc_mix = 2.0 * n * fast_convolve(a_mix).max()
                    if sc_mix < best_mix_score:
                        best_mix_score = sc_mix
                        best_mix = a_mix.copy()
                if best_mix is not None:
                    a = best_mix
                    eval_score(a)

            # Gradient step
            conv = fast_convolve(a)
            sm, w = smooth_max(conv, 20.0)
            grad = grad_conv_weighted(a, w)
            # Backtracking
            t_lr = 0.03
            for _ in range(10):
                a_new = project_simplex(a - t_lr * grad)
                conv_new = fast_convolve(a_new)
                if conv_new.max() < conv.max():
                    a = a_new
                    break
                t_lr *= 0.5
            else:
                a = project_simplex(a - t_lr * grad)
            eval_score(a)

            # Check if we're making progress
            if time.monotonic() > phase3_end - 60:
                break

    print("Phase 3 complete. Best: %.6f" % _best_score)

    # =========================================================
    # Phase 4: Alternative lengths and final polish
    # =========================================================
    print("\n=== Phase 4: Alt Lengths + Polish ===")
    phase4_end = _deadline - 30.0

    if _best_sequence is not None:
        n_best = len(_best_sequence)

        # Try nearby lengths
        alt_lengths = []
        for dl in [-32, -16, 16, 32, 64]:
            alt = n_best + dl
            if 64 <= alt <= 1024 and alt != n_best:
                alt_lengths.append(alt)

        for alt_n in alt_lengths[:3]:
            if time.monotonic() > phase4_end - 60:
                break

            # Interpolate best to new length
            a_interp = np.interp(np.linspace(0, 1, alt_n),
                                 np.linspace(0, 1, n_best),
                                 _best_sequence)
            a_interp = np.maximum(a_interp, 0)
            s = a_interp.sum()
            if s > 1e-10:
                a_interp /= s

            # Optimize
            a_opt = optimize_lbfgsb(a_interp, beta=10.0, maxiter=200)
            eval_score(a_opt)

            if time.monotonic() < phase4_end - 30:
                a_lpm = lpm_step_full(a_opt, n_lpm_steps=5, max_top_k=25)
                eval_score(a_lpm)

        # Final polish on best
        if time.monotonic() < phase4_end - 20:
            n = len(_best_sequence)

            # Very fine L-BFGS-B
            for beta in [2.0, 1.5, 1.0]:
                if time.monotonic() > phase4_end - 10:
                    break
                maxiter = int(min(300, max(50, (phase4_end - time.monotonic() - 10) / 3 * 15)))
                a_opt = optimize_lbfgsb(_best_sequence, beta, maxiter=maxiter)
                eval_score(a_opt)

            # PG with very small steps
            if time.monotonic() < phase4_end - 15:
                n_iters = int((phase4_end - time.monotonic() - 10) * 15)
                n_iters = max(50, min(500, n_iters))
                a_opt, _ = optimize_pg(_best_sequence, beta=8.0, lr=0.01, n_iter=n_iters)
                eval_score(a_opt)

            # Random perturbations
            if time.monotonic() < phase4_end - 10:
                for _ in range(3):
                    if time.monotonic() > phase4_end - 10:
                        break
                    scale = 0.002 + 0.003 * rng.uniform()
                    noise = rng.standard_normal(n) * scale
                    a_pert = project_simplex(_best_sequence + noise)
                    a_opt = optimize_lbfgsb(a_pert, beta=5.0, maxiter=100)
                    eval_score(a_opt)

    # Fallback
    if _best_sequence is None:
        _best_sequence = np.ones(128) / 128
        _best_score = float(evaluate_sequence(_best_sequence.tolist()))

    print("\n=== FINAL: %.6f (n=%d) ===" % (_best_score, len(_best_sequence)))
    return [float(x) for x in _best_sequence]
