"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient optimization
(log-sum-exp approximation of the peak), L-BFGS with temperature annealing,
LP-based direction finding, and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.special import logsumexp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlate(a, b, n):
    """Compute cross-correlation (a ⋆ b)[m] = sum_j a[j]*b[m+j] for m=0..n-1."""
    # This is the "reverse" convolution
    # (a ⋆ b)[m] = sum_{j=0}^{n-1} a[j] * b[m+j]
    # We can compute this as a cross-correlation using FFT
    # Actually, let's think of it as: flip a, then convolve with b, and extract the right part
    # (a_flipped * b)[k] = sum_j a_flipped[j]*b[k-j] = sum_j a[n-1-j]*b[k-j]
    # Let i = n-1-j: = sum_i a[i]*b[k-n+1+i]
    # So (a_flipped * b)[k] = sum_i a[i]*b[k-(n-1)+i]
    # We want: sum_i a[i]*b[m+i] for m=0..n-1
    # This equals (a_flipped * b)[m + n - 1]
    
    fft_size = 1 << (2*n - 1).bit_length()
    a_flipped = a[::-1]
    A = np.fft.rfft(a_flipped, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    # Extract positions n-1 to 2n-2
    return result[n-1:2*n-1]


def smoothed_objective_and_grad(a, T):
    """
    Compute the smoothed objective and its gradient.
    f_T(a) = 2*n*F_T(a) / sum(a)^2
    where F_T(a) = (1/T)*log(sum(exp(T*conv(a,a))))
    
    Returns (f_T, grad_f_T)
    """
    n = len(a)
    conv = fast_convolve(a)
    S = a.sum()
    
    # Log-sum-exp for numerical stability
    T_conv = T * conv
    F_T = logsumexp(T_conv) / T
    
    # p = softmax(T * conv)
    p = np.exp(T_conv - logsumexp(T_conv))
    
    # Gradient of F_T: 2 * cross_correlate(a, p)
    grad_F_T = 2.0 * cross_correlate(a, p, n)
    
    # Gradient of f_T = 2*n*F_T/S^2
    # df/da[m] = 2*n * (grad_F_T[m]*S - 2*F_T*a[m]) / S^3
    grad_f = 2.0 * n * (grad_F_T * S - 2.0 * F_T * a) / (S ** 3)
    
    f = 2.0 * n * F_T / (S ** 2)
    
    return f, grad_f


def exact_objective(a):
    """Compute the exact objective 2*n*max(conv)/(sum)^2."""
    n = len(a)
    conv = fast_convolve(a)
    S = a.sum()
    return 2.0 * n * conv.max() / (S ** 2)


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full cross-convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M):
    """
    Solve LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0.
    """
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def smoothed_lbfgs(a0, time_limit, T_start=50.0, T_end=2000.0, n_anneal_steps=50):
    """
    Optimize using L-BFGS-B on the smoothed objective with temperature annealing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_obj = exact_objective(a)
    
    t0 = time.monotonic()
    
    for step in range(n_anneal_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        frac = step / max(1, n_anneal_steps - 1)
        T = T_start * (T_end / T_start) ** frac
        
        # Define objective and gradient for L-BFGS
        def obj_func(x):
            x = np.maximum(x, 0.0)
            S_x = x.sum()
            if S_x < 1e-15:
                return 1e10
            x_norm = x / S_x
            f, _ = smoothed_objective_and_grad(x_norm, T)
            return f
        
        def grad_func(x):
            x = np.maximum(x, 0.0)
            S_x = x.sum()
            if S_x < 1e-15:
                return np.zeros(len(x))
            x_norm = x / S_x
            _, g = smoothed_objective_and_grad(x_norm, T)
            return g
        
        # Run L-BFGS-B with limited iterations
        max_iter = max(10, int((time_limit - (time.monotonic() - t0)) / 0.5))
        max_iter = min(max_iter, 50)
        
        result = minimize(obj_func, a, jac=grad_func, method='L-BFGS-B',
                         bounds=[(0, None)] * n,
                         options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                                   'ftol': 1e-12, 'gtol': 1e-8})
        
        if result.success or result.fun < 1e10:
            a_new = np.maximum(result.x, 0.0)
            S_new = a_new.sum()
            if S_new > 1e-15:
                a_new = a_new / S_new
                obj = exact_objective(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                a = a_new
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
(1) Multi-resolution scan with smoothed gradient (log-sum-exp) L-BFGS,
(2) Alternating smoothed L-BFGS + LP direction finding with line search,
(3) Coordinate descent polishing,
(4) Final LP refinement.
Exploits FFT-based fast convolution and the structure of the gradient
(gradient of smoothed max-conv = 2 * cross_correlation(a, softmax(T*conv))).
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.special import logsumexp


def fast_convolve(a):
    """Linear auto-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT. O(n log n)."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def cross_correlate_full(a, p):
    """
    Compute (a ⋆ p)[m] = sum_{j=0}^{len(a)-1} a[j] * p[m+j] for m = 0..len(a)-1.
    This is the gradient factor for the smoothed max-convolution.
    """
    n = len(a)
    # We need p padded to length 2n-1 (or we just use what we have)
    # (a ⋆ p)[m] = sum_j a[j]*p[m+j]
    # This is the cross-correlation of a with p.
    # Using FFT: flip a, convolve with p, extract the right part.
    fft_size = 1 << (2 * n - 1).bit_length()
    a_flip = a[::-1].copy()
    A = np.fft.rfft(a_flip, fft_size)
    P = np.fft.rfft(p, fft_size)
    result = np.fft.irfft(A * P, fft_size)
    # (a_flip * p)[k] = sum_j a_flip[j] * p[k-j] = sum_j a[n-1-j] * p[k-j]
    # Let i = n-1-j: = sum_i a[i] * p[k-(n-1)+i]
    # We want sum_i a[i] * p[m+i] for the coefficient of a[i]:
    # This is (a_flip * p)[m + n - 1] for m = 0, 1, ..., n-1
    return result[n - 1: 2 * n - 1]


def smoothed_obj_grad(a, T):
    """
    Compute smoothed objective f_T(a) = 2*n*F_T(a)/S^2 and its gradient.
    F_T(a) = (1/T) * logsumexp(T * conv(a,a))
    
    Returns (f, grad)
    """
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return 1e10, np.zeros(n)
    
    conv = fast_convolve(a)
    Tconv = T * conv
    
    # F_T = (1/T) * logsumexp(Tconv)
    max_Tconv = Tconv.max()
    log_sum_exp = max_Tconv + np.log(np.exp(Tconv - max_Tconv).sum())
    F_T = log_sum_exp / T
    
    # p = softmax(Tconv)
    p = np.exp(Tconv - log_sum_exp)
    
    # Gradient of F_T w.r.t. a: 2 * cross_correlate(a, p)
    # where cross_correlate(a, p)[m] = sum_j a[j]*p[m+j]
    grad_F_T = 2.0 * cross_correlate_full(a, p)
    
    # f = 2*n*F_T / S^2
    f = 2.0 * n * F_T / (S * S)
    
    # grad_f[m] = 2*n * (grad_F_T[m]*S - 2*F_T*a[m]) / S^3
    grad_f = 2.0 * n * (grad_F_T * S - 2.0 * F_T * a) / (S ** 3)
    
    return f, grad_f


def exact_obj(a):
    """Exact objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (S * S)


def normalize(a):
    """Normalize a to sum to 1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def build_cross_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def lp_direction(a, M):
    """
    LP: max sum(g) s.t. conv(a, g) <= M (element-wise), g >= 0.
    Returns normalized g and the optimal sum.
    """
    n = len(a)
    A_ub = build_cross_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return a.copy(), 1.0


def line_search(a, d, best_obj, n, max_evals=30):
    """
    Backtracking + golden section line search for a + t*d.
    d is a direction (may not be a descent direction for the exact obj).
    """
    best_t = 0.0
    best_f = best_obj
    
    # First try a range of step sizes
    candidates = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    
    for t in candidates:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        S = a_try.sum()
        if S < 1e-15:
            continue
        a_try = a_try / S
        f_try = exact_obj(a_try)
        if f_try < best_f - 1e-14:
            best_f = f_try
            best_t = t
    
    # Refine around best_t
    if best_t > 0.005:
        lo = max(0, best_t * 0.8)
        hi = min(1.5, best_t * 1.2)
        for _ in range(4):
            mid1 = lo + (hi - lo) / 3
            mid2 = hi - (hi - lo) / 3
            for t_try in [mid1, mid2]:
                a_try = a + t_try * d
                a_try = np.maximum(a_try, 0.0)
                S = a_try.sum()
                if S < 1e-15:
                    continue
                a_try = a_try / S
                f_try = exact_obj(a_try)
                if f_try < best_f - 1e-14:
                    best_f = f_try
                    best_t = t_try
            # Narrow
            a_try1 = a + mid1 * d
            a_try1 = np.maximum(a_try1, 0.0)
            f1 = exact_obj(a_try1 / a_try1.sum()) if a_try1.sum() > 1e-15 else 1e10
            a_try2 = a + mid2 * d
            a_try2 = np.maximum(a_try2, 0.0)
            f2 = exact_obj(a_try2 / a_try2.sum()) if a_try2.sum() > 1e-15 else 1e10
            if f1 < f2:
                hi = mid2
            else:
                lo = mid1
    
    return best_t, best_f


def smoothed_optimize(a0, time_limit, T_start=100.0, T_end=5000.0, n_anneal=30):
    """
    Smoothed gradient optimization with temperature annealing using L-BFGS-B.
    """
    a = normalize(a0)
    n = len(a)
    best_a = a.copy()
    best_obj = exact_obj(a)
    
    t0 = time.monotonic()
    
    for anneal_step in range(n_anneal):
        if time.monotonic() - t0 > time_limit:
            break
        
        frac = anneal_step / max(1, n_anneal - 1)
        T = T_start * (T_end / T_start) ** frac
        
        # L-BFGS on the smoothed problem
        remaining = time_limit - (time.monotonic() - t0)
        max_iter = max(5, int(remaining / 1.0))
        max_iter = min(max_iter, 30)
        
        def fun(x):
            x_pos = np.maximum(x, 0.0)
            S = x_pos.sum()
            if S < 1e-15:
                return 1e10
            xn = x_pos / S
            f, _ = smoothed_obj_grad(xn, T)
            return f
        
        def gfun(x):
            x_pos = np.maximum(x, 0.0)
            S = x_pos.sum()
            if S < 1e-15:
                return np.zeros(len(x))
            xn = x_pos / S
            _, g = smoothed_obj_grad(xn, T)
            return g
        
        try:
            result = minimize(fun, a, jac=gfun, method='L-BFGS-B',
                            bounds=[(0, None)] * n,
                            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
            if result.fun < 1e9:
                a_new = normalize(result.x)
                obj = exact_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                a = a_new
        except Exception:
            pass
    
    return best_a, best_obj


def lp_optimize(a0, time_limit, verbose=False):
    """
    LP-based iterative optimization.
    """
    a = normalize(a0)
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 100
    
    while time.monotonic() - t0 < time_limit:
        M_lp = M * 1.001
        g, g_sum = lp_direction(a, M_lp)
        
        if g_sum <= 1.0 + 1e-6:
            no_improve += 1
        else:
            d = g - a
            norm_d = np.linalg.norm(d)
            
            if norm_d < 1e-12:
                no_improve += 1
            else:
                t_best, obj_best = line_search(a, d, best_obj, n)
                
                if obj_best < best_obj - 1e-12:
                    a_new = a + t_best * d
                    a = normalize(a_new)
                    conv = fast_convolve(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        
        if no_improve >= max_no_improve:
            # Perturb and restart
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            noise = rng.standard_normal(n) * 0.005
            a = normalize(best_a + noise)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_no_improve = min(200, max_no_improve + 50)
            
    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_sweeps=100):
    """
    Coordinate descent: for each coordinate, find the best value.
    """
    a = normalize(a0)
    n = len(a)
    best_obj = exact_obj(a)
    
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Random order for diversity
        order = rng.permutation(n)
        
        for m in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[m]
            # Try a range of values
            candidates = [
                old_val,
                old_val * 1.5,
                old_val * 0.5,
                old_val * 2.0,
                old_val * 0.25,
                old_val + 0.001,
                old_val - 0.001,
                0.0,
                2.0 * best_obj * a.sum() ** 2 / (2.0 * n) / n  # approximate target
            ]
            candidates = [max(0.0, c) for c in candidates]
            
            best_val = old_val
            best_f = best_obj
            
            # Compute conv once for the base
            a_base = a.copy()
            a_base[m] = 0.0
            conv_base = fast_convolve(a_base) if a_base.sum() > 1e-15 else np.zeros(2*n-1)
            
            for x_val in candidates:
                a_try = a.copy()
                a_try[m] = x_val
                S = a_try.sum()
                if S < 1e-15:
                    continue
                a_try = a_try / S
                f_try = exact_obj(a_try)
                if f_try < best_f - 1e-14:
                    best_f = f_try
                    best_val = x_val
            
            if best_val != old_val:
                a[m] = best_val
                a = normalize(a)
                if best_f < best_obj:
                    best_obj = best_f
    
    return a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['raised_cos'] = rc / rc.sum()
    
    # Root raised cosine
    rrc = np.sqrt(rc)
    inits['root_raised_cos'] = rrc / rrc.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Sine
    s = (np.sin(np.pi * (x + 0.5) / n) + 0.1)
    inits['sine'] = s / s.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            pp = np.maximum(pp, 0)
            pps = pp.sum()
            if pps > 1e-10:
                inits['parent'] = pp / pps
    
    return inits


def upsample(a, new_n):
    """Upsample sequence a to length new_n using interpolation."""
    if len(a) == new_n:
        return a.copy()
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new
