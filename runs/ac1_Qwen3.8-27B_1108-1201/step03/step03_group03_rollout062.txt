                     top_k=30, temp_factor=0.015, decay=0.9998,
                     restart_interval=200, seed=42):
    """Projected subgradient with momentum and periodic restarts."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sg = compute_subgrad(a, conv, top_k=top_k,
                             temperature=conv.max() * temp_factor)
        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= decay

        if no_improve >= restart_interval:
            # Restart with perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, 0.001)
            conv = fast_convolve(a)
            no_improve = 0

    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, lr_init=0.002, top_k=30,
                            temp_factor=0.01, decay=0.9999, seed=42):
    """Mirror descent with KL divergence (exponential weighting)."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sg = compute_subgrad(a, conv, top_k=top_k,
                             temperature=conv.max() * temp_factor)
        # Mirror descent update: a_new[i] prop to a[i] * exp(-lr * sg[i])
        log_a = np.where(a > 1e-30, np.log(a + 1e-30), -30.0)
        log_a_new = log_a - lr * sg
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= a_new_sum

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= decay

        if no_improve >= 300:
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 0)
            a /= a.sum()
            lr = max(lr_init * 0.5, 0.0005)
            conv = fast_convolve(a)
            no_improve = 0

    return best_a, best_obj


def build_lp_matrix(n):
    """Build the constraint matrix for LP: (f*g)_k <= M for all k."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = 1.0  # placeholder, will be filled with f values
    return A


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': 15.0})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_direction_optimize(a0, time_limit, blend_init=0.3, n_lp_iters=50, seed=42):
    """LP-based directional optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init

    for it in range(n_lp_iters):
        if time.monotonic() - t0 > time_limit - 5:
            break
        M = conv.max()
        if M < 1e-15:
            break
        g = solve_lp_direction(a, M, n)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
        else:
            blend = max(0.01, blend * 0.7)

    return best_a, best_obj


def coordinate_descent(a_in, max_rounds=40, seed=42):
    """Coordinate descent: try transferring mass between pairs."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()

    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for i in order:
            if a[i] < 1e-8:
                continue
            # Try moving small fraction to a few random targets
            targets = rng.choice(n, size=min(10, n), replace=False)
            for j in targets:
                if j == i:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    delta = a[i] * frac
                    if delta < 1e-9:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-14:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
        if not improved:
            break

    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    g2 = (np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()

    # Geometric decay
    geo = np.exp(-0.5 * x / n)
    inits['geometric'] = geo / geo.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_resized = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(parent)), p)
                p_resized = np.maximum(p_resized, 0)
                p_rs = p_resized.sum()
                if p_rs > 1e-10:
                    inits['parent'] = p_resized / p_rs

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining subgradient, mirror descent, LP direction,
    and coordinate descent to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            if np.isfinite(par_score):
                best_score = par_score
                best_sequence = p.copy()
    except Exception:
        pass

    results = []

    # === Phase 1: Fast scan over lengths ===
    print("=== Phase 1: Fast Scan ===")
    t0 = time.monotonic()
    phase1_budget = min(350.0, (total_deadline - time.monotonic()) * 0.2)
    phase1_end = t0 + phase1_budget

    lengths = [256, 384, 512, 640, 768, 1024]
    # If parent exists, prioritize its length
    if parent is not None:
        pn = len(parent)
        # Insert parent length near the front
        insert_pos = min(2, len(lengths))
        lengths.insert(insert_pos, pn)
        lengths = sorted(set(lengths))
        # Limit to 7 lengths max
        lengths = lengths[:7]

    print(f"  Lengths to try: {lengths}")

    for n in lengths:
        if time.monotonic() > phase1_end - 15:
            break
        inits = gen_inits(n, parent, rng)
        # Priority: parent, uniform, random, triangle, gauss
        prio = ['parent', 'uniform', 'random', 'triangle', 'gauss', 'bimodal', 'geometric']
        names = [nm for nm in prio if nm in inits][:3]

        t_per = max(15.0, (phase1_end - time.monotonic() - 10) / max(1, len(names)))

        for name in names:
            if time.monotonic() > phase1_end - 8:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 5)
            if t_alloc < 8:
                break

            a0 = inits[name]
            # Fast subgradient
            a_opt, obj_opt = subgrad_optimize(a0, t_alloc * 0.7,
                                              lr_init=0.004, momentum=0.85,
                                              top_k=25, temp_factor=0.02,
                                              decay=0.9997, restart_interval=150,
                                              seed=int(rng.integers(1, 10000)))
            # Quick mirror descent
            a_opt2, obj2 = mirror_descent_optimize(a_opt, t_alloc * 0.3,
                                                   lr_init=0.001, top_k=25,
                                                   temp_factor=0.015,
                                                   decay=0.9998,
                                                   seed=int(rng.integers(1, 10000)))
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")

    # === Phase 2: Deep optimization on top candidates ===
    print("=== Phase 2: Deep Optimization ===")
    t1 = time.monotonic()

    top_list = []
    seen = set()
    for score, n, name, a_opt in results[:6]:
        key = (n, name)
        if key not in seen:
            top_list.append((n, a_opt))
            seen.add(key)
        if len(top_list) >= 4:
            break
    if best_sequence is not None and (len(best_sequence),) not in seen:
        top_list.append((len(best_sequence), best_sequence.copy()))

    phase2_budget = min(800.0, (total_deadline - time.monotonic()) * 0.45)
    phase2_end = t1 + phase2_budget
    t_per_cand = max(30.0, phase2_budget / max(1, len(top_list)))

    for ci, (n, a0) in enumerate(top_list):
        if time.monotonic() > phase2_end - 20:
            break
        t_alloc = min(t_per_cand, phase2_end - time.monotonic() - 15)
        if t_alloc < 15:
            break

        print(f"  Deep optimizing candidate {ci}: n={n}")

        # Subgradient with careful parameters
        t_sg = t_alloc * 0.4
        a_opt, obj_opt = subgrad_optimize(a0, t_sg,
                                          lr_init=0.003, momentum=0.88,
                                          top_k=40, temp_factor=0.012,
                                          decay=0.99985, restart_interval=300,
                                          seed=int(rng.integers(1, 10000)))

        # Mirror descent
        t_md = t_alloc * 0.25
        a_opt2, obj2 = mirror_descent_optimize(a_opt, t_md,
                                               lr_init=0.0008, top_k=40,
                                               temp_factor=0.008,
                                               decay=0.9999,
                                               seed=int(rng.integers(1, 10000)))
        if obj2 < obj_opt:
            a_opt, obj_opt = a_opt2, obj2

        # LP direction (if time allows)
        if n <= 640 and t_alloc > 50:
            t_lp = min(20.0, t_alloc * 0.2)
            a_opt3, obj3 = lp_direction_optimize(a_opt, t_lp,
                                                 blend_init=0.15,
                                                 n_lp_iters=20,
                                                 seed=int(rng.integers(1, 10000)))
            if obj3 < obj_opt:
                a_opt, obj_opt = a_opt3, obj2

        # Another round of subgradient from the best
        if time.monotonic() < phase2_end - 10:
            t_sg2 = min(15.0, (phase2_end - time.monotonic() - 10) * 0.3)
            if t_sg2 > 5:
                a_opt4, obj4 = subgrad_optimize(a_opt, t_sg2,
                                                lr_init=0.002, momentum=0.90,
                                                top_k=50, temp_factor=0.008,
                                                decay=0.9999, restart_interval=400,
                                                seed=int(rng.integers(1, 10000)))
                if obj4 < obj_opt:
                    a_opt, obj_opt = a_opt4, obj4

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            results.append((score, n, f'deep{ci}', a_opt.copy()))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep {ci} n={n}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 2 best: {best_score:.6f}, elapsed={time.monotonic()-t1:.1f}s")

    # === Phase 3: Coordinate descent + final polish ===
    print("=== Phase 3: Polish ===")
    t2 = time.monotonic()
    phase3_budget = min(300.0, (total_deadline - time.monotonic()) * 0.25)
    phase3_end = t2 + phase3_budget

    if best_sequence is not None:
        # Coordinate descent
        t_cd = min(phase3_budget * 0.4, 80.0)
        a_cd = coordinate_descent(best_sequence, max_rounds=50,
                                  seed=int(rng.integers(1, 10000)))
        score_cd = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(score_cd) and score_cd < best_score:
            best_score = score_cd
            best_sequence = a_cd.copy()
            print(f"  CD: {score_cd:.6f}")

        # Final subgradient refinement
        remaining = phase3_end - time.monotonic() - 10
        if remaining > 20:
            t_final = min(remaining * 0.5, 60.0)
            a_f, obj_f = subgrad_optimize(best_sequence, t_final,
                                          lr_init=0.001, momentum=0.92,
                                          top_k=50, temp_factor=0.005,
                                          decay=0.99995, restart_interval=500,
                                          seed=int(rng.integers(1, 10000)))
            score_f = float(evaluate_sequence(a_f.tolist()))
            if np.isfinite(score_f) and score_f < best_score:
                best_score = score_f
                best_sequence = a_f.copy()
                print(f"  Final SG: {score_f:.6f}")

        # Another CD round
        remaining = phase3_end - time.monotonic() - 5
        if remaining > 10:
            a_cd2 = coordinate_descent(best_sequence, max_rounds=30,
                                       seed=int(rng.integers(1, 10000)))
            score_cd2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(score_cd2) and score_cd2 < best_score:
                best_score = score_cd2
                best_sequence = a_cd2.copy()
                print(f"  CD2: {score_cd2:.6f}")

    # === Phase 4: Diverse exploration ===
    print("=== Phase 4: Diverse Exploration ===")
    t3 = time.monotonic()
    remaining = max(0, total_deadline - time.monotonic() - 15)
    if remaining > 30 and best_sequence is not None:
        n_b = len(best_sequence)
        # Try perturbations with different scales
        for ps in [0.002, 0.005, 0.01, 0.02]:
            if remaining < 10:
                break
            t_pert = min(remaining * 0.2, 15.0)
            if t_pert < 5:
                break
            for attempt in range(3):
                if time.monotonic() > total_deadline - 12:
                    break
                noise = rng.standard_normal(n_b) * ps
                a_p = project_simplex(best_sequence + noise)
                a_opt, obj_opt = subgrad_optimize(a_p, t_pert / 3.0,
                                                  lr_init=0.002, momentum=0.88,
                                                  top_k=30, temp_factor=0.01,
                                                  decay=0.9998, restart_interval=200,
                                                  seed=int(rng.integers(1, 10000)))
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb({ps}) attempt {attempt}: {score:.6f}")
                remaining = max(0, total_deadline - time.monotonic() - 12)

        # Try a different length with the best structure
        if remaining > 30:
            # Upsample or use the best structure at a different length
            alt_lengths = [n_b * 2, n_b // 2] if n_b > 128 else [n_b * 2]
            for alt_n in alt_lengths:
                if 64 <= alt_n <= 2048 and time.monotonic() < total_deadline - 20:
                    inits = gen_inits(alt_n, best_sequence, rng)
                    if 'parent' in inits:
                        t_alt = min(remaining * 0.3, 30.0)
                        if t_alt > 8:
                            a_alt, obj_alt = subgrad_optimize(
                                inits['parent'], t_alt,
                                lr_init=0.003, momentum=0.87,
                                top_k=30, temp_factor=0.015,
                                decay=0.9998, restart_interval=200,
                                seed=int(rng.integers(1, 10000)))
                            score_alt = float(evaluate_sequence(a_alt.tolist()))
                            if np.isfinite(score_alt) and score_alt < best_score:
                                best_score = score_alt
                                best_sequence = a_alt.copy()
                                print(f"  Alt length {alt_n}: {score_alt:.6f}")

    # Final validation
    print("=== Final ===")
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    total_time = time.monotonic() - t0
    print(f"FINAL: score={final_score:.6f}, n={len(best_sequence)}, time={total_time:.1f}s")
    return [float(x) for x in best_sequence]
