from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_minmax_lp(a, n, use_preproc=True):
    """
    Solve: min t s.t. conv(a,g)[k] <= t for all k, g>=0, sum(g)=1
    Returns g (the optimal direction).
    """
    # Variables: [g_0, ..., g_{n-1}, t] -> total n+1 variables
    # Objective: minimize t -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Constraints: conv(a,g)[k] - t <= 0 for k=0,...,2n-2
    # conv(a,g)[k] = sum_j a_valid[k-j] * g[j]
    # where a_valid[i] = a[i] if 0<=i<n else 0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    for k in range(n_constraints):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]
        A_ub[k, n] = -1.0  # coefficient of t
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = np.maximum(result.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g /= gs
            return g, result.x[n]
    except Exception:
        pass
    return None, None

def exponentiated_gradient(a0, time_limit, lr_init=0.01, decay=0.9995,
                           top_k=50, reset_every=100):
    """Exponentiated gradient (mirror descent) on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -(min(top_k, len(conv))))[-min(top_k, len(conv)):]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break
        
        # Subgradient: weighted sum of partner vectors
        # For peak position k, the subgradient component at j is proportional to a[k-j]
        sg = np.zeros(n)
        # Use softmax weighting over peaks
        temp = 0.01 * pmax
        w = np.exp((peak_vals - pmax) / temp)
        w /= w.sum()
        
        for idx in range(len(peaks)):
            if w[idx] < 1e-10:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]
        
        # Exponentiated gradient update
        a_new = a * np.exp(-lr * sg)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= s_new
        
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= reset_every:
            # Reset with perturbation
            a = project_simplex(a + np.random.default_rng().standard_normal(n) * 0.005)
            conv = fast_convolve(a)
            lr = max(0.001, lr * 0.5)
            no_improve = 0
        
        lr *= decay
    
    return best_a, best_obj


def subgradient_momentum(a0, time_limit, lr_init=0.02, momentum=0.9,
                         top_k=50, temp_scale=0.03):
    """Subgradient with momentum and adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        k_top = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break
        
        # Adaptive temperature
        temp = temp_scale * pmax
        w = np.exp((peak_vals - pmax) / temp)
        w /= w.sum()
        
        sg = np.zeros(n)
        for idx in range(k_top):
            if w[idx] < 1e-10:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg /= sg_norm
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 150:
            a = project_simplex(a + np.random.default_rng().standard_normal(n) * 0.008)
            conv = fast_convolve(a)
            v *= 0.3
            lr = max(0.003, lr * 0.6)
            no_improve = 0
        
        lr *= 0.9998
    
    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=50):
    """Pairwise coordinate descent: try moving mass between pairs."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        # Focus on coordinates with high values (they contribute most to peaks)
        active = np.where(a > 1e-6)[0]
        for i in active:
            delta = a[i] * 0.03
            if delta < 1e-8:
                continue
            # Try moving to a few random targets
            targets = np.random.default_rng(rnd * 1000 + i).choice(
                n, size=min(20, n), replace=False)
            for j in targets:
                if j == i:
                    continue
                a_test = a.copy()
                a_test[i] = a[i] - delta
                if a_test[i] < 0:
                    continue
                a_test[j] = a[j] + delta
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-13:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
                    break
        if not improved:
            break
    
    return a


def gen_inits(n, parent=None, rng=None):

    inits['triangle'] = tri / tri.sum()
    
    edge = np.zeros(n)
    e = max(1, n // 8)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Alternating high-low
    alt = np.ones(n)
    alt[::2] = 0.3
    inits['alternating'] = alt / alt.sum()
    
    # Whittaker-Shannon (sinc-like)
    sinc = np.sinc((x - n/2) / (n/4))
    sinc = np.maximum(sinc, 0.01)
    inits['sinc'] = sinc / sinc.sum()
    
    # Two-scale: coarse structure with fine modulation
    coarse = np.exp(-0.5 * ((x - n//2) / (n/6)) ** 2)
    fine = 1.0 + 0.3 * np.cos(2 * np.pi * x / (n/8))
    two_scale = np.maximum(coarse * fine, 0.01)
    inits['two_scale'] = two_scale / two_scale.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            elif len(parent) > n:
                p_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
                inits['parent_ds'] = p_interp / p_interp.sum()
            else:
                p_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
                inits['parent_up'] = p_interp / p_interp.sum()
    
    return inits
