import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_minmax_lp(a, n):
    """
    Solve: min M s.t. (a*g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt).
    """
    # Variables: g_0, ..., g_{n-1}, M
    # Objective: minimize M -> c = [0,...,0,1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: (a*g)_k - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = np.maximum(result.x[:n], 0.0)
            g = g / g.sum()
            M_opt = result.x[n]
            return g, M_opt
    except Exception:
        pass
    return None, None

def optimize_minmax_lp(a_init, time_limit, blend_init=0.3, blend_min=0.005,
                       max_stall=20):
    """Min-max LP based optimization with adaptive blending."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        g, M_g = solve_minmax_lp(a, n)
        if g is None:
            break
        
        # Blend
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= 8:
            blend = max(blend_min, blend * 0.6)
            stall = 0
            if blend <= blend_min * 1.01:
                break
        
        if stall > max_stall:
            # Perturbation
            rng_local = np.random.default_rng(int(time.monotonic() * 1e6) % 2**31)
            a = project_simplex(a + rng_local.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            stall = 0
    
    return best_a, best_obj

def optimize_mult_weight(a_init, time_limit, eta_init=0.1, eta_decay=0.999,
                         top_k=30, temp_scale=0.1):
    """Multiplicative weights (mirror descent) on the simplex."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = eta_init
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        # Find top-k peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        
        # Softmax weights
        temp = temp_scale * M
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w /= w.sum()
        
        # Subgradient
        G = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                G[j] += wi * a[k - j]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * G)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
        else:
            # Reduce learning rate on stall
            pass
        
        eta *= eta_decay
    
    return best_a, best_obj

    best_a = a.copy()

def coordinate_polish(a_in, max_rounds=50, n_trials=3):
    """Coordinate descent: try moving mass between pairs."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        # Try moving mass from each i to each j
        # Be smarter: only consider pairs where it might help
        peak_k = np.argmax(conv)
        
        for i in range(n):
            if a[i] < 1e-8:
                continue
            # The "pressure" on i from the peak
            j_start = max(0, peak_k - n + 1)
            j_end = min(n - 1, peak_k)
            if i < j_start or i > j_end:
                # i doesn't contribute to the peak, good source
                pass
            
            for j in range(n):
                if j == i:
                    continue
                delta = a[i] * 0.1
                if delta < 1e-9:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-14:
                    a = a_test
                    best_obj = obj_test
                    conv = conv_test
                    improved = True
                    break
        if not improved:
            break
    
    return a

def gen_inits(n, parent=None):
    """Generate diverse initializations."""
    rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Edge concentration
    edge = np.zeros(n)
    e = max(1, n // 4)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Sparse (random support)
    sparse = np.zeros(n)
    support = rng.choice(n, size=max(2, n // 4), replace=False)
    sparse[support] = 1.0
    inits['sparse'] = sparse / sparse.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                # Resample to length n
                idx = np.linspace(0, len(p) - 1, n).astype(int)
                p_resampled = p[idx]
                ps2 = p_resampled.sum()
                if ps2 > 1e-10:
                    inits['parent_rs'] = p_resampled / ps2
    
    return inits

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Min-max LP for optimal direction finding (minimizes peak of a*g directly)
2. Multiplicative weights (mirror descent) for fine-tuning near optimum

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(a, n):
    """Build constraint matrix for min-max LP. Returns (A_ub, b_ub, A_eq, b_eq, c, bounds)."""
    # Variables: g_0, ..., g_{n-1}, M (total n+1 variables)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = a[k - j]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # M can be any value
    
    return A_ub, b_ub, A_eq, b_eq, c, bounds

            g = np.maximum(result.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
            M_opt = max(result.x[n], 0.0)
            return g, M_opt
    except Exception:
        pass
    return None, None

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = eta_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        # Top-k peaks with softmax weights
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            peak_vals = conv
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
        
        temp = temp_scale * M
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w = w / w.sum()
        
        # Compute subgradient
        G = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                G[j] += wi * a[k - j]
        
        # Center the gradient (remove mean for stability)
        G -= G.mean()
        
        # Multiplicative update
        a_new = a * np.exp(-eta * G)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            break
        a_new = a_new / a_new_sum
        
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                eta *= 0.5
                stall = 0
                if eta < 1e-6:
                    break
        
        eta *= eta_decay
    
    return best_a, best_obj

def subgradient_optimize(a_init, time_limit, lr_init=0.015, momentum=0.92,
                         top_k=50, temp_scale=0.04):
    """Subgradient descent with momentum."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        temp = temp_scale * M
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w = w / w.sum()
        
        G = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            wi = w[idx] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                G[j] += wi * a[k - j]
        
        v = momentum * v - lr * G
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 300:
            v *= 0.3
            lr = max(lr * 0.5, 0.001)
            stall = 0
        
        lr *= 0.9997
    
    return best_a, best_obj

def targeted_coordinate_polish(a_in, time_limit=30.0):
    """Targeted coordinate descent: only try moves that reduce the peak."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    fracs = [0.02, 0.05, 0.1, 0.2]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        peak_k = int(np.argmax(conv))
        
        # Find coordinates that contribute to the peak
        j_start = max(0, peak_k - n + 1)
        j_end = min(n - 1, peak_k)
        
        # Try moving mass from peak-contributing coords to non-contributing ones
        contributors = []
        non_contributors = []
        for j in range(n):
            if a[j] < 1e-10:
                continue
            if j_start <= j <= j_end:
                contributors.append(j)
            else:
                non_contributors.append(j)
        
        # Sort contributors by their contribution (a[j] * a[peak_k-j])
        contributors.sort(key=lambda j: a[j] * a[peak_k - j], reverse=True)
        
        for frac in fracs:
            for i in contributors[:min(20, len(contributors))]:
                if time.monotonic() - t0 > time_limit:
                    return best_a if 'best_a' in dir() else a
                delta = a[i] * frac
                if delta < 1e-10:
                    continue
                for j in non_contributors[:min(10, len(non_contributors))]:
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-15:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try within contributors (redistribute)
            for frac in fracs[:2]:
                for i in range(min(15, len(contributors))):
                    if time.monotonic() - t0 > time_limit:
                        return a
                    ci = contributors[i] if i < len(contributors) else None
                    if ci is None:
                        continue
                    delta = a[ci] * frac
                    if delta < 1e-10:
                        continue
                    for j in range(len(contributors)):
                        if j == i or j >= len(contributors):
                            continue
                        cj = contributors[j]
                        a_test = a.copy()
                        a_test[ci] -= delta
                        a_test[cj] += delta
                        a_test = np.maximum(a_test, 0.0)
