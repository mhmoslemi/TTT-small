**Alternating Minimization Approach**:
- Fix the "support" (which elements are non-zero)
- Optimize the values on the support to minimize the max peak
- Periodically update the support

For a fixed support S (subset of {0,...,n-1}), the problem is:
minimize max_k sum_{i in S, m-i in S} a_i * a_{m-i}
subject to a_i >= 0 for i in S, sum(a) = 1

This is still non-convex, but with a fixed support, we can use:
- Bisection on t: check if there exists a >= 0 on S with sum=1 and all C_k <= t
- This is a feasibility problem with quadratic constraints

Alternatively, for a fixed support, we can use:
- Interior point methods (if we can formulate it)
- Or iterative: reduce the current max peak by adjusting the values

**Practical approach - "Peak shaving"**:
1. Compute C = conv(a, a)
2. Find the top peaks
3. For each peak at position m, identify the "hot" pairs (i, m-i) that contribute most to C[m]
4. Reduce a_i and a_{m-i} slightly (proportionally to their contribution)
5. Increase other a_j to maintain the sum
6. Repeat

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) fast multi-peak subgradient descent for quick progress,
(2) LP-based Frank-Wolfe for careful refinement, (3) coordinate descent polishing.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=3.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def local_obj(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    C = fast_convolve(a)
    S = a.sum()
    if S < 1e-15:
        return 1e10
    return 2.0 * n * C.max() / (S * S)


def line_search(a, b, n_trials=8):
    """Line search: find best t in (0,1) for (1-t)*a + t*b."""
    best_t = 0.0
    best_obj = local_obj(a)
    for t in np.linspace(0.05, 0.95, n_trials):
        candidate = (1.0 - t) * a + t * b
        obj = local_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj


def subgradient_descend(a0, time_limit, lr_init=0.005, momentum=0.92, top_k=7, temp_init=0.08):
    """Projected subgradient descent with multi-peak smoothing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S

    C = fast_convolve(a)
    best_obj = C.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    temperature = temp_init
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        C = fast_convolve(a)
        # Multi-peak subgradient
        k = min
        k = min(top_k, len(C) // 4)
        peaks = np.argpartition(C, -k)[-k:]
        peak_vals = C[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break
        weights = np.exp((peak_vals - pmax) / (temperature * pmax + 1e-15))
        weights /= weights.sum()

        g = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-10:
                continue
            m = int(peaks[i])
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            idx = np.arange(j_start, j_end + 1)
            g[idx] += 2.0 * weights[i] * a[m - idx]

        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new

        C = fast_convolve(a)
        cur_obj = C.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptivity
        lr *= 0.9997
        if stall > 200:
            # Perturb to escape
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.002
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            stall = 0
            temperature = temp_init

        if it % 500 == 0 and it > 0:
            temperature = max(0.02, temperature * 0.9)
            top_k = min(n // 2, top_k + 1)

        it += 1

    return best_a, best_obj


def fpw_lp_optimize(a0, time_limit, verbose=False, verbose_every=100):
    """Frank-Wolfe with LP oracle."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    C = fast_convolve(a)
    M = C.max()
    best_obj = M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit:
        # Solve LP
        b = lp_oracle(a, M, time_limit=2.0)
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb

        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
        else:
            t, obj_new = line_search(a, b, n_trials=6)
            if t < 0.01:
                no_improve += 1
            else:
                a = (1.0 - t) * a + t * b
                C = fast_convolve(a)
                M = C.max()
                if M < best_obj - 1e-14:
                    best_obj = M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        if no_improve >= max_stall:
            rng = np.random.default_rng(it + 7)
            noise = rng.standard_normal(n) * 0.004
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            C = fast_convolve(a)
            M = C.max()
            no_improve = 0
            max_stall = int(max_stall * 1.8)
            if max_stall > 500:
                break

        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"      FPW it={it}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    C = fast_convolve(a)
    best_obj = C.max()
    best_a = a.copy()
    t0 = time.monotonic()
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        C = fast_convolve(a)
        # Process coordinates in the order of their contribution to peaks
        top_peaks = np.argpartition(C, -10)[-10:]
        
        # Build priority: coordinates that appear in peak convolutions
        priority = np.zeros(n)
        for m in top_peaks:
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            for j in range(j_start, j_end + 1):
                priority[j] += C[m] * a[j] * a[m - j]
        
        order = np.argsort(-priority)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            # Optimize a[j] by bisection
            # Objective as function of a[j]: fix all other coords
            a_save = a[j]
            
            # The contribution of a[j] to conv[k] is a[j]*a[k-j] + a[j]*a[j-k] (roughly)
            # For a_j = x, conv changes at positions k=
            # k = j: add x * a[0]... this gets complex. Use golden section or ternary.
            
            # Try a few values
            best_local = C.max()
            best_x = a_save
            s_rest = S - a_save
            
            for x in [a_save * 0.5, a_save * 0.7, a_save * 0.85, a_save, a_save * 1.15, a_save * 1.3, a_save * 1.5, 0.0]:
                x = max(0.0, min(x, 1.0))  # Can't exceed 1 since sum is 1
                if x == a_save:
                    continue
                # Renormalize: if we change a[j] to x, do we renormalize?
                # Let's just set a[j] = x and renormalize the whole thing
                a[j] = x
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = a_save
                    continue
                a_norm = a / s_new
                C_test = fast_convolve(a_norm)
                obj_test = C_test.max()
                if obj_test < best_local:
                    best_local = obj_test
                    best_x = x
            
            if best_x != a_save:
                a[j] = best_x
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a / s_new
                improved = True
            
            a[j] = a_save  # restore for next trial (we'll set at end)

        if not improved:
            stagnation += 1
            if stagnation > 3:
                break
        else:
            stagnation = 0
            C = fast_convolve(a)
            if C.max() < best_obj - 1e-14:
                best_obj = C.max()
                best_a = a.copy()

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n) / n

    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()

    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri.copy()

    edge = max(1, n // 5)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc /= rc.sum()
    inits['raised_cos'] = rc.copy()

    # Sidelobe-like: 1 + 0.3*cos(4*pi*x/n)
    sl = 1.0 + 0.3 * np.cos(4 * np.pi * x / n)
    sl = np.maximum(sl, 0.01)
    sl /= sl.sum()
    inits['sidelobe'] = sl.copy()

    # Double frequency modulation
    df = 1.0 + 0.4 * np.cos(2 * np.pi * 2 * x / n) * np.exp(-x / (n/3))
    df = np.maximum(df, 0.01)
    df /= df.sum()
    inits['double_freq'] = df.copy()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sparse random
    sparse = rng.random(n)
    sparse[sparse < 0.4] = 0.0
    if sparse.sum() < 1e-10:
        sparse = rng.dirichlet(np.ones(n))
    else:
        sparse /= sparse.sum()
    inits['sparse'] = sparse.copy()

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits

            if t_alloc < 8:
                break
            
            if p % 2 == 0:
                a_opt, obj_opt = fpw_lp_optimize(best_sequence, time_limit=t_alloc)
            else:
                a_opt, obj_opt = subgradient_descend(best_sequence, time_limit=t_alloc,
                                                     lr_init=0.003, momentum=0.94,
                                                     top_k=10, temp_init=0.05)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Pass {p+1}: score={score:.6f} (improved!)")
                else:
                    print(f"  Pass {p+1}: score={score:.6f}")

    # === Phase 5: Perturbation restarts ===
    print("\n=== Phase 5: Perturbation Restarts ===")
    if best_sequence is not None and time.monotonic() < deadline - 20:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 12
        n_restarts = max(1, int(remaining // 12))
        n_restarts = min(n_restarts, 6)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            scale = 0.0015 * (1 + i * 0.7)
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(10.0, deadline - time.monotonic() - 8)
            if t_alloc < 4:
                break

            a_opt, obj_opt = subgradient_descend(a_pert, time_limit=t_alloc,
                                                 lr_init=0.003, momentum=0.93,
                                                 top_k=8, temp_init=0.06)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {i}: score={score:.6f} (improved!)")
                else:
                    print(f"  Restart {i}: score={score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
