```python
def gradient_polish(a0, time_limit, seed=0, verbose=False):
    """Polish using subgradient/smooth gradient of the max."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = 50.0  # starts soft, gets sharper
    lr = 0.1
    v = np.zeros(n)
    momentum = 0.9
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Softmax weights on the convolution peaks
        scaled = beta * (conv - M)
        w = np.exp(scaled)
        w /= w.sum()
        
        # Gradient: d/d a[j] of weighted sum of conv[k]
        # = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices
        # Use FFT
        fft_size = 1 << (2 * n).bit_length()
        A_fft = np.fft.fft(np.concatenate([a, np.zeros(fft_size - n)]))
        W_fft = np.fft.fft(np.concatenate([w, np.zeros(fft_size - len(w)])))
        
        # Term 1: sum_k w[k] * a[k-j] = cross-correlation
        grad1 = np.fft.ifft(A_fft * np.conj(W_fft)).real[:n]
        # Term 2: sum_k w[k] * a[j-k] = convolution
        grad2 = np.fft.ifft(A_fft * W_fft).real[:n]
        
        grad = grad1 + grad2
        
        # Normalize
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad /= g_norm
        
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.9
            if no_improve > 100:
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = 0.05
                beta = max(20, beta * 0.5)
                no_improve = 0
        
        # Increase beta (sharper max approximation)
        beta = min(5000.0, beta * 1.01)

    return best_a, best_obj
```
