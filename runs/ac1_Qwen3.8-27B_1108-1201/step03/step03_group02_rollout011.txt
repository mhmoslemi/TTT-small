Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating optimization
with adaptive step sizing, combined with log-sum-exp smoothed gradient descent.

import numpy as np

# --- Helper functions (all top-level, no closures) ---

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.nonzero(u * np.arange(1, n+1) - (cssv - 1) > 0)[0] + 1)
    theta = (cssv[rho-1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A

def solve_lp_min_M(a):
    n = len(a)
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_eq = np.array([[1.0]*n + [0.0]])

                  bounds=bounds, method='highs')

    conv = fast_convolve(a)
    scaled = conv / temp
    scaled -= scaled.max()
    ev = np.exp(scaled)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating optimization
with the min-M formulation, adaptive step sizing, diverse initializations,
and log-sum-exp smoothed gradient descent for final polishing.
Key improvements: direct min-M LP, adaptive step with backtracking logic,
efficient Toeplitz matrix construction, and multi-phase search strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_min_M(a):
    """Solve: min M s.t. conv(a,b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns (b_normalized, M) or (None, None) on failure."""
    n = len(a)
    A_mat = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A_mat, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    b = np.maximum(res.x[:n], 0.0)
    M = float(res.x[n])
    bs = b.sum()
    if bs < 1e-15:
        return None, None
    return b / bs, M


def lp_optimize(a0, time_limit, max_iters=500, t_init=0.3):
    """LP-based alternating optimization to minimize peak auto-correlation."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    t = t_init
    no_improve = 0
    consec_fail = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b, M_opt = solve_lp_min_M(a)
        if b is None:
            consec_fail += 1
            if consec_fail > 3:
                break
            continue
        consec_fail = 0

        if np.linalg.norm(b - a) < 1e-12:
            break

        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            t *= 0.5
            continue
        a_new = a_new / sn

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            t = min(0.75, t * 1.04)
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                t = max(0.03, t * 0.88)

        a = a_new
        conv = conv_new

        if no_improve > 60:
            t = t_init
            no_improve = 0
        if no_improve > 250:
            break

    return best_a, best_obj


def soft_max_objective_and_grad(a, temp):
    """Compute soft-max smoothed objective and gradient."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / temp
    scaled -= scaled.max()
    ev = np.exp(scaled)
    sum_ev = ev.sum()
    p = ev / sum_ev
    obj = (2.0 * n / temp) * (scaled.max() + np.log(sum_ev))
    grad = np.zeros(n)
    for j in range(n):
        end = min(2 * n - 1, j + n)
        if end > j:
            grad[j] = (4.0 * n / temp) * np.dot(p[j:end], a[:end - j])
    return obj, grad


def grad_optimize(a0, time_limit, temp_init=20.0, lr_init=0.002, momentum=0.92):
    """Projected gradient with momentum for soft-max smoothed objective."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    temp = temp_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        _, grad = soft_max_objective_and_grad(a, temp)
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 300 == 0:
            lr *= 0.6
        if it % 800 == 0 and temp < 200:
            temp *= 1.6
        if no_improve > 2000:
            break
        it += 1

    return best_a, best_obj


def make_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    c1 = np.maximum(1.0 + 0.7 * np.cos(2.0 * np.pi * x / n), 0)
    inits['cos1'] = c1 / c1.sum()

    c2 = np.maximum(1.0 + 0.5 * np.cos(4.0 * np.pi * x / n), 0)
    inits['cos2'] = c2 / c2.sum()

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    bi = np.exp(-((x - 0.25 * n) / (0.15 * n)) ** 2) + np.exp(-((x - 0.75 * n) / (0.15 * n)) ** 2)
    inits['bimodal'] = bi / bi.sum()

    tri = np.abs(1.0 - 2.0 * x / (n - 1))
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    return inits

            a_opt, _ = lp_optimize(inits2[name], time_limit=t_alloc, t_init=0.3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print("  New best: {:.6f} (n=1024)".format(score))

    print("Phase 5 done: best={:.6f}, elapsed={:.0f}s".format(best_score, time.monotonic() - t5_start))

    # Final safety
    if best_sequence is None:
        best_sequence = np.ones(N) / N

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: score={:.6f}, n={}".format(best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
