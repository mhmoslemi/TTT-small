"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an enhanced LP-based iterative
method with adaptive margins, log-space polishing, multi-peak targeting,
and diverse initialization strategies. Key improvements over prior approach:
(1) adaptive LP relaxation that tightens as convergence improves,

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def obj_from_conv(a, conv):
    """Compute objective 2n*max(conv)/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def line_search_golden(a, d, current_obj, n, tol=1e-7, max_iter=25):
    """Golden-section line search on t in [0, 1]. Returns (best_t, best_obj)."""
    best_t = 0.0
    best_obj = current_obj

    # First expand to find a bracket
    t_hi = 0.01
    a_try = np.maximum(a + t_hi * d, 0.0)
    s = a_try.sum()
    if s > 1e-15:
        a_try = a_try / s
        obj_try = obj_from_conv(a_try, fast_convolve(a_try))
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t_hi
            # Try expanding
            for t_exp in [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
                a_try = np.maximum(a + t_exp * d, 0.0)
                s = a_try.sum()
                if s > 1e-15:
                    a_try = a_try / s
                    obj_try = obj_from_conv(a_try, fast_convolve(a_try))
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_t = t_exp
                    else:
                        break

    # Golden section refinement around best_t
    if best_t > 0.005:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        t_lo = max(0.0, best_t * 0.3)
        t_hi2 = min(1.0, best_t * 1.7 + 0.05)
        if t_hi2 <= t_lo:
            t_lo, t_hi2 = 0.0, 1.0

        c = t_hi2 - gr * (t_hi2 - t_lo)
        dval = t_hi2 - gr * (t_hi2 - t_lo)
        # Evaluate at c and d
        for _ in range(max_iter):
            if time.monotonic() > line_search_golden._t_limit if hasattr(line_search_golden, '_t_limit') else True:
                break
            a_c = np.maximum(a + c * d, 0.0)
            sc = a_c.sum()
            obj_c = obj_from_conv(a_c / sc, fast_convolve(a_c / sc)) if sc > 1e-15 else np.inf

            a_d = np.maximum(a + dval * d, 0.0)
            sd = a_d.sum()
            obj_d = obj_from_conv(a_d / sd, fast_convolve(a_d / sd)) if sd > 1e-15 else np.inf

            if obj_c < obj_d:
                t_hi2 = dval
                dval = c
                c = t_hi2 - gr * (t_hi2 - t_lo)
            else:
                t_lo = c
                c = dval

            if abs(t_hi2 - t_lo) < tol:
                break

            if obj_c < best_obj:
                best_obj = obj_c
                best_t = c
            if obj_d < best_obj:
                best_obj = obj_d
                best_t = dval

    return best_t, best_obj

def peak_targeted_refine(a0, time_limit, n_steps=500):
    """
    Targeted peak reduction: identify the max convolution position and
    use coordinate descent to reduce it while monitoring other peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.01

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        peaks = np.where(conv >= conv.max() * 0.999)[0]

        # For each peak position, compute the "pressure" on each a_i
        grad = np.zeros(n)
        for k in peaks[:10]:  # Limit to top 10 peaks
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += 2.0 * a[k - idx]

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Update
        a_new = a - lr * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a = best_a.copy()
            continue

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.9

        if lr < 1e-6:
            break

    return best_a, best_obj

        # Subgradient in original space
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * a[k - idx]

        # Convert to log-space gradient (Jacobian of softmax)
        # If a = exp(x)/S, then da_i/dx_p = a_i*(delta_ip - a_p)
        # So grad_x = a * (grad_a - sum(a * grad_a))
        grad_x = a * (sg - np.dot(a, sg))

        # Update in log-space
        x = x - lr * grad_x
        a_new = np.exp(x - np.max(x))
        a_new = a_new / a_new.sum()

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve > 200:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    ramp = x + 1.0
    inits['ramp'] = ramp / ramp.sum()
    inits['ramp_rev'] = inits['ramp'][::-1].copy()

    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Raised cosine
    inits['raised_cos'] = (0.5 * (1 + np.cos(2 * np.pi * (x + 0.5) / n))) ** 2 + 0.01
    inits['raised_cos'] = inits['raised_cos'] / inits['raised_cos'].sum()

    # Fejer-kernel-like
    inits['fejer'] = (1.0 - np.abs(x - n / 2.0 + 0.5) / (n / 2.0)) ** 2 + 0.01
    inits['fejer'] = inits['fejer'] / inits['fejer'].sum()

    # Super-Gaussian
    g = np.exp(-((x - 0.5 * n) / (0.4 * n)) ** 4)
    inits['super_gauss'] = g / g.sum()

    # Sine
    inits['sine'] = (np.sin(np.pi * (x + 0.5) / n) ** 0.5 + 0.05)
    inits['sine'] = inits['sine'] / inits['sine'].sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining LP optimization,
    peak-targeted refinement, and log-space polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Phase 1: LP-based scan across lengths and initializations
    print("=== Phase 1: LP Scan ===")
    t_start = time.monotonic()
    phase1_limit = 400.0
    phase1_deadline = t_start + phase1_limit

    lengths = [256, 384, 512, 640]
    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_ds', 'parent_up', 'uniform', 'random', 'raised_cos', 'fejer', 'bimodal']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        t_per_init = max(12.0, (phase1_limit - 20.0) / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))

            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 2: Deep LP on top candidates
    print("=== Phase 2: Deep LP ===")
    t_start2 = time.monotonic()
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None:
        if (len(best_sequence), 'best') not in seen:
            top_configs.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 120.0
    n_candidates = max(1, len(top_configs))
    t_per_deep = max(15.0, remaining * 0.5 / n_candidates)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 60)
        if t_alloc < 10:
            break

        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: Peak-targeted refinement
    print("=== Phase 3: Peak-Targeted Refinement ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        if remaining > 15:
            t_alloc = min(remaining * 0.4, 60.0)
            if t_alloc > 5:
                a_opt, obj_opt = peak_targeted_refine(best_sequence, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Peak refactor: {score:.6f}")

            # LP polish after peak refactor
            remaining2 = deadline - time.monotonic() - 50.0
            if remaining2 > 10:
                t_alloc2 = min(30.0, remaining2 * 0.5)
                if t_alloc2 > 5:
                    a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc2)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  LP after peak: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Log-space refinement
    print("=== Phase 4: Log-Space Refinement ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50.0
        if remaining > 10:
            t_alloc = min(40.0, remaining * 0.5)
            if t_alloc > 5:
                a_opt, obj_opt = logspace_refine(best_sequence, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Log-space: {score:.6f}")

                # Brief LP after log-space
                remaining2 = deadline - time.monotonic() - 30.0
                if remaining2 > 8:
                    t_alloc2 = min(20.0, remaining2 * 0.6)
                    if t_alloc2 > 4:
                        a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc2)
                        score = float(evaluate_sequence(a_opt.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_opt.copy()
                            print(f"  LP after log: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Phase 5: Multi-perturbation restart
    print("=== Phase 5: Perturb & Reoptimize ===")
    t_start5 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 15:
            n_best = len(best_sequence)
            n_rounds = max(1, int(remaining / 12))
            t_per_round = remaining / n_rounds

            for pi in range(n_rounds):
                if time.monotonic() > deadline - 5:
                    break
                if t_per_round < 4:
                    break

                # Progressive perturbation
                scale = 0.002 * (pi + 1) + 0.001
                noise = rng.standard_normal(n_best) * scale
                a_pert = best_sequence + noise
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s

                t_alloc = min(t_per_round, deadline - time.monotonic() - 2)
                if t_alloc < 3:
                    break

                # Alternate between LP and peak refactor
                if pi % 2 == 0:
                    a_opt, obj_opt = lp_optimize(a_pert, time_limit=t_alloc,
                                                 margin_init=1.015, margin_min=1.001)
                else:
                    a_opt, obj_opt = peak_targeted_refine(a_pert, time_limit=t_alloc)
                    # Quick LP polish
                    if time.monotonic() < deadline - 3:
                        a_opt2, _ = lp_optimize(a_opt, time_limit=min(5.0, deadline - time.monotonic() - 2))
                        score2 = float(evaluate_sequence(a_opt2.tolist()))
                        if np.isfinite(score2):
                            a_opt = a_opt2

                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Round {pi} (n={n_best}): {score:.6f}")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # Final: one more LP polish with tight margin
    if best_sequence is not None and time.monotonic() < deadline - 5:
        t_final = min(15.0, deadline - time.monotonic() - 3)
        if t_final > 4:
            a_opt, _ = lp_optimize(best_sequence, time_limit=t_final,
                                   margin_init=1.003, margin_min=1.0003)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final polish: {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
