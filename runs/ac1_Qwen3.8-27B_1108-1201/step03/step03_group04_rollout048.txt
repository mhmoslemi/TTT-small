"""
Improved LP-based Frank-Wolfe with golden-section line search,
multi-length exploration, and adaptive restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix

def fast_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def local_obj(a):
    """Objective assuming sum(a) = 1: 2n * max(conv(a,a))"""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def build_conv_matrix(a):
    """Matrix mapping b -> conv(a, b), shape (2*len(a)-1, len(a))"""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp(a, M):
    """max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    A_sparse = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_sparse, b_ub=b_ub, method='highs',
                  bounds=bounds, options={'time_limit': 3.0, 'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

def golden_section_line_search(a, b, f, max_iter=20):
    """Golden section search for t in [0,1] minimizing f((1-t)*a + t*b)"""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ~0.618
    lo, hi = 0.0, 1.0
    c = lo + (1 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f((1 - c) * a + c * b)
    fd = f((1 - d) * a + d * b)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1 - gr) * (hi - lo)
            fc = f((1 - c) * a + c * b)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f((1 - d) * a + d * b)
    t = (lo + hi) / 2
    return t, f((1 - t) * a + t * b)

    best_a = a.copy()

    if tau_init is None:
        tau_init = conv.max() * 0.05

    v = np.zeros(n)
    momentum = 0.9
    lr = 0.01
    t0 = time.monotonic()
    it = 0
    n_eval = max(int(n / 64), 2)

    while time.monotonic() - t0 < time_limit:
        obj_val, g = smooth_gradient_obj(a, tau_init)
        # Normalized gradient on simplex
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm

        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s

        conv_new = fast_convolve(a_new)
        cur = 2.0 * n * conv_new.max()
        if cur < best_obj - 1e-13:
            best_obj = cur
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.7
            momentum = max(momentum * 0.98, 0.5)

        it += 1
        if verbose and it % 100 == 0:
            print(f"    SG it={it}, obj={best_obj:.7f}, lr={lr:.5f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse initializations of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    x = np.arange(n, dtype=np.float64)
    cx = 0.5 * (n - 1)

    # Edge concentration (various widths)
    for frac in [0.25, 0.35, 0.5]:
        width = max(1, int(n * frac / 2))
        ec = np.zeros(n)
        ec[:width] = 1.0
        ec[-width:] = 1.0
        ec /= ec.sum()
        inits[f'edge_{frac}'] = ec.copy()

    # Gaussian
    for sigma_frac in [0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - cx) / (sigma_frac * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sigma_frac}'] = g.copy()

    # W-shape (two peaks)
    peak_pos = n // 4
    w1 = np.exp(-0.5 * ((x - peak_pos) / (0.15 * n)) ** 2)
    w2 = np.exp(-0.5 * ((x - (n - 1 - peak_pos)) / (0.15 * n)) ** 2)
    ws = w1 + w2
    ws /= ws.sum()
    inits['w_shape'] = ws.copy()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Random sparse
    sparse = rng.random(n)
    sparse[sparse < 0.4] = 0.0
    if sparse.sum() < 1e-10:
        sparse = np.ones(n) / n
    else:
        sparse /= sparse.sum()
    inits['sparse'] = sparse.copy()

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(np.pi * (x - cx) / (n / 2)))
    cos_w = np.maximum(cos_w, 0.01)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w.copy()

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            pp = pp / ps
            if len(pp) == n:
                inits['parent'] = pp
            else:
                interp = np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, len(pp)), pp)
                interp /= interp.sum()
                inits['parent_interp'] = interp

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: parametric init scan -> LP Frank-Wolfe ->
    smoothed gradient polish -> deep FW -> restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    def update_best(a, label=""):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST [{label}]: score={score:.7f}, n={len(a)}")
        return score

    # === Phase 1: Quick scan of lengths and inits ===
    print("=== Phase 1: Scan ===")
    lengths = [256, 384, 512, 768]

    # If parent exists, prioritize its length
    if parent is not None and len(parent) in lengths:
        lengths.remove(len(parent))
        lengths.insert(0, len(parent))

    candidates = []  # list of (n, a_init)

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        # Prioritize
        prio = ['parent', 'parent_interp', 'edge_0.35', 'w_shape', 'gauss_0.3',
                'edge_0.25', 'cosine', 'uniform', 'random', 'sparse']
        init_names.sort(key=lambda x: prio.index(x) if x in prio else 99)

        for name in init_names[:5]:
            if time.monotonic() > deadline - 250:
                break
            a0 = inits[name].copy()
            # Quick FW run
            t_alloc = min(25.0, (deadline - time.monotonic() - 200) * 0.15)
            if t_alloc < 5:
                continue
            a_opt, obj_opt = fw_optimize(a0, time_limit=t_alloc)
            score = update_best(a_opt, f"P1 n={n} {name}")
            candidates.append((score, n, a_opt))

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best so far: {best_score:.7f}")

    # === Phase 2: Deep FW on top candidates ===
    print("=== Phase 2: Deep FW ===")
    seen_lengths = set()
    for score_c, n_c, a_c in candidates[:3]:
        if time.monotonic() > deadline - 200:
            break
        if n_c in seen_lengths:
            continue
        seen_lengths.add(n_c)
        t_alloc = min(150.0, (deadline - time.monotonic() - 150) * 0.3)
        if t_alloc < 20:
            break
        print(f"  Deep FW n={n_c}, t={t_alloc:.0f}s")
        a_opt, obj_opt = fw_optimize(a_c, time_limit=t_alloc, verbose=True)
        score = update_best(a_opt, f"P2 n={n_c}")

    # === Phase 3: Smoothed gradient polish ===
    if best_sequence is not None:
        print("=== Phase 3: Smooth Gradient Polish ===")
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            t_polish = min(80.0, remaining * 0.4)
            a_opt, _ = smooth_gradient_optimize(best_sequence, time_limit=t_polish,
                                                verbose=True)
            score = update_best(a_opt, "P3 polish")
            # Then one more FW pass
            if time.monotonic() < deadline - 80:
                t_fw = min(60.0, (deadline - time.monotonic() - 60) * 0.5)
                if t_fw > 10:
                    a_opt2, _ = fw_optimize(a_opt, time_limit=t_fw)
                    score = update_best(a_opt2, "P3b post-polish FW")

    # === Phase 4: Additional lengths ===
    if time.monotonic() < deadline - 120:
        print("=== Phase 4: More Lengths ===")
        extra_lengths = [1024, 192, 448, 640]
        for n in extra_lengths:
            if time.monotonic() > deadline - 80:
                break
            inits = generate_initializations(n, best_sequence, rng)
            for name in ['parent_interp', 'edge_0.35', 'w_shape', 'uniform']:
                if name not in inits:
                    continue
                if time.monotonic() > deadline - 60:
                    break
                t_alloc = min(40.0, (deadline - time.monotonic() - 50) * 0.4)
                if t_alloc < 10:
                    break
                a_opt, _ = fw_optimize(inits[name], time_limit=t_alloc)
                score = update_best(a_opt, f"P4 n={n} {name}")

    # === Phase 5: Restart perturbation on best ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("=== Phase 5: Restarts ===")
        n_best = len(best_sequence)
        n_restarts = max(1, int((deadline - time.monotonic() - 20) // 20))
        n_restarts = min(n_restarts, 6)
        for i in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            scale = 0.01 * (1 + 0.5 * i)
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(18.0, (deadline - time.monotonic() - 10) * 0.6)
            if t_alloc < 5:
                break
            a_opt, _ = fw_optimize(a_pert, time_limit=t_alloc)
            score = update_best(a_opt, f"P5 restart {i}")

    # === Phase 6: Final polish ===
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("=== Phase 6: Final ===")
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_opt, _ = fw_optimize(best_sequence, time_limit=remaining * 0.6)
            score = update_best(a_opt, "P6a")
            if time.monotonic() < deadline - 5:
                a_opt2, _ = smooth_gradient_optimize(best_sequence, time_limit=remaining * 0.3)
                score = update_best(a_opt2, "P6b")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"  Fallback used: score={best_score:.7f}")

    # Final check
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.7f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
