"""
Multi-phase optimizer combining spectral flattening (Gerchberg-Saxton),
LP-based direction finding, and adaptive subgradient descent to minimize
2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def spectral_flatten(a0, time_limit=60.0, n_iter_max=5000):
    """
    Gerchberg-Saxton style spectral flattening for non-negative sequences.
    Iteratively: FFT -> flatten magnitude -> IFFT -> project to non-negative.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = np.inf
    
    t0 = time.monotonic()
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT (zero-padded for linear convolution)
        fft_size = 2 * n
        A = np.fft.fft(a, fft_size)
        
        # Target: flat magnitude spectrum
        # The DC component is fixed (equals sum(a) = 1)
        # We want |A[k]| to be as uniform as possible for k >= 1
        mag = np.abs(A)
        # Keep DC, flatten the rest
        target_mag = np.ones(fft_size)
        target_mag[0] = mag[0]  # Keep DC
        # Scale to preserve total energy
        # Actually, for non-negative reals, the DC is the sum, which we fix to 1.
        # The "optimal" flat magnitude would have |A[k]| = sqrt(sum|A|^2 / fft_size) for k>0
        # But we need to be more careful. Let's just use a power-law flattening.
        
        # Blend toward flat magnitude
        alpha = 0.1  # Blending parameter
        new_mag = (1 - alpha) * mag + alpha * target_mag * (mag.mean() / target_mag.mean() if target_mag.mean() > 0 else 1)
        
        # Actually, let's use a simpler approach: raise magnitude to a power < 1 to compress dynamics
        # mag_new = mag^0.5 (compresses dynamic range)
        power = 0.7
        new_mag = mag ** power
        new_mag[0] = mag[0]  # Preserve DC
        
        # Reconstruct with same phase, new magnitude
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        
        # IFFT
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check objective
        conv = fast_convolve(a_new)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

    """

    Uses amplitude scaling that preserves the DC component and gradually

    """

        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

        phase = np.angle(A)

        # Enforce Hermitian symmetry
        A_new[fft_size // 2:] = np.conj(A_new[1:fft_size // 2 + 1][::-1]) if fft_size % 2 == 0 else np.conj(A_new[1:fft_size // 2][::-1])
        # Actually, let me just use rfft/irfft to handle this properly

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()

            break

def solve_lp_direction(a, M, A_ub=None):
    """
    Solve LP: maximize sum(g) s.t. a*g <= M (componentwise), g >= 0.
    Returns (g_normalized, g_sum, dual_vars).
    """
    n = len(a)
    if A_ub is None:
        A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum, result.ineqlin.marginals
    return a.copy(), 1.0, None

def subgradient_optimize_v2(a0, time_limit=100.0, lr_init=0.02, momentum=0.93, top_k=64):
    """
    Projected subgradient with momentum, using softmax-weighted peaks
    for a smoother subgradient estimate.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient using top peaks with softmax weighting
        k_max = n + top_k
        peaks = np.argpartition(conv, -k_max)[-k_max:]
        peak_vals = conv[peaks]
        
        # Softmax weighting
        temp = 0.001 * peak_vals.max()
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        w = w / w_sum
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * a[k - idx]
        
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        momentum = min(0.97, momentum + 0.0002)
        
        if no_improve >= 500:
            break
    
    return best_a, best_obj

    """Generate diverse initializations of length n."""
    inits = {}

    inits['gaussian'] = g / g.sum()

    anti = np.maximum(anti, 0.01)
    inits['anti_peak'] = anti / anti.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)

                if pps > 1e-10:

                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines spectral flattening (Gerchberg-Saxton), LP-based direction finding

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def build_conv_matrix_top(a, M, top_p=None):

def solve_lp_direction(a, M, A_ub=None):
    """
    Solve LP: maximize sum(g) s.t. a*g <= M (componentwise), g >= 0.
    Returns (g_normalized, g_sum, dual_vars).
    """
    n = len(a)
    if A_ub is None:
        A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

        g = result.x
        g_sum = g.sum()
        dual = result.ineqlin.marginals if hasattr(result, 'ineqlin') and result.ineqlin.marginals is not None else None
        if g_sum > 1e-10:
            return g / g_sum, g_sum, dual
    return a.copy(), 1.0, None

                    obj_try = 2.0 * n * fast_convolve(a_try).max()
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_t = t_try
                        improved = True
            step *= 0.5
            if not improved:
                break
    
    return best_t, best_obj


def lp_optimize(a0, time_limit=300.0, verbose=False):
    """
    LP-based iterative optimization with adaptive line search and
    perturbation-based restarts.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 150
    perturb_count = 0
    lp_fail_count = 0
    
    while time.monotonic() - t0 < time_limit:
        A_ub = build_conv_matrix(a)
        
        # Slight margin for numerical safety
        M_lp = M * 1.0005
        g, g_sum, dual = solve_lp_direction(a, M_lp, A_ub)
        
        if g_sum <= 1.0 + 1e-7:
            lp_fail_count += 1
            no_improve += 1
        else:
            lp_fail_count = 0
            d = g - a
            norm_d = np.linalg.norm(d)
            
            if norm_d < 1e-12:
                no_improve += 1
            else:
                t_best, obj_best = line_search(a, d, best_obj, n)
                
                if t_best > 0 and obj_best < best_obj - 1e-13:
                    a_new = np.maximum(a + t_best * d, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()
                    
                    conv = fast_convolve(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                    if verbose and iteration % 50 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s")
                else:
                    no_improve += 1
        
        iteration += 1
        
        # Handle stagnation
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 7 + 13)
            scale = 0.005 + 0.008 * min(perturb_count, 8)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(350, 150 + 25 * perturb_count)
            
            if perturb_count > 35:
                break
        
        if lp_fail_count > 40:
            rng = np.random.default_rng(iteration * 3 + 7)
            noise = rng.standard_normal(n) * 0.01
            a = np.maximum(best_a + noise, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            lp_fail_count = 0
            no_improve = 0
    
    return best_a, best_obj


def dual_guided_subgradient(a, conv, n, dual_vars=None, top_k=64):
    """
    Compute a weighted subgradient of max(conv) using either dual variables
    from the LP or softmax weighting of peaks.
    """
    if dual_vars is not None and np.sum(dual_vars) > 1e-10:
        # Use LP dual variables as weights
        w = np.abs(dual_vars)
        w /= w.sum()
    else:
        # Softmax on top peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        temp = 0.0005 * peak_vals.max()
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w_peaks = np.exp(scaled)
        w_peaks /= w_peaks.sum()
        
        w = np.zeros(len(conv))
        for i, pk in enumerate(peaks):
            w[int(pk)] = w_peaks[i]
    
    # Compute subgradient: sum_k w_k * a[shifted by k]
    sg = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w[k] * a[k - idx]
    
    return sg


def subgradient_optimize(a0, time_limit=100.0, lr_init=0.015, momentum=0.93, top_k=64):
    """
    Projected subgradient descent with momentum and dual-guided weighting.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        sg = dual_guided_subgradient(a, conv, n, top_k=top_k)
        
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iteration += 1
        lr *= 0.9995
        momentum = min(0.97, momentum + 0.0002)
        
        if no_improve >= 400:
            break
    
    return best_a, best_obj
