Let me think about what structure would help. The convolution c[m] = sum_i a[i]*a[m-i]. For this to be flat, we need the "products" a[i]*a[j] to be evenly spread across all "sums" i+j. This is like a "difference set" or " Sidon set" condition but for products.

If the a[i] values are all different and "randomly" assigned to positions, the products will be somewhat spread out. But to really flatten, we want a structured design.

import time
import numpy as np

# FFT-based convolution
def conv_fft(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

# Score computation
def compute_score(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = conv_fft(a)
    return 2.0 * n * c.max() / (S * S)

def coord_descent_pass(a, max_iter=10):
    n = len(a)
    a = a.copy()
    S = a.sum()
    c = conv_fft(a)
    
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            new_val, new_score = coord_step(a, i, c, S)
            if new_val != a[i]:

                # Update c incrementally
                d = np.zeros(2*n-1)

                c += (new_val - old) * d
                a[i] = new_val
                S += new_val - old
