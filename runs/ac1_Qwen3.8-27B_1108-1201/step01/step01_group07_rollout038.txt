```python
def solve_lp(f, M):
    n = len(f)
    # c = -ones(n) (we maximize sum(b), which is minimize -sum(b))
    c = -np.ones(n)
    
    # Constraints: (f*b)[k] <= M for k = 0..2n-2
    # (f*b)[k] = sum_j f[j] * b[k-j] (valid j)
    # For a fixed k, the coefficient of b[j] is f[k-j] if 0 <= k-j < n, i.e., k-n < j < k+1
    # So j ranges from max(0, k-n+1) to min(n-1, k)
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            A_ub[k, j] = f[k-j]
    
    b_ub = np.full(2*n-1, M)
    
    # Non-negativity: b >= 0 → bounds = (0, None)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

Actually, for the update f(t) = (1-t)*f + t*g, the convolution is:
conv_f(t) = (1-t)^2 * conv_f + 2t(1-t) * conv_fg + t^2 * conv_g

```python
        sg = np.zeros(n)
        thresh = y.max() * 0.001
        active_k = np.where(y > thresh)[0]

        for k_int in active_k:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                yk = y[k]
                seg_start = k - j_end
                seg_end = k - j_start
                sg[j_start:j_end + 1] += yk * a[seg_start:seg_end + 1][::-1]

        sg_mean = sg.mean()
        sg_centered = sg - sg_mean
        sg_norm = np.linalg.norm(sg_centered)
        if sg_norm < 1e-15:
            break
        sg_centered /= sg_norm

        # Multiplicative update
        step = eta
        log_factor = -step * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a /= a_sum

        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            no_improve = 0
            eta *= 0.6
            if eta < 0.01:
                eta = 0.15
                max_no_improve = 40

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussian bumps
    for sf in [0.1, 0.15, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Edge concentration
    for w_frac in [0.1, 0.2, 0.3]:
        w = max(1, int(n * w_frac))
        e = np.zeros(n)
        e[:w] = 1.0
        e[-w:] = 1.0
        e /= e.sum()
        inits[f'edge_{w_frac}'] = e

    # Cosine window
    cw = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        inits['cosine'] = cw / cw.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Random Dirichlet (more concentrated)
    inits['random_conc'] = rng.dirichlet(np.ones(n) * 0.1)

    # Parent-derived
    if parent is not None:
        if parent.size >= n:
            pp = parent[:n].copy()
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_interp'] = pp

    # Perturbed parent
    if parent is not None and 'parent' in inits:
        for eps in [0.05, 0.1, 0.2]:
            pp = inits['parent'] + eps * rng.standard_normal(n)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits[f'parent_pert_{eps}'] = pp

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search entry point. Uses LP-based iterative refinement combined with
    mirror descent to minimize the peak of the auto-convolution.
    Explores multiple sequence lengths and initializations in two phases.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Access parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    print(f"Parent length: {parent.size if parent is not None else 'N/A'}")

    # Phase 1: Quick scan over lengths and initializations
    print("Phase 1: Quick scan...")
    t_start = time.monotonic()

    lengths = [64, 128, 256, 512, 1024]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 120:
            break

        inits = generate_initializations(n, rng, parent)
        n_configs = len(inits)
        time_per_conf = min(15.0, max(3.0, 300.0 / max(1, n_configs)))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(time_per_conf, deadline - time.monotonic() - 120)
            if t_alloc < 2:
                break

            # Use mirror descent for quick scan (faster than LP)
            a_opt, obj_opt = mirror_descent_optimize(a0, time_limit=t_alloc,
                                                     rng=rng, t_param=40.0, eta0=0.4)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy(), obj_opt))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f}")

    print(f"  Phase 1 done in {time.monotonic() - t_start:.1f}s. Best: {best_score:.6f}")

    # Phase 2: Deep optimization with LP method on best candidates
    print("Phase 2: Deep LP optimization...")
    t_start2 = time.monotonic()

    scan_results.sort(key=lambda x: x[0])

    # Select top candidates with diversity
    top_candidates = []
    seen_lengths = set()
    for score, n, name, a_opt, obj_opt in scan_results:
        if len(top_candidates) < 10:
            top_candidates.append((n, name, a_opt, score))
            seen_lengths.add(n)
        elif n not in seen_lengths and len(top_candidates) < 14:
            top_candidates.append((n, name, a_opt, score))
            seen_lengths.add(n)

    # Always include current best
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), 'best', best_sequence.copy(), best_score))

    remaining = deadline - time.monotonic() - 60.0
    n_deep = len(top_candidates)
    time_per_deep = max(10.0, remaining / max(1, n_deep))

    for i, (n, name, a0, _) in enumerate(top_candidates):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 60)
        if t_alloc < 5:
            break

        # Try LP-based method
        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc * 0.6, rng=rng)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP n={n} {name}: {score:.6f}")

        # Polish with mirror descent
        if time.monotonic() < deadline - 60:
            t_remaining = deadline - time.monotonic() - 60
            a_opt2, _ = mirror_descent_optimize(a_opt, time_limit=min(30.0, t_remaining),
                                                rng=rng, t_param=80.0, eta0=0.3)
            score2 = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_sequence = a_opt2.copy()
                print(f"  MD polish n={n} {name}: {score2:.6f}")

    print(f"  Phase 2 done in {time.monotonic() - t_start2:.1f}s. Best: {best_score:.6f}")

    # Phase 3: Final deep dive on the best sequence
    print("Phase 3: Final deep dive...")
    t_start3 = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 30:
        n_best = len(best_sequence)
        remaining3 = deadline - time.monotonic() - 30

        # Try LP for a chunk of the remaining time
        chunk1 = remaining3 * 0.4
        a_opt, _ = lp_optimize(best_sequence, time_limit=chunk1, rng=rng)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep LP: {score:.6f}")

        # Mirror descent polish with various parameters
        for t_p in [60.0, 120.0, 200.0]:
            if time.monotonic() > deadline - 30:
                break
            t_left = deadline - time.monotonic() - 30
            a_opt2, _ = mirror_descent_optimize(best_sequence, time_limit=min(t_left * 0.25, 30.0),
                                                rng=rng, t_param=t_p, eta0=0.25)
            score2 = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_sequence = a_opt2.copy()
                print(f"  Deep MD t={t_p:.0f}: {score2:.6f}")

        # Dual LP mirror descent
        if time.monotonic() < deadline - 30:
            t_left = deadline - time.monotonic() - 30
            a_opt3, _ = dual_lp_mirror(best_sequence, time_limit=min(t_left * 0.3, 30.0),
                                       rng=rng, eta0=0.2)
            score3 = float(evaluate_sequence(a_opt3.tolist()))
            if np.isfinite(score3) and score3 < best_score:
                best_score = score3
                best_sequence = a_opt3.copy()
                print(f"  Dual LP MD: {score3:.6f}")

    # Phase 4: Try a few fresh random starts with the best length
    if time.monotonic() < deadline - 20:
        n_best = len(best_sequence) if best_sequence is not None else 128
        remaining4 = deadline - time.monotonic() - 20
        n_trials = max(1, int(remaining4 / 20))

        for trial in range(n_trials):
            if time.monotonic() > deadline - 20:
                break
            t_left = min(20.0, deadline - time.monotonic() - 20)
            if t_left < 3:
                break
            a_rand = rng.dirichlet(np.ones(n_best) * rng.uniform(0.1, 2.0))
            a_opt, _ = mirror_descent_optimize(a_rand, time_limit=t_left,
                                               rng=rng, t_param=60.0, eta0=0.4)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Fresh trial {trial}: {score:.6f}")

    print(f"  Phase 3+4 done in {time.monotonic() - t_start3:.1f}s.")

    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final evaluation
    final_score = float(evaluate_sequence([float(max(0, min(1000, x))) for x in best_sequence]))
    print(f"FINAL SCORE: {final_score:.6f} (length={len(best_sequence)})")
    return [float(max(0.0, min(1000.0, x))) for x in best_sequence]
```
