OK so my earlier analysis was correct - the two-level with e=1 is bad.

Now for the smooth cosine with eps=1: a[i] = (1 + cos(pi*i/(n-1))) normalized to sum n.
This goes from 2 (at i=0) to 0 (at i=n-1). It's a smooth "ramp" that's high on the left and low on the right.

The center convolution value is 1/(2n) as computed, giving score contribution of 1.
But other lags might be larger. Let me estimate.

For lag k near n-1 (say k = n-1-m):
c[n-1-m] = (1/n^2) * sum_i (1+cos(pi*i/(n-1)))(1+cos(pi*(n-1-m-i)/(n-1)))

Let u = pi*i/(n-1), v = pi*(n-1-m-i)/(n-1) = pi - u - pi*m/(n-1) = pi - u - delta (where delta = pi*m/(n-1)).

cos(v) = cos(pi - u - delta) = -cos(u+delta) = -cos(u)cos(delta) + sin(u)sin(delta)

So (1+cos(u))(1+cos(v)) = (1+cos(u))(1 - cos(u)cos(delta) + sin(u)sin(delta))

"""
Smoothed gradient descent with adaptive log-sum-exp temperature to minimize
the normalized autocorrelation peak of a non-negative sequence.
Uses FFT-accelerated convolution and gradient computation, simplex projection,
and multi-start diversification.
"""

import numpy as np
import time


def _convolve_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    N = 2 * n  # FFT size (enough for linear convolution)
    A = np.fft.rfft(a)
    C = np.fft.irfft(A * A)
    return C


def _lse(x, tau):
    """Log-sum-exp of x with temperature tau. Returns LSE(tau*x)/tau ≈ max(x)."""
    x_max = np.max(x)
    if x_max < -700 / tau:
        # Avoid underflow
        return x_max
    return x_max + np.log(np.sum(np.exp(tau * (x - x_max)))) / tau


def _lse_grad_weights(x, tau):
    """Compute softmax weights for LSE gradient: w_k = exp(tau*x_k) / sum(exp(tau*x_j))."""
    x_max = np.max(x)
    shifted = tau * (x - x_max)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    w = np.exp(shifted)
    w /= np.sum(w)
    return w


def _grad_objective(a, w):
    """
    Compute gradient of sum_k w_k * c_k where c = conv(a,a).
    grad_i = 2 * sum_j w_{i+j} * a[j] (for valid indices)
    Computed via FFT cross-correlation.
    """
    n = len(a)
    # We need g[i] = 2 * sum_j w[i+j] * a[j] for i in [0, n-1]
    # This is 2 * cross_correlation(w, a)[i]
    # cross_correlation(w, a)[i] = sum_j w[j] * a[j-i] (standard definition)
    # But we want sum_j w[j+i] * a[j] = sum_k w[k] * a[k-i] (with k=j+i)
    # = cross_correlation(w, a)[i] (with the convention that cross_corr[i] = sum_j w[j+i]*a[j])
    
    # Using FFT: cross_corr can be computed as ifft(fft(w) * conj(fft(a)))
    # with appropriate padding
    N = 1
    while N < len(w) + n:
        N *= 2
    
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    # ifft(fft(w) * conj(fft(a))) gives cross-correlation
    cross = np.fft.irfft(W * np.conj(A), N)
    # cross[i] = sum_j w[j] * a[(j-i) mod N]
    # With N >= len(w) + n - 1, this is the linear cross-correlation
    # cross[i] for i in [0, N-1]:
    # For i >= 0: cross[i] = sum_{j=i}^{len(w)-1} w[j] * a[j-i] (when j-i < n)
    # = sum_{k=0}^{min(len(w)-1-i, n-1)} w[i+k] * a[k]
    # This is exactly what we want: g[i] = 2 * cross[i]
    return 2.0 * cross[:n]


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score(a):
    """Compute the objective: 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = _convolve_fft(a)
    return 2.0 * n * np.max(c) / (S * S)


def _smoothed_score(a, tau):
    """Compute smoothed objective: 2n*LSE(tau*c)/S^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = _convolve_fft(a)
    return 2.0 * n * _lse(c, tau) / (S * S)


def _optimize_smoothed(a, deadline, rng, tau_init=10.0, tau_max=10000.0, 
                        lr_init=None, cool_rate=0.95, min_lr=1e-12):
    """
    Optimize using projected gradient descent on the smoothed objective.
    a is normalized to sum=1.
    """
    n = len(a)
    if lr_init is None:
        lr_init = 0.1
    
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(n) / n
    
    best = a.copy()
    best_score = _score(best)
    lr = lr_init
    tau = tau_init
    
    no_improve = 0
    patience = 200
    
    while time.monotonic() < deadline:
        # Compute convolution
        c = _convolve_fft(a)
        
        # Compute LSE weights
        w = _lse_grad_weights(c, tau)
        
        # Compute gradient
        grad = _grad_objective(a, w)
        
        # The objective is 2n/tau * LSE / S^2, but since we fixed S=1:
        # d/da [2n * LSE(tau*c)/tau] = 2n/tau * sum_k w_k * dc_k/da
        # For S fixed at 1, we just use the gradient of sum_k w_k * c_k
        # (the factor 2n/tau is a positive constant, doesn't affect direction)
        
        # Project gradient to tangent space (sum constraint): subtract mean
        grad = grad - np.mean(grad)
        
        # Normalized gradient step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Take step
        a_new = a - lr * grad / grad_norm
        
        # Project onto simplex
        a_new = _project_simplex(a_new, 1.0)
        
        # Check improvement
        c_new = _convolve_fft(a_new)
        cur_score = 2.0 * n * np.max(c_new)
        
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If no improvement, reduce learning rate and increase tau
            if no_improve % 50 == 0:
                lr *= cool_rate
                tau = min(tau * 1.5, tau_max)
            # Accept the step anyway (for exploration)
        
        a = a_new
        
        # Increase tau gradually
        tau = min(tau * 1.02, tau_max)
        
        if lr < min_lr:
            break
    
    return best, best_score / (1.0 ** 2)  # S=1, so just 2n*max_c


def _make_cosine_anti(n, eps):
    """Smooth anti-symmetric: 1 + eps*cos(pi*i/(n-1)), normalized."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_two_level(n, frac):
    """Two-level: first frac*n get higher value, rest get lower."""
    a = np.ones(n)
    cut = int(frac * n)
    a[:cut] = 2.0 / frac  # normalize so sum ≈ n
    a[cut:] = 2.0 / (1.0 - frac) * (1.0 - 2.0 * frac) if frac < 0.5 else 0.0
    # Actually let me just do: first part = p, second part = q, with p*cut + q*(n-cut) = 1
    # For anti-symmetry: p = (1+e)/n, q = (1-e)/n where e is asymmetry
    # Simpler: just make it anti-symmetric
    e = 2.0 * frac - 1.0  # range [-1, 1]
    a = np.ones(n)
    a[:cut] = (1.0 + e) / (2.0 * cut / n) if cut > 0 else 0
    a[cut:] = (1.0 - e) / (2.0 * (n - cut) / n) if n - cut > 0 else 0
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _perturb(a, rng, scale=0.05):
    """Perturb with smooth noise."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 4), size=min(5, n // 2))
    noise = np.zeros(n)
    i = np.arange(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * i / n + phase)
    noise /= (np.max(np.abs(noise)) + 1e-15)
    a_new = a + scale * noise
    a_new = _project_simplex(a_new, 1.0)
    return a_new
