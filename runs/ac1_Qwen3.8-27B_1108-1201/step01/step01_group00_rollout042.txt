import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def solve_lp(a, M):
    """Solve: max sum(g) s.t. a*g <= M (element-wise), g >= 0"""
    n = len(a)
    # Build constraint matrix: (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # But actually: (a*g)[k] = sum_j a[k-j]*g[j] for 0<=k-j<n
    # So A[k,j] = a[k-j] if 0 <= k-j < n
    # This is a convolution matrix ( Toeplitz-like )
    
    # For efficiency, build it as:
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] appears in c[k] for k = j, j+1, ..., j+n-1
        # coefficient is a[k-j]
        for k in range(j, j+n):
            A[k, j] = a[k-j]

    b_ub = np.full(2*n-1, M)

    if res.success:
        return res.x
    return np.ones(n) / n  # fallback

        M = np.max(c)
        
        # Solve LP

        if g_sum < 1e-10:

        g = g / g_sum * S

        a_new = (1 - t_blend) * a + t_blend * g

        M_new = np.max(c_new)
        score_new = 2.0 * n * M_new / (S * S)
        
        if score_new < best_score:

            a = a_new

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the score efficiently."""
    a = np.clip(a.astype(np.float64), 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def peak_and_grad(a):
    """Compute the peak value and its gradient with respect to a.
    For c = conv(a,a), peak at k*: dc[k*]/da[j] = 2*a[k*-j] if 0<=k*-j<n, else 0.

    n = len(a)
    c = fft_convolve(a, a)

    peak_val = c[k_star]
    # Gradient: grad[j] = 2*a[k_star - j] if 0 <= k_star - j < n

    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    return peak_val, grad, k_star, c
