def smooth_obj_and_grad(x, t, n):

    Objective and gradient in log-space (x = log(a)).
    a = softmax(x) = exp(x) / sum(exp(x))
    Returns (objective_value, gradient_wrt_x)

    a /= a.sum()

    scaled = t * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    smooth_max_val = lse / t

    w = np.exp(scaled - m)
    w /= w.sum()
    
    # Gradient of smooth_max w.r.t. a
    sg = conv_gradient_smooth(a, w)

    weighted_mean = np.dot(a, sg)

    obj = 2.0 * n * smooth_max_val
    
    # For tracking, also compute true max
    true_obj = 2.0 * n * conv.max()
    
    return true_obj, grad_x

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer in log-space
with adaptive temperature for smooth max approximation.
Explores multiple sequence lengths and initializations.

import numpy as np
import time
from scipy.optimize import minimize as sci_minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def conv_grad(a, w):
    """sg[j] = sum_k w[k] * a[k-j] via FFT."""
    n = len(a)
    lw = len(w)
    fft_size = 1 << (n + lw - 2).bit_length()
    wf = w[::-1]
    corr = np.fft.irfft(np.fft.rfft(wf, fft_size) * np.fft.rfft(a, fft_size), fft_size)
    return corr[lw - 1:lw - 1 + n]


def adam_run(a0, time_limit, t_init=40.0, lr_init=0.05):
    """Adam optimization in log-space with adaptive temperature."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a /= a.sum()
    
    x = np.log(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    t_s = t_init
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        max_c = conv.max()
        
        # Soft weights
        scaled = t_s * conv / (max_c + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        # Gradient
        sg = conv_grad(a, w)
        wm = np.dot(a, sg)
        gx = a * (sg - wm)
        
        # Adam update
        m = b1 * m + (1 - b1) * gx
        v = b2 * v + (1 - b2) * gx * gx
        mh = m / (1 - b1 ** step)
        vh = v / (1 - b2 ** step)
        
        dx = -lr * mh / (np.sqrt(vh) + eps)
        dx -= dx.max()
        
        x = x + dx
        a = np.exp(x - x.max())
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive params
        if stall == 40:
            lr *= 0.75
            stall = 0
        if stall == 100:
            t_s = min(t_s * 1.5, 500.0)
            lr = max(lr * 0.5, 0.005)
            stall = 0
        if stall > 300:
            break
    
    return best_a, best_obj


def lbfgs_run(a0, time_limit, t_init=50.0):
    """L-BFGS-B optimization in log-space."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a /= a.sum()
    x0 = np.log(a)
    
    state = {'t': t_init, 'count': 0, 't0': time.monotonic()}
    
    def func_grad(x):
        if time.monotonic() - state['t0'] > time_limit:
            raise OverflowError("time limit")
        state['count'] += 1
        xs = x - x.max()
        a = np.exp(xs)
        a /= a.sum()
        
        conv = fast_convolve(a)
        t = state['t']
        scaled = t * conv
        m = scaled.max()
        lse = m + np.log(np.exp(scaled - m).sum())
        
        w = np.exp(scaled - m)
        w /= w.sum()
        
        sg = conv_grad(a, w)
        wm = np.dot(a, sg)
        gx = a * (sg - wm)
        
        obj = 2.0 * n * lse / t
        return obj, gx
    
    try:
        res = sci_minimize(func_grad, x0, jac=True, method='L-BFGS-B',
                          options={'maxiter': 5000, 'maxfun': 5000,
                                   'gtol': 1e-8, 'ftol': 1e-10})
    except OverflowError:
        pass
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        return a, 2.0 * n * fast_convolve(a).max()
    
    x_opt = res.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt /= a_opt.sum()
    obj = 2.0 * n * fast_convolve(a_opt).max()
    return a_opt, obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g
    
    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    c /= c.sum()
    inits['cos'] = c
    
    # Raised cosine (Hanning)

    h /= h.sum()

    # Random smooth (low-pass filtered random)

    if r_smooth.sum() > 1e-10:
        r_smooth /= r_smooth.sum()
        inits['rand_smooth'] = r_smooth
    
    # Random Dirichlet
    inits['dirichlet'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Bimodal
    bi = np.zeros(n)
    bi[:n // 4] = 1.0
    bi[-n // 4:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Symmetric random
    half = n // 2
    rh = rng.random(half)
    rh = np.maximum(np.convolve(rh, np.ones(max(1, half//5)) / max(1, half//5), mode='same'), 0)
    sym = np.concatenate([rh, rh[::-1] if n % 2 == 0 else rh[:0:-1]])

Second term: `k-j = i`, i.e., `j = k-i`: gives `a_{k-i}` (if valid)

So `∂f/∂a_i = a_{k-i} + a_{k-i} = 2 a_{k-i}` (with `a_m = 0` outside `[0,n-1]`).

But wait! There's a subtlety. When `k-i = i` (i.e., `k = 2i`), both terms give the SAME contribution, and the term in the sum is `a_i \cdot a_i = a_i^2`. The derivative is `2a_i`. Our formula gives `2 \cdot a_{k-i} = 2 \cdot a_i`. ✓

When `k-i ≠ i`, the two terms are different terms in the sum: `a_i \cdot a_{k-i}` and `a_{k-i} \cdot a_i`. Each contributes `a_{k-i}` to the derivative. Total: `2a_{k-i}`. ✓

So the correct gradient IS `2 * a[k-i]`. The previous code was missing the factor of 2!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimizer in log-space
with adaptive temperature, heat restarts, L-BFGS-B refinement, and
multi-length exploration. Uses FFT-based convolution for efficiency.
"""

import numpy as np
import time
from scipy.optimize import minimize as sci_minimize


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def conv_grad(a, w):
    """Compute sg[j] = sum_k w[k] * a[k-j] via FFT (cross-correlation)."""
    n = len(a)
    lw = len(w)
    fft_size = 1 << (n + lw - 2).bit_length()
    wf = w[::-1]
    corr = np.fft.irfft(np.fft.rfft(wf, fft_size) * np.fft.rfft(a, fft_size), fft_size)
    return corr[lw - 1:lw - 1 + n]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def adam_run_v2(a0, time_limit, t_init=40.0, lr_init=0.1):
    """Adam in log-space with adaptive temperature and heat restarts."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a /= a.sum()

    x = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    t_s = t_init

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        if time.monotonic() - t0 > time_limit:
            break

        max_c = conv.max()

        # Soft peak weights
        scaled = t_s * conv / (max_c + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # Gradient in log-space
        sg = 2.0 * conv_grad(a, w)
        wm = np.dot(a, sg)
        gx = a * (sg - wm)

        # Adam update
        m = b1 * m + (1 - b1) * gx
        v = b2 * v + (1 - b2) * gx * gx
        mh = m / (1 - b1 ** step)
        vh = v / (1 - b2 ** step)
        dx = -lr * mh / (np.sqrt(vh) + eps)
        dx -= dx.max()
        x = x + dx

        a = np.exp(x - x.max())
        a = np.maximum(a, 1e-30)
        a /= a.sum()

        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-9:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive parameter adjustment
        if stall == 40:
            lr = max(lr * 0.7, 0.002)
            stall = 0
        elif stall == 100:
            t_s = min(t_s * 1.8, 400.0)
            lr = max(lr * 1.3, 0.02)
            m *= 0.3
            v *= 0.3
            stall = 0
        elif stall == 250:
            # Full restart with different temperature
            t_s = 30.0 + 20.0 * np.random.random()
            lr = 0.08
            m = np.zeros(n)
            v = np.zeros(n)
            step = 0
            stall = 0

    return best_a, best_obj


def lbfgs_run(a0, time_limit, t_val=60.0):
    """L-BFGS-B in log-space with smooth max objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a /= a.sum()
    x0 = np.log(a)

    state = {'t0': time.monotonic(), 'limit': time_limit}

    def func_grad(x):
        if time.monotonic() - state['t0'] > state['limit']:
            raise OverflowError("timeout")
        xs = x - x.max()
        a = np.exp(xs)
        a /= a.sum()

        conv = fast_convolve(a)
        t = t_val
        scaled = t * conv
        mval = scaled.max()
        lse = mval + np.log(np.exp(scaled - mval).sum())

        w = np.exp(scaled - mval)
        w /= w.sum()

        sg = 2.0 * conv_grad(a, w)
        wm = np.dot(a, sg)
        gx = a * (sg - wm)

        obj = 2.0 * n * lse / t
        return obj, gx

    try:
        res = sci_minimize(func_grad, x0, jac=True, method='L-BFGS-B',
                           options={'maxiter': 3000, 'maxfun': 3000,
                                    'gtol': 1e-7, 'ftol': 1e-10,
                                    'maxls': 40})
        x_opt = res.x
    except (OverflowError, Exception):
        return a, 2.0 * n * fast_convolve(a).max()

    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = np.maximum(a_opt, 1e-30)
    a_opt /= a_opt.sum()
    obj = 2.0 * n * fast_convolve(a_opt).max()
    return a_opt, obj


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    c = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    c /= c.sum()
    inits['cos'] = c

    h = 0.5 * (1 - np.cos(2 * np.pi * x / n))
    h = np.maximum(h, 1e-10)
    h /= h.sum()
    inits['han'] = h

    # Smoothed random
    r = rng.random(n)
    k = max(1, n // 8)
    r_s = np.convolve(r, np.ones(k) / k, mode='same')
    r_s = np.maximum(r_s, 1e-10)
    r_s /= r_s.sum()
    inits['rand_s'] = r_s

    # Dirichlet with concentration
    inits['dir'] = rng.dirichlet(np.ones(n) * 0.5)

    # Flat top with taper
    ft = np.ones(n)
    taper = max(1, n // 8)
    ft[:taper] = np.linspace(0.3, 1.0, taper)
    ft[-taper:] = np.linspace(1.0, 0.3, taper)
    ft /= ft.sum()
    inits['flat_taper'] = ft

    # Symmetric random
    half = n // 2
    rh = rng.random(half)
    rh = np.maximum(np.convolve(rh, np.ones(max(1, half // 4)) / max(1, half // 4), mode='same'), 1e-10)
    if n % 2 == 0:
        sym = np.concatenate([rh, rh[::-1]])
    else:
        sym = np.concatenate([rh, rh[:0:-1]])
    if len(sym) != n:
        sym = np.resize(sym, n)
    sym /= sym.sum()
    inits['sym'] = sym

    # Parent
    if parent is not None:
        if len(parent) >= n:
            pp = parent[:n].copy()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        pp = np.maximum(pp, 1e-10)
        pp /= pp.sum()
        inits['parent'] = pp

        # Perturbed parent
        pp2 = pp * (1.0 + 0.1 * rng.standard_normal(n))
        pp2 = np.maximum(pp2, 1e-10)
        pp2 /= pp2.sum()
        inits['parent_pert'] = pp2

    return inits
