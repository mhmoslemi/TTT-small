"""
Combined optimization for minimizing peak auto-convolution of non-negative sequences.
Uses smooth gradient descent (log-sum-exp + multiplicative updates), LP-based refinement,
and local search. Multiple lengths and initializations are explored.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    if n == 0:
        return float('inf')
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a.astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_parametric(n, family='raised_cosine', param=1.0):
    t = np.arange(n, dtype=np.float64)
    x = t / n
    if family == 'raised_cosine':
        a = (0.5 * (1.0 + np.cos(2.0 * np.pi * x))) ** param
    elif family == 'gaussian':
        center = 0.5
        sigma = param
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif family == 'parabolic':
        a = np.maximum(0.0, 1.0 - param * (2.0 * x - 1.0) ** 2)
    elif family == 'cosine_power':
        a = np.maximum(0.0, np.cos(np.pi * (x - 0.5))) ** param
    else:
        a = initial_raised_cosine(n)
    return normalize_nonneg(a)


def smooth_gradient_step(a, T, lr):
    """
    One step of smooth gradient descent with multiplicative updates.
    a: non-negative sequence (will be normalized internally)
    T: temperature for log-sum-exp smoothing
    lr: learning rate
    Returns: updated a (normalized, non-negative)
    """
    n = len(a)
    a = normalize_nonneg(a)
    
    # Compute convolution
    c = np.convolve(a, a)  # length 2n-1
    
    # Compute soft-max weights
    c_shifted = c - c.max()
    T_c = T * c_shifted
    # For numerical stability
    max_Tc = T_c.max()
    exp_Tc = np.exp(T_c - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    
    # Compute gradient: grad[m] = 2 * sum_j w[j] * a[j-m]
    # Using convolution: grad = 2 * convolve(w, a[::-1])[n-1:2n-1]
    flip_a = a[::-1]
    conv_wa = np.convolve(w, flip_a)  # length (2n-1) + n - 1 = 3n-2
    grad = 2.0 * conv_wa[n-1:2*n-1]
    
    # Multiplicative update
    a_new = a * np.exp(-lr * grad)
    a_new = np.maximum(0.0, a_new)
    a_new = normalize_nonneg(a_new)
    
    return a_new, c.max() / (a.sum() ** 2)  # return updated a and current peak


def smooth_gradient_descent(a, time_limit_s=300, verbose=True):
    """
    Run smooth gradient descent with annealing temperature.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"  SGD start: score={best_score:.6f}, n={n}")
    
    # Annealing schedule
    T_start = 5.0
    T_end = 200.0
    lr_start = 0.1
    lr_end = 0.01
    total_iters = 2000
    
    stagnant = 0
    for it in range(total_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        # Anneal parameters
        frac = it / total_iters
        T = T_start + (T_end - T_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac
        
        a_new, peak_val = smooth_gradient_step(a, T, lr)
        
        score = score_sequence(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stagnant = 0
            if verbose and (it % 100 == 0 or it < 5):
                print(f"    SGD iter {it:4d}: score={score:.6f}, T={T:.1f}, lr={lr:.4f}")
        else:
            stagnant += 1
            a = a_new
        
        if stagnant > 200:
            if verbose:
                print(f"    SGD stagnant at iter {it}, best={best_score:.6f}")
            break
    
    if verbose:
        print(f"  SGD done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix(a, n_rows=None):
    n = len(a)
    if n_rows is None:
        n_rows = 2 * n - 1
    # Build sparse matrix for convolution: row i, col j contributes a[j] to (conv)[i]
    # (conv)[i] = sum_j a[j] * a[i-j]
    # For LP, we need: for each row i (lag), the constraint sum_j M[i,j] * g[j] <= M
    # where M[i,j] = a[i-j] if 0 <= i-j < n, else 0
    col_idx = []
    row_idx = []
    data = []
    for j in range(n):
        # a[j] contributes to rows i where 0 <= i-j < n, i.e., j <= i <= j+n-1
        lo = j
        hi = min(n_rows - 1, j + n - 1)
        for i in range(lo, hi + 1):
            row_idx.append(i)
            col_idx.append(j)
            data.append(a[i - j])
    
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    """Solve LP: maximize sum(g) subject to conv(a,g) <= M, g >= 0"""
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a, n_rows)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, max_iters=200, verbose=False):
    """LP-based iterative refinement"""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"  LP start: score={best_score:.6f}, n={n}")
    
    stagnant = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 20 == 0 or it < 3):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        
        if stagnant > 30:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.2))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement at iter {it}")
            break
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        a = a_new
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"  LP done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=120.0, rng=None, verbose=False):
    """Local search with random perturbations"""
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"  LS start: score={best_score:.6f}")
    
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Try different perturbation strategies
        strategy = rng.integers(0, 3)
        candidate = a.copy()
        
        if strategy == 0:
            # Random segment scaling
            width = max(2, int(rng.integers(n // 200, n // 5)))
            start = int(rng.integers(0, max(1, n - width)))
            stop = min(n, start + width)
            candidate[start:stop] *= np.exp(rng.normal(0.0, 0.05, size=stop - start))
        elif strategy == 1:
            # Smooth random perturbation
            pert = np.cumsum(rng.normal(0.0, 0.01, size=n))
            pert -= pert.mean()
            candidate *= np.exp(pert)
        else:
            # Symmetric perturbation
            half = n // 2
            width = max(2, int(rng.integers(1, n // 4)))
            start = int(rng.integers(0, max(1, half - width)))
            stop = min(half, start + width)
            candidate[start:stop] *= np.exp(rng.normal(0.0, 0.03, size=stop - start))
            candidate[2*half - stop:2*half - start] = candidate[start:stop]
        
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 50 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    
    if verbose:
        print(f"  LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length"""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n)
    return np.interp(new_x, old_x, a)
