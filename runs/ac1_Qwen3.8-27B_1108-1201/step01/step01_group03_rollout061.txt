```python
Multi-phase optimization for minimizing the autocorrelation peak score.
Uses gradient descent with soft-max smoothing, LP-based direction finding,
and structured random perturbations.
"""

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def compute_grad(f, alpha):
    """Compute gradient of soft-max score w.r.t. f."""
    n = len(f)
    conv = np.convolve(f, f)
    m = conv.max()
    weights = np.exp(alpha * (conv - m))
    weights /= weights.sum()
    grad = 2.0 * np.correlate(weights, f, mode='valid')
    return grad

def project_simplex(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    # Simple projection: clip and normalize
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s

def grad_step(f, alpha, eta):
    """One gradient descent step on the simplex."""
    grad = compute_grad(f, alpha)
    f_new = f - eta * grad
    return project_simplex(f_new)

def build_conv_matrix(n, f):
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2*n-1, n)).tocsr()
    return A

def lp_direction(f, M, time_limit=5.0):
    """Solve LP to find the best direction g."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1 + np.cos(x))
    elif method == 'sin2':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)**2
    elif method == 'rand':
        a = rng.random(n) + 0.1
    elif method == 'bimodal':
        x = np.arange(n, dtype=float)
        s = max(1, n//8)
        a = np.exp(-0.5*((x-n//4)/s)**2) + np.exp(-0.5*((x-3*n//4)/s)**2)
    elif method == 'gaussian':
        x = np.arange(n, dtype=float) - n/2
        a = np.exp(-x**2 / (2*(n/4)**2))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def block_perturb(f, rng, width=None, sigma=0.3):
    n = len(f)
    cand = f.copy()
    if width is None:
        width = int(rng.integers(1, max(2, min(n//4, 64))))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= np.exp(rng.normal(0, sigma, stop-start))
    cand = np.maximum(cand, 0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s

            stall = 0
```

```python
"""
Multi-phase optimizer combining soft-max gradient descent, LP direction finding,
penalized LP, coordinate descent, and structured random search to minimize
the autocorrelation peak score: score = 2n * max(convolve(a,a)) / (sum(a))^2.
"""

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute score faithfully matching evaluate_sequence logic."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def compute_soft_max_grad(f, alpha):
    """Gradient of soft-max approximation to max(convolve(f,f))."""
    n = len(f)
    conv = np.convolve(f, f)
    m = conv.max()
    weights = np.exp(alpha * (conv - m))
    weights /= weights.sum()
    # grad[m] = 2 * sum_l f[l] * weights[l+m]
    grad = 2.0 * np.correlate(weights, f, mode='valid')
    return grad, conv, weights


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1} via clip-and-normalize."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def build_conv_matrix(n, f):
    """Sparse matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_standard(f, M, time_limit=5.0):
    """Standard LP: max sum(g) s.t. A@f * g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_penalized(f, M, lam, time_limit=5.0):
    """Penalized LP: max sum(g) - lam*sum(s) s.t. (f*g)_k - s_k <= M, s>=0, g>=0."""
    n = len(f)
    A_fg = build_conv_matrix(n, f)
    # Variables: [g (n), s (2n-1)]
    n_s = 2 * n - 1
    n_vars = n + n_s
    # Objective: -sum(g) + lam*sum(s)
    c = np.zeros(n_vars)
    c[:n] = -1.0
    c[n:] = lam
    # Constraints: A_fg @ g - I_s @ s <= M
    # Build as sparse
    from scipy.sparse import identity, hstack, csr_matrix
    A_s = -csr_matrix(identity(n_s))
    A_full = hstack([A_fg, A_s])
    b = np.full(n_s, M)
    bounds = [(0.0, None)] * n_vars
    try:
        res = linprog(c, A_ub=A_full, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            return g
    except Exception:
        pass
    return None


def make_initial(n, method, rng):
    """Generate initial non-negative sum-normalized vector."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0.0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
    elif method == 'sin2':
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'rand':
        a = rng.random(n) + 0.1
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        s = max(1.0, n / 8.0)
        a = np.exp(-0.5 * ((x - n / 4.0) / s) ** 2) + \
            np.exp(-0.5 * ((x - 3.0 * n / 4.0) / s) ** 2)
    elif method == 'gaussian':
        x = np.arange(n, dtype=np.float64) - n / 2.0
        a = np.exp(-x ** 2 / (2.0 * (n / 4.0) ** 2))
    elif method == 'triangle':
        x = np.arange(n, dtype=np.float64)
        a = 1.0 - 2.0 * np.abs(x - n / 2.0) / n
    elif method == 'edge_heavy':
        x = np.arange(n, dtype=np.float64) - n / 2.0
        a = x ** 2 + 1.0
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def block_perturb(f, rng, width=None, sigma=0.3):
    """Multiplicative block perturbation, renormalized."""
    n = len(f)
    cand = f.copy()
    if width is None:
        width = int(rng.integers(1, max(2, min(n // 4, 64))))
    width = min(width, n)
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= np.exp(rng.normal(0.0, sigma, stop - start))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s


def gradient_step(f, alpha, eta):
    """One projected gradient descent step."""
    grad, _, _ = compute_soft_max_grad(f, alpha)
    f_new = f - eta * grad
    return project_simplex(f_new)


def coordinate_optimize(f, i, n_iter=50):
    """Optimize a single coordinate f[i] to reduce score."""
    n = len(f)
    best_val = f[i]
    best_sc = fast_score(f)
    for _ in range(n_iter):
        # Binary search for optimal f[i] in a neighborhood
        lo = max(0.0, best_val * 0.1)
        hi = best_val * 10.0 + 0.01
        if best_val < 1e-8:
            lo = 0.0
            hi = 1.0
        for _ in range(15):
            mid = (lo + hi) / 2.0
            f[i] = mid
            s = f.sum()
            if s < 0.01:
                f[i] = best_val
                return best_val, best_sc
            conv = np.convolve(f, f)
            sc = 2.0 * n * conv.max() / (s * s)
            if sc < best_sc:
                best_sc = sc
                best_val = mid
                lo = mid
            else:
                hi = mid
        f[i] = best_val
    return best_val, best_sc


def run_gradient_phase(f0, budget_s, alpha=5.0, eta0=0.05, seed=0):
    """Gradient descent with adaptive step size and periodic perturbations."""
    f = f0.copy()

    t_start = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(seed + 7)

    while time.monotonic() - t_start < budget_s:

        f_new = gradient_step(f, alpha, eta)
        sc = fast_score(f_new)

        if sc < best_sc - 1e-10:
            best_sc = sc
            best = f_new.copy()
            stall = 0
            eta = min(eta * 1.08, 0.5)
        elif sc < best_sc * 1.001:
            stall = 0
        else:
            stall += 1

        f = f_new

            stall = 0

            sigma = float(rng.uniform(0.1, 0.6))
            f_cand = block_perturb(f, rng, width=width, sigma=sigma)
            sc = fast_score(f_cand)
            if sc < best_sc:
                best_sc = sc
                best = f_cand.copy()
                f = f_cand.copy()
            elif sc < best_sc * 1.01 and rng.random() < 0.3:
                f = f_cand.copy()  # accept slightly worse for exploration

    return best, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for the best non-negative sequence minimizing
    score = 2*n*max(convolve(a,a)) / (sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 30.0)

    best = None
    best_sc = np.inf

    # Try to use parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all() and parent.sum() > 0.01:
            p = parent / parent.sum()
            sc = fast_score(p)
            best = p.copy()
            best_sc = sc
            print(f"Parent: n={parent.size}, score={sc:.6f}", flush=True)
    except Exception:
        pass

    if best is None:
        best = np.ones(512)
        best_sc = fast_score(best)
        print(f"Default uniform: n=512, score={best_sc:.6f}", flush=True)

    def try_update(cand, sc):
        nonlocal best, best_sc
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = np.maximum(cand, 0.0).copy()
            print(f"  >>> NEW BEST: {sc:.6f} (n={best.size})", flush=True)
            return True
        return False

    # ============ Phase 1: Multi-start gradient (120s) ============
    print(f"\n=== Phase 1: Multi-start gradient (120s) ===", flush=True)
    p1_budget = 120.0
    p1_start = time.monotonic()

    n_parent = best.size
    configs = [
        (n_parent, 'uniform'), (n_parent, 'cosine'), (n_parent, 'sin2'),
        (n_parent, 'bimodal'), (n_parent, 'gaussian'), (n_parent, 'rand'),
        (n_parent, 'triangle'), (n_parent, 'edge_heavy'),
    ]
    if n_parent in (256, 512):
        configs.append((n_parent * 2, 'cosine'))
        configs.append((n_parent * 2, 'bimodal'))
    if n_parent > 256:
        configs.append((n_parent // 2, 'cosine'))

    per_config = p1_budget / len(configs)

    for (n, method) in configs:
        if time.monotonic() > p1_start + p1_budget:
            break
        f0 = make_initial(n, method, rng)
        c_best, c_sc = run_gradient_phase(f0, per_config * 0.9, alpha=5.0, eta0=0.04,
                                          seed=hash((n, method)) % 10000)
        try_update(c_best, c_sc)
        # If different length, note it
        print(f"  P1: n={n:4d} {method:12s} -> {c_sc:.6f}", flush=True)

    # ============ Phase 2: LP-based optimization (600s) ============
    print(f"\n=== Phase 2: LP optimization (600s) ===", flush=True)
    p2_budget = min(600.0, max(50.0, total_deadline - time.monotonic() - 700.0))
    p2_start = time.monotonic()

    # Run LP on the current best
    print(f"  LP on best (n={best.size}), budget={p2_budget*0.5:.0f}s", flush=True)
    c_best, c_sc = run_lp_phase(best, p2_budget * 0.5, lp_time=3.0, t_relax=0.4, seed=1)
    try_update(c_best, c_sc)

    # Run hybrid on best
    remaining = p2_budget - (time.monotonic() - p2_start)
    if remaining > 30:
        print(f"  Hybrid on best, budget={remaining*0.5:.0f}s", flush=True)
        c_best, c_sc = run_hybrid_phase(best, remaining * 0.5, lp_every=4,
                                        lp_time=2.5, seed=2)
        try_update(c_best, c_sc)

    # Try LP from a fresh cosine start
    remaining = p2_budget - (time.monotonic() - p2_start)
    if remaining > 30:
        f_fresh = make_initial(best.size, 'cosine', rng)
        print(f"  LP from cosine start, budget={remaining*0.5:.0f}s", flush=True)
        c_best, c_sc = run_lp_phase(f_fresh, remaining * 0.5, lp_time=2.0,
                                    t_relax=0.3, seed=3)
        try_update(c_best, c_sc)

    # ============ Phase 3: Intensive hybrid on best (500s) ============
    print(f"\n=== Phase 3: Intensive hybrid (500s) ===", flush=True)
    p3_budget = min(500.0, max(50.0, total_deadline - time.monotonic() - 250.0))
    p3_start = time.monotonic()

    # Multiple hybrid runs with different parameters
    hybrid_configs = [
        (4, 2.0, 0.3),
        (6, 3.0, 0.5),
        (3, 1.5, 0.2),
    ]
    per_h = p3_budget / len(hybrid_configs)

    for i, (lp_every, lp_time, t_init) in enumerate(hybrid_configs):
        if time.monotonic() > p3_start + p3_budget:
            break
        f_start = best.copy()
        if i > 0:
            f_start = block_perturb(best, rng, width=16, sigma=0.3)
        c_best, c_sc = run_hybrid_phase(f_start, per_h * 0.9, lp_every=lp_every,
                                        lp_time=lp_time, seed=100 + i)
        try_update(c_best, c_sc)
        print(f"  P3 config{i}: lp_every={lp_every}, lp_t={lp_time} -> {c_sc:.6f}",
              flush=True)

    # ============ Phase 4: Local search (200s) ============
    print(f"\n=== Phase 4: Local search (200s) ===", flush=True)
    p4_budget = min(200.0, max(30.0, total_deadline - time.monotonic() - 100.0))
    p4_start = time.monotonic()

    c_best, c_sc = run_local_search(best, p4_budget * 0.7, seed=42)
    try_update(c_best, c_sc)

    remaining = p4_budget - (time.monotonic() - p4_start)
    if remaining > 20:
        c_best, c_sc = run_local_search(best, remaining * 0.8, seed=99)
        try_update(c_best, c_sc)
        print(f"  P4 local search -> {c_sc:.6f}", flush=True)

    # ============ Phase 5: Final gradient polish (100s) ============
    print(f"\n=== Phase 5: Final polish ===", flush=True)
    p5_budget = max(10.0, total_deadline - time.monotonic() - 20.0)
    c_best, c_sc = run_gradient_phase(best, p5_budget, alpha=20.0, eta0=0.02, seed=777)
    try_update(c_best, c_sc)

    # ============ Final verification ============
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(256)

    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = np.ones(256)
        final_sc = float(evaluate_sequence(best.tolist()))

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: n={best.size}, fast={fast_score(best):.6f}, eval={final_sc:.6f}", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in best]
```
