"""
LP-based refinement combined with gradient optimization and local search
for minimizing the normalized peak autocorrelation of non-negative sequences.
Key innovation: Uses linear programming to find optimal descent directions
that provably control the peak, combined with multi-phase search strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))

def softmax_normalize(theta):
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def build_conv_matrix(f, n):
    """Build (2n-1) x n matrix C where C @ g = conv(f, g)."""
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = f
    return C

def smooth_max_value_and_weights(c, tau):
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    return value, weights

def grad_smooth_max_conv(a, tau):
    """Gradient of smooth_max(conv(a,a), tau) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = smooth_max_value_and_weights(c, tau)
    # d[j] = 2 * sum_l a[l] * w[l+j] = 2 * conv(a_rev, w)[n-1+j]
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = 2.0 * corr[n-1:2*n-1]
    return value, g, c

def optimize_gradient_nesterov(a0, time_limit, rng):
    """
    Gradient descent with Nesterov momentum on the smoothed objective.
    a is normalized to sum = sqrt(2n).
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    a = a / np.sum(a) * S  # normalize
    
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    
    t0 = time.monotonic()
    tau_start = 50.0 / n
    tau_end = 0.5 / n
    best_a = a.copy()
    best_score = local_score(a)
    step = 0
    lr = 0.1
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)
        
        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.clip(a_look, 0.0, None)
        
        value, g_a, c = grad_smooth_max_conv(a_look, tau)
        
        # Project gradient to maintain sum constraint
        # We want: new a has same sum. So subtract the mean of the gradient.
        g_projected = g_a - np.mean(g_a)
        
        step += 1
        # Adaptive learning rate
        lr = lr * 0.9995 + 0.001 * (1.0 / (1 + step * 0.01))
        
        v = mu * v - lr * g_projected
        a_new = a + v
        a_new = np.clip(a_new, 0.0, None)
        
        # Renormalize
        if np.sum(a_new) < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / np.sum(a_new) * S
        
        score = local_score(a_new)
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def lp_refine(a0, time_limit, rng):
    """
    LP-based refinement. Iteratively:
    1. Compute M = max(conv(a,a))
    2. Solve LP to find best direction g
    3. Blend a and g
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    a = np.clip(a, 0.0, None)
    a = a / np.sum(a) * S  # normalize so sum = S
    
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    # Precompute the convolution matrix
    # Actually, it changes with a, so we can't precompute it.
    # But we can precompute its structure.
    
    iteration = 0
    step_size = 0.3
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(a, a))
        
        # Build constraint matrix: C @ g <= M
        # C[k, j] = a[k-j] if 0 <= k-j < n
        C = np.zeros((2*n-1, n))
        for j in range(n):
            C[j:j+n, j] = a
        
        # Solve LP: maximize sum(g) s.t. C@g <= M, g >= 0
        c_obj = -np.ones(n)
        b = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c_obj, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                            options={'presolve': True, 'time_limit': 30})
        except Exception:
            break
        
        if not result.success or result.x is None:
            # Try with a larger M
            result2 = linprog(c_obj, A_ub=C, b_ub=np.full(2*n-1, 2*M), 
                             bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': 30})
            if not result2.success:
                break
            g = result2.x
        else:
            g = result.x
        
        # Check if g is useful
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        # Rescale g to have same sum as a
        g = g / g_sum * S
        
        # Line search for step size
        best_t = 0.1
        base_score = local_score(a)
        scores = []
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.clip(a_trial, 0.0, None)
            if np.sum(a_trial) < 1e-10:
                continue
            a_trial = a_trial / np.sum(a_trial) * S
            s = local_score(a_trial)
            scores.append((s, a_trial.copy(), t))
        
        if scores:
            scores.sort(key=lambda x: x[0])
            best_s, best_trial, best_t = scores[0]
            
            if best_s < best_score:
                best_score = best_s
                best_a = best_trial.copy()
            a = best_trial
        else:
            break
        
        iteration += 1
        if iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iteration}, score={best_score:.6f}, M={M:.4f}, t_elapsed={elapsed:.1f}s")
        
        # Check for convergence
        if M < best_score * S * S / (2.0 * n) * 1.001:
            # Very close to convergence
            if iteration > 20:
                # Try smaller steps
                step_size *= 0.5
                if step_size < 0.01:
                    break
    
    return best_a, best_score

def local_search_polish(a0, time_limit, rng):
    """Simulated annealing local search."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = a0.copy()
    a = np.clip(a, 0.0, None)
    a = a / np.sum(a) * S
    
    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.3:
            # Targeted: reduce peak contributors
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(2, n // 20)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(8, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.5:
            # Block
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.7:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 25))))
                start = int(rng.integers(0, n - width))
                candidate[start:start+width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.85:
            # Symmetrize
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.005, size=n))
        else:
            # Global
            candidate = a * np.exp(rng.normal(0.0, 0.005, size=n))
        
        candidate = np.clip(candidate, 0.0, None)
        if np.sum(candidate) < 1e-10:
            candidate = a.copy()
        else:
            candidate = candidate / np.sum(candidate) * S
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            a = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best_a, best_score

def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_multi_tone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k, amp in [(1, 0.2), (2, 0.15), (3, 0.1), (5, 0.08), (8, 0.05)]:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2*np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def propose_candidate(seed=42, budget_s=1800, **kwargs):

    (3) local search polish. Uses LP to find provably good descent directions.
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # Phase 1: Gradient optimization
    print("\n=== Phase 1: Gradient optimization ===")
    configs = [
        (256, 3),  # length, num_inits

        (512, 2),
        (512, 1),
        (768, 1),
    ]
    
    total_grad_time = 400.0
    time_per_grad = total_grad_time / len(configs)
    
    for n, n_inits in configs:
        if time.monotonic() > deadline - 600:
            break
        for init_i in range(n_inits):
            if time.monotonic() > deadline - 600:

            elif init_i == 1:

            else:

"""
LP-based refinement combined with gradient optimization and local search
for minimizing the normalized peak autocorrelation of non-negative sequences.
Key innovation: Uses linear programming to find descent directions that provably
control the cross-correlation peak, combined with adaptive line search and
multi-phase exploration. The LP direction maximizes total mass subject to the
convolution bound, providing a structurally informed step that accelerates
convergence near the optimum.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast computation of the score (same formula as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def smooth_max_value_and_weights(c, tau):
    """Compute soft-max value and weights for a vector c."""
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    return value, weights


def grad_smooth_max_conv(a, tau):
    """
    Compute value and gradient of soft_max(conv(a,a), tau) w.r.t. a.
    Gradient: d[j] = 2 * sum_l a[l] * w[l+j] = 2 * conv(a_rev, w)[n-1+j]
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = smooth_max_value_and_weights(c, tau)
    a_rev = a[::-1]
    corr = fft_convolve(a_rev, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g, c


def build_conv_matrix(f, n):
    """Build (2n-1) x n matrix C where C @ g = conv(f, g).
    C[k, j] = f[k-j] if 0 <= k-j < n.
    Column j is f shifted down by j positions.
    """
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def optimize_gradient_nesterov(a0, time_limit, rng):

    tau_start = 50.0 / n
    tau_end = 0.3 / n

    best_score = local_score(a)

    base_lr = 0.15

        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)

        a_look = np.clip(a_look, 0.0, None)
        if np.sum(a_look) < 1e-10:
            a_look = a.copy()
        else:
            a_look = a_look / np.sum(a_look) * S

        value, g_a, c = grad_smooth_max_conv(a_look, tau)

        g_proj = g_a - np.mean(g_a)

        step += 1
        lr = base_lr / (1.0 + 0.05 * np.sqrt(step))

    """
    LP-based refinement: iteratively solve LP to find descent direction,
    then blend with line search.
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.clip(a0, 0.0, None).astype(np.float64)
    a = a / np.sum(a) * S

    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    iteration = 0
    stuck_count = 0
    last_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 5:
            break

        c_aa = fft_convolve(a, a)
        M = float(np.max(c_aa))

        # Build constraint matrix
        C = build_conv_matrix(a, n)

        # Solve LP: maximize sum(g) s.t. C@g <= M, g >= 0
        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            result = linprog(c_obj, A_ub=C, b_ub=b, bounds=bounds,
                            method='highs', options={'presolve': True})
        except Exception:
            break

        if not result.success or result.x is None:
            break

        g = result.x
        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break

        g = g / g_sum * S

        current_score = best_score
        best_trial_score = np.inf
        best_trial = None

        for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.7, 0.8, 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.clip(a_trial, 0.0, None)
            if np.sum(a_trial) < 1e-10:
                continue
            a_trial = a_trial / np.sum(a_trial) * S
            s = local_score(a_trial)
            if s < best_trial_score:
                best_trial_score = s
                best_trial = a_trial.copy()

        if best_trial is None:
            break

        if best_trial_score < best_score - 1e-10:
            best_score = best_trial_score
            best_a = best_trial.copy()
            a = best_trial
            stuck_count = 0
        elif best_trial_score < current_score + 1e-10:
            # Accept if not worse than current (helps escaping plateaus)
            a = best_trial
            stuck_count += 1
        else:
            stuck_count += 1

        if stuck_count > 30:
            # Perturb and restart
            a = a * np.exp(rng.normal(0.0, 0.02, size=n))
            a = np.clip(a, 0.0, None)
            if np.sum(a) > 1e-10:
                a = a / np.sum(a) * S
            stuck_count = 0

        iteration += 1
        if iteration % 5 == 0:
            print(f"      LP iter={iteration}, score={best_score:.6f}, M={M:.4f}, "
                  f"g_sum={g_sum:.2f}, t={elapsed:.1f}s")

    return best_a, best_score


def local_search_polish(a0, time_limit, rng):
    """Simulated annealing local search with targeted peak reduction."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.clip(a0, 0.0, None).astype(np.float64)
    a = a / np.sum(a) * S

    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    temperature = 0.003
    cooling = 0.99975

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted: reduce peak contributors
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 16)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.035, 0.012, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.035, 0.012, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.7:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.82:
            # Partial symmetrization
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.008, size=n))
        elif r < 0.92:
            # Peak-directed: move mass from peak-region to anti-peak
            conv = fft_convolve(a, a)
            sorted_idx = np.argsort(conv)
            peak_regions = sorted_idx[-5:]
            low_regions = sorted_idx[:max(3, n // 10)]
            for pi in peak_regions:
                i = int(rng.integers(0, n))
                j = pi - i
                if 0 <= i < n and 0 <= j < n:
                    candidate[i] *= np.exp(-0.02)
                    lj = int(rng.choice(low_regions))
                    candidate[lj] *= np.exp(0.02)
        else:
            # Small global
            candidate = a * np.exp(rng.normal(0.0, 0.008, size=n))

        candidate = np.clip(candidate, 0.0, None)
        if np.sum(candidate) < 1e-10:
            candidate = a.copy()
        else:
            candidate = candidate / np.sum(candidate) * S

        new_score = local_score(candidate)

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            a = candidate
            current_score = new_score

        temperature *= cooling

    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    """Create a chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.015 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Create a multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k, amp in [(1, 0.22), (2, 0.15), (3, 0.10), (5, 0.07), (7, 0.04)]:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.015 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_quadratic_phase_init(n, rng):
    """Quadratic phase (chirp) with random modulation."""
    t = np.arange(n, dtype=np.float64)
    phase = 2.0 * np.pi * rng.uniform(0.3, 0.8) * t * t / n
    a = 1.0 + 0.3 * np.cos(phase) + 0.1 * np.sin(2.0 * phase)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for optimal non-negative step function:
    Phase 1: Gradient-based optimization (multiple inits, multiple lengths)
    Phase 2: LP-based refinement (structural descent directions)
    Phase 3: Local search polish (simulated annealing)
    Phase 4: Final LP refinement on best candidate
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Gradient optimization
    print("\n=== Phase 1: Gradient optimization ===")
    configs = [
        (256, ['provided', 'chirp', 'multi_tone']),
        (256, ['quad_phase', 'chirp_b']),
        (384, ['chirp', 'multi_tone']),
        (512, ['chirp', 'multi_tone']),
        (640, ['chirp']),
    ]

    time_per_grad = 60.0
    for n, init_types in configs:
        if time.monotonic() > deadline - 700:
            break
        for init_type in init_types:
            if time.monotonic() > deadline - 700:
                break

            # Generate initial sequence
            if init_type == 'provided' and start_seq is not None:
                if len(start_seq) == n:
                    a0 = start_seq.copy()
                else:
                    x_old = np.linspace(0, 1, len(start_seq))
                    x_new = np.linspace(0, 1, n)
                    a0 = np.interp(x_new, x_old, start_seq)
            elif init_type == 'chirp':
                a0 = make_chirp_init(n, 0.3 + 0.3 * rng.random(), rng)
            elif init_type == 'chirp_b':
                a0 = make_chirp_init(n, 0.5 + 0.2 * rng.random(), rng)
            elif init_type == 'multi_tone':
                a0 = make_multi_tone_init(n, rng)
            elif init_type == 'quad_phase':
                a0 = make_quadratic_phase_init(n, rng)
            else:
                a0 = make_chirp_init(n, 0.4, rng)

            a0 = np.clip(a0, 0.01, 1000.0)
            t0 = time.monotonic()
            a_best_g, score_g = optimize_gradient_nesterov(a0, time_per_grad, rng)
            elapsed = time.monotonic() - t0
            print(f"  n={n}, init='{init_type}', score={score_g:.6f}, time={elapsed:.1f}s")

            if score_g < best_score:
                best_score = score_g
                best_seq = a_best_g.copy()
                print(f"    *** New best: {best_score:.6f} ***")

    print(f"  Phase 1 complete. Best: {best_score:.6f}")

    # Phase 2: LP-based refinement
    print(f"\n=== Phase 2: LP refinement (start={best_score:.6f}) ===")
    if best_seq is not None:
        lp_time = min(800.0, max(60.0, deadline - time.monotonic() - 200.0))
        print(f"  LP budget: {lp_time:.1f}s")
        a_lp, score_lp = lp_refine(best_seq, lp_time, rng)
        print(f"  After LP: score={score_lp:.6f}")

        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Local search polish
    print(f"\n=== Phase 3: Local search polish ===")
    if best_seq is not None:
        polish_time = min(100.0, max(15.0, deadline - time.monotonic() - 60.0))
        print(f"  Polish budget: {polish_time:.1f}s")
        a_pol, score_pol = local_search_polish(best_seq, polish_time, rng)
        print(f"  After polish: score={score_pol:.6f}")

        if score_pol < best_score:
            best_score = score_pol
            best_seq = a_pol.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 4: Final LP refinement if time permits
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Phase 4: Final LP polish ===")
        final_lp_time = min(80.0, deadline - time.monotonic())
        a_final, score_final = lp_refine(best_seq, final_lp_time, rng)
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"    *** Final best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
