   LP-based iterative refinement combined with gradient polishing and simulated annealing
   for minimizing the normalized peak of autocorrelation of non-negative step functions.
   """

   import time
   from scipy.optimize import linprog
   try:
       import cvxpy as cp
       HAS_CVXPY = True
   except:
       HAS_CVXPY = False


   def fft_convolve(a, b):
       """Compute linear convolution using FFT."""
       n, m = len(a), len(b)
       size = 1
       while size < n + m - 1:
           size <<= 1
       return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


   def local_score(a):
       """Fast local computation of the score."""
       a = np.clip(a, 0, 1000)
       n = len(a)
       s = np.sum(a)
       if s < 0.01:
           return np.inf
       c = fft_convolve(a, a)
       return 2.0 * n * np.max(c) / (s * s)


   def lp_find_g(a, M, solver='auto'):
       """
       Find g >= 0 maximizing sum(g) subject to conv(a, g) <= M.
       Returns the optimal g or None if infeasible.
       """
       n = len(a)
       # conv(a, g)[k] = sum_{i,j: i+j=k} a[i]*g[j] for i,j in [0,n-1]
       # = sum_j a[k-j]*g[j] for valid j
       # Build constraint matrix C of shape (2n-1, n)
       # C[k, j] = a[k-j] if 0 <= k-j < n else 0
       
       # Use sparse representation for efficiency

       # C is a banded matrix

       # For j in range(max(0, k-n+1), min(k, n-1)+1): C[k,j] = a[k-j]
       
       # Let's build it row by row
       C = np.zeros((2*n-1, n))
       for k in range(2*n-1):

           for j in range(j_min, j_max + 1):

def simulated_annealing(a_init, time_limit, rng):
    """Simulated annealing with targeted peak reduction."""
    n = len(a_init)
    current = a_init.astype(np.float64).copy()
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temp = 0.005
    cooling = 0.99985
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted: reduce peak contributors
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, top_k))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(5, len(other))
                inc_idx = rng.choice(other, n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, n_inc))
        elif r < 0.7:
            # Local block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.04, width))
            # Compensate elsewhere
            comp_idx = int(rng.integers(0, n))
            if comp_idx < n:
                candidate[comp_idx] *= np.exp(rng.normal(0, 0.02))
        elif r < 0.85:
            # Symmetrize partially
            sym = 0.5 * (current + current[::-1])
            candidate = sym * np.exp(rng.normal(0, 0.01, n))
        else:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0, 0.015, n))
        
        candidate = np.clip(candidate, 0, None)
        cs = np.sum(candidate)
        if cs < 1e-10:
            continue
        candidate = candidate / cs
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            current = candidate
            current_score = new_score
        
        temp *= cooling
    
    return best, best_score


def make_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    return np.clip(a + 0.01 * rng.standard_normal(n), 0.01, None)


def make_multitone(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k, amp in [(1, 0.2), (2, 0.15), (3, 0.1), (5, 0.07)]:
        a += amp * np.cos(2 * np.pi * k * t / n + rng.uniform(0, 2*np.pi))
    return np.clip(a + 0.01 * rng.standard_normal(n), 0.01, None)

"""
LP-based iterative refinement with gradient polish and simulated annealing
for minimizing the normalized autocorrelation peak of non-negative step functions.
Uses the state-of-the-art LP direction finding: at each step, solve an LP to find
the optimal blending direction that keeps cross-correlation bounded, then blend.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the objective score for a sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def build_cross_conv_matrix(a):
    """Build sparse (2n-1) x n matrix C where (C@g)[k] = conv(a,g)[k]."""
    n = len(a)
    total_nonzeros = sum(min(k+1, n, 2*n-1-k) for k in range(2*n-1))
    rows = np.empty(total_nonzeros, dtype=np.int32)
    cols = np.empty(total_nonzeros, dtype=np.int32)
    vals = np.empty(total_nonzeros, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_find_g(a, M):
    """Solve LP: max sum(g) s.t. C(a) @ g <= M*1, g >= 0."""
    n = len(a)
    C = build_cross_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def gradient_polish(a_init, time_limit, rng):
    """Polish using L-BFGS-B with softmax-smoothed max, multiple tau values."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a)

    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()

    def make_fun_grad(tau):
        def fun_and_grad(x):
            x = np.maximum(x, 1e-12)
            s = np.sum(x)
            xn = x / s
            c = fft_convolve(xn, xn)
            cm = np.max(c)
            sh = (c - cm) / tau
            ev = np.exp(sh)
            se = np.sum(ev)
            sm = cm + tau * np.log(se)
            w = ev / se
            # Gradient via O(n^2) computation (fast for n<=1024)
            grads = np.zeros(n)
            for i in range(n):
                j_max = min(n, 2 * n - 1 - i)
                if j_max > 0:
                    grads[i] = 2.0 * np.dot(xn[:j_max], w[i:i + j_max])
            inner = np.dot(xn, grads)
            gx = (grads - inner * xn) / s
            val = 2.0 * n * sm
            return val, gx
        return fun_and_grad

    taus = [2.0 / n, 1.0 / n, 0.5 / n, 0.25 / n, 0.1 / n, 0.05 / n]
    time_per = time_limit / len(taus)

    for tau in taus:
        if time.monotonic() - t0 > time_limit:

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the objective score for a sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def build_cross_conv_matrix(a):
    """Build sparse (2n-1) x n matrix C where (C@g)[k] = conv(a,g)[k]."""
    n = len(a)
    total_nz = 0
    for k in range(2 * n - 1):
        total_nz += min(n, k + 1, 2 * n - 1 - k)
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    vals = np.empty(total_nz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_find_g(a, M):
    """Solve LP: max sum(g) s.t. C(a) @ g <= M*ones, g >= 0."""
    n = len(a)
    C = build_cross_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
