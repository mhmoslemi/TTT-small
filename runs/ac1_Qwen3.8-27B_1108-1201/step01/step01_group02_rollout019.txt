"""
Hybrid optimization for minimizing peak auto-convolution of non-negative sequences.
Combines LP-based refinement with line search, smooth multiplicative weights,
Gerchberg-Saxton frequency-domain iterations, and local search.
Uses FFT-based convolution for efficiency at larger sizes.

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

# Global best tracking
_best_score_global = float('inf')
_best_seq_global = None


def score(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s ** 2)


def normalize(a):
    a = np.maximum(0.0, a.astype(np.float64))
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    N = 1
    while N < 2 * n:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_asymmetric(n, rng):
    """Create an asymmetric initial sequence to break symmetry."""
    a = initial_raised_cosine(n)
    # Apply a linear ramp to create asymmetry
    ramp = np.linspace(0.5, 1.5, n)
    a = a * ramp
    # Add some random perturbation
    a += 0.1 * np.abs(rng.standard_normal(n))
    return normalize(a)


def initial_two_bump(n, rng):
    """Two Gaussian bumps at different positions with different widths."""
    a = np.zeros(n)
    c1 = n * rng.uniform(0.1, 0.4)
    c2 = n * rng.uniform(0.6, 0.9)
    w1 = n * rng.uniform(0.05, 0.2)
    w2 = n * rng.uniform(0.05, 0.2)
    t = np.arange(n, dtype=np.float64)
    a += np.exp(-0.5 * ((t - c1) / w1) ** 2)
    a += np.exp(-0.5 * ((t - c2) / w2) ** 2)
    return normalize(a)


def initial_known(seed_seq, target_n):
    """Interpolate a known sequence to a target length."""
    n = len(seed_seq)
    if n == target_n:
        return normalize(seed_seq.copy())
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    interpolated = np.interp(x_new, x_old, seed_seq)
    return normalize(interpolated)


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, min(j + n, n_rows), dtype=np.int32) for j in range(n)])
    data = np.concatenate([a[:n_rows - j] for j in range(n)])
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def line_search_blend(a, b, n_trials=20):
    """Find best t in [0,1] minimizing score((1-t)*a + t*b)."""
    global _best_score_global, _best_seq_global
    best_t = 0.0
    best_s = score(a)
    for i in range(1, n_trials + 1):
        t = i / n_trials
        candidate = (1.0 - t) * a + t * b
        candidate = normalize(candidate)
        s = score(candidate)
        if s < best_s:
            best_s = s
            best_t = t
    return (1.0 - best_t) * a + best_t * b, best_t, best_s


def lp_refine(a, time_limit_s=300.0, verbose=False):
    """LP-based iterative refinement with line search."""
    global _best_score_global, _best_seq_global
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    stagnant = 0

    it = 0
    while time.monotonic() - start_time < time_limit_s:
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = fft_convolve(a, a)
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.4))
        b = solve_lp_step(a, M, timeout=timeout)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-10:
            break
        b = b / sum_b

        # Line search for best blend
        a_new, t_used, s_new = line_search_blend(a, b, n_trials=15)

        if s_new < best_score - 1e-12:
            best_score = s_new
            best_a = a_new.copy()
            a = a_new.copy()
            stagnant = 0
            if s_new < _best_score_global:
                _best_score_global = s_new
                _best_seq_global = a_new.copy()
                if verbose:
                    print(f"    LP global best: {s_new:.6f} (n={n}, t={t_used:.3f})")
        else:
            stagnant += 1
            if s_new < score(a):
                a = a_new.copy()

        it += 1
        if stagnant > 30:
            break
        if verbose and (it % 20 == 0 or it < 5):
            print(f"    LP iter {it:4d}: score={best_score:.6f}, t={t_used:.3f}")

    return best_a, best_score


def smooth_mw_optimize(a, time_limit_s=300.0, verbose=False):
    """Smooth multiplicative weight optimization using log-sum-exp."""
    global _best_score_global, _best_seq_global
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score(a)

    N = 1
    while N < 3 * n - 2:
        N *= 2

    taus = [5.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    stagnant = 0

    it = 0
    while time.monotonic() - start_time < time_limit_s:
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        tau = taus[it % len(taus)]
        eta = 0.5 / (1.0 + 0.05 * it)

        conv = fft_convolve(a, a)
        conv_max = conv.max()
        # Log-sum-exp weights (numerically stable)
        exp_vals = np.exp(tau * (conv - conv_max))
        p = exp_vals / exp_vals.sum()

        # Compute smooth gradient via cross-correlation
        fp = np.fft.rfft(p, N)
        fa = np.fft.rfft(a, N)
        cross_corr = np.fft.irfft(fp * np.conj(fa), N)[:n]
        # Diagonal correction: a[j] * p[2j]
        diag = a * p[0::2]
        grad = cross_corr + diag

        # Try multiple step sizes
        best_s = best_score
        best_a_new = a
        for eta_try in [eta * 0.5, eta, eta * 2.0]:
            a_try = a * np.exp(-eta_try * np.maximum(0, grad))
            a_try = normalize(a_try)
            s_try = score(a_try)
            if s_try < best_s:
                best_s = s_try
                best_a_new = a_try

        if best_s < best_score - 1e-14:
            best_score = best_s
            best_a = best_a_new.copy()
            a = best_a_new.copy()
            stagnant = 0
            if best_score < _best_score_global:
                _best_score_global = best_score
                _best_seq_global = best_a.copy()
                if verbose and int(time.monotonic() - start_time) % 30 == 0:
                    print(f"    MW global best: {best_score:.6f} (n={n}, iter={it})")
        else:
            stagnant += 1
            # Accept if slightly better than current to avoid stagnation
            s_curr = score(a)
            if best_s < s_curr:
                a = best_a_new.copy()

        it += 1
        if stagnant > 100:
            # Restart with perturbation
            a = normalize(a + 0.01 * np.abs(np.random.default_rng(it).standard_normal(n)))
            stagnant = 0

    return best_a, best_score


def gs_optimize(a, time_limit_s=120.0, verbose=False):
    """Gerchberg-Saxton style frequency domain iterations."""
    global _best_score_global, _best_seq_global
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    N = 1
    while N < 2 * n:
        N *= 2

    it = 0
    stagnant = 0
    while time.monotonic() - start_time < time_limit_s:
        if time.monotonic() - start_time > time_limit_s:
            break
        # Forward FFT
        F = np.fft.rfft(a, N)
        # Target magnitude: flat (geometric mean)
        mag = np.abs(F)
        target_mag = np.sqrt(np.mean(mag ** 2))
        new_mag = 0.7 * mag + 0.3 * target_mag  # Blend to avoid oscillation
        # Inverse FFT with new magnitude, keep phase
        F_new = new_mag * np.exp(1j * np.angle(F))
        a_new = np.fft.irfft(F_new, N)[:n]
        # Project: non-negative, normalize
        a_new = np.maximum(0, a_new)
        a_new = normalize(a_new)

        s = score(a_new)
        if s < best_score - 1e-14:
            best_score = s
            best_a = a_new.copy()
            a = a_new.copy()
            stagnant = 0
            if best_score < _best_score_global:
                _best_score_global = best_score
                _best_seq_global = best_a.copy()
                if verbose:
                    print(f"    GS global best: {best_score:.6f} (n={n})")
        else:
            stagnant += 1
            a = a_new  # Still update to explore

        it += 1
        if stagnant > 50:
            break

    return best_a, best_score


def local_search(a, time_limit_s=120.0, rng=None, verbose=False):
    """Random local search with multiplicative perturbations."""
    global _best_score_global, _best_seq_global
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score(a)
    accept_count = 0

    while time.monotonic() - start_time < time_limit_s:
        # Random window perturbation
        width = max(2, int(rng.integers(max(2, n // 300), max(3, n // 5))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)

        candidate = a.copy()
        noise_scale = 0.05 + 0.1 * rng.random()
        candidate[start:stop] *= np.exp(rng.normal(0.0, noise_scale, size=stop - start))

        # Occasionally try anti-symmetric perturbation
        if rng.random() < 0.3:
            flip = np.linspace(0.7, 1.3, n)
            candidate *= flip
        elif rng.random() < 0.2:
            # Random coordinate reset
            idx = rng.integers(0, n)
            candidate[idx] = rng.random() * a.max() * 0.5

        candidate = normalize(candidate)
        s = score(candidate)
        if s < best_score:
            best_score = s
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if best_score < _best_score_global:
                _best_score_global = best_score
                _best_seq_global = best_a.copy()
                if verbose and accept_count % 30 == 0:
                    print(f"    LS global best: {best_score:.6f} (n={n}, accepts={accept_count})")

    return best_a, best_score


def propagate_to_larger(seed_seq, target_n):
    """Propagate a good seed to a larger size using zero-stuffing + smoothing."""
    n = len(seed_seq)
    ratio = target_n // n
    if ratio < 1:
        return initial_raised_cosine(target_n)
    # Upsample by factor ratio with zero insertion
    extended = np.zeros(target_n)
    extended[::ratio] = seed_seq
    # Smooth with a small kernel
    kernel_size = max(3, ratio)
    if kernel_size < target_n:
        kernel = np.ones(kernel_size) / kernel_size
        extended = np.convolve(extended, kernel, mode='same')
    return normalize(extended)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Hybrid approach: LP refinement + smooth multiplicative weights + GS + local search.
    """
    global _best_score_global, _best_seq_global
    _best_score_global = float('inf')
    _best_seq_global = None

    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 20.0)

    print(f"Starting search with budget {budget_s}s, deadline at {deadline - total_start:.0f}s")

    # Try to use known sequence
    known_seq = None
    try:
        known_seq = np.array(height_sequence_1, dtype=np.float64)
        known_seq = normalize(known_seq)
        known_score = score(known_seq)
        print(f"Known sequence: n={len(known_seq)}, score={known_score:.6f}")
        if known_score < _best_score_global:
            _best_score_global = known_score
            _best_seq_global = known_seq.copy()
    except (NameError, UnboundLocalError):
        pass

    # Define search configurations
    # (n, init_type, methods, time_frac)
    configs = []

    # Use known sequence at its native size
    if known_seq is not None:
        n_known = len(known_seq)
        configs.append((n_known, 'known', ['mw', 'lp', 'ls'], 0.15))
        # Larger sizes from known
        for n_large in [n_known * 2, n_known * 4]:
            if n_large <= 5000:
                configs.append((n_large, 'known_interp', ['mw', 'lp'], 0.12))

    # Various sizes and inits
    size_inits = [
        (300, 'raised_cosine'),
        (500, 'raised_cosine'),
        (500, 'asymmetric'),
        (500, 'two_bump'),
        (1000, 'raised_cosine'),
        (1000, 'asymmetric'),
        (1000, 'gaussian'),
        (1500, 'raised_cosine'),
        (1500, 'asymmetric'),
        (2000, 'raised_cosine'),
        (2000, 'asymmetric'),
        (2000, 'blackman'),
        (3000, 'raised_cosine'),
        (3000, 'asymmetric'),
    ]

    for n, init_type in size_inits:
        if known_seq is not None and n == len(known_seq) and init_type == 'raised_cosine':
            continue  # Already have 'known' at this size
        methods = ['mw', 'lp', 'ls'] if n <= 2000 else ['mw', 'ls']
        if n > 2000:
            methods = ['mw', 'ls']
        configs.append((n, init_type, methods, 0.0))

    # Distribute time budget
    total_weight = sum(1.0 for _ in configs)
    for i, (n, init_type, methods, tf) in enumerate(configs):
        if tf == 0.0:
            configs[i] = (n, init_type, methods, 0.8 * budget_s / len(configs))

    # Time allocation
    time_allocation = {}
    remaining_budget = budget_s - 40.0  # Reserve 40s for final polish
    per_config = remaining_budget / max(1, len(configs))
    for i, (n, init_type, methods, tf) in enumerate(configs):
        t_alloc = min(per_config, remaining_budget * 0.3) if tf == 0.0 else tf * budget_s
        time_allocation[i] = max(20.0, t_alloc)

    print(f"Number of configurations: {len(configs)}")
    print(f"Time per config: ~{per_config:.0f}s")

    for i, (n, init_type, methods, tf) in enumerate(configs):
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break
        t_alloc = min(time_allocation.get(i, per_config), remaining * 0.9)
        if t_alloc < 15.0:
            continue

        print(f"\n=== [{i+1}/{len(configs)}] n={n}, init={init_type}, alloc={t_alloc:.0f}s ===")

        # Initialize
        if init_type == 'known':
            a = known_seq.copy()
        elif init_type == 'known_interp':
            a = initial_known(known_seq, n)
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=0.25)
        elif init_type == 'blackman':
            a = initial_blackman(n)
        elif init_type == 'asymmetric':
            a = initial_asymmetric(n, rng)
        elif init_type == 'two_bump':
            a = initial_two_bump(n, rng)
        else:
            a = initial_raised_cosine(n)

        a = normalize(a)
        init_score = score(a)
        print(f"    Init: score={init_score:.6f}")

        if init_score < _best_score_global:
            _best_score_global = init_score
            _best_seq_global = a.copy()

        # Split time among methods
        n_methods = len(methods)
        method_time = t_alloc / n_methods

        for mi, method in enumerate(methods):
            remaining = deadline - time.monotonic()
            if remaining < 10.0:
                break
            mt = min(method_time, remaining * 0.9)
            if mt < 5.0:
                continue
            verbose = mt > 60

            if method == 'mw':
                a, s = smooth_mw_optimize(a, time_limit_s=mt, verbose=verbose)
                print(f"    After MW: score={s:.6f}")
            elif method == 'lp':
                if n <= 3000:  # LP is too slow for very large n
                    a, s = lp_refine(a, time_limit_s=mt, verbose=verbose)
                    print(f"    After LP: score={s:.6f}")
            elif method == 'gs':
                a, s = gs_optimize(a, time_limit_s=mt, verbose=verbose)
                print(f"    After GS: score={s:.6f}")
            elif method == 'ls':
                a, s = local_search(a, time_limit_s=mt, rng=rng, verbose=verbose)
                print(f"    After LS: score={s:.6f}")

            if s < _best_score_global:
                _best_score_global = s
                _best_seq_global = a.copy()

    # Final polishing phase
    if _best_seq_global is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")
            best_a = _best_seq_global.copy()
            best_n = len(best_a)

            # Local search polish
            ls_time = min(remaining * 0.4, 200.0)
            if ls_time > 10:
                best_a, s = local_search(best_a, time_limit_s=ls_time, rng=rng, verbose=True)
                print(f"  After final LS: {s:.6f}")

            # MW polish
            remaining2 = deadline - time.monotonic()
            mw_time = min(remaining2 * 0.4, 200.0)
            if mw_time > 10:
                best_a, s = smooth_mw_optimize(best_a, time_limit_s=mw_time, verbose=True)
                print(f"  After final MW: {s:.6f}")

            # LP polish (if size allows)
            remaining3 = deadline - time.monotonic()
            if best_n <= 2000 and remaining3 > 20:
                lp_time = min(remaining3 * 0.5, 200.0)
                if lp_time > 15:
                    best_a, s = lp_refine(best_a, time_limit_s=lp_time, verbose=True)
                    print(f"  After final LP: {s:.6f}")

            _best_seq_global = best_a.copy()
            _best_score_global = score(best_a)

    # Fallback
    if _best_seq_global is None:
        _best_seq_global = initial_raised_cosine(500)
        _best_score_global = score(_best_seq_global)
        print(f"  Fallback: score={_best_score_global:.6f}")

    best_seq = _best_seq_global
    print(f"\nFinal result: score={_best_score_global:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified score via evaluate_sequence: {verify_score:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_seq]
