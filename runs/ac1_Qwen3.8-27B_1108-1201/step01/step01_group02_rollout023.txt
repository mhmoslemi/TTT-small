import numpy as np
from scipy.optimize import minimize
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def smooth_score(a, T=1.0):
    n = len(a)
    conv = np.convolve(a, a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return 1e10
    # Log-sum-exp for numerical stability
    x = conv / (sum_a**2)
    x_max = x.max()
    return (1.0/T) * np.log(np.sum(np.exp(T * (x - x_max)))) + x_max

def smooth_score_grad(a, T=1.0):
    n = len(a)
    conv = np.convolve(a, a)
    sum_a = a.sum()
    x = conv / (sum_a**2)
    x_max = x.max()
    # Softmax weights
    weights = np.exp(T * (x - x_max))
    weights /= weights.sum()
    # d(c[k]/S^2)/da[i] = (a[k-i]*S^2 - c[k]*2S) / S^4 = (a[k-i]/S - 2*c[k]/S^2) / S
    # = (a[k-i] - 2*c[k]/S) / S^2
    # For each k, the contribution to gradient[i] is weights[k] * (a[k-i] - 2*conv[k]/S) / S^2 (if valid)
    # conv[k] = sum_i a[i]*a[k-i], so d(conv[k])/da[i] = a[k-i] (if 0<=k-i<n)
    # d(conv[k]/S^2)/da[i] = (a[k-i]*S^2 - conv[k]*2S)/S^4 = (a[k-i]*S - 2*conv[k])/S^3
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-15:
            continue
        # d(conv[k]/S^2)/da[i] = (a[k-i]*S - 2*conv[k])/S^3 for valid i
        # valid i: max(0, k-n+1) <= i <= min(k, n-1)
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min > i_max:
            continue
        # a[k-i] for i in [i_min, i_max]
        # k-i ranges from k-i_max to k-i_min
        a_vals = a[k-i_max : k-i_min+1][::-1]  # a[k-i] for i = i_min to i_max
        # Actually: for i = i_min, k-i = k-i_min; for i = i_max, k-i = k-i_max
        # So a[k-i] for i in [i_min..i_max] is a[k-i_min], a[k-i_min-1], ..., a[k-i_max]
        # = a[k-i_min : k-i_max-1 : -1]... let me just do it directly
        idx = np.arange(i_min, i_max+1)
        a_vals = a[k - idx]  # this gives a[k-i_min], a[k-(i_min+1)], ..., a[k-i_max]
        # Wait, k-idx = k - [i_min, i_min+1, ..., i_max] = [k-i_min, k-i_min-1, ..., k-i_max]
        # These should all be in [0, n-1] by construction
        contrib = (a_vals * sum_a - 2 * conv[k]) / (sum_a**3)
        grad[i_min:i_max+1] += w * contrib
    return 2.0 * n * grad

Let me also think about the gradient computation more efficiently. For large n, computing the full gradient naively is O(n^2). But we can use the fact that:

grad[i] = sum_k w_k * (a[k-i]*S - 2*c[k])/S^3 (for valid k-i)

= (1/S^3) * sum_k w_k * a[k-i]*S - (2/S^3) * sum_k w_k * c[k]

The second term is just a constant (doesn't depend on i): (2/S^3) * sum_k w_k * c[k] = (2/S^3) * E_w[c]

The first term: sum_k w_k * a[k-i] = (w * a_reverse)[something]... this is a convolution!

Let me define: for fixed i, sum_k w_k * a[k-i] = sum_{j=0}^{n-1} w_{i+j} * a[j] (where j = k-i, so k = i+j, valid when 0 <= i+j < 2n-1)

= (w_conv_a)[i] where w_conv_a is some cross-correlation.

Actually, let's define: for i in [0, n-1], g[i] = sum_{j=0}^{n-1} w_{i+j} * a[j]

This is the cross-correlation of w (padded) and a. We can compute this with FFT!

So: grad[i] = (1/S^3) * (S * g[i] - 2 * E_w[c])
= (S/S^3) * g[i] - (2/S^3) * E_w[c]
= g[i]/S^2 - 2*E_w[c]/S^3

"""
Gradient-based optimization with log-sum-exp smoothing to minimize peak autocorrelation.
Uses FFT for efficient convolution, L-BFGS-B for optimization, and temperature annealing.
Explores asymmetric sequences which can achieve score < 2.
"""

import numpy as np
from scipy.optimize import minimize
import time

def _pad_to_power_of_2(m):
    n = 1
    while n < m:
        n *= 2
    return n

def _convolve_fft(a):
    n = len(a)
    N = _pad_to_power_of_2(2*n - 1)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A)[:2*n-1]

def _smooth_objective_and_grad(a, T):
    """Compute smooth objective and gradient for L-BFGS."""
    n = len(a)
    N = _pad_to_power_of_2(2*n - 1)
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A)[:2*n-1]
    
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    
    x = conv / (S * S)
    x_max = x.max()
    shifted = T * (x - x_max)
    # Log-sum-exp
    max_shifted = shifted.max()
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    obj = (log_sum / T) + x_max
    
    # Gradient
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    
    # Cross-correlation: g[k] = sum_{j=0}^{n-1} w[k+j] * a[j]
    # Use FFT
    N_x = _pad_to_power_of_2(2*n)
    W_full = np.fft.fft(w, N_x)
    A_full = np.fft.fft(a, N_x)
    R = np.fft.ifft(W_full * np.conj(A_full)).real[:n]
    
    E_wc = np.dot(w, conv)
    grad = (R * S - 2.0 * E_wc) / (S * S * S)
    grad = 2.0 * n * grad
    
    return 2.0 * n * obj, grad

def _actual_score(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:

    return 2.0 * n * max_b / (sum_a * sum_a)

"""
Gradient-based optimization with log-sum-exp smoothing to minimize peak autocorrelation
of non-negative sequences. Uses FFT for efficient convolution/gradient, L-BFGS-B with
temperature annealing, and diverse asymmetric initializations. Exploits the key insight
that asymmetric sequences can achieve score < 2, approaching the theoretical bound of ~1.
"""

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _convolve_fft(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A)[:2 * n - 1]


def _smooth_obj_grad(a, T):
    """Compute smooth log-sum-exp objective and its gradient."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A)[:2 * n - 1]

    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)

    x = conv / (S * S)
    x_max = x.max()
    shifted = T * (x - x_max)
    smax = shifted.max()
    log_sum = smax + np.log(np.sum(np.exp(shifted - smax)))
    obj = (log_sum / T) + x_max
    obj = 2.0 * n * obj

    # Softmax weights
    w = np.exp(shifted - shifted.max())
    w /= w.sum()

    # Cross-correlation g[k] = sum_{j=0}^{n-1} w[k+j] * a[j]
    N_x = _next_pow2(2 * n)
    Wf = np.fft.fft(w, N_x)
    Af = np.fft.fft(a, N_x)
    R = np.fft.ifft(Wf * np.conj(Af)).real[:n]

    E_wc = np.dot(w, conv)
    grad = (R * S - 2.0 * E_wc) / (S * S * S)
    grad = 2.0 * n * grad

    # Add small regularization to break scale invariance
    reg_coeff = 1e-6
    obj += reg_coeff * (S - n) ** 2 / n
    grad += reg_coeff * 2.0 * (S - n) / n * np.ones(n)

    return obj, grad


def _actual_score(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


def _optimize_anneal(a0, time_budget, verbose=True):
    """Run L-BFGS-B with temperature annealing."""
    start_time = time.monotonic()
    a = np.maximum(0, a0.copy().astype(np.float64))
    n = len(a)
    best_score = _actual_score(a)
    best_a = a.copy()

    T_schedule = [20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.06, 0.04]

    for T in T_schedule:
        elapsed = time.monotonic() - start_time
        if elapsed > time_budget - 3:
            break
        remaining = time_budget - elapsed
        if remaining < 2:
            break

        def fun(g):
            return _smooth_obj_grad(g, T)

        max_iter = min(300, max(30, int(remaining * 5)))
        try:
            res = minimize(fun, a, jac=True, method='L-BFGS-B',
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                    'max_time': max(1.0, remaining * 0.85),
                                    'gtol': 1e-6})
            if res.x is not None:
                a_new = np.maximum(0, res.x)
                score = _actual_score(a_new)
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new.copy()
                    if verbose:
                        print(f"        T={T:.2f}: score={score:.6f} (iters={res.nit})")
                else:
                    # Try a damped step
                    a_try = 0.7 * a + 0.3 * np.maximum(0, res.x)
                    score_try = _actual_score(a_try)
                    if score_try < best_score - 1e-12:
                        best_score = score_try
                        best_a = a_try.copy()
                        a = a_try.copy()
        except Exception:
            pass

    return best_a, best_score

def _gen_sin_mod(n, rng):
    t = np.arange(n, dtype=np.float64)
    freq = rng.uniform(0.5, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * t / n + phase) + 1.0 * np.cos(np.pi * t / n)
    return np.maximum(0.01, a)


def _gen_random_smooth(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_comp = rng.integers(2, 5)
    for _ in range(n_comp):
        center = rng.uniform(0, n)
        width = rng.uniform(0.05, 0.25) * n
        amp = rng.uniform(-1.5, 3.0)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    return np.maximum(0.01, a)


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses gradient-based optimization (L-BFGS-B) with log-sum-exp smoothing and
    temperature annealing, exploring diverse asymmetric initializations across
    multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20  # reserve 20s for safety

    best_score = float('inf')
    best_seq = None

    def try_sequence(a, label=""):
        nonlocal best_score, best_seq
        a = np.maximum(0, a.astype(np.float64))
        if a.sum() < 0.01:
            return
        score = _actual_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - total_start
            print(f"  *** NEW BEST: score={score:.6f}, n={len(a)}, {label}, t={elapsed:.0f}s")

    # Try the parent sequence if available
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if len(parent) > 0:
            try_sequence(parent, "parent")
    except NameError:
        parent = None

    # Define configurations: (length, list_of_generators)
    lengths = [400, 500, 700, 1000, 1500, 2000]
    generators = [
        _gen_barbell, _gen_ramp, _gen_exp_decay,
        _gen_bump_left, _gen_bump_right, _gen_poly,
        _gen_two_bump, _gen_sin_mod, _gen_random_smooth,
    ]

    # Build schedule: for each length, try a few generators
    # Allocate time proportionally to number of (length, gen) pairs
    n_long = len(lengths)
    n_gen = min(4, len(generators))  # use 4 generators per length
    total_pairs = n_long * n_gen

    # Time allocation: 70% to main optimization, 20% to polishing, 10% buffer
    main_time = (budget_s - 20) * 0.65
    per_pair = main_time / total_pairs
    per_pair = max(20, min(per_pair, 120))  # clamp

    print(f"Config: {n_long} lengths x {n_gen} gens = {total_pairs} pairs, ~{per_pair:.0f}s each")

    pair_count = 0
    for n in lengths:
        for g_idx in range(n_gen):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            pair_count += 1

            # Generate initial sequence
            gen_fn = generators[g_idx]
            try:
                a0 = gen_fn(n, rng)
            except TypeError:
                a0 = gen_fn(n)
            a0 = np.maximum(0, a0).astype(np.float64)
            if a0.sum() < 0.01:
                continue
            a0 = a0 / a0.sum() * n  # normalize to sum = n

            init_score = _actual_score(a0)
            t_budget = min(per_pair, remaining - 10)
            if t_budget < 10:
                continue

            label = f"n={n}, gen={g_idx}, init={init_score:.4f}"
            print(f"  [{pair_count}/{total_pairs}] {label}, budget={t_budget:.0f}s")

            a_opt, score_opt = _optimize_anneal(a0, time_budget=t_budget, verbose=False)
            try_sequence(a_opt, label)

    # If we have the parent, also try optimizing it directly
    if parent is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_budget = min(200, remaining - 10)
            print(f"  Optimizing parent (n={len(parent)}), budget={t_budget:.0f}s")
            a_opt, score_opt = _optimize_anneal(parent, time_budget=t_budget, verbose=True)
            try_sequence(a_opt, "parent_opt")

    # Polish the best found so far
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            polish_time = min(remaining * 0.5, 300)
            print(f"\n  Polishing best (score={best_score:.6f}), time={polish_time:.0f}s")
            a_pol, score_pol = _polish_local(best_seq, polish_time, rng, verbose=True)
            try_sequence(a_pol, "polished")

            # Final gradient polish on the polished result
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                final_time = min(remaining2 * 0.6, 200)
                print(f"  Final gradient polish, time={final_time:.0f}s")
                T_fine = [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02]

                start_time = time.monotonic()
                a = best_seq.copy().astype(np.float64)
                for T in T_fine:
                    elapsed = time.monotonic() - start_time
                    if elapsed > final_time - 2:
                        break
                    remaining_this = final_time - elapsed
                    if remaining_this < 2:
                        break

                    def fun(g):
                        return _smooth_obj_grad(g, T)

                    try:
                        res = minimize(fun, a, jac=True, method='L-BFGS-B',
                                       bounds=[(0.0, None)] * len(a),
                                       options={'maxiter': 200, 'maxfun': 400,
                                                'max_time': max(1.0, remaining_this * 0.8)})
                        if res.x is not None:
                            a_new = np.maximum(0, res.x)
                            score_new = _actual_score(a_new)
                            if score_new < best_score - 1e-12:
                                best_score = score_new
                                best_seq = a_new.copy()
                                a = a_new.copy()
                                print(f"    Final T={T:.2f}: score={score_new:.6f}")
                            else:
                                a = np.maximum(0, a)
                    except Exception:
                        pass

    # Safety fallback
    if best_seq is None:
        best_seq = np.ones(500, dtype=np.float64)
        best_score = _actual_score(best_seq)
        print(f"  Fallback: uniform n=500, score={best_score:.6f}")

    # Verify
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"\n  Final: score={best_score:.6f}, verified={verify_score:.6f}, n={len(best_seq)}")
    print(f"  Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_seq]
