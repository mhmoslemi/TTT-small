    return a * S / s


def build_lp_matrix(a):
    """Build the convolution constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve(a, M):
    """
    Solve LP: maximize sum(g) subject to conv(a,g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    # Fallback: return zeros (degenerate case)
    return np.zeros(n)


def lp_refine(a, time_limit, verbose=True):
    """
    LP-based iterative refinement. Repeatedly solves the LP and blends.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))
    
    for iteration in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        
        g = lp_solve(a, M)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        c = normalize_seq(g)
        
        # Find best blend parameter t
        best_t = 0.0
        best_score_t = M
        for t_try in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            b = (1 - t_try) * a + t_try * c
            b_conv_max = np.max(fft_convolve(b, b))
            if b_conv_max <= M + 1e-10:
                best_t = t_try
                best_score_t = b_conv_max
                break
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * c
        a = np.clip(a, 0.0, None)
        a = normalize_seq(a, S)
        
        new_conv_max = np.max(fft_convolve(a, a))
        if new_conv_max < best_conv_max - 1e-12:
            best_conv_max = new_conv_max
            best_a = a.copy()
        
        if verbose and iteration % 100 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    LP iter {iteration}: conv_max={best_conv_max:.6f}, score={score:.6f}, t={best_t:.3f}")
        
        # Convergence check
        if best_t < 0.005:
            break
    
    return best_a


def gradient_step(a, lr):
    """
    One gradient step to reduce the peak of conv(a,a).
    Uses the subgradient at the peak lag.
    """
    n = len(a)
    conv = fft_convolve(a, a)
    peak_idx = int(np.argmax(conv))
    
    # Subgradient of conv(a,a)[peak_idx] w.r.t. a
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        i1 = peak_idx - j
        if 0 <= i1 < n:
            val += a[i1]
        i2 = j - peak_idx
        if 0 <= i2 < n:
            val += a[i2]
        grad[j] = val
    
    # Gradient step with projection
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    
    # Renormalize
    S = np.sqrt(2.0 * n)
    a_new = normalize_seq(a_new, S)
    
    return a_new, peak_idx


def gradient_optimize(a, time_limit, verbose=True):
    """
    Gradient-based optimization with adaptive step size.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = np.max(fft_convolve(a, a))
    
    lr = 0.5 / n
    step = 0
    stall_count = 0
    peak_lags = []
    
    while time.monotonic() - t0 < time_limit:
        if step % 100 == 0:
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            peak_lags.append(peak_idx)
            # Adaptive step size based on recent peak behavior
            if len(peak_lags) > 20:
                recent = peak_lags[-20:]
                if len(set(recent)) <= 3:
                    lr *= 0.9  # Stepping, reduce step size
                elif len(set(recent)) >= 10:
                    lr = min(lr * 1.1, 2.0 / n)  # Exploring, increase slightly
        
        a_new, peak_idx = gradient_step(a, lr)
        
        new_conv_max = np.max(fft_convolve(a_new, a_new))
        if new_conv_max < best_score:
            best_score = new_conv_max
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        elif new_conv_max < np.max(fft_convolve(a, a)):
            a = a_new
            stall_count += 1
        else:
            stall_count += 1
            lr *= 0.95
        
        if stall_count > 200:
            lr *= 0.5
            stall_count = 0
        
        step += 1
        
        if verbose and step % 500 == 0:
            score = 2.0 * n * best_score / (S * S)
            print(f"    Grad step {step}: score={score:.6f}, lr={lr:.4f}")
    
    return best_a


def targeted_search(a, time_limit, verbose=True):
    """
    Targeted local search: specifically tries to reduce the current peak.
    Uses coordinate and block updates.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]
        
        # Strategy 1: Reduce elements that contribute most to the peak
        # conv(a,a)[peak_idx] = sum_i a[i] * a[peak_idx - i] (with bounds)
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        # Reduce top contributors
        top_k = max(3, n // 12)
        top_idx = np.argsort(contribs)[-top_k:]
        
        for _ in range(5):
            candidate = a.copy()
            # Randomly reduce some top contributors and increase others
            reduce_mask = rng.random(len(top_idx)) < 0.6
            for idx in top_idx[reduce_mask]:
                candidate[idx] *= np.exp(rng.normal(-0.05, 0.02))
            
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = max(3, n // 20)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
            
            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize_seq(candidate, S)
            
            new_conv = fft_convolve(candidate, candidate)
            new_peak = np.max(new_conv)
            
            if new_peak < best_score:
                best_score = new_peak
                best_a = candidate.copy()
                a = candidate
            elif new_peak < current_peak * 1.05 and rng.random() < 0.3:
                a = candidate
        
        # Strategy 2: Block perturbation
        if rng.random() < 0.5:
            width = max(1, n // 30)
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize_seq(candidate, S)
            new_peak = np.max(fft_convolve(candidate, candidate))
            if new_peak < best_score:
                best_score = new_peak
                best_a = candidate.copy()
                a = candidate
        
        if verbose:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 10 == 0:
                score = 2.0 * n * best_score / (S * S)
                print(f"    Targeted t={elapsed:.0f}s: score={score:.6f}")
    
    return best_a


def make_initialization(n, init_type, rng):
    """Create various types of initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 4))
        amp = 0.2 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for _ in range(3):
            f = int(rng.integers(1, n // 4))
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif init_type == 'bimodal':
        center = n // 2
        sigma = n * (0.15 + 0.1 * rng.random())
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    else:
        a = np.ones(n)
    
    a = np.clip(a, 0.01, None)
    return a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. LP-based refinement from provided sequence and structured inits
    2. Gradient-based optimization
    3. Targeted local search polish
    Returns the best sequence found.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 20.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False
    
    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)
    
    # Phase 1: LP-based refinement
    print("\n=== Phase 1: LP-based refinement ===")
    
    # Try from provided sequence at its native length
    if provided_seq is not None:
        n = len(provided_seq)
        t_limit = min(120.0, (deadline - time.monotonic()) * 0.3)
        print(f"  LP refine provided seq (n={n}), time_limit={t_limit:.0f}s")
        result = lp_refine(provided_seq, t_limit, verbose=True)
        update_best(result)
        print(f"  After LP: score={best_score:.6f}")
    
    # Try other lengths from the provided sequence (interpolated)
    if provided_seq is not None:
        for n_try in [300, 384, 448, 512, 640, 768]:
            if time.monotonic() > deadline - 300:
                break
            t_per = 30.0
            a0 = interpolate_sequence(provided_seq, n_try)
            a0 = np.clip(a0, 0.01, None)
            result = lp_refine(a0, t_per, verbose=False)
            if update_best(result):
                print(f"  *** n={n_try}: new best score={best_score:.6f} ***")
    
    # Try structured initializations with LP refinement
    init_types = ['chirp', 'sine', 'multi_tone', 'random', 'bimodal']
    for n in [256, 320, 384, 448, 512]:
        if time.monotonic() > deadline - 200:
            break
        for init_type in init_types[:2]:  # Limit to save time
            if time.monotonic() > deadline - 180:
                break
            a0 = make_initialization(n, init_type, rng)
            t_per = min(20.0, (deadline - time.monotonic()) * 0.1)
            result = lp_refine(a0, t_per, verbose=False)
            if update_best(result):
                print(f"  *** n={n}, init={init_type}: new best score={best_score:.6f} ***")
    
    print(f"  Phase 1 best: {best_score:.6f}")
    
    # Phase 2: Gradient-based optimization on best candidate
    print("\n=== Phase 2: Gradient optimization ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(180.0, (deadline - time.monotonic()) * 0.35)
        print(f"  Gradient optimize (n={n}), time_limit={t_limit:.0f}s")
        result = gradient_optimize(best_seq, t_limit, verbose=True)
        update_best(result)
        print(f"  After gradient: score={best_score:.6f}")
    
    # Phase 3: Targeted local search
    print("\n=== Phase 3: Targeted local search ===")
    if best_seq is not None:
        t_limit = min(180.0, (deadline - time.monotonic()) * 0.4)
        print(f"  Targeted search, time_limit={t_limit:.0f}s")
        result = targeted_search(best_seq, t_limit, verbose=True)
        update_best(result)
        print(f"  After targeted: score={best_score:.6f}")
    
    # Phase 4: Another round of LP on the best
    print("\n=== Phase 4: LP refinement on best ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            t_limit = min(remaining, 120.0)
            print(f"  LP refine best, time_limit={t_limit:.0f}s")
            result = lp_refine(best_seq, t_limit, verbose=True)
            update_best(result)
            print(f"  After final LP: score={best_score:.6f}")
    
    # Phase 5: Final polish with targeted search
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            result = targeted_search(best_seq, remaining, verbose=False)
            update_best(result)
            print(f"\n=== Final polish: score={best_score:.6f} ===")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local_score={best_score:.6f} ===")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    
    return [1.0] * 256
