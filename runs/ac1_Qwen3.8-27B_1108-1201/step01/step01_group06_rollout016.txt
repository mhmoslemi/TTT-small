"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
  * Normalize to sum = sqrt(2*n) so objective = max of full self-convolution.
  * Use a smooth power-sum surrogate: (sum c_k^p)^(1/p) with increasing p.
  * Adam optimizer with exponential-weighted subgradient (soft-max of lags).
  * LP-based Frank-Wolfe step (CVXPY) for periodic big moves.
  * Multiple diverse initializations including asymmetric shapes.
  * Polish the global best with a fine-grained schedule.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import sys

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_smooth_subgradient(a, c, p, temperature):
    """Compute subgradient of (sum c^p)^(1/p) w.r.t. a."""
    n = len(a)
    # Weights proportional to c^p (normalized)
    cp = np.maximum(c, 0) ** p
    total = cp.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = cp / total
    grad = np.zeros(n)
    # For each lag k, dc[k]/da[m] = 2*a[k-m] (if valid)
    # Use the top lags for efficiency
    top_k = min(64, len(c))
    top_idx = np.argsort(c)[::-1][:top_k]
    for k in top_idx:
        w = weights[k]
        if w < 1e-15:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += w * 2.0 * a[k - hi : k - lo + 1][::-1]
    return grad

def compute_exp_subgradient(a, c, beta):
    """Exponentially-weighted subgradient: weights ~ exp(beta * c[k])."""
    n = len(a)
    # Shift c so max is at 0
    c_shifted = c - c.max()
    weights = np.exp(beta * c_shifted)
    total = weights.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights /= total
    grad = np.zeros(n)
    top_k = min(128, len(c))
    top_idx = np.argsort(c)[::-1][:top_k]
    for k in top_idx:
        w = weights[k]
        if w < 1e-15:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += w * 2.0 * a[k - hi : k - lo + 1][::-1]
    return grad

def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    return v * (S / s)

def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def run_smooth_adam(a_init, n, max_time, rng, p_init=4, p_growth=1.05, lr_init=None, beta=0.5):
    """Adam with smooth power-sum objective and exponential subgradient."""
    a = safe_normalize(a_init, n)
    S = np.sqrt(2.0 * n)
    if lr_init is None:
        lr_init = S / n * 0.3
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    p = p_init
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    patience = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > 5000:
                break
        
        # Compute subgradient
        grad = compute_exp_subgradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        
        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        lr = lr_init * (0.9995 ** t)
        a = a - lr * m / (np.sqrt(v) + 1e-8)
        a = project_nonneg_sum(a, S)
        
        # Increase p periodically
        if t % 2000 == 0:
            p = min(64.0, p * p_growth)
            beta = min(2.0, beta + 0.05)
    
    return best_local, best_score

def run_lp_frank_wolfe(a_init, n, max_time, rng, t_decay=0.99):

    try:
        import cvxpy as cp
        from cvxpy import Solver
        available_solver = None
        for s in ['ECOS', 'SCS', 'CLARABEL', 'MOSEK', 'GUROBI', 'GLPK', 'CBC']:
            try:

                test.solve(solver=s)

                break

                continue
        if available_solver is None:

    except ImportError:
        print("CVXPY not available, skipping LP-FW")
        return a_init.copy(), fft_convolve_self(a_init).max()
    
    a = safe_normalize(a_init, n)
    S = np.sqrt(2.0 * n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    t_step = 0.1
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        M = best_score  # Use best score as the bound
        
        # Build the LP
        g = cp.Variable(n, nonneg=True)
        # Build constraints: (a * g)[k] <= M for all k
        # (a * g)[k] = sum_j a[j] * g[k-j] for valid j, k-j
        # We need to build this efficiently
        # For each k, the constraint is: sum_{j} a[j] * g[k-j] <= M
        # where valid means 0 <= j < n and 0 <= k-j < n
        # i.e., max(0, k-n+1) <= j <= min(n-1, k)
        # Let i = k-j, so j = k-i, and the constraint is:

        # Build constraint matrix implicitly using cvxpy
        # This is O(n * 2n) = O(n^2) which is too large for n=8192
        # Instead, use a sampled subset of constraints
        
        # Sample constraints: use every m-th lag
        m_sample = max(1, (2*n-1) // 4000)  # Use at most 4000 constraints
        k_indices = np.arange(0, 2*n-1, m_sample)
        
        constraints = []
        for k in k_indices:
            # (a*g)[k] = sum_j a[j]*g[k-j] for valid j
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                # g indices: k-hi to k-lo
                g_lo = k - hi
                g_hi = k - lo
                # a indices: lo to hi
                # constraint: sum_{j=lo}^{hi} a[j] * g[k-j] <= M
                # g[k-j] for j from lo to hi: g[k-hi] to g[k-lo]
                a_slice = a[lo:hi+1]
                g_slice = g[g_lo:g_hi+1]
                # a[j]*g[k-j] = a[lo+j']*g[hi... ] 
                # j = lo + j', g index = k - j = k - lo - j'
                # So: sum_{j'=0}^{hi-lo} a[lo+j'] * g[k-lo-j']
                # This is a convolution of a_slice with flipped g_slice
                # = dot(a_slice, g_slice_flipped) but reversed
                # a[lo+j'] * g[k-lo-j']
                # Let's just build it as a sum
                coeffs = a_slice  # coefficients for each term
                # The terms are: a[lo]*g[k-lo] + a[lo+1]*g[k-lo-1] + ... + a[hi]*g[k-hi]
                # g indices: k-lo, k-lo-1, ..., k-hi
                # So g_slice_reversed = g[k-hi : k-lo+1][::-1]... 
                # Actually g[k-lo] + g[k-lo-1]... descending
                # Let me just use: cp.sum(cp.multiply(a[lo:hi+1], g[k-lo : k-hi-1 : -1]))
                # g[k-lo : k-hi-1 : -1] gives g[k-lo], g[k-lo-1], ..., g[k-hi]
                cons_expr = cp.sum(cp.multiply(a[lo:hi+1], g[k-hi:k-lo+1][::-1]))

        t0 = time.monotonic()

            a_trial = (1 - trial_t) * a + trial_t * g_rescaled

        a = a * (S / a.sum())
        
        t_step *= t_decay

    cur = c.max()

def fft_convolve_self(a):

    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def exp_weighted_subgradient(a, c, temperature, top_k=300):
    """Exponentially-weighted subgradient of max(c) using softmax over top lags."""
    n = len(a)
    c_max = c.max()
    top_idx = np.argsort(c)[::-1][:top_k]
    c_top = c[top_idx]
    scaled = (c_top - c_max) / temperature
    weights = np.exp(scaled)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    grad = np.zeros(n, dtype=np.float64)
    for w, k in zip(weights, top_idx):

        if lo > hi:

    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

        best_score = c.max()

            vals = 1.0 / (1.0 + np.exp(-steep * (tt - 0.5)))
            vals = np.maximum(vals, 0.05)
            inits.append((f"sigmoid_{n_try}_{steep}", safe_normalize(vals, n_try)))

    # Two asymmetric bumps
    for n_try in [n_main, n_sec]:
        tt = np.arange(n_try, dtype=np.float64) / n_try
        vals = (0.6 + 0.4 * np.exp(-((tt - 0.25) / 0.15) ** 2)
                + 0.3 * np.exp(-((tt - 0.75) / 0.15) ** 2))
        vals = np.maximum(vals, 0.05)
        inits.append((f"twobump_{n_try}", safe_normalize(vals, n_try)))

    # Power law
    for n_try in [n_main, n_sec]:
        for alpha in [0.2, 0.4]:
            tt = (np.arange(n_try, dtype=np.float64) + 1) / n_try
            vals = tt ** alpha
            inits.append((f"power_{n_try}_{alpha}", safe_normalize(vals, n_try)))

    # Asymmetric Gaussian (center offset)
    for n_try in [n_main, n_sec]:
        for center in [0.35, 0.4, 0.6]:
            tt = np.arange(n_try, dtype=np.float64) / n_try
            width = 0.3
            vals = np.exp(-((tt - center) / width) ** 2)
            vals = 0.3 + 0.7 * vals
            inits.append((f"gauss_{n_try}_{center}", safe_normalize(vals, n_try)))

    # Random
    for n_try in [n_main, n_sec]:
        inits.append((f"rand_{n_try}", safe_normalize(rng.random(n_try), n_try)))

    # Sine-based asymmetric
    for n_try in [n_main, n_sec]:
        tt = np.arange(n_try, dtype=np.float64) / n_try
        vals = 0.7 + 0.2 * np.sin(2 * np.pi * tt + 1.0) + 0.1 * np.cos(4 * np.pi * tt + 0.5)
        vals = np.maximum(vals, 0.05)
        inits.append((f"sine_{n_try}", safe_normalize(vals, n_try)))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for non-negative sequence minimizing evaluate_sequence."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 10.0)

    global_best = np.inf
    global_a = None

    def try_update(a, score):
        nonlocal global_best, global_a
        if score < global_best:
            global_best = score
            global_a = a.copy()
            return True
        return False

    # Phase 0: Generate and screen initializations
    inits = make_initializations(rng)
    print(f"Generated {len(inits)} initializations")
    print(f"Total budget: {budget_s}s, working until t={deadline - time.monotonic():.0f}s")

    # Quick screen
    for name, a in inits:
        c = fft_convolve_self(a)
        score = c.max()
        #try_update(a, score)
        if score < global_best:
            global_best = score
            global_a = a.copy()
    print(f"Best after screening: {global_best:.6f}")

    # Phase 1: Main optimization on top candidates
    # Sort by screen score, take top 6 diverse ones
    scored = []
    for name, a in inits:
        c = fft_convolve_self(a)
        scored.append((c.max(), name, a))
    scored.sort(key=lambda x: x[0])

    # Also add the global best as a candidate
    candidates = []
    seen_names = set()
    for score, name, a in scored:
        if name in seen_names:
            continue
        candidates.append((name, a.copy()))
        seen_names.add(name)
        if len(candidates) >= 6:
            break
    if global_a is not None:
        candidates.insert(0, ("global_best", global_a.copy()))

    print(f"\n=== Phase 1: Main optimization ===")
    phase1_budget = min(900.0, (deadline - time.monotonic()) * 0.5)
    n_cand = len(candidates)
    per_cand = phase1_budget / max(1, n_cand)

    for ci, (name, a) in enumerate(candidates):
        if time.monotonic() > deadline - 60:
            break
        n = len(a)
        alloc = min(per_cand, deadline - time.monotonic() - 60)
        if alloc < 5:
            break

        # Run Adam
        a1, s1 = run_adam_main(a, n, alloc * 0.6, rng,
                                lr_scale=0.4, temp_init=0.03, temp_final=0.8,
                                top_k=300, shake_interval=4000, shake_mag=0.012)
        try_update(a1, s1)
        print(f"  [{ci+1}/{n_cand}] {name} adam: {s1:.6f} (global: {global_best:.6f})")

        # Continue from best with mirror descent
        if time.monotonic() < deadline - 60:
            alloc2 = min(alloc * 0.4, deadline - time.monotonic() - 60)
            if alloc2 > 5:
                a2, s2 = run_mirror_descent(a1, n, alloc2, rng, eta_init=0.2, temp_init=0.1)
                try_update(a2, s2)
                print(f"  [{ci+1}/{n_cand}] {name} mirror: {s2:.6f} (global: {global_best:.6f})")

    # Phase 2: Peak rebalancing on the best
    print(f"\n=== Phase 2: Peak rebalancing ===")
    leftover = deadline - time.monotonic()
    if leftover > 20 and global_a is not None:
        n = len(global_a)
        pb_budget = min(300.0, leftover * 0.3)
        a3, s3 = run_peak_rebalance(global_a, n, pb_budget, rng,
                                     eta_init=0.03, eta_decay=0.9995)
        try_update(a3, s3)
        print(f"  Peak rebalance: {s3:.6f} (global: {global_best:.6f})")

        # Re-run Adam from the rebalanced point
        if time.monotonic() < deadline - 30:
            alloc = min(200.0, deadline - time.monotonic() - 30)
            if alloc > 10:
                a4, s4 = run_adam_main(a3, n, alloc, rng,
                                        lr_scale=0.2, temp_init=0.02, temp_final=0.5,
                                        top_k=200, shake_interval=5000, shake_mag=0.008)
                try_update(a4, s4)
                print(f"  Adam after rebalance: {s4:.6f} (global: {global_best:.6f})")

    # Phase 3: Polish
    print(f"\n=== Phase 3: Polish ===")
    leftover = deadline - time.monotonic()
    if leftover > 15 and global_a is not None:
        n = len(global_a)
        polish_budget = leftover * 0.8
        a5, s5 = run_adam_main(global_a, n, polish_budget, rng,
                                lr_scale=0.15, temp_init=0.01, temp_final=0.3,
                                top_k=150, shake_interval=8000, shake_mag=0.005,
                                cosine_iters=15000)
        try_update(a5, s5)
        print(f"  Polish: {s5:.6f} (global: {global_best:.6f})")

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    # Verify with the authoritative function
    final_score = evaluate_sequence(global_a.tolist())
    print(f"\nFinal verified score: {final_score:.6f}")

    if not np.isfinite(final_score):
        print("WARNING: final score is not finite, using fallback")
        global_a = np.ones(256, dtype=np.float64)
        final_score = evaluate_sequence(global_a.tolist())

    return [float(x) for x in global_a]
