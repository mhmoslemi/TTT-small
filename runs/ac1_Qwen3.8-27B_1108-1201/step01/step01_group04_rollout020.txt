Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-based gradient descent with top-k smoothing, momentum, and LP refinement.
Explores multiple lengths and starting points within the time budget.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def _convolve_fft(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = fft(a, n=size)
    c = np.real(ifft(A * A))
    return c[:2*n-1]

def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)

def _grad_topk(a, k=5, use_fft=False):
    """Compute gradient of top-k average of convolution."""
    n = len(a)
    if use_fft and n > 128:
        c = _convolve_fft(a)
    else:
        c = np.convolve(a, a)
    
    # Get top-k indices
    k = min(k, len(c))
    top_k_idx = np.argsort(c)[-k:]
    top_k_vals = c[top_k_idx]
    
    # Weight: uniform over top-k, or softmax
    w = np.zeros(len(c))
    w[top_k_idx] = 1.0 / k

def _grad_softmax_fft(a, tau=50.0):

    n = len(a)
    c = _convolve_fft(a)
    
    # Softmax weights

Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses top-k gradient descent with momentum on the simplex, LP refinement,
and multi-start strategy. Explores multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _obj(a):
    n = len(a)
    c = np.convolve(a, a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * float(c.max()) / (S * S)


def _norm(a, tgt=1.0):
    S = a.sum()
    if S < 1e-15:
        return a
    return a * (tgt / S)


def _lp_direction(f, M, n=None):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    if n is None:
        n = len(f)
    if n > 1024:
        return None
    
    c_obj = -np.ones(n)
    
    # Build A matrix (2n-1 rows, n cols)
    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def _lp_direction_fast(f, M, n=None):
    """Build LP matrix more efficiently using banded structure."""
    if n is None:
        n = len(f)
    if n > 2048:
        return None
    
    c_obj = -np.ones(n)
    
    # Efficient construction using the band structure
    A_rows = []
    A_cols = []
    A_vals = []
    
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A_rows.append(m)
            A_cols.append(j)
            A_vals.append(f[m - j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((A_vals, (A_rows, A_cols)), shape=(2*n-1, n))
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def _optimize_gradient(a0, deadline, k_start=10, k_end=1, momentum=0.85,
                       lr_init=0.15, max_iters=5000, kick_interval=500):
    """Optimize using top-k gradient with momentum on the simplex."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _norm(a, 1.0)  # Work on simplex
    
    best = a.copy()
    best_raw = 2.0 * n * float(np.convolve(a, a).max())
    
    v = np.zeros(n)
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 15:
            break
        
        # Animated k
        progress = min(1.0, it / max_iters)
        k = max(k_end, int(k_start * (1 - progress) + k_end * progress))
        k = min(k, 2 * n - 1)
        
        # Learning rate with cosine annealing
        lr = lr_init * 0.5 * (1 + np.cos(np.pi * progress))
        lr = max(lr, 0.0005)
        
        # Compute convolution
        c = np.convolve(a, a)
        
        # Top-k gradient
        k_eff = min(k, len(c))
        top_idx = np.argpartition(c, -k_eff)[-k_eff:]
        
        grad = np.zeros(n)
        for m in top_idx:
            m = int(m)
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            if hi >= lo:
                grad[lo:hi+1] += 2.0 * a[lo:hi+1][::-1]
        grad /= k_eff
        
        # Project to tangent space of simplex (sum=0)
        grad -= grad.mean()
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= s
        
        # Evaluate
        c_new = np.convolve(a_new, a_new)
        raw = 2.0 * n * float(c_new.max())
        
        if raw < best_raw:
            best_raw = raw
            best = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        a = a_new
        
        # Periodic kick to escape local minima
        if stagnant >= kick_interval and it < max_iters - 100:
            rng = np.random.default_rng(it)
            a = a + 0.05 * (rng.random(n) - 0.5)
            a = np.maximum(a, 0)
            a = _norm(a, 1.0)
            v = np.zeros(n)
            stagnant = 0
    
    return best, 2.0 * n * float(np.convolve(best, best).max())


def _optimize_lp(a0, deadline, max_lp_iters=100, k=3):
    """Optimize using LP-based direction finding."""
    n = len(a0)
    if n > 1024:
        return a0, _obj(a0)
    
    a = np.clip(a0.astype(np.float64), 0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _norm(a, 1.0)
    
    best = a.copy()
    best_score = 2.0 * n * float(np.convolve(a, a).max())
    
    for it in range(max_lp_iters):
        if time.monotonic() > deadline - 10:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        
        # Try slightly reduced M for more aggressive improvement
        for M_factor in [1.0, 0.95, 0.9]:
            g = _lp_direction_fast(a, M * M_factor, n)
            if g is None or g.sum() <= 0:
                continue
            
            # Blend
            for t in [0.5, 0.3, 0.1]:
                a_new = (1 - t) * a + t * g
                a_new = np.maximum(a_new, 0)
                a_new = _norm(a_new, 1.0)
                
                raw = 2.0 * n * float(np.convolve(a_new, a_new).max())
                if raw < best_score:
                    best_score = raw
                    best = a_new.copy()
                    a = a_new
                    break
        
        # Small gradient step to avoid getting stuck
        c = np.convolve(a, a)
        top_idx = np.argpartition(c, -k)[-k:]
        grad = np.zeros(n)
        for m in top_idx:
            m = int(m)
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            if hi >= lo:
                grad[lo:hi+1] += 2.0 * a[lo:hi+1][::-1]
        grad /= k
        grad -= grad.mean()
        a = a - 0.01 * grad
        a = np.maximum(a, 0)
        a = _norm(a, 1.0)
    
    return best, best_score

    best_seq = None
    best_score = float('inf')
    
    def _update(a, score):

        if score < best_score:

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses top-k gradient descent with momentum on the probability simplex,
LP-based direction finding, multi-start strategy, and length exploration.
Improvements over baseline: simplex formulation, animated top-k, Nesterov momentum,
aggressive LP with multiple tightness factors, periodic kicks, and resampling.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute objective value."""
    n = len(a)
    c = np.convolve(a, a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * c.max()) / (S * S)


def _norm_simplex(a):
    """Normalize to probability simplex (sum=1, non-negative)."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return np.ones_like(a) / len(a)
    return a / S


def _topk_gradient(a, c, k):
    """Compute gradient of top-k average of convolution values."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argpartition(c, -k)[-k:]
    grad = np.zeros(n)
    for m in top_idx:
        m = int(m)
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if hi >= lo:
            grad[lo:hi + 1] += 2.0 * a[lo:hi + 1][::-1]
    grad /= float(k)
    return grad


def _optimize_grad(a0, deadline, k_start=8, k_end=1, momentum=0.88,
                   lr_init=0.12, max_iters=3000, kick_every=400):
    """Gradient descent with momentum and animated top-k on the simplex."""
    n = len(a0)
    a = _norm_simplex(a0.astype(np.float64))
    if a.sum() < 1e-14:
        a = np.ones(n) / n
    c = np.convolve(a, a)
    best_raw = float(2.0 * n * c.max())
    best = a.copy()

    v = np.zeros(n)
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break

        progress = min(1.0, float(it) / float(max_iters))
        k = max(k_end, int(round(k_start * (1.0 - progress) + k_end * progress)))
        k = min(k, 2 * n - 1)

        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.0003)

        c = np.convolve(a, a)
        grad = _topk_gradient(a, c, k)
        grad -= grad.mean()

        # Nesterov-style: look-ahead then step
        a_look = a + momentum * v
        c_look = np.convolve(a_look, a_look)
        grad_look = _topk_gradient(a_look, c_look, k)
        grad_look -= grad_look.mean()

        v = momentum * v - lr * grad_look
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-14:
            a_new = np.ones(n) / n
        else:
            a_new /= s

        c_new = np.convolve(a_new, a_new)
        raw = float(2.0 * n * c_new.max())
        if raw < best_raw:
            best_raw = raw
            best = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        a = a_new

        if stagnant >= kick_every and it > 100:
            rng = np.random.default_rng(hash((n, it)) % (2**31))
            a = a + 0.03 * (rng.random(n) - 0.5)
            a = _norm_simplex(a)
            v = np.zeros(n)
            stagnant = 0

    return best, best_raw


def _build_lp_matrix(f, n):
    """Build constraint matrix for LP: A[m,j] = f[m-j] if valid."""
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            A[m, j] = f[m - j]
    return A


def _lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    if n > 1024:
        return None
    c_obj = -np.ones(n)
    A = _build_lp_matrix(f, n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _optimize_lp(a0, deadline, max_lp_iters=60, grad_k=3):
    """LP-based optimization with gradient assistance."""
    n = len(a0)
    if n > 1024:
        return a0, 2.0 * n * float(np.convolve(a0, a0).max())

    a = _norm_simplex(a0.astype(np.float64))
    c = np.convolve(a, a)
    best_raw = float(2.0 * n * c.max())
    best = a.copy()

    for it in range(max_lp_iters):
        if time.monotonic() > deadline - 10.0:
            break

        c = np.convolve(a, a)
        M = float(c.max())

        improved = False
        for M_frac in [1.0, 0.97, 0.93, 0.88]:
            g = _lp_direction(a, M * M_frac, n)
            if g is None or g.sum() < 1e-10:
                continue
            for t_blend in [0.6, 0.4, 0.2]:
                a_new = (1.0 - t_blend) * a + t_blend * g
                a_new = _norm_simplex(a_new)
                raw = float(2.0 * n * np.convolve(a_new, a_new).max())
                if raw < best_raw - 1e-10:
                    best_raw = raw
                    best = a_new.copy()
                    a = a_new
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Small gradient step to escape
            c = np.convolve(a, a)
            grad = _topk_gradient(a, c, grad_k)
            grad -= grad.mean()
            a = a - 0.005 * grad
            a = _norm_simplex(a)

    return best, best_raw


def _make_starts(rng, available_global):
    """Generate diverse starting sequences."""
    starts = []

    if available_global is not None:
        gs = np.clip(np.asarray(available_global, dtype=np.float64), 0, 1000)
        if gs.size >= 32 and np.all(np.isfinite(gs)):
            for L in [128, 192, 256, 384, 512, min(768, len(gs))]:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
            # Also try reversed and subset
            if len(gs) >= 512:
                starts.append(gs[:512][::-1].copy())
                starts.append(np.abs(gs[:256] - gs[256:512]))

    for n in [128, 256, 512]:
        starts.append(0.1 + 0.9 * rng.random(n))

    for n in [256, 512]:
        i = np.arange(n, dtype=np.float64)
        fn = float(n)
        starts.append(0.5 + 0.5 * np.cos(2.0 * np.pi * i * 7.0 / fn))
        starts.append(0.3 + 0.35 * np.cos(2.0 * np.pi * i * 3.0 / fn)
                      + 0.35 * np.sin(2.0 * np.pi * i * 11.0 / fn))
        starts.append(0.3 + 0.35 * np.cos(2.0 * np.pi * i * 5.0 / fn)
                      + 0.35 * np.sin(2.0 * np.pi * i * 13.0 / fn))
        starts.append(np.exp(-3.0 * i / fn) + 0.1)
        starts.append(np.exp(-8.0 * ((i / fn) - 0.25) ** 2)
                      + np.exp(-8.0 * ((i / fn) - 0.75) ** 2) + 0.05)
        # Golay-inspired (Walsh-Hadamard based)
        walsh = np.sign(np.cos(np.pi * np.outer(i // 1, i // 1) / fn))
        if walsh.size == n * n:
            starts.append(np.abs(walsh).mean(axis=0) + 0.1)

    starts.sort(key=lambda x: len(x))
    return starts


def _resample_sequence(a, new_n):
    """Resample sequence to a new length via interpolation."""
    old_n = len(a)
    if old_n < 2 or new_n < 2:
        return a.copy()
    t_old = np.linspace(0.0, 1.0, old_n)
    t_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(t_new, t_old, a)
