    s = float(p.sum())
    if s <= 1e-12 or p.size == 0:
        return np.full(max(1, p.size), 1.0/max(1,p.size), dtype=np.float64)
    return p / s


def _conv_full(p):
    n = int(np.asarray(p).size)
    if n == 0:
        return np.empty(0)
    L = next_fast_len(2*n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1].astype(np.float64)


def _internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(p.sum())
    if s <= 1e-12:
        return np.inf
    c = _conv_full(p)
    cmax = float(np.max(c)) if c.size > 0 else 0.0
    return float(2.0 * n * cmax / (s * s))


def _build_conv_matrix(n, a):
    """Build the (2n-1) x n sparse matrix A where A[k,i] = a[i] if 0<=k-i<n."""
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            data.append(a[i])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A


def _lp_best_response(a, M_ref=None):
    """
    Solve: min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0, t>=0.
    Returns (g_opt, t_opt).
    """
    n = int(len(a))
    if n <= 0:
        return a.copy(), np.inf
    if not _HAVE_LP:
        return a.copy(), np.inf

    A_conv = _build_conv_matrix(n, a)

    # Build full constraint matrix: [A_conv | -1] with shape (2n-1, n+1)
    # We need to add a column of -1's to A_conv
    from scipy.sparse import eye as speye
    neg_one_col = csr_matrix(np.full(2*n-1, -1.0), shape=(2*n-1, 1))
    A_ub = np.hstack([A_conv.toarray(), neg_one_col.toarray()])
    b_ub = np.zeros(2*n-1, dtype=np.float64)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimize t (last variable)
    c_obj = np.zeros(n+1, dtype=np.float64)
    c_obj[n] = 1.0

    bounds = [(0.0, None)] * (n+1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'disp': False, 'time_limit': 30.0})
    except Exception:
        return a.copy(), np.inf

    if res is None or not res.success:
        # Fallback: return a itself
        return a.copy(), M_ref if M_ref else _internal_score(a)

    g_opt = res.x[:n]
    t_opt = float(res.x[n])
    g_opt = np.where(np.isfinite(g_opt), g_opt, 0.0)
    g_opt = np.maximum(g_opt, 0.0)
    gs = float(g_opt.sum())
    if gs <= 1e-12:
        return a.copy(), M_ref if M_ref else np.inf
    g_opt = g_opt / gs
    return g_opt, t_opt


def _gradient_smoothed_max(a, T_factor=0.05):
    """Compute gradient of smoothed max of conv(a,a) w.r.t. a."""
    p = np.asarray(a, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0)

    c = _conv_full(p)
    cmax = float(np.max(c)) if c.size > 0 else 0.0
    if cmax <= 0:
        return np.zeros(n)

    T = max(T_factor * cmax, 1e-12)
    diff = (c - cmax) / T
    # Clip for numerical safety
    diff = np.clip(diff, -50, 50)
    e = np.exp(diff)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    # grad[i] = 2 * sum_k w[k] * a[k-i]  (cross-correlation of w and a)
    N = next_fast_len(n + 2*n - 1)
    Fw = fft(w, N)
    Fa = fft(p, N)
    cross = np.real(ifft(Fw * Fa))
    # cross[m] = sum_k w[k] * a[k-m] for m = 0, 1, ...
    grad = 2.0 * cross[:n]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-12:
        return np.full(n, 1.0/n)
    return out / s


def _make_geometric(n, r):
    x = np.arange(n, dtype=np.float64)
    p = np.power(r, x)
    return _safe_normalize(p)


def _make_geometric_rev(n, r):
    p = _make_geometric(n, r)
    return p[::-1].copy()


def _make_ramp_plateau(n, ramp_frac, plateau_h):
    """Ramp up, plateau, ramp down - asymmetric."""
    n = int(max(4, n))
    rf = int(n * ramp_frac)
    p = np.zeros(n)
    for i in range(n):
        if i < rf:
            p[i] = (i+1) / rf if rf > 0 else 1
        elif i < n - rf:
            p[i] = plateau_h
        else:
            p[i] = (n - i) / rf if rf > 0 else 1
    return _safe_normalize(p)


def _make_skew(n, alpha, beta):
    x = np.linspace(0.01, 0.99, n)
    p = np.power(x, alpha) * np.power(1-x, beta)
    return _safe_normalize(np.maximum(p, 0))


def _make_two_pole(n, mu1, sig1, w1, mu2, sig2, w2):
    x = np.linspace(0.0, 1.0, n)
    p = w1 * np.exp(-((x-mu1)**2) / max(1e-8, 2*sig1*sig1)) + \
        w2 * np.exp(-((x-mu2)**2) / max(1e-8, 2*sig2*sig2))
    return _safe_normalize(np.maximum(p, 0.01))


def _make_polynomial(n, coeffs):
    x = np.linspace(0.0, 1.0, n)
    p = np.zeros(n)
    for d, c in enumerate(coeffs):
        p += c * np.power(x, d)
    p = np.maximum(p, 0.01)
    return _safe_normalize(p)


def _make_random_asymmetric(n, rng):
    """Random smooth asymmetric shape."""
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + 0.5*rng.uniform() * np.sin(2*np.pi*rng.uniform(1,5)*x + rng.uniform(0, 2*np.pi))
    p += 0.3*rng.uniform() * np.sin(2*np.pi*rng.uniform(1,8)*x + rng.uniform(0, 2*np.pi))
    p += rng.uniform(0, 1.5) * x  # asymmetric bias
    p = np.maximum(p, 0.01)
    return _safe_normalize(p)


def _generate_inits(n, rng, templates=None):
    """Generate diverse initializations, emphasizing asymmetric shapes."""
    inits = []

    # Geometric (both directions, various rates)
    for r in [0.92, 0.95, 0.97, 0.985, 0.99, 0.995, 0.998]:
        inits.append(_make_geometric(n, r))
        inits.append(_make_geometric_rev(n, r))

    # Skew beta-like
    for a, b in [(0.5, 2), (1, 3), (2, 1), (0.3, 4), (3, 0.5), (0.7, 1.5), (1.5, 2.5)]:
        inits.append(_make_skew(n, a, b))

    # Ramp-plateau
    for rf in [0.1, 0.2, 0.3, 0.5]:
        for ph in [1.0, 1.5, 2.0]:
            inits.append(_make_ramp_plateau(n, rf, ph))

    # Two-pole
    inits.append(_make_two_pole(n, 0.2, 0.08, 1.0, 0.7, 0.12, 0.5))
    inits.append(_make_two_pole(n, 0.3, 0.15, 1.0, 0.8, 0.05, 0.8))
    inits.append(_make_two_pole(n, 0.15, 0.1, 1.5, 0.6, 0.08, 1.0))
    inits.append(_make_two_pole(n, 0.5, 0.2, 1.0, 0.9, 0.05, 1.0))
    inits.append(_make_two_pole(n, 0.1, 0.06, 2.0, 0.5, 0.15, 0.5))

    # Polynomials
    inits.append(_make_polynomial(n, [1, -2, 1.5]))  # peak in middle-left
    inits.append(_make_polynomial(n, [0.5, 1, -0.3]))
    inits.append(_make_polynomial(n, [2, -1, 0]))
    inits.append(_make_polynomial(n, [1, 0.5, -0.8, 0.3]))

    # Random asymmetric
    for _ in range(8):
        inits.append(_make_random_asymmetric(n, rng))

    # Templates (interpolated from smaller n)
    if templates:
        for tpl in templates:
            if tpl is not None and len(tpl) > 1:
                new_len = n
                old_len = len(tpl)
                if old_len != new_len:
                    old_x = np.linspace(0, 1, old_len)
                    new_x = np.linspace(0, 1, new_len)
                    interp = np.interp(new_x, old_x, tpl)
                    inits.append(_safe_normalize(interp))
                else:
                    inits.append(_safe_normalize(tpl))

    # Uniform as baseline
    inits.append(np.full(n, 1.0/n))

    return inits


def _lp_alternate(a, rng, max_iters, time_limit, label=""):
    """
    Main LP alternating loop with adaptive mixing.
    Returns (best_a, best_score, t_opt_final).
    """
    a = _safe_normalize(np.asarray(a, dtype=np.float64))
    n = int(a.size)
    best_a = a.copy()
    best_score = _internal_score(a)
    cur_score = best_score
    start = time.monotonic()

    mixing_ratios = [0.5, 0.35, 0.25, 0.65, 0.8, 0.4, 0.15, 0.7]
    ratio_idx = 0
    stuck_count = 0
    last_improve_iter = 0

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_limit:
            break

        # Solve LP
        g_opt, t_opt = _lp_best_response(a)
        if g_opt is None or not np.all(np.isfinite(g_opt)):
            break

        # Check convergence: if g ≈ a, we're at fixed point
        dist = float(np.sum(np.abs(g_opt - a)))
        if dist < 1e-6:
            break

        # Try multiple mixing ratios
        improved = False
        for ti in range(4):
            t_mix = mixing_ratios[(ratio_idx + ti) % len(mixing_ratios)]
            a_new = (1.0 - t_mix) * a + t_mix * g_opt
            a_new = _project_simplex(a_new)
            sc_new = _internal_score(a_new)

            if np.isfinite(sc_new) and sc_new < cur_score - 1e-10:
                a = a_new
                cur_score = sc_new
                improved = True
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    last_improve_iter = it
                    stuck_count = 0
                break

        if not improved:
            # Try pure LP solution
            a_check = _safe_normalize(g_opt)
            sc_check = _internal_score(a_check)
            if np.isfinite(sc_check) and sc_check < best_score:
                best_score = sc_check
                best_a = a_check.copy()
            stuck_count += 1
            if stuck_count > 15:
                # Perturb and reset
                a = _project_simplex(a + rng.normal(0, 0.05/n, n))
                cur_score = _internal_score(a)
                stuck_count = 0
                ratio_idx = (ratio_idx + 1) % len(mixing_ratios)

        if (it + 1) % 50 == 0 and label:
            print(f"  {label} iter={it} score={best_score:.8f} t_opt={t_opt:.6f} dist={dist:.6f}", flush=True)

        if best_score < 1.5027 and stuck_count > 5:
            break

    return best_a, best_score, t_opt


def _gradient_polish(a, time_limit, label=""):
    """Gradient-based polishing using smoothed max surrogate."""
    p = _safe_normalize(np.asarray(a, dtype=np.float64))
    n = int(p.size)
    best_p = p.copy()
    best_score = _internal_score(p)
    cur_score = best_score
    start = time.monotonic()
    T_factor = 0.08
    lr = 0.3
    prev_p = p.copy()

    while time.monotonic() - start < time_limit:
        g = _gradient_smoothed_max(p, T_factor)
        if not np.all(np.isfinite(g)):
            break

        # Projected gradient step with momentum
        v = p - lr * g + 0.3 * (p - prev_p)
        p_new = _project_simplex(v)

        c_new = _conv_full(p_new)
        cmax_new = float(np.max(c_new))
        s_new = float(p_new.sum())
        sc_new = float(2.0 * n * cmax_new / (s_new * s_new))

        # Smoothed max for acceptance
        T = max(T_factor * cmax_new, 1e-12)
        cmax_cur = float(np.max(_conv_full(p)))
        T_cur = max(T_factor * cmax_cur, 1e-12)
        diff_new = np.clip((c_new - cmax_new) / T, -50, 50)
        surr_new = cmax_new + T * np.log(float(np.exp(diff_new).sum()))

        diff_cur = np.clip((_conv_full(p) - cmax_cur) / T_cur, -50, 50)
        surr_cur = cmax_cur + T_cur * np.log(float(np.exp(diff_cur).sum()))

        if np.isfinite(sc_new) and (surr_new < surr_cur - 1e-12 or sc_new < best_score - 1e-10):
            prev_p = p
            p = p_new
            if sc_new < best_score:
                best_score = sc_new
                best_p = p_new.copy()
                lr = min(lr * 1.05, 2.0)
            else:
                lr = max(lr * 0.9, 0.01)
            cur_score = sc_new
        else:
            lr = max(lr * 0.8, 0.001)
            T_factor = max(T_factor * 0.95, 0.01)

        if best_score < 1.5027:
            break

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for non-negative step function minimizing 2n*max(conv)/(sum)^2."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 15.0)

    # Try to use parent sequence
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 16 and parent_arr.sum() > 0:
            parent = _safe_normalize(parent_arr)
    except Exception:
        parent = None

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = _safe_normalize(p)
        if not np.isfinite(sc):
            sc = _internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p.copy()
            print(f"  NEW BEST: score={sc:.10f} n={p.size}", flush=True)

    print(f"Starting search. budget={budget_s}s, parent={'yes' if parent is not None else 'no'}", flush=True)

    # Phase 1: Quick exploration on small n to find good templates
    templates = []
    small_ns = [40, 80, 160]
    time_per_small = min(30.0, (deadline - start_time - 300) * 0.08)

    for sn in small_ns:
        if time.monotonic() > deadline - 300:
            break
        inits = _generate_inits(sn, rng)
        scored = []
        for init in inits:
            sc = _internal_score(init)
            if np.isfinite(sc):
                scored.append((sc, init))
        scored.sort(key=lambda x: x[0])

        for sc, init in scored[:3]:
            a_best, sc_best, _ = _lp_alternate(init, rng, 80, time_per_small, f"n={sn}")
            try_record(a_best, sc_best)
            if a_best is not None:
                templates.append(a_best)
                print(f"  template n={sn} score={sc_best:.8f}", flush=True)

    # Phase 2: Main search on larger n values
    large_ns = [256, 384, 512, 640, 768, 900]

    # If parent exists, include its length
    if parent is not None:
        pn = len(parent)
        if pn not in large_ns and 100 <= pn <= 1200:
            large_ns.append(pn)
            large_ns.sort()

    time_per_large = min(200.0, (deadline - start_time - 120) / max(1, len(large_ns)))
    print(f"Phase 2: {len(large_ns)} n-values, ~{time_per_large:.0f}s each", flush=True)

    for ln in large_ns:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - 60 - time.monotonic()
        if remaining < 30:
            break

        # Add parent as template if close in length
        cur_templates = list(templates)
        if parent is not None:
            cur_templates.append(parent)

        inits = _generate_inits(ln, rng, cur_templates)
        scored = []
        for init in inits:
            sc = _internal_score(init)
            if np.isfinite(sc):
                scored.append((sc, init))
        scored.sort(key=lambda x: x[0])

        print(f"  n={ln}: {len(scored)} inits, best init score={scored[0][0]:.6f}", flush=True)

        # Run LP alternation on top initializations
        n_top = min(6, len(scored))
        time_each = max(20.0, min(120.0, remaining / (n_top * 1.5)))

        for idx in range(n_top):
            if time.monotonic() > deadline - 60:
                break
            sc_init, init = scored[idx]
            remaining_here = deadline - 60 - time.monotonic()
            if remaining_here < 25:
                break

            limit = min(time_each, remaining_here - 20)
            if limit < 20:
                limit = remaining_here * 0.7

            a_best, sc_best, _ = _lp_alternate(init, rng, 300, limit, f"n={ln}#{idx}")
            try_record(a_best, sc_best)

            if a_best is not None:
                # Add as template for subsequent n values
                if sc_best < best_score:
                    templates.append(a_best.copy())

        # Gradient polish on best for this n
        if time.monotonic() < deadline - 40:
            polish_limit = min(60.0, deadline - 40 - time.monotonic())
            if polish_limit > 10:
                a_pol, sc_pol = _gradient_polish(best_p, polish_limit, f"polish n={ln}")
                try_record(a_pol, sc_pol)

    # Phase 3: Final intensive polish on the best solution
    if best_p is not None and time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        print(f"Phase 3: Final polish, {remaining:.0f}s remaining", flush=True)

        # Multiple gradient polish passes with different T
        for T_factor in [0.05, 0.03, 0.01, 0.08]:
            if time.monotonic() > deadline - 15:
                break
            lim = min(remaining * 0.3, 80.0)
            if lim < 5:
                break
            a_pol, sc_pol = _gradient_polish(best_p, lim, f"final T={T_factor}")
            try_record(a_pol, sc_pol)

        # One more LP round if time permits
        if time.monotonic() < deadline - 30 and len(best_p) <= 1200:
            lim = min(60.0, deadline - 30 - time.monotonic())
            if lim > 20:
                a_lp, sc_lp, _ = _lp_alternate(best_p, rng, 100, lim, "FINAL-LP")
                try_record(a_lp, sc_lp)

    # Fallback
    if best_p is None:
        n_fb = 512
        best_p = np.full(n_fb, 1.0/n_fb)
        best_score = _internal_score(best_p)

    # Final validation
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s < 0.01:
        best_p = np.full(max(1, best_p.size), 1.0/max(1, best_p.size))

    final_score = _internal_score(best_p)
    print(f"\nFinal: score={final_score:.10f} n={len(best_p)}", flush=True)

    return [float(x) for x in best_p]
