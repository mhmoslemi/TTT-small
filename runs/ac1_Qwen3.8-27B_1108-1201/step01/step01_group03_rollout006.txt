```python
"""
Hybrid mirror-descent + LP optimization for minimizing autocorrelation peak.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

# --- Core functions ---

def autocorr(f):
    """Full linear autocorrelation."""
    return np.convolve(f, f)

def score_from_conv(f, conv):
    """Compute score from f and its autocorrelation."""
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * len(f) * conv.max() / (s * s)

def fast_score(f):
    conv = autocorr(f)
    return score_from_conv(f, conv)

# --- Mirror Descent ---

def softmax(h, max_exp=500):
    h_max = h.max()
    if h_max > max_exp:
        h = h - h_max
    e = np.exp(h - h.max())
    return e / e.sum()

def gradient_at_lag(f, k):
    """Compute d(f*f[k])/df for the given lag k."""
    n = len(f)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 2.0 * f
    else:
        # For j < n-k: contributes f[j+k]
        grad[:n-k] += f[k:]
        # For j >= k: contributes f[j-k]
        grad[k:] += f[:n-k]
    return grad

def mirror_descent_step(h, v, f, lr, beta, k):
    """One mirror descent step with Nesterov momentum."""
    grad = gradient_at_lag(f, k)
    v = beta * v + (1 - beta) * grad
    h_new = h - lr * v
    return h_new, v

def run_mirror_descent(f_init, lr, beta, n_iter, time_limit):
    """Run mirror descent for a fixed number of iterations or until time limit."""
    n = len(f_init)
    h = np.log(f_init + 1e-30)  # initial h such that softmax(h) ≈ f_init
    v = np.zeros(n)
    best_f = f_init.copy()
    best_sc = fast_score(f_init)
    start_time = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start_time > time_limit:
            break
        
        conv = autocorr(f_init)
        k_star = int(np.argmax(conv))
        
        h, v = mirror_descent_step(h, v, f_init, lr, beta, k_star)
        f_init = softmax(h)
        
        if it % 10 == 0:
            sc = score_from_conv(f_init, autocorr(f_init))
            if sc < best_sc:
                best_sc = sc
                best_f = f_init.copy()
    
    return best_f, best_sc
```

Hmm wait, I realize there's an issue with the mirror descent approach. The "gradient" I'm computing is for the max of f*f at a specific lag. But the actual objective is max_k (f*f)[k], which is the max over all k. The subgradient at the point where k* achieves the max is indeed the gradient at k*, but this only gives one subgradient (the max is not differentiable at points where multiple lags achieve the max).

A better approach: use a **weighted average of gradients** over the "active" lags (those where f*f[k] is close to the max). This gives a more informed descent direction.

Let me revise:
- At each step, find the top m lags (where m is small, like 5-20)
- Compute a weighted average of their gradients (weighted by how close they are to the max)
- Use this as the descent direction

This is similar to the "top-k" approach in robust optimization.

Also, I should consider using the **soft-max** approach more seriously. For a large but finite p:
- Objective: (sum_k (f*f[k])^p)^{1/p}
- This is smooth and its gradient uses ALL lags (weighted by (f*f[k])^{p-1})
- For p → ∞, this converges to the max

The advantage of using a finite p is that we get a smooth, well-defined gradient that considers all lags, not just the single worst one.

```python
def run_soft_mirror_descent(f_init, p, lr_init, decay, n_iter, time_limit, momentum=0.9):
    """Mirror descent on soft-max objective with Nesterov momentum."""
    n = len(f_init)
    h = np.log(f_init + 1e-30)
    v = np.zeros(n)
    f = f_init.copy()
    best_f = f.copy()
    best_sc = compute_score(f)
    start_time = time.monotonic()
    lr = lr_init
    
    for it in range(n_iter):
        if time.monotonic() - start_time > time_limit:
            break
        
        conv = autocorr(f)
        sc = 2.0 * n * conv.max() / (f.sum()**2)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
        
        # Compute soft gradient
        w = conv ** (p - 1)
        corr = np.correlate(w, f, mode='full')
        grad = 2.0 * corr[n-1 : n-1+n]
        
        # Normalize gradient (divide by sum(f)^2 for scale invariance)
        s = f.sum()
        if s > 1e-10:
            grad = grad / (s * s)
        
        # Nesterov momentum update
        v = momentum * v + (1 - momentum) * grad
        h = h - lr * v
        f = safe_softmax(h)
        
        # Adaptive learning rate (simple decay)
        if it % 100 == 99:
            lr *= decay
    
    return best_f, best_sc

def lp_refine(f, budget_s, t_init=0.3, t_min=0.02):
    """LP-based refinement similar to the previous approach."""
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_sc = compute_score(f)
    t = t_init
    start_time = time.monotonic()
    
    while time.monotonic() - start_time < budget_s:
        conv = autocorr(f)
        M = float(conv.max())
        if M < 1e-15:
            break
        
        # Build sparse constraint matrix
        # A[k, j] = f[k-j] for valid k-j, else 0
        # Shape: (2n-1) x n
        A = build_conv_matrix(n, f)
        
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'time_limit': 5.0, 'presolve': True})
            if res.status != 0:
                break
        except Exception:
            break
        
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        
        g = g / gs
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new = f_new / s
        
        sc = compute_score(f_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
        
        f = f_new
        t = max(t_min, t * 0.99)
    
    return best_f, best_sc

def build_conv_matrix(n, f):
    """Build sparse matrix A where A[k,j] = f[|k-j|] if |k-j| < n else 0."""
    # Actually A[k,j] = f[k-j] for 0 <= k-j < n
    # For the constraint (f*g)[k] <= M where (f*g) is full convolution:
    # (f*g)[k] = sum_j f[k-j] * g[j] (with f[x]=0 for x outside [0,n-1])
    # So A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
```

```python
def build_conv_matrix(n, f):
    # For each row k (0 to 2n-2) and column j (0 to n-1):
    # A[k,j] = f[k-j] if 0 <= k-j < n
    # k-j >= 0 => k >= j
    # k-j < n => k < n + j => j > k - n
    
    # So for row k, the valid columns j are: max(0, k-n+1) <= j <= min(n-1, k)
    
    # Vectorized construction:
    # For each j, the rows k where A[k,j] = f[k-j] are: k = j, j+1, ..., j+n-1
    # (since k-j ranges from 0 to n-1)
    
    # So column j has nonzeros at rows j, j+1, ..., j+n-1 with values f[0], f[1], ..., f[n-1]
    
    # This is a banded matrix! Each column j is a shifted copy of f.
    
    col_data = np.tile(f, n)  # n copies of f
    col_idx = np.repeat(np.arange(n), n)
    row_idx = col_idx + np.tile(np.arange(n), n)  # rows j, j+1, ..., j+n-1 for each j
    
    A = sp.coo_matrix((col_data, (row_idx, col_idx)), shape=(2*n-1, n)).tocsr()
    return A
```
