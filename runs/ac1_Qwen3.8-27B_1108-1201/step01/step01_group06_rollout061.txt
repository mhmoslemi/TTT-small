    a = safe_normalize(a, n)
    fsize = 1 << (2*n-1).bit_length()
    
    for _ in range(num_iters):
        A = fft(a, fsize)
        # Flatten: reduce large magnitudes, increase small ones
        mag = np.abs(A)
        # Scale: new_mag = mag * exp(-alpha * (mag^2 - mean(mag^2)) / (2*var(mag^2)))
        mag2 = mag ** 2
        mu = mag2.mean()
        sigma2 = mag2.var() + 1e-10

        # Inverse FFT and project

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new * (S / s)
        a = a_new
    return a

"""
Minimize 2*n*max_conv/sum^2 using spectral domain optimization,
improved proximal gradient with line search, and periodic structure exploitation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_subgradient_full(a, n, c, top_k=None):
    """Compute subgradient using top-k lags."""
    if top_k is None or top_k >= len(c):
        top_indices = np.argsort(c)[::-1]
    else:
        top_indices = np.argsort(c)[::-1][:top_k]
    
    grad = np.zeros(n, dtype=np.float64)
    total = c[top_indices].sum()
    if total <= 1e-15:
        return grad
    weights = c[top_indices] / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += w_val * 2.0 * a[k-hi:k-lo+1][::-1]
    return grad

def proximal_gradient(a_init, n, max_time, rng, lr_init=0.1, top_k=20, use_line_search=True):
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    lr = lr_init
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        grad = compute_subgradient_full(a, n, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm <= 1e-12:
            break
        grad_unit = grad / gnorm
        
        # Line search (Armijo)
        if use_line_search:
            step = lr
            f_cur = cur
            improved = False
            for _ in range(10):
                a_trial = project_simplex(a - step * grad_unit, S)
                c_trial = fft_convolve_self(a_trial)
                f_trial = c_trial.max()
                if f_trial < f_cur - 1e-6 * step * gnorm:
                    a = a_trial
                    improved = True
                    break
                step *= 0.5
            if not improved:
                lr *= 0.5
        else:
            a = project_simplex(a - lr * grad_unit, S)
            lr = lr_init * (0.99 ** t)
        
        # Periodic restart
        if t % 1000 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, 0.01 * S / n, n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_score

def spectral_optimize(a_init, n, max_time, alpha=0.5):
    """Iteratively flatten the spectrum."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    fsize = 1 << (2*n-1).bit_length()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        A = fft(a, fsize)
        mag2 = np.abs(A) ** 2
        mu = mag2.mean()
        sigma2 = mag2.var() + 1e-10
        # Flatten: reduce peaks in spectrum
        new_mag = np.abs(A) * np.exp(-alpha * 0.5 * (mag2 - mu) / sigma2)
        new_A = new_mag * np.exp(1j * np.angle(A))
        new_A[0] = A[0]  # Preserve DC
        a_new = np.real(ifft(new_A, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new * (S / s)
        a = a_new
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
    
    return best_local, best_score

def periodic_search(n, max_time, rng, p=8):
    """Optimize a periodic pattern of length p."""
    S = np.sqrt(2.0 * float(n))
    best_score = np.inf
    best_pattern = None
    
    # Initialize pattern
    pattern = rng.random(p) * 0.5 + 0.5
    
    start = time.monotonic()
    t = 0
    lr = S / (p * 10)
    
    while time.monotonic() - start < max_time:
        t += 1
        # Expand pattern to full length
        a = np.tile(pattern, n // p + 1)[:n]
        a = safe_normalize(a, n)
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_pattern = pattern.copy()
        
        # Compute gradient w.r.t. pattern
        grad = compute_subgradient_full(a, n, c, top_k=20)
        # Average gradient over periods
        pat_grad = np.zeros(p)
        for i in range(p):
            positions = np.arange(i, n, p)
            pat_grad[i] = grad[positions].mean()
        
        gnorm = np.linalg.norm(pat_grad)
        if gnorm <= 1e-12:
            break
        
        pattern = pattern - lr * pat_grad / gnorm
        pattern = np.maximum(pattern, 0.0)
        lr = lr * 0.999
    
    # Expand best pattern
    a = np.tile(best_pattern, n // p + 1)[:n]
    return safe_normalize(a, n), best_score

def run_mirror_descent(a_init, n, max_time, rng, eta_init=0.5, top_k=20):
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    eta = eta_init
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        grad = compute_subgradient_full(a, n, c, top_k)
        gmax = np.abs(grad).max()
        if gmax <= 1e-12:
            break
        grad_norm = grad / gmax
        
        a_new = a * np.exp(-eta * grad_norm)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new * (S / s)
        else:
            a = np.full(n, S/n)
        
        eta = eta_init / (1.0 + 0.002 * t)
    
    return best_local, best_score

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Algorithm:
1. Fast subgradient via argmax lag (O(n) per step)
2. Projected Gradient Descent with backtracking line search
3. Mirror Descent (multiplicative weights)
4. Spectral Domain Flattening (new approach)
5. Multi-phase search: initialize -> optimize -> polish
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_argmax(a, c, top_k=1):
    """Compute subgradient of max(c) using top_k argmax lags."""
    n = len(a)
    if top_k == 1:
        k = int(np.argmax(c))
        grad = np.zeros(n, dtype=np.float64)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        return grad
    else:
        # Use top-k with equal weights (or proportional to c values)
        top_idx = np.argsort(c)[::-1][:top_k]
        vals = c[top_idx]
        total = vals.sum()
        if total <= 1e-15:
            return np.zeros(n)
        weights = vals / total
        grad = np.zeros(n, dtype=np.float64)
        for w_val, k in zip(weights, top_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += w_val * 2.0 * a[k-hi:k-lo+1][::-1]
        return grad


def pgd_with_linesearch(a_init, n, max_time, rng, lr_init=1.0, top_k=1):
    """Projected Gradient Descent with backtracking line search."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    lr = lr_init
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            # Try a restart
            a = project_simplex(a + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)
            stall = 0
        
        grad = subgradient_argmax(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(15):
            a_trial = project_simplex(a + step * direction, S)
            c_trial = fft_convolve_self(a_trial)
            f_trial = c_trial.max()
            if f_trial < cur - 1e-8 * step * gnorm:
                a = a_trial
                lr = min(step * 2.0, lr_init * 10)
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break
    
    return best_local, best_score


def mirror_descent(a_init, n, max_time, rng, eta_init=1.0, top_k=1):
    """Mirror descent / multiplicative weights."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    eta = eta_init
    start = time.monotonic()
    t = 0
    stall = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1

            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            stall = 0

        else:
            a = np.full(n, S / n)

def spectral_flatten(a_init, n, max_time, alpha=0.5):
    """Iteratively flatten the power spectrum to reduce max convolution."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < max_time:
        A = fft(a, fsize)
        mag2 = np.abs(A) ** 2
        mu = mag2.mean()
        sigma2 = mag2.var() + 1e-10
        
        # Reduce spectral peaks
        scale = np.exp(-alpha * (mag2 - mu) / (2.0 * sigma2))
        new_A = np.abs(A) * scale * np.exp(1j * np.angle(A))
        new_A[0] = A[0]  # Preserve DC (maintain sum)
        
        a_new = np.real(ifft(new_A, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new * (S / s)
        a = a_new
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 500:
            # Reduce alpha and try again
            alpha *= 0.5
            stall = 0
            if alpha < 0.01:
                break
    
    return best_local, best_score


def fista(a_init, n, max_time, rng, lr_init=1.0, top_k=1):
    """FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) for this problem."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    z = a.copy()
    t_k = 1.0
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    lr = lr_init
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(z)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = z.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            z = project_simplex(z + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)
            a = z.copy()
            t_k = 1.0
            stall = 0
        
        grad = subgradient_argmax(z, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - a)
        a = project_simplex(y - lr * grad / gnorm, S)
        z = a
        t_k = t_new
        
        # Adaptive lr
        c_a = fft_convolve_self(a)
        if c_a.max() > cur + lr * gnorm * 0.1:
            lr *= 0.5
        elif c_a.max() < cur - 0.5 * lr * gnorm:
            lr = min(lr * 1.1, lr_init * 5)
    
    return best_local, best_score

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Algorithm:
- Fast O(n) subgradient via argmax lag for projected gradient descent
- Backtracking line search for adaptive step sizes
- FISTA acceleration for faster convergence
- Mirror descent (multiplicative weights) as complementary method
- Spectral domain flattening as a novel restart/escape mechanism
- Multi-phase search: diverse initializations -> parallel optimization -> polishing
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_topk(a, c, top_k):
    """Compute subgradient of max(c) using top_k argmax lags. O(n*top_k)."""
    n = len(a)
    if top_k <= 0:
        return np.zeros(n)
    top_idx = np.argsort(c)[::-1][:top_k]
    vals = c[top_idx]
    total = vals.sum()
    if total <= 1e-15:
        return np.zeros(n)
    weights = vals / total
    grad = np.zeros(n, dtype=np.float64)
    for w_val, k in zip(weights, top_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def subgradient_single(a, c):
    """Fast O(n) subgradient using only the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        grad[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def pgd_linesearch(a_init, n, max_time, rng, lr_init=1.0, top_k=1):
    """Projected Gradient Descent with backtracking line search."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    lr = lr_init
    start = time.monotonic()
    stall = 0
    best_stall_action = 0
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            # Escalating restart
            strength = 0.01 * min(stall / 2000.0, 0.5)
            a = project_simplex(a + rng.normal(0, strength * S / np.sqrt(n), n), S)
            stall = 0
            best_stall_action += 1
        
        if top_k == 1:
            grad = subgradient_single(a, c)
        else:
            grad = subgradient_topk(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        direction = -grad / gnorm
        
        # Backtracking line search
        step = lr
        improved = False
        cur_val = cur
        for _ in range(12):
            a_trial = project_simplex(a + step * direction, S)
            c_trial = fft_convolve_self(a_trial)
            f_trial = c_trial.max()
            if f_trial < cur_val - 1e-7:
                a = a_trial
                lr = min(step * 1.5, lr_init * 20.0)
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            if lr < 1e-12:
                break
    
    return best_local, best_score


def fista_optimize(a_init, n, max_time, rng, lr_init=1.0, top_k=1):
    """FISTA (accelerated proximal gradient) for this problem."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    z = a.copy()
    t_k = 1.0
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    lr = lr_init
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(z)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = z.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            z = project_simplex(z + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)
            a = z.copy()
            t_k = 1.0
            lr = lr_init
            stall = 0
        
        if top_k == 1:
            grad = subgradient_single(z, c)
        else:
            grad = subgradient_topk(z, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        y = z + momentum * (z - a)
        a_new = project_simplex(y - lr * grad / gnorm, S)
        
        # Simple backtracking on FISTA
        c_a = fft_convolve_self(a_new)
        if c_a.max() > cur + 0.5 * lr * gnorm:
            lr *= 0.5
            a_new = project_simplex(y - lr * grad / gnorm, S)
        elif lr < lr_init * 5.0:
            lr *= 1.05
        
        a = a_new
        z = a
        t_k = t_new
    
    return best_local, best_score


def mirror_descent_opt(a_init, n, max_time, rng, eta_init=1.0, top_k=1):
    """Mirror descent / multiplicative weights."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    eta = eta_init
    start = time.monotonic()
    t = 0
    stall = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            perturb = 0.005 * min(stall / 2000.0, 0.5)
            a = a * (1.0 + perturb * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * (S / s)
            stall = 0
        
        if top_k == 1:
            grad = subgradient_single(a, c)
        else:
            grad = subgradient_topk(a, c, top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        grad_norm = grad / gmax
        
        a_new = a * np.exp(-eta * grad_norm)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new * (S / s)
        else:
            a = np.full(n, S / n)
        
        eta = eta_init / (1.0 + 0.0005 * t)
    
    return best_local, best_score


def spectral_flatten(a_init, n, max_time, alpha=1.0):
    """Spectral domain flattening: reduce peaks in |FFT(a)|^2."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    stall = 0
    cur_alpha = alpha
    
    while time.monotonic() - start < max_time:
        A = fft(a, fsize)
        mag2 = np.abs(A) ** 2
        mu = mag2.mean()
        sigma2 = mag2.var() + 1e-10
        
        # Exponentially suppress spectral peaks
        scale = np.exp(-cur_alpha * (mag2 - mu) / (2.0 * sigma2))
        new_A = np.abs(A) * scale * np.exp(1j * np.angle(A))
        new_A[0] = A[0]  # Preserve DC component (sum)
        
        a_new = np.real(ifft(new_A, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new * (S / s)
        a = a_new
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            cur_alpha *= 0.7
            stall = 0
            if cur_alpha < 0.05:
                break
    
    return best_local, best_score


def adversarial_rebalance(a_init, n, max_time, rng, top_k=20):
    """Directly reduce high-convolution lags by moving mass."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    start = time.monotonic()
    lr = S / n
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Identify the lag with maximum convolution
        k_max = int(np.argmax(c))
        
        # Positions that contribute to c[k_max]: pairs (i, k_max - i)
        # Reduce a[i] for positions where a[i] is large and contributes to peak
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        
        if lo > hi:
            break
        
        # Get the contributing segment
        segment = a[lo:hi + 1]
        paired = a[k_max - hi:k_max - lo + 1][::-1]
        
        # Find the largest product in the segment (biggest contributor)
        products = segment * paired
        max_prod_idx = int(np.argmax(products))
        source_pos = lo + max_prod_idx
        
        # Move mass from source to a random position that doesn't contribute to peak
        # Pick a position far from the peak-contributing range
        all_positions = np.arange(n)
        contributing = set(range(lo, hi + 1))
        non_contributing = [p for p in range(n) if p not in contributing and k_max - p < 0 or k_max - p >= n]
        if not non_contributing:
            non_contributing = [p for p in range(n) if p not in contributing]
        if not non_contributing:
            break
        
        target_pos = int(rng.choice(non_contributing))
        
        move_amount = a[source_pos] * 0.05
        a[source_pos] -= move_amount
        a[target_pos] += move_amount
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)
    
    return best_local, best_score
