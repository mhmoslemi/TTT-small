def _fft_len(n):
    return int(next_fast_len(max(1, n)))

def convolve_self(p):
    """Full linear convolution of p with itself, length 2n-1."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.empty(0)
    L = _fft_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]

def compute_score(p):
    """Score for a nonnegative sequence (scale-invariant)."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.inf
    s = float(p.sum())
    if s < 1e-12:
        return np.inf
    c = convolve_self(p)
    cmax = float(c.max())
    return float(2.0 * n * cmax / (s * s))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(p, 0.0)
    p = np.where(np.isfinite(p), p, 0.0)
    s = p.sum()
    if s < 1e-300:
        return np.full(p.size, 1.0 / p.size)
    return p / s


def lp_best_response(f, M, max_time=30.0):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns g (normalized to sum 1) or None.
    """
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n == 0 or M <= 0:
        return None
    
    # Build sparse matrix A where A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    
    try:
        start = time.monotonic()
        res = linprog(
            -np.ones(n),  # minimize -sum(g) = maximize sum(g)
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': max_time, 'disp': False}
        )
        elapsed = time.monotonic() - start
        if res.status not in (0, 3):  # 0=optimal, 3=time limit
            return None
        g = res.x
        if g is None or not np.all(np.isfinite(g)):
            return None
        g = np.maximum(g, 0.0)
        s = g.sum()
        if s < 1e-300:
            return None
        return g / s
    except Exception:
        return None


def make_initials(n, rng, parent=None):
    """Generate diverse initial shapes."""
    inits = []
    
    if parent is not None and parent.size == n:
        inits.append(normalize(parent))
    
    # Uniform
    inits.append(np.full(n, 1.0/n))
    
    # Linear tapers
    for a in [0.5, 1.0, 2.0, 4.0]:
        x = np.linspace(0, 1, n)
        inits.append(normalize(1 + a * x))
        inits.append(normalize(1 + a * (1 - x)))
    
    # Quadratic (cap and cup)
    for a in [0.5, 1.0, 2.0, 4.0]:
        x = np.linspace(0, 1, n)
        inits.append(normalize(np.maximum(1 - a * (2*x - 1)**2, 0.01)))
        inits.append(normalize(1 + a * (2*x - 1)**2))
    
    # Exponential
    for a in [0.5, 1.0, 2.0, 5.0, 10.0]:
        x = np.linspace(0, 1, n)
        inits.append(normalize(np.exp(-a * x)))
        inits.append(normalize(np.exp(-a * (1-x))))
        inits.append(normalize(np.exp(-a * np.abs(2*x-1))))
        inits.append(normalize(np.exp(a * np.abs(2*x-1))))
    
    # Beta-like
    for alpha, beta in [(1,1),(2,1),(1,2),(3,1),(1,3),(5,1),(1,5),(2,2),(0.5,0.5),(0.3,0.3)]:
        x = np.linspace(1e-6, 1-1e-6, n)
        inits.append(normalize(x**alpha * (1-x)**beta))
    
    # Gaussian bumps
    for mu, sig in [(0.5, 0.1), (0.5, 0.2), (0.3, 0.1), (0.7, 0.1), (0.5, 0.05)]:
        x = np.linspace(0, 1, n)
        inits.append(normalize(np.exp(-0.5 * ((x-mu)/sig)**2)))
    
    # Two bumps
    for mu1, mu2, sig in [(0.3, 0.7, 0.08), (0.2, 0.8, 0.1), (0.25, 0.75, 0.05)]:
        x = np.linspace(0, 1, n)
        inits.append(normalize(np.exp(-0.5*((x-mu1)/sig)**2) + np.exp(-0.5*((x-mu2)/sig)**2)))
    
    # Random smooth
    for _ in range(8):
        x = np.linspace(0, 1, n)
        freqs = rng.integers(1, max(2, n//100), size=5)
        amps = rng.uniform(0.1, 0.8, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        p = 1.0
        for f, a, ph in zip(freqs, amps, phases):
            p += a * np.sin(2*np.pi*f*x + ph)
        inits.append(normalize(np.maximum(p, 0.01)))
    
    # Edge-concentrated (important for this problem)
    x = np.linspace(0, 1, n)
    for a in [2, 5, 10, 20, 50]:
        inits.append(normalize(np.exp(-a * x) + np.exp(-a * (1-x))))
    
    # Very edge-concentrated
    for a in [20, 50, 100, 200]:
        inits.append(normalize(np.exp(-a * x**2) + np.exp(-a * (1-x)**2)))
    
    return inits

    best_score = compute_score(p)
    current_score = best_score
    
    end = time.monotonic() + time_limit
    t_factor = 0.15  # temperature factor (T = t_factor * cmax)
    lp_count = 0
    grad_iter = 0
    stagnation = 0

    while time.monotonic() < end:
        # Decision: LP or gradient?
        # Do LP every few hundred gradient steps, or when stuck

"""
Step-function search minimizing 2n*max(conv(p,p))/sum(p)^2.
Uses iterative LP best-response combined with smoothed-max gradient descent.
Key improvements: adaptive temperature, backtracking line search, multi-length
search, aggressive LP exploitation, and diverse edge-concentrated initializations.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

TOTAL_BUDGET = 1800.0


def _fft_len(n):
    return int(next_fast_len(max(1, n)))


def convolve_self(p):
    """Full linear convolution of p with itself, length 2n-1."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.empty(0)
    L = _fft_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]


def compute_score(p):
    """Score for a nonnegative sequence (scale-invariant)."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.inf
    s = float(p.sum())
    if s < 1e-12:
        return np.inf
    c = convolve_self(p)
    cmax = float(c.max())
    return float(2.0 * n * cmax / (s * s))


def smoothed_max_and_grad(p, t_factor):
    """
    Compute smoothed max of conv(p,p) and its gradient w.r.t. p.
    Returns (surrogate_value, gradient, convolution, cmax, T_used).
    """
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    c = convolve_self(p)
    cmax = float(c.max())
    if cmax <= 0:
        return 0.0, np.zeros(n), c, 0.0, 0.0

    T = max(t_factor * cmax, 1e-10)
    diff = (c - cmax) / T
    diff = np.clip(diff, -60, 0)
    e = np.exp(diff)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(c.argmax())] = 1.0
        s = 1.0
    else:
        w = e / s

    # Gradient: 2 * sum_k w[k] * p[k-j] for each j
    L = _fft_len(3 * n - 2)
    Fp = fft(p, L)
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))

    G = h[2 * n - 2: n - 2: -1]
    if G.size != n:
        G2 = np.zeros(n)
        for j in range(n):
            k_start = j
            k_end = min(2 * n - 1, j + n)
            G2[j] = np.dot(w[k_start:k_end], p[0:k_end - k_start])
        G = G2

    grad = 2.0 * np.maximum(G, 0.0)
    surr_val = cmax + T * np.log(max(s, 1e-300))
    return surr_val, grad, c, cmax, T


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)

    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)

    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(p, 0.0)
    p = np.where(np.isfinite(p), p, 0.0)
    s = p.sum()
    if s < 1e-300:
        return np.full(p.size, 1.0 / p.size)
    return p / s


def build_conv_matrix(f, n):
    """Build sparse matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(f, M, max_time=25.0):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns g (normalized to sum 1) or None.
    """
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n == 0 or M <= 0 or n > 4000:
        return None

    A = build_conv_matrix(f, n)

    try:
        res = linprog(
            -np.ones(n),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': max_time, 'disp': False}
        )
        if res.status not in (0, 3):
            return None
        g = res.x
        if g is None or not np.all(np.isfinite(g)):
            return None
        g = np.maximum(g, 0.0)
        s = g.sum()
        if s < 1e-300:
            return None
        return g / s
    except Exception:
        return None


def make_initials(n, rng, parent=None):
    """Generate diverse initial shapes."""
    inits = []
    x = np.linspace(0, 1, n)

    if parent is not None and parent.size == n:
        inits.append(normalize(parent))

    inits.append(np.full(n, 1.0 / n))

    # Linear tapers
    for a in [0.3, 0.7, 1.5, 3.0, 6.0]:
        inits.append(normalize(1 + a * x))
        inits.append(normalize(1 + a * (1 - x)))

    # Quadratic
    for a in [0.3, 0.8, 1.5, 3.0]:
        inits.append(normalize(np.maximum(1 - a * (2 * x - 1) ** 2, 0.01)))
        inits.append(normalize(1 + a * (2 * x - 1) ** 2))

    # Exponential decay
    for a in [0.5, 1.0, 2.0, 5.0, 15.0]:
        inits.append(normalize(np.exp(-a * x)))
        inits.append(normalize(np.exp(-a * (1 - x))))
        inits.append(normalize(np.exp(-a * np.abs(2 * x - 1))))

    # Edge-concentrated (key for this problem)
    for a in [2, 5, 10, 30, 100]:
        inits.append(normalize(np.exp(-a * x) + np.exp(-a * (1 - x))))
    for a in [5, 15, 50, 150]:
        inits.append(normalize(np.exp(-a * x**2) + np.exp(-a * (1 - x)**2)))

    # Very edge-peaked
    core = np.zeros(n)
    for w in [1, 2, 3, 5, 10]:
        core2 = np.zeros(n)
        core2[:w] = 1.0
        core2[-w:] = 1.0
        core2[w:n-w] = 0.1
        inits.append(normalize(core2))

    # Beta-like
    for alpha, beta in [(1, 1), (2, 1), (1, 2), (4, 1), (1, 4), (0.5, 0.5), (0.3, 0.3), (3, 3)]:
        xx = np.linspace(1e-6, 1 - 1e-6, n)
        inits.append(normalize(xx**alpha * (1 - xx)**beta))

    # Gaussian bumps
    for mu, sig in [(0.5, 0.05), (0.5, 0.1), (0.5, 0.2), (0.3, 0.05), (0.7, 0.05)]:
        inits.append(normalize(np.exp(-0.5 * ((x - mu) / sig) ** 2)))

    # Two bumps (edge)
    for sig in [0.03, 0.05, 0.08, 0.12]:
        inits.append(normalize(np.exp(-0.5 * (x / sig) ** 2) + np.exp(-0.5 * ((1 - x) / sig) ** 2)))

    # Random smooth
    for _ in range(10):
        freqs = rng.integers(1, max(2, n // 50), size=5)
        amps = rng.uniform(0.1, 0.7, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        p = 1.0
        for f, a, ph in zip(freqs, amps, phases):
            p += a * np.sin(2 * np.pi * f * x + ph)
        inits.append(normalize(np.maximum(p, 0.01)))

    # Chebyshev-based shapes
    z = 2 * x - 1
    for d in [2, 4, 6, 8]:
        for sign in [1, -1]:
            if d % 2 == 0:
                p = 1.0 + 0.5 * sign * _cheb(d, z)
            else:
                p = 1.0 + 0.3 * sign * _cheb(d, z)
            inits.append(normalize(np.maximum(p, 0.01)))

    return inits


def _cheb(d, z):
    """Chebyshev polynomial of degree d."""
    if d == 0:
        return np.ones_like(z)
    if d == 1:
        return z.copy()
    T_nm = np.ones_like(z)
    T_n = z.copy()
    for _ in range(2, d + 1):
        T_np = 2 * z * T_n - T_nm
        T_nm = T_n
        T_n = T_np
    return T_n

def resample_sequence(p, n_target):
    """Resample a sequence to a different length via interpolation."""
    p = np.asarray(p, dtype=np.float64)
    n_src = p.size
    if n_src == n_target:
        return normalize(p)
    x_src = np.linspace(0, 1, n_src)
    x_tgt = np.linspace(0, 1, n_target)
    p_tgt = np.interp(x_tgt, x_src, p)
    return normalize(p_tgt)
