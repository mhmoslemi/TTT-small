```python
import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fast(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def lp_step(f, M, n):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)[k] <= M for all k.
    f is the current sequence (length n), M is the peak bound.
    Returns the optimal g (length n).
    """
    # (f*g)[k] = sum_i f[k-i] * g[i] for valid i
    # For each k = 0, ..., 2n-2:
    #   sum_{i: 0<=i<n, 0<=k-i<n} f[k-i] * g[i] <= M
    # 
    # This is A @ g <= M (vector of M's, length 2n-1)
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    #
    # Maximize sum(g) = -minimize -sum(g) = -minimize c @ g where c = -1
    # Subject to: A @ g <= M*ones(2n-1), g >= 0
    
    # Build constraint matrix A (2n-1 x n)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # i ranges from max(0, k-n+1) to min(k, n-1)
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            A[k, i] = f[k - i]
    
    # Actually, this is the same as the convolution matrix.
    # A @ g = convolve(f, g) (full mode)
    # We can build A more efficiently.
    
    c_vec = -np.ones(n)  # minimize -sum(g)
    b_vec = np.full(2*n-1, M)
    
    res = linprog(c_vec, A_ub=A, b_ub=b_vec, method='highs')
    if res.success:
        return res.x
    return None
```
