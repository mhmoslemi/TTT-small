```python
    """Linear auto-convolution via FFT."""
    n = len(a)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with log-sum-exp approximation, FFT-based gradients,
   multiplicative updates, and annealed temperature.
2. L-BFGS-B polish via softmax parameterization.
3. Multi-phase search over sequence lengths (128-1024) and diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def true_objective_normalized(a):
    """Compute 2*n*max(conv(a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smooth_obj_grad(a, t):
    """
    Compute f_t(a) = (1/t)*log(sum exp(t*conv(a))) and its gradient w.r.t. a.
    Returns (objective_value, gradient_vector_of_length_n).
    """
    n = len(a)
    conv = fast_convolve(a)

    # Numerically stable log-sum-exp
    max_conv = conv.max()
    shifted = t * (conv - max_conv)
    log_sum_exp = max_conv + np.log(np.exp(shifted).sum()) / t

    # Softmax weights
    w = np.exp(shifted)
    w /= w.sum()

    # Gradient: df/da[j] = 2 * sum_m w[m] * a[m-j]
    # This is 2 * cross_correlation(a, w)[j
    N = 1 << (3 * n).bit_length()
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(w, N)
    cc = np.fft.irfft(np.conj(A_fft) * W_fft, N)
    grad = 2.0 * cc[:n]

    return log_sum_exp, grad


def softmax(x):
    """Compute softmax with numerical stability."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def multiplicative_optimize(a0, time_limit, t_start, t_end, eta_start, seed=0):
    """
    Multiplicative mirror descent on the simplex with annealed temperature.
    a0 must be non-negative; will be normalized to sum=1.
    Returns (best_a, best_true_objective).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_obj = true_objective_normalized(a)
    best_a = a.copy()
    rng = np.random.default_rng(seed)

    t = t_start
    eta = eta_start
    patience = 0
    max_patience = 50
    t0 = time.monotonic()
    current_obj = best_obj

    while time.monotonic() - t0 < time_limit:
        smooth_val, grad = smooth_obj_grad(a, t)

        # Project gradient to tangent space of simplex
        grad_proj = grad - grad.mean()

        # Multiplicative update
        step = eta * grad_proj
        step -= step.max()
        a_new = a * np.exp(-step)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new /= a_new_sum

        # Evaluate true objective
        new_obj = true_objective_normalized(a_new)

        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            current_obj = new_obj
            patience = 0
            # Make progress: slowly increase temperature
            t = min(t * 1.005, t_end)
        elif new_obj < current_obj + 1e-9:
            a = a_new
            current_obj = new_obj
            patience += 1
        else:
            # Reject: reduce step
            eta = max(eta * 0.6, 0.005)
            patience += 1

        if patience >= max_patience:
            # Restart with noise
            patience = 0
            noise_scale = 0.005 * min(1.0, 50.0 / t)
            noise = rng.normal(0, noise_scale, n)
            a = np.maximum(a + noise, 0)
            a_sum = a.sum()
            if a_sum < 1e-12:
                a = best_a.copy()
            else:
                a /= a_sum
            current_obj = true_objective_normalized(a)
            t = max(t_start, t * 0.4)
            eta = max(eta_start * 0.5, eta)
            max_patience = 35

    return best_a, best_obj


def lbfgs_polish(a0, time_limit, t_val=100.0, maxiter=200):
    """
    L-BFGS-B polish using softmax parameterization.
    a0 must be non-negative; will be normalized to sum=1.
    Returns (best_a, best_true_objective).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return a0, true_objective_normalized(a0)
    a /= s

    # Initialize x in log space
    a_safe = np.maximum(a, 1e-15)
    x0 = np.log(a_safe / a_safe.sum())

    best_obj = true_objective_normalized(a)
    best_a = a.copy()

    def obj_func(x):
        a_s = softmax(x)
        val, _ = smooth_obj_grad(a_s, t_val)
        return val, a_s

    t_start = time.monotonic()
    call_count = [0]

    def objective_and_gradient(x):
        call_count[0] += 1
        if time.monotonic() - t_start > time_limit:
            return np.inf, np.zeros_like(x)
        a_s = softmax(x)
        val, grad_a = smooth_obj_grad(a_s, t_val)
        # Transform: df/dx_j = a_j * (grad[j] - dot(grad, a))
        grad_x = a_s * (grad_a - np.dot(grad_a, a_s))
        return val, grad_x

    try:
        result = scipy_minimize(objective_and_gradient, x0, method='L-BFGS-B',
                                jac=True, options={'maxiter': maxiter,
                                                   'maxfun': maxiter * 2,
                                                   'ftol': 1e-12, 'gtol': 1e-8})
        if result.x is not None:
            a_opt = softmax(result.x)
            obj_opt = true_objective_normalized(a_opt)
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_a = a_opt
    except Exception:
        pass

    return best_a, best_obj


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Cosine modulations
    for alpha in [0.3, 0.5, 0.7, 0.9]:
        for mode in [1, 2]:
            c = 1.0 + alpha * np.cos(2 * np.pi * mode * x / n)
            c = np.maximum(c, 0)
            if c.sum() > 1e-10:
                inits.append((f'cos_m{mode}_a{alpha}', c / c.sum()))

    # Gaussian bumps
    for sf in [0.1, 0.2, 0.35]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        inits.append((f'gauss_{sf}', g / g.sum()))

    # Two Gaussian bumps
    for sf in [0.1, 0.2]:
        g1 = np.exp(-0.5 * ((x - n * 0.25) / (sf * n)) ** 2)
        g2 = np.exp(-0.5 * ((x - n * 0.75) / (sf * n)) ** 2)
        g = g1 + g2
        inits.append(('bimodal_gauss', g / g.sum()))

    # Random Dirichlet
    for conc in [0.2, 0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * conc)
        inits.append((f'dirichlet_{conc}', r))

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(('ramp', ramp / ramp.sum()))

    # From parent
    if parent is not None:
        if parent.size == n:
            pp = np.maximum(parent, 0)
            if pp.sum() > 1e-10:
                inits.append(('parent', pp / pp.sum()))
        elif parent.size > n:
            idx = np.linspace(0, parent.size - 1, n).astype(int)
            pp = np.maximum(parent[idx], 0)
            if pp.sum() > 1e-10:
                inits.append(('parent_ds', pp / pp.sum()))
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits.append(('parent_us', pp / pp.sum()))

    return inits
```
