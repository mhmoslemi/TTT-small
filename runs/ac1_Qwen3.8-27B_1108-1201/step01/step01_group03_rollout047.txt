"""
Hybrid gradient + LP optimization for minimizing autocorrelation peak of
non-negative step functions.

Core idea: The score 2*n*max(conv(f,f))/(sum f)^2 is minimized by sequences
whose self-convolution has a low peak relative to its sum. We use:
1. Fast gradient-based 'peak shaving' (O(n) per step) as the main workhorse
2. LP refinement for global direction finding
3. Multiple structured initializations at various lengths
4. Momentum and adaptive step sizes in gradient phase
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute score = 2*n*max(conv(f,f))/(sum f)^2 efficiently."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def convolve_peak_info(f):
    """Return (conv, peak_val, peak_idx) for f*f."""
    f = np.asarray(f, dtype=np.float64)
    conv = np.convolve(f, f)
    peak_idx = int(np.argmax(conv))
    peak_val = float(conv[peak_idx])
    return conv, peak_val, peak_idx


def gradient_step(f, lr, top_k=1):
    """
    One projected gradient step to reduce the peak of f*f.
    Moves f in the direction that reduces the top convolution peaks.
    Returns updated f (normalized to sum=1).
    """
    n = len(f)
    conv, peak_val, peak_idx = convolve_peak_info(f)
    
    # Compute gradient contribution from top peaks (soft max)
    # grad[j] = sum over active peaks k of 2*f[k-j] / conv[k]
    if peak_val < 1e-30:
        return f
    
    # Use top_k peaks for a smoother gradient
    top_indices = np.argpartition(conv, -top_k)[-top_k:]
    weights = np.ones(top_k)
    # Weight by how close each peak is to the max
    peak_vals = conv[top_indices]
    max_pv = peak_vals.max()
    weights = peak_vals / max_pv  # normalize so max peak gets weight 1
    
    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        k = int(k)
        # d conv[k] / d f[j] = 2*f[k-j] if 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            j_range = np.arange(lo, hi)
            contrib = 2.0 * f[k - j_range] * w
            grad[lo:hi] += contrib
    
    # Projected gradient: subtract mean to maintain sum=1 (approximately)
    grad_mean = grad.mean()
    grad -= grad_mean
    
    # Step with clipping
    f_new = f - lr * grad
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f
    return f_new / s


def gradient_search(f, budget_s, lr_init=0.01, lr_min=0.0001, lr_decay=0.9999,
                    top_k=1, momentum_beta=0.9, stall_patience=200):
    """
    Run gradient-based peak shaving until budget exhausted.
    Returns best f found and its score.
    """
    f = np.asarray(f, dtype=np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = f / max(s, 1e-15)
    else:
        f = f / s
    
    lr = lr_init
    best = f.copy()
    best_sc = fast_score(f)
    momentum = np.zeros_like(f)
    stall = 0
    it = 0
    
    while time.monotonic() < budget_s:
        it += 1
        f_new = gradient_step(f, lr, top_k=top_k)
        
        # Momentum
        momentum = momentum_beta * momentum + (1 - momentum_beta) * (f_new - f)
        f_mom = f + momentum
        f_mom = np.maximum(f_mom, 0.0)
        sm = f_mom.sum()
        if sm > 1e-15:
            f_mom = f_mom / sm
            f = f_mom
        else:
            f = f_new
        
        # Adaptive learning rate
        lr = max(lr_min, lr * lr_decay)
        
        if it % 50 == 0:
            sc = fast_score(f)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
            
            if stall > stall_patience:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                perturb_idx = rng.integers(0, len(f))
                f[perturb_idx] *= rng.uniform(0.5, 2.0)
                f = np.maximum(f, 0)
                fs = f.sum()
                if fs > 1e-15:
                    f = f / fs
                momentum = np.zeros_like(f)
                lr = lr_init * 0.5
                stall = 0
    
    return best, best_sc


def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) with A[k,j] = f[k-j] for valid k-j."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_for_g(f, M, time_limit_s=5.0):
    """Solve max sum(g) s.t. A*f*g <= M, g >= 0."""
    n = int(len(f))
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine(f, budget_s, t_init=0.3, t_decay=0.99, lp_time=3.0):
    """Run LP-based refinement until budget exhausted."""
    f = np.asarray(f, dtype=np.float64).copy()
    f = f / f.sum()
    
    best = f.copy()
    best_sc = fast_score(f)
    t = t_init
    stall = 0
    
    while time.monotonic() < budget_s:
        conv, M, _ = convolve_peak_info(f)
        if M < 1e-15:
            break
        
        g = solve_lp_for_g(f, M, lp_time)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        if np.linalg.norm(g - f) < 1e-10:
            break
        
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f = f_new / s
        t = max(0.05, t * t_decay)
        
        sc = fast_score(f)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = f.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    
    return best, best_sc


def make_initializations(n, seed=42):
    """Generate several structured initializations of length n."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # 1. Raised cosine (Hann window)
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(x))
    a = np.maximum(a, 0.01)
    inits.append(('hann', a / a.sum()))
    
    # 2. Inverse Hann (high at edges, low in middle)
    a = 0.5 * (1 - np.cos(x))
    a = np.maximum(a, 0.01)
    inits.append(('inv_hann', a / a.sum()))
    
    # 3. Uniform + noise
    a = np.ones(n) + rng.normal(0, 0.3, n)
    a = np.maximum(a, 0.01)
    inits.append(('pert_uniform', a / a.sum()))
    
    # 4. Two Gaussian bumps
    s = max(1, n // 8)
    x_arr = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x_arr - n * 0.2) / s) ** 2) + \
        np.exp(-0.5 * ((x_arr - n * 0.8) / s) ** 2)
    a = np.maximum(a, 0.01)
    inits.append(('two_bumps', a / a.sum()))
    
    # 5. Symmetric peak (high in middle)
    dist = np.abs(x_arr - n / 2) / (n / 2)
    a = np.exp(-2 * dist ** 2)
    a = np.maximum(a, 0.01)
    inits.append(('center_peak', a / a.sum()))
    
    # 6. Linear ramp up then down (triangle)
    if n % 2 == 0:
        a = np.concatenate([np.linspace(0.1, 1, n // 2), np.linspace(1, 0.1, n // 2)])
    else:
        a = np.concatenate([np.linspace(0.1, 1, n // 2 + 1), np.linspace(1, 0.1, n // 2)])
    inits.append(('triangle', a / a.sum()))
    
    # 7. Sine-based (sin^2)
    a = np.sin(np.pi * x_arr / n) ** 2
    a = np.maximum(a, 0.01)
    inits.append(('sin2', a / a.sum()))
    
    # 8. Pure cosine taper
    a = np.cos(np.pi * (x_arr - n/2 + 0.5) / n) ** 2
    a = np.maximum(a, 0.01)
    inits.append(('cos_taper', a / a.sum()))
    
    return inits


def hybrid_search(f_init, budget_s, use_lp=True):
    """
    Hybrid search: gradient first, then LP if budget allows.

    """

    if np.isfinite(sc_grad) and sc_grad < best_sc:

    if use_lp and time.monotonic() < time.monotonic() + budget_s * 0.9:

        lp_end = time.monotonic() + lp_budget

def try_length_from_parent(parent, target_n, method='interpolate'):
    """Resize parent sequence to target_n."""
    if len(parent) == target_n:
        return parent.copy()
    if method == 'interpolate':
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, target_n)
        resized = np.interp(x_new, x_old, parent)
        return np.maximum(resized, 0.0)
    elif method == 'tile':
        reps = int(np.ceil(target_n / len(parent)))
        tiled = np.tile(parent, reps)[:target_n]
        return tiled
    return parent[:target_n] if len(parent) > target_n else np.pad(parent, (0, target_n - len(parent)))

            f_init = np.maximum(np.asarray(f_init, dtype=np.float64), 0)
            s_init = f_init.sum()
            if s_init < 1e-10:
                continue
            f_init = f_init / s_init
            
            sc_init = fast_score(f_init)
            if not np.isfinite(sc_init):
                continue
            
            # Run gradient search
            end = min(deadline, time.monotonic() + per_init_budget)
            f_res, sc_res = gradient_search(f_init, end, lr_init=0.005, top_k=2)
            
            if np.isfinite(sc_res) and sc_res < local_best_sc:
                local_best_sc = sc_res
                local_best = f_res.copy()
        
        if local_best is not None and local_best_sc < bs:
            update_best(local_best, local_best_sc)
        elif local_best is not None:
            print("  n=%d: best from gradient = %.6f (not improving global)" % (n, local_best_sc), flush=True)
    
    # Phase 2: LP refinement on best candidates
    print("\n=== Phase 2: LP refinement ===", flush=True)
    # Focus LP on the current best and a few good candidates
    n_best = len(best_global)
    lp_budget_per = max(30.0, (budget_s * 0.4) / 3)
    
    # LP on current best
    lp_end = min(deadline, time.monotonic() + lp_budget_per)
    f_lp, sc_lp = lp_refine(best_global, lp_end)
    if np.isfinite(sc_lp) and sc_lp < bs:
        update_best(f_lp, sc_lp)
    
    # LP on parent-resized to best length (if different)
    if parent_n == n_best:
        parent_resized = parent.copy()
    else:
        parent_resized = try_length_from_parent(parent, n_best)
    parent_resized = np.maximum(parent_resized, 0)
    ps = parent_resized.sum()
    if ps > 1e-10:
        parent_resized = parent_resized / ps
        lp_end2 = min(deadline, time.monotonic() + lp_budget_per * 0.7)
        if time.monotonic() < deadline:
            f_lp2, sc_lp2 = lp_refine(parent_resized, lp_end2)
            if np.isfinite(sc_lp2) and sc_lp2 < bs:
                update_best(f_lp2, sc_lp2)
    
    # Phase 3: Intensive gradient polishing with different hyperparameters
    print("\n=== Phase 3: Intensive gradient polishing ===", flush=True)
    n_polish = len(best_global)
    
    # Try different top_k and lr settings
    for top_k in [1, 2, 4, 8]:
        if time.monotonic() > deadline:
            break
        polish_budget = min(30.0, max(5.0, (deadline - time.monotonic()) * 0.2))
        f_p, sc_p = gradient_search(best_global, polish_budget, 
                                     lr_init=0.003 / top_k, top_k=top_k,
                                     lr_decay=0.99995, stall_patience=100)
        if np.isfinite(sc_p) and sc_p < bs:
            update_best(f_p, sc_p)
    
    # Phase 4: Try longer sequences (more "attack surface")
    print("\n=== Phase 4: Long sequence exploration ===", flush=True)
    for n_large in [max(n_best * 2, 2048), max(n_best * 4, 4096), 8192]:
        if time.monotonic() > deadline:
            break
        if n_large > 16384:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        # Create initial from best via interpolation
        f_init = try_length_from_parent(best_global, n_large)
        f_init = np.maximum(f_init, 0)
        fs = f_init.sum()
        if fs > 1e-10:
            f_init = f_init / fs
        
        # Also try a fresh structured init
        inits_large = make_initializations(n_large, seed=seed + 999 + n_large)
        
        grad_budget = min(remaining * 0.6, 120.0)
        per_init = grad_budget / (len(inits_large) + 1)
        
        local_best_sc = np.inf
        local_best = None
        
        # Gradient on interpolated parent
        end = min(deadline, time.monotonic() + per_init)
        f_r, sc_r = gradient_search(f_init, end, lr_init=0.003, top_k=2)
        if np.isfinite(sc_r) and sc_r < local_best_sc:
            local_best_sc = sc_r
            local_best = f_r.copy()
        
        # Gradient on structured inits
        for name, fi in inits_large[:3]:  # Limit to 3 to save time
            if time.monotonic() > deadline:
                break
            fi = np.maximum(fi, 0)
            fs2 = fi.sum()
            if fs2 < 1e-10:
                continue
            fi = fi / fs2
            end = min(deadline, time.monotonic() + per_init)
            f_r, sc_r = gradient_search(fi, end, lr_init=0.003, top_k=2)
            if np.isfinite(sc_r) and sc_r < local_best_sc:
                local_best_sc = sc_r
                local_best = f_r.copy()
        
        if local_best is not None and local_best_sc < bs:
            update_best(local_best, local_best_sc)
        elif local_best is not None:
            print("  n=%d: best = %.6f" % (n_large, local_best_sc), flush=True)
    
    # Phase 5: Final LP + gradient cycle on the global best
    print("\n=== Phase 5: Final refinement ===", flush=True)
    if time.monotonic() < deadline:
        # Brief LP
        lp_time_final = min(60.0, (deadline - time.monotonic()) * 0.5)
        if lp_time_final > 5:
            f_f, sc_f = lp_refine(best_global, lp_time_final, t_init=0.1, lp_time=2.0)
            if np.isfinite(sc_f) and sc_f < bs:
                update_best(f_f, sc_f)
        
        # Final gradient polish with very small steps
        if time.monotonic() < deadline:
            final_budget = deadline - time.monotonic()
            if final_budget > 2:
                f_ff, sc_ff = gradient_search(best_global, final_budget,
                                              lr_init=0.001, lr_min=0.00001,
                                              top_k=4, lr_decay=0.99999,
                                              stall_patience=50)
                if np.isfinite(sc_ff) and sc_ff < bs:
                    update_best(f_ff, sc_ff)
    
    # Final validation
    best_global = np.clip(np.asarray(best_global, dtype=np.float64), 0.0, 1000.0)
    if best_global.size == 0 or not np.isfinite(best_global).all() or best_global.sum() < 0.01:
        best_global = np.ones(256)
    
    # Verify with actual evaluator
    final_score = float(evaluate_sequence(best_global.tolist()))
    if not np.isfinite(final_score):
        # Fallback
        best_global = np.ones(256)
        final_score = float(evaluate_sequence(best_global.tolist()))
    
    print("\nFINAL: score = %.6f, n = %d" % (final_score, best_global.size), flush=True)
    
    return [float(x) for x in best_global]
