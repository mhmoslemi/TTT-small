def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    """Normalize to sum = target (default sqrt(2n))."""
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-15:
        return np.full(n, target / n)
    return a * (target / S)

def subgradient(a, m_star, top_k=1, tau=0):
    """Compute (sub)gradient of max conv at a, for peak position m_star.
    If top_k > 1 and tau > 0, uses soft combination."""
    n = len(a)
    if top_k == 1:
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        grad = np.zeros(n)
        grad[lo:hi+1] = 2.0 * a[m_star-hi:m_star-lo][::-1]

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Structured initializations (window functions, previous best)

import numpy as np

from scipy.optimize import linprog

def _compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def _normalize(a, target=None):
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-15:
        return np.full(n, target / n)
    return a * (target / S)

def _analytical_subgrad(a, m_star):
    """Compute subgradient of conv[m_star] w.r.t. a."""
    n = len(a)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[m_star - idx]
    return grad

def _soft_subgrad(a, top_k=5, tau=10.0):
    """Compute soft subgradient using top-k peaks."""
    n = len(a)
    c = np.convolve(a, a)
    # Get top-k indices
    top_idx = np.argsort(c)[-top_k:]
    top_vals = c[top_idx]
    # Softmax weights
    w = np.exp(tau * (top_vals - top_vals.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, m in zip(w, top_idx):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        idx = np.arange(lo, hi + 1)
        grad[idx] += wi * 2.0 * a[m - idx]
    return grad, c

def _grad_step_fast(a, lr, v, momentum, top_k, tau):
    """One accelerated projected gradient step."""
    n = len(a)
    target = np.sqrt(2.0 * n)
    a = _normalize(a, target)
    
    grad, _ = _soft_subgrad(a, top_k, tau)
    grad -= grad.mean()  # Project onto sum=const
    
    v = momentum * v - lr * grad
    a = a + v
    a = np.maximum(a, 0.0)
    a = _normalize(a, target)
    return a, v

def _optimize_grad(a0, deadline, max_iters=10000, top_k=1, tau=0.0):
    """Accelerated projected gradient descent."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    v = np.zeros(n)
    momentum = 0.9
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        # Learning rate: start large, decay
        lr = 1.0 / (1.0 + 0.01 * it)
        
        a, v = _grad_step_fast(a, lr, v, momentum, top_k, tau)
        
        if (it + 1) % 500 == 0:

            if s < best_score:
                best_score = s
                best = a.copy()
            # Periodic momentum reset

                v = np.zeros(n)
    
    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()
    
    return best, best_score

def _lp_refine(a, deadline, max_iters=100):
    """LP-based refinement: fix a, find g that maximizes sum(g) s.t. conv(a,g) <= M."""
    n = len(a)
    if n > 1500:
        return a, _compute_obj(a)
    
    a = _normalize(a)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    target = np.sqrt(2.0 * n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        
        # Solve LP: max sum(g) s.t. conv(a, g)[m] <= M for all m, g >= 0
        c_obj = -np.ones(n)
        A = np.zeros((2 * n - 1, n))
        for m in range(2 * n - 1):
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                A[m, j] = a[m - j]
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break
        
        if g is None or g.sum() < 1e-10:
            break
        
        # Try blending
        improved = False
        for t in [0.7, 0.4, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, target)
            s_new = _compute_obj(a_new)
            if s_new < best_score:

                best = a.copy()
                best_score = s_new

            a = a - lr * grad

def _coordinate_descent(a, deadline, max_sweeps=50):

    n = len(a)

    target = _normalize(a, None).sum()

    # i.e., m in [k, k+n-1] intersected with [0, 2n-2]
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline - 5.0:
            break
        
        # Random order
        order = np.random.permutation(n)
        
        for k in order:
            if time.monotonic() > deadline - 5.0:
                break
            
            # Find the optimal value of a[k] given all other a fixed
            # conv[m] as a function of a[k]:
            # Let b = a with a[k] = 0. Then conv[m](a) = conv[m](b) + contribution from a[k].
            # Contribution: a[k]*a[m-k] + a[m-k]*a[k] (if m-k ≠ k) or a[k]^2 (if m-k = k)
            # So conv[m] = base[m] + 2*a[k]*a[m-k] (if m≠2k) or base[m] + a[k]^2 (if m=2k)
            
            # For efficiency, compute base convolution (with a[k]=0)
            # Then find a[k] that minimizes max_m (base[m] + linear/quadratic term)

            # Simpler: use a small grid + interpolation
            # Current value: a[k]

            old_val = a[k]

            # If we change a[k] to v:
            # New conv[m] = old_conv[m] - old_contribution + new_contribution
            # old_contribution = terms involving a[k] = old_val
            # new_contribution = terms involving v

            # So the change in conv[m] is:

            # Current conv values (we need these)

            # For n=512, that's about 1.3M operations per sweep. Should be fast enough.
            
            best_v = old_val
