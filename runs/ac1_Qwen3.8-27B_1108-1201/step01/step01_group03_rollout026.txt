"""
Multi-strategy optimization for minimizing the autocorrelation peak ratio.
Combines LP-based refinement, adaptive step size, direct peak-shaving,
and simulated annealing for diversification.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute the score efficiently."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))


def build_conv_matrix(n, f):
    """Build sparse convolution matrix A where A[k,j] = f[k-j] if valid."""
    rows = []
    cols = []
    data = []
    for j in range(n):
        start = max(0, 0)  # k - j >= 0 => k >= j
        end = min(2*n-1, n-1+j)  # k - j <= n-1 => k <= n-1+j
        # Actually: A[k,j] = f[k-j] when 0 <= k-j <= n-1
        # So for fixed j, k ranges from j to j+n-1
        for k in range(j, j+n):
            rows.append(k)
            cols.append(j)
            data.append(f[k-j])
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))


def build_conv_matrix(n, f):
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2*n-1, n)).tocsr()
    return A


def solve_lp(f, M, time_limit=5.0):
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n

    except:
        pass
    return None

        return f, fast_score(f)

        return f, fast_score(f)

    if gs < 1e-10:
        return f, fast_score(f)
    
    g_norm = g / gs  # normalized to sum 1

def peak_shave(f, iterations=100, step=0.01):
    """Direct peak shaving: reduce contributions to the peak lag."""
    f = f.copy()
    n = len(f)
    best_sc = fast_score(f)
    
    for it in range(iterations):
        conv = np.convolve(f, f)
        k_star = int(np.argmax(conv))
        M = conv[k_star]
        
        # Find which pairs contribute to the peak
        j_min = max(0, k_star - (n-1))
        j_max = min(n-1, k_star)
        
        # For each j in [j_min, j_max], the contribution is f[j]*f[k_star-j]
        # We want to reduce these products
        # Strategy: find the pair with the largest product and reduce one of them
        j_vals = np.arange(j_min, j_max+1)
        products = f[j_vals] * f[k_star - j_vals]
        
        if products.max() < 1e-15:
            break
        
        # Find the j with the largest contribution
        j_best = j_vals[np.argmax(products)]
        
        # Transfer mass from f[j_best] to a position that doesn't contribute to the peak
        # The peak is at lag k_star, so f[j_best] contributes via f[j_best]*f[k_star-j_best]
        # If we reduce f[j_best], we should add mass to a position that's far from k_star-j_best
        # A safe position is one where both positions (pos and k_star-pos) have small values
        
        # Simple strategy: transfer to a random position far from j_best and k_star-j_best
        other = k_star - j_best
        # Find position with minimum f value that's not j_best or other
        candidates = np.arange(n)
        candidates = candidates[(candidates != j_best) & (candidates != other)]
        if len(candidates) == 0:
            break
        # Prefer positions where the "pair" position for the peak also has small value
        pair_vals = np.minimum(f[candidates], f[np.clip(k_star - candidates, 0, n-1)])
        target = candidates[np.argmin(pair_vals)]
        
        # Transfer a small amount
        delta = step * f[j_best]
        f[j_best] -= delta
        f[target] += delta
        f = np.maximum(f, 0)
        
        s = f.sum()
        if s < 1e-10:
            break
        f = f / s
        
        sc = fast_score(f)
        if sc < best_sc - 1e-12:
            best_sc = sc
    
    return f, best_sc

Multi-strategy optimization to minimize 2*n*max(conv(f,f))/sum(f)^2.
Combines adaptive LP refinement, direct peak-shaving, spectral initialization,
and simulated annealing-style diversification to escape local minima.
"""
import numpy as np
import scipy.sparse as sp

import time


def fast_score(f):
    """Compute score = 2*n*max(conv)/sum^2 efficiently."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    s = f.sum()
    if s < 0.01:

    conv = np.convolve(f, f)

def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) with A[k,j] = f[k-j] for valid k-j."""

        if it % 5 == 0:
            sc = fast_score(f)
            if sc < best_sc:
                best_sc = sc
                best_f = f.copy()
    
    sc = fast_score(f)
    if sc < best_sc:
        best_sc = sc
        best_f = f.copy()
    return best_f, best_sc


def make_initial(n, rng, method='uniform'):
    """Create a normalized non-negative initial vector."""
    if method == 'uniform':
        a = np.ones(n) + rng.normal(0, 0.4, n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x)) + rng.normal(0, 0.05, n)
    elif method == 'sinc':
        x = np.arange(n) - (n - 1) / 2.0
        a = np.sinc(x / (n / 5.0)) + 0.02
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        a = (np.exp(-0.5 * ((x - n/4) / (n/12))**2) +
             np.exp(-0.5 * ((x - 3*n/4) / (n/12))**2))
    elif method == 'plateau':
        a = np.ones(n)
        a[:n//5] = 0.2
        a[4*n//5:] = 0.2
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'alternating':
        a = np.ones(n)
        a[1::2] = 0.3
    elif method == 'spectral':
        # Use inverse FFT of a shaped power spectrum
        spec = np.random.exponential(1.0, n//2 + 1)
        spec[0] = 0  # zero DC for zero-mean
        inv = np.fft.irfft(spec * spec, n=n)
        a = np.abs(inv) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def lp_refine(f, deadline, multi_M=True, lp_time=3.0):
    """Iterative LP refinement until deadline."""
    best_f = f.copy()
    best_sc = fast_score(f)
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        t_cands = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5] if it < 50 else [0.03, 0.05, 0.1, 0.15, 0.2, 0.3]
        f_new, sc = adaptive_lp_step(f, t_candidates=t_cands, lp_time=lp_time,
                                      multi_M=multi_M if it < 100 else False)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f_new.copy()
            stall = 0
        else:
            f = f_new
            stall += 1
        f = f_new
        if stall > 40:
            break
    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Main search: find best non-negative sequence minimizing the score."""
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)
    
    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    except Exception:
        parent = np.ones(512)
    if parent.ndim != 1 or parent.size == 0 or parent.sum() < 0.01:
        parent = np.ones(512)
    parent = parent / parent.sum()
    
    bs = fast_score(parent)
    best = parent.copy()
    print(f"Baseline: score={bs:.6f}, n={len(parent)}", flush=True)
    
    # Phase 1: LP refinement starting from parent
    phase1_deadline = min(total_deadline, time.monotonic() + 400)
    print("Phase 1: LP refinement from parent", flush=True)
    f = parent.copy()
    while time.monotonic() < phase1_deadline:
        f, sc = lp_refine(f, min(total_deadline, time.monotonic() + 30),
                          multi_M=True, lp_time=3.0)
        if sc < bs - 1e-10:
            bs = sc
            best = f.copy()
            print(f"  P1: {bs:.6f} (n={len(f)})", flush=True)
        elif sc >= bs - 1e-6:
            # Try peak shaving as perturbation
            f_ps, sc_ps = peak_shave(f, iterations=100, step=0.015)
            if sc_ps < bs:
                bs = sc_ps
                best = f_ps.copy()
                f = f_ps
                print(f"  P1 shave: {bs:.6f}", flush=True)
            else:
                break
        else:
            f = f_ps if sc_ps < bs else f
            f_ps, sc_ps = peak_shave(f, iterations=150, step=0.02)
            if sc_ps < bs:
                bs = sc_ps
                best = f_ps.copy()
                f = f_ps
                print(f"  P1 shave: {bs:.6f}", flush=True)
            else:
                break
    
    # Phase 2: Try different lengths and initializations
    print("Phase 2: Multi-length exploration", flush=True)
    methods = ['uniform', 'cosine', 'sinc', 'bimodal', 'plateau', 'random', 'alternating']
    lengths = [256, 384, 512, 640, 768, 1024]
    
    #剩余时间分配
    remaining = total_deadline - time.monotonic()
    combos = [(n, m) for n in lengths for m in methods]
    budget_per = max(8.0, remaining / (len(combos) + 5))
    
    for (n, m) in combos:
        if time.monotonic() > total_deadline - 200:
            break
        start = make_initial(n, rng, m)
        local_dl = min(total_deadline - 200, time.monotonic() + budget_per)
        
        f_c, sc_c = lp_refine(start, local_dl, multi_M=(budget_per > 20), lp_time=2.5)
        if sc_c < bs:
            bs = sc_c
            best = f_c.copy()
            print(f"  P2 n={n} {m}: {bs:.6f}", flush=True)
        
        # Quick shave
        if time.monotonic() < local_dl and sc_c < bs + 0.05:
            f_ps, sc_ps = peak_shave(f_c, iterations=150, step=0.02)
            if sc_ps < bs:
                bs = sc_ps
                best = f_ps.copy()
                print(f"  P2 n={n} {m} shave: {bs:.6f}", flush=True)
    
    # Phase 3: Extended LP on best found so far with smaller steps
    print("Phase 3: Fine LP polish", flush=True)
    f = best.copy()
    stall = 0
    while time.monotonic() < total_deadline - 60:
        t_cands = [0.02, 0.03, 0.05, 0.07, 0.1, 0.15]
        f_new, sc = adaptive_lp_step(f, t_candidates=t_cands, lp_time=3.0, multi_M=False)
        if sc < bs - 1e-12:
            bs = sc
            best = f_new.copy()
            f = f_new
            stall = 0
            if bs < 1.51:
                print(f"  P3: {bs:.6f}", flush=True)
        else:
            f = f_new
            stall += 1
            if stall > 30:
                # Diversify with peak shave
                f_ps, sc_ps = peak_shave(f, iterations=200, step=0.01)
                if sc_ps < bs:
                    bs = sc_ps
                    best = f_ps.copy()
                    f = f_ps
                    print(f"  P3 shave: {bs:.6f}", flush=True)
                    stall = 0
                else:
                    break
    
    # Phase 4: Final peak shaving
    print("Phase 4: Final peak shaving", flush=True)
    f_final, sc_final = peak_shave(best, iterations=500, step=0.005)
    if sc_final < bs:
        bs = sc_final
        best = f_final.copy()
        print(f"  P4: {bs:.6f}", flush=True)
    
    # Verify with authoritative evaluator
    final_sc = float(evaluate_sequence(best.tolist()))
    if np.isfinite(final_sc):
        bs = min(bs, final_sc)
    
    # Ensure validity
    best = np.clip(best, 0.0, 1000.0)
    if best.size == 0 or best.sum() < 0.01:
        best = np.ones(256)
    
    print(f"FINAL: score={bs:.6f}, n={len(best)}", flush=True)
    return [float(x) for x in best]
