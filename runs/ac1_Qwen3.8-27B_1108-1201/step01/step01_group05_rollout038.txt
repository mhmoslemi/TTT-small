def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    v[~np.isfinite(v)] = 0.0
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (css - 1.0)
    if not cond.any():
        return np.full(n, 1.0 / n)
    k = np.count_nonzero(cond) - 1
    theta = (css[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s

def lp_best_response(p, time_limit=30.0):
    """
    Solve LP: max sum(g) s.t. p*g[k] <= M for all k, g >= 0.
    Returns normalized g or None.
    """
    if not _HAVE_LP:
        return None
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n < 2 or n > 1200:
        return None

    c = conv_full(p)
    M = c.max()
    if M <= 0:
        return None

    # Build constraint matrix A where A[k, j] = p[k-j] if 0 <= k-j < n else 0
    # A has shape (2n-1, n)
    k_vals = np.arange(2 * n - 1)
    j_vals = np.arange(n)
    k_col = k_vals[:, None]
    j_row = j_vals[None, :]
    valid = (k_col >= j_row) & (k_col < j_row + n)
    
    # Build sparse matrix
    rows = np.repeat(k_vals, n, dtype=np.int32)
    cols = np.tile(j_vals, 2 * n - 1, dtype=np.int32)
    mask = valid.ravel()
    rows = rows[mask]
    cols = cols[mask]
    # A[k, j] = p[k-j] when valid
    data = p[(k_col - j_row)[valid].ravel()].astype(np.float64)
    
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': time_limit})
    except Exception:
        return None

    if res is None or not res.success:
        return None

    g = res.x
    if g is None:
        return None
    g = np.maximum(g, 0)
    s = g.sum()
    if s < 1e-12:
        return None
    return g / s


def try_blend(p, g, time_budget_remaining):
    """Try blending p with g at various fractions, return best."""
    best_p = p
    best_s = score(p)
    for t in (1.0, 0.7, 0.5, 0.35, 0.2, 0.1):
        q = project_simplex((1 - t) * p + t * g)
        sc = score(q)
        if sc < best_s:
            best_s = sc
            best_p = q
    return best_p, best_s

            if gnorm > 1e-10:
                # Try several step sizes
                best_step = None
                best_step_score = score(p)
                for lr in (0.5 / gnorm, 0.2 / gnorm, 0.05 / gnorm, 0.01 / gnorm, 0.002 / gnorm):
                    lr = min(lr, 5.0)
                    q = project_simplex(p - lr * gvec)
                    sc = score(q)
                    if sc < best_step_score:
                        best_step_score = sc
                        best_step = q
                
                if best_step is not None:
                    if best_step_score < best_score:
                        best_score = best_step_score
                        best_p = best_step.copy()
                    p = best_step
                else:
                    # Small random perturbation
                    sigma = 0.5 / n
                    q = project_simplex(p + rng.normal(0, sigma, n))
                    sc = score(q)
                    if sc < best_score:
                        best_score = sc
                        best_p = q.copy()
                    # Always move a little to avoid getting stuck
                    p = project_simplex(0.9 * p + 0.1 * q)
        
        # Periodic restart from best
        if lp_iters > 0 and lp_iters % 50 == 0:
            p = normalize(0.7 * best_p + 0.3 * project_simplex(
                best_p + rng.normal(0, 1.0 / n, n)))
    
    # Phase 2: Final gradient polish
    remaining = end - time.monotonic()
    if remaining > 2:
        temp = 0.02
        for _ in range(min(500, int(remaining * 100))):
            if time.monotonic() > end - 1:
                break
            gvec, c, cmax, surr = smoothed_max_grad(p, temp)
            if not np.all(np.isfinite(gvec)) or np.linalg.norm(gvec) < 1e-12:
                break
            gnorm = np.linalg.norm(gvec)
            for lr in (0.3 / gnorm, 0.1 / gnorm, 0.02 / gnorm, 0.005 / gnorm):
                lr = min(lr, 3.0)
                q = project_simplex(p - lr * gvec)
                sc = score(q)
                if sc < score(p):
                    p = q
                    if sc < best_score:
                        best_score = sc
                        best_p = q.copy()
                    break
            temperature = 0.01
    
    return best_p, best_score


def make_analytic(n, kind, params):
    """Generate analytic shape."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        p = np.ones(n)
    elif kind == 'linear':
        a = params.get('a', 1.0)
        p = 1.0 + a * x
    elif kind == 'linear_rev':
        a = params.get('a', 1.0)
        p = 1.0 + a * (1 - x)
    elif kind == 'power':
        a = params.get('a', 1.0)
        p = np.power(x + 1e-8, a)
    elif kind == 'power_rev':
        a = params.get('a', 1.0)
        p = np.power(1 - x + 1e-8, a)
    elif kind == 'exponential':
        a = params.get('a', 2.0)
        p = np.exp(a * x)
    elif kind == 'exponential_rev':
        a = params.get('a', 2.0)
        p = np.exp(a * (1 - x))
    elif kind == 'gaussian':
        mu = params.get('mu', 0.3)
        sig = params.get('sig', 0.15)
        p = np.exp(-((x - mu) ** 2) / (2 * sig ** 2))
    elif kind == 'skew_gauss':
        a = params.get('a', 3.0)
        mu = params.get('mu', 0.25)
        b = params.get('b', 1.5)
        p = np.exp(-a * (x - mu) ** 2 + b * (x - mu))
    elif kind == 'beta':
        alpha = params.get('alpha', 1.0)
        beta = params.get('beta', 3.0)
        p = np.power(x + 1e-8, alpha) * np.power(1 - x + 1e-8, beta)
    elif kind == 'log_cosh':
        a = params.get('a', 2.0)
        mu = params.get('mu', 0.5)
        p = 1.0 / np.cosh(a * (x - mu))
    elif kind == 'saturate':
        # p(i) = 1 / (1 + exp(-a*(x-b)))  (sigmoid)
        a = params.get('a', 5.0)
        b = params.get('b', 0.3)
        p = 1.0 / (1.0 + np.exp(-a * (x - b)))
    else:
        p = np.ones(n)
    
    p = np.maximum(p, 1e-10)
    return p / p.sum()

def resample_sequence(p, new_n):
    """Resample sequence to new length using linear interpolation."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == new_n:
        return p
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, new_n)
    p_new = np.interp(new_x, old_x, p)
    p_new = np.maximum(p_new, 0)
    s = p_new.sum()
    if s < 1e-12:
        return np.full(new_n, 1.0 / new_n)
    return p_new / s


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative step function minimizing 2*n*max(conv*p)/(sum*p)^2.
    Uses LP best-response, gradient refinement, multi-start, and length adaptation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 15.0)
    
    # Get parent
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and parent_arr.sum() > 0:
            parent = parent_arr
    except Exception:
        parent = None
    
    best_p = None
    best_score = np.inf
    
    def try_update(p, sc=None):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = normalize(p)
        if sc is None or not np.isfinite(sc):
            sc = score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p.copy()
            print("  NEW BEST: score=%.10f, n=%d" % (sc, p.size), flush=True)
    
    print("Starting search. Parent available: %s" % (parent is not None), flush=True)
    if parent is not None:
        ps = score(parent)
        print("Parent score: %.10f, length: %d" % (ps, parent.size), flush=True)
        try_update(parent, ps)
    
    # Determine sequence lengths to try
    if parent is not None:
        base_n = parent.size
    else:
        base_n = 512
    
    # Try a few lengths around base_n
    lengths_to_try = sorted(set([
        base_n,
        max(64, base_n // 2),
        min(base_n * 2, 2048),
        256,
        512,
        768,
        1024,
        1536,
    ]))
    
    # Remove lengths that are too close to each other
    filtered = []
    for ln in lengths_to_try:
        if not filtered or abs(ln - filtered[-1]) > 50:
            filtered.append(ln)
    lengths_to_try = filtered
    
    print("Lengths to try: %s" % lengths_to_try, flush=True)
    
    # Allocate time to each length
    n_lengths = len(lengths_to_try)
    # Give more time to lengths near the parent
    weights = []
    for ln in lengths_to_try:
        if parent is not None:
            w = 1.0 / (1.0 + abs(ln - base_n) / base_n)
        else:
            w = 1.0
        weights.append(w)
    total_w = sum(weights)
    weights = [w / total_w for w in weights]
    
    available_time = deadline - start_time
    final_reserve = 120.0  # Reserve for final polish
    
    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - final_reserve:
            break
        
        remaining = deadline - final_reserve - time.monotonic()
        if remaining < 30:
            break
        
        time_for_length = min(remaining, available_time * weights[li] + 30)
        time_for_length = max(30, time_for_length)
        print("\n--- Length n=%d, time budget=%.0fs ---" % (n, time_for_length), flush=True)
        
        # Generate candidates for this length
        parent_for_len = None
        if parent is not None:
            parent_for_len = resample_sequence(parent, n)
        
        candidates = generate_candidates(n, rng, parent_for_len)
        
        # Score all candidates
        scored = []
        for c in candidates:
            c = normalize(c)
            sc = score(c)
            if np.isfinite(sc):
                scored.append((sc, c))
        scored.sort(key=lambda x: x[0])
        
        # Try top candidates
        top_k = min(len(scored), max(3, int(time_for_length / 40)))
        
        for si in range(top_k):
            if time.monotonic() > deadline - final_reserve - 10:
                break
            
            sc_init, p_init = scored[si]
            try_update(p_init, sc_init)
            
            if time.monotonic() > deadline - final_reserve - 20:
                break
            
            time_left = deadline - final_reserve - time.monotonic()
            if time_left < 15:
                break
            
            t_alloc = min(60.0, max(15.0, time_left / (top_k - si)))
            
            print("  Candidate %d/%d: init_score=%.6f, optimizing for %.0fs" % (
                si + 1, top_k, sc_init, t_alloc), flush=True)
            
            allow_lp = n <= 1000
            p_best, sc_best = optimize_sequence(p_init, rng, t_alloc, allow_lp)
            try_update(p_best, sc_best)
    
    # Final polish on the best found
    if best_p is not None and time.monotonic() < deadline - 10:
        print("\n--- Final polish ---", flush=True)
        remaining = deadline - time.monotonic()
        if remaining > 5:
            allow_lp = best_p.size <= 1000
            p_best, sc_best = optimize_sequence(best_p, rng, remaining, allow_lp)
            try_update(p_best, sc_best)
    
    # If we still have time, try LP on the best with more iterations
    if best_p is not None and time.monotonic() < deadline - 5:
        n = best_p.size
        if n <= 800:
            print("\n--- Extended LP refinement ---", flush=True)
            remaining = deadline - time.monotonic()
            p = best_p.copy()
            for it in range(20):
                if time.monotonic() > deadline - 3:
                    break
                g = lp_best_response(p, time_limit=min(15.0, remaining))
                if g is None:
                    break
                p_new, sc_new = try_blend(p, g, remaining)
                if sc_new < best_score - 1e-10:
                    p = p_new
                    try_update(p, sc_new)
                else:
                    sc_g = score(g)
                    if sc_g < best_score - 1e-10:
                        p = g.copy()
                        try_update(p, sc_g)
                    else:
                        # Try smaller blend
                        for t in (0.3, 0.15, 0.05):
                            p_try = project_simplex((1-t)*best_p + t*g)
                            sc_try = score(p_try)
                            if sc_try < best_score - 1e-10:
                                p = p_try
                                try_update(p, sc_try)
                                break
                        else:
                            break
    
    if best_p is None:
        # Fallback
        best_p = np.full(512, 1.0 / 512)
        best_score = 2.0
    
    # Final validation
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = best_p.sum()
    if s < 0.01:
        best_p = best_p * (0.01 / max(s, 1e-15))
    best_p = np.minimum(best_p, 1000.0)
    
    final_score = evaluate_sequence([float(x) for x in best_p])
    print("\nFinal score: %.10f (length=%d)" % (final_score, best_p.size), flush=True)
    
    return [float(x) for x in best_p]
