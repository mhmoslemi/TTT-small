The Frank-Wolfe method minimizes a convex function over a convex set by:
1. Computing a subgradient g at the current point.
2. Solving: min_{y in S} <g, y> (the linear minimization oracle).
3. Moving: x_new = (1-t)*x + t*y.

- The feasible set S = {a >= 0, sum(a) = 1} is the simplex.
- The linear minimization oracle: min <g, y> s.t. y >= 0, sum(y) = 1. Solution: y = e_j where j = argmin g.
- This gives a one-sparse step, which might be too aggressive.

The "LP version" from the prompt is different: instead of one-sparse, it finds the best y in the set {y >= 0, (f*y)_k <= M for all k}, which is a larger set that includes the simplex. This allows for "smoother" steps.

"""
Anti-symmetric step function optimizer using:
1. Multi-harmonic structured initialization with low-dimensional optimization
2. FFT-accelerated subgradient descent with Nesterov momentum
3. Multi-length exploration and diversification

import numpy as np
import time
from scipy.optimize import minimize_scalar, minimize


def fast_convolve(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:size]
    return result


def compute_score(a):
    """Compute 2*n*max_b for normalized sequence (sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * np.max(b)


def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def anti_sym_basis(n, mode, m=1):
    """Generate anti-symmetric shape: s_i = -s_{n-1-i}."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    if n <= 1:
        return np.zeros(n)

    if mode == 'linear':
        s = (i - center) / center
    elif mode == 'sin':
        s = np.sin(2 * np.pi * m * (i - center) / n)
    elif mode == 'square':
        s = np.sign(i - center)
    elif mode == 'cubic':
        s = ((i - center) / center) ** 3
    elif mode == 'tanh':
        s = np.tanh(4.0 * (i - center) / center)
    elif mode == 'sin_cos':
        s = np.sin(2 * np.pi * m * (i - center) / n) + 0.5 * np.sin(4 * np.pi * m * (i - center) / n)
    else:
        s = np.zeros(n)

    # Enforce exact anti-symmetry
    s = 0.5 * (s - s[::-1])
    return s


def optimize_1d_shape(n, s):
    """Optimize amplitude eps for a_i = (1/n)(1 + eps*s_i)."""
    max_s = np.max(np.abs(s))
    if max_s < 1e-12:
        return 0.0, 1.0 / n
    eps_max = 1.0 / max_s * 0.999

    def obj(eps):
        a = (1.0 / n) * (1.0 + eps * s)
        if np.any(a < -1e-14):
            return 1e10
        a = np.maximum(a, 0.0)
        b = fast_convolve(a)
        return np.max(b)

    result = minimize_scalar(obj, bounds=(0, eps_max), method='bounded',
                             options={'xatol': 1e-9, 'maxiter': 200})
    if result.success and result.fun < 1e9:
        return result.x, result.fun
    return 0.0, obj(0)


def multi_harmonic_score(params, n, M):
    """Score for a_i = (1/n)(1 + eps * sum_m c_m * sin(2*pi*m*(i-center)/n))."""
    c = params[:M]
    eps = params[M]

    # Check non-negativity constraint
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    s = np.zeros(n)
    for m in range(1, M + 1):
        s += c[m - 1] * np.sin(2 * np.pi * m * (i - center) / n)

    a = (1.0 / n) * (1.0 + eps * s)
    if np.any(a < -1e-12):
        return 1e10

    a = np.maximum(a, 0.0)
    b = fast_convolve(a)
    return np.max(b) * 2.0 * n  # Return the actual score


def optimize_multi_harmonic(n, M, x0, max_iter=500):
    """Nelder-Mead optimization of multi-harmonic parameters."""
    bounds_scale = 1.0  # c_m in [-1, 1], eps in [0, 1]

    def objective(params):
        return multi_harmonic_score(params, n, M)

    # Initial guess
    x0_full = np.zeros(M + 1)
    x0_full[:M] = x0[:M] if len(x0) >= M else x0
    x0_full[M] = x0[M] if len(x0) > M else 0.5

    # Constraint: |1 + eps * s| >= 0 for all i, roughly eps * max|s| <= 1
    # max|s| <= sum |c_m|, so eps * sum|c| <= 1
    # Let's just use bounds and penalty

    result = minimize(objective, x0_full, method='Nelder-Mead',
                      options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-10})
    return result


def subgradient_optimize(a, deadline, rng, M_peaks=30, init_eta=None, beta=0.7):
    """
    Subgradient descent with momentum to minimize max_k (a*a)_k
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
        S = 1.0

    best = a.copy()
    best_score = compute_score(best)

    if init_eta is None:
        init_eta = 0.8 / np.sqrt(n)

    eta = init_eta
    v = np.zeros(n)
    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline:
        b = fast_convolve(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Select peaks with softmax weighting
        if M_peaks >= len(b):
            k_idx = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            k_idx = np.argpartition(b, -M_peaks)[-M_peaks:]
            b_sel = b[k_idx]
            temp = max(b_max - np.min(b_sel), 1e-15)
            wts = np.exp(-8.0 * (b_max - b_sel) / temp)
            wts /= np.sum(wts)

        # Compute weighted gradient
        grad = np.zeros(n)
        for ki in range(len(k_idx)):
            w = wts[ki]
            if w < 1e-10:
                continue
            k = int(k_idx[ki])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * w * a[k - idx]

        # Center for sum constraint
        grad -= np.mean(grad)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Momentum update
        v = beta * v + grad / gn
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        a = a - eta * v / vn

        # Project onto simplex
        a = project_simplex(a, 1.0)
        S = np.sum(a)
        if S < 1e-10:
            a = best.copy()
            break

        # Adaptive learning rate
        iter_count += 1
        if no_improve >= 100:
            eta *= 0.75
            v *= 0.3
            no_improve = 0
            if eta < 1e-14:
                break
        elif iter_count % 500 == 0 and no_improve > 30:
            # Slight reheat
            eta *= 1.2
            no_improve = int(no_improve * 0.5)

    return best, best_score
