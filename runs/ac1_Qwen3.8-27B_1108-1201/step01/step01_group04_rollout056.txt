    best_score = compute_obj(a)
    stalled = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        c = np.convolve(a, a)
        M = float(c.max())

        # Try LP with different step fractions
        improved = False
        for sf in [0.5, 0.3, 0.15, 0.1]:
            d = lp_directed_step(a, c, M, step_frac=sf)
            if d is None:
                continue
            # Check if the direction actually helps
            for t in [0.5, 1.0, 0.3, 0.7]:
                a_new = np.maximum(a + t * d, 0.0)
                a_new = normalize_seq(a_new)
                s_new = compute_obj(a_new)
                if s_new < best_score:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Fallback: small subgradient step
            g = hard_peak_grad(a, c)
            g -= g.mean()
            for lr_try in [0.03, 0.01, 0.05, 0.005]:
                a_new = normalize_seq(np.maximum(a - lr_try * g, 0.0))
                s_new = compute_obj(a_new)
                if s_new < best_score:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    stalled = 0
                    improved = True
                    break
            if not improved:
                stalled += 1
                if stalled > 30:
                    # Try a random perturbation to escape
                    pert = np.random.randn(n) * 0.02 * a.max()
                    a = normalize_seq(np.maximum(a + pert, 0.0))
                    stalled = 0

        if verbose and (it + 1) % 50 == 0:
            print(f"    LP iter {it+1}: score={best_score:.6f}")

    return best, best_score


def combined_optimize(a0, deadline, verbose=False):
    """Combined optimization: subgradient then LP refinement then final polish."""
    n = len(a0)

    # Phase 1: Subgradient descent (use 50% of remaining time)
    t_now = time.monotonic()
    remaining = deadline - t_now
    phase1_deadline = t_now + 0.5 * remaining
    a, score = subgradient_optimize(a0, phase1_deadline,
                                    max_iters=6000, verbose=verbose)
    if verbose:
        print(f"  After subgradient: score={score:.6f}")

    # Phase 2: LP refinement (use 40% of remaining time)
    t_now = time.monotonic()
    remaining = deadline - t_now
    phase2_deadline = t_now + 0.7 * remaining
    if n <= 1500:
        a, score = lp_refine_optimize(a, phase2_deadline,
                                      max_iters=400, verbose=verbose)
        if verbose:
            print(f"  After LP refine: score={score:.6f}")

    # Phase 3: Final polish with alternating subgradient and LP
    t_now = time.monotonic()
    remaining = deadline - t_now
    if remaining > 20.0 and n <= 1500:
        phase3_deadline = t_now + 0.8 * remaining
        for polish_iter in range(200):
            if time.monotonic() > phase3_deadline:
                break
            # Small subgradient step
            c = np.convolve(a, a)
            tau = 50.0
            g = softmax_grad(a, c, tau)
            g -= g.mean()
            lr = 0.02 * (1.0 + np.cos(np.pi * polish_iter / 200.0))
            lr = max(lr, 0.001)
            a_new = normalize_seq(np.maximum(a - lr * g, 0.0))
            s_new = compute_obj(a_new)
            if s_new < score:
                a = a_new
                score = s_new
            else:
                # Try LP step
                M = float(c.max())
                d = lp_directed_step(a, c, M, step_frac=0.1)
                if d is not None:
                    for t in [0.3, 0.5, 0.1]:
                        a_try = normalize_seq(np.maximum(a + t * d, 0.0))
                        s_try = compute_obj(a_try)
                        if s_try < score:
                            a = a_try
                            score = s_try
                            break

        if verbose:
            print(f"  After polish: score={score:.6f}")

    return a, score


def try_single_sequence(a0, deadline, verbose=False):
    """Run full optimization pipeline on a single starting sequence."""
    if time.monotonic() > deadline - 15.0:
        return a0, compute_obj(a0)
    return combined_optimize(a0, deadline, verbose=verbose)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Explores multiple starting sequences of various lengths
    using subgradient descent, LP-based refinement, and combined optimization.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    def update_best(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  *** New best: score={best_score:.6f}, n={len(seq)}, "
                  f"time={time.monotonic()-t_start:.1f}s")

    # Generate starting sequences
    starts = []

    # Use height_sequence_1 if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            gs = np.clip(gs, 0.0, 1000.0)
            for L in [128, 256, 512, 768, min(1024, len(gs)), len(gs)]:
                if L <= len(gs) and L >= 16:
                    starts.append(gs[:L].copy())
    except NameError:
        pass

    # Random starts
    for n in [128, 256, 512, 768, 1024]:
        starts.append(rng.random(n) * 0.3 + 0.35)

    # Structured starts
    for n in [256, 512, 1024]:
        i = np.arange(n, dtype=np.float64)
        # Raised cosine
        starts.append(0.5 + 0.5 * np.cos(2.0 * np.pi * i / n))
        # Gaussian window
        center = n / 2.0
        starts.append(np.exp(-0.5 * ((i - center) / (0.3 * n)) ** 2))
        # Modulated
        starts.append(np.abs(np.sin(np.pi * i / n)) ** 2 + 0.1)
        # Exponential decay pair
        starts.append(np.concatenate([
            np.exp(-0.5 * (i[:n//2]) / (n / 8.0)),
            np.exp(-0.5 * (i[n//2:] - n//2) / (n / 8.0))
        ]))

    # Sort by length (shorter first for quick wins)
    starts.sort(key=lambda x: len(x))

    print(f"Proposing candidate with {len(starts)} initial sequences, "
          f"budget={budget_s}s")

    # Strategy: spend less time on shorter sequences, more on promising ones
    # First pass: quick optimization of all starts (limited time each)
    total_time_for_starts = 0.6 * (deadline - t_start)
    n_starts = len(starts)
    # Allocate time proportional to length (longer sequences have more potential)
    lengths = [len(s) for s in starts]
    max_len = max(lengths)
    weights = [min(1.0, l / max_len) for l in lengths]
    total_weight = sum(weights)
    time_allocs = [total_time_for_starts * w / total_weight for w in weights]

    for idx, (a0) in enumerate(starts):
        if time.monotonic() > deadline - 30.0:
            break
        alloc = min(time_allocs[idx], deadline - t_start - 30.0)
        if alloc < 5.0:
            continue
        seq_deadline = t_start + (idx + 1) / n_starts * (0.7 * (deadline - t_start))
        seq_deadline = min(seq_deadline, deadline - 20.0)

        # Quick check: evaluate the starting sequence
        s0 = compute_obj(normalize_seq(a0))
        print(f"  Start {idx}: n={len(a0)}, initial_score={s0:.6f}")

        if s0 < best_score:
            best_seq = normalize_seq(a0).copy()
            best_score = s0
            print(f"  *** Initial best: score={best_score:.6f}, n={len(a0)}")

        # Only do full optimization if remaining time allows
        if time.monotonic() > deadline - 60.0:
            break

        # For the first few starts, do quick optimization
        # For later starts (especially longer ones), do full optimization
        if idx < 3:
            # Quick: just subgradient
            a, s = subgradient_optimize(a0, seq_deadline, max_iters=2000)
            update_best(a, s)
        else:
            # Full pipeline
            a, s = try_single_sequence(a0, seq_deadline)
            update_best(a, s)

    # Second pass: focus remaining time on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        print(f"\nFocusing on best sequence (score={best_score:.6f}, "
              f"n={len(best_seq)})...")
        remaining_deadline = deadline
        # Multiple refinement passes
        for pass_i in range(10):
            if time.monotonic() > remaining_deadline - 15.0:
                break
            a, s = try_single_sequence(best_seq, remaining_deadline - 5.0,
                                       verbose=True)
            if s < best_score:
                best_seq = a
                best_score = s
                print(f"  Focus pass {pass_i}: score={best_score:.6f}")
            else:
                # Try a different perturbation
                n = len(best_seq)
                pert_scale = 0.01 * (1 - pass_i / 10.0)
                pert = rng.normal(0, pert_scale, size=n)
                a_pert = normalize_seq(np.maximum(best_seq + pert, 0.0))
                s_pert = compute_obj(a_pert)
                if s_pert < best_score:
                    best_seq = a_pert
                    best_score = s_pert
                    print(f"  Perturbation improved: score={best_score:.6f}")
                # Also try a fresh LP refinement
                if len(best_seq) <= 1500:
                    a_lp, s_lp = lp_refine_optimize(best_seq,
                                                    remaining_deadline - 5.0,
                                                    max_iters=100)
                    if s_lp < best_score:
                        best_seq = a_lp
                        best_score = s_lp
                        print(f"  LP refresh: score={best_score:.6f}")

    # Third pass: try a few new random starts with remaining time
    if time.monotonic() < deadline - 60.0:
        print(f"\nTrying new random starts with remaining time...")
        for attempt in range(5):
            if time.monotonic() > deadline - 30.0:
                break
            n_try = int(rng.choice([256, 512, 512, 768, 1024]))
            a_random = rngrandom = rng.random(n_try) * 0.5 + 0.25
            a_random = normalize_seq(a_random)
            s_random = compute_obj(a_random)

            if s_random < best_score * 1.05:
                # Promising, do quick optimization
                a_opt, s_opt = subgradient_optimize(
                    a_random, min(deadline - 20.0, t_start + 120.0),
                    max_iters=3000)
                update_best(a_opt, s_opt)
            elif s_random < best_score:
                update_best(a_random, s_random)

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal: n={len(best_seq)}, computed_score={final_score:.6f}")
        # Validate with evaluate_sequence
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
            # Fallback
            best_seq = np.ones(128)
            final_score = compute_obj(best_seq)
            print(f"Fallback score: {final_score:.6f}")
        return [float(x) for x in best_seq]

    # Ultimate fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
