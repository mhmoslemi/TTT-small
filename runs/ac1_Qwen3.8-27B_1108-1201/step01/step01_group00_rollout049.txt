import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = next_fast_len(n + m - 1)
    return np.real(ifft(fft(a, size) * fft(b, size), size)[:n+m-1])

def score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def lp_step(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    # Build constraint matrix A where A[k,j] = a[k-j] if 0 <= k-j < n
    # Number of constraints: 2n-1
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    # max sum(b) => min -sum(b)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```

```python
"""
LP-based convex optimization with gradient polish for minimizing
the normalized peak autocorrelation of non-negative step functions.
Uses the LP approach from the state of the art (maximize sum(b) 
subject to conv(a,b) <= M) combined with adaptive line search,
gradient-based polishing, and local search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = next_fast_len(n + m - 1)
    fa = fft(a, size)
    fb = fft(b, size)
    return np.real(ifft(fa * fb, size)[:n + m - 1])


def score_fast(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def build_constraint_matrix(a):
    """Build A where A[k,j] = a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_step(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_constraint_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def lp_optimize(a_init, time_limit, rng, verbose=True):
    """
    LP-based optimization with adaptive M and line search.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a_init, np.inf
    target_sum = np.sqrt(2.0 * n)
    a = a * target_sum / s

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    iteration = 0
    stuck_count = 0

    if verbose:
        print(f"  LP opt: n={n}, initial score={best_score:.6f}")

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        M = c.max()

        # Try different M tightness levels
        found_improvement = False
        for m_factor in [0.999, 0.998, 0.995, 0.99, 1.0]:
            M_tight = M * m_factor
            b = lp_step(a, M_tight)
            if b is not None and b.sum() > 1e-10:
                b = b * target_sum / b.sum()
                # Line search
                best_t = 0
                best_cand_score = best_score
                for t in [1.0, 0.8, 0.6, 0.4, 0.2]:
                    cand = (1 - t) * a + t * b
                    s_cand = score_fast(cand)
                    if s_cand < best_cand_score:
                        best_cand_score = s_cand
                        best_t = t
                if best_t > 0:
                    a = (1 - best_t) * a + best_t * b
                    if best_cand_score < best_score:
                        best_score = best_cand_score
                        best_a = a.copy()
                        stuck_count = 0
                        found_improvement = True
                    break

        iteration += 1
        if not found_improvement:
            stuck_count += 1
            if stuck_count > 20:
                break

        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iteration}, score={score_fast(a):.6f}, best={best_score:.6f}, t={elapsed:.1f}s")

    if verbose:
        print(f"  LP opt done: {iteration} iters, best={best_score:.6f}, time={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def grad_polish(a_init, time_limit, rng, verbose=True):
    """
    Gradient-based polish using soft-max approximation and Adam.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    target_sum = np.sqrt(2.0 * n)
    a = a * target_sum / s

    # Use log-parameterization: a_i = exp(theta_i) * scale
    # where scale = target_sum / sum(exp(theta))
    theta = np.log(np.maximum(a, 1e-10))
    theta = theta - theta.mean()  # center for numerical stability

    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    tau = 5.0 / n  # soft-max temperature
    tau_end = 0.5 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau * (tau_end / tau) ** 0  # decay later
        
        # Compute a from theta
        t_shifted = theta - theta.max()
        e = np.exp(t_shifted)
        a = e / e.sum() * target_sum

        # Compute gradient of soft_max(conv(a,a), tau) w.r.t. a
        c = fft_convolve(a, a)
        c_max = c.max()
        w = np.exp((c - c_max) / tau)
        w = w / w.sum()
        
        # Gradient: 2 * conv(a_rev, w)[n-1:2n-1]
        a_rev = a[::-1]
        corr = fft_convolve(a_rev, w)
        g_a = 2.0 * corr[n - 1:2 * n - 1]

        # Gradient w.r.t. theta (using the softmax parameterization)
        # a_i = exp(theta_i - logsumexp) * target_sum
        # da_i/dtheta_j = a_i * (delta_ij - a_j / sum(a))
        # grad_theta_j = sum_i g_a_i * da_i/dtheta_j
        #               = g_a_j * a_j - (sum_i g_a_i * a_i) * a_j / sum(a)
        inner = np.dot(g_a, a)
        a_norm = a / target_sum  # this sums to 1
        g_theta = a_norm * (g_a / target_sum - inner / (target_sum * target_sum))
        # Actually let me redo this more carefully.
        # a_i = S * exp(theta_i) / sum(exp(theta)) where S = target_sum
        # da_i/dtheta_j = S * exp(theta_i)/sum(exp) * (delta_ij - exp(theta_j)/sum(exp))
        #                = a_i * (delta_ij - a_j/S)
        # grad_theta_j = sum_i g_a_i * a_i * (delta_ij - a_j/S)
        #              = g_a_j * a_j - a_j/S * sum_i(g_a_i * a_i)
        #              = a_j * (g_a_j - inner/S)
        
        g_theta = a * (g_a - inner / target_sum)

        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1 - beta1 ** step)
        m2_hat = momentum2 / (1 - beta2 ** step)
        lr = 1.0 / np.sqrt(step + 1)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Check score
        t_shifted = theta - theta.max()
        e = np.exp(t_shifted)
        a_check = e / e.sum() * target_sum
        sc = score_fast(a_check)

        if sc < best_score:
            best_score = sc
            best_a = a_check.copy()

        # Decay temperature
        tau = max(tau_end, tau * 0.999)

    if verbose:
        print(f"  Grad polish done: best={best_score:.6f}, time={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def local_search_polish(a_init, time_limit, rng, verbose=True):
    """Simulated annealing with targeted peak reduction."""
    current = a_init.astype(np.float64).copy()
    n = len(current)
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9999

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.4:
            # Targeted: reduce peak contributors
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.75:
            # Symmetrize partially
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.015, size=n))
        elif r < 0.9:
            # Swap-based: swap two blocks
            w = max(1, n // 50)
            i = int(rng.integers(0, n - 2 * w))
            j = int(rng.integers(0, n - 2 * w))
            if abs(i - j) > w:
                candidate[i:i+w], candidate[j:j+w] = candidate[j:j+w].copy(), candidate[i:i+w].copy()
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = score_fast(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature = max(1e-6, temperature * cooling)

    if verbose:
        print(f"  Local search done: best={best_score:.6f}, time={time.monotonic()-t0:.1f}s")

    return best, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0
    for k, amp in [(1, 0.25), (3, 0.15), (5, 0.1), (8, 0.08), (13, 0.05), (21, 0.03)]:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_fast(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
