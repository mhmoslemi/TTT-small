def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    #---Security Checks---
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    # Convert each element to float, checking for negative
    sequence = [np.float64(x) for x in sequence]
    if np.any(x < 0 for x in sequence):
        return np.inf

    if np.max(sequence) > 1000.0:

    max_b = np.max(conv)

    # Protect against total sum being too close to zero (which might lead to division by zero)

        return np.inf
