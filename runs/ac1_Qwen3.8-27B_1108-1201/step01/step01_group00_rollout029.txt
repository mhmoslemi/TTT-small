```python
"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative step functions.
Uses binomial coefficient initialization, annealed softmax gradient descent,
LP-based directional refinement, and simulated annealing polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.special import comb as scipy_comb

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score_fast(a):
    """Fast local score computation."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def soft_max_and_weights(x, tau):
    """Compute soft-max value and weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = x_max + tau * np.log(sum_exp)
    return value, weights

def grad_soft_max_conv(a, tau):
    """Compute gradient of soft_max(conv(a,a), tau) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return value, g, c

def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)

def binomial_init(n):
    """Initialize with central binomial coefficients pattern."""
    k = np.arange(n, dtype=np.float64)
    # C(2k, k) / 4^k
    # Use log for numerical stability
    # log(C(2k,k)) = sum_{i=1}^{k} log(2k-i+1) - sum_{i=1}^{k} log(i)
    # = sum_{i=1}^{k} log((k+i)/i)
    # For large k, C(2k,k)/4^k ~ 1/sqrt(pi*k)
    a = scipy_comb(2 * k.astype(int), k.astype(int), exact=False) / (4.0 ** k)
    return a

def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def optimize_gradient(n, theta_init, time_limit, rng, lr_base=2.0):
    """Annealed softmax gradient descent."""
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = np.inf
    tau_start = 30.0 / n
    tau_end = 0.3 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed_frac * 1.2)
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        lr = lr_base / np.sqrt(step + 1)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 500 == 0:
            a_check = softmax_normalize(theta) * S
            score = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_theta = theta.copy()
    
    # Final check
    a_check = softmax_normalize(theta) * S
    score = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_theta = theta.copy()
    
    return best_theta, best_score

def build_lp_matrix(a):
    """Build the constraint matrix for the LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # A[k, j] = a[k-j] if 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        for j in range(lo, hi):
            A[k, j] = a[k - j]
    return A

def lp_refine(a, time_limit=60, lr_init=0.3):
    """LP-based directional refinement."""
    t0 = time.monotonic()
    n = len(a)
    S = np.sum(a)
    a = a.copy()
    lr = lr_init
    A_matrix = build_lp_matrix(a)  # Will need to rebuild if a changes significantly
    
    best_a = a.copy()
    best_score = local_score_fast(a)
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(a, a))
        
        # Rebuild LP matrix (needed because a changed)
        A_matrix = build_lp_matrix(a)
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A_matrix, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        g = res.x
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        # Rescale g to match sum of a
        g = g * S / sum_g
        
        # Update
        a_new = (1 - lr) * a + lr * g
        a_new = np.clip(a_new, 0, 1000)
        
        # Check improvement
        score_new = local_score_fast(a_new)
        if score_new < local_score_fast(a) * (1 + 1e-6):  # Allow tiny increase
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
        else:
            lr *= 0.9
    
    return best_a, best_score

    current = a_init.astype(np.float64).copy()
    current_score = local_score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.3:
            # Targeted peak reduction
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(peak_idx + 1, n)
            for i in range(lo, hi):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.5:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.7:
            # Pair swap (transfer mass)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = current[i] * rng.uniform(0.05, 0.15)
                candidate[i] = max(0, current[i] - delta)
                candidate[j] = current[j] + delta
        elif r < 0.85:
            # Symmetrize partially
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.015, size=n))
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score_fast(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score

                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score
```

```python
"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative step functions.
Uses binomial coefficient initialization (optimal infinite-sequence structure),
annealed softmax gradient descent, LP-based directional refinement,
and simulated annealing polish.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.special import comb as scipy_comb


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score_fast(a):
    """Fast local score computation (same as evaluate_sequence formula)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def soft_max_and_weights(x, tau):
    """Compute soft-max value and corresponding weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = x_max + tau * np.log(sum_exp)
    return value, weights


def grad_soft_max_conv(a, tau):
    """Compute gradient of soft_max(conv(a,a), tau) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return value, g, c


def softmax_normalize(theta):
    """Stable softmax returning probabilities summing to 1."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def binomial_init(n):
    """Initialize with central binomial coefficient pattern (optimal infinite seq)."""
    k = np.arange(n, dtype=np.int64)
    a = scipy_comb(2 * k, k, exact=False) / (4.0 ** k.astype(np.float64))
    return a


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_symmetric_decay_init(n, rng):
    """Symmetric decaying initialization."""
    t = np.linspace(0, 1, n)
    center = (n - 1) / 2.0
    x = np.abs(np.arange(n) - center) / center
    a = np.exp(-0.5 * x * x * 3.0)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def optimize_gradient(n, theta_init, time_limit, rng, lr_base=2.0, verbose=False):
    """Annealed softmax gradient descent with Adam optimizer."""
    S = np.sqrt(2.0 * n)
    theta = theta_init.astype(np.float64).copy()
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    best_theta = theta.copy()
    best_score = np.inf
    tau_start = 40.0 / n
    tau_end = 0.2 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed_frac * 1.15)
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        lr = lr_base / np.sqrt(step + 1)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 300 == 0:
            a_check = softmax_normalize(theta) * S
            score = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_theta = theta.copy()
                if verbose:
                    print(f"      step={step}, score={score:.6f}, tau={tau:.2e}")

    # Final check
    a_check = softmax_normalize(theta) * S
    score = 2.0 * n * np.max(fft_convolve(a_check, a_check)) / (S * S)
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_theta = theta.copy()

    return best_theta, best_score


def build_lp_matrix(a):
    """Build constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    k = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j = np.arange(n)[None, :].astype(np.int64)
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[idx[valid]]
    return A


def lp_refine(a, time_limit=60, lr_init=0.25, verbose=False):
    """
    LP-based directional refinement.
    At each step: find g>=0 maximizing sum(g) s.t. conv(a,g)<=M,
    then update a = (1-lr)*a + lr*g (normalized).
    """
    t0 = time.monotonic()
    n = len(a)
    S = np.sum(a)
    a = a.astype(np.float64).copy()
    lr = lr_init
    best_a = a.copy()
    best_score = local_score_fast(a)
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        # Build LP constraint matrix
        A_mat = build_lp_matrix(a)

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:
            break

        if not res.success:
            break

        g = res.x
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        # Rescale g to match sum of a
        g = g * S / sum_g

        # Update with clipping
        a_new = (1.0 - lr) * a + lr * g
        a_new = np.clip(a_new, 0.0, 1000.0)
        # Re-normalize to maintain sum
        sum_new = np.sum(a_new)
        if sum_new > 0.01:
            a_new = a_new * S / sum_new

        score_new = local_score_fast(a_new)
        if score_new <= best_score * (1 + 1e-8):
            a = a_new
            if score_new < best_score - 1e-10:
                best_score = score_new
                best_a = a.copy()
                stagnant = 0
                if verbose:
                    print(f"    LP iter: score={best_score:.6f}, lr={lr:.4f}")
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 20:
            lr *= 0.85
            stagnant = 0

    return best_a, best_score


def polish_local_search(a_init, time_limit, rng):
    """Simulated annealing with targeted peak reduction."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.003
    cooling = 0.9996

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.30:
            # Targeted: reduce peak contributors
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(peak_idx + 1, n)
            for i in range(lo, hi):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.06, 0.02, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(12, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.50:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        elif r < 0.65:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.80:
            # Partial symmetrization
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.02, size=n))
        elif r < 0.90:
            # Pair mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = current[i] * rng.uniform(0.05, 0.2)
                candidate[i] = max(0.0, current[i] - delta)
                candidate[j] = current[j] + delta
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score_fast(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling

    return best, best_score

    print("\n=== Phase 3: Simulated annealing polish ===")
    if best_seq is not None:
        polish_time = min(300.0, max(30.0, (deadline - time.monotonic()) * 0.4))
        print(f"  Available time={polish_time:.1f}s, starting score={best_score:.6f}")
        polished, polish_score = polish_local_search(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
        print(f"  After polish: {best_score:.6f}")

    # === Phase 4: Additional refinement ===
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print("\n=== Phase 4: Additional refinement ===")
        remaining = deadline - time.monotonic() - 15
        n = len(best_seq)

        # Gradient pass on current best with lower learning rate
        theta_init = np.log(np.clip(best_seq, 0.01, None) / np.sum(best_seq) + 1e-10)
        t_grad = min(60.0, remaining * 0.4)
        if t_grad > 10:
            theta_best, grad_score = optimize_gradient(n, theta_init, t_grad, rng, lr_base=1.0)
            a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
            actual_score = local_score_fast(a_best)
            print(f"  Gradient refinement: {actual_score:.6f}")
            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()

        # Short LP pass
        if time.monotonic() < deadline - 20:
            t_lp = min(40.0, deadline - time.monotonic() - 20)
            if t_lp > 10:
                a_lp3, lp_score3 = lp_refine(best_seq, time_limit=t_lp, lr_init=0.1)
                if lp_score3 < best_score:
                    best_score = lp_score3
                    best_seq = a_lp3.copy()
                    print(f"  Short LP: {best_score:.6f}")

        # Final short polish
        if time.monotonic() < deadline - 10:
            t_final = min(30.0, deadline - time.monotonic() - 5)
            if t_final > 10:
                polished2, ps2 = polish_local_search(best_seq, t_final, rng)
                if ps2 < best_score:
                    best_score = ps2
                    best_seq = polished2.copy()
                print(f"  Final polish: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
