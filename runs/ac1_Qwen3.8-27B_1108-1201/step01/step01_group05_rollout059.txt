    """Compute gradient of (1/|K|) * sum_{k in K} c[k] where c = p*p."""
    n = len(p)
    if c is None:
        c = convolve_full(p)
    
    # Create weight vector: 1/|K| for k in K, 0 otherwise
    w = np.zeros(len(c))
    for k in k_vals:
        w[k] = 1.0 / len(k_vals)
    
    # grad[m] = 2 * sum_j w[m+j] * p[j]
    # = 2 * full_convolve(w, p[::-1])[n-1 : n-1+n]
    L = next_fast_len(len(w) + n - 1)
    Fw = fft(w, L)
    Fp = fft(p[::-1], L)
    h = np.real(ifft(Fw * Fp))
    grad = 2.0 * h[n-1 : n-1+n]
    return grad, c

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssum - 1)
    k = np.count_nonzero(cond) - 1
    theta = (cssum[k] - 1) / (k + 1)
    out = np.maximum(v - theta, 0)
    s = out.sum()
    return out / s

def score(p):
    """Compute 2*n*max(conv(p,p)) / (sum(p))^2."""
    n = len(p)
    s = p.sum()
    if s < 1e-12:
        return np.inf
    c = convolve_full(p)
    return 2.0 * n * c.max() / (s * s)

**Alternating Minimization / Coordinate Descent:**

For each coordinate p[j], the objective max_k c[k] is a function of p[j]. If we fix all other coordinates, c[k] as a function of p[j] is:
c[k] = sum_{j' != j} p[j']*p[k-j'] + p[j]*p[k-j] + p[k-j]*p[j] + (if k=2j) p[j]^2
= c[k]_{-j} + 2*p[j]*p[k-j] + (if k=2j) p[j]^2

where c[k]_{-j} is the contribution to c[k] from all coordinates except j.

So for a fixed k, c[k] is a quadratic in p[j] (specifically, linear if k != 2j, quadratic if k = 2j).

The max over k is the max of these quadratics/linears. Minimizing this over p[j] is a 1D problem that can be solved by bisection.

This gives a **cyclical coordinate descent** algorithm:
1. For j = 0, 1, ..., n-1:
   a. Compute the current c (or just the values that depend on p[j]).
   b. Find the value of p[j] that minimizes max_k c[k] (given all other p[j']).
   c. Update p[j].
2. Renormalize (project onto simplex).
3. Repeat.

The 1D subproblem in step b:
Given all p[j'] for j' != j, find p* = argmin_t max_k f_k(t) where:
f_k(t) = c_k_base + 2*t*p[k-j] (if k != 2j)
f_{2j}(t) = c_{2j}_base + t^2 (if k = 2j, i.e., the self-term)

Wait, I need to be more careful. c[k] = sum_j' p[j']*p[k-j']. If we separate out the terms involving p[j]:
c[k] = (sum_{j' != j} p[j']*p[k-j']) + p[j]*p[k-j] (if k-j != j) + p[j]*p[j] (if k-j = j, i.e., k = 2j)

So:
- If k != 2j: c[k] = c_base[k] + p[j] * p[k-j] (linear in p[j])
- If k = 2j: c[k] = c_base[k] + p[j]^2 (quadratic in p[j])

where c_base[k] = sum_{j' != j} p[j']*p[k-j'].

To find the optimal p[j], we need to minimize:
max_k { c_base[k] + p[j]*p[k-j] (for k != 2j),
        c_base[2j] + p[j]^2 (for k = 2j) }
subject to p[j] >= 0.

- The function is convex in p[j] (max of convex functions).
- We can use bisection on the derivative.

- From the linear terms: p[k-j] for the k that achieves the max (among k != 2j)

At the optimum, 0 should be in the subdifferential. This means the maximum of the linear parts at t should be <= the quadratic part's contribution... it's getting complicated.

"""
Step-function search for low max autocorrelation using top-K gradient method.
Minimizes 2*n*max(conv(p,p))/(sum(p))^2 over nonnegative p.
Uses FFT-based gradient computation and projected gradient descent on the simplex.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len

def _conv_full(p):
    n = len(p)
    L = next_fast_len(2*n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def _score(p):
    n = len(p)
    s = float(p.sum())
    if s < 1e-12:
        return np.inf
    c = _conv_full(p)
    return float(2.0 * n * np.max(c) / (s * s))

def _project_simplex(v):
    v = np.asarray(v, float)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=float)
    cond = u * idx > (css - 1.0)
    k = int(np.count_nonzero(cond)) - 1
    if k < 0:
        return np.full(n, 1.0/n)
    theta = (css[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s < 1e-30:
        return np.full(n, 1.0/n)
    return out / s

def _grad_topk(p, c, K):
    """Gradient of (1/K) sum of top-K values of c, where c = p*p."""
    n = len(p)
    # Find top-K indices
    if K >= len(c):
        topk = np.arange(len(c))
    else:
        topk = np.argpartition(c, -K)[-K:]
    
    # Weight vector
    w = np.zeros(len(c))
    w[topk] = 1.0 / K
    
    # grad[m] = 2 * sum_j w[m+j] * p[j] (cross-correlation)
    # = 2 * full_conv(w, p[::-1])[n-1 : n-1+n]
    L = next_fast_len(len(w) + n - 1)
    Fw = fft(w, L)
    Fp = fft(p[::-1], L)
    h = np.real(ifft(Fw * Fp))
    grad = 2.0 * h[n-1 : n-1+n]
    return grad

def _optimize_local(p0, rng, time_limit, label=""):
    p = _project_simplex(p0.astype(float).copy())
    n = len(p)
    best_p = p.copy()
    best_sc = _score(p)
    cur_sc = best_sc
    
    end = time.monotonic() + time_limit
    it = 0
    stuck = 0
    K_init = max(1, n // 4)
    
    while time.monotonic() < end - 2:
        it += 1
        c = _conv_full(p)
        cmax = float(np.max(c))
        
        # Adaptive K
        frac = time.monotonic() - (end - time_limit)
        progress = min(1.0, frac / time_limit)
        K = max(1, int(K_init * (1 - 0.8 * progress)))
        
        grad = _grad_topk(p, c, K)
        
        # Adaptive step size
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            stuck += 1
            if stuck > 100:
                # Random restart
                p = _project_simplex(p + rng.normal(0, 2.0/n, n))
                stuck = 0
            continue
        
        # Step size: aim to reduce the max by a small fraction
        # The gradient of c[k] has L2 norm ~ 2*sqrt(K)*max(p) ~ 2*sqrt(K)/n
        # We want step * |grad| ~ 1/n (typical coordinate change)
        base_lr = 1.0 / (gnorm * n)
        
        accepted = False
        for lr_scale in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
            lr = base_lr * lr_scale
            p_new = _project_simplex(p - lr * grad)
            sc_new = _score(p_new)
            if sc_new < cur_sc:
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_p = p_new.copy()
                    stuck = 0
                else:
