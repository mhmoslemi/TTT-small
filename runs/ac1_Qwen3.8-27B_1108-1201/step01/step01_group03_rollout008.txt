def gradient_descent(f, max_iters=1000, temp_init=0.01, temp_end=0.001, 
                     lr_init=0.1, lr_end=0.001, budget_s=60.0):
    """Run gradient descent with annealed temperature and learning rate."""
    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    f = f / s  # normalize to sum 1
    
    best_f = f.copy()
    best_score = fast_score(f)
    start_time = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start_time > budget_s:
            break
        
        # Anneal parameters
        frac = it / max(1, max_iters)
        temp = temp_init * (temp_end / temp_init) ** frac
        lr = lr_init * (lr_end / lr_init) ** frac
        
        score, grad = score_and_gradient(f, temp)
        if not np.isfinite(score):
            break
        
        if score < best_score:
            best_score = score
            best_f = f.copy()
        
        # Gradient step with projection to non-negative and renormalization
        f_new = f - lr * grad
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:

"""
Hybrid soft-max gradient descent + LP polishing + local search
to minimize the autocorrelation peak of a non-negative step function.

The objective is score = 2*n*max(f*f)/(sum(f))^2, which is scale-invariant.
We use:
1. Soft-max gradient descent with annealing for fast exploration
2. LP-based local search for precise refinement
3. Block perturbation local search for final polishing
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fast_score(f):
    """Compute the score efficiently."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def score_and_softmax_weights(f, temp):
    """Compute score and softmax weights over the convolution."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    s = f.sum()
    conv = np.convolve(f, f)
    
    m = conv.max()
    shifted = (conv - m) / temp
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 0)
    exp_s = np.exp(shifted)
    S = exp_s.sum()
    sm = m + temp * np.log(S)
    w = exp_s / S
    
    score = 2.0 * n * sm / (s * s)
    return score, w, conv, sm


def compute_gradient(f, w, s, sm, n):
    """Compute gradient of score w.r.t. f using soft-max weights."""
    # grad_i = (4n/s^2) * sum_j w_{i+j} * f_j - 4n*sm/s^3
    # cross_corr[i] = sum_j w[i+j] * f[j] = (reversed_f * w)[i+n-1]
    u = f[::-1]
    u_w = np.convolve(u, w, mode='full')
    cross = u_w[n-1:2*n-1]
    
    grad = (4.0 * n / (s * s)) * cross - (4.0 * n * sm) / (s ** 3)
    return grad


def normalize_nonneg(f):
    """Project to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def softmax_gradient_descent(f_init, budget_s, temp_init=0.005, temp_end=0.001,
                              lr_init=0.05, momentum=0.9, 
                              check_interval=50):
    """Run soft-max gradient descent with momentum and temperature annealing."""
    f = normalize_nonneg(f_init.copy())
    n = len(f)
    s = f.sum()  # = 1 after normalization
    
    best_f = f.copy()
    best_score = fast_score(f)
    v = np.zeros(n)  # momentum
    
    start = time.monotonic()
    it = 0
    max_iters = 50000
    
    # Adaptive: estimate how many iterations we can do
    # Each iteration: conv (O(n log n) or O(n^2)) + gradient (O(n^2) or O(n log n))
    # For n=512, ~1ms per iteration, so ~50000 in 50s
    # For n=2048, ~10ms per iteration, so ~5000 in 50s
    
    lr = lr_init
    temp = temp_init
    
    while it < max_iters:
        if time.monotonic() - start > budget_s:
            break
        
        frac = min(1.0, it / max(1, max_iters * 0.8))
        temp = temp_init * (temp_end / temp_init) ** frac
        lr = max(1e-6, lr_init * (0.5 ** (it / 500)))
        
        score, w, conv, sm = score_and_softmax_weights(f, temp)
        if not np.isfinite(score):
            break
        
        if it % check_interval == 0:
            exact_sc = fast_score(f)
            if exact_sc < best_score:
                best_score = exact_sc
                best_f = f.copy()
        
        grad = compute_gradient(f, w, s, sm, n)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        f_new = f - lr * v
        f = normalize_nonneg(f_new)
        s = 1.0  # after normalization
        
        it += 1
    
    # Final check
    exact_sc = fast_score(f)
    if exact_sc < best_score:
        best_score = exact_sc
        best_f = f.copy()
    
    return best_f, best_score


def build_conv_matrix(n, f):
    """Sparse matrix A (2n-1 x n) for the LP: A g <= M."""
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    rows, cols, data = [], [], []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(f, M, time_limit_s=5.0):
    """Solve LP: max sum(g) s.t. A*g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_step(f, t=0.3, lp_time=5.0):
    """One LP-based update step."""
    f = normalize_nonneg(f)
    conv = np.convolve(f, f)
    M = conv.max()
    if M < 1e-15:
        return f
    
    g = solve_lp(f, M, lp_time)
    if g is None:
        return f
    gs = g.sum()
    if gs < 1e-10:
        return f
    g = g / gs
    
    f_new = (1 - t) * f + t * g
    return normalize_nonneg(f_new)


def block_perturb(f, rng, max_block_frac=0.1):
    """Perturb a random block of f."""

    cand[start:start+w] *= np.exp(rng.normal(0, 0.3, w))
    return normalize_nonneg(cand)

"""
Hybrid optimization for minimizing autocorrelation peak of non-negative step functions.
Uses soft-max gradient descent (fast exploration), LP-based alternating minimization
(exploitation), and local search polishing. The objective score = 2*n*max(f*f)/(sum(f))^2
is scale-invariant, so we always normalize to sum(f)=1.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fast_score(f):
    """Compute score = 2*n*max(conv)/(sum)^2 efficiently."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def normalize_nonneg(f):
    """Project to non-negative, normalize to unit sum."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def compute_cross_corr(f, w, n):
    """Compute cross_corr[i] = sum_j w[i+j]*f[j] using FFT-based convolution."""
    u = f[::-1]
    u_w = np.convolve(u, w, mode='full')
    return u_w[n-1:2*n-1]


def softmax_weights(conv, temp):
    """Compute soft-max value and softmax weight vector."""
    m = conv.max()
    shifted = (conv - m) / temp
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    S = exp_s.sum()
    sm = m + temp * np.log(S)
    w = exp_s / S
    return sm, w


def gradient_step(f, temp, n):
    """One gradient step. Returns (new_f, score, converged_flag)."""
    conv = np.convolve(f, f)
    sm, w = softmax_weights(conv, temp)
    score = 2.0 * n * sm  # since sum(f)=1
    cross = compute_cross_corr(f, w, n)
    grad = 4.0 * n * (cross - sm)
    return grad, score


def softmax_gd(f_init, budget_s, temp_init=0.01, temp_end=0.002,
               lr_init=0.1, lr_decay=0.9995, momentum=0.85,
               clamp_range=(1e-6, 10.0)):
    """Soft-max gradient descent with momentum, temperature annealing, and clamping."""
    f = normalize_nonneg(f_init.copy())
    n = len(f)
    v = np.zeros(n)
    best_f = f.copy()
    best_score = fast_score(f)
    start = time.monotonic()
    it = 0
    lr = lr_init
    temp = temp_init
    patience = 0
    patience_limit = 500

    while time.monotonic() - start < budget_s:
        it += 1
        # Anneal
        frac = min(1.0, it / 3000)
        temp = temp_init * (temp_end / temp_init) ** frac
        lr = max(1e-5, lr_init * (lr_decay ** it))

        grad, sm_score = gradient_step(f, temp, n)
        if not np.isfinite(sm_score):
            break

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        f_new = f - lr * v
        f_new = np.clip(f_new, clamp_range[0], clamp_range[1])
        f = normalize_nonneg(f_new)

        # Check improvement on exact score periodically
        if it % 100 == 0:
            exact_sc = fast_score(f)
            if exact_sc < best_score:
                best_score = exact_sc
                best_f = f.copy()
                patience = 0
            else:
                patience += 1
            if patience > patience_limit:
                # Reset momentum, maybe perturb
                v = np.zeros(n)
                f = normalize_nonneg(f + np.random.default_rng(it).normal(0, 0.01, n))
                patience = 0

    final_sc = fast_score(f)
    if final_sc < best_score:
        best_score = final_sc
        best_f = f.copy()
    return best_f, best_score


def build_sparsity(n):
    """Build sparse structure for convolution matrix (independent of f values)."""
    rows = np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    row_idx = cols + rows
    return rows, cols, row_idx


def solve_lp_for_g(f, M, time_limit_s=8.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    rows = np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    row_idx = cols + rows
    data = f[rows]
    A = sp.coo_matrix((data, (row_idx, cols)), shape=(2 * n - 1, n)).tocsr()
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine(f_init, budget_s, t_init=0.3, t_min=0.05, t_decay=0.97,
              lp_time=8.0, max_stall=15):
    """LP-based alternating minimization."""
    f = normalize_nonneg(f_init.copy())
    n = len(f)
    best_f = f.copy()
    best_score = fast_score(f)
    t = t_init
    stall = 0
    start = time.monotonic()
    rng = np.random.default_rng(7777)

    while time.monotonic() - start < budget_s:
        sc_current = fast_score(f)
        conv = np.convolve(f, f)
        M = conv.max()
        if M < 1e-12:
            break

        g = solve_lp_for_g(f, M, time_limit_s=lp_time)
        if g is None:
            stall += 1
            if stall > max_stall:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-10:
            stall += 1
            continue
        g = g / g_sum

        f_new = (1.0 - t) * f + t * g
        f_new = normalize_nonneg(f_new)
        sc_new = fast_score(f_new)

        if np.isfinite(sc_new) and sc_new < sc_current - 1e-9:
            f = f_new
            t = max(t_min, t * t_decay)
            stall = 0
            if sc_new < best_score:
                best_score = sc_new
                best_f = f.copy()
        else:
            stall += 1
            if stall % 5 == 0:
                # Perturb to escape
                w = max(2, n // 20)
                s = int(rng.integers(0, n - w))
                f = f.copy()
                f[s:s+w] *= np.exp(rng.normal(0, 0.5, w))
                f = normalize_nonneg(f)
            if stall > max_stall * 3:
                break

    return best_f, best_score


def block_perturb_local(f_init, budget_s, seed=42):
    """Local search with random block multiplicative perturbations + SA."""
    f = normalize_nonneg(f_init.copy())
    n = len(f)
    best_f = f.copy()
    best_score = fast_score(f)
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    T = 0.002
    T_min = 1e-7
    no_improve = 0

    while time.monotonic() - start < budget_s:
        # Random block
        frac = rng.choice([0.01, 0.02, 0.05, 0.1, 0.2])
        w = max(1, int(n * frac))
        s = int(rng.integers(0, max(1, n - w)))
        sigmas = rng.choice([0.1, 0.2, 0.3, 0.5])

        cand = f.copy()
        cand[s:s+w] *= np.exp(rng.normal(0, sigmas, w))
        cand = normalize_nonneg(cand)
        sc = fast_score(cand)

        if not np.isfinite(sc):
            continue

        if sc < best_score:
            best_score = sc
            best_f = cand.copy()
            f = cand
            no_improve = 0
        elif rng.random() < np.exp(-(sc - best_score) / max(T, T_min)):
            f = cand
            no_improve += 1
        else:
            no_improve += 1

        T = max(T_min, T * 0.9995)
        if no_improve > 2000:
            T = max(T, 0.001)
            no_improve = 0

    return best_f, best_score


def peak_reduction_step(f, n_steps=1):
    """Direct peak reduction: identify contributing elements at peak lag and adjust."""
    f = normalize_nonneg(f.copy())
    n = len(f)
    for _ in range(n_steps):
        conv = np.convolve(f, f)
        k_star = int(conv.argmax())
        peak_val = conv[k_star]

        # Find contributing pairs
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_min > j_max:
            break

        # Contributions: f[j] * f[k_star - j] for j in [j_min, j_max]
        js = np.arange(j_min, j_max + 1)
        partners = k_star - js
        contribs = f[js] * f[partners]
        total = contribs.sum()
        if total < 1e-15:
            break

        # Reduce elements with highest contribution, increase others
        top_frac = max(1, len(js) // 4)
        top_idx = js[np.argsort(contribs)[-top_frac:]]

        # Amount to reduce
        reduce_amt = 0.01 * total / len(top_idx)
        increase_total = reduce_amt * len(top_idx)
        other_mask = np.ones(n, dtype=bool)
        other_mask[top_idx] = False
        other_count = other_mask.sum()
        if other_count == 0:
            break
        increase_per = increase_total / other_count * f[other_mask].mean() / max(f[other_mask].mean(), 1e-10)

        f[top_idx] -= reduce_amt
        f[other_mask] += increase_per
        f = normalize_nonneg(f)

    return f


def make_initial(n, method='uniform'):
    """Generate initial sequences of various types."""
    if method == 'uniform':
        f = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0.0, np.pi, n)
        f = 0.5 * (1.0 + np.cos(x))
    elif method == 'gaussian':
        x = np.arange(n, dtype=np.float64)
        f = np.exp(-0.5 * ((x - n / 2.0) / (n / 6.0)) ** 2)
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        f = (np.exp(-0.5 * ((x - n / 4.0) / (n / 10.0)) ** 2) +
             np.exp(-0.5 * ((x - 3.0 * n / 4.0) / (n / 10.0)) ** 2))
    elif method == 'alternating':
        f = np.ones(n)
        f[::2] = 0.2
    elif method == 'random':
        rng = np.random.default_rng(123)
        f = rng.random(n) + 0.1
    elif method == 'arcsine':
        x = np.linspace(0.0, np.pi, n)
        f = np.sin(x) ** 2
    elif method == 'plateau':
        f = np.zeros(n)
        f[n//4:3*n//4] = 1.0
    else:
        f = np.ones(n)
    return normalize_nonneg(np.maximum(f, 0.01))


def multi_start_gd(f_starts, budget_per_start):
    """Run GD from multiple starting points."""
    best_f = None
    best_sc = np.inf
    for i, f0 in enumerate(f_starts):
        if time.monotonic() > f_starts[0][1] + budget_per_start * (i + 1):
            break
        f_res, sc_res = softmax_gd(f0[0], f0[1])
        if sc_res < best_sc:
            best_sc = sc_res
            best_f = f_res
