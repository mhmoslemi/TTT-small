    n = len(f)
    N = next_power_of_2(2*n - 1)
    F = np.fft.rfft(f, N)
    conv = np.fft.irfft(F * F, N)[:2*n-1]
    
    # Compute weights
    log_w = beta * conv - beta * conv.max()  # for numerical stability
    w = np.exp(log_w)
    w = w / w.sum()
    
    # Compute gradient via correlation
    W = np.fft.rfft(w, N)
    grad = 2 * np.fft.irfft(W * np.conj(F), N)[:n]
    
    # Update
    f_new = np.maximum(0, f - alpha * grad)
    s = f_new.sum()
    if s < 1e-10:
        return f, conv.max()
    f_new = f_new / s
    return f_new, conv.max()

        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0

        frac = i / max(1, n_iter - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad_mom = (1 - mu) * grad + mu * grad_prev
        grad_prev = grad.copy()

        s_new = f_new.sum()

            alpha *= 0.5
            continue
        f_new = f_new / s_new

        sc_new = fast_score(f_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_f = f_new.copy()
            f = f_new
        else:
            # Rejected: reduce alpha
            alpha *= 0.8
            # Keep f unchanged

"""
FFT-accelerated smoothed gradient descent for minimizing the autocorrelation
peak of a non-negative sequence. Uses continuation on the smoothing parameter
beta, adaptive step size with momentum, and multi-start with LP refinement.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Score using FFT for speed."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(f, N)
    conv = np.fft.irfft(F * F, N)[:2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def smooth_grad(f, beta):
    """Compute smoothed gradient and convolution."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(f, N)
    conv = np.fft.irfft(F * F, N)[:2 * n - 1]
    mx = conv.max()
    lw = beta * (conv - mx)
    w = np.exp(lw)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    W = np.fft.rfft(w, N)
    grad = 2.0 * np.fft.irfft(W * np.conj(F), N)[:n]
    return grad, conv
