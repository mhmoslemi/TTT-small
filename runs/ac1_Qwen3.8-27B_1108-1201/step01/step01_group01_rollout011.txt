def _score(a):

def _soft_max_grad(a, beta):
    """Compute gradient of soft max of convolution using FFT."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    weights = np.exp(beta * (b - b_max))
    weights /= np.sum(weights)
    
    # grad[j] = 2 * sum_m a[m] * weights[j+m] for j=0..n-1

    total_len = (2*n - 1) + n - 1  # len(weights) + n - 1

1. Find the current peak value T = max(b_k).
2. Find all k where b_k is close to T (within some tolerance).
3. Adjust a to decrease these peaks while keeping the sum fixed.

The adjustment direction is the negative gradient of the active peaks, projected to maintain the sum constraint.

This is essentially what the previous program does, but with the soft-max approach, we get a smoother signal.

Let me write the complete code:

"""
Soft-max gradient descent for minimizing the normalized autocorrelation peak
of a non-negative sequence. Uses FFT-based gradient computation and adaptive
beta schedule. Multiple anti-symmetric starting points with cosine profiles.
"""

import numpy as np
import time


def _compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    b = np.convolve(a, a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * np.max(b) / (sum_a ** 2)


def _compute_conv(a):
    """Compute full linear convolution of a with itself."""
    return np.convolve(a, a)


def _soft_max_gradient(a, beta, conv=None):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*b_k)) w.r.t. a.
    Uses FFT-based cross-correlation for O(n log n) computation.
    """
    n = len(a)
    if conv is None:
        conv = np.convolve(a, a)
    
    # Compute soft-max weights
    conv_max = np.max(conv)
    log_weights = beta * (conv - conv_max)
    # Numerical stability: subtract max
    log_weights -= np.max(log_weights)
    weights = np.exp(log_weights)
    weights /= np.sum(weights)
    
    # Gradient via FFT cross-correlation
    # grad[j] = 2 * sum_m a[m] * weights[j+m] for j = 0..n-1
    total_len = (2 * n - 1) + n - 1
    N = 1
    while N < total_len:
        N *= 2
    
    F_weights = np.fft.rfft(weights, N)
    a_flipped = a[::-1]
    F_af = np.fft.rfft(a_flipped, N)
    conv_full = np.fft.irfft(F_weights * F_af, N)[:total_len]
    grad = 2.0 * conv_full[n - 1:2 * n - 1]
    
    return grad


def _project_simplex(v, target=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_cosine_profile(n, eps, freq=1.0, phase=0.0):
    """
    Create an anti-symmetric cosine profile.
    a[i] = 1 + eps * cos(freq * pi * i / (n-1) + phase)
    Normalized to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(freq * np.pi * i / (n - 1) + phase)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_sigmoid_profile(n, steepness=5.0, offset=0.0):
    """
    Create a sigmoid-like anti-symmetric profile.
    a[i] = 1 / (1 + exp(-steepness * (2*i/(n-1) - 1 + offset)))
    Normalized to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / (n - 1) - 1.0 + offset
    a = 1.0 / (1.0 + np.exp(-steepness * x))
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_polyProfile(n, power=2.0):
    """
    Create a polynomial anti-symmetric profile.
    a[i] = (i/(n-1))^power + (1 - i/(n-1))^power  (symmetric, peaked at ends)
    Or a[i] = (i/(n-1))^power (anti-symmetric, increasing)
    """
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = t ** power
    a = np.maximum(a, 1e-10)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _gradient_descent(a, deadline, rng, beta_start=1.0, beta_end=50.0, 
                       lr=0.01, batch_peaks=50, evaluate_interval=500):
    """
    Gradient descent with soft-max objective and adaptive beta.
    """
    n = len(a)
    a = np.array(a, dtype=np.float64)
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _compute_score(best_a)
    best_conv = np.convolve(best_a, best_a)
    
    step = 0
    last_eval = time.monotonic()
    
    # Beta schedule: logarithmic ramp
    total_steps_est = 10000  # rough estimate
    
    while time.monotonic() < deadline:
        # Adaptive beta
        progress = min(1.0, step / max(1, total_steps_est))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient
        conv = np.convolve(a, a)
        grad = _soft_max_gradient(a, beta, conv)
        
        # Center gradient (to maintain sum constraint)
        grad -= np.mean(grad)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Update
        a_new = a - lr * grad / grad_norm
        
        # Project onto simplex
        a_new = _project_simplex(a_new, 1.0)
        
        # Check if sum is still valid
        if np.sum(a_new) < 1e-10:
            a = best_a.copy()
            break
        
        a = a_new
        step += 1
        
        # Periodic evaluation
        if step % evaluate_interval == 0 or time.monotonic() - last_eval > 5.0:
            cur_score = _compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"  Step {step}: New best score = {best_score:.6f} (beta={beta:.2f})")
            last_eval = time.monotonic()
    
    # Final evaluation
    cur_score = _compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    
    return best_a, best_score

def _perturb_smooth(a, rng, scale=0.05):
    """Add smooth random noise to a sequence and re-project."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 4), size=min(5, n // 2))
    noise = np.zeros(n)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
    max_noise = np.max(np.abs(noise))
    if max_noise > 1e-15:
        noise /= max_noise
    a_new = a + scale * noise
    a_new = _project_simplex(a_new, 1.0)
    return a_new
