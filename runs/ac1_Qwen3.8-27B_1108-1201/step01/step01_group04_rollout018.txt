import numpy as np
import time

def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def _compute_conv(a):
    """Compute full convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def _compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = _compute_conv(a)
    max_c = c.max()
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (S * S)

def _softmax_gradient(a, tau):
    """Compute the softmax-weighted gradient of the max of conv(a,a)."""
    n = len(a)
    c = _compute_conv(a)
    # Softmax weights
    c_shifted = c - c.max()
    w = np.exp(c_shifted / tau)
    w /= w.sum()
    # Gradient: grad[i] = sum_j w[i+j] * a[j]
    # Using the convolution identity: grad = conv(a[::-1], w)[n-1:2n-1]
    a_flip = a[::-1]
    full = _fft_convolve(a_flip, w)
    grad = full[n-1:2*n-1]
    return grad, c.max()

def _optimize(a0, deadline, n_iters=5000):
    """Optimize a single sequence using softmax gradient descent."""
    n = len(a0)
    a = np.array(a0, dtype=np.float64)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S  # normalize to sum 1
    
    best_a = a.copy()
    best_score = _compute_obj(a)
    
    # Annealing schedule
    tau_start = n * 0.1
    tau_end = 1.0
    lr = 0.1
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal temperature
        progress = it / n_iters
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Learning rate (cosine annealing)
        lr_it = lr * 0.5 * (1 + np.cos(np.pi * progress))
        lr_it = max(lr_it, 0.001)
        
        # Compute gradient
        grad, max_c = _softmax_gradient(a, tau)
        
        # Update
        a = a - lr_it * grad
        a = np.maximum(a, 0)
        
        # Renormalize
        S = a.sum()
        if S < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / S
        
        # Check improvement (every 100 iterations)
        if (it + 1) % 100 == 0:
            score = _compute_obj(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score
