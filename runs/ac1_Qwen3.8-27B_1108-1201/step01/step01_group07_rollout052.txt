```python
import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_S(a, w):
    """Compute S[j] = sum_k w[k] * a[k-j] for j = 0..n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    # W[k] = w[k] for 0<=k<n, 0 otherwise
    W = np.zeros(fft_size)
    W[:n] = w
    # A_rev[k] = a[n-1-k] for 0<=k<n, 0 otherwise
    A_rev = np.zeros(fft_size)
    A_rev[:n] = a[::-1]
    C = np.fft.irfft(np.fft.rfft(W) * np.fft.rfft(A_rev), fft_size)
    # S[j] = C[n-1+j]
    return C[n-1:n-1+n]

def log_softmax_obj(x, T):
    """Objective: 2n * soft_max(T, r(softmax(x)))."""
    # Shift for numerical stability
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    a = a / a_sum
    n = len(a)
    r = fast_convolve(a)
    # Soft max
    Tr = T * r
    Tr_max = Tr.max()
    log_sum_exp = Tr_max + np.log(np.sum(np.exp(Tr - Tr_max)))
    soft_max = log_sum_exp / T
    obj = 2.0 * n * soft_max
    return obj

def log_softmax_grad(x, T):
    """Gradient of log_softmax_obj w.r.t. x."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    a = a / a_sum
    n = len(a)
    r = fast_convolve(a)
    
    # Soft max weights
    Tr = T * r
    Tr_max = Tr.max()
    w = np.exp(Tr - Tr_max)
    w_sum = w.sum()
    w = w / w_sum  # normalized weights
    
    # S[j] = sum_k w[k] * a[k-j]
    S = compute_S(a, w)
    
    # w_dot_r = sum_k w[k] * r[k]
    w_dot_r = np.dot(w, r)
    
    # dF/dx[j] = 4n * a[j] * (S[j] - w_dot_r)
    grad = 4.0 * n * a * (S - w_dot_r)
    
    return grad

def optimize_log_softmax(a0, T_schedule, time_limit, method='L-BFGS-B'):
    """Optimize using log-softmax parametrization."""
    n = len(a0)
    x0 = np.log(np.maximum(a0, 1e-30))
    
    best_a = a0.copy()
    best_obj = 2.0 * n * fast_convolve(a0).max()
    
    t_start = time.monotonic()
    
    for T in T_schedule:
        if time.monotonic() - t_start > time_limit:
            break
        
        def obj(x):
            return log_softmax_obj(x, T)
        
        def grad(x):
            return log_softmax_grad(x, T)
        
        result = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                         options={'maxiter': 500, 'maxfun': 10000})
        
        # Convert back
        x_opt = result.x - result.x.max()
        a_opt = np.exp(x_opt)
        a_opt /= a_opt.sum()
        
        current_obj = 2.0 * n * fast_convolve(a_opt).max()
        
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a_opt.copy()
        
        # Use this as starting point for next T
        x0 = np.log(np.maximum(a_opt, 1e-30))
        
        print(f"  T={T:.1f}, obj={current_obj:.6f}, best={best_obj:.6f}, time={time.monotonic()-t_start:.1f}s")
    
    return best_a, best_obj
```

```python
def project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    # Find tau such that sum(max(v - tau, 0)) = 1
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    # Find rho = max{j: v_j - (sum_{i<=j} v_i)/j > 0}
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    tau = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - tau, 0)

def projected_gradient_step(a, T, eta):
    """One step of projected gradient descent."""
    n = len(a)
    r = fast_convolve(a)
    Tr = T * r
    Tr_max = Tr.max()
    w = np.exp(Tr - Tr_max)
    w /= w.sum()
    
    # S[i] = sum_k w[k] * a[k-i]
    S = compute_S(a, w)
    
    grad = 4.0 * n * S
    
    a_new = a - eta * grad
    return project_simplex(a_new)
```

```python
def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_S(a, w):
    """Compute S[i] = sum_k w[k] * a[k-i] for i = 0..n-1 using FFT."""
    n = len(a)
    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    W = np.zeros(fft_size)
    W[:n] = w
    A_rev = np.zeros(fft_size)
    A_rev[:n] = a[::-1]
    C = np.fft.irfft(np.fft.rfft(W) * np.fft.rfft(A_rev), fft_size)
    return C[n-1:n-1+n]


def project_simplex(v):
    """Project v onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    idx = np.arange(1, n+1)
    rho_mask = u * idx > (cv - 1)
    if not np.any(rho_mask):
        return np.ones(n) / n
    rho = np.nonzero(rho_mask)[0][-1]
    tau = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - tau, 0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Projected gradient descent on the simplex
- Log-softmax parametrization + L-BFGS-B (scipy)
- Multiplicative mirror descent
- Adaptive temperature annealing and restarts

Combines multiple optimizers with diverse initializations.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_cross_corr(a, w):
    """
    Compute S[i] = sum_k w[k] * a[k-i] for i = 0..n-1.
    (a[m] = 0 for m outside [0, n-1])
    Uses FFT for efficiency.
    """
    n = len(a)
    total = 2 * n
    fft_size = 1 << (total - 1).bit_length()
    W = np.zeros(fft_size)
    W[:n] = w
    A_rev = np.zeros(fft_size)
    A_rev[:n] = a[::-1]
    C = np.fft.irfft(np.fft.rfft(W) * np.fft.rfft(A_rev), fft_size)
    # S[i] = C[n-1+i]
    return C[n-1:n-1+n]


def project_simplex(v):
    """Project vector v onto the probability simplex {a: a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    mask = u * idx > (cv - 1)
    if not np.any(mask):
        return np.ones(n) / n
    rho = np.nonzero(mask)[0][-1]
    tau = (cv[rho] - 1) / (rho + 1)
    return np.maximum(v - tau, 0)


def get_obj_normalized(a):
    """Compute 2n * max(auto_conv(a)) assuming sum(a) = 1."""
    n = len(a)
    r = fast_convolve(a)
    return 2.0 * n * r.max()


def get_gradient_soft_max(a, T):
    """
    Compute gradient of 2n * soft_max(T, r(a)) w.r.t. a (without constraint).
    Returns (gradient, r, w) for reuse.
    """
    n = len(a)
    r = fast_convolve(a)
    Tr = T * r
    Tr_max = Tr.max()
    exp_Tr = np.exp(Tr - Tr_max)
    w_sum = exp_Tr.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(r))
        w[np.argmax(r)] = 1.0
    else:
        w = exp_Tr / w_sum
    
    S = compute_cross_corr(a, w)
    grad = 4.0 * n * S
    return grad, r, w


def optimize_projected_gradient(a0, time_limit, T0=50.0, eta0=0.05, max_T=2000.0):
    """
    Projected gradient descent on the probability simplex.
    Minimizes 2n * soft_max(T, r(a)) with annealing on T.
    """
    a = np.maximum(a0, 0).astype(np.float64)
    n = len(a)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.ones(n) / n
    else:
        a /= a_sum
    
    best_a = a.copy()
    best_obj = get_obj_normalized(a)
    
    t_start = time.monotonic()
    patience = 0
    max_patience = 80
    eta = eta0
    T = T0
    
    while time.monotonic() - t_start < time_limit:
        grad, r, w = get_gradient_soft_max(a, T)
        
        # Backtracking line search
        current_obj = get_obj_normalized(a)
        found = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            alpha = eta * factor
            a_trial = project_simplex(a - alpha * grad)
            obj_trial = get_obj_normalized(a_trial)
            if obj_trial < current_obj - 1e-12:
                a = a_trial
                found = True
                break
        
        if not found:
            eta *= 0.5
            if eta < 1e-8:
                break
        
        current_obj = get_obj_normalized(a)
        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            patience = 0
            T = min(T * 1.5, max_T)
            eta = max(eta * 0.7, 1e-6)
            # Optionally add noise to escape
            if T > 500:
                rng_escape = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                noise = rng_escape.random(n) * 0.02
                a = project_simplex(a + noise)
    
    return best_a, best_obj


def log_softmax_objective(x, T, n):
    """Objective function for log-softmax optimization."""
    x_s = x - x.max()
    a = np.exp(x_s)
    s = a.sum()
    if s < 1e-30:
        return 2.0 * n
    a = a / s
    r = fast_convolve(a)
    Tr = T * r
    Tr_m = Tr.max()
    lse = Tr_m + np.log(np.sum(np.exp(Tr - Tr_m)))
    return 2.0 * n * lse / T


def log_softmax_gradient(x, T, n):
    """Gradient of log-softmax objective w.r.t. x."""
    x_s = x - x.max()
    a = np.exp(x_s)
    s = a.sum()
    if s < 1e-30:
        return np.zeros_like(x)
    a = a / s
    r = fast_convolve(a)
    Tr = T * r
    Tr_m = Tr.max()
    exp_Tr = np.exp(Tr - Tr_m)
    w_sum = exp_Tr.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(r))
        w[np.argmax(r)] = 1.0
    else:
        w = exp_Tr / w_sum
    
    S = compute_cross_corr(a, w)
    w_dot_r = np.dot(w, r)
    return 4.0 * n * a * (S - w_dot_r)


def optimize_log_softmax(a0, time_limit, T_schedule):
    """
    Optimize using log-softmax parametrization + L-BFGS-B.
    a0 is the initial sequence (will be normalized).
    T_schedule is a list of temperature values (increasing = sharper).
    """
    n = len(a0)
    a_norm = np.maximum(a0, 0).astype(np.float64)
    s = a_norm.sum()
    if s < 1e-12:
        a_norm = np.ones(n) / n
    else:
        a_norm /= s
    
    x = np.log(np.maximum(a_norm, 1e-30))
    
    best_a = a_norm.copy()
    best_obj = get_obj_normalized(a_norm)
    
    t_start = time.monotonic()
    
    for T in T_schedule:
        if time.monotonic() - t_start > time_limit:
            break
        time_left = time_limit - (time.monotonic() - t_start)
        if time_left < 3:
            break
        
        maxiter = max(50, int(min(300, 50 * time_left)))
        
        result = minimize(
            log_softmax_objective, x,
            args=(T, n),
            jac=lambda xx: log_softmax_gradient(xx, T, n),
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8}
        )
        
        x_opt = result.x - result.x.max()
        a_opt = np.exp(x_opt)
        a_opt /= a_opt.sum()
        
        current_obj = get_obj_normalized(a_opt)
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a_opt.copy()
        
        x = np.log(np.maximum(a_opt, 1e-30))
        elapsed = time.monotonic() - t_start
        print(f"      LS: T={T}, obj={current_obj:.6f}, best={best_obj:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_obj
