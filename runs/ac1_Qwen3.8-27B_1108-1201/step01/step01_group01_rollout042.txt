Adaptive subgradient method with FFT-accelerated gradients for minimizing
the normalized autocorrelation peak of non-negative sequences.
Uses soft active-set weighting, adaptive step sizes, periodic reheating,
and multiple starting points.

import numpy as np

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    N = 1
    while N < n + m - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)
    return result[:n + m - 1]


def _fft_xcorr(w, a):
    """
    Compute cross-correlation c[i] = sum_j w[j+i] * a[j] for i = 0, ..., len(a)-1.
    Uses FFT for O(n log n) complexity.
    """
    n_w = len(w)
    n_a = len(a)
    N = 1
    while N < n_w + n_a - 1:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(W * np.conj(A), N)
    return c[:n_a]
