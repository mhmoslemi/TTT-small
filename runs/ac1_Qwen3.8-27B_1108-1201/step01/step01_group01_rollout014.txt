import numpy as np
import time

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _convolve_fft(a, b):
    """Compute linear convolution using FFT."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]

def _score(a):
    """Compute 2*n*max(convolve(a,a)) for normalized a."""
    n = len(a)
    c = _convolve_fft(a, a)
    return 2.0 * n * np.max(c)

def _grad_at_peak(a, k):
    """Gradient of convolve(a,a)[k] w.r.t. a."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros(n)
    grad[idx] = 2.0 * a[k - idx]
    return grad

def _subgrad_optimize(a, deadline, rng, M=10, eta_init=None, use_momentum=True):
    """Subgradient descent with momentum to minimize max autocorrelation."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    
    if eta_init is None:
        eta_init = 0.1 / np.sqrt(n)
    
    eta = eta_init
    best = a.copy()
    best_score = _score(best)
    v = np.zeros(n)  # momentum
    beta = 0.9
    no_improve = 0
    iteration = 0
    
    while time.monotonic() < deadline:
        c = _convolve_fft(a, a)
        c_max = np.max(c)
        cur_score = 2.0 * n * c_max
        
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive step size
        if no_improve > 200:
            eta *= 0.9
            v *= 0.5
            no_improve = 0
            if eta < 1e-12:
                break
        
        # Find top M peaks
        if M >= len(c):
            k_idx = np.arange(len(c))
            wts = np.ones(len(c)) / len(c)
        else:
            k_idx = np.argpartition(c, -M)[-M:]
            c_sel = c[k_idx]
            # Softmax weights
            wts = np.exp(-3.0 * (c_max - c_sel) / max(c_max, 1e-15))
            wts /= np.sum(wts)
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for ki in range(M):
            if wts[ki] < 1e-8:
                continue
            grad += wts[ki] * _grad_at_peak(a, k_idx[ki])
        
        # Center for sum constraint
        grad -= np.mean(grad)
        
        # Momentum update
        if use_momentum:
            v = beta * v + (1 - beta) * grad
            step = v
        else:
            step = grad
        
        step_norm = np.linalg.norm(step)
        if step_norm < 1e-15:
            break
        
        a_new = a - eta * step / step_norm
        a_new = _project_simplex(a_new, 1.0)
        
        # Check if we made progress
        a = a_new
        iteration += 1
    
    return best, best_score

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target} using bisection-free method."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            if time.monotonic() >= deadline:
                break
            for delta in steps:
                a_trial = a.copy()
                a_trial[i] = max(0.0, a_trial[i] + delta)
                # Adjust another coordinate to maintain sum
                j = (i + 1) % n
                adj = a[i] - a_trial[i]
                if abs(adj) > 1e-15:
                    a_trial[j] = max(0.0, a_trial[j] + adj)
                    S2 = np.sum(a_trial)
                    if S2 > 1e-10:
                        a_trial /= S2
                else:
                    continue
                sc = _score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break

        if not improved:
            # Try larger steps
            steps = [0.005, 0.002, 0.001, -0.005, -0.002, -0.001]

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-start FFT-accelerated subgradient descent with momentum,
    adaptive peak selection, and polishing phases.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    # Get parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)

    n = parent.size

    # Normalize parent
    parent_S = np.sum(parent)
    if parent_S > 1e-10:
        parent_norm = parent / parent_S
    else:
        parent_norm = np.ones(n) / n

    print(f"Sequence length: {n}")
    print(f"Parent score: {evaluate_sequence([float(x) for x in parent_norm]):.8f}")

    global_best = parent_norm.copy()
    global_best_score = _score_normalized(parent_norm)

    # Generate diverse starting points
    starts = [parent_norm.copy()]

    # Two-level anti-symmetric
    for eps in [0.3, 0.5, 0.7, 0.9]:
        starts.append(_make_two_level(n, eps))

    # Smooth cosine
    for freq, amp in [(3, 0.6), (5, 0.7), (7, 0.5), (2, 0.8)]:
        starts.append(_make_smooth_cos(n, freq, 0.0, amp))

    # Four-level
    starts.append(_make_four_level(n, 0.6))
    starts.append(_make_four_level(n, 0.9))

    # Multi-frequency random
    for _ in range(3):
        starts.append(_make_multi_freq(n, rng, n_freqs=4, max_amp=0.6))

    # Perturbed parent
    starts.append(_perturb_smooth(parent_norm, rng, scale=0.1))
    starts.append(_perturb_smooth(parent_norm, rng, scale=0.15))

    # Evaluate all starting points
    print("\nStarting point scores:")
    for i, s in enumerate(starts):
        sc = _score_normalized(s)
        print(f"  Start {i}: {sc:.8f}")
        if sc < global_best_score:
            global_best_score = sc
            global_best = s.copy()

    # Time allocation
    remaining = total_deadline - time.monotonic()
    n_starts = len(starts)

    # Phase 1: Multi-start optimization (60% of time)
    phase1_time = remaining * 0.6
    phase1_deadline = time.monotonic() + phase1_time

    # Sort starts by score (best first), give best more time
    start_scores = [_score_normalized(s) for s in starts]
    order = np.argsort(start_scores)

    # Allocate time: top 3 get 25% each, rest share remaining
    time_allocs = np.zeros(n_starts)
    top3_time = phase1_time * 0.75 / 3
    rest_time = phase1_time * 0.25 / max(1, n_starts - 3)
    for rank, idx in enumerate(order):
        if rank < 3:
            time_allocs[idx] = top3_time
        else:
            time_allocs[idx] = rest_time

    print(f"\nPhase 1: Multi-start optimization ({phase1_time:.0f}s)")
    for idx in order:
        if time.monotonic() >= phase1_deadline:
            break
        t_alloc = time_allocs[idx]
        start_deadline = min(time.monotonic() + t_alloc, phase1_deadline)
        label = f"Start{idx}"

        best_a, best_s = _subgrad_optimize(
            starts[idx], start_deadline, rng,
            M_init=5, M_max=50,
            eta_init=2.0 / np.sqrt(n),
            momentum_init=0.85,
            phase_label=label
        )

        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"  {label}: NEW BEST! Score: {global_best_score:.8f}")
        else:
            print(f"  {label}: Score {best_s:.8f} (best={global_best_score:.8f})")

    # Phase 2: Polish the best (25% of time)
    remaining = total_deadline - time.monotonic()
    phase2_time = remaining * 0.35
    phase2_deadline = time.monotonic() + phase2_time

    if remaining > 10:
        print(f"\nPhase 2: Polishing (M=30-80, small eta) ({phase2_time:.0f}s)")
        best_a, best_s = _subgrad_optimize(
            global_best, phase2_deadline, rng,
            M_init=20, M_max=80,
            eta_init=0.3 / np.sqrt(n),
            momentum_init=0.7,
            momentum_min=0.3,
            cool_rate=0.95,
            phase_label="Polish"
        )
        if best_s < global_best_score:
            global_best_score = best_s
            global_best = best_a.copy()
            print(f"  Polish: NEW BEST! Score: {global_best_score:.8f}")
        else:
            print(f"  Polish: Score {best_s:.8f} (best={global_best_score:.8f})")

    # Phase 3: Diversification + re-optimization (15% of time)
    remaining = total_deadline - time.monotonic()
    if remaining > 30:
        print(f"\nPhase 3: Diversification ({remaining:.0f}s)")
        n_div = max(1, int(remaining / 80))
        per_div = remaining / n_div

        for d in range(n_div):
            if time.monotonic() >= total_deadline:
                break
            # Different perturbation scales
            scale = 0.05 + 0.1 * d / max(1, n_div)
            a_pert = _perturb_smooth(global_best, rng, scale=scale)
            div_deadline = min(time.monotonic() + per_div, total_deadline)

            best_a_d, best_s_d = _subgrad_optimize(
                a_pert, div_deadline, rng,
                M_init=10, M_max=60,
                eta_init=1.0 / np.sqrt(n),
                momentum_init=0.8,
                phase_label=f"Div{d}"
            )
            if best_s_d < global_best_score:
                global_best_score = best_s_d
                global_best = best_a_d.copy()
                print(f"  Div{d}: NEW BEST! Score: {global_best_score:.8f}")
            else:
                print(f"  Div{d}: Score {best_s_d:.8f} (best={global_best_score:.8f})")

    # Final evaluation
    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.8f}")
    print(f"  Sequence length: {len(result)}")
    print(f"  Max element: {max(result):.6f}")
    print(f"  Min element: {min(result):.6f}")
    print(f"  Sum: {sum(result):.6f}")

    return result
