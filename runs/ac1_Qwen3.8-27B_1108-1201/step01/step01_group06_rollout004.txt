```python
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_pnorm_gradient(a, b, p):
    """Compute gradient of the p-norm of b = convolve(a,a)."""
    n = len(a)
    # Clip b to avoid issues with zero values
    b_safe = np.maximum(b, 1e-30)
    # Weights
    bp = b_safe ** (p - 1)
    # Normalization factor: (sum b^p)^(1/p - 1)
    norm = (np.sum(b_safe ** p)) ** (1.0 / p - 1.0)
    W = bp * norm  # length 2n-1
    
    # Compute gradient via FFT cross-correlation
    a_rev = a[::-1]
    N = 1 << (3 * n - 3).bit_length()
    A = fft(a_rev, N)
    Wf = fft(W, N)
    conv_full = np.real(ifft(A * Wf))
    R = conv_full[n - 1: 2 * n - 1]
    grad = 2.0 * R
    return grad

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
```

Minimize 2*n*max_conv/sum^2 using p-norm annealing with FFT-based gradients.

Key ideas:
- Replace the nonsmooth max with a smooth p-norm, annealing p from 2 to a large value.
- Compute gradients in O(n log n) using FFT cross-correlation.
- Use Adam optimizer with simplex projection.
- Try multiple initializations and lengths.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_pnorm_gradient(a, b, p):
    n = len(a)
    b_safe = np.maximum(b, 1e-30)
    bp_minus1 = b_safe ** (p - 1)
    norm_factor = (np.sum(b_safe ** p)) ** (1.0 / p - 1.0)
    W = bp_minus1 * norm_factor
    
    a_rev = a[::-1]
    N = 1 << (3 * n - 3).bit_length()
    A = fft(a_rev, N)
    Wf = fft(W, N)
    conv_full = np.real(ifft(A * Wf))
    R = conv_full[n - 1: 2 * n - 1]
    return 2.0 * R


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.full(n, S / n)
    return a * (S / s)


def optimize_pnorm(a_init, max_time, rng, p_schedule=None):
    """Optimize using p-norm annealing with Adam."""
    a = safe_normalize(a_init, len(a_init))
    n = len(a)
    S = np.sqrt(2.0 * float(n))
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    if p_schedule is None:
        p_schedule = [2, 3, 4, 6, 9, 14, 21, 32, 50, 80, 130, 200]
    
    iters_per_stage = 300
    
    for p in p_schedule:
        if time.monotonic() - start > max_time:
            break
        lr = 2.0 * S / n
        
        for _ in range(iters_per_stage):
            if time.monotonic() - start > max_time:
                break
            t += 1
            
            b = fft_convolve_self(a)
            cur_max = b.max()
            if cur_max < best_score:
                best_score = cur_max
                best_local = a.copy()
            
            grad = compute_pnorm_gradient(a, b, p)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            grad = grad / gnorm
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            # Learning rate decay

            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = project_simplex(a_new, S)
            
            # Occasional random perturbation
            if rng.random() < 0.01:
              a = project_simplex(a + rng.normal(0, 0.01 * S / n, n), S)
    
    return best_local, best_score

"""
Minimize 2*n*max_conv/sum^2 using p-norm annealing with FFT-based gradients.

Key ideas:
- Replace nonsmooth max with smooth p-norm, annealing p from 2 to large values.
- Compute gradients in O(n log n) using FFT cross-correlation.
- Adam optimizer with simplex projection and normalized gradient.
- Multiple initializations (given, triangular, smooth, random) and lengths.
- Periodic random perturbations to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_pnorm_gradient(a, b, p):
    """Compute gradient of (sum b[k]^p)^(1/p) where b = convolve(a,a)."""
    n = len(a)
    b_safe = np.maximum(b, 1e-30)
    # Compute weights w[k] proportional to b[k]^(p-1)
    if p <= 1.5:
        W = np.ones(2 * n - 1)
    else:
        log_b = np.log(b_safe)
        log_W = (p - 1) * log_b
        log_W_max = np.max(log_W)
        W = np.exp(log_W - log_W_max)
        W_sum = W.sum()
        if W_sum <= 0:
            W = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            W = W / W_sum

    # Compute gradient via FFT: grad[m] = 2 * sum_{j=0}^{n-1} W[m+j] * a[j]
    # This equals convolve(a_rev, W, 'full')[m + n - 1]
    a_rev = a[::-1]
    N = 1 << (3 * n - 3).bit_length()
    A = fft(a_rev, N)
    Wf = fft(W, N)
    conv_full = np.real(ifft(A * Wf))
    R = conv_full[n - 1: 2 * n - 1]
    return 2.0 * R


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.full(n, S / n)
    return a * (S / s)


def generate_triangular(n):
    """Generate a triangular (bell-shaped) sequence."""
    if n % 2 == 1:
        half = n // 2
        a = np.concatenate([np.arange(1, half + 1), np.arange(half, 0, -1)]).astype(np.float64)
    else:
        half = n // 2
        a = np.concatenate([np.arange(1, half + 1), np.arange(half + 1, 1, -1)]).astype(np.float64)
    return a


def generate_smooth_bell(n, rng):
    """Generate a smooth bell-shaped sequence with random width."""
    x = np.linspace(0, 1, n)
    width = rng.uniform(0.15, 0.4)
    center = rng.uniform(0.3, 0.7)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = a + 0.05 * rng.random(n)  # small random perturbation
    return np.maximum(a, 1e-6)


def generate_parabolic(n, rng):
    """Parabolic shape with random parameters."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(np.pi * (x - 0.5) * rng.uniform(0.5, 1.5))
    a = a + 0.03 * rng.random(n)
    return np.maximum(a, 1e-6)


def get_initializations(rng):
    """Collect diverse initializations."""
    inits = []

    # Given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000)))
    except Exception:
        pass

    # Triangular
    for n in (4096, 8192):
        inits.append(("tri_%d" % n, generate_triangular(n)))

    # Smooth bells
    for n in (4096, 8192):
        inits.append(("bell_%d" % n, generate_smooth_bell(n, rng)))

    # Parabolic
    inits.append(("para_8192", generate_parabolic(8192, rng)))

    # Random (but not pure noise - slight smoothing)
    a_rand = rng.random(8192)
    # Light exponential smoothing
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])

    inits.append(("rand_8192", a_rand))

    # Zurich-like: product of two smooth functions
    x = np.linspace(0, 1, 4096)
    a_zu = (0.5 + 0.5 * np.sin(2 * np.pi * x)) * (0.5 + 0.3 * np.cos(4 * np.pi * x))
    inits.append(("zu_4096", np.maximum(a_zu, 1e-6)))

    # Wide triangle (low peak)
    n = 8192
    x = np.linspace(0, 1, n)
    a_wide = 1.0 - np.abs(x - 0.5) * 1.2  # trapezoid-like
    inits.append(("trap_8192", np.maximum(a_wide, 0.05)))

    return inits


def optimize_pnorm(a_init, max_time, rng, verbose=True):
    """Optimize using p-norm annealing with Adam."""
    a = safe_normalize(a_init, len(a_init))
    n = len(a)
    S = np.sqrt(2.0 * float(n))

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps_adam = 0.8, 0.999, 1e-8

    # p-norm annealing schedule
    p_schedule = [2.0, 3.0, 4.0, 6.0, 10.0, 16.0, 25.0, 40.0, 64.0, 100.0, 160.0, 250.0]
    iters_per_stage = 400

    base_lr = 1.0 * S / n

    total_stages = len(p_schedule)
    for stage, p in enumerate(p_schedule):
        if time.monotonic() - start > max_time:
            break
        # LR decays across stages (coarser in later stages)
        stage_decay = 1.0 - 0.3 * (stage / total_stages)
        lr = base_lr * stage_decay

        for it in range(iters_per_stage):
            if time.monotonic() - start > max_time:
                break
            t += 1

            b = fft_convolve_self(a)
            cur_max = b.max()
            if cur_max < best_score:
                best_score = cur_max
                best_local = a.copy()

            grad = compute_pnorm_gradient(a, b, p)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad = grad / gnorm

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            # Warm start: use larger LR initially
            lr_t = lr * min(1.0, 5.0 / (it + 1.0))
            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = project_simplex(a_new, S)

            # Random perturbation to escape local minima
            if (it + 1) % 100 == 0 and rng.random() < 0.3:
                perturbation_scale = 0.05 * S / n * (1.0 + stage / 5.0)
                a = project_simplex(a + rng.normal(0, perturbation_scale, n), S)

        if verbose:
            elapsed = time.monotonic() - start
            print("    p=%.0f, best_max=%.6f, elapsed=%.1fs" % (p, best_score, elapsed))

    return best_local, best_score


def polish_sequence(a_best, max_time, rng):
    """Further refine the best sequence with finer steps."""
    n = len(a_best)
    a = safe_normalize(a_best, n)
    S = np.sqrt(2.0 * float(n))

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()

    # Use very high p (essentially max) with small steps
    p = 500.0
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    base_lr = 0.3 * S / n

    while time.monotonic() - start < max_time:
        t += 1
        b = fft_convolve_self(a)
        cur_max = b.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        grad = compute_pnorm_gradient(a, b, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr_t = base_lr * (0.9995 ** t)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = project_simplex(a_new, S)

    return best_local, best_score


def coordinate_polish(a, max_time, rng):
    """Coordinate-wise refinement: try small perturbations to individual coordinates."""
    n = len(a)
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a, n)

    start = time.monotonic()
    b = fft_convolve_self(a)
    best_score = b.max()
    best_local = a.copy()

    # Focus on coordinates where a[i] is large (they contribute most to peaks)
    # and lags where convolution is high
    while time.monotonic() - start < max_time:
        # Pick a random subset of coordinates to try
        n_try = min(100, n // 10)
        # Weight by a[i] (large coordinates are more impactful)
        weights = a + 1e-6
        weights = weights / weights.sum()
        indices = rng.choice(n, size=n_try, replace=False, p=weights)

        for idx in indices:
            if time.monotonic() - start > max_time:
                break
            if a[idx] < 1e-10:
                continue
            # Try reducing this coordinate (it likely contributes to peaks)
            for frac in (0.9, 0.7, 0.5, 0.3):
                a_trial = a.copy()
                a_trial[idx] *= frac
                a_trial = project_simplex(a_trial, S)
                b_trial = fft_convolve_self(a_trial)
                if b_trial.max() < best_score:
                    best_score = b_trial.max()
                    best_local = a_trial.copy()
                    a = a_trial.copy()
                    break

    return best_local, best_score
