Now, let me think about what the optimal structure looks like. We want to minimize:
`max_k sum_j a[j] * a[k-j]` subject to `sum a = S` (fixed) and `a >= 0`.

At the optimum, we expect the "active" lags (where the max is achieved) to have equal values (by a complementary slackness argument). If only one lag is active, we could potentially reduce it further.

For the uniform sequence, the peak is at the middle lag (k = n-1), and the value is n (all products are 1, n of them). The adjacent lags have value n-1, etc.

To reduce the peak, we need to make the products at the peak lag smaller. One way is to make the sequence non-uniform so that the products at the peak lag are smaller on average.

Let me think about this differently. Let's define the "correlation matrix" C where C[k,j] = a[j] * a[k-j] (with a[i] = 0 for i < 0 or i >= n). Then the convolution values are the row sums of C. We want to minimize the maximum row sum.

This is related to the problem of "balancing" the entries of C so that all row sums are as equal as possible.

The total sum of all entries of C is (sum a)^2 = S^2. There are 2n-1 rows. So the average row sum is S^2/(2n-1). If we could make all row sums equal, the max would be S^2/(2n-1), and the objective would be 2n * S^2/((2n-1)*S^2) = 2n/(2n-1) ≈ 1.

So the theoretical lower bound is approximately 1 (approached as n → ∞). The target of 1.5027 suggests we're getting close to this bound for moderate n.

Actually wait, we can't make all row sums equal because the first and last rows have only 1 entry (a[0]^2 and a[n-1]^2), while the middle row has n entries. So the structure of the problem imposes constraints.

Let me think about this more carefully. The k-th row of C has `min(k+1, n, 2n-1-k)` entries. For k = 0, there's 1 entry: a[0]^2. For k = n-1, there are n entries.

If we want all row sums to be equal to some value V, then:
- Row 0: a[0]^2 = V, so a[0] = sqrt(V)
- Row n-1: sum_{j=0}^{n-1} a[j]*a[n-1-j] = V

The first condition forces a[0] = sqrt(V). Similarly, a[n-1] = sqrt(V) (from row 2n-2).

"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses alternating LP to minimize the peak convolution directly, combined with
gradient refinement and multiple restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def _lp_min_peak_cross(f, S, n):
    """
    Solve LP: minimize M subject to (f * g)(k) <= M for all k,
    sum(g) = S, g >= 0.
    Returns the optimal g and M.
    """
    # Variables: g[0], ..., g[n-1], M
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize M
    
    # Inequality constraints: (f * g)(k) - M <= 0 for k = 0, ..., 2n-2
    # A_ub has shape (2n-1, n+1)
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    b_ub = np.zeros(n_constraints)
    
    for k in range(n_constraints):
        # (f * g)(k) = sum_{j} f[j] * g[k-j] where 0<=j<n and 0<=k-j<n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[k, j] = f[k - j]
        A_ub[k, -1] = -1.0  # coefficient of M
    
    # Equality constraint: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: g >= 0, M >= 0
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x[:n]
            M = res.x[-1]
            return g, M
    except Exception:
        pass
    return None, None


def _alternating_lp(a, S, max_iters=50, deadline=None):
    """
    Alternating LP: alternate between optimizing left and right factors.
    Minimizes peak of a*a.
    """
    n = len(a)
    best_score = _compute_obj(a)
    best = a.copy()
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Step 1: Fix left factor = a, optimize right factor
        g, M1 = _lp_min_peak_cross(a, S, n)
        if g is None:
            break
        
        # Step 2: Fix right factor = g, optimize left factor
        h, M2 = _lp_min_peak_cross(g, S, n)
        if h is None:
            break
        
        # Update: use h as the new sequence (it's the result of two alternating steps)
        # Blend for stability
        a_new = 0.5 * a + 0.5 * h
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S)
        
        score = _compute_obj(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
        
        # Also try using h directly
        score_h = _compute_obj(h)
        if score_h < best_score:
            best_score = score_h
            best = h.copy()
            a_new = h.copy()
        
        a = a_new
        
        # Check for convergence
        if it > 5:
            prev_score = _compute_obj(best)
            if abs(prev_score - best_score) < 1e-8:
                break
    
    return best, best_score


def _gradient_step(a, lr=0.1, S=None):

    n = len(a)
    if S is None:
        S = np.sqrt(2.0 * n)
    a = _normalize(a, S)
    
    c = np.convolve(a, a)
    k_star = int(c.argmax())
    
    # Gradient of c[k_star] with respect to a[j]
    # c[k_star] = sum_j a[j] * a[k_star - j]
    # d/d a[i] = a[k_star - i] + a[i - k_star] (with bounds, and factor of 2 when i = k_star - i)
    grad = np.zeros(n)
    for j in range(n):
        k_j = k_star - j
        if 0 <= k_j < n:
            grad[j] += a[k_j]

            # is a[k-i] (from the first factor) + a[i-k] (from the second factor)

    # but wait, a is only defined on [0,n), so we need k-i in [0,n) for the first term
    # and i-k in [0,n) for the second... but i-k = -(k-i), so if k-i >= 0 then i-k <= 0.
    # So the two terms are: a[k-i] (if 0 <= k-i < n) + a[i-k] (if 0 <= i-k < n)
    # Note: k-i >= 0 means i <= k, and i-k >= 0 means i >= k.
    # So for i < k: derivative = a[k-i] (second term requires i-k >= 0, which is false)
    # For i > k: derivative = a[i-k] (first term requires k-i >= 0, which is false)
    # For i = k: derivative = a[0] + a[0] = 2*a[0]... wait that doesn't seem right.
    
    # Let me redo this. c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
    # Let's write it as sum_j a[j] * a[k-j] where the sum is over valid j.
    # d/d(a[i]) = (d a[i]/d a[i]) * a[k-i] + sum over j where k-j = i, i.e., j = k-i:
    #   a[j] * (d a[k-j]/d a[i]) = a[k-i] * 1
    # So d(c[k])/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i] if i <= k and k-i < n

    # So d(c[k])/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i] if 0 <= k-i < n and 0 <= i < n

    # i.e., (grad c[k])[i] = a[k-i] + a[i-k] (with a[m] = 0 for m < 0 or m >= n)

    # Project out the mean (to maintain the sum constraint)
    grad -= grad.mean()
    
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = _normalize(a, S)
    return a


def _softmax_gradient_step(a, lr=0.1, tau=30.0, S=None):
    """Gradient step using softmax-weighted peak (smoother)."""
    n = len(a)
    if S is None:
        S = np.sqrt(2.0 * n)
    a = _normalize(a, S)
    
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    
    # Gradient: sum_k w[k] * 2*a[k-i] (with bounds)
    grad = np.zeros(n)
    for i in range(n):
        # grad[i] = sum_k w[k] * 2 * a[k-i] (if 0 <= k-i < n)
        k_lo = max(0, i)  # k-i >= 0 -> k >= i
        k_hi = min(2*n-2, i + n - 1)  # k-i < n -> k < i+n
        if k_hi >= k_lo:
            grad[i] = 2.0 * np.dot(w[k_lo:k_hi+1], a[:k_hi-k_lo+1])
    
    grad -= grad.mean()
    a = a - lr * grad
    a = np.maximum(a, 0.0)
    a = _normalize(a, S)
    return a


def _optimize_with_lp(a0, deadline, max_lp_iters=100, max_grad_iters=500):

    n = len(a0)

    if a.sum() < 1e-6:

    a = _normalize(a, S)
    
    best = a.copy()
    best_score = _compute_obj(a)
    
    # Phase 1: Alternating LP
    for it in range(max_lp_iters):
        if time.monotonic() > deadline - 30.0:
            break
        
        g, M1 = _lp_min_peak_cross(a, S, n)
        if g is None:
            break
        h, M2 = _lp_min_peak_cross(g, S, n)
        if h is None:
            break
        
        # Try different blends
        for t in [1.0, 0.7, 0.5, 0.3]:
            a_new = (1.0 - t) * a + t * h
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, S)
            s = _compute_obj(a_new)
            if s < best_score:
                best_score = s
                best = a_new.copy()
                a = a_new
                break
        else:
            # No improvement from blending, try gradient step
            a = _gradient_step(a, lr=0.05, S=S)
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    # Phase 2: Gradient refinement
    lr = 0.05
    for it in range(max_grad_iters):
        if time.monotonic() > deadline - 10.0:
            break
        lr = 0.05 * (1.0 + np.cos(np.pi * it / max_grad_iters)) * 0.5
        lr = max(lr, 0.002)
        a = _gradient_step(a, lr=lr, S=S)
        if (it + 1) % 50 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    
    # Phase 3: More LP iterations
    for it in range(50):
        if time.monotonic() > deadline - 5.0:
            break
        g, M1 = _lp_min_peak_cross(a, S, n)
        if g is None:
            break
        h, M2 = _lp_min_peak_cross(g, S, n)
        if h is None:
            break
        a_new = 0.5 * a + 0.5 * h
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, S)
        s = _compute_obj(a_new)
        if s < best_score:
            best_score = s
            best = a_new.copy()
            a = a_new
    
    return best, best_score


def _generate_starts(n, rng):
    """Generate diverse starting sequences of length n."""
    starts = []
    i = np.arange(n, dtype=np.float64)
    
    # Uniform
    starts.append(np.ones(n))
    
    # Linearly increasing
    starts.append(np.linspace(0.1, 1.0, n))
    
    # Linearly decreasing
    starts.append(np.linspace(1.0, 0.1, n))
    
    # Quadratic
    starts.append((i / n) ** 2 + 0.1)
    
    # Exponential
    starts.append(np.exp(2.0 * i / n) + 0.1)
    
    # Cosine
    starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i / n))
    
    # Two peaks
    starts.append(0.1 + np.exp(-0.5 * ((i - n*0.25) / (n*0.1)) ** 2) + np.exp(-0.5 * ((i - n*0.75) / (n*0.1)) ** 2))
    
    # Random
    starts.append(rng.random(n) * 0.5 + 0.25)
    
    # Triangular
    starts.append(1.0 - 2.0 * np.abs(i / n - 0.5))
    
    # Power law
    starts.append((i + 1) ** 0.5 + 0.1)
    
    # Sine modulated
    starts.append(0.5 + 0.4 * np.sin(2 * np.pi * i * 3.0 / n) + 0.1)
    
    return starts

- `a[j] >= 0`

- Start with a feasible point `a^(0)`.

- Solve the resulting LP.
- Update and repeat.

This is the **successive linear programming (SLP)** approach. It should converge to a local minimum.
