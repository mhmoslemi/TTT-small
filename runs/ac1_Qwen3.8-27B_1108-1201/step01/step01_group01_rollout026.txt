def _fft_conv2(a):
    """Compute np.convolve(a, a) using FFT for speed."""
    n = len(a)
    N = 2 * n - 1
    fa = np.fft.rfft(a, n=N)
    return np.fft.irfft(fa * fa, n=N)


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) / sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    b = _fft_conv2(a)
    return 2.0 * n * np.max(b) / (S * S)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _gradient_at_lag(a, k):
    """Gradient of b_k w.r.t. a: grad[j] = 2*a[k-j] if 0<=k-j<n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def _subgradient_topk(a, K=10, temperature=5.0):
    """Compute weighted subgradient from top-K active lags."""
    b = _fft_conv2(a)
    b_max = np.max(b)
    n_b = len(b)
    
    if K >= n_b:
        k_idx = np.arange(n_b)
        wts = np.ones(n_b) / n_b
    else:
        k_idx = np.argpartition(b, -K)[-K:]
        b_sel = b[k_idx]
        wts = np.exp(-temperature * (b_max - b_sel) / max(b_max, 1e-15))
        wts /= np.sum(wts)
        k_idx = k_idx[np.argsort(wts)[::-1]]
    
    n = len(a)
    grad = np.zeros(n)
    for ki in range(len(k_idx)):

        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]
    
    return grad, b_max


def _subgradient_single(a):
    """Subgradient from the single maximum lag."""
    b = _fft_conv2(a)
    k_max = np.argmax(b)
    b_max = b[k_max]
    n = len(a)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    grad = np.zeros(n)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    # Center for sum constraint
    grad -= np.mean(grad)
    return grad, b_max


def _frank_wolfe_step(a, grad):
    """Standard Frank-Wolfe: move toward the vertex minimizing grad^T v."""
    # The vertex of the simplex minimizing grad^T v is e_j where j = argmin(grad)
    j = np.argmin(grad)
    return j


def _mirror_descent_step(a, grad, eta=0.1):
    """Mirror descent with KL divergence (exponential weighting)."""
    # v[j] = a[j] * exp(-grad[j]/eta) / sum
    log_a = np.log(np.maximum(a, 1e-30))
    log_v = log_a - grad / eta
    log_v -= np.max(log_v)  # numerical stability
    v = np.exp(log_v)
    v = v / np.sum(v)
    return v


def _make_two_level(n, eps):
    """Anti-symmetric two-level: first half (1+eps), second half (1-eps)."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_smooth_anti(n, eps):
    """Smooth anti-symmetric: 1 + eps*cos(pi*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_four_level(n, eps):
    """Four-level anti-symmetric profile."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2 * q] = 1.0 + eps / 3.0
    a[2 * q:3 * q] = 1.0 - eps / 3.0
    a[3 * q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_multiscale(n, base, scales):
    """Multi-scale anti-symmetric: combine multiple frequency components."""
    a = np.ones(n)
    i = np.arange(n, dtype=np.float64)
    for freq, amp in scales:
        a += amp * np.cos(2 * np.pi * freq * i / n)
    a = np.maximum(a, 1e-6)
    # Make anti-symmetric about the mean
    a = a - np.mean(a)
    a = a + abs(np.min(a)) + 1e-6
    S = np.sum(a)
    return a / S


def _optimize_subgrad_fw(a, deadline, rng, K=10, use_fw=True,温度=5.0):
    """
    Combined subgradient + Frank-Wolfe optimization.
    Returns (best_a, best_score).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    best = a.copy()
    best_score = _compute_score(best)
    
    eta = 0.5 / np.sqrt(n)
    fw_iter = 0
    no_improve = 0
    start_time = time.monotonic()
    
    while time.monotonic() < deadline:
        # Compute subgradient
        grad, b_max = _subgradient_topk(a, K=K, temperature=温度)
        grad_centered = grad - np.mean(grad)
        
        # Frank-Wolfe step combination
        if use_fw:
            # FW direction: toward vertex e_j
            j = np.argmin(grad_centered)
            e_j = np.zeros(n)
            e_j[j] = 1.0
            fw_dir = e_j - a
        else:
            fw_dir = -grad_centered
        
        # Compute step size with backtracking
        # Try the FW step size or a damped version
        t_fw = 2.0 / (fw_iter + 2)
        
        # Combine: use a mix of gradient and FW directions
        # Primary: FW step
        a_trial = (1 - t_fw) * a + t_fw * e_j if use_fw else a - eta * grad_centered / (np.linalg.norm(grad_centered) + 1e-15)
        
        if use_fw:
            a_trial = _project_simplex(a_trial, 1.0)
        else:
            a_trial = _project_simplex(a_trial, 1.0)
        
        # Evaluate
        score_trial = _compute_score(a_trial)
        
        if score_trial < best_score:
            best = a_trial.copy()
            best_score = score_trial
            no_improve = 0
        else:
            no_improve += 1
            # Reduce step size
            t_fw *= 0.5
            a_trial = (1 - t_fw) * a + t_fw * e_j if use_fw else a
            a_trial = _project_simplex(a_trial, 1.0)
        
        a = a_trial
        fw_iter += 1
        
        # Periodic cooling
        if no_improve >= 200:
            eta *= 0.7
            K = max(3, K // 2)
            no_improve = 0
            if eta < 1e-12:
                break
        
        # Every 1000 iterations, try mirror descent
        if fw_iter % 1000 == 0 and time.monotonic() < deadline:
            a_md = _mirror_descent_step(a, grad_centered, eta * 10)
            score_md = _compute_score(a_md)
            if score_md < best_score:
                best = a_md.copy()
                best_score = score_md
                a = a_md.copy()
        
        # Every 5000 iterations, try a random restart from best
        if fw_iter % 5000 == 0 and time.monotonic() < deadline - 100:
            a_pert = best + 0.02 * (np.random.random(n) - 0.5)
            a_pert = _project_simplex(a_pert, 1.0)
            score_pert = _compute_score(a_pert)
            if score_pert < best_score * 1.01:
                a = a_pert.copy()
    
    return best, best_score


def _optimize_lp_fw(a, deadline, rng, max_lp_time=30.0):
    """
    LP-based Frank-Wolfe optimization.
    Uses the LP oracle when available, falls back to standard FW.
    """
    try:
        import cvxpy as cp
        have_cvxpy = True
    except ImportError:
        have_cvxpy = False
    
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    
    best = a.copy()
    best_score = _compute_score(best)
    fw_iter = 0
    
    while time.monotonic() < deadline:
        b = _fft_conv2(a)
        M = float(np.max(b))
        cur_score = 2.0 * n * M
        
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            fw_iter_reset = 0
        fw_iter += 1
        
        # Try LP-based FW step
        g_dir = None
        if have_cvxpy and n <= 8000:
            lp_start = time.monotonic()
            try:
                g_var = cp.Variable(n)
                constraints = [g_var >= 0]
                # Build convolution constraints
                # (a*g)[k] = sum_j a[k-j]*g[j] <= M
                # A[k,j] = a[k-j] if 0 <= k-j < n else 0
                for k in range(2 * n - 1):
                    lo = max(0, k - n + 1)
                    hi = min(n - 1, k)
                    if lo <= hi:
                        idx = np.arange(lo, hi + 1)
                        coeff = a[k - idx]
                        constraints.append(coeff @ g_var[lo:hi + 1] <= M)
                
                obj = cp.Maximize(cp.sum(g_var))
                prob = cp.Problem(obj, constraints)
                
                # Solve with appropriate solver
                try:
                    prob.solve(solver=cp.GLPK, verbose=False)
                except Exception:
                    try:
                        prob.solve(solver=cp.CBC, verbose=False)
                    except Exception:
                        try:
                            prob.solve(solver=cp.PDLP, verbose=False)
                        except Exception:
                            prob.solve(verbose=False)
                
                if g_var.value is not None:
                    g_dir = np.asarray(g_var.value, dtype=np.float64)
                    g_dir = np.maximum(g_dir, 0)
                    S_g = np.sum(g_dir)
                    if S_g > 1e-10:
                        g_dir = g_dir / S_g
                    else:
                        g_dir = None
                
                lp_time = time.monotonic() - lp_start
                if lp_time > max_lp_time:
                    have_cvxpy = False  # Disable LP for rest of this run
            except Exception:
                g_dir = None
        
        # If LP failed, use standard FW
        if g_dir is None:
            grad, _ = _subgradient_single(a)
            j = np.argmin(grad - np.mean(grad))
            g_dir = np.zeros(n)
            g_dir[j] = 1.0
        
        # FW step
        t = 2.0 / (fw_iter + 2)
        a = (1 - t) * a + t * g_dir
        a = _project_simplex(a, 1.0)
        
        if fw_iter % 100 == 0:
            print(f"  LP-FW iter {fw_iter}: score={cur_score:.6f}, best={best_score:.6f}")
    
    return best, best_score


def _coordinate_descent_polish(a, deadline, rng):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, the objective is a convex piecewise-quadratic function.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    
    best = a.copy()
    best_score = _compute_score(best)
    no_improve = 0
    
    while time.monotonic() < deadline:
        improved = False
        # Cycle through coordinates (random order for diversity)
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() >= deadline:
                break
            
            # Try small perturbations of coordinate i
            a_orig_i = a[i]
            a_new_sum = S  # current sum (should be ~1)
            
            # The score as a function of a[i] (with others fixed):
            # We need to re-normalize, so effectively changing a[i] changes the whole thing
            # Simple approach: try a few values and pick the best
            candidates = [
                a_orig_i * 0.5,
                a_orig_i * 0.75,
                a_orig_i * 0.9,
                a_orig_i,
                a_orig_i * 1.1,
                a_orig_i * 1.25,
                a_orig_i * 1.5,
                a_orig_i + 0.01 / n,
                a_orig_i - 0.01 / n,
                0.0,
            ]
            
            best_cand = a_orig_i
            best_cand_score = np.inf
            
            for c in candidates:
                if c < -1e-10:
                    c = 0.0
                a[i] = c
                new_S = np.sum(a)
                if new_S < 1e-10:
                    a[i] = a_orig_i
                    continue
                # Normalized score
                b = _fft_conv2(a)
                sc = 2.0 * n * np.max(b) / (new_S * new_S)
                if sc < best_cand_score:
                    best_cand_score = sc
                    best_cand = c
            
            a[i] = best_cand
            a = _project_simplex(a, 1.0)
            
            cur_score = _compute_score(a)
            if cur_score < best_score - 1e-12:
                best = a.copy()
                best_score = cur_score
                no_improve = 0
                improved = True
            elif not improved:
                no_improve += 1
                if no_improve > 3 * n:
                    break
        
        if no_improve > 3 * n:
            break
    
    return best, best_score


def _refined_subgradient(a, deadline, rng):
    """
    Refined subgradient with adaptive parameters and restarts.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    
    best = a.copy()
    best_score = _compute_score(best)
    
    # Adaptive parameters
    K_list = [20, 10, 5, 3, 2]
    K_idx = 0
    temp_list = [3.0, 5.0, 8.0, 12.0, 20.0]
    temp_idx = 0
    
    eta = 0.3 / np.sqrt(n)
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() < deadline:
        # Compute subgradient
        grad, b_max = _subgradient_topk(a, K=K_list[K_idx], temperature=temp_list[temp_idx])
        grad_c = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_c)
        
        if grad_norm < 1e-15:
            break
        
        # Normalized gradient step
        step = eta / grad_norm
        a = a - step * grad_c
        a = _project_simplex(a, 1.0)
        
        # Evaluate
        score = _compute_score(a)
        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        iter_count += 1
        
        # Adaptive parameter changes
        if no_improve >= 500:
            # Try different K and temperature
            K_idx = (K_idx + 1) % len(K_list)
            temp_idx = (temp_idx + 1) % len(temp_list)
            eta *= 0.8
            no_improve = 0
            
            # Occasional restart
            if iter_count % 5000 == 0 and time.monotonic() < deadline - 50:
                a = best + 0.01 * (np.random.random(n) - 0.5)
                a = _project_simplex(a, 1.0)
            
            if eta < 1e-14:
                break
    
    return best, best_score


def _try_length(n, initial_a, deadline, rng):
    """Try optimizing with a specific length and initial sequence."""
    a = np.asarray(initial_a, dtype=np.float64).copy()
    if len(a) != n:
        a = np.ones(n) / n
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    best = a.copy()
    best_score = _compute_score(best)
    
    # Phase 1: Subgradient + FW (60% of time)
    t1 = min(deadline - 60, time.monotonic() + 0.6 * (deadline - time.monotonic()))
    a1, s1 = _optimize_subgrad_fw(a, t1, rng, K=10, use_fw=True, 温度=5.0)
    if s1 < best_score:
        best, best_score = a1, s1
    
    # Phase 2: Refined subgradient (30% of time)
    t2 = min(deadline - 30, time.monotonic() + 0.3 * (deadline - time.monotonic()))
    a2, s2 = _refined_subgradient(best, t2, rng)
    if s2 < best_score:
        best, best_score = a2, s2
    
    # Phase 3: Coordinate descent polish (10% of time)
    t3 = deadline
    a3, s3 = _coordinate_descent_polish(best, t3, rng)
    if s3 < best_score:
        best, best_score = a3, s3
    
    return best, best_score
