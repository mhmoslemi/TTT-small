"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase strategy:
1. Mirror descent (exponential gradient) with log-sum-exp smoothed objective
2. LP-based Frank-Wolfe method
3. Conflict-aware coordinate descent
4. Multi-length exploration with interpolation
5. Simulated annealing restarts

Key innovation: Mirror descent on the simplex with adaptive log-sum-exp
temperature provides a smooth, non-oscillatory path toward minimizing
the peak autocorrelation, complementing the LP-FW approach.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_lse_gradient(a, T):
    """Compute gradient of log-sum-exp(conv(a)/T) and the softmax weights."""
    n = len(a)
    conv = fast_convolve(a)
    shifted = conv / T
    shifted -= shifted.max()
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    # grad[j] = 2 * sum_{k=j}^{min(2n-2, j+n-1)} w[k] * a[k-j]
    grad = np.zeros(n)
    for j in range(n):
        k_hi = min(2 * n - 2, j + n - 1)
        length = k_hi - j + 1
        if length > 0:
            grad[j] = 2.0 * np.dot(w[j:j + length], a[:length])
    return grad, conv, w


def mirror_descent_optimize(a0, time_limit, T_init=None, lr_init=0.4,
                            verbose=False):
    """Mirror descent (exponential gradient) with LSE smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv0 = fast_convolve(a)
    if T_init is None:
        T_init = max(conv0.max() * 0.015, 0.00005)
    T_min = T_init * 0.03

    best_obj = conv0.max()
    best_a = a.copy()
    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        grad, conv, w = compute_lse_gradient(a, T)

        # Mirror descent: a_new[j] proportional to a[j] * exp(-lr * grad_normalized)
        grad_centered = grad - grad.mean()
        grad_max = np.max(np.abs(grad_centered))
        if grad_max < 1e-15:
            no_improve += 1
            if no_improve > 100:
                break
            continue

        effective_lr = lr * min(1.0, 0.1 / (grad_max * n ** 0.5 + 1e-10))
        log_ratio = -effective_lr * grad_centered / grad_max
        log_ratio = np.clip(log_ratio, -25, 25)

        a_new = a * np.exp(log_ratio)
        # Add small floor to prevent values from going to exact zero
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and it % 500 == 0:

        else:
            no_improve += 1
            if no_improve % 5 == 0:
                a = a_new

        # Annealing schedule
        T = max(T_min, T * 0.9996)
        lr = max(0.05, lr * 0.99995)

        if no_improve > 250:

            conv = fast_convolve(a)
            T = T_init * 0.4

            no_improve = 0

    return best_a, best_obj


def solve_lp_max_sum(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def conflict_cd(f0, time_limit, verbose=False):
    """
    Conflict-aware coordinate descent: identifies coordinates that
    contribute to multiple peaks and reduces them, redistributing
    to coordinates that contribute to fewer peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Find top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        # Build conflict map: which coordinates contribute to which peaks
        conflict_count = np.zeros(n)
        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            for j in range(js, je + 1):
                conflict_count[j] += 1.0 * f[j]

        # For each peak, try adjusting its contributing coordinates
        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            contributing = list(range(js, je + 1))

            if not contributing:
                continue

            # Find the most "conflicted" coordinate with largest value
            contrib_vals = [(j, f[j], conflict_count[j]) for j in contributing]
            contrib_vals.sort(key=lambda x: x[1] * x[2], reverse=True)

            for (j, old_val, conf) in contrib_vals[:3]:
                if time.monotonic() - t0 > time_limit:
                    break
                if old_val < 1e-12:
                    continue

                # Try reducing this coordinate
                for mult in [0.5, 0.65, 0.8, 1.2, 1.4, 1.6]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()

                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 5:
                # Random exploration
                rng = np.random.default_rng(passes * 31415)
                for _ in range(min(30, n // 20)):
                    if time.monotonic() - t0 > time_limit:
                        break
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.5, 0.7, 1.3, 1.5]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.5:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break
                    if improved:
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.003, momentum=0.93,
                         top_k=40, temp=0.00005, verbose=False):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj

        n_best = len(best_seq)

        # Quick mirror descent with very low T
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            conv_best = fast_convolve(best_seq)
            t_init = max(conv_best.max() * 0.003, 0.00001)
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=min(rem, 60),
                                               T_init=t_init, lr_init=0.3)
            try_seq(a_opt, "md_polish")

        # Quick LP-FW
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 50), use_min_peak=True)
            try_seq(a_opt, "lpfw_polish")

        # Perturbations with short MD
        for p in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.0008 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = mirror_descent_optimize(f_pert, time_limit=min(rem, 20),
                                               T_init=max(fast_convolve(f_pert).max() * 0.005, 0.00002),
                                               lr_init=0.4)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
