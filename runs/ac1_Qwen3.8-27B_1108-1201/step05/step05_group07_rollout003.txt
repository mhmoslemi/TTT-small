# Helper functions at top level

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def lsmax_weights(conv, T):
    """Softmax weights over convolution values for smooth max approximation."""
    x = conv / T
    x -= x.max()
    w = np.exp(x)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w

def conv_grad_fft(w, a):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Result: grad_a[i] = 2 * sum_j w[i+j] * a[j]
    """
    n = len(a)
    N = 1 << (2 * n - 2).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    grad = 2.0 * np.real(np.fft.ifft(W * np.conj(A)))[:n]
    return grad

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def peak_targeting(f0, time_limit, top_k=20, verbose=False):
    """
    Directly target the peaks: for each peak, try to reduce it by adjusting
    the contributing elements.
    """
    n = len(f0)
    f = np.maximum(f0, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Get top peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        improved = False
        for k in sorted(peaks, key=lambda x: -conv[x]):
            if time.monotonic() - t0 > time_limit:
                break
            
            k = int(k)
            # Find contributing pairs
            js_start = max(0, k - n + 1)
            js_end = min(n - 1, k)
            js = np.arange(js_start, js_end + 1)
            
            # Contribution of each pair (j, k-j)
            contribs = f[js] * f[k - js]
            total = contribs.sum()
            
            if total < 1e-15:
                continue
            
            # Try to reduce the largest contributor
            idx_max = js[np.argmax(contribs)]
            idx_min = js[np.argmin(contribs)]
            
            if f[idx_max] < 1e-10 or idx_max == idx_min:
                continue
            
            # Transfer mass from idx_max to idx_min
            for frac in [0.05, 0.1, 0.15, 0.2]:
                amt = f[idx_max] * frac
                f[idx_max] -= amt
                f[idx_min] += amt
                f = np.maximum(f, 0)
                f /= f.sum()
                
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    # Revert
                    f[idx_min] -= amt
                    f[idx_max] += amt
                    f /= f.sum()
        
        if not improved:
            # Try random perturbations
            rng = np.random.default_rng(it * 104729 + 3)
            for _ in range(20):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or f[i] < 1e-10:
                    continue
                amt = f[i] * (0.05 + 0.15 * rng.random())
                f[i] -= amt
                f[j] += amt
                f = np.maximum(f, 0)
                f_s = f.sum()
                if f_s < 1e-15:
                    f = best_f.copy()
                    break
                f /= f_s
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()
                    improved = True
                    break
            
            if not improved:
                break
    
    return best_f, best_obj
