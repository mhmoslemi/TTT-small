```python
        i_end = min(n - 1, k)
        j_start = k - i_end  # = max(k-n+1, 0)
        j_end = k - i_start  # = min(k, n-1)
        g[i_start:i_end+1] += 2.0 * a[j_start:j_end+1][::-1]
    return g / len(peaks)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Exact subgradient descent with Polyak step size (top-k peaks)
- Mirror descent (multiplicative weights) on the simplex
- FW-LP with LP oracle for global improvements
- Adaptive simulated annealing for final polish
- Multi-n exploration from 256 to 4096

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peaks):
    """Compute subgradient of max_k conv_k(a) at given peak indices.
    d conv[k]/d a[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    g = np.zeros(n)
    for k in peaks:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        j_start = k - i_end
        j_end = k - i_start
        g[i_start:i_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
    return g / len(peaks)


def subgradient_polyak(a0, time_limit, top_k=50, cooling=1.0, verbose=False):
    """Polyak subgradient descent with top-k peak selection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    f_lb = 1.0 / (2 * n - 1)  # Lower bound on max(conv) when sum(a)=1

    conv = fast_convolve(a)
    f_val = conv.max()
    best_a = a.copy()
    best_f = f_val

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    effective_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        f_val = conv.max()

        # Select top-k peaks
        if effective_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -effective_k)[-effective_k:]

        g = compute_subgradient(a, peaks)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Polyak step size with damping
        polyak_eta = (f_val - f_lb) / (g_norm ** 2)
        # Damp and cap the step
        eta = min(polyak_eta * cooling * 0.5, 0.1)

        # Backtracking: try several step sizes
        best_candidate = a
        best_trial_val = f_val
        for trial in [eta, eta * 0.5, eta * 0.25, eta * 0.1, eta * 2.0]:
            candidate = project_simplex(a - trial * g)
            cv = fast_convolve(candidate)
            val = cv.max()
            if val < best_trial_val:
                best_trial_val = val
                best_candidate = candidate

        a = best_candidate
        if best_trial_val < best_f - 1e-15:
            best_f = best_trial_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt k and cooling
        if no_improve > 50:
            effective_k = max(10, effective_k // 2)
            cooling = max(0.3, cooling * 0.9)
        if no_improve > 200:
            # Perturb
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            effective_k = top_k
            cooling = 1.0
            no_improve = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it}, f={best_f:.10f}, k={effective_k}, t={elapsed:.1f}s")

    return best_a, best_f


def mirror_descent(a0, time_limit, T_init=0.01, verbose=False):
    """Mirror descent (multiplicative weights) for the convex problem.
    a_new[i] = a[i] * exp(-eta * g[i]) / sum(...)
    This is the natural update for simplex-constrained problems.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    f_lb = 1.0 / (2 * n - 1)

    conv = fast_convolve(a)
    f_val = conv.max()
    best_a = a.copy()
    best_f = f_val

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        f_val = conv.max()

        # Softmax weights for peak selection
        scaled = (conv - conv.max()) / T
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = p / p_sum

        # Compute weighted subgradient: g[i] = 2 * sum_k p[k] * a[k-i]
        # Use FFT: this is 2 * (p_reversed convolved with a) restricted to [0, n-1]
        # Actually: g[i] = 2 * sum_k p[k] * a[k-i] * 1_{0<=k-i<n}
        # This is a cross-correlation. Let's compute via FFT.
        # Define q as length 2n-1, q[k] = p[k]
        # Then g[i] = 2 * sum_k q[k] * a[k-i] for valid indices
        # = 2 * (q_flipped * a)[offset] ... let me just do it directly for clarity
        # Using the fact that it's a correlation: g = 2 * correlate(a, q, 'full')[n-1:n-1+n]
        # Wait, let me think again.
        # g[i] = 2 * sum_{k} p[k] * a[k-i] where 0 <= k-i < n
        # Let j = k-i, so k = i+j. g[i] = 2 * sum_j p[i+j] * a[j] where 0<=j<n and 0<=i+j<2n-1
        # This is the cross-correlation of p and a: (p ⊛ a)[i] = sum_j p[i+j] * a[j]
        # So g = 2 * (p ⊛ a)
        # np.correlate(p, a, 'full') gives sum_j p[j] * a[j+i]... let me be careful.
        # Actually let me just use the FFT approach:
        # g[i] = 2 * sum_k p[k] * a[k-i] * 1_{0<=k-i<n}
        # = 2 * sum_{j=0}^{n-1} a[j] * p[i+j] (where i+j < 2n-1)
        # This is the cross-correlation. Use FFT:
        fft_len = 1 << (2 * (2 * n - 1) - 1).bit_length()
        P = np.fft.rfft(p, fft_len)
        A = np.fft.rfft(a, fft_len)
        # Cross-correlation: sum_j a[j] * p[i+j] = IFFT(FFT(a) * conj(FFT(p)))... 
        # Actually: (a ⊛ p)[m] = sum_j a[j] * conj(p[j-m]) ... this is getting confusing.
        # Let me just do it the simple way for correctness:
        # g[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j] for i+j < 2n-1
        # Build a large array and use convolution
        # p_conv[i] = sum_j a[j] * p[i+j] = (reversed_a * p)[offset]
        # Let ar = a[::-1], then (ar * p)[n-1+i] = sum_j ar[j] * p[n-1+i-j] = sum_j a[n-1-j] * p[n-1+i-j]
        # Let k = n-1-j: = sum_k a[k] * p[i+k] ✓ (for valid i+k)
        ar = a[::-1]
        conv_ap = np.convolve(ar, p, mode='full')
        g = 2.0 * conv_ap[n - 1:n - 1 + n]
        # Normalize by sum(p) which should be 1
        # Actually p is already normalized to sum to 1

        # Polyak step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        eta = (f_val - f_lb) / (g_norm ** 2)
        eta = min(eta, 0.1)

        # Multiplicative weights update
        log_a = np.log(a + 1e-30) - eta * g
        log_a -= log_a.max()  # for numerical stability
        a_new = np.exp(log_a)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = project_simplex(a - eta * g)
            a = a_new
        else:
            a = a_new / a_new_sum

        f_new = fast_convolve(a).max()
        if f_new < best_f - 1e-15:
            best_f = f_new
            best_a = a.copy()
            no_improve = 0
            # If improving, maybe increase T slightly
            if it % 100 == 0:
                T *= 1.02
        else:
            no_improve += 1
            if no_improve > 30:
                T *= 1.1
            if no_improve > 150:
                T *= 0.7
            if no_improve > 400:
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                T = T_init
                no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it}, f={best_f:.10f}, T={T:.4f}, t={elapsed:.1f}s")

def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.2:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp, g >= 0
        g = None
        try:
            c = -np.ones(n)
            A_ub = np.zeros((2 * n - 1, n))
            for j in range(n):
                A_ub[j:j + n, j] = f
            b_ub = np.full(2 * n - 1, M * m_exp)
            bounds = [(0, None)] * n
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1, remaining - 1)})
            if res.success:
                g = np.maximum(res.x, 0.0)
        except Exception:
            pass

        if g is None or g.sum() < 1e-15:
            m_exp *= 1.3
            if m_exp > 5.0:
                m_exp = 1.0
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.2, 5.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 48271 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def simulated_annealing(a0, time_limit, T_init=0.002, T_min=0.0001, verbose=False):
    """Simple simulated annealing with 2-swap moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    f_val = conv.max()
    best_a = a.copy()
    best_f = f_val

    T = T_init
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # 2-swap: pick two random indices, swap some mass
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Move a fraction of mass from i to j (or vice versa)
        delta = rng.random() * 0.01 * max(a[i], a[j])
        if rng.random() < 0.5:
            if a[i] > delta:
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
            else:
                continue
        else:
            if a[j] > delta:
                a_new = a.copy()
                a_new[j] -= delta
                a_new[i] += delta
            else:
                continue

        f_new = fast_convolve(a_new).max()
        delta_f = f_new - f_val
        if delta_f < 0 or rng.random() < np.exp(-delta_f / T):
            a = a_new
            f_val = f_new
            if f_val < best_f:
                best_f = f_val
                best_a = a.copy()

        # Cool down
        T = max(T_min, T * 0.9997)

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA it={it}, f={best_f:.10f}, T={T:.6f}, t={elapsed:.1f}s")

    return best_a, best_f


def neighborhood_refinement(a0, time_limit, window=20, verbose=False):

        print(f"  Subgrad n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_polyak(start, time_limit=t_alloc, top_k=50)
        try_seq(a_opt, f"sg_n{n_try}")

        # Quick mirror descent
        remaining = deadline - time.monotonic() - 1050
        if remaining > 20:
            t_alloc2 = min(remaining, 25)
            a_opt2, _ = mirror_descent(a_opt, time_limit=t_alloc2, T_init=0.01)
            try_seq(a_opt2, f"md_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: FW-LP on best candidate (~400s) ===
    print("\n=== Phase 2: FW-LP Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Use FW-LP if n is not too large (LP gets slow)
        if n_best <= 1536:
            remaining = deadline - time.monotonic() - 700
            if remaining > 30:
                t_alloc = min(remaining, 200)
                print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
                a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "fw_main")

                # Perturbed FW-LP
                remaining = deadline - time.monotonic() - 500
                if remaining > 30:
                    for scale in [0.005, 0.015]:
                        remaining = deadline - time.monotonic() - 400
                        if remaining < 20:
                            break
                        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
                        t_alloc2 = min(remaining, 80)
                        a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc2)
                        try_seq(a_opt2, f"fw_pert_{scale}")
        else:
            # For large n, use subgradient instead
            remaining = deadline - time.monotonic() - 700
            if remaining > 30:
                t_alloc = min(remaining, 200)
                print(f"  Subgrad (large n) n={n_best}, t={t_alloc:.0f}s")
                a_opt, _ = subgradient_polyak(best_seq, time_limit=t_alloc, top_k=100)
                try_seq(a_opt, "sg_large")

    print(f"  Phase 2 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Mirror descent + refined subgradient (~450s) ===
    print("\n=== Phase 3: Mirror Descent & Refined Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Mirror descent with different temperatures
        configs = [
            {'T_init': 0.005},
            {'T_init': 0.02},
            {'T_init': 0.05},
        ]
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, 120)
            print(f"  MD T={cfg['T_init']}, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, T_init=cfg['T_init'])
            try_seq(a_opt, f"md_{ci}")

        # Refined subgradient with smaller top_k for focus
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Subgrad fine k=20, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_polyak(best_seq, time_limit=t_alloc, top_k=20, cooling=0.8)
            try_seq(a_opt, "sg_fine")

        # Subgradient with larger k for global view
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 80)
            print(f"  Subgrad wide k=200, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_polyak(best_seq, time_limit=t_alloc, top_k=200, cooling=1.2)
            try_seq(a_opt, "sg_wide")

    print(f"  Phase 3 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Neighborhood refinement + Simulated Annealing (~350s) ===
    print("\n=== Phase 4: Local Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Neighborhood refinement
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  Neighborhood, t={t_alloc:.0f}s")
            a_opt, _ = neighborhood_refinement(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "neighborhood")

        # Simulated annealing
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  Simulated Annealing, t={t_alloc:.0f}s")
            a_opt, _ = simulated_annealing(best_seq, time_limit=t_alloc,
                                           T_init=0.001, T_min=0.00005)
            try_seq(a_opt, "anneal")

        # One more subgradient pass on the result
        remaining = deadline - time.monotonic() - 60
        if remaining > 20 and best_seq is not None:
            t_alloc = min(remaining, 60)
            a_opt, _ = subgradient_polyak(best_seq, time_limit=t_alloc, top_k=30, cooling=0.5)
            try_seq(a_opt, "sg_final")

    print(f"  Phase 4 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Try different n values from best structure (~150s) ===
    print("\n=== Phase 5: Cross-n Transfer ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Try nearby n values
        for n_try in [n_best + 128, n_best - 128, n_best * 2, n_best // 2]:
            if n_try < 128 or n_try > 4096:
                continue
            remaining = deadline - time.monotonic() - 30
            if remaining < 20:
                break
            # Interpolate
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_best), best_seq)
            start = np.maximum(start, 0)
            start = start / start.sum()
            t_alloc = min(remaining, 40)
            a_opt, _ = subgradient_polyak(start, time_limit=t_alloc, top_k=50)
            try_seq(a_opt, f"xfer_n{n_try}")

    print(f"  Phase 5 done: best={best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Final: One last polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_opt, _ = subgradient_polyak(best_seq, time_limit=min(remaining, 20),
                                          top_k=10, cooling=0.3)
            try_seq(a_opt, "final_polish")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
