import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_active_gradient(f, active_indices, weights=None):
    """Compute weighted sum of gradients of c[k] for k in active_indices.
    grad_k[i] = 2*f[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(f)
    if weights is None:
        weights = np.ones(len(active_indices))
    grad = np.zeros(n)
    for idx, k in enumerate(active_indices):
        w = weights[idx]
        if w < 1e-15:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += 2.0 * w * f[k - ms]
    return grad
