import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:

    return np.maximum(v - theta, 0.0)

    return 2.0 * n * conv.max()

def solve_lp(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    K = len(active_k)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    for row, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Free-Set Frank-Wolfe with
improved constraint selection, L-BFGS-B polishing on smoothed max objective,
multi-length exploration, and aggressive coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def build_lp_matrix(f, n, active_k):
    """Build the LP constraint matrix for given active constraints."""
    K = len(active_k)
    A_ub = np.zeros((K, n))
    for row, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    return A_ub


def solve_lp(f, M, n, active_k):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    K = len(active_k)
    A_ub = np.zeros((K, n))
    for row, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    b_ub = np.full(K, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_all(f, M, n):
    """Solve LP with all 2n-1 constraints."""
    size = 2 * n - 1
    A_ub = np.zeros((size, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(size, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def line_search(f, g, max_iter=30):
    """Golden section line search for best t mixing f and g."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)

    def obj(t):
        x = (1.0 - t) * f + t * g
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x = x / s
        return fast_convolve(x).max()

    fc = obj(c)
    fd = obj(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = obj(d)
    t_best = (lo + hi) / 2.0
    return t_best, obj(t_best)


def lpfw_optimize(f0, time_limit, top_k=50, use_all_every=40, verbose=False):
    """LP-Frank-Wolfe with adaptive constraints and periodic full constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        if time.monotonic() - t0 > time_limit - 3:
            break

        # Adaptive constraint selection
        if it % use_all_every == 0:
            active_k = list(range(2 * n - 1))
        else:
            sorted_idx = np.argsort(conv)[::-1]
            k_sel = min(top_k, 2 * n - 1)
            active_k = sorted_idx[:k_sel].tolist()

        g = solve_lp(f, M * m_exp, n, active_k)
        if g is None:
            m_exp *= 1.15
            if m_exp > 6.0:
                m_exp = 1.0
                active_k = list(range(2 * n - 1))
                g = solve_lp_all(f, M, n)
            if g is None:
                no_improve += 1
                if no_improve > 100:
                    break
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 100:
                break
            continue

        t_best, obj_best = line_search(f, g)

        if t_best > 1e-10:
            f_new = (1.0 - t_best) * f + t_best * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()

            if M_new < best_obj - 1e-13:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                M = M_new
                m_exp = 1.0
                if verbose and it % 30 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
            f = f_new
            conv = conv_new
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        # Periodic perturbation to escape local minima
        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.004 * min(no_improve // 60, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def smooth_max_obj_and_grad(a, k=30, temp=0.005):
    """Smoothed max of autocorrelation and its gradient."""
    conv = fast_convolve(a)
    n = len(a)
    top_k_idx = np.argsort(conv)[::-1][:min(k, len(conv))]
    top_k_vals = conv[top_k_idx]
    shifted = (top_k_vals - top_k_vals[0]) / temp
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / exp_shifted.sum()
    obj = top_k_vals[0] + temp * (max_shifted + np.log(exp_shifted.sum()))

    grad = np.zeros(n)
    for i, k_idx in enumerate(top_k_idx):
        if weights[i] < 1e-14:
            continue
        k = int(k_idx)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * weights[i] * a[k - idx]
    return obj, grad


def bfgs_polish(f0, time_limit, temp=0.003, k=40):
    """L-BFGS-B polishing on smoothed max objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    def objective_and_grad(x):
        x = np.maximum(x, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            return 1e10, np.zeros(len(x))
        x = x / sx
        obj, grad = smooth_max_obj_and_grad(x, k=k, temp=temp)
        return obj, grad

    best_iter = [0]

    def callback(xk):
        best_iter[0] += 1
        if best_iter[0] % 20 == 0:
            el = fast_convolve(np.maximum(xk, 0)).max()
            nonlocal best_obj, best_f
            if el < best_obj:
                best_obj = el
                best_f = np.maximum(xk, 0).copy()
                best_f = best_f / best_f.sum()

    try:
        res = minimize(objective_and_grad, f, method='L-BFGS-B',
                       options={'maxiter': 500, 'maxfun': 2000,
                                'gtol': 1e-10, 'iprint': 0},
                       callback=callback)
    except Exception:
        pass

    if res.success or hasattr(res, 'fun'):
        x_opt = np.maximum(res.x, 0.0)
        sx = x_opt.sum()
        if sx > 1e-15:
            x_opt = x_opt / sx
            obj_opt = fast_convolve(x_opt).max()
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_f = x_opt.copy()

    # Also check current f
    cur_obj = fast_convolve(f).max()
    if cur_obj < best_obj:
        best_obj = cur_obj
        best_f = f.copy()

    return best_f, best_obj


def coord_desc(f0, time_limit, multipliers=None):
    """Coordinate descent with multipliers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    if multipliers is None:
        multipliers = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit - 1:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit - 2:
                break
            old_val = f[j]
            if old_val < 1e-13:
                continue

            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0 - 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = (1.0 - scale) * f
                f_trial[j] = new_val
                f_trial = np.maximum(f_trial, 0.0)
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig[:n])
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def sg_polish(f0, time_limit, lr=0.001, mom=0.94, top_k=40, temp=0.00003):
    """Subgradient polishing with softmax-weighted active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it_count = 0

    while time.monotonic() - t0 < time_limit - 2:
        it_count += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        cur_lr *= 0.9998
        cur_temp *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(it_count * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj

        restart_configs.append(("small_noise", project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)))
        restart_configs.append(("rev_sym", project_simplex(0.6 * best_seq + 0.4 * best_seq[::-1])))
        restart_configs.append(("flat_blend", project_simplex(0.8 * best_seq + 0.2 * np.ones(n_best) / n_best)))
        restart_configs.append(("gauss_center", project_simplex(
            np.exp(-2.5 * (np.arange(n_best) - n_best / 2.0) ** 2 / (0.5 * n_best) ** 2))))

        for i, (name, f_init) in enumerate(restart_configs):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem / max(1, len(restart_configs) - i + 1), 100)
            if t_alloc < 20:
                break
            print(f"  Restart [{name}] n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = lpfw_optimize(f_init, time_limit=t_alloc, top_k=55, use_all_every=25, verbose=False)
            try_seq(a_opt, f"deep_{name}")

    # === Phase 5: Coordinate Descent ===
    print(f"=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_cd = min(rem * 0.3, 200)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coord_desc(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

            # Second CD
            rem = deadline - time.monotonic()
            if rem > 30:
                t_cd2 = min(rem * 0.25, 120)
                a_opt2, _ = coord_desc(a_opt, time_limit=t_cd2,
                                       multipliers=[0.8, 0.9, 0.95, 1.0, 1.05, 1.1, 1.2])
                try_seq(a_opt2, "cd2")

    # === Phase 6: BFGS + SG Polish ===
    print(f"=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # BFGS with very small temperature
        rem = deadline - time.monotonic()
        if rem > 30:
            t_b = min(rem * 0.25, 80)
            a_opt, _ = bfgs_polish(best_seq, time_limit=t_b, temp=0.0005, k=20)
            try_seq(a_opt, "bfgs_fine")

        # SG polish
        rem = deadline - time.monotonic()
        if rem > 20:
            t_sg = min(rem * 0.3, 100)
            a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                                 lr=0.0008, mom=0.95,
                                 top_k=max(20, n_best // 6),
                                 temp=0.00002)
            try_seq(a_opt, "sg")

            # Final BFGS on SG result
            rem = deadline - time.monotonic()
            if rem > 20:
                a_opt2, _ = bfgs_polish(a_opt, time_limit=min(40, rem * 0.3),
                                        temp=0.0003, k=15)
                try_seq(a_opt2, "bfgs_final")

    # === Phase 7: Palindromic attempt ===
    print(f"=== Phase 7: Palindromic (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        half_n = (n_best + 1) // 2
        pal_half = (best_seq[:half_n] + best_seq[n_best - 1: n_best - half_n - 1: -1]) / 2.0
        pal_full = project_simplex(make_palindromic(pal_half, n_best))
        try_seq(pal_full, "pal_init")

        rem = deadline - time.monotonic()
        if rem > 30:
            t_pal = min(rem * 0.3, 80)
            a_opt, _ = lpfw_optimize(pal_full, time_limit=t_pal, top_k=50, use_all_every=30)
            try_seq(a_opt, "pal_lpfw")

    # === Phase 8: Final perturbations ===
    print(f"=== Phase 8: Final Perturbations (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for i in range(5):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            noise_scale = 0.0005 + 0.001 * i
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.2, 30)
            a_opt, _ = lpfw_optimize(f_pert, time_limit=t_pert, top_k=40, use_all_every=20)
            try_seq(a_opt, f"perturb_{i}")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
