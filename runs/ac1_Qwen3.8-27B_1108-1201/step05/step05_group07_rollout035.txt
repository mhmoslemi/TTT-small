def fast_convolve(a):

    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_full_peaks(a, conv, temperature):
    """Compute subgradient using ALL peaks with exponential weighting."""
    n = len(a)
    # Use all conv values with exponential weighting
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        # Fallback: use only the argmax
        k = np.argmax(conv)
        return subgradient_single_peak(a, conv, k)
    w /= ws
    
    sg = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[k] * a[k - idx]
    return sg

def subgradient_top_k(a, conv, top_k, temperature):
    """Compute subgradient using top-K peaks with exponential weighting."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        k = int(peaks[np.argmax(pv)])
        sg = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * a[k - idx]
        return sg
    w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg

def mu_optimize(a0, time_limit, top_k=50, temperature=5e-5, lr=0.003,
                momentum=0.85, verbose=False, label=""):
    """
    Multiplicative-weights subgradient optimization with Nesterov momentum.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        f_nest = f * np.exp(0.5 * momentum * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn
        
        conv_nest = fast_convolve(f_nest)
        if cur_top_k < len(conv_nest):
            sg = subgradient_top_k(f_nest, conv_nest, cur_top_k, cur_temp)
        else:
            sg = subgradient_full_peaks(f_nest, conv_nest, cur_temp)
        
        # Update momentum
        v = momentum * v + (1 - momentum) * sg
        
        # Multiplicative weights update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive schedule
        cur_lr *= 0.9998
        if cur_lr < 0.00003:
            cur_lr = 0.00003
        cur_temp *= 1.005
        if it % 1000 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.97))
        
        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng((it * 104729 + 7) % 2**31)
            noise_scale = 0.002 + 0.002 * min(no_improve // 1000, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0002)
            cur_temp = temperature * 0.3
            conv = fast_convolve(f)
            no_improve = 0
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MU iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")
    
    return best_f, best_obj

def coordinate_descent_exact(a0, time_limit, verbose=False):
    """
    Coordinate descent with exact 1D optimization.
    For each coordinate i, find the optimal value that minimizes the max
    of affected convolution values, accounting for re-normalization.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Process coordinates in order of contribution to peaks
        # (heuristic: center-out)
        mid = n // 2
        order = list(range(n))
        # Sort by distance from center (center first)
        order.sort(key=lambda i: abs(i - mid))
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[i]
            if old_val < 1e-14:
                # Try increasing from 0
                continue
            
            # Try a grid of values around current
            best_v = old_val
            best_local_obj = best_obj
            
            # The optimal v should be near the current value or slightly different
            # Try multiplicative factors
            for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                new_v = old_val * mult
                if new_v < 0 or new_v > 0.99:
                    continue
                
                # After changing f[i] to new_v, re-normalize
                # New sum = S - old_val + new_v = 1 - old_val + new_v
                new_sum = 1.0 - old_val + new_v
                if new_sum < 1e-10:
                    continue
                
                # Instead of full re-normalization (which changes all coords),
                # just evaluate the effect: the objective is 2*n*max_conv/S^2
                # But since we work with sum=1, changing one coord and re-normalizing
                # is equivalent to scaling all other coords.
                
                # Simpler: just set f[i] = new_v and re-normalize the whole vector
                f[i] = new_v
                f = np.maximum(f, 0)
                fs = f.sum()
                if fs < 1e-10:
                    f[i] = old_val
                    continue
                f /= fs
                
                conv_new = fast_convolve(f)
                obj_new = conv_new.max()
                
                if obj_new < best_local_obj - 1e-14:
                    best_local_obj = obj_new
                    best_v = new_v
                    # Keep this f
                else:
                    # Revert
                    f[i] = old_val
                    f = np.maximum(f, 0)
                    fs = f.sum()
                    f /= fs
            
            if best_v != old_val:
                if best_local_obj < best_obj - 1e-13:
                    best_obj = best_local_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            # Try a different order or break
            break
    
    return best_f, best_obj

def pair_search(a0, time_limit, verbose=False):
    """Pair exchange local search focused on peak reduction."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        
        # Find top peaks
        n_peaks = min(20, 2*n-1)
        top_peaks_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        top_peaks_idx = top_peaks_idx[np.argsort(conv[top_peaks_idx])[::-1]]
        
        # For each top peak, try to reduce it by transferring mass
        for k in top_peaks_idx[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            partner_idx = k - idx
            contribs = f[idx] * f[partner_idx]
            
            # Find the pair with largest contribution
            max_contrib_pos = np.argmax(contribs)
            hi_idx = idx[max_contrib_pos]
            
            # Try moving some mass from hi_idx to a random other position
            for _ in range(5):
                j = rng.integers(0, n)
                if j == hi_idx:
                    continue
                if f[hi_idx] < 1e-10:
                    break
                for frac in [0.1, 0.2, 0.4]:
                    amt = f[hi_idx] * frac
                    f[hi_idx] -= amt
                    f[j] += amt
                    conv_new = fast_convolve(f)
                    obj_new = conv_new.max()
                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[hi_idx] += amt
                        f[j] -= amt
                        break
        
        if not improved:
            # Random perturbations
            for _ in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or f[i] < 1e-10:
                    continue
                amt = f[i] * 0.15
                f[i] -= amt
                f[j] += amt
                conv_new = fast_convolve(f)
                if conv_new.max() < best_obj - 1e-13:
                    best_obj = conv_new.max()
                    best_f = f.copy()
                    improved = True
                else:
                    f[i] += amt
                    f[j] -= amt
            
            if not improved:
                break
    
    return best_f, best_obj

def make_tilted_init(n, rng, tilt_amount=0.3):
    """Create a tilted (non-symmetric) initial sequence."""
    x = np.linspace(0, 1, n)
    # Base: slightly peaked
    base = 1.0 + 0.5 * np.sin(np.pi * x)
    # Add tilt
    base += tilt_amount * (x - 0.5) * 2  # linear tilt
    base = np.maximum(base, 0.01)
    # Add small noise
    base += rng.standard_normal(n) * 0.02
    base = np.maximum(base, 0.0)
    base /= base.sum()
    return base

def make_gaussian_init(n, rng, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    center = n / 2.0 + rng.standard_normal() * n * 0.1
    sigma = n * sigma_frac
    x = np.arange(n)
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    g = np.maximum(g, 1e-10)
    g /= g.sum()
    return g

def make_dual_peak_init(n, rng):
    """Two peaks at different heights (breaks symmetry)."""
    c1 = n * (0.3 + rng.random() * 0.1)
    c2 = n * (0.6 + rng.random() * 0.1)
    x = np.arange(n)
    h1 = 1.0
    h2 = 0.5 + rng.random() * 0.8
    g1 = h1 * np.exp(-0.5 * ((x - c1) / (n * 0.15)) ** 2)
    g2 = h2 * np.exp(-0.5 * ((x - c2) / (n * 0.15)) ** 2)
    g = g1 + g2 + 1e-10
    g /= g.sum()
    return g

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj_normalized(a):
    """Compute max of auto-convolution for a normalized (sum=1) sequence."""
    conv = fast_convolve(a)
    return conv.max()


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_compute(a, conv, top_k, temperature):
    """Compute weighted subgradient of max(conv) using top-K peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    if k_actual == len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        k = int(np.argmax(conv))
        sg = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * a[k - idx]
        return sg
    w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def mu_subgradient_optimize(a0, time_limit, top_k=50, temperature=5e-5,
                            lr=0.003, momentum=0.87, tag=""):
    """
    Multiplicative-weights subgradient optimization with Nesterov momentum.
    Returns (best_sequence_normalized, best_max_conv).
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(0.5 * momentum * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        sg = subgradient_compute(f_nest, conv_nest, cur_top_k, cur_temp)

        # Update momentum
        v = momentum * v + (1.0 - momentum) * sg

        # Multiplicative weights update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_lr *= 0.9998
        if cur_lr < 0.00003:
            cur_lr = 0.00003
        cur_temp *= 1.006
        if it % 800 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.96))

        # Restart if stuck
        if no_improve > 400:
            seed_val = (it * 104729 + 31) % (2**31)

            noise_scale = 0.002 + 0.002 * min(no_improve // 800, 5)

            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0002)
            cur_temp = temperature * 0.4
            conv = fast_convolve(f)
            no_improve = 0

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{tag}] iter={it}, t={elapsed:.0f}s, obj={best_obj:.10f}", flush=True)

def coordinate_descent_polish(a0, time_limit, tag=""):
    """
    Coordinate descent: for each coordinate, try multiple multiplicative
    factors and keep the best. Uses center-out ordering.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    mid = n // 2
    order = sorted(range(n), key=lambda i: abs(i - mid))

    passes = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[i]
            if old_val < 1e-13:
                continue

            best_local_v = old_val
            best_local_obj = best_obj
            best_local_f = None

            # Try multiplicative factors
            factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
            for mult in factors:
                new_v = old_val * mult
                if new_v < 0 or new_v > 0.999:
                    continue

                f[i] = new_v
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-10:
                    f[i] = old_val
                    f = np.maximum(f, 0.0)
                    f /= f.sum()
                    continue
                f /= fs

                conv_new = fast_convolve(f)
                obj_new = conv_new.max()

                if obj_new < best_local_obj - 1e-14:
                    best_local_obj = obj_new
                    best_local_v = new_v
                    best_local_f = f.copy()

            # Restore if we found improvement, else keep original
            if best_local_f is not None:
                f = best_local_f.copy()
                if best_local_obj < best_obj - 1e-13:
                    best_obj = best_local_obj
                    best_f = f.copy()
                    improved = True
            else:
                f[i] = old_val
                f = np.maximum(f, 0.0)
                f /= f.sum()

        if not improved:
            break

    return best_f, best_obj


def peak_targeted_pair_search(a0, time_limit, tag=""):
    """
    Local search: identify peak-contributing pairs and try mass transfers
    to reduce the maximum convolution value.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    stall_count = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)

        # Top peaks
        n_peaks = min(15, 2 * n - 1)
        if n_peaks < 2 * n - 1:
            top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(conv))
        top_idx = top_idx[np.argsort(conv[top_idx])[::-1]]

        for k in top_idx[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            pidx = k - idx
            contribs = f[idx] * f[pidx]

            # Try reducing the top contributors
            top_contrib_pos = np.argsort(contribs)[::-1][:3]
            for cp in top_contrib_pos:
                hi = int(idx[cp])
                if f[hi] < 1e-12:
                    continue

                # Try transferring to random target
                for attempt in range(8):
                    j = int(rng.integers(0, n))
                    if j == hi:
                        continue
                    frac = 0.1 + 0.1 * attempt
                    amt = f[hi] * frac
                    if amt < 1e-14:
                        break

                    f[hi] -= amt
                    f[j] += amt
                    conv_new = fast_convolve(f)
                    obj_new = conv_new.max()

                    if obj_new < best_obj - 1e-13:
                        best_obj = obj_new
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[hi] += amt
                        f[j] -= amt

        # Random pair exchanges
        if not improved:
            for _ in range(30):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] < 1e-12:
                    continue
                amt = f[i] * (0.05 + 0.1 * rng.random())
                f[i] -= amt
                f[j] += amt
                conv_new = fast_convolve(f)
                if conv_new.max() < best_obj - 1e-13:
                    best_obj = conv_new.max()
                    best_f = f.copy()
                    improved = True
                else:
                    f[i] += amt
                    f[j] -= amt

        if not improved:
            stall_count += 1
            if stall_count > 5:
                break
        else:
            stall_count = 0

    return best_f, best_obj


def make_tilted_init(n, rng, tilt=0.4):
    """Non-symmetric tilted initial sequence."""
    x = np.linspace(0, 1, n)
    base = 1.0 + 0.3 * np.sin(np.pi * x)
    base += tilt * (x - 0.5) * 2.0
    base += rng.standard_normal(n) * 0.02
    base = np.maximum(base, 0.01)
    base /= base.sum()
    return base


def make_skewed_gaussian(n, rng):
    """Skewed Gaussian (gamma-like) initial sequence."""
    center = n * (0.35 + rng.random() * 0.3)
    sigma = n * (0.15 + rng.random() * 0.15)
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    # Add skew
    skew = 0.3 + rng.random() * 0.4
    g *= (1.0 + skew * (x - center) / n)
    g = np.maximum(g, 1e-10)
    g /= g.sum()
    return g


def make_dual_peak(n, rng):
    """Two asymmetric peaks."""
    c1 = n * (0.25 + rng.random() * 0.15)
    c2 = n * (0.55 + rng.random() * 0.15)
    h1 = 1.0
    h2 = 0.4 + rng.random() * 0.8
    s1 = n * (0.1 + rng.random() * 0.1)
    s2 = n * (0.1 + rng.random() * 0.1)
    x = np.arange(n, dtype=np.float64)
    g = h1 * np.exp(-0.5 * ((x - c1) / s1) ** 2) + h2 * np.exp(-0.5 * ((x - c2) / s2) ** 2)
    g = np.maximum(g, 1e-10)
    g /= g.sum()
    return g


def interpolate_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                                                top_k=max(30, n_best // 4),
                                                temperature=2e-5, lr=0.002,
                                                momentum=0.90, tag="p3_mir")
            track(a_opt, "p3_mirror")

        # Average of best and its mirror, then optimize
        rem = phase3_deadline - time.monotonic()
        if rem > 20:
            f_avg = 0.5 * best_seq + 0.5 * best_seq[::-1]
            f_avg = project_simplex(f_avg)
            t_alloc = min(rem * 0.5, 80)
            print(f"  P3 Mirror-avg opt: t={t_alloc:.0f}s", flush=True)
            a_opt, _ = mu_subgradient_optimize(f_avg, time_limit=t_alloc,
                                                top_k=max(30, n_best // 4),
                                                temperature=3e-5, lr=0.002,
                                                momentum=0.88, tag="p3_avg")
            track(a_opt, "p3_mirror_avg")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length Exploration (250s, best={best_score:.6f}) ===", flush=True)
    phase4_deadline = time.monotonic() + 250.0

    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = []
        for tl in [512, 640, 768, 896, 1024, 1152, 1280, 1536]:
            if tl != n_best:
                target_lengths.append(tl)

        for target_n in target_lengths:
            rem = phase4_deadline - time.monotonic()
            if rem < 15:
                break
            f_interp = interpolate_length(best_seq, target_n)
            t_alloc = min(rem * 0.25, 80)
            if t_alloc < 8:
                break
            print(f"  P4 n={target_n}, t={t_alloc:.0f}s", flush=True)
            a_opt, _ = mu_subgradient_optimize(f_interp, time_limit=t_alloc,
                                                top_k=max(25, target_n // 6),
                                                temperature=4e-5, lr=0.003,
                                                momentum=0.87, tag=f"p4_{target_n}")
            track(a_opt, f"p4_n{target_n}")

    # === Phase 5: Local search ===
    print(f"\n=== Phase 5: Local Search (250s, best={best_score:.6f}) ===", flush=True)
    phase5_deadline = time.monotonic() + 250.0

    if best_seq is not None:
        rem = phase5_deadline - time.monotonic()
        if rem > 20:
            t_pex = min(rem * 0.5, 100)
            print(f"  P5 Pair search: t={t_pex:.0f}s", flush=True)
            a_opt, _ = peak_targeted_pair_search(best_seq, time_limit=t_pex, tag="p5")
            track(a_opt, "p5_pair")

        rem = phase5_deadline - time.monotonic()
        if rem > 15:
            t_cd = min(rem * 0.7, 100)
            print(f"  P5 Coord descent: t={t_cd:.0f}s", flush=True)
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd, tag="p5cd")
            track(a_opt, "p5_cd")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (180s, best={best_score:.6f}) ===", flush=True)
    phase6_deadline = time.monotonic() + 180.0

    if best_seq is not None:
        n_best = len(best_seq)

        # Ultra-fine subgradient
        rem = phase6_deadline - time.monotonic()
        if rem > 20:
            t_fine = min(rem * 0.4, 70)
            print(f"  P6 Fine subgrad: t={t_fine:.0f}s", flush=True)
            a_opt, _ = mu_subgradient_optimize(best_seq, time_limit=t_fine,
                                                top_k=n_best, temperature=5e-7,
                                                lr=0.0002, momentum=0.96, tag="p6fine")
            track(a_opt, "p6_fine")

        # Pair search on refined
        rem = phase6_deadline - time.monotonic()
        if rem > 15:
            a_opt, _ = peak_targeted_pair_search(best_seq, time_limit=min(rem, 50), tag="p6pex")
            track(a_opt, "p6_pair")

        # Multiple small perturbations
        rem = phase6_deadline - time.monotonic()
        for trial in range(8):
            if rem < 10:
                break
            for scale in [0.0005, 0.002, 0.004]:
                rem = phase6_deadline - time.monotonic()
                if rem < 8:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)
                t_quick = min(rem, 12)
                a_opt, _ = mu_subgradient_optimize(f_pert, time_limit=t_quick,
                                                    top_k=max(20, n_best // 5),
                                                    temperature=2e-5, lr=0.001,
                                                    momentum=0.92, tag=f"p6pert{trial}")
                track(a_opt, f"p6_pert{trial}_{scale}")

        # Final coordinate descent
        rem = phase6_deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem, 30), tag="p6cd")
            track(a_opt, "p6_cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
