"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-Frank-Wolfe with
sparse constraint matrices, exact line search, multi-scale optimization,
and coordinate descent polish. Exploits the Toeplitz structure of the
convolution constraint matrix for faster LP solves.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.sparse import diags
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(f, n):
    """Build sparse Toeplitz matrix F where (Fg)[k] = (f*g)[k]."""
    # F[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a lower-Hessenberg-like Toeplitz matrix
    # Row k has non-zeros at columns max(0, k-n+1) to min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js)
            vals.extend(f[k - js])
    F = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return F


def build_conv_matrix_dense(f, n):
    """Build dense Toeplitz matrix F where (Fg)[k] = (f*g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated uniform initialization."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def gaussian_init(n, sigma_frac=0.25):
    """Gaussian-shaped initialization."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def triangle_init(n):
    """Triangular initialization."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 - np.abs(j - center) / (center + 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def raised_cosine_init(n):
    """Raised cosine window initialization."""
    j = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (j + 0.5) / n))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def solve_lp_direction(f, M, n, use_sparse=True):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k=0..2n-2, g >= 0."""
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    if use_sparse:
        A_ub = build_conv_matrix_sparse(f, n)
    else:
        A_ub = build_conv_matrix_dense(f, n)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, use_sparse=True,
                   max_linesearch_points=16, restart_interval=80):
    """LP-based Frank-Wolfe with exact line search and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    lp_fail_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP direction
        g = solve_lp_direction(f, M * m_exp, n, use_sparse=use_sparse)
        if g is None:
            m_exp = min(m_exp * 1.15, 5.0)
            lp_fail_count += 1
            if lp_fail_count > 10:
                m_exp = 1.0
                lp_fail_count = 0
            continue

        lp_fail_count = 0
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Check if direction is meaningful
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > restart_interval:
                # Noisy restart
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.002 + 0.002 * min(no_improve // restart_interval, 5)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            else:
                m_exp = min(m_exp * 1.03, 3.0)
            continue

        # Exact line search: golden section + binary refinement
        best_alpha = 0.0
        best_val = M

        # Candidate alphas
        fw_alpha = 2.0 / (it + 2.0)
        candidates = [0.01, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5,
                      fw_alpha, 0.6, 0.7, 0.8, 0.9, 1.0]
        candidates = sorted(set(max(0.0, min(1.0, c)) for c in candidates))

        for alpha in candidates:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Binary refinement around best
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)

        # Check time
        if time.monotonic() - t0 > time_limit:
            break

    return best_f, best_obj


def greedy_direction(f, M, n, passes=3):
    """Greedy approximation to LP direction: max sum(g) s.t. f*g <= M, g>=0."""
    g = np.zeros(n)
    conv_fg = np.zeros(2 * n - 1)  # f * g

    for p in range(passes):
        for j in range(n):
            # Max delta we can add to g[j]
            # (f*g)[k] increases by f[k-j]*delta for k in [j, j+n-1]
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            if k_lo > k_hi:
                continue
            # Available slack at each k
            slack = M - conv_fg[k_lo:k_hi + 1]
            # f[k-j] for k in [k_lo, k_hi]
            f_slice = f[k_lo - j:k_hi - j + 1]
            # delta_max = min(slack / f_slice) where f_slice > 0
            positive = f_slice > 1e-15
            if not positive.any():
                continue
            ratios = slack[positive] / f_slice[positive]
            delta = np.min(ratios)
            if delta > 1e-12:
                cap = 1.0 / n  # Don't let any single g[j] exceed 1/n too much
                delta = min(delta, cap)
                g[j] += delta
                # Update conv_fg
                conv_fg[j:j + n] += f[:n] * delta

    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    return g / g_sum


def nesterov_multipeak(f0, time_limit, lr0=0.005, mom0=0.85,
                       top_k=40, temp0=0.0001):
    """Nesterov-accelerated subgradient with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Multi-peak subgradient
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * w[i] * f[k - indices]

        # Nesterov momentum update
        v = cur_mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        y = f_new + cur_mom * (f_new - f)
        y = project_simplex(y)
        f = f_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        cur_lr *= 0.9997
        cur_temp *= 1.003
        cur_mom = min(0.97, cur_mom * 1.0002)

        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.7, 0.001)
            cur_temp = temp0
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with golden section search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    no_pass_improve = 0

    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            lo = 0.0
            hi = min(0.999, 2.5 * old_val + 0.01)

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            def eval_coord(x, jj=j):
                ft = f.copy()
                ft[jj] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[jj] = False
                ft[mask] = f[mask] * (ft2 / denom)
                ss = ft.sum()
                if ss < 1e-15:
                    return float('inf')
                ft = ft / ss
                return fast_convolve(ft).max()

            f1 = eval_coord(c1)
            f2 = eval_coord(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_coord(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)

            cur_val = fast_convolve(f).max()
            if opt_val < cur_val - 1e-14:
                ft = f.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = f[mask] * (ft2 / denom)
                ft = ft / ft.sum()
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()
                    if verbose and pass_improved:
                        print(f"    CD pass {passes}: obj={best_obj:.10f}")

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Use cubic interpolation for smoother results
    if old_n >= 4:
        from scipy.interpolate import PchipInterpolator
        interp = PchipInterpolator(old_x, f, extrapolate=False)
        f_interp = interp(new_x)
    else:
        f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP-Frank-Wolfe (primary), Nesterov subgradient, coordinate descent,
    and multi-scale exploration from diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    # === Phase 0: Try diverse initializations ===
    print("=== Phase 0: Initialization ===")
    init_pool = []

    # Parent
    init_pool.append(("parent", parent))

    # Cosine modulated at parent length
    n_parent = len(parent)
    for freq, amp in [(1, 0.3), (2, 0.25), (3, 0.2), (1, 0.5), (2, 0.15)]:
        init_pool.append((f"cos_{freq}_{amp}", cosine_modulated_init(n_parent, freq, amp)))

    # Other shapes
    init_pool.append(("gauss", gaussian_init(n_parent, 0.25)))
    init_pool.append(("triangle", triangle_init(n_parent)))
    init_pool.append(("raised_cos", raised_cosine_init(n_parent)))

    # Different lengths
    for n_try in [512, 768, 1024]:
        if n_try != n_parent:
            init_pool.append((f"cos512_{n_try}", cosine_modulated_init(n_try, 2, 0.25)))
            init_pool.append((f"tri_{n_try}", triangle_init(n_try)))

    # Evaluate all initializations
    for label, init_a in init_pool:
        s = try_seq(init_a, label)
        if np.isfinite(s) and s < best_score + 0.3:
            print(f"  Init {label}: score={s:.8f}")

    # === Phase 1: LP-FW on the best initialization ===
    print(f"\n=== Phase 1: LP-FW (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 900
    t_lpfw = min(rem, 600)
    if t_lpfw > 30:
        print(f"  Running LP-FW for {t_lpfw:.0f}s on n={len(base)}")
        a_opt, obj = lp_fw_optimize(base, time_limit=t_lpfw, verbose=True,
                                     use_sparse=True, restart_interval=60)
        try_seq(a_opt, "lpfw_main")

    # === Phase 2: Multi-scale extension ===
    print(f"\n=== Phase 2: Multi-scale (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        target_lens = []
        if n_best <= 600:
            target_lens = [768, 1024]
        elif n_best <= 800:
            target_lens = [1024, 1280]
        else:
            target_lens = [1024]

        for target_n in target_lens:
            rem = deadline - time.monotonic() - 500
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            s_interp = try_seq(f_interp, f"interp_n{target_n}")

            if np.isfinite(s_interp) and s_interp < best_score + 0.1:
                t_alloc = min(rem * 0.5, 200)
                if t_alloc > 20:
                    print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
                    a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc,
                                               verbose=False, use_sparse=True,
                                               restart_interval=50)
                    try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 3: Second LP-FW pass on best ===
    print(f"\n=== Phase 3: LP-FW pass 2 (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t2 = min(rem * 0.7, 300)
            print(f"  LP-FW pass 2 for {t2:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2,
                                       verbose=True, use_sparse=True,
                                       restart_interval=50)
            try_seq(a_opt, "lpfw_pass2")

    # === Phase 4: Nesterov subgradient ===
    print(f"\n=== Phase 4: Nesterov (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_ns = min(rem * 0.6, 200)
            top_k = max(20, n_best // 5)
            print(f"  Nesterov t={t_ns:.0f}s, n={n_best}, top_k={top_k}")
            a_opt, _ = nesterov_multipeak(best_seq, time_limit=t_ns,
                                           lr0=0.006, mom0=0.85,
                                           top_k=top_k, temp0=0.00008)
            try_seq(a_opt, "nesterov1")

            # Second Nesterov pass with finer params
            rem = deadline - time.monotonic() - 150
            if rem > 15:
                a_opt, _ = nesterov_multipeak(a_opt, time_limit=min(rem, 100),
                                               lr0=0.002, mom0=0.9,
                                               top_k=max(15, n_best // 6),
                                               temp0=0.00003)
                try_seq(a_opt, "nesterov2")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: CD Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            # Limit CD to reasonable time; for large n, do fewer passes
            t_cd = min(rem * 0.8, 150)
            print(f"  CD t={t_cd:.0f}s, n={n_best}")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "cd_polish")

    # === Phase 6: Final short LP-FW + perturbations ===
    print(f"\n=== Phase 6: Final (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Short LP-FW
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_final = min(rem, 60)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final,
                                       use_sparse=True, restart_interval=30)
            try_seq(a_opt, "final_lpfw")

        # Perturbation + short Nesterov
        for p in range(3):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem, 20)
            a_opt, _ = nesterov_multipeak(f_pert, time_limit=t_pert,
                                           lr0=0.001, mom0=0.92,
                                           top_k=max(15, n_best // 5),
                                           temp0=0.00001)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
