def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def golden_section_cd(f0, time_limit, n_multipliers=5):
    """Coordinate descent with golden section search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    no_pass_improve = 0

    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Golden section search on x = f[j], with others scaled to maintain sum=1
            # Objective: max_k conv(f)[k] where f[j]=x, f[others] = f[others] * (1-x)/denom
            lo = 0.0
            hi = min(0.999, 2.0 * old_val)
            # First evaluate at current point
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            def eval_at(x):
                ft = f.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                return fast_convolve(ft).max()

            f1 = eval_at(c1)
            f2 = eval_at(c2)

            for _ in range(12):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_at(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)

            # Also try the current value and a few multipliers
            cur_val = fast_convolve(f).max()
            if opt_val < cur_val - 1e-14:
                ft = f.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                ft = ft / ft.sum()
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
