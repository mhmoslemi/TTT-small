import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP: A[k,m] = f[k-m] if 0<=k-m<n."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, res
    return None, None

def freq_domain_smooth(f0, time_limit, n_peaks=5, smooth_iters=50):
    """Iteratively smooth the frequency domain to reduce conv peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.05

    for it in range(smooth_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Get top peaks
        k = min(n_peaks, len(conv))
        peak_indices = np.argpartition(conv, -k)[-k:]
        peak_vals = conv[peak_indices]
        max_pv = peak_vals.max()
        if max_pv < 1e-15:
            break

        # Compute gradient: weighted sum of shifted a
        grad = np.zeros(n)
        for i in range(len(peak_indices)):
            ki = int(peak_indices[i])
            weight = (peak_vals[i] / max_pv) ** 2  # emphasize largest peaks
            m_lo = max(0, ki - (n - 1))
            m_hi = min(n - 1, ki)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                grad[indices] += 2.0 * weight * f[ki - indices]

        # Gradient step with projection
        f_new = project_simplex(f - lr * grad)
        new_conv = fast_convolve(f_new)
        new_obj = new_conv.max()

        if new_obj < best_obj:
            best_obj = new_obj
            best_f = f_new.copy()
            f = f_new
            lr = min(lr * 1.02, 0.15)
        else:
            lr *= 0.7
            if lr < 0.001:
                # Try perturbation
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                lr = 0.05

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Cyclic coordinate descent with golden section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_pass_improve = 0
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR

    while time.monotonic() - t0 < time_limit:
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            lo = 0.0
            hi = min(0.9999, 3.0 * old_val)
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            def eval_coord(x):
                ft = f.copy()
                ft[j] = x
                remainder = 1.0 - x
                if remainder < 0:
                    return 999.0
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = f[mask] * (remainder / denom)
                return fast_convolve(ft).max()

            f1 = eval_coord(c1)
            f2 = eval_coord(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_coord(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = fast_convolve(f).max()

            if opt_val < cur_val - 1e-13:
                ft = f.copy()
                ft[j] = opt_x
                remainder = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = f[mask] * (remainder / denom)
                ft = np.maximum(ft, 0.0)
                ft = ft / ft.sum()
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def cosine_init(n, freq, amp):
    """Cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def exp_window_init(n, decay):
    """Exponentially weighted initialization."""
    j = np.arange(n)
    a = np.exp(-decay * j / n) * np.exp(-decay * (n - 1 - j) / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def window_init(n, win_type='blackman'):
    """Window function initialization."""
    j = np.arange(n)
    if win_type == 'blackman':
        a = np.blackman(n)
    elif win_type == 'hann':
        a = np.hanning(n)
    elif win_type == 'hamming':
        a = np.hamming(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"Starting optimization, parent n={n_parent}, score={best_score:.6f}")

    # === Phase 1: AFW on parent ===
    print("\n=== Phase 1: AFW on parent (n={}) ===".format(n_parent))
    t1 = min(200, max(60, (deadline - time.monotonic()) - 1000))
    if t1 > 30:
        print(f"  Allocating {t1:.0f}s")
        a_opt, obj = away_step_fw(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, f"afw_parent_n{n_parent}")

    # === Phase 2: Multi-initialization AFW ===
    print("\n=== Phase 2: Multi-init AFW ===")
    inits_to_try = []

    # Cosine modulated at various frequencies and lengths
    for n_try in [n_parent]:
        for freq in [1, 2, 3, 4, 5, 7, 11]:
            for amp in [0.2, 0.4, 0.6]:
                inits_to_try.append((n_try, cosine_init(n_try, freq, amp), f"cos_f{freq}_a{amp}"))

    # Window functions
    for n_try in [n_parent]:
        inits_to_try.append((n_try, exp_window_init(n_try, 2.0), "expw2"))
        inits_to_try.append((n_try, exp_window_init(n_try, 5.0), "expw5"))
        inits_to_try.append((n_try, window_init(n_try, 'blackman'), "blackman"))
        inits_to_try.append((n_try, window_init(n_try, 'hann'), "hann"))

    # Perturbed parent
    for p in range(3):
        noise_scale = 0.01 * (p + 1)
        pert = parent + rng.standard_normal(n_parent) * noise_scale
        inits_to_try.append((n_parent, project_simplex(pert), f"pert{p}"))

    # Blend parent with structured inits
    for n_try in [n_parent]:
        blend1 = 0.7 * parent + 0.3 * cosine_init(n_try, 3, 0.4)
        inits_to_try.append((n_try, project_simplex(blend1), "blend_cos3"))
        blend2 = 0.8 * parent + 0.2 * exp_window_init(n_try, 3.0)
        inits_to_try.append((n_try, project_simplex(blend2), "blend_exp"))

    # Score all inits quickly
    scored_inits = []
    for n_try, init, label in inits_to_try:
        s = try_seq(init, f"init_{label}")
        scored_inits.append((s, init, label))

    scored_inits.sort(key=lambda x: x[0])

    # Run AFW on top inits
    for s_val, init, label in scored_inits[:6]:
        rem = deadline - time.monotonic() - 500
        if rem < 15:
            break
        t_alloc = min(rem * 0.2, 60)
        if t_alloc < 10:
            t_alloc = max(t_alloc, 15)
        print(f"  AFW on {label} (score={s_val:.6f}), t={t_alloc:.0f}s")
        a_opt, obj = away_step_fw(init, time_limit=t_alloc)
        try_seq(a_opt, f"afw_{label}")

    # === Phase 3: Frequency-domain smoothing on best ===
    print(f"\n=== Phase 3: Freq smoothing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_fs = min(rem * 0.3, 120)
            a_opt, obj = freq_domain_smooth(best_seq, time_limit=t_fs, n_peaks=8)
            try_seq(a_opt, "freq_smooth")

            # Second pass
            rem = deadline - time.monotonic() - 300
            if rem > 15:
                a_opt, obj = freq_domain_smooth(a_opt, time_limit=min(rem, 60), n_peaks=12)
                try_seq(a_opt, "freq_smooth2")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = []
        n_best_cur = len(best_seq)
        for tl in [512, 640, 768, 896, 1024, 1280]:
            if tl != n_best_cur:
                target_lengths.append(tl)

        for tl in target_lengths:
            rem = deadline - time.monotonic() - 300
            if rem < 20:
                break
            f_interp = interpolate_to_length(best_seq, tl)
            # Quick AFW
            t_alloc = min(rem * 0.25, 50)
            a_opt, obj = away_step_fw(f_interp, time_limit=t_alloc)
            s = try_seq(a_opt, f"afw_n{tl}")
            # If it's competitive, do more
            if np.isfinite(s) and s < best_score + 0.01:
                rem = deadline - time.monotonic() - 200
                if rem > 30:
                    t_alloc2 = min(rem * 0.3, 80)
                    a_opt2, _ = away_step_fw(a_opt, time_limit=t_alloc2)
                    try_seq(a_opt2, f"afw_n{tl}_2")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: CD polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_cd = min(rem * 0.5, 150)
            print(f"  CD for {t_cd:.0f}s, n={len(best_seq)}")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "cd_polish")

    # === Phase 6: Final AFW + smoothing combo ===
    print(f"\n=== Phase 6: Final combo (best={best_score:.6f}) ===")
    if best_seq is not None:
        # AFW
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(rem * 0.4, 100)
            a_opt, _ = away_step_fw(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_afw")

        # Freq smooth
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = freq_domain_smooth(best_seq, time_limit=min(rem, 50), n_peaks=10)
            try_seq(a_opt, "final_freq")

        # Short CD
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_cd")

    # === Phase 7: Tiny perturbations ===
    print(f"\n=== Phase 7: Perturbations (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            # Different perturbation strategies
            if p < 3:
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003 * (p + 1))
            elif p < 5:
                # Frequency perturbation
                f_pert = best_seq.copy()
                f_pert[rng.choice(n_best, size=max(5, n_best // 50), replace=False)] *= 1.01
                f_pert = project_simplex(f_pert)
            elif p < 7:
                # Swap pairs
                f_pert = best_seq.copy()
                i1 = rng.integers(0, n_best)
                i2 = rng.integers(0, n_best)
                f_pert[i1], f_pert[i2] = f_pert[i2], f_pert[i1]
                f_pert = project_simplex(f_pert)
            else:
                # Local blur
                f_pert = best_seq.copy()
                center = rng.integers(10, n_best - 10)
                w = 5
                local = f_pert[center-w:center+w+1]
                f_pert[center-w:center+w+1] = np.convolve(local, np.ones(3)/3, 'same')
                f_pert = project_simplex(f_pert)

            # Quick AFW on perturbed
            a_opt, _ = away_step_fw(f_pert, time_limit=min(rem, 15))
            try_seq(a_opt, f"pert_{p}")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
