23. **Implementation details for coordinate pair descent**:

For a given peak k and a pair (p, q) where p + q = k:
- Fix all a[j] for j ≠ p, q
- Let s = a[p] + a[q] (constant)
- Parameterize: a[p] = t, a[q] = s - t, where t ∈ [0, s]
- The auto-convolution b_new[j] as a function of t:
  - For j = k: b[k] = old_b[k] - a[p]*a[q] + t*(s-t) = old_b[k] - a[p]*a[q] + t*s - t^2
    (since the old contribution of a[p]*a[q] to b[k] is a[p]*a[q], and new is t*(s-t))
    
    Wait, actually b[k] = sum_i a[i]*a[k-i]. The terms involving a[p] and a[q] (where p+q=k):
    - i=p: a[p]*a[k-p] = a[p]*a[q]
    - i=q: a[q]*a[k-q] = a[q]*a[p]
    So the total contribution to b[k] from the pair (p,q) is 2*a[p]*a[q] (if p≠q) or a[p]^2 (if p=q, i.e., k=2p).
    
    If p ≠ q: new contribution is 2*t*(s-t) = 2*t*s - 2*t^2
    Old contribution: 2*a[p]*a[q]
    Δb[k] = 2*t*(s-t) - 2*a[p]*a[q] = 2*(t*s - t^2 - a[p]*a[q])
    
  - For j = 2p: b[2p] contains the term a[p]^2 (from i=p, k-i=p). New: t^2. Δb[2p] = t^2 - a[p]^2
  - For j = 2q: b[2q] contains the term a[q]^2. New: (s-t)^2. Δb[2q] = (s-t)^2 - a[q]^2
  - For other j: b[j] contains terms a[p]*a[j-p] (from i=p) and a[q]*a[j-q] (from i=q).
    New: t*a[j-p] + (s-t)*a[j-q]
    Old: a[p]*a[j-p] + a[q]*a[j-q]
    Δb[j] = (t - a[p])*a[j-p] + ((s-t) - a[q])*a[j-q] = δ*a[j-p] - δ*a[j-q] where δ = t - a[p]
    
    (Since s - t - a[q] = a[p] + a[q] - t - a[q] = a[p] - t = -δ)

So all the b[j] values are at most quadratic in t (or δ = t - a[p]). The max of these is a piecewise-quadratic function of t, and we want to minimize it over t ∈ [0, s].

We can find the optimal t by:
1. Evaluate max_j b_new[j] at a grid of t values.
2. Find the approximate minimum.
3. Refine with ternary search or golden section.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j]*g[k-j] <= M
    # The constraint matrix A_ub has shape (2n-1, n)
    # A_ub[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A_ub[k, j] = f[j] if k-n+1 <= j <= k and 0 <= j < n
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in (f*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j]
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)

        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp *= 1.3
            if m_exp > 5.0:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.003 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coord_pair_descent(a0, time_limit, top_peaks=30, grid_size=200):
    """
    Coordinate pair descent: for each of the top peaks, find the dominant
    coordinate pair and optimize the 1D subproblem.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        
        # Get top peaks
        if top_peaks >= len(conv):
            peak_indices = np.arange(len(conv))
        else:
            peak_indices = np.argpartition(conv, -top_peaks)[-top_peaks:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        improved_this_round = False

        for k in peak_indices:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            k = int(k)
            # Find valid range of i for b[k] = sum_i a[i]*a[k-i]
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            
            if i_min > i_max:
                continue
            
            # Find the dominant pair
            # b[k] = sum_{i=i_min}^{i_max} a[i]*a[k-i]
            # The dominant pair is the one with max a[i]*a[k-i]
            products = a[i_min:i_max+1] * a[k-i_max:k-i_min+1][::-1]
            local_best = int(np.argmax(products))
            p = i_min + local_best
            q = k - p
            
            if p == q:
                # Single coordinate optimization
                # b[k] contains a[p]^2, and other b[j] are affected linearly
                # Try small perturbations
                ap = a[p]
                if ap < 1e-10:
                    continue
                # Try reducing a[p] and distributing to neighbors
                for scale in [0.1, 0.2, 0.3, 0.5]:
                    delta = ap * scale
                    a_try = a.copy()
                    a_try[p] = ap - delta
                    # Distribute delta to i and i_max
                    a_try[i_min] += delta / 2
                    a_try[i_max] += delta / 2
                    a_try = np.maximum(a_try, 0)
                    a_try = a_try / a_try.sum()
                    obj = fast_convolve(a_try).max()
                    if obj < best_obj - 1e-15:
                        a = a_try
                        best_obj = obj
                        best_a = a.copy()
                        improved_this_round = True
                        break
                continue
            
            # Pair optimization: a[p] = t, a[q] = s_pq - t where s_pq = a[p] + a[q]
            s_pq = a[p] + a[q]
            if s_pq < 1e-15:
                continue
            
            # Compute the effect on all b[j]
            # For each j, b_new[j] = b_old[j] + delta_j(t)
            # where delta_j depends on t
            
            # The affected b[j] values:
            # b[k]: change = 2*(t*(s_pq-t) - a[p]*a[q]) if p≠q
            # b[2p]: change = t^2 - a[p]^2
            # b[2q]: change = (s_pq-t)^2 - a[q]^2
            # b[j] for other j: linear in t
            
            # Let's just do a grid search over t
            t_grid = np.linspace(0, s_pq, grid_size)
            best_t = a[p]
            best_max = conv.max()
            
            for t_val in t_grid:
                a_try = a.copy()
                a_try[p] = t_val
                a_try[q] = s_pq - t_val
                if a_try[p] < 0 or a_try[q] < 0:
                    continue
                obj = fast_convolve(a_try).max()
                if obj < best_max:
                    best_max = obj
                    best_t = t_val
            
            if best_t != a[p] and best_max < best_obj - 1e-15:
                a[p] = best_t
                a[q] = s_pq - best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_obj = fast_convolve(a).max()
                best_a = a.copy()
                improved_this_round = True

        if not improved_this_round and it > 3:
            # Try random pair perturbations
            rng = np.random.default_rng(it * 104729 + 7)
            for _ in range(10):
                p = int(rng.integers(n))
                q = int(rng.integers(n))
                if p == q:
                    continue
                s_pq = a[p] + a[q]
                if s_pq < 1e-10:
                    continue
                t_new = s_pq * rng.uniform(0.2, 0.8)
                a_try = a.copy()
                a_try[p] = t_new
                a_try[q] = s_pq - t_new
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj = fast_convolve(a_try).max()
                if obj < best_obj - 1e-15:
                    a = a_try
                    best_obj = obj
                    best_a = a.copy()
                    break
            else:
                if it > 20:
                    break

def smoothed_grad_nesterov(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent with Nesterov momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    # Nesterov variables
    a_mom = a.copy()
    v = np.zeros(n)
    T = T_init
    lr = lr_init
    beta = 0.9

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Nesterov lookahead
        a_look = project_simplex(a_mom - beta * v)
        
        conv = fast_convolve(a_look)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient
        grad = 2.0 * np.correlate(p, a_look, mode='valid')

        # Update
        v = beta * v + lr * grad
        a_new = project_simplex(a_mom - v)
        
        # Wait, Nesterov: v_{k+1} = beta*v_k + lr*grad(x_k - beta*v_k)
        # x_{k+1} = x_k - v_{k+1}
        # Let me redo this properly.
        
        # Standard Nesterov:
