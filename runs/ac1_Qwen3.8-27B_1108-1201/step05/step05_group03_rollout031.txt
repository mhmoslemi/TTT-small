"""

1. Entropic (multiplicative weights) subgradient with adaptive temperature
2. Conjugate LP-Frank-Wolfe with relaxed constraints and multi-direction

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(f, M, n, relaxation=1.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M*relaxation for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M * relaxation)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

        conv = fast_convolve(f)
        k_peaks = min(10, 2 * n - 1)
        peak_positions = np.argpartition(conv, -k_peaks)[-k_peaks:]

        # Build priority list of pairs to try
        pair_set = set()
        for k in peak_positions:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            top_local = np.argpartition(products, -min(5, len(idx)))[-min(5, len(idx)):]
            for li in top_local:
                i = idx[li]
                j = k - i
                if i != j:
                    pair_set.add((min(i, j), max(i, j)))

        # Add some random pairs
        for _ in range(n // 8):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pair_set.add((min(i, j), max(i, j)))

        pairs = list(pair_set)
        rng.shuffle(pairs)

        pass_improved = False
        max_pairs = min(len(pairs), 5000)

        for pi in range(max_pairs):
            if time.monotonic() - t0 > time_limit:
                break

            i, j = pairs[pi]
            c = f[i] + f[j]
            if c < 1e-12:
                continue

            # Quick scan of transfer values
            best_t = f[i] / c
            best_val = fast_convolve(f).max()

            candidates = [best_t, 0.0, 1.0, 0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]
            if 0.05 < best_t < 0.95:
                candidates.extend([best_t - 0.15, best_t - 0.05, best_t + 0.05, best_t + 0.15])

            for t in candidates:
                t = np.clip(t, 0.0, 1.0)
                if abs(t - best_t) < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[i] = t * c
                f_trial[j] = (1 - t) * c
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_t = t

            if best_val < fast_convolve(f).max() - 1e-13:
                f[i] = best_t * c
                f[j] = (1 - best_t) * c
                f = f / f.sum()
                if best_val < best_obj - 1e-14:
                    best_obj = best_val
                    best_f = f.copy()
                pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 6:
                # Try a structural reset
                if best_obj < 1.51:
                    break
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve_passes = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    # Light smoothing
    if new_n > 16:
        kernel_size = max(3, new_n // 64)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.ones(kernel_size) / kernel_size
        f_interp = np.convolve(f_interp, kernel, mode='same')
        f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, seed=0):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    init_type = seed % 6

    if init_type == 0:
        # Smooth bump
        x = np.linspace(0, 1, n)
        f = np.exp(-5 * (x - 0.5) ** 2)
    elif init_type == 1:
        # Bimodal
        x = np.linspace(0, 1, n)
        f = np.exp(-8 * (x - 0.25) ** 2) + np.exp(-8 * (x - 0.75) ** 2)
    elif init_type == 2:
        # Cosine window
        x = np.linspace(0, 1, n)
        f = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif init_type == 3:
        # Random smooth
        f = np.ones(n)
        for _ in range(5):
            center = rng.random() * n
            width = rng.uniform(n * 0.1, n * 0.4)
            amp = rng.uniform(-0.5, 0.5)
            x = np.arange(n)
            f += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
        f = np.maximum(f, 0)
    elif init_type == 4:
        # Geometric decay from center
        x = np.arange(n)
        center = n // 2
        f = 0.7 ** np.abs(x - center)
    else:
        # Sinusoidal modulation
        x = np.arange(n)
        f = 1 + 0.3 * np.cos(2 * np.pi * 3 * x / n) + 0.1 * np.cos(2 * np.pi * 7 * x / n)
        f = np.maximum(f, 0)

    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    else:
        f = np.ones(n) / n
    return f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining entropic subgradient, conjugate LP-FW,
    peak-targeted coordinate descent, segment moves, and pairwise polish.
    Explores multiple lengths and diverse initializations.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = structured_init(512, seed=42)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Entropic subgradient on parent ===
    print(f"\n=== Phase 1: Entropic on parent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 800
    t1 = min(rem, 350)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s, n={len(parent)}")
        a_opt, _ = entropic_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "entropic_p1")

    # === Phase 2: Conjugate LP-FW on best ===
    print(f"\n=== Phase 2: Conj LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        t2 = min(rem, 250)
        if t2 > 20:
            print(f"  Time: {t2:.0f}s, n={len(best_seq)}")
            a_opt, _ = lp_fw_conjugate(best_seq, time_limit=t2, verbose=True)
            try_seq(a_opt, "lpfw_conj_p2")

    # === Phase 3: Peak coordinate descent ===
    print(f"\n=== Phase 3: Peak CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        t3 = min(rem, 150)
        if t3 > 15:
            print(f"  Time: {t3:.0f}s, n={len(best_seq)}")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "peakcd_p3")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = [384, 512, 640, 768, 896, 1024]
    target_lengths = [tn for tn in target_lengths if tn != len(base)]

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic() - 300
        if rem < 40:
            break

        f_interp = interpolate_sequence(base, target_n)

        # Quick entropic
        t_alloc = min(rem * 0.3, 80)
        if t_alloc > 10:
            print(f"  Entropic n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = entropic_optimize(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"entropic_n{target_n}")

        # Quick LP-FW
        rem = deadline - time.monotonic() - 250
        if rem > 15 and 'a_opt' in dir():
            t_lpfw = min(rem * 0.5, 100)
            if t_lpfw > 10:
                print(f"  LP-FW n={target_n}, t={t_lpfw:.0f}s")
                a_opt2, _ = lp_fw_conjugate(a_opt, time_limit=t_lpfw, verbose=False)
                try_seq(a_opt2, f"lpfw_n{target_n}")

    # === Phase 5: Diverse structured starts ===
    print(f"\n=== Phase 5: Diverse starts (best={best_score:.8f}) ===")
    n_base = len(best_seq) if best_seq is not None else 768

    for p in range(4):
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        t_alloc = min(rem / max(1, 4 - p), 120)

        if p == 0 and best_seq is not None:
            f_init = best_seq.copy()
        elif p == 1:
            f_init = structured_init(n_base, seed=100 + p)
        elif p == 2:
            f_init = structured_init(n_base, seed=200 + p)
        else:
            if best_seq is not None:
                rng = np.random.default_rng(500 + p)
                f_init = project_simplex(best_seq + rng.standard_normal(n_base) * 0.02)
            else:
                f_init = structured_init(n_base, seed=300 + p)

        # Alternate methods
        if p % 2 == 0:
            a_opt, _ = entropic_optimize(f_init, time_limit=min(t_alloc, 100), verbose=(p == 0))
        else:
            a_opt, _ = lp_fw_conjugate(f_init, time_limit=min(t_alloc, 120), verbose=(p == 0))
        try_seq(a_opt, f"diverse_{p}")

    # === Phase 6: Segment moves ===
    print(f"\n=== Phase 6: Segment moves (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_seg = min(rem * 0.6, 80)
            print(f"  Time: {t_seg:.0f}s, n={len(best_seq)}")
            a_opt, _ = segment_move_optimize(best_seq, time_limit=t_seg)

    # === Phase 7: Intensive pairwise polish ===
    print(f"\n=== Phase 7: Pairwise polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80

            t_pw = min(rem * 0.7, 200)
            print(f"  Time: {t_pw:.0f}s, n={len(best_seq)}")
            a_opt, _ = pairwise_optimize(best_seq, time_limit=t_pw, verbose=True)

    # === Phase 8: Final deep optimization on best ===
    print(f"\n=== Phase 8: Final deep opt (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Conjugate LP-FW final push

        # Entropic final

            a_opt, _ = entropic_optimize(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "entropic_final")

        for trial in range(5):

            if rem < 8:
                break
            rng = np.random.default_rng(700 + trial * 13)

            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = entropic_optimize(f_pert, time_limit=min(rem, 15))

        # Final pairwise

        if rem > 8:

    # Fallback
    if best_seq is None:

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
