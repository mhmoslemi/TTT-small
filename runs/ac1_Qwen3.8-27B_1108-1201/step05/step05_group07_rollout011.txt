"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using peak-directed subgradient descent
with targeted pair moves, frequency-domain spectral flattening, multi-peak
simultaneous reduction, simulated annealing local search, and multi-length
exploration. Key innovations: (1) exact gradient computation for each peak
identifying contributing pairs, (2) danger-map based mass redistribution that
avoids creating new peaks, (3) frequency-domain Gerchberg-Saxton style
flattening, (4) adaptive multi-length search to find optimal n.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    power = A.real * A.real + A.imag * A.imag
    return np.fft.irfft(power, fft_size)[:size]


def objective_from_conv(a, conv):
    """Compute objective given sequence and its autocorrelation."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)


def project_nonneg_sum1(v):
    """Project to {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(len(v)) / len(v)
    return v / s


def compute_peak_gradients(a, conv, top_k, temperature):
    """Compute weighted subgradient of max autocorrelation using top-k peaks."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / max(temperature, 1e-20)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * w[i] * a[k - idx]
    return grad


def peak_directed_descent(a0, time_limit, top_k=30, temperature=1e-5, lr=0.005,
                          momentum=0.85, verbose=False, label=""):
    """
    Peak-directed subgradient descent with momentum and adaptive learning rate.
    Uses exact subgradient computation for top-k peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at current point
        conv = fast_convolve(a)
        sg = compute_peak_gradients(a, conv, cur_top_k, cur_temp)

        # Momentum update
        v = momentum * v + (1.0 - momentum) * sg

        # Nesterov-like step: evaluate at lookahead
        a_look = a * np.exp(-cur_lr * 0.5 * v)
        sl = a_look.sum()
        if sl < 1e-30:
            v *= 0.5
            a_look = a.copy()
        else:
            a_look = a_look / sl

        conv_look = fast_convolve(a_look)
        sg_look = compute_peak_gradients(a_look, conv_look, cur_top_k, cur_temp)
        v = momentum * v + (1.0 - momentum) * sg_look

        # Multiplicative update (maintains non-negativity)
        exponent = -cur_lr * v
        exponent -= exponent.max()
        a = a * np.exp(exponent)
        as_ = a.sum()
        if as_ < 1e-30:
            a = best_a.copy()
            v = np.zeros(n)
            cur_lr *= 0.5
            continue
        a /= as_

        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_lr *= 0.9998
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp *= 1.01

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((no_improve * 7919 + int(t0 * 1e6)) % 1000000)
            noise = rng.standard_normal(n) * (0.003 + 0.002 * min(no_improve // 400, 3))
            a = project_nonneg_sum1(best_a + noise)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            cur_temp = temperature * 0.5
            no_improve = 0

        if verbose and (int((time.monotonic() - t0) * 2) % 10 == 0):
            print(f"    PD iter t={time.monotonic()-t0:.1f}s, obj={best_obj:.10f}")

    return best_a, best_obj


def targeted_local_search(a0, time_limit, verbose=False):
    """
    Targeted local search: identifies contributing pairs for each peak and
    moves mass from high contributors to low-danger positions.
    Uses simulated annealing acceptance for escape from local minima.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = conv.max()
    t0 = time.monotonic()
    temperature = 0.0001
    rng = np.random.default_rng(12345)

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        conv = fast_convolve(a)
        current_max = conv.max()

        # Find top peaks
        top_k = min(15, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        # Compute danger map: for each position, total contribution risk to top peaks
        danger = np.zeros(n)
        for ki in peak_indices[:5]:
            k = int(ki)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            danger[idx] += conv[k] * a[k - idx]

        # Find best move: decrease a contributor, increase a low-danger position
        best_move_gain = 0
        best_decrease_pos = -1
        best_increase_pos = -1
        best_frac = 0.0

        # Try decreasing from top contributors at the main peak
        main_k = int(peak_indices[0])
        lo = max(0, main_k - n + 1)
        hi = min(n - 1, main_k)
        idx = np.arange(lo, hi + 1)
        contribs = a[idx] * a[main_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:10]

        # Candidate increase positions (low danger)
        safe_positions = np.argsort(danger)[:20]

        for ci in top_contrib:
            j = int(idx[ci])
            if a[j] < 1e-8:
                continue
            for si in safe_positions:
                m = int(safe_positions[si])
                if m == j:
                    continue
                # Check if increasing m creates a new peak
                # m contributes to peaks at positions m+j' for j' in range
                # Quick check: does m pair with any top peak contributor?
                increase_risk = 0.0
                for ki2 in peak_indices[:3]:
                    k2 = int(ki2)
                    partner = k2 - m
                    if 0 <= partner < n:
                        increase_risk += a[partner] * conv[k2]
                if increase_risk > contribs[ci] * 0.5:
                    continue

                for frac in [0.15, 0.3, 0.5]:
                    amt = a[j] * frac
                    a[j] -= amt
                    a[m] += amt
                    new_conv = fast_convolve(a)
                    new_max = new_conv.max()
                    a[j] += amt
                    a[m] -= amt

                    if new_max < current_max - 1e-15:
                        gain = current_max - new_max
                        if gain > best_move_gain:
                            best_move_gain = gain
                            best_decrease_pos = j
                            best_increase_pos = m
                            best_frac = frac
                    elif new_max < best_obj + temperature:
                        # Accept with probability (simulated annealing)
                        prob = np.exp(-(new_max - current_max) / max(temperature, 1e-10))
                        if rng.random() < prob:
                            best_move_gain = 0.001
                            best_decrease_pos = j
                            best_increase_pos = m
                            best_frac = frac
                            break

        # Apply best move
        if best_decrease_pos >= 0:
            amt = a[best_decrease_pos] * best_frac
            a[best_decrease_pos] -= amt
            a[best_increase_pos] += amt
            a = np.maximum(a, 0)
            a /= a.sum()

            conv = fast_convolve(a)
            if conv.max() < best_obj - 1e-14:
                best_obj = conv.max()
                best_a = a.copy()

        # Cool down
        temperature *= 0.9995
        if temperature < 1e-8:
            temperature = 1e-8

        # Periodic random restart if no improvement for a long time
        if iteration % 100 == 0 and conv.max() > best_obj * 1.001:
            a = project_nonneg_sum1(best_a + rng.standard_normal(n) * 0.002)

        if verbose and iteration % 200 == 0:
            print(f"    TLS iter {iteration}, t={time.monotonic()-t0:.1f}s, obj={best_obj:.10f}")

    return best_a, best_obj


def frequency_flatten(a0, time_limit, alpha=0.3, verbose=False):
    """
    Gerchberg-Saxton style: flatten the power spectrum to reduce autocorrelation peaks.
    Alternates between time-domain non-negativity projection and frequency-domain
    spectral flattening.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = conv.max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Frequency domain: flatten magnitude
        A = np.fft.rfft(a, 2 * n - 1)
        mag = np.abs(A)
        # Target: flat spectrum (exclude DC and Nyquist for real signal)
        target = np.ones_like(mag)
        target[0] = mag[0]  # Keep DC (sum constraint)
        if n > 1:
            target[-1] = mag[-1]  # Keep Nyquist
        # Blend
        new_mag = (1.0 - alpha) * mag + alpha * target * (mag.mean() / (target[target > 0].mean() + 1e-30))
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, 2 * n - 1)[:n]

        # Time domain: enforce non-negativity
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-30:
            break
        a_new /= sn

        conv_new = fast_convolve(a_new)
        if conv_new.max() < best_obj - 1e-14:
            best_obj = conv_new.max()
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        target_lengths = [t for t in target_lengths if t != len(best_seq)]

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 200
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 80)

            # Quick peak-directed descent
            a_opt, _ = peak_directed_descent(f_interp, time_limit=t_alloc,
                                              top_k=max(20, target_n // 10),
                                              temperature=1e-5, lr=0.004,
                                              momentum=0.87)
            try_seq(a_opt, f"len_{target_n}")

            # Quick local search
            rem = deadline - time.monotonic() - 180
            if rem > 15:
                a_opt2, _ = targeted_local_search(a_opt, time_limit=min(rem * 0.5, 40))
                try_seq(a_opt2, f"len_{target_n}_ls")

    # === Phase 4: Aggressive local search on best ===
    print(f"\n=== Phase 4: Aggressive Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_mps = min(rem * 0.4, 80)
            print(f"  Multi-pair search: t={t_mps:.0f}s")
            a_opt, _ = multi_pair_search(best_seq, time_limit=t_mps)
            try_seq(a_opt, "mps")

        rem = deadline - time.monotonic() - 80
        if rem > 10:
            t_tls2 = min(rem * 0.5, 60)
            print(f"  TLS pass 2: t={t_tls2:.0f}s")
            a_opt, _ = targeted_local_search(best_seq, time_limit=t_tls2)
            try_seq(a_opt, "tls2")

    # === Phase 5: Final fine-tuning ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine peak-directed
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            a_opt, _ = peak_directed_descent(best_seq, time_limit=min(rem, 40),
                                              top_k=n_best, temperature=1e-7,
                                              lr=0.0005, momentum=0.95)
            try_seq(a_opt, "fine_pd")

        # Final local search
        rem = deadline - time.monotonic() - 30
        if rem > 8:
            a_opt, _ = targeted_local_search(best_seq, time_limit=min(rem, 25))
            try_seq(a_opt, "fine_tls")

        # Small perturbations + quick optimize
        for _ in range(6):
            rem = deadline - time.monotonic() - 20
            if rem < 8:
                break
            for scale in [0.001, 0.002, 0.004]:
                rem = deadline - time.monotonic() - 15
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_nonneg_sum1(best_seq + noise)
                a_opt, _ = peak_directed_descent(f_pert, time_limit=min(rem, 8),
                                                  top_k=max(20, n_best // 5),
                                                  temperature=5e-6, lr=0.001,
                                                  momentum=0.92)
                try_seq(a_opt, f"pert_{scale}_{_}")

        # Ultimate final local search
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_opt, _ = targeted_local_search(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "ultimate_tls")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
