import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}"""
    n = len(v)
    # First project onto {x >= 0, sum(x) = 1}, then scale
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.where(mu * np.arange(1, n+1) > (cumsum - target_sum))[0]
    if len(rho) == 0:
        rho = np.where(mu * np.arange(1, n+1) > (cumsum - 1))[0]
        if len(rho) == 0:
            return np.ones(n) * target_sum / n
        theta = (cumsum[rho[-1]] - 1) / (rho[-1] + 1)
        result = np.maximum(v - theta, 0)
    else:
        theta = (cumsum[rho[-1]] - target_sum) / (rho[-1] + 1)
        result = np.maximum(v - theta, 0)
    s = result.sum()
    if s > 1e-15:
        result = result * target_sum / s
    return result

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using Adam optimization on a smoothed
(log-sum-exp) objective with adaptive temperature, combined with coordinate
descent polish and multi-dimension exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(conv, T):
    """Compute softmax weights over convolution values with temperature T."""
    c = conv / T
    c -= c.max()
    w = np.exp(c)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w


def compute_gradient_fft(a, p):
    """Compute cross-correlation gradient: result[i] = sum_k p[k] * a[k-i].
    Uses FFT for O(n log n) computation."""
    n = len(a)
    # Need L >= (2n-1) + n - 1 = 3n - 2 for aperiodic cross-correlation
    L = 1 << (3 * n - 2).bit_length()
    b = np.zeros(L)
    b[:len(p)] = p
    a_pad = np.zeros(L)
    a_pad[:n] = a
    # ifft(fft(b) * conj(fft(a_pad)))[i] = sum_j b[i+j] * a[j] (aperiodic)
    result = np.fft.ifft(np.fft.fft(b) * np.fft.conj(np.fft.fft(a_pad)))
    return result.real[:n]


def smoothed_gradient(a, T):
    """Compute gradient of T * log(sum(exp(conv/T))) w.r.t. a.
    Returns gradient of shape (n,)."""
    n = len(a)
    conv = fast_convolve(a)
    p = softmax_weights(conv, T)
    # grad[i] = 2 * sum_k p[k] * a[k-i] = 2 * cross_corr(p, a)[i]
    return 2.0 * compute_gradient_fft(a, p)


def adam_optimize(f0, time_limit, T_init=0.01, T_final=0.0005,
                  lr=0.05, verbose=True):
    """Adam optimizer on the smoothed objective, with simplex projection.
    f0: initial sequence (will be normalized to sum=1).
    Returns (best_f, best_smoothed_obj, best_true_obj)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    conv = fast_convolve(f)
    best_true_obj = conv.max()
    best_f = f.copy()
    best_smoothed = T_init * np.log(np.sum(np.exp((conv - conv.max()) / T_init)) + 1)

    t0 = time.monotonic()
    total_time = time_limit
    it = 0
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < total_time:
        it += 1
        # Adaptive temperature
        progress = (time.monotonic() - t0) / total_time
        T = T_init * (T_final / T_init) ** progress

        # Compute gradient
        conv = fast_convolve(f)
        p = softmax_weights(conv, T)
        grad = 2.0 * compute_gradient_fft(f, p)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project onto simplex
        f_new = project_simplex(f_new)

        # Evaluate
        conv_new = fast_convolve(f_new)
        true_obj = conv_new.max()
        smoothed_obj = T * np.log(np.sum(np.exp((conv_new - conv_new.max()) / T)))

        f = f_new

        if true_obj < best_true_obj - 1e-14:
            best_true_obj = true_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam iter {it}: true_obj={best_true_obj:.10f}, T={T:.6f}, elapsed={elapsed:.1f}s")
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 500:
            cur_lr *= 0.5
            no_improve = 0
            if cur_lr < 0.001:
                # Try a small perturbation
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            if verbose:
                print(f"    Adam: no improve for 500 iters, lr={cur_lr:.6f}")

        # Reduce learning rate slowly
        cur_lr *= 0.99995

    return best_f, best_true_obj

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == 1:
        return np.ones(new_n) / new_n
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def lp_fw_step(f, n):
    """One LP-FW step: solve min t s.t. (f*g)[k] <= t, g>=0, sum(g)=1.
    Returns the optimal g (normalized to sum=1)."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g /= gs
        else:
            g = np.ones(n) / n
        return g
    return None

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using Adam optimization on a smoothed
(log-sum-exp) objective with adaptive temperature, combined with LP-FW steps,
coordinate descent polish, and multi-dimension exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_weights(conv, T):
    """Compute softmax weights over convolution values with temperature T."""
    c = conv / T
    c -= c.max()
    w = np.exp(c)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w


def compute_gradient_fft(a, p):
    """Compute cross-correlation: result[i] = sum_k p[k] * a[k-i] using FFT."""
    n = len(a)
    L = 1 << (3 * n - 2).bit_length()
    b = np.zeros(L)
    b[:len(p)] = p
    a_pad = np.zeros(L)
    a_pad[:n] = a
    result = np.fft.ifft(np.fft.fft(b) * np.fft.conj(np.fft.fft(a_pad)))
    return result.real[:n]


def smoothed_gradient(a, T):
    """Gradient of T*logsumexp(conv/T) w.r.t. a (up to constant factor)."""
    conv = fast_convolve(a)
    p = softmax_weights(conv, T)
    return 2.0 * compute_gradient_fft(a, p)


def adam_optimize(f0, time_limit, T_init=0.01, T_final=0.0005,
                  lr=0.05, verbose=True):
    """Adam optimizer on smoothed objective with simplex projection and annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    conv = fast_convolve(f)
    best_true_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    total_time = max(time_limit, 1.0)
    it = 0
    no_improve = 0
    cur_lr = lr

    while time.monotonic() - t0 < total_time:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        T = T_init * (T_final / T_init) ** progress

        grad = smoothed_gradient(f, T)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        f_new = f - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        true_obj = conv_new.max()
        f = f_new

        if true_obj < best_true_obj - 1e-14:
            best_true_obj = true_obj
            best_f = f.copy()
            no_improve = 0

                elapsed = time.monotonic() - t0

        else:
            no_improve += 1

        if no_improve > 800:
            cur_lr *= 0.4
            no_improve = 0
            if cur_lr < 0.002:
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.001 + 0.002 * min((it // 800), 5)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                m = np.zeros(n)
                v = np.zeros(n)
                if verbose:
                    print(f"    Adam: restart at iter {it}, scale={scale:.5f}")

    return best_f, best_true_obj


def coordinate_descent_polish(f0, time_limit, verbose=True):

    f = np.maximum(f0.astype(np.float64), 0.0)

    best_f = f.copy()

    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peak_pos = int(np.argmax(conv))
        k = peak_pos
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)

        improved = False

        # Try top 3 contributors

                diff = old_val - new_val
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                others = mask & (f > 1e-10)
                if others.sum() > 0:
                    f_trial[others] += diff * f_trial[others] / f_trial[others].sum()

                    best_obj = val

                    improved = True

def lp_fw_step(f, n):
    """One LP-FW step: min t s.t. (f*g)[k] <= t, g>=0, sum(g)=1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = np.maximum(res.x[:n], 0)
        gs = g.sum()
        if gs > 1e-15:
            g /= gs
        else:
            g = np.ones(n) / n
        return g
    return None


def lp_fw_optimize(f0, time_limit, verbose=True):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = M
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_fw_step(f, n)
        if g is None:
            break
        if np.max(np.abs(g - f)) < 1e-10:
            break
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 1e-10:
            lo, hi = max(0.0, best_alpha - 0.05), min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft /= st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0)
            f /= f.sum()
            M = best_val
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 5 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, t={time.monotonic()-t0:.0f}s")
        else:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length using linear interpolation."""
    old_n = len(f)
    if old_n == 1:
        return np.ones(new_n) / new_n
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_cosine_window(n):
    """Create a cosine window of length n, normalized to sum=1."""
    x = np.linspace(0, np.pi, n)
    w = np.cos(x / 2.0) ** 2
    w = np.maximum(w, 0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    else:
        w = np.ones(n) / n
    return w


def make_beta_window(n, alpha=2.0, beta=2.0):
    """Create a beta-distribution shaped window."""
    x = np.linspace(0.01, 0.99, n)
    w = x ** (alpha - 1) * (1 - x) ** (beta - 1)
    w = np.maximum(w, 0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    else:
        w = np.ones(n) / n
    return w


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: Adam on smoothed objective, LP-FW,
    coordinate descent, dimension search, and diverse initializations.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_cosine_window(832)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default cosine: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Adam on parent (primary) ===
    print(f"\n=== Phase 1: Adam on n={base_n} ===")
    rem = deadline - time.monotonic()
    t1 = min(500, rem * 0.35)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f_opt, _ = adam_optimize(parent, time_limit=t1, T_init=0.01, T_final=0.0004, lr=0.06)
        try_seq(f_opt, "adam_main")

    # Second Adam run with different params
    rem = deadline - time.monotonic()
    t1b = min(200, rem * 0.15)
    if t1b > 20 and best_seq is not None:
        f_opt, _ = adam_optimize(best_seq, time_limit=t1b, T_init=0.005, T_final=0.0002, lr=0.03)
        try_seq(f_opt, "adam_fine")

    # === Phase 2: Dimension search ===
    print(f"\n=== Phase 2: Dimension search (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    candidate_lengths = sorted(set([
        base_n - 24, base_n - 12, base_n + 12, base_n + 24,
        base_n + 48, base_n + 64, base_n + 96, base_n + 128,
    ]))
    candidate_lengths = [l for l in candidate_lengths if 300 <= l <= 1500 and l != base_n][:4]

    for target_n in candidate_lengths:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_alloc = min(rem * 0.25, 250)
        if t_alloc < 25:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  Adam n={target_n}, t={t_alloc:.0f}s")
        f_opt, _ = adam_optimize(f_interp, time_limit=t_alloc, T_init=0.008, T_final=0.0003, lr=0.05)
        try_seq(f_opt, f"dim_{target_n}")

    # === Phase 3: LP-FW refinement ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.2, 300)
        if t_lp > 30:
            print(f"  n={len(best_seq)}, t={t_lp:.0f}s")
            f_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp)
            try_seq(f_opt, "lpfw")
            rem = deadline - time.monotonic()
            if rem > 30:
                f_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem * 0.3, 150))
                try_seq(f_opt, "lpfw_2")

    # === Phase 4: Diverse initializations ===
    print(f"\n=== Phase 4: Diverse starts (best={best_score:.8f}) ===")
    n_try = len(best_seq) if best_seq is not None else base_n
    diverse_inits = [
        (make_cosine_window(n_try), "cosine"),
        (make_beta_window(n_try, 2.0, 2.0), "beta_2_2"),
        (make_beta_window(n_try, 1.5, 1.5), "beta_1_5"),
        (make_beta_window(n_try, 3.0, 3.0), "beta_3_3"),
    ]
    for f_init, label in diverse_inits:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.1, 120)
        if t_alloc < 20:
            break
        # Blend with best
        blend = 0.3 * f_init + 0.7 * (best_seq if best_seq is not None else f_init)
        blend = np.maximum(blend, 0)
        blend /= blend.sum()
        f_opt, _ = adam_optimize(blend, time_limit=t_alloc, T_init=0.006, T_final=0.0003, lr=0.04)
        try_seq(f_opt, f"div_{label}")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: CD Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.25, 300)
        if t_cd > 20:
            print(f"  n={len(best_seq)}, t={t_cd:.0f}s")
            f_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(f_opt, "cd_main")

    # === Phase 6: Final combined polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Small Adam
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            f_opt, _ = adam_optimize(best_seq, time_limit=min(rem * 0.4, 150),
                                     T_init=0.003, T_final=0.0001, lr=0.015)
            try_seq(f_opt, "adam_final")

        # CD again
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            f_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem * 0.4, 100))
            try_seq(f_opt, "cd_final")

        # Tiny Adam
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            f_opt, _ = adam_optimize(best_seq, time_limit=min(rem * 0.5, 80),
                                     T_init=0.001, T_final=0.00005, lr=0.008)
            try_seq(f_opt, "adam_tiny")

        # Symmetrized
        sym = (best_seq + best_seq[::-1]) / 2.0
        sym = np.maximum(sym, 0)
        ss = sym.sum()
        if ss > 1e-10:
            sym /= ss
            try_seq(sym, "symmetric")
            # Polish symmetrized
            rem = deadline - time.monotonic() - 15
            if rem > 10:
                f_opt, _ = adam_optimize(sym, time_limit=min(rem * 0.6, 60),
                                         T_init=0.002, T_final=0.0001, lr=0.01)
                try_seq(f_opt, "sym_polish")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
