def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_reduced(f, M_target, active_indices, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M_target for k in active_indices, g >= 0.
    Returns normalized g or None."""
    m = len(active_indices)
    c = -np.ones(n)
    A_ub = np.zeros((m, n))
    for i, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based Frank-Wolfe with reduced active set (top-K constraints)
2. Subgradient descent with adaptive softmax-weighted peaks
3. Peak-targeting coordinate descent with analytical sub-optimization
4. Multi-length exploration (768, 832, 896, 1024)
5. Aggressive perturbation and restart strategy

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix for (f*g)[k] <= M constraints.
    A[i, l] = f[indices[i] - l] if 0 <= indices[i]-l < n, else 0."""
    m = len(indices)
    A = np.zeros((m, n))
    for i in range(m):
        k = indices[i]
        l_start = max(0, k - n + 1)
        l_end = min(n - 1, k)
        f_start = max(0, k - (n - 1))
        f_end = min(n - 1, k)
        # A[i, l] = f[k-l] for l in [l_start, l_end]
        # When l = l_start: f[k - l_start]
        # When l = l_end: f[k - l_end]
        # f indices go from f[k-l_end] to f[k-l_start]
        # k - l_end = k - min(n-1, k)
        # k - l_start = k - max(0, k-n+1)
        a_start = k - l_end
        a_end = k - l_start
        # A[i, l_start:l_end+1] = f[a_start:a_end+1][::-1]
        A[i, l_start:l_end + 1] = f[a_start:a_end + 1][::-1]
    return A


def solve_lp_active(f, active_idx, M_val, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M_val for k in active_idx, g >= 0.
    Returns normalized g or None."""
    m = len(active_idx)
    if m == 0:
        return np.ones(n) / n
    
    c = -np.ones(n)
    A_ub = build_constraint_matrix(f, active_idx, n)
    b_ub = np.full(m, M_val)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            gs = res.x.sum()
            if gs > 1e-15:
                return res.x / gs
    except Exception:
        pass
    return None

def sg_optimize(f0, time_limit, lr=0.003, momentum=0.94, top_k=50, temp=0.00005,
                verbose=False):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # (f*g)[k] = sum_j f[j]*g[k-j], subgradient w.r.t. g at position l is f[k-l]
            # So sg[l] += w[i] * 2 * f[k-l] for valid l
            l_start = max(0, k - n + 1)
            l_end = min(n - 1, k)
            sg[l_start:l_end + 1] += 2.0 * w[i] * f[k - l_end:k - l_start + 1][::-1]
        
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new
        
        cur_lr *= 0.9999
        cur_temp *= 1.003
        
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0005)
            no_improve = 0
    
    return best_f, best_obj


def peak_cd(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        
        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            # Try modifying coordinates in the range [js, je]
            # Focus on coordinates with largest contribution
            contrib = np.zeros(je - js + 1)
            for j in range(js, je + 1):
                kg = peak_k - j
                if 0 <= kg < n:
                    contrib[j - js] = f[j] * f[kg]
            
            # Sort by contribution (descending)
            order = np.argsort(contrib)[::-1]
            n_try = min(30, len(order))
            
            for oi in range(n_try):
                if time.monotonic() - t0 > time_limit:
                    break
                j = js + int(order[oi])
                old_val = f[j]
                if old_val < 1e-12:
                    continue
                
                best_j_val = old_val
                best_j_obj = conv.max()
                
                for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val
                
                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                    
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)
            
            if improved and time.monotonic() - t0 > time_limit:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based Frank-Wolfe with min-peak LP and reduced active set
2. Subgradient descent with adaptive softmax-weighted peaks

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, indices, n):
    """Build matrix where A[i,l] = f[indices[i]-l] if 0<=indices[i]-l<n else 0."""
    m = len(indices)
    A = np.zeros((m, n))
    for i in range(m):
        k = indices[i]
        l_start = max(0, k - n + 1)
        l_end = min(n - 1, k)
        a_start = k - l_end
        a_end = k - l_start
        A[i, l_start:l_end + 1] = f[a_start:a_end + 1][::-1]
    return A


def solve_lp_min_peak(f, active_idx, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for k in active_idx, g>=0, sum(g)=1.
    Returns (g_normalized, t) or (None, None)."""
    m = len(active_idx)
    if m == 0:
        return np.ones(n) / n, 0.0
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = build_constraint_matrix(f, active_idx, n)
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            g = np.maximum(res.x[:n], 0.0)
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
                t = float(res.x[-1])
                return g, t
    except Exception:
        pass
    return None, None


def solve_lp_max_sum(f, active_idx, M_val, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M_val for k in active_idx, g>=0.
    Returns normalized g or None."""
    m = len(active_idx)
    if m == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    A_ub = build_constraint_matrix(f, active_idx, n)
    b_ub = np.full(m, M_val)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None

def sg_optimize(f0, time_limit, lr=0.003, momentum=0.94, top_k=50, temp=0.00005,
                verbose=False):
    """Subgradient method with softmax-weighted peaks and Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient efficiently
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            l_start = max(0, k - n + 1)
            l_end = min(n - 1, k)
            a_start = k - l_end
            a_end = k - l_start
            sg[l_start:l_end + 1] += 2.0 * w[i] * f[a_start:a_end + 1][::-1]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.99992
        cur_temp *= 1.004

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0004)
            no_improve = 0

    return best_f, best_obj


def peak_cd(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Focus on high-contribution coordinates
            contribs = []
            for j in range(js, je + 1):
                kg = peak_k - j
                if 0 <= kg < n:
                    contribs.append((f[j] * f[kg], j))
            contribs.sort(key=lambda x: -x[0])

            n_try = min(40, len(contribs))
            for ci in range(n_try):
                if time.monotonic() - t0 > time_limit:
                    break
                j = contribs[ci][1]
                old_val = f[j]
                if old_val < 1e-12:
                    continue

                best_j_val = old_val
                best_j_obj = conv.max()

                for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 6:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            rem = deadline - time.monotonic()
            if rem > 30:
                a_opt, _ = sg_optimize(a_opt, time_limit=min(rem * 0.5, 70),
                                       lr=0.001, momentum=0.94,
                                       top_k=70, temp=0.00003)
                best_seq, best_score = try_update(a_opt, best_seq, best_score, "sg_896")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick LP-FW on current best
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 100), top_k=100)
            best_seq, best_score = try_update(a_opt, best_seq, best_score, "lpfw_polish")

        # Quick SG
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = sg_optimize(best_seq, time_limit=min(rem, 60),
                                   lr=0.0008, momentum=0.95,
                                   top_k=max(30, n_best // 12),
                                   temp=0.00002)
            best_seq, best_score = try_update(a_opt, best_seq, best_score, "sg_polish")

        # Quick CD
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = peak_cd(best_seq, time_limit=min(rem, 40))
            best_seq, best_score = try_update(a_opt, best_seq, best_score, "cd_polish")

        # Perturbations with short optimization
        for p in range(10):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise_scale = 0.001 + 0.0006 * p
            noise = rng.standard_normal(len(best_seq)) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = sg_optimize(f_pert, time_limit=min(rem, 18),
                                   lr=0.0005, momentum=0.95,
                                   top_k=max(20, len(best_seq) // 15),
                                   temp=0.00001)
            best_seq, best_score = try_update(a_opt, best_seq, best_score, f"pert_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
