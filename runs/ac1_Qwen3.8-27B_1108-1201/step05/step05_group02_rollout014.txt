"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smooth gradient optimization
with log-sum-exp peak approximation, multi-length exploration, and
coordinate descent polishing.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_multiply(w, f):
    """Compute Hankel(w) @ f where Hankel(w)[i,j] = w[i+j].
    w: array of length 2n-1
    f: array of length n
    Returns: array of length n
    Uses FFT for O(n log n) computation.
    """
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    f_rev = f[::-1].copy()
    W = np.fft.rfft(w, fft_size)
    FR = np.fft.rfft(f_rev, fft_size)
    conv = np.fft.irfft(W * FR, fft_size)
    return conv[n - 1: n - 1 + n]


def softmax(x, tau):
    """Stable softmax with temperature."""
    x_scaled = x / tau
    x_scaled -= x_scaled.max()
    e = np.exp(x_scaled)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def smooth_objective(f, tau):
    """Compute the smooth objective (log-sum-exp of convolution)."""
    conv = fast_convolve(f)
    return tau * np.log(np.sum(np.exp(conv / tau)))


def optimize_smooth(f0, time_limit, tau=0.001, lr=0.01, verbose=False):
    """Gradient descent on the smooth log-sum-exp objective."""
    f = project_simplex(f0.astype(np.float64).copy())
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break
        
        conv = fast_convolve(f)
        weights = softmax(conv, tau)
        grad = 2.0 * hankel_multiply(weights, f)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Use a fixed step size scaled by problem size
        step = lr / max(grad_norm, 1e-10)
        # Clamp step to avoid too large moves
        step = min(step, 0.5)
        
        f_new = project_simplex(f - step * grad)
        
        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"      smooth iter {it}: obj={best_obj:.10f}")
        else:
            f = f_new
            no_improve += 1
        
        # Decay learning rate
        lr = max(lr * 0.999, 0.0001)
        # Increase tau slowly (anneal)
        tau = min(tau * 1.001, 0.01)
        
        if no_improve > 200:
            # Restart with noise
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            no_improve = 0
            lr = 0.01
            tau = 0.001
    
    return best_f, best_obj / (2.0 * n)  # Return just max(conv) for normalized f


def optimize_subgrad(f0, time_limit, top_k=50, verbose=False):
    """Subgradient method on the actual peak."""
    f = project_simplex(f0.astype(np.float64).copy())
    n = len(f)
    
    conv = fast_convolve(f)
    best_peak = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    lr = 0.005
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break
        
        conv = fast_convolve(f)
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                kg = k - j
                if 0 <= kg < n:
                    sg[j] += 2.0 * f[kg]
        
        # Normalize and update
        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        peak_new = conv_new.max()
        
        if peak_new < best_peak - 1e-15:
            best_peak = peak_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"      subgrad iter {it}: peak={best_peak:.10f}, obj={2*n*best_peak:.10f}")
        else:
            no_improve += 1
        
        f = f_new
        lr *= 0.9999
        momentum = min(momentum * 1.0001, 0.95)
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = 0.003
            no_improve = 0
    
    return best_f, 2.0 * n * best_peak


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent: for each coordinate, find optimal value."""
    f = project_simplex(f0.astype(np.float64).copy())
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Try each coordinate (or a subset for large n)
        if n > 2000:
            # Only try coordinates near the peak
            peak_pos = int(np.argmax(conv))
            js = max(0, peak_pos - n + 1)
            je = min(n - 1, peak_pos)
            coords = list(range(js, je + 1))
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            coords.extend([int(rng.integers(0, n)) for _ in range(min(100, n // 20))])
        else:
            coords = list(range(n))
        
        for j in coords:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_val = old_val
            best_local_obj = best_obj
            
            # Try a few candidate values
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                
                # Rescale others to maintain sum=1
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                s = f_trial.sum()
                if s < 1e-15:
                    continue
                f_trial /= s
                
                val = 2.0 * n * fast_convolve(f_trial).max()
                if val < best_local_obj - 1e-14:
                    best_local_obj = val
                    best_val = new_val
            
            if best_val != old_val:
                # Apply the change
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    f[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    s = f.sum()
                    if s > 1e-15:
                        f /= s
                
                if best_local_obj < best_obj - 1e-14:
                    best_obj = best_local_obj
                    best_f = f.copy()
                    conv = fast_convolve(f)
                    improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result


def gaussian_init(n, sigma_frac=0.3):
    """Create a Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    sigma = sigma_frac
    center = 0.5
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    g = np.maximum(g, 0)
    s = g.sum()
    if s > 1e-15:
        g /= s
    return g

    best_seq = None
