where b_m = sum_j w_{m+j} * a_j (the cross-correlation computed above) and φ_β = (1/β)*log(sum exp(β*C_k)).

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed-objective L-BFGS-B
optimization with progressive temperature annealing, multi-start exploration,
FFT-based gradient computation, and targeted local refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smoothed_max_and_weights(conv, beta):
    """Compute smoothed max (log-sum-exp) and softmax weights."""
    b = beta * conv
    b -= b.max()
    exp_b = np.exp(b)
    w = exp_b / exp_b.sum()
    phi = np.log(exp_b.sum()) / beta
    return phi, w


def compute_crosscorr(w, a):
    """Compute b_m = sum_j w[m+j]*a[j] for m=0..n-1 using FFT."""
    n = len(a)
    w_rev = w[::-1]
    N = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w_rev, N)
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(W * A, N)
    return conv[n - 1:2 * n - 1][::-1]


def obj_and_grad(a, beta):
    """Compute the smoothed objective 2n*phi/S^2 and its gradient."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return 2.0, np.zeros(n)

    conv = fast_convolve(a)
    phi, w = smoothed_max_and_weights(conv, beta)

    obj = 2.0 * n * phi / (S * S)

    # b_m = sum_j w[m+j]*a[j]
    b = compute_crosscorr(w, a)

    # grad_m = 4n/S^2 * (b_m - phi/S)
    grad = 4.0 * n / (S * S) * (b - phi / S)

    return obj, grad


def lbfgs_optimize(f0, beta, time_limit, max_iter=3000):
    """Optimize the smoothed objective using L-BFGS-B with time callback."""
    a0 = np.maximum(f0, 0.0).copy()
    S = a0.sum()
    if S < 1e-15:
        a0 = np.ones(len(f0)) / len(f0)
        S = 1.0

    state = {'t0': time.monotonic(), 'done': False}

    def fun(x):
        if state['done']:
            return 2.0
        o, _ = obj_and_grad(x, beta)
        return o

    def jac(x):
        if state['done']:
            return np.zeros(len(x))
        _, g = obj_and_grad(x, beta)
        return g

    def callback(xk):
        if time.monotonic() - state['t0'] > time_limit:
            state['done'] = True
        return state['done']

    result = minimize(fun, a0, jac=jac, method='L-BFGS-B',
                      bounds=[(0, None)] * len(a0),
                      callback=callback,
                      options={'maxiter': max_iter, 'ftol': 1e-12,
                               'gtol': 1e-7, 'maxfun': 50000,
                               'disp': False})

    return np.maximum(result.x, 0.0)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descent_refine(f0, time_limit):
    """Coordinate descent on the true (non-smoothed) objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    S = f.sum()
    if S < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / S
    n = len(f)

    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    t0 = time.monotonic()

    mid = n // 2
    order = []
    for d in range(mid + 1):
        if mid + d < n:
            order.append(mid + d)
        if mid - d >= 0 and d > 0:
            order.append(mid - d)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue

            cur_conv_max = fast_convolve(f).max()
            cur_obj = 2.0 * n * cur_conv_max

            best_j_val = old_val
            best_j_obj = cur_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                f[j] = new_val
                S_new = S - old_val + new_val
                if S_new < 1e-10:
                    f[j] = old_val
                    continue
                f_norm = f / S_new
                val = 2.0 * n * fast_convolve(f_norm).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
                f[j] = old_val

            if best_j_val != old_val:
                f[j] = best_j_val
                S = S - old_val + best_j_val
                f = f / S
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f


def peak_targeted_search(f0, time_limit, max_rounds=200):
    """Target mass transfer from high-contributing to low-contributing coordinates near peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    S = f.sum()
    if S < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / S
    n = len(f)

    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for _ in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        cur_max = conv.max()

        # Get top peaks
        num_peaks = min(20, 2 * n - 1)
        top_idx = np.argpartition(conv, -num_peaks)[-num_peaks:]

        improved = False
        for k in top_idx:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]

            if len(idx) < 2:
                continue

            hi_pos = idx[np.argmax(contribs)]
            lo_pos = idx[np.argmin(contribs)]

            if hi_pos == lo_pos or f[hi_pos] < 1e-10:
                continue

            # Try transferring mass
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                amt = f[hi_pos] * frac
                f[hi_pos] -= amt
                f[lo_pos] += amt
                new_obj = 2.0 * n * fast_convolve(f / f.sum()).max()
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    S = f.sum()
                    f = f / S
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[hi_pos] += amt
                    f[lo_pos] -= amt
                    break

        # Also try random transfers
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or f[i] < 1e-10:
                continue
            for frac in [0.05, 0.1, 0.2]:
                amt = f[i] * frac
                f[i] -= amt
                f[j] += amt
                new_obj = 2.0 * n * fast_convolve(f / f.sum()).max()
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    S = f.sum()
                    f = f / S
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] += amt
                    f[j] -= amt
                    break

        if not improved:
            # Try perturbation
            noise = rng.standard_normal(n) * 0.003
            f_pert = project_simplex(f + noise)
            pert_obj = 2.0 * n * fast_convolve(f_pert).max()
            if pert_obj < best_obj:
                f = f_pert
                best_obj = pert_obj
                best_f = f.copy()
            else:
                break

    return best_f


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    center = n / 2.0
    sigma = n / (4.0 + rng.random() * 4.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 100)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Super-Gaussian (sharper)
    power = 3.0 + rng.random() * 5.0
    supg = np.exp(-((np.arange(n) - center) / (n * 0.25)) ** power)
    inits.append(supg / supg.sum())

    return inits[:count]
