def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        # FISTA momentum update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = candidate + (t_k - 1.0) / t_next * (candidate - a)
        y = project_simplex(y)
        a = candidate
        t_k = t_next
        
        # Check improvement
        conv_a = fast_convolve(a)
        M_a = conv_a.max()
        if M_a < best_M - 1e-14:
            best_M = M_a
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart
        if no_improve > 100:
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            noise = 0.001 + 0.004 * min(1.0, (no_improve - 100) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            y = a.copy()
            t_k = 1.0
            no_improve = 0
            best_M = fast_convolve(a).max()
        
    return best_a, best_M
