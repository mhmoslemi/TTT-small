def project_simplex(v):
    v=np.asarray(v,dtype=np.float64)
    n=len(v)
    if n==0: return v
    if n==1: return np.array([max(0.0,v[0])])
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.arange(1,n+1)
    cond = u*rho > (cssv-1.0)
    if not cond.any():
        return np.ones(n)/n
    r=int(np.nonzero(cond)[0][-1])
    theta=(cssv[r]-1.0)/(r+1.0)
    out=np.maximum(v-theta,0.0)
    s=out.sum()
    if s>0: out/=s # ensure sum 1? Projection with theta should sum1, but numeric.
    else: out=np.ones(n)/n
    return out
```

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    c = np.maximum(c, 0.0)
    return c

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    if n == 0: return v
    if n == 1: return np.array([max(0.0, float(v[0]))])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
       out = np.ones(n, dtype=np.float64)/n
    else:
       r = int(np.nonzero(cond)[0][-1])
       theta = (cssv[r] - 1.0)/(r+1.0)
       out = np.maximum(v - theta, 0.0)
       s = float(out.sum())
       if s <= 0.0:
          out = np.ones(n)/n
       else:
          out /= s
    return out
```

def interpolate_to_length(f, new_n):
    f=normalize_nonneg(f,1.0)
    old_n=len(f)
    if new_n <=0: return np.array([])

    if old_n == 1: return np.full(new_n, f[0]/new_n)
    old_x=np.linspace(0.0,1.0,old_n)
    new_x=np.linspace(0.0,1.0,new_n)
    y=np.interp(new_x, old_x, f)

def compute_subgradient(a, conv, temp=1e-4, top_k=64):
   n=len(a); m=len(conv)
   if m==0: return np.zeros(n)
   k=int(top_k)
   if k <=0 or k >= m: peaks=np.arange(m)
   else: peaks=np.argpartition(conv, -k)[-k:]
   pv=conv[peaks]
   t=float(temp)
   if not np.isfinite(t) or t <=0: t=1e-12
   scaled=pv/t
   if scaled.size:
       scaled = scaled - float(np.max(scaled))
       w=np.exp(scaled)
       ws=float(np.sum(w))
       if ws <= 1e-300 or not np.isfinite(ws):
           w=np.zeros_like(w); w[int(np.argmax(pv))]=1.0
       else: w/=ws
   else: return np.zeros(n)
   sg=np.zeros(n,dtype=np.float64)
   for ii in range(peaks.size):
       wi=float(w[ii])
       if wi < 1e-15: continue
       kk=int(peaks[ii])
       js=max(0, kk-n+1)
       je=min(n-1, kk)
       if js > je: continue
       idx=np.arange(js, je+1)
       sg[idx] += 2.0 * wi * a[kk - idx]
   return sg

def conv_after_transfer(a, conv, src, dst, amount):
    if src == dst or amount <= 0.0: return None
    n=len(a)
    if amount > a[src] + 1e-12: return None
    new_conv=conv.copy()
    two_amount = 2.0*amount
    new_conv[src:src+n] -= two_amount * a
    new_conv[dst:dst+n] += two_amount * a
    d2=amount*amount
    new_conv[2*src] += d2
    new_conv[2*dst] += d2
    new_conv[src+dst] -= 2.0*d2
    new_conv = np.maximum(new_conv, 0.0)
    return new_conv

"""
Searches for non-negative step-function heights minimizing
    2 * n * max(conv(a, a)) / (sum(a))^2
by combining:
  - projected smooth subgradient descent,
  - linear-programming directions (full and active-set),
  - Frank-Wolfe-style gradient LP directions,
  - exact-ish star-coordinate polishing by scalar minimization,
  - fast sparse pair-transfer local search,
  - symmetric / perturbation / interpolation restarts.
"""

import numpy as np
import time

try:
    from scipy import optimize as sp_opt
    HAS_SCIPY = True
except Exception:
    HAS_SCIPY = False


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    c[c < 0.0] = 0.0
    return c


def internal_score(a, conv=None):
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if s <= 1e-15 or not np.isfinite(s):
        return float("inf")
    if conv is None:
        conv = fast_convolve(a)
    if conv.size == 0:
        return float("inf")
    m = float(np.max(conv))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * len(a) * m / (s * s))


def actual_score(a):
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
        vals = [float(x) for x in arr]
        return float(evaluate_sequence(vals))
    except Exception:
        return float("inf")


def normalize_nonneg(a, target=1.0):
    arr = np.asarray(a, dtype=np.float64).ravel().copy()
    if arr.size == 0:
        return arr
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-300:
        return np.full(arr.shape, float(target) / arr.size)
    return arr * (float(target) / s)


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(0.0, float(v[0]))])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        out = np.ones(n, dtype=np.float64) / n
    else:
        r = int(np.nonzero(cond)[0][-1])
        theta = (cssv[r] - 1.0) / (r + 1.0)
        out = np.maximum(v - theta, 0.0)
        s = float(out.sum())
        if s <= 0.0:
            out = np.ones(n, dtype=np.float64) / n
        else:
            out /= s
    return out


def symmetrize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    out = 0.5 * (a + a[::-1])
    return normalize_nonneg(out, 1.0)


def interpolate_to_length(f, new_n):
    f = normalize_nonneg(f, 1.0)
    old_n = len(f)
    if new_n <= 0:
        return np.empty(0, dtype=np.float64)
    if old_n == new_n:
        return f.copy()
    if old_n == 0:
        return np.full(new_n, 1.0 / new_n)
    if old_n == 1:
        return np.full(new_n, float(f[0]) / new_n)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    y = np.interp(new_x, old_x, f)
    return normalize_nonneg(y, 1.0)


def compute_subgradient(a, conv, temp=1e-4, top_k=64):
    n = len(a)
    m = len(conv)
    if m == 0:
        return np.zeros(n, dtype=np.float64)

    k = int(top_k)
    if k <= 0 or k >= m:
        peaks = np.arange(m)
    else:
        peaks = np.argpartition(conv, -k)[-k:]

    pv = conv[peaks]
    t = float(temp)
    if not np.isfinite(t) or t <= 0.0:
        t = 1e-12

    scaled = pv / t
    scaled = scaled - float(np.max(scaled))
    w = np.exp(scaled)
    ws = float(np.sum(w))
    if ws <= 1e-300 or not np.isfinite(ws):
        w = np.zeros_like(w)
        w[int(np.argmax(pv))] = 1.0
    else:
        w /= ws

    sg = np.zeros(n, dtype=np.float64)
    for ii in range(peaks.size):
        wi = float(w[ii])
        if wi < 1e-15:
            continue
        kk = int(peaks[ii])
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * wi * a[kk - idx]
    return sg


def subgrad_refine(a0, time_limit=60.0, lr=0.02, momentum=0.90,
                   temp_fraction=0.03, top_k=None, noise_scale=0.002,
                   seed=1, verbose=False):
    a = normalize_nonneg(a0, 1.0)
    n = len(a)
    if top_k is None:
        top_k = max(32, min(192, n // 4))

    conv = fast_convolve(a)
    current_obj = internal_score(a, conv)
    best = a.copy()
    best_obj = current_obj

    t0 = time.monotonic()
    v = np.zeros(n, dtype=np.float64)
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0
    cur_lr = float(lr)

    while time.monotonic() - t0 < time_limit:
        if time_limit - (time.monotonic() - t0) < 0.5:
            break
        it += 1
        M = float(np.max(conv)) if conv.size else 0.0
        temp = max(1e-12, temp_fraction * M)
        sg = compute_subgradient(a, conv, temp, top_k)
        if not np.all(np.isfinite(sg)):
            sg = np.zeros(n, dtype=np.float64)

        v = momentum * v - cur_lr * sg
        if not np.all(np.isfinite(v)):
            v = np.zeros(n, dtype=np.float64)

        accepted = False
        for scale in (1.0, 0.5, 0.25, 0.125):
            cand = project_simplex(a + scale * v)
            if np.linalg.norm(cand - a) < 1e-16:
                continue
            convc = fast_convolve(cand)
            objc = internal_score(cand, convc)
            if objc < current_obj - 1e-13:
                a = cand
                conv = convc
                current_obj = objc
                accepted = True
                if objc < best_obj - 1e-14:
                    best_obj = objc
                    best = a.copy()
                no_improve = 0
                break

        if not accepted:
            no_improve += 1

        if no_improve > 120:
            sn = noise_scale * min(3.0, 1.0 + no_improve / 100.0)
            a = project_simplex(best + rng.standard_normal(n) * sn)
            conv = fast_convolve(a)
            current_obj = internal_score(a, conv)
            v = np.zeros(n, dtype=np.float64)
            no_improve = 0
            cur_lr = float(lr) * 0.7

        cur_lr *= 0.99995

        if verbose and it % 200 == 0:
            print(f"      subgrad it={it} obj={best_obj:.8f}")

    return best, best_obj


def star_move_phi(x, B, b_valid, diag_idx):
    om = 1.0 - x
    vals = (om * om) * B + (2.0 * x * om) * b_valid
    if 0 <= diag_idx < len(vals):
        vals[diag_idx] += x * x
    return float(np.max(vals))


def coordinate_polish(a0, time_limit=60.0, samples=32, refine_samples=12,
                      seed=2, no_improve_limit=2, verbose=False):
    a = normalize_nonneg(a0, 1.0)
    n = len(a)
    L = 2 * n - 1

    conv = fast_convolve(a)
    current_max = float(np.max(conv))
    best = a.copy()
    best_max = current_max

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    passes = 0
    no_pass = 0

    large = n > 1200
    sub_size = min(n, 768) if large else n

    while time.monotonic() - t0 < time_limit:
        passes += 1

        if large:
            order = rng.choice(n, size=sub_size, replace=False)
        else:
            order = np.arange(n)
            if passes > 1:
                rng.shuffle(order)

        pass_improved = False

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            x_old = float(a[j])
            s_others = 1.0 - x_old
            if s_others < 1e-12 or x_old < -1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_convolve(b)
            b_valid = np.zeros(L, dtype=np.float64)
            b_valid[j:j + n] = b
            diag_idx = 2 * j

            xs = np.linspace(0.0, 1.0, max(16, samples))
            xs = np.unique(np.concatenate((xs, np.array([x_old]))))

            best_val = float("inf")
            best_x = x_old
            for xx in xs:
                val = star_move_phi(float(xx), B, b_valid, diag_idx)
                if val < best_val:
                    best_val = val
                    best_x = float(xx)

            half = 2.0 / max(2, samples)
            lo = max(0.0, best_x - half)
            hi = min(1.0, best_x + half)
            if hi - lo > 1e-8:
                xs2 = np.linspace(lo, hi, max(8, refine_samples))
                for xx in xs2:
                    val = star_move_phi(float(xx), B, b_valid, diag_idx)
                    if val < best_val:
                        best_val = val
                        best_x = float(xx)

            if best_val < current_max - 1e-12:
                x_new = best_x
                factor = (1.0 - x_new) / s_others
                a[j] = x_new
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= factor
                a = np.maximum(a, 0.0)
                a = normalize_nonneg(a, 1.0)

                conv = fast_convolve(a)
                current_max = float(np.max(conv))
                if current_max < best_max - 1e-13:
                    best_max = current_max
                    best = a.copy()
                    pass_improved = True

        if pass_improved:
            no_pass = 0
        else:
            no_pass += 1
            if no_pass >= no_improve_limit:
                break
            a = project_simplex(best + rng.standard_normal(n) * 0.0015)
            conv = fast_convolve(a)
            current_max = float(np.max(conv))

        if verbose and passes % 5 == 0:
            print(f"      coord pass={passes} obj={2.0 * n * best_max:.8f}")

    return best, internal_score(best)


def conv_after_transfer(a, conv, src, dst, amount):
    if src == dst or amount <= 0.0:
        return None
    n = len(a)
    if amount > a[src] + 1e-12:
        return None

    new_conv = conv.copy()
    two_amount = 2.0 * amount
    new_conv[src:src + n] -= two_amount * a
    new_conv[dst:dst + n] += two_amount * a

    d2 = amount * amount
    new_conv[2 * src] += d2
    new_conv[2 * dst] += d2
    new_conv[src + dst] -= 2.0 * d2

    new_conv = np.maximum(new_conv, 0.0)
    return new_conv


def score_from_conv(n, conv, s=1.0):
    if conv.size == 0:
        return float("inf")
    m = float(np.max(conv))
    if not np.isfinite(m) or s <= 0.0:
        return float("inf")
    return float(2.0 * n * m / (s * s))


def pair_polish(a0, time_limit=30.0, seed=3, fracs=None, stall_limit=600,
                verbose=False):
    a = normalize_nonneg(a0, 1.0)
    n = len(a)
    if fracs is None:
        fracs = (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9)

    conv = fast_convolve(a)
    current_obj = internal_score(a, conv)
    best = a.copy()
    best_obj = current_obj

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        src = int(rng.integers(0, n))
        dst = int(rng.integers(0, n))
        if src == dst:
            continue

        improved = False

        for s, d in ((src, dst), (dst, src)):
            base_amt = float(a[s])
            if base_amt <= 1e-15:
                continue
            for fr in fracs:
                amount = fr * base_amt
                nc = conv_after_transfer(a, conv, s, d, amount)
                if nc is None:
                    continue
                obj = score_from_conv(n, nc)
                if obj < current_obj - 1e-13:
                    a[s] -= amount
                    a[d] += amount
                    a = np.maximum(a, 0.0)
                    a = normalize_nonneg(a, 1.0)
                    conv = fast_convolve(a)
                    current_obj = internal_score(a, conv)
                    if current_obj < best_obj - 1e-14:
                        best_obj = current_obj
                        best = a.copy()
                    improved = True

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                a = project_simplex(best + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(a)
                current_obj = internal_score(a, conv)
                stall = 0

    return best, best_obj


def build_full_lp_matrix(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def build_active_lp_matrix(f, k_idx):
    n = len(f)
    if k_idx is None or len(k_idx) == 0:
        return np.zeros((0, n), dtype=np.float64)

    k_idx = np.asarray(k_idx, dtype=int).ravel()
    K = len(k_idx)
    A = np.zeros((K, n), dtype=np.float64)

    for r, kk in enumerate(k_idx):
        kk = int(kk)
        if kk < 0 or kk >= 2 * n - 1:
            continue
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        if js <= je:
            A[r, js:je + 1] = f[kk - je:kk - js + 1][::-1]
    return A


def solve_lp(f, M, k_idx=None, full=False, c=None, grad_mode=False,
             time_limit=2.0, rng=None):
    if not HAS_SCIPY:
        return None
    M = float(M)
    if M <= 0.0 or not np.isfinite(M):
        return None

    n = len(f)
    if full:
        A = build_full_lp_matrix(f)
    else:
        if k_idx is None or len(k_idx) == 0:
            return None
        A = build_active_lp_matrix(f, k_idx)

    if A.size == 0:
        return None

    b = np.full(A.shape[0], M, dtype=np.float64)

    if grad_mode:
        c_use = np.asarray(c, dtype=np.float64).ravel()
        if c_use.shape[0] != n:
            c_use = np.zeros(n, dtype=np.float64)
        A_eq = np.ones((1, n), dtype=np.float64)
        b_eq = np.ones(1, dtype=np.float64)
        bounds = (0.0, 1.0)
    else:
        c_use = -np.ones(n, dtype=np.float64)
        A_eq = None
        b_eq = None
        bounds = None if full else (0.0, 100.0)

    if rng is not None:
        c_use = c_use + 1e-10 * rng.standard_normal(n)

    res = None
    try:
        res = sp_opt.linprog(
            c_use, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
            bounds=bounds, method="highs",
            options={"time_limit": float(max(0.5, time_limit))}
        )
    except Exception:
        try:
            res = sp_opt.linprog(
                c_use, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method="interior-point",
                options={"maxiter": 150}
            )
        except Exception:
            try:
                res = sp_opt.linprog(
                    c_use, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds
                )
            except Exception:
                res = None

    if res is None:
        return None

    g = getattr(res, "x", None)
    if g is None:
        return None

    g = np.asarray(g, dtype=np.float64).ravel()
    if not np.all(np.isfinite(g)):
        return None

    g = np.maximum(g, 0.0)
    s = float(np.sum(g))
    if s <= 1e-14:
        return None
    g = g / s

    if A.shape[0] > 0:
        viol = float(np.max(A @ g - b))
        if not np.isfinite(viol) or viol > 1e-6 * max(1.0, M) + 1e-12:
            return None

    return g


def lp_refine(a0, time_limit=60.0, active_k=256, use_grad=True, use_full=False,
              max_full_n=850, full_every=4, seed=4, verbose=False):
    a = normalize_nonneg(a0, 1.0)
    n = len(a)

    conv = fast_convolve(a)
    current_obj = internal_score(a, conv)
    best = a.copy()
    best_obj = current_obj

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    alphas = (0.1, 0.2, 0.05, 0.3, 0.02, 0.5, 0.01, 0.4)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if time_limit - elapsed < 5.0:
            break

        it += 1
        conv = fast_convolve(a)
        M = float(np.max(conv)) + 1e-12

        k = min(len(conv), max(32, int(active_k)))
        if k <= 0 or k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -k)[-k:]

        peaks = np.unique(np.concatenate((peaks, np.array([0, len(conv) - 1]))))
        if peaks.size > 2 * max(32, int(active_k)):
            peaks = peaks[np.argsort(conv[peaks])[::-1][:max(32, int(active_k))]]

        g = None
        remaining_local = time_limit - (time.monotonic() - t0)

        if use_full and n <= max_full_n and (it == 1 or it % max(2, full_every) == 0):
            tl = min(8.0, max(1.0, 0.5 * remaining_local))
            g = solve_lp(a, M, full=True, time_limit=tl, rng=rng)
        elif use_grad and (it % 2 == 1):
            tl = min(2.0, max(0.5, 0.25 * remaining_local))
            sg = compute_subgradient(a, conv, max(1e-12, 0.03 * M),
                                     min(256, max(32, int(active_k))))
            g = solve_lp(a, M, k_idx=peaks, c=sg, grad_mode=True,
                         time_limit=tl, rng=rng)
        else:
            tl = min(2.0, max(0.5, 0.25 * remaining_local))
            g = solve_lp(a, M, k_idx=peaks, full=False, grad_mode=False,
                         time_limit=tl, rng=rng)

        if g is None:
            no_improve += 3
            if no_improve > 20:
                break
            a = project_simplex(best + rng.standard_normal(n) * 0.0015)
            continue

        dist = float(np.linalg.norm(g - a))
        accepted = False
        if dist >= 1e-6:
            for alpha in alphas:
                cand = project_simplex((1.0 - alpha) * a + alpha * g)
                objc = internal_score(cand)
                if objc < current_obj - 1e-14:
                    a = cand
                    current_obj = objc
                    accepted = True
                    if objc < best_obj - 1e-14:
                        best_obj = objc
                        best = a.copy()
                    no_improve = 0
                    break

        if not accepted:
            no_improve += 1

        if no_improve > 12:
            a = project_simplex(best + rng.standard_normal(n) * 0.002)
            current_obj = internal_score(a)
            no_improve = 0

        if verbose and it % 10 == 0:
            print(f"      lp it={it} obj={best_obj:.8f}")

    return best, best_obj


def refine_sequence(a0, time_limit=60.0, verbose=False, use_lp=True, seed=0):
    a = normalize_nonneg(a0, 1.0)
    n = len(a)

    conv = fast_convolve(a)
    best = a.copy()
    best_obj = internal_score(a, conv)

    t0 = time.monotonic()
    cycle = 0
    stale = 0
    rng = np.random.default_rng(seed)
    use_lp = use_lp and HAS_SCIPY

    if verbose:
        print(f"    refine n={n} start obj={best_obj:.8f}")

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        rem = time_limit - elapsed
        if rem < 10.0:
            break

        phase = cycle % 6
        a_seed = int(seed + cycle * 101 + phase * 7)

        if use_lp:
            if phase == 0:
                t = min(40.0, 0.35 * rem)
                a_opt, obj = lp_refine(
                    a, time_limit=t,
                    active_k=min(700, max(128, n)),
                    use_grad=True, use_full=True, full_every=5,
                    seed=a_seed
                )
            elif phase == 1:
                t = min(40.0, 0.35 * rem)
                a_opt, obj = subgrad_refine(
                    a, time_limit=t,
                    top_k=min(256, max(32, n // 4)),
                    seed=a_seed
                )
            elif phase == 2:
                t = min(50.0, 0.40 * rem)
                samples = 36 if n <= 1000 else 18
                a_opt, obj = coordinate_polish(a, time_limit=t, samples=samples,
                                               seed=a_seed)
            elif phase == 3:
                t = min(30.0, 0.30 * rem)
                a_opt, obj = lp_refine(
                    a, time_limit=t,
                    active_k=min(350, max(64, n // 2)),
                    use_grad=False, use_full=False,
                    seed=a_seed
                )
            elif phase == 4:
                t = min(30.0, 0.30 * rem)
                a_opt, obj = pair_polish(a, time_limit=t, seed=a_seed,
                                         stall_limit=300)
            else:
                t = min(30.0, 0.30 * rem)
                noisy = project_simplex(a + rng.standard_normal(n) * 0.002)
                a_opt, obj = subgrad_refine(noisy, time_limit=t, seed=a_seed + 123)
        else:
            if phase == 0:
                t = min(40.0, 0.35 * rem)
                a_opt, obj = subgrad_refine(a, time_limit=t, seed=a_seed)
            elif phase == 1:
                t = min(50.0, 0.40 * rem)
                samples = 36 if n <= 1000 else 18
                a_opt, obj = coordinate_polish(a, time_limit=t, samples=samples,
                                               seed=a_seed)
            elif phase == 2:
                t = min(30.0, 0.30 * rem)
                a_opt, obj = pair_polish(a, time_limit=t, seed=a_seed,
                                         stall_limit=300)
            else:
                t = min(30.0, 0.30 * rem)
                noisy = project_simplex(a + rng.standard_normal(n) * 0.002)
                a_opt, obj = subgrad_refine(noisy, time_limit=t, seed=a_seed + 123)

        if obj < best_obj - 1e-14:
            a = a_opt.copy()
            best_obj = obj
            best = a.copy()
            stale = 0
            if verbose:
                print(f"      refine cycle={cycle} obj={best_obj:.8f}")
        else:
            stale += 1
            if stale % 3 == 0:
                a = project_simplex(best + rng.standard_normal(n) * 0.002)
            else:
                a = best.copy()

        cycle += 1

    rem = max(0.0, time_limit - (time.monotonic() - t0))
    if rem > 5.0:
        a_final, obj_final = coordinate_polish(
            best, time_limit=min(rem, 20.0), samples=24, seed=int(seed + 999)
        )
        if obj_final < best_obj:
            best_obj = obj_final
            best = a_final

    return best, best_obj


def make_variants(base, rng, max_variants=12):
    out = []
    n = len(base)
    base = normalize_nonneg(base, 1.0)

    out.append(("base", base.copy()))
    out.append(("rev", normalize_nonneg(base[::-1])))
    out.append(("sym", symmetrize(base)))

    for p in (0.7, 1.5):
        arr = base ** p
        out.append((f"pow{p}", normalize_nonneg(arr)))

    if n >= 8:
        for shift in (1, n // 4, n // 2, 3 * n // 4):
            if shift != 0:
                out.append((f"roll{shift}", normalize_nonneg(np.roll(base, shift))))

    for scale in (0.01, 0.03):
        out.append((f"noise{scale}",
                    project_simplex(base + rng.standard_normal(n) * scale)))

    x = np.linspace(0.0, 1.0, n)
    for bandwidth in (4.0, 12.0):
        g = np.exp(-bandwidth * (x - 0.5) ** 2)
        out.append((f"gauss{int(bandwidth)}",
                    normalize_nonneg(0.6 * base + 0.4 * g)))

    e = np.abs(x - 0.5) ** 2
    out.append(("edge", normalize_nonneg(0.7 * base + 0.3 * e)))

    return out


def consider_actual(best_a, best_score, cand, label):
    cand = normalize_nonneg(cand, 1.0).copy()
    score = actual_score(cand)
    if np.isfinite(score) and score < best_score:
        print(f"NEW ACTUAL BEST [{label}]: {score:.8f} (n={len(cand)})")
        return cand, score
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(seed)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
        p = np.maximum(p, 0.0)
        if p.size > 0 and p.sum() > 1e-10:
            p = p / p.sum()
            parent = p
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768, dtype=np.float64) / 768.0

    print(f"Starting parent n={len(parent)}")

    best_a = parent.copy()
    best_i = internal_score(best_a)
    best_actual_a = parent.copy()
    best_actual = actual_score(parent)
    print(f"Initial internal={best_i:.8f} actual={best_actual:.8f}")

    # Main optimization on parent length.
    rem = deadline - time.monotonic()
    if rem > 60.0:
        t1 = min(700.0, max(60.0, rem - 500.0))
        if t1 > 30.0:
            a_opt, obj_opt = refine_sequence(
                best_a, time_limit=t1, verbose=True, use_lp=True, seed=101
            )
            if obj_opt < best_i:
                best_i = obj_opt
                best_a = a_opt.copy()
            best_actual_a, best_actual = consider_actual(
                best_actual_a, best_actual, a_opt, "phase1"
            )

    # Short same-length restarts / variants.
    rem = deadline - time.monotonic()
    if rem > 250.0:
        variants = make_variants(best_a, rng, max_variants=14)
        vi = 0
        for label, cand in variants:
            if label == "base":
                continue

            rem = deadline - time.monotonic()
            if rem < 150.0:
                break

            cand = normalize_nonneg(cand, 1.0)
            if len(cand) != len(best_a):
                cand = interpolate_to_length(cand, len(best_a))

            t = max(30.0, min(80.0, 0.25 * rem))
            a_opt, obj_opt = refine_sequence(
                cand, time_limit=t, verbose=False, use_lp=True,
                seed=200 + vi * 17
            )
            if obj_opt < best_i:
                best_i = obj_opt
                best_a = a_opt.copy()
            best_actual_a, best_actual = consider_actual(
                best_actual_a, best_actual, a_opt, f"variant_{label}"
            )

            vi += 1
            if vi >= 5:
                break

    # Try longer lengths.
    rem = deadline - time.monotonic()
    for target_n in (1024, 1536):
        rem = deadline - time.monotonic()
        if rem < 180.0:
            break

        f = interpolate_to_length(best_a, target_n)

        t_quick = max(40.0, min(100.0, 0.30 * rem))
        a_quick, obj_quick = refine_sequence(
            f, time_limit=t_quick, verbose=False, use_lp=True,
            seed=300 + (target_n % 100)
        )

        a_candidate = a_quick
        obj_candidate = obj_quick

        improved_quick = obj_candidate < best_i - 1e-7
        if improved_quick or (target_n <= 1024 and rem > 300.0):
            rem = deadline - time.monotonic()
            if rem > 100.0:
                t_deep = max(60.0, min(350.0, 0.55 * rem))
                a_deep, obj_deep = refine_sequence(
                    a_quick, time_limit=t_deep, verbose=True,
                    use_lp=(target_n <= 900),
                    seed=400 + (target_n % 100)
                )
                if obj_deep < obj_candidate:
                    a_candidate = a_deep
                    obj_candidate = obj_deep

        if obj_candidate < best_i:
            best_i = obj_candidate
            best_a = a_candidate.copy()

        best_actual_a, best_actual = consider_actual(
            best_actual_a, best_actual, a_candidate, f"length_{target_n}"
        )

    # Final polish.
    rem = deadline - time.monotonic()
    if rem > 40.0:
        t = max(20.0, min(150.0, rem - 20.0))

        a_opt, obj_opt = coordinate_polish(
            best_a, time_limit=0.5 * t, samples=40, seed=555
        )
        if obj_opt < best_i:
            best_i = obj_opt
            best_a = a_opt.copy()

        rem = deadline - time.monotonic()
        if rem > 30.0:
            t2 = max(10.0, min(80.0, rem - 20.0))
            a_opt, obj_opt = pair_polish(
                best_a, time_limit=t2, seed=666, stall_limit=500
            )
            if obj_opt < best_i:
                best_i = obj_opt
                best_a = a_opt.copy()

        rem = deadline - time.monotonic()
        if rem > 60.0 and HAS_SCIPY and len(best_a) <= 900:
            a_opt, obj_opt = lp_refine(
                best_a,
                time_limit=min(rem - 30.0, 120.0),
                active_k=512,
                use_grad=True,
                use_full=True,
                seed=777
            )
            if obj_opt < best_i:
                best_i = obj_opt
                best_a = a_opt.copy()

    # Final exact score selection.
    final_candidates = [best_a, best_actual_a, parent]
    final_best = None
    final_score = float("inf")

    for cand in final_candidates:
        cand = normalize_nonneg(cand, 1.0).copy()
        sc = actual_score(cand)
        if np.isfinite(sc) and sc < final_score:
            final_score = sc
            final_best = cand

    if final_best is None:
        final_best = normalize_nonneg(best_a, 1.0)

    print(f"FINAL SCORE {final_score:.8f} (n={len(final_best)})")
    return [float(x) for x in final_best]
