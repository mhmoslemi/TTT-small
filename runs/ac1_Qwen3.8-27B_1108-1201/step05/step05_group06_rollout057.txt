```python
from scipy.optimize import linprog

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    # Constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j] <= M
    # For each k in [0, 2n-2], the constraint is:
    # sum_{j: 0<=k-j<n} f[j] * g[k-j] <= M
    # In terms of g variables: sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} f[k-j] * g[j] <= M
    
    # Build constraint matrix A (shape: (2n-1, n))
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    # Maximize sum(g) = -minimize -sum(g)
    c = -np.ones(n)
    # Constraints: A @ g <= M (vector of M's)
    b = np.full(2*n-1, M)
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x
    else:
        # Fallback: return f normalized
        return f / f.sum()
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding with
line search, combined with coordinate descent polishing and multi-restart search.

The key idea: at each step, solve an LP to find the best direction g that
maximizes the sum (i.e., "mass") while keeping the cross-convolution f*g 
bounded by the current peak M. Then take a line search step in that direction.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(f, n):
    """Build matrix A where (A @ g)[k] = (f * g)[k] for all k."""
    # A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For variable g[j], it contributes f[k-j] to constraint k
        # k-j must be in [0, n-1], so k in [j, j+n-1]
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        f_slice = f[k_start - j : k_end - j]  # f[0 : k_end-j]
        A[k_start:k_end, j] = f_slice
    return A


def solve_direction_lp(f, M, method='highs'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method=method)
    if result.success:
        return result.x
    else:
        return f.copy()


def peak_at_t(conv_f, cross_fg, conv_g, t):
    """Compute max of (1-t)^2 * conv_f + 2*t*(1-t) * cross_fg + t^2 * conv_g."""
    vals = (1 - t) ** 2 * conv_f + 2 * t * (1 - t) * cross_fg + t * t * conv_g
    return np.max(vals)


def find_max_t(conv_f, cross_fg, conv_g, M, max_iter=50):
    """Find largest t in [0,1] such that peak_at_t(t) <= M."""
    # peak(0) = max(conv_f) = M (by definition)
    # Check t=1
    if peak_at_t(conv_f, cross_fg, conv_g, 1.0) <= M + 1e-12:
        return 1.0
    
    lo, hi = 0.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        if peak_at_t(conv_f, cross_fg, conv_g, mid) <= M + 1e-12:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-12:
            break
    return lo


def lp_optimize(f0, time_limit, verbose=False):
    """LP-based optimization with line search."""
    a = f0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv_f = fast_convolve(a)
    M = conv_f.max()
    best_a = a.copy()
    best_M = M
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Solve LP
        g = solve_direction_lp(a, M)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum  # normalize to sum=1
        
        # Compute cross-convolution and g*g
        cross_fg = cross_convolve(a, g)
        conv_g = fast_convolve(g)
        
        # Find max step
        t_max = find_max_t(conv_f, cross_fg, conv_g, M)
        
        if t_max < 1e-8:
            # Can't make progress; try perturbing
            if verbose:
                print(f"    Iter {it}: t_max={t_max:.2e}, stuck")
            break
        
        # Take the step
        a = (1 - t_max) * a + t_max * g
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a /= s
        
        conv_f = fast_convolve(a)
        new_M = conv_f.max()
        
        # Due to normalization, M might change slightly
        # Renormalize M
        # Actually, after normalization, the peak changes. Let's just use the new peak.
        M = new_M
        
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            if verbose and it % 10 == 0:
                score = 2 * n * M  # since sum=1
                print(f"    Iter {it}: M={M:.10f}, score={score:.10f}, t={t_max:.4f}")
    
    return best_a, best_M
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding with
line search (bisection), combined with coordinate descent polishing, multi-restart
search over different lengths, and adaptive perturbation strategies.

Key idea: At each iteration, solve an LP to find direction g maximizing sum(g)
subject to (f*g)[k] <= M (current peak) and g >= 0. Then take a bisection line
search step. This gives a guaranteed non-increasing peak trajectory.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(f, n):
    """Build matrix A where (A @ g)[k] = (f * g)[k] for all k in [0, 2n-2]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        A[k_start:k_end, j] = f[:length]
    return A


def solve_direction_lp(f, M, tol=1e-10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, success)."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tol': tol,
                                  'primal_feasibility_tol': tol})
        if result.success:
            return result.x, True
        else:
            return f.copy(), False
    except Exception:
        return f.copy(), False


def peak_at_t(conv_f, cross_fg, conv_g, t):
    """Compute max of (1-t)^2 * conv_f + 2*t*(1-t) * cross_fg + t^2 * conv_g."""
    vals = (1.0 - t) ** 2 * conv_f + 2.0 * t * (1.0 - t) * cross_fg + t * t * conv_g
    return np.max(vals)


def find_max_t(conv_f, cross_fg, conv_g, M, max_iter=60):
    """Find largest t in [0,1] such that peak_at_t(t) <= M."""
    if peak_at_t(conv_f, cross_fg, conv_g, 1.0) <= M * (1.0 + 1e-10) + 1e-15:
        return 1.0
    lo = 0.0
    hi = 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        if peak_at_t(conv_f, cross_fg, conv_g, mid) <= M * (1.0 + 1e-10) + 1e-15:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-13:
            break
    return lo


def lp_optimize(f0, time_limit, verbose=False, relaxation=1.0):
    """LP-based optimization with bisection line search.
    f0 is normalized to sum=1."""
    a = f0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv_f = fast_convolve(a)
    M = conv_f.max()
    best_a = a.copy()
    best_M = M

    t0 = time.monotonic()
    it = 0
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Solve LP with slight relaxation to escape degeneracy
        M_relaxed = M * relaxation
        g, success = solve_direction_lp(a, M_relaxed)
        if not success:
            g = a.copy()

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Skip if direction is essentially the same as current
        cos_sim = np.dot(a, g)
        if cos_sim > 0.9999:
            stuck_count += 1
            if stuck_count > 20:
                break
            continue

        stuck_count = 0

        # Compute cross-convolution and g*g
        cross_fg = cross_convolve(a, g)
        conv_g = fast_convolve(g)

        # Find max step
        t_max = find_max_t(conv_f, cross_fg, conv_g, M)

        if t_max < 1e-10:
            stuck_count += 1
            if stuck_count > 30:
                break
            # Try with larger relaxation
            relaxation = min(relaxation * 1.05, 1.5)
            continue

        relaxation = max(1.0, relaxation * 0.98)

        # Take the step
        a = (1.0 - t_max) * a + t_max * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a /= s

        conv_f = fast_convolve(a)
        new_M = conv_f.max()
        M = new_M

        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            if verbose and it % 20 == 0:
                score = 2.0 * n * M
                print(f"    LP iter {it}: M={M:.10f}, score={score:.10f}, t={t_max:.5f}")

    return best_a, best_M


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent: for each coordinate, find optimal value via ternary search."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-14:
                continue

            x_old = a[j]
            others_sum = 1.0 - x_old
            if others_sum < 1e-12:
                continue

            # b = a with coordinate j zeroed, normalized
            b = a.copy()
            b[j] = 0.0
            b = b / others_sum
            B = fast_convolve(b)
            conv_len = 2 * n - 1

            # b_valid[k] = b[k-j] if 0 <= k-j < n, else 0
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            # Ternary search on x in [0, 1]
            lo = 0.0
            hi = 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                om1 = 1.0 - m1
                f1 = om1 * om1 * B + 2.0 * m1 * om1 * b_valid
                if 2 * j < conv_len:
                    f1[2 * j] += m1 * m1
                peak1 = np.max(f1)

                om2 = 1.0 - m2
                f2 = om2 * om2 * B + 2.0 * m2 * om2 * b_valid
                if 2 * j < conv_len:
                    f2[2 * j] += m2 * m2
                peak2 = np.max(f2)

                if peak1 < peak2:
                    hi = m2
                else:
                    lo = m1

            x_new = (lo + hi) / 2.0

            if abs(x_new - x_old) < 1e-12:
                continue

            # Apply
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / others_sum
            a = np.maximum(a, 0.0)
            sc = a.sum()
            if sc > 1e-15:
                a /= sc
            else:
                a[j] = x_old
                continue

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_obj


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        if s_try < 0.01:
            return
        score = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_try)})")

    # === Phase 1: LP optimize parent ===
    print(f"\n=== Phase 1: LP optimize parent (n={len(parent)}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_alloc = min(rem, 400)
        print(f"  t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(parent, time_limit=t_alloc, verbose=True)
        update_best(a_opt, "parent_lp")

    # === Phase 2: Coordinate descent polish ===
    if best_a is not None:
        rem = deadline - time.monotonic() - 700
        if rem > 20:
            t_alloc = min(rem, 150)
            print(f"\n=== Phase 2: CD polish (t={t_alloc:.0f}s) ===")
            a_opt, _ = coordinate_descent(best_a, time_limit=t_alloc)
            update_best(a_opt, "cd_polish")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    target_lengths = []
    n_base = len(base)
    for n_try in [n_base - 128, n_base + 64, n_base + 128, n_base - 64, n_base + 256]:
        if 256 <= n_try <= 3000 and n_try != n_base:
            target_lengths.append(n_try)

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic() - 500
        if rem < 40:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.4, 180)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(f_interp, time_limit=t_alloc, verbose=True)
        update_best(a_opt, f"n{target_n}")

    # === Phase 4: Deep optimization with restarts ===
    print(f"\n=== Phase 4: Deep restarts (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        restart_configs = [
            ("noise3", 0.003),
            ("noise1", 0.001),
            ("noise5", 0.005),
            ("rev_mix", None),
            ("sym", None),
        ]
        for idx, (label, nscale) in enumerate(restart_configs):
            rem = deadline - time.monotonic() - 300 + idx * 40
            if rem < 25:
                break
            t_alloc = min(rem, 120)
            if nscale is not None:
                rng_r = np.random.default_rng(seed + 500 + idx * 17)
                f_init = project_simplex(best_a + rng_r.standard_normal(n_best) * nscale)
            elif label == "rev_mix":
                f_init = project_simplex(0.7 * best_a + 0.3 * best_a[::-1])
            else:  # sym
                f_init = project_simplex(0.5 * (best_a + best_a[::-1]))

            print(f"  Restart {label}, t={t_alloc:.0f}s")
            a_opt, _ = combined_optimize(f_init, time_limit=t_alloc, verbose=(idx == 0))
            update_best(a_opt, f"deep_{label}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem, 200)
            print(f"  CD polish, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_a, time_limit=t_alloc)
            update_best(a_opt, "final_cd")

        rem = deadline - time.monotonic() - 15
        if rem > 10 and best_a is not None:
            # Tiny perturbations + LP
            n_best = len(best_a)
            for p in range(4):
                rem = deadline - time.monotonic() - 10
                if rem < 8:
                    break
                rng_p = np.random.default_rng(seed + 2000 + p * 7)
                f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.0005)
                t_alloc = min(rem, 20)
                a_opt, _ = lp_optimize(f_pert, time_limit=t_alloc)
                update_best(a_opt, f"tiny_lp_{p}")

        # Final CD on best
        rem = deadline - time.monotonic() - 5
        if rem > 5 and best_a is not None:
            t_alloc = min(rem, 30)
            a_opt, _ = coordinate_descent(best_a, time_limit=t_alloc)
            update_best(a_opt, "final_cd2")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
```
