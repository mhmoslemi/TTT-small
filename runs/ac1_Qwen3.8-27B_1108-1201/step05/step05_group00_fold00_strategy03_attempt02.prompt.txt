    """

    Returns np.inf if the input is invalid.

        if np.isnan(x) or np.isinf(x):

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]. p has length 2n-1, a has length n."""
    return np.correlate(p, a, mode='valid')


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def smoothed_grad_descent(a0, time_limit, T_init=0.0005, lr_init=0.05, verbose=False):
    """
    Smoothed gradient descent using softmax-weighted peaks.
    The gradient uses p = softmax(conv/T) which gives a smooth surrogate
    for the max, enabling more stable convergence than pure subgradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        conv = fast_convolve(a)

        # Softmax weights with adaptive temperature
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: 2 * correlate(p, a)
        grad = 2.0 * hankel_mv(p, a)

        # Projected gradient with backtracking
        best_candidate = a
        best_val = conv.max()

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt hyperparameters
        if no_improve > 20:
            T *= 1.05
            lr *= 0.97
        if no_improve > 40:
            T *= 0.8  # Slightly cool down to refocus
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def qp_subproblem(p, n, max_iter=200):
    """
    Solve min a^T M(p) a s.t. a >= 0, sum(a) = 1 using active set method.
    M(p)_{j,m} = p[j+m].
    """
    active = list(range(n))
    inactive = set()

    for _ in range(max_iter):
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[active[0]] = 1.0
            return a

        S = active
        k = len(S)

        # Build submatrix
        M_SS = np.array([[p[S[i] + S[j]] for j in range(k)] for i in range(k)])

        # Solve M_SS x = 1
        try:
            ones_vec = np.ones(k)
            x = np.linalg.solve(M_SS, ones_vec)
        except np.linalg.LinAlgError:
            x, _, _, _ = np.linalg.lstsq(M_SS, np.ones(k), rcond=None)

        x_sum = x.sum()
        if x_sum <= 1e-15:
            x = np.ones(k) / k
            x_sum = 1.0
        a_S = x / x_sum

        # Check for negative entries -> remove from active set
        removed = False
        new_active = []
        for i, idx in enumerate(S):
            if a_S[i] < -1e-12:
                inactive.add(idx)
                removed = True
            else:
                new_active.append(idx)

        if removed:
            active = new_active
            continue

        # Compute full solution
        a = np.zeros(n)
        for i, idx in enumerate(S):
            a[idx] = max(0.0, a_S[i])

        # Compute lambda: from 2Ma + lambda = 0 on active set
        Ma = hankel_mv(p, a)
        # Use average over active set for robustness
        lambda_val = -2.0 * np.mean([Ma[idx] for idx in S])

        # Check KKT for inactive: 2*Ma[j] + lambda >= 0
        added = False
        for j in sorted(inactive):
            if 2.0 * Ma[j] + lambda_val < -1e-10:
                inactive.remove(j)
                active.append(j)
                added = True

        if not added:
            return a

    # Fallback: uniform over active
    a = np.zeros(n)
    a[active] = 1.0 / len(active)
    return a


def game_iterate(a0, time_limit, T_init=0.001, verbose=False):
    """
    Game-theoretic primal-dual iteration:
    - Dual: p = softmax(conv(a) / T)
    - Primal: solve min a'^T M(p) a' s.t. a' in simplex (QP)
    This is equivalent to smoothed gradient but with a more accurate inner solve.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Solve QP subproblem
        a_qp = qp_subproblem(p, n)

        # Blend: use full QP solution (or partial if time-constrained)
        a = a_qp

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            T = T_init
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    game_iterate it={it}, obj={best_obj:.8f}, T={T:.6f}")

    return best_a, best_obj


def multi_peak_sg(a0, time_limit, top_k=20, lr_init=0.003, mom=0.93):
    """Multi-peak subgradient with momentum and adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        # Top-k peaks with inverse-rank weighting
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 2.0
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        # Momentum update
        v = mom * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        mom = min(0.98, mom + 0.0001)

        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 200) / 400.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.6
            no_improve = 0

    return best_a, best_obj


def coordinate_peak_polish(a0, time_limit):
    """
    For each of the top peaks, try to locally reduce it by adjusting
    a small neighborhood of variables.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        # Find the single highest peak
        k_peak = int(np.argmax(conv))

        # Determine which variables affect this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = list(range(js, je + 1))

        if len(affected) == 0:
            break

        # Try small adjustments to the affected variables
        # For each pair of affected vars, try shifting weight
        improved = False
        for _ in range(5):  # Try a few random perturbations
            rng = np.random.default_rng((it * 12345 + k_peak) % (2**31))
            perturbation = rng.standard_normal(n) * 0.0005
            # Zero out perturbation for non-affected vars
            for j in range(n):
                if j not in affected:
                    perturbation[j] = 0
            candidate = project_simplex(a + perturbation)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if not improved and it % 50 == 0:
            # Try a broader perturbation
            rng = np.random.default_rng((it * 67890 + 7) % (2**31))
            candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
            else:
                # No progress in a while, break
                if it > 200:
                    break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x / (n - 1) - 0.5)
    tri = np.maximum(tri, 0.1)
    inits.append(tri / tri.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits
