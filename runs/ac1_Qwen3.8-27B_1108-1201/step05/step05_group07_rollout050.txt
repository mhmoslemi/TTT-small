import numpy as np
import time
from scipy.optimize import minimize_scalar


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient focusing on top-k peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def mirror_descent_optimize(f0, time_limit, top_k=50, temperature=0.00005, lr=0.003,
                           momentum=0.85, use_symmetry=False, verbose=False):
    """
    Mirror descent (multiplicative weights) with Nesterov momentum and adaptive scheduling.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    # Optionally enforce symmetry
    if use_symmetry:
        f = 0.5 * (f + f[::-1])
        f /= f.sum()

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        if use_symmetry:
            f_nest = 0.5 * (f_nest + f_nest[::-1])
            f_nest /= f_nest.sum()

        conv_nest = fast_convolve(f_nest)
        sg = compute_subgradient_topk(f_nest, conv_nest, cur_top_k, cur_temp)

        if use_symmetry:
            sg = 0.5 * (sg + sg[::-1])

        # Update momentum
        v = momentum * v + (1 - momentum) * sg

        # Multiplicative weights update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)

        if use_symmetry:
            f = 0.5 * (f + f[::-1])

        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_lr *= 0.9997
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp *= 1.005
        if it % 1000 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.9))

        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.001 + 0.0005 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            if use_symmetry:
                f = 0.5 * (f + f[::-1])
                f /= f.sum()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_temp = temperature * 0.5
            no_improve = 0

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def peak_directed_cd(f0, time_limit, verbose=False):
    """
    Peak-directed coordinate descent: for the dominant peak, find the optimal
    redistribution among contributing coordinates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        peak_lag = int(np.argmax(conv))
        current_max = conv[peak_lag]

        # Find contributing indices for the peak lag
        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        idx = np.arange(js, je + 1)
        
        if len(idx) < 2:
            break

        # Try to equalize the contributions a[i]*a[peak_lag-i] for i in idx
        # The optimal for minimizing the peak (with sum constraint on these coords)
        # would be to make all a[i]*a[peak_lag-i] equal
        # This means a[i] = a[peak_lag-i] for all i (symmetric around peak_lag/2)
        
        # Simple approach: try adjusting the most imbalanced pair
        contribs = f[idx] * f[peak_lag - idx]
        max_ci = np.argmax(contribs)
        min_ci = np.argmin(contribs)
        
        hi_idx = idx[max_ci]
        lo_idx = idx[min_ci]
        
        if hi_idx == lo_idx:
            break
        
        # Try transferring mass
        improved = False
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
            amt = f[hi_idx] * frac
            if f[hi_idx] - amt < 0:
                continue
            f[hi_idx] -= amt
            f[lo_idx] += amt
            f /= f.sum()
            
            new_conv = fast_convolve(f)
            new_max = new_conv.max()
            
            if new_max < best_obj - 1e-14:
                best_obj = new_max
                best_f = f.copy()
                improved = True
            else:
                f[hi_idx] += amt
                f[lo_idx] -= amt
                f /= f.sum()
                break
        
        if not improved:
            # Try random coordinate descent
            rng = np.random.default_rng(int(time.monotonic() * 1e9) % 2**32)
            j = int(rng.integers(0, n))
            old_val = f[j]
            if old_val < 1e-10:
                break
            
            best_new_val = old_val
            best_new_obj = current_max
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                
                val = fast_convolve(f_trial).max()
                if val < best_new_obj - 1e-14:
                    best_new_obj = val
                    best_new_val = new_val
            
            if best_new_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_new_val) / denom
                    f[j] = best_new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                    
                    if best_new_obj < best_obj - 1e-13:
                        best_obj = best_new_obj
                        best_f = f.copy()
            else:
                break

    return best_f, best_obj


def pair_exchange_search(f0, time_limit, max_trials_per_iter=200, verbose=False):
    """Pair-exchange local search with adaptive trial counts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        top_peaks = np.argpartition(conv, -min(20, 2 * n - 1))[-min(20, 2 * n - 1):]
        
        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1e9) % 2**32)
        
        for _ in range(max_trials_per_iter):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Choose a peak and try to reduce it
            k = int(top_peaks[int(rng.integers(0, len(top_peaks)))])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js >= je:
                continue
            
            i = int(rng.integers(js, je + 1))
            j = int(rng.integers(js, je + 1))
            if i == j:
                continue
            
            old_i = f[i]
            old_j = f[j]
            if old_i < 1e-10:
                continue
            
            for frac in [0.1, 0.2, 0.4]:
                amt = old_i * frac
                if old_i - amt < -1e-15:
                    continue
                f[i] = max(0, old_i - amt)
                f[j] = old_j + amt
                f /= f.sum()
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] = old_j
                    break
        
        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    # Gaussian
    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 5.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Bump with plateau
    w = n // (8 + int(rng.random() * 8))
    bump = np.zeros(n)
    center_idx = n // 2
    bump[center_idx - w:center_idx + w] = 1.0
    # Smooth edges
    edge = w // 3
    for d in range(edge):
        bump[center_idx - w + d] = d / edge
        bump[center_idx + w - 1 - d] = d / edge
    bump = np.maximum(bump, 1e-10)
    inits.append(bump / bump.sum())

    # Cosine taper (like a Hamming window)
    hamming = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    hamming = np.maximum(hamming, 1e-10)
    inits.append(hamming / hamming.sum())

    # Two-peak
    if n > 4:
        two_peak = np.zeros(n)
        q1, q2 = n // 3, 2 * n // 3
        s1 = n // 20
        s2 = n // 20
        two_peak = np.exp(-0.5 * ((np.arange(n) - q1) / s1) ** 2) + np.exp(-0.5 * ((np.arange(n) - q2) / s2) ** 2)
        two_peak = np.maximum(two_peak, 1e-10)
        inits.append(two_peak / two_peak.sum())

    return inits[:count]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent (multiplicative
weights) with Nesterov momentum, peak-directed coordinate descent, pair-exchange local
search, and multi-length exploration. Key improvements: longer deep-optimization phases,
adaptive peak weighting with dual temperature, symmetry exploitation, and peak-pair

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient focusing on top-k peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def compute_subgradient_exact(a, conv, top_k, temperature):
    """Subgradient using only the top-k peaks with uniform weighting (no softmax)."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    # Use a softer weighting: proportional to (peak - median)^2
    med = np.median(pv)
    w = np.maximum(pv - med, 0) ** 2
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg

def peak_directed_cd(f0, time_limit, verbose=False):
    """
    Peak-directed coordinate descent: targets the dominant peak by redistributing
    mass among its contributing coordinates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        peak_lag = int(np.argmax(conv))
        current_max = conv[peak_lag]

        # Find contributing indices for the peak lag
        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        idx = np.arange(js, je + 1)

        if len(idx) < 2:
            break

        improved = False

        # Strategy 1: Equalize contributions a[i]*a[peak_lag-i]
        contribs = f[idx] * f[peak_lag - idx]
        max_ci = int(np.argmax(contribs))
        min_ci = int(np.argmin(contribs))
        hi_idx = int(idx[max_ci])
        lo_idx = int(idx[min_ci])

        if hi_idx != lo_idx and f[hi_idx] > 1e-10:
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                amt = f[hi_idx] * frac
                f[hi_idx] -= amt
                f[lo_idx] += amt
                f /= f.sum()
                new_max = fast_convolve(f).max()
                if new_max < best_obj - 1e-14:
                    best_obj = new_max
                    best_f = f.copy()
                    improved = True
                else:
                    f[hi_idx] += amt
                    f[lo_idx] -= amt
                    f /= f.sum()
                    break

        # Strategy 2: Coordinate descent on random coordinate
        if not improved:
            rng_c = np.random.default_rng(int(time.monotonic() * 1e9) % (2**32))
            j = int(rng_c.integers(0, n))
            old_val = f[j]
            if old_val < 1e-10:
                stall += 1
                if stall > 200:
                    break
                continue

            best_new_val = old_val
            best_new_obj = current_max

            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_new_obj - 1e-14:
                    best_new_obj = val
                    best_new_val = new_val

            if best_new_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_new_val) / denom
                    f[j] = best_new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                    if best_new_obj < best_obj - 1e-13:
                        best_obj = best_new_obj
                        best_f = f.copy()
                        improved = True
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

        if not improved and stall > 300:
            break

    return best_f, best_obj

                break

            k = int(top_peaks[int(rng_p.integers(0, len(top_peaks)))])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js >= je:
                continue

            i = int(rng_p.integers(js, je + 1))
            j = int(rng_p.integers(js, je + 1))
            if i == j:
                continue

            old_i = f[i]
            old_j = f[j]
            if old_i < 1e-10:
                continue

            for frac in [0.08, 0.15, 0.25, 0.4]:
                amt = old_i * frac
                if old_i - amt < -1e-15:
                    continue
                f[i] = max(0.0, old_i - amt)
                f[j] = old_j + amt
                f /= f.sum()
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] = old_j
                    break

        if not improved:
            break

    return best_f, best_obj


def peak_pair_search(f0, time_limit, verbose=False):
    """
    Targets the interaction between the top two peaks simultaneously.
    Tries moves that reduce both peaks at once.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        # Get top 2 peaks
        top2 = np.argsort(conv)[-2:][::-1]
        k1 = int(top2[0])
        k2 = int(top2[1])

        # For peak 1, find the worst contributor
        js1 = max(0, k1 - n + 1)
        je1 = min(n - 1, k1)
        idx1 = np.arange(js1, je1 + 1)
        contribs1 = f[idx1] * f[k1 - idx1]
        worst1 = int(idx1[np.argmax(contribs1)])

        # Find an index NOT contributing to either peak
        all_indices = set(range(n))
        contributes_to_k1 = set(range(js1, je1 + 1)) | set(k1 - np.arange(js1, je1 + 1))
        js2 = max(0, k2 - n + 1)
        je2 = min(n - 1, k2)
        contributes_to_k2 = set(range(js2, je2 + 1)) | set(k2 - np.arange(js2, je2 + 1))
        safe = all_indices - contributes_to_k1 - contributes_to_k2

        if not safe:
            break

        safe_list = list(safe)
        best_target = safe_list[0]
        best_target_val = 0

        # Find safe index with smallest value (to minimize disruption)
        for si in safe_list[:50]:
            if f[si] < best_target_val or best_target_val == 0:
                best_target = si
                best_target_val = f[si]

        # Try transferring from worst1 to best_target
        improved = False
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
            amt = f[worst1] * frac
            if f[worst1] - amt < 0:
                continue
            f[worst1] -= amt
            f[best_target] += amt
            f /= f.sum()
            new_max = fast_convolve(f).max()
            if new_max < best_obj - 1e-14:
                best_obj = new_max
                best_f = f.copy()
                improved = True
            else:
                f[worst1] += amt
                f[best_target] -= amt
                f /= f.sum()
                break

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Light smoothing
    if new_n > 10:
        kernel = np.ones(3) / 3.0
        f_interp = np.convolve(f_interp, kernel, mode='same')
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    # Hamming window
    hamming = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    hamming = np.maximum(hamming, 1e-10)
    inits.append(hamming / hamming.sum())

    # Gaussian
    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 5.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Bump with smooth edges
    w = n // (6 + int(rng.random() * 10))
    bump = np.zeros(n)
    center_idx = n // 2
    bump[center_idx - w:center_idx + w] = 1.0
    edge = max(1, w // 4)
    for d in range(edge):
        frac = d / edge
        bump[center_idx - w + d] = 0.5 * (1 - np.cos(np.pi * frac))
        bump[center_idx + w - 1 - d] = 0.5 * (1 - np.cos(np.pi * frac))
    bump = np.maximum(bump, 1e-10)
    inits.append(bump / bump.sum())

    # Two-peak
    if n > 8:
        t1, t2 = n // 3.0, 2.0 * n // 3.0
        s1 = n / 25.0
        s2 = n / 25.0
        two_peak = np.exp(-0.5 * ((np.arange(n) - t1) / s1) ** 2) + np.exp(-0.5 * ((np.arange(n) - t2) / s2) ** 2)
        two_peak = np.maximum(two_peak, 1e-10)
        inits.append(two_peak / two_peak.sum())

    return inits[:count]


def try_evaluate(a, best_state, label=""):
    """Evaluate sequence and update best state (list: [best_seq, best_score])."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state[1]:
        best_state[1] = score
        best_state[0] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses adaptive mirror descent with multiple peak-weighting strategies,
    peak-directed local search, pair-pair interaction search, and multi-length
    exploration to find high-quality non-negative sequences.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_state = [None, float('inf')]

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state[0] = parent.copy()
            best_state[1] = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent, using uniform n=768")
        best_state[0] = parent.copy()
        best_state[1] = float(evaluate_sequence(parent.tolist()))

    n_parent = len(parent)

    # === Phase 1: Broad exploration with multiple inits ===
    print(f"\n=== Phase 1: Exploration (parent n={n_parent}, score={best_state[1]:.6f}) ===")

    candidates = [parent]

    # Symmetrized parent
    sym_p = 0.5 * (parent + parent[::-1])
    sym_p /= sym_p.sum()
    candidates.append(sym_p)

    # Structured inits at parent length
    for ci in structured_inits(n_parent, rng, count=3):
        candidates.append(ci)

    # Different lengths
    for n_try in [640, 896]:
        if n_try != n_parent:
            for ci in structured_inits(n_try, rng, count=2):
                candidates.append(ci)

    for idx, cand in enumerate(candidates):
        rem = deadline - time.monotonic() - 400
        if rem < 15:
            break
        n_cand = len(cand)
        if n_cand == n_parent:
            t_alloc = min(rem * 0.18, 120)
        else:
            t_alloc = min(rem * 0.10, 50)
        if t_alloc < 8:
            t_alloc = min(rem * 0.25, 60)

        use_sym = (idx == 1)
        print(f"  Cand {idx}: n={n_cand}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent_optimize(cand, time_limit=t_alloc,
                                            top_k=max(30, n_cand // 6),
                                            temperature=0.00004, lr=0.003,
                                            momentum=0.86, use_symmetry=use_sym)
        try_evaluate(a_opt, best_state, f"explore_{idx}")

    # === Phase 2: Deep optimization with diverse configurations ===
    print(f"\n=== Phase 2: Deep Optimization (best={best_state[1]:.6f}) ===")
    if best_state[0] is not None:
        n_best = len(best_state[0])

        configs = [
            # (top_k_frac, temp, lr, momentum, sg_mode, sym, init_mode)
            (0.25, 0.00002, 0.002, 0.91, "softmax", False, "direct"),
            (0.15, 0.00008, 0.004, 0.80, "softmax", False, "noise"),
            (0.10, 0.000005, 0.001, 0.95, "quadratic", False, "direct"),
            (0.33, 0.00015, 0.005, 0.75, "softmax", False, "reverse"),
            (1.0, 0.00001, 0.0012, 0.93, "quadratic", True, "sym"),
            (0.20, 0.00003, 0.0025, 0.88, "softmax", False, "blend"),
        ]

        for p, (tk_frac, temp, lr, mom, sg_mode, sym, init_mode) in enumerate(configs):
            rem = deadline - time.monotonic() - 350
            if rem < 15:
                break
            t_alloc = min(rem * 0.38, 250)

            if init_mode == "direct":
                f_init = best_state[0].copy()
            elif init_mode == "noise":
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_state[0] + noise)
            elif init_mode == "reverse":
                f_init = 0.65 * best_state[0] + 0.35 * best_state[0][::-1]
                f_init = project_simplex(f_init)
            elif init_mode == "sym":
                f_init = 0.5 * (best_state[0] + best_state[0][::-1])
                f_init /= f_init.sum()
            else:
                f_init = 0.85 * best_state[0] + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)

            tk = max(10, int(n_best * tk_frac))
            print(f"  Deep {p}: n={n_best}, t={t_alloc:.0f}s, tk={tk}, sg={sg_mode}, sym={sym}")
            a_opt, _ = mirror_descent_optimize(f_init, time_limit=t_alloc,
                                                top_k=tk, temperature=temp,
                                                lr=lr, momentum=mom, use_symmetry=sym,
                                                sg_mode=sg_mode)
            try_evaluate(a_opt, best_state, f"deep_{p}")

    # === Phase 3: Peak-directed and pair local search ===
    print(f"\n=== Phase 3: Local Search (best={best_state[1]:.6f}) ===")
    if best_state[0] is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_pds = min(rem * 0.3, 80)
            print(f"  Peak-directed t={t_pds:.0f}s")
            a_opt, _ = peak_directed_cd(best_state[0], time_limit=t_pds)
            try_evaluate(a_opt, best_state, "peak_cd")

        rem = deadline - time.monotonic() - 200
        if rem > 15:
            t_pp = min(rem * 0.3, 50)
            print(f"  Peak-pair t={t_pp:.0f}s")
            a_opt, _ = peak_pair_search(best_state[0], time_limit=t_pp)
            try_evaluate(a_opt, best_state, "peak_pair")

        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_pex = min(rem * 0.4, 60)
            print(f"  Pair exchange t={t_pex:.0f}s")
            a_opt, _ = pair_exchange_search(best_state[0], time_limit=t_pex)
            try_evaluate(a_opt, best_state, "pair_exch")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length (best={best_state[1]:.6f}) ===")
    if best_state[0] is not None:
        n_best = len(best_state[0])
        target_lengths = sorted(set([
            n_best - 128, n_best - 64, n_best + 64, n_best + 128, n_best + 256
        ]))
        target_lengths = [tl for tl in target_lengths if 256 <= tl <= 2560]

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 150
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_state[0], target_n)
            t_alloc = min(rem * 0.3, 100)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent_optimize(f_interp, time_limit=t_alloc,
                                                top_k=max(30, target_n // 6),
                                                temperature=0.00003, lr=0.0025,
                                                momentum=0.88)
            try_evaluate(a_opt, best_state, f"len_{target_n}")

            # Quick local search on the result
            rem = deadline - time.monotonic() - 100
            if rem > 10:
                a_opt2, _ = peak_directed_cd(a_opt, time_limit=min(rem, 20))
                try_evaluate(a_opt2, best_state, f"len_{target_n}_cd")

    # === Phase 5: Final intensive polishing ===
    print(f"\n=== Phase 5: Final Polish (best={best_state[1]:.6f}) ===")
    if best_state[0] is not None:
        n_best = len(best_state[0])

        # Ultra-fine mirror descent
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            a_opt, _ = mirror_descent_optimize(best_state[0], time_limit=min(rem, 100),
                                                top_k=n_best, temperature=0.000001,
                                                lr=0.0004, momentum=0.97, sg_mode="quadratic")
            try_evaluate(a_opt, best_state, "ultra_fine")

        # Peak-directed on best
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = peak_directed_cd(best_state[0], time_limit=min(rem, 40))
            try_evaluate(a_opt, best_state, "polish_pcd")

        # Pair exchange
        rem = deadline - time.monotonic() - 25
        if rem > 8:
            a_opt, _ = pair_exchange_search(best_state[0], time_limit=min(rem, 25))
            try_evaluate(a_opt, best_state, "polish_pex")

        # Short perturbations with quick optimization
        for trial in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            scale = [0.0003, 0.001, 0.003, 0.006][trial % 4]
            noise = rng.standard_normal(n_best) * scale
            f_pert = project_simplex(best_state[0] + noise)
            t_short = min(rem, 12)
            a_opt, _ = mirror_descent_optimize(f_pert, time_limit=t_short,
                                                top_k=max(30, n_best // 5),
                                                temperature=0.00002, lr=0.001,
                                                momentum=0.92)
            try_evaluate(a_opt, best_state, f"pert{trial}")

        # Final peak-directed pass
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = peak_directed_cd(best_state[0], time_limit=min(rem, 15))
            try_evaluate(a_opt, best_state, "final_pcd")

    # Fallback
    if best_state[0] is None:
        best_state[0] = np.ones(768) / 768
        best_state[1] = float(evaluate_sequence(best_state[0].tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_state[0].tolist()))
    if np.isfinite(fs):
        best_state[1] = fs

    print(f"\nFINAL SCORE: {best_state[1]:.6f} (n={len(best_state[0])})")
    return [float(x) for x in best_state[0]]
