3. Mirror descent (multiplicative) variant
4. Multi-length exploration with simulated annealing
5. Peak-averaged subgradient for multi-peak handling
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad(f, w, n_peaks=None):
    """Compute gradient of sum_k w[k]*c_k using np.correlate.
    c_k = (f*f)[k], dc_k/df[i] = 2*f[k-i] (if in range)
    grad[i] = 2 * sum_j w[i+j] * f[j] = 2 * np.correlate(w, f, 'valid')[i]
    """
    n = len(f)
    if n_peaks is not None and n_peaks < len(w):
        # Focus on top peaks
        idx = np.argpartition(w, -n_peaks)[-n_peaks:]
        w_focused = np.zeros(len(w))
        w_focused[idx] = w[idx]
    else:
        w_focused = w
    return 2.0 * np.correlate(w_focused, f, mode='valid')

def lse_optimize(f0, time_limit, beta_start=20.0, beta_max=20000.0, 
                 lr_init=None, backtracking=True, top_k=None):
    """LSE gradient descent with backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if lr_init is None:
        lr_init = 2.0 / n
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        c_max = conv.max()
        
        # LSE weights
        beta_c = beta * conv
        beta_c_shifted = beta_c - beta_c.max()
        exp_bc = np.exp(beta_c_shifted)
        w = exp_bc / exp_bc.sum()
        
        # Gradient
        grad = compute_grad(f, w, n_peaks=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_dir = -grad / gnorm  # descent direction
        
        # Line search
        if backtracking:
            step = lr
            best_val = best_obj
            best_f_new = f.copy()
            for _ in range(20):
                f_new = project_simplex(f + step * grad_dir * gnorm)
                # Wait, grad_dir is already normalized, so the actual gradient magnitude matters
                # Let me redo: update = f - lr * grad (then project)
                # Actually, let's do: f_new = project_simplex(f - step * grad)
                f_new = project_simplex(f - step * grad)
                f_new = np.maximum(f_new, 0)
                if f_new.sum() < 1e-15:
                    step *= 0.5
                    continue
                f_new /= f_new.sum()
                val = fast_convolve(f_new).max()
                if val < best_val:
                    best_val = val
                    best_f_new = f_new.copy()
                step *= 0.7  # shrink
                
                if step * gnorm < 1e-10:
                    break
        
        else:
            f_new = project_simplex(f - lr * grad)
            f_new = np.maximum(f_new, 0)
            f_new /= max(f_new.sum(), 1e-15)
            val = fast_convolve(f_new).max()
            best_f_new = f_new
            best_val = val
        
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_f = best_f_new.copy()
            f = best_f_new
            no_improve = 0
            lr = min(lr * 1.01, 10.0 / n)
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve > 50:
                # Perturb
                rng = np.random.default_rng(it * 7 + 3)
                scale = 0.001 + 0.003 * min(1.0, (no_improve - 50) / 50.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                f = np.maximum(f, 0)
                f /= max(f.sum(), 1e-15)
                v = np.zeros(n)
                lr = lr_init
                beta = max(beta_start, beta * 0.5)
                no_improve = 0
        
        if beta < beta_max:
            beta *= 1.005
    
    return best_f, best_obj

def mirror_descent(f0, time_limit, beta_start=50.0, beta_max=10000.0, 
                   lr_init=None, top_k=None):
    """Mirror descent with entropy (multiplicative) updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if lr_init is None:
        lr_init = 0.5 / np.sqrt(n)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        beta_c = beta * conv
        beta_c_shifted = beta_c - beta_c.max()
        exp_bc = np.exp(beta_c_shifted)
        w = exp_bc / exp_bc.sum()
        
        grad = compute_grad(f, w, n_peaks=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Multiplicative update: f_new[j] = f[j] * exp(-lr * grad[j]) / Z
        log_update = -lr * grad / gnorm
        # Clip to prevent overflow
        log_update -= log_update.max()
        update = np.exp(log_update)
        f_new = f * update
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            no_improve += 100
            break
        f_new /= f_new_sum
        
        val = fast_convolve(f_new).max()
        if val < best_obj - 1e-14:
            best_obj = val
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.95
            if no_improve > 100:
                rng = np.random.default_rng(it * 13 + 1)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                f = np.maximum(f, 0)
                f /= max(f.sum(), 1e-15)
                lr = lr_init
                beta = beta_start
                no_improve = 0
        
        if beta < beta_max:
            beta *= 1.003
    
    return best_f, best_obj

def peak_average_subgrad(f, n_peaks=5, tol=1e-6):
    """Compute average subgradient over peaks within tol of max."""
    conv = fast_convolve(f)
    c_max = conv.max()
    active = np.where(conv >= c_max - tol * c_max)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    elif len(active) > n_peaks:
        # Take the top n_peaks from active
        vals = conv[active]
        top_idx = np.argsort(vals)[::-1][:n_peaks]
        active = active[top_idx]
    
    # Average gradient over active peaks
    n = len(f)
    grad = np.zeros(n)
    for k in active:
        # dc_k/df[i] = 2*f[k-i] if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += f[k - hi: k - lo][::-1] / len(active)
    return 2.0 * grad, conv

def subgrad_optimize(f0, time_limit, n_peaks=5, tol=1e-5, lr_init=None):
    """Subgradient method with peak averaging."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if lr_init is None:
        lr_init = 1.0 / n
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        c_max = conv.max()
        active = np.where(conv >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        elif len(active) > n_peaks:
            vals = conv[active]
            top_idx = np.argsort(vals)[::-1][:n_peaks]
            active = active[top_idx]
        
        # Average subgradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += f[k - hi: k - lo][::-1]
        grad /= len(active)
        grad *= 2.0  # factor of 2 from the quadratic form
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking
        step = lr
        best_val = best_obj
        best_f_new = f.copy()
        for _ in range(15):
            f_new = project_simplex(f - step * grad)
            f_new = np.maximum(f_new, 0)
            if f_new.sum() < 1e-15:
                step *= 0.5
                continue
            f_new /= f_new.sum()
            val = fast_convolve(f_new).max()
            if val < best_val:
                best_val = val
                best_f_new = f_new.copy()
            step *= 0.6
            if step * gnorm < 1e-12:
                break
        
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_f = best_f_new.copy()
            f = best_f_new
            no_improve = 0
            lr = min(lr * 1.005, 5.0 / n)
        else:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 31 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                f = np.maximum(f, 0)
                f /= max(f.sum(), 1e-15)
                lr = lr_init
                no_improve = 0
    
    return best_f, best_obj

def interpolate_to_length(f, new_n):
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
