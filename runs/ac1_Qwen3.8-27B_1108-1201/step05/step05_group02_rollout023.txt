import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, top_k, temperature):
    """Compute weighted subgradient of max_k (a*a)_k using top-k peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        # d/d a_m (a*a)_k = 2 * a_{k-m} if 0 <= k-m < n
        # So sg_m += 2 * w_i * a_{k-m} for valid m
        # m ranges from max(0, k-n+1) to min(n-1, k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # For m in [js, je], k-m is in [k-je, k-js] = [max(0, n-1-k), min(n-1, k)]
        # So a_{k-m} for m in [js, je] is a[k-m] where k-m ranges from k-je to k-js
        idx = np.arange(js, je + 1)
        kg = k - idx  # k - m values
        sg[idx] += 2.0 * w[i] * a[kg]
    
    return sg

        t = np.arange(n)
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif shape == "sine":
        t = np.arange(n)
        a = np.sin(np.pi * (t + 0.5) / n)
    elif shape == "raised_cosine":
        t = np.arange(n) / max(n - 1, 1)
        a = np.sin(np.pi * t) ** 2
    elif shape == "triangle":
        t = np.arange(n) / max(n - 1, 1)
        a = 1.0 - np.abs(2.0 * t - 1.0)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    if boundary_zeros > 0 and boundary_zeros * 2 < n:
        a[:boundary_zeros] = 0
        a[-boundary_zeros:] = 0

    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Multi-shape, multi-length exploration with quick subgradient
    Phase 2: Intensive subgradient on top candidates
    Phase 3: LP-FW local refinement
    Phase 4: Coordinate descent polish
    Phase 5: Perturbation-based final search
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Multi-shape, multi-length exploration ===
    print(f"\n=== Phase 1: Exploration (best={best_score:.8f}) ===")

    shapes = ["uniform", "cosine", "gaussian", "sine", "raised_cosine", "triangle"]
    lengths = [512, 768, 1024, 1536, 2048, 3072]
    boundary_options = [0, 5, 10]

    candidates = []  # (score, sequence, label)

    t_phase1_start = time.monotonic()
    phase1_time = 180  # seconds for exploration

    # Quick evaluation: just a few SG iterations to rank candidates
    quick_time = 8

    for n in lengths:
        if time.monotonic() - t_phase1_start > phase1_time:
            break
        for shape in shapes:
            if time.monotonic() - t_phase1_start > phase1_time:
                break
            for bz in boundary_options[:2]:  # Skip bz=10 for speed in phase 1
                if time.monotonic() - t_phase1_start > phase1_time:
                    break
                a_init = make_initial_sequence(n, shape, bz)
                # Quick subgradient to get a rough score
                a_quick, obj_quick = sg_optimize(a_init, time_limit=quick_time,
                                                  lr=0.008, momentum=0.9,
                                                  top_k=30, temp=0.001)
                score_quick = float(evaluate_sequence(a_quick.tolist()))
                if np.isfinite(score_quick):
                    candidates.append((score_quick, a_quick, f"{shape}_n{n}_bz{bz}"))

    # Sort by score and keep top ones
    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates evaluated in {time.monotonic()-t_phase1_start:.0f}s")
    if candidates:
        print(f"  Best Phase 1: {candidates[0][0]:.8f} ({candidates[0][2]})")

    # Update best from phase 1
    for score_c, seq_c, label_c in candidates[:5]:
        if score_c < best_score:
            try_seq(seq_c, label_c)

    # === Phase 2: Intensive subgradient on top candidates ===
    print(f"\n=== Phase 2: Intensive SG (best={best_score:.8f}) ===")

    # Build list of starting points for intensive optimization
    intensive_starts = []

    # Add best from phase 1
    if candidates:
        intensive_starts.append((candidates[0][1], candidates[0][2]))

    # Add parent if it's good
    if parent is not None and len(parent) >= 256:
        intensive_starts.append((parent, "parent"))

    # Add a few more diverse candidates
    for i in range(1, min(4, len(candidates))):
        intensive_starts.append((candidates[i][1], candidates[i][2]))

    # Also try some manually designed starting points at larger n
    for n_large in [2048, 4096]:
        for shape in ["cosine", "gaussian"]:
            for bz in [0, 10, 20]:
                a_init = make_initial_sequence(n_large, shape, bz)
                intensive_starts.append((a_init, f"manual_{shape}_n{n_large}_bz{bz}"))

    # Limit to 8 intensive starts
    intensive_starts = intensive_starts[:8]

    t_phase2_start = time.monotonic()
    phase2_time = 800  # seconds for intensive optimization

    for idx, (a_start, label) in enumerate(intensive_starts):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.4, 200)
        if t_alloc < 30:
            t_alloc = min(rem - 60, 100)
        if t_alloc < 20:
            break

        n_cur = len(a_start)
        # Adjust parameters based on n
        top_k = max(30, min(80, n_cur // 16))
        lr_init = max(0.002, 0.8 / n_cur * 10)
        temp_init = 0.0005 * (n_cur / 1024)

        print(f"  SG [{label}] n={n_cur}, t={t_alloc:.0f}s")
        a_opt, obj = sg_optimize(a_start, time_limit=t_alloc,
                                  lr=lr_init, momentum=0.93,
                                  top_k=top_k, temp=temp_init, verbose=True)
        try_seq(a_opt, f"sg2_{label}")

        # If this improved, do another round with smaller lr
        if best_score < 1.5049 and time.monotonic() - t_phase2_start < phase2_time:
            rem = deadline - time.monotonic()
            if rem > 60:
                t_alloc2 = min(rem * 0.3, 150)
                if t_alloc2 > 20:
                    print(f"  SG-fine [{label}] t={t_alloc2:.0f}s")
                    a_opt2, obj2 = sg_optimize(best_seq, time_limit=t_alloc2,
                                                lr=lr_init * 0.3, momentum=0.95,
                                                top_k=top_k, temp=temp_init * 0.3)
                    try_seq(a_opt2, f"sg2fine_{label}")

    # === Phase 3: LP-FW local refinement ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")

    t_phase3_start = time.monotonic()
    if best_seq is not None and len(best_seq) <= 2048:
        rem = deadline - time.monotonic()
        t_lpfw = min(rem * 0.3, 300)
        if t_lpfw > 30:
            print(f"  LP-FW n={len(best_seq)}, t={t_lpfw:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_lpfw)
            try_seq(a_opt, "lpfw_main")
    else:
        # If best is too large for LP, try a smaller version
        if best_seq is not None:
            a_small = interpolate_sequence(best_seq, 1024)
            rem = deadline - time.monotonic()
            if rem > 100:
                print(f"  LP-FW on interpolated n=1024")
                a_opt, obj = lp_fw_optimize(a_small, time_limit=min(rem - 100, 200))
                try_seq(a_opt, "lpfw_interp1024")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Peak CD (best={best_score:.8f}) ===")

    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.3, 250)
        if t_cd > 20:
            n_cur = len(best_seq)
            print(f"  CD n={n_cur}, t={t_cd:.0f}s")
            a_opt, obj = peak_cd_optimize(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_main")

    # === Phase 5: Perturbation-based final search ===
    print(f"\n=== Phase 5: Perturbations (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(10):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            noise_scale = 0.002 + 0.002 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.3, 60)
            if t_alloc < 15:
                continue
            a_opt, _ = sg_optimize(f_pert, time_limit=t_alloc,
                                    lr=0.002, momentum=0.95,
                                    top_k=max(20, n_best // 20),
                                    temp=0.00005)
            try_seq(a_opt, f"perturb_{p}")

    # === Phase 6: Try larger sequences from best ===
    print(f"\n=== Phase 6: Larger sequences (best={best_score:.8f}) ===")

    if best_seq is not None:
        for target_n in [len(best_seq) * 2, len(best_seq) + 512]:
            if target_n > 8192 or target_n < 256:
                continue
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            a_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 120)
            if t_alloc < 20:
                continue
            print(f"  SG n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = sg_optimize(a_interp, time_limit=t_alloc,
                                    lr=0.003, momentum=0.93,
                                    top_k=max(30, target_n // 20),
                                    temp=0.0002)
            try_seq(a_opt, f"larger_n{target_n}")

    # === Final: Quick polish on the absolute best ===
    print(f"\n=== Final Polish (best={best_score:.8f}) ===")

    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            n_best = len(best_seq)
            a_opt, _ = sg_optimize(best_seq, time_limit=min(rem - 10, 40),
                                    lr=0.0005, momentum=0.97,
                                    top_k=max(10, n_best // 25),
                                    temp=0.00002)
            try_seq(a_opt, "final_polish")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
