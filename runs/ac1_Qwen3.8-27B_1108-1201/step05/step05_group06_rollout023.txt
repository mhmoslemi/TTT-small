import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(f, g):
    """Compute f * g (full linear convolution)."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]

def lp_direction(f, M, n):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns the optimal g (unnormalized) or None if infeasible.
    """
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_{i: max(0, k-n+1) <= i <= min(n-1, k)} f[i] * g[k-i]
    # For each k, the constraint is a linear form in g.

    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # equivalently: A_ub[k, j] = f[i] where i = k-j, valid if 0 <= i < n
    
    # A_ub has shape (2n-1, n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = f[i]
    
    b_ub = np.full(2*n-1, M)
    
    # g >= 0: bounds
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-guided direction search

Key innovation: Uses linear programming to find descent directions that are
more informed than subgradient methods, exploiting the structure of the

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(f, g):
    """Full linear convolution f * g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]

def build_conv_matrix(f, n):
    """Build matrix C such that (C @ g)[k] = (f*g)[k] for all k."""
    C = np.zeros((2*n - 1, n))
    for j in range(n):
        start = j
        end = min(j + n, 2*n - 1)
        C[start:end, j] = f[:end - start]
    return C

    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns optimal g or None.
    """

    c = -np.ones(n)  # minimize -sum(g)

    C = build_conv_matrix(f, n)
    b_ub = np.full(2*n - 1, M)

    return None

            b_shifted[start:end] = b[:end - start]
            
            e_shifted = np.zeros(conv_len)
            if 2*j < conv_len:
                e_shifted[2*j] = 1.0
            
            # f_new * f_new at param x:
            # = (1-x)^2 * B + 2x(1-x) * b_shifted + x^2 * e_shifted
            # = B - 2x*B + x^2*B + 2x*b_shifted - 2x^2*b_shifted + x^2*e_shifted
            # = B + x*(2*b_shifted - 2*B) + x^2*(B - 2*b_shifted + e_shifted)
            
            # We want to minimize max_k of this quadratic in x
            # For each k: q_k(x) = B[k] + x*(2*b_shifted[k] - 2*B[k]) + x^2*(B[k] - 2*b_shifted[k] + e_shifted[k])
            
            # The max of quadratics is convex in x, so we can use ternary search
            lo = 0.0
            hi = 1.0
            
            # Check if x_old is already good
            for _ in range(60):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                
                # Compute max_k q_k(m1)
                q1 = B + m1*(2*b_shifted - 2*B) + m1*m1*(B - 2*b_shifted + e_shifted)
                obj1 = q1.max()
                
                q2 = B + m2*(2*b_shifted - 2*B) + m2*m2*(B - 2*b_shifted + e_shifted)
                obj2 = q2.max()
                
                if obj1 < obj2:
                    hi = m2
                else:
                    lo = m1
            
            x_new = (lo + hi) / 2
            
            if abs(x_new - x_old) < 1e-13:
                continue
            
            # Apply
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] = b[mask] * (1 - x_new)
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a /= ss
            
            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:

            a = np.maximum(a + rng.standard_normal(n) * 0.001, 0)
            a /= a.sum()
    
    return best_a, best_obj

def lp_optimize(f, time_limit, verbose=False):
    """Main LP-guided optimization loop."""
    f = f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP
        g0 = lp_find_direction(f, M)
        
        if g0 is None or g0.sum() < 1e-10:
            # LP infeasible or trivial, try coordinate descent
            break
        
        g = g0 / g0.sum()  # normalize to sum = 1
        
        # Line search
        best_t, obj_after = line_search(f, g, t_max=0.5)
        
        if best_t > 0 and obj_after < best_obj - 1e-14:
            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            ss = f.sum()
            if ss > 1e-15:
                f /= ss
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP iter {it}: obj={best_obj:.10f}, t={best_t:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 20:

            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 0)

            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            if verbose:
                print(f"    LP iter {it}: restart, obj={M:.10f}")
    
    return best_f, best_obj

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a combination of:

2. Smooth gradient optimization via L-BFGS on softmax parameterization

  maximize sum(g) s.t. g >= 0, (f*g)[k] <= M_bound for all k
This is more aggressive than subgradient methods and can escape local minima.

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Full linear convolution f * g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def build_lp_matrix(f, n):
    """Build matrix C where (C @ g)[k] = (f*g)[k]."""
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        C[start:end, j] = f[:end - start]
    return C


def solve_lp_direction(f, M_bound):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M_bound."""
    n = len(f)
    c = -np.ones(n)
    C = build_lp_matrix(f, n)
    b_ub = np.full(2 * n - 1, M_bound)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 30})
        if result.success and result.x is not None:
            g = result.x
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def smooth_obj_grad(p, T):
    """Compute smooth objective and gradient for softmax parameterization."""
    n = len(p)
    p_max = p.max()
    exp_p = np.exp(p - p_max)
    a = exp_p / exp_p.sum()
    
    conv = fast_convolve(a)
    Tconv = T * conv
    Tconv_max = Tconv.max()
    shifted = Tconv - Tconv_max
    exp_shifted = np.exp(shifted)
    log_sum_exp = Tconv_max + np.log(exp_shifted.sum())
    
    # Weights
    w = exp_shifted / exp_shifted.sum()
    
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] for 0 <= k-j < n
    # = 2 * cross_correlation(w, a)[j]
    # Use FFT: cross_corr[j] = sum_i a[i] * w[i+j]
    # = (a_rev * w)[j + n - 1] where a_rev = a[::-1]
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = np.fft.rfft(a[::-1], fft_size)
    W = np.fft.rfft(w, fft_size)
    cross_full = np.fft.irfft(A_rev * W, fft_size)[:conv_len]
    grad_a = 2.0 * cross_full[n-1:n-1+n]
    
    # Chain rule for softmax
    mean_grad = np.dot(a, grad_a)
    grad_p = a * (grad_a - mean_grad)
    
    return log_sum_exp, grad_p


def smooth_optimize(f, time_limit, T_start=5.0, T_end=50.0, verbose=False):
    """Smooth gradient optimization using L-BFGS-B on softmax parameterization."""
    f = f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    # Initialize p from f: p[j] = log(f[j]) (with small epsilon for zeros)

    p0 = np.log(f_safe)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    # Multi-temperature schedule
    temps = [T_start, T_start * 2, T_start * 4, min(T_start * 8, T_end), T_end]
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        t_remaining = time_limit - (time.monotonic() - t0)
        if t_remaining < 5:
            break

            val, _ = smooth_obj_grad(p, T)
            return val
        def gfun(p):
            _, g = smooth_obj_grad(p, T)
            return g
        
        try:
            result = minimize(fun, p0, jac=gfun, method='L-BFGS-B',
                             options={'maxiter': 200, 'maxfun': 500,

        except Exception:
            break
        
        # Extract result

            best_obj = obj
            best_f = a_opt.copy()
            p0 = p_opt.copy()
            if verbose:
                print(f"    Smooth T={T}: obj={best_obj:.10f}")
    
    return best_f, best_obj


def coord_descent(a, time_limit, max_passes=200):
    """Coordinate descent with ternary search per coordinate."""
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    for _pass in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-13:
                continue
            
            x_old = a[j]
            others = a.copy()
            others[j] = 0.0
            others_sum = others.sum()
            if others_sum < 1e-14:
                continue
            b = others / others_sum
            
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            # b_shifted[k] = b[k-j] if 0 <= k-j < n
            b_shifted = np.zeros(conv_len)
            st = j
            en = min(j + n, conv_len)
            b_shifted[st:en] = b[:en - st]
            
            e_shifted = np.zeros(conv_len)
            if 2 * j < conv_len:
                e_shifted[2 * j] = 1.0
            
            # q(x) = (1-x)^2 * B + 2x(1-x) * b_shifted + x^2 * e_shifted
            # = B + x*(2*b_shifted - 2*B) + x^2*(B - 2*b_shifted + e_shifted)
            lin = 2.0 * b_shifted - 2.0 * B
            quad = B - 2.0 * b_shifted + e_shifted
            
            # Ternary search for min of max_k q_k(x)
            lo = 0.0
            hi = 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                q1 = B + m1 * lin + m1 * m1 * quad
                q2 = B + m2 * lin + m2 * m2 * quad
                if q1.max() < q2.max():
                    hi = m2
                else:
                    lo = m1
            
            x_new = (lo + hi) / 2.0
            
            if abs(x_new - x_old) < 1e-13:
                continue
            
            # Apply
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] = b[mask] * (1.0 - x_new)
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss < 1e-15:
                a = best_a.copy()
                break
            a /= ss
            
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Perturb to escape
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.001, 0)
            a /= a.sum()
            if not improved:
                # Check if we've been stuck
                break
    
    return best_a, best_obj


def interpolate_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result
