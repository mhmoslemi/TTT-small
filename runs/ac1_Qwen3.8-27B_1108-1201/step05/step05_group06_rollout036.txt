"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent
with adaptive log-sum-exp temperature, exact coordinate bisection polishing,
basin hopping for global exploration, and multi-length search.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_grad_step(a, beta):
    """Compute the smoothed gradient of log-sum-exp(beta * conv(a,a))."""
    n = len(a)
    c = fast_convolve(a)
    
    # Numerically stable softmax
    c_shifted = c - c.max()
    exp_c = np.exp(beta * c_shifted)
    p = exp_c / exp_c.sum()
    
    # Gradient: grad[j] = 2 * sum_i a[i] * p[j+i]
    # = 2 * cross_correlation(a, p)[j]
    # = 2 * ifft(fft(p, N) * conj(fft(a, N)))[j]
    N = 1 << (3 * n - 1).bit_length()
    P_fft = np.fft.fft(p, N)
    A_fft = np.fft.fft(a, N)
    grad = 2.0 * np.real(np.fft.ifft(P_fft * np.conj(A_fft)))[:n]
    
    return grad, c.max()


def smoothed_gradient_descent(a0, time_limit=60.0, beta_start=5.0, beta_end=500.0,
                               lr_start=0.1, decay_lr=0.9995, decay_beta=1.005,
                               restart_every=200):
    """Gradient descent with adaptive log-sum-exp temperature."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_obj = float('inf')
    best_a = a.copy()
    curr = a.copy()
    beta = beta_start
    lr = lr_start
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, c_max = smoothed_grad_step(curr, beta)
        
        # Take gradient step
        curr_new = curr - lr * grad
        curr = project_simplex(curr_new)
        
        # Check improvement
        c = fast_convolve(curr)
        obj = c.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = curr.copy()
        
        # Adapt parameters
        beta = min(beta * decay_beta, beta_end)
        lr *= decay_lr
        
        # Periodic restart from best with noise
        if it % restart_every == 0:
            rng = np.random.default_rng(it)
            noise_scale = 0.01 * max(0.1, 1.0 - it / (2 * restart_every))
            curr = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            lr = lr_start * 0.5
            beta = max(beta_start, beta * 0.5)
        
        if lr < 1e-6:
            break
    
    return best_a, best_obj


def coordinate_bisection(a0, time_limit=30.0, max_passes=100):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-14:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            lo, hi = 0.0, 1.0
            x_new = x_old
            
            for _ in range(50):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:

                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                if deriv > 1e-13:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-13:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break
            
            if abs(x_new - x_old) < 1e-14:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

            rng = np.random.default_rng(int(time.monotonic() * 1000))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            fast_convolve(a)  # just to ensure valid state
    
    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def basin_hopping(a0, time_limit=60.0, n_hops=50):
    """Basin hopping: random perturbation + quick local optimization."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    for hop in range(n_hops):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Random perturbation
        noise_scale = 0.05 * (1.0 + hop / n_hops)
        a_tri = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
        
        # Quick local optimization (short CD)
        a_opt, obj = coordinate_bisection(a_tri, time_limit=min(5.0, (time_limit - (time.monotonic() - t0)) * 0.5), max_passes=3)
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt
    
    return best_a, best_obj


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

If j != k-j: dc_k/da_j = a[k-j] + a[k-j] = 2*a[k-j]
If j == k-j (k=2j): dc_k/da_j = 2*a[j]
In all cases: dc_k/da_j = 2*a[k-j] if 0<=k-j<=n-1, else 0.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam-optimized smoothed gradient
descent with adaptive log-sum-exp temperature, exact coordinate bisection polishing,
basin hopping for global exploration, multi-length search, and diverse initializations.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, beta):
    """Compute gradient of log-sum-exp(beta * conv(a,a)) using FFT."""
    n = len(a)
    c = fast_convolve(a)
    c_shifted = c - c.max()
    exp_c = np.exp(beta * c_shifted)
    p = exp_c / exp_c.sum()
    N = 1 << (3 * n - 1).bit_length()
    P_fft = np.fft.fft(p.astype(np.float64), N)
    A_fft = np.fft.fft(a.astype(np.float64), N)
    grad = 2.0 * np.real(np.fft.ifft(P_fft * np.conj(A_fft)))[:n]
    return grad, c.max(), p


def adam_smoothed_gd(a0, time_limit=60.0, beta_start=10.0, beta_end=3000.0,
                     lr_start=0.1, beta_decay=1.008, lr_decay=0.9997,
                     adam_b1=0.9, adam_b2=0.999, adam_eps=1e-8,
                     restart_interval=150, verbose=False):
    """Adam-optimized gradient descent with adaptive log-sum-exp temperature."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    beta = beta_start
    lr = lr_start
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_adam += 1
        grad, c_max, _ = compute_smoothed_gradient(a, beta)

        m = adam_b1 * m + (1.0 - adam_b1) * grad
        v = adam_b2 * v + (1.0 - adam_b2) * (grad * grad)
        m_hat = m / (1.0 - adam_b1 ** t_adam)
        v_hat = v / (1.0 - adam_b2 ** t_adam)
        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        a_new = project_simplex(a - step)
        a = a_new

        c = fast_convolve(a)
        obj = c.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        beta = min(beta * beta_decay, beta_end)
        lr *= lr_decay

        if it % restart_interval == 0:
            rng = np.random.default_rng(it * 7 + 3)
            noise_scale = 0.02 * max(0.1, 1.0 - 0.02 * it / restart_interval)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            lr = lr_start * 0.5
            beta = max(beta_start, beta * 0.6)

        if verbose and it % 500 == 0:
            print(f"    Adam-GD iter {it}: obj={best_obj:.10f}, beta={beta:.1f}")

    return best_a, best_obj


def coordinate_bisection(a0, time_limit=30.0, max_passes=50):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    for _pass in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo, hi = 0.0, 1.0
            x_new = x_old

            for _ in range(50):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                if deriv > 1e-13:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-13:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.0004)

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def basin_hopping(a0, time_limit=60.0, n_hops=40):
    """Basin hopping: random perturbation + quick local optimization."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)

    for hop in range(n_hops):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        noise_scale = 0.03 + 0.04 * (hop / max(1, n_hops - 1))
        a_tri = project_simplex(best_a + rng.standard_normal(n) * noise_scale)

        t_alloc = min(4.0, remaining * 0.6)
        a_opt, obj = coordinate_bisection(a_tri, time_limit=t_alloc, max_passes=3)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt

    return best_a, best_obj


def peak_shaving(a0, time_limit=30.0):
    """Direct peak shaving: reduce mass at elements contributing to the peak."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(8888)

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        c_max = c.max()
        if c_max < best_obj - 1e-14:
            best_obj = c_max
            best_a = a.copy()

        peak_idx = int(np.argmax(c))
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)

        contributors = np.arange(js, je + 1)
        contrib_vals = a[contributors] * a[peak_idx - contributors]
        total_contrib = contrib_vals.sum()
        if total_contrib < 1e-15:
            break

        # Find least-contributing element and reduce it
        min_local = np.argmin(contrib_vals)
        idx_reduce = contributors[min_local]
        if a[idx_reduce] < 1e-8:
            a[idx_reduce] = 0
            continue

        # Transfer mass to random non-contributing elements
        non_contrib = np.setdiff1d(np.arange(n), contributors)
        if len(non_contrib) == 0:
            break

        # Weighted random selection for transfer target
        weights = a[non_contrib] + 0.001
        weights /= weights.sum()
        target = non_contrib[rng.choice(len(non_contrib), p=weights)]

        transfer = 0.05 * a[idx_reduce]
        a[idx_reduce] -= transfer
        a[target] += transfer
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-15:
            a /= ss

    return best_a, best_obj

        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
