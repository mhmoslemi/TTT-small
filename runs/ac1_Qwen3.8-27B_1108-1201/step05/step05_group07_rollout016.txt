import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_full(a, conv, threshold_frac=0.85):
    """Compute subgradient using all peaks above threshold_frac * max."""
    n = len(a)
    M = conv.max()
    if M < 1e-30:
        return np.zeros(n)
    threshold = threshold_frac * M
    peaks = np.where(conv >= threshold)[0]
    if len(peaks) == 0:
        peaks = np.array([np.argmax(conv)])

    # Weight peaks by how close they are to the max
    weights = (conv[peaks] - threshold) / (M - threshold + 1e-30)
    weights += 0.1  # small base weight
    weights /= weights.sum()

    sg = np.zeros(n)
    for i, k in enumerate(peaks):
        k = int(k)
        w = weights[i]
        if w < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w * a[k - idx]
    return sg


def compute_subgradient_topk(a, conv, top_k=30, temperature=0.00003):
    """Compute softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def mu_weights_optimize(f0, time_limit, top_k=40, temperature=0.00004,

            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj

def block_rebalance(f0, time_limit, verbose=False):

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    num_blocks = max(4, n // 32)
    block_size = n // num_blocks

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        M = conv.max()

        # For each block, compute its "contribution" to the max peaks
        # and try to reduce it
        for b in range(num_blocks):
            if time.monotonic() - t0 > time_limit:
                break
            start = b * block_size
            end = min((b + 1) * block_size, n)
            if end - start < 2:
                continue

            # Try: reduce the max element in this block, increase the min
            block = f[start:end]
            max_idx_in_block = start + int(np.argmax(block))
            min_idx_in_block = start + int(np.argmin(block))

            if f[max_idx_in_block] < 1e-12:
                continue

            for frac in [0.05, 0.1, 0.2, 0.35]:

                    f[min_idx_in_block] -= amt
                    # Try a random target in a different block

                    os = other_block * block_size

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using smooth interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    # Use smooth interpolation
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Smooth a bit
    kernel = np.array([0.25, 0.5, 0.25])
    f_interp = np.convolve(f_interp, kernel, mode='same')
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate structured initial sequences with diverse shapes."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Slightly asymmetric uniform (offset cosine)
    phase = rng.uniform(0, 2 * np.pi)
    asym = 1.0 + 0.3 * np.cos(2 * np.pi * x + phase)
    asym = np.maximum(asym, 0.01)
    inits.append(asym / asym.sum())

    # Dual-peak (two Gaussians)
    c1 = rng.uniform(0.2, 0.4)
    c2 = rng.uniform(0.6, 0.8)
    sig = n / (6.0 + rng.random() * 4.0)
    g1 = np.exp(-0.5 * ((np.arange(n) - c1 * n) / sig) ** 2)
    g2 = np.exp(-0.5 * ((np.arange(n) - c2 * n) / sig) ** 2)
    dual = g1 + g2 + 0.1
    inits.append(dual / dual.sum())

    # Power-law decay from edges
    edges = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    power = 0.5 + rng.random() * 2.0
    pl = (n / 2 - edges) ** power + 0.1
    inits.append(pl / pl.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 0.01)
    inits.append(r / r.sum())

    # Triangular with noise
    tri = 1.0 - np.abs(2 * x - 1)
    noise = rng.standard_normal(n) * 0.1
    tri_noisy = np.maximum(tri + noise, 0.01)
    inits.append(tri_noisy / tri_noisy.sum())

    return inits[:count]

        print(f"No parent found, using uniform n=512, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Quick exploration with different inits and lengths ===
    print("\n=== Phase 1: Quick Exploration ===")
    exploration_lengths = [n_parent, n_parent - 64, n_parent + 64, n_parent + 128]
    exploration_lengths = [max(128, l) for l in exploration_lengths if l >= 128]

    for n_try in exhibition_lengths[:4]:
        rem = deadline - time.monotonic() - 600
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 60)
        if t_alloc < 10:
            continue

        inits = structured_inits(n_try, rng, count=4)
        for idx, init in enumerate(inits):
            rem = deadline - time.monotonic() - 500
            if rem < 15:
                break
            t_sub = min(rem * 0.25, 30)
            if t_sub < 5:
                break
            a_opt, _ = mu_weights_optimize(init, time_limit=t_sub,
                                            top_k=max(20, n_try // 10),
                                            temperature=0.00005, lr=0.004,
                                            momentum=0.82, use_full=True)
            try_seq(a_opt, f"exp_{n_try}_{idx}")

    # === Phase 2: Deep optimization on best ===
    print(f"\n=== Phase 2: Deep Optimization (best={best_score:.6f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {"top_k": max(30, n_best // 4), "temp": 0.00002, "lr": 0.002,
             "mom": 0.88, "use_full": False, "thresh": 0.9},
            {"top_k": max(20, n_best // 8), "temp": 0.00006, "lr": 0.003,
             "mom": 0.82, "use_full": True, "thresh": 0.85},
            {"top_k": max(50, n_best // 3), "temp": 0.00001, "lr": 0.0015,
             "mom": 0.92, "use_full": False, "thresh": 0.95},
            {"top_k": max(15, n_best // 12), "temp": 0.0001, "lr": 0.005,
             "mom": 0.78, "use_full": True, "thresh": 0.80},
        ]

        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 400
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 120)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init /= f_init.sum()
            else:
                # Slightly shift the sequence
                shift = int(rng.integers(-3, 4))
                f_shifted = np.roll(best_seq, shift)
                f_init = 0.7 * best_seq + 0.3 * f_shifted
                f_init = project_simplex(f_init)

            print(f"  Deep {p}: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = mu_weights_optimize(f_init, time_limit=t_alloc,
                                            top_k=cfg["top_k"], temperature=cfg["temp"],
                                            lr=cfg["lr"], momentum=cfg["mom"],
                                            use_full=cfg["use_full"],
                                            threshold_frac=cfg["thresh"])
            try_seq(a_opt, f"deep_{p}")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_cd = min(rem * 0.4, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_analytic(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_1")

            # Second pass with perturbation
            noise = rng.standard_normal(len(best_seq)) * 0.002
            f_pert = project_simplex(best_seq + noise)
            rem = deadline - time.monotonic() - 200
            if rem > 20:
                a_opt, _ = coordinate_descent_analytic(f_pert, time_limit=min(rem, 60))
                try_seq(a_opt, "cd_2")

    # === Phase 4: Pair exchange + block rebalancing ===
    print(f"\n=== Phase 4: Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_pex = min(rem * 0.3, 80)
            print(f"  Pair exchange t={t_pex:.0f}s")
            a_opt, _ = pair_exchange_search(best_seq, time_limit=t_pex)
            try_seq(a_opt, "pex_1")

            rem = deadline - time.monotonic() - 150
            if rem > 20:
                t_br = min(rem * 0.4, 80)
                print(f"  Block rebalance t={t_br:.0f}s")
                a_opt, _ = block_rebalance(best_seq, time_limit=t_br)
                try_seq(a_opt, "block_1")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = sorted(set([n_best - 128, n_best - 64, n_best + 64,
                                     n_best + 128, n_best + 256, n_best + 384]))
        target_lengths = [l for l in target_lengths if 256 <= l <= 4096]

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 100
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 80)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = mu_weights_optimize(f_interp, time_limit=t_alloc,
                                            top_k=max(30, target_n // 6),
                                            temperature=0.00004, lr=0.003,
                                            momentum=0.86, use_full=True)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 6: Final intensive polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine subgradient
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            a_opt, _ = mu_weights_optimize(best_seq, time_limit=min(rem, 60),
                                            top_k=n_best, temperature=0.000001,
                                            lr=0.0003, momentum=0.95, use_full=True,
                                            threshold_frac=0.95)
            try_seq(a_opt, "sg_fine")

        # Pair exchange
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = pair_exchange_search(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "pex_fine")

        # Multiple perturbation rounds
        for pround in range(6):
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break
            for scale in [0.0005, 0.001, 0.002]:
                rem = deadline - time.monotonic() - 10
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)
                t_sub = min(rem, 8)
                a_opt, _ = mu_weights_optimize(f_pert, time_limit=t_sub,
                                                top_k=max(20, n_best // 5),
                                                temperature=0.00002, lr=0.001,
                                                momentum=0.92, use_full=True)
                try_seq(a_opt, f"pert_{pround}_{scale}")

        # Coordinate descent as final step
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, _ = coordinate_descent_analytic(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "cd_final")

        # One more pair exchange
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = pair_exchange_search(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "pex_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
