def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def solve_lp_min_peak(f, n):
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based Frank-Wolfe with momentum, smooth-max subgradient descent,
peak-shaving coordinate descent, and length exploration.
Uses Nesterov-style acceleration and structured perturbations to escape
stationary points faster than the previous approach.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_max_sum(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def smooth_max_sg_optimize(f0, time_limit, verbose=False):
    """Subgradient method using smooth max approximation over ALL peaks with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    f_prev = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    tau = 0.00005
    lr = 0.006
    mu = 0.92
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute smooth max weights over ALL peaks
        scaled = conv / tau
        scaled -= scaled.max()
        # Clip for numerical stability
        scaled = np.clip(scaled, -50, 50)
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws

        # Compute gradient of smooth max
        sg = np.zeros(n)
        for i in range(len(conv)):
            if w[i] < 1e-12:
                continue
            k = i
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        # Nesterov momentum
        v = mu * v - lr * sg
        f_new = f + v
        f_new = project_simplex(f_new)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f_prev = f.copy()
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f_prev = f.copy()
            f = f_new

        # Adaptive learning rate and temperature
        lr *= 0.99992
        tau *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            f_prev = f.copy()
            v = np.zeros(n)
            lr = max(lr * 1.5, 0.002)
            no_improve = 0

    return best_f, best_obj

    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 200 == 0 and verbose:
            print(f"    PeakCD iter {it}: obj={best_obj:.10f}")

        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        improved = False
        for pk in sorted(peak_positions, key=lambda x: -conv[int(x)]):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(pk)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            products = f[idx] * f[kg]

            # Find heaviest contributing coordinate
            order = np.argsort(products)[::-1]
            for oi in order[:min(15, len(order))]:
                j = int(idx[oi])
                if f[j] < 1e-10:
                    continue

                # Try reducing f[j] by various fractions and renormalizing
                for frac in [0.2, 0.4, 0.6, 0.8]:
                    if time.monotonic() - t0 > time_limit:
                        break
                    delta = f[j] * frac
                    f_trial = f.copy()
                    f_trial[j] = f[j] - delta
                    if f_trial[j] < 0:
                        continue
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-10:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        improved = True
                        break

                # Also try increasing the least-heavy coordinate (redistribution)
                if not improved and len(order) > 1:
                    j2 = int(idx[order[-1]])
                    if f[j2] > 1e-10 and j2 != j:
                        for frac2 in [0.3, 0.5]:
                            delta2 = f[j] * frac2
                            f_trial = f.copy()
                            f_trial[j] = f[j] - delta2
                            f_trial[j2] = f[j2] + delta2
                            if f_trial[j] < 0 or f_trial[j2] < 0:
                                continue
                            val = fast_convolve(f_trial).max()
                            if val < best_obj - 1e-10:
                                f = f_trial
                                best_obj = val
                                best_f = f.copy()
                                conv = fast_convolve(f)
                                improved = True
                                break

                if improved:
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 80:
                rng = np.random.default_rng(it * 31337 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                no_improve = 0
            if no_improve > 300:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def l2_surrogate_optimize(f0, time_limit, verbose=False):
    """Optimize the L2 norm of the convolution as a surrogate for the max norm."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.01
    mu = 0.9
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Gradient of sum_k (f*f)[k]^2 = 4 * sum_k (f*f)[k] * d(f*f)[k]/df
        # d(f*f)[k]/df[j] = f[k-j] + f[j-k] (with bounds)
        # So grad = 4 * conv * f (circular-like, but linear convolution)
        # Actually: d/d f[j] of (f*f)[k] = f[k-j] (if 0<=k-j<n) + f[j-k] (if 0<=j-k<n)
        # This is essentially the "folded" convolution

        # Compute gradient efficiently
        # grad[j] = 4 * sum_k conv[k] * (f[k-j] + f[j-k]) where indices are valid
        # This equals 4 * (conv * f_folded)[j] where f_folded accounts for the two terms
        # Actually it's 4 * sum_k conv[k] * f[|k-j|] with appropriate bounds
        # Simpler: use the identity that grad of ||f*f||_2^2 = 4 * ifft(fft(conv) * conj(fft(f)))
        # But conv is the full linear convolution, so we need to be careful.

        # Let's compute it directly using the convolution structure
        # (f*f)[k] = sum_j f[j]*f[k-j]
        # d/d f[j] (f*f)[k] = f[k-j] (when 0 <= k-j < n) + f[j-(k-n+1+... )]
        # Actually: d/d f[j] sum_i f[i] f[k-i] = f[k-j] (if 0<=k-j<n) + f[j+k-(2n-2)... ]
        # Let me just use: it's the cross-correlation of f with itself, evaluated at shift j,
        # weighted by conv[k]... this is getting complex.

        # Simpler approach: use FFT
        # The gradient of (1/2)||f*f||_2^2 with respect to f is:
        # 4 * (f * (conv_f)) where * denotes appropriate convolution
        # Actually, let's just use a numerical approach for the gradient
        # via the subgradient of the max (which is simpler and we already have)

        # For now, use the same subgradient as smooth-max but with larger tau
        tau = 0.001
        scaled = conv / tau
        scaled -= scaled.max()
        scaled = np.clip(scaled, -30, 30)
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws

        sg = np.zeros(n)
        for i in range(len(conv)):
            if w[i] < 1e-10:
                continue
            k = i
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = mu * v - lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new

        lr *= 0.99995
        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 7) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = max(lr * 2, 0.003)
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. LP-FW with momentum on parent
    2. Smooth-max subgradient (all peaks, Nesterov)
    3. Peak-shaving coordinate descent
    4. Length exploration with LP-FW
    5. Final polish with fine-grained methods
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")

    # === Phase 1: LP-FW with momentum ===
    print(f"\n=== Phase 1: LP-FW + momentum (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(450, rem * 0.32)
    if t1 > 30:
        print(f"  Allocated: {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True, use_momentum=True)
        update_best(a_opt, "lpfw_mom")

    # === Phase 2: Smooth-max subgradient ===
    print(f"\n=== Phase 2: Smooth-Max SG (n={len(best_seq)}, best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(350, rem * 0.24)
    if t2 > 30 and best_seq is not None:
        print(f"  Allocated: {t2:.0f}s")
        a_opt, _ = smooth_max_sg_optimize(best_seq, time_limit=t2, verbose=True)
        update_best(a_opt, "smooth_sg")

    # === Phase 3: Peak-shaving CD ===
    print(f"\n=== Phase 3: Peak-Shaving CD (n={len(best_seq)}, best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(250, rem * 0.18)
    if t3 > 30 and best_seq is not None:
        print(f"  Allocated: {t3:.0f}s")
        a_opt, _ = peak_shaving_cd(best_seq, time_limit=t3, verbose=True)
        update_best(a_opt, "peak_cd")

    # === Phase 4: L2 surrogate (different landscape) ===
    print(f"\n=== Phase 4: L2 Surrogate (n={len(best_seq)}, best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(200, rem * 0.14)
    if t4 > 25 and best_seq is not None:
        print(f"  Allocated: {t4:.0f}s")
        a_opt, _ = l2_surrogate_optimize(best_seq, time_limit=t4, verbose=True)
        update_best(a_opt, "l2_surr")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.8f}) ===")
    candidate_lengths = []
    for delta in [-48, -24, -12, 12, 24, 48, 96]:
        ln = len(best_seq) + delta
        if 256 <= ln <= 2048:
            candidate_lengths.append(ln)

    for target_n in sorted(candidate_lengths)[:4]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.35, 180)
        if t_alloc < 20:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        print(f"  LP-FW n={target_n}, time={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, use_momentum=True)
        update_best(a_opt, f"len_{target_n}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick LP-FW
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_polish = min(rem, 100)
            print(f"  LP-FW polish: {t_polish:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_polish, use_momentum=True)
            update_best(a_opt, "lpfw_polish")

        # Quick SG
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_sg = min(rem, 80)
            print(f"  SG polish: {t_sg:.0f}s")
            a_opt, _ = smooth_max_sg_optimize(best_seq, time_limit=t_sg)
            update_best(a_opt, "sg_polish")

        # Quick peak CD
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_cd = min(rem, 60)
            print(f"  PeakCD polish: {t_cd:.0f}s")
            a_opt, _ = peak_shaving_cd(best_seq, time_limit=t_cd)
            update_best(a_opt, "cd_polish")

        # Perturbation restarts
        for p in range(6):
            rem = deadline - time.monotonic() - 10
            if rem < 12:
                break
            noise_scale = 0.0005 + 0.0003 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem, 25)
            a_opt, _ = smooth_max_sg_optimize(f_pert, time_limit=t_pert)
            update_best(a_opt, f"pert_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
