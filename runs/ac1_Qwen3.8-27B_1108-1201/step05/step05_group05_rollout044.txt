import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_palindromic(a, n):
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])

def interp_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def flat_spectrum_init(n, seed=0):
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(np.fft.irfft(spectrum, fft_size)[:n])
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig

def binary_init(n, seed=0, density=0.5):
    """Initialize with random binary sequence."""
    rng = np.random.default_rng(seed)
    sig = (rng.random(n) < density).astype(float)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.ones(n) / n
    else:
        sig /= ss
    return sig

    """Initialize with a few scattered peaks."""
    rng = np.random.default_rng(seed)
    sig = np.ones(n) * 0.1
    peak_positions = rng.choice(n, size=n_peaks, replace=False)
    sig[peak_positions] = 1.0
    ss = sig.sum()
    sig /= ss
    return sig

def solve_lp(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for k in active_k, g >= 0."""
    if active_k is None:
        active_k = list(range(2 * n - 1))
    
    c = -np.ones(n)
    K = len(active_k)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    
    for row, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-Frank-Wolfe with ALL constraints (full 2n-1 constraint set)
- Multi-start diverse initialization (flat spectrum, binary, scattered peaks, parent)
- Multi-length exploration and interpolation
- Block coordinate descent
- Subgradient polishing with adaptive temperature
- Peak equalization at convergence
"""
import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(np.fft.irfft(spectrum, fft_size)[:n])
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def binary_init(n, seed=0, density=0.4):
    rng = np.random.default_rng(seed)
    sig = (rng.random(n) < density).astype(float)
    sig += 0.05
    ss = sig.sum()
    sig /= ss
    return sig


def two_level_init(n, seed=0, ratio=3.0):
    rng = np.random.default_rng(seed)
    sig = np.ones(n)
    mask = rng.random(n) < 0.5
    sig[mask] = ratio
    ss = sig.sum()
    sig /= ss
    return sig


def build_full_conv_matrix(n, f):
    """Build the (2n-1) x n matrix where row k gives conv(f,.:) [k] = sum_j f[...]*g[k-j]."""
    rows = 2 * n - 1
    A = np.zeros((rows, n))
    for k in range(rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[k, j] = f[k - j]
    return A


def solve_lp_full(f, M, n, A_matrix):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. A is precomputed (2n-1) x n."""
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_matrix, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None


def solve_lp_subset(f, M, n, A_matrix, active_indices):
    """Solve LP with only active constraints."""
    c = -np.ones(n)
    K = len(active_indices)
    A_active = A_matrix[active_indices]
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_active, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, use_all_constraints=True,
                   top_k=60, step_mode='diminishing'):
    """LP-Frank-Wolfe with either all constraints or adaptive top-k."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    # Precompute the full convolution matrix
    if use_all_constraints:
        A_full = build_full_conv_matrix(n, f)

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Determine constraints to use
        if use_all_constraints:
            # Rebuild matrix since f changed
            A_full = build_full_conv_matrix(n, f)
            g = solve_lp_full(f, M * m_exp, n, A_full)
        else:
            k_sorted = np.argsort(conv)[::-1][:top_k]
            A_full = build_full_conv_matrix(n, f)
            g = solve_lp_subset(f, M * m_exp, n, A_full, k_sorted)

        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                m_exp = 1.0
                # Try with relaxed M
                continue
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 5.0)
            continue

        # Step size
        if step_mode == 'diminishing':
            alpha = 1.0 / (it + 2)
        else:
            # Quick 3-point line search
            alphas = [0.3, 0.5, 0.7, 1.0]
            best_alpha = 0.5
            best_alpha_obj = np.inf
            for a_try in alphas:
                f_try = project_simplex(f + a_try * (g - f))
                obj_try = fast_convolve(f_try).max()
                if obj_try < best_alpha_obj:
                    best_alpha_obj = obj_try
                    best_alpha = a_try
            alpha = best_alpha

        f_new = f + alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-15:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
            M = M_new
            m_exp = 1.0
            if verbose and it % 20 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")
        else:
            no_improve += 1

        f = f_new
        conv = conv_new

        if no_improve > 60:
            # Diversify
            rng = np.random.default_rng(it * 104729 + 1)
            scale = 0.002 + 0.004 * min(no_improve // 80, 8)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def block_coord_descent(f0, time_limit, block_size=3):
    """Block coordinate descent: optimize blocks of consecutive coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                break
            end = min(j + block_size, n)
            block = f[j:end].copy()
            block_sum = block.sum()
            if block_sum < 1e-12:
                continue

            # Try redistributing within the block
            for trial in range(5):
                if trial == 0:
                    candidate_block = block * 1.1
                elif trial == 1:
                    candidate_block = block * 0.9
                elif trial == 2:
                    new_block = block.copy()
                    if len(new_block) > 1:
                        new_block[0] *= 1.3
                        rest = new_block[1:]
                        rest *= 1.0
                        new_block[1:] = rest
                    candidate_block = new_block
                elif trial == 3:
                    candidate_block = np.ones(end - j) / (end - j) * block_sum
                else:
                    rng = np.random.default_rng(j * 31 + trial)
                    candidate_block = block * (1.0 + rng.standard_normal(end - j) * 0.1)
                    candidate_block = np.maximum(candidate_block, 0)

                cb_sum = candidate_block.sum()
                if cb_sum < 1e-12:
                    continue

                f_trial = f.copy()
                f_trial[j:end] = candidate_block
                f_trial = np.maximum(f_trial, 0)
                f_trial_sum = f_trial.sum()
                if f_trial_sum < 1e-12:
                    continue
                f_trial /= f_trial_sum

                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < best_obj - 1e-14:
                    f = f_trial
                    best_obj = obj_trial
                    best_f = f.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.001, mom=0.94, top_k=50, temp=0.00003):
    """Subgradient polishing with softmax-weighted active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(it_count * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            no_improve = 0

    return best_f, best_obj


def peak_equalization(f0, time_limit, n_rounds=20):
    """Try to equalize the top peaks of the convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Find all peaks within 1% of max
        threshold = best_obj * 0.999
        active_peaks = np.where(conv >= threshold)[0]

        if len(active_peaks) < 2:
            break

        # For each active peak, compute the "pressure" on each coordinate
        pressure = np.zeros(n)
        for k in active_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            pressure[idx] += f[k - idx]

        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure /= p_max

        # Move f away from high-pressure regions
        step = 0.01 / (1 + rnd * 0.5)
        f_new = f - step * pressure
        f_new = np.maximum(f_new, 0)
        f_sum = f_new.sum()
        if f_sum < 1e-12:
            break
        f_new /= f_sum

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            f = f_new
            best_obj = obj_new
            best_f = f_new.copy()
            conv = conv_new
        else:
            # Try smaller step
            f_new = f - step * 0.3 * pressure
            f_new = np.maximum(f_new, 0)
            f_sum = f_new.sum()
            if f_sum > 1e-12:
                f_new /= f_sum
                obj_new = fast_convolve(f_new).max()
                if obj_new < best_obj - 1e-14:
                    f = f_new
                    best_obj = obj_new
                    best_f = f_new.copy()
                    conv = fast_convolve(f)

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):

    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP-Frank-Wolfe with full constraints, multi-start, multi-length exploration,

    """

    best_seq = None
    best_score = float('inf')

            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    try:

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = flat_spectrum_init(768, seed=123)

    # === Phase 1: LP-FW on parent with FULL constraints ===
    print("\n=== Phase 1: LP-FW full constraints on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.35)
    if t1 > 60:
        print(f"  Time budget: {t1:.0f}s")
        a_opt, obj = lp_fw_optimize(parent, time_limit=t1, verbose=True,
                                     use_all_constraints=True)
        try_seq(a_opt, "lpfw_full_parent")

    # === Phase 2: Diverse multi-start LP-FW ===
    print("\n=== Phase 2: Multi-start diverse LP-FW ===")
    n_base = len(best_seq) if best_seq is not None else len(parent)
    base = best_seq if best_seq is not None else parent

    inits = []
    # Interpolated parent
    inits.append(("interp", interpolate_seq(base, n_base)))
    # Flat spectrum
    inits.append(("flat_spec", flat_spectrum_init(n_base, seed=999)))
    # Binary
    inits.append(("binary", binary_init(n_base, seed=777, density=0.35)))
    # Two-level
    inits.append(("two_level", two_level_init(n_base, seed=555, ratio=4.0)))
    # Noisy parent
    inits.append(("noisy", project_simplex(base + rng.standard_normal(n_base) * 0.005)))
    # Reversed blend
    inits.append(("rev_blend", project_simplex(0.7 * base + 0.3 * base[::-1])))

    for name, f_init in inits:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.15, 150)
        print(f"  Multi-start [{name}], t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=False,
                                   use_all_constraints=True)
        try_seq(a_opt, f"multi_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [512, 640, 896, 1024]
        for tn in target_lengths:

            # Also try pure interpolation

    # === Phase 4: Peak equalization ===

                try_seq(a_opt2, "block_cd_2")

    # === Phase 6: Subgradient polishing ===
    print(f"\n=== Phase 6: SG Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 20:
            t_sg = min(rem * 0.2, 150)
            a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                                 lr=0.0008, mom=0.95,
                                 top_k=max(30, n_best // 4),
                                 temp=0.00002)
            try_seq(a_opt, "sg_1")

            rem = deadline - time.monotonic()
            if rem > 15:
                a_opt2, _ = sg_polish(a_opt, time_limit=min(rem * 0.2, 80),
                                      lr=0.0004, mom=0.96,
                                      top_k=max(20, n_best // 6),
                                      temp=0.00001)
                try_seq(a_opt2, "sg_2")

    # === Phase 7: Final LP-FW deep dive ===
    print(f"\n=== Phase 7: Final Deep LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 50:
            t_final = min(rem * 0.4, 300)
            print(f"  Final LP-FW, t={t_final:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final, verbose=True,
                                       use_all_constraints=True)
            try_seq(a_opt, "final_lpfw")

    # === Phase 8: Micro-perturbations ===
    print(f"\n=== Phase 8: Micro-perturbations (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for i in range(6):

            noise_scale = 0.0005 + 0.0005 * i

                                       use_all_constraints=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
