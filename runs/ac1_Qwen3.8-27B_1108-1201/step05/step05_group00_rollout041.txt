import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]. p has length 2n-1, a has length n."""
    return np.correlate(p, a, mode='valid')


def qp_solve_hankel(p, n, max_iters=500):
    """
    Solve min a^T M a s.t. sum(a)=1, a>=0 where M_{i,j} = p[i+j].
    Uses active set method with closed-form solution on the support.
    """
    active = list(range(n))
    
    for _ in range(max_iters):
        T = active
        k = len(T)
        if k == 0:
            return np.ones(n) / n
        if k == 1:
            a = np.zeros(n)
            a[T[0]] = 1.0
            return a
        
        T_arr = np.array(T)
        # Build M_T,T: (M_T,T)[l,m] = p[T[l] + T[m]]
        M_sub = p[T_arr[:, None] + T_arr[None, :]]
        
        # Solve M_sub @ a_T = -mu * 1, sum(a_T) = 1
        # => a_T = M_sub^{-1} @ 1 / (1^T M_sub^{-1} @ 1)
        ones = np.ones(k)
        try:
            sol = np.linalg.solve(M_sub, ones)
        except np.linalg.LinAlgError:
            sol, _, _, _ = np.linalg.lstsq(M_sub, ones, rcond=None)
        
        denom = sol.sum()
        if abs(denom) < 1e-15:
            a = np.zeros(n)
            a[T] = 1.0 / k
            return a
        
        a_T = sol / denom
        
        # Check positivity
        if a_T.min() > -1e-10:
            a_T = np.maximum(a_T, 0.0)
            a_T /= a_T.sum()
            
            a = np.zeros(n)
            a[T] = a_T
            
            # Check KKT for inactive
            Ma = hankel_mv(p, a)
            ma_T = Ma[T]
            inactive = set(range(n)) - set(T)
            
            add_idx = -1
            best_violation = 1e-10
            for i in inactive:
                viol = ma_T.min() - Ma[i]  # Should be >= 0 for KKT
                if viol > best_violation:
                    best_violation = viol
                    add_idx = i
            
            if add_idx >= 0:
                active.append(add_idx)
                continue
            
            return a
        else:
            # Remove most negative
            worst_pos = int(np.argmin(a_T))
            active.remove(T[worst_pos])
    
    # Fallback
    a = np.zeros(n)
    a[active] = 1.0 / len(active)
    return a


def game_iteration(a0, time_limit, T_init=0.001):
    """
    Game-theoretic primal-dual iteration.
    Primal: solve QP min a^T M(p) a s.t. a in simplex
    Dual: p = softmax(conv(a) / T)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        
        # Dual update: p = softmax(conv / T)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()
        
        # Primal update: solve QP
        a_qp = qp_solve_hankel(p, n)
        a = a_qp
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        
        # Adapt temperature
        if it % 100 == 0:
            T *= 0.95  # Slowly decrease
            T = max(T, 0.0001)
    
    return best_a, best_obj

    n = len(a)

    best_obj = fast_convolve(a).max()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]. p has length 2n-1, a has length n."""
    return np.correlate(p, a, mode='valid')

    for _ in range(max_iters):

        k = len(T)

        T_arr = np.where(a > support_thresh)[0].astype(np.int64)
        t = len(T_arr)

        if t == 0 or s == 0:
            break

        # Build initial guess
        a_T_init = a[T_arr].copy()
        # Initial lambda: proportional to peak values (or uniform)
        lam_init = np.full(s, 1.0 / s)
        # Initial mu: from the E2 equations
        # 2 * sum_j lam_j * a[S[j]-i] + mu = 0
        # mu = -2 * mean over i in T of sum_j lam_j * a[S[j]-i]
        Ma_approx = np.zeros(t)
        for j in range(s):
            shifted = S_arr[j] - T_arr
            valid = (shifted >= 0) & (shifted < n)
            Ma_approx += lam_init[j] * np.where(valid, a_full := np.zeros(n), 0.0)
        # Recompute properly
        a_full_temp = np.zeros(n)
        a_full_temp[T_arr] = a_T_init
        Ma_approx = np.zeros(t)
        for j in range(s):
            k = S_arr[j]
            shifted = k - T_arr
            valid = (shifted >= 0) & (shifted < n)
            vals = np.zeros(t)
            vals[valid] = a_full_temp[shifted[valid]]
            Ma_approx += lam_init[j] * vals
        mu_init = -2.0 * Ma_approx.mean()

        x0 = np.concatenate([a_T_init, lam_init, [mu_init, c_max]])

        # Solve with Newton
        max_newton = 30
        x = x0.copy()

        for n_it in range(max_newton):
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            F = kkt_residual(x, S_arr, T_arr, n, s, t)
            F_norm = np.max(np.abs(F))

            if F_norm < 1e-14:
                break

            J = kkt_jacobian(x, S_arr, T_arr, n, s, t)

            try:
                dx = np.linalg.solve(J, -F)
            except np.linalg.LinAlgError:
                dx, _, _, _ = np.linalg.lstsq(J, -F, rcond=None)

            # Line search
            alpha = 1.0
            for _ in range(20):
                x_trial = x + alpha * dx
                F_trial = kkt_residual(x_trial, S_arr, T_arr, n, s, t)
                F_trial_norm = np.max(np.abs(F_trial))
                if F_trial_norm < F_norm:
                    break
                alpha *= 0.5

            x = x + alpha * dx

        # Extract solution
        a_new_T = x[:t]
        lam_new = x[t:t+s]
        t_val_new = x[t+s+1]

        # Check feasibility
        a_full_new = np.zeros(n)
        a_full_new[T_arr] = np.maximum(a_new_T, 0.0)
        a_full_new = a_full_new / a_full_new.sum()

        conv_new = fast_convolve(a_full_new)
        obj_new = conv_new.max()

        # Check KKT consistency
        kkt_ok = True
        # Check: all peaks outside S are <= t_val
        if np.max(conv_new[~np.isin(np.arange(2*n-1), S_arr)]) > t_val_new + 1e-8:
            kkt_ok = False
        # Check: lambda >= 0
        if lam_new.min() < -1e-6:
            kkt_ok = False
        # Check: a >= 0
        if a_new_T.min() < -1e-6:
            kkt_ok = False

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_full_new.copy()

        if kkt_ok and F_norm < 1e-10:
            # Converged!
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_full_new.copy()
            break

        # If not converged, update a and adjust active set
        a = a_full_new
        # Broaden active set if needed
        conv_check = fast_convolve(a)
        cm = conv_check.max()
        S_new = np.where(conv_check >= cm * (1.0 - 1e-6))[0]
        if len(S_new) > 15:
            top_idx = np.argsort(conv_check[S_new])[::-1][:15]
            S_new = S_new[top_idx]
        if len(S_new) != len(S_arr) or not np.array_equal(np.sort(S_new), np.sort(S_arr)):
            # Active set changed, restart Newton with new set
            continue

        # If no improvement in a while, try a different active set
        if outer_it % 10 == 0:
            # Try with slightly different threshold
            conv_check = fast_convolve(a)
            cm = conv_check.max()
            S_new = np.where(conv_check >= cm * (1.0 - 1e-5))[0]
            if len(S_new) > 10:
                top_idx = np.argsort(conv_check[S_new])[::-1][:10]
                S_new = S_new[top_idx]
            if len(S_new) >= 1:
                S_arr = S_new.astype(np.int64)
                s = len(S_arr)

    return best_a, best_obj


def fw_lp_step(f, M, n):
    """Single FW step using LP oracle."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, max_iters=200):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        g = fw_lp_step(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.3
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

    return best_f, best_obj


def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.05):
    """Smoothed gradient descent with softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()

        grad = 2.0 * hankel_mv(p, a)
        candidate = project_simplex(a - lr * grad)
        val = fast_convolve(candidate).max()

        if val < best_obj - 1e-15:
            a = candidate
            best_obj = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                candidate2 = project_simplex(a - lr * factor * grad)
                val2 = fast_convolve(candidate2).max()
                if val2 < best_obj - 1e-15:
                    a = candidate2
                    best_obj = val2
                    best_a = a.copy()
                    no_improve = 0
                    break

        if no_improve > 30:
            T *= 1.05
            lr *= 0.95
        if no_improve > 100:
            T *= 0.7
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search:
    Phase 1: Quick FW-LP (~150s)
    Phase 2: Game iteration with exact QP (~450s)
    Phase 3: KKT Newton polish (~1100s)
    Phase 4: Final refinement with multiple active sets (~100s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    n = len(parent)
    print(f"Working with n={n}")

    # === Phase 1: Quick FW-LP ===
    print("\n=== Phase 1: Quick FW-LP ===")
    t_start = time.monotonic()
    t_alloc = min(120, deadline - time.monotonic() - 1400)
    if t_alloc > 10:
        a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc)
        try_seq(a_opt, "fw_lp")
        print(f"  Phase 1 done ({time.monotonic()-t_start:.0f}s), best={best_score:.8f}")

    # === Phase 2: Game iteration ===
    print("\n=== Phase 2: Game Iteration ===")
    t_start = time.monotonic()
    if best_seq is not None:
        t_alloc = min(400, deadline - time.monotonic() - 900)
        if t_alloc > 20:
            a_opt, _ = game_iteration(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "game_iter")
            print(f"  Game iteration done ({time.monotonic()-t_start:.0f}s), best={best_score:.8f}")

    # === Phase 3: KKT Newton ===
    print("\n=== Phase 3: KKT Newton Polish ===")
    t_start = time.monotonic()
    if best_seq is not None:
        t_alloc = min(900, deadline - time.monotonic() - 200)
        if t_alloc > 30:
            print(f"  Allocating {t_alloc:.0f}s for Newton")
            a_opt, _ = kkt_newton_solve(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "newton")
            print(f"  Newton done ({time.monotonic()-t_start:.0f}s), best={best_score:.8f}")

    # === Phase 4: Additional refinement ===
    print("\n=== Phase 4: Additional Refinement ===")
    t_start = time.monotonic()
    if best_seq is not None:
        # Try game iteration again from the Newton result
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            t_alloc = min(remaining, 200)
            a_opt, _ = game_iteration(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "game2")

        # Try another Newton pass
        remaining = deadline - time.monotonic() - 50
        if remaining > 30:
            t_alloc = min(remaining, 200)
            a_opt, _ = kkt_newton_solve(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "newton2")

        # Quick smoothed gradient for fine-tuning
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=min(remaining, 50),
                                             T_init=0.0005, lr_init=0.01)
            try_seq(a_opt, "smooth_fine")

        # Perturbation + Newton
        remaining = deadline - time.monotonic() - 20
        if remaining > 30:
            for scale in [0.001, 0.005]:
                remaining = deadline - time.monotonic() - 10
                if remaining < 20:
                    break
                noise = rng.standard_normal(n) * scale
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = kkt_newton_solve(f_pert, time_limit=min(remaining, 60))
                try_seq(a_opt, f"pert_newton_{scale}")

    print(f"  Phase 4 done ({time.monotonic()-t_start:.0f}s)")

    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
