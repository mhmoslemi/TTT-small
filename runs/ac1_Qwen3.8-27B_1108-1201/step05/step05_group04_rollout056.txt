"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using adaptive LSE gradient descent
with Adam optimizer, two-point local search, and multi-n exploration.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_peak_gradient(f, conv, k_peak):
    """Compute gradient of conv[k_peak] w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    lo = max(0, k_peak - n + 1)
    hi = min(n - 1, k_peak)
    if lo <= hi:
        # grad[i] = 2 * f[k_peak - i] for i in [lo, hi]
        indices = np.arange(lo, hi + 1)
        grad[indices] = 2.0 * f[k_peak - indices]
    return grad

def compute_lse_gradient(f, conv, beta, top_k=None):
    """Compute gradient of LSE(beta * conv) w.r.t. f.
    Uses top_k peaks for efficiency."""
    n = len(f)
    conv_len = len(conv)
    
    # Compute weights
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    if top_k is not None and top_k < conv_len:
        # Keep only top_k weights
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_top = w[idx]
        w_top = w_top / w_top.sum()
        # Rebuild sparse weight vector
        w_sparse = np.zeros(conv_len)
        w_sparse[idx] = w_top
    else:
        w_sparse = w
    
    # Compute gradient by summing over peaks
    grad = np.zeros(n)
    # Only process non-zero weights
    nonzero = w_sparse > 1e-15
    peak_indices = np.nonzero(nonzero)[0]
    
    for k in peak_indices:
        wk = w_sparse[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            grad[indices] += wk * 2.0 * f[k - indices]
    
    return grad

def lse_adam(f0, time_limit, beta_start=20.0, beta_end=5000.0, top_k=200):
    """LSE gradient descent with Adam optimizer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 1.0 / np.sqrt(n)  # Adam learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t_step = 0
    t0 = time.monotonic()
    no_improve = 0
    patience = 100
    
    while time.monotonic() - t0 < time_limit:
        t_step += 1
        conv = fast_convolve(f)
        grad = compute_lse_gradient(f, conv, beta, top_k=top_k)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                # Reset and try different parameters
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                beta = min(beta * 2.0, beta_end)
                lr = lr * 0.5
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.002
        
        # Periodic learning rate decay
        if t_step % 500 == 0:
            lr *= 0.95
    
    return best_f, best_obj

def two_point_search(f0, time_limit, n_trials=50000):
    """Local search: transfer mass between two points."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    best_score = float('inf')
    
    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    trials = 0
    
    while time.monotonic() - t0 < time_limit and trials < n_trials:
        # Pick two random indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        
        # Try transferring mass from j to i (or vice versa)
        # Amount: proportional to min(f[i], f[j])
        eps = 0.01 * min(f[i], f[j]) + 0.001 / n
        
        for sign in [1, -1]:
            f_trial = f.copy()
            f_trial[i] += sign * eps
            f_trial[j] -= sign * eps
            if f_trial[i] < 0 or f_trial[j] < 0:
                continue
            f_trial = np.maximum(f_trial, 0)
            f_trial /= f_trial.sum()
            
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_f = f_trial.copy()
                f = f_trial
                trials += 1
        
        trials += 1
        if trials % 10000 == 0:
            print(f"    2-point: trial={trials}, obj={best_obj:.10f}, t={time.monotonic()-t0:.1f}s")
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, n_sweeps=100):
    """Coordinate descent: optimize one coordinate at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit and sweep < n_sweeps:
        sweep += 1
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 5:
                break
            
            # Try adjusting f[i]
            # The contribution of f[i] to conv[k] is f[i]*f[k-i] for valid k-i
            # We want to find f[i] that minimizes max(conv)
            
            # Binary search for optimal f[i]
            lo = 0.0
            hi = min(2.0 * f[i] + 1.0/n, 1.0)  # reasonable range
            
            best_fi = f[i]
            best_obj_i = best_obj
            
            for _ in range(8):
                mid = (lo + hi) / 2.0
                f_trial = f.copy()
                f_trial[i] = mid
                f_trial /= f_trial.sum()
                conv_trial = fast_convolve(f_trial)
                obj_trial = conv_trial.max()
                if obj_trial < best_obj_i:
                    best_obj_i = obj_trial
                    best_fi = mid
                    hi = mid
                else:
                    lo = mid
            
            if best_obj_i < best_obj - 1e-14:
                f[i] = best_fi
                f /= f.sum()
                conv = fast_convolve(f)
                best_obj = conv.max()
                best_f = f.copy()
    
    return best_f, best_obj

def project_simplex(v):

    n = len(v)

    """Interpolate normalized sequence to new length."""
    x_old = np.linspace(0, 1, len(f))

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        try_sequence(parent, best_tracker)

    # Phase 1: LSE-Adam
    print("\n=== Phase 1: LSE-Adam ===")
    t1 = min(350, deadline - time.monotonic())
    if t1 > 30:
        f1, _ = lse_adam(best_tracker[1], time_limit=t1, beta_start=10.0, beta_end=3000.0, top_k=200)
        try_sequence(f1, best_tracker)

    # Phase 2: Two-point SA
    print("\n=== Phase 2: Two-Point SA ===")
    t2 = min(350, deadline - time.monotonic())
    if t2 > 30:
        f2, _ = two_point_sa(best_tracker[1], time_limit=t2, temp_start=0.004, temp_end=0.0001, seed=seed)
        try_sequence(f2, best_tracker)

    # Phase 3: Peak-aligned gradient descent
    print("\n=== Phase 3: Peak-Targeted Gradient Descent ===")
    t3 = min(250, deadline - time.monotonic())
    if t3 > 20:
        f3, _ = peak_grad_descent(best_tracker[1], time_limit=t3, n_peaks=6)
        try_sequence(f3, best_tracker)

    # Phase 4: LSE-Adam with higher beta
    print("\n=== Phase 4: LSE-Adam High Beta ===")
    t4 = min(250, deadline - time.monotonic())
    if t4 > 20:
        f4, _ = lse_adam(best_tracker[1], time_limit=t4, beta_start=100.0, beta_end=8000.0, top_k=100)
        try_sequence(f4, best_tracker)

    # Phase 5: Two-point SA again (lower temperature)
    print("\n=== Phase 5: Two-Point SA (Cold) ===")
    t5 = min(250, deadline - time.monotonic())
    if t5 > 20:
        f5, _ = two_point_sa(best_tracker[1], time_limit=t5, temp_start=0.001, temp_end=0.00002, seed=seed + 1000)
        try_sequence(f5, best_tracker)

    # Phase 6: Try nearby lengths
    print("\n=== Phase 6: Nearby Lengths ===")
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        offsets = [64, -64, 128, -128, 32, -32, 96, -96]
        for off in offsets:
            n_try = n_best + off
            if n_try < 128 or n_try > 1500:
                continue
            t_rem = deadline - time.monotonic() - 20
            if t_rem < 40:
                break
            t_alloc = min(80, t_rem * 0.35)
            if t_alloc < 20:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_adam(f_init, time_limit=t_alloc * 0.6, beta_start=20.0, beta_end=2000.0, top_k=150)
            try_sequence(f_opt, best_tracker)
            t_rem2 = deadline - time.monotonic() - 20
            t_sa = min(40, t_rem2 * 0.4)
            if t_sa > 10:
                f_opt2, _ = two_point_sa(f_opt, time_limit=t_sa, temp_start=0.002, temp_end=0.00005, seed=seed + n_try)
                try_sequence(f_opt2, best_tracker)

    # Phase 7: Final polish
    print("\n=== Phase 7: Final Polish ===")
    if best_tracker[1] is not None:
        t7 = min(120, deadline - time.monotonic() - 10)
        if t7 > 15:
            f7, _ = two_point_sa(best_tracker[1], time_limit=t7, temp_start=0.0005, temp_end=0.00001, seed=seed + 9999)
            try_sequence(f7, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Fallback")
    return [float(x) for x in (np.ones(768) / 768)]
