import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, peak_positions):
    """
    Compute the gradient of max(b) w.r.t. a, where b = conv(a,a).
    For a peak at position pos, the gradient contribution is:
    grad[m] += 2 * a[pos-m] for 0 <= pos-m < n
    """
    n = len(a)
    grad = np.zeros(n)
    for pos in peak_positions:
        pos = int(pos)
        m_start = max(0, pos - n + 1)
        m_end = min(n, pos + 1)
        # a[pos-m] for m in [m_start, m_end)
        # indices: pos-m_start, pos-m_start-1, ..., pos-(m_end-1)
        idx = np.arange(pos - (m_end - 1), pos - m_start + 1)
        grad[m_start:m_end] += 2.0 * a[idx]
    return grad

    """
    Peak-flattening optimization using softmax-weighted peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    v = np.zeros(n)
    no_improve = 0
    it = 0
    current_T = T
    current_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Softmax weights on peaks
        scaled = conv / current_T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-15:
            break
        p = p / p_sum
        
        # Only use top-k peaks for efficiency
        if top_k < len(p):
            top_idx = np.argpartition(p, -top_k)[-top_k:]
            p_masked = np.zeros_like(p)
            p_masked[top_idx] = p[top_idx]
            p_masked /= p_masked.sum()
            p = p_masked
        
        # Compute gradient using all non-negligible peaks
        significant = p > 1e-8
        peak_positions = np.where(significant)[0]
        
        grad = np.zeros(n)
        for pos in peak_positions:
            w = p[pos]
            pos_int = int(pos)
            m_start = max(0, pos_int - n + 1)
            m_end = min(n, pos_int + 1)
            idx = np.arange(pos_int - (m_end - 1), pos_int - m_start + 1)
            grad[m_start:m_end] += w * 2.0 * a[idx]
        
        # Momentum update
        v = mom * v - current_lr * grad
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt hyperparameters
        if no_improve > 30:
            current_T *= 1.03
            current_lr *= 0.98
        if no_improve > 80:
            current_T *= 0.8
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            current_T = T
            current_lr = lr
            no_improve = 0
        
        if it % 100 == 0 and current_lr < lr * 0.1:
            current_lr = lr * 0.5
            current_T = T
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Anti-symmetric and structured initializations to minimize central peak
2. Softmax-weighted peak-flattening gradient descent with adaptive temperature
3. Variance-aware gradient (penalizes spread of peaks, not just the max)
4. Multi-n exploration with interpolation
5. LP-based polishing and coordinate descent for final refinement
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weighted_grad(a, weights, positions):
    """
    Compute weighted gradient of sum_i w_i * b[pos_i] w.r.t. a.
    db[pos]/da[m] = 2*a[pos-m] if 0 <= pos-m < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    for i in range(len(positions)):
        w = weights[i]
        if w < 1e-15:
            continue
        pos = int(positions[i])
        m_start = max(0, pos - n + 1)
        m_end = min(n, pos + 1)
        if m_start >= m_end:
            continue
        # a[pos-m] for m in [m_start, m_end)
        # m=m_start -> a[pos-m_start], m=m_end-1 -> a[pos-(m_end-1)]
        idx = np.arange(pos - (m_end - 1), pos - m_start + 1)
        grad[m_start:m_end] += w * 2.0 * a[idx]
    return grad


def peak_flatten_optimize(a0, time_limit, T=0.001, lr=0.05, top_k=15, mom=0.92, var_weight=0.1):
    """
    Peak-flattening optimization with variance-aware gradient.
    Minimizes max(b) + var_weight * std(b) using softmax-weighted subgradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()

    t0 = time.monotonic()
    v = np.zeros(n)
    no_improve = 0
    it = 0
    current_T = T
    current_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        mean_b = 1.0 / (2 * n - 1)

        # Softmax weights with variance-aware adjustment
        scaled = (conv - mean_b) / current_T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            break
        p = p / p_sum

        # Focus on top-k
        if top_k < len(p):
            top_idx = np.argpartition(p, -top_k)[-top_k:]
            p_foc = np.zeros_like(p)
            p_foc[top_idx] = p[top_idx]
            p_foc /= p_foc.sum()
            p = p_foc

        # Compute gradient
        significant = p > 1e-10
        peak_positions = np.where(significant)[0]
        peak_weights = p[peak_positions]

        grad = compute_weighted_grad(a, peak_weights, peak_positions)

        # Add variance penalty gradient (smooth component)
        if var_weight > 0:
            excess = conv - mean_b
            excess /= (np.std(conv) + 1e-15)
            sig2 = excess > 0.1
            if sig2.any():
                var_positions = np.where(sig2)[0]
                var_weights = excess[var_positions]
                var_weights /= np.abs(var_weights).sum() + 1e-15
                var_weights *= var_weight
                grad += compute_weighted_grad(a, var_weights, var_positions)

        # Adaptive step size with line search
        best_candidate = a
        best_val = best_obj
        for trial_mult in [1.0, 0.5, 0.25, 0.1]:
            trial_lr = current_lr * trial_mult
            v_new = mom * v - trial_lr * grad
            candidate = project_simplex(a + v_new)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                v = v_new
                break

        a = best_candidate

        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt hyperparameters
        if no_improve > 20:
            current_T *= 1.02
        if no_improve > 50:
            current_lr *= 0.95
        if no_improve > 100:
            current_T *= 0.9
        if no_improve > 250:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - 250) / 400.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            current_T = T
            current_lr = lr
            no_improve = 0

    return best_a, best_obj


def anti_sym_init(n, rng, variant=0):
    """Generate anti-symmetric initializations that reduce b[n-1]."""
    x = np.arange(n, dtype=np.float64)
    t = x / max(n - 1, 1)

    if variant == 0:
        # Linear ramp (strongly anti-symmetric)
        a = 0.5 + t
    elif variant == 1:
        # Quadratic anti-symmetric
        a = 1.0 + 1.5 * (t - 0.5)
    elif variant == 2:
        # Moderate linear
        a = 1.0 + 0.8 * (t - 0.5)
    elif variant == 3:
        # Exponential with small tilt
        a = np.exp(0.5 * (t - 0.5))
    elif variant == 4:
        # S-curve anti-symmetric
        a = 1.0 + 0.6 * np.sin(np.pi * t)
    elif variant == 5:
        # Random smooth
        a = 0.5 + 0.5 * np.sin(2 * np.pi * rng.uniform(0.3, 0.8) * t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.1)
    elif variant == 6:
        # Two-hump anti-symmetric
        a = np.exp(-2.0 * (t - 0.3)**2) + 0.5 * np.exp(-2.0 * (t - 0.7)**2)
    else:
        a = 1.0 + 0.3 * rng.standard_normal(n)
        a = np.maximum(a, 0.1)

    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def lp_polish(f0, time_limit, max_iters=50):
    """LP-based Frank-Wolfe step for final polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1, time_limit - (time.monotonic() - t0) - 5)})
            if not res.success:
                break
            g = np.maximum(res.x, 0.0)
        except Exception:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            break

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()

    return best_f, best_obj

def generate_initializations(n, parent, rng):
    """Generate diverse initializations with emphasis on anti-symmetric patterns."""
    inits = []

    # Anti-symmetric variants
    for v in range(7):
        inits.append(anti_sym_init(n, rng, variant=v))

    # Standard initializations
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    if parent is not None and len(parent) > 1:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits
