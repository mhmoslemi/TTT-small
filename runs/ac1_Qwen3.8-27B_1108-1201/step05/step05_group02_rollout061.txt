"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_smooth_grad(conv, T, n):
    """Compute gradient of T*log(sum(exp(conv/T))) w.r.t. a (assuming sum(a)=1)."""
    max_c = conv.max()
    log_w = conv / T - max_c / T
    w = np.exp(log_w)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # = 2 * (w * a_rev)[i + n - 1]
    a_rev = np.zeros(2 * n)  # we'll use the pieces we need
    # Actually we need the a passed in
    return w  # return w, we'll compute grad separately


def smooth_grad_full(conv, T, a, n):
    """Full gradient computation using FFT."""
    max_c = conv.max()
    log_w = conv / T - max_c / T
    w = np.exp(log_w)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    a_rev = a[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Ar = np.fft.rfft(a_rev, fft_size)
    conv_result = np.fft.irfft(W * Ar, fft_size)
    g = conv_result[n-1 : n-1+n] * 2.0
    return g


def mirror_descent_optimize(f0, time_limit, T_start=0.001, T_end=0.00001, 
                            eta_start=0.5, verbose=False):
    """Mirror descent (multiplicative weights) with log-sum-exp annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    total_iters = 0
    no_improve = 0
    
    # Annealing parameters
    n_expected = 5000  # expected total iterations
    
    while time.monotonic() - t0 < time_limit:
        total_iters += 1
        
        # Current temperature
        frac = min(1.0, total_iters / n_expected)
        T = T_start * (T_end / T_start) ** frac
        eta = eta_start * (1.0 - 0.3 * frac)
        
        # Compute smooth gradient
        g = smooth_grad_full(conv, T, f, n)
        
        # Mirror descent update (multiplicative weights)
        # f_new[i] = f[i] * exp(-eta * g[i]) / Z
        log_updates = -eta * g
        log_updates -= log_updates.max()  # for numerical stability
        updates = np.exp(log_updates)
        f_new = f * updates
        Z = f_new.sum()
        if Z < 1e-300:
            f_new = f.copy()
        else:
            f_new /= Z
        
        # Line search: try several step sizes
        best_eta = eta
        best_val = fast_convolve(f_new).max()
        
        # Try smaller steps
        for eta_try in [eta * 0.5, eta * 0.25, eta * 2.0]:
            log_updates_t = -eta_try * g
            log_updates_t -= log_updates_t.max()
            f_try = f * np.exp(log_updates_t)
            zt = f_try.sum()
            if zt < 1e-300:
                continue
            f_try /= zt
            val = fast_convolve(f_try).max()
            if val < best_val:
                best_val = val
                best_eta = eta_try
        
        if best_eta > 0:
            log_updates_best = -best_eta * g
            log_updates_best -= log_updates_best.max()
            f = f * np.exp(log_updates_best)
            Z = f.sum()
            if Z > 1e-300:
                f /= Z
            
            conv = fast_convolve(f)
            cur_obj = conv.max()
            
            if cur_obj < best_obj - 1e-15:
                best_obj = cur_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                
            if verbose and total_iters % 200 == 0:
                print(f"    MD iter {total_iters}: T={T:.6f}, obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        # Periodic restart
        if no_improve > 500:
            rng = np.random.default_rng(total_iters * 31 + 7)
            noise = rng.standard_normal(n) * 0.002
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            no_improve = 0
    
    return best_f, best_obj


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def sg_optimize(f0, time_limit, lr=0.003, momentum=0.94, top_k=50, 
                T_start=0.0001, T_end=0.00001, verbose=False):
    """Adaptive subgradient with annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    n_expected = 8000
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = min(1.0, it / n_expected)
        T = T_start * (T_end / T_start) ** frac
        cur_lr = lr * (1.0 - 0.5 * frac)
        
        # Compute softmax weights over peaks
        max_c = conv.max()
        log_w = conv / T - max_c / T
        w = np.exp(log_w)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient using FFT
        a_rev = f[::-1]
        size = len(w) + n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        Ar = np.fft.rfft(a_rev, fft_size)
        conv_result = np.fft.irfft(W * Ar, fft_size)
        sg = conv_result[n-1 : n-1+n] * 2.0
        
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 104729 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            no_improve = 0
    
    return best_f, best_obj


def joint_peak_cd(f0, time_limit, verbose=False):
    """Coordinate descent that jointly reduces top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        # Get top peaks
        n_peaks = min(5, len(conv))
        peak_positions = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_positions = peak_positions[conv[peak_positions] > 0.95 * best_obj]
        
        if len(peak_positions) == 0:
            peak_positions = np.array([conv.argmax()])
        
        # For each peak, find the coordinate that most contributes
        for pk in peak_positions:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            
            # Find coordinate with highest contribution to this peak
            contributions = np.array([f[j] * f[pk - j] for j in range(js, je + 1)])
            max_idx = js + int(np.argmax(contributions))
            
            # Try reducing this coordinate
            old_val = f[max_idx]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[max_idx] = new_val
                mask = np.ones(n, dtype=bool)
                mask[max_idx] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[max_idx] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[max_idx] = False
                    f[mask] *= scale
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
                    conv = fast_convolve(f)
        
        # Also try random coordinates
        if not improved:
            rng = np.random.default_rng(passes * 54321)
            for _ in range(min(30, n // 5)):
                j = int(rng.integers(0, n))
                old_val = f[j]
                if old_val < 1e-10:
                    continue
                for mult in [0.4, 0.6, 1.5, 1.8]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial /= f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        improved = True
                        break
                if improved:
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break
    
    return best_f, best_obj

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_softmax_weights(conv, T):
    max_c = conv.max()
    log_w = conv / T - max_c / T
    w = np.exp(log_w)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    return w


def smooth_grad_fft(conv, T, f, n):
    w = compute_softmax_weights(conv, T)
    a_rev = f[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Ar = np.fft.rfft(a_rev, fft_size)
    conv_result = np.fft.irfft(W * Ar, fft_size)
    return conv_result[n - 1:n - 1 + n] * 2.0

                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[max_idx] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[max_idx] = False
                    f[mask] *= scale
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
                    conv = fast_convolve(f)

        if not improved:
            rng = np.random.default_rng(passes * 54321)
            for _ in range(min(20, n // 8)):
                j = int(rng.integers(0, n))
                old_val = f[j]
                if old_val < 1e-10:
                    continue
                for mult in [0.4, 0.6, 1.5, 1.8]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial /= f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        improved = True
                        break
                if improved:
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining mirror descent, LP-FW, subgradient,
    evolutionary search, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Mirror Descent on parent ===
    print("\n=== Phase 1: Mirror Descent (parent) ===")
    rem = deadline - time.monotonic()
    t1 = min(350, rem * 0.22)
    if t1 > 60:
        print(f"  n={len(parent)}, t={t1:.0f}s")
        a_opt, _ = mirror_descent_optimize(parent, time_limit=t1,
                                           T_start=0.0005, T_end=0.00001,
                                           eta_start=0.3, verbose=True)
        a_opt = np.maximum(a_opt, 0.0)
        if a_opt.sum() > 1e-10:
            a_opt = a_opt / a_opt.sum()
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [md]: {sc:.8f}")

    # === Phase 2: LP-FW ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(250, rem * 0.18)
    if t2 > 40 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        a_opt = np.maximum(a_opt, 0.0)
        if a_opt.sum() > 1e-10:
            a_opt = a_opt / a_opt.sum()
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [lpfw]: {sc:.8f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    candidate_lengths = set()
    for delta in [-128, -64, -32, 32, 64, 128, 256]:
        ln = len(base) + delta
        if 256 <= ln <= 2048:
            candidate_lengths.add(ln)
    for ln in [512, 768, 1024, 1280, 1536]:
        if 256 <= ln <= 2048:
            candidate_lengths.add(ln)

    for target_n in sorted(candidate_lengths)[:5]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.2, 120)
        if t_alloc < 30:
            break
        print(f"  MD n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent_optimize(f_interp, time_limit=t_alloc,
                                           T_start=0.0003, T_end=0.00001,
                                           eta_start=0.2)
        a_opt = np.maximum(a_opt, 0.0)
        if a_opt.sum() > 1e-10:
            a_opt = a_opt / a_opt.sum()
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [md_n{target_n}]: {sc:.8f}")

        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc2 = min(rem * 0.15, 60)
        if t_alloc2 > 20:
            a_opt2, _ = lp_fw_optimize(a_opt, time_limit=t_alloc2)
            a_opt2 = np.maximum(a_opt2, 0.0)
            if a_opt2.sum() > 1e-10:
                a_opt2 = a_opt2 / a_opt2.sum()
                sc = float(evaluate_sequence(a_opt2.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_opt2.copy()
                    print(f"  NEW BEST [lpfw_n{target_n}]: {sc:.8f}")

    # === Phase 4: Evolutionary Search ===
    print(f"\n=== Phase 4: Evolutionary (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(300, rem * 0.25)
    if t4 > 40 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t4:.0f}s")
        a_opt, _ = evolutionary_search(best_seq, time_limit=t4, pop_size=14, verbose=True)
        a_opt = np.maximum(a_opt, 0.0)
        if a_opt.sum() > 1e-10:
            a_opt = a_opt / a_opt.sum()
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [evo]: {sc:.8f}")

    # === Phase 5: Subgradient refinement ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(200, rem * 0.2)
    if t5 > 30 and best_seq is not None:
        n_best = len(best_seq)
        print(f"  n={n_best}, t={t5:.0f}s")
        a_opt, _ = sg_optimize(best_seq, time_limit=t5,
                               lr=0.0015, momentum=0.95,
                               T_start=0.00004, T_end=0.000004, verbose=True)
        a_opt = np.maximum(a_opt, 0.0)
        if a_opt.sum() > 1e-10:
            a_opt = a_opt / a_opt.sum()
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [sg]: {sc:.8f}")

    # === Phase 6: Joint Peak CD ===
    print(f"\n=== Phase 6: Joint CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(150, rem * 0.2)
    if t6 > 25 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t6:.0f}s")
        a_opt, _ = joint_peak_cd(best_seq, time_limit=t6)
        a_opt = np.maximum(a_opt, 0.0)
        if a_opt.sum() > 1e-10:
            a_opt = a_opt / a_opt.sum()
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [cd]: {sc:.8f}")

    # === Phase 7: Final polish with perturbations ===
    print(f"\n=== Phase 7: Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick MD
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=min(rem, 60),
                                               T_start=0.00005, T_end=0.00001,
                                               eta_start=0.08)
            a_opt = np.maximum(a_opt, 0.0)
            if a_opt.sum() > 1e-10:
                a_opt = a_opt / a_opt.sum()
                sc = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [md_polish]: {sc:.8f}")

        # Quick LP-FW
        rem = deadline - time.monotonic() - 25
        if rem > 20:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 50))
            a_opt = np.maximum(a_opt, 0.0)
            if a_opt.sum() > 1e-10:
                a_opt = a_opt / a_opt.sum()
                sc = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [lpfw_polish]: {sc:.8f}")

        # Perturbations with quick optimization
        for p in range(10):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise_scale = 0.0003 + 0.0002 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = sg_optimize(f_pert, time_limit=min(rem, 15),
                                   lr=0.0004, momentum=0.96,
                                   T_start=0.00001, T_end=0.000003)
            a_opt = np.maximum(a_opt, 0.0)
            if a_opt.sum() > 1e-10:
                a_opt = a_opt / a_opt.sum()
                sc = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"  NEW BEST [perturb_{p}]: {sc:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
