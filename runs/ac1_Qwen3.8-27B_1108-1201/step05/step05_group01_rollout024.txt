"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Smooth log-sum-exp subgradient with momentum (fast gradient-based optimization)
2. LP-based Frank-Wolfe for direction correction when gradient stalls
3. Multi-length exploration with structured initializations
4. Golden-section coordinate descent for final polish
Key innovation: FFT-accelerated smooth gradient that targets peaks with adaptive
temperature annealing, combined with periodic LP-FW corrections.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cosine_init(n, freq, amp, seed=0):
    """Cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def gaussian_init(n, width_factor=1.0, seed=0):
    """Gaussian-window initialization."""
    j = np.arange(n)
    center = n / 2.0
    width = n / width_factor * 0.5
    a = np.exp(-0.5 * ((j - center) / width) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def raised_cosine_init(n):
    """Raised cosine window initialization."""
    j = np.arange(n)
    a = 0.5 * (1 + np.cos(2.0 * np.pi * j / n))
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_to(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_smooth_grad(a, w, sigma):
    """
    Compute gradient of log-sum-exp((conv(a,a)-t)/sigma) w.r.t. a.
    grad[m] = (2/sigma) * sum_{j=0}^{n-1} w[m+j] * a[j]
    This is the valid cross-correlation of w (len 2n-1) and a (len n).
    """
    n = len(a)
    # Use FFT for efficiency when n is large
    corr_full_len = (2 * n - 1) + n - 1  # full correlation length
    fft_size = 1 << (corr_full_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Cross-correlation: R[m] = sum_j w[m+j] * a[j]
    # This equals ifft(conj(W) * A) shifted
    corr_full = np.fft.irfft(np.conj(W) * A, fft_size)[:corr_full_len]
    # Valid part: R[m] for m=0..n-1
    # Full correlation index offset: the 'valid' part starts at index n-1
    R = corr_full[n - 1: n - 1 + n]
    return (2.0 / sigma) * R


def smooth_momentum_opt(a0, time_limit, sigma_init=0.05, lr_init=0.01,
                        mom_init=0.85, sigma_decay=0.999, lr_decay=0.9998,
                        verbose=False, label=""):
    """
    Smooth log-sum-exp subgradient with momentum.
    Minimizes max(conv(a,a)) approximately via smooth approximation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_autoconv(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()

    sigma = sigma_init * M
    lr = lr_init
    mom = mom_init
    v = np.zeros(n)

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    patience = 300

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute soft weights over convolution peaks
        scaled = (conv - M) / sigma
        scaled = np.maximum(scaled, -30.0)  # clip for numerical stability
        exp_s = np.exp(scaled)
        w = exp_s / exp_s.sum()

        # Compute gradient
        grad = compute_smooth_grad(a, w, sigma)

        # Momentum update
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        # Evaluate
        conv = fast_autoconv(a_new)
        cur_M = conv.max()

        if cur_M < best_obj - 1e-14:
            best_obj = cur_M
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"  SM[{label}] iter {it}: obj={best_obj:.10f}, n={n}")
        else:
            no_improve += 1

        a = a_new

        # Adaptive parameters
        sigma *= sigma_decay
        lr *= lr_decay
        mom = min(0.97, mom * 1.0001)

        # If stuck, perturb and/or increase sigma
        if no_improve > patience:
            rng = np.random.default_rng(it * 7 + 3)
            noise_scale = 0.002 * min(no_improve // patience, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            conv = fast_autoconv(a)
            M = conv.max()
            sigma = 0.05 * M
            lr = lr_init * 0.5
            v = np.zeros(n)
            no_improve = 0
            mom = mom_init

        # Update M tracking
        if cur_M < M:
            M = cur_M

    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                           'primal_feasibility_tolerance': 1e-9,
                           'dual_objective_limit': 1e50, 'primal_objective_limit': 1e50})
    if res.success:
        return res.x
    return None

    M = conv.max()
    best_obj = M

    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        g = solve_lp_direction(a, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 6.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - a)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        fw_alpha = 2.0 / (it + 2.0)
        alphas.append(max(0.01, min(1.0, fw_alpha)))
        alphas = sorted(set(alphas))

        for alpha in alphas:
            at = a + alpha * (g - a)
            at = np.maximum(at, 0.0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_autoconv(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Binary refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                at = a + mid * (g - a)
                at = np.maximum(at, 0.0)
                st = at.sum()
                if st < 1e-15:
                    break
                at = at / st
                val = fast_autoconv(at).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            a_new = a + best_alpha * (g - a)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            conv = fast_autoconv(a_new)
            M = conv.max()
            m_exp = 1.0
            a = a_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"  LPFW[{label}] iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 4.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 23 + 7)
            scale = 0.003 + 0.002 * min(no_improve // 50, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            conv = fast_autoconv(a)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_a, best_obj


def hybrid_optimize(a0, time_limit, verbose=False, label=""):
    """
    Hybrid: alternate between smooth gradient (fast) and LP-FW (direction correction).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_autoconv(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    lp_counter = 0
    lp_interval = 50  # do LP every 50 gradient steps

    # Gradient state
    sigma = 0.05 * M
    lr = 0.005
    mom = 0.85
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it_step = lp_counter  # rough step count
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        # Decide: gradient step or LP-FW step
        do_lp = (lp_counter % lp_interval == 0) and (rem > 2.0)

        if do_lp:
            # LP-FW step
            g = solve_lp_direction(a, M * 1.02, n)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                    # Quick line search
                    best_alpha = 0.3
                    best_val = fast_autoconv(a).max()
                    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7]:
                        at = np.maximum(a + alpha * (g - a), 0.0)
                        at = at / at.sum()
                        val = fast_autoconv(at).max()
                        if val < best_val:
                            best_val = val
                            best_alpha = alpha
                    a_new = np.maximum(a + best_alpha * (g - a), 0.0)
                    a_new = a_new / a_new.sum()
                    conv = fast_autoconv(a_new)
                    cur_M = conv.max()
                    if cur_M < best_obj - 1e-14:
                        best_obj = cur_M
                        best_a = a_new.copy()
                        M = cur_M
                        no_improve = 0
                        if verbose:
                            print(f"  HYB[{label}] LP step: obj={best_obj:.10f}")
                    a = a_new
                    v = np.zeros(n)  # reset momentum after LP
                else:
                    no_improve += 1
            else:
                no_improve += 1
            lp_counter += 1
        else:
            # Gradient step
            scaled = (conv - M) / sigma
            scaled = np.maximum(scaled, -30.0)
            exp_s = np.exp(scaled)
            w = exp_s / exp_s.sum()

            grad = compute_smooth_grad(a, w, sigma)
            v = mom * v - lr * grad
            a_new = project_simplex(a + v)
            conv = fast_autoconv(a_new)
            cur_M = conv.max()

            if cur_M < best_obj - 1e-14:
                best_obj = cur_M
                best_a = a_new.copy()
                M = cur_M
                no_improve = 0
            else:
                no_improve += 1

            a = a_new
            sigma *= 0.9995
            lr *= 0.9998
            mom = min(0.96, mom * 1.0002)

        # Stuck detection
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            conv = fast_autoconv(a)
            M = conv.max()
            sigma = 0.05 * M
            lr = 0.005
            mom = 0.85
            v = np.zeros(n)
            no_improve = 0
            lp_counter = 0  # reset LP interval

    return best_a, best_obj


def golden_cd(a0, time_limit):
    """Golden-section coordinate descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_autoconv(a).max()
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        if no_pass >= 3:
            break
        pass_imp = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-13:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            lo = 0.0
            hi = min(0.999, 2.5 * old_val)
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            def eval_golden(x):
                ft = a.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return 1e10
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                return fast_autoconv(ft).max()

            f1 = eval_golden(c1)
            f2 = eval_golden(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_golden(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_golden(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = fast_autoconv(a).max()

            if opt_val < cur_val - 1e-13:
                ft = a.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                ft = ft / ft.sum()
                a = ft
                pass_imp = True
                new_obj = fast_autoconv(a).max()
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()

        if pass_imp:
            no_pass = 0
        else:
            no_pass += 1

    return best_a, best_obj


def evaluate_score(a):
    """Score a sequence using evaluate_sequence."""
    a_list = [float(x) for x in np.maximum(a, 0.0)]
    return float(evaluate_sequence(a_list))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP-FW on parent for quick baseline improvement
    Phase 2: Smooth momentum optimization on parent
    Phase 3: Multi-length exploration with diverse initializations
    Phase 4: Hybrid optimization on best candidates
    Phase 5: Coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence([float(x) for x in a]))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence([float(x) for x in parent]))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence([float(x) for x in parent]))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n_parent}) ===")
    t1 = min(100, deadline - time.monotonic() - 800)
    if t1 > 10:
        print(f"  Allocating {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True, label="p1")
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Smooth momentum on best ===
    print(f"\n=== Phase 2: Smooth momentum (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 700
    if rem > 20:
        t2 = min(rem * 0.4, 180)
        print(f"  Allocating {t2:.0f}s, n={len(base)}")
        a_opt, _ = smooth_momentum_opt(base, time_limit=t2,
                                       sigma_init=0.03, lr_init=0.008,
                                       mom_init=0.85, sigma_decay=0.999,
                                       lr_decay=0.9997, verbose=True, label="sm2")
        try_seq(a_opt, "smooth_mom")

        # Second pass with different params
        rem = deadline - time.monotonic() - 500
        if rem > 20 and best_seq is not None:
            t2b = min(rem * 0.3, 120)
            a_opt, _ = smooth_momentum_opt(best_seq, time_limit=t2b,
                                           sigma_init=0.01, lr_init=0.003,
                                           mom_init=0.90, sigma_decay=0.9995,
                                           lr_decay=0.9999, verbose=False, label="sm2b")
            try_seq(a_opt, "smooth_mom_fine")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.8f}) ===")
    init_configs = [
        (512, 1, 0.4), (512, 2, 0.3), (512, 3, 0.2),
        (640, 1, 0.35), (640, 2, 0.25),
        (768, 1, 0.4), (768, 2, 0.3), (768, 3, 0.2), (768, 4, 0.15),
        (1024, 1, 0.35), (1024, 2, 0.25), (1024, 3, 0.2),
        (1280, 1, 0.3), (1280, 2, 0.2),
        (1536, 1, 0.3),
    ]

    # Also try Gaussian and raised-cosine inits
    special_inits = [
        (gaussian_init, 768, 1.0),
        (gaussian_init, 1024, 1.5),
        (raised_cosine_init, 768, 0),
        (raised_cosine_init, 1024, 0),
    ]

    for n_init, freq, amp in init_configs:
        rem = deadline - time.monotonic() - 400
        if rem < 15:
            break
        a_init = cosine_init(n_init, freq, amp)
        s_init = try_seq(a_init, f"cos_{n_init}_f{freq}")
        if np.isfinite(s_init) and s_init < best_score + 0.05:
            t_alloc = min(rem * 0.15, 60)
            if t_alloc > 8:
                a_opt, _ = smooth_momentum_opt(a_init, time_limit=t_alloc,
                                               sigma_init=0.03, lr_init=0.006,
                                               mom_init=0.85, verbose=False,
                                               label=f"cos_{n_init}")
                try_seq(a_opt, f"smooth_cos_{n_init}_f{freq}")

    for init_fn, n_init, param in special_inits:
        rem = deadline - time.monotonic() - 350
        if rem < 15:
            break
        a_init = init_fn(n_init, param)
        s_init = try_seq(a_init, f"spec_{n_init}")
        if np.isfinite(s_init) and s_init < best_score + 0.05:
            t_alloc = min(rem * 0.12, 45)
            if t_alloc > 8:
                a_opt, _ = smooth_momentum_opt(a_init, time_limit=t_alloc,
                                               sigma_init=0.03, lr_init=0.006,
                                               mom_init=0.85, verbose=False,
                                               label=f"spec_{n_init}")
                try_seq(a_opt, f"smooth_spec_{n_init}")

    # Try interpolating best to different lengths
    if best_seq is not None:
        for target_n in [1024, 1280]:
            rem = deadline - time.monotonic() - 300
            if rem < 20:
                break
            f_interp = interpolate_to(best_seq, target_n)
            s_int = try_seq(f_interp, f"interp_{target_n}")
            if np.isfinite(s_int):
                t_alloc = min(rem * 0.2, 80)
                if t_alloc > 10:
                    a_opt, _ = smooth_momentum_opt(f_interp, time_limit=t_alloc,
                                                   sigma_init=0.02, lr_init=0.004,
                                                   mom_init=0.88, verbose=False,
                                                   label=f"int_{target_n}")
                    try_seq(a_opt, f"smooth_interp_{target_n}")

    # === Phase 4: Hybrid optimization on best ===
    print(f"\n=== Phase 4: Hybrid optimization (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t4 = min(rem * 0.5, 250)
            print(f"  Hybrid t={t4:.0f}s, n={n_best}")
            a_opt, _ = hybrid_optimize(best_seq, time_limit=t4, verbose=True, label="hyb")
            try_seq(a_opt, "hybrid")

            # Second hybrid with different starting perturbation
            rem = deadline - time.monotonic() - 150
            if rem > 20:
                noise = rng.standard_normal(n_best) * 0.003
                a_pert = project_simplex(best_seq + noise)
                t4b = min(rem * 0.6, 100)
                a_opt, _ = hybrid_optimize(a_pert, time_limit=t4b, verbose=False, label="hyb2")
                try_seq(a_opt, "hybrid_pert")

    # === Phase 5: LP-FW on best (if not already done) ===
    print(f"\n=== Phase 5: LP-FW polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t5 = min(rem * 0.5, 100)
            print(f"  LP-FW t={t5:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t5, verbose=False, label="lp5")
            try_seq(a_opt, "lpfw_polish")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t6 = min(rem * 0.6, 120)
            print(f"  CD t={t6:.0f}s")
            a_opt, _ = golden_cd(best_seq, time_limit=t6)
            try_seq(a_opt, "golden_cd")

    # === Phase 7: Final quick attempts ===
    print(f"\n=== Phase 7: Final attempts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        attempts = [
            ("smooth_final", lambda: smooth_momentum_opt(best_seq, time_limit=min(30, deadline - time.monotonic() - 10),
                                                         sigma_init=0.005, lr_init=0.002,
                                                         mom_init=0.92, sigma_decay=0.9998,
                                                         lr_decay=0.99995, verbose=False, label="fin")),
            ("lpfw_final", lambda: lp_fw_optimize(best_seq, time_limit=min(20, deadline - time.monotonic() - 10),
                                                 verbose=False, label="fin")),
        ]

        # Multi-perturbation
        for attempt_i in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.001
            a_p = project_simplex(best_seq + noise)
            t_alloc = min(rem * 0.3, 25)
            a_opt, _ = smooth_momentum_opt(a_p, time_limit=t_alloc,
                                           sigma_init=0.008, lr_init=0.002,
                                           mom_init=0.93, sigma_decay=0.9998,
                                           lr_decay=0.99995, verbose=False, label=f"pert{attempt_i}")
            try_seq(a_opt, f"perturb_{attempt_i}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence([float(x) for x in best_seq]))

    # Final verification
    fs = float(evaluate_sequence([float(x) for x in np.maximum(best_seq, 0.0)]))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in np.maximum(best_seq, 0.0)]
