def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)

            continue
        exponent -= emax
        f_new = f * np.exp(exponent)
        fs = f_new.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            no_improve += 50
            continue
        f_new /= fs
        f = np.maximum(f_new, 0.0)
        fsum = f.sum()
        if fsum < 1e-15:
            f = best_f.copy()
            v = np.zeros(n)
            continue
        f /= fsum

        if it % skip_eval == 0 or time.monotonic() - t0 > time_limit - 1:
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 800:
            rng = np.random.default_rng((it * 7919 + 17) % 1000000)
            noise_scale = 0.002 + 0.001 * min(no_improve // 1000, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            no_improve = 0

        if verbose and it % 2000 == 0:
            print(f"      MU iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, "
                  f"tau={tau:.2e}, eta={eta:.5f}")

    return best_f, best_obj


def optimize_fista(f0, time_limit, tau0=0.0003, tau1=0.000001,
                   eta0=0.008, eta_decay=3.0, verbose=False):
    """
    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) with
    simplex projection for the constraint a>=0, sum(a)=1.
    Uses the smooth approximation 2n*tau*log(sum exp(c_k/tau)).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    y = f.copy()
    t_nesterov = 1.0
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1.0), 1.0)

        tau = tau0 * (tau1 / tau0) ** progress
        eta = eta0 / (1.0 + eta_decay * progress)

        conv_y = fast_convolve(y)
        sg = full_softmax_subgradient(y, conv_y, tau)
        # Scale gradient by 2n (the full objective is 2n * smooth_max)
        sg *= (2.0 * n)

        # Gradient step
        z = y - eta * sg
        # Project onto simplex
        x_new = project_simplex(z)

        # Nesterov update
        t_prev = t_nesterov
        t_nesterov = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        y = x_new + ((t_prev - 1.0) / t_nesterov) * (x_new - f)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = x_new.copy()
        else:
            y /= ys

        f = x_new

        if it % 5 == 0:
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 2000:
            rng = np.random.default_rng((it * 3301 + 7) % 1000000)
            noise_scale = 0.003
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            y = f.copy()
            t_nesterov = 1.0
            no_improve = 0

        if verbose and it % 2000 == 0:
            print(f"      FISTA iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def peak_coordinate_descent(f0, time_limit, verbose=False):
    """
    Coordinate descent targeting peak-contributing coordinates.
    For the peak lag k*, identifies coordinates with large a[i]*a[k*-i]
    and transfers mass to low-contributor coordinates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    best_peak_lag = int(np.argmax(conv))

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_max = conv.max()
        # Get top 5 peak lags
        top_lags = np.argpartition(conv, -5)[-5:]

        improved = False
        for k_star in top_lags:
            k_star = int(k_star)
            if time.monotonic() - t0 > time_limit:
                break

            # Compute contributions to this peak
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k_star - idx]

            if len(contribs) < 2:
                continue

            # Sort by contribution
            order = np.argsort(contribs)[::-1]
            hi_idx = idx[order[:min(5, len(order))]]
            lo_idx = idx[order[-min(5, len(order)):]]

            for hi in hi_idx:
                for lo in lo_idx:
                    if hi == lo or f[hi] < 1e-10:
                        continue
                    for frac in [0.05, 0.1, 0.15, 0.2]:
                        amt = f[hi] * frac
                        f[hi] -= amt
                        f[lo] += amt
                        val = fast_convolve(f).max()
                        if val < best_obj - 1e-14:
                            best_obj = val
                            best_f = f.copy()
                            improved = True
                            break
                        else:
                            f[hi] += amt
                            f[lo] -= amt
                            break

        # Also try random pair exchanges
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or f[i] < 1e-10:
                continue
            for frac in [0.05, 0.1, 0.2]:
                amt = f[i] * frac
                f[i] -= amt
                f[j] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] += amt
                    f[j] -= amt
                    break

        if not improved:
            # Try a more aggressive restart
            f = f.copy()
            noise = rng.standard_normal(n) * 0.002
            f = project_simplex(f + noise)

    return best_f, best_obj


def coordinated_sweep(f0, time_limit, verbose=False):
    """
    Sweep individual coordinates: for each coordinate, try several
    multipliers and re-normalize the rest proportionally.
    Focus on coordinates near the peak contributing lags.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()

    # Determine priority order: coordinates contributing most to peak
    peak_lag = int(np.argmax(conv))
    js = max(0, peak_lag - n + 1)
    je = min(n - 1, peak_lag)
    idx = np.arange(js, je + 1)
    contribs = f[idx] * f[peak_lag - idx]

    # Build priority order
    priority = list(idx[np.argsort(contribs)[::-1]])
    remaining = [i for i in range(n) if i not in set(priority)]
    order = priority + remaining

    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]

    sweeps = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    fsum = f.sum()
                    if fsum < 1e-15:
                        f = best_f.copy()
                        continue
                    f /= fsum

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        sweeps += 1
        if not improved:
            break
        if verbose and sweeps % 3 == 0:
            print(f"      CoordSweep pass {sweeps}, obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length via linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate structured initial sequences for diversity."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 5.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    gauss = np.maximum(gauss, 1e-10)
    inits.append(gauss / gauss.sum())

    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 80)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Chirp-like (linear frequency sweep)
    chirp = 1.0 + 0.5 * np.sin(2 * np.pi * rng.random() * 3 * x)
    chirp = np.maximum(chirp, 1e-10)
    inits.append(chirp / chirp.sum())

    return inits[:count]


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining multiplicative-weights subgradient,
    FISTA with simplex projection, peak-targeted coordinate descent,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})", flush=True)
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}", flush=True)
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform n=768", flush=True)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 1: Multi-start optimization ===
    print("\n=== Phase 1: Multi-Start Optimization ===", flush=True)

    candidates = [parent]
    for n_try in [768, 1024]:
        inits = structured_inits(n_try, rng, count=3)
        candidates.extend(inits)

    # Deduplicate by length and skip if too many
    seen_lengths = {len(parent)}
    filtered = [candidates[0]]
    for c in candidates[1:]:
        if len(c) not in seen_lengths:
            seen_lengths.add(len(c))
            filtered.append(c)
        elif len(filtered) < 6:
            filtered.append(c)
    candidates = filtered[:6]

    for idx, cand in enumerate(candidates):
        rem = deadline - time.monotonic() - 600
        if rem < 15:
            break
        n_cand = len(cand)
        t_alloc = min(rem * 0.15, 80)
        if t_alloc < 5:
            t_alloc = min(rem * 0.3, 40)

        print(f"  Cand {idx}: n={n_cand}, t={t_alloc:.0f}s", flush=True)

        # Try both MU weights and FISTA
        if t_alloc > 20:
            t_half = t_alloc * 0.5
            a_opt, _ = optimize_mu_weights(cand, time_limit=t_half,
                                           tau0=0.0004, tau1=0.000001,
                                           eta0=0.004, eta_decay=2.0,
                                           mu0=0.85, mu1=0.95,
                                           use_full_softmax=True)
            try_seq(a_opt, f"mu_{idx}")

            rem2 = deadline - time.monotonic() - 500
            if rem2 > 10:
                a_opt2, _ = optimize_fista(cand, time_limit=min(rem2 * 0.1, t_half * 0.8),
                                           tau0=0.0004, tau1=0.000001,
                                           eta0=0.006, eta_decay=2.5)
                try_seq(a_opt2, f"fista_{idx}")
        else:
            a_opt, _ = optimize_mu_weights(cand, time_limit=t_alloc,
                                           tau0=0.0003, tau1=0.000002,
                                           eta0=0.003, eta_decay=2.0,
                                           mu0=0.88, mu1=0.95)
            try_seq(a_opt, f"mu_{idx}_short")

    # === Phase 2: Deep optimization on best ===
    print(f"\n=== Phase 2: Deep Optimization (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {"method": "mu", "tau0": 0.0002, "tau1": 0.0000005,
             "eta0": 0.003, "eta_decay": 3.0, "mu0": 0.90, "mu1": 0.97,
             "full": True, "top_k": 200},
            {"method": "mu", "tau0": 0.0005, "tau1": 0.000001,
             "eta0": 0.005, "eta_decay": 1.5, "mu0": 0.82, "mu1": 0.93,
             "full": True, "top_k": 100},
            {"method": "fista", "tau0": 0.0002, "tau1": 0.0000005,
             "eta0": 0.005, "eta_decay": 3.5},
            {"method": "mu", "tau0": 0.0001, "tau1": 0.0000003,
             "eta0": 0.002, "eta_decay": 4.0, "mu0": 0.93, "mu1": 0.98,
             "full": True, "top_k": 50},
        ]

        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 300
            if rem < 15:
                break
            t_alloc = min(rem * 0.25, 120)

            # Perturb the best for diversity
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise)

            print(f"  Deep {p}: n={n_best}, t={t_alloc:.0f}s, method={cfg['method']}", flush=True)

            if cfg["method"] == "mu":
                a_opt, _ = optimize_mu_weights(f_init, time_limit=t_alloc,
                                               tau0=cfg["tau0"], tau1=cfg["tau1"],
                                               eta0=cfg["eta0"], eta_decay=cfg["eta_decay"],
                                               mu0=cfg["mu0"], mu1=cfg["mu1"],
                                               use_full_softmax=cfg.get("full", True),
                                               top_k=cfg.get("top_k", 100))
            else:
                a_opt, _ = optimize_fista(f_init, time_limit=t_alloc,
                                          tau0=cfg["tau0"], tau1=cfg["tau1"],
                                          eta0=cfg["eta0"], eta_decay=cfg["eta_decay"])
            try_seq(a_opt, f"deep_{p}")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length Exploration (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        target_lengths = [640, 1024, 1280, 1536]
        for target_n in target_lengths:
            if target_n == len(best_seq):
                continue
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.25, 90)
            print(f"  n={target_n}, t={t_alloc:.0f}s", flush=True)

            a_opt, _ = optimize_mu_weights(f_interp, time_limit=t_alloc,
                                           tau0=0.0003, tau1=0.000001,
                                           eta0=0.004, eta_decay=2.5,
                                           mu0=0.88, mu1=0.96,
                                           use_full_softmax=True)
            try_seq(a_opt, f"len_{target_n}")

            # Quick FISTA on the result
            rem2 = deadline - time.monotonic() - 150
            if rem2 > 15:
                a_opt2, _ = optimize_fista(a_opt, time_limit=min(rem2 * 0.2, 40),
                                           tau0=0.0001, tau1=0.0000005,
                                           eta0=0.004, eta_decay=3.0)
                try_seq(a_opt2, f"fista_len_{target_n}")

    # === Phase 4: Peak-targeted coordinate descent ===
    print(f"\n=== Phase 4: Peak Coordinate Descent (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_pcd = min(rem * 0.5, 80)
            print(f"  Peak coord descent t={t_pcd:.0f}s", flush=True)
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_pcd)
            try_seq(a_opt, "peak_cd")

            rem2 = deadline - time.monotonic() - 60
            if rem2 > 10:
                t_sweep = min(rem2 * 0.7, 60)
                print(f"  Coordinated sweep t={t_sweep:.0f}s", flush=True)
                a_opt2, _ = coordinated_sweep(best_seq, time_limit=t_sweep)
                try_seq(a_opt2, "coord_sweep")

    # === Phase 5: Final fine-tuning ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine subgradient with low temperature
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = optimize_mu_weights(best_seq, time_limit=min(rem, 50),
                                           tau0=0.00001, tau1=0.0000001,
                                           eta0=0.0005, eta_decay=3.0,
                                           mu0=0.95, mu1=0.99,
                                           use_full_softmax=True)
            try_seq(a_opt, "sg_fine")

        # FISTA fine
        rem = deadline - time.monotonic() - 25
        if rem > 8:
            a_opt, _ = optimize_fista(best_seq, time_limit=min(rem, 30),
                                      tau0=0.00001, tau1=0.0000001,
                                      eta0=0.001, eta_decay=4.0)
            try_seq(a_opt, "fista_fine")

        # Small perturbations + quick optimize
        for _ in range(3):
            rem = deadline - time.monotonic() - 15
            if rem < 8:
                break
            for scale in [0.001, 0.002]:
                rem = deadline - time.monotonic() - 10
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = optimize_mu_weights(f_pert, time_limit=min(rem, 10),
                                               tau0=0.000005, tau1=0.0000002,
                                               eta0=0.0004, eta_decay=3.0,
                                               mu0=0.96, mu1=0.99,
                                               use_full_softmax=True)
                try_seq(a_opt, f"pert_{scale}")

        # Final coordinate descent
        rem = deadline - time.monotonic() - 8
        if rem > 5:
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "final_pcd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
