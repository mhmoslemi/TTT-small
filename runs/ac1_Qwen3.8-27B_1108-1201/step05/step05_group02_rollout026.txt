Actually, let me reconsider the spectral approach more carefully.

The auto-convolution c = a * a (full convolution). In frequency domain: C(ω) = |A(ω)|^2 where A(ω) = FFT(a).

We want to minimize max(c). Equivalently, we want |A(ω)|^2 to be as flat as possible (since c is the IFFT of |A|^2).

The "flatness" of |A|^2 is measured by max(|A|^2) / mean(|A|^2) (in the frequency domain, this isn't exactly the same as max(c)/mean(c) due to the FFT/IFFT relationship, but it's related).

Actually, by Parseval: sum(c[k]^2) = (1/N) sum(|C(ω)|^2) where N is the FFT size. And sum(c[k]) = sum(a)^2 = C(0) = |A(0)|^2.

The relationship between time-domain peak and frequency-domain flatness is:
- A flat |A(ω)|^2 means A(ω) has constant magnitude → a is a "bandlimited" signal with constant envelope → in time domain, this gives a sinc-like function.
- But we need a >= 0, which constrains us.

The spectral approach for iteratively flattening:
1. Compute A = FFT(a)
2. Compute G = |A|^2 (this is the power spectrum)
3. We want the IFFT of G to be as flat as possible
4. Modify G: G_new = G^(1/2) (reduces dynamic range) or G_new = exp(log(G)/2) or G_new = 1 + (G - mean(G)) * alpha
5. A_new = sqrt(G_new) * exp(i*angle(A)) (preserve phase, flatten magnitude)
6. a_new = IFFT(A_new)
7. Project a_new onto simplex (non-negative, sum=1)
8. Repeat

This is similar to the "Gerchberg-Saxton" or "error reduction" algorithm used in optics (for phase retrieval and beam shaping).

Let me implement this!

The key question is: what modification to G gives the best flattening? 

If I use G_new = G^gamma where 0 < gamma < 1, this compresses the dynamic range. gamma=0.5 is the "square root" (makes |A| flatter). gamma closer to 0 makes it even flatter but might introduce oscillations in the time domain.

Let me try gamma = 0.5 to 0.8 and see what works.

1. Spectral flattening for initialization
2. LP-based Frank-Wolfe with adaptive top-K peak formulation
3. Fine-grained coordinate descent for peak reduction
4. Multi-length exploration and polishing

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a, n_iters=300, gamma=0.5):
    """Spectral flattening via iterative magnitude modification."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    a_cur = a.astype(np.float64).copy()
    s = a_cur.sum()
    if s < 1e-15:
        a_cur = np.ones(n) / n
    else:
        a_cur = a_cur / s
    
    for it in range(n_iters):
        A = np.fft.fft(a_cur, N)
        G = np.abs(A) ** 2
        # Compress dynamic range
        G_max = G.max()
        G_min = G.min()
        if G_max < 1e-15:
            break
        # Use power compression
        G_new = G ** gamma
        phase = A / np.where(np.abs(A) > 1e-15, np.abs(A), 1.0)
        A_new = np.sqrt(G_new) * phase
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a_cur = a_new
    
    return a_cur


def solve_topk_lp(f, peaks, n):
    """Solve LP: min t s.t. conv(f, g)[k] <= t for k in peaks, g>=0, sum(g)=1.
    Returns (g, t) or (None, None)."""
    K = len(peaks)
    # Variables: g[0..n-1], t (n+1 total)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: conv(f, g)[k_i] - t <= 0 for each peak k_i
    A_ub = np.zeros((K, n + 1))
    for i, k in enumerate(peaks):
        A_ub[i, -1] = -1.0  # -t
        # conv(f, g)[k] = sum_j f[k-j] * g[j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[i, j] = f[k - j]
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    b_ub = np.zeros(K)
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def line_search(f, g, base_conv, n):
    """Find best alpha in [0,1] to minimize max conv of (1-alpha)*f + alpha*g."""
    best_alpha = 0.0
    best_val = base_conv.max()
    
    # Coarse search
    for alpha in [0.03, 0.05, 0.08, 0.12, 0.2, 0.3, 0.5, 0.7]:
        ft = (1 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    # Fine search (bisection)
    if best_alpha > 0:
        lo = max(0.0, best_alpha - 0.05)
        hi = min(1.0, best_alpha + 0.05)
        for _ in range(15):
            mid = (lo + hi) / 2.0
            ft = (1 - mid) * f + mid * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                break
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid
    
    return best_alpha, best_val


def lp_fw_optimize(f0, time_limit, K_start=3, K_end=20, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    stuck_threshold = 50
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        # Adaptive K: start small, grow over time
        K = min(K_start + int(progress * (K_end - K_start)), K_end)
        K = max(1, min(K, 2 * n - 2))
        
        # Find top K peaks
        if K >= 2 * n - 1:
            peaks = list(range(2 * n - 1))
        else:
            peaks = np.argpartition(conv, -K)[-K:]
        
        # Solve LP
        g, lp_t = solve_topk_lp(f, peaks, n)
        if g is None:
            g = np.ones(n) / n
        else:
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs < 1e-15:
                g = np.ones(n) / n
            else:
                g = g / gs
        
        # Check if g is close to f (convergence)
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        # Line search
        alpha, best_val = line_search(f, g, conv, n)
        
        if alpha > 1e-8:
            f_new = (1 - alpha) * f + alpha * g
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 500 == 0:
                    score = 2 * n * best_obj
                    print(f"    LP-FW iter {it}: score={score:.8f}, K={K}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > stuck_threshold:
            rng = np.random.default_rng(it * 104729 + 7)
            noise_scale = 0.001 * min(no_improve // stuck_threshold, 5)
            f_pert = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            f = f_pert
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            stuck_threshold = max(20, stuck_threshold + 10)
    
    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.005, momentum=0.93, top_k=30, temp=0.0001, verbose=False):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]
        
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 1000 == 0:
                print(f"    SG iter {it}: score={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            f = f_new
        
        cur_lr *= 0.9999
        cur_temp *= 1.002
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0
    
    return best_f, best_obj


def coord_descent_peaks(f0, time_limit, verbose=False):
    """Coordinate descent that targets peak-reducing moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    passes = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        # Find top 3 peaks
        top_k = min(3, 2 * n - 1)
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        
        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            
            # Find contributing coordinates
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            # Try reducing high-mass coordinates that contribute to this peak
            # Sort by contribution to this peak
            contributions = []
            for j in range(js, je + 1):
                contrib = f[j] * f[peak_k - j]
                contributions.append((contrib, j))
            contributions.sort(reverse=True)
            
            # Try reducing top contributors
            for contrib, j in contributions[:8]:
                if f[j] < 1e-10:
                    continue
                
                # Try reducing f[j] and redistributing
                for mult in [0.5, 0.7, 0.85, 1.2, 1.4]:
                    new_val = f[j] * mult
                    if new_val < 0 or new_val >= 1.0:
                        continue
                    
                    f_trial = f.copy()
                    delta = new_val - f[j]
                    f_trial[j] = new_val
                    # Redistribute delta to other coordinates (or absorb into remainder)
                    if abs(delta) > 1e-12:
                        # Redistribute to random other coordinates
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        other_idx = np.where(mask)[0]
                        # Redistribute proportionally
                        f_trial[other_idx] -= delta * f[other_idx] / f[other_idx].sum()
                        f_trial = np.maximum(f_trial, 0)
                        st = f_trial.sum()
                        if st < 1e-15:
                            continue
                        f_trial = f_trial / st
                    
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        improved = True
                        break
        
        if not improved:
            # Random exploration
            for _ in range(min(50, n // 20)):
                if time.monotonic() - t0 > time_limit:
                    break
                j = int(rng.integers(0, n))
                if f[j] < 1e-8:
                    continue
                for mult in [0.6, 0.8, 1.25, 1.5]:
                    new_val = f[j] * mult
                    if new_val < 0 or new_val >= 0.5:
                        continue
                    f_trial = f.copy()
                    delta = new_val - f[j]
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    other_idx = np.where(mask)[0]
                    f_trial[other_idx] -= delta * f[other_idx] / (f[other_idx].sum() + 1e-15)
                    f_trial = np.maximum(f_trial, 0)
                    st = f_trial.sum()
                    if st < 1e-15:
                        continue
                    f_trial = f_trial / st
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        improved = True
                        break
            if not improved:
                no_improve += 1
                if no_improve > 20:
                    break
        else:
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
