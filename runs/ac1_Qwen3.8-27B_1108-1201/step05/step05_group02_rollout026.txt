Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based Frank-Wolfe, subgradient descent, and intensive coordinate descent
with simulated annealing for final polish.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_obj(f):
    """Compute the objective value (assuming sum(f)=1)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

        scaled -= scaled.max()
        w = np.exp(scaled)

        if ws < 1e-15:

            w[np.argmax(pv)] = 1.0

                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)

            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

Let me try the log-parameterization approach:
- f[j] = exp(w[j]) / sum(exp(w))
- The objective is smooth in w
- We can use standard gradient descent on w (no projection needed!)

The gradient w.r.t. w is:
d/dw[j] obj = sum_k (d obj/d f[k']) * (d f[k']/dw[j]) for all k'

Wait, let me be more careful. Let F = sum_k w_probs[k] * conv[k] (the smooth objective part).

dF/dw[j] = sum_i (dF/df[i]) * (df[i]/dw[j])

df[i]/dw[j] = f[i] * (delta(i,j) - f[j])

So dF/dw[j] = sum_i (dF/df[i]) * f[i] * (delta(i,j) - f[j])
= f[j] * dF/df[j] - f[j] * sum_i (dF/df[i]) * f[i]
= f[j] * (grad[j] - sum_i grad[i]*f[i])

where grad[i] = dF/df[i] is what we computed above.

So the w-gradient is: fw * (grad - mean_weighted) where mean_weighted = sum_i grad[i]*f[i].

This is nice because it's automatically "centered" and the optimization in w-space doesn't need projection.

    return best_f

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses smooth gradient descent (new), LP-Frank-Wolfe, and intensive coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None


def smooth_gd(f0, time_limit, verbose=False):
    """Smooth gradient descent in log-parameterization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)
    
    conv0 = fast_convolve(f)
    max0 = conv0.max()
    temps = max0 * np.array([0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005])
    
    best_f = f.copy()
    best_score = 2.0 * n * max0
    
    t0 = time.monotonic()
    time_per_temp = time_limit / len(temps)
    
    for ti, temp in enumerate(temps):
        t_start = time.monotonic()
        w = np.log(np.maximum(f, 1e-30))
        lr = 5.0 / n
        lr_min = 0.1 / n
        momentum_v = np.zeros(n)
        momentum = 0.9
        
        for it in range(50000):
            if time.monotonic() - t_start > time_per_temp:
                break
            if time.monotonic() - t0 > time_limit:
                break
            
            w_shift = w - w.max()
            f = np.exp(w_shift)
            f_sum = f.sum()
            if f_sum < 1e-30:
                break
            f = f / f_sum
            
            conv = fast_convolve(f)
            shifted = conv / temp
            shifted -= shifted.max()
            e = np.exp(shifted)
            e_sum = e.sum()
            if e_sum < 1e-30:
                break
            w_probs = e / e_sum
            
            # Compute gradient
            h_rev = w_probs[::-1]
            fft_size = 1 << (3 * n - 1).bit_length()
            F = np.fft.rfft(f, fft_size)
            H = np.fft.rfft(h_rev, fft_size)
            conv_fhr = np.fft.irfft(F * H, fft_size)[:3 * n - 2]
            grad_f = 2.0 * conv_fhr[n - 1:2 * n - 1][::-1]
            
            mean_grad = np.dot(grad_f, f)
            grad_w = f * (grad_f - mean_grad)
            
            # Momentum step
            momentum_v = momentum * momentum_v - lr * grad_w
            w = w + momentum_v
            
            # Adaptive LR
            if it % 500 == 499:
                lr = max(lr * 0.9, lr_min)
            
            # Track best
            score = 2.0 * n * conv.max()
            if score < best_score - 1e-12:
                best_score = score
                best_f = f.copy()
                if verbose and it % 500 == 499:
                    print(f"      SGD temp={temp:.6f} it={it}: {best_score:.10f}")
        
        f = best_f.copy()
    
    return best_f


def fw_iterate(f0, time_limit, perturb_every=50, verbose=False):
    """LP-based Frank-Wolfe iteration."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        g = solve_lp_min_peak(f, n)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0
            continue
        
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs
        
        if np.max(np.abs(g - f)) < 1e-14:
            no_improve += 1
            if no_improve > perturb_every:
                f = project_simplex(best_f + rng.standard_normal(n) * (0.001 + 0.001 * no_improve // 100))
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = fast_convolve(f).max()
        
        # Coarse grid
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            ft = np.maximum(f + alpha * (g - f), 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine bisection
        if 0 < best_alpha < 1:
            lo = max(0, best_alpha - 0.1)
            hi = min(1, best_alpha + 0.1)
            for _ in range(15):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                ft1 = np.maximum(f + m1 * (g - f), 0)
                ft1 /= ft1.sum()
                ft2 = np.maximum(f + m2 * (g - f), 0)
                ft2 /= ft2.sum()
                v1 = fast_convolve(ft1).max()
                v2 = fast_convolve(ft2).max()
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            best_alpha = (lo + hi) / 2
            ft = np.maximum(f + best_alpha * (g - f), 0)
            ft /= ft.sum()
            best_val = fast_convolve(ft).max()
        
        if best_alpha > 1e-12:
            f = np.maximum(f + best_alpha * (g - f), 0)
            f = f / f.sum()
            score = 2.0 * n * best_val
            it += 1
            if score < best_score - 1e-12:
                best_score = score
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    print(f"    FW it={it}: {best_score:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > perturb_every:
            scale = 0.001 + 0.001 * min(no_improve // 50, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            no_improve = 0
    
    return best_f


def coord_descent(f0, time_limit, num_trials=5, verbose=False):
    """Intensive coordinate descent with multiple trials per coordinate."""
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_score = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve_passes = 0
    mults = np.array([0.3, 0.5, 0.7, 0.85, 1.2, 1.4, 1.7])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        
        # Process coordinates in order of contribution to peak
        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        
        # Sort coordinates by their contribution to the peak
        contributions = np.zeros(n)
        for j in range(js, je + 1):
            kg = peak_k - j
            if 0 <= kg < n:
                contributions[j] = f[j] * f[kg]
        
        order = np.argsort(-contributions)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue
            
            old_val = f[j]
            best_j_obj = best_score
            best_j_val = old_val
            
            for mult in mults:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = 2.0 * n * fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_score - 1e-14:
                    best_score = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 5:
                break
        else:
            no_improve_passes = 0
    
    return best_f
