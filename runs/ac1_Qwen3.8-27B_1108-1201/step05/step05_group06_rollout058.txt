Key insights:
1. The objective is `2*n*max_b / sum_a^2` where `max_b = max(conv(a,a))` and `sum_a = sum(a)`.
2. Since the objective is scale-invariant (if we scale `a` by `c`, both `max_b` scales by `c^2` and `sum_a^2` scales by `c^2`), we can normalize `sum(a) = 1` and minimize `2*n*max(conv(a,a))`.
3. So effectively we're minimizing `max(conv(a,a))` subject to `sum(a) = 1` and `a >= 0`.

This is equivalent to finding a non-negative sequence with unit sum that minimizes the peak of its auto-convolution. This is related to the "flat-top" problem in signal processing - we want the autocorrelation to be as flat as possible.

The theoretical lower bound: if `conv(a,a)` were perfectly flat, each entry would be `1/(2n-1)` (since sum of all entries of conv = sum(a)^2 = 1), so `max_b >= 1/(2n-1)`, giving objective `>= 2n/(2n-1) ≈ 1`. But we can't achieve this with non-negative sequences easily.

For the specific structure: the convolution has `2n-1` entries, sum = 1. The minimum possible max is at least `1/(2n-1)`. The objective would be `2n/(2n-1)`.

    n = len(v)
    if n == 1:

I can derive that when changing coordinate `j` from `old` to `t` (with other coordinates scaled proportionally to maintain sum=1), the new convolution entries are **quadratic** in `t`: `conv_new[k] = (A[k] + B[k]*t + C[k]*t²) / (1-old)²`. This means I can use fast vectorized ternary search (40 iterations of O(n) vectorized ops) instead of the expensive bisection+FFT approach.

This makes each coordinate step ~10x faster, enabling many more passes and restarts in the same time budget.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using rapid coordinate descent
(ternary search on quadratic per-coordinate updates), iterated local search
with adaptive perturbations, smooth gradient descent, and multi-length exploration.
Key: per-coordinate update is quadratic in the new value, enabling fast vectorized
ternary search without FFT calls inside the inner loop.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_descent(a_init, conv_init, n, time_limit, tern_iters=40, verbose=False):
    """
    Rapid coordinate descent with ternary search.
    For each coordinate j, the new convolution is quadratic in t (new value of a[j]).
    Minimize max_k conv_new[k] via ternary search on the upper envelope of quadratics.
    """
    a = a_init.copy().astype(np.float64)
    conv = conv_init.copy().astype(np.float64)
    best_obj = conv.max()
    best_a = a.copy()
    conv_len = 2 * n - 1
    t0 = time.monotonic()
    passes_done = 0

    for p in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        passes_done += 1
        improved = False

        for j in range(n):
            old = a[j]
            if old < 1e-13:
                continue
            one_minus_old = 1.0 - old
            if one_minus_old < 1e-13:
                continue

            # Precompute quadratic coefficients:
            # conv_new[k] = (A[k] + B[k]*t + C[k]*t^2) / one_minus_old^2
            A_arr = conv.copy()
            B_arr = -2.0 * conv.copy()
            C_arr = conv.copy()

            # Cross terms: k in [j, j+n-1], k != 2j
            k_lo = j
            k_hi = min(j + n, conv_len)
            if k_lo < k_hi:
                a_vals = a[k_lo - j:k_hi - j]
                B_arr[k_lo:k_hi] = 2.0 * (1.0 + old) * a_vals - 2.0 * conv[k_lo:k_hi]
                C_arr[k_lo:k_hi] = conv[k_lo:k_hi] - 2.0 * a_vals

            # Diagonal: k = 2j
            if 0 <= 2 * j < conv_len:
                c2j = conv[2 * j]
                A_arr[2 * j] = c2j - old * old
                B_arr[2 * j] = -2.0 * (c2j - old * old)
                C_arr[2 * j] = 1.0 - 2.0 * old + c2j

            # Ternary search for optimal t in [0, 1]
            lo = 0.0
            hi = 1.0
            for _ in range(tern_iters):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(A_arr + B_arr * m1 + C_arr * m1 * m1)
                f2 = np.max(A_arr + B_arr * m2 + C_arr * m2 * m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            t_new = (lo + hi) / 2.0
            if abs(t_new - old) < 1e-13:
                continue

            # Update a
            s = (1.0 - t_new) / one_minus_old
            a[j] = t_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= s
            np.maximum(a, 0.0, out=a)

            # Update conv
            conv = (A_arr + B_arr * t_new + C_arr * t_new * t_new) / (one_minus_old * one_minus_old)
            np.maximum(conv, 0.0, out=conv)

            obj = conv.max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                improved = True

        if verbose and passes_done % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD pass {passes_done}: obj={best_obj:.12f}, elapsed={elapsed:.1f}s")

        if not improved:
            break

    return best_a, conv, best_obj

    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    v = np.zeros(n)
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Compute softmax weights over convolution peaks
        scaled = conv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        # Gradient: d(conv[k])/da[j] involves a[k-j] for valid range
        sg = np.zeros(n)
        conv_len = len(conv)
        for k in range(conv_len):
            wk = w[k]
            if wk < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                sg[js:je + 1] += 2.0 * wk * a[k - js:k - js + (je - js + 1)]

        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        cur_lr *= 0.9998
        cur_temp *= 1.005

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr_init * 0.5
            no_improve = 0

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric and renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym
