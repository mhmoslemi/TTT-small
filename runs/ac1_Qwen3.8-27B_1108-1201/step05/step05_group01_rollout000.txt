    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max() * 2 * n

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr0
    cur_mom = momentum

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Adaptive temperature: anneal from T0 to T_end
        frac = min(1.0, elapsed / (time_limit * 0.7))
        T = T0 + (T_end - T0) * frac ** 2

        # Use top-k subgradient for speed when T is large
        if T > 15.0:
            sg, conv = compute_subgradient_topk(a, top_k)
        else:
            sg, conv = compute_smoothed_gradient(a, T)

        # Nesterov update
        y = a + cur_mom * v
        # Compute gradient at y (or at a for stability)
        if T > 15.0:
            sg_y, _ = compute_subgradient_topk(y, top_k)
        else:
            sg_y, _ = compute_smoothed_gradient(y, T)

        v = cur_mom * v - cur_lr * sg_y
        a_new = project_simplex(a + v)

        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt learning rate
        if no_improve > 200:
            cur_lr *= 0.9
            no_improve = 0
        elif no_improve > 1000:
            # Restart with perturbation
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            y = a.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.7, 0.001)
            no_improve = 0

        # Periodic momentum decay
        cur_mom = min(0.9, momentum + 0.0001 * it)

        if verbose and it % 5000 == 0:
            print(f"    iter={it}, obj={best_obj:.8f}, T={T:.1f}, lr={cur_lr:.6f}")

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, n_trials=200):
    """
    Coordinate descent: for each coordinate, try scaling it by various factors
    and keep the best. More efficient than golden section for this problem.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max() * 2 * n

    t0 = time.monotonic()
    multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 2.0])

    pass_count = 0
    no_pass_improve = 0

    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        pass_improved = False

        # Process coordinates in random order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                a_test = a.copy()
                a_test[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_test[mask] = a_test[mask] * ((1.0 - new_val) / denom)
                a_test = np.maximum(a_test, 0.0)
                s_test = a_test.sum()
                if s_test < 1e-15:
                    continue
                a_test /= s_test
                obj_test = fast_convolve(a_test).max()
                if obj_test < best_obj - 1e-13:
                    best_obj = obj_test
                    best_a = a_test.copy()
                    a = a_test.copy()
                    old_val = new_val
                    denom = 1.0 - old_val
                    pass_improved = True

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_a, best_obj


def generate_ini_windowed(n, window_type='hamming'):
    """Generate a windowed initialization."""
    j = np.arange(n)
    x = j / max(n - 1, 1)
    if window_type == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    elif window_type == 'hanning':
        a = 0.5 - 0.5 * np.cos(2 * np.pi * x)
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif window_type == 'gaussian':
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    elif window_type == 'triangular':
        center = (n - 1) / 2.0
        a = 1.0 - np.abs(j - center) / (center + 1)
    elif window_type == 'raised_cosine':
        # Broader peak
        center = (n - 1) / 2.0
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (j - center) / n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def generate_ini_concatenated(n, base_n=128):
    """
    Generate initialization by tiling a short optimal-like pattern.
    Uses a pattern that has low autocorrelation peaks.
    """
    # Start with a pattern that approximately minimizes autocorrelation
    # Use a pseudo-random but structured pattern
    rng = np.random.default_rng(42)
    base = rng.dirichlet(np.ones(base_n) * 0.5)
    # Tile and blend
    n_tiles = n // base_n + 1
    a = np.tile(base, n_tiles)[:n]
    # Smooth the tile boundaries
    kernel_len = min(16, base_n // 4)
    if kernel_len > 1:
        kernel = np.ones(kernel_len) / kernel_len
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Generate diverse initializations (windowed, parent, structured)
    2. Quick evaluation to pick best starting point
    3. Smoothed subgradient with Nesterov and temperature annealing
    4. Length exploration via interpolation
    5. Coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Generate and evaluate initializations ===
    print("=== Phase 1: Initialization ===")
    n_parent = len(parent) if parent is not None else 768

    init_candidates = []

    # Parent (if available)
    if parent is not None:
        init_candidates.append(('parent', parent))

    # Windowed inits at various lengths
    for n in [512, 768, 1024, 1536]:
        for wt in ['hamming', 'hanning', 'blackman', 'gaussian', 'triangular', 'raised_cosine']:
            init_candidates.append((f'win_{wt}_n{n}', generate_ini_windowed(n, wt)))

    # Concatenated pattern
    for n in [768, 1024]:
        init_candidates.append((f'concat_n{n}', generate_ini_concatenated(n)))

    # Evaluate all initializations
    print(f"  Evaluating {len(init_candidates)} initializations...")
    for label, a in init_candidates:
        try_seq(a, label)

    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"  Best init score: {best_score:.6f}")

    # === Phase 2: Main optimization on best init ===
    print(f"\n=== Phase 2: Smoothed subgradient (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_main = min(rem, 500)
        print(f"  Duration: {t_main:.0f}s, n={n_best}")
        a_opt, obj = optimized_smoothed_grad(
            best_seq, time_limit=t_main,
            T0=2.0, T_end=40.0,
            lr0=0.015, momentum=0.75, top_k=10,
            verbose=True
        )
        try_seq(a_opt, "smoothed_main")

    # === Phase 3: Second pass with different params ===
    print(f"\n=== Phase 3: Second pass (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 20:
        t2 = min(rem, 200)
        a_opt, obj = optimized_smoothed_grad(
            best_seq, time_limit=t2,
            T0=5.0, T_end=50.0,
            lr0=0.008, momentum=0.85, top_k=5,
            verbose=False
        )
        try_seq(a_opt, "smoothed_pass2")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length exploration (best={best_score:.6f}) ===")
    base = best_seq
    for target_n in [640, 896, 1024, 1280, 1536, 2048]:
        rem = deadline - time.monotonic() - 200
        if rem < 15:
            break
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")
        t_alloc = min(rem * 0.2, 60)
        if t_alloc > 10:
            a_opt, obj = optimized_smoothed_grad(
                f_interp, time_limit=t_alloc,
                T0=3.0, T_end=30.0,
                lr0=0.01, momentum=0.8, top_k=10,
                verbose=False
            )
            try_seq(a_opt, f"opt_interp_n{target_n}")
        base = best_seq  # Always interpolate from current best

    # === Phase 5: Multi-start perturbation ===
    print(f"\n=== Phase 5: Multi-start (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    for p in range(6):
        rem = deadline - time.monotonic() - 150
        if rem < 15:
            break
        t_alloc = min(rem * 0.25, 50)

        if p == 0:
            f_init = best_seq.copy()
        elif p == 1:
            noise = rng.standard_normal(n_best) * 0.01
            f_init = project_simplex(best_seq + noise)
        elif p == 2:
            f_init = 0.8 * best_seq + 0.2 * np.ones(n_best) / n_best
            f_init = project_simplex(f_init)
        elif p == 3:
            f_init = 0.7 * best_seq + 0.3 * generate_ini_windowed(n_best, 'gaussian')
            f_init = project_simplex(f_init)
        elif p == 4:
            # Reverse
            f_init = project_simplex(best_seq[::-1])
        else:
            noise = rng.standard_normal(n_best) * 0.02
            f_init = project_simplex(best_seq + noise)

        a_opt, obj = optimized_smoothed_grad(
            f_init, time_limit=t_alloc,
            T0=3.0, T_end=35.0,
            lr0=0.01, momentum=0.8, top_k=8,
            verbose=False
        )
        try_seq(a_opt, f"multi_{p}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: CD Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 80
    if rem > 15:
        t_cd = min(rem, 120)
        print(f"  Duration: {t_cd:.0f}s")
        a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_cd)
        try_seq(a_opt, "cd_polish")

    # === Phase 7: Final fine-tuned subgradient ===
    print(f"\n=== Phase 7: Final (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_final = min(rem, 80)
        a_opt, obj = optimized_smoothed_grad(
            best_seq, time_limit=t_final,
            T0=10.0, T_end=60.0,
            lr0=0.003, momentum=0.9, top_k=5,
            verbose=False
        )
        try_seq(a_opt, "final_sm")

        # Another CD pass
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
