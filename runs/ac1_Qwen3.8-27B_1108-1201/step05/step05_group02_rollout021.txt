def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

However, we can use a **semi-infinite programming** approach or **successive convex approximation**:
1. Start with a feasible point
2. Linearize the constraints at the current point
3. Solve the resulting LP
4. Update

This is essentially what the LP-FW approach does! The previous code's `solve_lp_min_peak` is solving exactly this LP at each iteration.

So the LP-FW approach is actually the right framework. The question is how to make it more effective.

**Improvements to LP-FW:**
1. Better initialization (using spectral methods or known good patterns)
2. Adaptive step size
3. Multiple LP formulations (sometimes max-sum works better than min-peak)
4. Periodic restarts with perturbations
5. Multi-length search

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Soft-max gradient descent (smooth approximation of the max)
2. LP-based Frank-Wolfe with adaptive step
3. Spectral initialization
4. Multi-length exploration
5. Final coordinate descent polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    # Clamp tiny negative values from FFT numerical error
    result[result < 0] = 0
    return result


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def soft_max_gradient(a, T):
    """
    Compute the gradient of (1/T)*log(sum_k exp(T*(a*a)[k])) w.r.t. a.
    This is a smooth approximation of the subgradient of max_k (a*a)[k].
    Returns (gradient, weights) where weights are the softmax probabilities.
    """
    n = len(a)
    conv = fast_convolve(a)  # length 2n-1
    
    # Compute softmax weights with numerical stability
    scaled = T * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    w = exp_vals / exp_vals.sum()
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i in [0, n-1])
    # This is 2 * cross_correlation(w, a)[i]
    # Using FFT: cross_correlation(w, a) = ifft(FFT(w) * conj(FFT(a)))
    # Need FFT size >= (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    
    # corr[i] = sum_k w[k] * a[k-i] for i = 0, ..., n-1
    g = 2.0 * corr[:n]
    return g, w


def adam_soft_max_optimize(f0, time_limit, T_init=50.0, T_max=5000.0,
                           lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8,
                           verbose=False):
    """
    Adam optimizer using soft-max gradient for the max-of-convolution objective.
    Operates on the simplex (sum=1, a>=0).
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(f0)) / len(f0)
    else:
        a = a / s
    n = len(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0  # Time step
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    
    T = T_init
    cur_lr = lr
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        g, w = soft_max_gradient(a, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Update in the "dual" space, then project
        a_raw = a - cur_lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = project_simplex(a_raw)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new
        
        # Anneal temperature
        T *= 1.005
        T = min(T, T_max)
        
        # Decay learning rate
        cur_lr *= 0.9999
        
        # Periodic restart
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            cur_lr = lr * 0.7
            T = T_init * (1 + 0.5 * (no_improve // 500))
            no_improve = 0
        
        if verbose and t % 1000 == 0:
            print(f"    SoftMax iter {t}: obj={best_obj:.10f}, T={T:.1f}, no_improve={no_improve}")
        
        # Early stop if very stuck
        if no_improve > 3000:
            break
    
    return best_a, best_obj


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def lp_fw_optimize(f0, time_limit, verbose=False, step_sizes=None):
    """LP-based Frank-Wolfe with adaptive step and line search."""
    if step_sizes is None:
        step_sizes = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            # Fallback: use the current f
            g = f.copy()
        
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in step_sizes:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new

            if M < best_obj - 1e-15:

                best_f = f.copy()

            no_improve = 0
    
    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.005, momentum=0.93, top_k=50,
                         temp=0.0001, verbose=False):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]
        
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new
        
        cur_lr *= 0.9998
        cur_temp *= 1.003
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent polish targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        
        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            # Try modifying each coordinate that contributes to this peak
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-12:
                    continue
                
                best_j_val = old_val
                best_j_obj = best_obj
                multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]
                
                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    ss = f_trial.sum()
                    if ss < 1e-15:
                        continue
                    f_trial /= ss
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val
                
                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f /= f.sum()
                    
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                # Random coordinate exploration
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
                found = False
                for _ in range(30):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.4, 0.6, 1.5, 2.0]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.5:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        ss = f_trial.sum()
                        if ss < 1e-15:
                            continue
                        f_trial /= ss
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            found = True
                            improved = True
                            break
                    if found:
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj
