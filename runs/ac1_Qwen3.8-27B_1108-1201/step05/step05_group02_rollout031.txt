import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_min_peak(f, n):
    """LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

import numpy as np

def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_min_peak(f, n):
    """LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        f_new = f * np.exp(-cur_lr * sg)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = f / f.sum()
        else:
            f_new = f_new / f_new_sum

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    MD it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        cur_lr *= lr_decay
        cur_temp *= temp_growth

        if no_improve > 400:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            cur_lr = max(lr_init * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj


def cd_optimize(f0, time_limit, verbose=False):
    """Golden-section coordinate descent targeting peak contributors."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    passes = 0
    no_improve = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        top_k = min(4, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-14:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue

                lo = 0.0
                hi = min(0.4, old_val * 3.0)
                if hi - lo < 1e-10:
                    continue

                a1 = lo + (1.0 - gr) * (hi - lo)
                a2 = lo + gr * (hi - lo)

                f_trial = f.copy()
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= (1.0 - a1) / denom
                f_trial[j] = a1
                f1 = fast_convolve(f_trial).max()

                f_trial[mask] *= (1.0 - a2) / denom
                f_trial[j] = a2
                f2 = fast_convolve(f_trial).max()

                for _ in range(7):
                    if f1 < f2:
                        hi = a2
                        a2 = a1
                        f2 = f1
                        a1 = lo + (1.0 - gr) * (hi - lo)
                        f_trial[mask] *= (1.0 - a1) / denom
                        f_trial[j] = a1
                        f1 = fast_convolve(f_trial).max()
                    else:
                        lo = a1
                        a1 = a2
                        f1 = f2
                        a2 = lo + gr * (hi - lo)
                        f_trial[mask] *= (1.0 - a2) / denom
                        f_trial[j] = a2
                        f2 = fast_convolve(f_trial).max()

                if f1 < f2:
                    best_cv = f1
                    best_jv = a1
                else:
                    best_cv = f2
                    best_jv = a2

                if best_cv < best_obj - 1e-13 and best_jv != old_val:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= (1.0 - best_jv) / denom
                    f[j] = best_jv
                    best_obj = best_cv
                    best_f = f.copy()
                    conv = fast_convolve(f)
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                rng = np.random.default_rng(passes * 104729)
                for _ in range(min(40, n // 4)):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-12:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    break_found = False
                    for mult in [0.3, 0.5, 0.7, 1.3, 1.6, 2.2, 3.0]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.4:
                            continue
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial = f.copy()
                        f_trial[mask] *= (1.0 - new_val) / denom
                        f_trial[j] = new_val
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-13:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break_found = True
                            break
                    if break_found:
                        break
            if no_improve > 12:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def anneal_optimize(f0, time_limit, n_steps_per_iter=5, init_temp=0.003,
                    cool_rate=0.9995, verbose=False):
    """Simulated annealing with coordinate mutations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_obj = best_obj
    temp = init_temp
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        for _ in range(n_steps_per_iter):
            n_mut = rng.integers(1, 5)
            idxs = rng.integers(0, n, size=n_mut)
            f_trial = f.copy()
            for idx in idxs:
                old_val = f_trial[idx]
                if old_val < 1e-14:
                    continue
                factor = rng.uniform(0.4, 2.5)
                new_val = old_val * factor
                if new_val > 0.4:
                    new_val = 0.4
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                f_trial[mask] *= (1.0 - new_val) / denom
                f_trial[idx] = new_val

            trial_obj = fast_convolve(f_trial).max()
            delta = trial_obj - cur_obj
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                f = f_trial
                cur_obj = trial_obj
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_f = f.copy()
                    if verbose and it % 200 == 0:
                        print(f"    SA it={it}: obj={best_obj:.10f}, T={temp:.6f}")

        temp *= cool_rate

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def evaluate_and_track(a, best_seq, best_score, label):
    """Evaluate sequence and return updated best."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return best_seq, best_score
    a_norm = a / a.sum()
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_seq = a_norm.copy()
        best_score = score
        print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: Enhanced FW, KL Mirror Descent, golden-section CD,
    simulated annealing, and multi-length exploration to minimize the autocorrelation
    peak objective.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.10f}")

    # === Phase 1: Enhanced FW on parent (strongest proven method) ===
    print("\n=== Phase 1: Enhanced FW on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(350, rem * 0.30)
    if t1 > 50:
        print(f"  n={len(parent)}, t={t1:.0f}s")
        a_opt, _ = fw_optimize(parent, time_limit=t1, verbose=True, use_shrink=True)
        best_seq, best_score = evaluate_and_track(a_opt, best_seq, best_score, "fw_parent")

    # === Phase 2: KL Mirror Descent (different optimization landscape) ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.10f}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic()
    t2 = min(250, rem * 0.20)
    if t2 > 40:
        print(f"  n={len(base)}, t={t2:.0f}s")
        a_opt, _ = md_optimize(base, time_limit=t2, lr_init=0.004, lr_decay=0.9998,
                               top_k=60, temp_init=0.00004, temp_growth=1.004, verbose=True)
        best_seq, best_score = evaluate_and_track(a_opt, best_seq, best_score, "md_main")

        rem = deadline - time.monotonic()
        t2b = min(120, rem * 0.12)
        if t2b > 30:
            a_opt, _ = md_optimize(base, time_limit=t2b, lr_init=0.0015, lr_decay=0.9999,
                                   top_k=30, temp_init=0.00001, temp_growth=1.002)
            best_seq, best_score = evaluate_and_track(a_opt, best_seq, best_score, "md_fine")

    # === Phase 3: Golden-section Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(250, rem * 0.20)
    if t3 > 40:
        print(f"  n={len(best_seq)}, t={t3:.0f}s")
        a_opt, _ = cd_optimize(best_seq, time_limit=t3, verbose=True)
        best_seq, best_score = evaluate_and_track(a_opt, best_seq, best_score, "cd_main")

    # === Phase 4: Simulated Annealing (escape local minima) ===
    print(f"\n=== Phase 4: Annealing (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(150, rem * 0.12)
    if t4 > 30:
        print(f"  n={len(best_seq)}, t={t4:.0f}s")
        a_opt, _ = anneal_optimize(best_seq, time_limit=t4, n_steps_per_iter=3,
                                   init_temp=0.004, cool_rate=0.9993, verbose=True)
        best_seq, best_score = evaluate_and_track(a_opt, best_seq, best_score, " anneal")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.10f}) ===")
    base_n = len(best_seq)
    candidate_lengths = []
    for delta in [-96, -64, -32, 32, 64, 96, 128]:
        ln = base_n + delta
        if 256 <= ln <= 2048:
            candidate_lengths.append(ln)

    for target_n in candidate_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_alloc = min(rem * 0.35, 100)
        if t_alloc < 25:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        print(f"  FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = fw_optimize(f_interp, time_limit=t_alloc, use_shrink=True)
        best_seq, best_score = evaluate_and_track(a_opt, best_seq, best_score, f"fw_n{target_n}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        a_opt, _ = fw_optimize(best_seq, time_limit=min(rem * 0.35, 100), use_shrink=True)
        best_seq, best_score = evaluate_and_track(a_opt, best_seq, best_score, "polish_fw")

        rem = deadline - time.monotonic()
        if rem > 20:
            a_opt, _ = cd_optimize(best_seq, time_limit=min(rem * 0.3, 60))
            best_seq, best_score = evaluate_and_track(a_opt, best_seq, best_score, "polish_cd")

        rem = deadline - time.monotonic()
        if rem > 30:
            a_opt, _ = md_optimize(best_seq, time_limit=min(rem * 0.4, 80),
                                   lr_init=0.001, lr_decay=0.9999, top_k=20,
                                   temp_init=0.00001, temp_growth=1.002)
            best_seq, best_score = evaluate_and_track(a_opt, best_seq, best_score, "polish_md")

        # Random restarts from best
        for p in range(4):
            rem = deadline - time.monotonic() - 5
            if rem < 12:
                break
            noise = rng.standard_normal(len(best_seq)) * (0.0008 + 0.0004 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = fw_optimize(f_pert, time_limit=min(rem, 30), use_shrink=True)
            best_seq, best_score = evaluate_and_track(a_opt, best_seq, best_score, f"restart_{p}")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = np.ones(512) / 512
            best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
