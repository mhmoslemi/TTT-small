def subgrad_multi_peak(a, top_k=10):
    """Compute weighted subgradient from top-k peaks of the auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    order = np.argsort(pv)[::-1]
    weights = np.zeros(len(peaks))
    for rank, idx in enumerate(order):
        weights[idx] = 1.0 / (rank + 1) ** 1.5
    ws = weights.sum()
    if ws > 0:
        weights /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        sg[idx_arr] += weights[i] * a[k - idx_arr]
    return sg, conv

def solve_lp_oracle(f, M, method='highs'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    # Build the constraint matrix efficiently
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method,
                      options={'presolve': True, 'max_time': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if it % 100 == 0:
            print(f"    AS-Newton it={it}, obj={best_obj:.10f}")

        conv = fast_convolve(a)
        M = conv.max()
        support = np.where(a > 1e-10)[0]
        if len(support) == 0:
            break

        # Find active peaks (within 1% of max)
        threshold = M * (1.0 - 0.001)
        active_peaks = np.where(conv >= threshold)[0]

        # If only one active peak, use simple gradient step
        if len(active_peaks) <= 1:
            break

        # Build linear system for KKT on support
        # For j in support: sum_{k in active} mu_k * a[k-j] = nu
        # We solve for mu (and nu) by least squares
        # A_system[i, k_idx] = a[active_peaks[k_idx] - support[i]] (if valid)
        # b_system[i] = nu (unknown, so we add it as a column)

        m = len(support)
        p = len(active_peaks)

        # Build matrix: A_sys is m x (p+1), last column is ones (for nu)
        A_sys = np.zeros((m, p + 1))
        for i, j in enumerate(support):
            A_sys[i, -1] = 1.0  # nu column
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = a[idx]

        # We want A_sys @ [mu; nu] = 0 (homogeneous, since RHS is 0)
        # Actually, the KKT condition is: sum_k mu_k * a[k-j] = nu
        # So: sum_k mu_k * a[k-j] - nu = 0
        # This gives: A_sys @ [mu; -nu] = 0
        # We want a non-trivial solution, so we use SVD
        try:
            U, S_vals, Vt = np.linalg.svd(A_sys)
            # Smallest singular vector
            sol = Vt[-1, :]
            mu_vals = sol[:p]
            nu_val = -sol[p]
        except np.linalg.LinAlgError:
            break

        # Ensure mu >= 0 (dual feasibility)
        mu_vals = np.maximum(mu_vals, 0)
        mu_sum = mu_vals.sum()
        if mu_sum < 1e-15:
            break
        mu_vals /= mu_sum

        # Use mu as weights for a gradient step
        # Gradient direction: -sum_k mu_k * a[k-j]
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu_vals[k_idx] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= mu_vals[k_idx] * a[k - idx_arr]

        # Take a small step
        best_candidate = a
        best_val = M
        for step in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            candidate = project_simplex(a + step * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()

        # Check convergence
        if abs(M - best_val) < 1e-14:
            # Check if we should continue with tighter tolerance
            if best_obj < M * (1.0 + 1e-10):
                break

    return best_a, best_obj


def softmax_smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.01):
    """Smoothed gradient using softmax-weighted peaks for better convergence."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init

    v = np.zeros(n)
    mom = 0.92
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        # Softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient
        grad = 2.0 * hankel_mv(p, a)

        # Momentum step with backtracking
        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.9

        # Adapt
        if no_improve > 20:
            T *= 1.08
            lr *= 0.96
        if no_improve > 50:
            T *= 0.85
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 200) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved. Returns (best_seq, best_score)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score, iters = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score, iters)


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search exploiting problem structure.
    Uses symmetry, active-set Newton, and adaptive subgradient methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    total_iters = 0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Symmetry check and quick optimization (~100s) ===
    print("\n=== Phase 1: Symmetry & Quick Opt ===")
    t_start = time.monotonic()

    # Try symmetrized version
    sym_parent = (parent + parent[::-1]) / 2.0
    sym_parent = sym_parent / sym_parent.sum()
    (best_seq, best_score, total_iters) = try_and_update(sym_parent, (best_seq, best_score, total_iters), "sym_parent")

    # Quick PGD on parent length
    remaining = deadline - time.monotonic() - 1400
    if remaining > 10:
        t_alloc = min(remaining, 60)
        a_opt, _ = pgd_momentum(best_seq, time_limit=t_alloc, top_k=20, lr_init=0.03)
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), "pgd_quick")

    # Quick FW-LP
    remaining = deadline - time.monotonic() - 1300
    if remaining > 10:
        t_alloc = min(remaining, 50)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), "fw_quick")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Multi-n exploration (~300s) ===
    print("\n=== Phase 2: Multi-n Exploration ===")
    t_start = time.monotonic()

    n_best = len(best_seq)
    candidate_ns = [n_best]
    for n_try in [n_best // 2, n_best * 2, 384, 512, 640, 768, 896, 1024]:
        if n_try not in candidate_ns and 128 <= n_try <= 2048:
            candidate_ns.append(n_try)

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 800
        if remaining < 30:
            break

        if n_try == n_best:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        t_alloc = min(remaining, 80)
        print(f"  Exploring n={n_try}, t={t_alloc:.0f}s")

        # PGD
        a_opt, _ = pgd_momentum(start, time_limit=t_alloc * 0.4, top_k=15, lr_init=0.02)
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), f"pgd_n{n_try}")

        # FW-LP
        remaining = deadline - time.monotonic() - 700
        if remaining > 20:
            a_opt2, _ = fw_lp_optimize(a_opt, time_limit=min(remaining, t_alloc * 0.4))
            (best_seq, best_score, total_iters) = try_and_update(a_opt2, (best_seq, best_score, total_iters), f"fw_n{n_try}")

        # Smoothed grad
        remaining = deadline - time.monotonic() - 650
        if remaining > 15:
            a_opt3, _ = softmax_smoothed_grad(a_opt, time_limit=min(remaining, t_alloc * 0.2), T_init=0.001)
            (best_seq, best_score, total_iters) = try_and_update(a_opt3, (best_seq, best_score, total_iters), f"sg_n{n_try}")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Deep FW-LP refinement (~400s) ===
    print("\n=== Phase 3: Deep FW-LP ===")
    t_start = time.monotonic()

    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 500
    if remaining > 30:
        t_alloc = min(remaining, 300)
        print(f"  Deep FW-LP n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), "deep_fw")

    # Perturbed FW restarts
    for scale in [0.003, 0.007, 0.015]:
        remaining = deadline - time.monotonic() - 350
        if remaining < 20:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining, 80)
        a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), f"fw_pert_{scale}")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Active-set Newton + Smoothed gradient (~350s) ===
    print("\n=== Phase 4: Active-Set & Smoothed ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Active-set Newton
    remaining = deadline - time.monotonic() - 300
    if remaining > 20:
        t_alloc = min(remaining, 150)
        print(f"  Active-set Newton, t={t_alloc:.0f}s")
        a_opt, _ = active_set_newton(best_seq, time_limit=t_alloc)
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), "as_newton")

    # Smoothed gradient with different temperatures
    for T_init in [0.0005, 0.002, 0.005]:
        remaining = deadline - time.monotonic() - 200
        if remaining < 15:
            break
        t_alloc = min(remaining, 100)
        a_opt, _ = softmax_smoothed_grad(best_seq, time_limit=t_alloc, T_init=T_init, lr_init=0.008)
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), f"smooth_T{T_init}")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Intensive local refinement (~250s) ===
    print("\n=== Phase 5: Local Refinement ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Multiple PGD configs
    for top_k, lr_init, mom_init in [(10, 0.01, 0.95), (30, 0.005, 0.93), (50, 0.003, 0.96)]:
        remaining = deadline - time.monotonic() - 150
        if remaining < 15:
            break
        t_alloc = min(remaining, 80)
        a_opt, _ = pgd_momentum(best_seq, time_limit=t_alloc, top_k=top_k, lr_init=lr_init, mom_init=mom_init)
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), f"pgd_k{top_k}")

    # Active-set Newton again (on refined solution)
    remaining = deadline - time.monotonic() - 100
    if remaining > 15:
        t_alloc = min(remaining, 60)
        a_opt, _ = active_set_newton(best_seq, time_limit=t_alloc)
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), "as_newton2")

    # Fine smoothed gradient
    remaining = deadline - time.monotonic() - 60
    if remaining > 10:
        t_alloc = min(remaining, 40)
        a_opt, _ = softmax_smoothed_grad(best_seq, time_limit=t_alloc, T_init=0.0003, lr_init=0.005)
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), "fine_smooth")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Final polishing (~150s) ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Tiny perturbations with quick optimization
    for _ in range(5):
        remaining = deadline - time.monotonic() - 30
        if remaining < 10:
            break
        noise = rng.standard_normal(n_best) * 0.001
        f_pert = project_simplex(best_seq + noise)
        # Quick PGD
        t_alloc = min(remaining, 20)
        a_opt, _ = pgd_momentum(f_pert, time_limit=t_alloc, top_k=15, lr_init=0.01)
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), "perturb")

        # Active-set Newton on the perturbed result
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            a_opt2, _ = active_set_newton(a_opt, time_limit=min(remaining, 15))
            (best_seq, best_score, total_iters) = try_and_update(a_opt2, (best_seq, best_score, total_iters), "perturb_as")

    # One more FW-LP
    remaining = deadline - time.monotonic() - 15
    if remaining > 10:
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(remaining, 30))
        (best_seq, best_score, total_iters) = try_and_update(a_opt, (best_seq, best_score, total_iters), "final_fw")

    # Symmetry final check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    (best_seq, best_score, total_iters) = try_and_update(sym_final, (best_seq, best_score, total_iters), "final_sym")

    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
