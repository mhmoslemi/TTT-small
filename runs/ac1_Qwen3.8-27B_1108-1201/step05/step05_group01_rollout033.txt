from scipy.signal import correlate
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Compute gradient of log-sum-exp of auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    # p_k = exp(conv[k] / T) / sum exp(conv[j] / T)
    scaled = conv / temperature
    scaled -= scaled.max()  # for numerical stability
    p = np.exp(scaled)
    p /= p.sum()
    # grad[i] = 2 * sum_j a[j] * p[i+j]
    # This is correlate(p, a, mode='valid')
    grad = 2.0 * correlate(p, a, mode='valid')[:n]
    return grad, conv.max()

        b -= b.max()  # numerical stability
        a = np.exp(b)
        a /= a.sum()
        
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            
        T *= T_decay
        if T < 0.001:
            T = 0.001
    
    return best_a, best_obj

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using Adam-softmax with temperature
annealing, multi-start exploration, and coordinate descent polishing.
"""

import numpy as np
from scipy.signal import correlate
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_max_grad(a, T):
    """Compute gradient of log-sum-exp of auto-convolution w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    # grad[i] = 2 * sum_j a[j] * p[i+j] = 2 * correlate(p, a, 'valid')[:n]
    grad = 2.0 * correlate(p, a, mode='valid')[:n]
    return grad

def adam_softmax(a0, time_limit, lr=0.5, T_init=0.05, T_min=0.005, T_decay=0.9997, 
                 beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """Adam optimization with softmax parameterization and temperature annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-30)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    b = np.log(a)
    b -= b.mean()  # center for numerical stability
    
    T = T_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        grad_a = smoothed_max_grad(a, T)
        
        # Transform to b-space
        mean_grad = np.dot(a, grad_a)
        grad_b = a * (grad_a - mean_grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * grad_b ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Compute a from b
        b_max = b.max()
        a = np.exp(b - b_max)
        a_sum = a.sum()
        if a_sum < 1e-30:
            a = np.ones(n) / n
            b = np.zeros(n)
        else:
            a = a / a_sum
        
        # Evaluate
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and t % 1000 == 0:
                print(f"    iter {t}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
        
        # Temperature annealing
        T = max(T_min, T * T_decay)
        
        # If stuck, briefly increase temperature
        if no_improve > 5000:
            T = min(T * 5.0, 0.5)
            no_improve = 0
            
        # Periodic re-evaluation
        if t % 5000 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 1:
                break
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, n_iter_per_coord=8):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old = a[j]
            if old < 1e-12:
                continue
            denom = 1.0 - old
            if denom < 1e-10:
                continue
            
            # Try a few multipliers
            best_mult = 1.0
            best_val = best_obj
            for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                new_j = old * mult
                if new_j > 1.0:
                    continue
                ft = a.copy()
                ft[j] = new_j
                rest_sum = 1.0 - new_j
                if rest_sum < 1e-12:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = a[mask] * (rest_sum / denom)
                val = fast_convolve(ft).max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_mult = mult
            
            if best_mult != 1.0:
                new_j = old * best_mult
                ft = a.copy()
                ft[j] = new_j
                rest_sum = 1.0 - new_j
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = a[mask] * (rest_sum / denom)
                a = ft
                improved = True
                if best_val < best_obj - 1e-12:
                    best_obj = best_val
                    best_a = ft.copy()
        
        if not improved:
            break
    
    return best_a, best_obj

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using Adam optimizer with
temperature-annealed smoothed max, multi-start exploration, and
coordinate descent polishing. Uses fast FFT-based convolution and
scipy.signal.correlate for efficient gradient computation.
"""

import numpy as np
from scipy.signal import correlate
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(v - u * 0 + u)[::-1]  # just cumsum of sorted
    # Redo properly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, T):
    """Compute gradient of log-sum-exp of auto-convolution w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    # grad[i] = 2 * sum_j a[j] * p[i+j] for i=0..n-1
    # This is cross-correlation of p and a in 'valid' mode
    grad = 2.0 * correlate(p, a, mode='valid')[:n]
    return grad


def topk_grad(a, T, k_ratio=0.1):
    """Compute gradient focusing on top-k peaks (sharper than full softmax)."""
    n = len(a)
    conv = fast_convolve(a)
    k = max(1, int(len(conv) * k_ratio))
    top_indices = np.argsort(conv)[-k:]
    p = np.zeros_like(conv)
    peak_vals = conv[top_indices]
    scaled = peak_vals / T
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    for idx, w in zip(top_indices, weights):
        p[idx] = w
    grad = 2.0 * correlate(p, a, mode='valid')[:n]
    return grad


def adam_optimize(a0, time_limit, lr=0.05, T_init=0.05, T_min=0.003,
                  T_decay=0.9998, beta1=0.9, beta2=0.999, eps=1e-8,
                  use_topk=False, k_ratio=0.1, verbose=False, label=""):
    """Adam optimizer with smoothed max objective and simplex projection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    T = T_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1

        if use_topk:
            grad_a = topk_grad(a, T, k_ratio)
        else:
            grad_a = smooth_grad(a, T)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_a
        v = beta2 * v + (1 - beta2) * grad_a ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        # Evaluate
        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and t % 5000 == 0:
                rem = time_limit - (time.monotonic() - t0)
                print(f"    {label} iter {t}: obj={best_obj:.10f}, T={T:.6f}, rem={rem:.0f}s")
        else:
            no_improve += 1

        a = a_new
        T = max(T_min, T * T_decay)

        # If stuck, temporarily increase temperature
        if no_improve > 4000:
            T = min(max(T * 3.0, 0.05), 0.5)
            no_improve = 0

        if t % 10000 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 1:
                break

    return best_a, best_obj


def adam_softmax_optimize(a0, time_limit, lr=0.5, T_init=0.05, T_min=0.003,
                          T_decay=0.9998, beta1=0.9, beta2=0.999, eps=1e-8,
                          verbose=False, label=""):
    """Adam optimizer with softmax parameterization (avoids projection)."""
    a = np.maximum(a0.astype(np.float64), 1e-30)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    b = np.log(a)
    b -= b.mean()

    T = T_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1

        grad_a = smooth_grad(a, T)

        # Transform gradient from a-space to b-space
        mean_grad = np.dot(a, grad_a)
        grad_b = a * (grad_a - mean_grad)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * grad_b ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        b = b - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Compute a from b
        b_shift = b - b.max()
        a = np.exp(b_shift)
        a_sum = a.sum()
        if a_sum < 1e-30:
            a = np.ones(n) / n
            b = np.zeros(n)
        else:
            a = a / a_sum

        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and t % 5000 == 0:
                rem = time_limit - (time.monotonic() - t0)
                print(f"    {label} iter {t}: obj={best_obj:.10f}, T={T:.6f}, rem={rem:.0f}s")
        else:
            no_improve += 1

        T = max(T_min, T * T_decay)

        if no_improve > 5000:
            T = min(max(T * 4.0, 0.1), 0.5)
            no_improve = 0

        if t % 10000 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 1:
                break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Cyclic coordinate descent with multiplier search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    cur_obj = best_obj
    t0 = time.monotonic()
    passes = 0

    multipliers = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old = a[j]
            if old < 1e-13:
                continue
            denom = 1.0 - old
            if denom < 1e-10:
                continue

            best_mult = 1.0
            best_val = cur_obj

            for mult in multipliers:
                new_j = old * mult
                if new_j >= 1.0 or new_j < 0:
                    continue
                ft = a.copy()
                ft[j] = new_j
                rest_sum = 1.0 - new_j
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = a[mask] * (rest_sum / denom)
                val = fast_convolve(ft).max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_mult = mult

            if best_mult != 1.0:
                new_j = old * best_mult
                ft = a.copy()
                ft[j] = new_j
                rest_sum = 1.0 - new_j
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = a[mask] * (rest_sum / denom)
                a = ft
                cur_obj = fast_convolve(a).max()
                pass_improved = True
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()

        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

        if not pass_improved:
            # Try finer multipliers around 1.0
            fine_mults = [0.95, 0.97, 0.99, 1.01, 1.03, 1.05]
            fine_improved = False
            for j in range(n):
                if time.monotonic() - t0 > time_limit:
                    break
                old = a[j]
                if old < 1e-13:
                    continue
                denom = 1.0 - old
                if denom < 1e-10:
                    continue
                bestMult = 1.0
                bestVal = cur_obj
                for mult in fine_mults:
                    new_j = old * mult
                    if new_j >= 1.0 or new_j < 0:
                        continue
                    ft = a.copy()
                    ft[j] = new_j
                    rest_sum = 1.0 - new_j
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    ft[mask] = a[mask] * (rest_sum / denom)
                    val = fast_convolve(ft).max()
                    if val < bestVal - 1e-13:
                        bestVal = val
                        bestMult = mult
                if bestMult != 1.0:
                    new_j = old * bestMult
                    ft = a.copy()
                    ft[j] = new_j
                    rest_sum = 1.0 - new_j
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    ft[mask] = a[mask] * (rest_sum / denom)
                    a = ft
                    cur_obj = fast_convolve(a).max()
                    fine_improved = True
                    if cur_obj < best_obj - 1e-12:
                        best_obj = cur_obj
                        best_a = a.copy()

            if not fine_improved:
                break

    return best_a, best_obj


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

            elif p == 5:
                noise = rng.standard_normal(n_best) * 0.01
                f_init = project_simplex(best_seq + noise)
            elif p == 6:
                # Shift the sequence
                shift = rng.integers(1, n_best // 4)
                f_init = np.roll(best_seq, shift)
                f_init = project_simplex(f_init)
            else:
                # Blend with uniform
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)

            # Alternate between direct and softmax
            if p % 2 == 0:
                a_opt, _ = adam_optimize(f_init, time_limit=t5, lr=0.04, T_init=0.03,
                                          T_min=0.003, T_decay=0.9998, verbose=False, label=f"ms_{p}")
            else:
                a_opt, _ = adam_softmax_optimize(f_init, time_limit=t5, lr=0.4, T_init=0.03,
                                                  T_min=0.003, T_decay=0.9998, verbose=False, label=f"ms_{p}")
            _, best_score, best_seq = evaluate_and_track(a_opt, best_score, best_seq, f"multi_{p}")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        target_lengths = [512, 640, 896, 1024, 1280]
        for target_n in target_lengths:
            if target_n == len(best_seq):
                continue
            rem = deadline - time.monotonic()
            t6 = min(rem * 0.1, 100)
            if t6 < 15:
                break
            f_interp = interpolate_to_length(best_seq, target_n)
            print(f"  n={target_n}, t={t6:.0f}s")
            a_opt, _ = adam_optimize(f_interp, time_limit=t6, lr=0.05, T_init=0.04,
                                      T_min=0.003, T_decay=0.9998, verbose=False, label=f"len_{target_n}")
            _, best_score, best_seq = evaluate_and_track(a_opt, best_score, best_seq, f"interp_n{target_n}")
            # Quick CD on the interpolated result
            if np.isfinite(best_score) and target_n == len(best_seq):
                rem = deadline - time.monotonic()
                t6b = min(rem * 0.05, 40)
                if t6b > 10:
                    a_opt, _ = coordinate_descent(best_seq, time_limit=t6b)
                    _, best_score, best_seq = evaluate_and_track(a_opt, best_score, best_seq, f"cd_n{target_n}")

    # === Phase 7: Final low-temperature polish ===
    print(f"\n=== Phase 7: Final low-T polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t7 = min(rem * 0.25, 200)
        if t7 > 20:
            print(f"  Adam direct, T_min=0.001, t={t7:.0f}s")
            a_opt, _ = adam_optimize(best_seq, time_limit=t7, lr=0.02, T_init=0.01,
                                      T_min=0.001, T_decay=0.9999,
                                      beta1=0.95, beta2=0.999, verbose=True, label="final")
            _, best_score, best_seq = evaluate_and_track(a_opt, best_score, best_seq, "final_adam")

        rem = deadline - time.monotonic()
        t7b = min(rem * 0.2, 100)
        if t7b > 15:
            print(f"  Final CD, t={t7b:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t7b)
            _, best_score, best_seq = evaluate_and_track(a_opt, best_score, best_seq, "final_cd")

    # === Phase 8: Very fine perturbations ===
    print(f"\n=== Phase 8: Fine perturbations (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for _ in range(5):
            rem = deadline - time.monotonic()
            t8 = min(rem * 0.06, 50)
            if t8 < 8:
                break
            noise = rng.standard_normal(n_best) * 0.005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = adam_optimize(f_pert, time_limit=t8, lr=0.01, T_init=0.005,
                                      T_min=0.001, T_decay=0.9999,
                                      beta1=0.95, beta2=0.999, verbose=False, label="pert")
            _, best_score, best_seq = evaluate_and_track(a_opt, best_score, best_seq, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
