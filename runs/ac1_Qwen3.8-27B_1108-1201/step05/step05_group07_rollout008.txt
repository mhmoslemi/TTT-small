Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization of a 
smooth log-sum-exp approximation to the max-convolution objective, with 
softmax parameterization for non-negativity, multi-length exploration, and 
coordinate descent polishing.

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def logsumexpconv(z, T):
    """Compute T * log(sum(exp(C/T))) where C = conv(softmax(z), softmax(z))."""
    a = softmax(z)
    C = fast_convolve(a)
    C_max = C.max()
    return C_max + T * np.log(np.sum(np.exp((C - C_max) / T)))

def logsumexpconv_grad(z, T):
    """Gradient of T * log(sum(exp(C/T))) w.r.t. z."""
    a = softmax(z)
    C = fast_convolve(a)
    C_max = C.max()
    exp_C = np.exp((C - C_max) / T)
    w = exp_C / exp_C.sum()
    
    n = len(a)
    # grad_a[j] = 2 * sum_m a[m] * w[m+j]
    # = 2 * correlate(w, a, 'full')[n-1+j]
    conv_wa = np.correlate(w, a, mode='full')
    grad_a = 2.0 * conv_wa[n-1:n-1+n]
    
    # dL/dz[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a)
    return grad_z

def objective_and_grad(z, T):
    obj = logsumexpconv(z, T)
    g = logsumexpconv_grad(z, T)
    return obj, g
