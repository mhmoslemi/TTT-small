import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def compute_conv_grad_at_peak(f, k):
    """Compute gradient of (f*f)[k] w.r.t. f.
    Result: g[m] = 2*f[k-m] if 0 <= k-m < n, else 0.
    """
    n = len(f)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        ms = np.arange(lo, hi + 1)
        g[ms] = 2.0 * f[k - ms]
    return g


def lse_nesterov(f0, time_limit, beta_start=30.0, beta_end=10000.0, 
                  max_peaks=300, lr_init=None, verbose=False):
    """Nesterov-accelerated LSE gradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 1.5 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    v = np.zeros(n)  # velocity for Nesterov
    y = f.copy()  # lookahead point
    t_nest = 1.0  # Nesterov momentum parameter
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    backtracks = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute LSE weights
        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Compute gradient at y using weighted peaks
        conv_len = 2 * n - 1
        if len(w) <= max_peaks:
            sig_idx = np.argsort(w)[::-1]
        else:
            sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
            sig_idx = sig_idx[w[sig_idx] > 1e-15 * max(w.max(), 1e-30)]
        
        grad = np.zeros(n)
        for k in sig_idx:
            wk = w[k]
            if wk < 1e-15:
                continue
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                ms = np.arange(lo, hi + 1)
                grad[ms] += wk * 2.0 * y[k - ms]
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        t_nest = t_new
        
        z = y + theta * (y - f)  # extrapolation
        f_candidate = project_simplex(z - lr * grad)
        
        obj_candidate = fast_convolve(f_candidate).max()
        
        # Backtracking line search
        accepted = False
        for bt in range(10):
            if obj_candidate < best_obj or obj_candidate < fast_convolve(f).max() * (1 - 1e-6):
                accepted = True
                break
            lr *= 0.7
            f_candidate = project_simplex(z - lr * grad)
            obj_candidate = fast_convolve(f_candidate).max()
            backtracks += 1
        
        if accepted:
            y = f_candidate
            f = f_candidate
            if obj_candidate < best_obj - 1e-15:
                best_obj = obj_candidate
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            # Gentle learning rate increase
            lr = min(lr * 1.002, 3.0 / n)
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve > 50:
                # Reset with higher beta
                f = best_f.copy()
                y = f.copy()
                v = np.zeros(n)
                t_nest = 1.0
                beta = min(beta * 2.0, beta_end)
                lr = max(lr_init * 0.5, 0.0005)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.005

    return best_f, best_obj


def peak_directed(f0, time_limit, n_peaks=8, lr_init=0.002, 
                  momentum=0.93, verbose=False):
    """Directly target top peaks with weighted subgradient and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    v = np.zeros(n)
    cur_mom = momentum
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Select top peaks with exponential weighting
        k_top = min(n_peaks, conv_len)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        
        # Softmax weighting on peak values (emphasize highest)
        pv_scaled = pv - pv.max()
        w = np.exp(pv_scaled * 5.0)  # temperature controls emphasis
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(k_top)
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum

        # Compute weighted gradient
        grad = np.zeros(n)
        for i, k in enumerate(peaks):
            wk = w[i]
            if wk < 1e-15:
                continue
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                ms = np.arange(lo, hi + 1)
                grad[ms] += wk * 2.0 * f[k - ms]
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        v = cur_mom * v - cur_lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            cur_lr = min(cur_lr * 1.001, lr_init * 2.0)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                cur_lr *= 0.95
                cur_mom = max(0.85, cur_mom - 0.003)
            if no_improve > 200:
                rng = np.random.default_rng((it * 104729 + 3) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.0003)
                no_improve = 0

    return best_f, best_obj

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def chirp_init(n, alpha=1.0):

def random_structured_init(n, seed, num_modes=3):
    """Initialize with random sum of sinusoids."""

    f = np.ones(n)
    x = np.arange(n)
    for _ in range(num_modes):

        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.2)
        f += amp * np.cos(2 * np.pi * freq * x / n + phase)
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def fw_lp(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle - improved with better line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        A_cache = build_lp_matrix(f)
        g = solve_lp_direction(f, M * m_exp, A_cache)
        if g is None:

            if m_exp > 10.0:

            continue

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:

        g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 25:
                break
            m_exp = min(m_exp * 1.05, 5.0)
            continue

        # Thorough line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.02, 0.04, 0.06, 0.08, 0.1, 0.12, 0.15, 0.2, 0.25,

            val = fast_convolve(ft).max()

def lse_nesterov(f0, time_limit, beta_start=30.0, beta_end=10000.0,
                 max_peaks=300, lr_init=None, verbose=False):
    """Nesterov-accelerated LSE gradient descent with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 1.5 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    v = np.zeros(n)
    y = f.copy()
    t_nest = 1.0
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        conv_len = 2 * n - 1
        if len(w) <= max_peaks:
            sig_idx = np.argsort(w)[::-1]
        else:
            sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
            sig_idx = sig_idx[w[sig_idx] > 1e-15 * max(w.max(), 1e-30)]

        grad = np.zeros(n)
        for k in sig_idx:
            wk = w[k]
            if wk < 1e-15:
                continue
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                ms = np.arange(lo, hi + 1)
                grad[ms] += wk * 2.0 * y[k - ms]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        t_nest = t_new

        z = y + theta * (y - f)
        f_candidate = project_simplex(z - lr * grad)
        obj_candidate = fast_convolve(f_candidate).max()

        # Backtracking
        accepted = False
        obj_f = fast_convolve(f).max()
        for bt in range(8):
            if obj_candidate < best_obj * (1.0 + 1e-8) or obj_candidate < obj_f:
                accepted = True
                break
            lr *= 0.7
            f_candidate = project_simplex(z - lr * grad)
            obj_candidate = fast_convolve(f_candidate).max()

        if accepted:
            y = f_candidate
            f = f_candidate
            if obj_candidate < best_obj - 1e-15:
                best_obj = obj_candidate
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            lr = min(lr * 1.003, 3.0 / n)
        else:
            no_improve += 1
            lr *= 0.85
            if no_improve > 40:
                f = best_f.copy()
                y = f.copy()
                t_nest = 1.0
                beta = min(beta * 2.0, beta_end)
                lr = max(lr_init * 0.5, 0.0005)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.006

    return best_f, best_obj


def peak_directed(f0, time_limit, n_peaks=8, lr_init=0.002,
                  momentum=0.93, verbose=False):
    """Directly target top peaks with weighted subgradient and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    v = np.zeros(n)
    cur_mom = momentum
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        k_top = min(n_peaks, conv_len)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]

        # Exponential weighting emphasizing highest peaks
        pv_scaled = pv - pv.max()
        w = np.exp(pv_scaled * 8.0)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(k_top)
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum

        grad = np.zeros(n)
        for i in range(k_top):
            wk = w[i]
            if wk < 1e-15:
                continue
            k = int(peaks[i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                ms = np.arange(lo, hi + 1)
                grad[ms] += wk * 2.0 * f[k - ms]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        v = cur_mom * v - cur_lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            cur_lr = min(cur_lr * 1.001, lr_init * 3.0)
        else:
            no_improve += 1
            if no_improve % 25 == 0:
                cur_lr *= 0.94
                cur_mom = max(0.85, cur_mom - 0.004)
            if no_improve > 150:
                rng = np.random.default_rng((it * 104729 + 3) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0006)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.0003)
                no_improve = 0

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def chirp_init(n, alpha=1.0):
    """Initialize with chirp-like pattern."""
    x = np.arange(n)
    phase = 2.0 * np.pi * alpha * x * x / n
    f = 1.0 + 0.3 * np.cos(phase)
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f


def random_structured_init(n, seed, num_modes=4):
    """Initialize with random sum of sinusoids."""
    rng = np.random.default_rng(seed)
    f = np.ones(n)
    x = np.arange(n)
    for _ in range(num_modes):
        freq = rng.uniform(1, n / 3)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.25)
        f += amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f


def local_perturbation_search(f0, best_score, time_limit, n_trials=200):
    """Try small random perturbations to find local improvements."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        # Random perturbation with decreasing scale
        scale = 0.005 * (1.0 - 0.5 * trial / max(n_trials, 1))
        f_new = project_simplex(best_f + rng.standard_normal(n) * scale)
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main multi-phase optimization:
    Phase 1: Multi-length LSE exploration with diverse inits
    Phase 2: FW-LP on best sequence
    Phase 3: High-beta Nesterov LSE
    Phase 4: Peak-directed with multiple peak counts
    Phase 5: Ultra-fine polish + local perturbation
    Phase 6: Try nearby lengths
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Multi-length exploration ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    t1_end = deadline - 950

    n_candidates = sorted(set([n_parent, 384, 448, 512, 576, 640, 768, 896, 1024]))
    n_candidates = [nc for nc in n_candidates if 256 <= nc <= 1024]

    tried = 0
    for nc in n_candidates:
        t_remaining = t1_end - time.monotonic()
        if t_remaining < 20 or tried >= 7:
            break

        if nc == n_parent and parent is not None:
            f_init = parent.copy()
        else:
            r = rng.random()
            if r < 0.35:
                f_init = interpolate_to_length(parent, nc)
            elif r < 0.55:
                f_init = chirp_init(nc, alpha=float(rng.uniform(0.5, 2.5)))
            elif r < 0.75:
                f_init = random_structured_init(nc, seed=int(rng.integers(1000000)))
            else:
                f_init = np.ones(nc) / nc

            # Quick warmup
            t_warm = min(30, t_remaining * 0.15)
            if t_warm > 8:
                f_init, _ = lse_nesterov(f_init, time_limit=t_warm,
                                          beta_start=10.0, beta_end=400.0)

        t_alloc = min(100, t_remaining * 0.4)
        if t_alloc < 12:
            break
        tried += 1
        print(f"  LSE n={nc}, t={t_alloc:.0f}s")
        f_opt, _ = lse_nesterov(f_init, time_limit=t_alloc,
                                 beta_start=20.0, beta_end=3000.0)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: FW-LP on best so far ===
    print(f"\n=== Phase 2: FW-LP Optimization ===")
    t2_end = deadline - 550
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    t_alloc = min(450, t2_end - time.monotonic())
    if t_alloc > 40:
        print(f"  FW-LP n={n_cur}, t={t_alloc:.0f}s")
        f_opt, obj = fw_lp(current, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: High-beta LSE refinement ===
    print(f"\n=== Phase 3: High-Beta LSE ===")
    t3_end = deadline - 350
    current = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = min(250, t3_end - time.monotonic())
    if t_alloc > 20:
        print(f"  LSE n={len(current)}, t={t_alloc:.0f}s, beta_end=30000")
        f_opt, _ = lse_nesterov(current, time_limit=t_alloc,
                                 beta_start=100.0, beta_end=30000.0,
                                 lr_init=0.8 / len(current),
                                 max_peaks=400)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Peak-directed optimization ===
    print(f"\n=== Phase 4: Peak-Directed ===")
    t4_end = deadline - 200
    current = best_tracker[1] if best_tracker[1] is not None else parent

    peak_configs = [(4, 0.0012, 0.94), (8, 0.0015, 0.93), (12, 0.002, 0.92),
                    (20, 0.0025, 0.91), (32, 0.003, 0.90)]
    for npk, lr_p, mom_p in peak_configs:
        t_alloc = min(50, t4_end - time.monotonic())
        if t_alloc < 12:
            break
        print(f"  Peak-directed n_peaks={npk}, lr={lr_p}, t={t_alloc:.0f}s")
        f_opt, _ = peak_directed(current, time_limit=t_alloc,
                                   n_peaks=npk, lr_init=lr_p,
                                   momentum=mom_p)
        try_sequence(f_opt, best_tracker)
        current = best_tracker[1] if best_tracker[1] is not None else current

    # === Phase 5: Ultra-fine polish ===
    print(f"\n=== Phase 5: Ultra-Fine Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent

    # Very high beta LSE with tiny steps
    t_alloc = min(100, deadline - time.monotonic() - 80)
    if t_alloc > 15:
        print(f"  Ultra-fine LSE t={t_alloc:.0f}s")
        f_opt, _ = lse_nesterov(current, time_limit=t_alloc,
                                 beta_start=10000.0, beta_end=100000.0,
                                 lr_init=0.15 / len(current),
                                 max_peaks=500)
        try_sequence(f_opt, best_tracker)

    # Peak-directed with very few peaks
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(40, deadline - time.monotonic() - 40)
    if t_alloc > 10:
        print(f"  Ultra-fine peak-directed t={t_alloc:.0f}s")
        f_opt, _ = peak_directed(current, time_limit=t_alloc,
                                   n_peaks=3, lr_init=0.0004, momentum=0.96)
        try_sequence(f_opt, best_tracker)

    # Local perturbation search
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(30, deadline - time.monotonic() - 10)
    if t_alloc > 5:
        print(f"  Local perturbation t={t_alloc:.0f}s")
        f_opt, _ = local_perturbation_search(current, best_tracker[0],
                                              time_limit=t_alloc, n_trials=500)
        try_sequence(f_opt, best_tracker)

    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Nearby Lengths ===")
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        offsets = [32, -32, 64, -64, 128, -128, 16, -16]
        for off in offsets:
            n_try = n_best + off
            if n_try < 256 or n_try > 1100:
                continue
            t_remaining = deadline - time.monotonic() - 5
            if t_remaining < 15:
                break
            t_alloc = min(30, t_remaining * 0.4)
            if t_alloc < 8:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_nesterov(f_init, time_limit=t_alloc,
                                     beta_start=50.0, beta_end=5000.0)
            try_sequence(f_opt, best_tracker)

    # === Return ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(448) / 448
    return [float(x) for x in fallback]
