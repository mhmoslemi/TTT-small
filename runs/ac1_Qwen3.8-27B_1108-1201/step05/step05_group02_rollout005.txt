import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def log_sum_exp(conv, beta):
    """Compute log(sum(exp(beta * conv))) and the softmax weights."""
    shifted = beta * conv - beta * conv.max()
    exp_shifted = np.exp(shifted)
    log_val = np.log(exp_shifted.sum()) + beta * conv.max()
    weights = exp_shifted / exp_shifted.sum()
    return log_val, weights

def compute_gradient_lse(f, conv, beta):
    """Compute gradient of log-sum-exp(beta * conv) with respect to f.
    conv = f * f (full convolution)
    grad_j = 2 * sum_k weights[k] * f[k-j] (where valid)
    """
    n = len(f)
    log_val, weights = log_sum_exp(conv, beta)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-15:
            continue
        w = weights[k]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            k_minus_j = k - j
            if 0 <= k_minus_j < n:
                grad[j] += 2.0 * w * f[k_minus_j]
    return log_val, grad

def lse_gradient_descent(f0, time_limit, beta_start=10, beta_end=100, 
                          lr=0.01, momentum=0.9, verbose=False):
    n = len(f0)
    f = f0.copy()
    f = project_simplex(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    total_iters_target = 5000
    # We'll adapt beta over time
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Anneal beta
        progress = min(1.0, it / total_iters_target)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute softmax weights
        shifted = beta * conv - beta * conv.max()
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()
        
        # Compute gradient via cross-correlation
        # grad[j] = 2 * sum_k w[k] * f[k-j] for valid k
        # This is 2 * np.correlate(w, f, 'valid')
        fft_size = 1 << (3 * n - 2 - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        F = np.fft.rfft(f, fft_size)
        grad = 2.0 * np.fft.irfft(W * np.conj(F), fft_size)[:n]
        
        # Update with momentum
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Evaluate
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new  # still move (for momentum)
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.9
            if no_improve > 500:
                # Restart from best
                f = best_f + np.random.default_rng(it).standard_normal(n) * 0.001
                f = project_simplex(f)
                v = np.zeros(n)
                no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    LSE-GD iter {it}: obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}")

def project_simplex(v):

def softmax_weights(conv, beta):

    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:

        j_end = min(n - 1, max_pk)
        js = np.arange(j_start, j_end + 1)
        products = f[js] * f[max_pk - js]
        max_j = int(js[np.argmax(products)])
        partner = max_pk - max_j
        
        # Try to reduce the product by moving mass from the larger of the two
        if f[max_j] >= f[partner]:
            source = max_j
        else:
            source = partner
        
        if f[source] < 1e-8:
            continue
        
        # Target: a position that contributes to a low-conv position
        # Heuristic: position where f is smallest
        target = int(np.argmin(f))
        if target == source:
            # Second smallest
            sorted_idx = np.argsort(f)
            target = int(sorted_idx[1])
        
        for frac in [0.005, 0.01, 0.02, 0.05]:
            amount = f[source] * frac
            f_trial = f.copy()
            f_trial[source] -= amount
            f_trial[target] += amount
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            if obj_trial < best_obj - 1e-12:
                f = f_trial
                conv = conv_trial
                best_obj = obj_trial
                best_f = f.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                # Random restart from best
                rng = np.random.default_rng(it * 31)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                conv = fast_convolve(f)
                no_improve = 0
                if it > 10000:
                    break
        else:
            no_improve = 0
    
    return best_f, best_obj
