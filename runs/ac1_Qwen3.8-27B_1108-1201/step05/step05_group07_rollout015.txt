import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    v = np.maximum(v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_peak_context(conv, a, k):
    """Get the contributing pairs for peak at position k."""
    n = len(a)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    return idx, a[idx] * a[k - idx]


def compute_subgradient(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg, peaks[:5]

            v = np.zeros(n)

    best_obj = fast_convolve(f).max()
    cur_obj = best_obj
    t0 = time.monotonic()
    cur_temp = temperature
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            cur_temp *= 0.999
        else:
            no_improve += 1
            cur_temp *= 0.9995

        # Find top peaks
        k_actual = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]

        # For the top peaks, try mass transfers
        try_count = min(80, 20000 // max(1, it // 10))
        for _ in range(try_count):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick a peak (weighted by height)
            peak_weights = conv[peaks] / conv[peaks].sum()
            peak_idx = int(np.random.choice(peaks, p=peak_weights))

            js = max(0, peak_idx - n + 1)
            je = min(n - 1, peak_idx)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[peak_idx - idx]
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue

            # Pick a contributor pair weighted by contribution
            pair_weights = contribs / total_contrib
            i_pos = int(np.random.choice(idx, p=pair_weights))
            j_pos = peak_idx - i_pos

            if i_pos == j_pos:
                # Self-pair: reduce f[i_pos]^2 by spreading mass
                i = i_pos
                # Find a low-contributing target
                target_weights = np.ones(n)
                target_weights[peaks] = 0.1
                target_weights /= target_weights.sum()
                j = int(np.random.choice(n, p=target_weights))
                if j == i or f[i] < 1e-12:
                    continue
            else:
                i = i_pos
                j = j_pos
                # Transfer from the larger to the smaller to equalize
                if f[i] < f[j]:
                    i, j = j, i
                if f[i] < 1e-12:
                    continue

            # Golden section search for optimal transfer amount
            max_transfer = f[i] * 0.5
            if max_transfer < 1e-14:
                continue

            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            a_lo = 0.0
            a_hi = max_transfer
            a1 = a_hi - gr * (a_hi - a_lo)
            a2 = a_lo + gr * (a_hi - a_lo)

            def eval_transfer(amt):
                old_i = f[i]
                old_j = f[j]
                f[i] = old_i - amt
                f[j] = old_j + amt
                fs = f.sum()
                if fs < 1e-15:
                    f[i] = old_i
                    f[j] = old_j
                    return cur_obj + 1.0
                f /= fs
                val = fast_convolve(f).max()
                f[i] *= fs
                f[j] *= fs
                f[i] = old_i
                f[j] = old_j
                return val

            val1 = eval_transfer(a1)
            val2 = eval_transfer(a2)
            for _ in range(8):
                if a_hi - a_lo < 1e-12:
                    break
                if val1 < val2:
                    a_hi = a2
                    a2 = a1
                    val2 = val1
                    a1 = a_hi - gr * (a_hi - a_lo)
                    val1 = eval_transfer(a1)
                else:
                    a_lo = a1
                    a1 = a2
                    val1 = val2
                    a2 = a_lo + gr * (a_hi - a_lo)
                    val2 = eval_transfer(a2)

            best_amt = (a_lo + a_hi) / 2
            best_val = min(val1, val2)

            # Accept/reject with simulated annealing
            if best_val < cur_obj:
                # Always accept improvement
                old_i = f[i]
                old_j = f[j]
                f[i] = old_i - best_amt
                f[j] = old_j + best_amt
                f /= f.sum()
                cur_obj = best_val
            else:
                # Simulated annealing acceptance
                delta = best_val - cur_obj
                if cur_temp > 1e-8 and np.random.random() < np.exp(-delta / cur_temp):
                    old_i = f[i]
                    old_j = f[j]
                    f[i] = old_i - best_amt
                    f[j] = old_j + best_amt
                    f /= f.sum()
                    cur_obj = best_val

        # Periodic reheat
        if no_improve > 200:
            reset_temp = 0.003 + 0.002 * min(no_improve // 400, 5)
            cur_temp = reset_temp
            if no_improve > 500:
                # Strong restart
                noise = rng.standard_normal(n) * 0.005
                f = project_simplex(best_f + noise)
                cur_obj = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_optimal(f0, time_limit, rng, label=""):
    """
    Coordinate descent with optimal step computation.
    For each coordinate, the max convolution is a piecewise-linear convex function
    of that coordinate (when others are rescaled). We find the minimizer.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Process coordinates in order of their contribution to the top peaks
    conv = fast_convolve(f)
    top_peaks = np.argpartition(conv, -min(10, 2 * n - 1))[-min(10, 2 * n - 1):]

    # Compute contribution score for each coordinate
    scores = np.zeros(n)
    for k in top_peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        contribs = f[idx] * f[k - idx]
        scores[idx] += contribs

    order = np.argsort(scores)[::-1]

    while time.monotonic() - t0 < time_limit:
        improved = False
        round_count = 0

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            round_count += 1
            old_val = f[j]
            if old_val < 1e-14:
                # Try turning on
                for frac in [0.001, 0.003, 0.01]:
                    f_trial = f.copy()
                    f_trial[j] = frac
                    denom = 1.0 - 0.0
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= (1.0 - frac)
                    st = f_trial.sum()
                    if st < 1e-15:
                        continue
                    f_trial /= st
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        improved = True
                        break
                continue

            # Try multiple multipliers with rescaling
            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.3, 0.5, 0.7, 1.3, 1.6, 2.0, 0.1, 3.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            # Try random coordinates that weren't high-scoring
            for _ in range(min(50, n)):
                if time.monotonic() - t0 > time_limit:
                    break
                j = int(rng.integers(0, n))
                old_val = f[j]
                if old_val < 1e-14:
                    continue
                for mult in [0.5, 1.5]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.99:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = np.maximum(f_trial, 0)
                    st = f_trial.sum()
                    if st < 1e-15:
                        continue
                    f_trial /= st
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        improved = True
                        break

        if not improved:
            break

    return best_f, best_obj


def pair_transfer_search(f0, time_limit, rng, n_peaks=15, max_tries_per_peak=5):
    """
    Systematic pair transfer: for each peak, find the worst-contributing pair
    and transfer mass optimally.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()

        k_actual = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]

        improved = False
        for k in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]

            # Sort by contribution, try to reduce top contributors
            sorted_idx = idx[np.argsort(contribs)[::-1]]

            for attempt in range(max_tries_per_peak):
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(sorted_idx[attempt % len(sorted_idx)])
                if f[src] < 1e-12:
                    continue

                # Find target: coordinate NOT involved in this peak and with low value
                target_mask = np.ones(n, dtype=bool)
                target_mask[k - idx] = False  # exclude coords involved in this peak
                target_mask[src] = False
                if target_mask.sum() == 0:
                    continue
                target_candidates = np.where(target_mask)[0]
                # Pick from lowest value candidates
                if len(target_candidates) > n // 4:
                    target_candidates = target_candidates[np.argsort(f[target_candidates])[:n // 4]]
                dest = int(rng.choice(target_candidates))

                if f[src] < 1e-12:
                    continue

                # Try several transfer fractions
                for frac in [0.05, 0.15, 0.3, 0.5, 0.7]:
                    amt = f[src] * frac
                    old_s = f[src]
                    old_d = f[dest]
                    f[src] = old_s - amt
                    f[dest] = old_d + amt
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[src] = old_s
                        f[dest] = old_d
                        if val > cur_obj:
                            break

        if not improved:
            break

    return best_f, best_obj


def fourier_inject(f0, rng, n_modes=5):
    """
    Create a perturbation in the Fourier domain to add structured diversity.
    This injects oscillatory patterns that may reduce convolution peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    # Work in Fourier domain
    A = np.fft.rfft(f)
    # Add small random phase perturbations to non-DC, non-Nyquist modes
    perturbation = np.zeros_like(A)
    for _ in range(n_modes):
        mode = int(rng.integers(1, len(A) - 1))
        amp = rng.uniform(0.001, 0.005)
        phase = rng.uniform(0, 2 * np.pi)
        perturbation[mode] += amp * np.exp(1j * phase)
        if mode < len(A) - 1 - mode:
            perturbation[len(A) - 1 - mode] += amp * np.exp(-1j * phase)

    A_new = A + perturbation
    f_new = np.fft.irfft(A_new, n)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    else:
        f_new = f.copy()

    return f_new


def mirror_blend(f0, rng, alpha_range=(0.2, 0.8)):
    """Blend a sequence with its mirror (reverse)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    alpha = rng.uniform(*alpha_range)
    f_mirror = f[::-1].copy()
    f_blend = alpha * f + (1 - alpha) * f_mirror
    f_blend = np.maximum(f_blend, 0)
    ss = f_blend.sum()
    if ss > 1e-15:
        f_blend /= ss
    return f_blend


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    center = n / 2.0
    sigma = n / (4.0 + rng.random() * 4.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 100)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Two-peak structure
    peak1 = int(n * (0.25 + rng.random() * 0.1))
    peak2 = int(n * (0.65 + rng.random() * 0.1))
    two_peak = np.zeros(n)
    two_peak[peak1] = 1.0
    two_peak[peak2] = 1.0
    sigma2 = n / 20
    for i in range(n):
        two_peak[i] = np.exp(-0.5 * ((i - peak1) / sigma2) ** 2) + np.exp(-0.5 * ((i - peak2) / sigma2) ** 2)
    two_peak = np.maximum(two_peak, 1e-10)
    inits.append(two_peak / two_peak.sum())

    # Periodic-like
    period = max(2, int(n / (16 + rng.integers(0, 16))))
    base = np.ones(period) * (0.5 + rng.random() * 0.5)
    periodic = np.tile(base, n // period + 1)[:n]
    periodic = np.maximum(periodic, 1e-10)
    inits.append(periodic / periodic.sum())

    return inits[:count]


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining subgradient, peak-directed search,
    coordinate descent, simulated annealing, and evolutionary diversity.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform n=512")

    n_parent = len(parent)

    # === Phase 1: Quick subgradient on parent ===
    print("\n=== Phase 1: Subgradient on parent ===")
    rem = deadline - time.monotonic() - 900
    t_p1 = min(rem, 60)
    if t_p1 > 5:
        a_opt, _ = mu_weights_optimize(parent, time_limit=t_p1,
                                        top_k=max(20, n_parent // 8),
                                        temperature=0.00004, lr=0.003,
                                        momentum=0.87)
        try_seq(a_opt, "sg_parent")

    # === Phase 2: Peak-directed search with simulated annealing ===
    print(f"\n=== Phase 2: Peak-Directed Search (best={best_score:.6f}) ===")
    base_f = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic() - 600
    t_p2 = min(rem, 300)
    if t_p2 > 10:
        a_opt, _ = peak_directed_search(base_f, time_limit=t_p2, rng=rng,
                                         temperature=0.002, n_peaks=25)
        try_seq(a_opt, "peak_sa")

        # Second round with different params
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            a_opt, _ = peak_directed_search(base_f, time_limit=min(rem, 150), rng=rng,
                                             temperature=0.001, n_peaks=10)
            try_seq(a_opt, "peak_sa2")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        t_p3 = min(rem, 200)
        if t_p3 > 10:
            a_opt, _ = coordinate_descent_optimal(best_seq, time_limit=t_p3, rng=rng)
            try_seq(a_opt, "cd1")

            rem = deadline - time.monotonic() - 200
            if rem > 10:
                a_opt, _ = coordinate_descent_optimal(best_seq, time_limit=min(rem, 100), rng=rng)
                try_seq(a_opt, "cd2")

    # === Phase 4: Pair transfer search ===
    print(f"\n=== Phase 4: Pair Transfer (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        t_p4 = min(rem, 150)
        if t_p4 > 10:
            a_opt, _ = pair_transfer_search(best_seq, time_limit=t_p4, rng=rng,
                                             n_peaks=20, max_tries_per_peak=8)
            try_seq(a_opt, "pair1")

            a_opt, _ = pair_transfer_search(best_seq, time_limit=min(rem - t_p4, 100), rng=rng,
                                             n_peaks=10, max_tries_per_peak=12)
            try_seq(a_opt, "pair2")

    # === Phase 5: Diversity injection + re-optimize ===
    print(f"\n=== Phase 5: Diversity + Re-optimize (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        diversity_candidates = []

        # Fourier injection
        for _ in range(3):
            diversity_candidates.append(fourier_inject(best_seq, rng, n_modes=3))

        # Mirror blend
        for _ in range(2):
            diversity_candidates.append(mirror_blend(best_seq, rng))

        # Random perturbation
        for scale in [0.002, 0.005, 0.01]:
            noise = rng.standard_normal(n_best) * scale
            diversity_candidates.append(project_simplex(best_seq + noise))

        # Structured inits at same length
        diversity_candidates.extend(structured_inits(n_best, rng, count=2))

        # Evaluate diversity candidates
        for i, dcand in enumerate(diversity_candidates):
            rem = deadline - time.monotonic() - 100
            if rem < 5:
                break
            try_seq(dcand, f"div_{i}")

            # Quick optimize the best diversity candidate
            if dcand.sum() > 0:
                dcand_norm = dcand / dcand.sum()
                t_quick = min(rem * 0.15, 30)
                if t_quick > 3:
                    a_opt, _ = mu_weights_optimize(dcand_norm, time_limit=t_quick,
                                                    top_k=max(15, n_best // 8),
                                                    temperature=0.00003, lr=0.002,
                                                    momentum=0.88)
                    try_seq(a_opt, f"divopt_{i}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [n_parent + 64, n_parent + 128, n_parent * 12 // 10,
                         n_parent * 11 // 10, n_parent + 256]
        target_lengths = sorted(set(l for l in target_lengths if 128 <= l <= 3000))

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.25, 80)
            print(f"  n={target_n}, t={t_alloc:.0f}s")

            a_opt, _ = mu_weights_optimize(f_interp, time_limit=min(t_alloc, 40),
                                            top_k=max(20, target_n // 6),
                                            temperature=0.00003, lr=0.003,
                                            momentum=0.87)
            try_seq(a_opt, f"len_{target_n}_sg")

            rem = deadline - time.monotonic() - 40
            if rem > 10:
                a_opt, _ = peak_directed_search(a_opt, time_limit=min(rem, 30), rng=rng,
                                                 temperature=0.001, n_peaks=15)
                try_seq(a_opt, f"len_{target_n}_peak")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = mu_weights_optimize(best_seq, time_limit=min(rem, 40),
                                            top_k=n_best, temperature=0.000001,
                                            lr=0.0002, momentum=0.95)
            try_seq(a_opt, "fine_sg")

        # Fine peak-directed
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            a_opt, _ = peak_directed_search(best_seq, time_limit=min(rem, 30), rng=rng,
                                             temperature=0.0003, n_peaks=5)
            try_seq(a_opt, "fine_peak")

        # Final coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem > 3:
            a_opt, _ = coordinate_descent_optimal(best_seq, time_limit=min(rem, 15), rng=rng)
            try_seq(a_opt, "cd_final")

        # One more pair transfer
        rem = deadline - time.monotonic() - 3
        if rem > 3:
            a_opt, _ = pair_transfer_search(best_seq, time_limit=min(rem, 10), rng=rng,
                                             n_peaks=5, max_tries_per_peak=3)
            try_seq(a_opt, "pair_final")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
