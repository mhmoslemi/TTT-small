def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_peak_gradient(a, k, n):
    """Compute gradient of c[k] = sum_j a[j]*a[k-j] w.r.t. a.
    Result: grad[m] = 2*a[k-m] if 0<=k-m<n, else 0."""

    grad = np.zeros(n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Spectral (Gerchberg-Saxton-like) optimization for fast peak reduction
- Projected subgradient descent with multi-peak weighting
- LP-Frank-Wolfe for periodic high-quality direction updates

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Ensure non-negative and sum to 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_optimize(a0, time_limit, verbose=False, clip_frac=0.4, soft_beta=2.0):
    """Gerchberg-Saxton style spectral optimization for flat convolution."""
    a = normalize(a0.astype(np.float64))
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_clip = clip_frac

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        A = np.fft.rfft(a, N)
        C = A * A
        c_full = np.fft.irfft(C, N)[:2 * n - 1]

        c_max = c_full.max()
        if c_max < 1e-15:
            break

        # Soft clipping: reduce peaks smoothly
        threshold = cur_clip * c_max
        excess = np.maximum(c_full - threshold, 0.0)
        c_new = c_full - (1.0 - np.exp(-soft_beta * excess / (c_max + 1e-30))) * excess
        c_new = np.maximum(c_new, 0.0)

        C_new = np.fft.rfft(c_new, N)

        # Square root in frequency domain
        mag = np.sqrt(np.maximum(np.abs(C_new), 0.0))
        phase = np.angle(C_new) * 0.5
        A_new = mag * np.cos(phase) + 1j * mag * np.sin(phase)

        a_new = np.fft.irfft(A_new, N)[:n].real
        a_new = normalize(a_new)

        obj = fast_convolve(a_new).max()
        a = a_new

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Spectral iter {it}: obj={best_obj:.10f}, clip={cur_clip:.3f}")
        else:
            no_improve += 1

        # Adapt clip fraction
        if no_improve > 50:
            cur_clip = min(0.9, cur_clip * 1.05)
            no_improve = 0
        if no_improve > 500:
            break

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, verbose=False, lr0=0.02, decay=0.9998, top_k=15, temp0=0.002):
    """Projected subgradient descent with multi-peak softmax weighting and backtracking."""
    a = normalize(a0.astype(np.float64))
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = lr0
    cur_temp = temp0
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        c = fast_convolve(a)
        c_max = c.max()

        k_actual = min(cur_top_k, len(c))
        peak_idx = np.argpartition(c, -k_actual)[-k_actual:]
        peak_vals = c[peak_idx]

        scaled = peak_vals / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peak_idx))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        # Compute weighted gradient
        grad = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-14:
                continue
            k = int(peak_idx[i])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                grad[indices] += 2.0 * w[i] * a[k - indices]

        # Backtracking line search
        best_step_a = a.copy()
        best_step_val = c_max

        for lr_mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            lr = cur_lr * lr_mult
            a_try = project_simplex(a - lr * grad)
            c_try = fast_convolve(a_try)
            val = c_try.max()
            if val < best_step_val - 1e-15:
                best_step_val = val
                best_step_a = a_try

        if best_step_val < best_obj - 1e-14:
            a = best_step_a
            best_obj = best_step_val
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}")
        else:
            no_improve += 1
            cur_lr *= 0.9

        cur_lr *= decay
        cur_temp *= 1.003
        cur_top_k = min(50, top_k + it // 5000)

        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 5)
            a = project_simplex(best_a + rng.standard_normal(n) * (0.003 + 0.002 * min(no_improve // 300, 5)))
            cur_lr = max(lr0 * 0.5, 0.003)
            no_improve = 0
            if no_improve > 2000:
                break

    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def make_cosine_init(n, freq, amp):
    """Cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    return normalize(a)


def make_exponential_init(n, alpha, direction=1):
    """Exponential ramp initialization."""
    j = np.arange(n, dtype=np.float64)
    if direction == 1:
        a = np.exp(alpha * j / n)
    elif direction == -1:
        a = np.exp(-alpha * j / n)
    else:
        a = np.exp(-alpha * np.abs(j - n / 2.0) / n)
    return normalize(a)


def make_bump_init(n, center_frac, width_frac):
    """Gaussian bump initialization."""
    j = np.arange(n, dtype=np.float64)
    c = center_frac * n
    w = max(width_frac * n, 1.0)
    a = np.exp(-((j - c) ** 2) / (2.0 * w * w))
    return normalize(a)


def make_vshape_init(n, depth):
    """V-shaped initialization (low in middle, high at edges)."""
    j = np.arange(n, dtype=np.float64)
    mid = n / 2.0
    a = 1.0 + depth * np.abs(j - mid) / mid
    return normalize(a)


def make_inverted_v_init(n, depth):
    """Inverted V (high in middle, low at edges) - like a tent."""
    j = np.arange(n, dtype=np.float64)
    mid = n / 2.0
    a = 1.0 - depth * np.abs(j - mid) / mid
    a = np.maximum(a, 0.01)
    return normalize(a)

        n_fin = len(best_seq)

        # Short spectral
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            a_opt, _ = spectral_optimize(best_seq, time_limit=min(rem, 60), clip_frac=0.6, soft_beta=1.0)
            try_seq(a_opt, "final_spec")

        # Short subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem, 50),
                                             lr0=0.005, top_k=10, temp0=0.0005)
            try_seq(a_opt, "final_sg")

        # Short LP-FW
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 40), max_iters=100)
            try_seq(a_opt, "final_lpfw")

        # Tiny CD
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
