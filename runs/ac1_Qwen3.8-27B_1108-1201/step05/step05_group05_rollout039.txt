import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_smooth(a, alpha=0.95):
    """Apply low-pass filtering in frequency domain to smooth the sequence.
    alpha controls how much high-frequency to remove (closer to 1 = more smoothing).
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    freqs = np.fft.rfftfreq(fft_size)
    # Low-pass: keep low frequencies, attenuate high
    # Use a raised cosine transition
    cutoff = 0.5  # Nyquist
    transition = 0.1
    mask = np.ones(len(A))
    for i in range(len(A)):
        if freqs[i] > cutoff - transition:
            if freqs[i] > cutoff:
                mask[i] = alpha
            else:
                t = (freqs[i] - (cutoff - transition)) / transition
                mask[i] = 1.0 - t * (1.0 - alpha)
    A *= mask
    result = np.fft.irfft(A, fft_size)[:n]
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result


def spectral_smooth_simplex(a, bandwidth=0.3):
    """Smooth in frequency domain, keeping only low frequencies up to bandwidth.
    Then project back to simplex.
    """
    n = len(a)
    fft_size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    freqs = np.fft.rfftfreq(fft_size)
    mask = (freqs <= bandwidth).astype(float)
    # Smooth transition
    for i in range(len(A)):
        if bandwidth < freqs[i] < bandwidth + 0.1:
            mask[i] = 1.0 - (freqs[i] - bandwidth) / 0.1
    A *= mask
    result = np.fft.irfft(A, fft_size)[:n]
    return project_simplex(result)


def backtracking_line(f, g, n, c0, f0_val, max_iter=30):
    """Backtracking line search for alpha in [0,1] mixing f and g.
    f is current (sum=1), g is direction (sum=1).
    Returns best alpha found.
    """
    alpha = 1.0
    c = c0  # current conv max
    best_alpha = 0.0
    best_c = c0
    
    for _ in range(max_iter):
        if alpha < 1e-6:
            break
        f_new = f + alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            alpha *= 0.5
            continue
        f_new /= s
        c_new = fast_convolve(f_new).max()
        if c_new < best_c:
            best_c = c_new
            best_alpha = alpha
        # Armijo condition: c_new < c0 - beta * alpha * (something)
        # For simplicity, just use: c_new < c0 - 0.01 * alpha * (c0 - c0*0.99)
        if c_new < c0 * (1 - 0.01 * alpha):
            return best_alpha  # Accept
        alpha *= 0.7
    
    return best_alpha


def solve_lp_full(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (sum-normalized) or None.
    """
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j] (with valid indices)
    # For each k in 0..2n-2: sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * g[j] <= M
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None


def solve_lp_active(f, M, n, active_k):
    """Solve LP with only active constraints."""
    c = -np.ones(n)
    K = len(active_k)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    for row, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None

def coord_descent_v2(f0, time_limit, mult_list=None):
    """Coordinate descent with more multiplier options and early stopping."""
    if mult_list is None:
        mult_list = [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 1.8, 2.2, 3.0, 4.0]
    
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in mult_list:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-12 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0
    
    return best_f, best_obj


def sg_polish_v2(f0, time_limit, lr=0.001, mom=0.93, top_k=30, temp=0.00003):
    """Subgradient polishing with adaptive temperature and peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it_count += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate and temperature
        cur_lr *= 0.9999
        cur_temp *= 1.003
        
        if no_improve > 300:
            rng = np.random.default_rng((it_count * 7919 + 13) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0
    
    return best_f, best_obj


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    # Enforce Hermitian symmetry for real signal
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def gaussian_window_init(n, center_frac=0.5, width_frac=0.5, seed=0):
    """Initialize with Gaussian window (centered, adjustable width)."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    center = center_frac * (n - 1)
    width = width_frac * (n - 1)
    g = np.exp(-0.5 * ((x - center) / (width / 2.355)) ** 2)  # FWHM = width
    # Add small random perturbation
    g = g + rng.standard_normal(n) * 0.01
    g = np.maximum(g, 0)
    return g / g.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_palindromic(a, n):
    """Create a palindromic sequence of length n."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
- LP-Frank-Wolfe with backtracking line search and periodic full-constraint LP
- Spectral smoothing (frequency-domain low-pass filtering)
- Iteratively reweighted quadratic optimization (IRL-QP)
- Diverse combinatorial and spectral initializations
- Multi-scale (multi-length) exploration with interpolation
- Adaptive coordinate descent
- Subgradient polishing with softmax-weighted peaks

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_smooth_simplex(a, bandwidth=0.35):
    """Smooth in frequency domain with raised-cosine low-pass, project to simplex."""
    n = len(a)
    fft_size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    freqs = np.fft.rfftfreq(fft_size)
    mask = np.ones(len(A))
    trans_width = 0.08
    for i in range(len(A)):
        f = freqs[i]
        if f > bandwidth:
            if f > bandwidth + trans_width:
                mask[i] = 0.05
            else:
                t = (f - bandwidth) / trans_width
                mask[i] = 0.5 * (1.0 + np.cos(np.pi * t))
    A *= mask
    result = np.fft.irfft(A, fft_size)[:n]
    return project_simplex(result)


def backtracking_line(f, g, c0, max_iter=25):
    """Backtracking line search for alpha in [0,1] mixing f and g (both sum=1)."""
    best_alpha = 0.0
    best_c = c0
    alpha = 1.0
    for _ in range(max_iter):
        if alpha < 1e-7:
            break
        f_new = f + alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            alpha *= 0.5
            continue
        f_new /= s
        c_new = fast_convolve(f_new).max()
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_alpha = alpha
        if c_new < c0 * (1.0 - 0.005 * alpha):
            return best_alpha
        alpha *= 0.65
    return best_alpha


def solve_lp_full(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k=0..2n-2, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None


def solve_lp_active(f, M, n, active_k):
    """Solve LP with only active (top-K) constraints."""
    c = -np.ones(n)
    K = len(active_k)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    for row, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None


def lpfw_optimize(f0, time_limit, use_spectral=True, spectral_bw=0.35,
                  full_interval=15, top_k=30, verbose=False):
    """LP-Frank-Wolfe with backtracking, spectral smoothing, periodic full LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        use_full = (it % full_interval == 0) or (it <= 2)
        if use_full:
            g = solve_lp_full(f, M * m_exp, n)
        else:
            k_sorted = np.argsort(conv)[::-1][:top_k]
            g = solve_lp_active(f, M * m_exp, n, k_sorted.tolist())

        if g is None:
            m_exp *= 1.12
            if m_exp > 4.0:
                m_exp = 1.0
                g = solve_lp_full(f, M, n)
                if g is None:
                    break
            else:
                continue

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.06, 3.0)
            continue

        alpha = backtracking_line(f, g, M)
        if alpha < 1e-7:
            alpha = 0.08

        f_new = f + alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()

        if use_spectral and it % 4 == 0:
            f_smooth = spectral_smooth_simplex(f_new, bandwidth=spectral_bw)
            c_sm = fast_convolve(f_smooth).max()
            c_new = fast_convolve(f_new).max()
            if c_sm < c_new:
                f_new = f_smooth

        conv = fast_convolve(f_new)
        M_new = conv.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
            M = M_new
            m_exp = 1.0
            if verbose and it % 20 == 0:
                print(f"      LPFW it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 3.5)

        f = f_new

        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.003 * min(no_improve // 80, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def iql_refine(f0, time_limit, n_iter_max=200):
    """Iteratively reweighted quadratic: min a^T W a s.t. sum(a)=1, a>=0.
    Where W is Hankel with entries W[j,m] = w_{j+m}, w_k = 1/(c[k]+eps).
    Uses projected gradient with backtracking.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = 0.01
    eps = 1e-6

    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute weights
        w = 1.0 / (conv + eps)
        w = w / w.sum()  # normalize

        # Build Hankel matrix W[j,m] = w[j+m] (for valid j+m < 2n-1)
        # Gradient of a^T W a = 2 * W @ a
        # But W is (n x n) with W[j,m] = w[j+m] if j+m < 2n-1 else 0
        # Actually W[j,m] = sum_k w_k * 1[k = j+m] = w[j+m]
        # So W is the top-left n x n submatrix of the full Hankel matrix
        W = np.zeros((n, n))
        for j in range(n):
            for m in range(n):
                k = j + m
                if k < 2 * n - 1:
                    W[j, m] = w[k]

        grad = 2.0 * (W @ f)

        # Backtracking projected gradient
        step = lr
        for _ in range(15):
            f_trial = project_simplex(f - step * grad)
            c_trial = fast_convolve(f_trial).max()
            c_cur = fast_convolve(f).max()
            if c_trial < c_cur - 1e-15:
                break
            step *= 0.5

        f = project_simplex(f - step * grad)
        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            break

        lr = max(lr * 0.995, 0.001)

    return best_f, best_obj


def coord_desc(f0, time_limit):
    """Coordinate descent with multi-scale multipliers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    multipliers = [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.2, 3.0, 5.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-12 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.001, mom=0.94, top_k=30, temp=0.00003):
    """Subgradient polishing with softmax-weighted active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9999
        cur_temp *= 1.004

        if no_improve > 350:
            rng = np.random.default_rng((it_count * 7919 + 13) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum (random phases)."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def gaussian_init(n, center=0.5, width=0.55, seed=0):
    """Gaussian window initialization with small noise."""
    rng = np.random.default_rng(seed)
    x = np.arange(n) / max(1, n - 1)
    c = center
    w = width
    g = np.exp(-4.0 * ((x - c) / w) ** 2)
    g += rng.standard_normal(n) * 0.005
    g = np.maximum(g, 0)
    return g / g.sum()


def cosine_init(n, power=1.0):
    """Raised cosine window."""
    x = np.arange(n) / max(1, n - 1)
    c = 0.5 * (1 + np.cos(np.pi * (2 * x - 1))) ** power
    c = np.maximum(c, 1e-10)
    return c / c.sum()


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, maintain simplex."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    return fi


def make_palindromic(a, n):
    """Create palindromic sequence of length n from first half of a."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])

            print(f"  Intensive [{name}] n={len(f_init)}, t={t_alloc:.0f}s")
            a_opt, _ = lpfw_optimize(f_init, time_limit=t_alloc,
                                     use_spectral=use_spec, spectral_bw=bw,
                                     full_interval=full_int, top_k=tk)
            _try(a_opt, f"intense_{name}")

    # === Phase 3: IRL-QP refinement (150s) ===
    print(f"\n=== Phase 3: IRL-QP (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_irl = min(rem * 0.3, 100)
            print(f"  IRL ref, t={t_irl:.0f}s")
            a_opt, _ = iql_refine(best_seq, time_limit=t_irl, n_iter_max=150)
            _try(a_opt, "irl")

            # LP-FW after IRL
            rem = deadline - time.monotonic()
            if rem > 15:
                t_after = min(rem * 0.3, 60)
                a_opt2, _ = lpfw_optimize(a_opt, time_limit=t_after,
                                          use_spectral=True, spectral_bw=0.35,
                                          full_interval=10, top_k=30)
                _try(a_opt2, "irl_then_lpfw")

    # === Phase 4: Multi-length exploration (200s) ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        base = best_seq
        base_n = len(base)
        target_ns = sorted(set([
            base_n, base_n + 32, base_n + 64, base_n + 128,
            base_n - 64 if base_n > 384 else 384,
            512, 640, 768, 832, 896, 1024, 1152
        ]))
        target_ns = [t for t in target_ns if 192 <= t <= 1536 and t != base_n]

        for tn in target_ns[:5]:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            fi = interpolate_seq(base, tn)
            fi = project_simplex(fi + rng.standard_normal(tn) * 0.003)
            t_alloc = min(rem * 0.25, 70)
            print(f"  MultLen n={tn}, t={t_alloc:.0f}s")
            a_opt, _ = lpfw_optimize(fi, time_limit=t_alloc,
                                     use_spectral=True, spectral_bw=0.35,
                                     full_interval=12, top_k=25)
            _try(a_opt, f"ml_{tn}")

    # === Phase 5: Coordinate descent (250s) ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.4, 180)
            print(f"  CD pass 1, t={t_cd:.0f}s")
            a_opt, _ = coord_desc(best_seq, time_limit=t_cd)
            _try(a_opt, "cd1")

            rem = deadline - time.monotonic()
            if rem > 20:
                t_cd2 = min(rem * 0.3, 100)
                print(f"  CD pass 2, t={t_cd2:.0f}s")
                a_opt2, _ = coord_desc(a_opt, time_limit=t_cd2)
                _try(a_opt2, "cd2")

    # === Phase 6: Subgradient polish (150s) ===
    print(f"\n=== Phase 6: SG Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 20:
            t_sg = min(rem * 0.4, 100)
            a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                                 lr=0.0008, mom=0.94,
                                 top_k=max(20, n_best // 6), temp=0.00002)
            _try(a_opt, "sg1")

            rem = deadline - time.monotonic()
            if rem > 15:
                t_sg2 = min(rem * 0.3, 60)
                a_opt2, _ = sg_polish(a_opt, time_limit=t_sg2,
                                      lr=0.0005, mom=0.95,
                                      top_k=max(10, n_best // 10), temp=0.000008)
                _try(a_opt2, "sg2")

    # === Phase 7: Final LP-FW + perturbations (remaining time) ===
    print(f"\n=== Phase 7: Final (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 15:
            t_f = min(rem * 0.35, 60)
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_f,
                                     use_spectral=True, spectral_bw=0.3,
                                     full_interval=8, top_k=40)
            _try(a_opt, "final_lpfw")

        for i in range(4):
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            ns = 0.001 + 0.0015 * i
            fp = project_simplex(best_seq + rng.standard_normal(n_best) * ns)
            tp = min(rem * 0.2, 25)
            a_opt, _ = lpfw_optimize(fp, time_limit=tp,
                                     use_spectral=False, full_interval=8, top_k=25)
            _try(a_opt, f"fpert_{i}")

        # Spectral variants
        rem = deadline - time.monotonic()
        for bw in [0.25, 0.4, 0.55]:
            rem = deadline - time.monotonic()
            if rem < 5:
                break
            fs = spectral_smooth_simplex(best_seq, bandwidth=bw)
            a_opt, _ = lpfw_optimize(fs, time_limit=min(rem * 0.3, 20),
                                     use_spectral=False, full_interval=5, top_k=20)
            _try(a_opt, f"spec_{bw}")

    # Fallback
    if best_seq is None:
        best_seq = parent

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\nFINAL: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
