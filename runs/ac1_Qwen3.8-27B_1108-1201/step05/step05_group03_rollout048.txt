def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(f, T, top_k=50):

    n = len(f)
    conv = fast_convolve(f)

    pv = conv[peaks]
    max_pv = pv.max()
    scaled = T * (pv - max_pv)
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws

    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue

def lp_fw_step(f, M, n):
    """Single LP-FW step: solve LP and return direction."""
    c = -np.ones(n)
    # Build constraint matrix sparsely
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs > 1e-15:
            return g / gs
    return None

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def structured_init(n, kind='raised_cosine'):
    if kind == 'raised_cosine':
        x = np.linspace(0, np.pi, n)
        return np.maximum(np.sin(x) ** 2, 0) / np.maximum(np.sin(x) ** 2, 1e-10).sum()
    elif kind == 'gaussian':
        x = np.linspace(-3, 3, n)
        g = np.exp(-x**2 / 2)
        return g / g.sum()
    elif kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'triangular':
        x = np.linspace(0, 1, n)
        t = np.maximum(1 - 2 * np.abs(x - 0.5), 0)
        return t / t.sum()
    elif kind == 'plateau':
        x = np.linspace(0, 1, n)
        p = np.ones(n)
        fade = max(1, n // 8)
        for i in range(fade):
            p[i] = i / fade
            p[-1-i] = i / fade
        return p / p.sum()
    else:
        return np.ones(n) / n
