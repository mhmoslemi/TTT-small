def solve_lp_topk(a, M, k_list):
    """Solve max sum(g) s.t. (a*g)[k] <= M for k in k_list, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A_ub = np.zeros((len(k_list), n))
    for i, k in enumerate(k_list):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[i, js:je+1] = a[k - np.arange(js, je+1)]
    b_ub = np.full(len(k_list), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, p):
    """Compute smoothed gradient: 2 * sum_k p[k] * a[k-m] for k>=m.
    Returns gradient of smooth approximation to max_k (a*a)[k]."""
    n = len(a)
    grad = np.zeros(n)
    for m in range(n):
        grad[m] = 2.0 * np.dot(p[m:], a[:n - m])
    return grad


def solve_lp_oracle(a, M, k_list=None):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in k_list, g >= 0.
    If k_list is None, use all constraints."""
    n = len(a)
    if k_list is None:
        k_list = list(range(n))
    k_list = sorted(k_list)
    c = -np.ones(n)
    A_ub = np.zeros((len(k_list), n))
    for i, k in enumerate(k_list):
        A_ub[i, :k + 1] = a[k::-1]
    b_ub = np.full(len(k_list), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def get_top_k_indices(conv, k, min_ratio=0.85):
    """Get indices of top-k peaks, plus any peaks above min_ratio * max."""
    n = len(conv)
    max_val = conv.max()
    # Include all peaks above threshold
    threshold = max_val * min_ratio
    above = np.where(conv >= threshold)[0]
    if len(above) <= k:
        return above
    # Otherwise pick top k
    top_k = np.argpartition(conv, -k)[-k:]
    return np.sort(top_k)

            break

        # Get top-k peaks for LP
        k_idx = get_top_k_indices(conv, k_max, min_ratio=0.85)
        # Only use first n lags (0 to n-1) since conv is symmetric
        k_idx = k_idx[k_idx < n]
        if len(k_idx) < 3:
            k_idx = np.arange(min(10, n))

        g = solve_lp_oracle(a, M, k_list=list(k_idx))
        if g is None or g.sum() < 1e-15:
            no_improve += 1
            if no_improve > 50:
                # Restart with perturbation
                rng = np.random.default_rng(it * 7919 + 13)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0
            continue
        g = g / g.sum()

        if np.max(np.abs(g - a)) < 1e-12:
            no_improve += 1
            if no_improve > 60:
                rng = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            at = (1.0 - alpha) * a + alpha * g
            at = np.maximum(at, 0.0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and it % 100 == 0:
            print(f"    FW-LP it={it}, best_obj={best_obj:.10f}, M={M:.10f}")

    return best_a, best_obj


def gradient_optimize(a0, time_limit, T_init=None, vr=0.9, verbose=False):
    """Fast smoothed gradient descent with adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    if T_init is None:
        T_init = max(conv.max() / 10.0, 1e-8)
    T = T_init
    lr = 0.5
    best_a = a.copy()
    best_obj = conv.max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.02:
            break

        conv = fast_convolve(a)

        # Adaptive temperature: focus on top peaks
        max_cv = conv.max()
        if max_cv > 0:
            T = max(T * 0.999, max_cv / 50.0)

        # Softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones_like(conv) / len(conv)
        else:
            p = p / p_sum

        # Compute smoothed gradient
        # Only need first n lags
        p_use = p[:n]
        grad = compute_smooth_grad(a, p_use)

        # Adaptive learning rate with backtracking
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                lr = trial_lr * 1.05
                break

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95

        # Reset LR periodically
        if no_improve > 50:
            lr = 0.5
            T = max_cv / 20.0

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise_scale = 0.002 + 0.003 * min(1.0, (no_improve - 200) / 500.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = 0.5
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    Grad it={it}, best_obj={best_obj:.10f}, T={T:.8f}, lr={lr:.4f}")

    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=False):
    """Fine coordinate-wise optimization targeting the highest peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv[:n]))  # Only consider first half

        # Find which coordinates affect this peak
        # (a*a)[k_peak] = sum_{j=0}^{k_peak} a[j] * a[k_peak - j]
        # Gradient at coordinate m: 2*a[k_peak - m] for m <= k_peak
        # So coordinates 0..k_peak affect this peak

        # Try to reduce this peak by moving mass from high-contribution
        # coordinates to low-contribution ones
        if k_peak > 0 and k_peak < n - 1:
            # Compute contribution of each coordinate to this peak
            contrib = np.zeros(n)
            contrib[:k_peak + 1] = 2.0 * a[k_peak::-1]

            # Find coordinates with low contribution to ALL top peaks
            top_peaks = np.argpartition(conv[:n], -5)[-5:]
            top_peaks = top_peaks[top_peaks < n]
            low_contrib = np.ones(n)
            for kp in top_peaks:
                low_contrib[:kp + 1] += a[kp - np.arange(kp + 1)]**2

            # Try moving mass from high-contrib to low-contrib coordinates
            high_idx = np.argmax(contrib[:k_peak + 1])
            # Move small amount from high_idx to a low-contribution coordinate
            low_mask = low_contrib > best_obj * 0.1
            low_candidates = np.where(low_mask)[0]
            if len(low_candidates) > 0:
                # Pick the one with lowest total contribution
                target = low_candidates[np.argmin(low_contrib[low_candidates])]
                if target != high_idx:
                    for eps in [0.001, 0.005, 0.01, 0.02]:
                        candidate = a.copy()
                        amt = min(eps, a[high_idx])
                        candidate[high_idx] -= amt
                        candidate[target] += amt
                        candidate = np.maximum(candidate, 0)
                        cs = candidate.sum()
                        if cs < 1e-15:
                            continue
                        candidate = candidate / cs
                        cv = fast_convolve(candidate).max()
                        if cv < best_obj - 1e-15:
                            a = candidate
                            best_obj = cv
                            best_a = a.copy()
                            break

        # Also try random small perturbations
        if it % 10 == 0:
            rng = np.random.default_rng((it * 12345 + 7) % (2**31))
            pert = rng.standard_normal(n) * 0.001
            candidate = project_simplex(a + pert)
            cv = fast_convolve(candidate).max()
            if cv < best_obj - 1e-15:
                a = candidate
                best_obj = cv
                best_a = a.copy()

    return best_a, best_obj


def mixed_optimize(a0, time_limit, verbose=False):
    """Alternates between gradient steps and FW-LP steps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    T = max(conv.max() / 10.0, 1e-8)
    lr = 0.5

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    mode = "grad"  # alternate between grad and fw

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Switch mode every 50 iterations or based on progress
        if it % 50 == 0:
            mode = "fw" if mode == "grad" else "grad"

        if mode == "grad" and remaining > 2.0:
            # Gradient step
            conv = fast_convolve(a)
            max_cv = conv.max()
            T = max(T * 0.999, max_cv / 50.0)

            scaled = conv / T
            scaled -= scaled.max()
            p = np.exp(scaled)
            p_sum = p.sum()
            if p_sum < 1e-30:
                p = np.ones_like(conv) / len(conv)
            else:
                p = p / p_sum

            grad = compute_smooth_grad(a, p[:n])

            best_candidate = a
            best_val = conv.max()
            for trial_lr in [lr, lr * 0.5, lr * 0.25]:
                candidate = project_simplex(a - trial_lr * grad)
                c = fast_convolve(candidate)
                val = c.max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_candidate = candidate
                    lr = trial_lr * 1.02
                    break

            a = best_candidate
            obj = best_val

        else:
            # FW-LP step (with time limit on LP)
            conv = fast_convolve(a)
            M = conv.max()
            k_idx = get_top_k_indices(conv, 30, min_ratio=0.85)
            k_idx = k_idx[k_idx < n]
            if len(k_idx) < 3:
                k_idx = np.arange(min(10, n))

            g = solve_lp_oracle(a, M, k_list=list(k_idx))
            if g is not None and g.sum() > 1e-15:
                g = g / g.sum()
                best_alpha = 0.0
                best_val = M
                for alpha in [0.1, 0.2, 0.4, 0.6, 1.0]:
                    at = (1.0 - alpha) * a + alpha * g
                    at = np.maximum(at, 0.0)
                    st = at.sum()
                    if st < 1e-15:
                        continue
                    at = at / st
                    val = fast_convolve(at).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = alpha
                if best_alpha > 1e-10:
                    a = (1.0 - best_alpha) * a + best_alpha * g
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    obj = fast_convolve(a).max()
                else:
                    obj = M
            else:
                obj = M

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart logic
        if no_improve > 300:
            rng = np.random.default_rng((it * 31415 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = max(fast_convolve(a).max() / 10.0, 1e-8)
            lr = 0.5
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    Mixed it={it}, mode={mode}, best_obj={best_obj:.10f}")

    return best_a, best_obj


def generate_patterns(n, rng, parent=None):
    """Generate diverse initial patterns."""
    patterns = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    patterns.append(np.ones(n) / n)

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.05
    patterns.append(c / c.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / (0.1 * n))**2)
    g2 = np.exp(-0.5 * ((x - 0.7 * (n - 1)) / (0.1 * n))**2)
    patterns.append((g1 + g2) / (g1 + g2).sum())

    # Three peaks
    g3 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.08 * n))**2) + \
         np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.08 * n))**2) + \
         np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.08 * n))**2)
    patterns.append(g3 / g3.sum())

    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)**0.5
    patterns.append(edge_w / edge_w.sum())

    # Truncated Gaussian (wide)
    g_wide = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n))**2)
    patterns.append(g_wide / g_wide.sum())

    # Truncated Gaussian (narrow)
    g_narrow = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n))**2)
    patterns.append(g_narrow / g_narrow.sum())

    # Periodic
    per = np.abs(np.sin(2 * np.pi * 3 * x / n)) + 0.1
    patterns.append(per / per.sum())

    # Random Dirichlet
    patterns.append(rng.dirichlet(np.ones(n) * 0.5))
    patterns.append(rng.dirichlet(np.ones(n) * 0.1))

    # Interpolated parent
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            patterns.append(px / ps)

    return patterns


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for optimal sequence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    best_n = 0

    def try_seq(a, label=""):
        nonlocal best_seq, best_score, best_n
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            best_n = len(a_norm)
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    # === Phase 1: Quick multi-n exploration (~200s) ===
    print("\n=== Phase 1: Quick Multi-n Exploration ===")
    t_phase_start = time.monotonic()

    # Candidate n values around parent
    candidate_ns = sorted(set([
        max(128, n_parent // 2),
        n_parent,
        n_parent * 2 if n_parent * 2 <= 4096 else max(256, n_parent + 128),
        512, 768, 1024, 1536, 2048,
    ]))
    # Filter to reasonable range
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 30:
            break

        # Generate initial patterns
        inits = generate_patterns(n_try, rng, parent)

        best_n_init = None
        best_n_score = float('inf')

        for idx, init in enumerate(inits[:5]):  # Try top 5 patterns
            remaining = deadline - time.monotonic() - 1150
            if remaining < 10:
                break
            t_alloc = min(remaining, 25)
            a_opt, _ = gradient_optimize(init, time_limit=t_alloc)
            score = fast_convolve(a_opt).max()
            if score < best_n_score:
                best_n_score = score
                best_n_init = a_opt

        if best_n_init is not None:
            try_seq(best_n_init, f"grad_n{n_try}")

            # Quick FW-LP polish
            remaining = deadline - time.monotonic() - 1100
            if remaining > 15:
                a_opt2, _ = fw_lp_optimize(best_n_init, time_limit=min(remaining, 30), k_max=30)
                try_seq(a_opt2, f"fw_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 2: Deep optimization on best n (~600s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Mixed optimize (alternates gradient + FW-LP)
        remaining = deadline - time.monotonic() - 600
        if remaining > 50:
            t_alloc = min(remaining, 300)
            print(f"  Mixed optimize n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = mixed_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "mixed_1")

        # Pure gradient with higher iterations
        remaining = deadline - time.monotonic() - 500
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print(f"  Gradient optimize n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = gradient_optimize(best_seq, time_limit=t_alloc, T_init=1e-6)
            try_seq(a_opt, "grad_deep")

        # FW-LP with more constraints
        remaining = deadline - time.monotonic() - 350
        if remaining > 30:
            t_alloc = min(remaining, 150)
            print(f"  FW-LP (k=50) n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, k_max=50)
            try_seq(a_opt, "fw_deep")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 3: Try different n with best structure (~300s) ===
    print("\n=== Phase 3: Alternative n values ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        # Try neighboring n values
        for n_try in [best_n - 64, best_n + 64, best_n + 128, best_n * 2 // 3]:
            if n_try < 256 or n_try > 4096 or n_try == best_n:
                continue
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break

            # Interpolate best to new n
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, best_n), best_seq)
            a_interp = np.maximum(a_interp, 0)
            a_interp = a_interp / a_interp.sum()

            t_alloc = min(remaining, 40)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            a_opt, _ = gradient_optimize(a_interp, time_limit=t_alloc)
            try_seq(a_opt, f"alt_n{n_try}")

    # Also try a few completely different initializations at best_n
    if best_seq is not None:
        for trial in range(3):
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break
            inits = generate_patterns(best_n, rng, best_seq)
            # Pick a random one different from parent
            idx = rng.integers(0, len(inits))
            t_alloc = min(remaining, 30)
            a_opt, _ = gradient_optimize(inits[idx], time_limit=t_alloc)
            try_seq(a_opt, f"rand_init_{trial}")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 4: Fine polishing (~250s) ===
    print("\n=== Phase 4: Fine Polishing ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        # Coordinate polish
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 60)
            print(f"  Coordinate polish, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_polish")

        # Very fine gradient (small T, small LR)
        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 50)
            print(f"  Fine gradient, t={t_alloc:.0f}s")
            a_opt, _ = gradient_optimize(best_seq, time_limit=t_alloc, T_init=1e-7)
            try_seq(a_opt, "fine_grad")

        # Multiple short FW-LP runs with perturbation
        for pert_scale in [0.001, 0.003, 0.01]:
            remaining = deadline - time.monotonic() - 40
            if remaining < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(best_n) * pert_scale)
            t_alloc = min(remaining, 30)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, k_max=30)
            try_seq(a_opt, f"fw_pert_{pert_scale}")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 5: Final attempts (~150s) ===
    print("\n=== Phase 5: Final Attempts ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        # One more mixed run
        remaining = deadline - time.monotonic() - 30
        if remaining > 30:
            t_alloc = min(remaining, 80)
            print(f"  Final mixed, t={t_alloc:.0f}s")
            a_opt, _ = mixed_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_mixed")

        # Tiny perturbations
        for _ in range(5):
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            noise = rng.standard_normal(best_n) * 0.0005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = gradient_optimize(f_pert, time_limit=min(remaining, 10), T_init=1e-7)
            try_seq(a_opt, "tiny_pert")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
