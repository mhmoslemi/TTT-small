"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Iterative Linearized QP (ILQ) with active-set LP
- FW-LP with full constraints for global improvements
- Game iterate (primal-dual) for convergence acceleration
- Peak-shaving coordinate descent for final polish
- Multi-restart with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def convolve_with_fixed(f, u):
    """Compute full convolution f*u where f is fixed. Shape: (2n-1,) for u of length n."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    U = np.fft.rfft(u, fft_size)
    return np.fft.irfft(F * U, fft_size)[:size]

def ilq_optimize(a0, time_limit, active_frac=0.01, verbose=False):
    """Iterative Linearized QP: solve small LPs on active set."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find active set (peaks within active_frac relative to max)
        threshold = M * (1.0 - active_frac)
        active = np.where(conv >= threshold)[0]
        
        # Cap active set size for efficiency
        if len(active) > 50:
            # Keep only the top 50
            active_vals = conv[active]
            top_idx = np.argsort(active_vals)[-50:]
            active = active[top_idx]
        
        # Build LP:
        # min t
        # s.t. 2*(a*u)[k] - t <= conv[k] for k in active
        #      u >= 0, sum(u) = 1
        #
        # Variables: [u_0, ..., u_{n-1}, t] (n+1 variables)
        # Objective: minimize t => c = [0,...,0, 1]
        # Inequality: 2*B_a[k,:] @ u - t <= conv[k]
        #   Row: [2*B_a[k,0], ..., 2*B_a[k,n-1], -1] <= conv[k]
        # Equality: sum(u) = 1 => [1,...,1, 0] = 1
        # Bounds: u >= 0, t free (or t >= 0)
        
        n_active = len(active)
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize t
        
        A_ub = np.zeros((n_active, n + 1))
        b_ub = np.zeros(n_active)
        
        for i, k in enumerate(active):
            # (a*u)[k] = sum_j a[j] * u[k-j] for valid j
            # This equals sum_j a[k-j] * u[j] (reindexing)
            # So the row is: 2*a[k-j] for j = 0,...,n-1 (where 0<=k-j<n)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # For u[j], the coefficient is 2*a[k-j]
            # k-j ranges from k-je to k-js
            coeffs = 2.0 * a[k - np.arange(js, je + 1)]
            A_ub[i, js:je+1] = coeffs
            A_ub[i, n] = -1.0
            b_ub[i] = conv[k]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(None, None)]  # u >= 0, t free
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
            if res.success:
                u = np.maximum(res.x[:n], 0.0)
                u_sum = u.sum()
                if u_sum < 1e-15:
                    continue
                u = u / u_sum
                
                # Line search
                best_alpha = 0.0
                best_val = M
                for alpha in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
                    candidate = (1.0 - alpha) * a + alpha * u
                    candidate = np.maximum(candidate, 0.0)
                    cs = candidate.sum()
                    if cs < 1e-15:
                        continue
                    candidate = candidate / cs
                    val = fast_convolve(candidate).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = alpha
                
                if best_alpha > 1e-8:
                    a = (1.0 - best_alpha) * a + best_alpha * u
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    if best_val < best_obj - 1e-14:
                        best_obj = best_val
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1
        
        # Adaptive active set
        if no_improve > 20:
            active_frac = min(0.1, active_frac * 1.5)
            no_improve = 0
        if no_improve > 50:
            active_frac = 0.01
            # Restart with perturbation
            rng = np.random.default_rng(it * 7919)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    ILQ it={it}, obj={best_obj:.8f}, active={len(active)}")
    
    return best_a, best_obj

def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def game_iterate(a0, time_limit, T_init=0.0005, verbose=False):
    """Primal-dual game iteration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Solve QP: min a'^T M(p) a' s.t. a' in simplex
        # M(p)[j,m] = p[j+m]
        # Use projected gradient with backtracking for the QP
        a_qp = a.copy()
        for _ in range(50):  # Inner iterations
            grad = 2.0 * convolve_with_fixed(p, a_qp)
            # Projected gradient step
            for lr in [0.5, 0.25, 0.1, 0.05, 0.01]:
                candidate = project_simplex(a_qp - lr * grad)
                obj_new = candidate @ convolve_with_fixed(p, candidate)
                obj_old = a_qp @ convolve_with_fixed(p, a_qp)
                if obj_new < obj_old:
                    a_qp = candidate
                    break
        
        a = a_qp
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            T *= 1.2
        if no_improve > 200:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0
    
    return best_a, best_obj

def smoothed_adagrad(a0, time_limit, T_init=0.0003, lr_init=0.02):
    """Smoothed gradient with AdaGrad-style adaptive steps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init
    G = np.zeros(n)  # Accumulated squared gradients
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        grad = 2.0 * convolve_with_fixed(p, a)
        G += grad ** 2
        adaptive_lr = lr / np.sqrt(G + 1e-8)
        
        candidate = project_simplex(a - adaptive_lr)
        val = fast_convolve(candidate).max()
        
        if val < best_obj - 1e-14:
            a = candidate
            best_obj = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            candidate = project_simplex(a - adaptive_lr * 0.5)
            val = fast_convolve(candidate).max()
            if val < fast_convolve(a).max() - 1e-14:
                a = candidate
        
        if no_improve > 30:
            T *= 1.05
        if no_improve > 100:
            T *= 0.8
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            G = np.zeros(n)
            no_improve = 0
    
    return best_a, best_obj

"""
Minimizes 2*n*max(auto-convolution(a))/(sum(a))^2 using a hybrid approach:
- Iterative Linearized QP (ILQ): solves small LPs on active peak sets
- Frank-Wolfe with full LP oracle for global improvements

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_fixed(f, u):
    """Full convolution f*u where f is fixed (precompute FFT of f outside for speed)."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    U = np.fft.rfft(u, fft_size)
    return np.fft.irfft(F * U, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_ilq_lp(a, conv, active, n):
    """
    Solve the linearized LP for ILQ.
    min t
    s.t. 2*(a*u)[k] - t <= conv[k] for k in active
         u >= 0, sum(u) = 1
    Returns (u, t_opt) or (None, None) if infeasible.
    """
    n_active = len(active)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t

    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)

    for i, k in enumerate(active):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            A_ub[i, n] = -1.0
            b_ub[i] = conv[k]
            continue
        coeffs = 2.0 * a[k - np.arange(js, je + 1)]
        A_ub[i, js:je + 1] = coeffs
        A_ub[i, n] = -1.0
        b_ub[i] = conv[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 2.0})
        if res.success:
            u = np.maximum(res.x[:n], 0.0)
            u_sum = u.sum()
            if u_sum < 1e-15:
                return None, None
            u = u / u_sum
            return u, float(res.fun)
    except Exception:
        pass
    return None, None

            continue

        # Line search

                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            a = (1.0 - best_alpha) * a + best_alpha * u
            a = np.maximum(a, 0.0)
            a = a / a.sum()

            if best_val < best_obj - 1e-14:

                best_a = a.copy()
                no_improve = 0

def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        remaining = time_limit - (time.monotonic() - t0)

        g = solve_lp_oracle(f, M * m_exp)

        g_sum = g.sum()
        if g_sum < 1e-15:

            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 3.0)

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

            if val < best_val:

def game_iterate(a0, time_limit, T_init=0.0004, verbose=False):
    """
    Primal-dual game iteration:
    - Dual: p = softmax(conv(a)/T)
    - Primal: projected gradient on min a^T M(p) a
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Inner: minimize a^T M(p) a via projected gradient
        a_new = a.copy()
        lr_inner = 0.3
        for _ in range(30):
            grad = 2.0 * convolve_fixed(p, a_new)
            a_trial = project_simplex(a_new - lr_inner * grad)
            obj_new = a_new @ convolve_fixed(p, a_new)
            obj_trial = a_trial @ convolve_fixed(p, a_trial)
            if obj_trial < obj_new:
                a_new = a_trial
                lr_inner = min(lr_inner * 1.1, 1.0)
            else:
                lr_inner *= 0.5
                if lr_inner < 0.01:
                    break

        a = a_new
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt temperature
        if no_improve > 40:
            T *= 1.15
        if no_improve > 100:
            T *= 0.75
        if no_improve > 300:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0

    return best_a, best_obj


def smoothed_adagrad(a0, time_limit, T_init=0.0003, lr_init=0.015, verbose=False):
    """Smoothed gradient descent with AdaGrad adaptive step sizes."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init
    G = np.zeros(n)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    mom = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * convolve_fixed(p, a)

        # Momentum
        mom = 0.9 * mom + 0.1 * grad
        G += mom ** 2
        step = lr / np.sqrt(G + 1e-10)

        # Try multiple step scales
        best_candidate = a
        best_val = conv.max()
        for scale in [1.0, 0.5, 0.25, 0.1]:
            candidate = project_simplex(a - scale * step)
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            T *= 1.05
            lr *= 0.99
        if no_improve > 100:
            T *= 0.85
        if no_improve > 400:

            G = np.zeros(n)

def peak_shave_optimize(a0, time_limit, verbose=False):
    """

    variable adjustments using a small local QP.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        conv = fast_convolve(a)
        M = conv.max()
        k_peak = int(np.argmax(conv))

        # Variables that affect this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        n_aff = len(affected)

        if n_aff == 0 or n_aff > n:
            break

        # Build small LP: reduce (a*u)[k_peak] while keeping sum=1, a>=0
        # Linearized: minimize 2*(a*u)[k_peak] s.t. u in simplex
        # This is just: minimize sum_j a[k_peak-j]*u[j] s.t. u in simplex
        # Optimal: put all mass on the j with smallest a[k_peak-j]
        costs = a[k_peak - affected]
        min_cost = costs.min()
        
        # But we also need to not increase other peaks too much
        # Use a regularized version: min costs @ u + reg * ||u - a[affected]||^2
        # Simpler: just try moving mass to reduce the peak
        # Find the variable with highest contribution to the peak
        contribs = a[affected] * a[k_peak - affected]
        
        # Try transferring mass from high-contrib vars to low-contrib vars
        if n_aff >= 2:
            max_c_idx = np.argmax(contribs)
            min_c_idx = np.argmin(contribs)
            if max_c_idx != min_c_idx:
                src = affected[max_c_idx]
                dst = affected[min_c_idx]
                # Try different transfer amounts
                best_alpha = 0.0
                best_val = M
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    transfer = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= transfer
                    a_trial[dst] += transfer
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    val = fast_convolve(a_trial).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = transfer

                if best_alpha > 0:
                    a[src] -= best_alpha
                    a[dst] += best_alpha
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    val = fast_convolve(a).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 100:
            rng = np.random.default_rng(it * 48271 + 1)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations for multi-start."""
    inits = []
    if parent is not None:
        inits.append(parent.copy())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random Dirichlet with various concentration
    for alpha_val in [0.5, 1.0, 3.0, 10.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha_val))
    
    # Edge concentration
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())
    
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())
    
    # Sawtooth
    saw = (x % (n // 8 + 1)) / (n // 8 + 1)
    inits.append(saw / saw.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for minimizing
    2*n*max(auto-convolution)/(sum)^2 with non-negative sequences.
    
    Phases:
    1. Quick exploration with ILQ on diverse initializations
    2. Deep ILQ refinement on best found
    3. FW-LP for global structure
    4. Game iterate + smoothed gradient
    5. Peak shaving + final ILQ polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_best = len(parent)

    # === Phase 1: Quick multi-init exploration with ILQ (~400s) ===
    print(f"\n=== Phase 1: ILQ Multi-Init (n={n_best}) ===")
    t_phase = time.monotonic()
    
    inits = generate_diverse_inits(n_best, parent, rng)
    # Sort by initial objective
    init_objs = []
    for init in inits:
        obj = fast_convolve(init).max()
        init_objs.append((obj, init))
    init_objs.sort(key=lambda x: x[0])

    for (init_obj, init_a) in init_objs[:5]:
        remaining = deadline - time.monotonic() - 1300
        if remaining < 20:
            break
        t_alloc = min(remaining, 100)
        label = "ilq_init"
        a_opt, obj = ilq_optimize(init_a, time_limit=t_alloc, active_frac=0.01)
        try_seq(a_opt, f"ilq_init_{obj:.6f}")

    print(f"  Phase 1 done: best={best_score:.8f}, elapsed={time.monotonic()-t_phase:.0f}s")

    # === Phase 2: Deep ILQ on best (~450s) ===
    print(f"\n=== Phase 2: Deep ILQ ===")
    t_phase = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 800
        if remaining > 50:
            t_alloc = min(remaining, 450)
            print(f"  Deep ILQ t={t_alloc:.0f}s")
            a_opt, obj = ilq_optimize(best_seq, time_limit=t_alloc, active_frac=0.005)
            try_seq(a_opt, "deep_ilq")

        # Second pass with different frac
        remaining = deadline - time.monotonic() - 500
        if remaining > 50:
            t_alloc = min(remaining, 200)
            a_opt, obj = ilq_optimize(best_seq, time_limit=t_alloc, active_frac=0.002)
            try_seq(a_opt, "ilq_fine")

    print(f"  Phase 2 done: best={best_score:.8f}, elapsed={time.monotonic()-t_phase:.0f}s")

    # === Phase 3: FW-LP (~300s) ===
    print(f"\n=== Phase 3: FW-LP ===")
    t_phase = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 400
        if remaining > 30:
            t_alloc = min(remaining, 300)
            print(f"  FW-LP t={t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw_lp")

        # Perturbed FW
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            for scale in [0.005, 0.015]:
                remaining = deadline - time.monotonic() - 150
                if remaining < 20:
                    break
                pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
                a_opt, obj = fw_lp_optimize(pert, time_limit=min(remaining, 100))
                try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 3 done: best={best_score:.8f}, elapsed={time.monotonic()-t_phase:.0f}s")

    # === Phase 4: Game iterate + Smoothed gradient (~250s) ===
    print(f"\n=== Phase 4: Game + Smoothed ===")
    t_phase = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 120)
            a_opt, obj = game_iterate(best_seq, time_limit=t_alloc, T_init=0.0003)
            try_seq(a_opt, "game")

        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            t_alloc = min(remaining, 100)
            a_opt, obj = smoothed_adagrad(best_seq, time_limit=t_alloc,
                                         T_init=0.0002, lr_init=0.01)
            try_seq(a_opt, "adagrad")

    print(f"  Phase 4 done: best={best_score:.8f}, elapsed={time.monotonic()-t_phase:.0f}s")

    # === Phase 5: Peak shaving + final polish (~remaining) ===
    print(f"\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 20:
            t_alloc = min(remaining, 80)
            a_opt, obj = peak_shave_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_shave")

        # Final ILQ with very tight active set
        remaining = deadline - time.monotonic() - 40
        if remaining > 15:
            a_opt, obj = ilq_optimize(best_seq, time_limit=min(remaining, 60),
                                      active_frac=0.003)
            try_seq(a_opt, "ilq_tight")

        # Tiny perturbations + quick ILQ
        for _ in range(3):
            remaining = deadline - time.monotonic() - 15
            if remaining < 10:
                break
            noise = rng.standard_normal(n_best) * 0.001
            pert = project_simplex(best_seq + noise)
            a_opt, obj = ilq_optimize(pert, time_limit=min(remaining, 20),
                                      active_frac=0.005)
            try_seq(a_opt, "perturb_ilq")

    print(f"  Phase 5 done: best={best_score:.8f}, elapsed={time.monotonic()-t_phase:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
