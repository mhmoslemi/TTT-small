import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """Make sequence symmetric: a[i] = (a[i] + a[n-1-i]) / 2, then renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_subgrad(a, conv, top_k=50, temp=0.001):
    """Compute weighted subgradient using softmax over top peaks."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
    
    return sg


def subgradient_optimize(a0, time_limit, lr=0.01, mom=0.9, top_k=50,
                         temp=0.0005, decay=0.9998, batch=1):
    """Projected subgradient descent with momentum."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgrad(a, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        cur_lr *= decay
        cur_temp *= 1.005
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 7)
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 4)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0
    
    return best_a, best_obj

def ant_colony_search(a0, time_limit, pop_size=20, verbose=False):
    """Ant-colony-like population search with block mutations."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    population = [best_a.copy()]
    
    gen = 0
    while time.monotonic() - t0 < time_limit:
        gen += 1
        
        # Generate new candidates
        for r in range(pop_size):
            if time.monotonic() - t0 > time_limit:
                break
            
            parent = population[r % len(population)]
            
            # Random mutation strategy
            strategy = rng.integers(0, 4)
            
            if strategy == 0:
                # Gaussian noise
                noise_scale = 0.005 * (1.0 + gen * 0.001)
                a_new = project_simplex(parent + rng.standard_normal(n) * noise_scale)

def ftrl_optimize(a0, time_limit, lr=0.01, top_k=50, temp=0.001):
    """Follow-the-Regularized-Leader (FTRL) optimization."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    Z = np.zeros(n)  # cumulative gradients
    G = np.zeros(n)  # cumulative absolute gradients
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgrad(a, conv, top_k=top_k, temp=temp)
        
        Z += sg
        G += np.abs(sg)
        
        # FTRL update: a_i = argmin_{x>=0, sum=1} sum_j Z_j * x_j + (1/eta) * |G_j| * x_j^2
        # With eta adapted per coordinate
        eta = 0.1 / np.sqrt(G + 1e-10)
        
        # Solve: minimize sum_j (Z_j * x_j + (1/eta_j) * x_j^2) subject to x >= 0, sum(x) = 1
        # KKT: Z_j + 2*(1/eta_j) * x_j = lambda for x_j > 0
        # x_j = max(0, (lambda - Z_j) * eta_j / 2)
        
        # Bisection on lambda
        lo = Z.min() - 1
        hi = Z.max() + 1
        
        for _ in range(50):
            mid = (lo + hi) / 2
            x_trial = np.maximum((mid - Z) * eta / 2, 0)
            if x_trial.sum() > 1:
                hi = mid
            else:
                lo = mid
        
        a_new = x_trial / x_trial.sum()
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        temp *= 1.003
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 7 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(a)
            Z = np.zeros(n)
            G = np.zeros(n)
            no_improve = 0
    
    return best_a, best_obj


def peak_shaving(a0, time_limit):
    """Directly target the peak: iteratively reduce the coordinates that contribute most to the peak."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find top peaks
        top_k = 10
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # For each peak, compute which coordinates to reduce
        reduce_mask = np.zeros(n)
        add_mask = np.zeros(n)
        
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # Contribution of coordinate i to peak k: a[i] * a[k-i]
            # We want to reduce coordinates where this product is large
            for j in range(js, je + 1):
                partner = k - j
                prod = a[j] * a[partner]
                reduce_mask[j] += prod
                # The partner also contributes
                if partner != j:
                    reduce_mask[partner] += prod * 0.5
        
        # Normalize masks
        rr = reduce_mask.sum()
        if rr > 1e-15:
            reduce_mask /= rr
        else:
            reduce_mask = np.ones(n) / n
        
        # Reduce high-contribution, add to low-contribution
        threshold = np.median(reduce_mask)
        reduce_coords = reduce_mask > threshold * 2
        add_coords = reduce_mask < threshold * 0.5
        
        if not add_coords.any():
            add_coords = reduce_mask < threshold
        
        # Transfer mass
        total_reduce = a[reduce_coords].sum() * 0.1
        n_add = add_coords.sum()
        if n_add > 0 and total_reduce > 1e-15:
            a_trial = a.copy()
            a_trial[reduce_coords] -= total_reduce / reduce_coords.sum()
            a_trial[add_coords] += total_reduce / n_add
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial /= ss
            
            conv_trial = fast_convolve(a_trial)
            obj_trial = conv_trial.max()
            
            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                conv = conv_trial
            else:
                # Try smaller transfer
                a_trial2 = a.copy()
                a_trial2[reduce_coords] -= total_reduce * 0.3 / reduce_coords.sum()
                a_trial2[add_coords] += total_reduce * 0.3 / n_add
                a_trial2 = np.maximum(a_trial2, 0)
                ss2 = a_trial2.sum()
                if ss2 > 1e-15:
                    a_trial2 /= ss2
                conv_trial2 = fast_convolve(a_trial2)
                obj_trial2 = conv_trial2.max()
                if obj_trial2 < best_obj - 1e-14:
                    best_obj = obj_trial2
                    best_a = a_trial2.copy()
                    a = a_trial2.copy()
                    conv = conv_trial2
        
        # Periodic restart
        if it % 500 == 0:
            rng_restart = np.random.default_rng(it)
            a = project_simplex(best_a + rng_restart.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
    
    return best_a, best_obj

def binary_search_polish(a0, time_limit):
    """Final polish: binary search on individual coordinates."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < 200:
        passes += 1
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-15:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            # Binary search for optimal a[j]
            lo = 0.0
            hi = 1.0
            x_new = x_old
            
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                
                # Compute conv with a[j] = mid, others scaled by om/s_others
                a_trial = a * (om / s_others)
                a_trial[j] = mid
                conv_trial = fast_convolve(a_trial)
                obj_trial = conv_trial.max()
                
                # Compare with current
                if obj_trial < conv.max() - 1e-14:
                    lo = mid
                    x_new = mid
                else:
                    hi = mid
            
            if abs(x_new - x_old) > 1e-13:
                a[j] = x_new
                others_mask = np.ones(n, dtype=bool)
                others_mask[j] = False
                s_others_new = 1.0 - x_new
                if s_others_new > 1e-14:
                    a[others_mask] *= s_others_new / s_others
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a /= ss
                
                conv = fast_convolve(a)
                obj = conv.max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    pass_improved = True
        
        if not pass_improved:
            if passes > 5:
                break
            # Small perturbation
            rng = np.random.default_rng(passes * 53 + 19)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            conv = fast_convolve(a)
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. FFT-based subgradient descent with adaptive temperature
2. Block coordinate descent (joint optimization of coordinate groups)
3. Direct peak shaving (targeted reduction of peak-contributing coordinates)
4. Population-based search with diverse mutations
5. Binary search polish for final convergence
6. Multi-length exploration and symmetry exploitation

Key improvements: block-level moves for faster convergence, direct peak targeting,
and better time allocation toward exploitation in later stages.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """Make sequence symmetric: a[i] = (a[i] + a[n-1-i]) / 2, then renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_subgrad(a, conv, top_k, temp):
    """Compute weighted subgradient using softmax over top peaks."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

    return sg


def subgradient_optimize(a0, time_limit, lr, top_k, temp, decay, mom):
    """Projected subgradient descent with momentum and adaptive restarts."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgrad(a, conv, top_k, cur_temp)
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        cur_lr *= decay
        cur_temp *= 1.004

        if no_improve > 150:
            noise_scale = 0.003 + 0.002 * min(no_improve // 100, 4)
            a = project_simplex(best_a + np.random.default_rng(it * 13 + 7).standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0

    return best_a, best_obj


def block_bcd(a0, time_limit, block_size):
    """Block coordinate descent: optimize groups of coordinates jointly."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    pass_count = 0
    no_improve = 0

    block_starts = list(range(0, n, block_size))

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        pass_improved = False

        for bs in block_starts:
            if time.monotonic() - t0 > time_limit:
                break

            be = min(bs + block_size, n)
            block_idx = np.arange(bs, be)
            block_vals = a[block_idx].copy()
            rest_sum = 1.0 - block_vals.sum()
            m = be - bs
            target_block_sum = 1.0 - rest_sum

            if target_block_sum < 1e-14 or target_block_sum > 1.0 + 1e-12:
                continue

            # Inner optimization: gradient descent on the block
            cur_block = np.maximum(block_vals, 0)
            cs = cur_block.sum()
            if cs > 1e-15:
                cur_block *= target_block_sum / cs
            else:
                cur_block = np.ones(m) * target_block_sum / m

            for inner_it in range(20):
                a_trial = a.copy()
                a_trial[block_idx] = cur_block
                conv_trial = fast_convolve(a_trial)
                obj_trial = conv_trial.max()

                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_a = a_trial.copy()
                    pass_improved = True

                # Compute subgradient for block coordinates
                k_peak = int(np.argmax(conv_trial))
                sg_block = np.zeros(m)
                for i_local, i in enumerate(block_idx):
                    ki = k_peak - i
                    if 0 <= ki < n:
                        sg_block[i_local] = 2.0 * a_trial[ki]

                # Gradient step
                lr_inner = 0.008
                cur_block = cur_block - lr_inner * sg_block
                cur_block = np.maximum(cur_block, 0)
                cs2 = cur_block.sum()
                if cs2 > 1e-15:
                    cur_block *= target_block_sum / cs2
                else:
                    cur_block = np.ones(m) * target_block_sum / m

            a[block_idx] = cur_block

        if not pass_improved:
            no_improve += 1
            if no_improve > 2:
                a = project_simplex(best_a + np.random.default_rng(pass_count * 31 + 17).standard_normal(n) * 0.003)
                conv = fast_convolve(a)
                no_improve = 0
                if pass_count > 30:
                    break

    return best_a, best_obj


def peak_shaving(a0, time_limit):
    """Directly target and reduce the peak of the auto-convolution."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Find top peaks and compute contribution masks
        top_k = min(10, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]

        reduce_scores = np.zeros(n)
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                partner = k - j
                prod = a[j] * a[partner]
                reduce_scores[j] += prod
                if partner != j and 0 <= partner < n:
                    reduce_scores[partner] += 0.5 * prod

        # Identify coordinates to reduce and add
        rr = reduce_scores.sum()
        if rr > 1e-15:
            reduce_scores_n = reduce_scores / rr
        else:
            reduce_scores_n = np.ones(n) / n

        threshold_med = np.median(reduce_scores_n)
        reduce_coords = reduce_scores_n > threshold_med * 2.5
        add_coords = reduce_scores_n < threshold_med * 0.4

        if not add_coords.any():
            add_coords = reduce_scores_n <= threshold_med

        if not reduce_coords.any() or not add_coords.any():
            continue

        n_reduce = reduce_coords.sum()
        n_add = add_coords.sum()
        total_mass = a[reduce_coords].sum() * 0.08

        if total_mass < 1e-15:
            continue

        # Try different transfer fractions
        for frac in [1.0, 0.6, 0.3]:
            transfer = total_mass * frac
            a_trial = a.copy()
            a_trial[reduce_coords] -= transfer / n_reduce
            a_trial[add_coords] += transfer / n_add
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-14:
                continue
            a_trial /= ss

            conv_trial = fast_convolve(a_trial)
            obj_trial = conv_trial.max()

            if obj_trial < best_obj - 1e-14:
                best_obj = obj_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                conv = conv_trial
                break

        # Periodic restart to avoid stagnation
        if it % 300 == 0:
            a = project_simplex(best_a + np.random.default_rng(it * 7).standard_normal(n) * 0.004)
            conv = fast_convolve(a)

    return best_a, best_obj


def population_search(a0, time_limit, pop_size):
    """Population-based search with diverse mutation strategies."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    population = [best_a.copy()]
    gen = 0

    while time.monotonic() - t0 < time_limit:
        gen += 1

        for r in range(pop_size):
            if time.monotonic() - t0 > time_limit:
                break

            parent = population[r % len(population)]
            strategy = rng.integers(0, 5)

            if strategy == 0:
                # Gaussian noise
                noise_scale = 0.004 + 0.002 * (gen // 50)
                a_new = project_simplex(parent + rng.standard_normal(n) * noise_scale)
            elif strategy == 1:
                # Block swap
                b1 = rng.integers(0, max(1, n - 16))
                b2 = rng.integers(0, max(1, n - 16))
                sz = min(16, n // 5)
                if abs(b1 - b2) < sz:
                    b2 = (b1 + n // 2) % max(1, n - sz)
                a_new = parent.copy()
                tmp = a_new[b1:b1 + sz].copy()
                a_new[b1:b1 + sz] = a_new[b2:b2 + sz]
                a_new[b2:b2 + sz] = tmp
                a_new = np.maximum(a_new, 0)
                ss = a_new.sum()
                if ss > 1e-15:
                    a_new /= ss
                else:
                    a_new = parent.copy()
            elif strategy == 2:
                # Crossover
                other = population[(r + 1 + rng.integers(0, len(population) - 1)) % len(population)]
                mask = rng.random(n) > 0.5
                a_new = np.where(mask, parent, other)
                a_new = np.maximum(a_new, 0)
                ss = a_new.sum()
                if ss > 1e-15:
                    a_new /= ss
                else:
                    a_new = parent.copy()
            elif strategy == 3:
                # Peak reduction: reduce top contributors to peak
                conv_p = fast_convolve(parent)
                k_peak = int(np.argmax(conv_p))
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                a_new = parent.copy()
                n_reduce = min(8, je - js + 1)
                for j in range(js, je + 1):
                    a_new[j] *= (1.0 - 0.1 * n_reduce / max(1, je - js + 1))
                a_new = np.maximum(a_new, 0)
                ss = a_new.sum()
                if ss > 1e-15:
                    a_new /= ss
            else:
                # Self-crossover: mix with reversed
                mix = 0.3 + 0.4 * rng.random()
                a_new = (1.0 - mix) * parent + mix * parent[::-1]
                a_new = np.maximum(a_new, 0)
                ss = a_new.sum()
                if ss > 1e-15:
                    a_new /= ss

            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()

            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                population.insert(0, best_a.copy())
                if len(population) > pop_size:
                    population.pop()
            elif len(population) < pop_size:
                population.append(a_new)

        # Ensure best is in population
        if gen % 10 == 0:
            population.insert(0, best_a.copy())
            if len(population) > pop_size:
                population.pop()

    return best_a, best_obj


def binary_polish(a0, time_limit):
    """Binary search polish on individual coordinates."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < 100:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-15:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            # Binary search
            lo = 0.0
            hi = 1.0
            x_new = x_old
            cur_obj = conv.max()

            for _ in range(35):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                a_trial = a * (om / s_others)
                a_trial[j] = mid
                conv_trial = fast_convolve(a_trial)
                obj_trial = conv_trial.max()

                if obj_trial < cur_obj - 1e-15:
                    lo = mid
                    x_new = mid
                    cur_obj = obj_trial
                else:
                    hi = mid

            if abs(x_new - x_old) > 1e-14:
                a[j] = x_new
                others_mask = np.ones(n, dtype=bool)
                others_mask[j] = False
                s_others_new = 1.0 - x_new
                if s_others_new > 1e-14:
                    a[others_mask] *= s_others_new / s_others
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a /= ss
                conv = fast_convolve(a)
                obj = conv.max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved:
            if passes > 5:
                break
            a = project_simplex(best_a + np.random.default_rng(passes * 53 + 19).standard_normal(n) * 0.001)
            conv = fast_convolve(a)

    return best_a, best_obj


def combined_v2(a0, time_limit, verbose):
    """Multi-method combined optimization."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    phase = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 8:
            break

        phase_mod = phase % 5
        t_alloc = min(remaining * 0.35, 100)

        if phase_mod == 0:
            # Subgradient
            lr = 0.005 / (1.0 + 0.15 * (phase // 5))
            a_opt, obj = subgradient_optimize(
                best_a, t_alloc, lr=lr,
                top_k=max(30, n // 6), temp=0.0003 + 0.0002 * (phase // 5),
                decay=0.9997, mom=0.9
            )
            label = f"SG{phase}"
        elif phase_mod == 1:
            # Block BCD
            a_opt, obj = block_bcd(best_a, t_alloc, block_size=8)
            label = f"BCD{phase}"
        elif phase_mod == 2:
            # Peak shaving
            a_opt, obj = peak_shaving(best_a, t_alloc)
            label = f"PEAK{phase}"
        elif phase_mod == 3:
            # Population search
            a_opt, obj = population_search(best_a, t_alloc, pop_size=12)
            label = f"POP{phase}"
        else:
            # Binary polish (short)
            a_opt, obj = binary_polish(best_a, t_alloc)
            label = f"BIN{phase}"

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt
            if verbose:
                print(f"  [{label}] obj={best_obj:.10f}")

        phase += 1

    return best_a, best_obj


def make_block_fix_restart(a, seed_val):
    """Create a restart with some blocks set to uniform."""
    n = len(a)
    a_new = a.copy()
    rng = np.random.default_rng(seed_val)
    num_blocks = min(3, n // 32)
    block_size = max(8, n // 30)
    for _ in range(num_blocks):
        start = rng.integers(0, max(1, n - block_size))
        end = min(start + block_size, n)
        a_new[start:end] = a_new[start:end].sum() / max(1, end - start)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def make_center_restart(a, factor):
    """Create a restart with center coordinates scaled."""
    n = len(a)
    a_new = a.copy()
    center = n // 2
    width = max(8, n // 10)
    for i in range(max(0, center - width), min(n, center + width)):
        a_new[i] *= factor
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new


def make_anti_peak_restart(a):

def make_reverse_mixed_restart(a, mix_ratio):

    a_new = np.maximum(a_new, 0)

        a_new /= s
    return a_new

        if rem < 50:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.3, 120)
        if t_alloc < 20:
            t_alloc = 20
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = combined_v2(f_interp, t_alloc, verbose=True)
        try_update(a_opt, f"len{target_n}")

    # === Phase 4: Diverse restarts ===
    print(f"\n=== Phase 4: Restarts (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)

        restarts_info = [
            ("noise_003", 0),
            ("noise_007", 1),
            ("block_fix", 2),
            ("center_1.05", 3),
            ("center_0.95", 4),
            ("anti_peak", 5),
            ("rev_mix_0.3", 6),
            ("rev_mix_0.5", 7),
            ("sym", 8),
        ]

        for idx, (label, code) in enumerate(restarts_info):
            rem = deadline - time.monotonic() - 300 + idx * 40
            if rem < 25:
                break
            t_alloc = min(rem, 100)

            if code == 0:
                f_init = project_simplex(best_a + np.random.default_rng(101).standard_normal(n_best) * 0.003)
            elif code == 1:
                f_init = project_simplex(best_a + np.random.default_rng(202).standard_normal(n_best) * 0.007)
            elif code == 2:
                f_init = make_block_fix_restart(best_a, 303)
            elif code == 3:
                f_init = make_center_restart(best_a, 1.05)
            elif code == 4:
                f_init = make_center_restart(best_a, 0.95)
            elif code == 5:
                f_init = make_anti_peak_restart(best_a)
            elif code == 6:
                f_init = make_reverse_mixed_restart(best_a, 0.3)
            elif code == 7:
                f_init = make_reverse_mixed_restart(best_a, 0.5)
            else:
                f_init = symmetrize(best_a)

            print(f"  {label}, t={t_alloc:.0f}s")
            a_opt, _ = combined_v2(f_init, t_alloc, verbose=True)
            try_update(a_opt, f"r_{label}")

    # === Phase 5: Intensive final polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        # Long binary polish
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 300)
            print(f"  Binary polish long, t={t_alloc:.0f}s")
            a_opt, _ = binary_polish(best_a, t_alloc)
            try_update(a_opt, "bin_long")

        # Combined on the polished result
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  Combined final, t={t_alloc:.0f}s")
            a_opt, _ = combined_v2(best_a, t_alloc, verbose=True)
            try_update(a_opt, "combined_final")

        # Short binary polish again
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_alloc = min(rem, 150)
            print(f"  Binary polish short, t={t_alloc:.0f}s")
            a_opt, _ = binary_polish(best_a, t_alloc)
            try_update(a_opt, "bin_short")

        # Tiny perturbations + quick optimize
        rem = deadline - time.monotonic() - 15
        for p in range(4):
            if rem < 10:
                break
            t_alloc = min(rem - 5, 20)
            if t_alloc < 5:
                break
            pert = project_simplex(best_a + np.random.default_rng(500 + p).standard_normal(len(best_a)) * 0.001)
            a_opt, _ = binary_polish(pert, t_alloc)
            try_update(a_opt, f"perturb_{p}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
