            grad_da[idx] += 4.0 * n * w * a[k - idx]

    # Chain rule: d obj / dz[j] = sum_i (d obj/da[j]) * da[i]/dz[j]
    # da[i]/dz[j] = a[i]*(delta_ij - a[j])
    dot = np.dot(grad_da, a)
    grad_z = grad_da * a - a * dot
    return obj, grad_z


def smooth_lbfgs(a0, time_limit, beta_start=5.0, beta_end=2000.0,
                 max_iter=500, tol=1e-12):
    """Optimize using L-BFGS with adaptive beta on the smooth approximation."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Init z from a: z = log(a) (with small epsilon for zeros)
    z = np.log(np.maximum(a, 1e-30))
    # Center z (invariance of softmax to adding constant)
    z -= z.mean()

    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    n_elec = min(20, n)
    for iteration in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        # Adaptive beta
        frac = min(1.0, iteration / (max_iter * 0.7))
        beta = beta_start * (beta_end / beta_start) ** frac

        res = minimize(
            obj_and_grad_z, z, args=(beta,),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': 100, 'gtol': tol, 'maxls': 50,
                     'maxcor': n_elec}
        )
        z = res.x
        a_new = softmax(z)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj


def coordinate_bisection(a0, time_limit, max_passes=2000):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        any_change = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(50):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-13:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-13:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                any_change = True
                continue

            any_change = True
            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved and passes > 3:
            break
        if not any_change:
            break

    return best_a, best_obj


def pair_optimize(a0, time_limit, rng):
    """Optimize pairs of coordinates jointly using small grid search."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        x_i = a[i]
        x_j = a[j]
        if x_i < 1e-10 or x_j < 1e-10:
            continue

        # Try several transfer fractions
        best_local = best_obj
        best_frac = 0.0
        for delta_frac in np.linspace(-0.5, 0.5, 11):
            delta = delta_frac * min(x_i, x_j)
            a_trial = a.copy()
            a_trial[i] = x_i - delta
            a_trial[j] = x_j + delta
            if a_trial[i] < 0 or a_trial[j] < 0:
                continue
            st = a_trial.sum()
            if st < 1e-12:
                continue
            a_trial /= st
            obj = fast_convolve(a_trial).max()
            if obj < best_local:
                best_local = obj
                best_frac = delta
                a_trial_best = a_trial.copy()

        if best_frac != 0.0:
            a = a_trial_best.copy()
            if best_local < best_obj - 1e-12:
                best_obj = best_local
                best_a = a.copy()

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using cubic interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    if old_n >= 4:
        from scipy.interpolate import CubicSpline
        cs = CubicSpline(old_x, f, bc_type='not-a-knot')
        f_interp = cs(new_x)
    else:
        f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def combined_optimize(a0, time_limit, verbose=False):
    """Multi-phase: smooth L-BFGS -> coordinate bisection -> pair optimize."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    # Phase 1: Smooth L-BFGS (adaptive beta)
    rem = time_limit - (time.monotonic() - t0) - 30
    if rem > 10:
        t1 = min(rem * 0.4, 120)
        if verbose:
            print(f"    L-BFGS phase, t={t1:.0f}s")
        a_opt, obj = smooth_lbfgs(best_a, time_limit=t1,
                                  beta_start=3.0, beta_end=500.0,
                                  max_iter=200)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt

        # Second L-BFGS pass with higher beta
        rem = time_limit - (time.monotonic() - t0) - 30
        if rem > 10:
            t1b = min(rem * 0.3, 60)
            a_opt, obj = smooth_lbfgs(best_a, time_limit=t1b,
                                      beta_start=50.0, beta_end=2000.0,
                                      max_iter=100)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"    L-BFGS2: obj={best_obj:.10f}")

    # Phase 2: Coordinate bisection
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        t2 = min(rem * 0.35, 120)
        if verbose:
            print(f"    CD phase, t={t2:.0f}s")
        a_opt, obj = coordinate_bisection(best_a, time_limit=t2, max_passes=500)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt
            if verbose:
                print(f"    CD: obj={best_obj:.10f}")

    # Phase 3: Pair optimization
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        t3 = min(rem * 0.2, 30)
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
        a_opt, obj = pair_optimize(best_a, time_limit=t3, rng=rng)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt

    # Phase 4: Final CD polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 3:
        a_opt, obj = coordinate_bisection(best_a, time_limit=rem, max_passes=200)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining smooth L-BFGS gradient descent,
    exact coordinate bisection, pair optimization, multi-restart with
    diverse initializations, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 40.0)

    best_a = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    def update_best(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        score = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_try)})")

    # === Phase 1: Optimize parent ===
    print(f"=== Phase 1: Optimize parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem, 250)
        print(f"  Combined optimize parent, t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(parent, time_limit=t_alloc, verbose=True)
        update_best(a_opt, "parent")

    # === Phase 2: Symmetric variant ===
    print(f"\n=== Phase 2: Symmetric (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            a_sym = symmetrize(best_a)
            t_alloc = min(rem, 100)
            a_opt, _ = combined_optimize(a_sym, time_limit=t_alloc)
            update_best(a_opt, "symmetric")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    target_lengths = [512, 640, 896, 1024, 1280]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 250
        if rem < 40:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.4, 120)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(f_interp, time_limit=t_alloc)
        update_best(a_opt, f"n{target_n}")
        if best_a is not None:
            # Quick polish at this length
            rem = deadline - time.monotonic() - 200
            if rem > 20:
                t_polish = min(rem * 0.3, 40)
                a_opt2, _ = coordinate_bisection(best_a, time_limit=t_polish, max_passes=100)
                update_best(a_opt2, f"n{target_n}_polish")

    # === Phase 4: Diverse restarts ===
    print(f"\n=== Phase 4: Diverse restarts (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        restart_defs = [
            ("noise3", 0.003),
            ("noise7", 0.007),
            ("noise15", 0.015),
            ("reverse_mix", None),
            ("flat_mix", None),
            ("edge_boost", None),
            ("center_dip", None),
        ]
        for idx, (label, noise_scale) in enumerate(restart_defs):
            rem = deadline - time.monotonic() - 150 + idx * 30
            if rem < 20:
                break
            if noise_scale is not None:
                f_init = project_simplex(best_a + rng.standard_normal(n_best) * noise_scale)
            elif label == "reverse_mix":
                f_init = project_simplex(0.7 * best_a + 0.3 * best_a[::-1])
            elif label == "flat_mix":
                f_init = project_simplex(0.85 * best_a + 0.15 * np.ones(n_best) / n_best)
            elif label == "edge_boost":
                f_init = best_a.copy()
                f_init[0] *= 1.5
                f_init[-1] *= 1.5
                f_init = f_init / f_init.sum()
            elif label == "center_dip":
                f_init = best_a.copy()
                mid = n_best // 2
                f_init[max(0, mid - n_best // 10):mid + n_best // 10] *= 0.5
                f_init = f_init / f_init.sum()
            else:
                f_init = best_a.copy()

            t_alloc = min(rem, 100)
            print(f"  Restart {label}, t={t_alloc:.0f}s")
            a_opt, _ = combined_optimize(f_init, time_limit=t_alloc)
            update_best(a_opt, f"r_{label}")

    # === Phase 5: Intensive final polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        # Multiple rounds of CD with tiny perturbations
        for round_i in range(5):
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break

            # Tiny random perturbation to escape potential local min
            if round_i > 0:
                n_best = len(best_a)
                scale = 0.001 * (round_i + 1)
                f_pert = project_simplex(best_a + rng.standard_normal(n_best) * scale)
            else:
                f_pert = best_a.copy()

            t_alloc = min(rem * 0.6, 60)
            a_opt, _ = coordinate_bisection(f_pert, time_limit=t_alloc, max_passes=300)
            update_best(a_opt, f"polish_{round_i}")

            # Quick L-BFGS
            rem = deadline - time.monotonic() - 15
            if rem > 8:
                t_lb = min(rem * 0.4, 20)
                a_opt2, _ = smooth_lbfgs(best_a, time_limit=t_lb,
                                         beta_start=100.0, beta_end=5000.0,
                                         max_iter=50)
                update_best(a_opt2, f"polish_lbfgs_{round_i}")

    # === Phase 6: Try completely different structure ===
    print(f"\n=== Phase 6: Alternative structures (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 30:
        # Try a "ramp" sequence
        n_try = n_parent
        a_ramp = np.linspace(0.01, 1.0, n_try)
        a_ramp = a_ramp / a_ramp.sum()
        t_alloc = min(rem * 0.3, 60)
        a_opt, _ = combined_optimize(a_ramp, time_limit=t_alloc)
        update_best(a_opt, "ramp")

        # Try "U-shaped"
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            x = np.arange(n_try) / (n_try - 1)
            a_u = (x - 0.5) ** 2 + 0.01
            a_u = a_u / a_u.sum()
            t_alloc = min(rem * 0.5, 40)
            a_opt, _ = combined_optimize(a_u, time_limit=t_alloc)
            update_best(a_opt, "u_shape")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
