def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                candidates.extend(tc.tolist())
    
    # Critical point from k = 2i
    if has_2i:
        c_2i = conv[2*i] - s * s
        # t_2i = (c_2i - s*S) / (S - s)  [derived above]
        denom_2i = S - s
        if abs(denom_2i) > 1e-15:
            t_2i = (c_2i - s * S) / denom_2i
            if t_2i > 1e-14:
                candidates.append(t_2i)
    
    # Evaluate all candidates
    best_t = s
    best_h = h(s)
    
    for tc in candidates:
        tc = max(0.0, tc)
        hc = h(tc)
        if hc < best_h:
            best_h = hc
            best_t = tc
    
    # Golden section refinement around best_t
    if best_t > 1e-10:
        # Determine a bracket
        width = max(best_t * 0.1, 1e-4)
        lo = max(0.0, best_t - width)
        hi = best_t + width
        
        # Verify bracket (h should decrease from boundaries toward center)
        h_lo = h(lo)
        h_hi = h(hi)
        
        expand = 0
        while expand < 8 and (h_lo < best_h or h_hi < best_h):
            width *= 2
            lo = max(0.0, best_t - width)
            hi = best_t + width
            h_lo = h(lo)
            h_hi = h(hi)
            expand += 1
        
        if h_lo > best_h and h_hi > best_h:
            # Standard golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(60):
                if hi - lo < 1e-13 * max(1.0, best_t):
                    break
                d1 = hi - gr * (hi - lo)
                d2 = lo + gr * (hi - lo)
                h1 = h(d1)
                h2 = h(d2)
                if h1 < h2:
                    hi = d2
                else:
                    lo = d1
            mid = (lo + hi) / 2.0
            h_mid = h(mid)
            if h_mid < best_h:
                best_h = h_mid
                best_t = mid
    
    return max(0.0, best_t)
