"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Adaptive SCA (Successive Convex Approximation) with LP oracle
- Mirror descent (multiplicative weight updates)
- Spectral analysis for initialization

import numpy as np
from scipy.optimize import linprog
from scipy.special import softmax
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute sum_j p[j+m] * a[j] for all m (Hankel matrix-vector product)."""
    return np.correlate(p, a, mode='valid')


def solve_lp_oracle(f, M, n):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g normalized to sum 1, or None if infeasible.
    """
    c = -np.ones(n)
    # Build constraint matrix: A_ub[k, j] = f[j] if 0 <= k-j < n, else 0
    # (f*g)[k] = sum_j f[j] * g[k-j] where g is indexed 0..n-1
    # So (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
    # Let i = k-j, then j = k-i, so (f*g)[k] = sum_i f[k-i] * g[i]
    # A_ub[k, i] = f[k-i] if 0 <= k-i < n, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # g[i] appears in (f*g)[k] with coefficient f[k-i]
        # Valid when 0 <= k-i < n, i.e., k-n < i <= k and 0 <= i < n
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            A_ub[k, i_start:i_end + 1] = f[k - i_end:k - i_start + 1][::-1]
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def sca_optimize(f0, time_limit, M_init=None, verbose=False):
    """
    SCA (Successive Convex Approximation) optimization.
    Iteratively solves LP subproblems with adaptive M.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max() if M_init is None else M_init
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_decrease = 0.995  # Gradually decrease M target
    m_increase = 1.05   # Increase M if infeasible

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Solve LP with current M
        g = solve_lp_oracle(f, M, n)
        
        if g is None:
            # Infeasible: increase M
            M *= m_increase
            no_improve += 1
            if no_improve > 100:
                # Reset to current max
                M = fast_convolve(f).max() * 1.01
                no_improve = 0

def mirror_descent(a0, time_limit, eta_init=0.1, verbose=False):
    """
    Multiplicative weight / mirror descent for the max-conv problem.
    Update: a_new[j] = a[j] * exp(-eta * grad[j]), then normalize.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find top peaks and compute weighted subgradient
        top_k = min(20, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        # Softmax weights over peaks
        pw = softmax(peak_vals / (0.01 * max_val + 1e-10))
        
        # Compute subgradient: grad[j] = 2 * sum_k pw[k] * a[k-j] (where valid)
        grad = np.zeros(n)
        for idx, k in enumerate(peaks):
            w = pw[idx]
            if w < 1e-10:
                continue
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            grad[js:je + 1] += w * 2.0 * a[k_int - je:k_int - js + 1][::-1]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Line search: try blending
        best_candidate = a
        best_val = max_val
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            candidate = (1 - alpha) * a + alpha * a_new
            candidate = np.maximum(candidate, 0)
            candidate = candidate / candidate.sum()
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt eta
        if no_improve > 30:
            eta *= 0.95
        if no_improve > 100:
            eta *= 0.9
        if eta < 1e-4:
            eta = eta_init * 0.1
            no_improve = 0
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            eta = eta_init
            no_improve = 0

    return best_a, best_obj


def spectral_smooth(a0, time_limit, verbose=False):
    """
    Frequency-domain approach: dampen high-power frequency components
    that contribute to time-domain peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    gamma = 0.01  # Smoothing parameter

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2

        # Identify which frequencies cause peaks in time domain
        # The time-domain signal is ifft(power), and we want to reduce its max
        # Dampen frequencies that contribute most to the peak
        peak_time = np.argmax(conv)
        # The contribution of frequency k to time t is |A[k]|^2 * exp(2*pi*i*k*t/fft_size)
        # For the peak, we want to reduce the constructive interference
        
        # Simple approach: slightly flatten the power spectrum
        # (reduce the dynamic range of |A|^2)
        log_power = np.log(power + 1e-20)
        mean_lp = log_power.mean()
        # Compress: log_power_new = mean + gamma * (log_power - mean)
        # This reduces the dynamic range
        log_power_new = mean_lp + (1 - gamma) * (log_power - mean_lp)
        power_new = np.exp(log_power_new)
        
        # Scale to maintain DC
        dc_ratio = power_new[0] / power[0] if power[0] > 0 else 1.0
        
        # Reconstruct: we need a such that |FFT(a)|^2 ≈ power_new
        # This is a phase retrieval problem - use iterative amplitude adjustment
        a_new = a.copy()
        for inner in range(20):
            A_new = np.fft.rfft(a_new, fft_size)
            # Adjust amplitude to match target power
            target_amp = np.sqrt(power_new / (power + 1e-30))
            # Blend
            A_new *= (1 - 0.5) + 0.5 * target_amp * np.ones_like(A_new)  # This isn't right
            # Actually, let's just scale
            A_new_scaled = A_new * np.sqrt(np.maximum(power_new / (power + 1e-30), 0))
            a_new = np.fft.irfft(A_new_scaled, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            if a_new.sum() < 1e-15:
                break
            a_new = a_new / a_new.sum()
        
        if a_new.sum() > 1e-10:
            a_new = a_new / a_new.sum()
            val = fast_convolve(a_new).max()
            if val < best_obj:
                best_obj = val
                best_a = a_new.copy()
        
        gamma *= 1.01  # Gradually increase smoothing
        if gamma > 0.5:
            break

    return best_a, best_obj


def subgradient_with_line_search(a0, time_limit, top_k=10, verbose=False):
    """
    Subgradient method with exact line search on the max.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.05

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        
        # Top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            # Inverse value weighting (emphasize higher peaks)
            weights = pv / pv.sum()
        
        # Subgradient
        grad = np.zeros(n)
        for idx, k in enumerate(peaks):
            w = weights[idx]
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            grad[js:je + 1] += w * 2.0 * a[k_int - je:k_int - js + 1][::-1]
        
        # Projected gradient with line search
        direction = project_simplex(a - lr * grad)
        diff = direction - a
        
        # Binary search for best alpha in [0, 1]
        lo, hi = 0.0, 1.0
        best_alpha = 0.0
        best_val = conv.max()
        
        for _ in range(15):
            mid = (lo + hi) / 2.0
            candidate = a + mid * diff
            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()
            if cs < 1e-15:
                hi = mid
                continue
            candidate = candidate / cs
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid
        
        if best_alpha > 1e-10:
            a = a + best_alpha * diff
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        lr *= 0.998
        if lr < 0.001:
            lr = 0.05
        
        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0
            lr = 0.05

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng, num=8):
    """Generate diverse initializations for the optimizer."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 5.0))
    
    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 2
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.08 * n)) ** 2)
    inits.append(g2 / g2.sum())
    
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / (n - 1))) ** 0.5 + 0.1
    inits.append(c1 / c1.sum())
    
    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Random perturbation of parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px + rng.standard_normal(n) * 0.05, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits[:num]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Adaptive SCA (Successive Convex Approximation) with LP oracle
- Mirror descent (multiplicative weight updates)
- Subgradient with exact line search
- Multi-n exploration with diverse initializations
- Peak-directed redistribution

import numpy as np
from scipy.optimize import linprog
from scipy.special import softmax
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Precompute the sparsity pattern for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, n, row_idx=None, col_idx=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g (unnormalized) or None.
    """
    if row_idx is None or col_idx is None:
        row_idx, col_idx = build_lp_matrix(n)

    c = -np.ones(n)
    # A_ub[k, i] = f[k-i] for valid (k,i) pairs
    vals = f[col_idx[::-1].copy()]
    # We need A_ub where A_ub[k, i] = f[k-i]
    # row_idx[t] = k, col_idx[t] = i, value = f[k-i]
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]

    from scipy.sparse import csr_matrix
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def mirror_descent(a0, time_limit, eta_init=0.05):
    """Multiplicative weight mirror descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Top peaks with softmax weights
        top_k = min(15, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        pw = softmax(pv / (0.005 * max_val + 1e-10))

        # Subgradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = pw[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Blend with line search
        best_candidate = a
        best_val = max_val
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            eta *= 0.95
        if eta < 1e-4:
            eta = eta_init * 0.5
            no_improve = 0

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            eta = eta_init
            no_improve = 0

    return best_a, best_obj


def subgradient_ls(a0, time_limit, top_k=10):
    """Subgradient with exact binary line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.05

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            weights = pv / pv.sum()

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = weights[idx]
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        # Direction
        direction = project_simplex(a - lr * grad) - a

        # Binary line search
        lo, hi = 0.0, 1.0
        best_alpha = 0.0
        best_val = conv.max()
        for _ in range(12):
            mid = (lo + hi) / 2.0
            cand = a + mid * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                hi = mid
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid

        if best_alpha > 1e-10:
            a = a + best_alpha * direction
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        lr *= 0.999
        if lr < 0.002:
            lr = 0.05

        if no_improve > 80:
            rng = np.random.default_rng((it * 1299709 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0
            lr = 0.05

    return best_a, best_obj


def peak_shave(a0, time_limit):
    """
    Targeted peak shaving: directly reduce the highest peak by
    redistributing mass from positions that contribute to it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        # Positions contributing to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        # Contribution of each position j to the peak: a[j] * a[k_peak - j]
        contrib = np.zeros(n)
        for j in range(js, je + 1):
            contrib[j] = a[j] * a[k_peak - j]

        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Fraction to redistribute from each position
        # Move mass proportional to contribution
        frac = 0.1  # Try to move 10% of the peak-causing mass
        # Remove mass from high-contrib positions, add to low-contrib positions
        move_amount = np.zeros(n)
        for j in range(n):
            if contrib[j] > 0:
                move_amount[j] = -frac * a[j] * (contrib[j] / total_contrib)

        # Determine where to add: positions NOT in the peak span
        outside = np.array([j for j in range(n) if j < js or j > je])
        if len(outside) == 0:
            outside = np.array([j for j in range(n) if j != js and j != je])

        if len(outside) > 0:
            total_move = -move_amount.sum()
            if total_move > 0:
                # Distribute to outside positions proportionally to their current value (or uniformly)
                ext_vals = a[outside]
                ext_sum = ext_vals.sum()
                if ext_sum > 1e-15:
                    add_amount = total_move * ext_vals / ext_sum
                else:
                    add_amount = np.full(len(outside), total_move / len(outside))
                for i, j in enumerate(outside):
                    move_amount[j] += add_amount[i]

        a_new = a + move_amount
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        new_obj = fast_convolve(a_new).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng((it * 9973 + 5) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # From parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Uniform
    inits.append(np.ones(n) / n)

    # Dirichlet varied
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Edge-heavy (small center)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.5
    inits.append(edge_w / edge_w.sum())

    # Anti-symmetric-ish (helps reduce center peak)
    t = x / (n - 1)
    asym = (1.0 + 0.5 * np.sin(2 * np.pi * t)) ** 2 + 0.1
    inits.append(asym / asym.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.08 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Random perturbation of parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px + rng.standard_normal(n) * 0.03, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

                t_alloc2 = min(remaining, 150)
                a_opt2, _ = sca_optimize(a_opt, time_limit=t_alloc2)
                try_seq(a_opt2, "deep_sca2")

    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: Mirror Descent + Peak Shave (300s) ===
    print("\n=== Phase 3: Mirror Descent + Peak Shave ===")
    t3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 250
        if remaining > 30:
            t_alloc = min(remaining, 120)
            print(f"  Mirror t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, eta_init=0.03)
            try_seq(a_opt, "mirror")

            remaining = deadline - time.monotonic() - 150
            if remaining > 20:
                t_alloc = min(remaining, 60)
                a_opt2, _ = peak_shave(a_opt, time_limit=t_alloc)
                try_seq(a_opt2, "peak_shave")

            remaining = deadline - time.monotonic() - 80
            if remaining > 20:
                t_alloc = min(remaining, 60)
                a_opt3, _ = mirror_descent(a_opt2, time_limit=t_alloc, eta_init=0.01)
                try_seq(a_opt3, "mirror2")

    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Subgradient with line search (200s) ===
    print("\n=== Phase 4: Subgradient LS ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for k in [5, 15, 30]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 20:
                break
            t_alloc = min(remaining, 60)
            print(f"  Subgrad k={k}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_ls(best_seq, time_limit=t_alloc, top_k=k)
            try_seq(a_opt, f"subgrad_k{k}")

    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Diversified restarts (150s) ===
    print("\n=== Phase 5: Diversified Restarts ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        scales = [0.003, 0.008, 0.015, 0.03, 0.06]
        for scale in scales:
            remaining = deadline - time.monotonic() - 50
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(25, remaining - 30)
            if t_alloc < 5:
                break
            print(f"  Restart scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"restart_{scale}")

    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Final polish (remaining) ===
    print("\n=== Phase 6: Final Polish ===")
    t6 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  Final SCA t={t_alloc:.0f}s")
            a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_sca")

        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            a_opt, _ = subgradient_ls(best_seq, time_limit=min(remaining, 20), top_k=5)
            try_seq(a_opt, "final_sg")

        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_opt, _ = peak_shave(best_seq, time_limit=min(remaining, 10))
            try_seq(a_opt, "final_peak")

    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
