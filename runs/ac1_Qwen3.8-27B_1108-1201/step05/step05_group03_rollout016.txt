def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

    max_iters = int(time_limit / 0.012) if time_limit > 0 else 10000

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max(1, max_iters * 0.85))
        T = T_start * (T_end / T_start) ** progress
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 - 0.4 * progress)

        conv = fast_convolve(f)
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Adam update with Nesterov
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * (sg * sg)
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)

        # Nesterov lookahead
        grad_nest = nu * (m_hat / (np.sqrt(v_hat) + eps)) + (1 - nu) * (m_prev / (np.sqrt(v) + eps))
        m_prev = m_hat
        step = lr * grad_nest

        f_new = project_simplex(f - step)

        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Adam it={it}: obj={best_obj:.10f}, T={T:.6f}, k={top_k}")
        else:
            no_improve += 1
            if no_improve > 400:
                rng = np.random.default_rng(it * 41 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                m = np.zeros(n)
                v = np.zeros(n)
                m_prev = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def pairwise_optimize(f0, time_limit, verbose=False):
    """Optimize pairs of coordinates by mass transfer with golden section."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(98765)

    offsets = [1, 2, 3, 5, 7, 11, 13, 21, 34, 55, 89, 144]
    passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        order = rng.permutation(n) if passes % 3 == 1 else np.arange(n)

        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            i = order[idx_i]
            for offset in offsets:
                j = order[idx_i + offset] if idx_i + offset < n else order[abs(idx_i - offset)]
                if i == j:
                    continue

                c = f[i] + f[j]
                if c < 1e-12:
                    continue

                best_t = f[i] / c
                best_val = fast_convolve(f).max()

                # Quick scan with more points
                candidates = np.linspace(0, 1, 11)
                if 0.05 < best_t < 0.95:
                    candidates = np.concatenate([candidates, [best_t - 0.02, best_t - 0.01, best_t + 0.01, best_t + 0.02]])
                    candidates = np.unique(np.clip(candidates, 0, 1))

                for t in candidates:
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1 - t) * c
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_val - 1e-13:
                        best_val = val
                        best_t = t

                if best_val < best_obj - 1e-12:
                    # Golden section refinement
                    lo, hi = 0.0, 1.0
                    t_mid = best_t
                    gr = (np.sqrt(5) - 1) / 2

                    def eval_t(tt):
                        ft = f.copy()
                        ft[i] = tt * c
                        ft[j] = (1 - tt) * c
                        ft = ft / ft.sum()
                        return fast_convolve(ft).max()

                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    fc1 = eval_t(c1)
                    fc2 = eval_t(c2)
                    for _ in range(6):
                        if hi - lo < 1e-5:
                            break
                        if fc1 < fc2:
                            hi = c2
                            c2 = c1
                            fc2 = fc1
                            c1 = hi - gr * (hi - lo)
                            fc1 = eval_t(c1)
                        else:
                            lo = c1
                            c1 = c2
                            fc1 = fc2
                            c2 = lo + gr * (hi - lo)
                            fc2 = eval_t(c2)

                    t_opt = (lo + hi) / 2
                    val_opt = eval_t(t_opt)
                    if val_opt < best_obj - 1e-12:
                        f[i] = t_opt * c
                        f[j] = (1 - t_opt) * c
                        f = f / f.sum()
                        best_obj = val_opt
                        best_f = f.copy()
                        pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                break

    return best_f, best_obj


def structured_perturb_search(f0, time_limit, verbose=False):
    """Try structured perturbations: sine waves, ramps, cosine windows."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    # Generate structured perturbation templates
    templates = []
    for freq in range(1, min(20, n // 4) + 1):
        templates.append(np.sin(2 * np.pi * freq * np.arange(n) / n))
        templates.append(np.cos(2 * np.pi * freq * np.arange(n) / n))
    templates.append(np.arange(n) / n - 0.5)  # ramp
    templates.append(np.cos(np.pi * np.arange(n) / n) - np.cos(np.pi))  # cosine window
    templates.append((np.arange(n) / n) ** 2 - 0.5)  # quadratic
    templates.append(np.exp(-((np.arange(n) / n - 0.5) ** 2) * 8) - 0.3)  # gaussian bump
    for freq in range(2, 8):
        templates.append(np.sin(2 * np.pi * freq * np.arange(n) / n) * np.exp(-((np.arange(n) / n - 0.5) ** 2) * 4))

    scales = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]

    for scale in scales:
        if time.monotonic() - t0 > time_limit:
            break
        for tmpl in templates:
            if time.monotonic() - t0 > time_limit:
                break
            for sign in [1, -1]:
                f_trial = project_simplex(best_f + sign * scale * tmpl)
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_f = f_trial.copy()

    # Random Gaussian perturbations
    num_trials = max(100, int((time_limit - (time.monotonic() - t0)) / 0.02))
    for _ in range(num_trials):
        if time.monotonic() - t0 > time_limit:
            break
        scale = rng.uniform(0.001, 0.05)
        f_trial = project_simplex(best_f + rng.standard_normal(n) * scale)
        f_trial = f_trial / f_trial.sum()
        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_trial.copy()

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_seq_wrapper(a, best_state, label=""):
    """Evaluate and update best state. Returns (best_f, best_score, improved)."""
    a = np.maximum(a, 0.0)
    try:
        sc = float(evaluate_sequence(a.tolist()))
    except Exception:
        return best_state[0], best_state[1], False
    if np.isfinite(sc) and sc < best_state[1]:
        print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
        return a.copy(), sc, True
    return best_state[0], best_state[1], False


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: spectral shaping, LP-FW, Adam, pairwise,
    structured search, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_f = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_f, best_score
        a = np.maximum(a, 0.0)
        try:
            sc = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_f = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_f = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.10f}")

    n_parent = len(parent)

    # === Phase 1: Spectral shaping on parent ===
    print(f"\n=== Phase 1: Spectral Shaping (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 800
    t1 = min(rem, 250)
    if t1 > 30:
        print(f"  Peak-suppression: n={n_parent}, t={t1:.0f}s")
        a_opt, _ = spectral_shape(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "spec_shape")

        rem = deadline - time.monotonic() - 750
        t1b = min(rem, 150)
        if t1b > 20:
            print(f"  Blend: n={n_parent}, t={t1b:.0f}s")
            a_opt, _ = spectral_blend(parent, time_limit=t1b, verbose=True)
            try_seq(a_opt, "spec_blend")

    # === Phase 2: LP-FW on best ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 600
    t2 = min(rem, 250)
    if t2 > 20:
        print(f"  LP-FW: n={len(best_f)}, t={t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_f, time_limit=t2, verbose=True)
        try_seq(a_opt, "lpfw_p2")

    # === Phase 3: Adam gradient ===
    print(f"\n=== Phase 3: Adam (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 400
    t3 = min(rem, 200)
    if t3 > 20:
        print(f"  Adam: n={len(best_f)}, t={t3:.0f}s")
        a_opt, _ = adam_optimize(best_f, time_limit=t3, verbose=True)
        try_seq(a_opt, "adam_p3")

    # === Phase 4: Second LP-FW pass ===
    print(f"\n=== Phase 4: LP-FW v2 (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 300
    t4 = min(rem, 180)
    if t4 > 15:
        print(f"  LP-FW v2: n={len(best_f)}, t={t4:.0f}s")
        a_opt, _ = lp_fw_optimize(best_f, time_limit=t4, verbose=True)
        try_seq(a_opt, "lpfw_p4")

    # === Phase 5: Spectral on current best ===
    print(f"\n=== Phase 5: Spectral v2 (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 200
    t5 = min(rem, 150)
    if t5 > 15:
        print(f"  Spectral v2: n={len(best_f)}, t={t5:.0f}s")
        a_opt, _ = spectral_shape(best_f, time_limit=t5, verbose=True)
        try_seq(a_opt, "spec_v2")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.10f}) ===")
    for target_n in [512, 640, 896, 1024]:
        rem = deadline - time.monotonic() - 150
        if rem < 40:
            break
        if target_n == len(best_f):
            continue
        f_interp = interpolate_sequence(best_f, target_n)

        t_alloc = min(rem * 0.3, 100)
        if t_alloc > 15:
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"lpfw_n{target_n}")

        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_a = min(rem * 0.4, 80)
            if t_a > 10:
                a_opt, _ = adam_optimize(a_opt, time_limit=t_a, verbose=False)
                try_seq(a_opt, f"adam_n{target_n}")

    # === Phase 7: Structured perturbation search ===
    print(f"\n=== Phase 7: Structured Search (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 100
    t7 = min(rem, 100)
    if t7 > 10:
        a_opt, _ = structured_perturb_search(best_f, time_limit=t7, verbose=False)
        try_seq(a_opt, "struct")

    # === Phase 8: Pairwise optimization ===
    print(f"\n=== Phase 8: Pairwise (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 60
    t8 = min(rem, 150)
    if t8 > 15:
        print(f"  Pairwise: n={len(best_f)}, t={t8:.0f}s")
        a_opt, _ = pairwise_optimize(best_f, time_limit=t8, verbose=True)
        try_seq(a_opt, "pairwise")

    # === Phase 9: Final Adam polish ===
    print(f"\n=== Phase 9: Final Adam (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 30
    t9 = min(rem, 80)
    if t9 > 10:
        a_opt, _ = adam_optimize(best_f, time_limit=t9, verbose=False)
        try_seq(a_opt, "adam_final")

    # === Phase 10: Diverse restarts + quick LP-FW ===
    print(f"\n=== Phase 10: Diverse (best={best_score:.10f}) ===")
    n_best = len(best_f)
    for p in range(6):
        rem = deadline - time.monotonic() - 15
        if rem < 10:
            break
        t_alloc = min(rem / max(1, 6 - p), 60)

        if p == 0:
            f_init = best_f.copy()
        elif p == 1:
            rng = np.random.default_rng(200 + p)
            f_init = project_simplex(best_f + rng.standard_normal(n_best) * 0.005)
        elif p == 2:
            f_init = best_f[::-1].copy() / best_f.sum()
        elif p == 3:
            f_init = project_simplex(0.6 * best_f + 0.4 * (np.ones(n_best) / n_best))
        elif p == 4:
            rng = np.random.default_rng(500 + p)
            f_init = project_simplex(best_f + rng.standard_normal(n_best) * 0.01)
        else:
            # Scale asymmetry
            f_init = best_f.copy()
            f_init[:n_best // 2] *= 1.02
            f_init[n_best // 2:] *= 0.98
            f_init = project_simplex(f_init)

        if p % 2 == 0:
            a_opt, _ = lp_fw_optimize(f_init, time_limit=min(t_alloc, 50), verbose=False)
        else:
            a_opt, _ = adam_optimize(f_init, time_limit=min(t_alloc, 40), verbose=False)
        try_seq(a_opt, f"diverse_{p}")

    # === Phase 11: Final pairwise + spectral ===
    print(f"\n=== Phase 11: Final Polish (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        a_opt, _ = pairwise_optimize(best_f, time_limit=min(rem, 40), verbose=False)
        try_seq(a_opt, "pw_final")

    rem = deadline - time.monotonic() - 3
    if rem > 8:
        a_opt, _ = spectral_shape(best_f, time_limit=min(rem, 25), verbose=False)
        try_seq(a_opt, "spec_final")

    # Final verification
    if best_f is None:
        best_f = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_f.tolist()))

    fs = float(evaluate_sequence(best_f.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_f)})")
    return [float(x) for x in best_f]
