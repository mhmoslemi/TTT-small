import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u - (cssv - target_sum) / np.arange(1, n+1)) > 0
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_fft(f, p):
    """Compute gradient of smooth max: 2 * cross_correlation(p, f)."""
    n = len(f)
    q = p  # length 2n-1
    q_rev = q[::-1]
    conv_size = (2*n-1) + n - 1  # = 3n-2
    S = 1 << (conv_size - 1).bit_length()

    F_f = np.fft.fft(f, S)
    conv_result = np.fft.ifft(F_qrev * F_f).real[:conv_size]
    grad = 2.0 * conv_result[n-1:2*n-1][::-1]
    return grad

    v = np.zeros(n)  # momentum
    it = 0
    no_improve = 0

import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_max_gradient(f, p):
    """
    Compute gradient of smooth max of conv(f,f).
    grad[m] = 2 * sum_j p[j+m] * f[j]
    Computed via FFT for O(n log n) complexity.
    """
    n = len(f)
    q_rev = p[::-1]  # reverse the softmax weights (length 2n-1)
    conv_size = (2 * n - 1) + n - 1  # 3n - 2
    S = 1 << (conv_size - 1).bit_length()
    F_qrev = np.fft.fft(q_rev, S)
    F_f = np.fft.fft(f, S)
    conv_result = np.fft.ifft(F_qrev * F_f).real[:conv_size]
    grad = 2.0 * conv_result[n-1:2*n-1][::-1]
    return grad

def smooth_max_optimize(f0, time_limit, T_start=20.0, T_end=200.0, 
                        lr_start=0.05, momentum=0.9, verbose=False):
    """
    Optimize using smooth-max gradient with adaptive temperature and momentum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum velocity
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        remaining_frac = 1.0 - min(1.0, elapsed / time_limit)
        
        # Temperature schedule: increase over time
        T = T_start + (T_end - T_start) * (1.0 - remaining_frac)
        T = max(T, T_start)
        
        # Learning rate: decrease over time
        lr = lr_start * (0.5 ** (it / 200.0))
        lr = max(lr, lr_start * 0.01)
        
        # Compute gradient
        conv = fast_convolve(f)
        exp_Tc = np.exp(T * conv - T * conv.max())
        p = exp_Tc / exp_Tc.sum()
        grad = compute_smooth_max_gradient(f, p)
        
        # Gradient step with momentum
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        # Accept or reject
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    iter {it}: obj={best_obj:.10f}, T={T:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        f = f_new
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(42 + it)
            scale = 0.001 + 0.002 * min(no_improve // 200, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            no_improve = 0
    
    return best_f, best_obj

def spectral_peak_reduction(f0, time_limit, n_iter_max=500, verbose=False):
    """
    Iteratively reduce spectral peaks by adjusting the power spectrum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1
        
        # Compute current spectrum and autocorrelation
        N = 1 << (2 * n - 2).bit_length()
        F = np.fft.fft(f, N)
        power = np.abs(F) ** 2
        conv = np.fft.ifft(power).real[:2*n-1]
        
        # Find top peaks
        n_peaks = min(10, 2*n-1)
        peak_indices = np.argsort(conv)[-n_peaks:]
        
        # For each peak, compute the sensitivity to each frequency bin
        # and reduce the power at bins that contribute most
        reduction = np.zeros(N)
        for k in peak_indices:
            k = int(k)
            # conv[k] = (1/N) * sum_m power[m] * exp(2πi*k*m/N)
            # Sensitivity: (1/N) * exp(2πi*k*m/N)
            # Real part: (1/N) * cos(2π*k*m/N)
            sensitivity = np.cos(2.0 * np.pi * k * np.arange(N) / N)
            # Only reduce where sensitivity is positive (contributing to peak)
            reduction += np.maximum(sensitivity, 0) * (conv[k] - conv.mean())
        
        # Apply reduction (soft)
        reduction = np.maximum(reduction, 0)
        if reduction.max() > 0:
            reduction = reduction / reduction.max()
            # Reduce power proportionally
            factor = 1.0 - 0.1 * reduction
            factor = np.maximum(factor, 0.5)
            power_new = power * factor
        else:
            power_new = power
        
        # Recover time domain
        F_new = np.sqrt(power_new) * np.exp(1j * np.angle(F))
        f_new = np.fft.ifft(F_new).real[:n]
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    spectral iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 100:
                break
    
    return best_f, best_obj

def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def raised_cosine_init(n):
    """Raised cosine window initialization."""
    j = np.arange(n) / (n - 1)
    a = 0.5 * (1 + np.cos(np.pi * (2 * j - 1)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
