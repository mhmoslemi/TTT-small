        obj_smooth = T * np.log(np.exp((c - c.max()) / T).sum()) + c.max()

        # Backtracking line search
        lr = 1.0 / (np.linalg.norm(grad, np.inf) + 1e-12)
        lr = min(lr, 0.5)
        found = False
        for trial in range(10):
            candidate = project_simplex(a - lr * grad)
            c_new = fast_convolve(candidate)
            obj_new = c_new.max()
            if obj_new < best_obj - 1e-15:
                a = candidate
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
                found = True
                break
            lr *= 0.5

        if not found:
            no_improve += 1

        # Adapt temperature
        if no_improve > 10:
            T *= 0.92
        elif no_improve > 40:
            T *= 0.85
        elif no_improve > 100:
            T *= 1.05  # Slightly warm up to escape
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 17) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    SGD it={it}, obj={best_obj:.8f}, T={T:.6f}, no_improve={no_improve}")

    return best_a, best_obj


def adam_multi_peak(a0, time_limit, top_k=20, lr_init=0.002, decay=0.999,
                    beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """
    Multi-peak subgradient with Adam optimizer.
    Uses top-K peaks with inverse-rank squared weights.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_convolve(a)

        # Top-k peaks
        if top_k >= len(c):
            peaks = np.arange(len(c))
        else:
            peaks = np.argpartition(c, -top_k)[-top_k:]

        pv = c[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = m_hat / (np.sqrt(v_hat) + eps)

        candidate = project_simplex(a - lr * step)
        obj = fast_convolve(candidate).max()

        if obj < best_obj - 1e-15:
            a = candidate
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Decay learning rate
        lr *= decay
        if no_improve > 50:
            lr = max(lr, lr_init * 0.3)
        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 3) % (2**31))
            noise_scale = 0.002 + 0.003 * min(1.0, (no_improve - 150) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = lr_init * 0.8
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    Adam it={it}, obj={best_obj:.8f}, lr={lr:.6f}")

    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=False):
    """
    Coordinate descent: for the top peaks, try to reduce them by
    shifting mass from contributing variables to non-contributing ones.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_convolve(a)
        # Top 3 peaks
        top_peaks = np.argsort(c)[::-1][:3]

        improved = False
        for k_peak in top_peaks:
            k_peak = int(k_peak)
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            affected = set(range(js, je + 1))

            # Find best single-variable shift
            best_gain = 1e-15
            best_src = -1
            best_dst = -1
            best_amt = 0.0

            # Try moving mass from affected to non-affected
            for _ in range(30):
                src = np.random.choice(list(affected))
                if a[src] < 1e-8:
                    continue
                dst_candidates = [j for j in range(n) if j not in affected and a[j] < 0.5]
                if not dst_candidates:
                    continue
                dst = np.random.choice(dst_candidates)
                for amt_frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                    amt = a[src] * amt_frac
                    if amt < 1e-10:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    c_trial = fast_convolve(a_trial)
                    gain = c_trial.max() - c[k_peak]
                    if -gain > best_gain:
                        # Check overall max doesn't increase too much
                        if c_trial.max() < best_obj + 0.01 * best_obj:
                            best_gain = -gain
                            best_src = src
                            best_dst = dst
                            best_amt = amt

            if best_src >= 0 and best_gain > 0:
                a[best_src] -= best_amt
                a[best_dst] += best_amt
                a = np.maximum(a, 0)
                a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 100:
            # Try broader random perturbations
            rng = np.random.default_rng((it * 31415 + 7) % (2**31))
            for _ in range(5):
                pert = rng.standard_normal(n) * 0.001
                candidate = project_simplex(a + pert)
                obj = fast_convolve(candidate).max()
                if obj < best_obj - 1e-15:
                    a = candidate
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
            if not improved and it > 300:
                break

    return best_a, best_obj


def mirror_refine(a0, time_limit):
    """
    Try mirroring and averaging: if a is good, (a + reverse(a))/2
    might be better due to symmetry.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    # Try mirror average
    a_mirror = (a + a[::-1]) / 2.0
    a_mirror = np.maximum(a_mirror, 0)
    a_mirror = a_mirror / a_mirror.sum()
    obj_m = fast_convolve(a_mirror).max()
    if obj_m < best_obj:
        best_obj = obj_m
        best_a = a_mirror.copy()

    # Try shifted versions
    for shift in [1, 2, 3]:
        a_shift = np.roll(a, shift)
        # Wrap-around issue: the convolution changes with shift
        # Actually np.roll changes the sequence, so conv changes
        obj_s = fast_convolve(a_shift).max()
        if obj_s < best_obj:
            best_obj = obj_s
            best_a = a_shift.copy()

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate many diverse initializations."""
    inits = []

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Parent interp
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(('parent_interp', px / ps))
        inits.append(('parent_direct', parent.copy()) if len(parent) == n else ('parent_interp2', px / ps))

    # Cosine variations
    for alpha in [0.3, 0.5, 0.7, 0.9]:
        for power in [1, 2, 3]:
            inits.append((f'cos_a{alpha}_p{power}', make_cosine_init(n, alpha, power)))

    # Gaussian
    for sigma in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        inits.append((f'gauss_{sigma}', make_gaussian_init(n, sigma)))

    # Bimodal
    for (p1, p2) in [(0.2, 0.8), (0.25, 0.75), (0.3, 0.7), (0.1, 0.9)]:
        for sigma in [0.05, 0.1, 0.15]:
            inits.append((f'bimod_{p1}_{p2}_{sigma}', make_bimodal_init(n, p1, p2, sigma)))

    # Edge concentration
    for power in [1, 2, 3, 5]:
        inits.append((f'edge_p{power}', make_edge_init(n, power)))

    # W-shape
    for depth in [0.3, 0.5, 0.7, 0.9]:
        inits.append((f'w_{depth}', make_w_init(n, depth)))

    # Periodic
    for period in [2, 3, 4, 5, 8]:
        for duty in [0.3, 0.4, 0.5]:
            inits.append((f'per_{period}_{duty}', make_periodic_init(n, period, duty)))

    # Random Dirichlet
    for concentration in [0.5, 1.0, 2.0, 5.0]:
        inits.append((f'dirichlet_{concentration}', rng.dirichlet(np.ones(n) * concentration)))

    # Random smoothed
    for scale in [0.01, 0.05, 0.1]:
        raw = rng.standard_normal(n)
        # Smooth it
        kernel = np.ones(5) / 5
        raw_s = np.convolve(raw, kernel, mode='same')
        raw_s = np.maximum(raw_s, 0)
        rs = raw_s.sum()
        if rs > 1e-10:
            inits.append((f'rand_smooth_{scale}', raw_s / rs))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search:
    Phase 0: Diverse initialization and quick screening
    Phase 1: Smoothed gradient on top candidates
    Phase 2: Adam multi-peak refinement
    Phase 3: Coordinate polish + mirror refine
    Phase 4: Final micro-optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 0: Diverse initialization and screening ===
    print("\n=== Phase 0: Diverse Initialization ===")
    t0 = time.monotonic()

    # Determine candidate n values
    n_candidates = [n_parent]
    for n_try in [n_parent // 2, n_parent * 2, 256, 384, 512, 640, 1024]:
        if n_try >= 64 and n_try <= 2048 and n_try not in n_candidates:
            n_candidates.append(n_try)

    # For each n, generate and screen initializations
    screened = []  # List of (score, a, label, n)

    for n in n_candidates:
        if time.monotonic() > deadline - 1200:
            break

        inits = generate_diverse_inits(n, parent, rng)
        print(f"  n={n}: testing {len(inits)} initializations...")

        for label, a_init in inits:
            if time.monotonic() > deadline - 1200:
                break
            a_init = np.maximum(a_init, 0)
            s = a_init.sum()
            if s < 1e-15:
                continue
            a_init = a_init / s
            c = fast_convolve(a_init)
            obj = c.max()
            # Score estimate (without calling evaluate_sequence for speed)
            score_est = 2.0 * n * obj
            screened.append((score_est, a_init, label, n))

        print(f"  n={n}: done screening (elapsed {time.monotonic()-t0:.1f}s)")

    # Sort by estimated score, keep top candidates
    screened.sort(key=lambda x: x[0])
    print(f"  Total screened: {len(screened)}, top score est: {screened[0][0]:.6f}")

    # Pick top diverse candidates (different n values and structures)
    top_candidates = []
    seen_ns = set()
    for score_est, a, label, n in screened:
        if len(top_candidates) >= 8:
            break
        # Prefer diversity
        if n in seen_ns and len(top_candidates) >= 4:
            continue
        seen_ns.add(n)
        top_candidates.append((score_est, a, label, n))

    # Always include parent
    if parent is not None:
        top_candidates.append((2.0 * n_parent * fast_convolve(parent).max(),
                               parent.copy(), "parent", n_parent))

    # === Phase 1: Smoothed gradient on top candidates ===
    print(f"\n=== Phase 1: Smoothed Gradient (top {len(top_candidates)} candidates) ===")
    t_phase = time.monotonic()

    results = []
    for ci, (score_est, a_init, label, n) in enumerate(top_candidates):
        remaining = deadline - time.monotonic() - 800
        if remaining < 30:
            break
        t_alloc = min(remaining / max(1, len(top_candidates) - ci), 120)
        print(f"  [{ci+1}/{len(top_candidates)}] {label} (n={n}), t={t_alloc:.0f}s")

        a_opt, obj = smooth_gradient_descent(a_init, time_limit=t_alloc, T_init=0.001)
        results.append((obj, a_opt, f"sgd_{label}", n))
        try_seq(a_opt, f"sgd_{label}")

    # Sort results
    results.sort(key=lambda x: x[0])
    if results:
        print(f"  Phase 1 best: est_obj={results[0][0]:.8f} ({results[0][2]})")

    # === Phase 2: Adam multi-peak refinement ===
    print(f"\n=== Phase 2: Adam Multi-Peak ===")
    t_phase = time.monotonic()

    # Refine top 3 from Phase 1
    for ri in range(min(3, len(results))):
        remaining = deadline - time.monotonic() - 400
        if remaining < 20:
            break
        _, a_cur, label, n = results[ri]
        t_alloc = min(remaining / max(1, 3 - ri), 150)
        print(f"  Adam on {label} (n={n}), t={t_alloc:.0f}s")

        for top_k in [10, 30]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 15:
                break
            a_opt, obj = adam_multi_peak(a_cur, time_limit=min(remaining, 80),
                                         top_k=top_k, lr_init=0.002)
            try_seq(a_opt, f"adam_{label}_k{top_k}")
            if obj < results[0][0]:
                results[0] = (obj, a_opt, f"adam_{label}_k{top_k}", n)

    # === Phase 3: Coordinate polish + mirror ===
    print(f"\n=== Phase 3: Coordinate Polish & Mirror ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Mirror refine
        remaining = deadline - time.monotonic() - 200
        if remaining > 10:
            a_mirror, obj_m = mirror_refine(best_seq, time_limit=min(remaining, 30))
            try_seq(a_mirror, "mirror")
            best_for_polish = a_mirror if obj_m <= fast_convolve(best_seq).max() else best_seq
        else:
            best_for_polish = best_seq

        # Coordinate polish
        remaining = deadline - time.monotonic() - 150
        if remaining > 15:
            t_alloc = min(remaining, 100)
            print(f"  Coord polish (n={n_best}), t={t_alloc:.0f}s")
            a_pol, obj_p = coordinate_polish(best_for_polish, time_limit=t_alloc)
            try_seq(a_pol, "coord_polish")

        # Another Adam pass on the polished result
        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 60)
            a_opt, obj = adam_multi_peak(best_for_polish, time_limit=t_alloc,
                                         top_k=15, lr_init=0.001, decay=0.9995)
            try_seq(a_opt, "adam_polished")

    # === Phase 4: Final micro-optimization ===
    print(f"\n=== Phase 4: Final Micro-Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Tiny smoothed gradient with very low T
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            t_alloc = min(remaining, 40)
            a_fine, _ = smooth_gradient_descent(best_seq, time_limit=t_alloc, T_init=0.0003)
            try_seq(a_fine, "fine_smooth")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 20
        if remaining > 5:
            for _ in range(5):
                remaining = deadline - time.monotonic() - 10
                if remaining < 3:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                # Quick polish
                a_q, _ = smooth_gradient_descent(f_pert, time_limit=min(remaining, 5), T_init=0.0005)
                try_seq(a_q, "micro_perturb")

        # One more coordinate polish
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            a_cp, _ = coordinate_polish(best_seq, time_limit=min(remaining, 10))
            try_seq(a_cp, "final_coord")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - t0:.1f}s")
    return [float(x) for x in best_seq]
