```python
def project_simplex(v):
    n = len(v)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based conditional gradient descent,
fast subgradient-bisection coordinate optimization, structured initializations,
and multi-length exploration. Key innovation: solving a linear program at each step
to find the best feasible descent direction (Frank-Wolfe style), combined with
exact 1D coordinate optimization via convex subgradient bisection.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_norm(a):
    """Objective assuming sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def build_lp_matrix(f, n):
    """Build constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = f[k - j_start:k - j_end + 1]
    return A


def solve_lp_direction(f):
    """Solve LP: max sum(g) s.t. g>=0, f*g <= M elementwise. Return g."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()

    A = build_lp_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': 10, 'presolve': True})
        if res.success:
            return -res.x
    except Exception:
        pass
    return f.copy()


def line_search_1d(f, g, n_evals_max=80):
    """Golden section search for best t in [0,1] for (1-t)f + t*g."""
    n = len(f)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)

    def eval_t(t):
        x = (1.0 - t) * f + t * g
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-15:
            return 1e10
        x /= s
        return 2.0 * n * fast_convolve(x).max()

    fc = eval_t(c_)
    fd = eval_t(d_)
    nevals = 2

    for _ in range(50):
        if nevals > n_evals_max:
            break
        if fc < fd:
            b = d_
            d_ = c_
            fd = fc
            c_ = b - gr * (b - a)
            fc = eval_t(c_)
            nevals += 1
        else:
            a = c_
            c_ = d_
            fc = fd
            d_ = a + gr * (b - a)
            fd = eval_t(d_)
            nevals += 1

    t_best = (a + b) / 2.0
    x_best = (1.0 - t_best) * f + t_best * g
    x_best = np.maximum(x_best, 0)
    s = x_best.sum()
    if s > 1e-15:
        x_best /= s
    return t_best, x_best, 2.0 * n * fast_convolve(x_best).max()


def optimize_coord_subgrad(a, j, n_iters=55):
    """Optimize coordinate j via subgradient bisection. Returns optimal x_j."""
    n = len(a)
    x_old = a[j]
    denom = 1.0 - x_old
    if denom < 1e-14:
        return x_old

    b = a.copy()
    b[j] = 0.0
    s_b = b.sum()
    if s_b < 1e-14:
        return x_old
    b = b / s_b

    B = fast_convolve(b)
    C = np.zeros(2 * n - 1)
    j_start = max(0, j)
    j_end = min(2 * n - 1, j + n)
    C[j_start:j_end] = b[:j_end - j_start]

    D = np.zeros(2 * n - 1)
    if 2 * j < 2 * n - 1:
        D[2 * j] = 1.0

    lo, hi = 0.0, 1.0
    for _ in range(n_iters):
        mid = (lo + hi) / 2.0
        s = (1.0 - mid) / denom
        vals = s * s * B + 2.0 * s * mid * C + mid * mid * D
        k_max = int(np.argmax(vals))
        sg = (-2.0 * s * B[k_max] / denom
              + 2.0 * C[k_max] * (s - mid / denom)
              + 2.0 * mid * D[k_max])
        if sg < -1e-15:
            lo = mid
        elif sg > 1e-15:
            hi = mid
        else:
            break
    return (lo + hi) / 2.0


def coordinate_descent(a0, time_limit, verbose=False):
    """Full coordinate descent with subgradient bisection."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a /= s
    n = len(a)

    best_obj = obj_norm(a)
    best_a = a.copy()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-15:
                continue

            x_new = optimize_coord_subgrad(a, j)
            if abs(x_new - a[j]) < 1e-14:
                continue

            a[j] = x_new
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
                continue
            a /= s

            obj = obj_norm(a)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                improved = True

        if not improved:
            if pass_count > 1:
                break
            else:
                break

    return best_a, best_obj


def lp_descent(f0, time_limit, verbose=False, max_stall=8):
    """LP-based conditional gradient descent."""
    f = f0.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    n = len(f)

    best_obj = obj_norm(f)
    best_f = f.copy()
    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f)
        gs = g.sum()
        if gs < 1e-15:
            g = f.copy()
        else:
            g /= gs

        t_best, x_best, val_best = line_search_1d(f, g)

        if val_best < best_obj - 1e-13:
            best_obj = val_best
            best_f = x_best.copy()
            f = x_best.copy()
            stall = 0
            if verbose and it % 5 == 0:
                print(f"    LP iter {it}: obj={best_obj:.10f}, t={t_best:.4f}")
        else:
            stall += 1
            if stall >= max_stall:
                break
            # Try partial step
            t_p = 0.3 * t_best
            x_p = (1.0 - t_p) * f + t_p * g
            x_p = np.maximum(x_p, 0)
            sp = x_p.sum()
            if sp > 1e-15:
                x_p /= sp
                val_p = obj_norm(x_p)
                if val_p < best_obj - 1e-13:
                    best_obj = val_p
                    best_f = x_p.copy()
                    f = x_p.copy()
                    stall = 0

    return best_f, best_obj


def interp_to_length(f, new_n):
    """Linear interpolation to new length, renormalize."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new


def fejer_init(n):
    """Fejér-kernel-like initialization."""
    k = np.arange(n)
    center = (n - 1) / 2.0
    a = np.maximum(0, 1.0 - np.abs(k - center) / (center + 0.5))
    a *= (1.0 - (np.cos(2 * np.pi * k / n) + 1) / 4.0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def gaussian_init(n):
    """Gaussian-shaped initialization."""
    k = np.arange(n)
    center = (n - 1) / 2.0
    sigma = n / 5.0
    a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    a /= a.sum()
    return a


def hamming_init(n):
    """Hamming window initialization."""
    k = np.arange(n)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * k / (n - 1))
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def get_parent():
    """Get parent sequence from global."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p /= p.sum()
            return p
    except Exception:
        pass
    return None


def symmetrize(a):
    """Symmetrize and renormalize."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP-based conditional gradient descent,
    exact coordinate subgradient bisection, structured initializations,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(60.0, float(budget_s) - 30.0)

    best_a = None
    best_score = float('inf')

    parent = get_parent()
    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.8f}")
        if np.isfinite(p_score) and p_score < best_score:
            best_score = p_score
            best_a = parent.copy()
    else:
        parent = np.ones(768) / 768
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent. Uniform: score={p_score:.8f}")
        best_score = p_score
        best_a = parent.copy()

    def update(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 0.01:
            return
        a_norm = a_try / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_try)})")

    # === Phase 1: LP descent from parent ===
    print(f"\n=== Phase 1: LP descent from parent (n={len(parent)}) ===")
    rem = deadline - time.monotonic() - 700
    if rem > 60:
        t_alloc = min(rem, 400)
        print(f"  t={t_alloc:.0f}s")
        a_opt, _ = lp_descent(parent, time_limit=t_alloc, verbose=True)
        update(a_opt, "LP_parent")

        # Quick CD polish
        rem = deadline - time.monotonic() - 650
        if rem > 30:
            t_alloc = min(rem, 100)
            a_opt2, _ = coordinate_descent(a_opt, time_limit=t_alloc)
            update(a_opt2, "LP+CD_parent")

    # === Phase 2: Structured initializations ===
    print(f"\n=== Phase 2: Structured inits (best={best_score:.8f}) ===")
    n_base = len(parent)
    inits = [
        ("fejer", fejer_init(n_base)),
        ("gauss", gaussian_init(n_base)),
        ("hamming", hamming_init(n_base)),
        ("tri", _triangle_init(n_base)),
    ]

    for label, init_f in inits:
        rem = deadline - time.monotonic() - 500
        if rem < 40:
            break
        t_alloc = min(rem * 0.4, 120)
        print(f"  {label}: t={t_alloc:.0f}s")
        a_opt, _ = lp_descent(init_f, time_limit=t_alloc)
        update(a_opt, f"LP_{label}")

    # === Phase 3: Symmetric variant ===
    print(f"\n=== Phase 3: Symmetric (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            a_sym = symmetrize(best_a)
            t_alloc = min(rem, 150)
            print(f"  Symmetric: t={t_alloc:.0f}s")
            a_opt, _ = lp_descent(a_sym, time_limit=t_alloc)
            update(a_opt, "LP_sym")

            rem = deadline - time.monotonic() - 380
            if rem > 30:
                a_opt2, _ = coordinate_descent(a_opt, time_limit=min(rem, 80))
                update(a_opt2, "LP_sym_CD")

    # === Phase 4: Multi-length ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_a
    for target_n in [512, 1024, 1536]:
        rem = deadline - time.monotonic() - 300
        if rem < 50:
            break
        f_interp = interp_to_length(base, target_n)
        t_alloc = min(rem * 0.3, 150)
        print(f"  n={target_n}: t={t_alloc:.0f}s")
        a_opt, _ = lp_descent(f_interp, time_limit=t_alloc)
        update(a_opt, f"LP_n{target_n}")

        rem = deadline - time.monotonic() - 280
        if rem > 20:
            a_opt2, _ = coordinate_descent(a_opt, time_limit=min(rem, 60))
            update(a_opt2, f"LP_n{target_n}_CD")

    # === Phase 5: Random restarts + LP ===
    print(f"\n=== Phase 5: Restarts (best={best_score:.8f}) ===")
    n_best = len(best_a)
    for p in range(4):
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 100)
        noise_scale = 0.005 * (1.0 + 0.5 * p)
        rng_p = np.random.default_rng(seed + 500 + p * 7)
        f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * noise_scale)
        print(f"  Restart {p} (noise={noise_scale:.4f}): t={t_alloc:.0f}s")
        a_opt, _ = lp_descent(f_pert, time_limit=t_alloc)
        update(a_opt, f"restart_{p}")

    # === Phase 6: Final intensive CD on best ===
    print(f"\n=== Phase 6: Final CD (best={best_score:.8f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_alloc = min(rem, 200)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_a, time_limit=t_alloc)
            update(a_opt, "final_CD")

        # Tiny perturbations + CD
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            n_best = len(best_a)
            for p in range(3):
                rem = deadline - time.monotonic() - 15
                if rem < 10:
                    break
                rng_p = np.random.default_rng(seed + 2000 + p * 13)
                f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.001)
                t_alloc = min(rem, 20)
                a_opt, _ = coordinate_descent(f_pert, time_limit=t_alloc)
                update(a_opt, f"perturb_{p}")

        # One more LP round on the polished result
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 40)
            print(f"  Final LP polish: t={t_alloc:.0f}s")
            a_opt, _ = lp_descent(best_a, time_limit=t_alloc)
            update(a_opt, "final_LP")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]


def _triangle_init(n):
    """Triangular (Bartlett) window initialization."""
    k = np.arange(n)
    center = (n - 1) / 2.0
    a = np.maximum(0, 1.0 - np.abs(k - center) / (center + 0.5))
    a /= a.sum()
    return a
```
