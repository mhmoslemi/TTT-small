"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed subgradient optimization
with peak balancing, coordinate descent with golden-section line search, and
multi-length exploration. Key innovation: explicit peak equalization step that
solves a least-squares subproblem to simultaneously reduce the top convolution
peaks, combined with adaptive annealing and diversified restarts.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_subgradient(a, conv, top_k, temperature):
    """Compute smoothed subgradient of max(conv) using softmax weighting over top peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg, peaks


def peak_balance_direction(a, top_peaks, n):
    """Compute direction that simultaneously reduces all top peaks."""
    d = np.zeros(n)
    m = len(top_peaks)
    if m == 0:
        return d
    for k in top_peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            d[i] += a[k - i]
    d /= m
    d -= d.mean()
    dn = np.linalg.norm(d)
    if dn > 1e-15:
        d /= dn
    return d


def subgradient_optimize(f0, time_limit, top_k=50, temperature=1e-5, lr=0.004,
                         momentum=0.87, balance_interval=200, verbose=False, label=""):
    """
    Smoothed subgradient optimization with Nesterov momentum and peak balancing.
    Uses exponential (multiplicative) weights update for automatic non-negativity.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(0.5 * momentum * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        sg, peaks = smoothed_subgradient(f_nest, conv_nest, cur_top_k, cur_temp)

        # Update momentum
        v = momentum * v + (1.0 - momentum) * sg

        # Peak balancing every balance_interval steps
        if balance_interval > 0 and it % balance_interval == 0:
            conv_cur = fast_convolve(f)
            cur_peaks = np.argpartition(conv_cur, -min(20, 2 * n - 1))[-min(20, 2*n-1):]
            bal_dir = peak_balance_direction(f, cur_peaks, n)
            v = v + 0.5 * bal_dir

        # Multiplicative weights update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f_new = f * np.exp(exponent)
        fs = f_new.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 1e-5)
            no_improve = 0
            continue
        f_new /= fs
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()
        f = f_new

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_lr *= 0.9998
        if cur_lr < 5e-5:
            cur_lr = 5e-5
        cur_temp *= 1.005
        if it % 800 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.93))

        # Restart if stuck
        if no_improve > 400:
            rng = np.random.default_rng((it * 7919 + 17) % 1000000)
            noise_scale = 0.001 + 0.001 * min(no_improve // 400, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 1e-4)
            cur_temp = temperature * 0.3
            no_improve = 0

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SG iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def coord_descent_golden(f0, time_limit, verbose=False):
    """Coordinate descent with golden-section line search on each coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Search order: center-out
    mid = n // 2
    order = []
    for d in range(mid + 1):
        if mid + d < n:
            order.append(mid + d)
        if mid - d >= 0 and (d > 0 or mid == 0):
            order.append(mid - d)

    golden_ratio = (np.sqrt(5.0) - 1.0) / 2.0

    sweeps = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        sweeps += 1
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            # Golden section search for optimal value of f[j]
            # Keep others proportional
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            def obj_at(val):
                if val < 0 or val > 0.999:
                    return 1e10
                scale = (1.0 - val) / denom
                f_trial = f.copy()
                f_trial[j] = val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    return 1e10
                f_trial /= st
                return fast_convolve(f_trial).max()

            # Search in [max(0, old_val*0.1), min(0.5, old_val*5)]
            lo = max(0.0, old_val * 0.1)
            hi = min(0.5, old_val * 5.0)
            if hi - lo < 1e-10:
                continue

            # Golden section
            c = hi - golden_ratio * (hi - lo)
            d_pt = lo + golden_ratio * (hi - lo)
            fc = obj_at(c)
            fd = obj_at(d_pt)
            for _ in range(15):
                if fc < fd:
                    hi = d_pt
                    d_pt = c
                    fd = fc
                    c = hi - golden_ratio * (hi - lo)
                    fc = obj_at(c)
                else:
                    lo = c
                    c = d_pt
                    fc = fd
                    d_pt = lo + golden_ratio * (hi - lo)
                    fd = obj_at(d_pt)

            best_val = c if fc < fd else d_pt
            best_obj_j = min(fc, fd)

            if best_obj_j < best_obj - 1e-13:
                scale = (1.0 - best_val) / denom
                f[j] = best_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = np.maximum(f, 0)
                f /= f.sum()
                best_obj = best_obj_j
                best_f = f.copy()
                improved = True

        if not improved:
            break

    return best_f, best_obj


def pair_exchange_grad(f0, time_limit, verbose=False):
    """Gradient-informed pair exchange: transfer mass from high-gradient to low-gradient positions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

        f = f / s
    n = len(f)

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:

        top_k_peaks = min(30, 2 * n - 1)
        top_peaks = np.argpartition(conv, -top_k_peaks)[-top_k_peaks:]

        # Compute gradient of max (using top peak)
        k_max = int(np.argmax(conv))
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        grad = np.zeros(n)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * f[k_max - idx]

        # Find best pair to swap: high f with high grad, low f with low grad
        f_sorted = np.argsort(f)
        g_sorted = np.argsort(grad)

        improved = False
        # Try a few candidate pairs
        for attempt in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            # Source: high value, high gradient (contributes to peak)
            src_candidates = g_sorted[-min(50, n):]
            src_candidates = src_candidates[f[src_candidates] > 1e-8]
            if len(src_candidates) == 0:
                break
            # Destination: low value, low gradient
            dst_candidates = f_sorted[:min(50, n)]
            dst_candidates = dst_candidates[grad[dst_candidates] < grad.max()]

            src = src_candidates[np.random.randint(len(src_candidates))]
            dst = dst_candidates[np.random.randint(len(dst_candidates))]
            if src == dst:
                continue

            # Optimal transfer: solve for t that minimizes max(conv) at f + t*(e_dst - e_src)
            # Approximate: use first-order condition
            # db_k/dt = 2*(f[k-dst]*(-1) + f[k-src]*(1)) for the max peak
            # We want to reduce the max, so transfer from src (which increases peak) to dst
            old_src = f[src]
            old_dst = f[dst]
            if old_src < 1e-10:
                continue

            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = old_src * frac
                f[src] = old_src - amt
                f[dst] = old_dst + amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[src] = old_src
                    f[dst] = old_dst
                    break

        if not improved:
            # Try systematic approach: for each top peak, find worst contributor
            for k in top_peaks[:8]:
                if time.monotonic() - t0 > time_limit:
                    break
                k = int(k)
                js2 = max(0, k - n + 1)
                je2 = min(n - 1, k)
                idx2 = np.arange(js2, je2 + 1)
                contribs = f[idx2] * f[k - idx2]
                if len(contribs) < 2:
                    continue
                hi = idx2[np.argmax(contribs)]
                lo = idx2[np.argmin(contribs)]
                if hi == lo or f[hi] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[hi] * frac
                    f[hi] -= amt
                    f[lo] += amt
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-13:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[hi] += amt
                        f[lo] -= amt
                        break
            if not improved:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length and normalize."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    # Gaussian centered
    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 5.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Asymmetric: concentrated on one side
    asym = np.exp(-np.arange(n) * (2.0 + rng.random() * 3.0) / n)
    inits.append(asym / asym.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 80)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - 0.5)
    tri = np.maximum(tri, 1e-10)
    inits.append(tri / tri.sum())

    return inits[:count]

                balance_interval=cfg["bal"])
            try_seq(a_opt, f"deep_{p}")

    # === Phase 3: Coordinate descent + peak balancing ===
    print(f"\n=== Phase 3: Coord Descent + Peak Balance (best={best_score:.8f}) ===")
    t_phase3_end = time.monotonic() + 380.0

    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent
        rem = t_phase3_end - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.5, 150)
            print(f"  Coord descent t={t_cd:.0f}s")
            a_opt, _ = coord_descent_golden(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_1")

        # Peak balance + re-optimize
        rem = t_phase3_end - time.monotonic()
        if rem > 20 and best_seq is not None:
            n_best = len(best_seq)
            conv = fast_convolve(best_seq)
            top_peaks = np.argpartition(conv, -20)[-20:]
            bal_dir = peak_balance_direction(best_seq, top_peaks, n_best)
            for step in [0.001, 0.005, 0.01, 0.02]:
                f_bal = project_simplex(best_seq + step * bal_dir)
                a_opt, _ = subgradient_optimize(
                    f_bal, time_limit=min(30, rem * 0.2),
                    top_k=max(15, n_best // 10), temperature=2e-5,
                    lr=0.002, momentum=0.90, balance_interval=100)
                try_seq(a_opt, f"bal_{step}")
                rem = t_phase3_end - time.monotonic()
                if rem < 15:
                    break

    # === Phase 4: Pair exchange local search ===
    print(f"\n=== Phase 4: Pair Exchange (best={best_score:.8f}) ===")
    t_phase4_end = time.monotonic() + 280.0

    if best_seq is not None:
        rem = t_phase4_end - time.monotonic()
        if rem > 20:
            t_pex = min(rem * 0.7, 150)
            print(f"  Pair exchange t={t_pex:.0f}s")
            a_opt, _ = pair_exchange_grad(best_seq, time_limit=t_pex)
            try_seq(a_opt, "pex")

            # Another round of coord descent after pair exchange
            rem = t_phase4_end - time.monotonic()
            if rem > 15 and best_seq is not None:
                t_cd2 = min(rem * 0.6, 80)
                print(f"  Coord descent 2 t={t_cd2:.0f}s")
                a_opt, _ = coord_descent_golden(best_seq, time_limit=t_cd2)
                try_seq(a_opt, "cd_2")

    # === Phase 5: Final fine polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    t_phase5_end = deadline

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine subgradient
        rem = t_phase5_end - time.monotonic()
        if rem > 20:
            t_fine = min(rem * 0.3, 60)
            print(f"  Fine SG t={t_fine:.0f}s")
            a_opt, _ = subgradient_optimize(
                best_seq, time_limit=t_fine,
                top_k=n_best, temperature=1e-6,
                lr=0.0005, momentum=0.95, balance_interval=500)
            try_seq(a_opt, "sg_fine")

        # Perturbation + re-optimize
        rem = t_phase5_end - time.monotonic()
        if rem > 20:
            a_pert, _ = refine_with_perturbation(
                best_seq, n_best, rng,
                time_limit=time.monotonic() + min(rem * 0.3, 80),
                temperature_base=5e-6)
            try_seq(a_pert, "pert")

        # Final coord descent
        rem = t_phase5_end - time.monotonic()
        if rem > 10 and best_seq is not None:
            t_cd3 = min(rem * 0.5, 40)
            print(f"  Final CD t={t_cd3:.0f}s")
            a_opt, _ = coord_descent_golden(best_seq, time_limit=t_cd3)
            try_seq(a_opt, "cd_final")

        # Last resort: tiny peak balance
        rem = t_phase5_end - time.monotonic()
        if rem > 5 and best_seq is not None:
            n_best = len(best_seq)
            conv = fast_convolve(best_seq)
            top_peaks = np.argpartition(conv, -10)[-10:]
            bal_dir = peak_balance_direction(best_seq, top_peaks, n_best)
            for step in [0.0005, 0.001, 0.002]:
                f_bal = project_simplex(best_seq + step * bal_dir)
                try_seq(f_bal, f"bal_final_{step}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
