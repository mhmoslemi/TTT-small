**Key insight for the optimizer:**

This is a sparse gradient! Only the positions that pair with k* get non-zero gradient.

This is essentially a weighted superposition of shifted copies of a.

```python
import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, top_k):
    n = len(a)
    n_conv = len(conv)
    k_actual = min(top_k, n_conv)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    # Weight by peak height
    w = conv[peaks]
    w = w / w.sum() if w.sum() > 0 else np.ones(len(peaks)) / len(peaks)
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        kg = k - idx
        mask = (kg >= 0) & (kg < n)
        sg[idx[mask]] += 2.0 * w[i] * a[kg[mask]]
    
    return sg

def sg_descent(a0, time_limit, lr=0.005, momentum=0.9, top_k=20, 
               anneal_top_k=True, verbose=False):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        sg = compute_subgradient(a, conv, top_k)
        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)
        
        conv = fast_convolve(a_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.8
            if no_improve > 500:
                # Restart with perturbation
                rng = np.random.default_rng(it * 7919)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                lr = max(0.001, lr * 2)
                conv = fast_convolve(a)
                no_improve = 0
        
        if verbose and it % 1000 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
    
    return best_a, best_obj
```

```python
import numpy as np
```

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(a)
    n_conv = len(conv)
    k_actual = min(top_k, n_conv)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        kg = k - idx
        mask = (kg >= 0) & (kg < n)
        sg[idx[mask]] += 2.0 * w[i] * a[kg[mask]]
    
    return sg

def peak_equalization(a0, time_limit, verbose=False):
    """Iteratively try to equalize the top peaks by adjusting contributing coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)

        if no_improve > 200:
            break
        
        # Find top peaks
        top_k = 5
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        mean_peak = peak_vals.mean()
        
        improved = False
        for pk in peaks:
            k = int(pk)
            if conv[k] <= mean_peak * 1.01:
                continue
            
            # Find contributing coordinates
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Compute contribution of each a[j] to this peak
            for j in range(js, je + 1):
                if a[j] < 1e-10:
                    continue
                # Try reducing a[j] and redistributing
                reduction = a[j] * 0.1
                if reduction < 1e-10:
                    continue
                
                # Redistribute to coordinates that don't contribute to this peak
                mask = np.ones(n, dtype=bool)
                mask[js:je+1] = False  # don't redistribute to same peak contributors
                # Also avoid coordinates that contribute to other top peaks
                for pk2 in peaks:
                    k2 = int(pk2)
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    mask[js2:je2+1] = False
                
                if mask.sum() == 0:
                    continue
                
                f_trial = a.copy()
                f_trial[j] -= reduction
                if f_trial[j] < 0:
                    continue
                # Redistribute to masked positions proportional to current values
                target_sums = f_trial[mask].sum()
                if target_sums < 1e-10:
                    f_trial[mask] += reduction / mask.sum()
                else:
                    f_trial[mask] += f_trial[mask] * (reduction / target_sums)
                
                if f_trial.sum() < 1e-10:
                    continue
                f_trial = f_trial / f_trial.sum()
                
                new_conv = fast_convolve(f_trial)
                new_obj = new_conv.max()
                
                if new_obj < best_obj - 1e-12:
                    a = f_trial
                    conv = new_conv
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
        
        if not improved:
            no_improve += 1
            # Try random coordinate perturbation

                j = int(rng.integers(0, n))

                    continue

                    old = f_trial[j]
                    f_trial[j] = old * factor
                    rest_sum = f_trial.sum() - f_trial[j]
                    if rest_sum < 1e-10:
                        continue
                    mask = np.ones(n, dtype=bool)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_geometric_init(n, ratio=0.97, side='left'):
    """Create a geometric sequence initialization."""
    if side == 'left':
        a = ratio ** np.arange(n)
    else:
        a = ratio ** np.arange(n - 1, -1, -1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def make_random_init(n, seed, sparsity=0.5):
    """Create a random sparse initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    if sparsity < 1.0:
        mask = rng.random(n) < sparsity
        a[~mask] = 0
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def make_binary_init(n, seed):
    """Create a random binary initialization."""
    rng = np.random.default_rng(seed)
    a = rng.integers(0, 2, n).astype(np.float64)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a
