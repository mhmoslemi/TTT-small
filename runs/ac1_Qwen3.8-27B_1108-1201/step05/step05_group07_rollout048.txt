def fft_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    u = np.sort(v)[::-1]
    n = len(u)
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_of_max_conv(a, conv, top_k, temperature):
    """Compute subgradient of max(conv) using softmax weighting of top peaks."""
    n = len(a)
    k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k_idx = int(peaks[i])
        js = max(0, k_idx - n + 1)
        je = min(n - 1, k_idx)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k_idx - idx]
    return sg

def fft_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_sg(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k_idx = int(peaks[i])
        js = max(0, k_idx - n + 1)
        je = min(n - 1, k_idx)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k_idx - idx]
    return sg

            if best_j_val != old_val:
                # Reapply the best value
                old_val = f[j]
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 3.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Periodic modulation
    p = rng.integers(2, max(3, n // 64))
    eps = 0.3 + rng.random() * 0.4
    a = 1.0 + eps * np.cos(2 * np.pi * p * np.arange(n) / n)
    a = np.maximum(a, 1e-8)
    inits.append(a / a.sum())

    # Double periodic
    p1 = rng.integers(2, max(3, n // 100))
    p2 = rng.integers(p1 + 1, max(p1 + 2, n // 50))
    e1 = 0.2 + rng.random() * 0.3
    e2 = 0.1 + rng.random() * 0.2
    a = 1.0 + e1 * np.cos(2 * np.pi * p1 * np.arange(n) / n) + e2 * np.cos(2 * np.pi * p2 * np.arange(n) / n)
    a = np.maximum(a, 1e-8)
    inits.append(a / a.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kl = max(3, n // 80)
    kernel = np.ones(kl) / kl
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Two-peak (edges high, center low)
    center_dist = np.abs(np.arange(n) - n / 2.0)
    u_shape = 1.0 + 0.5 * (center_dist / (n / 2.0)) ** 2
    inits.append(u_shape / u_shape.sum())

    return inits[:count]


def anneal_perturb(f, n_pert, scale_base, rng, verbose=False):
    """Simulated annealing style perturbation: accept worse with probability."""
    best_f = f.copy()
    best_obj = fft_convolve(f).max()
    cur_f = f.copy()
    cur_obj = best_obj

    for i in range(n_pert):
        scale = scale_base * (1.0 - i / n_pert) + 0.0001
        perturbed = cur_f + rng.standard_normal(len(f)) * scale
        perturbed = project_simplex(perturbed)

        new_obj = fft_convolve(perturbed).max()

        if new_obj < cur_obj:
            cur_f = perturbed
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_f = perturbed.copy()
        else:
            # Accept with probability exp(-(new_obj - cur_obj) / scale)
            delta = (new_obj - cur_obj) / max(scale * 10, 1e-10)
            if delta < 50 and rng.random() < math.exp(-delta):
                cur_f = perturbed
                cur_obj = new_obj
            else:
                cur_f = best_f.copy()
                cur_obj = best_obj

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining accelerated subgradient, peak-pair search,
    multi-peak balancing, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform n=512")
        best_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()

    n_parent = len(parent)

    # === Phase 1: Coarse optimization on parent and structured inits ===
    print(f"\n=== Phase 1: Coarse Optimization (best={best_score:.8f}) ===")

    candidates = [parent]
    # Add structured inits at parent length
    candidates.extend(structured_inits(n_parent, rng, count=3))
    # Add structured inits at nearby lengths
    for n_try in [n_parent // 2, n_parent * 3 // 2]:
        if n_try >= 64:
            candidates.extend(structured_inits(n_try, rng, count=2))

    for idx, cand in enumerate(candidates):
        rem = deadline - time.monotonic() - 400
        if rem < 15:
            break
        t_alloc = min(rem * 0.1, 60)
        if t_alloc < 5:
            t_alloc = min(rem * 0.2, 40)

        print(f"  Cand {idx}: n={len(cand)}, t={t_alloc:.0f}s")
        a_opt, _ = accelerated_sg(cand, time_limit=t_alloc,
                                  top_k=max(20, len(cand) // 6),
                                  temperature=5e-5, lr=0.003,
                                  momentum=0.85, label=f"c{idx}")
        try_seq(a_opt, f"coarse_{idx}")

    # === Phase 2: Deep optimization on current best ===
    print(f"\n=== Phase 2: Deep Optimization (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {"top_k": max(30, n_best // 3), "temp": 1e-5, "lr": 0.002, "mom": 0.90},
            {"top_k": max(15, n_best // 8), "temp": 5e-6, "lr": 0.001, "mom": 0.93},
            {"top_k": max(50, n_best // 2), "temp": 1e-4, "lr": 0.004, "mom": 0.80},
            {"top_k": max(10, n_best // 12), "temp": 1e-6, "lr": 0.0008, "mom": 0.95},
        ]

        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 300
            if rem < 10:
                break
            t_alloc = min(rem * 0.3, 120)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init)

            print(f"  Deep {p}: t={t_alloc:.0f}s")
            a_opt, _ = accelerated_sg(f_init, time_limit=t_alloc,
                                      top_k=cfg["top_k"], temperature=cfg["temp"],
                                      lr=cfg["lr"], momentum=cfg["mom"],
                                      label=f"d{p}")
            try_seq(a_opt, f"deep_{p}")

    # === Phase 3: Peak-pair targeted search ===
    print(f"\n=== Phase 3: Peak-Pair Search (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_pp = min(rem * 0.6, 150)
            print(f"  Peak-pair: t={t_pp:.0f}s")
            a_opt, _ = peak_pair_search(best_seq, time_limit=t_pp)
            try_seq(a_opt, "peak_pair")

            rem = deadline - time.monotonic() - 150
            if rem > 15:
                t_mb = min(rem * 0.5, 80)
                print(f"  Multi-peak balance: t={t_mb:.0f}s")
                a_opt, _ = multi_peak_balance(best_seq, time_limit=t_mb)
                try_seq(a_opt, "balance")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Length Exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        target_lengths = []
        n_best = len(best_seq)
        for factor in [0.75, 1.25, 1.5, 2.0]:
            tl = int(n_best * factor)
            if tl >= 32 and tl <= 4096 and tl not in target_lengths:
                target_lengths.append(tl)

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            f_interp = interpolate_seq(best_seq, target_n)
            t_alloc = min(rem * 0.3, 100)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = accelerated_sg(f_interp, time_limit=t_alloc,
                                      top_k=max(20, target_n // 5),
                                      temperature=5e-5, lr=0.0025,
                                      momentum=0.87, label=f"L{target_n}")
            try_seq(a_opt, f"len_{target_n}")

            # Follow up with peak-pair on the new length if it improved
            rem = deadline - time.monotonic() - 100
            if rem > 10 and best_seq is not None and len(best_seq) == target_n:
                a_opt, _ = peak_pair_search(best_seq, time_limit=min(rem, 40))
                try_seq(a_opt, f"pp_len_{target_n}")

    # === Phase 5: Annealing perturbation ===
    print(f"\n=== Phase 5: Annealing (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_anneal = min(rem * 0.5, 60)
            # Convert time to number of perturbation steps
            n_pert = int(t_anneal * 200)  # rough estimate
            n_pert = max(100, min(n_pert, 5000))
            print(f"  Annealing: {n_pert} steps")
            a_opt, _ = anneal_perturb(best_seq, n_pert=n_pert,
                                       scale_base=0.002, rng=rng)
            try_seq(a_opt, "anneal")

    # === Phase 6: Fine polishing ===
    print(f"\n=== Phase 6: Fine Polishing (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine subgradient
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_fine = min(rem * 0.4, 50)
            print(f"  Fine SG: t={t_fine:.0f}s")
            a_opt, _ = accelerated_sg(best_seq, time_limit=t_fine,
                                      top_k=n_best, temperature=1e-7,
                                      lr=0.0003, momentum=0.96, label="fine")
            try_seq(a_opt, "sg_fine")

        # Peak-pair polish
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_pp = min(rem * 0.5, 30)
            print(f"  Peak-pair polish: t={t_pp:.0f}s")
            a_opt, _ = peak_pair_search(best_seq, time_limit=t_pp)
            try_seq(a_opt, "pp_fine")

        # Coordinate descent
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_cd = min(rem * 0.7, 40)
            print(f"  Coord descent: t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

        # Another round of fine SG after CD
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, _ = accelerated_sg(best_seq, time_limit=min(rem, 15),
                                      top_k=max(5, n_best // 4),
                                      temperature=1e-8, lr=0.0002,
                                      momentum=0.97, label="final")
            try_seq(a_opt, "sg_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
