def project_simplex(v):

    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]

    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def fourier_refine(a0, time_limit, n_fourier=12):
    """Optimize over Fourier coefficients for structured improvement."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    # Initial Fourier coefficients
    A = np.fft.rfft(a)
    # Use only the first n_fourier components (plus DC)
    M = min(n_fourier, n // 2)
    
    def fourier_to_seq(coeffs):
        # coeffs: [dc, re1, im1, re2, im2, ...]
        # Actually, let's parametrize differently
        # a_i = (1/n) * (1 + sum_{m=1}^{M} c_m * cos(2*pi*m*i/n) + d_m * sin(2*pi*m*i/n))
        x = np.arange(n, dtype=np.float64)
        base = np.ones(n) / n
        for m in range(1, M + 1):
            c = coeffs[2*(m-1)]
            d = coeffs[2*(m-1)+1]
            base += (c * np.cos(2 * np.pi * m * x / n) + d * np.sin(2 * np.pi * m * x / n)) / n
        base = np.maximum(base, 0.0)
        s = base.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return base / s
    
    def objective(coeffs):
        seq = fourier_to_seq(coeffs)
        conv = fast_convolve(seq)
        return conv.max()
    
    # Start from current sequence's Fourier components
    x = np.arange(n, dtype=np.float64)
    a_centered = a - 1.0 / n
    init_coeffs = np.zeros(2 * M)
    for m in range(1, M + 1):
        init_coeffs[2*(m-1)] = 2.0 * np.sum(a_centered * np.cos(2 * np.pi * m * x / n))
        init_coeffs[2*(m-1)+1] = 2.0 * np.sum(a_centered * np.sin(2 * np.pi * m * x / n))
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    from scipy.optimize import minimize
    
    t0 = time.monotonic()
    # Multiple restarts with different initial points
    n_restarts = 3
    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        if r == 0:
            x0 = init_coeffs
        else:
            rng = np.random.default_rng(r * 12345 + 7)
            x0 = init_coeffs + rng.standard_normal(2 * M) * 0.01
        
        try:
            res = minimize(objective, x0, method='Nelder-Mead',
                          options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12})
            if res.fun < best_obj:
                best_obj = res.fun
                best_a = fourier_to_seq(res.x)
        except Exception:
            pass
    
    return best_a, best_obj


def targeted_peak_reduction(a0, time_limit):
    """Reduce the highest peak by targeted perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Find which indices contribute to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        
        # Compute contribution of each pair
        # conv[k_peak] = sum_{j=js}^{je} a[j] * a[k_peak - j]
        # To reduce this, we want to decrease a[j] * a[k_peak - j] for the largest terms
        # This means we want to move mass from high-contribution indices to low-contribution ones
        
        # Strategy: move mass from the index with highest a[j]*a[k_peak-j] 
        # to an index that doesn't contribute to this peak
        
        affected = np.arange(js, je + 1)
        contributions = a[affected] * a[k_peak - affected]
        
        # Find the index with highest contribution
        j_max = affected[np.argmax(contributions)]
        
        # Find an index that doesn't contribute to this peak (outside [js, je])
        # and has low value (to minimize increase elsewhere)
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        
        # Choose the outside index with lowest a value
        j_min_cand = outside[np.argmin(a[outside])]
        
        # Try moving a small amount from j_max to j_min_cand
        improved = False
        for delta in [0.001, 0.002, 0.005, 0.01, 0.02]:
            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min_cand] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate / candidate.sum()
            
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try a broader perturbation
            if it % 50 == 0:
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                elif it > 500:
                    break
            elif it > 1000:
                break
    
    return best_a, best_obj

        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        best_candidate = a
        best_val = conv.max()

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T *= 1.05
            lr *= 0.97
        if no_improve > 40:
            T *= 0.8
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def fourier_to_seq(coeffs, n, M):
    """Convert Fourier coefficients to a sequence."""
    x = np.arange(n, dtype=np.float64)
    base = np.ones(n) / n
    for m in range(1, M + 1):
        c = coeffs[2 * (m - 1)]
        d = coeffs[2 * (m - 1) + 1]
        base += (c * np.cos(2.0 * np.pi * m * x / n) +
                 d * np.sin(2.0 * np.pi * m * x / n)) / n
    base = np.maximum(base, 0.0)
    s = base.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return base / s


def fourier_objective(coeffs, n, M):
    """Objective for Fourier parametrization."""
    seq = fourier_to_seq(coeffs, n, M)
    conv = fast_convolve(seq)
    return conv.max()


def fourier_refine(a0, time_limit, n_fourier=10):
    """Optimize over Fourier coefficients for structured improvement."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    M = min(n_fourier, n // 2)
    x = np.arange(n, dtype=np.float64)

    # Extract initial Fourier coefficients
    a_centered = a - 1.0 / n
    init_coeffs = np.zeros(2 * M)
    for m in range(1, M + 1):
        init_coeffs[2 * (m - 1)] = 2.0 * np.sum(a_centered * np.cos(2.0 * np.pi * m * x / n))
        init_coeffs[2 * (m - 1) + 1] = 2.0 * np.sum(a_centered * np.sin(2.0 * np.pi * m * x / n))

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    n_restarts = max(2, int((time_limit // 30)) + 1)

    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        if r == 0:
            x0 = init_coeffs.copy()
        else:
            rng = np.random.default_rng(r * 12345 + 7)
            scale = 0.005 * (1 + r * 0.3)
            x0 = init_coeffs + rng.standard_normal(2 * M) * scale

        try:
            res = minimize(fourier_objective, x0, args=(n, M), method='Nelder-Mead',
                           options={'maxiter': 8000, 'xatol': 1e-10, 'fatol': 1e-12,
                                    'maxfev': 50000})
            if res.fun < best_obj:
                best_obj = res.fun
                best_a = fourier_to_seq(res.x, n, M)
                print("    Fourier restart {}: obj={:.10f}".format(r, best_obj))
        except Exception:
            pass

    return best_a, best_obj


def targeted_peak_reduction(a0, time_limit):
    """Reduce the highest peak by targeted coordinate perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)

        if len(affected) == 0:
            break

        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        j_min_cand = outside[np.argmin(a[outside])]

        improved = False
        for delta in [0.0005, 0.001, 0.002, 0.005, 0.01]:
            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min_cand] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate / candidate.sum()
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 30 == 0:
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                elif stale > 500:
                    break

    return best_a, best_obj


def multi_peak_sg(a0, time_limit, top_k, lr_init, mom):
    """Multi-peak subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 2.0
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        mom = min(0.98, mom + 0.0001)

        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 200) / 400.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.6
            no_improve = 0

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    for freq in [2, 3, 5, 7]:
        h = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * x / n)
        h = np.maximum(h, 0.01)
        inits.append(h / h.sum())

    return inits


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved. state = [best_score, best_seq]."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search:
    Phase 1: Quick multi-n exploration with FW-LP
    Phase 2: Deep FW-LP refinement on best n
    Phase 3: Fourier-parametrized structured optimization
    Phase 4: Smoothed gradient + multi-peak subgradient
    Phase 5: Targeted peak reduction + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = [float('inf'), None]  # [best_score, best_seq]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: Quick Multi-n Exploration (~200s) ===
    print("\n=== Phase 1: Quick Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, 512, 640, 768, 896, 1024, 1280]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1400
        if remaining < 15:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(remaining, 40)
        print("  FW-LP n={}, t={:.0f}s".format(n_try, t_alloc))
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        try_and_update(a_opt, state, "fw_n{}".format(n_try))

        remaining = deadline - time.monotonic() - 1350
        if remaining > 10:
            a_opt2, _ = smoothed_grad_descent(a_opt, time_limit=min(remaining, 15),
                                              T_init=0.001, lr_init=0.03)
            try_and_update(a_opt2, state, "sg_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep FW-LP (~500s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 900
        if remaining > 30:
            t_alloc = min(remaining, 400)
            print("  Deep FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "deep_fw")

            # Perturbed restarts
            for scale in [0.003, 0.008, 0.015]:
                remaining = deadline - time.monotonic() - 700
                if remaining < 15:
                    break
                f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
                t_alloc = min(remaining, 60)
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
                try_and_update(a_opt2, state, "fw_pert_{}".format(scale))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Fourier-parametrized optimization (~250s) ===
    print("\n=== Phase 3: Fourier Refinement ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 500
        if remaining > 20:
            t_alloc = min(remaining, 200)
            print("  Fourier refine, t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_fourier=12)
            try_and_update(a_opt, state, "fourier_12")

        remaining = deadline - time.monotonic() - 350
        if remaining > 15:
            t_alloc = min(remaining, 100)
            print("  Fourier refine (more modes), t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_fourier=20)
            try_and_update(a_opt, state, "fourier_20")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Smoothed gradient + multi-peak SG (~350s) ===
    print("\n=== Phase 4: Gradient Methods ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Multiple temperature settings
        for T_init, lr_init in [(0.0003, 0.04), (0.001, 0.03), (0.003, 0.02)]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break
            t_alloc = min(remaining, 100)
            print("  Smoothed grad T={}, lr={}, t={:.0f}s".format(T_init, lr_init, t_alloc))
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "smooth_T{}".format(T_init))

        # Multi-peak subgradient
        sg_configs = [
            (10, 0.002, 0.94),
            (30, 0.003, 0.92),
            (50, 0.001, 0.95),
        ]
        for top_k, lr, mom in sg_configs:
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            t_alloc = min(remaining, 60)
            print("  Multi-peak SG k={}, t={:.0f}s".format(top_k, t_alloc))
            a_opt, _ = multi_peak_sg(state[1], time_limit=t_alloc,
                                     top_k=top_k, lr_init=lr, mom=mom)
            try_and_update(a_opt, state, "mpsg_k{}".format(top_k))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Targeted peak reduction + final polish (~250s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Targeted peak reduction
        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 100)
            print("  Targeted peak reduction, t={:.0f}s".format(t_alloc))
            a_opt, _ = targeted_peak_reduction(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "targeted")

        # One more FW-LP pass
        remaining = deadline - time.monotonic() - 60
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print("  Final FW-LP, t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "final_fw")

        # Fine smoothed gradient
        remaining = deadline - time.monotonic() - 30
        if remaining > 5:
            t_alloc = min(remaining, 30)
            print("  Fine smooth grad, t={:.0f}s".format(t_alloc))
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=0.0001, lr_init=0.005)
            try_and_update(a_opt, state, "fine_smooth")

        # Tiny perturbations with quick polish
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            for _ in range(5):
                remaining = deadline - time.monotonic() - 5
                if remaining < 3:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0005
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = smoothed_grad_descent(f_pert, time_limit=min(remaining, 5),
                                                 T_init=0.0002, lr_init=0.003)
                try_and_update(a_opt, state, "perturb_fine")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
