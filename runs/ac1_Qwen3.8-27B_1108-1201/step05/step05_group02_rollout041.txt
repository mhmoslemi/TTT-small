```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based Frank-Wolfe with adaptive peak weighting
2. Accelerated subgradient with Nesterov momentum
3. Multi-length exploration with interpolation

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n, weights=None):
    """Solve LP: min weighted sum of peaks or min t s.t. (f*g)[k] <= t.
    If weights given: min sum_k w_k * (f*g)[k] s.t. g >= 0, sum(g) = 1.
    Returns (g, obj) or (None, None)."""
    if weights is not None:
        # Minimize weighted sum: min sum_k w_k * (f*g)[k]
        # = min sum_j g[j] * sum_k w_k * f[k-j]
        # So c[j] = sum_k w_k * f[k-j] for valid k
        c = np.zeros(n)
        for j in range(n):
            # (f*g)[k] = sum_i f[i] * g[k-i], so coefficient of g[j] in (f*g)[k] is f[k-j]
            # sum_k w_k * f[k-j] where 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
            ks = np.arange(j, min(j + n, 2 * n - 1))
            idx = ks - j  # = 0, 1, ..., min(n-1, 2n-2-j)
            c[j] = np.dot(weights[ks], f[idx])
        
        res = linprog(c, A_eq=np.ones((1, n)), b_eq=np.array([1.0]),
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x, res.fun
        return None, None
    else:
        # Min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        n_constraints = 2 * n - 1
        A_ub = np.zeros((n_constraints, n + 1))
        for j in range(n):
            A_ub[j:j + n, j] = f
        A_ub[:, -1] = -1.0
        b_ub = np.zeros(n_constraints)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
        return None, None

        f = f / s
    n = len(f)

    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if use_weighted:
            # Compute weights based on current peaks
            k_actual = min(top_k_weights, len(conv))
            peaks = np.argpartition(conv, -k_actual)[-k_actual:]
            pv = conv[peaks]
            # Softmax weights with temperature
            temp = 0.01 * M
            scaled = pv / temp
            scaled -= scaled.max()
            w_all = np.zeros(len(conv))
            w_peaks = np.exp(scaled)
            w_peaks /= w_peaks.sum()
            w_all[peaks] = w_peaks
            g, _ = solve_lp_min_peak(f, n, weights=w_all)
        else:
            g, _ = solve_lp_min_peak(f, n)
        
        if g is None:
            g = np.ones(n) / n
        else:
            g = np.maximum(g, 0)
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine line search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()

def accelerated_sg_optimize(f0, time_limit, lr=0.003, momentum=0.93, top_k=50, temp=0.0001, verbose=False):
    """Accelerated subgradient with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    y = f.copy()  # Nesterov lookahead
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    beta = 1.0  # Nesterov beta

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov update
        beta_new = (1 + np.sqrt(1 + 4 * beta * beta)) / 2
        y = f + (beta / beta_new - 1) * (f - y) if it > 1 else f.copy()
        beta = beta_new

        conv_y = fast_convolve(y)
        k_actual = min(top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -k_actual)[-k_actual:]
        pv = conv_y[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient of max_k conv(f,f)(k)
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            # d/d f[j] conv(f,f)[k] = 2*f[k-j] if 0 <= k-j <= n-1
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx  # = k - j for j in idx
            sg[idx] += 2.0 * w[i] * y[kg]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(y + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 100 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def peak_cd_optimize(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Try adjusting coordinates that contribute most to this peak
            contrib = []
            for j in range(js, je + 1):
                cj = f[j] * f[peak_k - j]
                contrib.append((cj, j))
            contrib.sort(reverse=True)
            
            for cj, j in contrib[:10]:  # Top 10 contributors
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-13:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj
                multipliers = [0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.3, 1.5, 2.0]

                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.8:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    ss = f_trial.sum()
                    if ss < 1e-15:
                        continue
                    f_trial = f_trial / ss

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_f, best_obj
```

```python
def spectral_shaping(f0, time_limit, verbose=False):
    """Iterative spectral shaping to flatten |F(ω)|^2."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    alpha = 0.1  # relaxation parameter
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute spectrum
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        F = np.fft.rfft(f, fft_size)
        P = np.abs(F) ** 2
        
        # Target: flat spectrum = min(P)
        # Or: reduce the max
        P_max = P.max()
        P_min = P.min()
        
        # Create target: flatten toward the mean
        P_mean = P.mean()
        # Only reduce the peaks above mean
        P_target = np.minimum(P, P_mean * 1.2)
        
        # Inverse: F_target = sqrt(P_target) * phase(F)
        phase = np.angle(F)
        F_target = np.sqrt(P_target) * np.exp(1j * phase)
        
        # Inverse FFT
        f_new = np.fft.irfft(F_target, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f_new = f_new / ss
        
        # Relaxation
        f_relaxed = f + alpha * (f_new - f)
        f_relaxed = np.maximum(f_relaxed, 0.0)
        ss = f_relaxed.sum()
        if ss < 1e-15:
            break
        f_relaxed = f_relaxed / ss
        
        conv = fast_convolve(f_relaxed)
        co = conv.max()
        
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_relaxed.copy()
            f = f_relaxed
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Spectral iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        # Adapt alpha
        if no_improve > 50:
            alpha *= 0.9
            no_improve = 0
        if alpha < 0.001:
            # Restart
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            alpha = 0.1
            no_improve = 0
        
        if no_improve > 500:
            break
    
    return best_f, best_obj
```

```python
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()

    epsilon = 0.001  # transfer amount
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Pick random pair (i, j)
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        
        # Determine direction: transfer from larger to smaller, or random
        if f[i] < epsilon:
            continue
        if np.random.random() < 0.5:
            di = -epsilon  # decrease f[i]
            dj = +epsilon  # increase f[j]
        else:
            di = +epsilon
            dj = -epsilon
            if f[j] < epsilon:
                continue
        
        # Compute change in convolution (first order)
        # δ(conv)(k) = 2*(di*f[k-i] + dj*f[k-j]) for valid k
        delta_conv = np.zeros(2*n-1)
        # Contribution from δf[i] = di
        ki_start = max(0, i)
        ki_end = min(2*n-2, i + n - 1)
        for k in range(ki_start, ki_end + 1):
            kj = k - i
            if 0 <= kj <= n-1:
                delta_conv[k] += 2 * di * f[kj]
        # Contribution from δf[j] = dj
        kj_start = max(0, j)
        kj_end = min(2*n-2, j + n - 1)
        for k in range(kj_start, kj_end + 1):
            ki = k - j
            if 0 <= ki <= n-1:
                delta_conv[k] += 2 * dj * f[ki]
        
        # New max
        new_conv = conv + delta_conv
        new_obj = np.max(new_conv)
        old_obj = conv.max()
        delta_obj = new_obj - old_obj
        
        # Accept or reject
        if delta_obj < 0 or np.random.random() < np.exp(-delta_obj / T):
            f[i] += di
            f[j] += dj
            f = np.maximum(f, 0)
            ss = f.sum()
            f = f / ss
            conv = fast_convolve(f)  # Recompute exactly (first order might be wrong)
            cur_obj = conv.max()
            if cur_obj < best_obj - 1e-15:
                best_obj = cur_obj
                best_f = f.copy()
        
        # Cool down
        T *= 0.9999
        if T < 0.0001:
            T = 0.001  # reheat
            f = best_f.copy()
            conv = fast_convolve(f)
```

Actually, I think the issue is that I was computing the derivative of conv(k) with respect to f[i], but for the change in conv when f changes by δ, I should use:
δ conv(k) ≈ sum_i δf[i] * d/d f[i] conv(k)

And d/d f[i] conv(k) = f[k-i] (if 0<=k-i<=n-1 and k≠2i) or 2*f[i] (if k=2i).

So:
δ conv(k) ≈ di * f[k-i] + dj * f[k-j] (for the generic case where k≠2i and k≠2j)
           ≈ di * 2*f[i] + dj * f[k-j] (if k=2i and k≠2j)
           etc.

The vectorized version would be:
delta_conv = di * (f reversed and shifted) + dj * (f reversed and shifted)

Specifically:
- The vector of d/d f[i] conv(k) for all k is: a vector of length 2n-1 where the k-th element is f[k-i] if 0<=k-i<=n-1, else 0. But with a factor of 2 at position k=2i.

This is essentially a "smeared" version of f, shifted by i positions.

Let me just implement it directly:

```python
def compute_subgrad_conv(f, i_val, j_val, di, dj, n):
    """Compute first-order change in conv(f,f) when f[i] += di, f[j] += dj."""
    delta = np.zeros(2*n-1)
    # Effect of changing f[i] by di:
    # d conv(k)/d f[i] = f[k-i] for 0<=k-i<=n-1, with factor 2 at k=2i
    k_lo = max(0, i)  # k-i >= 0 => k >= i
    k_hi = min(2*n-2, i + n - 1)  # k-i <= n-1 => k <= i+n-1
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi + 1)
        delta[ks] += di * f[ks - i]
        if 2*i <= 2*n-2:  # k=2i is in range
            delta[2*i] += di * f[i]  # extra factor for the diagonal
    # Effect of changing f[j] by dj:
    k_lo = max(0, j)
    k_hi = min(2*n-2, j + n - 1)
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi + 1)
        delta[ks] += dj * f[ks - j]
        if 2*j <= 2*n-2:
            delta[2*j] += dj * f[j]
    return delta
```

This is O(n) per evaluation, which is fast. But we still need to track the full conv vector to know the current max.

Actually, for a proper SA/MCMC approach, we need to:
1. Maintain the current conv vector
2. When we make a move (i, j, ε), compute δconv in O(n)
3. Update conv += δconv
4. Check if new max < old max (or accept with probability)
5. If rejected, revert

This gives us O(n) per step, which for n=832 is about 832 operations. In Python, with vectorized numpy, this should take about 1-5 microseconds per step. So we can do about 200K-1M steps per second. That's a lot!
