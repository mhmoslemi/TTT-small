"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP (bilinear
minimization) with subgradient polishing and multi-length exploration.
The key insight is that fixing one copy of the sequence turns the bilinear
auto-correlation peak into a linear function, solvable as an LP.
"""

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_matrix(b, n):
    """Build the matrix C such that C @ a = b * a (linear convolution)."""
    # C[k, i] = b[k - i] if 0 <= k - i < n, else 0
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            C[k, i] = b[k - i]
    return C

    Minimizes max_k (a*b)[k] by alternately solving LPs with one variable fixed.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    b = np.maximum(b_init, 0).astype(np.float64).copy()
    sa = a.sum()
    sb = b.sum()
    if sa < 1e-15:
        a = np.ones(n) / n
        sa = 1.0
    else:
        a /= sa
    if sb < 1e-15:
        b = np.ones(n) / n
        sb = 1.0
    else:
        b /= sb

    # Current objective
    conv = fast_convolve(a)
    cur_obj = conv.max()
    best_obj = cur_obj
    best_a = a.copy()

    # LP variables
    a_cvx = cp.Variable(n, nonneg=True)
    t_cvx = cp.Variable()
    obj = cp.Minimize(t_cvx)
    constraints = [cp.sum(a_cvx) == 1]

    t0 = time.monotonic()
    it = 0
    last_improve = 0

    while it < max_iter and (time.monotonic() - t0) < time_limit:
        it += 1

        # Solve LP: given b, find a minimizing max_k (a*b)[k]
        C = compute_conv_matrix(b, n)
        constraints = [cp.sum(a_cvx) == 1] + [C[k] @ a_cvx <= t_cvx for k in range(2*n-1)]
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                try:
                    prob.solve(solver=cp.CLARABEL)
                except Exception:
                    break

        if a_cvx.value is None:
            break

        a_new = np.maximum(a_cvx.value, 0)
        a_new = a_new / a_new.sum()

        # Check improvement
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            last_improve = it

        # Update b and swap
        b = a_new.copy()
        a = a_new.copy()

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ALT-LP iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, cur={obj_new:.10f}")

            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(best_a + noise)
            b = a.copy()
            last_improve = it

    return best_a, best_obj


def subgradient_polish(a_init, time_limit=60.0, top_k=100, temperature=0.00005, lr=0.003, momentum=0.85, verbose=False):
    """Subgradient optimization for polishing."""
    f = np.maximum(a_init, 0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        k_actual = min(cur_top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]

        v = momentum * v + (1 - momentum) * sg

        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp *= 1.008
        if it % 500 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.95))

        if no_improve > 300:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.002 + 0.001 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0002)
            cur_temp = temperature * 0.5
            no_improve = 0

    return best_f, best_obj


def pair_search(a_init, time_limit=30.0):
    """Pair exchange local search."""
    f = np.maximum(a_init, 0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        improved = False
        trials = 300
        for _ in range(trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or f[i] < 1e-10:
                continue
            for frac in [0.1, 0.3, 0.5]:
                amt = f[i] * frac
                f[i] -= amt
                f[j] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] += amt
                    f[j] -= amt
                    break
        if not improved:
            break

    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP (bilinear
minimization), subgradient polishing with Nesterov momentum, peak-deflation
local search, and multi-length exploration. The key innovation is the
alternating LP: fixing one copy of the sequence makes the auto-correlation
peak a linear function, solvable as a small LP. This provides excellent
initial points for subsequent non-convex refinement.
"""

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_matrix(b, n):
    """Build C such that C @ a = b * a (linear convolution). Vectorized."""
    idx = np.arange(2 * n - 1)[:, None] - np.arange(n)[None, :]
    mask = (idx >= 0) & (idx < n)
    idx_clipped = np.clip(idx, 0, n - 1)
    return np.where(mask, b[idx_clipped], 0.0)

    while it < max_iter and (time.monotonic() - t0) < time_limit:
        it += 1

        C = compute_conv_matrix(b, n)
        num_cons = 2 * n - 1
        constraints = [cp.sum(a_cvx) == 1]
        for k in range(num_cons):
            constraints.append(C[k] @ a_cvx <= t_cvx)
        prob = cp.Problem(obj, constraints)

        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                solved = True
                solver_used = s
                break
            except Exception:
                continue

        if not solved or a_cvx.value is None:
            break

        a_new = np.maximum(np.asarray(a_cvx.value, dtype=np.float64), 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new /= sn

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            last_improve_iter = it
            no_improve = 0
        else:
            no_improve += 1

        b = a_new.copy()

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ALT-LP iter {it}, t={elapsed:.1f}s, best={best_obj:.10f}, cur={obj_new:.10f}")

        if no_improve >= 30:
            rng = np.random.default_rng((it * 104729 + 31) % 1000000)
            noise = rng.standard_normal(n) * 0.004
            a_new = project_simplex(best_a + noise)
            b = a_new.copy()
            no_improve = 0

    return best_a, best_obj


def subgradient_optimize(f_init, time_limit=60.0, top_k=100, temperature=0.00004,
                         lr=0.003, momentum=0.87, verbose=False):
    """Multiplicative-weights subgradient optimization with Nesterov momentum."""
    f = np.maximum(f_init, 0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        k_actual = min(cur_top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]

        v = momentum * v + (1.0 - momentum) * sg

        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp *= 1.008
        if it % 500 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.95))

        if no_improve > 400:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.002 + 0.001 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0002)
            cur_temp = temperature * 0.5
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def peak_deflation_search(f_init, time_limit=30.0):
    """Peak-deflation local search targeting the highest convolution peaks."""
    f = np.maximum(f_init, 0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        current_max = conv.max()

        # Target top peaks
        top_k = min(20, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]

        for pk in sorted(peaks, key=lambda x: -conv[x])[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(pk)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            order = np.argsort(contribs)[::-1]

            for oi in order[:4]:
                i = int(idx[oi])
                if f[i] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    alt = f.copy()
                    amt = f[i] * frac
                    alt[i] -= amt
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    rest_sum = alt[mask].sum()
                    if rest_sum < 1e-15:
                        break
                    alt[mask] += amt * (alt[mask] / rest_sum)
                    val = fast_convolve(alt).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = alt.copy()
                        f = alt.copy()
                        improved = True
                        break
            if improved:
                break

        if not improved:
            # Random pair exchange
            rng = np.random.default_rng(int(time.monotonic() * 1e9) % 999999)
            for _ in range(300):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or f[i] < 1e-10:
                    continue
                for frac in [0.1, 0.2, 0.4]:
                    amt = f[i] * frac
                    f[i] -= amt
                    f[j] += amt
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[i] += amt
                        f[j] -= amt
                        break
            if not improved:
                break

    return best_f, best_obj


def coordinate_descent_search(f_init, time_limit=60.0):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f_init, 0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    mid = n // 2
    order = []
    for d in range(mid + 1):
        if mid + d < n:
            order.append(mid + d)
        if mid - d >= 0 and d > 0:
            order.append(mid - d)
    if mid > 0 and (mid - 0) not in order:
        order.append(0)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_conv = fast_convolve(f)
            cur_peak = cur_conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def get_evaluation(a):
    """Wrapper for evaluation that returns score."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: alternating LP + subgradient + peak deflation + coordinate
    descent, with multi-length exploration and iterative refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform n=768")

    # === Phase 1: Alternating LP on multiple lengths ===
    print("\n=== Phase 1: Alternating LP (multi-length) ===")
    t_start = time.monotonic()

    lengths = [512, 768, 1024, 1280]
    for n_try in lengths:
        rem = deadline - time.monotonic() - 400
        if rem < 40:
            break

        # Interpolate parent
        f_interp = interpolate_sequence(parent, n_try)

        t_alloc = min(rem * 0.15, 150)
        print(f"  ALT-LP n={n_try}, t={t_alloc:.0f}s")
        a_opt, obj_val = alternating_lp(f_interp, f_interp,
                                         max_iter=400, time_limit=t_alloc,
                                         verbose=(n_try == 768))
        try_seq(a_opt, f"altlp_{n_try}")

        # Second init: Hann window
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            x = np.linspace(0, 1, n_try)
            hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
            hann = np.maximum(hann, 1e-10)
            hann /= hann.sum()
            t_alloc2 = min(rem * 0.1, 80)
            print(f"  ALT-LP n={n_try} hann, t={t_alloc2:.0f}s")
            a_opt2, _ = alternating_lp(hann, hann,
                                        max_iter=200, time_limit=t_alloc2,
                                        verbose=False)
            try_seq(a_opt2, f"altlp_hann_{n_try}")

        # Third init: Gaussian
        rem = deadline - time.monotonic() - 250
        if rem > 25:
            center = n_try / 2.0
            sigma = n_try / 6.0
            gauss = np.exp(-0.5 * ((np.arange(n_try) - center) / sigma) ** 2)
            gauss = np.maximum(gauss, 1e-10)
            gauss /= gauss.sum()
            t_alloc3 = min(rem * 0.1, 60)
            a_opt3, _ = alternating_lp(gauss, gauss,
                                        max_iter=150, time_limit=t_alloc3,
                                        verbose=False)
            try_seq(a_opt3, f"altlp_gauss_{n_try}")

    # === Phase 2: Subgradient polish on best ===
    print(f"\n=== Phase 2: Subgradient Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {"top_k": max(50, n_best // 4), "temp": 0.00003, "lr": 0.002, "mom": 0.88},
            {"top_k": max(30, n_best // 6), "temp": 0.00001, "lr": 0.001, "mom": 0.93},
            {"top_k": max(100, n_best // 3), "temp": 0.0001, "lr": 0.004, "mom": 0.80},
            {"top_k": max(20, n_best // 10), "temp": 0.000005, "lr": 0.0008, "mom": 0.95},
        ]

        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 250
            if rem < 25:
                break
            t_alloc = min(rem * 0.2, 120)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init)

            print(f"  SG cfg{p}: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(f_init, time_limit=t_alloc,
                                             top_k=cfg["top_k"], temperature=cfg["temp"],
                                             lr=cfg["lr"], momentum=cfg["mom"])
            try_seq(a_opt, f"sg_{p}")

    # === Phase 3: More alternating LP on best ===
    print(f"\n=== Phase 3: ALT-LP Deep (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            n_best = len(best_seq)
            t_alloc = min(rem * 0.4, 200)
            print(f"  ALT-LP deep: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = alternating_lp(best_seq, best_seq,
                                       max_iter=500, time_limit=t_alloc,
                                       verbose=True)
            try_seq(a_opt, "altlp_deep")

            # With perturbation
            rem = deadline - time.monotonic() - 120
            if rem > 30:
                noise = rng.standard_normal(n_best) * 0.01
                f_pert = project_simplex(best_seq + noise)
                t_alloc2 = min(rem * 0.5, 100)
                a_opt2, _ = alternating_lp(f_pert, f_pert,
                                            max_iter=300, time_limit=t_alloc2,
                                            verbose=False)
                try_seq(a_opt2, "altlp_pert")

    # === Phase 4: Peak deflation + coordinate descent ===
    print(f"\n=== Phase 4: Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_pd = min(rem * 0.35, 80)
            print(f"  Peak deflation: t={t_pd:.0f}s")
            a_opt, _ = peak_deflation_search(best_seq, time_limit=t_pd)
            try_seq(a_opt, "peakdef")

        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_cd = min(rem * 0.6, 90)
            print(f"  Coord descent: t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_search(best_seq, time_limit=t_cd)
            try_seq(a_opt, "coord")

    # === Phase 5: Length refinement ===
    print(f"\n=== Phase 5: Length Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Try nearby lengths
        for delta in [64, -64, 128, -128]:
            target_n = n_best + delta
            if target_n < 128 or target_n > 2048:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 80)
            print(f"  Len n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = alternating_lp(f_interp, f_interp,
                                       max_iter=200, time_limit=t_alloc,
                                       verbose=False)
            try_seq(a_opt, f"len_{target_n}")
            # Quick SG polish
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                a_opt2, _ = subgradient_optimize(a_opt, time_limit=min(rem, 30),
                                                  top_k=max(30, target_n // 5),
                                                  temperature=0.00003, lr=0.002,
                                                  momentum=0.88)
                try_seq(a_opt2, f"len_{target_n}_sg")

    # === Phase 6: Final combined polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Ultra-fine SG
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem, 60),
                                             top_k=n_best, temperature=0.000002,
                                             lr=0.0003, momentum=0.96)
            try_seq(a_opt, "ultra_sg")

        # Peak deflation
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = peak_deflation_search(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "peakdef_fine")

        # Perturbation series
        for iter_p in range(5):
            rem = deadline - time.monotonic() - 15
            if rem < 12:
                break
            for scale in [0.001, 0.002, 0.004]:
                rem = deadline - time.monotonic() - 10
                if rem < 8:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)
                t_alloc = min(rem, 10)
                a_opt, _ = subgradient_optimize(f_pert, time_limit=t_alloc,
                                                 top_k=max(20, n_best // 8),
                                                 temperature=0.00002, lr=0.001,
                                                 momentum=0.92)
                try_seq(a_opt, f"pert_{iter_p}_{scale}")

        # Final coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = coordinate_descent_search(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, time={total_time:.1f}s)")
    return [float(x) for x in best_seq]
