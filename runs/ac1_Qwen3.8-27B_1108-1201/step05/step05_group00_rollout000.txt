"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast game-theoretic
primal-dual iterations with FFT-based operations, alternating LP
for lower bounds, and adaptive multi-resolution search.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    n = len(a)
    # (M(p) @ a)_j = sum_m a_m * p[j+m]
    # This is the 'valid' correlation of p and a (reversed)
    # p has length 2n-1, a has length n
    # For j=0: sum_m a_m * p[0+m] = a . p[0:n]
    # For j=n-1: sum_m a_m * p[n-1+m] = a . p[n-1:2n-1]
    return np.correlate(p, a[::-1], mode='valid')


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def game_iterate_fast(a0, time_limit, T_init=0.001, verbose=False):
    """
    Fast game-theoretic primal-dual iteration:
    - Dual: p ∝ exp(conv(a)/T) (softmax)
    - Primal: accelerated gradient on a^T H(p) a over simplex
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Dual update: p = softmax(conv(y) / T)
        conv_y = fast_convolve(y)
        scaled = conv_y / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Primal: one step of accelerated gradient on f(a) = a^T H(p) a
        # Gradient at y: 2 * H(p) @ y
        grad = 2.0 * hankel_mv(p, y)
        
        # Nesterov update
        if it == 1:
            t_k = 1.0
        else:
            momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum**2)) / 2.0
            t_k = (1.0 + momentum) / 2.0
        
        # Step size (Polyak or fixed)
        lr = 1.0 / (it + 1.0)
        
        y_old = y.copy()
        y = y + (t_k - 1.0) / t_k * (y - y_old) - lr / t_k * grad
        a_new = project_simplex(y)
        
        # Evaluate
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adapt temperature
        if no_improve > 50:
            T *= 1.02
        if no_improve > 200:
            T *= 0.8  # Refocus
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            y = a.copy()
            T = T_init
            momentum = 0.0
            no_improve = 0
        
        if verbose and it % 1000 == 0:
            print(f"    game it={it}, obj={best_obj:.10f}, T={T:.6f}")
    
    return best_a, best_obj

def solve_lp_minmax(fixed, n):
    """
    Solve: min t s.t. (x * fixed)[k] <= t for all k, sum x = 1, x >= 0
    where * denotes linear convolution.
    """
    # Variables: x_0, ..., x_{n-1}, t (total n+1)
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: (x * fixed)[k] - t <= 0 for k = 0, ..., 2n-2
    # (x * fixed)[k] = sum_j x_j * fixed[k-j] (valid range)
    # So: sum_j x_j * fixed[k-j] - t <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = fixed[idx]
        A_ub[k, n] = -1.0  # coefficient of t
    
    # Equality constraint: sum x_j = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: x_j >= 0, t >= 0
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return np.maximum(res.x[:n], 0.0)
    except Exception:
        pass
    
    return None
