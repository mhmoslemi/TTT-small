Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax-parameterized L-BFGS on LSE-smoothed objective (A* via FFT)
2. Multi-length exploration with diverse structured initializations
3. Exact subgradient method with line search for final polish
4. Population-based restarts to escape local minima
"""

import numpy as np
from scipy.optimize import minimize as scipy_minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(t):
    """Numerically stable softmax."""
    t_shifted = t - t.max()
    e = np.exp(t_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(t)) / len(t)
    return e / s


def compute_grad_wrt_t(t, beta):
    """Compute gradient of LSE-smoothed objective w.r.t. t (softmax param)."""
    n = len(t)
    a = softmax(t)
    c = fast_convolve(a)
    
    # LSE weights
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Gradient w.r.t. a: g[j] = (rev(w)*a)[2n-2-j] + (w*a)[j]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    conv_wa = np.fft.irfft(W * A, fft_size)
    conv_wra = np.fft.irfft(np.fft.rfft(w[::-1], fft_size) * A, fft_size)
    
    g1 = conv_wra[2*n-2 - (n-1) : 2*n-2 + 1][::-1]
    # g1[j] = conv_wra[2n-2-j] for j=0..n-1
    # So indices in conv_wra: 2n-2, 2n-3, ..., n-1
    # In reverse order: conv_wra[n-1 : 2n-1] reversed
    g1 = conv_wra[n-1 : 2*n-1][::-1]
    g2 = conv_wa[0 : n]
    g = g1 + g2
    
    # Gradient w.r.t. t
    ag = np.dot(a, g)
    grad_t = a * (g - ag)
    
    return grad_t


def lse_value(t, beta):
    """Compute LSE-smoothed objective value."""
    a = softmax(t)
    c = fast_convolve(a)
    bc = beta * c
    bc_max = bc.max()
    return (bc_max + np.log(np.exp(bc - bc_max).sum())) / beta


def true_objective(a):
    """Compute the actual objective: max(conv(a,a)) (assuming sum(a)=1)."""
    c = fast_convolve(a)
    return c.max()


def init_uniform(n):
    return np.zeros(n)  # t=0 gives uniform softmax


def init_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - (n-1)/2.0
    sigma = sigma_frac * n
    t = -0.5 * (x / sigma) ** 2
    return t - t.max()


def init_cosine(n, freq=1.0):
    x = np.arange(n)
    t = -np.cos(2 * np.pi * freq * x / n)
    return t - t.max()

    t = np.log(a) - np.log(a).max()

def interpolate_t(t_old, n_new):
    """Interpolate from old softmax params to new length."""
    a_old = softmax(t_old)
    x_old = np.linspace(0, 1, len(a_old))
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return init_from_sequence(a_new)


def lbfgs_run(t, beta, maxiter=300, gtol=1e-8):
    """Run L-BFGS-B for a single beta value."""
    def fun(tt):
        a = softmax(tt)
        c = fast_convolve(a)
        bc = beta * c
        bc_max = bc.max()
        return (bc_max + np.log(np.exp(bc - bc_max).sum())) / beta
    
    def jac(tt):
        return compute_grad_wrt_t(tt, beta)
    
    result = scipy_minimize(fun, t, jac=jac, method='L-BFGS-B',

    return result.x, result.fun, result.nit, result.nfev


def subgradient_polish(a, time_limit, n_peaks=1, lr_init=0.01):

            w = np.zeros(2*n-1)

    sig = (n - 1) / 4.0
    t = -0.5 * ((x - c) / sig) ** 2 - 0.5 * ((x + c) / sig) ** 2
    return t - t.max()


def init_plateau(n, width_frac=0.6):
    x = np.arange(n) - (n - 1) / 2.0
    w = width_frac * n / 2.0
    t = np.maximum(0.0, 1.0 - np.abs(x) / w)
    return t - t.max()


def t_from_seq(a):
    """Convert probability vector to softmax parameters."""
    a = np.maximum(a, 1e-30)
    t = np.log(a) - np.log(a).max()
    return t


def interp_t(t_old, n_new):
    """Interpolate softmax parameters to new length."""
    a = softmax_fn(t_old)
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return t_from_seq(a_new)


def subgrad_polish(a, t_max, n_peaks=1, lr0=0.02):
    """Exact subgradient method with line search targeting top peaks."""
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = true_obj(a)
    t0 = time.monotonic()
    lr = lr0
    stagn = 0
    fft_sz = 1 << (3 * n - 2 - 1).bit_length()

    while time.monotonic() - t0 < t_max:
        c = fast_convolve(a)
        if n_peaks == 1:
            ks = int(c.argmax())
            w = np.zeros(2 * n - 1)
            w[ks] = 1.0
        else:
            pk = np.argpartition(c, -n_peaks)[-n_peaks:]
            w = np.zeros(2 * n - 1)
            pv = c[pk]
            w[pk] = pv
            wsum = w.sum()
            if wsum > 0:
                w /= wsum
            else:
                w[c.argmax()] = 1.0

        A = np.fft.rfft(a, fft_sz)
        W = np.fft.rfft(w, fft_sz)
        Wr = np.fft.rfft(w[::-1], fft_sz)
        g1 = np.fft.irfft(Wr * A, fft_sz)[n - 1:2 * n - 1][::-1]
        g2 = np.fft.irfft(W * A, fft_sz)[0:n]
        g = g1 + g2

        found = False
        for mult in [1.0, 0.5, 0.25, 2.0, 0.1, 4.0]:
            alpha = lr * mult
            a_new = a - alpha * g
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new /= s
            v = true_obj(a_new)
            if v < best_obj - 1e-14:
                a = a_new
                best_obj = v
                best_a = a.copy()
                lr = alpha
                stagn = 0
                found = True
                break

        if not found:
            stagn += 1
            if stagn % 15 == 0:
                lr *= 0.93
            if stagn > 400:
                break

    return best_a, best_obj


def coordinate_polish(a, t_max):
    """Cycle through coordinates, locally optimizing each."""
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = true_obj(a)
    t0 = time.monotonic()

    while time.monotonic() - t0 < t_max:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > t_max:
                break
            # Save current
            saved = a[j]
            # Try different values
            candidates = [saved * 0.5, saved * 0.75, saved * 1.0, saved * 1.25, saved * 1.5,
                          max(0, saved - 0.001), max(0, saved + 0.001), 0.0]
            best_j_val = saved
            best_j_obj = best_obj
            for cv in candidates:
                a[j] = cv
                s = a.sum()
                if s < 1e-15:
                    a[j] = saved
                    continue
                a_norm = a / s
                v = true_obj(a_norm)
                if v < best_j_obj - 1e-12:
                    best_j_obj = v
                    best_j_val = cv
            a[j] = best_j_val
            a /= a.sum()
            if best_j_obj < best_obj - 1e-14:
                best_obj = best_j_obj
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main multi-phase optimization:
    Phase 1: Multi-n L-BFGS exploration with diverse inits
    Phase 2: Intensive L-BFGS refinement at best n (high beta annealing)
    Phase 3: Subgradient + coordinate polish
    Phase 4: Nearby-n exploration and final squeeze
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        try_seq(parent, best_tracker)

    n_parent = len(parent)
    t_parent = t_from_seq(parent)

    # === Phase 1: Multi-n L-BFGS Exploration ===
    print("\n=== Phase 1: Multi-n L-BFGS Exploration ===")
    t1_end = time.monotonic() + 350

    n_list = sorted(set([n_parent, 256, 384, 512, 640, 720, 768, 832, 896, 1024, 1152, 1280]))
    n_list = [n for n in n_list if 256 <= n <= 1400]

    init_fns = [
        ('parent', lambda n: interp_t(t_parent, n)),
        ('gauss04', lambda n: init_gaussian(n, 0.4)),
        ('gauss06', lambda n: init_gaussian(n, 0.6)),
        ('gauss08', lambda n: init_gaussian(n, 0.8)),
        ('parab', lambda n: init_parabolic(n)),
        ('bipolar', lambda n: init_bimodal(n, 0.3)),
        ('plateau', lambda n: init_plateau(n, 0.6)),
        ('uniform', lambda n: init_uniform(n)),
    ]

    beta_sched_1 = [30, 150, 800, 4000]

    for nc in n_list:
        t_rem = t1_end - time.monotonic()
        if t_rem < 40:
            break
        # Try a subset of inits for each n
        inits_to_try = init_fns[:4]
        for iname, ifn in inits_to_try:
            t_rem = t1_end - time.monotonic()
            if t_rem < 20:
                break
            t = ifn(nc)
            for beta in beta_sched_1:
                t_rem = t1_end - time.monotonic()
                if t_rem < 8:
                    break
                maxit = min(200, max(30, int(t_rem * 15)))
                t, val, nit = lbfgs_step(t, nc, beta, maxiter=maxit)
            a = softmax_fn(t)
            try_seq(a, best_tracker)
            if best_tracker[1] is not None and len(best_tracker[1]) == nc:
                t_parent = t.copy()

    # === Phase 2: Intensive Refinement ===
    print(f"\n=== Phase 2: Intensive Refinement (best so far: {best_tracker[0]:.8f}) ===")
    t2_end = time.monotonic() + 350

    if best_tracker[1] is not None:
        a_cur = best_tracker[1].copy()
    else:
        a_cur = parent.copy()
    n_cur = len(a_cur)
    t_cur = t_from_seq(a_cur)

    # High beta annealing with restarts
    beta_sched_2 = [100, 500, 2000, 8000, 30000, 100000, 300000]
    for beta in beta_sched_2:
        t_rem = t2_end - time.monotonic()
        if t_rem < 10:
            break
        maxit = min(400, max(50, int(t_rem * 8)))
        t_cur, val, nit = lbfgs_step(t_cur, n_cur, beta, maxiter=maxit)
        a_cur = softmax_fn(t_cur)
        try_seq(a_cur, best_tracker)

        # Occasional restart with noise
        if t_rem > 30 and nit < maxit * 0.3:
            t_noise = t_cur + rng.standard_normal(n_cur) * 0.05
            t_cur, val2, _ = lbfgs_step(t_noise, n_cur, beta, maxiter=100)
            a_noise = softmax_fn(t_cur)
            try_seq(a_noise, best_tracker)
            if best_tracker[1] is not None and len(best_tracker[1]) == n_cur:
                t_cur = t_from_seq(best_tracker[1])

    # === Phase 3: Subgradient + Coordinate Polish ===
    print(f"\n=== Phase 3: Subgradient + Coordinate Polish ===")
    t3_end = time.monotonic() + 200

    if best_tracker[1] is not None:
        a_cur = best_tracker[1].copy()

    t_alloc = min(100, t3_end - time.monotonic())
    if t_alloc > 5:
        print(f"  Subgradient (1 peak), n={len(a_cur)}, t={t_alloc:.0f}s")
        a_pol, _ = subgrad_polish(a_cur, time_limit=t_alloc, n_peaks=1, lr0=0.03)
        try_seq(a_pol, best_tracker)

    if best_tracker[1] is not None:
        a_cur = best_tracker[1].copy()

    t_alloc = min(60, t3_end - time.monotonic())
    if t_alloc > 5:
        print(f"  Subgradient (5 peaks), n={len(a_cur)}, t={t_alloc:.0f}s")
        a_pol, _ = subgrad_polish(a_cur, time_limit=t_alloc, n_peaks=5, lr0=0.01)
        try_seq(a_pol, best_tracker)

    if best_tracker[1] is not None:
        a_cur = best_tracker[1].copy()

    t_alloc = min(40, t3_end - time.monotonic())
    if t_alloc > 5:
        print(f"  Coordinate polish, n={len(a_cur)}, t={t_alloc:.0f}s")
        a_pol, _ = coordinate_polish(a_cur, time_limit=t_alloc)
        try_seq(a_pol, best_tracker)

    # === Phase 4: Nearby-n and Final Squeeze ===
    print(f"\n=== Phase 4: Nearby-n & Final Squeeze ===")

    if best_tracker[1] is not None:
        a_best = best_tracker[1]
        n_best = len(a_best)
        t_best = t_from_seq(a_best)

        # Try nearby n values
        offsets = [64, -64, 128, -128, 32, -32, 192, -192]
        for dn in offsets:
            n_try = n_best + dn
            if n_try < 256 or n_try > 1500:
                continue
            t_rem = deadline - time.monotonic() - 10
            if t_rem < 30:
                break
            t_try = interp_t(t_best, n_try)
            for beta in [200, 2000, 20000]:
                t_rem = deadline - time.monotonic() - 5
                if t_rem < 10:
                    break
                maxit = min(300, max(50, int(t_rem * 8)))
                t_try, val, nit = lbfgs_step(t_try, n_try, beta, maxiter=maxit)
            a_try = softmax_fn(t_try)
            try_seq(a_try, best_tracker)

        # Multiple restarts at best n
        n_restart = len(best_tracker[1]) if best_tracker[1] is not None else n_best
        t_base = t_from_seq(best_tracker[1]) if best_tracker[1] is not None else t_best

        for restart in range(20):
            t_rem = deadline - time.monotonic() - 10
            if t_rem < 20:
                break
            scale = 0.02 + 0.01 * restart
            t_try = t_base + rng.standard_normal(n_restart) * scale
            for beta in [500, 5000]:
                t_rem = deadline - time.monotonic() - 5
                if t_rem < 10:
                    break
                maxit = min(200, max(30, int(t_rem * 8)))
                t_try, val, nit = lbfgs_step(t_try, n_restart, beta, maxiter=maxit)
            a_try = softmax_fn(t_try)
            try_seq(a_try, best_tracker)

            # Subgradient polish on the result
            t_rem = deadline - time.monotonic() - 5
            if t_rem > 5:
                a_pol, _ = subgrad_polish(a_try, time_limit=min(10, t_rem), n_peaks=1, lr0=0.01)
                try_seq(a_pol, best_tracker)

    # Final L-BFGS at very high beta
    if best_tracker[1] is not None:
        t_rem = deadline - time.monotonic() - 5
        if t_rem > 10:
            a_fin = best_tracker[1].copy()
            n_fin = len(a_fin)
            t_fin = t_from_seq(a_fin)
            for beta in [500000, 2000000]:
                t_rem = deadline - time.monotonic() - 3
                if t_rem < 5:
                    break
                maxit = min(200, max(20, int(t_rem * 5)))
                t_fin, val, nit = lbfgs_step(t_fin, n_fin, beta, maxiter=maxit)
                a_fin = softmax_fn(t_fin)
                try_seq(a_fin, best_tracker)

            # Final subgradient
            t_rem = deadline - time.monotonic() - 3
            if t_rem > 3 and best_tracker[1] is not None:
                a_pol, _ = subgrad_polish(best_tracker[1].copy(), time_limit=t_rem,
                                          n_peaks=1, lr0=0.005)
                try_seq(a_pol, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using fallback")
    fallback = np.ones(512) / 512
    return [float(x) for x in fallback]
