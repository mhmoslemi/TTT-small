"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-Frank-Wolfe with
adaptive constraint cycling, Nesterov acceleration, multi-scale n-sweep,
structured initializations (difference-set inspired, Gaussian taper, flat
spectrum), pairwise coordinate descent, and adaptive subgradient polishing.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def solve_lp_direction(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    if active_k is None or len(active_k) >= 2 * n - 1:
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def golden_section_line(f, g, n, max_iter=30):
    """Golden section search for best alpha mixing f and g."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    f_c = project_simplex(f + c * (g - f))
    f_d = project_simplex(f + d * (g - f))
    fc = fast_convolve(f_c).max()
    fd = fast_convolve(f_d).max()
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            f_c = project_simplex(f + c * (g - f))
            fc = fast_convolve(f_c).max()
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            f_d = project_simplex(f + d * (g - f))
            fd = fast_convolve(f_d).max()
    return (lo + hi) / 2.0


def lpfw_optimize(f0, time_limit, verbose=False, use_palindrome=False,
                  top_k=35, nesterov_mom=0.0):
    """LP-based Frank-Wolfe with adaptive constraints and optional Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_fac = 1.0
    momentum = np.zeros(n)
    y = f.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptive constraint selection with cycling
        active_k = None
        if it % 100 == 0:

        elif it > 3:

            k_sorted = np.argsort(conv)[::-1][:top_k]
            active_k = k_sorted.tolist()

        g_raw = solve_lp_direction(f, M * m_fac, n, active_k=active_k)
        if g_raw is None:
            m_fac *= 1.15
            if m_fac > 8.0:
                m_fac = 1.0
            continue

        g_sum = g_raw.sum()
        if g_sum < 1e-15:
            break
        g = g_raw / g_sum

            g = project_simplex(f + momentum)

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 80:

            m_fac = min(m_fac * 1.05, 5.0)
            continue

        alpha = golden_section_line(f, g, n)

        if alpha > 1e-9:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            if use_palindrome:
                f_new = project_simplex(make_palindromic(f_new, n))

            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            conv = conv_new

            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                M = M_new
                m_fac = 1.0
            else:
                no_improve += 1
            f = f_new
        else:
            no_improve += 1
            m_fac = min(m_fac * 1.1, 6.0)

            conv = fast_convolve(f)
            M = conv.max()
            m_fac = 1.0
            no_improve = 0
            momentum = np.zeros(n)

    return best_f, best_obj


def coord_desc_fine(f0, time_limit, max_rounds=500):
    """Coordinate descent with multi-scale local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    no_improve_passes = 0

    multipliers = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 1.8, 2.2, 3.0]

    while time.monotonic() - t0 < time_limit and passes < max_rounds:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-13:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.93, top_k=40, temp=5e-5):

    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]

        scaled -= scaled.max()

            w[np.argmax(pv)] = 1.0
        else:

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9999
        cur_temp *= 1.005

        if no_improve > 500:
            rng = np.random.default_rng(it_count * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def gaussian_init(n, sigma_frac=0.4):
    """Gaussian-shaped initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-15:
        a /= ss
    return a


def tapered_uniform_init(n, taper_frac=0.3):
    """Uniform with raised-cosine taper at edges."""
    a = np.ones(n)
    m = max(1, int(taper_frac * n))
    for i in range(m):
        w = 0.5 * (1 - np.cos(np.pi * i / m))
        a[i] = w
        a[n - 1 - i] = w
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-15:
        a /= ss
    return a


def difference_set_init(n, p=None):
    """Initialize based on quadratic residues mod p (if p divides or is near n)."""
    if p is None:
        # Find a prime near n
        p = n
        while p > 2:
            if all(p % i for i in range(2, int(np.sqrt(p)) + 1)):
                break
            p -= 1
        if p < 5:
            p = n
            while p > 2:
                if all(p % i for i in range(2, int(np.sqrt(p)) + 1)):
                    break
                p += 1
    # Quadratic residues mod p
    residues = set()
    for x in range(p):
        residues.add((x * x) % p)
    residues.discard(0)
    # Map to sequence
    a = np.zeros(n)
    for r in residues:
        if r < n:
            a[r] = 1.0
    # Also add complementary for balance
    for r in residues:
        val = (n - 1 - r) % n
        if val < n:
            a[val] = max(a[val], 0.5)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-15:
        a /= ss
    else:
        a = np.ones(n) / n
    return a


def pairwise_cd(f0, time_limit, max_rounds=200):
    """Pairwise coordinate descent: optimize pairs of coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Precompute which lags are affected by each pair
    # For pair (i,j), the affected convolution indices are those k where
    # k-i is in [0,n-1] or k-j is in [0,n-1]
    # i.e., k in [i, i+n-1] union [j, j+n-1]

    rng = np.random.default_rng(42)

    round_num = 0
    while time.monotonic() - t0 < time_limit and round_num < max_rounds:
        round_num += 1
        improved = False
        # Sample random pairs (biased toward nearby indices for locality)
        n_pairs = min(n, 50)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            # j is near i with some probability, random otherwise
            if rng.random() < 0.5:
                j = (i + rng.integers(1, max(2, n // 10))) % n
            else:
                j = rng.integers(0, n)
            if i == j:
                continue

            old_i = f[i]
            old_j = f[j]
            if old_i < 1e-13 and old_j < 1e-13:
                continue

            best_obj_pair = best_obj
            best_ii = old_i
            best_jj = old_j

            # Try various ratios of mass redistribution
            for ratio in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0, 1.5, 2.0, 3.0]:
                # Move mass from i to j or vice versa
                for direction in [1, -1]:
                    delta = direction * ratio * min(old_i, old_j) * 0.5
                    new_i = old_i + delta
                    new_j = old_j - delta
                    if new_i < 0 or new_j < 0:
                        continue
                    new_sum = 1.0  # sum is preserved
                    f_trial = f.copy()
                    f_trial[i] = new_i
                    f_trial[j] = new_j
                    # Renormalize (should be ~1 already)
                    ss = f_trial.sum()
                    if ss < 1e-10:
                        continue
                    f_trial /= ss
                    val = fast_convolve(f_trial).max()
                    if val < best_obj_pair - 1e-15:
                        best_obj_pair = val
                        best_ii = new_i
                        best_jj = new_j

            if best_ii != old_i or best_jj != old_j:
                f[i] = best_ii
                f[j] = best_jj
                f = f / f.sum()
                if best_obj_pair < best_obj - 1e-14:
                    best_obj = best_obj_pair
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP-Frank-Wolfe, n-sweep, diverse
    initializations, pairwise CD, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-10:
            return float('inf')
        a_norm = a / ss
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = tapered_uniform_init(768)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default init: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(350, rem * 0.22)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = lpfw_optimize(parent, time_limit=t1, verbose=True, top_k=40)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: N-sweep with interpolation ===
    print(f"\n=== Phase 2: N-Sweep ===")
    base = best_seq if best_seq is not None else parent
    n_candidates = [480, 576, 640, 720, 768, 832, 896, 960, 1024, 1200, 1400]
    # Sort by proximity to current best n
    n_candidates.sort(key=lambda x: abs(x - len(base)))

    for target_n in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        if target_n == len(base):
            continue
        f_init = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.08, 80)
        if t_alloc < 20:
            break
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(f_init, time_limit=t_alloc, top_k=35)
        try_seq(a_opt, f"nweep_{target_n}")

    # === Phase 3: Palindromic optimization ===
    print(f"\n=== Phase 3: Palindromic ===")
    if best_seq is not None:
        n_best = len(best_seq)
        half_n = (n_best + 1) // 2
        pal_half = (best_seq[:half_n] + best_seq[n_best - 1: n_best - half_n - 1: -1]) / 2.0
        pal_full = project_simplex(make_palindromic(pal_half, n_best))
        try_seq(pal_full, "pal_init")

        rem = deadline - time.monotonic()
        if rem > 50:
            t2 = min(rem * 0.12, 150)
            print(f"  t={t2:.0f}s")
            a_opt, _ = lpfw_optimize(pal_full, time_limit=t2, use_palindrome=True, top_k=40)
            try_seq(a_opt, "pal_lpfw")

    # === Phase 4: Diverse initializations ===
    print(f"\n=== Phase 4: Diverse Initializations ===")
    n_best = len(best_seq) if best_seq is not None else 768

    inits = []
    inits.append(("gauss", gaussian_init(n_best, 0.35)))
    inits.append(("tapered", tapered_uniform_init(n_best, 0.25)))
    inits.append(("flat_spec", flat_spectrum_init(n_best, seed=123)))
    inits.append(("diffset", difference_set_init(n_best)))
    inits.append(("blend_best", project_simplex(0.7 * best_seq + 0.3 * tapered_uniform_init(n_best, 0.2))))
    inits.append(("noise", project_simplex(best_seq + rng.standard_normal(n_best) * 0.02)))

    for name, f_init in inits:
        rem = deadline - time.monotonic()
        if rem < 35:
            break
        t_alloc = min(rem * 0.06, 60)
        if t_alloc < 15:
            break
        print(f"  [{name}] t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(f_init, time_limit=t_alloc, top_k=35)
        try_seq(a_opt, f"div_{name}")

    # === Phase 5: Pairwise CD ===
    print(f"\n=== Phase 5: Pairwise CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.12, 120)
            print(f"  t={t_cd:.0f}s")
            a_opt, _ = pairwise_cd(best_seq, time_limit=t_cd)
            try_seq(a_opt, "pairwise_cd")

            # Follow up with LP-FW on the result
            rem = deadline - time.monotonic()
            if rem > 25:
                t_lpfw = min(rem * 0.08, 80)
                a_opt2, _ = lpfw_optimize(a_opt, time_limit=t_lpfw, top_k=35)
                try_seq(a_opt2, "pairwise_cd+lpfw")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.15, 180)
            print(f"  t={t_cd:.0f}s")
            a_opt, _ = coord_desc_fine(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

            rem = deadline - time.monotonic()
            if rem > 25:
                t_cd2 = min(rem * 0.1, 100)
                a_opt2, _ = coord_desc_fine(a_opt, time_limit=t_cd2)
                try_seq(a_opt2, "cd2")

    # === Phase 7: Subgradient polish ===
    print(f"\n=== Phase 7: SG Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        n_best = len(best_seq)
        if rem > 20:
            t_sg = min(rem * 0.12, 120)
            print(f"  t={t_sg:.0f}s (top_k={max(25, n_best//5)})")
            a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                                 lr=0.001, mom=0.94,
                                 top_k=max(25, n_best // 5),
                                 temp=4e-5)
            try_seq(a_opt, "sg")

            rem = deadline - time.monotonic()
            if rem > 15:
                t_sg2 = min(rem * 0.08, 60)
                a_opt2, _ = sg_polish(a_opt, time_limit=t_sg2,
                                      lr=0.0005, mom=0.95,
                                      top_k=max(15, n_best // 8),
                                      temp=1e-5)
                try_seq(a_opt2, "sg_fine")

    # === Phase 8: Final LP-FW with full constraints + Nesterov ===
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        n_best = len(best_seq)
        if rem > 20:
            t_final = min(rem * 0.2, 120)
            print(f"  Full-constraint LP-FW, t={t_final:.0f}s")
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_final, top_k=2 * n_best,
                                     nesterov_mom=0.3)
            try_seq(a_opt, "final_lpfw_nesterov")

            # Perturbation restarts
            rem = deadline - time.monotonic()
            for i in range(5):
                rem = deadline - time.monotonic()
                if rem < 12:
                    break
                noise = rng.standard_normal(n_best) * (0.0008 + 0.001 * i)
                f_pert = project_simplex(best_seq + noise)
                t_pert = min(rem * 0.15, 50)
                a_opt, _ = lpfw_optimize(f_pert, time_limit=t_pert, top_k=30)
                try_seq(a_opt, f"perturb_{i}")

            # Tiny SG on final
            rem = deadline - time.monotonic()
            if rem > 10:
                a_opt, _ = sg_polish(best_seq, time_limit=min(rem * 0.4, 40),
                                     lr=0.0004, mom=0.95,
                                     top_k=max(10, n_best // 10),
                                     temp=8e-6)
                try_seq(a_opt, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = tapered_uniform_init(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
