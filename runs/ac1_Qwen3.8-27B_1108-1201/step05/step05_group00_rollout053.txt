def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)

    # where b_flipped[m] = b[-m] (not quite right for finite sequences).
    
    # Easier: just use np.correlate or compute via FFT.
    # corr[j] = sum_k p[k] * a[k-j]
    # Let's compute: for j = 0, ..., n-1
    #   corr[j] = sum_k p[k] * a[k-j]
    #   = sum_{i=max(0,j-n+1)}^{min(n-1, ...)} ... 
    
    # Actually, the cleanest way: 
    # corr[j] = sum_i a[i] * p[i+j]  (where p[m] = 0 for m >= 2n-1)
    # This is the cross-correlation of a and p.
    # In numpy: np.correlate(p, a, mode='valid') doesn't quite work because of the size mismatch.
    
    # Let me just use FFT:
    # corr[j] = sum_i a[i] * p[i+j] for j = 0, ..., 2n-2
    # This is the cross-correlation of a (length n) and p (length 2n-1).
    # Using FFT: ifft(fft(a, N) * conj(fft(p, N))) gives corr[j] for j = 0, ..., N-1
    # where corr[j] = sum_i a[i] * p[(i+j) % N]... hmm, that's circular.
    
    # For linear cross-correlation:
    # corr[j] = sum_i a[i] * p[i+j]
    # This is equivalent to: (a * flip(p))[j] where * is convolution and flip(p)[k] = p[len(p)-1-k]
    
    # Let me just do it directly with a loop-free approach:
    # For j in [0, n-1]: corr[j] = sum_{i=0}^{n-1} a[i] * p[i+j]
    # We can compute all of these at once using a matrix, but that's O(n^2).
    
    # Better: use the fact that this is a linear operator.
    # corr = H @ a where H is a Hankel matrix with H[j,i] = p[i+j].
    # We can compute this using FFT-based convolution:
    # corr[j] = sum_i a[i] * p[i+j] = (a * p_flipped)[j + (2n-2) - (n-1)]... 
    
    # OK let me just be explicit:
    # Define p_ext of length 2n (pad p to length 2n with zeros at end if needed, but p already has length 2n-1)
    # corr[j] = sum_{i=0}^{n-1} a[i] * p[i+j] for j = 0, ..., n-1
    # 
    # Consider the convolution c = conv(a, p) where a has length n and p has length 2n-1.
    # c[k] = sum_i a[i] * p[k-i] for valid i, k-i.
    # c[k] = sum_i a[i] * p[k-i]
    # We want corr[j] = sum_i a[i] * p[i+j] = sum_i a[i] * p[i+j]
    # Let m = i+j, so i = m-j: corr[j] = sum_m a[m-j] * p[m] = conv(a, p)[j + (n-1)]... 
    # Hmm, let me just use the correlate function.
    
    # np.correlate(in1, in2, mode) computes:
    #   out[k] = sum_i in1[i+k] * conj(in2[i])
    # So np.correlate(p, a, mode='full') gives:
    #   out[k] = sum_i p[i+k] * a[i] for valid i, i+k
    # For k = 0: out[0] = sum_i p[i] * a[i] (i from 0 to min(n, 2n-1)-1 = n-1)
    # For k = j: out[j] = sum_i p[i+j] * a[i] (i from 0 to n-1, i+j from 0 to 2n-2)
    #   = sum_{i=0}^{n-1} a[i] * p[i+j] (where p[i+j] is defined only for i+j < 2n-1)
    # This is exactly corr[j]! (for j = 0, ..., 2n-2)
    
    # So corr = np.correlate(p, a, mode='full')
    # But this is O(n * (2n)) = O(n^2) for the correlate call.
    # For n = 768, that's ~1.2M multiplications per call. With FFT it'd be O(n log n).
    
    # Let's use FFT:
    # corr[j] = sum_{i=0}^{n-1} a[i] * p[i+j]
    # This is the cross-correlation. Using FFT:
    # corr = ifft(fft(p, N) * conj(fft(a, N))) won't work directly because of the definition.
    # 
    # numpy.correlate(a, b) = sum_k a[k] * conj(b[k]) for the overlap...
    # Let me just check: np.correlate([1,2,3], [4,5], mode='full')
    # = [1*4, 1*5+2*4, 2*5+3*4, 3*5] = [4, 13, 22, 15]
    # So out[k] = sum_i a[i+k] * b[i] (for valid indices)
    # 
    # We want corr[j] = sum_i a[i] * p[i+j]
    # = sum_i p[i+j] * a[i]
    # This is np.correlate(p, a, mode='full')[j] (checking: out[j] = sum_i p[i+j] * a[i]) ✓
    
    # For FFT: correlate(p, a) = conv(p_flipped, a) where p_flipped[k] = p[len(p)-1-k]
    # So: corr = conv(flip(p), a)
    # flip(p) has length 2n-1, a has length n, so conv has length 3n-2.
    # corr[j] = conv(flip(p), a)[j] for j = 0, ..., 2n-2 (the first 2n-1 values)
    
    # With FFT:
    N = 1 << ((3*n - 2) - 1).bit_length()  # next power of 2 >= 3n-2
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    # conv(flip(p), a) = ifft(fft(flip(p)) * fft(a))
    # fft(flip(p)) = conj(fft(p)) * exp(-2j*pi*k*(len(p)-1)/N)... this is getting complicated.
    
    # Simpler: just compute flip(p) and do the convolution.
    pf = p[::-1]  # flip
    conv_pa = np.fft.irfft(np.fft.rfft(pf, N) * np.fft.rfft(a, N), N)
    # conv_pa[k] = sum_i pf[i] * a[k-i] = sum_i p[len(p)-1-i] * a[k-i]
    # = sum_i p[2n-2-i] * a[k-i]
    # Let j' = 2n-2-i, so i = 2n-2-j':
    # = sum_{j'} p[j'] * a[k-(2n-2-j')] = sum_{j'} p[j'] * a[k-2n+2+j']
    # Hmm, this doesn't simplify nicely.
    
    # Let me just use the direct correlate for n up to ~1000.
    # For n=768, np.correlate(p, a, mode='full') should be fast enough (it uses C code).
    
    corr = np.correlate(p, a, mode='full')
    # corr has length n + (2n-1) - 1 = 3n-2
    # corr[j] = sum_i p[i+j] * a[i] for j = 0, ..., 2n-2 (first 2n-1 entries)
    # We need corr[0:n] for the gradient (j = 0, ..., n-1)
    
    grad = 2.0 * corr[:n]
    max_val = conv.max()
    return grad, p, max_val


def fista_optimize(a0, time_limit, T_start=0.01, T_end=0.0001, verbose=False):
    """
    FISTA (accelerated projected gradient) with annealed temperature.
    Minimizes the smoothed objective f_T(a) = T * logsumexp(conv(a,a)/T)
    which approximates max(conv(a,a)) as T -> 0.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Initialize
    T = T_start
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # FISTA parameters
    k = 1  # momentum counter
    y = a.copy()
    z = a.copy()  # previous iterate
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Anneal temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        # Compute gradient at y
        grad, p, _ = compute_smoothed_gradient(y, T)
        
        # Step size (using Lipschitz constant estimate)
        # The Lipschitz constant of the gradient is bounded by 2 * ||p||_1 * max something...
        # For safety, use a small step size with backtracking.
        # ||grad||_2 gives a rough scale; use step = 1 / (2 * max_grad) or similar
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-15:
            step = 0.1
        else:
            step = 0.5 / gnorm
        step = min(step, 0.1)  # Cap the step size
        
        # Gradient step
        candidate = y - step * grad
        
        # Project onto simplex
        a_new = project_simplex(candidate)
        
        # FISTA momentum update
        k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k * k))
        y = a_new + ((k - 1) / k_new) * (a_new - z)
        z = a_new
        a = a_new
        k = k_new
        
        # Evaluate
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            noise_scale = 0.002 + 0.005 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            z = a.copy()
            y = a.copy()
            k = 1
            no_improve = 0
            T = max(T_start * 0.5, T)  # Slightly increase T on restart
        
        if verbose and it % 1000 == 0:
            print(f"    FISTA it={it}, obj={best_obj:.10f}, T={T:.6f}, k={k:.1f}")
    
    return best_a, best_obj


def fw_fast_optimize(a0, time_limit, T_init=0.005, verbose=False):
    """
    Fast Frank-Wolfe with smoothed subgradient oracle.
    Oracle: one-hot at argmin of smoothed gradient.
    Uses backtracking line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = T_init
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Compute smoothed gradient
        grad, p, cur_obj = compute_smoothed_gradient(a, T)
        
        # FW oracle: vertex at argmin of gradient
        j_star = int(np.argmin(grad))
        oracle = np.zeros(n)
        oracle[j_star] = 1.0
        
        # Line search: find best alpha in [0, 1]
        direction = oracle - a
        best_alpha = 2.0 / (it + 2)  # Default FW step size
        best_val = cur_obj
        
        # Try a few values including the default
        candidates = [best_alpha, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0]
        for alpha in candidates:
            if alpha == 0 or alpha > 1:
                continue
            a_trial = a + alpha * direction
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            val = fast_convolve(a_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Apply update
        if best_alpha > 1e-10:
            a = a + best_alpha * direction
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        
        # Track best
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt temperature
        if no_improve > 50:
            T *= 0.95  # Decrease T to sharpen the gradient
            if T < 0.0001:
                T = 0.0001
        if no_improve > 200:
            # Restart
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, T={T:.6f}")
    
    return best_a, best_obj


def adaptive_fw_optimize(a0, time_limit, verbose=False):
    """
    Adaptive FW that combines:
    - Smoothed gradient direction (softmax-weighted peaks)
    - Simplex projection step (not just vertex, but projected)
    - Adaptive step size with backtracking
    - Periodic temperature changes
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    T = 0.005
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = 0.05
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Compute smoothed gradient
        grad, p, cur_obj = compute_smoothed_gradient(a, T)
        
        # Projected gradient step with backtracking
        improved = False
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < cur_obj - 1e-15:
                a = candidate
                cur_obj = val
                improved = True
                break
        
        if not improved:
            # Try FW step as fallback
            j_star = int(np.argmin(grad))
            for alpha in [1.0/(it+2), 0.01, 0.05, 0.1]:
                a_trial = (1-alpha)*a + alpha*np.eye(n)[j_star]
                val = fast_convolve(a_trial).max()
                if val < cur_obj - 1e-15:
                    a = a_trial
                    cur_obj = val
                    improved = True
                    break
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        if no_improve > 30:
            lr *= 0.95
            T *= 0.97
        if no_improve > 150:
            T = max(0.0001, T * 0.5)
        if no_improve > 400:
            rng = np.random.default_rng((it * 48271) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * (0.002 + 0.003*min(1,(no_improve-400)/200)))
            lr = 0.05
            T = 0.005
            no_improve = 0
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=False):
    """
    Coordinate-level polish: for the top peaks, try to reduce them by
    adjusting the contributing coordinates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Find coordinates that contribute to this peak
        # conv[k_peak] = sum_i a[i] * a[k_peak - i]
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        
        if js > je:
            break
        
        # Try to reduce the peak by redistributing mass among contributing coords
        # Strategy: for the pair (i, k_peak-i) with the largest product,
        # try to reduce one and increase the other (or a non-contributing coord)
        
        improved = False
        for _ in range(10):
            rng = np.random.default_rng((it * 12345 + k_peak + _) % (2**31))
            # Small perturbation focused on the peak region
            pert = rng.standard_normal(n) * 0.001
            # Focus on the affected region
            for j in range(n):
                if j < js or j > je:
                    pert[j] *= 0.1  # Reduce perturbation outside the peak region
            
            candidate = project_simplex(a + pert)
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-15:
                a = candidate
                best_obj = val
                best_a = a.copy()
                improved = True
                break
        
        if not improved and it % 100 == 0:
            # Broader perturbation
            rng = np.random.default_rng((it * 67890) % (2**31))
            candidate = project_simplex(a + rng.standard_normal(n) * 0.002)
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-15:
                a = candidate
                best_obj = val
                best_a = a.copy()
            elif it > 500:
                break
    
    return best_a, best_obj


def multi_scale_optimize(a0, time_limit, verbose=False):
    """
    Multi-scale optimization:
    1. Coarsen the sequence (average pairs of adjacent elements)
    2. Optimize the coarse sequence
    3. Interpolate back and refine
    Repeat with finer scales.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    # Find a good coarsening factor
    # Coarsen by factor 4, optimize, then interpolate and refine
    factor = 4
    n_coarse = n // factor
    if n_coarse < 16:
        return a, fast_convolve(a).max()
    
    # Coarsen
    a_coarse = a[:n_coarse * factor].reshape(n_coarse, factor).mean(axis=1)
    a_coarse = a_coarse / a_coarse.sum()
    
    # Optimize coarse (briefly)
    t_coarse = time_limit * 0.3
    a_coarse_opt, _ = adaptive_fw_optimize(a_coarse, time_limit=t_coarse)
    
    # Interpolate back to fine scale
    a_fine = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_coarse), a_coarse_opt)
    a_fine = np.maximum(a_fine, 0)
    a_fine = a_fine / a_fine.sum()
    
    # Refine fine scale
    t_fine = time_limit * 0.7
    a_opt, obj = adaptive_fw_optimize(a_fine, time_limit=t_fine)
    
    return a_opt, obj

    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
