def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interp_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

import numpy as np

import time


def fast_conv(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_conv_grad(a, w):
    """Compute sum_k w[k] * d(conv[k])/da[j] efficiently via FFT.
    w has length 2n-1 (same as conv). Returns gradient of length n.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 1).bit_length()
    # grad[j] = sum_k w[k] * a[k-j] = (a_rev * w)[n-1+j]
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    return 2.0 * conv_result[n - 1: 2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_to_length(f, new_n):
    """Interpolate sequence to new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, f[0])
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def lse_objective(a, tau):
    """Log-sum-exp of auto-convolution (smooth surrogate for max)."""
    conv = fast_conv(a)
    conv_max = conv.max()
    shifted = conv - conv_max
    return conv_max + tau * np.log(np.sum(np.exp(shifted / tau)))


def lse_gradient(a, tau):
    """Gradient of log-sum-exp of auto-convolution."""
    n = len(a)
    conv = fast_conv(a)
    conv_max = conv.max()
    shifted = (conv - conv_max) / tau
    w = np.exp(shifted)
    w /= w.sum()
    return fast_conv_grad(a, w)


def fista_lse(a0, time_limit, tau_start=0.5, tau_end=0.001, verbose=False):
    """FISTA (accelerated projected gradient) on log-sum-exp objective with tau annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_val = fast_conv(a).max()
    t0 = time.monotonic()

    tau = tau_start
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:

        # FISTA parameters

        step = 1.0 / max(L, 1e-6)
        step = min(step, 0.5)

        y = a.copy()

            # Gradient step
            grad = lse_gradient(y, tau)
            z = y - step * grad
            # Project to simplex
            x_new = project_simplex(z)
            # FISTA update

            y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)

            if sy > 1e-15:

            t_k = t_new

        lse_val = lse_objective(a, tau)

            best_a = a.copy()
            no_improve = 0

def multiplicative_weights(a0, time_limit, verbose=False):

    it = 0
    peak_frac = 0.02

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv(a)
        cur_peak = conv.max()

        # Find active peaks (within peak_frac of max)
        threshold = cur_peak * (1.0 - peak_frac)
        mask = conv >= threshold
        active_k = np.where(mask)[0]

        if len(active_k) == 0:
            active_k = np.array([np.argmax(conv)])

        w_k = np.ones(len(active_k)) / len(active_k)
        sg = np.zeros(n)
        for i in range(len(active_k)):
            k = int(active_k[i])
            js = max(0, n - 1 - k)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                sg[j] += 2.0 * w_k[i] * a[k - j]

        # Multiplicative update with clipping
        log_update = -eta * sg
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new /= s_new

        new_conv = fast_conv(a_new)
        new_peak = new_conv.max()

        if new_peak < best_val - 1e-13:
            best_val = new_peak
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and it % 200 == 0:
                print(f"    MW iter {it}: peak={best_val:.10f}, eta={eta:.5f}")
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                a = a_new  # Move anyway to explore

        # Adaptive eta
        if new_peak < cur_peak:
            eta = min(eta * 1.005, 0.3)
        else:
            eta *= 0.98

        # Widen peak window as we converge
        if it > 100:
            peak_frac = min(peak_frac + 0.0001, 0.05)

        if no_improve > 800:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.001 * min(no_improve // 200, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            eta = 0.05
            no_improve = 0

    return best_a, best_val


def active_set_newton(a0, time_limit, verbose=False):
    """Newton-style optimization on the active set of peak positions.
    At optimum, all active peaks should be equal. We optimize to equalize them.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_conv(a)
    best_val = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv(a)
        cur_peak = conv.max()
        cur_min = conv.min()

        # Find active peaks (within 1% of max)
        threshold = cur_peak * 0.99
        active_mask = conv >= threshold
        active_k = np.where(active_mask)[0]

        if len(active_k) < 2:
            active_k = np.array([np.argmax(conv)])
            active_mask = np.zeros(len(conv), dtype=bool)
            active_mask[active_k] = True

        # Compute the "excess" at each active peak
        active_vals = conv[active_k]
        mean_val = active_vals.mean()

        # For each coordinate j, compute how much it contributes to each active peak
        # and adjust to reduce the spread
        n_active = len(active_k)
        contributions = np.zeros((n, n_active))
        for i in range(n_active):
            k = int(active_k[i])
            js = max(0, n - 1 - k)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                contributions[j, i] = a[k - j]

        # Direction: reduce contributions to peaks that are above mean,
        # increase contributions to peaks that are below mean
        peak_weights = active_vals - mean_val  # Positive for above-mean peaks
        direction = contributions.T @ peak_weights  # Length n
        # direction[j] > 0 means j contributes more to above-mean peaks -> should decrease
        # direction[j] < 0 means j contributes more to below-mean peaks -> can increase

        # Normalize direction
        d_norm = np.linalg.norm(direction)
        if d_norm < 1e-15:
            no_improve += 1
        else:
            direction /= d_norm

            # Line search
            best_alpha = 0.0
            best_trial = best_val
            # Try decreasing j where direction > 0, and vice versa
            # Use: a_new = a - alpha * direction, projected
            for alpha in [0.001, 0.002, 0.005, 0.01, 0.02]:
                a_trial = a - alpha * direction
                a_trial = project_simplex(a_trial)
                val = fast_conv(a_trial).max()
                if val < best_trial:
                    best_trial = val
                    best_alpha = alpha

            if best_alpha > 0:
                a = a - best_alpha * direction
                a = project_simplex(a)
                if best_trial < best_val - 1e-13:
                    best_val = best_trial
                    best_a = a.copy()
                    no_improve = 0
                    if verbose and it % 50 == 0:
                        print(f"    Newton iter {it}: peak={best_val:.10f}")
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 100:
            break

    return best_a, best_val


def coordinate_polish(a0, time_limit, verbose=False):
    """Fine coordinate descent: try small multiplicative changes to each coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_conv(a)
    best_val = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    order = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        np.random.default_rng(int(time.monotonic() * 1000)).shuffle(order)
        improved = False

        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(idx)
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_val
            # Try multiplicative changes
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                # Adjust others to maintain sum=1
                remainder = 1.0 - new_val
                others_sum = 1.0 - old_val
                if others_sum < 1e-10:
                    continue
                scale = remainder / others_sum
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-10:
                    continue
                a_trial /= st

                val = fast_conv(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                remainder = 1.0 - best_j_val
                others_sum = 1.0 - old_val
                if others_sum > 1e-10:
                    scale = remainder / others_sum
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    st = a.sum()
                    if st > 1e-10:
                        a /= st
                    if best_j_obj < best_val - 1e-14:
                        best_val = best_j_obj
                        best_a = a.copy()
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_a, best_val

            print(f"  n={len(best_seq)}, t={t4:.0f}s")
            a_opt, _ = active_set_newton(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "newton")

    # === Phase 5: Coordinate polish ===
    print(f"\n=== Phase 5: Coordinate Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t5 = min(200, rem * 0.3)
        if t5 > 15:
            print(f"  n={len(best_seq)}, t={t5:.0f}s")
            a_opt, _ = coordinate_polish(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "cd_polish")

    # === Phase 6: Final combined refinement ===
    print(f"\n=== Phase 6: Final Refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Quick FISTA with very small tau
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t6 = min(rem, 100)
            a_opt, _ = fista_lse(best_seq, time_limit=t6, tau_start=0.05, tau_end=0.001)
            try_seq(a_opt, "fista_fine")

        # Quick MW
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = multiplicative_weights(best_seq, time_limit=min(rem, 80))
            try_seq(a_opt, "mw_fine")

        # Coordinate polish again
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = coordinate_polish(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "cd_final")

        # Try a few random perturbations followed by quick optimization
        for p in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 12:
                break
            noise = rng.standard_normal(len(best_seq)) * (0.0005 + 0.0003 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = multiplicative_weights(f_pert, time_limit=min(rem, 30))
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
