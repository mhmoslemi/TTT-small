def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_active_lp(f, M, active_indices, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0."""
    n_act = len(active_indices)
    if n_act == 0:
        return np.ones(n) / n

    A_sub = np.zeros((n_act, n))
    for i, k in enumerate(active_indices):
        js = np.arange(max(0, k - n + 1), min(n, k + 1))
        if len(js) > 0:
            A_sub[i, js] = f[k - js]

    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sub, b_ub=np.full(n_act, M),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_active(f0, time_limit, verbose=False, active_tol=0.1, max_active=50):
    """LP-Frank-Wolfe using only active constraints for speed."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 200000:
            break

        # Identify active constraints (lags where conv is close to max)
        threshold = M * (1.0 - active_tol)
        active = np.where(conv >= threshold)[0]

        # Limit active set size for speed - keep the largest
        if len(active) > max_active:
            sorted_idx = active[np.argsort(conv[active])[::-1]]
            active = sorted_idx[:max_active]

        # Solve LP with active constraints
        g = solve_active_lp(f, M, active, n)

        if g is None:
            # Fallback: use subgradient direction
            k_star = int(np.argmax(conv))
            g = np.zeros(n)
            js = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
            g[js] = f[k_star - js]
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n
        else:
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        # Check convergence
        if np.max(np.abs(g - f)) < 1e-14:
            no_improve += 1
            if no_improve > 20:
                break
            # Try with wider active set
            active_tol = min(active_tol * 1.5, 0.5)
            continue

        # Line search (efficient: few evaluations)
        best_alpha = 0.0
        best_val = M
        fw_alpha = 2.0 / (it + 2.0)
        candidates = sorted(set([max(fw_alpha, 0.01), 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]))

        for alpha in candidates:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Binary refinement (6 steps)
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.5, best_alpha + 0.15)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            active_tol = max(active_tol * 0.98, 0.02)

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW-A iter {it}: obj={best_obj:.10f}, |active|={len(active)}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Periodic restart with noise if stuck
        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 80, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            active_tol = 0.15

    return best_f, best_obj


def kkt_polish(a0, time_limit, n_top=15, verbose=False):
    """
    KKT-based polishing: minimize max of top-k conv values by adjusting a.
    Uses scipy minimize on a reparameterized smooth objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / ss
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_f = a.copy()

    t0 = time.monotonic()

    # Identify top lags
    top_lags = np.argsort(conv)[::-1][:n_top]

    def objective(x):
        """Objective: max of selected conv values + penalty for constraint violations."""
        total = x.sum()
        if total <= 0:
            return 1e10
        a_local = x / total
        conv_local = fast_convolve(a_local)
        # Max of top lags plus a smoothed max of all
        vals = conv_local[top_lags]
        # Smooth max via log-sum-exp
        v_max = vals.max()
        smooth_max = v_max + np.log(np.sum(np.exp(vals - v_max))) / len(vals)
        # Also include the overall max (clipped)
        overall_max = conv_local.max()
        return max(smooth_max, overall_max)

    # Use L-BFGS-B with multiple random restarts
    best_x = a.copy()
    best_val = objective(a)

    n_restarts = 5
    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        if r == 0:
            x0 = a.copy()
        else:
            rng = np.random.default_rng(r * 1000 + 42)
            x0 = project_simplex(a + rng.standard_normal(n) * 0.003)

        try:
            res = minimize(objective, x0, method='L-BFGS-B',
                           options={'maxiter': 200, 'ftol': 1e-14})
            a_res = res.x / res.x.sum() if res.x.sum() > 0 else a
            a_res = np.maximum(a_res, 0)
            a_res = a_res / a_res.sum()
            val = objective(a_res)
            if val < best_val:
                best_val = val
                best_x = a_res.copy()
        except Exception:
            pass

    conv_best = fast_convolve(best_x)
    obj = conv_best.max()
    if obj < best_obj:
        best_obj = obj
        best_f = best_x.copy()

    return best_f, best_obj


def nesterov_subgrad(f0, time_limit, lr0=0.005, mom0=0.85, top_k=30, temp0=0.0001):
    """Nesterov-accelerated projected subgradient with multi-peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * w[i] * f[k - indices]

        v = cur_mom * v - cur_lr * sg
        f_new = project_simplex(y + v)
        y = f_new + cur_mom * (f_new - f)
        y = project_simplex(y)
        f = f_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.005
        cur_mom = min(0.97, cur_mom * 1.0004)

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.5, 0.001)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, n_samples=50):
    """Randomized coordinate descent with line search for final polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Pick a random coordinate
        j = rng.integers(0, n)
        old_val = f[j]
        if old_val < 1e-14:
            continue

        denom = 1.0 - old_val
        if denom < 1e-12:
            continue

        # Try a range of multipliers
        best_mult = 1.0
        best_val = fast_convolve(f).max()

        for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 1.7, 2.0]:
            new_val = old_val * mult
            if new_val < 0 or new_val > 0.9999:
                continue
            ft = f.copy()
            ft[j] = new_val
            ft2 = 1.0 - new_val
            if ft2 <= 0:
                continue
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            ft[mask] = ft[mask] * (ft2 / denom)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_mult = mult

        if best_mult != 1.0:
            new_val = old_val * best_mult
            ft = f.copy()
            ft[j] = new_val
            ft2 = 1.0 - new_val
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            ft[mask] = ft[mask] * (ft2 / denom)
            ft = ft / ft.sum()
            f = ft
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: active-set LP-FW, KKT polishing,
    Nesterov subgradient, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(640) / 640

    n_parent = len(parent)

    # === Phase 1: Quick LP-FW on parent ===
    print("=== Phase 1: LP-FW on parent ===")
    t1 = min(100, max(30, (deadline - time.monotonic()) * 0.12))
    if t1 > 15:
        a_opt, _ = lp_fw_active(parent, time_limit=t1, verbose=True, active_tol=0.1, max_active=40)
        try_seq(a_opt, "lpfw_p1")

    # === Phase 2: Multiple initializations at various lengths ===
    print(f"\n=== Phase 2: Multi-init (best={best_score:.6f}) ===")
    best_base = best_seq if best_seq is not None else parent

    init_configs = []
    for n_try in [n_parent, 512, 768, 1024, 1280]:
        if n_try < 64 or n_try > 2048:
            continue
        for freq in [1, 2, 3, 5]:
            for amp in [0.15, 0.3, 0.5]:
                init_configs.append((n_try, freq, amp))
        # Random init
        init_configs.append((n_try, -1, -1))

    # Deduplicate and limit
    seen = set()
    unique_configs = []
    for cfg in init_configs:
        key = (cfg[0], cfg[1], cfg[2])
        if key not in seen:
            seen.add(key)
            unique_configs.append(cfg)

    max_inits = 12
    count = 0
    for n_try, freq, amp in unique_configs:
        if count >= max_inits:
            break
        rem = deadline - time.monotonic() - 500
        if rem < 20:
            break
        count += 1

        if freq == -1:
            a_init = project_simplex(np.abs(rng.standard_normal(n_try)) + 0.1)
        else:
            a_init = cosine_modulated_init(n_try, freq, amp)

        # Also try starting from interpolated best
        if best_seq is not None and n_try != len(best_seq):
            a_interp = interpolate_sequence(best_seq, n_try)
            # Blend
            a_try = 0.5 * a_init + 0.5 * a_interp
            a_try = project_simplex(a_try)
        else:
            a_try = a_init

        s = try_seq(a_try, f"init_n{n_try}_f{freq}")

        # Short LP-FW on promising inits
        if np.isfinite(s) and s < best_score + 0.05:
            t_alloc = min((deadline - time.monotonic() - 450) * 0.15, 50)
            if t_alloc > 10:
                a_opt, _ = lp_fw_active(a_try, time_limit=t_alloc, active_tol=0.1, max_active=30)
                try_seq(a_opt, f"lpfw_init_n{n_try}_f{freq}")

    # === Phase 3: Intensive LP-FW on best ===
    print(f"\n=== Phase 3: Intensive LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t3 = min(rem * 0.7, 400)
            print(f"  Intensive LP-FW: n={len(best_seq)}, t={t3:.0f}s")
            a_opt, _ = lp_fw_active(best_seq, time_limit=t3, verbose=True,
                                    active_tol=0.05, max_active=30)
            try_seq(a_opt, "lpfw_intensive")

            # Second pass with different params
            rem = deadline - time.monotonic() - 200
            if rem > 20 and best_seq is not None:
                t3b = min(rem * 0.5, 150)
                a_opt, _ = lp_fw_active(a_opt, time_limit=t3b, verbose=False,
                                        active_tol=0.03, max_active=20)
                try_seq(a_opt, "lpfw_intensive2")

    # === Phase 4: KKT-based polish ===
    print(f"\n=== Phase 4: KKT Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t4 = min(rem * 0.6, 120)
            print(f"  KKT polish: n={len(best_seq)}, t={t4:.0f}s")
            a_opt, _ = kkt_polish(best_seq, time_limit=t4, n_top=12)
            try_seq(a_opt, "kkt_polish")

    # === Phase 5: Nesterov subgradient ===
    print(f"\n=== Phase 5: Nesterov (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t5 = min(rem * 0.5, 120)
            print(f"  Nesterov: n={n_best}, t={t5:.0f}s")
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t5,
                                         lr0=0.006, mom0=0.87,
                                         top_k=max(20, n_best // 8),
                                         temp0=0.00005)
            try_seq(a_opt, "nesterov_1")

            # Fine Nesterov
            rem = deadline - time.monotonic() - 80
            if rem > 15:
                a_opt, _ = nesterov_subgrad(a_opt, time_limit=min(rem, 60),
                                             lr0=0.002, mom0=0.92,
                                             top_k=max(10, n_best // 12),
                                             temp0=0.00001)
                try_seq(a_opt, "nesterov_fine")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: CD Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t6 = min(rem * 0.7, 80)
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t6)
            try_seq(a_opt, "cd_polish")

    # === Phase 7: Multi-length interpolation ===
    print(f"\n=== Phase 7: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [len(best_seq) * 2 // 3, len(best_seq) * 3 // 2, len(best_seq) * 2]:
            if target_n < 128 or target_n > 3000:
                continue
            rem = deadline - time.monotonic() - 50
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.4, 60)
            if t_alloc > 8:
                a_opt, _ = lp_fw_active(f_interp, time_limit=t_alloc,
                                        active_tol=0.08, max_active=25)
                try_seq(a_opt, f"interp_n{target_n}")

    # === Phase 8: Final micro-polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            # Very short Nesterov
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=min(rem * 0.5, 30),
                                         lr0=0.001, mom0=0.93,
                                         top_k=max(8, len(best_seq) // 20),
                                         temp0=0.000005)
            try_seq(a_opt, "final_nest")

            rem = deadline - time.monotonic() - 10
            if rem > 8:
                a_opt, _ = lp_fw_active(best_seq, time_limit=min(rem, 20),
                                        active_tol=0.02, max_active=15)
                try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
