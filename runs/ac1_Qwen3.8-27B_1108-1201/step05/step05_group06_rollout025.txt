import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.ones(new_n) / new_n
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def compute_subgradient(a, conv, n_peaks=50, delta_frac=0.01):
    """
    Compute subgradient using all peaks within delta_frac of max.
    Returns (subgradient, n_active_peaks).
    """
    n = len(a)
    cmax = conv.max()
    threshold = cmax * (1 - delta_frac)
    active = np.where(conv >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    
    # Weight by how close to max (exponential weighting)
    weights = np.exp((conv[active] - cmax) / (cmax * 0.01 + 1e-15))
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for i in range(len(active)):
        k = int(active[i])
        w = weights[i]
        if w < 1e-15:
            continue
        # (P_k a) has (P_k a)[j] = a[k-j] if valid
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je+1] += 2.0 * w * a[k - np.arange(js, je+1)]
    
    return sg, len(active)

def line_search(a, g, conv, n, t_max):
    """
    Find optimal step size t in [0, t_max] that minimizes max(conv(a + t*g)).
    Uses adaptive candidate evaluation + refinement.
    """
    def obj(t):
        a_t = a + t * g
        if np.any(a_t < -1e-15):
            return np.inf
        a_t = np.maximum(a_t, 0)
        c_t = fast_convolve(a_t)
        return c_t.max()
    
    # Adaptive candidates
    candidates = [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.7 * t_max, 0.95 * t_max]
    candidates = [t for t in candidates if 0 <= t <= t_max]
    if not candidates:
        candidates = [t_max * 0.5]
    
    best_t = 0
    best_val = obj(0)
    
    for t in candidates:
        v = obj(t)
        if v < best_val:
            best_val = v
            best_t = t
    
    # Refine with golden section around best_t
    # Determine bracket
    if best_t == 0:
        lo, hi = 0, max(0.001, t_max * 0.1)
    elif best_t >= t_max * 0.9:
        lo, hi = max(0, t_max * 0.5), t_max
    else:
        span = min(best_t, t_max - best_t) * 0.5 + 0.001
        lo = max(0, best_t - span)
        hi = min(t_max, best_t + span)
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    a_gs = lo + (1 - gr) * (hi - lo)
    b_gs = lo + gr * (hi - lo)
    fa = obj(a_gs)
    fb = obj(b_gs)
    
    for _ in range(20):
        if fa < fb:
            hi = b_gs
            b_gs = a_gs
            fb = fa
            a_gs = lo + (1 - gr) * (hi - lo)
            fa = obj(a_gs)
        else:
            lo = a_gs
            a_gs = b_gs
            fa = fb
            b_gs = lo + gr * (hi - lo)
            fb = obj(b_gs)
    
    best_t = (lo + hi) / 2
    best_val = obj(best_t)
    
    return best_t, best_val

def coordinate_bisection(a0, time_limit, verbose=False, max_passes=200):

    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        time_left = time_limit - (time.monotonic() - t0)
        time_per_coord = time_left / (n + 1)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            # Build normalized sequence without j
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            # b_valid[k] = b[k-j] if valid
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            if start < end:
                b_valid[start:end] = b[:end - start]
            
            # Find optimal x in [0, 1]
            lo = 0.0
            hi = 1.0
            x_new = x_old
            
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                
                # Derivative at the max point
                k_max = int(np.argmax(f_vals))
                b_kj = b[k_max - j] if 0 <= k_max - j < n else 0.0
                B_k = B[k_max]
                ind = 1.0 if (2 * j < conv_len and k_max == 2 * j) else 0.0
                
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                
                if deriv > 1e-13:
                    hi = mid
                elif deriv < -1e-13:
                    lo = mid
                else:
                    break
            
            x_new = (lo + hi) / 2.0
            x_new = max(0.0, min(1.0, x_new))
            
            if abs(x_new - x_old) < 1e-14:
                continue
            
            # Apply
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            
            conv = fast_convolve(a)
            obj = conv.max()
            
            if obj < best_obj - 1e-13:

                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:

                rng = np.random.default_rng(passes * 47 + 11)

            else:

    return best_a, best_obj

def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

def structured_perturb(a, rng):
    """Apply a structured perturbation."""
    n = len(a)
    a_new = a.copy()
    op = rng.integers(0, 4)
    
    if op == 0:
        # Chunk rotation: rotate a segment
        length = rng.integers(max(4, n // 10), max(5, n // 3))
        start = rng.integers(0, n - length)
        shift = rng.integers(1, length)
        seg = a_new[start:start+length].copy()
        a_new[start:start+length] = np.roll(seg, shift)
        
    elif op == 1:
        # Local Gaussian reshaping
        center = rng.integers(0, n)
        sigma = rng.uniform(n * 0.02, n * 0.1)
        x = np.arange(n)
        bump = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        bump -= bump.mean()  # Center to preserve sum
        a_new = a_new + 0.02 * bump
        
    elif op == 2:
        # Edge taper modification
        edge_len = max(4, n // 8)
        alpha = rng.uniform(0.8, 1.5)
        a_new[:edge_len] *= alpha
        a_new[-edge_len:] *= alpha
