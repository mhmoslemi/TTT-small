Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based LSE gradient descent
with adaptive beta, momentum, direct subgradient polishing, and multi-length
exploration. Key improvements: vectorized FFT gradient, Nesterov momentum,
progressive beta schedule, exact subgradient at convergence, and systematic
length exploration.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(a, w):
    """
    Compute gradient of LSE objective using FFT.
    w: weights of length 2n-1 (LSE soft-max weights over convolution positions)
    Returns gradient of shape (n,).
    
    grad[m] = 2 * sum_j a_j * w[j+m] (cross-correlation)
    = 2 * conv(a, w_rev)[2n-2-m]
    """
    n = len(a)
    w_len = len(w)  # should be 2n-1
    w_rev = w[::-1]
    
    # Compute conv(a, w_rev) via FFT
    conv_len = n + w_len - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:conv_len]
    
    # Extract: grad[m] = 2 * conv[2n-2-m] for m = 0, ..., n-1
    indices = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv[indices]
    return grad

with adaptive beta, momentum, direct subgradient polishing, and multi-length

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(a, w):
    """
    Compute gradient of LSE objective w.r.t. a using FFT.
    w: soft-max weights of length 2n-1.
    Returns gradient of shape (n,).
    
    Formula: grad[m] = 2 * sum_j a_j * w[j+m] = 2 * conv(a, w_rev)[2n-2-m]
    """
    n = len(a)
    w_len = len(w)
    w_rev = w[::-1]
    conv_len = n + w_len - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:conv_len]
    indices = 2 * n - 2 - np.arange(n)

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"Initial best: {best_tracker[0]:.10f} (n={n_parent})")

    # === Phase 1: Multi-length exploration (350s) ===
    print("\n=== Phase 1: Multi-length exploration ===")
    t1_budget = min(350.0, (deadline - time.monotonic()) * 0.22)
    t1_end = time.monotonic() + t1_budget

    lengths = sorted(set([
        n_parent,
        max(512, n_parent - 128),
        min(2048, n_parent + 128),
        max(448, n_parent // 2 + n_parent // 4),
        min(1536, n_parent * 2 // 3),
        640, 896, 1024
    ]))
    lengths = [l for l in lengths if 384 <= l <= 2048]

    n_try_phase1 = min(5, len(lengths))
    for i in range(n_try_phase1):
        t_remaining = t1_end - time.monotonic()
        if t_remaining < 25:
            break
        nc = lengths[i]
        t_alloc = min(70, t_remaining * 0.85)
        if t_alloc < 20:
            break

        if nc == n_parent:
            f_init = parent.copy()
        else:
            f_init = interpolate_to_length(parent, nc)

        print(f"  LSE n={nc}, t={t_alloc:.0f}s")
        f_opt, _ = lse_optimize(f_init, time_limit=t_alloc,
                                  beta_start=30.0, beta_end=8000.0,
                                  momentum_init=0.90, lr_scale=4.0)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Deep progressive LSE (900s) ===
    print(f"\n=== Phase 2: Deep LSE optimization ===")
    t2_budget = min(900.0, (deadline - time.monotonic()) * 0.5)
    t2_end = time.monotonic() + t2_budget

    current = best_tracker[1].copy()

    # Round 1: moderate beta
    print(f"  Round 1: n={len(current)}, beta 50->5000")
    t_alloc = min(250, t2_end - time.monotonic())

        f_opt, _ = lse_optimize(current, time_limit=t_alloc,

    current = best_tracker[1].copy()

    # Round 3: very high beta
    current = best_tracker[1].copy()
    print(f"  Round 3: n={len(current)}, beta 2000->80000")

    if t_alloc > 30:
        f_opt, _ = lse_optimize(current, time_limit=t_alloc,

    current = best_tracker[1].copy()

    print(f"\n=== Phase 3: Subgradient polishing ===")

    if best_tracker[1] is not None and time.monotonic() < deadline - 40:

        for n_try in [n_best + 64, n_best - 64, n_best + 128]:
            if n_try < 384 or n_try > 2500:

            t_remaining = deadline - time.monotonic() - 10

            f_init = interpolate_to_length(best_tracker[1], n_try)

    if best_tracker[1] is not None and time.monotonic() < deadline - 15:

                                          n_peaks=4, lr_init=0.0008, momentum_init=0.95)
            try_sequence(f_opt, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using fallback uniform")
    return [float(x) for x in parent]
