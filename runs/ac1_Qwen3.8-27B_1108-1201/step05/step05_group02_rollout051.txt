"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using improved multi-phase optimization:
1. RCC-FW (Retrospective Convex Combination Frank-Wolfe) with diverse LP directions

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def solve_lp_min_peak(f, n, M_init=None):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def solve_lp_max_sum(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def get_lp_directions(f, n, M, verbose=False):
    """Get multiple LP directions for diversity."""
    directions = []
    
    # Direction 1: min-peak LP
    g, t = solve_lp_min_peak(f, n)
    if g is not None:
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
            directions.append(('minpeak', g))
    
    # Direction 2: max-sum LP at current M
    g2 = solve_lp_max_sum(f, M, n)
    if g2 is not None:
        gs2 = g2.sum()
        if gs2 > 1e-15:
            g2 = g2 / gs2
            directions.append(('maxsum_M', g2))
    
    # Direction 3: max-sum LP at slightly higher M
    g3 = solve_lp_max_sum(f, M * 1.1, n)
    if g3 is not None:
        gs3 = g3.sum()
        if gs3 > 1e-15:
            g3 = g3 / gs3
            directions.append(('maxsum_1.1M', g3))
    
    return directions

def rcc_fw_optimize(f0, time_limit, verbose=False):
    """RCC-FW optimization with diverse LP directions and better line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    f_curr = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    pull_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        
        # Periodically pull current toward best
        if no_improve > 20 and no_improve % 10 == 0:
            alpha = 0.1 + 0.1 * min((no_improve // 20), 5)
            f_curr = (1 - alpha) * best_f + alpha * f_curr
            f_curr = project_simplex(f_curr)
        
        # Get LP directions
        directions = get_lp_directions(f_curr, n, M)
        
        if not directions:
            directions = [('uniform', np.ones(n) / n)]
        
        # Pick the best direction via line search
        best_alpha = 0.0
        best_val = M
        best_g = None
        
        for name, g in directions:
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs < 1e-15:
                continue
            g = g / gs
            
            # Quick check: does this direction improve?
            # Try a few alphas
            for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
                ft = (1 - alpha) * f_curr + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
                    best_g = g
        
        if best_g is None or best_alpha < 1e-10:
            no_improve += 1
            if no_improve > 50:
                # Restart with perturbation

                scale = 0.003 + 0.002 * min(no_improve // 30, 5)
                f_curr = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f_curr)
                M = conv.max()
                no_improve = 0
            continue
        
        # Fine line search
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        for _ in range(15):
            mid = (lo + hi) / 2.0
            ft = (1 - mid) * f_curr + mid * best_g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                break
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid
        
        # Update
        f_new = (1 - best_alpha) * f_curr + best_alpha * best_g
        f_new = np.maximum(f_new, 0.0)

        if sn < 1e-15:

        else:

        if M < best_obj - 1e-14:

            best_f = f_curr.copy()
            no_improve = 0
            if verbose and it % 10 == 0:

def sg_optimize(f0, time_limit, lr=0.003, momentum=0.93, top_k=40, temp=0.00008,
                verbose=False):
    """Subgradient method with softmax-weighted peaks and averaging."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    # Nesterov averaging
    avg_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]
        
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        conv = fast_convolve(f_new)
        co = conv.max()
        
        # Nesterov averaging
        alpha_avg = 2.0 / (it + 2.0)
        avg_f = (1 - alpha_avg) * avg_f + alpha_avg * f_new
        
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new
        
        # Also check the averaged solution
        avg_obj = fast_convolve(avg_f).max()
        if avg_obj < best_obj - 1e-15:
            best_obj = avg_obj
            best_f = avg_f.copy()
            no_improve = 0
        
        cur_lr *= 0.9999
        cur_temp *= 1.003
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            avg_f = f.copy()
            no_improve = 0
    
    return best_f, best_obj

def pairwise_cd(f0, time_limit, verbose=False):
    """Pairwise coordinate descent: adjust pairs of coordinates jointly."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    passes = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        # Find top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        
        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            # Get contributing coordinates
            contrib = list(range(js, je + 1))
            if len(contrib) < 2:
                continue
            
            # Try pairwise adjustments
            # Pick pairs that are "far apart" in the convolution to have different effects
            n_pairs = min(10, len(contrib) * (len(contrib) - 1) // 2)
            tried = set()
            for _ in range(n_pairs):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(0, len(contrib)))
                j = int(rng.integers(0, len(contrib)))
                if i == j or (i, j) in tried:
                    continue
                tried.add((i, j))
                ci, cj = contrib[i], contrib[j]
                
                # Try various mixing ratios
                for ratio in [0.3, 0.5, 0.7, 1.3, 1.5]:
                    old_i, old_j = f[ci], f[cj]
                    if old_i < 1e-12 and old_j < 1e-12:
                        continue
                    
                    # Move mass from one to the other

def fast_convolve(a):

    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()

    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)

    if ss > 1e-15:

        kg = k - idx
        mask = (kg >= 0) & (kg < n)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using improved multi-phase optimization:
1. Structured Frank-Wolfe with 4 diverse directions (full LP, restricted LP, subgradient, max-sum LP)
2. Multi-length exploration with interpolation
3. Nesterov-averaged subgradient descent
4. Pairwise coordinate descent targeting active peaks
5. Final polish with short FW + SG

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """LP: min t s.t. (f*g)[k] <= t for all k, g>=0, sum(g)=1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_con = 2 * n - 1
    A_ub = np.zeros((n_con, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_con)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def solve_lp_max_sum_topk(f, M, n, top_k):
    """LP: max sum(g) s.t. (f*g)[k] <= M for top_k peaks, g>=0."""
    conv = fast_convolve(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    n_con = len(peaks)
    c = -np.ones(n)
    A_ub = np.zeros((n_con, n))
    for idx, k in enumerate(peaks):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[idx, j] = f[k - j]
    b_ub = np.full(n_con, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def compute_subgradient(f, conv, top_k=30, temp=0.0001):
    """Compute softmax-weighted subgradient of the max convolution."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        kg = k - idx
        mask = (kg >= 0) & (kg < n)
        sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]
    return sg
