import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2*n - 1).bit_length()  # next power of 2
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max(conv, temperature):
    """Log-sum-exp approximation of max."""
    m = conv.max()
    return m + temperature * np.log(np.sum(np.exp((conv - m) / temperature)))

def compute_subgradient(a, conv, top_k=50, temperature=0.01):
    """Compute subgradient of smoothed max of convolution."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / temperature
    w = np.exp(scaled)
    w /= w.sum()
    
    # Build the weighted peak indicator L
    L = np.zeros(len(conv))
    for i in range(k_actual):
        L[peaks[i]] = w[i]
    
    # Compute subgradient via FFT
    a_flip = a[::-1]  # reversed
    size = 1 << (len(conv) + n - 1).bit_length()
    L_fft = np.fft.rfft(L, size)
    af_fft = np.fft.rfft(a_flip, size)
    corr = np.fft.irfft(L_fft * af_fft, size)
    
    # subgrad[m] = corr[n-1+m] for m=0..n-1
    subgrad = 2.0 * corr[n-1 : 2*n-1]
    return subgrad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth projected subgradient
with FFT-based gradients, LP-Frank-Wolfe, and multi-length exploration.
Key insight: asymmetric sequences beat 2.0; the peaked autocorrelation is
reduced by iteratively chasing convolution peaks via smoothed max optimization.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, conv, top_k, temperature):
    """Compute gradient of log-sum-exp smoothed max of convolution via FFT."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / temperature
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    # Build weighted peak indicator L of length 2n-1
    L = np.zeros(2 * n - 1)
    for i in range(k_actual):
        L[peaks[i]] = w[i]

    # subgrad[m] = 2 * sum_k L[k] * a[k-m] (valid indices)
    # = 2 * (L * a_flip)[n-1+m]
    a_flip = a[::-1]
    size = 1 << (2 * n - 1 + n - 1).bit_length()
    L_fft = np.fft.rfft(L, size)
    af_fft = np.fft.rfft(a_flip, size)
    corr = np.fft.irfft(L_fft * af_fft, size)
    subgrad = 2.0 * corr[n - 1:2 * n - 1]
    return subgrad

        if g_sum < 1e-15:

            m_relax = 1.0

def interp_seq(f, new_n):
    """Linear interpolation of a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    out = np.maximum(out, 0.0)
    ss = out.sum()
    if ss > 1e-15:
        out /= ss
    else:
        out = np.ones(new_n) / new_n
    return out


def cosine_init(n, freq, amp):
    """Cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def exp_bump_init(n, center, width):
    """Exponential bump initialization."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((j - center) / width) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        blend = 0.5 * best_seq + 0.5 * a_init
        blend = project_simplex(blend)
        t6 = min(60, rem * 0.08)
        if t6 > 10:
            a_opt, _ = smooth_subgrad_opt(blend, time_limit=t6,
                                           lr0=0.05, mom0=0.8,
                                           top_k=80, temp0=0.08,
                                           temp_decay=1.003, lr_decay=0.998)
            s = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_seq = a_opt.copy()
                print(f"  NEW BEST [init{idx}]: {s:.6f} (n={len(a_opt)})")

    # === Phase 6: Final LP-FW multi-start ===
    print(f"\n=== Phase 6: LP-FW multi-start (best={best_str(best_score)}) ===")
    n_best = len(best_seq)
    for p in range(8):
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        t7 = min(80, rem * 0.09)
        if t7 < 10:
            break

        if p == 0:
            f_init = best_seq.copy()
        elif p == 1:
            noise = rng.standard_normal(n_best) * 0.004
            f_init = project_simplex(best_seq + noise)
        elif p == 2:
            f_init = 0.9 * best_seq + 0.1 * cosine_init(n_best, 1, 0.3)
            f_init = project_simplex(f_init)
        elif p == 3:
            f_init = 0.85 * best_seq + 0.15 * best_seq[::-1]
            f_init = project_simplex(f_init)
        elif p == 4:
            noise = rng.standard_normal(n_best) * 0.01
            f_init = project_simplex(best_seq + noise)
        elif p == 5:
            f_init = 0.8 * best_seq + 0.2 * exp_bump_init(n_best, n_best * 0.3, n_best * 0.1)
            f_init = project_simplex(f_init)
        elif p == 6:
            f_init = 0.95 * best_seq + 0.05 * np.ones(n_best) / n_best
            f_init = project_simplex(f_init)
        else:
            noise = rng.standard_normal(n_best) * 0.002
            f_init = project_simplex(best_seq + noise)

        a_opt, _ = lp_fw_opt(f_init, time_limit=t7)
        s = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a_opt.copy()
            print(f"  NEW BEST [lpfw_ms{p}]: {s:.6f} (n={len(a_opt)})")

    # === Phase 7: Ultra-fine subgradient polish ===
    print(f"\n=== Phase 7: Final polish (best={best_str(best_score)}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(100, rem * 0.4)
        a_opt, _ = smooth_subgrad_opt(best_seq, time_limit=t8,
                                       lr0=0.01, mom0=0.9,
                                       top_k=300, temp0=0.02,
                                       temp_decay=1.005, lr_decay=0.999)
        s = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a_opt.copy()
            print(f"  NEW BEST [final_subg]: {s:.6f} (n={len(a_opt)})")

    # One more LP-FW pass
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(60, rem * 0.4)
        a_opt, _ = lp_fw_opt(best_seq, time_limit=t9)
        s = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a_opt.copy()
            print(f"  NEW BEST [final_lpfw]: {s:.6f} (n={len(a_opt)})")

    # Tiny perturbations
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    for _ in range(6):
        rem = deadline - time.monotonic()
        if rem < 12:
            break
        noise = rng.standard_normal(n_best) * 0.001
        f_pert = project_simplex(best_seq + noise)
        t10 = min(15, rem * 0.5)
        a_opt, _ = smooth_subgrad_opt(f_pert, time_limit=t10,
                                       lr0=0.005, mom0=0.92,
                                       top_k=200, temp0=0.01,
                                       temp_decay=1.005, lr_decay=0.999)
        s = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a_opt.copy()
            print(f"  NEW BEST [perturb]: {s:.6f} (n={len(a_opt)})")

    # Final check
    final_s = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_s):
        best_score = final_s

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]


def best_str(x):
    """Format score for display."""
    return f"{x:.6f}"
