import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_subgradient(a, conv, beta):
    """Compute softmax-weighted subgradient of max(conv(a,a)) using all peaks."""
    n = len(a)
    scaled = beta * (conv - conv.max())
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # All weight on the max
        peaks = np.where(conv == conv.max())[0]
        w = np.zeros(len(conv))
        w[peaks] = 1.0 / len(peaks)
    else:
        w /= w_sum

    sg = np.zeros(n)
    for k in range(len(conv)):

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[k] * a[k - idx]
    return sg

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_subgradient(f, conv, beta):
    """Softmax-weighted subgradient of max(conv(f,f))."""
    n = len(f)
    b = beta * (conv - conv.max())
    w = np.exp(b)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for k in range(len(conv)):
        wk = w[k]
        if wk < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * wk * f[k - idx]
    return sg


def subgradient_optimize(f0, time_limit, beta=200.0, lr=0.005,
                         momentum=0.9, verbose=False, label=""):
    """Soft subgradient with momentum and adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f /= s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_beta = beta

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_soft_subgradient(f, conv, cur_beta)
        v = momentum * v + (1.0 - momentum) * sg

        f_trial = f - cur_lr * v
        f_trial = project_simplex(f_trial)
        f_trial = np.maximum(f_trial, 0.0)
        f_trial /= f_trial.sum()

        conv_trial = fast_convolve(f_trial)
        obj = conv_trial.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_trial.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                cur_lr *= 0.8
                if no_improve > 50:
                    f = best_f.copy()
                    v *= 0.5
                    cur_lr = lr * 0.7
            f = f  # keep old
        f = f_trial if no_improve <= 20 else f

        if it % 50 == 0:
            cur_beta = min(cur_beta * 1.08, 5000.0)
        if it % 200 == 0:
            cur_lr *= 0.97

        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            v = np.zeros(n)
            cur_lr = lr * 0.6
            cur_beta = beta * 0.5
            conv = fast_convolve(f)
            no_improve = 0

        conv = fast_convolve(f)

        if verbose and it % 1000 == 0:
            print(f"    SG {it} t={time.monotonic()-t0:.0f}s obj={best_obj:.10f}")

    return best_f, best_obj


def update_conv(f, conv, j, old_val, new_val, n):
    """Incrementally update convolution after changing f[j]."""
    diff = new_val - old_val
    if abs(diff) < 1e-18:
        return conv
    # For k in [j, j+n-1]: conv[k] += 2*diff*f[k-j]
    # But for k=2j: conv[2j] += diff*(f[2j-j] + f[2j-j]) = 2*diff*f[j]
    # Wait, let me be more careful.
    # conv[k] = sum_i f[i]*f[k-i]
    # Terms involving f[j]:
    #   i=j: f[j]*f[k-j], valid if 0<=k-j<n, i.e., j<=k<j+n
    #   k-i=j => i=k-j: f[k-j]*f[j], valid if 0<=k-j<n, same range
    # If j != k-j (k!=2j): change = diff*f[k-j] + f[k-j]*diff = 2*diff*f[k-j]
    # If k=2j: change = diff*f[j] + f[j]*diff ... wait no.
    #   Old: f[old_j]*f[old_j] = old_val^2
    #   New: f[new_j]*f[new_j] = new_val^2
    #   Change: new_val^2 - old_val^2 = diff*(old_val + new_val)
    
    # So for k in [j, min(j+n-1, 2n-2)]:
    if j <= 2*n - 2:
        ks = np.arange(j, min(j + n, 2*n))
        # For k != 2j: conv[k] += 2*diff*f[k-j]
        # For k == 2j: conv[2j] += diff*(old_val + new_val)
        
        k2j = 2 * j
        mask = ks != k2j
        if mask.any():
            conv[ks[mask]] += 2.0 * diff * f[ks[mask] - j]
        if 0 <= k2j <= 2*n - 2:
            conv[k2j] += diff * (old_val + new_val)
    
    # Ensure non-negative (numerical)
    np.maximum(conv, 0, out=conv)
    return conv

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft subgradient + coordinate descent with scipy 1D minimization
+ randomized restarts + multi-length exploration.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]
