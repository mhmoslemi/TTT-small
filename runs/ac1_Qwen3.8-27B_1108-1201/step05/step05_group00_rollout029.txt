"""
Coordinate descent with exact 1D subproblem solves for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses multiple initializations, 
simulated annealing restarts, and adaptive strategies.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Find optimal value of a[i] to minimize max(conv(a,a))/(sum(a))^2,
    with all other coordinates fixed.

    S0 = a.sum() - ai
    
    if S0 < 1e-15:
        return ai  # Can't normalize properly
    
    # Precompute C, d, e arrays
    # conv[k] = C[k] + d[k]*t + e[k]*t^2 where t = a[i]

    # d[k] = 2*a[k-i] if 0 <= k-i <= n-1 and k != 2*i, else 0

    k_start = max(0, i)

        j_indices = indices - i  # = k - i, ranges from 0 to n-1

"""
Coordinate descent with exact 1D subproblem minimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses golden section search on each coordinate
with proper normalization accounting, multiple diverse initializations,
and simulated-annealing-style perturbations to escape local minima.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective_raw(a):
    """Compute max(conv(a,a)) / (sum(a))^2. The full objective is 2n * this."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def optimize_coordinate(a, i):
    """
    Find optimal value of a[i] to minimize max(conv(a,a))/(sum(a))^2,
    with all other coordinates fixed. Uses golden section search.
    """
    n = len(a)
    ai = a[i]
    S0 = a.sum() - ai

    if S0 < 1e-15:
        return ai

    # Precompute C, d, e arrays so that conv[k] = C[k] + d[k]*t + e[k]*t^2
    conv = fast_convolve(a)
    conv_len = 2 * n - 1

    e_arr = np.zeros(conv_len)
    d_arr = np.zeros(conv_len)

    ki = 2 * i
    if 0 <= ki < conv_len:
        e_arr[ki] = 1.0

    k_start = max(0, i)
    k_end = min(conv_len - 1, i + n - 1)

    if k_start <= k_end:
        indices = np.arange(k_start, k_end + 1)
        j_indices = indices - i
        d_arr[indices] = 2.0 * a[j_indices]
        if 0 <= ki < conv_len:
            d_arr[ki] = 0.0

    C_arr = conv - d_arr * ai - e_arr * ai * ai

    # Minimize f(t) = max_k(C_k + d_k*t + e_k*t^2) / (S0 + t)^2 for t >= 0
    def f(t):
        t = max(t, 0.0)
        S = S0 + t
        if S < 1e-15:
            return np.inf
        vals = C_arr + d_arr * t + e_arr * t * t
        return vals.max() / (S * S)

    # Coarse grid search to find a good bracket
    lo = 0.0
    hi = max(ai * 3.0, S0 / n * 3.0, 0.01)

    # Expand hi if needed
    for _ in range(15):
        if f(hi) > f(hi * 0.5):
            hi *= 2.0
        else:
            break

    # Coarse grid to find best region
    n_grid = 40
    grid = np.linspace(lo, hi, n_grid)
    f_vals = np.array([f(x) for x in grid])
    best_idx = np.argmin(f_vals)

    # Refine bracket around best grid point
    if best_idx == 0:
        lo = 0.0
        hi = grid[1]
    elif best_idx == n_grid - 1:
        lo = grid[-2]
        hi = grid[-1]
    else:
        lo = grid[max(0, best_idx - 1)]
        hi = grid[min(n_grid - 1, best_idx + 1)]

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = lo + gr * (hi - lo)
    b1 = lo + (1.0 - gr) * (hi - lo)
    fa = f(a1)
    fb = f(b1)

    for _ in range(60):
        if hi - lo < 1e-14:
            break
        if fa < fb:
            hi = b1
            b1 = a1
            fb = fa
            a1 = lo + gr * (hi - lo)
            fa = f(a1)
        else:
            lo = a1
            a1 = b1
            fa = fb
            b1 = lo + (1.0 - gr) * (hi - lo)
            fb = f(b1)

    t_opt = (lo + hi) / 2.0
    return max(t_opt, 0.0)


def coordinate_descent(a0, time_limit, verbose=False):
    """Cyclic coordinate descent with exact 1D subproblem solves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = compute_objective_raw(a)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2.0:
            break

        # Full pass
        for i in range(n):
            t_opt = optimize_coordinate(a, i)
            a[i] = t_opt

        # Normalize
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        obj = compute_objective_raw(a)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")

        if no_improve >= 4:
            break

    return best_a, best_obj

    if np.isfinite(fs) and fs < best_score[0]:
        best_score[0] = fs

    print(f"\nFINAL SCORE: {best_score[0]:.8f} (n={len(best_score[1])})")
    return [float(x) for x in best_score[1]]
