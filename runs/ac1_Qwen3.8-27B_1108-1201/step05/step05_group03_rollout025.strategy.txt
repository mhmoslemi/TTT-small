def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a list of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    if not isinstance(sequence, list):
        return np.inf
    if len(sequence) == 0:
        return np.inf
    for x in sequence:
        if not isinstance(x, (int, float)) or isinstance(x, bool):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
        # Ensure values after conversion are in bounds
        if x < 0:
            return np.inf
    # Convert to float and convert if necessary
    sequence = [float(x) for x in sequence]
    
    # Defensive clipping
    normalized = [x if x>0 else 0.0001 for x in sequence]  # Ensure at least a small value
    normalized = [min(x, 1000.0) for x in normalized]     # Cap at 1000 to prevent overflow
    
    n = len(normalized)
    total_sum = sum(normalized)
    conv_seq = np.zeros(2*n-1)
    for i in range(2*n-1):
        # Ensure indices are valid
        low_idx = max(0, i - n + 1)
        high_idx = min(i, n-1)
        if high_idx < low_idx:
            continue
        for j in range(low_idx, high_idx+1):
            if (i - j) < len(normalized) and (i - j) >=0:
                conv_seq[i] += normalized[j] * normalized[i-j]
    
    max_val = max(conv_seq)
    return (2*n*max_val)/(total_sum**2)
