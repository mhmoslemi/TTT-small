    ga = grad_a * a
    grad_x = a * (grad_a - ga.sum())

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def logsumexp(b, t):
    """Compute logsumexp(b/t) * t = smooth max."""
    b_scaled = b / t
    b_max = b_scaled.max()
    return t * (b_max + np.log(np.sum(np.exp(b_scaled - b_max))))


def soft_max_obj(x, t):
    """Objective: logsumexp(conv(softmax(x))/t) * t"""
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    a = ex / ex.sum()
    b = fast_convolve(a)
    return logsumexp(b, t)


def soft_max_grad(x, t):
    """Gradient of logsumexp(conv(softmax(x))/t) * t w.r.t. x"""
    n = len(x)
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    a = ex / ex.sum()
    b = fast_convolve(a)
    
    # Weights for logsumexp
    b_scaled = b / t
    b_max = b_scaled.max()
    w = np.exp(b_scaled - b_max)
    w /= w.sum()
    w = w / t  # This gives dw/db_k = w_k, and d(obj)/db_k = w_k
    
    # grad_a[j] = sum_k w[k] * db[k]/da[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # = 2 * cross_correlation(w, a)[j]
    # where cross_correlation(w, a)[j] = sum_m a[m] * w[j+m]
    
    # Compute using FFT
    fft_size2 = 1 << (3 * n - 2 - 1).bit_length() if n > 1 else 2
    a_pad = np.zeros(fft_size2)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size2)
    w_pad[:2*n-1] = w
    Ah = np.fft.rfft(a_pad)
    Wh = np.fft.rfft(w_pad)
    cross = np.fft.irfft(Ah * np.conj(Wh), fft_size2)
    grad_a = 2.0 * cross[:n]
    
    # Convert to grad_x via softmax Jacobian
    ga = grad_a * a
    grad_x = a * (grad_a - ga.sum())
    
    return grad_x


def logsumexp_lbfgs(a_init, time_limit, t_start=0.01, t_end=0.0005, verbose=False):
    """Optimize using logsumexp objective with L-BFGS and temperature annealing."""
    n = len(a_init)
    # Initial x for softmax parameterization
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()
    x = np.log(a_init)
    
    best_x = x.copy()
    best_obj = soft_max_obj(x, t_start)
    t0 = time.monotonic()
    
    # Annealing schedule: several rounds with decreasing temperature
    temperatures = np.geomspace(t_start, t_end, 8)
    
    for t in temperatures:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Run L-BFGS with current temperature
        result = minimize(
            soft_max_obj, x, args=(t,),
            jac=soft_max_grad, method='L-BFGS-B',
            options={'maxiter': 200, 'gtol': 1e-8, 'maxfun': 500}
        )
        
        x = result.x
        obj = soft_max_obj(x, t)
        
        # Check with the actual max objective
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()
        b = fast_convolve(a)
        actual_obj = b.max()
        
        if actual_obj < best_obj:  # Note: best_obj tracks the actual max, not logsumexp
            best_obj = actual_obj
            best_x = x.copy()
            if verbose:
                print(f"    t={t:.6f}: actual_max={actual_obj:.10f}, lbfgs_obj={result.fun:.10f}")
    
    # Return the best sequence
    x_shifted = best_x - best_x.max()
    a_best = np.exp(x_shifted)
    a_best = a_best / a_best.sum()
    return a_best

    fft_size = 1 << (size - 1).bit_length()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()

    return a_sym

def _sm_obj(x, t):

# --- Multi-peak subgradient ---

def compute_multi_peak_subgrad(a, conv, top_k=30, temp=0.001):
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += 2.0 * w[i] * a[k - idx_arr]
    
    return sg


def subgradient_optimize(a0, time_limit, lr=0.01, mom=0.9, top_k=30,
                         temp=0.001, decay=0.9998, verbose=False):
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_multi_peak_subgrad(a, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        cur_lr *= decay
        cur_temp *= 1.002
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 5)
            noise_scale = 0.002 + 0.002 * min(no_improve // 100, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0
    
    return best_a, best_obj

def coordinate_bisection(a0, time_limit, verbose=False, max_passes=1000):
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        any_change = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-14:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            lo = 0.0
            hi = 1.0
            
            for _ in range(50):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                if deriv > 1e-13:
                    hi = mid
                elif deriv < -1e-13:
                    lo = mid
                else:
                    break
            
            x_new = (lo + hi) / 2.0
            
            if abs(x_new - x_old) < 1e-13:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            
            any_change = True
            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            if passes > 2:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            else:
                break
    
    return best_a, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Logsumexp-smoothed L-BFGS with softmax parameterization and temperature annealing

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_to_length(f, new_n):
    """Linear interpolation to new length, normalized."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Symmetrize and normalize."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def _sm_obj(x, t):
    """Logsumexp objective for L-BFGS with softmax parameterization."""
    n = len(x)
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    a = ex / ex.sum()
    b = fast_convolve(a)
    b_scaled = b / t
    b_max = b_scaled.max()
    return t * (b_max + np.log(np.sum(np.exp(b_scaled - b_max))))


def _sm_grad(x, t):
    """Gradient of logsumexp objective w.r.t. log-parameterization."""
    n = len(x)
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    a = ex / ex.sum()
    b = fast_convolve(a)

    b_scaled = b / t
    b_max = b_scaled.max()
    w = np.exp(b_scaled - b_max)
    w /= w.sum()
    w = w / t

    # grad_a[j] = 2 * sum_m a[m] * w[j+m]
    if n > 1:
        need = 3 * n - 2
        fft_size2 = 1 << (need - 1).bit_length()
    else:
        fft_size2 = 2
    a_pad = np.zeros(fft_size2)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size2)
    w_pad[:2 * n - 1] = w
    Ah = np.fft.rfft(a_pad, fft_size2)
    Wh = np.fft.rfft(w_pad, fft_size2)
    cross = np.fft.irfft(Ah * np.conj(Wh), fft_size2)
    grad_a = 2.0 * cross[:n]

    ga = grad_a * a
    grad_x = a * (grad_a - ga.sum())
    return grad_x


def logsumexp_optimize(a_init, time_limit, t_start=0.005, t_end=0.0002, num_stages=10, verbose=False):
    """Optimize using logsumexp + L-BFGS with temperature annealing."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()
    x = np.log(a_init)

    best_a = a_init.copy()
    best_max_b = fast_convolve(a_init).max()
    t0 = time.monotonic()

    temperatures = np.geomspace(t_start, t_end, num_stages)

    for idx in range(num_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        t = temperatures[idx]
        maxiter = max(30, int(remaining * 80))

        try:
            result = minimize(
                _sm_obj, x, args=(t,),
                jac=_sm_grad, method='L-BFGS-B',
                options={'maxiter': maxiter, 'gtol': 1e-10, 'maxfun': maxiter * 4,
                         'ftol': 1e-12}
            )
            x = result.x
        except Exception:
            break

        x_shifted = x - x.max()
        a_new = np.exp(x_shifted)
        a_new = a_new / a_new.sum()
        max_b_new = fast_convolve(a_new).max()

        if max_b_new < best_max_b:
            best_max_b = max_b_new
            best_a = a_new.copy()

        if verbose:
            print(f"    LBF stage {idx+1}/{num_stages}, t={t:.6f}: max_b={max_b_new:.10f}, best={best_max_b:.10f}")

    return best_a


def compute_multi_peak_subgrad(a, conv, top_k=30, temp=0.001):
    """Weighted subgradient using softmax over top peaks."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    if k_actual <= 0:
        return np.zeros(n)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += 2.0 * w[i] * a[k - idx_arr]

    return sg


def subgradient_optimize(a0, time_limit, lr=0.01, mom=0.9, top_k=30,
                         temp=0.001, decay=0.9998, verbose=False):
    """Projected subgradient descent with momentum."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_multi_peak_subgrad(a, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        cur_lr *= decay
        cur_temp *= 1.003

        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 5)
            noise_scale = 0.002 + 0.002 * min(no_improve // 100, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0
            if verbose:
                print(f"    SG restart at iter {it}")

    return best_a, best_obj


def coordinate_bisection(a0, time_limit, verbose=False, max_passes=500):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            # Compute B = conv(b, b) where b = a with a[j]=0, renormalized
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            # Bisection
            lo = 0.0
            hi = 1.0
            for _ in range(45):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                if deriv > 1e-13:
                    hi = mid
                elif deriv < -1e-13:
                    lo = mid
                else:
                    break

            x_new = (lo + hi) / 2.0

            if abs(x_new - x_old) < 1e-13:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            else:
                break

    return best_a, best_obj


def block_coordinate_descent(a0, time_limit, block_size=4, verbose=False):
    """Block coordinate descent: optimize small blocks simultaneously."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    # For each block of consecutive coordinates, do a small L-BFGS optimization
    for start in range(0, n, block_size):
        if time.monotonic() - t0 > time_limit:
            break
        end = min(start + block_size, n)
        block_len = end - start

        # Get current values
        a_block = a[start:end].copy()
        if a_block.sum() < 1e-14:
            continue

        # Optimize the block keeping others fixed
        # The objective is max of quadratic functions in the block variables
        # Use a simple approach: try to reduce the max by adjusting the block

        # Compute the gradient at the current point
        a_curr = a.copy()
        b_curr = fast_convolve(a_curr)
        max_k = int(np.argmax(b_curr))

        # Subgradient at the block coordinates
        sg_block = np.zeros(block_len)
        for k in [max_k]:
            for i in range(block_len):
                j_global = start + i
                k_minus_j = k - j_global
                if 0 <= k_minus_j < n:
                    sg_block[i] += 2.0 * b_curr[k] * a_curr[k_minus_j] / b_curr.max()

        # Simple step
        lr = 0.001
        a_trial = a_curr.copy()
        a_trial[start:end] = a_block - lr * sg_block
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st > 1e-15:
            a_trial /= st
            obj = fast_convolve(a_trial).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a_trial.copy()
                a = a_trial

    return best_a, best_obj


def pair_exchange_optimize(a0, time_limit, verbose=False):
    """Optimize by exchanging mass between coordinate pairs via bisection."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        x_i = a[i]
        x_j = a[j]
        if x_i < 1e-12 or x_j < 1e-12:
            continue

        # Bisection on transfer amount: try to find optimal delta
        # a[i] -> x_i + delta, a[j] -> x_j - delta, others rescaled
        lo = -x_i + 1e-10
        hi = x_j - 1e-10

        for _ in range(35):
            mid = (lo + hi) / 2.0
            a_trial = a.copy()
            a_trial[i] = x_i + mid
            a_trial[j] = x_j - mid
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-12:
                lo = mid
                continue
            a_trial /= st
            obj = fast_convolve(a_trial).max()

            # Compare with current
            if obj < conv.max() - 1e-14:
                lo = mid
            else:
                hi = mid

        # Try the best found
        a_trial = a.copy()
        a_trial[i] = x_i + lo
        a_trial[j] = x_j - lo
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st > 1e-12:
            a_trial /= st
            obj = fast_convolve(a_trial).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a_trial.copy()
                a = a_trial
                conv = fast_convolve(a)

    return best_a, best_obj


def combined_optimize(a0, time_limit, verbose=False):
    """Alternate multiple optimization phases."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    phase = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        if phase % 6 == 0:
            # Logsumexp L-BFGS
            t_alloc = min(50, remaining)
            scale = 1.0 + 0.2 * (phase // 6)
            a_opt = logsumexp_optimize(best_a, time_limit=t_alloc,
                                       t_start=0.003 / scale, t_end=0.0001 / scale,
                                       num_stages=5, verbose=False)
            obj = fast_convolve(a_opt).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  P{phase} LBF: {best_obj:.10f}")

        elif phase % 6 == 1:
            # Coordinate bisection
            t_alloc = min(70, remaining)
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc, max_passes=300)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  P{phase} CD: {best_obj:.10f}")

        elif phase % 6 == 2:
            # Subgradient
            t_alloc = min(25, remaining)
            lr = 0.004 / (1.0 + 0.15 * (phase // 6))
            a_opt, obj = subgradient_optimize(best_a, time_limit=t_alloc,
                                              lr=lr, top_k=max(20, n // 12),
                                              temp=0.0004, decay=0.9999)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  P{phase} SG: {best_obj:.10f}")

        elif phase % 6 == 3:
            # Noise + CD
            rng = np.random.default_rng(phase * 101 + 7)
            noise_scale = 0.002 * max(0.3, 1.0 - phase * 0.02)
            a_noisy = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            t_alloc = min(35, remaining)
            a_opt, obj = coordinate_bisection(a_noisy, time_limit=t_alloc, max_passes=150)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  P{phase} N+CD: {best_obj:.10f}")

        elif phase % 6 == 4:
            # Symmetrize + L-BFGS
            a_sym = symmetrize(best_a)
            t_alloc = min(30, remaining)
            a_opt = logsumexp_optimize(a_sym, time_limit=t_alloc,
                                       t_start=0.002, t_end=0.0002,
                                       num_stages=4, verbose=False)
            obj = fast_convolve(a_opt).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  P{phase} SYM: {best_obj:.10f}")

        else:
            # Pair exchange
            t_alloc = min(20, remaining)
            a_opt, obj = pair_exchange_optimize(best_a, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  P{phase} PE: {best_obj:.10f}")

        phase += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: logsumexp L-BFGS, coordinate bisection,
    subgradient descent, pair exchange, multi-length, and multiple restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    # Storage for best result (avoids closure)
    best_a = None
    best_score = float('inf')

    # --- Get parent ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    # Helper to check and update best (inline to avoid closure)
    def _try_update(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        score = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: score={score:.6f} (n={len(a_try)})")

    # === Phase 1: L-BFGS from parent ===
    print(f"\n=== Phase 1: L-BFGS parent (n={len(parent)}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(rem, 350)
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt = logsumexp_optimize(parent, time_limit=t_alloc,
                                   t_start=0.006, t_end=0.0001,
                                   num_stages=12, verbose=True)
        _try_update(a_opt, "lbfgs_parent")

    # === Phase 2: Coordinate bisection polish ===
    print(f"\n=== Phase 2: CD polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 450
        if rem > 10:
            t_alloc = min(rem, 250)
            print(f"  Allocating {t_alloc:.0f}s")
            a_opt, _ = coordinate_bisection(best_a, time_limit=t_alloc, max_passes=600)
            _try_update(a_opt, "cd1")

    # === Phase 3: Combined multi-phase ===
    print(f"\n=== Phase 3: Combined (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t_alloc = min(rem, 250)
            print(f"  Allocating {t_alloc:.0f}s")
            a_opt, _ = combined_optimize(best_a, time_limit=t_alloc, verbose=True)
            _try_update(a_opt, "combined")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    target_lengths = [640, 896, 1024, 1280, 1536]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.35, 100)
        if t_alloc < 15:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt = logsumexp_optimize(f_interp, time_limit=t_alloc,
                                   t_start=0.004, t_end=0.0003,
                                   num_stages=5, verbose=False)
        _try_update(a_opt, f"lbfgs_n{target_n}")

        rem = deadline - time.monotonic() - 150
        if rem > 10:
            t_alloc2 = min(rem * 0.35, 50)
            if t_alloc2 > 5:
                a_opt2, _ = coordinate_bisection(a_opt, time_limit=t_alloc2, max_passes=200)
                _try_update(a_opt2, f"cd_n{target_n}")

    # === Phase 5: Structured restarts ===
    print(f"\n=== Phase 5: Restarts (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        restart_labels = ["noise2", "noise5", "sym", "flatmix", "revmix"]
        for idx in range(len(restart_labels)):
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            label = restart_labels[idx]
            if label == "noise2":
                rng_r = np.random.default_rng(seed + 100 + idx)
                a_init = project_simplex(best_a + rng_r.standard_normal(n_best) * 0.002)
            elif label == "noise5":
                rng_r = np.random.default_rng(seed + 200 + idx)
                a_init = project_simplex(best_a + rng_r.standard_normal(n_best) * 0.005)
            elif label == "sym":
                a_init = symmetrize(best_a)
            elif label == "flatmix":
                a_init = project_simplex(0.85 * best_a + 0.15 * np.ones(n_best) / n_best)
            else:
                a_init = project_simplex(0.6 * best_a + 0.4 * best_a[::-1])

            t_alloc = min(rem * 0.4, 60)
            if t_alloc < 10:
                continue
            a_opt = logsumexp_optimize(a_init, time_limit=t_alloc,
                                       t_start=0.003, t_end=0.0002,
                                       num_stages=4, verbose=False)
            _try_update(a_opt, f"r_{label}")

            rem = deadline - time.monotonic() - 80
            if rem > 5:
                t_alloc2 = min(rem, 30)
                a_opt2, _ = coordinate_bisection(a_opt, time_limit=t_alloc2, max_passes=100)
                _try_update(a_opt2, f"r_{label}_cd")

    # === Phase 6: Final intensive CD ===
    print(f"\n=== Phase 6: Final CD (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 15
