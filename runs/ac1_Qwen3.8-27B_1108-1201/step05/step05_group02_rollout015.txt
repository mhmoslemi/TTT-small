For a sequence `a[i]`, the DFT is `A[ω] = sum a[i] * exp(-2πiωi/n)`. The power spectrum is `|A[ω]|^2`. The circular autocorrelation is the inverse DFT of the power spectrum. """
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Soft-max (log-sum-exp) gradient descent with beta annealing for fast convergence
2. Hard-max subgradient method with adaptive softmax-weighted peaks for final polish
3. Multi-length exploration with diverse initializations (flat-spectrum, parametric, random)
4. Iterative restart strategy to escape local minima
"""

import numpy as np
import time


def fast_auto_conv(a):
    """Fast linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_softmax_grad(f, beta):
    """Compute gradient of log-sum-exp(beta * conv(f,f)) / beta.
    Returns (gradient, current_max, conv)."""
    n = len(f)
    conv = fast_auto_conv(f)
    scaled = beta * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    weights = exp_vals / exp_vals.sum()

    g = np.zeros(n)
    # Only process peaks with significant weight
    threshold = 1e-8
    active = weights > threshold
    for k in range(len(conv)):
        if not active[k]:
            continue
        w = 2.0 * weights[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        kg = k - idx
        g[idx] += w * f[kg]

    return g, conv.max(), conv


def softmax_gradient_descent(f0, time_limit, beta_init=5.0, beta_final=500.0,
                             lr_init=0.02, momentum=0.92, verbose=False):
    """
    Gradient descent on smooth log-sum-exp approximation of max(conv).
    Anneals beta from beta_init to beta_final over the time budget.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    best_obj = fast_auto_conv(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / max(time_limit, 1e-6))
        # Smooth annealing
        beta = beta_init * (beta_final / beta_init) ** (progress ** 0.7)
        lr = lr_init * (0.5 + 0.5 * (1.0 - progress))

        g, cur_max, _ = compute_softmax_grad(f, beta)

        v = momentum * v - lr * g
        f_new = project_simplex(f + v)

        obj = fast_auto_conv(f_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()

        f = f_new

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMGD iter {it}: obj={best_obj:.10f}, beta={beta:.1f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def hardmax_subgradient(f0, time_limit, top_k=None, temp=0.0005, lr=0.004,
                        momentum=0.93, verbose=False):
    """
    Subgradient method for hard max(conv) with softmax-weighted peaks.
    Uses top-k peaks with exponential weighting by temperature.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    best_obj = fast_auto_conv(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_auto_conv(f)
        cur_max = conv.max()

        k_actual = min(top_k or (len(conv) // 4), len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = (pv - pv.max()) / max(temp, 1e-10)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[int(np.argmax(pv))] = 1.0
        else:
            w /= ws

        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            ww = 2.0 * w[i]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            g[idx[mask]] += ww * f[kg[mask]]

        v = momentum * v - lr * g
        f_new = project_simplex(f + v)

        obj = fast_auto_conv(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive: increase temp if stuck (explore more peaks)
        if no_improve > 200:
            temp *= 1.5
            lr *= 0.8
            no_improve = 0

        # Periodic reset to best

            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)

def generate_init(n, method, seed=42):
    """Generate an initial sequence of length n using various methods."""
    rng = np.random.default_rng(seed)

    if method == 'uniform':
        f = np.ones(n) / n
    elif method == 'sine':
        x = np.arange(n)
        freq = rng.uniform(0.5, 3.0)
        f = 1.0 + 0.7 * np.cos(2.0 * np.pi * freq * x / n)
        f = np.maximum(f, 0.01)
        f /= f.sum()
    elif method == 'bimodal':
        x = np.arange(n)
        f = (np.exp(-((x - 0.2 * n) ** 2) / (2 * (0.08 * n) ** 2)) +
             np.exp(-((x - 0.8 * n) ** 2) / (2 * (0.08 * n) ** 2)))
        f /= f.sum()
    elif method == 'lognormal':
        f = np.exp(rng.standard_normal(n))
        f /= f.sum()
    elif method == 'chirp':
        x = np.arange(n)
        rate = rng.uniform(2.0, 8.0)
        f = 1.0 + 0.6 * np.abs(np.sin(np.pi * rate * x / n))
        f /= f.sum()
    elif method == 'flat_spectrum':
        # Generate with approximately flat power spectrum
        n_fft = 1 << (2 * n - 1).bit_length()
        half = n_fft // 2
        A = np.zeros(n_fft, dtype=np.complex128)
        # DC and Nyquist are real
        A[0] = 1.0
        if n_fft % 2 == 0:
            A[half] = 1.0
        for k in range(1, half):
            mag = 1.0 + 0.2 * rng.standard_normal()
            mag = max(mag, 0.3)
            ph = rng.uniform(0, 2 * np.pi)
            A[k] = mag * np.exp(1j * ph)
            A[n_fft - k] = np.conj(A[k])
        f = np.fft.irfft(A)[:n]
        f = np.maximum(f, 0.0)
        ss = f.sum()
        if ss > 1e-15:
            f /= ss
        else:
            f = np.ones(n) / n
    elif method == 'barkar_like':
        # Alternating with noise
        x = np.arange(n)
        f = 1.0 + 0.5 * np.cos(np.pi * x) + 0.3 * rng.standard_normal(n)
        f = np.maximum(f, 0.0)
        ss = f.sum()
        if ss > 1e-15:
            f /= ss
        else:
            f = np.ones(n) / n
    elif method == 'gaussian_ring':
        # Ring-like: high at two symmetric positions
        x = np.arange(n)
        center = n / 2.0
        width = n * 0.15
        f = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        f2 = 1.0 / (1.0 + ((x - center) / (n * 0.3)) ** 4)
        f = f + 0.5 * f2
        f /= f.sum()
    else:
        f = np.ones(n) / n

    return f


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

            print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # === Get parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            best_seq = parent.copy()
            best_score = p_score
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(832) / 832
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Optimize parent (two-phase) ===
    print("\n=== Phase 1: Two-phase on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(350, rem * 0.2)
    if t1 > 40:
        n_p = len(parent)
        print(f"  n={n_p}, time={t1:.0f}s")
        a_opt, obj = two_phase_optimize(parent, time_limit=t1, n=n_p)
        try_seq(a_opt, "twophase_parent")

    # === Phase 2: Try diverse initializations at parent length ===
    print("\n=== Phase 2: Diverse inits at parent length ===")
    n_parent = len(best_seq)
    inits = [
        ('sine', 101),
        ('bimodal', 202),
        ('lognormal', 303),
        ('flat_spectrum', 404),
        ('chirp', 505),
    ]
    for method, seed in inits:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.15, 150)
        if t_alloc < 30:
            break
        f_init = generate_init(n_parent, method, seed)
        print(f"  Method={method}, time={t_alloc:.0f}s")
        a_opt, obj = two_phase_optimize(f_init, time_limit=t_alloc, n=n_parent)
        try_seq(a_opt, f"init_{method}")

    # === Phase 3: Length exploration ===
    print("\n=== Phase 3: Length exploration ===")
    base = best_seq if best_seq is not None else parent
    lengths = [512, 640, 704, 768, 832, 896, 960, 1024, 1152, 1280, 1408, 1536, 1792, 2048]
    lengths = sorted(set(l for l in lengths if l >= 256))
    # Remove lengths we already tried
    lengths = [l for l in lengths if l != n_parent]

    for target_n in lengths:
        rem = deadline - time.monotonic()
        if rem < 100:
            break

        # Interpolate best to new length
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.25, 200)

        if t_alloc < 40:
            continue

        print(f"  n={target_n}, time={t_alloc:.0f}s")
        a_opt, obj = two_phase_optimize(f_interp, time_limit=t_alloc, n=target_n)
        try_seq(a_opt, f"interp_n{target_n}")

    # === Phase 4: Focus on best length with more aggressive search ===
    print(f"\n=== Phase 4: Aggressive search (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()

        # Multiple restarts from best with perturbation
        for restart in range(5):
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.2, 120)
            if t_alloc < 30:
                break

            noise_scale = 0.002 + 0.003 * restart
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)

            a_opt, obj = two_phase_optimize(f_pert, time_limit=t_alloc, n=n_best)
            try_seq(a_opt, f"restart_{restart}")

    # === Phase 5: Try longer sequences with flat-spectrum init ===
    print(f"\n=== Phase 5: Long sequences (best={best_score:.8f}) ===")
    for target_n in [1280, 1536, 2048, 2560]:
        rem = deadline - time.monotonic()
        if rem < 120:
            break

        f_init = generate_init(target_n, 'flat_spectrum', seed=target_n)
        t_alloc = min(rem * 0.3, 250)
        if t_alloc < 50:
            continue

        print(f"  n={target_n} (flat-spectrum init), time={t_alloc:.0f}s")
        a_opt, obj = two_phase_optimize(f_init, time_limit=t_alloc, n=target_n)
        try_seq(a_opt, f"long_n{target_n}")

        # Also try interpolated best
        rem = deadline - time.monotonic()
        if rem > 60 and best_seq is not None:
            f_interp = interpolate_to_length(best_seq, target_n)
            t_alloc2 = min(rem * 0.3, 150)
            if t_alloc2 > 40:
                a_opt2, obj2 = two_phase_optimize(f_interp, time_limit=t_alloc2, n=target_n)
                try_seq(a_opt2, f"interp_long_n{target_n}")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final fine-tuning (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very careful HSG with low temp and small lr
        for trial, (temp, lr, mk) in enumerate([(0.0001, 0.001, 0.96),
                                                 (0.00005, 0.0005, 0.97),
                                                 (0.00002, 0.0002, 0.98)]):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 80)
            if t_alloc < 20:
                break
            a_opt, obj = hardmax_subgradient(best_seq, time_limit=t_alloc,
                                             top_k=max(5, n_best // 20),
                                             temp=temp, lr=lr, momentum=mk)
            try_seq(a_opt, f"finetune_{trial}")

    # === Fallback ===
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback: score={best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        if fs < best_score:
            best_score = fs
    else:
        print("WARNING: best_seq failed evaluation, using fallback")
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
