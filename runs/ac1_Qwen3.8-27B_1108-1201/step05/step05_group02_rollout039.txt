Minimizes 2*n*max(conv(a,a))/sum(a)^2 using log-sum-exp smooth optimization
with adaptive temperature annealing, FFT-based gradients, diverse initialization,
length exploration, and coordinate descent polish.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*c_k))) where c=conv(a,a).
    Returns (gradient, c_max)."""
    n = len(a)
    c = fast_convolve(a)
    beta_c = beta * c
    beta_c -= beta_c.max()
    p = np.exp(beta_c)
    p /= p.sum()

                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial /= f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break
            if no_improve > 15:
                break

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate (or downsample) sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window (bell-shaped, high in middle)
    x = np.linspace(0, np.pi, n)
    a = np.cos(0.5 * x) ** 2
    a = np.maximum(a, 0)
    a /= a.sum()
    inits.append(a)

    # U-shaped (high at edges)
    t = np.linspace(0, 1, n)
    a = 0.3 + 0.7 * (2 * t - 1) ** 2
    a /= a.sum()
    inits.append(a)

    # Raised cosine
    t = np.linspace(0, 1, n)
    a = 1.0 - np.cos(2 * np.pi * t)
    a = np.maximum(a, 0.01)
    a /= a.sum()
    inits.append(a)

    # Two-peak
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.25) / 0.15) ** 2) + np.exp(-((x - 0.75) / 0.15) ** 2)
    a /= a.sum()
    inits.append(a)

    # Gaussian centered
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) / 0.2) ** 2)
    a /= a.sum()
    inits.append(a)

    # Random
    for _ in range(3):
        a = rng.random(n)
        a /= a.sum()
        inits.append(a)

    # Parent (interpolated if available)
    if parent is not None:
        p = interpolate_seq(parent, n)
        inits.append(p)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Diverse initialization with quick LSE optimization
    2. Main LSE optimization with annealing
    3. Length exploration
    4. Coordinate descent polish
    5. Final attempts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-10:
            return float('inf')
        a_norm = a / ss
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(832) / 832
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Diverse initialization (250s) ===
    print(f"\n=== Phase 1: Diverse Init (best={best_score:.8f}) ===")
    t1 = 250
    if best_score > 1.5027:
        # Try multiple initializations at the parent length
        inits = make_diverse_inits(base_n, rng, parent)
        for i, init in enumerate(inits[:7]):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(t1 / len(inits[:7]), rem - 10)
            if t_alloc < 10:
                break
            print(f"  Init {i}: running LSE for {t_alloc:.0f}s...")
            a_opt, _ = lse_optimize(init, time_limit=t_alloc,
                                    beta_start=2.0, beta_end=300.0,
                                    max_iters=5000, seed=seed + i)
            try_seq(a_opt, f"init_{i}")

    # === Phase 2: Main LSE optimization (650s) ===
    print(f"\n=== Phase 2: Main LSE (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(650, rem * 0.4)
    if t2 > 50 and best_seq is not None:
        n_best = len(best_seq)
        print(f"  n={n_best}, t={t2:.0f}s")

        # First main run
        a_opt, _ = lse_optimize(best_seq, time_limit=t2 * 0.55,
                                beta_start=2.0, beta_end=600.0,
                                verbose=True, max_iters=30000, seed=seed)
        try_seq(a_opt, "lse_main1")

        # Second run with different params
        rem = deadline - time.monotonic()
        t2b = min(t2 * 0.45, rem - 20)
        if t2b > 30:
            a_opt, _ = lse_optimize(best_seq, time_limit=t2b,
                                    beta_start=5.0, beta_end=1000.0,
                                    max_iters=25000, seed=seed + 100)
            try_seq(a_opt, "lse_main2")

        # Third: start from perturbed version
        rem = deadline - time.monotonic()
        t2c = min(120, rem - 20)
        if t2c > 20 and best_seq is not None:
            perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.01)
            a_opt, _ = lse_optimize(perturbed, time_limit=t2c,
                                    beta_start=3.0, beta_end=500.0,
                                    max_iters=15000, seed=seed + 200)
            try_seq(a_opt, "lse_perturbed")

    # === Phase 3: Length exploration (450s) ===
    print(f"\n=== Phase 3: Length Exploration (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    # Try lengths around the parent
    candidate_lengths = set()
    for delta in [-192, -128, -64, -32, 32, 64, 128, 192, 256, 320, 384]:
        ln = len(base) + delta
        if 400 <= ln <= 2048:
            candidate_lengths.add(ln)
    # Also try some "round" numbers
    for ln in [512, 640, 768, 832, 896, 960, 1024, 1152, 1280, 1408]:
        if 400 <= ln <= 2048:
            candidate_lengths.add(ln)
    candidate_lengths = sorted(candidate_lengths)

    # Prioritize lengths close to the current best length
    n_best = len(base)
    candidate_lengths.sort(key=lambda x: abs(x - n_best))
    candidate_lengths = candidate_lengths[:10]  # top 10 closest

    for target_n in candidate_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if target_n == len(base):
            continue
        t_alloc = min(rem * 0.35, 100)
        if t_alloc < 25:
            break
        f_interp = interpolate_seq(base, target_n)
        print(f"  Trying n={target_n} (t={t_alloc:.0f}s)...")
        a_opt, _ = lse_optimize(f_interp, time_limit=t_alloc,
                                beta_start=3.0, beta_end=500.0,
                                max_iters=10000, seed=seed + target_n)
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 4: Coordinate descent (300s) ===
    print(f"\n=== Phase 4: CD Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t4 = min(300, rem * 0.3)
        if t4 > 25:
            print(f"  n={len(best_seq)}, t={t4:.0f}s")
            a_opt, _ = cd_optimize(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "cd_polish")

            # Second CD with slightly different start
            rem = deadline - time.monotonic()
            t4b = min(100, rem - 20)
            if t4b > 15:
                pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.005)
                a_opt, _ = cd_optimize(pert, time_limit=t4b)
                try_seq(a_opt, "cd_polish2")

    # === Phase 5: Final combined attempts (remaining time) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick LSE
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t5 = min(rem, 150)
            a_opt, _ = lse_optimize(best_seq, time_limit=t5,
                                    beta_start=10.0, beta_end=2000.0,
                                    max_iters=20000, seed=seed + 999)
            try_seq(a_opt, "lse_final")

        # Quick CD
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, _ = cd_optimize(best_seq, time_limit=min(rem, 80))
            try_seq(a_opt, "cd_final")

        # Multiple small perturbations + LSE
        for p in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break
            noise = rng.standard_normal(n_best) * (0.003 + 0.002 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lse_optimize(f_pert, time_limit=min(rem, 40),
                                    beta_start=5.0, beta_end=800.0,
                                    max_iters=8000, seed=seed + 1000 + p)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(832) / 832
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(832) / 832
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
