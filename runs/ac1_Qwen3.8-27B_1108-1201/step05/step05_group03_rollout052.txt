import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_weighted(a, peaks, weights):
    """Compute weighted subgradient of auto-convolution max."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            segment = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * w * segment
    return grad

def lp_direction_fast(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_stagewise(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    else:
        f = f / s
    n = len(f)

    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time_limit - (time.monotonic() - t0) < 5:
            break
        it += 1

        g = lp_direction_fast(f, M, n)
        if g is None:
            g = lp_direction_fast(f, M * 1.01, n)
            if g is None:
                g = lp_direction_fast(f, M * 1.05, n)
            if g is None:
                no_improve += 1
                if no_improve > 60:
                    break
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 60:
                rng = np.random.default_rng(it * 17 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                continue
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.001 * min(no_improve // 40, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate-wise optimization with golden section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42424)

    no_improve_passes = 0
    pass_num = 0

    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        if no_improve_passes > 3:
            break

        pass_improved = False
        order = rng.permutation(n)

        for ii in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[ii]

            # Try transferring mass to several partners
            for offset in [1, 2, 3, 5, 8]:
                j = order[(ii + offset) % n]
                if i == j:
                    continue

                c = f[i] + f[j]
                if c < 1e-12:
                    continue

                # Golden section search for optimal split
                lo = 0.0
                hi = 1.0
                f_a = f.copy()
                f_b = f.copy()
                gr = (np.sqrt(5) - 1) / 2

                x1 = hi - gr * (hi - lo)
                x2 = lo + gr * (hi - lo)

                f_a[i] = x1 * c
                f_a[j] = (1 - x1) * c
                va = fast_convolve(f_a).max()

                f_b[i] = x2 * c
                f_b[j] = (1 - x2) * c
                vb = fast_convolve(f_b).max()

                for _ in range(8):
                    if hi - lo < 1e-7:
                        break
                    if va < vb:
                        hi = x2
                        x2 = x1
                        vb = va
                        x1 = hi - gr * (hi - lo)
                        f_a = f.copy()
                        f_a[i] = x1 * c
                        f_a[j] = (1 - x1) * c
                        va = fast_convolve(f_a).max()
                    else:
                        lo = x1
                        x1 = x2
                        va = vb
                        x2 = lo + gr * (hi - lo)
                        f_b = f.copy()
                        f_b[i] = x2 * c
                        f_b[j] = (1 - x2) * c
                        vb = fast_convolve(f_b).max()

                x_opt = (lo + hi) / 2
                f_trial = f.copy()
                f_trial[i] = x_opt * c
                f_trial[j] = (1 - x_opt) * c
                val = fast_convolve(f_trial).max()

                if val < best_obj - 1e-13:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if verbose:
                print(f"  CoordDescent pass {pass_num}: no improvement")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_init(n):
    """Cosine-shaped initialization."""
    x = np.arange(n)
    f = np.cos(np.pi * x / (2 * n)) ** 2 + 0.01
    f = np.maximum(f, 0)
    f /= f.sum()
    return f


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.arange(n) - n / 2
    sigma = n * sigma_frac
    f = np.exp(-x ** 2 / (2 * sigma ** 2))
    f /= f.sum()
    return f


def triangle_init(n):
    """Triangular initialization."""
    x = np.arange(n)
    f = 1.0 - np.abs(2.0 * x / n - 1.0)
    f = np.maximum(f, 0.01)
    f /= f.sum()
    return f


def random_bimodal_init(n, seed=123):
    """Random bimodal initialization."""
    rng = np.random.default_rng(seed)
    f = rng.standard_normal(n) ** 2
    f += 0.1
    f /= f.sum()
    return f

        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return float('inf')
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Adam-Adaptive on parent ===
    print(f"\n=== Phase 1: Adam-Adaptive on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 800
    t1 = min(rem, 300)
    if t1 > 30:
        print(f"  Allocating {t1:.0f}s")

    # === Phase 2: LP stagewise on best ===
    print(f"\n=== Phase 2: LP Stagewise (best={best_score:.8f}) ===")

    # === Phase 3: Multi-scale exploration ===
    print(f"\n=== Phase 3: Multi-scale (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent

    # Try different target lengths
    target_lengths = [512, 768, 1024, 896]

            try_seq(a_opt, f"adam_n{target_n}")

                t_lp = min(rem * 0.4, 80)

    return [float(x) for x in best_seq]
