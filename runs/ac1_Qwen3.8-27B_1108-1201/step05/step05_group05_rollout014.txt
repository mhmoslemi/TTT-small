import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_palindromic(a, n):
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])

def solve_lp_direction(f, M, n, active_k=None):
    c = -np.ones(n)
    if active_k is None or len(active_k) >= 2 * n - 1:
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix for the LP constraints."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_full(f, M, n, A_cache=None):
    """Solve LP with all 2n-1 constraints."""
    c = -np.ones(n)
    if A_cache is None:
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
    else:
        A_ub = A_cache
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_sub(f, M, n, active_indices):
    """Solve LP with subset of constraints."""
    c = -np.ones(n)
    K = len(active_indices)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    for row, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def peak_directed_cd(f0, time_limit, verbose=False):
    """Coordinate descent that specifically targets the peak lags."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f0)) / len(f0)

        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        # Find top peaks
        top_k = min(10, len(conv))
        peak_lags = np.argsort(conv)[::-1][:top_k]
        
        improved = False
        for k in peak_lags:
            if time.monotonic() - t0 > time_limit:
                break
            # For lag k, the contributing pairs are (j, k-j)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Try reducing each contributing coordinate
            for j in range(js, je + 1):
                partner = k - j
                if time.monotonic() - t0 > time_limit:
                    break
                
                if f[j] < 1e-10:
                    continue
                
                # Try reducing f[j] by various amounts
                for frac in [0.1, 0.25, 0.5]:
                    delta = f[j] * frac
                    if delta < 1e-12:
                        continue
                    # Redistribute delta proportionally to others
                    f_trial = f.copy()
                    f_trial[j] -= delta
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    other_sum = f_trial[mask].sum()
                    if other_sum < 1e-15:
                        continue
                    f_trial[mask] += delta * f_trial[mask] / other_sum
                    f_trial = f_trial / f_trial.sum()
                    
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-15:
                        # Accept
                        f = f_trial
                        best_obj = obj

                        improved = True
                        break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3:

                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
            elif no_improve > 10:

        if verbose:
            pass  # Too verbose

def smooth_cd(f0, time_limit, multipliers=None):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

        f = f / s

    if multipliers is None:

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_mult = 1.0
            for mult in multipliers:
                new_val = f[j] * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - f[j]
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_mult = mult

            if best_j_mult != 1.0:
                new_val = f[j] * best_j_mult
                denom = 1.0 - f[j]
                if denom > 1e-10:
                    scale = (1.0 - new_val) / denom
                    f[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj

                    improved = True

        if improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def two_level_init(n, gap=0.5, seed=42):
    """Initialize with a two-level pattern."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) * 0.5
    # Create a pattern where ~50% are high and ~50% are low
    threshold = gap
    for i in range(0, n, 2):
        if i + 1 < n:
            a[i] = 1.0 - threshold
            a[i + 1] = threshold
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


def sinusoidal_init(n, freq=3, amplitude=0.3, seed=42):
    """Initialize with sinusoidal modulation."""
    x = np.arange(n) / n
    a = 1.0 + amplitude * np.sin(2 * np.pi * freq * x)
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


def freq_bump_init(n, center=0.1, width=0.15, seed=42):
    """Initialize with frequency-domain bump (concentrated power spectrum)."""
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    rng = np.random.default_rng(seed)
    freqs = np.linspace(0, 1, fft_size // 2 + 1)
    # Create a bump in the frequency domain
    bump = np.exp(-((freqs - center) ** 2) / (2 * width ** 2))
    # Add random phases
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = bump * np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0.0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig
