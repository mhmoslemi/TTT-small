import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def lse_smooth_max(conv, tau):
    """Log-sum-exp smoothing of max."""
    shifted = conv / tau - conv.max() / tau
    w = np.exp(shifted)
    return tau * (np.log(w.sum()) + conv.max() / tau)


def lse_weights(conv, tau):
    """Weights for log-sum-exp smoothing."""
    shifted = conv / tau - conv.max() / tau
    w = np.exp(shifted)
    return w / w.sum()


def subgradient_step(f, conv, weights, lr):
    """Compute subgradient of smoothed max autocorrelation."""
    n = len(f)
    sg = np.zeros(n)
    k_max = len(conv)
    # Only process top-weighted lags for efficiency
    order = np.argsort(weights)[::-1]
    active = order[:min(60, k_max)]
    for k in active:
        w = weights[k]
        if w < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w * f[k - idx]
    return sg


def lp_solve_direction(f, M, n, active_set=None):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for k in active_set, g >= 0."""
    c = -np.ones(n)
    total_lags = 2 * n - 1
    if active_set is None:
        active_set = np.arange(total_lags)
    
    K = len(active_set)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    
    for row, k in enumerate(active_set):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def sg_optimize(f0, time_limit, lr=0.002, mom=0.92, top_k=50, tau_init=None):

    if tau_init is None:
        tau = M0 * 0.008
    else:
        tau = tau_init
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        weights = lse_weights(conv, tau)
        sg = subgradient_step(f, conv, weights, 1.0)
        
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive params
        cur_lr *= 0.9999
        tau *= 1.003
        
        if no_improve > 300:

            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            conv = fast_convolve(f)
            best_obj = best_obj  # keep
            cur_lr = max(lr * 0.7, 0.0003)
            tau = M0 * 0.005
            no_improve = 0

    return best_f, best_obj


def coord_descent(f0, time_limit):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    multipliers = [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.0]
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-13:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or 1.0 - new_val < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial[np.arange(n) != j] *= scale
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                f[np.arange(n) != j] *= scale
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def diverse_init(n, method, seed=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    if method == "uniform":
        return np.ones(n) / n
    elif method == "cosine":
        j = np.arange(n, dtype=float)
        a = np.cos(np.pi * j / (2 * (n - 1)))
        a = np.maximum(a, 0)
        a /= a.sum()
        return a
    elif method == "exponential":
        alpha = 1.0 + rng.uniform(0, 2)
        j = np.arange(n, dtype=float)
        a = np.exp(-alpha * j / n)
        a /= a.sum()
        return a
    elif method == "gaussian":
        j = np.arange(n, dtype=float)
        center = n * (0.3 + 0.4 * rng.uniform())
        width = n * (0.15 + 0.2 * rng.uniform())
        a = np.exp(-0.5 * ((j - center) / width) ** 2)
        a = np.maximum(a, 1e-10)
        a /= a.sum()
        return a
    elif method == "flat_spectrum":
        fft_size = 1 << (2 * n - 2).bit_length()
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        spectrum = np.exp(1j * phases)
        spectrum[0] = 1.0
        if fft_size % 2 == 0 and fft_size // 2 < len(spectrum):
            spectrum[fft_size // 2] = np.exp(1j * phases[fft_size // 2])
        sig = np.fft.irfft(spectrum, fft_size)
        sig = sig[:n]
        sig = np.maximum(sig, 0)
        ss = sig.sum()
        if ss < 1e-10:
            sig = np.abs(sig[:n])
            ss = sig.sum()
        if ss > 1e-10:
            sig /= ss
        else:
            sig = np.ones(n) / n
        return sig
    elif method == "random":
        a = rng.exponential(1.0, n)
        a /= a.sum()
        return a
    elif method == "two_bump":
        j = np.arange(n, dtype=float)
        c1 = n * 0.25
        c2 = n * 0.75
        w = n * 0.15
        a = np.exp(-0.5 * ((j - c1) / w) ** 2) + np.exp(-0.5 * ((j - c2) / w) ** 2)
        a = np.maximum(a, 1e-10)
        a /= a.sum()
        return a
    elif method == "sawtooth":
        j = np.arange(n, dtype=float)
        period = max(4, n // (2 + rng.integers(0, 3)))
        a = np.abs((j % period) / period - 0.5) + 0.1
        a /= a.sum()
        return a
    else:
        return np.ones(n) / n

    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_cd = min(rem * 0.3, 200)
            print(f"  CD time: {t_cd:.0f}s")
            a_opt, _ = coord_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")
            
            # Second pass
            rem = deadline - time.monotonic()
            if rem > 30:
                t_cd2 = min(rem * 0.2, 100)
                a_opt2, _ = coord_descent(a_opt, time_limit=t_cd2)
                try_seq(a_opt2, "cd2")
    
    # === Phase 6: Final SG polishing ===
    print(f"\n=== Phase 6: Final SG (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 20:
            t_sg = min(rem * 0.35, 150)
            print(f"  SG time: {t_sg:.0f}s")
            a_opt, _ = sg_optimize(best_seq, time_limit=t_sg,
                                   lr=0.0006, mom=0.96,
                                   top_k=max(30, n_best // 4),
                                   tau_init=fast_convolve(best_seq).max() * 0.003)
            try_seq(a_opt, "final_sg")
            
            rem = deadline - time.monotonic()
            if rem > 15:
                a_opt2, _ = sg_optimize(a_opt, time_limit=min(rem * 0.3, 60),
                                        lr=0.0003, mom=0.97,
                                        top_k=max(20, n_best // 6),
                                        tau_init=fast_convolve(a_opt).max() * 0.001)
                try_seq(a_opt2, "final_sg2")
    
    # === Phase 7: Final LP-FW polish ===
    print(f"\n=== Phase 7: Final LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_final = min(rem * 0.5, 80)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final,
                                      verbose=False, top_k=60)
            try_seq(a_opt, "final_lpfw")
    
    # === Phase 8: Tiny perturbations and evaluation ===
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        for i in range(6):
            if rem < 5:
                break
            rem = deadline - time.monotonic()
            noise = rng.standard_normal(n_best) * (0.0005 + 0.0005 * i)
            f_pert = project_simplex(best_seq + noise)
            try_seq(f_pert, f"perturb_{i}")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
