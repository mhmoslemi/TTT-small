def project_simplex(v):
    n = len(v)
    if n == 1:

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(f, w):
    """Compute gradient of sum_k w[k]*conv(f,f)[k] w.r.t. f.
    Uses np.correlate for efficiency.
    grad[j] = 2 * sum_k w[k] * f[k-j] * [0<=k-j<n]
    """
    n = len(f)
    # grad[j]/2 = sum_{m=0}^{n-1} f[m] * w[j+m] (with w[k]=0 for k >= len(w))
    # This is np.correlate(w, f, 'full')[j] for j=0,...,n-1
    corr = np.correlate(w, f, mode='full')
    grad = 2.0 * corr[:n]
    return grad


def compute_grad_top(f, w, max_peaks=200):
    """Compute gradient using only top max_peaks entries of w."""
    n = len(f)
    conv_len = len(w)
    if len(w) <= max_peaks:
        sig_idx = np.argsort(w)[::-1]
    else:
        sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
    
    w_sparse = np.zeros(conv_len)
    w_sparse[sig_idx] = w[sig_idx]
    return compute_grad(f, w_sparse)


def lse_weights(conv, beta):
    """Compute LSE weights from convolution values."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    return exp_bc / exp_bc.sum()


def mirror_descent_step(f, grad, lr):
    """Multiplicative weights update on the simplex."""
    log_mult = -lr * grad
    log_mult_max = log_mult.max()
    f_new = f * np.exp(log_mult - log_mult_max)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    f_new /= s
    return f_new


def lse_mirror_descent(f0, time_limit, beta_start=5.0, beta_end=10000.0,
                       lr_start=0.2, lr_min=1e-6, momentum=0.0,
                       max_peaks=200, verbose=False):
    """LSE-smoothed mirror descent with optional momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = lr_start
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_lr = lr_start
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        w = lse_weights(conv, beta)
        
        if momentum > 0:
            # Nesterov-like: extrapolate
            y = f + momentum * v
            y_s = y.sum()
            if y_s < 1e-15:
                y = f
            else:
                y = y / y_s
            grad = compute_grad_top(y, w, max_peaks=max_peaks)
        else:
            grad = compute_grad_top(f, w, max_peaks=max_peaks)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm
        
        if momentum > 0:
            v = grad  # store for next extrapolation
            # Actually, for mirror descent, momentum is different
            # Let's use: f_new = mirror_step(f - lr * (grad + momentum * (grad - grad_prev)))
            # Simplified: just use the current gradient with effective lr
            lr_eff = lr / (1.0 + momentum)
        else:
            lr_eff = lr
        
        f_new = mirror_descent_step(f, grad, lr_eff)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slight lr increase
            lr = min(lr * 1.002, lr_start * 2.0)
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.95
            if no_improve > 80:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_start * 0.5, lr_min)
                beta = min(beta * 1.5, beta_end)
                no_improve = 0
                if verbose:
                    print(f"  MD restart it={it}, obj={best_obj:.10f}, beta={beta:.1f}")
        
        # Anneal beta
        if beta < beta_end:
            beta *= 1.005
        
        if verbose and it % 500 == 0:
            print(f"  MD it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}")
    
    return best_f, best_obj


def lse_projected_descent(f0, time_limit, beta_start=5.0, beta_end=10000.0,
                          lr_start=0.01, momentum=0.9, max_peaks=200, verbose=False):
    """LSE-smoothed gradient descent with Euclidean projection and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = lr_start
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        w = lse_weights(conv, beta)
        grad = compute_grad_top(f, w, max_peaks=max_peaks)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 0.05)
        else:
            no_improve += 1
            if no_improve >= 8:
                lr *= 0.93
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_start * 0.5, 0.001)
                beta = min(beta * 1.6, beta_end)
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.004
    
    return best_f, best_obj


def fourier_parametrize(c, n):
    """Convert Fourier coefficients to a non-negative sequence.
    c = [A1, B1, A2, B2, ..., AM, BM]
    a_i = exp(sum_m A_m*cos(2*pi*m*i/n) + B_m*sin(2*pi*m*i/n)) / Z
    """
    M = len(c) // 2
    if M == 0:
        return np.ones(n) / n
    x = np.arange(n)
    log_a = np.zeros(n)
    for m in range(1, M + 1):
        A = c[2 * (m - 1)]
        B = c[2 * (m - 1) + 1]
        log_a += A * np.cos(2.0 * np.pi * m * x / n) + B * np.sin(2.0 * np.pi * m * x / n)
    log_a -= log_a.max()
    a = np.exp(log_a)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    a /= s
    return a


def fourier_optimize(n, M=8, time_limit=120, n_restarts=10, seed=42):
    """Optimize Fourier parameters to minimize peak autocorrelation."""
    rng = np.random.default_rng(seed)
    best_f = None
    best_obj = np.inf
    
    for restart in range(n_restarts):
        scale = 0.3 / (restart + 1)
        c0 = rng.standard_normal(2 * M) * scale
        t_start = time.monotonic()
        
        def obj_func(c):
            f = fourier_parametrize(c, n)
            conv = fast_convolve(f)
            return conv.max()
        
        try:
            res = minimize(obj_func, c0, method='Nelder-Mead',
                           options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12,
                                    'adaptive': True})
            f = fourier_parametrize(res.x, n)
            obj = fast_convolve(f).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
                print(f"    Fourier restart {restart}: obj={obj:.10f}")
        except Exception:
            pass
        
        if time.monotonic() - t_start > time_limit / n_restarts:
            break
        if time.monotonic() - t_start > time_limit * 0.8:
            break
    
    return best_f, best_obj


def slsqp_refine(f0, max_iter=30, time_limit=60):
    """Local refinement using SLSQP."""
    n = len(f0)
    f = f0.copy().astype(np.float64)
    
    t0 = time.monotonic()
    
    def objective(x):
        if time.monotonic() - t0 > time_limit:
            raise TimeoutError()
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-10:
            return 1e10
        x = x / s
        conv = fast_convolve(x)
        return conv.max()
    
    def objective_grad(x):
        if time.monotonic() - t0 > time_limit:
            raise TimeoutError()
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-10:
            return np.zeros(n)
        x = x / s
        conv = fast_convolve(x)
        k = int(conv.argmax())
        grad = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            grad[j] = 2.0 * x[k - j]
        return grad
    
    def constraint_eq(x):
        return x.sum() - 1.0
    
    constraints = [{'type': 'eq', 'fun': constraint_eq, 'jac': lambda x: np.ones(n)}]
    bounds = [(0, None)] * n
    
    try:
        res = minimize(objective, f, jac=objective_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            x = np.maximum(res.x, 0)
            s = x.sum()
            if s > 1e-10:
                x /= s
                obj = fast_convolve(x).max()
                return x, obj
    except (TimeoutError, Exception):
        pass
    
    return f, objective(f)


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score
