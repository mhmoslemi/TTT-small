"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
LP-based Frank-Wolfe for fast initial progress, Nesterov-accelerated projected
subgradient with iterate averaging for long runs, and multi-length/symmetric

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_subgradient(a, conv, top_k=50, temperature=0.001):
    """Compute smoothed subgradient of max_k conv(a,a)[k] using top-k softmax weighting."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w[i] * a[k - indices]
    return sg


def make_init(n, kind, seed=0):
    """Create various initializations."""
    j = np.arange(n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / (n - 1)))
    elif kind == 'blackman':
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * j / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * j / (n - 1))
    elif kind == 'cos1':
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * j / n)
    elif kind == 'cos2':
        a = 1.0 + 0.25 * np.cos(4.0 * np.pi * j / n)
    elif kind == 'cos3':
        a = 1.0 + 0.2 * np.cos(6.0 * np.pi * j / n)
    elif kind == 'bathtub':
        a = 1.0 + 0.4 * np.sin(np.pi * j / (n - 1))
    elif kind == 'random':
        rng = np.random.default_rng(seed)
        a = rng.random(n) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    return normalize(a)

def _solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f@g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 5, 'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _solve_lp_symmetric(f, M, n, half_n):
    """Solve LP for symmetric g: max sum(g) s.t. (f@g)[k] <= M, g symmetric, g >= 0."""
    c = -np.ones(half_n)
    # Build constraint matrix for symmetric g
    # g[j] = g[n-1-j], so variable h[j] represents g[j] = g[n-1-j] for j in [0, half_n)
    # conv(f, g)[k] = sum_j f[j]*g[k-j]
    # For symmetric g: g[m] = h[min(m, n-1-m)]
    A_ub = np.zeros((n, half_n))  # By symmetry, only need k=0..n-1
    for j in range(n):
        for m in range(half_n):
            # h[m] contributes to g[m] and g[n-1-m]
            # g[m] contributes to conv[k] when k-j=m, i.e., k=j+m
            # g[n-1-m] contributes to conv[k] when k-j=n-1-m, i.e., k=j+n-1-m
            if j + m < n:
                A_ub[j + m, m] += f[j]
            if j + n - 1 - m < n and (n - 1 - m) != m:
                A_ub[j + n - 1 - m, m] += f[j]
    b_ub = np.full(n, M)
    bounds = [(0, None)] * half_n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 5, 'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def nesterov_subgrad(f0, time_limit, lr0=0.01, mom0=0.9, top_k=50, temp0=0.0005):
    """Nesterov-accelerated projected subgradient with iterate averaging."""
    f = normalize(f0.astype(np.float64).copy())
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    avg_f = np.zeros(n)
    count = 0

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute smoothed subgradient
        sg = compute_subgradient(f, conv, top_k=top_k, temperature=cur_temp)

        # Nesterov update
        v = cur_mom * v - cur_lr * sg
        f_new = normalize(y + v)
        y = f_new + cur_mom * (f_new - f)
        y = normalize(y)
        f = f_new

        # Track average (running average for convergence)
        avg_f += f
        count += 1

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Also check the average
        if count % 1000 == 0:
            avg_check = normalize(avg_f / count)
            avg_conv = fast_convolve(avg_check)
            avg_obj = avg_conv.max()
            if avg_obj < best_obj - 1e-15:
                best_obj = avg_obj
                best_f = avg_check.copy()
                no_improve = 0

        # Adaptive parameters
        cur_lr *= 0.9998
        cur_temp *= 1.003
        cur_mom = min(0.98, cur_mom * 1.0002)

        if no_improve > 500:
            rng = np.random.default_rng(hash((it, int(time.monotonic() * 1000))) % 2**31)
            scale = 0.003 + 0.002 * min(no_improve // 500, 5)
            f = normalize(best_f + rng.standard_normal(n) * scale)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.7, 0.002)
            cur_temp = temp0
            no_improve = 0
            conv = fast_convolve(f)

    # Final: compare last iterate with average
    avg_f = normalize(avg_f / max(1, count))
    avg_obj = fast_convolve(avg_f).max()
    if avg_obj < best_obj:
        best_f = avg_f
        best_obj = avg_obj

    return best_f, best_obj


def subgrad_simple(f0, time_limit, lr0=0.005, top_k=100, temp0=0.001):
    """Simple projected subgradient with decreasing step size and averaging."""
    f = normalize(f0.astype(np.float64).copy())
    n = len(f)
    avg_f = np.zeros(n)
    count = 0

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    cur_temp = temp0

    while time.monotonic() - t0 < time_limit:
        it += 1

        sg = compute_subgradient(f, conv, top_k=top_k, temperature=cur_temp)

        # Decreasing step size: lr_t = C / sqrt(t)
        lr = lr0 / (1.0 + 0.01 * it) ** 0.5
        f = normalize(f - lr * sg)

        avg_f += f
        count += 1

        if it % 500 == 0:
            conv = fast_convolve(f)
            co = conv.max()
            if co < best_obj - 1e-15:
                best_obj = co
                best_f = f.copy()

            avg_check = normalize(avg_f / count)
            avg_conv = fast_convolve(avg_check)
            if avg_conv.max() < best_obj - 1e-15:
                best_obj = avg_conv.max()
                best_f = avg_check.copy()

        cur_temp *= 1.002

    # Final check
    avg_f = normalize(avg_f / max(1, count))
    avg_obj = fast_convolve(avg_f).max()
    if avg_obj < best_obj:
        best_f = avg_f
        best_obj = avg_obj

    conv_f = fast_convolve(best_f)
    return best_f, conv_f.max()


def spectral_init(n, num_iters=100):
    """
    POCS-style spectral method: iterate between making the spectrum flat
    and projecting onto the simplex.
    """
    a = make_init(n, 'hann')
    for _ in range(num_iters):
        # Compute spectrum
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2

        # Make flatter: clip the power spectrum
        P_mean = P.mean()
        P_clipped = np.minimum(P, 2.0 * P_mean)
        # Ensure the zero frequency component is preserved (it relates to sum)
        # a_new has the same "shape" in frequency but flattened magnitude
        A_new = np.sqrt(P_clipped) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]

        # Project onto simplex
        a = normalize(a_new)

    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

            try_seq(a_opt, "perturb_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
