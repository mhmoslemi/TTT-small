def project_simplex(v):

def coordinate_descent(f0, time_limit):
    """Cyclic coordinate descent with candidate evaluation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        cur_obj = best_obj
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                candidates = [0.002 / n, 0.01 / n]
            else:
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                candidates = [0.0, 0.3 * old_val, 0.5 * old_val, 0.7 * old_val,
                              1.3 * old_val, 1.7 * old_val]
                candidates = [max(0.0, min(c, 0.9999)) for c in candidates]

            for c in candidates:
                ft = f.copy()
                ft[j] = c
                st = 1.0 - c
                if st < 1e-14:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                others_sum = ft[mask].sum()
                if others_sum > 1e-14:
                    ft[mask] = ft[mask] * st / others_sum
                else:
                    ft[mask] = st / max(n - 1, 1)
                val = fast_convolve(ft).max()
                if val < cur_obj - 1e-14:
                    f = ft
                    cur_obj = val
                    improved = True

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()

        if not improved:
            break

    return best_f, best_obj


def cosine_init(n, freq, amp):
    """Cosine-modulated initialization on simplex."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return a / ss


def triangular_init(n):
    """Gaussian-shaped initialization."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((j - center) / (n * 0.25))**2)
    ss = a.sum()
    return a / ss


def uniform_init(n):
    return np.ones(n) / n


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length, normalize to simplex."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_evaluate(a, best_state):
    """Evaluate sequence and update best state (dict with 'seq' and 'score')."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return float('inf')
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state['score']:
        best_state['score'] = score
        best_state['seq'] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses Nesterov subgradient, Frank-Wolfe, coordinate descent,
    and multi-start with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_state = {'seq': None, 'score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(parent)}, sum={parent.sum():.4f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(256)

    n_parent = len(parent)
    parent_norm = parent / parent.sum()
    try_evaluate(parent_norm, best_state)

    # Try uniform sequences of various lengths
    for n_try in [n_parent, 128, 256, 384, 512, 768, 1024, 1536, 2048]:
        if n_try < 2 or n_try > 10000:
            continue
        try_evaluate(uniform_init(n_try), best_state)

    # Phase 1: Nesterov on best
    print(f"\n=== Phase 1: Nesterov (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 20 and best_state['seq'] is not None:
        t_alloc = min(rem * 0.3, 120)
        n_b = len(best_state['seq'])
        print(f"  Nesterov n={n_b}, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_optimize(best_state['seq'], time_limit=t_alloc,
                                      lr0=0.02, mom0=0.86, top_k=3)
        try_evaluate(a_opt, best_state)

    # Phase 2: Frank-Wolfe
    print(f"\n=== Phase 2: Frank-Wolfe (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 15 and best_state['seq'] is not None:
        t_alloc = min(rem * 0.2, 60)
        n_b = len(best_state['seq'])
        print(f"  FW n={n_b}, t={t_alloc:.0f}s")
        a_opt, _ = frank_wolfe_optimize(best_state['seq'], time_limit=t_alloc)
        try_evaluate(a_opt, best_state)

    # Phase 3: Diverse initializations
    print(f"\n=== Phase 3: Diverse inits (best={best_state['score']:.6f}) ===")
    base = best_state['seq'] if best_state['seq'] is not None else parent_norm
    n_base = len(base)

    diverse_inits = [
        cosine_init(n_base, 2, 0.3),
        cosine_init(n_base, 3, 0.2),
        cosine_init(n_base, 5, 0.15),
        triangular_init(n_base),
    ]

    for idx, init in enumerate(diverse_inits):
        rem = deadline - time.monotonic() - 300
        if rem < 10:
            break
        s = try_evaluate(init, best_state)
        if np.isfinite(s) and s < best_state['score'] + 0.5:
            t_alloc = min(rem * 0.15, 35)
            if t_alloc > 8:
                a_opt, _ = nesterov_optimize(init, time_limit=t_alloc,
                                              lr0=0.015, mom0=0.88, top_k=3)
                try_evaluate(a_opt, best_state)

    # Phase 4: Coordinate descent (only for smaller n)
    print(f"\n=== Phase 4: Coord descent (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic() - 250
    if rem > 15 and best_state['seq'] is not None:
        n_b = len(best_state['seq'])
        if n_b <= 1024:
            t_cd = min(rem * 0.35, 80)
            print(f"  CD n={n_b}, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_state['seq'], time_limit=t_cd)
            try_evaluate(a_opt, best_state)

    # Phase 5: Fine-tuning
    print(f"\n=== Phase 5: Fine-tune (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 15 and best_state['seq'] is not None:
        t_alloc = min(rem * 0.5, 80)
        a_opt, _ = nesterov_optimize(best_state['seq'], time_limit=t_alloc,
                                      lr0=0.005, mom0=0.92, top_k=5)
        try_evaluate(a_opt, best_state)

        rem = deadline - time.monotonic() - 60
        if rem > 10:
            a_opt, _ = nesterov_optimize(best_state['seq'], time_limit=min(rem, 40),
                                          lr0=0.002, mom0=0.94, top_k=8)
            try_evaluate(a_opt, best_state)

    # Phase 6: Multi-length exploration
    print(f"\n=== Phase 6: Multi-length (best={best_state['score']:.6f}) ===")
    if best_state['seq'] is not None:
        for mult in [2, 3]:
            target_n = len(best_state['seq']) * mult
            if target_n > 8000:
                continue
            rem = deadline - time.monotonic() - 50
            if rem < 15:
                break
            f_interp = interpolate_to_length(best_state['seq'], target_n)
            t_alloc = min(rem * 0.3, 50)
            a_opt, _ = nesterov_optimize(f_interp, time_limit=t_alloc,
                                          lr0=0.01, mom0=0.87, top_k=3)
            try_evaluate(a_opt, best_state)

    # Phase 7: Final polish
    print(f"\n=== Phase 7: Final polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 5 and best_state['seq'] is not None:
        a_opt, _ = nesterov_optimize(best_state['seq'], time_limit=min(rem, 15),
                                      lr0=0.001, mom0=0.95, top_k=10)
        try_evaluate(a_opt, best_state)

    # Final short FW pass
    rem = deadline - time.monotonic() - 5
    if rem > 5 and best_state['seq'] is not None:
        a_opt, _ = frank_wolfe_optimize(best_state['seq'], time_limit=min(rem, 10))
        try_evaluate(a_opt, best_state)

    # Fallback
    if best_state['seq'] is None:
        best_state['seq'] = np.ones(256)
        best_state['score'] = float(evaluate_sequence(best_state['seq'].tolist()))

    print(f"\nFINAL: score={best_state['score']:.8f}, n={len(best_state['seq'])}")
    return [float(x) for x in best_state['seq']]
