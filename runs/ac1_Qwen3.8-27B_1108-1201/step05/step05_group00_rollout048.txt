import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad(a, T):
    """Compute gradient of log-sum-exp smoothed max(conv(a,a)).
    Returns (gradient, convolution)."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = T * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    
    # Cross-correlation corr(p, a) via FFT
    # corr[m] = sum_k p[k+m]*a[k] for m >= 0
    N = 1 << (3 * n - 1).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(P) * A, N)
    grad = 2.0 * corr[:n]
    return grad, conv

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)
    if len(rho[0]) == 0:
        return np.ones(n) / n
    rho = rho[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                          eps=1e-8, T_decay=0.9999, T_min=0.001):
    """Adam optimizer on the log-sum-exp smoothed objective."""
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = compute_grad(a, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = project_simplex(a - step)
        a = a_new
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Decay temperature
        T = max(T * T_decay, T_min)
        
        # Periodic exact check
        if t % 100 == 0:
            exact_obj = fast_convolve(a).max()
            if exact_obj < best_obj:
                best_obj = exact_obj
                best_a = a.copy()
    
    return best_a, best_obj

def subgradient_refine(a0, time_limit, lr_init=0.01):
    """Subgradient method on the exact max objective."""

    a = np.maximum(a, 0.0)

    best_a = a.copy()

        grad = np.zeros(n)
        for i in range(n):

            if 0 <= k_max - i < n:
                gi += a[k_max - i]
            if 0 <= i - k_max < n:

        a = a_new

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        lr *= 0.9999
    
    return best_a, best_obj

The key insight: the maximum of conv(a,a) is at least the average of conv(a,a) over all lags. The average is (1/(2n-1)) * sum_k conv[k] = (1/(2n-1)) * (sum_j a[j])^2 = 1/(2n-1).

But actually, a better lower bound: by the pigeonhole principle or by considering specific lags.

The value at lag n-1 is sum_j a[j]*a[n-1-j]. This is NOT the same as sum_j a[j]^2 unless a is symmetric!

For a symmetric sequence (a[j] = a[n-1-j]), conv[n-1] = sum_j a[j]^2 >= 1/n.
For a non-symmetric sequence, conv[n-1] = sum_j a[j]*a[n-1-j] which can be different.

So the trick is to make a NON-symmetric so that the peak of the convolution is NOT at the center!

```python
def compute_grad(a, T):
    n = len(a)
    conv = fast_convolve(a)
    scaled = T * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    
    # Need FFT size >= (2n-1) + n - 1 = 3n-2 for cross-correlation without wraparound
    N = 1 << (3*n - 1).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    
    # conv(p,a) = IFFT(P * A)
    conv_pa = np.fft.irfft(P * A, N)
    
    # corr(p,a)[i] = sum_k p[k+i]*a[k] = IFFT(conj(P) * A) for the positive lags
    corr_pa = np.fft.irfft(np.conj(P) * A, N)
    
    # grad[i] = corr_pa[i] + conv_pa[i] for i = 0..n-1
    grad = corr_pa[:n] + conv_pa[:n]
    return grad, conv
```
