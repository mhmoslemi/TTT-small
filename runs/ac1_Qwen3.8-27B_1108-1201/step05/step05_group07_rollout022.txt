import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1

    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(f, M):
    """
    Solve the LP to find the flattest non-negative g_0 such that f*g_0 <= M.
    Returns g_0 (unnormalized).
    """
    n = len(f)
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is the cross-convolution matrix
    # A_ub has shape (2n-1, n)
    
    # Build A_ub efficiently
    # For each k, A_ub[k, j] = f[k-j] if 0 <= k-j < n
    # j ranges from max(0, k-n+1) to min(n-1, k)
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A_ub[k, j_start:j_end+1] = f[k - j_end : k - j_start : -1]
    
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g_0)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: return uniform
        return np.ones(n)


def lp_optimize(f0, time_limit, max_iters=5000, verbose=False):
    """
    LP-based optimization: iteratively find the flattest direction
    compatible with the current convolution peak, and update.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    t = 0.3  # step size
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()
        
        # Solve LP
        g_0 = lp_direction(f, M)
        g_sum = g_0.sum()
        
        if g_sum < 1e-10:
            # LP gives trivial solution, try relaxed constraint
            g_0 = lp_direction(f, M * 1.5)
            g_sum = g_0.sum()
            if g_sum < 1e-10:
                break  # Can't make progress
        
        g = g_0 / g_sum
        
        # Line search for optimal step size
        best_t = t
        best_new_obj = best_obj
        for t_candidate in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_trial = (1 - t_candidate) * f + t_candidate * g
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial /= st
            new_conv = fast_convolve(f_trial)
            new_obj = new_conv.max()
            if new_obj < best_new_obj:
                best_new_obj = new_obj
                best_t = t_candidate
        
        # Update
        f = (1 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        f /= f.sum()
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            # Adapt step size
            if best_t < 0.2:
                t = max(0.05, best_t * 0.8)
            elif best_t > 0.5:
                t = min(0.8, best_t * 1.1)
        else:
            no_improve += 1
            if no_improve > 50:
                # Try larger relaxation
                t = min(1.0, t * 1.2)
                if no_improve > 200:
                    break
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, step={best_t:.3f}")
    
    return best_f, best_obj

def compute_max_subgradient(f, conv, top_k):
    """Compute subgradient of max(conv) using top-k peaks."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    # Uniform weights over top-k (simple subgradient)
    w = np.ones(len(peaks)) / len(peaks)
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        k = int(peaks[i])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        grad[i_vals] += 2.0 * w[i] * f[k - i_vals]
    return grad


def smoothed_lbfgs(f0, time_limit, T_init=0.001, T_min=0.00001, verbose=False):
    """
    Smoothed optimization using log-sum-exp approximation with L-BFGS.
    Projects onto simplex after each step.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    T = T_init
    it = 0
    
    while time.monotonic() - t0 < time_limit and T > T_min:
        it += 1
        # Define objective and gradient with current T
        def obj_and_grad(x):
            x = np.maximum(x, 0)
            xs = x.sum()
            if xs < 1e-15:
                return 1e10, np.zeros_like(x)
            x = x / xs
            obj, grad = smooth_logsumexp_obj(x, T)
            return obj, grad
        
        # Run L-BFGS for a few iterations
        result = minimize(
            lambda x: obj_and_grad(x)[0],
            f.copy(),
            jac=lambda x: obj_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 50, 'gtol': 1e-6, 'ftol': 1e-8}
        )
        
        f_new = np.maximum(result.x, 0)
        f_new = f_new / f_new.sum()
        
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
        
        # Decrease temperature
        T *= 0.7
        
        f = f_new
    
    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, try to reduce the max peak
    by adjusting that coordinate.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    conv = fast_convolve(f)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Process coordinates in order of their contribution to peaks
        # Find top peaks and prioritize coordinates that contribute to them
        top_k = min(20, 2*n-1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        # Get coordinates contributing to these peaks
        coord_priority = set()
        for k in peaks:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            coord_priority.update(range(i_start, i_end + 1))
        
        # Try coordinates in priority order
        coords = sorted(coord_priority) + [i for i in range(n) if i not in coord_priority]
        
        for j in coords:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            # Try different values for coordinate j
            for mult in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                
                # Renormalize: keep sum = 1
                # f[j] = new_val, rest scaled proportionally
                rest_sum = 1.0 - old_val
                if rest_sum < 1e-10:
                    continue
                scale = (1.0 - new_val) / rest_sum
                if scale < 0:
                    continue
                
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] = f[mask] * scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                rest_sum = 1.0 - old_val
                if rest_sum > 1e-10:
                    scale = (1.0 - best_j_val) / rest_sum
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] = f[mask] * scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
                    conv = fast_convolve(f)
        
        if not improved:
            break
    
    return best_f, best_obj


def get_parent():

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""

(Frank-Wolfe style), with adaptive step sizing, multi-length interpolation,
cross-correlation-accelerated smoothed gradients, and coordinate descent.
The LP at each step finds the flattest non-negative direction g_0 such that
f*g_0 <= M (current peak), providing a more directed update than subgradient.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build the A_ub matrix for the LP: A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = f[k - j_end:k - j_start:-1]
    return A

def lp_optimize(f0, time_limit, max_iters=10000, verbose=False):
    """
    LP-based Frank-Wolfe optimization. At each step, finds the flattest
    non-negative direction compatible with the current convolution peak,
    then performs a line search update.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    A_ub_cache = build_lp_matrix(f)

    t0 = time.monotonic()
    no_improve = 0
    step_t = 0.3
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        M = conv.max()

        # Solve LP
        g_0 = None
        try:
            A_ub = build_lp_matrix(f)
            b_ub = np.full(2 * n - 1, M)
            c_obj = -np.ones(n)
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                             bounds=[(0, None)] * n,
                             method='highs', options={'presolve': True})
            if result.success:
                g_0 = result.x
        except Exception:
            g_0 = None

        if g_0 is None:
            # Try relaxed constraint
            try:
                A_ub = build_lp_matrix(f)
                b_ub = np.full(2 * n - 1, M * 2.0)
                c_obj = -np.ones(n)
                result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                                 bounds=[(0, None)] * n,
                                 method='highs', options={'presolve': True})
                if result.success:
                    g_0 = result.x
            except Exception:
                pass

        if g_0 is None or g_0.sum() < 1e-15:
            no_improve += 1
            if no_improve > 100:
                break
            continue

        g = g_0 / g_0.sum()

        # Check if direction is close to current (stagnation)
        if np.linalg.norm(g - f, ord=1) < 1e-8:
            no_improve += 1
            if no_improve > 100:
                break
            continue

        # Line search
        best_t = 0.1
        best_new_obj = best_obj
        for t_cand in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_trial = (1.0 - t_cand) * f + t_cand * g
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial /= st
            new_conv = fast_convolve(f_trial)
            new_obj = new_conv.max()
            if new_obj < best_new_obj - 1e-15:
                best_new_obj = new_obj
                best_t = t_cand

        if best_new_obj >= best_obj - 1e-15:
            no_improve += 1
            if no_improve > 200:
                break
            # Try to adjust step size
            if no_improve % 50 == 0:
                step_t = min(1.0, step_t * 1.3)
            continue

        # Update
        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            # Adapt step
            if best_t < 0.15:
                step_t = max(0.03, step_t * 0.9)
            elif best_t > 0.6:
                step_t = min(1.0, step_t * 1.15)
        else:
            no_improve += 1

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, step={best_t:.3f}, no_impr={no_improve}")

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, top_k=50, temperature=0.0001, lr=0.002,
                         momentum=0.9, verbose=False):
    """
    Multiplicative-weights subgradient optimization with Nesterov momentum.
    Used as complement to LP when LP stagnates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)

        # Compute subgradient with softmax weighting
        k_actual = min(top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]

        # Update momentum
        v = momentum * v + (1 - momentum) * sg

        # Multiplicative weights update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, 0.0001)
            no_improve += 1
            if no_improve > 200:
                break
            continue
        f /= fs
        f = np.maximum(f, 0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_lr *= 0.9997
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp *= 1.005
        if it % 500 == 0:
            top_k = max(5, int(top_k * 0.95))

        if no_improve > 400:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.002 + 0.001 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_temp = temperature * 0.5
            conv = fast_convolve(f)
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent with peak-aware ordering."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    conv = fast_convolve(f)
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        top_k = min(30, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Priority: coordinates contributing most to peaks
        coord_score = np.zeros(n)
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            coord_score[idx] += contribs

        order = np.argsort(-coord_score)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 0.2]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                rest_sum = 1.0 - old_val
                if rest_sum < 1e-10:
                    continue
                scale = (1.0 - new_val) / rest_sum
                if scale < 0:
                    continue

                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                rest_sum = 1.0 - old_val
                if rest_sum > 1e-10:
                    scale = (1.0 - best_j_val) / rest_sum
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
                    conv = fast_convolve(f)

        if not improved:
            break

    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            a_opt, obj_opt = lp_optimize(f_init, time_limit=t_alloc, max_iters=10000, verbose=False)
            try_seq(a_opt, f"lp_r{restart}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [1024, 1280, 1536]:
            rem = deadline - time.monotonic() - 300
            if rem < 50:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 100)
            print(f"  LP on n={target_n}, budget={t_alloc:.0f}s")
            a_opt, obj_opt = lp_optimize(f_interp, time_limit=t_alloc, max_iters=8000, verbose=False)
            try_seq(a_opt, f"lp_n{target_n}")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.5, 100)
            print(f"  CD, budget={t_cd:.0f}s")
            a_opt, obj_opt = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_p5")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_final = min(rem * 0.6, 60)
            print(f"  Final LP, budget={t_final:.0f}s")
            a_opt, obj_opt = lp_optimize(best_seq, time_limit=t_final, max_iters=5000, verbose=False)
            try_seq(a_opt, "lp_final")

    # === Phase 7: Perturbation + quick LP ===
    print(f"\n=== Phase 7: Perturbation Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for scale in [0.001, 0.002, 0.005, 0.01]:
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            for trial in range(3):
                rem = deadline - time.monotonic() - 10
                if rem < 8:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)
                t_alloc = min(rem, 20)
                a_opt, obj_opt = lp_optimize(f_pert, time_limit=t_alloc, max_iters=2000, verbose=False)
                try_seq(a_opt, f"pert_{scale}_{trial}")

    # === Phase 8: Final coordinate descent ===
    print(f"\n=== Phase 8: Final CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, obj_opt = coordinate_descent_polish(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
