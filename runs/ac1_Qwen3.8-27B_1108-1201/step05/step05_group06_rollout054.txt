def lp_direction(f, M):
    """Solve LP to find best direction.
    Maximize sum(b) s.t. b >= 0, (f*b)[k] <= M for all k.
    """
    n = len(f)
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
    # (f*b)[k] = sum_i f[i] * b[k-i] (with appropriate bounds)
    # This is a sparse constraint matrix
    
    # For constraint k: sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * b[k-i] <= M
    # The coefficient of b[j] in constraint k is f[k-j] if 0 <= k-j <= n-1, else 0
    
    # Build sparse matrix
    from scipy import sparse
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_vec = np.full(2*n-1, M)
    
    # Solve LP
    result = scipy.optimize.linprog(c, A_ub=A, b_ub=b_vec, bounds=[(0, None)]*n, method='hihgs')
    
    if result.status == 0:
        return result.x
    else:
        return None

1. Compute gradient g = grad f(x)
2. Find s = argmin_{v in P} g^T v (the linear minimization oracle)
3. Update: x_new = (1-t)*x + t*s

Wait, that's for the standard simplex. The Frank-Wolfe step would be to move towards the coordinate with the smallest gradient. This is actually just a variant of coordinate descent!

More specifically, at each step:
- Compute gradient g = grad f(x)
- Find j* = argmin_j g_j (the coordinate where the function most wants to decrease)
- Move: x_new = (1-t)*x + t*e_{j*}

This is exactly the "greedy coordinate descent" where we always pick the coordinate with the best subgradient.

- If x has support S, try both:
  a. Move towards e_{j*} (j* = argmin g)
  b. Move away from e_{j_away} (j_away = argmax_{j in S} g)
- Pick the better one

This is known to converge faster than vanilla FW.

Uses scipy HiGHS LP solver for direction finding, combined with fast coordinate
descent and multi-length exploration.
"""

import numpy as np
import time
import scipy.optimize
from scipy import sparse

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(f, n):
    """Build (2n-1)x n matrix where A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_weighted_gradient(a, conv, top_k=50, temp=0.001):
    """Compute weighted subgradient of max(conv)."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    g = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        # grad q_k / da_j = 2 * a[k-j] if 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * w[i] * a[k - idx]
    return g

def lp_find_direction(a, M, max_time=5.0):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a, n)
    b_vec = np.full(2*n-1, M)
    
    try:
        result = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b_vec,
            bounds=[(0, None)]*n,
            method='hihgs',  # typo? let me check
        )
    except Exception:
        result = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b_vec,
            bounds=[(0, None)]*n,
            method='highs',
        )
    
    if result.status == 0 and result.x is not None:
        return result.x
    return None

def lp_guided_descent(a0, time_limit, verbose=False):
    """Main optimization using LP-guided direction finding."""
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    
    # Step size schedule
    base_t = 0.3
    t_decay = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        # Recompute convolution
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Restart with noise
            rng = np.random.default_rng(iteration * 31 + 7)
            noise_scale = 0.01 * min(no_improve / 100, 3.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            no_improve = 0
            base_t = 0.3
            continue
        
        # Solve LP for direction
        M = cur_obj
        b_dir = lp_find_direction(a, M)
        
        if b_dir is None or b_dir.sum() < 1e-10:
            # Fallback to subgradient
            g = compute_weighted_gradient(a, conv, top_k=50, temp=0.001)
            b_dir = project_simplex(-g)
        else:
            b_dir = normalize(b_dir)
        
        # Line search: try several step sizes
        cur_t = base_t
        best_t = 0.0
        best_trial_obj = cur_obj
        
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            a_trial = normalize((1-t)*a + t*b_dir)
            trial_conv = fast_convolve(a_trial)
            trial_obj = trial_conv.max()
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_t = t
        
        if best_t > 0:
            a = normalize((1-best_t)*a + best_t*b_dir)
            conv = fast_convolve(a)
            if conv.max() < best_obj - 1e-10:
                best_obj = conv.max()
                best_a = a.copy()
                no_improve = 0
                base_t = min(0.5, base_t * 1.1)  # increase step size
            else:
                base_t *= 0.9  # decrease step size
        else:
            no_improve += 1
            base_t *= 0.8
        
        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration}: obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_obj

def afw_descent(a0, time_limit, verbose=False, temp=0.001, top_k=50):
    """Away-step Frank-Wolfe descent."""
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        cur_temp *= 1.002
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            rng = np.random.default_rng(iteration * 17 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
            cur_temp = temp
            continue
        
        # Compute gradient
        g = compute_weighted_gradient(a, conv, top_k=top_k, temp=cur_temp)
        
        # Find towards direction
        j_star = int(np.argmin(g))
        
        # Find away direction (among non-zero coords)
        support_mask = a > 1e-10
        if support_mask.sum() > 1:
            g_support = np.where(support_mask, g, -np.inf)
            j_away = int(np.argmax(g_support))
        else:
            j_away = None
        
        # Try towards step
        best_trial_obj = cur_obj
        best_update = None
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_trial = a.copy()
            a_trial[j_star] += t * (1.0 - a[j_star])  # move towards e_{j_star}
            # Actually, FW step: a_new = (1-t)*a + t*e_{j_star}
            a_trial = (1-t)*a
            a_trial[j_star] += t
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_update = ('towards', t, j_star)
        
        # Try away step
        if j_away is not None:
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                a_trial = a.copy()
                # Move away from e_{j_away}: a_new = (1-t/(n-1))*a + ... 
                # AFW away step: a_new = (1 - t/(1-t_frac))*a ... 
                # Simpler: a_new[j_away] = (1-t)*a[j_away], redistribute to others
                # Standard AFW: a_new = (1-t)*a, then a_new[j_away] = 0, rescale
                # Actually the standard away step is:
                # a_new = (1 - t/(n-1)) * a - t/(n-1) * a[j_away] * ... 
                # Let me just do: reduce a[j_away] by t*a[j_away], add proportionally to others
                delta = t * a[j_away]
                a_trial = a.copy()
                a_trial[j_away] -= delta
                others = support_mask.copy()
                others[j_away] = False
                if others.sum() > 0:
                    a_trial[others] += delta / others.sum()
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial /= st
                trial_obj = fast_convolve(a_trial).max()
                if trial_obj < best_trial_obj:
                    best_trial_obj = trial_obj
                    best_update = ('away', t, j_away)
        
        if best_update is not None:
            kind, t, j = best_update
            if kind == 'towards':
                a = (1-t)*a
                a[j] += t
            else:
                delta = t * a[j]
                a[j] -= delta
                others = np.ones(n, dtype=bool)
                others[j] = False
                a[others] += delta / (n-1)
                a = np.maximum(a, 0)
                st = a.sum()
                if st > 1e-15:
                    a /= st
            
            conv = fast_convolve(a)
            if conv.max() < best_obj - 1e-12:
                best_obj = conv.max()
                best_a = a.copy()
                no_improve = 0
        
        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AFW iter {iteration}: obj={best_obj:.10f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_obj

def coordinate_bisection(a0, time_limit, verbose=False, max_passes=200):
    """Exact coordinate descent with bisection per coordinate."""
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-14:
                continue
            
            s_others = 1.0 - a[j]
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end-start]
            
            lo = 0.0
            hi = 1.0
            x_new = a[j]
            
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om*om*B + 2.0*mid*om*b_valid
                if 2*j < conv_len:
                    f_vals[2*j] += mid*mid
                k_max = int(np.argmax(f_vals))
                b_kj = b[k_max-j] if 0 <= k_max-j < n else 0.0
                B_k = B[k_max]
                ind = 1.0 if (2*j < conv_len and k_max == 2*j) else 0.0
                deriv = 2.0*(b_kj - B_k + mid*(B_k - 2.0*b_kj + ind))
                if deriv > 1e-13:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-13:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break
            
            if abs(x_new - a[j]) < 1e-14:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0)
            st = a.sum()
            if st > 1e-15:
                a /= st
            
            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            if passes > 2:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0005)
            else:
                break
    
    return best_a, best_obj

def greedy_coordinate_descent(a0, time_limit, verbose=False):
    """Greedy coordinate descent: always pick the best coordinate to update."""
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            rng = np.random.default_rng(iter_count * 13 + 5)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
            temp = 0.001
            continue
        
        temp *= 1.003
        
        # Compute gradient and find best coordinate
        g = compute_weighted_gradient(a, conv, top_k=30, temp=temp)
        j_star = int(np.argmin(g))
        
        # Binary search for optimal value of a[j_star]
        j = j_star
        x_old = a[j]
        if x_old < 1e-14:
            # Try setting it to a small value
            a[j] = 1.0/n
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - a[j]) / (1.0 - 0)  # approximately
            a = np.maximum(a, 0)
            st = a.sum()
            if st > 1e-15:
                a /= st
            continue
        
        s_others = 1.0 - x_old
        if s_others < 1e-12:
            continue
        
        b = a.copy()
        b[j] = 0.0
        b = b / s_others
        B = fast_convolve(b)
        conv_len = 2 * n - 1
        b_valid = np.zeros(conv_len)
        start = j
        end = min(j + n, conv_len)
        b_valid[start:end] = b[:end-start]
        
        lo = 0.0
        hi = 1.0
        x_new = x_old
        for _ in range(35):
            mid = (lo + hi) / 2.0
            om = 1.0 - mid
            f_vals = om*om*B + 2.0*mid*om*b_valid
            if 2*j < conv_len:
                f_vals[2*j] += mid*mid
            k_max = int(np.argmax(f_vals))
            b_kj = b[k_max-j] if 0 <= k_max-j < n else 0.0
            B_k = B[k_max]
            ind = 1.0 if (2*j < conv_len and k_max == 2*j) else 0.0
            deriv = 2.0*(b_kj - B_k + mid*(B_k - 2.0*b_kj + ind))
            if deriv > 1e-13:
                hi = mid
                x_new = mid
            elif deriv < -1e-13:
                lo = mid
                x_new = mid
            else:
                x_new = mid
                break
        
        a[j] = x_new
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        a[mask] *= (1.0 - x_new) / s_others
        a = np.maximum(a, 0)
        st = a.sum()
        if st > 1e-15:
            a /= st
    
    return best_a, best_obj

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def symmetrize(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

"""
LP-guided descent with AFW polishing and coordinate bisection for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses scipy HiGHS solver for LP direction finding,
away-step Frank-Wolfe for fast intermediate optimization, and exact coordinate
bisection for final polishing. Multi-length exploration and restarts for robustness.
"""

import numpy as np
import time
import scipy.optimize
from scipy import sparse


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(f, n):
    """Build sparse (2n-1)x n matrix where A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            k = i + j
            if f[i] != 0.0:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def build_conv_matrix_dense(f, n):
    """Build dense (2n-1)x n convolution matrix."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def compute_weighted_gradient(a, conv, top_k=50, temp=0.001):
    """Compute weighted subgradient of max(conv)."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    g = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * w[i] * a[k - idx]
    return g


def solve_direction_lp(a, M, use_sparse=False):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    n = len(a)
    c = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)

    if use_sparse:
        A = build_conv_matrix_sparse(a, n)
    else:
        A = build_conv_matrix_dense(a, n)

    try:
        result = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b_vec,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': 10}
        )
    except Exception:
        return None

    if result.status == 0 and result.x is not None:
        return result.x
    return None


def lp_guided_descent(a0, time_limit, verbose=False, max_lp_iters=None):
    """LP-guided descent: solve LP for direction, line search for step size."""
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    base_t = 0.3
    lp_iters = 0
    use_sparse = n > 512

    while time.monotonic() - t0 < time_limit:
        iteration += 1

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(iteration * 31 + 7)
            noise_scale = 0.005 + 0.01 * min(no_improve // 200, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            no_improve = 0
            base_t = 0.3
            continue

        M = cur_obj
        b_dir = solve_direction_lp(a, M, use_sparse=use_sparse)
        lp_iters += 1

        if b_dir is not None and b_dir.sum() > 1e-10:
            b_dir = normalize(b_dir)
        else:
            g = compute_weighted_gradient(a, conv, top_k=50, temp=0.001)
            b_dir = project_simplex(-g)

        cur_t = base_t
        best_t = 0.0
        best_trial_obj = cur_obj

        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_trial = normalize((1 - t) * best_a + t * b_dir)
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_t = t

        if best_t > 0:
            a = normalize((1 - best_t) * best_a + best_t * b_dir)
            conv = fast_convolve(a)
            cur_obj = conv.max()
            if cur_obj < best_obj - 1e-10:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
                base_t = min(0.8, base_t * 1.05)
            else:
                base_t = max(0.05, base_t * 0.85)
                a = best_a.copy()
                conv = fast_convolve(a)
        else:
            no_improve += 5
            base_t = max(0.05, base_t * 0.8)

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration} ({lp_iters} LPs): obj={best_obj:.10f}, t={best_t:.3f}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def afw_descent(a0, time_limit, verbose=False, temp_init=0.001, top_k=50):
    """Away-step Frank-Wolfe descent with adaptive temperature."""
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    cur_temp = temp_init

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        cur_temp = min(cur_temp * 1.003, 0.1)

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            rng = np.random.default_rng(iteration * 17 + 3)
            noise_scale = 0.002 + 0.003 * min(no_improve // 100, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            no_improve = 0
            cur_temp = temp_init
            continue

        g = compute_weighted_gradient(a, conv, top_k=top_k, temp=cur_temp)
        j_star = int(np.argmin(g))

        support_mask = a > 1e-10
        n_support = support_mask.sum()

        best_trial_obj = cur_obj
        best_update = None

        for t in [0.15, 0.3, 0.5, 0.7, 0.9]:
            a_trial = (1 - t) * best_a
            a_trial[j_star] += t
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_update = ('towards', t, j_star)

        if n_support > 2:
            g_support = np.where(support_mask, g, np.inf)
            j_away = int(np.argmin(-g_support))
            if a[j_away] > 1e-8:
                for t in [0.1, 0.2, 0.4, 0.6, 0.8]:
                    delta = t * best_a[j_away]
                    a_trial = best_a.copy()
                    a_trial[j_away] -= delta
                    others = support_mask.copy()
                    others[j_away] = False
                    n_others = others.sum()
                    if n_others > 0:
                        a_trial[others] += delta / n_others
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 1e-15:
                        a_trial /= st
                    trial_obj = fast_convolve(a_trial).max()
                    if trial_obj < best_trial_obj:
                        best_trial_obj = trial_obj
                        best_update = ('away', t, j_away)

        if best_update is not None:
            kind, t, j = best_update
            if kind == 'towards':
                a = (1 - t) * best_a
                a[j] += t
            else:
                delta = t * best_a[j]
                a = best_a.copy()
                a[j] -= delta
                others = support_mask.copy()
                others[j] = False
                n_others = others.sum()
                if n_others > 0:
                    a[others] += delta / n_others
                a = np.maximum(a, 0)
                st = a.sum()
                if st > 1e-15:
                    a /= st

            conv = fast_convolve(a)
            if conv.max() < best_obj - 1e-12:
                best_obj = conv.max()
                best_a = a.copy()
                no_improve = 0

        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AFW iter {iteration}: obj={best_obj:.10f}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def coord_bisect_single(a, j, n, B, b_valid, conv_len):
    """Bisection to find optimal value for coordinate j given others fixed."""
    x_old = a[j]
    s_others = 1.0 - x_old
    if s_others < 1e-12:
        return x_old

    lo = 0.0
    hi = 1.0
    x_new = x_old

    for _ in range(40):
        mid = (lo + hi) / 2.0
        om = 1.0 - mid
        f_vals = om * om * B + 2.0 * mid * om * b_valid
        if 2 * j < conv_len:
            f_vals[2 * j] += mid * mid
        k_max = int(np.argmax(f_vals))
        b_kj = b[k_max - j]] if 0 <= k_max - j < n else 0.0
        B_k = B[k_max]
        ind = 1.0 if (2 * j < conv_len and k_max == 2 * j) else 0.0
        deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
        if deriv > 1e-13:
            hi = mid
            x_new = mid
        elif deriv < -1e-13:
            lo = mid
            x_new = mid
        else:
            x_new = mid
            break

    return x_new


def coordinate_bisection(a0, time_limit, verbose=False, max_passes=300):
    """Exact coordinate descent with bisection per coordinate."""
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    conv_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-14:
                continue

            s_others = 1.0 - a[j]
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            x_new = coord_bisect_single(a, j, n, B, b_valid, b, conv_len)

            if abs(x_new - a[j]) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0)
            st = a.sum()
            if st > 1e-15:
                a /= st

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0004)
            else:
                break

    return best_a, best_obj


def greedy_cd(a0, time_limit, verbose=False):

    a = normalize(a0.copy().astype(np.float64))

    conv_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        cur_temp = min(temp * (1 + 0.002 * iteration), 0.05)

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(iteration * 13 + 5)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
            continue

        g = compute_weighted_gradient(a, conv, top_k=30, temp=cur_temp)
        j = int(np.argmin(g))

        if a[j] < 1e-14:

            others_sum = 1.0 - a[j]
            if others_sum > 1e-12:
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= others_sum / a[mask].sum() if a[mask].sum() > 1e-15 else 0

        s_others = 1.0 - a[j]

        b = a.copy()
        b[j] = 0.0

        b_valid = np.zeros(conv_len)
        start = j
        end = min(j + n, conv_len)
        b_valid[start:end] = b[:end - start]

        if abs(x_new - a[j]) > 1e-10:

            mask = np.ones(n, dtype=bool)

            a[mask] *= (1.0 - x_new) / s_others
            a = normalize(a)

        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Greedy CD iter {iteration}: obj={best_obj:.10f}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new
