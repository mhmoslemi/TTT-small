Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-Frank-Wolfe with

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated uniform initialization on the simplex."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def build_lp_matrix(f, n):
    """Build dense (2n-1) x n constraint matrix for LP direction finding.
    A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_solve_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def nesterov_subgrad(f0, time_limit=120, lr0=0.005, mom0=0.85, top_k=30, temp0=0.0001):
    """Nesterov-accelerated projected subgradient with multi-peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Multi-peak weighted subgradient
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * w[i] * f[k - indices]

        # Nesterov momentum update
        v = cur_mom * v - cur_lr * sg
        f_candidate = y + v
        f_new = project_simplex(f_candidate)
        y = project_simplex(f_new + cur_mom * (f_new - f))
        f = f_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003
        cur_mom = min(0.97, cur_mom * 1.0002)

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999997)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.7, 0.001)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def coordinate_descent(f0, time_limit=60, max_passes=10):
    """Cyclic coordinate descent with golden section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    no_pass = 0

    for _pass in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old = f[j]
            if old < 1e-14:
                continue
            denom = 1.0 - old
            if denom < 1e-12:
                continue

            lo = 0.0
            hi = min(0.999, 2.5 * old)
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            # Cache f without j
            f_noj = np.ones(n)
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            f_noj[mask] = f[mask] / denom

            def eval_cd(x):
                ft = f_noj * (1.0 - x)
                ft[j] = x
                return fast_convolve(ft).max()

            f1 = eval_cd(c1)
            f2 = eval_cd(c2)
            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_cd(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_cd(c2)

            opt_x = (lo + hi) / 2.0
            cur_obj = fast_convolve(f).max()
            if min(f1, f2) < cur_obj - 1e-13:
                ft = f_noj * (1.0 - opt_x)
                ft[j] = opt_x
                ft = ft / ft.sum()
                f = ft
                improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_f = f.copy()

        if improved:
            no_pass = 0
        else:
            no_pass += 1
            if no_pass >= 2:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            elif p == 1:
                f_init = 0.8 * best_seq + 0.2 * cosine_modulated_init(n_best, 2, 0.3)
                f_init = project_simplex(f_init)
            elif p == 2:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init)
            else:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)

            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"multi_{p}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=min(rem, 40),
                                         lr0=0.001, mom0=0.93,
                                         top_k=max(10, n_best // 6),
                                         temp0=0.00001)
            try_seq(a_opt, "final_nest")

        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 25))
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
