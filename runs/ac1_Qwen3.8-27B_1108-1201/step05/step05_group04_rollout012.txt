def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent_step(f, grad, lr):
    """Exponential update on the simplex (mirror descent)."""
    # f_new[i] = f[i] * exp(-lr * grad[i]) / Z

    unnormalized = log_f - lr * grad
    max_val = unnormalized.max()
    exp_vals = np.exp(unnormalized - max_val)
    f_new = exp_vals / exp_vals.sum()
    return f_new

def compute_grad_fft(f, w):
    """Compute gradient: grad[j] = 2 * sum_k w[k] * f[k-j] for valid indices.
    w has length 2n-1."""
    n = len(f)

Optimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient computation for the LSE-smoothed objective
2. Mirror descent (exponential updates on simplex) + Euclidean projection hybrid
3. Adaptive backtracking line search

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_fft(f, w):
    """Compute gradient of sum_k w[k] * (f*f)[k] using FFT.
    w has length 2n-1. Returns gradient of length n.
    grad[j] = 2 * sum_k w[k] * f[k-j] for valid k-j in [0,n-1].
    """
    n = len(f)
    conv_len = 2 * n - 1
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    fw = np.fft.rfft(np.flip(w), fft_size)
    fa = np.fft.rfft(f, fft_size)
    conv = np.fft.irfft(fw * fa, fft_size)
    c = conv[2 * n - 2 : n - 2 : -1]
    return 2.0 * c

def lse_objective_and_grad(f, beta, top_k=None):
    """Compute LSE-smoothed objective and gradient.
    Returns (lse_value, gradient).
    """
    n = len(f)
    conv = fast_convolve(f)
    conv_len = 2 * n - 1
    
    # LSE
    scaled = beta * conv
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    lse_val = (max_s + np.log(exp_s.sum())) / beta
    
    # Weights for gradient
    w = exp_s / exp_s.sum()
    
    # Optionally restrict to top-k peaks
    if top_k is not None and top_k < conv_len:
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_masked = np.zeros(conv_len)
        w_masked[idx] = w[idx]
        w_masked /= w_masked.sum()
        w = w_masked
    
    grad = compute_grad_fft(f, w)
    return lse_val, grad

def mirror_step(f, grad, lr):
    """Mirror descent step: f_new = f * exp(-lr * grad) normalized."""
    log_f = np.log(f + 1e-30)
    unnorm = log_f - lr * grad
    max_u = unnorm.max()
    exp_u = np.exp(unnorm - max_u)
    f_new = exp_u / exp_u.sum()
    return f_new

def euclidean_step(f, grad, lr):
    """Euclidean projection step."""
    return project_simplex(f - lr * grad)

def optimize_lse(f0, time_limit, beta_start=10.0, beta_end=10000.0, 
                 step_type='mirror', lr_init=None, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    if lr_init is None:
        lr_init = 1.0 / np.sqrt(n)
    
    conv = fast_convolve(f)
    best_conv_max = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        conv_max = conv.max()
        
        # LSE weights
        scaled = beta * conv
        max_s = scaled.max()
        exp_s = np.exp(scaled - max_s)
        w = exp_s / exp_s.sum()
        
        # Gradient
        grad = compute_grad_fft(f, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm  # normalize gradient
        
        # Try step
        f_new = mirror_step(f, grad, lr) if step_type == 'mirror' else euclidean_step(f, grad, lr)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_conv_max - 1e-12:
            best_conv_max = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slowly increase lr
            lr = min(lr * 1.01, 2.0 / np.sqrt(n))
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 100:
                # Restart from best with perturbation
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                lr = lr_init
                no_improve = 0

    s = f_new.sum()

    else:

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/(sum(a))²`. Key insights:

1. The objective is scale-invariant, so I normalize to sum=1
2. FFT-based gradient computation gives O(n log n) per step (much faster than O(n*peaks))
3. Mirror descent (exponential updates) is naturally adapted to the simplex
4. Larger n with similar relative optimization quality gives better scores

"""
Optimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based gradient computation (O(n log n) per step)
2. Mirror descent + Euclidean projection hybrid on simplex
3. Adaptive beta schedule + backtracking line search
4. Multi-length exploration (up to n=3000) with coarse-to-fine
5. Directed peak refinement and high-beta final polishing
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(f, w):
    """Compute gradient of sum_k w[k]*(f*f)[k] using FFT. O(n log n).
    w has length 2n-1. Returns gradient of length n.
    """
    n = len(f)
    conv_len = 2 * n - 1
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    fw = np.fft.rfft(np.flip(w), fft_size)
    fa = np.fft.rfft(f, fft_size)
    conv = np.fft.irfft(fw * fa, fft_size)
    c = conv[2 * n - 2 : n - 2 : -1]
    return 2.0 * c


def mirror_step(f, grad, lr):
    """Mirror descent step on simplex."""
    log_f = np.log(f + 1e-30)
    unnorm = log_f - lr * grad
    max_u = unnorm.max()
    exp_u = np.exp(unnorm - max_u)
    return exp_u / exp_u.sum()


def euclidean_step(f, grad, lr):
    """Euclidean projection step on simplex."""
    return project_simplex(f - lr * grad)


def optimize_lse(f0, time_limit, beta_start=10.0, beta_end=10000.0,
                 step_type='mirror', lr_init=None, top_k=None, verbose=False):
    """LSE-smoothed optimization with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 1.0 / np.sqrt(n)

    conv = fast_convolve(f)
    best_conv_max = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # LSE weights
        scaled = beta * conv
        max_s = scaled.max()
        exp_s = np.exp(scaled - max_s)
        w = exp_s / exp_s.sum()

        # Optionally focus on top-k peaks
        if top_k is not None and top_k < 2 * n - 1:
            idx = np.argpartition(w, -top_k)[-top_k:]
            w_masked = np.zeros(2 * n - 1)
            w_masked[idx] = w[idx]
            s = w_masked.sum()
            if s > 1e-15:
                w = w_masked / s

        grad = compute_grad_fft(f, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        f_new = mirror_step(f, grad, lr) if step_type == 'mirror' else euclidean_step(f, grad, lr)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_conv_max - 1e-13:
            best_conv_max = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.005, 3.0 / np.sqrt(n))
        else:
            no_improve += 1
            lr *= 0.96
            if no_improve > 80:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.003 + 0.004 * min(1.0, (no_improve - 80) / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                lr = lr_init
                no_improve = 0

        # Beta schedule
        if beta < beta_end:
            beta *= 1.003
            if it % 50 == 0:
                beta = min(beta * 1.15, beta_end)

    return best_f, best_conv_max


def optimize_directed(f0, time_limit, n_peaks=15, lr_init=None, momentum=0.92,
                      step_type='mirror'):
    """Directed peak optimization focusing on top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 0.5 / np.sqrt(n)

    conv = fast_convolve(f)
    best_conv_max = conv.max()
    best_f = f.copy()

    lr = lr_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        k_top = min(n_peaks, 2 * n - 1)
        peaks_idx = np.argpartition(conv, -k_top)[-k_top:]
        peak_vals = conv[peaks_idx]

        # Exponential weighting emphasizing highest peaks
        w_peaks = np.exp((peak_vals - peak_vals.max()) * 8.0)
        w_peaks /= w_peaks.sum()

        w_full = np.zeros(2 * n - 1)
        w_full[peaks_idx] = w_peaks

        grad = compute_grad_fft(f, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        v = momentum * v - lr * grad
        if step_type == 'mirror':
            f_new = mirror_step(f, grad, lr)
        else:
            f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_conv_max - 1e-13:
            best_conv_max = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.98
            if no_improve > 150:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0

    return best_f, best_conv_max


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    return score
