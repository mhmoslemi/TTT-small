def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                # Random exploration
                rng = np.random.default_rng(it * 2654435761)
                for _ in range(min(30, n // 8)):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.6, 0.8, 1.2, 1.5]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.5:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        ss = f_trial.sum()
                        if ss < 1e-15:
                            continue
                        f_trial /= ss
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved_this_round = True
                            break
                    if improved_this_round:
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def subgradient_optimize(a0, time_limit, lr=0.003, momentum=0.94,
                         top_k=50, start_temp=0.00005, verbose=False):
    """Subgradient descent with Nesterov momentum and softmax peaks."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = start_temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = compute_subgradient(a, peaks, w)
        v = momentum * v - cur_lr * sg
        a_new = project_simplex(a + v)

        conv = fast_convolve(a_new)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            a = a_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(it * 69069 + 17)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_a, best_obj


def propose_candidate():
    """
    Multi-phase optimization:
    1. Mirror descent (main, ~700s)
    2. Subgradient refinement (~300s)
    3. LP-FW polish (~200s)
    4. Length exploration (~300s)
    5. Final CD polish (~150s)
    """
    rng = np.random.default_rng(12345)
    budget = 1800.0
    run_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - run_start
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)}, t={elapsed:.0f}s)")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Mirror Descent (main optimization) ===
    print("\n=== Phase 1: Mirror Descent ===")
    rem = budget - (time.monotonic() - run_start)
    t1 = min(700, rem * 0.4)
    if t1 > 50:
        print(f"  n={len(parent)}, t={t1:.0f}s")
        # Run multiple MD rounds with different top_k
        round_time = t1 / 3
        for rnd in range(3):
            rem = budget - (time.monotonic() - run_start)
            if rem < 30:
                break
            t_alloc = min(round_time, rem)
            tk = [40, 80, 120][rnd] if rnd < 3 else 60
            et = [0.004, 0.003, 0.002][rnd]
            a_opt, _ = mirror_descent_optimize(parent if rnd == 0 else best_seq,
                                               time_limit=t_alloc,
                                               eta0=et, top_k=tk)
            try_seq(a_opt, f"md_r{rnd}_tk{tk}")
            parent = best_seq  # warm start next round

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Subgradient (best={best_score:.8f}) ===")
    rem = budget - (time.monotonic() - run_start)
    t2 = min(300, rem * 0.2)
    if t2 > 30 and best_seq is not None:
        n_best = len(best_seq)
        print(f"  n={n_best}, t={t2:.0f}s")
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t2,
                                        lr=0.003, momentum=0.94,
                                        top_k=max(30, n_best // 8),
                                        start_temp=0.00004)
        try_seq(a_opt, "sg_main")

        # Second pass with finer params
        rem = budget - (time.monotonic() - run_start)
        t2b = min(150, rem * 0.15)
        if t2b > 20:
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t2b,
                                            lr=0.001, momentum=0.95,
                                            top_k=max(20, n_best // 10),
                                            start_temp=0.00002)
            try_seq(a_opt, "sg_fine")

    # === Phase 3: LP-FW polish ===
    print(f"\n=== Phase 3: LP-FW Polish (best={best_score:.8f}) ===")
    rem = budget - (time.monotonic() - run_start)
    t3 = min(200, rem * 0.15)
    if t3 > 30 and best_seq is not None:
        n_best = len(best_seq)
        print(f"  n={n_best}, t={t3:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
        try_seq(a_opt, "lpfw")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length Exploration (best={best_score:.8f}) ===")
    rem = budget - (time.monotonic() - run_start)
    base = best_seq if best_seq is not None else parent
    base_n = len(base)

    # Try nearby lengths
    candidate_lengths = []
    for delta in [-64, -32, 32, 64, 128, 192, 256]:
        ln = base_n + delta
        if 256 <= ln <= 2048:
            candidate_lengths.append(ln)
    candidate_lengths = sorted(set(candidate_lengths))

    for target_n in candidate_lengths[:5]:
        rem = budget - (time.monotonic() - run_start)
        if rem < 60:
            break
        t_alloc = min(rem * 0.25, 150)
        if t_alloc < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  MD n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent_optimize(f_interp, time_limit=t_alloc,
                                           eta0=0.003, top_k=60)
        try_seq(a_opt, f"len_{target_n}")

        # Quick SG on the best from this length
        rem = budget - (time.monotonic() - run_start)
        if rem > 40 and best_seq is not None and len(best_seq) == target_n:
            t_sg = min(rem * 0.3, 80)
            if t_sg > 20:
                a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg,
                                                lr=0.002, momentum=0.94,
                                                top_k=max(20, target_n // 8),
                                                start_temp=0.00003)
                try_seq(a_opt, f"sg_len_{target_n}")

    # === Phase 5: Final combined polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    rem = budget - (time.monotonic() - run_start)
    if best_seq is not None and rem > 50:
        n_best = len(best_seq)

        # Mirror descent polish
        t_md = min(rem * 0.3, 120)
        if t_md > 20:
            print(f"  MD polish n={n_best}, t={t_md:.0f}s")
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_md,
                                               eta0=0.001, top_k=40)
            try_seq(a_opt, "md_polish")

        # Coordinate descent
        rem = budget - (time.monotonic() - run_start)
        t_cd = min(rem * 0.3, 120)
        if t_cd > 20 and best_seq is not None:
            print(f"  CD polish n={len(best_seq)}, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_polish")

        # Small perturbations + MD
        for p in range(4):
            rem = budget - (time.monotonic() - run_start)
            if rem < 20:
                break
            noise = rng.standard_normal(len(best_seq)) * (0.0005 + 0.0003 * p)
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.2, 40)
            if t_pert > 10:
                a_opt, _ = mirror_descent_optimize(f_pert, time_limit=t_pert,
                                                   eta0=0.001, top_k=30)
                try_seq(a_opt, f"perturb_{p}")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    elapsed = time.monotonic() - run_start
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, elapsed={elapsed:.0f}s")
    return [float(x) for x in best_seq]
