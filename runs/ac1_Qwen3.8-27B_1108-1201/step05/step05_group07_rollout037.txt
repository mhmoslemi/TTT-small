```python
def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_shaping(f0, time_limit, alpha_init=0.7, alpha_final=0.4, 
                     n_iter_max=50000, verbose=False):
    """
    Iterative time-frequency projection:
    - Time domain: enforce non-negativity and normalization
    - Frequency domain: compress the power spectrum (flatten peaks)
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Forward FFT
        A = np.fft.rfft(f, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compute power spectrum
        power = mag ** 2
        
        # Compress: raise to power alpha < 1 to reduce dynamic range
        # Adaptive alpha: start high (less compression) and decrease
        progress = min(it / n_iter_max, 1.0)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        
        # Normalize power before compression
        power_norm = power / power.sum()
        power_compressed = power_norm ** alpha
        power_compressed = power_compressed / power_compressed.sum()
        
        # New magnitude (geometric interpolation with original)
        # mag_new = mag^alpha * (normalized)
        mag_new = np.sqrt(power_compressed * power.sum())
        
        # Blend with original magnitude (to preserve some structure)
        blend = 0.5  # 50% new, 50% original
        mag_new = blend * mag_new + (1 - blend) * mag
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project onto non-negative, sum=1
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f = best_f.copy()
            no_improve += 100
            continue
        f = f_new / s_new
        
        # Evaluate
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and it % 1000 == 0:
            print(f"    SpecShp iter {it}, obj={best_obj:.10f}, alpha={alpha:.3f}")
        
        # Restart if stuck
        if no_improve > 200:
            f = best_f.copy()
            no_improve = 0
            alpha = alpha_final
    
    return best_f, best_obj

def mu_weights_optimize_v2(f0, time_limit, top_k=50, temperature=0.00005, 
                           lr=0.003, momentum=0.85, verbose=False):
    """Improved multiplicative-weights with adaptive peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn
        
        conv_nest = fast_convolve(f_nest)
        
        # Compute subgradient with adaptive peak selection
        k_actual = min(cur_top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        
        # Weight peaks by how much they exceed the mean
        mean_conv = conv_nest.mean()
        excess = np.maximum(pv - mean_conv, 0)
        if excess.sum() < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = excess / excess.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]
        
        # Update momentum
        v = momentum * v + (1 - momentum) * sg
        
        # Multiplicative weights update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive schedule
        cur_lr *= 0.9998
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp = max(cur_temp * 1.005, 1e-6)
        
        if it % 1000 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.9))
        
        # Restart
        if no_improve > 500:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.003 + 0.001 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0
    
    return best_f, best_obj

def equalization_search(f0, time_limit, verbose=False):
    """
    Local search based on equalization: repeatedly pick pairs (i,j) where
    a[i] >> a[j] and partially equalize them.
    Also includes random mass transfers.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Strategy 1: Equalize the most imbalanced pairs
        # Find the pair with maximum ratio a[i]/a[j] (among significant elements)
        significant = np.where(f > best_f.max() * 0.01)[0]
        if len(significant) > 1:
            vals = f[significant]
            # Try a few random pairs from significant elements
            for _ in range(20):
                idx1 = rng.integers(0, len(significant))
                idx2 = rng.integers(0, len(significant))
                if idx1 == idx2:
                    continue
                i = significant[idx1]
                j = significant[idx2]
                
                if f[i] < f[j]:
                    i, j = j, i
                
                if f[j] < 1e-12:
                    target = f[i] * 0.5
                else:
                    target = np.sqrt(f[i] * f[j])
                
                # Move towards equalization
                delta = (f[i] - target) * 0.5
                if abs(delta) < 1e-14:
                    continue
                f[i] -= delta
                f[j] += delta
                f = np.maximum(f, 0)
                f /= f.sum()
                
                obj = fast_convolve(f).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                else:
                    f[i] += delta
                    f[j] -= delta
                    f = np.maximum(f, 0)
                    f /= f.sum()
        
        # Strategy 2: Random mass transfer guided by convolution peaks
        conv = fast_convolve(f)
        top_k = min(5, 2*n-1)
        peaks = np.argsort(conv)[-top_k:]
        
        for _ in range(30):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(peaks[rng.integers(0, top_k)])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            
            if contribs.max() < 1e-15:
                continue
            
            # Transfer from highest contributor to random other position
            hi = idx[np.argmax(contribs)]
            lo = rng.integers(0, n)
            if lo == hi or f[hi] < 1e-10:
                continue
            
            for frac in [0.05, 0.1, 0.2, 0.4]:
                amt = f[hi] * frac
                f[hi] -= amt
                f[lo] += amt
                obj = fast_convolve(f).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[hi] += amt
                    f[lo] -= amt
                    break
        
        # Strategy 3: Swap values
        for _ in range(10):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            tmp = f[i]
            f[i] = f[j]
            f[j] = tmp
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                improved = True
            else:
                f[j] = f[i]
                f[i] = tmp
        
        if not improved:
            # Try a bigger perturbation
            noise = rng.standard_normal(n) * 0.005
            f = project_simplex(best_f + noise)
            obj = fast_convolve(f).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
            else:
                f = best_f.copy()
            # If still stuck, break
            if time.monotonic() - t0 > time_limit * 0.8:
                break
    
    return best_f, best_obj

def coordinate_descent_v2(f0, time_limit, verbose=False):
    """Coordinate descent with golden-section search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    mid = n // 2
    order = []
    for d in range(mid + 1):
        if mid + d < n:
            order.append(mid + d)
        if mid - d >= 0 and d > 0:
            order.append(mid - d)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()
            
            # Try a few multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def structured_inits(n, rng, count=5):
    inits = []
    inits.append(np.ones(n) / n)
    
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())
    
    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 5.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())
    
    # "Ring" pattern: high in middle, low at edges, with oscillation
    omega = 2 * np.pi * (3 + rng.integers(0, 5)) / n
    ring = 0.5 + 0.3 * np.cos(omega * np.arange(n))
    ring = np.maximum(ring, 1e-10)
    inits.append(ring / ring.sum())
    
    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())
    
    return inits[:count]
```

Let me also add one more idea: **alternating minimization**. 

The objective is `max_k conv[k]` (for normalized sequences). We can write this as:
`min_a max_k sum_i a[i] a[k-i]`

This is a min-max problem. By von Neumann's minimax theorem (or Sion's theorem for this convex-concave case), we can write:
`min_a max_k f(a,k) = max_p min_a E_p[f(a,k)]`
where p is a distribution over k.

The inner minimization (for fixed p) is:
`min_a E_p[f(a,k)] = min_a sum_k p[k] sum_i a[i] a[k-i] = min_a a^T Q(p) a`
where Q(p) is a matrix with Q[i,j] = sum_k p[k] * [i+j=k] = p[i+j] (if i+j < 2n-1, else 0).

So Q(p)[i,j] = p[i+j] (for i+j valid). This is a "Hankel" matrix constructed from p.

The inner problem is: `min_a a^T Q(p) a` subject to `a >= 0, sum(a) = 1`.

This is a quadratic program! We can solve it with scipy or cvxpy.

The outer problem is: `max_p a(p)^T Q(p) a(p)` where a(p) is the optimal solution of the inner problem.

This is a maximin formulation. In practice, we can use a Frank-Wolfe or multiplicative weights approach on p (which lives on the simplex over 2n-1 dimensions).

A Hankel matrix is PSD if and only if the sequence p corresponds to the moments of a positive measure (by the moment problem). In general, a random probability vector p will NOT give a PSD Hankel matrix.

The key observation is that for a sequence a of length n, the convolution has 2n-1 values, and their sum is (sum a)^2. So the average convolution value is (sum a)^2/(2n-1).

For a normalized sequence (sum=1), the average is 1/(2n-1) ≈ 1/(2n).

The objective is 2n * max_conv. If max_conv = c/(2n) for some constant c, then objective = c.

The minimum possible c is 1 (if all convolution values are equal). For non-negative sequences, c must be > 1 because the convolution at the edges is always small (only 1 term in the sum), while the center has n terms.

Specifically, conv[0] = a[0]^2 and conv[n-1] = sum_i a[i] a[n-1-i]. For a "spread out" sequence, a[0]^2 is very small but conv[n-1] is around 1/n. So the max is at the center.

By AM-GM, for each pair (a[i], a[n-1-i]), we have a[i]*a[n-1-i] <= (a[i] + a[n-1-i])^2 / 4.

But sum_i (a[i] + a[n-1-i]) = 2 (since each a[j] appears exactly once as a[i] and once as a[n-1-i]).

By Cauchy-Schwarz: sum_i (a[i] + a[n-1-i])^2 >= (sum_i (a[i] + a[n-1-i]))^2 / n = 4/n.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Spectral shaping (iterative time-frequency projection) - novel
2. Alternating minimization (peak weighting + weighted convolution minimization)
3. Multiplicative-weights subgradient with Nesterov momentum
4. Equalization-based and swap local search
5. Coordinate descent polishing
6. Multi-length exploration
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_shaping(f0, time_limit, alpha_init=0.8, alpha_final=0.5,
                     blend=0.5, n_iter_max=100000, verbose=False):
    """
    Iterative time-frequency projection:
    - Frequency domain: compress the power spectrum (flatten peaks)
    - Time domain: enforce non-negativity and normalization
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    fft_size = 1 << (2 * n - 2).bit_length()

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0

    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.rfft(f, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        power = mag ** 2

        progress = min(float(it) / float(n_iter_max), 1.0)
        alpha = alpha_init + (alpha_final - alpha_init) * progress
        b = blend + 0.3 * progress

        power_norm = power / (power.sum() + 1e-30)
        power_comp = power_norm ** alpha
        pc_sum = power_comp.sum()
        if pc_sum < 1e-30:
            continue
        power_comp = power_comp / pc_sum * power.sum()
        mag_new = np.sqrt(power_comp)
        mag_new = b * mag_new + (1.0 - b) * mag

        A_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(A_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f = best_f.copy()
            no_improve += 50
            continue
        f = f_new / s_new

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 5000 == 0:
            print(f"    SpecShp {it}: obj={best_obj:.10f}, alpha={alpha:.3f}")

        if no_improve > 300:
            f = best_f.copy()
            no_improve = 0

    return best_f, best_obj


def alternating_minimize(f0, time_limit, n_outer=50, grad_iters=30,
                         temperature=0.01, lr_init=0.01, verbose=False):
    """
    Alternating: (1) compute peak weights from current convolution,
    (2) minimize weighted sum of convolution values via projected gradient.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    ss = f.sum()
    if ss < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / ss
    n = len(f)
    fft_size = 1 << (2 * n - 2).bit_length()

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    for outer in range(n_outer):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        # Compute peak weights
        scaled = conv / max(temperature, 1e-10)
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(len(conv)) / len(conv)
        else:
            p /= p_sum

        # Minimize sum_k p[k] * conv(a,a)[k] via projected gradient
        # Gradient of a^T Q a where Q[i,j] = p[i+j]
        # grad[i] = 2 * sum_j p[i+j] * a[j] = 2 * (p * a)[i] (cross-correlation)
        # Compute via FFT: ifft(fft(p_padded) * conj(fft(a_padded)))... 
        # Actually: sum_j p[i+j] a[j] is a cross-correlation
        # cross[i] = sum_j p[i+j] a[j]
        # = sum_j p_conv[i+j] a[j] where p_conv = p
        # This is like: if we define b[j] = a[-j] (reversed), then cross = conv(p, b_reversed)
        # Simpler: use the fact that sum_j p[i+j] a[j] = (p cross-corr a)[i]
        # We can compute: IFFT(FFT(p) * conj(FFT(a))) gives cross-correlation
        
        P = np.fft.rfft(p, fft_size)
        Fa = np.fft.rfft(f, fft_size)
        # cross[i] = sum_j p[i+j] a[j]
        # This equals the correlation: ifft(FFT(p) * conj(FFT(a_shifted)))
        # Actually let's just compute it directly via the convolution identity:
        # sum_j p[i+j] a[j] for valid j means we need p shifted
        # Alternative: sum_j p[i+j] a[j] = sum_k p[k] a[k-i] = (p * reverse(a))[i + n - 1]... 
        # Let me just do it with FFT properly:
        # Define g[k] = p[k] for k in [0, 2n-2], 0 otherwise (already the case)
        # h[k] = a[k] for k in [0, n-1], 0 otherwise
        # We want: cross[i] = sum_j g[i+j] h[j]
        # = sum_j g[i+j] h[j]
        # Let m = i+j, j = m-i: sum_m g[m] h[m-i]
        # This is the cross-correlation of g and h at lag i
        # = ifft(FFT(g) * conj(FFT(h))) at index i (with appropriate shifting)
        
        cross = np.fft.irfft(P * np.conj(Fa), fft_size)[:2*n-1]
        # cross[k] = sum_j p[j] a[j-k]... hmm let me verify
        # Actually irfft(FFT(g)*conj(FFT(h))[n] gives g cross-correlated with h
        # cross_correlate[g,h][k] = sum_j g[j+k] h[j] (for non-negative lags)
        # So cross[k] = sum_j p[j+k] a[j] which is what we want!
        
        grad = 2.0 * cross[:n]  # gradient of a^T Q a

        # Projected gradient descent
        lr = lr_init / (1.0 + 0.1 * outer)
        f_new = f - lr * grad
        f_new = project_simplex(f_new)

        # Check improvement
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
        f = f_new

        # Decay temperature
        temperature *= 0.95

        if verbose and outer % 10 == 0:
            print(f"    AltMin outer {outer}: obj={best_obj:.10f}, T={temperature:.6f}")

    return best_f, best_obj


def mu_weights_optimize(f0, time_limit, top_k=50, temperature=0.00005,
                        lr=0.003, momentum=0.85, verbose=False):
    """Multiplicative-weights subgradient with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        k_actual = min(cur_top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]

        # Weight by excess over median
        med = np.median(pv)
        excess = np.maximum(pv - med, 0.0)
        if excess.sum() < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = excess / excess.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]

        v = momentum * v + (1.0 - momentum) * sg
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        if f.sum() > 0:
            f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        if it % 800 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.92))

        if no_improve > 400:
            rng = np.random.default_rng((it * 6271 + 17) % 1000000)
            noise = 0.002 + 0.001 * min(no_improve // 500, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0002)
            no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"    MU iter {it}: obj={best_obj:.10f}, lr={cur_lr:.7f}")

    return best_f, best_obj


def local_search(f0, time_limit, verbose=False):
    """Combined local search: equalization, swaps, and guided transfers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    ss = f.sum()
    if ss < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / ss
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        top_k = min(20, 2 * n - 1)
        peaks = np.argsort(conv)[-top_k:]

        # Guided transfers from peak contributors
        for _ in range(40):
            if time.monotonic() - t0 > time_limit:
                break
            k = int(peaks[rng.integers(0, top_k)])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            if contribs.max() < 1e-15:
                continue
            hi = idx[np.argmax(contribs)]
            lo = rng.integers(0, n)
            if lo == hi or f[hi] < 1e-10:
                continue
            for frac in [0.05, 0.1, 0.2, 0.35, 0.5]:
                amt = f[hi] * frac
                f[hi] -= amt
                f[lo] += amt
                obj = fast_convolve(f).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
                f[hi] += amt
                f[lo] -= amt
                break

        # Equalization moves
        for _ in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or f[i] < 1e-12 or f[j] < 1e-12:
                continue
            if f[i] < f[j]:
                i, j = j, i
            target = np.sqrt(f[i] * f[j])
            if target >= f[i] * 0.99:
                continue
            delta = (f[i] - target) * 0.4
            f[i] -= delta
            f[j] += delta
            f = np.maximum(f, 0)
            f /= f.sum()
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                improved = True
            else:
                f = best_f.copy()

        # Swaps
        for _ in range(15):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            tmp = f[i]
            f[i] = f[j]
            f[j] = tmp
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                improved = True
            else:
                f[j] = f[i]
                f[i] = tmp

        if not improved:
            stall += 1
            if stall > 5:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                obj = fast_convolve(f).max()
                if obj < best_obj:
                    best_obj = obj
                    best_f = f.copy()
                else:
                    f = best_f.copy()
                stall = 0
            if stall > 20:
                break
        else:
            stall = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiplier search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    ss = f.sum()
    if ss < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / ss
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    mid = n // 2
    order = []
    for d in range(mid + 1):
        if mid + d < n:
            order.append(mid + d)
        if mid - d >= 0 and d > 0:
            order.append(mid - d)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 5.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    omega = 2.0 * np.pi * float(3 + rng.integers(0, 6)) / float(n)
    osc = 0.6 + 0.3 * np.cos(omega * np.arange(n))
    osc = np.maximum(osc, 1e-10)
    inits.append(osc / osc.sum())

    r = np.abs(rng.standard_normal(n))
    kl = max(3, n // 50)
    kernel = np.ones(kl) / kl
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    return inits[:count]


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining spectral shaping, alternating minimization,
    multiplicative weights, local search, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform n=512")

    n_parent = len(parent)

    # === Phase 1: Spectral Shaping (novel approach) ===
    print("\n=== Phase 1: Spectral Shaping ===")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t = min(rem * 0.35, 180)
        print(f"  SpecShp parent: n={n_parent}, t={t:.0f}s")
        a_opt, _ = spectral_shaping(parent, time_limit=t, alpha_init=0.8, alpha_final=0.5, blend=0.5)
        try_seq(a_opt, "spec_parent")

    # Spectral on structured inits at parent length
    inits = structured_inits(n_parent, rng, count=3)
    for idx, init in enumerate(inits):
        rem = deadline - time.monotonic() - 400
        if rem < 15:
            break
        t = min(rem * 0.2, 90)
        a_opt, _ = spectral_shaping(init, time_limit=t, alpha_init=0.7, alpha_final=0.4, blend=0.6)
        try_seq(a_opt, f"spec_init{idx}")

    # Try a longer length with spectral
    n_long = n_parent + 256
    rem = deadline - time.monotonic() - 350
    if rem > 20:
        f_interp = interpolate_sequence(parent, n_long)
        t = min(rem * 0.3, 120)
        print(f"  SpecShp longer: n={n_long}, t={t:.0f}s")
        a_opt, _ = spectral_shaping(f_interp, time_limit=t, alpha_init=0.75, alpha_final=0.5, blend=0.5)
        try_seq(a_opt, f"spec_long_{n_long}")

    # === Phase 2: Alternating Minimization (novel) ===
    print(f"\n=== Phase 2: Alternating Minimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t = min(rem * 0.4, 120)
            print(f"  AltMin: n={n_best}, t={t:.0f}s")
            a_opt, _ = alternating_minimize(best_seq, time_limit=t, temperature=0.05, lr_init=0.02)
            try_seq(a_opt, "altmin")

        # AltMin from perturbed start
        rem = deadline - time.monotonic() - 250
        if rem > 15:
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)
            t = min(rem * 0.4, 90)
            a_opt, _ = alternating_minimize(f_pert, time_limit=t, temperature=0.1, lr_init=0.015)
            try_seq(a_opt, "altmin_pert")

    # === Phase 3: Multiplicative Weights ===
    print(f"\n=== Phase 3: MU Weights (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            {"top_k": max(30, n_best // 5), "lr": 0.003, "mom": 0.85},
            {"top_k": max(60, n_best // 3), "lr": 0.005, "mom": 0.80},
            {"top_k": max(15, n_best // 10), "lr": 0.001, "mom": 0.93},
            {"top_k": max(100, n_best // 2), "lr": 0.008, "mom": 0.75},
        ]

        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 200
            if rem < 12:
                break
            t = min(rem * 0.25, 120)
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.004)
            elif p == 2:
                f_init = project_simplex(0.8 * best_seq + 0.2 * np.ones(n_best) / n_best)
            else:
                f_init = project_simplex(0.5 * best_seq + 0.5 * best_seq[::-1])

            a_opt, _ = mu_weights_optimize(f_init, time_limit=t,
                                            top_k=cfg["top_k"], lr=cfg["lr"],
                                            momentum=cfg["mom"])
            try_seq(a_opt, f"mu_{p}")

    # === Phase 4: Length exploration ===
    print(f"\n=== Phase 4: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        targets = sorted(set([n_best - 128, n_best + 64, n_best + 192, n_best + 512]))
        targets = [t for t in targets if t > 128 and t < 4096]

        for tn in targets[:3]:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, tn)
            # Quick spectral + MU
            t1 = min(rem * 0.3, 80)
            a_opt, _ = spectral_shaping(f_interp, time_limit=t1, alpha_init=0.8, alpha_final=0.5)
            try_seq(a_opt, f"spec_{tn}")

            rem = deadline - time.monotonic() - 120
            if rem > 15:
                t2 = min(rem * 0.4, 70)
                a_opt, _ = mu_weights_optimize(f_interp, time_limit=t2, top_k=max(30, tn // 6))
                try_seq(a_opt, f"mu_{tn}")

    # === Phase 5: Local Search ===
    print(f"\n=== Phase 5: Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t = min(rem * 0.5, 100)
            print(f"  LocalSearch: t={t:.0f}s")
            a_opt, _ = local_search(best_seq, time_limit=t)
            try_seq(a_opt, "local")

    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Coord Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t = min(rem * 0.6, 80)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t)
            try_seq(a_opt, "cd")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick spectral
        rem = deadline - time.monotonic() - 30
        if rem > 8:
            a_opt, _ = spectral_shaping(best_seq, time_limit=min(rem, 30),
                                         alpha_init=0.9, alpha_final=0.8, blend=0.3)
            try_seq(a_opt, "spec_polish")

        # Quick local
        rem = deadline - time.monotonic() - 20
        if rem > 6:
            a_opt, _ = local_search(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "loc_polish")

        # Quick CD
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "cd_polish")

        # Fine MU
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = mu_weights_optimize(best_seq, time_limit=min(rem, 12),
                                            top_k=n_best, lr=0.0003, momentum=0.96)
            try_seq(a_opt, "mu_fine")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
