    return best_a, best_obj


def coordinate_linesearch(a0, time_limit, verbose=False):
    """Coordinate descent with trial perturbations and rescaling."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    rng = np.random.default_rng(42)

    # Pre-compute fractional multipliers to try
    fracs = [0.5, 0.6, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 1.8, 2.0]

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Process coordinates in random order for diversity
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-10:
                # Try setting it to a small value
                for frac in [0.1, 0.5, 1.0]:
                    x_trial = 0.01 / n
                    a[j] = x_trial
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= (1.0 - x_trial) / (1.0 - x_old + 1e-30)
                    a = np.maximum(a, 0.0)
                    ss = a.sum()
                    if ss > 1e-15:
                        a /= ss
                    obj = fast_convolve(a).max()
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                # Reset
                a = best_a.copy()
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            for frac in fracs:
                x_new = x_old * frac
                x_new = min(max(x_new, 0.0), 1.0 - 1e-12)
                if abs(x_new - x_old) < 1e-12:
                    continue

                a[j] = x_new
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= (1.0 - x_new) / s_others
                a = np.maximum(a, 0.0)
                ss = a.sum()
                if ss < 1e-15:
                    a = best_a.copy()
                    break
                a /= ss

                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                else:
                    a = best_a.copy()

        if not improved:
            if passes > 1:
                noise_scale = 0.001 * (1.0 + 0.1 * passes)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            else:
                break

        if verbose and passes % 5 == 0:
            print(f"    CDL pass {passes}: obj={best_obj:.10f}")

    return best_a, best_obj


def multinational_descent(a0, time_limit, verbose=False):
    """
    Alternating between Nesterov smoothed descent and coordinate line search,
    with periodic restarts and temperature adaptation.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    phase = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        if phase % 3 == 0:
            # Nesterov smoothed descent
            t_alloc = min(60, remaining * 0.5)
            T_cur = 0.003 + 0.002 * max(0, 5 - phase // 3) * 0.3
            lr_cur = 0.04 / (1.0 + 0.3 * (phase // 3))
            a_opt, obj = nesterov_smoothed(
                best_a, time_limit=t_alloc, T_init=T_cur,
                T_min=0.0003, lr_init=lr_cur, top_k=max(50, n // 6)
            )
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} NAG: obj={best_obj:.10f}")

        elif phase % 3 == 1:
            # Coordinate line search
            t_alloc = min(40, remaining * 0.4)
            a_opt, obj = coordinate_linesearch(best_a, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} CDL: obj={best_obj:.10f}")

        else:
            # Noise + quick Nesterov
            noise_scale = 0.004 * max(0.3, 1.0 - 0.1 * (phase // 3))
            a_noisy = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            t_alloc = min(30, remaining * 0.3)
            a_opt, obj = nesterov_smoothed(
                a_noisy, time_limit=t_alloc, T_init=0.005,
                T_min=0.0005, lr_init=0.03, top_k=80
            )
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} Noise+NAG: obj={best_obj:.10f}")

        phase += 1

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence palindromic: a[i] = (a[i] + a[n-1-i])/2, renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining Nesterov smoothed gradient descent,
    coordinate line search, multi-length exploration, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 40.0)

    best_a = None
    best_score = float('inf')

    def update_best(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        score = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_try)})")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    # === Phase 1: Deep Nesterov optimization on parent ===
    print(f"\n=== Phase 1: Nesterov on parent (n={len(parent)}) ===")
    rem = deadline - time.monotonic() - 900
    if rem > 60:
        t_alloc = min(rem, 600)
        print(f"  Nesterov smooth, t={t_alloc:.0f}s")
        a_opt, _ = multinational_descent(parent, time_limit=t_alloc, verbose=True)
        update_best(a_opt, "parent_nag")

    # === Phase 2: Symmetric variant ===
    print(f"\n=== Phase 2: Symmetric variant (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            a_sym = symmetrize(best_a)
            t_alloc = min(rem, 180)
            print(f"  Symmetric, t={t_alloc:.0f}s")
            a_opt, _ = multinational_descent(a_sym, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "symmetric")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    target_lengths = [512, 640, 768, 896, 1024, 1280]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 50:
            break
        if target_n == len(base):
            continue
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.4, 150)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = multinational_descent(f_interp, time_limit=t_alloc, verbose=False)
        update_best(a_opt, f"n{target_n}")

    # === Phase 4: Diverse restarts on best ===
    print(f"\n=== Phase 4: Diverse restarts (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        restart_configs = [
            ("noise3", 0.003),
            ("noise5", 0.005),
            ("noise8", 0.008),
            ("reverse_blend", None),
            ("flat_blend", None),
            ("bimodal", None),
        ]
        for idx, (label, noise) in enumerate(restart_configs):
            rem = deadline - time.monotonic() - 200 + idx * 30
            if rem < 30:
                break
            if noise is not None:
                f_init = project_simplex(best_a + rng.standard_normal(n_best) * noise)
            elif label == "reverse_blend":
                f_init = project_simplex(0.7 * best_a + 0.3 * best_a[::-1])
            elif label == "flat_blend":
                f_init = project_simplex(0.85 * best_a + 0.15 * np.ones(n_best) / n_best)
            elif label == "bimodal":
                center = n_best // 2
                widths = np.arange(n_best)
                f_gauss = np.exp(-((widths - center) ** 2) / (2 * (n_best / 4) ** 2))
                f_init = project_simplex(0.7 * best_a + 0.3 * f_gauss)
            else:
                f_init = best_a.copy()

            t_alloc = min(rem, 150)
            a_opt, _ = multinational_descent(f_init, time_limit=t_alloc, verbose=False)
            update_best(a_opt, f"restart_{label}")

    # === Phase 5: Final intensive polishing ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        # Long coordinate line search
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  CDL polish, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_linesearch(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "final_cdl")

        # Very long Nesterov with tiny learning rate
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem, 100)
            print(f"  Final NAG fine-tune, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_smoothed(
                best_a, time_limit=t_alloc, T_init=0.002,
                T_min=0.0002, lr_init=0.01, top_k=50
            )
            update_best(a_opt, "final_nag_fine")

        # Tiny perturbations + quick CDL
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            n_best = len(best_a)
            for p in range(4):
                rem = deadline - time.monotonic() - 10
                if rem < 8:
                    break
                rng_p = np.random.default_rng(seed + 500 + p * 111)
                f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.001)
                t_alloc = min(rem, 12)
                a_opt, _ = coordinate_linesearch(f_pert, time_limit=t_alloc)
                update_best(a_opt, f"perturb_{p}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
