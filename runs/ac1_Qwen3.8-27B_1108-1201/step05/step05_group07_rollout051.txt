def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def smooth_max(x, T):
    """Smooth approximation of max using logsumexp."""
    x_shifted = x / T - np.max(x) / T
    return T * (np.log(np.sum(np.exp(x_shifted))) + np.max(x) / T)

def smooth_max_gradient(x, T):
    """Gradient of smoothed max (softmax)."""
    x_shifted = x / T - np.max(x) / T
    w = np.exp(x_shifted)
    w /= w.sum()
    return w  # This is the softmax weights

def adam_update(params, grads, m, v, t, beta1=0.9, beta2=0.999, lr=0.01, eps=1e-8):
    """Adam optimizer update."""
    m = beta1 * m + (1 - beta1) * grads
    v = beta2 * v + (1 - beta2) * grads**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    return params - update, m, v

    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(len(a)) / len(a)
    return a

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Structured initialization with multiple diverse starting points
2. Logsumexp-smoothed subgradient optimization with Adam
3. Precise coordinate descent with multi-trial evaluation
4. Length exploration with interpolation
5. Ultra-fine polishing with very low learning rates

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def logsumexp_sf(x, T):
    """Compute logsumexp(x/T)*T in a stable way."""
    x_max = np.max(x)
    return T * np.log(np.sum(np.exp((x - x_max) / T))) + x_max

def softmax_sf(x, T):
    """Compute softmax(x/T) in a stable way."""
    x_shifted = x / T - np.max(x) / T
    w = np.exp(x_shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(x))
        w[np.argmax(x)] = 1.0
    else:
        w /= s
    return w

def compute_subgradient_smooth(a, conv, T, n_peaks=None):
    """
    Compute the gradient of logsumexp(conv/T)*T with respect to a.
    This is sum_k softmax_k * d(conv[k])/d(a)
    """
    n = len(a)
    w = softmax_sf(conv, T)
    
    # Only consider significant weights for efficiency
    if n_peaks is not None:
        top = np.argpartition(w, -n_peaks)[-n_peaks:]
        w_masked = np.zeros_like(w)
        w_masked[top] = w[top]
        w = w_masked
    
    sg = np.zeros(n)
    significant = np.where(w > 1e-12)[0]
    for k in significant:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[k] * a[k - idx]
    return sg

def optimize_logsumexp(a0, time_limit, T0=0.05, T_end=0.0001, lr=0.01,
                        beta1=0.9, beta2=0.999, n_peaks=100,
                        restart_interval=200, restart_scale=0.005):
    """
    Optimize using logsumexp smoothing with Adam.
    T decreases geometrically from T0 to T_end.
    """
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    t0 = time.monotonic()
    no_improve = 0
    
    # T schedule: geometric from T0 to T_end over the time limit
    # We'll adjust based on actual iterations
    t_decay = 0.999  # per-iteration decay
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        T = max(T_end, T0 * (t_decay ** t))
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gradient of logsumexp
        sg = compute_subgradient_smooth(a, conv, T, n_peaks)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = project_simplex(a_new)
        
        # Restart if stuck
        if no_improve > restart_interval:
            rng = np.random.default_rng((t * 7919) % 1000000)
            noise = rng.standard_normal(n) * restart_scale
            a = project_simplex(best_a + noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            t_decay = 0.9995  # slow down T decay after restart
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE: it={t}, T={T:.6f}, obj={best_obj:.10f}, no_imp={no_improve}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def coordinate_descent(f0, time_limit, trials_per_coord=5, max_passes=50):
    """
    Coordinate descent: for each coordinate, try several multipliers
    and keep the best.
    """
    f = project_simplex(f0.astype(np.float64).copy())
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    multipliers = [0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]
    
    passes = 0
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj  # use global best as threshold
            
            for mult in multipliers:
                if mult == 1.0:
                    continue
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                
                # Scale others to maintain sum=1
                others_sum = 1.0 - old_val
                if others_sum < 1e-12:
                    continue
                new_others_sum = 1.0 - new_val
                scale = new_others_sum / others_sum
                
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial[:j] *= scale
                f_trial[j+1:] *= scale
                f_trial = np.maximum(f_trial, 0)
                s = f_trial.sum()
                if s < 1e-15:
                    continue
                f_trial /= s
                
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                others_sum = 1.0 - old_val
                if others_sum > 1e-12:
                    new_others_sum = 1.0 - best_j_val
                    scale = new_others_sum / others_sum
                    f[j] = best_j_val
                    f[:j] *= scale
                    f[j+1:] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj

def pair_swap_search(f0, time_limit, max_trials=5000):
    """
    Try swapping or transferring mass between pairs of coordinates.
    """
    f = project_simplex(f0.astype(np.float64).copy())
    n = len(f)

    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    rng = np.random.default_rng(42)
    trials = 0
    
    while time.monotonic() - t0 < time_limit and trials < max_trials:
        trials += 1
        
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        old_i = f[i]
        old_j = f[j]
        
        if old_i < 1e-12 and old_j < 1e-12:
            continue
        
        # Try various transfers
        for frac in [0.2, 0.5, 0.8]:

            amt = old_i * frac
            if amt > old_i:

            f[i] = old_i - amt
            f[j] = old_j + amt
            s = f.sum()
            if s > 0:
                f /= s
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                break
            else:
                # Restore
                f = best_f.copy() if obj >= best_obj else f
                break
        
        # Occasionally try swap

                best_f = f.copy()

    return best_f, best_obj

def structured_inits(n, rng):
    """Generate diverse structured initial sequences."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramp
    ramp = np.arange(1, n+1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    # Quadratic ramp
    quad = np.arange(1, n+1, dtype=np.float64)**2
    inits.append(quad / quad.sum())
    
    # Cubic ramp
    cubic = np.arange(1, n+1, dtype=np.float64)**3
    inits.append(cubic / cubic.sum())
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(np.pi * (x - 0.5) * 2))
    cos_w = np.maximum(cos_w, 1e-10)
    inits.append(cos_w / cos_w.sum())
    
    # Two peaks
    peak1 = np.exp(-0.5 * ((np.arange(n) - n*0.2) / (n*0.05))**2)
    peak2 = np.exp(-0.5 * ((np.arange(n) - n*0.8) / (n*0.05))**2)
    two = peak1 + peak2
    inits.append(two / two.sum())
    
    # Random smooth
    r = np.abs(rng.standard_normal(n))
    klen = max(3, n // 50)
    kernel = np.ones(klen) / klen
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())
    
    # Exponential decay
    exp_d = np.exp(-3.0 * np.arange(n) / n)
    inits.append(exp_d / exp_d.sum())
    
    # Sharp exponential
    exp_sharp = np.exp(-10.0 * np.arange(n) / n)
    inits.append(exp_sharp / exp_sharp.sum())
    
    return inits

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
