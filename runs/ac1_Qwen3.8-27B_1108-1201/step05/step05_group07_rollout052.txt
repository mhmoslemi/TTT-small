"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed gradient
optimization with Adam, multi-length exploration, symmetry exploitation,
peak-shaving local search, and structured restarts.
Key innovations: smooth LSE loss for true gradients, adaptive temperature
cooling, symmetry-aware updates, and peak-targeted local search.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_optimize(f0, time_limit, init_alpha=500.0, min_alpha=5000.0,
                 lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                 top_k=30, sym_every=50, verbose=False, label=""):
    """
    Log-sum-exp smoothed optimization with Adam.
    Minimizes (1/alpha)*log(sum exp(alpha*b[k])) as approximation to max(b).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    m = np.zeros(n)  # Adam first moment
    v = np.zeros(n)  # Adam second moment
    t_step = 0
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_alpha = init_alpha

    while time.monotonic() - t0 < time_limit:
        t_step += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.01:
            break

        # Compute convolution
        b = fast_convolve(f)

        # Compute LSE gradient using top-K peaks
        scaled = cur_alpha * b
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        p = exp_scaled / exp_scaled.sum()

        # Use top-K for gradient computation
        k_actual = min(top_k, len(b))
        top_idx = np.argpartition(p, -k_actual)[-k_actual:]
        p_top = p[top_idx]
        p_top_sum = p_top.sum()
        if p_top_sum > 1e-15:
            p_top = p_top / p_top_sum
        else:
            p_top = np.zeros(k_actual)
            p_top[np.argmax(p_top)] = 1.0

        grad = np.zeros(n)
        for idx in range(k_actual):
            if p_top[idx] < 1e-12:
                continue
            k = int(top_idx[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            a_idx_start = k - i_end
            a_idx_end = k - i_start
            grad[i_start:i_end + 1] += 2.0 * p_top[idx] * f[a_idx_start:a_idx_end + 1]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        update = m_hat / (np.sqrt(v_hat) + eps)

        f_new = f - cur_lr * update
        f_new = np.maximum(f_new, 0.0)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = best_f.copy()
            cur_lr *= 0.5
        else:
            f_new = f_new / fs

        # Periodic symmetrization
        if sym_every > 0 and t_step % sym_every == 0:
            f_new = 0.5 * (f_new + f_new[::-1])
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()

        f = f_new

        # Evaluate
        b_new = fast_convolve(f)
        obj = b_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive cooling: increase alpha (sharper approximation to max)
        cur_alpha *= 1.02
        # Decay learning rate
        cur_lr *= 0.9998

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((t_step * 31 + 7) % 100000000)
            noise_scale = 0.003 + 0.002 * min(no_improve // 200, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.001)
            cur_alpha = max(init_alpha * 0.5, 100.0)
            t_step = 0
            no_improve = 0

        if verbose and t_step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE {label} iter {t_step}, t={elapsed:.1f}s, obj={best_obj:.10f}, alpha={cur_alpha:.0f}, lr={cur_lr:.6f}")

    return best_f, best_obj


def peak_shaving_search(f0, time_limit, verbose=False):
    """
    Peak-shaving local search: iteratively find the peak, then redistribute
    mass from high-contributing coordinates to low-contributing ones.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        peak_lag = int(np.argmax(b))
        current_max = b[peak_lag]

        # Find contributing indices for the peak
        i_start = max(0, peak_lag - n + 1)
        i_end = min(n - 1, peak_lag)
        idx = np.arange(i_start, i_end + 1)
        a_idx = peak_lag - idx
        contribs = f[idx] * f[a_idx]
        total_contrib = contribs.sum()

        if total_contrib < 1e-15:
            break

        normalized_contribs = contribs / total_contrib
        # Sort by contribution (highest first)
        order = np.argsort(contribs)[::-1]

        improved = False
        # Try moving mass from high contributors to low contributors or random positions
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            if improved:
                break

            src_pos = order[rng.integers(0, min(10, len(order)))]
            # Choose destination: random position not contributing to peak
            dst = rng.integers(0, n)
            if dst in idx:
                continue

            amt = f[int(idx[src_pos])] * 0.2
            if amt < 1e-12:
                continue

            old_src = f[int(idx[src_pos])]
            old_dst = f[dst]
            f[int(idx[src_pos])] = old_src - amt
            f[dst] = old_dst + amt

            b_new = fast_convolve(f)
            new_max = b_new.max()

            if new_max < best_obj - 1e-14:
                best_obj = new_max
                best_f = f.copy()
                improved = True
            else:
                f[int(idx[src_pos])] = old_src
                f[dst] = old_dst

        if not improved:
            # Try moving mass from peak contributors to non-peak areas more aggressively
            found = False
            for src_rank in range(min(5, len(order))):
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[order[src_rank]])
                if f[src] < 1e-10:
                    continue
                for dst in range(n):
                    if dst in idx or f[dst] > f[src]:
                        continue
                    for frac in [0.1, 0.3, 0.5]:
                        amt = f[src] * frac
                        old_s, old_d = f[src], f[dst]
                        f[src] = old_s - amt
                        f[dst] = old_d + amt
                        b_new = fast_convolve(f)
                        if b_new.max() < best_obj - 1e-14:
                            best_obj = b_new.max()
                            best_f = f.copy()
                            found = True
                            break
                        else:
                            f[src] = old_s
                            f[dst] = old_d
                    if found:
                        break
                if found:
                    break

            if not found:
                # Try random perturbations
                for _ in range(20):
                    if time.monotonic() - t0 > time_limit:
                        break
                    i = rng.integers(0, n)
                    j = rng.integers(0, n)
                    if i == j or f[i] < 1e-10:
                        continue
                    for frac in [0.05, 0.1, 0.2, 0.4]:
                        amt = f[i] * frac
                        old_i, old_j = f[i], f[j]
                        f[i] = old_i - amt
                        f[j] = old_j + amt
                        b_new = fast_convolve(f)
                        if b_new.max() < best_obj - 1e-14:
                            best_obj = b_new.max()
                            best_f = f.copy()
                            found = True
                            break
                        f[i] = old_i
                        f[j] = old_j
                    if found:
                        break

            if not found:
                stall += 1
                if stall > 5:
                    break
        else:
            stall = 0

    return best_f, best_obj


def coordinate_descent_search(f0, time_limit, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    # Center-out order
    mid = n // 2
    order = []
    for d in range(mid + 1):
        if mid + d < n:
            order.append(mid + d)
        if d > 0 and mid - d >= 0:
            order.append(mid - d)

    while time.monotonic() - t0 < time_limit:
        improved_cycle = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            b_current = fast_convolve(f)
            cur_peak = b_current.max()
            best_j_val = old_val
            best_j_obj = cur_peak

            # Try different multipliers
            for mult in [0.3, 0.5, 0.7, 1.3, 1.6, 2.0, 3.0, 0.1]:
                new_val = old_val * mult
                if new_val <= 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved_cycle = True

        if not improved_cycle:
            # Try randomized order
            rng_order = rng.permutation(n)
            improved2 = False
            for j in rng_order[:n // 2]:
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-12:
                    continue
                for mult in [0.5, 0.8, 1.2, 1.5, 2.0, 0.3]:
                    new_val = old_val * mult
                    if new_val <= 0 or new_val > 0.99:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = np.maximum(f_trial, 0)
                    f_trial /= f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved2 = True
                        break
            if not improved2:
                break
        if not improved_cycle:
            break

    return best_f, best_obj


def make_initial(n, kind, rng):
    """Create various structured initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'hann':
        x = np.linspace(0, 1, n)
        w = 0.5 * (1 - np.cos(2 * np.pi * x))
        w = np.maximum(w, 1e-10)
        return w / w.sum()
    elif kind == 'cosine':
        x = np.linspace(0, 1, n)
        w = np.cos(np.pi * x / 2) ** 2
        w = np.maximum(w, 1e-10)
        return w / w.sum()
    elif kind == 'gaussian':
        center = n / 2.0
        sigma = n / (3.0 + rng.random() * 3.0)
        w = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        w = np.maximum(w, 1e-10)
        return w / w.sum()
    elif kind == 'random_smooth':
        w = np.abs(rng.standard_normal(n))
        klen = max(3, n // 50)
        kernel = np.ones(klen) / klen
        w = np.convolve(w, kernel, mode='same')
        w = np.maximum(w, 1e-10)
        return w / w.sum()
    elif kind == 'bimodal':
        c1 = n * (0.25 + 0.1 * rng.random())
        c2 = n * (0.75 - 0.1 * rng.random())
        sigma = n / 6
        w = np.exp(-0.5 * ((np.arange(n) - c1) / sigma) ** 2) + \
            np.exp(-0.5 * ((np.arange(n) - c2) / sigma) ** 2)
        w = np.maximum(w, 1e-10)
        return w / w.sum()
    elif kind == 'inverse_cosine':
        x = np.linspace(0.01, 0.99, n)
        w = 1.0 / (x * (1 - x)) ** 0.3
        w = np.maximum(w, 1e-10)
        return w / w.sum()
    else:
        return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LSE smoothed gradient with Adam,
    peak-shaving local search, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, uniform n=512: score={best_score:.6f}")

    # === Phase 1: Quick multi-length/multi-init exploration ===
    print("\n=== Phase 1: Quick Exploration ===")
    configs_p1 = [
        (len(parent), 'hann'),
        (len(parent), 'cosine'),
        (len(parent), 'bimodal'),
        (512, 'gaussian'),
        (1024, 'hann'),
        (1024, 'cosine'),
        (1536, 'uniform'),
        (768, 'inverse_cosine'),
    ]

    for (n_try, kind) in configs_p1:
        rem = deadline - time.monotonic() - 600
        if rem < 15:
            break
        init = make_initial(n_try, kind, rng)
        t_alloc = min(rem * 0.12, 60)
        print(f"  n={n_try}, kind={kind}, t={t_alloc:.0f}s")
        a_opt, _ = lse_optimize(init, time_limit=t_alloc,
                                 init_alpha=300.0, min_alpha=3000.0,
                                 lr=0.008, top_k=30, sym_every=40)
        try_seq(a_opt, f"p1_{kind}_{n_try}")

    # Also optimize parent directly
    rem = deadline - time.monotonic() - 500
    if rem > 20:
        t_alloc = min(rem * 0.2, 80)
        print(f"  Parent direct, t={t_alloc:.0f}s")
        a_opt, _ = lse_optimize(parent, time_limit=t_alloc,
                                 init_alpha=500.0, lr=0.006, top_k=40, sym_every=30)
        try_seq(a_opt, "p1_parent")

    # === Phase 2: Deep optimization on best ===
    print(f"\n=== Phase 2: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        deep_configs = [
            {"alpha": 400, "lr": 0.005, "top_k": 50, "sym": 30, "label": "warm"},
            {"alpha": 800, "lr": 0.003, "top_k": 30, "sym": 50, "label": "cold"},
            {"alpha": 200, "lr": 0.01, "top_k": 60, "sym": 20, "label": "hot"},

        for p, cfg in enumerate(deep_configs):

            t_alloc = min(rem * 0.3, 200)

            elif p == 1:

            elif p == 2:

                f_init = project_simplex(best_seq + noise)
            else:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init)

            print(f"  Deep {cfg['label']}: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = lse_optimize(f_init, time_limit=t_alloc,
                                     init_alpha=cfg["alpha"], lr=cfg["lr"],
                                     top_k=cfg["top_k"], sym_every=cfg["sym"])
            try_seq(a_opt, f"deep_{cfg['label']}")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Try extending and shrinking
        lengths_to_try = sorted(set([
            len(best_seq) // 2,
            len(best_seq),
            len(best_seq) + 128,
            len(best_seq) + 256,
            1024,
            1280,
            1536,

        for target_n in lengths_to_try[:4]:

                continue
            rem = deadline - time.monotonic() - 200
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 120)

            a_opt, _ = lse_optimize(f_interp, time_limit=t_alloc,
                                     init_alpha=400.0, lr=0.006,
                                     top_k=max(20, target_n // 10), sym_every=30)
            try_seq(a_opt, f"len_{target_n}")

    print(f"\n=== Phase 4: Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Peak shaving
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_ps = min(rem * 0.5, 120)
            print(f"  Peak shaving t={t_ps:.0f}s")

            print(f"  Coord descent t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_search(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 5: Final fine-tuning ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine LSE
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_fine = min(rem * 0.4, 60)
            print(f"  Fine LSE t={t_fine:.0f}s")
            a_opt, _ = lse_optimize(best_seq, time_limit=t_fine,

        # Peak shaving again
        rem = deadline - time.monotonic() - 40
        if rem > 10:

            try_seq(a_opt, "ps_final")

        # Perturbation + quick optimize
        for pidx in range(5):
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break
            for scale in [0.002, 0.005, 0.01]:
                rem = deadline - time.monotonic() - 15
                if rem < 8:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)

                                         init_alpha=1000.0, lr=0.003,
                                         top_k=25, sym_every=0)

        # Final coordinate descent
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, _ = coordinate_descent_search(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
