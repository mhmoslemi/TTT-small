def coordinate_descent_step(f, i, num_iters=80):
    """Find optimal f[i] given all other coordinates fixed."""
    n = len(f)
    
    # Compute convolution with f[i] = 0
    f_zero = f.copy()
    f_zero[i] = 0
    conv_zero = fast_convolve(f_zero)
    
    # For the current f[i] = t:
    # (f*f)[k] = conv_zero[k] + extra
    # where extra = t*f[k-i] + f[k-i]*t (if k-i != i) or t^2 (if k=i and k-i=i)
    
    t = f[i]
    
    # Define h(t) = max(conv_zero[k] + 2*t*f[k-i] for valid k != 2i, 
    #                   conv_zero[2i] + t^2 if 2i < 2n-1,
    #                   conv_zero[k] for invalid k)
    
    # For efficiency, precompute the valid k indices and their f[k-i] values
    ks = []
    coeffs = []
    for k in range(2*n - 1):
        ki = k - i
        if 0 <= ki < n and ki != i:  # linear term
            ks.append(k)
            coeffs.append(2.0 * f[ki])
    
    # Quadratic term at k = 2i (if in range)
    has_quad = (2*i < 2*n - 1)
    quad_base = conv_zero[2*i] if has_quad else 0
    
    # Constant terms (k where no extra)
    const_terms = []
    for k in range(2*n - 1):
        ki = k - i
        if not (0 <= ki < n and ki != i) and not (k == 2*i and has_quad):
            const_terms.append(conv_zero[k])
    
    max_const = max(const_terms) if const_terms else 0
    
    def h(t_val):
        # Linear part
        if ks:
            lin_vals = conv_zero[ks] + t_val * np.array(coeffs)
            max_lin = lin_vals.max()
        else:
            max_lin = 0
        # Quadratic part
        q_val = quad_base + t_val * t_val if has_quad else 0
        # Constant part
        return max(max_lin, q_val, max_const)
    
    # Golden section search on [0, t_hi]
    # t_hi: we know f[i] is currently t, and the optimum shouldn't be too far
    # Upper bound: when t is too large, the quadratic dominates
    # h(t) >= t^2 + quad_base, so if t > sqrt(h(0) - quad_base), h(t) > h(0)
    h0 = h(0)
    if has_quad and h0 > quad_base:
        t_hi = np.sqrt(h0 - quad_base)
    else:
        t_hi = 2.0 * max(t, 0.01)
    
    # Actually, let's use a wider range
    t_hi = max(t_hi, 2.0 * t, 0.01)
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = t_hi
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    fc1 = h(c1)
    fc2 = h(c2)
    
    for _ in range(num_iters):
        if hi - lo < 1e-12:
            break
        if fc1 < fc2:
            hi = c2
            c2 = c1
            fc2 = fc1
            c1 = hi - gr * (hi - lo)
            fc1 = h(c1)
        else:
            lo = c1
            c1 = c2
            fc1 = fc2
            c2 = lo + gr * (hi - lo)
            fc2 = h(c2)
    
    t_opt = (lo + hi) / 2
    return t_opt

Actually, one more thought: the coordinate descent I described finds the LOCAL minimum along each coordinate. It's like a block coordinate descent where each block is a single coordinate. This should converge to a locally optimal solution (KKT point) for the problem of minimizing max(f*f) subject to f >= 0, sum(f) = 1.

The KKT conditions are:
- There exists a set S of "active" peak positions (where (f*f)[k] = M)
- For each i with f[i] > 0: sum_{k in S} (contribution of f[i] to (f*f)[k]) = constant (Lagrange multiplier)
- For each i with f[i] = 0: sum_{k in S} (contribution) >= constant

The coordinate descent should converge to a point satisfying these conditions (approximately).

The question is whether this local minimum is the global minimum. For the problem of minimizing the max of a convex function (max of convex functions is convex), the problem is convex, so any local minimum is global!

Wait, is max_k (f*f)[k] convex in f? (f*f)[k] is a quadratic form in f (specifically, f^T A_k f where A_k is the matrix representing the convolution at position k). A quadratic form f^T A f is convex if A is positive semidefinite. The matrix A_k for the k-th convolution coefficient is a symmetric matrix with 1's on the k-th super/sub-diagonal (and 1/2 on the diagonal if k=0). This is not necessarily positive semidefinite!

For example, for k=1: A_1 has 1's on the first super and sub-diagonal. The eigenvalues of this matrix range from -2 to 2 (for large n), so it's NOT positive semidefinite. So (f*f)[k] is not convex in f.

This means max_k (f*f)[k] is not convex, and the problem is non-convex. So coordinate descent might get stuck in local minima.

This is why the previous approach used multiple initialization strategies and random restarts.

Given this non-convexity, the key to finding the global minimum is:
1. Good initialization (structured patterns)
2. Multiple restarts

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-structured initialization (U-shaped, cosine-modulated)
2. Exact coordinate descent (1D minimax per coordinate) - main workhorse
3. Adam-SG with log-sum-exp smoothing and temperature annealing
4. LP-based Frank-Wolfe for polishing
5. Multiplicative weights (mirror descent) as alternative
6. Multi-length exploration and diverse restarts

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coord_descent(f, num_passes=100, num_gs=45, time_limit=None, reheat_interval=20):
    """Exact coordinate descent minimizing max(f*f) with simplex renormalization."""
    n = len(f)
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    conv = fast_convolve(f)
    best_max = conv.max()
    best_f = f.copy()

    t_start = time.monotonic()
    pass_num = 0

    for p in range(num_passes):
        if time_limit is not None and (time.monotonic() - t_start) > time_limit:
            break

        pass_num += 1
        pass_improved = False

        # Reheat: add small mass to zero coordinates
        if pass_num % reheat_interval == 0:
            zeros = np.where(f < 1e-12)[0]
            if len(zeros) > 0:
                eps = 0.001 / n
                for z in zeros[:min(len(zeros), n // 4)]:
                    f[z] = eps
                f = f / f.sum()
                conv = fast_convolve(f)

        # Determine coordinate order (random or sequential)
        if pass_num % 3 == 1:
            order = np.random.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            t_old = f[i]

            # Build linear terms: k where 0 <= k-i < n and k != 2i
            k_start = max(0, i)
            k_end = min(2 * n - 1, i + n)
            if k_end > k_start:
                valid_ks = np.arange(k_start, k_end)
                mask = valid_ks != (2 * i)
                lin_ks = valid_ks[mask]
                if len(lin_ks) > 0:
                    lin_coeffs = 2.0 * f[lin_ks - i]
                    lin_bases = conv[lin_ks].copy()
                else:
                    lin_ks = np.array([], dtype=np.int64)
                    lin_coeffs = np.array([])
                    lin_bases = np.array([])
            else:
                lin_ks = np.array([], dtype=np.int64)
                lin_coeffs = np.array([])
                lin_bases = np.array([])

            # Constant part: max of conv[k] for k not in linear range and k != 2i
            const_max = 0.0
            if i > 0:
                const_max = max(const_max, float(conv[:i].max()))
            if i + n <= 2 * n - 2:
                const_max = max(const_max, float(conv[i + n:].max()))

            has_quad = (2 * i < 2 * n - 1)
            quad_conv = conv[2 * i] if has_quad else 0.0

            # h_delta(delta) = max of:
            #   lin_bases + delta * lin_coeffs  (linear terms)
            #   quad_conv + 2*t_old*delta + delta^2  (quadratic)
            #   const_max  (constant)
            # We search delta in [-t_old, 0] since h is non-decreasing for delta >= 0

            def h_delta(delta):
                if len(lin_ks) > 0:
                    max_lin = float((lin_bases + delta * lin_coeffs).max())
                else:
                    max_lin = 0.0
                if has_quad:
                    q_val = quad_conv + 2.0 * t_old * delta + delta * delta
                else:
                    q_val = 0.0
                return max(max_lin, q_val, const_max)

            # Check if delta=0 is already optimal (subgradient test)
            h0 = h0_val = h_delta(0.0)
            # Right derivative at 0: max of active gradients
            # Left derivative at 0: min of active gradients
            # If left derivative >= 0, then delta=0 is optimal
            active_grads = []
            if len(lin_ks) > 0:
                active_mask = lin_bases >= h0_val - 1e-14
                if active_mask.any():
                    active_grads.extend(lin_coeffs[active_mask].tolist())
            if has_quad:
                if quad_conv >= h0_val - 1e-14:
                    active_grads.append(2.0 * t_old)
            if const_max >= h0_val - 1e-14:
                active_grads.append(0.0)

            if active_grads:
                left_deriv = min(active_grads)
                if left_deriv > 1e-12:
                    # Function is increasing to the left, so delta=0 is optimal
                    # (shouldn't happen since 0 is always in active_grads if const is active)
                    continue

            # Golden section search on [-t_old, 0]
            delta_lo = -t_old
            delta_hi = 0.0

            if delta_hi - delta_lo < 1e-15:
                continue

            gr = (np.sqrt(5) - 1) / 2
            lo = delta_lo
            hi = delta_hi
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            fc1 = h_delta(c1)
            fc2 = h_delta(c2)

            for _ in range(num_gs):
                if hi - lo < 1e-15:
                    break
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    fc1 = h_delta(c1)
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    fc2 = h_delta(c2)

            delta_opt = (lo + hi) / 2
            t_new = t_old + delta_opt
            if t_new < 0:
                t_new = 0.0

            if abs(t_new - t_old) > 1e-14:
                delta = t_new - t_old
                f[i] = t_new
                # Update conv incrementally
                if len(lin_ks) > 0:
                    conv[lin_ks] += 2.0 * delta * f[lin_ks - i]
                if has_quad:
                    conv[2 * i] += 2.0 * t_old * delta + delta * delta
                pass_improved = True

        # Renormalize
        f_sum = f.sum()
        if f_sum > 1e-15:
            f = f / f_sum
        else:
            f = np.ones(n) / n
            conv = fast_convolve(f)
            best_max = conv.max()
            best_f = f.copy()
            continue

        conv = fast_convolve(f)
        current_max = conv.max()

        if current_max < best_max - 1e-14:
            best_max = current_max
            best_f = f.copy()
        elif not pass_improved:
            break

    return best_f, best_max


def adam_sg_optimize(f0, time_limit, verbose=False):
    """Adam optimizer on log-sum-exp smoothed objective with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.003

    T_start = 8.0 / n
    T_end = 0.3 / n
    top_k_start = max(100, n)
    top_k_end = max(10, n // 10)

    max_iters = int(time_limit / 0.012)
    no_improve = 0
    rng = np.random.default_rng(42)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max(1, int(max_iters * 0.85)))
        T = T_start * (T_end / T_start) ** progress
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 - 0.6 * progress)

        conv = fast_convolve(f)

        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idxs = np.arange(js, je + 1)
            sg[idxs] += 2.0 * w[idx] * f[k - idxs]

        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)

        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
            if no_improve > 500:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

    no_improve = 0
    it = 0
    rng = np.random.default_rng(77)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        g = lp_direction(f, M, n)
        if g is None:
            g = lp_direction(f, M * 1.01, n)
            if g is None:
                no_improve += 1
                if no_improve > 50:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    conv = fast_convolve(f)
                    M = conv.max()
                    no_improve = 0
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                continue
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def mirror_descent(f0, time_limit, verbose=False):
    """Multiplicative weights / mirror descent for the smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    # Mirror descent: f_new[i] = f[i] * exp(-eta * grad[i]) / Z
    eta = 0.5
    T = 5.0 / n
    max_iters = int(time_limit / 0.01)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max(1, int(max_iters * 0.8)))
        eta = 0.5 * (1.0 - 0.5 * progress)
        T = (5.0 / n) * (0.5 / n) ** progress

        conv = fast_convolve(f)
        k_actual = min(max(30, n // 4), 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idxs = np.arange(js, je + 1)
            sg[idxs] += 2.0 * w[idx] * f[k - idxs]

        # Multiplicative update
        update = np.exp(-eta * sg)
        f_new = f * update
        fs = f_new.sum()
        if fs < 1e-15:
            break
        f_new = f_new / fs
        f = f_new

        new_M = fast_convolve(f).max()
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Mirror iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            if no_improve > 400:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, kind='cosine', alpha=0.5, seed=0):
    """Create structured initializations."""
    x = np.arange(n) / max(1, n - 1)
    if kind == 'uniform':
        f = np.ones(n)
    elif kind == 'cosine':
        f = 1.0 + alpha * np.cos(2 * np.pi * x)
    elif kind == 'u_shape':
        f = x ** 2 + (1 - x) ** 2 + 0.1
    elif kind == 'bimodal':
        f = np.exp(-8.0 * (x - 0.2) ** 2) + np.exp(-8.0 * (x - 0.8) ** 2) + 0.05
    elif kind == 'triangular_inv':
        f = 1.0 / (0.3 + np.abs(x - 0.5) * 2.0)
    elif kind == 'random':
        rng = np.random.default_rng(seed)
        f = rng.random(n)
    elif kind == 'symmetric_cos':
        f = 1.0 + alpha * np.cos(4 * np.pi * x * 0.5)
    else:
        f = np.ones(n)
    f = np.maximum(f, 0.01)
    f = f / f.sum()
    return f
