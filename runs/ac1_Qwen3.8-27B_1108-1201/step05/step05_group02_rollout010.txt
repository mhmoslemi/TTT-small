```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using FFT-accelerated subgradient descent
with adaptive temperature, momentum, and multi-length exploration.
Key insight: non-palindromic sequences can have auto-correlation peaks below 1/n.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_fft(f, w, sg_fft_size):
    """Compute subgradient of sum_k w[k] * (f*f)[k] using FFT cross-correlation.
    sg[j] = 2 * sum_{m=0}^{n-1} w[j+m] * f[m]
    """
    W = np.fft.rfft(w, sg_fft_size)
    F = np.fft.rfft(f, sg_fft_size)
    xc = np.fft.irfft(W * np.conj(F), sg_fft_size)
    return 2.0 * xc[:len(f)]


def sg_optimize(f0, time_limit, lr=0.005, momentum=0.92, temperature=0.00003,
                verbose=False, verbose_interval=500):
    """FFT-accelerated subgradient descent with momentum and adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(len(f0)) / len(f0)
    n = len(f)
    sg_fft_size = 1 << (3 * n - 2 - 1).bit_length()
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temperature
    it = 0
    best_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Adaptive softmax weights
        scaled = conv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws > 1e-30:
            w /= ws
        else:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        
        # Subgradient via FFT
        sg = subgradient_fft(f, w, sg_fft_size)
        
        # Project to simplex tangent
        active = f > 1e-14
        if active.any():
            sg[~active] = 0.0
            sg_mean = sg[active].mean()
            sg[active] -= sg_mean
        
        # Momentum update
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % verbose_interval == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new
        
        # Slow decay
        cur_lr *= 0.99997
        cur_temp *= 1.0008
        
        # Periodic restart
        if no_improve > 400:
            rng = np.random.default_rng((it * 2654435761) % (2**31))
            noise_scale = 0.001 + 0.002 * min(no_improve / 400.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(best_lr * 0.7, 0.0008)
            cur_temp = max(temperature * 0.5, 1e-7)
            no_improve = 0
        
        if no_improve > 2000:
            break
    
    return best_f, best_obj

                if not improved and no_improve > 20:
                    break
    
    return best_f, best_obj


def lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = res.x[:n]
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        else:
            g = np.ones(n) / n
        return g
    return None


def lp_fw_step(f, n):
    """One Frank-Wolfe step using min-peak LP."""
    g = lp_min_peak(f, n)
    if g is None:
        return None
    g = np.maximum(g, 0)
    gs = g.sum()
    if gs > 1e-15:
        g = g / gs
    return g
