   import numpy as np
   import time

   def fast_convolve(a):
       """Linear auto-convolution via FFT."""
       n = len(a)
       size = 2 * n - 1
       fft_size = 1 << (size - 1).bit_length()
       A = np.fft.rfft(a, fft_size)
       return np.fft.irfft(A * A, fft_size)[:size]

   def project_simplex(v):
       """Euclidean projection onto the probability simplex."""
       n = len(v)
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u)
       cond = u * np.arange(1, n + 1) > (cssv - 1)
       if not cond.any():
           return np.ones(n) / n
       rho = np.nonzero(cond)[0][-1]
       theta = (cssv[rho] - 1.0) / (rho + 1.0)
       return np.maximum(v - theta, 0.0)

   def compute_lse_obj_grad(a, lam):
       """Compute log-sum-exp objective and gradient.
       Returns (objective, gradient) where objective = (1/lam)*log(sum exp(lam*conv[k]))
       and gradient = d(objective)/da.
       """
       conv = fast_convolve(a)
       max_c = conv.max()
       shifted = lam * (conv - max_c)
       exp_shifted = np.exp(shifted)
       sum_exp = exp_shifted.sum()
       
       if sum_exp < 1e-300:
           obj = max_c  # fallback
           g = np.zeros(len(a))
           return obj, g
       
       log_sum = np.log(sum_exp)
       obj = (log_sum + lam * max_c) / lam  # = max_c + log(sum_exp)/lam
       
       # p[k] = exp(lam*(conv[k]-max_c)) / sum_exp
       p = exp_shifted / sum_exp
       
       # gradient: g[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
       # = 2 * np.correlate(p, a, mode='valid')
       g = 2.0 * np.correlate(p, a, mode='valid')
       
       return obj, g

   def adam_simplex_optimize(a0, time_limit, lam_init=1.0, lam_final=50.0,
                             lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,

           a = np.ones(len(a0)) / len(a0)
       else:
           a = a / s
       n = len(a)
       
       m = np.zeros(n)  # first moment
       v = np.zeros(n)  # second moment
       t = 0
       
       best_obj = np.inf
       best_a = a.copy()
       
       t0 = time.monotonic()
       
       while time.monotonic() - t0 < time_limit:
           t += 1
           
           # Anneal lambda
           progress = min(1.0, t / lam_ramp_iters)
           lam = lam_init + (lam_final - lam_init) * progress
           
           # Compute objective and gradient
           obj, g = compute_lse_obj_grad(a, lam)
           
           # Adam update
           m = beta1 * m + (1 - beta1) * g
           v = beta2 * v + (1 - beta2) * g * g
           m_hat = m / (1 - beta1**t)
           v_hat = v / (1 - beta2**t)
           
           a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
           
           # Project to simplex
