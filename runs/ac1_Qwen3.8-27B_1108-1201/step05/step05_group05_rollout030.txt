Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-method optimization:
log-sum-exp projected gradient with annealing, Gerchberg-Saxton spectral
iteration, coordinate descent, and diverse restarts.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    plt = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, plt)
    return np.fft.irfft(A * A, plt)[:size]

def project_simplex(v, s=1.0):
    """Euclidean projection onto {x>=0, sum(x)=s}."""
    n = len(v)
    if n == 0:
        return v
    if s <= 0:
        return np.zeros(n)
    x = v / s
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)

def compute_lse_grad(a, tau):
    """
    Compute gradient of tau * log(sum(exp(b_k/tau))) where b = a*a.
    Returns (gradient array of length n, b array, p array).
    """
    n = len(a)
    b = fast_convolve(a)
    
    # Softmax weights for the log-sum-exp
    b_scaled = b / tau
    b_scaled -= b_scaled.max()
    exp_b = np.exp(b_scaled)
    p_sum = exp_b.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
    else:
        p = exp_b / p_sum
    
    # Gradient: g_j = 2 * sum_k p[k] * a[k-j] (valid k-j in [0,n-1])
    # This is 2 * cross_correlation(p, a)
    # Using FFT: corr = IFFT(FFT(p) * conj(FFT(a)))
    corr_size = len(p) + n - 1  # 3n-2
    plt = 1 << (corr_size - 1).bit_length()

    A = np.fft.rfft(a, plt)
    corr = np.fft.irfft(P * np.conj(A), plt)

    # We want the part where j is in [0, n-1]
    # The cross-correlation corr[j] for j=0,...,n-1 is what we need
    g = 2.0 * corr[:n]
    
    return g, b, p

def gerchberg_saxton(a_init, time_limit, n_iter_target=None, verbose=False):
    """
    Gerchberg-Saxton style iteration: alternate between enforcing
    flat spectrum and enforcing non-negativity.
    """
    a = project_simplex(a_init.copy())
    n = len(a)
    size = 2 * n - 1
    plt = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Forward FFT
        A = np.fft.rfft(a, plt)
        
        # Target: flat magnitude spectrum
        # |A|^2 should be proportional to 1 (flat power spectrum)
        # |A| should be constant
        target_mag = np.sqrt(size / plt)  # normalized
        # Actually, let's use the average magnitude as target
        mag = np.abs(A)
        avg_mag = mag.mean()
        
        # Blend: move magnitude toward flat, keep phase
        # Soft enforcement
        phase = np.angle(A)
        new_mag = 0.7 * mag + 0.3 * avg_mag  # blend toward flat
        A_new = new_mag * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, plt)[:n]
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Evaluate
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose and it % 50 == 0:
                print(f"      GS iter {it}: obj={best_obj:.10f}")
        
        a = a_new
    
    return best_a, best_obj

def gerchberg_saxton_v2(a_init, time_limit, verbose=False):
    """
    Improved GS: use the actual convolution flatness as the spectral target.
    Target b_k = 1/(2n-1) for all k (flat convolution).
    This corresponds to |A(w)|^2 being a specific function of frequency.
    """
    a = project_simplex(a_init.copy())
    n = len(a)
    size = 2 * n - 1
    plt = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Check time
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current spectrum
        A = np.fft.rfft(a, plt)
        B = A * A  # power spectrum (for real signal, B = A^2, not |A|^2)
        
        # Actually, for real a: FFT(a * a) = IFFT(FFT(a) * FFT(a)) (circular)
        # But we want linear convolution, so this isn't exact.
        # Let me use a different approach.
        
        # Modified GS: 
        # 1. Compute b = a*a (linear convolution)
        # 2. Modify b to be flatter
        # 3. Find a' such that a'*a' ≈ b_modified and a' >= 0
        
        # This is more complex. Let me use the spectral approach differently.
        # 
        # Alternative: minimize variance of b
        # b should be flat at level 1/(2n-1) (when sum(a)=1)
        # 
        # In frequency domain: b = IFFT(FFT(a)^2) (linear, with zero padding)
        # We want b to be flat, so we want FFT(a)^2 to be the FFT of a flat signal.
        # FFT of flat signal (length 2n-1, value 1/(2n-1)) is a delta at frequency 0.
        # So we want A(w) to be concentrated at w=0, which means a is a delta.
        # That contradicts non-negativity giving a spread-out sequence.
        
        # OK let me abandon this approach and just do the simple spectral flattening.
        
        # Simple approach: blend current spectrum toward its average magnitude
        mag = np.abs(A)
        phase = np.angle(A)
        avg = mag.mean()
        
        # Use a "soft" flattening - push magnitudes toward average
        # but allow some variation
        alpha_blend = 0.5 + 0.3 * min(it / 1000.0, 1.0)  # increase over time
        new_mag = (1 - alpha_blend) * mag + alpha_blend * avg
        A_new = new_mag * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_new, plt)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def coord_descent(a_init, time_limit, verbose=False):
    """
    Coordinate descent: optimize one coordinate at a time.
    For each coordinate j, find the optimal value given all others fixed.
    """
    a = project_simplex(a_init.copy())
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        pass_improve = 0
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            # Try several multipliers
            best_val = old_val
            best_obj_j = best_obj
            
            # The multipliers to try
            for mult in [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0]:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-12 or new_val < 0:
                    continue
                
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                
                scale = remaining / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial /= a_trial.sum()
                
                obj_j = fast_convolve(a_trial).max()
                if obj_j < best_obj_j - 1e-15:
                    best_obj_j = obj_j
                    best_val = new_val
            
            if best_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_val) / denom
                    a[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a /= a.sum()
                
                if best_obj_j < best_obj - 1e-14:
                    best_obj = best_obj_j
                    best_a = a.copy()
                improved = True
                pass_improve += 1
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
            if verbose:
                print(f"      CD pass: {pass_improve} coords improved, obj={best_obj:.10f}")
    
    return best_a, best_obj

def diverse_init(n, method, seed=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    
    if method == 'uniform':
        a = np.ones(n) / n
    elif method == 'random':
        a = rng.random(n)
        a /= a.sum()
    elif method == 'gaussian':
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
        a /= a.sum()
    elif method == 'bilinear':
        # Triangular (bilinear) - like a Fejer kernel
        x = np.arange(n)
        a = np.maximum(0, 1 - 2*abs(x - (n-1)/2) / n)
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        a /= a.sum()
    elif method == 'barker_like':
        # Based on Barker sequence pattern (non-negative version)
        # Barker length 13: +1 +1 +1 -1 +1 -1 -1 ... 
        barker = np.array([1, 1, 1, -1, 1, -1, -1])
        # Tile and smooth
        n_tiles = (n + len(barker) - 1) // len(barker)
        pattern = np.tile(barker, n_tiles)[:n]
        a = (pattern + 2) / 3.0  # map to [1/3, 1]
        a /= a.sum()
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = np.abs(np.cos(x)) ** 2
        a /= a.sum()
    elif method == 'sinc':
        x = (np.arange(n) - n/2) / 5.0
        a = np.sinc(x) ** 2
        a = np.maximum(a, 0)
        a /= a.sum()
    else:
        a = np.ones(n) / n
    
    return a

"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    plt = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, plt)
    return np.fft.irfft(A * A, plt)[:size]


def project_simplex(v, s=1.0):
    """Euclidean projection onto {x>=0, sum(x)=s}."""
    n = len(v)
    if n == 0:
        return v
    if s <= 0:
        return np.zeros(n)
    x = v / s
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def compute_lse_gradient(a, tau):
    """
    Compute gradient of tau*log(sum(exp(b_k/tau))) where b = a*a.
    Returns (gradient of length n, b array, p array, obj value).
    """
    n = len(a)
    b = fast_convolve(a)

    b_scaled = b / tau
    b_scaled -= b_scaled.max()
    exp_b = np.exp(b_scaled)
    p_sum = exp_b.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
    else:
        p = exp_b / p_sum

    # Gradient: g_j = 2 * sum_k p[k] * a[k-j] (valid k-j in [0,n-1])
    # cross_correlation(p, a)[j] = sum_k p[k] * a[k-j]
    corr_size = len(p) + n - 1
    plt = 1 << (corr_size - 1).bit_length()
    P = np.fft.rfft(p, plt)
    A = np.fft.rfft(a, plt)
    corr = np.fft.irfft(P * np.conj(A), plt)

    g = 2.0 * corr[:n]
    obj = b.max()
    return g, b, p, obj


def lse_projected_grad(a_init, time_limit, tau_init=0.05, tau_end=0.0005,
                       lr_init=0.005, lr_decay=0.9995, restart_every=300,
                       verbose=False):
    """Projected gradient descent on log-sum-exp of auto-convolution."""
    a = project_simplex(a_init.copy())
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (tau_end / tau_init) ** progress

        g, b, p, obj = compute_lse_gradient(a, tau)

        a_new = a - lr * g
        a_new = project_simplex(a_new)

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      LSE-PGD iter {it}: obj={best_obj:.10f}, tau={tau:.6f}")
        else:
            no_improve += 1
            if no_improve > restart_every:
                rng = np.random.default_rng(it * 7 + 3)
                noise = rng.standard_normal(n) * 0.008
                a = project_simplex(best_a + noise)
                a_new = a.copy()
                no_improve = 0
                lr = lr_init * 0.4
                tau = max(tau * 3.0, tau_end * 20.0)

        a = a_new
        lr *= lr_decay
        if lr < 1e-9:
            lr = 1e-9

    return best_a, best_obj


def mirror_descent_lse(a_init, time_limit, tau_init=0.05, tau_end=0.0005,
                       lr_init=0.003, verbose=False):
    """Mirror descent (multiplicative weights) on log-sum-exp of auto-convolution."""
    a = project_simplex(a_init.copy())
    n = len(a)
    a = a + 1e-15
    a = a / a.sum()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (tau_end / tau_init) ** progress

        g, b, p, obj = compute_lse_gradient(a, tau)

        # Mirror descent step: a_new[j] prop to a[j] * exp(-lr * g[j])
        log_step = -lr * g
        log_step -= log_step.max()
        a_new = a * np.exp(log_step)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      Mirror iter {it}: obj={best_obj:.10f}, tau={tau:.6f}")
        else:
            no_improve += 1
            if no_improve > 400:
                rng = np.random.default_rng(it * 13 + 5)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                a = a + 1e-15
                a = a / a.sum()
                a_new = a.copy()
                no_improve = 0
                lr = lr_init * 0.3
                tau = max(tau * 2.5, tau_end * 15.0)

        a = a_new
        lr *= 0.9997

    return best_a, best_obj


def gerchberg_saxton_v2(a_init, time_limit, verbose=False):
    """GS spectral iteration: alternate between flattening spectrum and enforcing non-negativity."""
    a = project_simplex(a_init.copy())
    n = len(a)
    size = 2 * n - 1
    plt = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        A = np.fft.rfft(a, plt)
        mag = np.abs(A)
        phase = np.angle(A)
        avg = mag.mean()

        # Adaptive blending: start gentle, get stronger
        alpha_blend = 0.3 + 0.4 * min(it / 2000.0, 1.0)
        new_mag = (1.0 - alpha_blend) * mag + alpha_blend * avg
        A_new = new_mag * np.exp(1j * phase)

        a_new = np.fft.irfft(A_new, plt)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose and it % 100 == 0:
                print(f"      GS iter {it}: obj={best_obj:.10f}")

        a = a_new

    return best_a, best_obj


def coord_descent(a_init, time_limit, verbose=False):
    """Coordinate descent: optimize one coordinate at a time with geometric search."""
    a = project_simplex(a_init.copy())
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    multipliers = [0.4, 0.6, 0.75, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-13:
                continue

            best_val = old_val
            best_obj_j = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-12 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = remaining / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial /= a_trial.sum()

                obj_j = fast_convolve(a_trial).max()
                if obj_j < best_obj_j - 1e-15:
                    best_obj_j = obj_j
                    best_val = new_val

            if best_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_val) / denom
                    a[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a /= a.sum()
                if best_obj_j < best_obj - 1e-14:
                    best_obj = best_obj_j
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
            if verbose:
                print(f"      CD: obj={best_obj:.10f}")

    return best_a, best_obj


def diverse_init(n, method, seed=0):
    """Generate diverse initial sequences of length n."""
    rng = np.random.default_rng(seed)

    if method == 'uniform':
        a = np.ones(n) / n
    elif method == 'random':
        a = rng.random(n)
        a /= a.sum()
    elif method == 'gaussian':
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n / 2.0) / (n / 4.0)) ** 2)
        a /= a.sum()
    elif method == 'bilinear':
        x = np.arange(n)
        a = np.maximum(0.0, 1.0 - 2.0 * np.abs(x - (n - 1) / 2.0) / n)
        if a.sum() < 1e-10:
            a = np.ones(n) / n
        a /= a.sum()
    elif method == 'barker_like':
        barker = np.array([1, 1, 1, -1, 1, -1, -1])
        n_tiles = (n + len(barker) - 1) // len(barker)
        pattern = np.tile(barker, n_tiles)[:n]
        a = (pattern + 2.0) / 3.0
        a /= a.sum()
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = np.abs(np.cos(x)) ** 2
        a /= a.sum()
    elif method == 'sinc':
        x = (np.arange(n) - n / 2.0) / 5.0
        a = np.sinc(x) ** 2
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif method == 'flat_spectrum':
        # Generate via flat-magnitude spectrum
        plt = 1 << (2 * n - 2).bit_length()
        phases = rng.uniform(0, 2 * np.pi, plt // 2 + 1)
        spectrum = np.exp(1j * phases)
        sig = np.fft.irfft(spectrum, plt)[:n]
        sig = np.maximum(sig, 0.0)
        if sig.sum() < 1e-10:
            sig = np.abs(sig[:n])
        ss = sig.sum()
        if ss > 1e-10:
            a = sig / ss
        else:
            a = np.ones(n) / n
    else:
        a = np.ones(n) / n

    return a


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def variance_grad_step(a):

    b = fast_convolve(a)

    corr = np.fft.irfft(B * np.conj(A), plt)

    g = 2.0 * corr[:n]
    return g, b

    print("\n=== Phase 6: Diverse Inits ===")
    if best_a is not None:
        n_best = len(best_a)
        init_configs = [
            ('flat_spec', 'flat_spectrum', 111),
            ('bilinear', 'bilinear', 42),
            ('cosine', 'cosine', 17),
            ('sinc', 'sinc', 99),
            ('barker', 'barker_like', 7),
            ('gaussian', 'gaussian', 55),
        ]

        for name, method, s in init_configs:
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            a_init = diverse_init(n_best, method, seed=s)
            a_init = project_simplex(0.4 * a_init + 0.6 * best_a)
            t_alloc = min(rem * 0.08, 80)
            if t_alloc < 12:
                continue
            print(f"  Diverse [{name}] n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = lse_projected_grad(a_init, time_limit=t_alloc,
                                           tau_init=0.08, tau_end=0.001,
                                           lr_init=0.008, lr_decay=0.999,
                                           restart_every=150)
            try_seq(a_opt, f"diverse_{name}")

    # === Phase 7: Coordinate Descent ===
    print("\n=== Phase 7: Coordinate Descent ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_cd = min(rem * 0.25, 200)
            print(f"  CD n={len(best_a)}, t={t_cd:.0f}s")
            a_opt, _ = coord_descent(best_a, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "cd")

            rem = deadline - time.monotonic()
            if rem > 25:
                t_cd2 = min(rem * 0.15, 100)
                a_opt2, _ = coord_descent(a_opt, time_limit=t_cd2)
                try_seq(a_opt2, "cd2")

    # === Phase 8: Fine LSE with small tau ===
    print("\n=== Phase 8: Fine LSE ===")
    if best_a is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t8 = min(rem * 0.2, 120)
            print(f"  Fine LSE n={len(best_a)}, t={t8:.0f}s")
            a_opt, _ = lse_projected_grad(best_a, time_limit=t8,
                                           tau_init=0.004, tau_end=0.0001,
                                           lr_init=0.0015, lr_decay=0.9998,
                                           restart_every=400)
            try_seq(a_opt, "fine_lse")

    # === Phase 9: Final perturbations and polish ===
    print("\n=== Phase 9: Final Polish ===")
    if best_a is not None:
        n_best = len(best_a)
        rng = np.random.default_rng(seed + 777)
        for i in range(4):
            rem = deadline - time.monotonic()
            if rem < 12:
                break
            noise = rng.standard_normal(n_best) * (0.0015 + 0.001 * i)
            a_pert = project_simplex(best_a + noise)
            t_pert = min(rem * 0.12, 50)
            print(f"  Perturb {i}, t={t_pert:.0f}s")
            a_opt, _ = lse_projected_grad(a_pert, time_limit=t_pert,
                                           tau_init=0.002, tau_end=0.00005,
                                           lr_init=0.001, lr_decay=0.9999,
                                           restart_every=150)
            try_seq(a_opt, f"perturb_{i}")

        # Final coordinate descent
        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = coord_descent(best_a, time_limit=min(rem * 0.25, 30))
            try_seq(a_opt, "final_cd")

        # Tiny mirror descent
        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = mirror_descent_lse(best_a, time_limit=min(rem * 0.2, 20),
                                           tau_init=0.003, tau_end=0.0001,
                                           lr_init=0.001)
            try_seq(a_opt, "final_mirror")

    # Fallback
    if best_a is None:
        best_a = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_a = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
