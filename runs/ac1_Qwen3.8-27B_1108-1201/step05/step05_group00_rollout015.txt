def peak_balance_lp(a, M, time_limit, verbose=False, label=""):
    """
    Find direction g that minimizes the maximum convolution peak.
    Solves: min max_k (a*g)[k] s.t. g >= 0, sum(g) = 1
    using bisection on M and LP.
    Returns the minimizing g and its max peak value.
    """
    n = len(a)
    # The LP: min t s.t. (a*g)[k] <= t for all k, g >= 0, sum(g) = 1
    # Using bisection on t:
    lo = 0.0
    hi = M * 1.1

    # First, find a feasible upper bound
    g_uniform = np.ones(n) / n
    hi = fast_convolve(g_uniform).max()  # This is 1/n
    
    # Actually we want the LP direction, not the objective direction
    # Let's solve: min max_k sum_j a[j]*g[k-j] s.t. g >= 0, sum(g)=1
    # By bisection: for a given t, check feasibility of sum_j a[j]*g[k-j] <= t for all k

    t0 = time.monotonic()
    for _ in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        mid = (lo + hi) / 2.0

        # LP: find g >= 0, sum(g) = 1, (a*g)[k] <= mid for all k
        # Variables: g[0], ..., g[n-1]
        # Constraints: sum_j a[j]*g[k-j] <= mid for k=0,...,2n-2
        #              sum(g) = 1
        #              g >= 0

        c = np.zeros(n)  # minimize 0
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            # (a*g)[k] = sum_j a[j] * g[k-j], valid when 0 <= k-j < n
            # For fixed j, k-j is valid when j <= k <= j+n-1
            # So A_ub[k, j] = a[j] if j <= k <= j+n-1
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            A_ub[k_start:k_end + 1, j] = a[j]
        b_ub = np.full(2 * n - 1, mid)
        A_eq = np.ones((1, n))
        b_eq = np.ones(1)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 2.0})
            if res.success:
                hi = mid
            else:
                lo = mid
        except Exception:
            break

    # Now solve at the found hi to get the actual g
    try:
        c = np.zeros(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            A_ub[k_start:k_end + 1, j] = a[j]
        b_ub = np.full(2 * n - 1, hi)
        A_eq = np.ones((1, n))
        b_eq = np.ones(1)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs
                return g, hi
    except Exception:
        pass
    return None, hi

def kkt_polish(a0, time_limit, verbose=True, label=""):

    n = len(a)

    it = 0

            break

        conv = fast_convolve(a)
        t_cur = conv.max()

            active_peaks = np.argsort(conv)[-5:]

        # For each active peak k, the "gradient direction" is:
        # d_k[i] = sum_j H_k[i,j] * a[j] = a[k-i] (for valid indices)
        # We want to find a direction d such that for all active k:
        #   sum_i d_k[i] * delta[i] = 0 (keep peaks equal)
        #   sum_i delta[i] = 0 (keep sum constant)
        #   delta moves in the direction that reduces the max

        # Build the system: for each active peak, the contribution vector
        # c_k[i] = number of pairs (j, m) with j+m=k, j=i or m=i
        # Actually, d/d a[i] of (a*a)[k] = sum_j (H_k)_{i,j} * 2 * a[j] (since symmetric)
        # = 2 * a[k-i] (if valid)

        # We want: for all active k, sum_i (2*a[k-i])*delta[i] = 0
        # This gives a linear constraint on delta.
        # We also want sum_i delta[i] = 0.
        # The remaining degrees of freedom: minimize the increase in any non-active peak.

        # Simple approach: solve for delta that equalizes active peaks
        # and minimizes ||delta||^2 (steepest descent in the feasible direction)

        # Build constraint matrix C where C[k, i] = 2*a[active_peaks[k]-i] (if valid)
        n_active = len(active_peaks)
        C = np.zeros((n_active + 1, n))
        for idx, k in enumerate(active_peaks):
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    C[idx, i] = 2.0 * a[j]
        C[n_active, :] = 1.0  # sum constraint

        # Find the minimum-norm delta in the null space of C
        # that decreases the objective
        # Use SVD to find null space
        try:
            U, S, Vt = np.linalg.svd(C, full_matrices=True)
        except np.linalg.LinAlgError:
            break

        # Null space starts at index len(C)
        null_start = C.shape[0]
        if null_start >= Vt.shape[0]:
            break
        null_space = Vt[null_start:].T  # n x (n - null_start)

        # Find direction in null space that decreases the max
        # Gradient of max ≈ gradient at the active peaks (equalized)
        # d_obj/da_i ≈ 2 * sum_{k active} a[k-i] (uniform weighting)
        grad_approx = np.zeros(n)
        for k in active_peaks:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad_approx[i] += 2.0 * a[j]
        grad_approx /= n_active

        # Project gradient onto null space
        grad_proj = null_space @ (null_space.T @ grad_approx)

        # Move in the negative gradient direction (within null space)
        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-15:
            no_improve += 1
            if no_improve > 20:
                # Try small random perturbation
                rng = np.random.default_rng((it * 12345 + 7) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                no_improve = 0
            continue

        # Line search
        best_step = 0.0
        best_val = t_cur
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            delta = -step * grad_proj / gnorm
            candidate = project_simplex(a + delta)
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_step = step

        if best_step > 0:
            delta = -best_step * grad_proj / gnorm
            a = project_simplex(a + delta)
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng((it * 67891 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    [{label}] KKT it={it}, obj={best_obj:.10f}, n_active={n_active}")

    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=True, label=""):
    """
    Coordinate-wise polish: for each coordinate, find the optimal value
    given all others fixed, using golden section search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve_total = 0

    while time.monotonic() - t0 < time_limit - 0.5:
        no_improve_pass = 0
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            # Fix all coordinates except a[i]
            # a[i] is constrained: a[i] >= 0, and sum(a) = 1
            # If we change a[i] to v, the sum changes by (v - a[i])
            # To maintain sum=1, we could either:
            #   (a) Just change a[i] and renormalize (but this changes all coords)
            #   (b) Change a[i] and distribute the difference to other coords
            # For simplicity, use (a): change a[i] to v, then renormalize

            if a[i] < 1e-10:
                continue

            # Golden section search for optimal a[i]
            # Try a range around current value
            lo = max(0.0, a[i] * 0.5)
            hi = a[i] * 1.5 + 0.01 / n

            # Save current
            a_save = a.copy()
            a_best_i = a[i]
            obj_best_i = best_obj

            golden = 0.6180339887498949
            c1 = hi - golden * (hi - lo)
            c2 = lo + golden * (hi - lo)

            # Evaluate at c1 and c2
            a[i] = c1
            a[i] = c1 / (1.0 + (c1 - a_save[i]))  # renormalize approximately
            # Actually just set and renormalize whole thing
            a[i] = c1
            a = a / a.sum()
            f1 = fast_convolve(a).max()

            a = a_save.copy()
            a[i] = c2
            a = a / a.sum()
            f2 = fast_convolve(a).max()

            # A few golden section iterations
            for _ in range(5):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - golden * (hi - lo)
                    a = a_save.copy()
                    a[i] = c1
                    a = a / a.sum()
                    f1 = fast_convolve(a).max()
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + golden * (hi - lo)
                    a = a_save.copy()
                    a[i] = c2
                    a = a / a.sum()
                    f2 = fast_convolve(a).max()

            # Take the better
            if f1 < f2:
                a = a_save.copy()
                a[i] = c1
                a = a / a.sum()
                f_best = f1
            else:
                a = a_save.copy()
                a[i] = c2
                a = a / a.sum()
                f_best = f2

            if f_best < best_obj - 1e-15:
                best_obj = f_best
                best_a = a.copy()
                no_improve_pass = 0
            else:
                a = a_save.copy()
                no_improve_pass += 1

        if no_improve_pass < n // 2:
            no_improve_total = 0
        else:
            no_improve_total += 1

        if no_improve_total > 5:
            break

        if verbose:
            print(f"    [{label}] Coord polish: obj={best_obj:.10f}")

    return best_a, best_obj


def generate_structured_init(n, parent=None, rng=None):
    """Generate a structured initialization."""
    if rng is None:
        rng = np.random.default_rng(42)

    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    if parent is not None:
        # Interpolate parent to length n
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Two Gaussians
    g2 = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.7 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.1
    inits.append(c / c.sum())

    # Dirichlet random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-stage optimization:
    Stage 1: Quick exploration of n values with smoothed gradient
    Stage 2: Deep Nesterov smoothed gradient on best n
    Stage 3: FW with peak balancing LP
    Stage 4: KKT Newton polish
    Stage 5: Coordinate polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
    n_parent = len(parent)

    # === Stage 1: Quick multi-n exploration (~300s) ===
    print("\n=== Stage 1: Quick Multi-n Exploration ===")
    t_start = time.monotonic()
    stage1_budget = 300.0

    candidate_ns = sorted(set([
        n_parent,
        n_parent - 128 if n_parent > 256 else n_parent,
        n_parent + 128,
        512, 640, 768, 896, 1024, 1280, 1536
    ]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]

    for n_try in candidate_ns:
        if time.monotonic() - t_start > stage1_budget - 30:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        # Quick smoothed gradient
        t_alloc = min(30.0, stage1_budget - (time.monotonic() - t_start) - 20.0)
        if t_alloc < 5:
            break
        a_opt, _ = nesterov_smoothed_grad(start, time_limit=t_alloc,
                                          T_init=0.002, lr_init=0.05,
                                          verbose=False, label=f"exp_n{n_try}")
        try_seq(a_opt, f"exp_n{n_try}")

    print(f"  Stage 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    if best_seq is None:
        best_seq = parent.copy()
        best_score = evaluate_sequence(best_seq.tolist())

    # === Stage 2: Deep Nesterov smoothed gradient (~500s) ===
    print("\n=== Stage 2: Deep Nesterov Smoothed Gradient ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Run with different temperature settings
    configs = [
        {'T_init': 0.0005, 'lr_init': 0.1, 'mom_init': 0.85, 't': 200},
        {'T_init': 0.001, 'lr_init': 0.08, 'mom_init': 0.9, 't': 150},
        {'T_init': 0.0003, 'lr_init': 0.06, 'mom_init': 0.88, 't': 150},
    ]

    for ci, cfg in enumerate(configs):
        remaining = stage2_budget_remaining = 500.0 - (time.monotonic() - t_start)
        if remaining < 20:
            break
        t_alloc = min(cfg['t'], remaining)
        print(f"  Config {ci}: T={cfg['T_init']}, lr={cfg['lr_init']}, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_smoothed_grad(best_seq, time_limit=t_alloc,
                                          T_init=cfg['T_init'], lr_init=cfg['lr_init'],
                                          mom_init=cfg['mom_init'],
                                          verbose=True, label=f"nest_{ci}")
        try_seq(a_opt, f"nesterov_{ci}")

    print(f"  Stage 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Stage 3: FW with Peak Balancing LP (~400s) ===
    print("\n=== Stage 3: FW with Peak Balancing ===")
    t_start = time.monotonic()
    stage3_budget = 400.0

    # Run FW with peak balance
    t_alloc = min(200.0, stage3_budget - 20.0)
    print(f"  FW-peak t={t_alloc:.0f}s")
    a_opt, _ = fw_with_peak_balance(best_seq, time_limit=t_alloc,
                                     alpha_init=0.2, verbose=True, label="fw_peak")
    try_seq(a_opt, "fw_peak")

    # Run again from the result with smaller alpha
    remaining = stage3_budget - (time.monotonic() - t_start) - 10.0
    if remaining > 30:
        t_alloc = min(remaining, 150.0)
        print(f"  FW-peak fine t={t_alloc:.0f}s")
        a_opt2, _ = fw_with_peak_balance(best_seq, time_limit=t_alloc,
                                          alpha_init=0.1, verbose=True, label="fw_fine")
        try_seq(a_opt2, "fw_peak_fine")

    print(f"  Stage 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Stage 4: KKT Newton Polish (~300s) ===
    print("\n=== Stage 4: KKT Newton Polish ===")
    t_start = time.monotonic()
    stage4_budget = 300.0

    t_alloc = min(200.0, stage4_budget - 20.0)
    print(f"  KKT polish t={t_alloc:.0f}s")
    a_opt, _ = kkt_polish(best_seq, time_limit=t_alloc,
                          verbose=True, label="kkt")
    try_seq(a_opt, "kkt")

    # Second KKT pass
    remaining = stage4_budget - (time.monotonic() - t_start) - 10.0
    if remaining > 20:
        t_alloc = min(remaining, 100.0)
        a_opt2, _ = kkt_polish(best_seq, time_limit=t_alloc,
                               verbose=True, label="kkt2")
        try_seq(a_opt2, "kkt2")

    print(f"  Stage 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Stage 5: Final Polish (~270s) ===
    print("\n=== Stage 5: Final Polish ===")
    t_start = time.monotonic()
    stage5_budget = 270.0

    # Nesterov with very small T (near-exact max)
    remaining = stage5_budget - 20.0
    if remaining > 20:
        t_alloc = min(100.0, remaining)
        print(f"  Fine Nesterov t={t_alloc:.0f}s")
        a_opt, _ = nesterov_smoothed_grad(best_seq, time_limit=t_alloc,
                                          T_init=0.0001, lr_init=0.02,
                                          mom_init=0.95,
                                          verbose=True, label="fine_nest")
        try_seq(a_opt, "fine_nest")

    # Coordinate polish
    remaining = stage5_budget - (time.monotonic() - t_start) - 10.0
    if remaining > 20:
        t_alloc = min(80.0, remaining)
        print(f"  Coord polish t={t_alloc:.0f}s")
        a_opt2, _ = coordinate_polish(best_seq, time_limit=t_alloc,
                                      verbose=True, label="coord")
        try_seq(a_opt2, "coord")

    # Final tiny perturbations
    remaining = stage5_budget - (time.monotonic() - t_start) - 5.0
    if remaining > 10:
        n_best = len(best_seq)
        for perturb_scale in [0.0005, 0.001, 0.002, 0.003, 0.005]:
            remaining = stage5_budget - (time.monotonic() - t_start) - 3.0
            if remaining < 5:
                break
            for _ in range(3):
                remaining = stage5_budget - (time.monotonic() - t_start) - 2.0
                if remaining < 3:
                    break
                noise = rng.standard_normal(n_best) * perturb_scale
                f_pert = project_simplex(best_seq + noise)
                a_opt3, _ = nesterov_smoothed_grad(f_pert, time_limit=min(remaining, 8.0),
                                                   T_init=0.0002, lr_init=0.01,
                                                   mom_init=0.9, verbose=False,
                                                   label="perturb")
                try_seq(a_opt3, f"perturb_{perturb_scale}")

    print(f"  Stage 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
