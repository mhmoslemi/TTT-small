def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, w_sparse):
    """
    Compute gradient: grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w_sparse[j+m]
    using cross-correlation.
    """
    n = len(a)
    # w_sparse has length >= n, a has length n
    # We want corr[j] = sum_{m=0}^{n-1} a[m] * w_sparse[j+m] for j=0,...,n-1
    # This is np.correlate(w_sparse, a, 'valid') if len(w_sparse) >= len(a)
    if len(w_sparse) < n:
        w_sparse = np.pad(w_sparse, (0, n - len(w_sparse)))
    corr = np.correlate(w_sparse[:2*n-1], a, mode='valid')
    # corr should have length min(len(w_sparse[:2n-1]), len(a)) = n (if 2n-1 >= n)
    if len(corr) < n:
        corr = np.pad(corr, (0, n - len(corr)))
    return 2.0 * corr[:n]


def get_smoothed_weights(conv, temperature, top_k=None):
    """Get softmax weights over convolution peaks."""
    n_conv = len(conv)
    if top_k is not None and top_k < n_conv:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
    else:
        idx = np.arange(n_conv)
        vals = conv
    
    scaled = temperature * vals
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w_new = np.zeros(n_conv)
        w_new[
            np.array(idx)[np.argmax(vals)]
        ] = 1.0
        return w_new
    
    w_new = np.zeros(n_conv)
    w_new[idx] = w / ws
    return w_new


def mu_optimize(f0, time_limit, top_k=50, temperature=0.0001, lr=0.004,
                momentum=0.87, temp_growth=1.01, lr_decay=0.9995,
                verbose=False, label=""):
    """
    Multiplicative-weights subgradient with Nesterov momentum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        exp_arg = momentum * 0.5 * v
        exp_arg -= exp_arg.max()
        f_nest = f * np.exp(exp_arg)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn
        
        conv_nest = fast_convolve(f_nest)
        w_sparse = get_smoothed_weights(conv_nest, cur_temp, cur_top_k)
        sg = compute_gradient(f_nest, w_sparse)
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * sg
        
        # Multiplicative weights update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30 or not np.isfinite(fs):
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0001)
            no_improve = min(no_improve, 100)
            continue
        f = np.maximum(f, 0.0)
        f /= f.sum()
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive schedule
        cur_lr *= lr_decay
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp *= temp_growth
        
        # Periodic top_k reduction
        if it % 1000 == 0:
            cur_top_k = max(3, int(cur_top_k * 0.9))
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.001 + 0.002 * min(no_improve // 400, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_temp = temperature * 0.3
            no_improve = 0
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MU iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, lr={cur_lr:.6f}, temp={cur_temp:.2e}")
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, try scaling it by various factors
    and redistribute the rest proportionally.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    # Process coordinates in an order that prioritizes high values
    order = np.argsort(f)[::-1]
    
    multipliers = [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            old_val = f[j]
            if old_val < 1e-10:
                continue
            
            best_mult = 1.0
            best_local_obj = fast_convolve(f).max()
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                # Redistribute: scale all other coordinates to maintain sum=1
                other_sum = 1.0 - old_val
                if other_sum < 1e-10:
                    continue
                new_other_sum = 1.0 - new_val
                if new_other_sum < 0:
                    continue
                scale = new_other_sum / other_sum
                
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                
                val = fast_convolve(f_trial).max()
                if val < best_local_obj - 1e-14:
                    best_local_obj = val
                    best_mult = mult
            
            if best_mult != 1.0:
                new_val = old_val * best_mult
                other_sum = 1.0 - old_val
                new_other_sum = 1.0 - new_val
                scale = new_other_sum / other_sum
                f[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = np.maximum(f, 0)
                f /= f.sum()
                
                if best_local_obj < best_obj - 1e-14:
                    best_obj = best_local_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def peak_pair_search(f0, time_limit, verbose=False):
    """
    For each top peak, find the pair of coordinates contributing most,
    and try to reduce their product by transferring mass.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        top_k = min(10, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        improved = False
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            
            if products.max() < 1e-20:
                continue
            
            # Find the pair with the largest product
            hi_local = np.argmax(products)
            hi = int(idx[hi_local])
            lo = int(k - hi)  # The paired index
            
            if hi == lo or f[hi] < 1e-10:
                continue
            
            # Try transferring a fraction of f[hi] to f[lo]
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = f[hi] * frac
                f[hi] -= amt
                f[lo] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[hi] += amt
                    f[lo] -= amt
                    break
        
        # Also try random transfers guided by peaks
        if not improved:
            trials = 100
            for _ in range(trials):
                if time.monotonic() - t0 > time_limit:
                    break
                # Pick a random high-value coordinate and transfer to a random low-value one
                hi_candidates = np.where(f > np.median(f))[0]
                lo_candidates = np.where(f < np.median(f))[0]
                if len(hi_candidates) == 0 or len(lo_candidates) == 0:
                    break
                hi = int(rng.choice(hi_candidates))
                lo = int(rng.choice(lo_candidates))
                if hi == lo:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[hi] * frac
                    f[hi] -= amt
                    f[lo] += amt
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[hi] += amt
                        f[lo] -= amt
                        break
            
            if not improved:
                # Try a larger perturbation and re-evaluate
                noise = rng.standard_normal(n) * 0.005
                f_trial = project_simplex(f + noise)
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                else:
                    break  # No improvement, stop
        
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
