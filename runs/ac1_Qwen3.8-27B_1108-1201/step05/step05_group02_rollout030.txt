Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a hybrid strategy:
1. LP-based Frank-Wolfe with symmetric constraint reduction
2. L-BFGS in softmax parameterization for smooth optimization
3. Subgradient descent with adaptive peak weighting

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    if old_n == 1:
        return np.full(new_n, f[0])
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def solve_lp_min_peak_symmetric(f, n, top_k=None):
    """Solve LP: min t s.t. (f*g)[k] <= t for k=0..n-1 (using symmetry), g>=0, sum(g)=1.
    Only uses first n constraints (the rest are symmetric)."""
    if top_k is not None:
        # Precompute which constraints to use (but we don't know the peak of f*g yet)
        # Just use all n constraints for simplicity
        pass
    
    n_vars = n + 1  # n components of g + 1 for t
    c = np.zeros(n_vars)
    c[-1] = 1.0  # minimize t
    
    n_constraints = n  # only k = 0, ..., n-1
    A_ub = np.zeros((n_constraints, n_vars))
    for j in range(n):
        # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * g[j]
        # For fixed k, the constraint is: sum_j f[k-j] * g[j] - t <= 0
        # where f[k-j] is valid only when 0 <= k-j <= n-1
        for k in range(n_constraints):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[k, j] = f[k - j]
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None
