"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smoothed projected gradient
with Nesterov momentum, LP refinement, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, t):
    """
    Compute gradient of F(a) = (1/t) * log(sum_k exp(t * c[k]))
    where c = convolve(a, a).
    
    Returns (gradient, F_value, c_array).
    """
    n = len(a)
    c = fast_convolve(a)
    
    # Compute softmax weights
    tc = t * c
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    p = exp_tc / exp_tc.sum()
    
    # Gradient: grad[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
    # = cross-correlation of p and a, first n values
    # Using FFT: ifft(fft(p) * conj(fft(a)))
    size = 2 * n - 1
    fft_size = 1
    while fft_size < 2 * size - 1:
        fft_size *= 2
    
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(P * np.conj(A), fft_size)
    grad = cross[:n]
    
    # F value
    F = (tc_max + np.log(exp_tc.sum())) / t
    
    return grad, F, c, p


def smoothed_optimize(f0, time_limit, t_start=50.0, t_end=5000.0, 
                      lr=0.01, momentum=0.9, verbose=False):
    """
    Smoothed projected gradient with Nesterov momentum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)  # momentum
    y = f.copy()     # Nesterov lookahead
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    cur_t = t_start
    cur_lr = lr
    cur_mom = momentum
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    # Schedule parameters
    total_iters_expected = max(1, int(time_limit / 0.05))  # rough estimate
    t_growth = (t_end / t_start) ** (1.0 / max(1, total_iters_expected))
    lr_decay = 0.9999
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        f_new = project_simplex(y - cur_mom * v)
        
        # Compute gradient at f_new
        grad, F_val, c_new, p = compute_lse_gradient(f_new, cur_t)
        
        # Update velocity
        v_new = cur_mom * v + grad
        v = v_new
        
        # Update y
        y = f_new - cur_mom * v_new
        
        # Project
        f_trial = project_simplex(y)
        
        # Evaluate actual objective
        c_trial = fast_convolve(f_trial)
        obj = c_trial.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_trial.copy()
            no_improve = 0
            f = f_trial
            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smoothed iter {it}: obj={best_obj:.10f}, t={cur_t:.1f}, time={elapsed:.0f}s")
        else:
            no_improve += 1
            f = f_trial
        
        # Schedule
        cur_t = min(cur_t * t_growth, t_end)
        cur_lr *= lr_decay
        
        if no_improve > 200:
            # Restart
            rng = np.random.default_rng(it * 31 + 17)
            scale = 0.001 * min(1.0, no_improve / 200.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            y = f.copy()
            cur_t = max(t_start, cur_t * 0.5)
            no_improve = 0
    
    return best_f, best_obj


def lp_refine(f0, time_limit, verbose=False):
    """
    LP-based Frank-Wolfe: at each step, solve min t s.t. (f*g)[k] <= t, 
    g >= 0, sum(g) = 1. Then line search between f and g.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1
        # Variables: g[0..n-1], t
        # c = [0, ..., 0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        n_cons = 2 * n - 1
        A_ub = np.zeros((n_cons, n + 1))
        for j in range(n):
            if f[j] != 0:
                A_ub[j:j+n, j] = f
        A_ub[:, -1] = -1.0
        b_ub = np.zeros(n_cons)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0, None)] * n + [(None, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        
        if not res.success:
            g = np.ones(n) / n
        else:
            g = np.maximum(res.x[:n], 0)
            gs = g.sum()
            if gs < 1e-15:
                g = np.ones(n) / n
            else:
                g = g / gs
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Binary search refinement
            lo = max(0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            f = f_new
            M = best_val
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP iter {it}: obj={best_obj:.10f}, time={elapsed:.0f}s")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 80:
            rng = np.random.default_rng(it * 1009 + 7)
            scale = 0.001 + 0.002 * min(no_improve // 50, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate in the active set, optimize its value.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        improved = False
        
        # Find active peaks
        k_actual = min(5, len(conv))
        peak_positions = np.argpartition(conv, -k_actual)[-k_actual:]
        
        # Collect coordinates that contribute to peaks
        coord_set = set()
        for pk in peak_positions:
            pk = int(pk)
            for j in range(max(0, pk - n + 1), min(n, pk + 1)):
                coord_set.add(j)
        
        # Sort by importance (contribution to peak)
        coords = sorted(coord_set, key=lambda j: -f[j])
        
        for j in coords[:min(50, len(coords))]:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            # Try a range of multipliers
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial[:j] *= scale
                f_trial[j+1:] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                f[:j] *= scale
                f[j+1:] *= scale
                f = f / f.sum()
                
                conv = fast_convolve(f)
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
