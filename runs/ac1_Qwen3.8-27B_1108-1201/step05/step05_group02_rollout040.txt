def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def raised_cosine(n, alpha=0.5):
    """Raised cosine window."""
    t = np.arange(n) / (n - 1)
    w = np.ones(n)
    m = int(alpha * (n - 1) / 2)
    # Apply raised cosine at edges
    for i in range(min(m, n)):
        w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        w[n-1-i] = w[i]
    w = np.maximum(w, 0)
    w /= w.sum()
    return w

def tukey_window(n, r=0.5):
    """Tukey (trapezoidal) window."""
    w = np.ones(n)
    m = int(r * n / 2)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

def solve_lp_min_peak(f, n):
    """LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    return None

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        
        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                
                old_val = f[j]
                if old_val < 1e-13:
                    continue
                
                best_j_val = old_val
                best_j_obj = best_obj
                multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
                
                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-16:
                        best_j_obj = val
                        best_j_val = new_val
                
                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                    
                    if best_j_obj < best_obj - 1e-15:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)
                if not improved and passes > 5:
                    # Try random coordinates
                    rng = np.random.default_rng(passes * 31 + j)
                    rj = int(rng.integers(0, n))
                    old_val = f[rj]
                    if old_val > 1e-10:
                        for mult in [0.5, 1.5]:
                            new_val = old_val * mult
                            if 0 < new_val < 0.5:
                                denom = 1.0 - old_val
                                if denom > 1e-10:
                                    scale = (1.0 - new_val) / denom
                                    f_trial = f.copy()
                                    f_trial[rj] = new_val
                                    mask = np.ones(n, dtype=bool)
                                    mask[rj] = False
                                    f_trial[mask] *= scale
                                    f_trial = f_trial / f_trial.sum()
                                    val = fast_convolve(f_trial).max()
                                    if val < best_obj - 1e-15:
                                        f = f_trial
                                        best_obj = val
                                        best_f = f.copy()
                                        conv = fast_convolve(f)
                                        improved = True
                                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj

Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a focused multi-phase optimization:
Phase 1: LP-based Frank-Wolfe (primary optimizer)
Phase 2: Subgradient refinement
Phase 3: Multi-length exploration
Phase 4: Coordinate descent polish + final perturbations

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_max_sum(f, M, n):
    """LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        if use_min_peak:
            g, lp_t = solve_lp_min_peak(f, n)
            if g is None:
                use_min_peak = False
                continue
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n
        else:
            M_cur = fast_convolve(f).max()
            g = solve_lp_max_sum(f, M_cur * 1.03, n)
            if g is None:
                g = solve_lp_max_sum(f, M_cur * 1.1, n)
            if g is None:
                use_min_peak = True
                continue
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n
            if it % 7 == 0:
                use_min_peak = True

        g = np.maximum(g, 0)
        gs = g.sum()
        g = g / gs if gs > 1e-15 else np.ones(n) / n

        if np.max(np.abs(g - f)) < 1e-14:
            no_improve += 1
            if no_improve > max_no_improve:
                break
            continue

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M

        coarse_alphas = [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.25, 0.3,
                         0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
        for alpha in coarse_alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(14):
                if hi - lo < 1e-6:
                    break
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                ft = ft / st if st > 1e-15 else np.ones(n) / n
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            f = f_new / sn if sn > 1e-15 else np.ones(n) / n
            M = best_val

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.12f} ({time.monotonic()-t0:.0f}s)")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            rng = np.random.default_rng(it * 6151 + 17)
            r = rng.random()
            if r < 0.35:
                noise = rng.standard_normal(n) * 0.002
                f = project_simplex(best_f + noise)
            elif r < 0.65:
                noise = rng.standard_normal(n) * 0.004
                noise = np.convolve(noise, np.ones(7) / 7, mode='same')
                f = project_simplex(best_f + noise)
            else:
                tilt = np.linspace(-1, 1, n) * 0.003
                f = project_simplex(best_f + tilt + rng.standard_normal(n) * 0.001)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            use_min_peak = True

    return best_f, best_obj


def sg_optimize(f0, time_limit, lr=0.002, momentum=0.94, top_k=50, temp=0.00004,
                verbose=False, max_no_improve=400):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s if s > 1e-15 else np.ones(len(f0)) / len(f0)
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-16:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 3000 == 0:
                print(f"    SG iter {it}: obj={best_obj:.12f}")
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.99993
        cur_temp *= 1.005

        if no_improve > max_no_improve:
            rng = np.random.default_rng(abs(hash(int(time.monotonic() * 9973))) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.4, 0.0003)
            no_improve = 0

    return best_f, best_obj


def smooth_penalty_optimize(f0, time_limit, rho_init=50.0, lr_init=0.005,
                            momentum=0.9, verbose=False):
    """Smooth penalty method: minimize max_k (a*a)_k + rho * sum[(a*a)_k - t]+^2."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s if s > 1e-15 else np.ones(len(f0)) / len(f0)
    n = len(f)

    conv = fast_convolve(f)
    t = conv.max()
    best_obj = t
    best_f = f.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    rho = rho_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        excess = np.maximum(conv - t, 0.0)
        n_excess = np.count_nonzero(excess > 1e-15)
        if n_excess == 0:
            t = conv.max() * 0.999
            conv = fast_convolve(f)
            excess = np.maximum(conv - t, 0.0)

        # Gradient of penalty w.r.t. a_j: sum_k 2*rho*excess[k]*2*a_{k-j}
        grad = np.zeros(n)
        excess_indices = np.where(excess > 1e-12)[0]
        for k in excess_indices:
            ek = excess[int(k)]
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            idx = np.arange(js, je + 1)
            kg = int(k) - idx
            mask = (kg >= 0) & (kg < n)
            grad[idx[mask]] += 2.0 * rho * ek * 2.0 * f[kg[mask]]

        # Add gradient of max term (subgradient at the argmax)
        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        kg = peak_k - idx
        mask = (kg >= 0) & (kg < n)
        grad[idx[mask]] += 2.0 * f[kg[mask]]

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    SmoothPen iter {it}: obj={best_obj:.12f} ({time.monotonic()-t0:.0f}s)")
        else:
            no_improve += 1
            f = f_new

        # Gradual parameter updates
        rho *= 1.02
        lr *= 0.9998

        if no_improve > 300:
            rng = np.random.default_rng(abs(hash(int(time.monotonic() * 7919))) % 791901)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            rho = rho_init * 0.5
            lr = lr_init
            no_improve = 0

    return best_f, best_obj


def coord_descent(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s if s > 1e-15 else np.ones(len(f0)) / len(f0)
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        top_k = min(4, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-13:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj
                multipliers = [0.4, 0.6, 0.75, 0.9, 1.1, 1.3, 1.6, 2.0]

                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-16:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                    if best_j_obj < best_obj - 1e-15:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            rng = np.random.default_rng(passes * 104729)
            for _ in range(min(20, n // 15)):
                j = int(rng.integers(0, n))
                old_val = f[j]
                if old_val < 1e-10:
                    continue
                for mult in [0.5, 0.7, 1.4, 1.7]:
                    new_val = old_val * mult
                    if 0 < new_val < 0.5:
                        denom = 1.0 - old_val
                        if denom > 1e-10:
                            scale = (1.0 - new_val) / denom
                            f_trial = f.copy()
                            f_trial[j] = new_val
                            mask = np.ones(n, dtype=bool)
                            mask[j] = False
                            f_trial[mask] *= scale
                            f_trial = f_trial / f_trial.sum()
                            val = fast_convolve(f_trial).max()
                            if val < best_obj - 1e-15:
                                f = f_trial
                                best_obj = val
                                best_f = f.copy()
                                conv = fast_convolve(f)
                                improved = True
                                break
                if improved:
                    break
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-FW, smooth penalty, subgradient,
    coordinate descent, multi-length exploration, and final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)
    print(f"Starting: n={base_n}, score={best_score:.8f}")

    # === Phase 1: LP-FW (primary, most time) ===
    print(f"\n=== Phase 1: LP-FW min-peak (budget: best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(550, rem * 0.35)
    if t1 > 40:
        print(f"  n={len(best_seq)}, t={t1:.0f}s")
        a_opt, _ = lpfw_optimize(best_seq, time_limit=t1, verbose=True, max_no_improve=40)
        try_seq(a_opt, "lpfw1")

    # === Phase 2: Smooth penalty optimization ===
    print(f"\n=== Phase 2: Smooth Penalty (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(rem * 0.18, 250)
        if t2 > 30:
            print(f"  n={len(best_seq)}, t={t2:.0f}s")
            a_opt, _ = smooth_penalty_optimize(best_seq, time_limit=t2,
                                                rho_init=30.0, lr_init=0.004,
                                                momentum=0.9, verbose=True)
            try_seq(a_opt, "smooth_pen")

            rem = deadline - time.monotonic()
            t2b = min(rem * 0.1, 120)
            if t2b > 20:
                a_opt, _ = smooth_penalty_optimize(best_seq, time_limit=t2b,
                                                    rho_init=80.0, lr_init=0.002,
                                                    momentum=0.92, verbose=True)
                try_seq(a_opt, "smooth_pen2")

    # === Phase 3: LP-FW again on updated best ===
    print(f"\n=== Phase 3: LP-FW round 2 (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(rem * 0.22, 350)
        if t3 > 30:
            print(f"  n={len(best_seq)}, t={t3:.0f}s")
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t3, verbose=True, max_no_improve=35)
            try_seq(a_opt, "lpfw2")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t4 = min(rem * 0.18, 200)
        if t4 > 25:
            print(f"  n={n_best}, t={t4:.0f}s")
            a_opt, _ = sg_optimize(best_seq, time_limit=t4,
                                   lr=0.0015, momentum=0.94,
                                   top_k=max(40, n_best // 6),
                                   temp=0.00003, verbose=True, max_no_improve=350)
            try_seq(a_opt, "sg1")

            rem = deadline - time.monotonic()
            t4b = min(rem * 0.08, 80)
            if t4b > 15:
                a_opt, _ = sg_optimize(best_seq, time_limit=t4b,
                                       lr=0.0008, momentum=0.95,
                                       top_k=max(25, n_best // 10),
                                       temp=0.000015, max_no_improve=250)
                try_seq(a_opt, "sg2")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        candidate_lengths = set()
        for delta in [-96, -48, 48, 96, 192, 320]:
            ln = len(base) + delta
            if 384 <= ln <= 2048:
                candidate_lengths.add(ln)
        candidate_lengths = sorted(set((ln // 32) * 32 for ln in candidate_lengths))

        for target_n in candidate_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_alloc = min(rem * 0.15, 120)
            if t_alloc < 15:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lpfw_optimize(f_interp, time_limit=t_alloc,
                                      use_min_peak=True if hasattr(lpfw_optimize, 'use_min_peak') else True,
                                      max_no_improve=25)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t6 = min(rem * 0.2, 150)
        if t6 > 15:
            print(f"  n={len(best_seq)}, t={t6:.0f}s")
            a_opt, _ = coord_descent(best_seq, time_limit=t6)
            try_seq(a_opt, "cd")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick LP-FW
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = lpfw_optimize(best_seq, time_limit=min(rem, 80), max_no_improve=20)
            try_seq(a_opt, "polish_lpfw")

        # Smooth penalty quick
        rem = deadline - time.monotonic() - 15
        if rem > 12:
            a_opt, _ = smooth_penalty_optimize(best_seq, time_limit=min(rem, 50),
                                                rho_init=100.0, lr_init=0.001,
                                                momentum=0.92)
            try_seq(a_opt, "polish_sp")

        # SG quick
        rem = deadline - time.monotonic() - 12
        if rem > 10:
            a_opt, _ = sg_optimize(best_seq, time_limit=min(rem, 40),
                                   lr=0.0005, momentum=0.95,
                                   top_k=max(15, n_best // 12),
                                   temp=0.00001, max_no_improve=150)
            try_seq(a_opt, "polish_sg")

        # Perturbations
        for p in range(6):
            rem = deadline - time.monotonic() - 8
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.0004 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lpfw_optimize(f_pert, time_limit=min(rem, 25), max_no_improve=12)
            try_seq(a_opt, f"pert{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
