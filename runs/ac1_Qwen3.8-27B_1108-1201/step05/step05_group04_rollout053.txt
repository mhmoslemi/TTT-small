def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with exact 1D subproblems (primary method)
2. LSE smoothed gradient descent for complementary refinement
3. Multi-start with adaptive random perturbations
4. Adaptive length search and interpolation
The coordinate descent solves each 1D subproblem exactly via golden section
search, making it far more efficient per-iteration than LP-based methods.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_1d_subproblem(f, i, conv):
    """
    Solve the 1D subproblem for coordinate i given current f (sum=1) and its convolution.
    Returns optimal t >= 0.
    """
    n = len(f)
    S = 1.0 - f[i]
    if S < 1e-15:
        return 0.0  # All mass is at i, can't do better

    # Compute C_l = conv[l] - contribution of f[i] to lag l
    # Affected lags: l in [i, i+n-1] intersect [0, 2n-2]
    l_start = max(0, i)
    l_end = min(2 * n - 2, i + n - 1)

    # M_other: max of unaffected lags
    # We'll compute it as max(conv) excluding the affected range
    if l_start == 0 and l_end == 2 * n - 2:
        M_other = 0.0
    else:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[l_start:l_end + 1] = False
        if mask.any():
            M_other = conv[mask].max()
        else:
            M_other = 0.0

    vi = f[i]

    def compute_contribution(l, v):
        """Contribution of value v at coordinate i to convolution lag l."""
        if l < i or l > i + n - 1:
            return 0.0
        if l == 2 * i:
            return v * v
        idx = l - i
        if 0 <= idx < n:
            return 2.0 * v * f[idx]
        return 0.0

    # Build the affected lags and their base values C_l
    affected_lags = np.arange(l_start, l_end + 1)
    C_vals = np.empty(len(affected_lags))
    g_coeffs = np.empty(len(affected_lags))  # linear coefficient (for 2*v*f[l-i])
    g_quad = 0.0  # coefficient of v^2 (1 if this is the 2i lag)
    quad_lag_idx = -1

    for idx, l in enumerate(affected_lags):
        if l == 2 * i:
            C_vals[idx] = conv[l] - vi * vi
            g_coeffs[idx] = 0.0
            g_quad = 1.0
            quad_lag_idx = idx
        else:
            idx2 = l - i
            if 0 <= idx2 < n:
                C_vals[idx] = conv[l] - 2.0 * vi * f[idx2]
                g_coeffs[idx] = 2.0 * f[idx2]
            else:
                C_vals[idx] = conv[l]
                g_coeffs[idx] = 0.0

    def F(t):
        """Compute F(t) = max_l value_l(t) / (S+t)^2."""
        if t < 0:
            t = 0.0
        denom = (S + t)
        if denom < 1e-15:
            return 1e10
        # h(t) = max over affected lags
        if g_quad > 0 and quad_lag_idx >= 0:
            h_quad = C_vals[quad_lag_idx] + t * t
        else:
            h_quad = -np.inf
        # Linear terms
        h_lin = C_vals + g_coeffs * t
        h = np.maximum(h_lin, h_quad)
        h_max = h.max()
        val = max(M_other, h_max) / (denom * denom)
        return val

    # Golden section search
    # Determine bracket
    t_current = vi
    t_left = 0.0
    t_right = max(2.0 * vi, S * 0.5)
    if t_right < 1e-10:
        t_right = 1.0

    # Expand bracket if needed
    F_left = F(t_left)
    F_right = F(t_right)
    F_mid = F(t_current)

    # Make sure the bracket contains the minimum
    # F is roughly unimodal: decreases then increases (or just decreases)
    best_t = t_current
    best_F = F_mid

    # Try a few sample points
    samples = np.array([0.0, vi * 0.5, vi, vi * 1.5, vi * 2.0, S * 0.25, S * 0.5, S])
    samples = samples[(samples >= 0) & (samples <= 10.0)]
    if len(samples) == 0:
        return vi

    for s in samples:
        Fs = F(s)
        if Fs < best_F:
            best_F = Fs
            best_t = s

    # Golden section search around best_t
    # Determine search range
    search_lo = max(0.0, best_t - 0.5 * vi - 0.1 * S)
    search_hi = best_t + 0.5 * vi + 0.1 * S
    search_hi = min(search_hi, 10.0)
    if search_hi <= search_lo:
        return best_t

    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g = search_lo
    b_g = search_hi
    c_g = b_g - gr * (b_g - a_g)
    d_g = a_g + gr * (b_g - a_g)
    fc = F(c_g)
    fd = F(d_g)

    for _ in range(40):
        if b_g - a_g < 1e-12 * max(1.0, abs(best_t)):
            break
        if fc < fd:
            b_g = d_g
            d_g = c_g
            fd = fc
            c_g = b_g - gr * (b_g - a_g)
            fc = F(c_g)
        else:
            a_g = c_g
            c_g = d_g
            fc = fd
            d_g = a_g + gr * (b_g - a_g)
            fd = F(d_g)

    t_opt = (a_g + b_g) / 2.0
    F_opt = F(t_opt)
    if F_opt > best_F:
        t_opt = best_t
    return max(0.0, t_opt)


def coord_descent(f_init, time_limit, rng, verbose=False):
    """
    Coordinate descent with exact 1D subproblems and renormalization.
    """
    f = np.maximum(f_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)

        for i in order:
            t_opt = compute_1d_subproblem(f, i, conv)
            if t_opt < 0:
                t_opt = 0.0
            S = 1.0 - f[i]
            f[i] = t_opt
            denom = S + t_opt
            if denom < 1e-15:
                f = np.ones(n) / n
                conv = fast_convolve(f)
            else:
                f = f / denom
                conv = conv / (denom * denom)

        # Periodic full recompute for numerical stability
        if sweep % 10 == 0:
            conv = fast_convolve(f)
            s = f.sum()
            if s < 1e-15:
                f = np.ones(n) / n
                conv = fast_convolve(f)
            else:
                f = f / s
                conv = fast_convolve(f)

        obj = conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Restart with perturbation
                scale = 0.001 + 0.003 * min(1.0, (no_improve - 50) / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                obj = conv.max()
                if obj < best_obj:
                    best_obj = obj
                    best_f = f.copy()
                no_improve = 0

        if verbose and sweep % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep {sweep}, obj={best_obj:.12f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def compute_grad_weighted(f, w, max_peaks=200):
    """Compute gradient of sum_k w[k] * (f*f)[k] w.r.t. f."""
    n = len(f)
    conv_len = 2 * n - 1

    if len(w) <= max_peaks:
        sig_idx = np.argsort(w)[::-1]
    else:
        sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
        sig_idx = sig_idx[w[sig_idx] > 1e-15 * w.max()]

    grad = np.zeros(n)
    for k in sig_idx:
        wk = w[k]
        if wk < 1e-15:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += wk * f[k - ms]

        ms2_lo = max(0, k - hi)
        ms2_hi = min(n - 1, k - lo)
        if ms2_lo <= ms2_hi:
            ms2 = np.arange(ms2_lo, ms2_hi + 1)
            grad[ms2] += wk * f[k - ms2]

    return grad


def lse_optimize(f_init, time_limit, beta_start=20.0, beta_end=5000.0, max_peaks=200):
    """LSE smoothed gradient descent on simplex."""
    f = np.maximum(f_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 1.5 / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_grad_weighted(f, w, max_peaks=max_peaks)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 200
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 4.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.92
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 80:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(1.5 / n, 0.001)
                beta = min(beta * 1.5, beta_end)
                momentum = 0.88
                no_improve = 0

        if beta < beta_end:
            beta *= 1.003

    return best_f, best_obj


def directed_peak_opt(f_init, time_limit, n_peaks=12, lr_init=0.002, momentum=0.93):
    """Directly target top peaks with weighted subgradient."""
    f = np.maximum(f_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    n = len(f)

    no_improve = 0
    cur_lr = lr_init
    v = np.zeros(n)
    cur_mom = momentum

        conv = fast_convolve(f)

        k_top = min(n_peaks, conv_len)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = pv - pv.min()
        threshold = scaled.max() * 0.3
        w = np.where(scaled > threshold, np.exp(scaled - scaled.max()), 0.0)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum

        w_full = np.zeros(conv_len)
        w_full[peaks] = w

        grad = compute_grad_weighted(f, w_full, max_peaks=k_top)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 200
            break
        grad /= grad_norm

        v = cur_mom * v - cur_lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:

        else:
            no_improve += 1

                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.0002)
                no_improve = 0

    return best_f, best_obj


def try_sequence(a, best_tracker):

def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
