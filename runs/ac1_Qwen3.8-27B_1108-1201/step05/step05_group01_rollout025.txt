def fast_convolve(a):

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_gradient(conv, tau, f):
    """Compute gradient of log-sum-exp approximation of max(conv).
    
    Returns (gradient, smoothed_value, weights)
    """
    # For numerical stability
    shifted = conv / tau
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    smooth_val = tau * (shifted.max() + np.log(exp_vals.sum()))
    
    # Gradient: sum_k w_k * d(conv[k])/d(f)
    # d(conv[k])/d(f[m]) = 2*f[k-m] if 0 <= k-m < n, else 0
    n = len(f)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-15:
            continue
        # k-m in [0, n-1] => m in [k-(n-1), k] intersect [0, n-1]
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            idx = np.arange(m_lo, m_hi + 1)
            grad[idx] += 2.0 * weights[k] * f[k - idx]
    
    return grad, smooth_val, weights


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lpfw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = lp_direction(f, M * m_relax, n)
        if g is None:
            m_relax = min(m_relax * 1.2, 10.0)
            if m_relax >= 10.0:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            m_relax = min(m_relax * 1.05, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        # FW step size
        fw_a = 2.0 / (it + 2.0)
        candidates = sorted(set([0.05, 0.1, 0.15, fw_a, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]))
        
        for alpha in candidates:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha
        
        # Binary refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.08, 5.0)
        
        if no_improve > 80:
            # Restart with perturbation
            rng = np.random.default_rng(hash((it, n)) % 2**31)
            noise = rng.standard_normal(n) * 0.005
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            no_improve = 0
    
    return best_f, best_obj


def gradient_optimize(f0, time_limit, tau0=0.001, tau_final=0.00005, lr0=0.01):
    """Log-sum-exp smoothed gradient with Nesterov acceleration and backtracking."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # Nesterov state
    y = f.copy()
    v = np.zeros(n)
    mom = 0.85
    lr = lr0
    tau = tau0
    
    total_iter_budget = int(time_limit / 0.05)  # rough estimate
    decay = (tau_final / tau0) ** (1.0 / max(total_iter_budget, 1))
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        # Adaptive temperature
        tau = max(tau_final, tau0 * (tau_final / tau0) ** (elapsed / time_limit))
        
        # Compute smoothed gradient at y (Nesterov lookahead)
        conv_y = fast_convolve(y)
        grad, sv, w = smooth_max_gradient(conv_y, tau, y)
        
        # Nesterov momentum update
        v = mom * v - lr * grad
        y_new = project_simplex(y + v)
        
        # Backtracking line search
        f_new = y_new
        f_val = fast_convolve(f_new).max()
        
        # Try a few step sizes
        best_f_candidate = f_new
        best_val = f_val
        
        for step in [1.0, 0.5, 0.25, 0.1]:
            ft = (1 - step) * f + step * (y_new - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st > 1e-15:
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_f_candidate = ft
        
        f = best_f_candidate
        y = f + mom * (f - y_new)  # Update Nesterov lookahead (simplified)
        y = project_simplex(y)
        
        # Update best
        f_val = fast_convolve(f).max()
        if f_val < best_obj - 1e-14:
            best_obj = f_val
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.9
            mom = min(0.95, mom * 1.001)
            no_improve = 0
        
        if no_improve > 1000:
            # Diversify
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            v = np.zeros(n)
            lr = lr0 * 0.5
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, max_passes=50):
    """Cyclic coordinate descent with exact 1D minimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    for _pass in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old = f[j]
            if old < 1e-15:
                continue
            
            # Try scaling f[j] by various factors
            # When we change f[j] to x, others scale by (1-x)/(1-old)
            denom = 1.0 - old
            if denom < 1e-12:
                continue
            
            best_x = old
            best_v = fast_convolve(f).max()
            
            # Try a range of multipliers
            for mult in [0.3, 0.5, 0.7, 0.8, 1.2, 1.3, 1.5, 2.0]:
                x_try = old * mult
                if x_try < 0 or x_try > 1 - 1e-6:
                    continue
                ft = f.copy()
                ft[j] = x_try
                scale = (1.0 - x_try) / denom
                mask = np.arange(n) != j
                ft[mask] *= scale
                ft = np.maximum(ft, 0.0)
                ftsum = ft.sum()
                if ftsum < 1e-15:
                    continue
                ft = ft / ftsum
                val = fast_convolve(ft).max()
                if val < best_v - 1e-15:
                    best_v = val
                    best_x = x_try
                    best_ft = ft
            
            if best_x != old:
                f = best_ft
                pass_improved = True
                if fast_convolve(f).max() < best_obj - 1e-14:
                    best_obj = fast_convolve(f).max()
                    best_f = f.copy()
        
        if not pass_improved:
            break
    
    return best_f, best_obj


def interpolate_to(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0.0)
    s = fi.sum()
    if s > 1e-15:
        fi /= s
    return fi
