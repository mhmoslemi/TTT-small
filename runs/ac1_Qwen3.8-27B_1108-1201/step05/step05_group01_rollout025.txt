   Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed
   gradient descent with continuation in smoothing parameter, multi-length
   exploration, and periodic polynomial refinement.
   """
   
   import numpy as np
   import time
   
   def fast_convolve(a):
       n = len(a)
       size = 2 * n - 1
       fft_size = 1 << (size - 1).bit_length()
       A = np.fft.rfft(a, fft_size)
       return np.fft.irfft(A * A, fft_size)[:size]
   
   def project_simplex(v):
       n = len(v)
       if n == 1:
           return np.array([max(v[0], 0.0)])
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u)
       cond = u * np.arange(1, n + 1) > (cssv - 1)
       if not cond.any():
           return np.ones(n) / n
       rho = np.nonzero(cond)[0][-1]
       theta = (cssv[rho] - 1.0) / (rho + 1.0)
       return np.maximum(v - theta, 0.0)
   
   def compute_lse_gradient(a, lam):
       """Compute gradient of (1/lam)*log(sum exp(lam*conv[k])) at a."""
       n = len(a)
       conv = fast_convolve(a)
       shifted = lam * (conv - conv.max())
       exp_vals = np.exp(shifted)
       p = exp_vals / exp_vals.sum()
       # grad[m] = 2 * sum_{j=0}^{n-1} p[m+j] * a[j]
       # = 2 * conv_full(a[::-1], p)[n-1 : 2n-1]
       a_rev = a[::-1]
       fft_size = 1 << (3 * n - 3).bit_length()
       A = np.fft.rfft(a_rev, fft_size)
       P = np.fft.rfft(p, fft_size)
       c = np.fft.irfft(A * P, fft_size)
       g = c[n-1 : 2*n-1]
       return 2.0 * g, conv
   
   def lse_optimize(f0, time_limit, lam_init=10.0, lam_max=500.0, verbose=False):

       else:

       y = f.copy()
       t_k = 1.0
       lam = lam_init

       lr = 0.1 / (lam * best_obj + 0.1)
       
       t0 = time.monotonic()
       it = 0
       no_improve = 0
       lam_counter = 0

           # FISTA step
           y_new = project_simplex(y - lr * grad)

           f_new = project_simplex(f_new)
