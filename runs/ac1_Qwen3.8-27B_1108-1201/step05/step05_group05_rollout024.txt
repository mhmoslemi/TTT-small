def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using peak-directed pair coordinate descent,
LP-Frank-Wolfe with adaptive constraints, mirror descent, multi-length exploration,
and structured restarts. Key improvements: pair-directed moves that jointly optimize
two coordinates contributing to the peak lag, incremental convolution updates for

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_after_swap(a, i, j, d, conv):
    """Update convolution after changing a[i]+=d, a[j]-=d. Returns new conv array."""
    n = len(a)
    new_conv = conv.copy()
    # For each lag k, the change is:
    # d * (a[k-i] if 0<=k-i<n else 0) + d * (a[k-i] if k-i==i special) 
    # - d * (a[k-j] if 0<=k-j<n else 0)
    # Plus cross terms when k-i==j or k-j==i
    # Simple approach: just recompute affected lags
    # Lags affected: those k where k-i in [0,n) or k-j in [0,n) or (k-i)==j or (k-j)==i
    # That's lags in range [0, i+n-1] union [0, j+n-1] = [0, max(i,j)+n-1]
    # Actually all lags from 0 to 2n-2 could be affected in worst case.
    # For simplicity with small n, just recompute fully.
    A = np.fft.rfft(a, 1 << (2 * n - 2).bit_length())
    new_conv = np.fft.irfft(A * A, 1 << (2 * n - 2).bit_length())[:2 * n - 1]
    return new_conv


def pair_cd_optimize(f, time_limit, top_k_lags=10, top_pairs=20, verbose=False):
    """
    Peak-directed pair coordinate descent.
    For each peak lag, find contributing pairs and do line search on mass transfer.
    """
    f = np.maximum(f.astype(np.float64), 0.0).copy()
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        peak_val = conv.max()
        peak_k = int(np.argmax(conv))

        # Get top contributing pairs for the peak lag
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            break
        paired_js = np.arange(js, je + 1)
        contributions = f[paired_js] * f[peak_k - paired_js]

        # Sort by contribution
        order = np.argsort(contributions)[::-1][:top_pairs]

        improved_this_iter = False
        for idx in order:
            j = int(paired_js[idx])
            k_other = peak_k - j
            if j == k_other:
                continue
            a_j = f[j]
            a_k = f[k_other]
            if a_j < 1e-12 or a_k < 1e-12:
                continue

            # Try transferring mass from the larger of (j, k_other) to other positions
            # We want to reduce f[j]*f[k_other]
            # Strategy: reduce the larger one, put mass at a position with low autocorrelation impact
            if a_j > a_k:
                src = j
                dst_candidates = np.argsort(f)[:min(10, n // 4)]
                src_val = a_j
            else:
                src = k_other
                dst_candidates = np.argsort(f)[:min(10, n // 4)]
                src_val = a_k

            best_move_obj = best_obj
            best_move = None

            for frac in [0.1, 0.25, 0.4, 0.6, 0.8]:
                d = src_val * frac
                if d < 1e-12:
                    continue
                for m in dst_candidates:
                    if m == src or m == (peak_k - src):
                        continue
                    f_trial = f.copy()
                    f_trial[src] -= d
                    f_trial[m] += d
                    if f_trial[src] < 0:
                        continue
                    f_trial = np.maximum(f_trial, 0)
                    f_trial = f_trial / f_trial.sum()
                    conv_trial = fast_convolve(f_trial)
                    obj_trial = conv_trial.max()
                    if obj_trial < best_move_obj - 1e-14:
                        best_move_obj = obj_trial
                        best_move = (src, m, d)

            if best_move is not None:
                src, m, d = best_move
                f[src] -= d
                f[m] += d
                f = np.maximum(f, 0)
                f = f / f.sum()
                conv = fast_convolve(f)
                if conv.max() < best_obj - 1e-14:
                    best_obj = conv.max()
                    best_f = f.copy()
                    improved_this_iter = True

        if improved_this_iter:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

        if verbose and it % 200 == 0:
            print(f"    PairCD iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def single_cd_optimize(f, time_limit, multipliers=None, verbose=False):
    """
    Single-coordinate descent with continuous line search per coordinate.
    More effective than fixed multipliers.
    """
    if multipliers is None:
        multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0, 3.0]

    f = np.maximum(f.astype(np.float64), 0.0).copy()
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-13:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                remaining = 1.0 - new_val
                if remaining <= 1e-12:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-12:
                    continue
                f_trial = f_trial / ss

                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                best_obj = best_j_obj
                best_f = f.copy()
                conv = fast_convolve(f)
                improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

        if verbose:
            print(f"    SingleCD pass done: obj={best_obj:.10f}")

    return best_f, best_obj


def lp_fw_direction(f, M, n, active_k=None):
    """Solve LP for Frank-Wolfe direction."""
    c = -np.ones(n)
    if active_k is None:
        n_constr = 2 * n - 1
        A_ub = np.zeros((n_constr, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(n_constr, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

        active_k = None

            active_k = k_sorted.tolist()
        if it % 40 == 0:
            active_k = None

        g = lp_fw_direction(f, M, n, active_k)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1

            continue

        # Simple fixed step or small line search
        for alpha in [0.3, 0.5, 0.7, 1.0, 0.15]:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0)

def mirror_descent(f0, time_limit, lr_init=0.001, lr_decay=0.9997, top_k=30,
                   temp_init=0.0001, temp_growth=1.003):
    """Mirror descent with entropy regularization and softmax-weighted subgradients."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    temp = temp_init
    mom = 0.9
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= lr_decay
        temp *= temp_growth

        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            lr = max(lr * 0.5, 0.0001)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def structured_init(n, mode=0, seed=0):
    """Various structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)

    if mode == 0:
        # Gaussian centered
        f = np.exp(-4.0 * ((x - n / 2.0) / (0.4 * n)) ** 2)
    elif mode == 1:
        # Raised cosine
        f = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        f[n // 2] = 1.0
    elif mode == 2:
        # Two bumps
        f = np.exp(-8.0 * ((x - n * 0.25) / (0.15 * n)) ** 2) + \
            np.exp(-8.0 * ((x - n * 0.75) / (0.15 * n)) ** 2)
    elif mode == 3:
        # Linear ramp down
        f = 1.0 - x / n
        f = f ** 1.5
    elif mode == 4:
        # Random with smoothing
        f = rng.random(n)
        kernel = np.ones(5) / 5
        f = np.convolve(f, kernel, mode='same')
    else:
        f = np.ones(n) / n

    f = np.maximum(f, 0)
    ss = f.sum()
    if ss < 1e-10:
        f = np.ones(n) / n
    else:
        f /= ss
    return f


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining pair-directed CD, LP-FW, mirror descent,
    and multi-length exploration to minimize the autocorrelation peak ratio.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-10:
            return float('inf')
        a_norm = a / ss
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}, t={time.monotonic():.0f})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    # === Phase 1: Quick LP-FW on parent ===
    print("=== Phase 1: LP-FW on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(200, rem * 0.2)
    if t1 > 30:
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, top_k_constraints=35)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Pair-directed CD ===
    print("\n=== Phase 2: Pair CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(250, rem * 0.2)
        if t2 > 30:
            a_opt, _ = pair_cd_optimize(best_seq, time_limit=t2, top_k_lags=5,
                                        top_pairs=30, verbose=True)
            try_seq(a_opt, "pair_cd")

    # === Phase 3: Single-coordinate CD with better multipliers ===
    print("\n=== Phase 3: Single CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(200, rem * 0.15)
        if t3 > 30:
            a_opt, _ = single_cd_optimize(best_seq, time_limit=t3,
                                          multipliers=[0.4, 0.6, 0.75, 0.85, 0.92,
                                                        1.08, 1.15, 1.3, 1.5, 2.0, 3.0])
            try_seq(a_opt, "single_cd")

    # === Phase 4: Mirror descent ===
    print("\n=== Phase 4: Mirror Descent ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t4 = min(150, rem * 0.12)
        if t4 > 20:
            a_opt, _ = mirror_descent(best_seq, time_limit=t4,
                                      lr_init=0.0008, lr_decay=0.9998,
                                      top_k=max(20, n_best // 8),
                                      temp_init=0.00005, temp_growth=1.004)
            try_seq(a_opt, "mirror")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-length exploration ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = [640, 768, 896, 1024]

    for tl in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        f_init = interpolate_sequence(base, tl)
        # Mix with flat spectrum for diversity
        fs = flat_spectrum_init(tl, seed=tl)
        f_mix = project_simplex(0.75 * f_init + 0.25 * fs)

        t_alloc = min(rem * 0.15, 150)
        if t_alloc < 20:
            continue
        print(f"  n={tl}, t={t_alloc:.0f}s")

        # LP-FW on this length
        a_opt, _ = lp_fw_optimize(f_mix, time_limit=t_alloc * 0.6, top_k_constraints=30)
        try_seq(a_opt, f"lpfw_n{tl}")

        # Quick CD on result
        rem = deadline - time.monotonic()
        if rem > 30:
            a_cd, _ = single_cd_optimize(a_opt, time_limit=min(rem * 0.1, 60),
                                         multipliers=[0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0])
            try_seq(a_cd, f"cd_n{tl}")

    # === Phase 6: Focused optimization on best so far ===
    print(f"\n=== Phase 6: Focused optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Aggressive pair CD
        rem = deadline - time.monotonic()
        if rem > 50:
            t6a = min(rem * 0.2, 200)
            print(f"  Pair CD t={t6a:.0f}s")
            a_opt, _ = pair_cd_optimize(best_seq, time_limit=t6a,
                                        top_k_lags=8, top_pairs=40)
            try_seq(a_opt, "pair_cd2")

        # LP-FW with full constraints periodically
        rem = deadline - time.monotonic()
        if rem > 40:
            t6b = min(rem * 0.2, 150)
            print(f"  LP-FW t={t6b:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t6b, top_k_constraints=25)
            try_seq(a_opt, "lpfw_focus")

        # Mirror descent with different params
        rem = deadline - time.monotonic()
        if rem > 30:
            t6c = min(rem * 0.15, 100)
            a_opt, _ = mirror_descent(best_seq, time_limit=t6c,
                                      lr_init=0.0005, lr_decay=0.9999,
                                      top_k=max(15, n_best // 10),
                                      temp_init=0.00002, temp_growth=1.005)
            try_seq(a_opt, "mirror_fine")

    # === Phase 7: Diverse restarts ===
    print(f"\n=== Phase 7: Diverse restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        restarts = [
            ("noise_small", best_seq + rng.standard_normal(n_best) * 0.002),
            ("noise_med", best_seq + rng.standard_normal(n_best) * 0.005),
            ("rev_blend", 0.7 * best_seq + 0.3 * best_seq[::-1]),
            ("flat_blend", 0.85 * best_seq + 0.15 * np.ones(n_best) / n_best),
            ("peak_trim", np.where(fast_convolve(best_seq) > best_score / (2 * n_best) * 1.01,
                                   best_seq * 0.9, best_seq)),
        ]

        for name, f_init in restarts:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            f_init = project_simplex(np.maximum(f_init, 0))
            t_alloc = min(rem * 0.15, 100)
            print(f"  Restart [{name}] t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc * 0.5, top_k_constraints=30)
            try_seq(a_opt, f"restart_{name}")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_cd, _ = single_cd_optimize(a_opt, time_limit=min(rem * 0.3, 40),
                                             multipliers=[0.6, 0.8, 0.9, 1.1, 1.3, 1.6])
                try_seq(a_cd, f"restart_{name}_cd")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t8 = min(rem * 0.3, 100)
            print(f"  Pair CD final t={t8:.0f}s")
            a_opt, _ = pair_cd_optimize(best_seq, time_limit=t8,
                                        top_k_lags=5, top_pairs=25)
            try_seq(a_opt, "final_pair")

        rem = deadline - time.monotonic()
        if rem > 15:
            t8b = min(rem * 0.3, 60)
            a_opt, _ = single_cd_optimize(best_seq, time_limit=t8b,
                                          multipliers=[0.85, 0.9, 0.95, 1.05,
                                                        1.1, 1.15, 1.25])
            try_seq(a_opt, "final_cd")

        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = mirror_descent(best_seq, time_limit=min(rem * 0.5, 40),
                                      lr_init=0.0002, lr_decay=0.9999,
                                      top_k=15, temp_init=0.00001, temp_growth=1.005)
            try_seq(a_opt, "final_md")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, t={time.monotonic():.0f})")
    return [float(x) for x in best_seq]
