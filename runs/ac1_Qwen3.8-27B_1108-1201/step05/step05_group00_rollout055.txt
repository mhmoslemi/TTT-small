def spectral_move(a, alpha=0.5):
    """
    Frequency-domain move: modify the spectrum to reduce auto-correlation peaks.
    Works by reducing spectral components that align with time-domain peaks.
    """
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)

    # Current auto-correlation
    conv = fast_convolve(a)
    # Find peaks
    n_peaks = max(5, n // 20)
    if n_peaks >= len(conv):
        peak_positions = np.arange(len(conv))
    else:
        peak_positions = np.argpartition(conv, -n_peaks)[-n_peaks:]

    # The contribution of freq m to conv[k] involves the phase alignment
    # Simple heuristic: reduce magnitude at frequencies where the spectral
    # "energy" correlates with peak positions
    freqs = np.arange(len(mag))
    # Compute correlation between |A|^2 profile and peak indicator

    for k in peak_positions:
        # Frequency bin k/(2n-1) * N corresponds to time-domain position k
        m_idx = min(int(k * N / (2 * n - 1)), len(mag) - 1)
        peak_indicator[m_idx] += 1.0

    peak_indicator /= (peak_indicator.max() + 1e-10)
    # Reduce magnitude at high-alignment frequencies

    new_A = new_mag * np.exp(1j * phase)

    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def fw_lp_oracle(f, M, n_constraints=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for key k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix (sparse: each row has at most n nonzeros)
    if n_constraints is None:
        ks = np.arange(2 * n - 1)
    else:
        # Only use the most important constraints
        conv_current = fast_convolve(f)
        ks = np.argpartition(conv_current, -n_constraints)[-n_constraints:]

    A_ub = np.zeros((len(ks), n))
    for i, k in enumerate(ks):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[i, js:je + 1] = f[js:je + 1][:je + 1 - min(0, k - n + 1)][::-1] if False else \
            np.array([f[k - j] for j in range(js, je + 1)])
    b_ub = np.full(len(ks), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    """
    Adaptive smoothed gradient descent with log-sum-exp objective.
    Includes spectral moves for diversity.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    rng = np.random.default_rng(seed)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    spectral_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        grad, conv, p = compute_smooth_grad(a, T)
        current_max = conv.max()

        # Projected gradient step with backtracking
        accepted = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < current_max - 1e-15:
                a = candidate
                current_max = val
                accepted = True
                break

        if not accepted:
            # Try spectral move
            if use_spectral and it % 50 == 0:
                a_spec = spectral_move(a, alpha=0.3 + 0.3 * min(1.0, no_improve / 200.0))
                c_spec = fast_convolve(a_spec)
                if c_spec.max() < current_max - 1e-15:
                    a = a_spec
                    current_max = c_spec.max()
                    spectral_count += 1
                    accepted = True

        if current_max < best_obj - 1e-14:
            best_obj = current_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            a = best_a.copy()  # Revert to best

        # Adaptive temperature and learning rate
        if no_improve < 50:
            T *= 0.995
        elif no_improve < 100:
            T *= 1.005
        else:
            T *= 1.02

        T = max(T, T_min)
        lr = max(0.001, lr * 0.999)

        # Restart mechanism
        if no_improve > 200:
            noise_scale = 0.002 + 0.005 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init * 0.5
            lr = lr_init * 0.5
            no_improve = 0
            spectral_count = 0

        if no_improve > 500:
            # Aggressive restart
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def multi_peak_descent(a0, time_limit, K=20, seed=42):
    """
    Multi-peak balanced descent: explicitly try to equalize the top K peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        # Get top K peaks
        k_eff = min(K, len(conv))
        peak_indices = np.argpartition(conv, -k_eff)[-k_eff:]
        peak_values = conv[peak_indices]

        # Target: mean of top peaks
        target = peak_values.mean()

        # Compute weighted gradient
        weights = np.maximum(peak_values - target, 0)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            # All peaks are balanced; try to reduce them all
            weights = np.ones(len(peak_indices))
            w_sum = 1.0
        else:
            weights /= w_sum

        grad = np.zeros(n)
        for i, k in enumerate(peak_indices):
            if weights[i] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            # grad_contribution[j] = 2 * a[k-j] for j in [js, je]
            for j in range(js, je + 1):
                grad[j] += 2.0 * weights[i] * a[k - j]

        # Projected step
        accepted = False
        for trial_lr in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            if c.max() < current_max - 1e-15:
                a = candidate
                current_max = c.max()
                accepted = True
                break

        if current_max < best_obj - 1e-14:
            best_obj = current_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if not accepted:
                a = best_a.copy()

        if no_improve > 100:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, seed=42):
    """Frank-Wolfe with LP oracle (relaxed version for speed)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    rng = np.random.default_rng(seed)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Solve LP with relaxed M
        g = fw_lp_oracle(f, M * 1.05, n_constraints=min(2 * n - 1, 200))
        if g is None or g.sum() < 1e-15:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj


def peak_balancing_polish(a0, time_limit, seed=42):
    """
    Final polish: repeatedly identify the top peak and try to reduce it
    by adjusting only the variables that contribute to it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        # Find variables contributing to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = list(range(js, je + 1))

        if len(affected) <= 1:
            break

        # Try to redistribute among affected variables to reduce the peak
        # The peak is sum_{j in affected} a[j] * a[k_peak - j]
        # To reduce it, we want to make the product smaller.
        # Strategy: equalize the affected variables (reduces max product)
        current_vals = a[affected]
        mean_val = current_vals.mean()
        # Move toward equalization
        for _ in range(3):
            rng_local = np.random.default_rng((it * 31 + k_peak) % (2**31))
            # Perturb toward equalization
            direction = np.zeros(n)
            for j in affected:
                direction[j] = mean_val - a[j]
            # Add small random perturbation
            direction[affected] += rng_local.standard_normal(len(affected)) * 0.001

            for scale in [0.1, 0.05, 0.01]:
                candidate = project_simplex(a + scale * direction)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    break

        # Also try random perturbation of affected vars
        rng_local = np.random.default_rng((it * 7919 + k_peak) % (2**31))
        pert = np.zeros(n)
        pert[affected] = rng_local.standard_normal(len(affected)) * 0.002
        candidate = project_simplex(a + pert)
        c = fast_convolve(candidate)
        if c.max() < best_obj - 1e-15:
            a = candidate
            best_obj = c.max()
            best_a = a.copy()

        if it % 500 == 0 and it > 500:
            # Check if stuck
            conv_check = fast_convolve(a)
            if conv_check.max() > best_obj * 1.001:
                a = best_a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining:
    - Smoothed gradient with adaptive temperature
    - Spectral domain moves
    - Multi-peak balanced descent
    - FW-LP for global moves
    - Peak balancing polish
    - n exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    # === Phase 1: Quick multi-n exploration ===
    print("\n=== Phase 1: Quick Multi-n Exploration ===")
    t_phase_start = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 128),
        n_parent + 128,
        max(128, n_parent - 256),
        n_parent + 256,
    ]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 30:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(remaining, 40)
        print(f"  Smooth grad n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = smoothed_gradient_optimize(start, time_limit=t_alloc,
                                               T_init=0.002, lr_init=0.1,
                                               use_spectral=True, seed=seed + n_try)
        try_seq(a_opt, f"smooth_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 2: Deep smoothed gradient on best n ===
    print("\n=== Phase 2: Deep Smoothed Gradient ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple temperature sweeps
        configs = [
            {'T_init': 0.005, 'lr_init': 0.08, 'label': 'T0.005'},
            {'T_init': 0.001, 'lr_init': 0.05, 'label': 'T0.001'},
            {'T_init': 0.0003, 'lr_init': 0.03, 'label': 'T0.0003'},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 600
            if remaining < 20:
                break
            t_alloc = min(remaining, 180)
            print(f"  Smooth grad {cfg['label']}, n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_gradient_optimize(best_seq, time_limit=t_alloc,
                                                  T_init=cfg['T_init'],
                                                  lr_init=cfg['lr_init'],
                                                  use_spectral=True,
                                                  seed=seed + 100 + ci)
            try_seq(a_opt, f"deep_{cfg['label']}")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 3: Multi-peak descent + FW-LP ===
    print("\n=== Phase 3: Multi-Peak + FW-LP ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multi-peak descent
        remaining = deadline - time.monotonic() - 400
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  Multi-peak K=30, t={t_alloc:.0f}s")
            a_opt, _ = multi_peak_descent(best_seq, time_limit=t_alloc, K=30,
                                          seed=seed + 200)
            try_seq(a_opt, "multi_peak_30")

        remaining = deadline - time.monotonic() - 300
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Multi-peak K=10, t={t_alloc:.0f}s")
            a_opt, _ = multi_peak_descent(best_seq, time_limit=t_alloc, K=10,
                                          seed=seed + 201)
            try_seq(a_opt, "multi_peak_10")

        # FW-LP
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  FW-LP, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, seed=seed + 300)
            try_seq(a_opt, "fw_lp")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 4: Peak balancing polish ===
    print("\n=== Phase 4: Peak Balancing Polish ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 200)
            print(f"  Peak polish, t={t_alloc:.0f}s")
            a_opt, _ = peak_balancing_polish(best_seq, time_limit=t_alloc,
                                             seed=seed + 400)
            try_seq(a_opt, "peak_polish")

        # Additional smoothed gradient with very low temperature
        remaining = deadline - time.monotonic() - 60
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print(f"  Fine smooth T=0.0001, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_gradient_optimize(best_seq, time_limit=t_alloc,
                                                  T_init=0.0001, lr_init=0.02,
                                                  use_spectral=False,
                                                  seed=seed + 500)
            try_seq(a_opt, "fine_smooth")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 5: Final refinement with perturbations ===
    print("\n=== Phase 5: Final Refinement ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Try small perturbations with quick optimization
        for pert_scale in [0.001, 0.002, 0.003, 0.005]:
            remaining = deadline - time.monotonic() - 20
            if remaining < 5:
                break
            t_alloc = min(remaining, 30)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * pert_scale)
            a_opt, _ = smoothed_gradient_optimize(f_pert, time_limit=t_alloc,
                                                  T_init=0.0002, lr_init=0.01,
                                                  use_spectral=True,
                                                  seed=seed + 600 + int(pert_scale * 10000))
            try_seq(a_opt, f"perturb_{pert_scale}")

        # One more FW-LP
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(remaining, 20),
                                       seed=seed + 700)
            try_seq(a_opt, "final_fw")

        # Final peak polish
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_opt, _ = peak_balancing_polish(best_seq, time_limit=min(remaining, 15),
                                             seed=seed + 800)
            try_seq(a_opt, "final_polish")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
