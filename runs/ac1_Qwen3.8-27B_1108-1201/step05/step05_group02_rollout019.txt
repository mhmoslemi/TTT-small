def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def solve_lp_min_peak(f, n, M_hint=None):
    """Solve LP: min t s.t. (f*g)[k] <= t, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

    This is equivalent to finding g that minimizes the weighted autocorrelation."""
    # The objective is: sum_k w[k] * sum_j f[j]*g[k-j] = sum_j g[j] * sum_k w[k]*f[k-j]

from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlate(w, f):
    """Compute c[j] = sum_k w[k] * f[k-j] (valid range) using FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    # cross-correlation: c[j] = sum_k w[k] f[k-j]
    # This is the correlation of w with f
    corr = np.fft.irfft(W * np.conj(F), fft_size)[:size]
    # corr[k] = sum_j w[j] f[j+k-n+1]... let me be more careful
    # Actually np.correlate(w, f, 'full')[k] = sum_j w[j+k] * conj(f[j])
    # We want c[j] = sum_k w[k] * f[k-j]
    # = sum_m w[m+j] * f[m] where m = k-j
    # This is cross-correlation of f and w with lag j
    # np.correlate(f, w, 'full')[j] = sum_m f[m] * conj(w[m+j-n_w+1])... 
    # Let me just use direct approach for correctness
    # c[j] = sum over valid k of w[k] * f[k-j]
    # k ranges from max(0, j) to min(2n-2, 2n-2), and k-j ranges from 0 to n-1
    # So k-j = i means k = i+j, i in [0, n-1], k = i+j in [0, 2n-2]
    # c[j] = sum_i f[i] * w[i+j] for i=0..n-1 where i+j < 2n-1
    # This is the cross-correlation (f * w)[j] in the valid range
    # Which is exactly the full convolution of f and reversed w... 
    # Actually let's just use FFT properly:
    # c[j] = sum_i f[i] * w[i+j]
    # Let's define: this is the cross-correlation of f and w at lag j
    # = (f ⊛ w)[j] where ⊛ denotes cross-correlation
    # Using FFT: corr = ifft(conj(FFT(f)) * FFT(w))
    # Hmm, let me just do it directly with FFT of the convolution
    # c[j] = sum_i f[i] w[i+j]
    # Define u[i] = f[i], v[k] = w[k]
    # c[j] = sum_i u[i] v[i+j]
    # This is the cross-correlation. In terms of convolution:
    # If we define v_flipped[k] = v[-k] (padded), then c = u * v_flipped
    # More simply: c[j] = sum_i u[i] v[i+j] = (full_conv(u, v_rev))[j + n - 1]
    # where v_rev is v reversed and padded
    # Actually, the simplest correct approach:
    # c[j] = sum_i f[i] * w[i+j] for i=0..n-1, i+j < len(w)
    # This equals the cross-correlation, which via FFT is:
    # corr = ifft(conj(fft(f, N)) * fft(w, N))
    # where corr[j] gives the cross-correlation at lag j
    # Let me verify: conj(FFT(f)) * FFT(w) -> ifft gives sum_i conj(f[i]) w[i+j]
    # Since f is real, conj(f[i]) = f[i], so we get sum_i f[i] w[i+j] = c[j] ✓
    F_full = np.fft.rfft(f, fft_size)
    W_full = np.fft.rfft(w, fft_size)
    corr = np.fft.irfft(np.conj(F_full) * W_full, fft_size)
    return corr[:n]  # c[j] for j=0..n-1 (lags 0 to n-1)


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={"presolve": True, "time_limit": 30.0})
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        return g, res.x[-1]
    return None, None


def fw_min_vertex_direction(f, conv, beta):
    """Find FW direction using softmax-weighted minimum vertex oracle (FFT-accelerated)."""
    n = len(f)
    # Softmax weights on peaks
    scaled = conv * beta
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    # c[j] = sum_k w[k] * f[k-j] (cross-correlation)
    c = cross_correlate(w, f)
    j_star = int(np.argmin(c))
    g = np.zeros(n)
    g[j_star] = 1.0
    return g, c[j_star]


def lp_fw_optimize(f0, time_limit, verbose=False, use_lp=True, beta_start=50.0):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    beta = beta_start
    step_boost = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Choose direction
        if use_lp and n <= 2500:
            g, lp_t = solve_lp_min_peak(f, n)
            if g is None:
                g, _ = fw_min_vertex_direction(f, conv, beta)
        else:
            g, _ = fw_min_vertex_direction(f, conv, beta)

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search with bisection
        best_alpha = 0.0
        best_val = M

        # First pass: coarse grid
        alphas = [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Fine bisection
            if best_alpha < 1.0:
                lo = max(0.0, best_alpha - 0.03)
                hi = min(1.0, best_alpha + 0.03)
            else:
                lo = 0.8
                hi = 1.0
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    hi = mid
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"      FW iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve > 30:
            beta *= 1.5
            step_boost = 1.0
        elif no_improve > 80:
            # Restart with perturbation
            rng = np.random.default_rng(it * 6151 + 7)
            scale = 0.002 + 0.003 * min(no_improve // 100, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            beta = beta_start
            no_improve = 0

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.003, momentum=0.93, top_k=50, temp=0.0005,
                         verbose=False):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient using FFT for efficiency
        # sg[i] = 2 * sum_k w[k] * f[k-i] (for peaks k)
        # Build a peak-weight vector and use cross-correlation
        w_full = np.zeros(len(conv))
        w_full[peaks] = w
        sg = 2.0 * cross_correlate(w_full, f)

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        co = conv_new.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        f = f_new
        conv = conv_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        if time.monotonic() - t0 > time_limit:
            break
        improved = False

        # Find top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-14:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj
                multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.4, 1.8, 2.5]

                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 2:
                # Random exploration
                rng = np.random.default_rng(passes * 104729 + 3)
                for _ in range(min(30, n // 5)):
                    if time.monotonic() - t0 > time_limit:
                        break
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-12:
                        continue
                    for mult in [0.4, 0.7, 1.5, 2.0]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.4:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break
                    if improved:
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-FW, subgradient, coordinate descent,
    multi-length exploration, and final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: LP-FW on parent (main workhorse) ===
    print(f"\n=== Phase 1: LP-FW on parent (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(500, rem * 0.35)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True, use_lp=True, beta_start=30.0)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Continue LP-FW with different beta ===
    print(f"\n=== Phase 2: LP-FW refined (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(300, rem * 0.2)
        if t2 > 30:
            print(f"  n={len(best_seq)}, Time: {t2:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True, use_lp=True, beta_start=100.0)
            try_seq(a_opt, "lpfw_refined")

    # === Phase 3: Try different lengths ===
    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        base_n = len(base)
        # Try nearby and longer lengths
        candidate_lengths = []
        for delta in [-128, -64, 64, 128, 256, 512, 1024]:
            ln = base_n + delta
            if 256 <= ln <= 4096:
                candidate_lengths.append(ln)
        # Sort by proximity to base
        candidate_lengths.sort(key=lambda x: abs(x - base_n))

        for target_n in candidate_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 100:
                break
            t_alloc = min(rem * 0.2, 180)
            if t_alloc < 30:
                break
            f_interp = interpolate_sequence(base, target_n)

            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, use_lp=(target_n <= 2500), beta_start=50.0)

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")

        n_best = len(best_seq)
        rem = deadline - time.monotonic()

            print(f"  n={n_best}, Time: {t_sg:.0f}s")

                                             lr=0.002, momentum=0.94,

                                             temp=0.0004)
            try_seq(a_opt, "sg_main")

            # Second round with different params
            rem = deadline - time.monotonic()
            t_sg2 = min(rem * 0.12, 120)

    print(f"\n=== Phase 5: CD Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.2, 200)
        if t_cd > 20:
            print(f"  n={len(best_seq)}, Time: {t_cd:.0f}s")
            a_opt, _ = coord_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_polish")

            # Another CD round
            rem = deadline - time.monotonic()
            t_cd2 = min(rem * 0.1, 80)
            if t_cd2 > 15:
                a_opt, _ = coord_descent_polish(best_seq, time_limit=t_cd2)
                try_seq(a_opt, "cd_polish2")

    # === Phase 6: Final LP-FW + perturbations ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 60
        if rem > 30:

            if t_final > 20:

        for p in range(5):

            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.4, 60)
            if t_pert > 15:

                                           use_lp=(n_best <= 2500), beta_start=50.0)
                try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
