def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, active_peaks, weights):
    """Compute subgradient of max_k conv(a,a)[k] at given active peaks.
    
    d/da_j [(a*a)[k]] = 2*a[k-j] if 0 <= k-j < n, else 0
    """
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(active_peaks, weights):
        if w < 1e-15:

        # j ranges over 0..n-1, k-j ranges over 0..n-1
        # So j ranges over k-(n-1) to k, intersected with [0, n-1]
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            grad[j] += 2.0 * w * a[k - j]
    return grad

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a, b=None):
    """Linear convolution of a and b (or auto-convolution if b is None)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_fw_solve(g_init, f, n, theta, inner_iters=20, inner_lr=0.01):
    """Solve: min_g LSE(theta * (f*g)) s.t. g >= 0, sum(g) = 1.
    Uses projected gradient descent."""
    g = np.maximum(g_init, 1e-15)
    g /= g.sum()
    
    for _ in range(inner_iters):
        # Compute f*g
        fg = fast_convolve(f, g)
        
        # Compute softmax weights
        scaled = theta * fg
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()
        
        # Compute gradient of LSE w.r.t. g
        # d/dg[j] LSE = sum_k p[k] * f[k-j] (if 0 <= k-j < n)
        # This is cross-correlation of p and f
        grad = np.zeros(n)
        for j in range(n):
            k_lo = max(j, 0)
            k_hi = min(j + n - 1, 2 * n - 2)
            if k_lo <= k_hi:
                grad[j] = np.dot(p[k_lo:k_hi+1], f[k_lo-j:k_hi-j+1])
        
        # Gradient descent step
        g = g - inner_lr * grad
        g = project_simplex(g)
    
    return g

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LSE-based Frank-Wolfe with
adaptive temperature, mirror descent, coordinate descent polish, and
multi-length exploration. Key innovation: log-sum-exp inner problem
replaces LP for faster, smoother direction finding.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a, b=None):
    """Linear convolution of a and b via FFT."""
    if b is None:
        b = a
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(p, f, n):
    """Compute gradient of LSE objective w.r.t. g.
    grad[j] = sum_k p[k] * f[k-j] for valid 0 <= k-j < n."""
    grad = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        if k_start <= k_end:
            length = k_end - k_start + 1
            grad[j] = np.dot(p[k_start:k_start + length], f[0:length])
    return grad


def lse_fw_optimize(f0, time_limit, theta_start=15.0, theta_end=300.0,

def mirror_descent_optimize(f0, time_limit, verbose=True):
    """Exponentiated gradient (mirror descent) on the simplex."""
    n = len(f0)
    a = np.maximum(f0, 1e-15).astype(np.float64)
    a /= a.sum()

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    eta = 0.05
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    peak_temp = 0.005

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)

        # Softmax-weighted subgradient
        scaled = conv / peak_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute subgradient: weighted combination of active peak gradients
        grad = np.zeros(n)
        significant = np.where(w > 1e-6)[0]
        for idx in significant:
            k = int(idx)
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1)
            grad[js] += 2.0 * w[idx] * a[k - js]

        # Mirror descent: update in log-space
        a = np.maximum(a - eta * grad, 1e-15)
        a /= a.sum()

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    MD it={it}: obj={best_obj:.10f} eta={eta:.5f}")
        else:
            no_improve += 1

        eta *= 0.9997
        peak_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(hash(it * 31) % (2**31))
            a = np.maximum(best_a + rng.standard_normal(n) * 0.001, 1e-15)
            a /= a.sum()
            eta = max(eta * 2, 0.02)
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=True):
    """Coordinate descent targeting peak-contributing coordinates."""
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64)
    f /= f.sum()

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        conv = fast_convolve(f)

        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            j_start = max(0, peak_k - (n - 1))
            j_end = min(n - 1, peak_k)

            for j in range(j_start, j_end + 1):
                if f[j] < 1e-14:
                    continue
                # Try reducing f[j] and redistributing
                for frac in [0.5, 0.7, 0.85, 1.2, 1.5]:
                    new_val = f[j] * frac
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - f[j]
                    if denom < 1e-12:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial /= f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True

        if not improved:
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0

    return best_f, best_obj

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: min t s.t. (f*g)[k] <= t, g >= 0, sum(g) = 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        n_constr = 2 * n - 1
        A_ub = np.zeros((n_constr, n + 1))
        for j in range(n):
            A_ub[j:j + n, j] = f
        A_ub[:, -1] = -1.0
        b_ub = np.zeros(n_constr)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'time_limit': 30})
        except Exception:
            break

        if not res.success:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        g = res.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = f.copy()
        else:
            g /= gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = best_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            s = ft.sum()
            if s < 1e-15:
                continue
            ft /= s
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW it={it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LSE-FW (primary), mirror descent, LP-FW,
    coordinate descent polish, and multi-length exploration.
    """
    rng = np.random.default_rng(42)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: LSE-FW (primary, ~50% time) ===
    print(f"\n=== Phase 1: LSE-FW (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 600)
    if t1 > 30:
        print(f"  Allocating {t1:.0f}s")
        a_opt, _ = lse_fw_optimize(parent, time_limit=t1,
                                   theta_start=10.0, theta_end=200.0,
                                   inner_steps=5, verbose=True)
        try_seq(a_opt, "lse_fw_main")

        # Second pass with different params
        rem = deadline - time.monotonic()
        t1b = min(rem * 0.2, 150)
        if t1b > 20 and best_seq is not None:
            print(f"  LSE-FW pass 2: {t1b:.0f}s")
            a_opt, _ = lse_fw_optimize(best_seq, time_limit=t1b,
                                       theta_start=50.0, theta_end=500.0,
                                       inner_steps=8, verbose=True)
            try_seq(a_opt, "lse_fw_pass2")

    # === Phase 2: Mirror Descent (~15% time) ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(rem * 0.2, 300)
        if t2 > 20:
            print(f"  Allocating {t2:.0f}s")
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t2, verbose=True)
            try_seq(a_opt, "mirror_descent")

    # === Phase 3: LP-FW backup (~15% time) ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        if n_best <= 1024:
            rem = deadline - time.monotonic()
            t3 = min(rem * 0.2, 250)
            if t3 > 20:
                print(f"  Allocating {t3:.0f}s (n={n_best})")
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
                try_seq(a_opt, "lp_fw")

    # === Phase 4: Multi-length exploration (~10% time) ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        candidate_lengths = set()
        for delta in [-64, -32, 32, 64, 128]:
            ln = len(base) + delta
            if 256 <= ln <= 2048:
                candidate_lengths.add(ln)

        for target_n in sorted(candidate_lengths)[:3]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.3, 150)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  LSE-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lse_fw_optimize(f_interp, time_limit=t_alloc,
                                       theta_start=20.0, theta_end=200.0,
                                       inner_steps=5, verbose=False)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 5: Coordinate Descent Polish (~10% time) ===
    print(f"\n=== Phase 5: CD Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.4, 200)
        if t5 > 15:
            print(f"  Allocating {t5:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "cd_polish")

    # === Phase 6: Final perturbations ===
    print(f"\n=== Phase 6: Final Perturbations (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise_scale = 0.0005 + 0.0005 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            # Quick LSE-FW refinement
            t_pert = min(rem, 30)
            if t_pert > 5:
                a_opt, _ = lse_fw_optimize(f_pert, time_limit=t_pert,
                                           theta_start=50.0, theta_end=300.0,
                                           inner_steps=3, verbose=False)
                try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
