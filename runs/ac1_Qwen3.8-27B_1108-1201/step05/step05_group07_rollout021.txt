def spectral_project(a, flatness=1.0):
    """Project a onto the set of sequences with flat auto-convolution spectrum."""
    n = len(a)
    L = 1 << (2*n - 1 - 1).bit_length()  # Next power of 2 >= 2n-1
    a_pad = np.zeros(L)
    a_pad[:n] = a
    A = np.fft.fft(a_pad)
    target_mag = np.sqrt(np.sum(a**2))  # = ||a||_2
    # Blend between flat and original magnitude for stability
    A_new = np.zeros_like(A)
    mag = np.abs(A)
    for i in range(L):
        if mag[i] > 1e-15:
            new_mag = (1-flatness) * mag[i] + flatness * target_mag
            A_new[i] = new_mag * A[i] / mag[i]

    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_project(a, flatness=0.5):
    """
    Geometric-magnitude spectral projection toward flat auto-convolution spectrum.
    Makes |FFT(a_padded)| more uniform, which flattens the linear auto-convolution.
    """
    n = len(a)
    L = 1 << (2 * n - 2).bit_length()
    if L < 2 * n - 1:
        L *= 2
    a_pad = np.zeros(L)
    a_pad[:n] = a
    A = np.fft.fft(a_pad)
    l2_norm = np.sqrt(np.sum(a * a))
    if l2_norm < 1e-15:
        return a.copy()
    target_mag = l2_norm
    mag = np.abs(A)
    eps = 1e-30
    log_mag = np.log(mag + eps)
    log_target = np.log(target_mag + eps)
    log_new_mag = (1.0 - flatness) * log_mag + flatness * log_target
    new_mag = np.exp(log_new_mag)
    phase = A / (mag + eps)
    A_new = new_mag * phase
    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def spectral_subgradient_loop(f0, time_limit, top_k=50, temperature=0.0001,
                              lr=0.003, momentum=0.85, spectral_prob=0.3,
                              flatness=0.4, verbose=False, label=""):
    """
    Combined spectral projection + multiplicative-weights subgradient optimization.
    Alternates between spectral flattening and gradient-based peak reduction.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k
    cur_flat = flatness

    while time.monotonic() - t0 < time_limit:
        it += 1
        do_spectral = (it % 10 < int(10 * spectral_prob))

        if do_spectral:
            f_spec = spectral_project(f, cur_flat)
            if f_spec.sum() > 1e-10:
                f = f_spec
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            f_nest = f * np.exp(momentum * 0.5 * v)
            fn = f_nest.sum()
            if fn < 1e-30:
                f_nest = np.ones(n) / n
            else:
                f_nest /= fn

            conv_nest = fast_convolve(f_nest)
            k_actual = min(cur_top_k, len(conv_nest))
            peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
            pv = conv_nest[peaks]
            scaled = pv / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(pv)] = 1.0
            else:
                w /= ws

            sg = np.zeros(n)
            for i in range(len(peaks)):
                if w[i] < 1e-13:
                    continue
                k = int(peaks[i])
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                sg[idx] += 2.0 * w[i] * f_nest[k - idx]

            v = momentum * v + (1.0 - momentum) * sg
            exponent = -cur_lr * v
            exponent -= exponent.max()
            f = f * np.exp(exponent)
            fs = f.sum()
            if fs < 1e-30:
                f = best_f.copy()
                v = np.zeros(n)
                cur_lr = max(lr * 0.3, 0.0001)
                continue
            f /= fs
            f = np.maximum(f, 0.0)
            f /= f.sum()

            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        cur_lr *= 0.9998
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp *= 1.006
        if it % 800 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.92))
        cur_flat = min(0.7, cur_flat * 1.003)

        if no_improve > 400:
            rng = np.random.default_rng((it * 7919 + 17) % 1000000)
            noise_scale = 0.003 + 0.002 * min(no_improve // 500, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            f = 0.7 * f + 0.3 * spectral_project(best_f, 0.5)
            f = np.maximum(f, 0)
            f /= f.sum()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_temp = temperature * 0.5
            cur_flat = 0.3
            conv = fast_convolve(f)
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SSGL iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def pure_subgradient(f0, time_limit, top_k=50, temperature=0.00005,
                     lr=0.003, momentum=0.85):

    f = np.maximum(f0.astype(np.float64), 0.0)

        fn = f_nest.sum()
        if fn < 1e-30:

        pv = conv_nest[peaks]

        scaled -= scaled.max()

        ws = w.sum()
        if ws < 1e-15:

            js = max(0, k - n + 1)
            je = min(n - 1, k)

            cur_lr = 0.00005
        cur_temp *= 1.005

        if no_improve > 350:
            rng = np.random.default_rng((it * 6271 + 7) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0002)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative perturbation per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    mid = n // 2
    order = []
    for d in range(mid + 1):
        if mid + d < n:
            order.append(mid + d)
        if mid - d >= 0 and d > 0:
            order.append(mid - d)
    if mid - d >= 0:
        order.append(mid - d)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def pair_exchange_search(f0, time_limit, max_rounds=200):
    """Targeted pair-exchange local search focused on peak-reducing transfers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for _ in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(f)
        top_peaks = np.argpartition(conv, -min(20, len(conv)))[
            -min(20, len(conv)):]
        improved = False

        for k in top_peaks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            if contribs.max() < 1e-15:
                continue
            hi_pos = np.argmax(contribs)
            lo_pos = np.argmin(contribs)
            hi = idx[hi_pos]
            lo = idx[lo_pos]
            if hi == lo or f[hi] < 1e-10:
                continue
            for frac in [0.15, 0.3, 0.5]:
                amt = f[hi] * frac
                f[hi] -= amt
                f[lo] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[hi] += amt
                    f[lo] -= amt
                    break

        if not improved:
            for _ in range(100):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or f[i] < 1e-10:
                    continue
                for frac in [0.1, 0.3, 0.5]:
                    amt = f[i] * frac
                    f[i] -= amt
                    f[j] += amt
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-13:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[i] += amt
                        f[j] -= amt
                        break

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    center = (n - 1) / 2.0
    sigma = n / (5.0 + rng.random() * 3.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    r = np.abs(rng.standard_normal(n))
    klen = max(3, n // 80)
    kernel = np.ones(klen) / klen
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Raised cosine squared
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    rc = np.maximum(rc, 1e-10)
    inits.append(rc / rc.sum())

    # Spectral init from uniform
    spec = spectral_project(np.ones(n) / n, 0.6)
    inits.append(spec)

    # Bi-modal
    sig2 = n / 8.0
    left = np.exp(-0.5 * ((np.arange(n) - n * 0.25) / sig2) ** 2)
    right = np.exp(-0.5 * ((np.arange(n) - n * 0.75) / sig2) ** 2)
    bim = left + right
    bim = np.maximum(bim, 1e-10)
    inits.append(bim / bim.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining spectral projection, subgradient descent,
    and local search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent, using uniform n=512")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 1: Diversified spectral + subgradient initialization ===
    print("\n=== Phase 1: Diversified Initialization ===")
    n_parent = len(parent)
    all_inits = list(make_structured_inits(n_parent, rng))
    all_inits.append(parent)

    # Also try a few other lengths
    for alt_n in [640, 896]:
        if alt_n != n_parent:
            all_inits.append(interpolate_sequence(parent, alt_n))

    for idx, cand in enumerate(all_inits):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        n_cand = len(cand)
        t_alloc = min(rem * 0.15, 60)
        if t_alloc < 10:
            t_alloc = min(rem * 0.3, 40)

        print(f"  Init {idx}: n={n_cand}, t={t_alloc:.0f}s")
        a_opt, _ = spectral_subgradient_loop(
            cand, time_limit=t_alloc,
            top_k=max(20, n_cand // 8),
            temperature=0.00005, lr=0.003,
            momentum=0.85, spectral_prob=0.3, flatness=0.4)
        try_seq(a_opt, f"init_{idx}")

    # === Phase 2: Deep optimization with multiple configs ===
    print(f"\n=== Phase 2: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {"top_k": max(30, n_best // 4), "temp": 0.00003, "lr": 0.002,
             "mom": 0.90, "spec_p": 0.25, "flat": 0.3},
            {"top_k": max(20, n_best // 6), "temp": 0.0001, "lr": 0.004,
             "mom": 0.82, "spec_p": 0.4, "flat": 0.5},
            {"top_k": max(50, n_best // 3), "temp": 0.00001, "lr": 0.001,
             "mom": 0.93, "spec_p": 0.15, "flat": 0.2},
            {"top_k": max(40, n_best // 2), "temp": 0.00008, "lr": 0.005,
             "mom": 0.78, "spec_p": 0.5, "flat": 0.6},
        ]

        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            t_alloc = min(rem * 0.35, 150)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.7 * best_seq + 0.3 * spectral_project(best_seq, 0.5)
                f_init = np.maximum(f_init, 0)
                f_init /= f_init.sum()
            else:
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = project_simplex(f_init)

            print(f"  Deep {p}: n={n_best}, t={t_alloc:.0f}s, spec_p={cfg['spec_p']}")
            a_opt, _ = spectral_subgradient_loop(
                f_init, time_limit=t_alloc,
                top_k=cfg["top_k"], temperature=cfg["temp"],
                lr=cfg["lr"], momentum=cfg["mom"],
                spectral_prob=cfg["spec_p"], flatness=cfg["flat"])
            try_seq(a_opt, f"deep_{p}")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [700, 832, 1024, 1280]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            f_interp = 0.8 * f_interp + 0.2 * spectral_project(f_interp, 0.4)
            f_interp = np.maximum(f_interp, 0)
            f_interp /= f_interp.sum()
            t_alloc = min(rem * 0.25, 80)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = spectral_subgradient_loop(
                f_interp, time_limit=t_alloc,
                top_k=max(30, target_n // 6),
                temperature=0.00004, lr=0.003,
                momentum=0.87, spectral_prob=0.3, flatness=0.35)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 4: Local search ===
    print(f"\n=== Phase 4: Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_pex = min(rem * 0.35, 60)
            print(f"  Pair exchange t={t_pex:.0f}s")
            a_opt, _ = pair_exchange_search(best_seq, time_limit=t_pex)
            try_seq(a_opt, "pair_exch")

        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_cd = min(rem * 0.5, 80)
            print(f"  Coord descent t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 5: Fine-tuning with spectral + very fine gradient ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine subgradient
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            a_opt, _ = pure_subgradient(
                best_seq, time_limit=min(rem, 50),
                top_k=n_best, temperature=0.000005,
                lr=0.0005, momentum=0.96)
            try_seq(a_opt, "sg_fine")

        # Spectral + gradient hybrid
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = spectral_subgradient_loop(
                best_seq, time_limit=min(rem, 35),
                top_k=n_best, temperature=0.000002,
                lr=0.0004, momentum=0.95,
                spectral_prob=0.4, flatness=0.6)
            try_seq(a_opt, "spec_fine")

        # Pair exchange on best
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = pair_exchange_search(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "pex_fine")

        # Tiny perturbations
        for p in range(6):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            scale = 0.0005 + 0.0005 * p
            noise = rng.standard_normal(n_best) * scale
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem, 10)
            a_opt, _ = pure_subgradient(
                f_pert, time_limit=t_pert,
                top_k=max(20, n_best // 4),
                temperature=0.00001, lr=0.001, momentum=0.93)
            try_seq(a_opt, f"pert_{p}")

        # Final coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
