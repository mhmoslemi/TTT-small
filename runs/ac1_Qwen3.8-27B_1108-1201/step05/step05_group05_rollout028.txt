            cols.append(j)
            vals.append(f[k - j])

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe with
sparse LP, adaptive step sizes, multi-start diversity, smooth gradient descent,
and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n, active_k):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_k, g >= 0."""
    c = -np.ones(n)
    K = len(active_k)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(f[k - j])
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coord_descent(f0, time_limit):
    """Coordinate descent with golden section line search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 3:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                # Try setting it to a small positive value
                candidates = [1e-4, 5e-4, 1e-3, 5e-3]
            else:
                # Try scaling the current value
                candidates = [f[j] * m for m in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]]
            
            best_j = f[j]
            best_j_obj = best_obj
            
            for new_val in candidates:
                if new_val < 0 or new_val > 0.99:
                    continue
                remaining = 1.0 - new_val
                denom = 1.0 - f[j]
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial[np.arange(n) != j] *= scale
                f_trial = f_trial / f_trial.sum()
                
                obj_val = fast_convolve(f_trial).max()
                if obj_val < best_j_obj - 1e-15:
                    best_j_obj = obj_val
                    best_j = new_val
            
            if best_j != f[j]:
                denom = 1.0 - f[j]
                if denom > 1e-10:
                    scale = (1.0 - best_j) / denom
                    f[j] = best_j
                    f[np.arange(n) != j] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
