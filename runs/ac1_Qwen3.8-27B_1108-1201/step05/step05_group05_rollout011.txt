import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n from a."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def objective(f):
    """Compute the objective value for normalized f (sum=1)."""
    conv = fast_convolve(f)
    return conv.max()


def solve_lp_full(f, M, n):
    """Solve LP with all constraints."""
    c = -np.ones(n)
    K = 2 * n - 1
    A_ub = np.zeros((K, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_adaptive(f, M, n, top_k):
    """Solve LP with top-k constraints (most binding)."""
    conv = fast_convolve(f)
    k_sorted = np.argsort(conv)[::-1][:top_k]
    c = -np.ones(n)
    K = len(k_sorted)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    for row, k in enumerate(k_sorted):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[int(k) - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coord_descent_golden(f0, time_limit, max_inner=30):
    """Coordinate descent with golden-section search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

        passes += 1
        improved = False
        phi = (np.sqrt(5.0) - 1.0) / 2.0

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            # Golden section search for coordinate j
            # Fix all other coordinates proportionally
            lo = 0.0
            hi = 1.0 - 1e-10

            c = hi - phi * (hi - lo)
            d = lo + phi * (hi - lo)

            def eval_coord(val):
                if val < 0 or val > 1 - 1e-10:
                    return float('inf')
                denom = 1.0 - old_val
                if denom < 1e-12:
                    return float('inf')
                scale = (1.0 - val) / denom
                f_trial = f.copy()
                f_trial[j] = val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                return fast_convolve(f_trial).max()

            fc = eval_coord(c)
            fd = eval_coord(d)

            for _ in range(max_inner):
                if time.monotonic() - t0 > time_limit:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - phi * (hi - lo)
                    fc = eval_coord(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + phi * (hi - lo)
                    fd = eval_coord(d)

            best_val = (lo + hi) / 2.0
            best_val_obj = eval_coord(best_val)

            if best_val_obj < best_obj - 1e-14 and best_val > 0:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_val) / denom
                    f[j] = best_val

                    f[mask] *= scale

                    best_obj = best_val_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe with
backtracking line search, thorough golden-section coordinate descent,
multi-length exploration, palindromic symmetry, frequency-domain initialization,

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n from a."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def solve_lp_full(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    K = 2 * n - 1
    A_ub = np.zeros((K, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_adaptive(f, conv, M, n, top_k):
    """Solve LP with top-k most binding constraints."""
    k_sorted = np.argsort(conv)[::-1][:top_k]
    c = -np.ones(n)
    K = len(k_sorted)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    for row, k in enumerate(k_sorted):
        ki = int(k)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        for j in range(js, je + 1):
            A_ub[row, j] = f[ki - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coord_descent_golden(f0, time_limit, max_inner=20):
    """Coordinate descent with golden-section search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    no_improve = 0
    phi = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            lo = 0.0
            hi = 1.0 - 1e-10

            c = hi - phi * (hi - lo)
            d = lo + phi * (hi - lo)

            def eval_coord(val):
                if val < 0 or val > 1 - 1e-10:
                    return float('inf')
                scale = (1.0 - val) / denom
                f_trial = f.copy()
                f_trial[j] = val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                return fast_convolve(f_trial).max()

            fc = eval_coord(c)
            fd = eval_coord(d)

            for _ in range(max_inner):
                if time.monotonic() - t0 > time_limit:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - phi * (hi - lo)
                    fc = eval_coord(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + phi * (hi - lo)
                    fd = eval_coord(d)

            best_val = (lo + hi) / 2.0
            best_val_obj = eval_coord(best_val)

            if best_val_obj < best_obj - 1e-14 and best_val > 1e-12:
                scale = (1.0 - best_val) / denom
                f[j] = best_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if f.sum() < 1e-15:
                    break
                best_obj = best_val_obj
                best_f = f.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 2:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum (good for low peak autocorrelation)."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def subgradient_polish(f0, time_limit, lr=0.0015, mom=0.94, top_k=40, temp=0.00003):
    """Subgradient polishing with softmax-weighted active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_temp *= 1.003

            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)

                np.exp(-2.5 * ((np.arange(n_best) - n_best / 2.0) / (0.45 * n_best)) ** 2))),
            ("bimodal", project_simplex(
                np.exp(-4.0 * ((np.arange(n_best) - 0.25 * n_best) / (0.15 * n_best)) ** 2) +
                np.exp(-4.0 * ((np.arange(n_best) - 0.75 * n_best) / (0.15 * n_best)) ** 2))),
        ]

        for i, (name, f_init) in enumerate(restart_configs):
            rem = deadline - time.monotonic()
            if rem < 35:
                break
            t_alloc = min(rem / max(1, len(restart_configs) - i), 130)
            print(f"  Restart [{name}], t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=False, top_k=30, full_every=50)
            try_seq(a_opt, f"restart_{name}")

    # === Phase 6: Second CD pass ===
    print(f"\n=== Phase 6: Coordinate Descent Pass 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 50:
            t_cd2 = min(rem * 0.3, 250)
            print(f"  CD t={t_cd2:.0f}s")
            a_opt, _ = coord_descent_golden(best_seq, time_limit=t_cd2, max_inner=22)
            try_seq(a_opt, "cd2")

            # Third pass if time allows
            rem = deadline - time.monotonic()
            if rem > 30:
                t_cd3 = min(rem * 0.25, 150)
                a_opt3, _ = coord_descent_golden(a_opt, time_limit=t_cd3, max_inner=25)
                try_seq(a_opt3, "cd3")

    # === Phase 7: Subgradient polish ===
    print(f"\n=== Phase 7: SG Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 30:
            t_sg = min(rem * 0.25, 150)
            a_opt, _ = subgradient_polish(best_seq, time_limit=t_sg,
                                          lr=0.0012, mom=0.94,
                                          top_k=max(25, n_best // 5),
                                          temp=0.00002)
            try_seq(a_opt, "sg")

            rem = deadline - time.monotonic()
            if rem > 20:
                t_sg2 = min(rem * 0.2, 80)
                a_opt2, _ = subgradient_polish(a_opt, time_limit=t_sg2,
                                               lr=0.0006, mom=0.95,
                                               top_k=max(15, n_best // 8),
                                               temp=0.000008)
                try_seq(a_opt2, "sg_fine")

    # === Phase 8: Final LP-FW + perturbations ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 25:
            t_final = min(rem * 0.3, 100)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final,
                                      verbose=False, top_k=50, full_every=10)
            try_seq(a_opt, "final_lpfw")

        rng2 = np.random.default_rng(seed + 9999)
        for i in range(4):
            rem = deadline - time.monotonic()
            if rem < 12:
                break
            noise = rng2.standard_normal(n_best) * (0.001 + 0.0015 * i)
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.25, 45)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_pert,
                                      verbose=False, top_k=30, full_every=50)
            try_seq(a_opt, f"final_pert_{i}")

        # Tiny CD on final
        rem = deadline - time.monotonic()
        if rem > 15:
            t_final_cd = min(rem * 0.4, 50)
            a_opt, _ = coord_descent_golden(best_seq, time_limit=t_final_cd, max_inner=15)
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
