def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_multi_peak_subgrad(a, conv, top_k=30, temp=0.001):
    """Compute weighted subgradient using softmax over top peaks."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

    return sg

def coordinate_bisection(a0, time_limit, verbose=False, max_passes=500):

                continue

            b[j] = 0.0

            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            # Bisection on x in [0, 1-x_old*0.5, x_old*1.5] clipped to [0,1]
            # Actually just [0, 1] is fine since sum constraint handles it
            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(45):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                # f_k(mid) = om^2 * B[k] + 2*mid*om * b_valid[k] + mid^2 * [k==2j]
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))

                # Derivative of f_{k_max} at mid
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                    hi = mid
                    x_new = mid
                elif deriv < -1e-13:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

                continue

            a[j] = x_new

                a /= s_check

                a = a0.copy()
                continue

def pair_swap_polish(a0, time_limit, verbose=False):
    """Try pair swaps: transfer mass between coordinate pairs."""
    a = a0.copy().astype(np.float64)

    while time.monotonic() - t0 < time_limit:
        # Pick a random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        x_i = a[i]
        x_j = a[j]
        if x_i < 1e-12 or x_j < 1e-12:
            continue

        # Try transferring some mass from i to j (and vice versa)
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            # Transfer frac * x_i from i to j
            delta = frac * x_i
            a_trial = a.copy()
            a_trial[i] = x_i - delta
            a_trial[j] = x_j + delta
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-12:
                continue
            a_trial /= st

            obj = fast_convolve(a_trial).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_trial.copy()
                a = a_trial.copy()
                conv = fast_convolve(a)

            # Transfer frac * x_j from j to i
            delta2 = frac * x_j
            a_trial2 = a.copy()
            a_trial2[j] = x_j - delta2
            a_trial2[i] = x_i + delta2
            a_trial2 = np.maximum(a_trial2, 0.0)
            st2 = a_trial2.sum()
            if st2 < 1e-12:
                continue
            a_trial2 /= st2

            obj2 = fast_convolve(a_trial2).max()
            if obj2 < best_obj - 1e-12:
                best_obj = obj2
                best_a = a_trial2.copy()
                a = a_trial2.copy()

def symmetrize(a):
    """Make sequence symmetric: a[i] = (a[i] + a[n-1-i]) / 2, then renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym
