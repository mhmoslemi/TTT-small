import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_with_dual(f, M, n, active_k=None):
    """Solve LP and return both primal and dual solutions."""
    c = -np.ones(n)
    if active_k is None:
        K = 2 * n - 1
        k_range = range(K)
    else:
        K = len(active_k)
        k_range = active_k
    
    # Build constraint matrix
    # (f * g)[k] = sum_j f[k-j] * g[j] for valid j
    A_ub = np.zeros((K, n))
    for row, k in enumerate(k_range):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, res.ineqlin.marginals  # marginals are the dual variables
    return None, None

def compute_subgradient(f, conv, top_k=50, temp=0.001):
    """Compute softmax-weighted subgradient at active peaks."""
    n = len(f)
    # Select top-k peaks
    k_sel = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_sel)[-k_sel:]
    # Softmax weights
    pv = conv[peaks]

    w = np.exp(scaled)

        k = int(peaks[i])
        # gradient of (f*f)[k] w.r.t. f[j] is 2*f[k-j] (with zero-padding)

        je = min(n - 1, k)

        sg[j_vals] += 2.0 * w[i] * f[k - j_vals]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-Frank-Wolfe with
dual-guided subgradient steps, Adam optimizer, multi-length exploration,
and structured restarts. Key innovations over previous approach:
- Combined LP-FW + Adam-subgradient alternating scheme
- Multi-length exploration to find optimal sequence length
- Structured perturbations based on current solution features
- No forced palindromic symmetry (exploits asymmetric structure)
- Adaptive temperature subgradient with full and reduced LP alternation
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_subgradient(f, conv, top_k=50, temp=0.001):
    """Compute softmax-weighted subgradient of max(conv) at active peaks."""
    n = len(f)
    k_sel = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_sel)[-k_sel:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(k_sel):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j_vals = np.arange(js, je + 1)
        sg[j_vals] += 2.0 * w[i] * f[k - j_vals]
    return sg


def solve_lp_reduced(f, M, n, active_k):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    K = len(active_k)
    A_ub = np.zeros((K, n))
    for row, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_full(f, M, n):
    """Solve LP with all constraints."""
    c = -np.ones(n)
    K = 2 * n - 1
    A_ub = np.zeros((K, n))
    for k in range(K):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_direction(f, n, M, top_k=50, use_full=False):
    """Get FW direction via LP solve."""
    conv = fast_convolve(f)
    if use_full:
        g = solve_lp_full(f, M, n)
    else:
        k_sorted = np.argsort(conv)[::-1][:top_k]
        g = solve_lp_reduced(f, M, n, k_sorted.tolist())
    if g is not None and g.sum() > 1e-10:
        return g / g.sum()
    return None


def line_search(f, g, n, M, n_trials=8):
    """Backtracking line search for mixing f with g."""
    best_alpha = 0.0
    best_val = M
    alphas = np.linspace(0.05, 1.0, n_trials)
    for alpha in alphas:
        f_trial = f + alpha * (g - f)
        f_trial = np.maximum(f_trial, 0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        trial_conv = fast_convolve(f_trial)
        trial_M = trial_conv.max()
        if trial_M < best_val:
            best_val = trial_M
            best_alpha = alpha
    return best_alpha, best_val


def adam_subgradient_step(f, conv, adam_state, lr, top_k, temp):
    """Adam step using subgradient of max convolution."""
    n = len(f)
    sg = compute_subgradient(f, conv, top_k=top_k, temp=temp)
    t = adam_state['t'] + 1
    adam_state['t'] = t
    m = adam_state['m']
    v = adam_state['v']
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = beta1 * m + (1 - beta1) * sg
    v = beta2 * v + (1 - beta2) * sg ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = -lr * m_hat / (np.sqrt(v_hat) + eps)
    f_new = project_simplex(f + step)
    adam_state['m'] = m
    adam_state['v'] = v
    return f_new


def deep_optimize(f0, time_limit, verbose=False):
    """Deep optimization: alternating LP-FW and Adam-subgradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)

    adam_state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    lr = 0.05
    temp = 0.002
    top_k_fw = min(50, 2 * n - 1)
    top_k_sg = min(40, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Every 4th iteration: full LP for quality direction
        use_full = (it % 4 == 0 and n <= 1200)

        # Pattern: FW, SG, SG, FW(full), SG, SG, FW, SG, ...
        phase = it % 4
        if phase == 0 or phase == 3:
            # FW step
            g = fw_direction(f, n, M, top_k=top_k_fw, use_full=use_full)
            if g is not None:
                if np.max(np.abs(g - f)) < 1e-12:
                    no_improve += 1
                else:
                    alpha, trial_M = line_search(f, g, n, M)
                    if alpha > 1e-6:
                        f_new = f + alpha * (g - f)
                        f_new = np.maximum(f_new, 0)
                        f_new = f_new / f_new.sum()
                        conv_new = fast_convolve(f_new)
                        M_new = conv_new.max()
                        if M_new < best_obj - 1e-15:
                            best_obj = M_new
                            best_f = f_new.copy()
                            M = M_new
                            no_improve = 0
                        else:
                            no_improve += 1
                        f = f_new
                        conv = conv_new
                    else:
                        no_improve += 1
            else:
                no_improve += 1
        else:
            # Adam subgradient step
            f_new = adam_subgradient_step(f, conv, adam_state, lr,
                                          top_k_sg, temp)
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f_new.copy()
                M = M_new
                no_improve = 0
            else:
                no_improve += 1
            f = f_new
            conv = conv_new

        # Adapt parameters
        lr *= 0.9995
        temp *= 1.003
        if lr < 0.001:
            lr = 0.001

        # Restart logic
        if no_improve > 60:
            scale = 0.002 + 0.003 * min((no_improve - 60) // 30, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            adam_state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            lr = 0.03
            temp = 0.002
            no_improve = 0

        if verbose and it % 20 == 0:
            print(f"    iter {it}, t={elapsed:.0f}s, obj={best_obj:.8f}, "
                  f"no_improve={no_improve}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_multipliers=8):
    """Multiplier-based coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0 / 0.85, 1.3, 1.6, 2.5])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-13:
                continue
            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                remaining = 1.0 - new_val
                denom = 1.0 - old_val
                if denom < 1e-10 or remaining < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 2:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def fine_sg_polish(f0, time_limit, lr=0.008, top_k=30, temp=0.0005):
    """Fine subgradient polishing with Adam."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(42)
    adam_state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        f_new = adam_subgradient_step(f, conv, adam_state, cur_lr,
                                      top_k, cur_temp)
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        conv = conv_new
        cur_lr *= 0.9997
        cur_temp *= 1.002

        if no_improve > 200:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            adam_state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
            cur_lr = max(lr * 0.5, 0.001)
            no_improve = 0

    return best_f, best_obj

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification

    if np.isfinite(fs):
        best_score = fs

    return [float(x) for x in best_seq]
