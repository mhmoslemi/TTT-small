import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, temp=0.0001, top_k=None):
    """Compute the subgradient of max(conv) (or softmax approximation) w.r.t. a."""
    n = len(a)
    if top_k is not None and top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(len(conv))
    
    # Exponential weighting
    pv = conv[peaks].copy()
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        kg = k - idx
        mask = (kg >= 0) & (kg < n)
        sg[idx[mask]] += 2.0 * w[i] * a[kg[mask]]
    
    return sg

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated subgradient
descent with adaptive parameters, peak-targeting coordinate descent polishing,
and multi-length exploration. Avoids expensive LPs in favor of fast FFT-based
gradient methods for better time efficiency.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def nesterov_subgrad(a0, time_limit, lr=0.008, momentum=0.92, temp=5e-5,
                     top_k=40, verbose=False, label=""):
    """
    Nesterov-accelerated subgradient descent with adaptive parameters.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    a_prev = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    best_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute softmax-weighted subgradient
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            continue
        w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * a[kg[mask]]

        # Momentum update
        v = momentum * v - cur_lr * sg
        a_new = project_simplex(a + v)

        conv = fast_convolve(a_new)

def peak_coord_descent(a0, time_limit, verbose=False, label=""):

    For each element contributing to the peak, try to reduce it and

        improved = False

        # Find the top peaks

                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # For each contributing coordinate, try to reduce it
            # and redistribute to a random non-contributing coordinate
            for _ in range(min(3, je - js + 1)):
                if time.monotonic() - t0 > time_limit:
                    break
                j = int(rng.integers(js, je + 1))
                k_comp = peak_k - j
                if k_comp < 0 or k_comp >= n:
                    continue

                old_val = a[j]
                if old_val < 1e-10:
                    continue

                # Try several reduction factors
                best_factor = 1.0
                best_trial_obj = best_obj

                for factor in [0.5, 0.7, 0.85, 0.95, 0.6, 0.4, 1.2, 1.3]:
                    new_val = old_val * factor
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)

                    if ss < 1e-15:

                    trial_obj = trial_conv.max()
                    if trial_obj < best_trial_obj - 1e-15:

                    if denom > 1e-10:
                        scale = (1.0 - new_val) / denom
                        a[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        a[mask] *= scale
                        ss = a.sum()
                        if ss > 1e-15:
                            a /= ss
                        conv = fast_convolve(a)
                        if conv.max() < best_obj - 1e-15:
                            best_obj = conv.max()
                            best_a = a.copy()
                            improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Try random coordinate adjustments
                for _ in range(min(5, n // 50)):
                    if time.monotonic() - t0 > time_limit:
                        break
                    j = int(rng.integers(0, n))

                        scale = (1.0 - new_val) / denom

                        a_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        a_trial[mask] *= scale
                        ss = a_trial.sum()
                        if ss < 1e-15:

                        a_trial /= ss
                        trial_obj = fast_convolve(a_trial).max()
                        if trial_obj < best_obj - 1e-15:
                            a = a_trial
                            best_obj = trial_obj
                            best_a = a.copy()

            a_opt, _ = nesterov_subgrad(f_pert, time_limit=t_pert, lr=0.003,
                                        momentum=0.94, temp=2e-5, top_k=30,
                                        label=f"restart{p}")
            try_seq(a_opt, f"restart_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
