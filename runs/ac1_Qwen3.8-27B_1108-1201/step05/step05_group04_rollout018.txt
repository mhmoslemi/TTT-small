```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def lse_grad_fft(f, w):
    """Compute gradient of LSE(beta * (f*f)) using FFT cross-correlation.
    w: softmax weights of length 2n-1 (already computed)
    Returns gradient of length n.
    """
    n = len(f)
    # grad[m] = 2 * sum_k w[k] * f[k-m] (valid when 0 <= k-m < n)
    # = 2 * cross_correlation(w, f)[m]
    # = 2 * (w * rev(f))[n-1 + m] for m in [0, n-1]
    
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n-1 : 2*n-1]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LSE-FISTA with FFT-based gradient (O(n log n) per step)
2. Peak-targeted coordinate descent with analytic 1D minimization
3. Multi-length exploration with adaptive time allocation
4. Nesterov-accelerated momentum LSE for final polishing
5. Active-set peak equalization refinement
Key innovation: O(n log n) gradient via cross-correlation FFT identity,
plus coordinate descent that exploits the quadratic structure of each (f*f)[k].
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lse_grad_fft(f, w):
    """Compute LSE gradient via FFT cross-correlation.
    grad[m] = 2 * sum_k w[k] * f[k-m] for valid pairs.
    w has length 2n-1, f has length n. Returns array of length n.
    """
    n = len(f)
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f[::-1], fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_fista(f0, time_limit, beta_start=20.0, beta_end=8000.0, verbose=False):
    """FISTA on LSE-smoothed max(auto-conv) with FFT gradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    t_mom = 1.0
    y = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute LSE gradient at y
        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = lse_grad_fft(y, w)

        # Lipschitz constant: L = 4 * beta * ||f||_2^2
        L = 4.0 * beta * np.dot(f, f) + 1e-12
        step = 1.0 / L

        # FISTA update
        f_new = project_simplex(y - step * grad)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = f_new + ((t_mom - 1.0) / t_new) * (f_new - f)
        if (y < 0).any() or y.sum() > 1.0 + 1e-9:
            y = project_simplex(y)
        f = f_new
        t_mom = t_new

        # Check improvement
        conv_f = fast_convolve(f)
        obj = conv_f.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive beta
        if beta < beta_end:
            beta = min(beta * 1.005, beta_end)
        if no_improve > 100:
            beta = min(beta * 1.5, beta_end)
            t_mom = 1.0
            y = f.copy()
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    LSE-FISTA it={it}, obj={obj:.10f}, beta={beta:.0f}")

    return best_f, best_obj


def momentum_lse(f0, time_limit, beta_start=5.0, beta_end=10000.0,
                 lr_init=None, momentum_init=0.90, max_iter=None):
    """Momentum-based LSE gradient descent (ADAM-like)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 3.0 / n
    if max_iter is None:
        max_iter = 100000

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        conv_y = fast_convolve(f)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = lse_grad_fft(f, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_n = grad / gnorm

        v = mom * v - lr * grad_n
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 10.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.92
                mom = max(0.80, mom - 0.003)
            if no_improve > 80:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.001)
                beta = min(beta * 2.0, beta_end)
                mom = 0.88
                no_improve = 0

        if beta < beta_end:
            beta = min(beta * 1.003, beta_end)

    return best_f, best_obj


def coord_descent_peak(f0, time_limit, n_sweeps=None, verbose=False):
    """Coordinate descent targeting peak reduction.
    For each coordinate, try small perturbations and keep improvement.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0
    perturb_scale = 0.005

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved_this_sweep = False

        # Get current peaks
        peak_idx = np.argmax(conv)
        # Try perturbing coordinates that appear in the peak convolution
        peak_k = int(peak_idx)
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        candidates = list(range(lo, hi + 1))

        # Also add random coordinates
        if len(candidates) < 20:
            extra = np.random.default_rng(sweep * 7 + 3).integers(0, n, size=20 - len(candidates))
            candidates.extend(extra.tolist())

        rng = np.random.default_rng(sweep * 13 + 17)
        for m in candidates:
            if time.monotonic() - t0 > time_limit:
                break
            x_cur = f[m]
            if x_cur < 1e-12:
                continue

            # Try a few candidate values
            deltas = [
                -0.3 * x_cur, -0.15 * x_cur, -0.05 * x_cur,
                0.05 * x_cur, 0.15 * x_cur, 0.3 * x_cur
            ]
            # Add some random scale
            scale = perturb_scale * (0.5 + rng.random())
            deltas = [x_cur + d for d in deltas]
            deltas = [max(0.0, d) for d in deltas]

            best_x = x_cur
            best_local = best_obj
            # Quick eval: compute new conv incrementally
            for x_new in deltas:
                if x_new == x_cur:
                    continue
                # Update conv: (f*f)[k] changes by
                # delta_k = 2*(x_new - x_cur)*f[k-m]*I(valid) + (x_new^2 - x_cur^2)*I(k=2m)
                diff = x_new - x_cur
                conv_mod = conv.copy()
                # Part 1: 2*diff*f[k-m] for valid k-m
                # k-m in [0, n-1] means k in [m, m+n-1], intersect [0, 2n-2]
                k_lo = max(0, m)
                k_hi = min(2 * n - 2, m + n - 1)
                if k_lo <= k_hi:
                    ks = np.arange(k_lo, k_hi + 1)
                    conv_mod[ks] += 2.0 * diff * f[ks - m]
                # Part 2: (x_new^2 - x_cur^2) at k=2m
                k2m = 2 * m
                if k2m < 2 * n - 1:
                    conv_mod[k2m] += x_new * x_new - x_cur * x_cur

                # Renormalize: new sum = s - x_cur + x_new (but s=1)
                new_sum = 1.0 - x_cur + x_new
                if new_sum < 1e-10:
                    continue
                obj = conv_mod.max() / (new_sum * new_sum)
                if obj < best_local - 1e-14:
                    best_local = obj
                    best_x = x_new

            if best_x != x_cur:
                f[m] = best_x
                # Recompute conv
                conv = fast_convolve(f)
                new_sum = f.sum()
                if new_sum > 1e-10:
                    f = f / new_sum
                    conv = fast_convolve(f)
                    obj = conv.max()
                else:
                    obj = best_obj
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                improved_this_sweep = True

        if not improved_this_sweep:
            no_improve_sweeps += 1
            perturb_scale *= 0.95
            if no_improve_sweeps > 5:
                break
        else:
            no_improve_sweeps = 0
            perturb_scale = max(0.001, perturb_scale * 0.99)

        if verbose and sweep % 10 == 0:
            print(f"    CoordDescent sweep={sweep}, obj={best_obj:.10f}")

    return best_f, best_obj


def peak_equalize(f0, time_limit, n_peaks=8, verbose=False):
    """Iteratively equalize the top peaks using weighted gradient.
    Focus all optimization effort on the current maximum peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = 2.0 / n
    mom = 0.92
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        # Get top peaks
        k_sel = min(n_peaks, len(conv))
        peak_positions = np.argpartition(conv, -k_sel)[-k_sel:]
        peak_vals = conv[peak_positions]

        # Weight: emphasize peaks above the median of selected
        median_pv = np.median(peak_vals)
        w_sel = np.maximum(peak_vals - median_pv + 1e-10, 0.0)
        if w_sel.sum() < 1e-15:
            w_sel = np.ones(k_sel) / k_sel
        else:
            w_sel = w_sel / w_sel.sum()

        # Build full weight vector
        w_full = np.zeros(len(conv))
        w_full[peak_positions] = w_sel

        # Compute gradient
        grad = lse_grad_fft(f, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_n = grad / gnorm

        v = mom * v - lr * grad_n
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 8:
                lr *= 0.93
                mom = max(0.82, mom - 0.002)
            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr * 0.5, 0.1 / n)
                no_improve = 0
                n_peaks = min(n_peaks + 2, 30)  # widen the target set

    return best_f, best_obj

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def additive_correlation_init(n, n_terms=5, freqs=None):
    """Initialize with sum of shifted cosines for better structure."""
    rng = np.random.default_rng(42)
    if freqs is None:
        freqs = rng.uniform(1, 15, size=n_terms)
    phases = rng.uniform(0, 2 * np.pi, size=n_terms)
    f = np.ones(n)
    for i in range(n_terms):
        f += 0.1 * np.cos(2 * np.pi * freqs[i] * np.arange(n) / n + phases[i])
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f

            t_remaining = deadline - time.monotonic() - 20.0
            if t_remaining < 40:
                break
            t_alloc = min(50.0, t_remaining * 0.3)
            if t_alloc < 15:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  LSE-FISTA n={n_try}, t={t_alloc:.0f}s")
            f_opt, obj = lse_fista(f_init, time_limit=t_alloc,
                                    beta_start=10.0, beta_end=3000.0)
            try_sequence(f_opt, best_tracker)
            # Quick momentum polish
            t_alloc2 = min(30.0, deadline - time.monotonic() - 20.0)
            if t_alloc2 > 10:
                f_opt2, _ = momentum_lse(f_opt, time_limit=t_alloc2,
                                          beta_start=20.0, beta_end=5000.0)
                try_sequence(f_opt2, best_tracker)

    # === Phase 7: Final squeeze - alternate on best ===
    print(f"\n=== Phase 7: Final Squeeze ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    cycles = 0
    while time.monotonic() < deadline - 10.0 and cycles < 6:
        cycles += 1
        remaining = deadline - 10.0 - time.monotonic()
        if remaining < 25:
            break
        t_alloc = min(60.0, remaining * 0.5)

        if cycles % 3 == 1:
            print(f"  Final cycle {cycles}: momentum LSE t={t_alloc:.0f}s")
            f_opt, _ = momentum_lse(current, time_limit=t_alloc,
                                     beta_start=500.0, beta_end=50000.0,
                                     lr_init=0.5 / len(current))
            try_sequence(f_opt, best_tracker)
        elif cycles % 3 == 2:
            print(f"  Final cycle {cycles}: peak eq t={t_alloc:.0f}s")
            f_opt, _ = peak_equalize(current, time_limit=t_alloc, n_peaks=40)
            try_sequence(f_opt, best_tracker)
        else:
            print(f"  Final cycle {cycles}: coord desc t={t_alloc:.0f}s")
            f_opt, _ = coord_descent_peak(current, time_limit=t_alloc)
            try_sequence(f_opt, best_tracker)

        current = best_tracker[1] if best_tracker[1] is not None else current

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using fallback")
    fallback = np.ones(768) / 768
    return [float(x) for x in fallback]
