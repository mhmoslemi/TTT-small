def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lsq_objective(conv, T):
    """Log-sum-exp approximation of max."""
    conv_shifted = conv - conv.max()
    return conv.max() + np.log(np.sum(np.exp(T * conv_shifted))) / T


def lsq_gradient(a, conv, T, n):
    """Gradient of the log-sum-exp objective with respect to a."""
    Tconv = T * (conv - conv.max())
    w = np.exp(Tconv)
    w /= w.sum()
    
    grad = np.zeros(n)
    # For each lag k with weight w[k], the contribution to grad[j] is
    # w[k] * (a[j+k] + a[j-k]) (with bounds)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        # r[k] = sum_j a[j] * a[j+k] for valid j
        # d r[k] / d a[j] = a[j+k] (if j+k < n) + a[j-k] (if j-k >= 0 and k > 0)
        # For k=0: d r[0]/d a[j] = 2*a[j]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            contrib = 0.0
            if k == 0:
                contrib = 2.0 * a[j]
            else:
                if j + k < n:
                    contrib += a[j + k]
                if j - k >= 0:
                    contrib += a[j - k]
            grad[j] += w[k] * contrib
    return grad

```python
def logsumexp_gradient_descent(f0, time_limit, T_init=100, T_final=2000, lr_init=0.01, momentum=0.9):
    """Log-sum-exp projected gradient descent with annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)  # momentum
    T = T_init
    lr = lr_init
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    fft_size = 1 << (3 * n).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gradient
        # w = softmax(T * (conv - conv.max()))
        Tconv = T * (conv - conv.max())
        w = np.exp(Tconv)
        w /= w.sum()
        
        # grad = 2 * ifft(fft(w) * conj(fft(f)))[:n]
        W = np.fft.rfft(w, fft_size)
        F = np.fft.rfft(f, fft_size)
        grad = 2.0 * np.fft.irfft(W * np.conj(F), fft_size)[:n]
        
        # Update with momentum
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        f = f_new
        
        # Anneal
        T *= 1.002
        lr *= 0.9995
        
        if no_improve > 200:
            # Restart with noise
            rng = np.random.default_rng(it * 137 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            T = T_init * 2
            lr = lr_init
            no_improve = 0
        
        if it % 1000 == 0:
            print(f"  LSGD iter {it}: obj={cur_obj:.10f}, best={best_obj:.10f}, T={T:.1f}, lr={lr:.6f}")
    
    return best_f, best_obj
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-scale log-sum-exp projected
gradient descent with FFT-based gradients, adaptive annealing, Nesterov momentum,
LP-Frank-Wolfe initialization, coordinate descent polishing, and multi-start
diverse template exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def multi_scale_gradient(f, conv, T_values, n, fft_size):
    """Compute multi-scale gradient: weighted sum of LSE gradients at different T."""
    F = np.fft.rfft(f, fft_size)
    total_grad = np.zeros(n)
    total_weight = 0.0
    for T in T_values:
        Tconv = T * (conv - conv.max())
        w = np.exp(Tconv)
        w_sum = w.sum()
        if w_sum < 1e-30:
            continue
        w = w / w_sum
        W = np.fft.rfft(w, fft_size)
        grad = 2.0 * np.fft.irfft(W * np.conj(F), fft_size)[:n]
        # Weight by effective number of active peaks (more peaks = more important)
        ep = 1.0 / (np.sum(w * w) * len(w) + 1e-30)
        total_grad += ep * grad
        total_weight += ep
    if total_weight > 1e-30:
        total_grad /= total_weight
    return total_grad


def lsgd_optimize(f0, time_limit, T_init=50.0, T_max=5000.0, lr_init=0.005,
                  momentum_init=0.9, verbose=False):
    """Multi-scale log-sum-exp projected gradient descent with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    fft_size = 1 << (3 * n).bit_length()

    best_f = f.copy()
    best_conv = fast_convolve(f)
    best_obj = best_conv.max()

    v = np.zeros(n)
    m = momentum_init
    lr = lr_init
    T = T_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    restart_count = 0

    # Adaptive T values for multi-scale
    def get_T_values(base_T):
        return [base_T * r for r in [0.1, 0.5, 1.0, 5.0, 20.0]]

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            best_conv = conv.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 5000 == 0:
            print(f"    LSGD iter {it}: obj={cur_obj:.10f}, best={best_obj:.10f}, T={T:.1f}")

        # Compute multi-scale gradient
        T_vals = get_T_values(T)
        grad = multi_scale_gradient(f, conv, T_vals, n, fft_size)

        # Nesterov momentum update
        v_new = m * v - lr * grad
        # Nesterov: evaluate gradient at extrapolated point
        f_look = f + m * v_new
        f_look = np.maximum(f_look, 0.0)
        s_look = f_look.sum()
        if s_look > 1e-15:
            f_look = f_look / s_look
            conv_look = fast_convolve(f_look)
            T_vals_look = get_T_values(T)
            grad_look = multi_scale_gradient(f_look, conv_look, T_vals_look, n, fft_size)
            v_new = m * v - lr * grad_look

        f = project_simplex(f + v_new)
        v = v_new

        # Annealing
        T = min(T * 1.003, T_max)
        lr *= 0.9997
        m = min(m * 1.00005, 0.98)

        # Adaptive restart
        if no_improve > 500:
            restart_count += 1
            scale = 0.002 * min(1.0 + 0.3 * restart_count, 3.0)
            rng = np.random.default_rng(it * 7919 + restart_count * 31)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            T = max(T_init, T * 0.5)
            lr = lr_init * 0.8
            m = momentum_init
            no_improve = 0
            if restart_count > 15:
                break

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, top_k=30, verbose=False):
    """LP-based Frank-Wolfe with adaptive constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Select active constraints (top-k peaks)
        if it > 10 and top_k < 2 * n - 1:
            k_sorted = np.argsort(conv)[::-1][:top_k]
            A_ub = np.zeros((top_k, n))
            b_ub = np.full(top_k, M)
            for row, k in enumerate(k_sorted):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    A_ub[row, j] = f[k - j]
        else:
            A_ub = np.zeros((2 * n - 1, n))
            for j in range(n):
                A_ub[j:j + n, j] = f
            b_ub = np.full(2 * n - 1, M)

        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max(0.1, (time_limit - (time.monotonic() - t0)) * 0.3)})
        if not res.success:
            break
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Line search (simple)
        best_alpha = 0.0
        best_line_obj = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            f_trial = project_simplex(f + alpha * (g - f))
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_line_obj:
                best_line_obj = obj_trial
                best_alpha = alpha

        if best_alpha > 0:
            f = project_simplex(f + best_alpha * (g - f))
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            if verbose and it % 20 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def coord_desc(f0, time_limit, n_multipliers=8):
    """Coordinate descent with geometric multipliers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    # Use fewer multipliers for speed
    if n > 1000:
        multipliers = np.geomspace(0.3, 3.0, n_multipliers)
    else:
        multipliers = np.geomspace(0.25, 4.0, n_multipliers)

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            old_val = f[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale

                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < best_j_obj - 1e-15:
                    best_j_obj = obj_trial
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def peak_reduction_step(f, n_steps=5):
    """Targeted peak reduction: reduce coordinates contributing most to the max peak."""
    f = f.copy()
    n = len(f)
    for _ in range(n_steps):
        conv = fast_convolve(f)
        k_star = np.argmax(conv)
        # Pressure on coordinate j: contribution to peak at k_star
        pressure = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            pressure[j] = f[k_star - j]
        # Reduce high-pressure coordinates
        max_p = pressure.max()
        if max_p < 1e-15:
            break
        reduction = pressure / max_p * 0.05 * f
        f_new = f - reduction
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f = f_new / s
    return f


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def diverse_templates(n, seed=42):
    """Generate diverse initial templates."""
    rng = np.random.default_rng(seed)
    templates = []

    # Uniform
    templates.append(np.ones(n) / n)

    # Flat spectrum
    templates.append(flat_spectrum_init(n, seed=seed))

    # Gaussian center
    x = np.arange(n) / n
    g = np.exp(-4.0 * (x - 0.5) ** 2)
    templates.append(g / g.sum())

    # Bimodal
    b = np.exp(-20.0 * (x - 0.2) ** 2) + np.exp(-20.0 * (x - 0.8) ** 2)
    templates.append(b / b.sum())

    # Sine-modulated
    s = 0.5 + 0.4 * np.sin(2 * np.pi * n * x / n)
    s = np.maximum(s, 0.01)
    templates.append(s / s.sum())

    # Cosine taper
    c = 0.5 * (1 + np.cos(2 * np.pi * x))
    c = np.maximum(c, 0.01)
    templates.append(c / c.sum())

    # Random smooth
    t = rng.standard_normal(n)
    # Smooth with a kernel
    kernel = np.ones(10) / 10
    t_smooth = np.convolve(t, kernel, mode='same')
    t_smooth = np.maximum(t_smooth, 0)
    ss = t_smooth.sum()
    if ss > 1e-10:
        templates.append(t_smooth / ss)

    # Weighted random (power law)
    w = rng.uniform(0.1, 1.0, n)
    w = w ** 2
    templates.append(w / w.sum())

    return templates

            noise_scale = 0.001 * (i + 1)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_pert = min(rem * 0.3, 50)
            a_opt, _ = lsgd_optimize(f_pert, time_limit=t_pert, T_init=100.0,
                                     T_max=5000.0, lr_init=0.002, momentum_init=0.9)
            try_seq(a_opt, f"perturb_{i}")

        # Peak reduction + LSGD on final
        rem = deadline - time.monotonic()
        if rem > 20:
            a_pr = peak_reduction_step(best_seq, n_steps=10)
            try_seq(a_pr, "peak_red_final")
            t_pr = min(rem * 0.4, 60)
            a_opt, _ = lsgd_optimize(a_pr, time_limit=t_pr, T_init=80.0, T_max=5000.0,
                                     lr_init=0.002, momentum_init=0.9)
            try_seq(a_opt, "peak_red_final_lsgd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
