```python
def solve_qp_simplex(p, n, max_iter=100):
    """Solve min a^T M(p) a s.t. a >= 0, sum(a) = 1.
    M(p) is the Hankel matrix with M(p)[j,m] = p[j+m].
    At optimum: M(p)@a = c (constant) on active set, M(p)@a >= c on inactive set.
    """
    active = set(range(n))
    inactive = set()
    c = 0.0
    
    for iteration in range(max_iter):
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:
            a = np.zeros(n)
            a[list(active)[0]] = 1.0
            return a
        
        S = sorted(active)
        k = len(S)
        
        # Build M_SS: M_SS[i,j] = p[S[i] + S[j]]
        M_SS = np.array([[p[S[i] + S[j]] for j in range(k)] for i in range(k)])
        
        # Solve M_SS @ x = c * 1 where c is unknown
        # We need: M_SS @ x = c * ones AND sum(x) = 1
        # => [M_SS, -ones; ones^T, 0] @ [x; c] = [c*ones... wait
        # Actually: M_SS @ x = c * ones, sum(x) = 1
        # => M_SS @ x - c * ones = 0
        # => [M_SS, -ones] @ [x; c] = 0
        # => augment with sum(x) = 1: [M_SS, -ones; ones^T, 0] @ [x; c] = [0; 1]
        
        # This is a (k+1) x (k+1) system
        A = np.zeros((k+1, k+1))
        A[:k, :k] = M_SS
        A[:k, k] = -1.0
        A[k, :k] = 1.0
        b = np.zeros(k+1)
        b[k] = 1.0
        
        try:
            sol = np.linalg.solve(A, b)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(A, b, rcond=None)[0]
        
        x = sol[:k]
        c = sol[k]
        
        # Check KKT conditions
        # For active: x >= 0 (should be satisfied by construction)
        # For inactive: (M(p)@a_full)[j] >= c (i.e., the "reduced cost" is non-negative)
        
        # Compute full M(p)@a
        a_full = np.zeros(n)
        for i, idx in enumerate(S):
            a_full[idx] = x[i]
        
        Ma = hankel_mv(p, a_full)
        
        # Check for negative x (active set violation)
        min_x_idx = -1
        min_x_val = float('inf')
        for i in range(k):
            if x[i] < min_x_val:
                min_x_val = x[i]
                min_x_idx = i
        
        if min_x_val < -1e-10:
            # Remove the most negative from active
            active.remove(S[min_x_idx])
            inactive.add(S[min_x_idx])
            continue
        
        # Check if any inactive should be added
        # For inactive j: Ma[j] - c should be >= 0 (if not, add j to active)
        worst_inactive = -1
        worst_val = float('inf')
        for j in inactive:
            reduced_cost = Ma[j] - c
            if reduced_cost < worst_val:
                worst_val = reduced_cost
                worst_inactive = j
        
        if worst_inactive >= 0 and worst_val < -1e-10:
            inactive.remove(worst_inactive)
            active.add(worst_inactive)
            continue
        
        # KKT satisfied
        a_full = np.zeros(n)
        for i, idx in enumerate(S):
            a_full[idx] = max(0.0, x[i])
        a_full = a_full / a_full.sum()
        return a_full
    
    # Fallback
    a = np.ones(n) / n
    return a
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Continuation method with L-BFGS-B in log-space (global optimization)
2. Primal-dual game iteration (refinement)
3. Frank-Wolfe with LP oracle (global improvements)
4. Multi-n exploration and final polish
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute (M(p) @ a)[m] = sum_j p[m+j] * a[j]. p length 2n-1, a length n."""
    return np.correlate(p, a, mode='valid')


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def continuation_lbfgsb(a0, time_limit, T_schedule=None):
    """
    Continuation method: start with smooth objective (small T), gradually increase T.
    At each T, use L-BFGS-B to minimize the smoothed objective in log-space.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if T_schedule is None:
        # Start smooth, end sharp
        T_schedule = np.geomspace(0.5, 100.0, 20)
    
    best_a = a.copy()
    best_conv = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    for T in T_schedule:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        
        # Work in log-space: x = log(a) (with a = softmax(x))
        # Initialize x from current a
        a_safe = np.maximum(a, 1e-30)
        x0 = np.log(a_safe) - np.log(a_safe.sum())
        
        # Define objective and gradient in log-space
        def obj_grad(x):
            a_local = softmax(x)
            conv = fast_convolve(a_local)
            
            # Smoothed max
            scaled = T * conv
            scaled -= scaled.max()
            log_sum_exp = scaled.max() + np.log(np.sum(np.exp(scaled)))
            obj = log_sum_exp / T
            
            p = np.exp(scaled)
            p = p / p.sum()
            
            # Gradient w.r.t. a
            grad_a = 2.0 * hankel_mv(p, a_local)
            
            # Gradient w.r.t. x (chain rule through softmax)
            ga = grad_a * a_local
            grad_x = ga - a_local * ga.sum()
            
            return obj, grad_x
        
        # Try L-BFGS-B
        try:
            res = minimize(obj_grad, x0, jac=True, method='L-BFGS-B',
                          options={'maxiter': 200, 'maxfun': 500,
                                  'ftol': 1e-12, 'gtol': 1e-8})
            a_new = softmax(res.x)
        except Exception:
            a_new = a
        
        # Evaluate true objective
        conv_new = fast_convolve(a_new)
        if conv_new.max() < best_conv - 1e-12:
            best_conv = conv_new.max()
            best_a = a_new.copy()
            a = a_new
        
        print(f"    T={T:.2f}, smoothed_obj={obj_grad(x0)[0]:.8f}, true_max={best_conv:.8f}")
    
    return best_a, best_conv


def game_iterate_v2(a0, time_limit, T_init=0.01):
    """
    Improved game iterate: alternate between primal (QP via gradient steps)
    and dual (softmax) updates, with adaptive temperature.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Dual update: p = softmax(conv / T)
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Primal update: minimize a^T M(p) a via projected gradient
        # Use multiple steps with decreasing step size
        for step in range(20):
            grad = 2.0 * hankel_mv(p, a)
            lr = 0.1 / (step + 1)
            a = project_simplex(a - lr * grad)
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt temperature
        if no_improve > 30:
            T *= 0.9  # Decrease T (sharper)
        if no_improve > 100:
            T *= 1.1  # Increase T (smoother) to escape
        T = max(0.001, min(T, 1.0))
        
        if no_improve > 300:
            rng = np.random.default_rng((it * 7919 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            no_improve = 0
    
    return best_a, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # LP oracle: max sum(g) s.t. (f*g)[k] <= M*m_exp for all k, g >= 0
        g = solve_lp_oracle(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.3
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 4.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

def solve_lp_oracle(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None
```

```python
def peak_equalization_step(a, top_k=5):
    """Find a direction that equalizes the top autocorrelation peaks."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Get the top peaks
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]
    
    # For each peak k, the gradient w.r.t. a is 2*a[k-j] (valid j)
    # We want to find d (with sum(d)=0, ||d||=1) that minimizes:
    # max over peaks k of sum_j d[j] * 2*a[k-j] (first-order approximation of the change in peak k)
    # This is a minmax problem in n variables with 2 constraints.
    # For a first-order approach, we can use the subgradient of the max:
    # subgrad = weighted average of gradients at the active peaks
    # (weights proportional to how close each peak is to the max)
    
    peak_vals = conv[peak_indices]
    max_val = peak_vals.max()
    # Weights: exponential decay from the max
    weights = np.exp(-(max_val - peak_vals) / 0.001)  # Sharp: only the top ones matter
    weights = weights / weights.sum()
    
    # Combined gradient
    grad = np.zeros(n)
    for i, k in enumerate(peak_indices):
        if weights[i] < 1e-10:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += weights[i] * 2.0 * a[k - j]
    
    # Project out the mean (to preserve sum)
    grad = grad - grad.mean()
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a, 0.0
    
    d = -grad / grad_norm
    
    # Line search
    best_t = 0.0
    best_val = max_val
    for t in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
        a_new = project_simplex(a + t * d * grad_norm)  # Scale by grad_norm for better step
        val = fast_convolve(a_new).max()
        if val < best_val:
            best_val = val
            best_t = t
    
    if best_t > 0:
        return project_simplex(a + best_t * d * grad_norm), best_val
    return a, best_val
```
