"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Dual-guided Frank-Wolfe with LP oracle
- Fast peak-peeling local search (no LP needed)
- Multigrid coarsening/de-coarsening
- Multi-n exploration with structured restarts
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP oracle.
    F[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    Size: (2n-1) x n
    """
    F = np.zeros((2 * n - 1, n))
    for j in range(n):
        # F[k, j] = f[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        k_start = j
        k_end = j + n  # exclusive
        F[k_start:k_end, j] = f
    return F


def solve_lp_oracle(f, M, F=None):
    """Solve LP: max sum(g) s.t. F @ g <= M, g >= 0.
    Returns (g, dual_vars) or (None, None) on failure.
    """
    n = len(f)
    if F is None:
        F = build_lp_matrix(f, n)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=F, b_ub=np.full(2 * n - 1, M), bounds=bounds,
                      method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            g = np.maximum(res.x, 0.0)
            duals = res.ineqlin.marginals if hasattr(res, 'ineqlin') and res.ineqlin is not None else None
            return g, duals
    except Exception:
        pass
    return None, None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """
    Frank-Wolfe with LP oracle and dual-guided M adaptation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    # Pre-build LP matrix (it changes when f changes, so we rebuild each time)
    # Actually F depends on f, so we can't pre-build. But the structure is the same.
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_margin = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Build LP matrix for current f
        F = build_lp_matrix(f, n)
        
        # Solve LP with current M (with margin)
        g, duals = solve_lp_oracle(f, M * m_margin, F)
        
        if g is None:
            m_margin *= 1.1
            if m_margin > 5.0:
                m_margin = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_margin = min(m_margin * 1.02, 3.0)
            continue

        # Line search (golden section)
        best_alpha = 0.0
        best_val = M
        lo, hi = 0.0, 1.0
        gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
        
        for _ in range(15):  # 15 iterations of golden section
            a1 = hi - gr * (hi - lo)
            a2 = lo + gr * (hi - lo)
            
            f1 = np.maximum((1 - a1) * f + a1 * g, 0.0)
            f1 = f1 / f1.sum()
            v1 = fast_convolve(f1).max()
            
            f2 = np.maximum((1 - a2) * f + a2 * g, 0.0)
            f2 = f2 / f2.sum()
            v2 = fast_convolve(f2).max()
            
            if v1 < v2:
                hi = a2
                best_val = min(best_val, v1)
                best_alpha = a1
            else:
                lo = a1
                best_val = min(best_val, v2)
                best_alpha = a2
        
        if best_alpha > 1e-10:
            f = np.maximum((1 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_margin = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_margin = min(m_margin * 1.05, 5.0)

        if verbose and it % 100 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, M={M:.10f}, margin={m_margin:.2f}")

        # Perturb if stuck
        if no_improve > 50:
            rng = np.random.default_rng(it * 6151 + 7)
            scale = 0.002 + 0.005 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_margin = 1.0
            no_improve = 0

    return best_f, best_obj


def peak_peel(a0, time_limit, verbose=False):
    """
    Fast local search that directly targets the top peaks of the convolution.
    No LP needed - uses greedy mass redistribution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Get top peaks
        k_peak = int(np.argmax(conv))

        # Determine valid j range for this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        if js > je:
            break

        # Compute the contribution of each pair
        j_arr = np.arange(js, je + 1)
        partner = k_peak - j_arr
        products = a[j_arr] * a[partner]
        
        # Find the dominant pair
        idx_max = np.argmax(products)
        j1 = int(j_arr[idx_max])
        j2 = int(k_peak - j1)

        # Try to reduce this peak by moving mass
        # Option 1: Move mass from j1 to some j' where a[k_peak - j'] is small
        # Option 2: Move mass from j2 to some j' where a[k_peak - j'] is small
        
        # For option 1: effect on conv[k_peak] ≈ -a[j2] * delta + a[k_peak - j'] * delta
        # We want a[k_peak - j'] < a[j2]
        # For option 2: effect on conv[k_peak] ≈ -a[j1] * delta + a[k_peak - j'] * delta
        # We want a[k_peak - j'] < a[j1]

        # Find best j' for option 1
        # a[k_peak - j'] must be small, and j' must be valid (0 <= j' < n)
        # Also, moving to j' shouldn't create a new peak elsewhere
        
        # Simple heuristic: try moving to the position with smallest a value
        # that is far from the peak
        candidates = np.arange(n)
        # Score: we want small a[k_peak - j'] (to reduce this peak)
        # and large a[j'] (so we're not concentrating too much)
        # Actually, we want to minimize the maximum increase in other conv[k]
        
        # Simplified: move to the position with smallest a[k_peak - j']
        # j' must satisfy 0 <= k_peak - j' < n, i.e., k_peak - n + 1 <= j' <= k_peak
        valid_jp = np.arange(max(0, k_peak - n + 1), min(n, k_peak + 1))
        if len(valid_jp) == 0:
            break
        
        partner_vals = a[k_peak - valid_jp]
        # We want to move TO a position j' where partner_vals is small
        # But we also don't want to increase a[j'] too much
        scores = partner_vals + 0.1 * a[valid_jp]  # small weight on current value
        best_jp_idx = np.argmin(scores)
        jp = int(valid_jp[best_jp_idx])

        # Decide: move from j1 or from j2?
        # Moving from j1 reduces conv[k_peak] by a[j2]*delta
        # Moving from j2 reduces conv[k_peak] by a[j1]*delta
        # Choose the larger reduction
        if a[j1] >= a[j2]:
            src = j2  # move from j2 (smaller value, larger effect on peak)
        else:
            src = j1

        # Step size
        delta = min(a[src], a[src] * 0.05)  # move at most 5% of source mass
        
        # Try the move
        a_try = a.copy()
        a_try[src] -= delta
        a_try[jp] += delta
        a_try = project_simplex(a_try)
        
        conv_try = fast_convolve(a_try)
        obj_try = conv_try.max()
        
        if obj_try < best_obj - 1e-15:
            a = a_try
            best_obj = obj_try
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic random restart
        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(a)
            no_improve = 0

        # Check top few peaks and try to reduce them too
        if it % 10 == 0:
            top_k = np.argpartition(conv, -5)[-5:]
            for k2 in top_k:
                if k2 == k_peak:
                    continue
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                if js2 > je2:
                    continue
                j_arr2 = np.arange(js2, je2 + 1)
                partner2 = int(k2) - j_arr2
                products2 = a[j_arr2] * a[partner2]
                idx_max2 = np.argmax(products2)
                j1_2 = int(j_arr2[idx_max2])
                j2_2 = int(k2 - j1_2)
                
                valid_jp2 = np.arange(max(0, int(k2) - n + 1), min(n, int(k2) + 1))
                if len(valid_jp2) == 0:
                    continue
                partner_vals2 = a[int(k2) - valid_jp2]
                scores2 = partner_vals2 + 0.1 * a[valid_jp2]
                best_jp_idx2 = np.argmin(scores2)
                jp2 = int(valid_jp2[best_jp_idx2])
                
                if a[j1_2] >= a[j2_2]:
                    src2 = j2_2
                else:
                    src2 = j1_2
                
                delta2 = min(a[src2], a[src2] * 0.03)
                a_try2 = a.copy()
                a_try2[src2] -= delta2
                a_try2[jp2] += delta2
                a_try2 = project_simplex(a_try2)
                conv_try2 = fast_convolve(a_try2)
                if conv_try2.max() < best_obj - 1e-15:
                    a = a_try2
                    best_obj = conv_try2.max()
                    best_a = a.copy()
                    no_improve = 0
                    break

    return best_a, best_obj


def multigrid_refine(a0, time_limit, verbose=False):
    """
    Multigrid: coarsen, optimize the coarse sequence, de-coarsen.
    This can escape local optima that are visible at the fine scale
    but not at the coarse scale.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()

    # Coarsen by factor 2
    n_coarse = n // 2
    if n_coarse < 16:
        return best_a, best_obj

    # Coarsen: average pairs
    a_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        a_coarse[i] = (a[2*i] + a[2*i+1]) / 2.0
    a_coarse = a_coarse / a_coarse.sum()

    # Optimize the coarse sequence (FW-LP for a short time)
    t_coarse = min(time_limit * 0.5, 60.0)
    a_coarse_opt, _ = fw_lp_optimize(a_coarse, time_limit=t_coarse)
    
    # De-coarsen: interpolate back
    a_de = np.zeros(n)
    a_de[0::2] = a_coarse_opt
    a_de[1::2] = a_coarse_opt  # same value for both (piecewise constant)
    # Actually, better: linear interpolation
    x_fine = np.linspace(0, 1, n)
    x_coarse = np.linspace(0, 1, n_coarse)
    a_de = np.interp(x_fine, x_coarse, a_coarse_opt)
    a_de = np.maximum(a_de, 0)
    a_de = a_de / a_de.sum()

    obj_de = fast_convolve(a_de).max()
    if obj_de < best_obj:
        best_a = a_de.copy()
        best_obj = obj_de

    # Now refine: blend with original
    remaining = time_limit - (time.monotonic() - t0) - 5
    if remaining > 5:
        for alpha in [0.3, 0.5, 0.7]:
            blend = (1 - alpha) * a0 + alpha * best_a
            blend = np.maximum(blend, 0)
            blend = blend / blend.sum()
            obj_blend = fast_convolve(blend).max()
            if obj_blend < best_obj:
                best_a = blend.copy()
                best_obj = obj_blend

        # Quick peak-peel on the best
        if time.monotonic() - t0 < time_limit - 5:
            best_a, best_obj = peak_peel(best_a, time_limit=max(0, time_limit - (time.monotonic() - t0)))

    return best_a, best_obj


def smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.03):
    """Smoothed gradient descent with adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: 2 * correlate(p, a) = 2 * hankel_mv(p, a)
        # hankel_mv(p, a)[j] = sum_{m} p[j+m] * a[m]
        grad = np.correlate(p, a, mode='valid')
        grad = 2.0 * grad

        a_new = project_simplex(a - lr * grad)
        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-15:
            a = a_new
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Backtrack
            lr *= 0.9

        if no_improve > 30:
            T *= 1.1
            lr = lr_init * 0.5
        if no_improve > 100:
            rng = np.random.default_rng((it * 41 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj
