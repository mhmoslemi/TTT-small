delta_c[k] = d_i * (2*a[k-i] - 2*a[k-j]) (with appropriate index validity)

And the new max is max_k (conv[k] + delta_c[k]).

To minimize the new max, we want to find t that minimizes max_k (conv[k] + d_i * (2*a[k-i] - 2*a[k-j])).

This is a 1D minimization problem (over t), which can be solved by ternary search or by finding the optimal t analytically.

Actually, for a given pair (i,j), the function t → max_k (conv[k] + (t-s)*c * (2*a[k-i] - 2*a[k-j])) is the maximum of linear functions of t, so it's convex in t. The minimum can be found by:
1. Computing the slopes: slope_k = c * (2*a[k-i] - 2*a[k-j]) (with validity)
2. The optimal t is where the maximum of the lines is minimized, which occurs at the "water level" where the top lines cross.

This is essentially a 1D LP, which is very fast to solve.

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):

    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)

    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    k = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k)[-k:]

    w = np.exp(scaled)

    if wsum < 1e-30:

"""
Focused Adam + exact coordinate descent for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses adaptive log-sum-exp smoothing, precise 1D pair optimization via ternary search,
and multi-start diversification to escape local minima.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def max_conv(f):
    """Returns max of auto-convolution of f (assumes sum(f)=1)."""
    return fast_convolve(f).max()


def smooth_grad(f, T, top_k):
    """Smoothed gradient of max(conv) via log-sum-exp weighting over top-k peaks."""
    n = len(f)
    conv = fast_convolve(f)
    k = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k)[-k:]
    pv = conv[peaks]
    scaled = -(pv.max() - pv) / max(T, 1e-15)
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(k)
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws
    grad = np.zeros(n)
    for i in range(k):
        if w[i] < 1e-15:
            continue
        kk = int(peaks[i])
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        if js > je:
            continue
        idxs = np.arange(js, je + 1)
        grad[idxs] += 2.0 * w[i] * f[kk - idxs]
    return grad


def exact_subgrad(f):
    """Exact subgradient at the argmax of convolution."""
    n = len(f)
    conv = fast_convolve(f)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    idxs = np.arange(js, je + 1)
    grad[idxs] = 2.0 * f[k_star - idxs]
    return grad


def adam_run(f0, time_limit, T_init, T_final, k_init, k_final, lr_init, lr_final, use_exact_tail=False):
    """Adam optimizer with adaptive temperature, top-k, and learning rate schedule."""
    f = f0.copy()
    s = f.sum()
    f = f / s
    n = len(f)
    best_f = f.copy()
    best_m = max_conv(f)
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    it = 0
    no_imp = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        p = min(1.0, elapsed / time_limit)
        T = T_init * (T_final / T_init) ** p
        k = max(2, int(k_init + (k_final - k_init) * p))
        lr = lr_init * (lr_final / lr_init) ** p

        if use_exact_tail and p > 0.85:
            g = exact_subgrad(f)
        else:
            g = smooth_grad(f, T, k)

        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        f_new = project_simplex(f - lr * mh / (np.sqrt(vh) + eps))
        if np.max(np.abs(f_new - f)) < 1e-15:
            no_imp += 1
            if no_imp > 400:
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                m = np.zeros(n)
                v = np.zeros(n)
                no_imp = 0
                continue
            continue
        f = f_new
        nm = max_conv(f)
        if nm < best_m - 1e-14:
            best_m = nm
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
    return best_f


def coord_descent_run(f0, time_limit, verbose=False):
    """Exact coordinate descent: optimize random pairs via 1D ternary search."""
    f = f0.copy()
    s = f.sum()
    f = f / s
    n = len(f)
    best_f = f.copy()
    best_m = max_conv(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    no_imp = 0
    it = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        nz_mask = f > 1e-8
        nz_idx = np.nonzero(nz_mask)[0]
        if len(nz_idx) < 2:
            break
        i = int(nz_idx[rng.integers(len(nz_idx))])
        j = int(nz_idx[rng.integers(len(nz_idx))])
        if i == j:
            continue
        c = f[i] + f[j]
        if c < 1e-12:
            continue
        s_i = f[i] / c

        conv = fast_convolve(f)

        # Build pattern: delta_conv[k] = d * pattern[k]
        pattern = np.zeros(2 * n - 1)
        lo_i = i
        hi_i = min(2 * n - 2, i + n - 1)
        pattern[lo_i:hi_i + 1] += 2.0 * f[0:hi_i - lo_i + 1]
        lo_j = j
        hi_j = min(2 * n - 2, j + n - 1)
        pattern[lo_j:hi_j + 1] -= 2.0 * f[0:hi_j - lo_j + 1]

        # Minimize max(conv + d*pattern) for d in [-f[i], f[j]]
        d_lo = -f[i]
        d_hi = f[j]

        lo, hi = d_lo, d_hi
        for _ in range(25):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            v1 = (conv + m1 * pattern).max()
            v2 = (conv + m2 * pattern).max()
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        d_opt = (lo + hi) / 2.0
        new_m = (conv + d_opt * pattern).max()

        if new_m < best_m - 1e-15:
            f[i] += d_opt
            f[j] -= d_opt
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            best_m = new_m
            best_f = f.copy()
            no_imp = 0
            total_improvements += 1
        else:
            no_imp += 1

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD iter {it}: max_conv={best_m:.12f}, improvements={total_improvements}, t={elapsed:.0f}s")

        if no_imp > 20000:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            no_imp = 0

    return best_f


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                try_it(r1, f"ml_adam_{target_n}")
            rem = deadline - time.monotonic()
            if rem > 15 and t_c > 10:
                r2 = coord_descent_run(r1 if 'r1' in dir() else f_interp, min(rem * 0.5, t_c))
                try_it(r2, f"ml_cd_{target_n}")

    # === Phase 7: Final intensive polish (remaining time) ===
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\n=== Phase 7: Final polish (time={rem:.0f}s) ===")
        t_cd = min(rem * 0.5, 100)
        if t_cd > 15:
            result = coord_descent_run(best_seq, t_cd)
            try_it(result, "final_cd")
        rem = deadline - time.monotonic()
        if rem > 15:
            t_ad = min(rem * 0.5, 80)
            result = adam_run(best_seq, t_ad, 0.3 / len(best_seq), 0.02 / len(best_seq), 5, 2, 0.0005, 0.0001, use_exact_tail=True)
            try_it(result, "final_adam")
        rem = deadline - time.monotonic()
        if rem > 10:
            result = coord_descent_run(best_seq, min(rem * 0.7, 50))
            try_it(result, "final_cd2")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n  FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
