def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_param_gradient(b, T=1.0):
    """
    Given b (unconstrained), a = softmax(b) (i.e., a[i] = exp(b[i]-max)/sum).
    Compute gradient of smoothed objective with respect to b.
    Returns: (a, grad_b)
    """
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    a = exp_b / exp_b.sum()
    return a


def smoothed_obj_and_grad(a, T):
    """
    Compute smoothed objective and its gradient with respect to a.
    Smoothed obj: -T * log(sum(exp(c[k]/T))) ≈ max(c)
    Gradient: sum_k p[k] * grad c[k] where p = softmax(c/T)
    """
    c = fast_convolve(a)
    c_shifted = c / T - (c / T).max()
    p = np.exp(c_shifted)
    p /= p.sum()
    # Smoothed objective (up to constant): T * log(sum(exp(c/T)))
    # = T * log(sum(exp(c/T - c.max()/T))) + c.max()
    # For our purposes, we just need the gradient direction
    grad = conv_gradient(a, p)
    return c.max(), grad


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def smooth_gradient_descent(a0, time_limit, T_init=0.001, lr_init=0.03,
                             top_k_peaks=10, use_momentum=True):
    """
    Primary optimization: smoothed gradient descent with adaptive temperature,
    multiple peak weighting, and momentum.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    v = np.zeros(n) if use_momentum else None
    mom = 0.9 if use_momentum else 0.0

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_convolve(a)

        # Multi-peak softmax weights
        if top_k_peaks >= len(c):
            peaks = np.arange(len(c))
        else:
            peaks = np.argpartition(c, -top_k_peaks)[-top_k_peaks:]

        pv = c[peaks]
        pv_shifted = pv / T - (pv / T).max()
        pw = np.exp(pv_shifted)
        pw /= pw.sum()

        p = np.zeros(len(c))
        p[peaks] = pw

        # Gradient
        grad = conv_gradient(a, p)

        # Update with momentum
        if use_momentum:
            v = mom * v - lr * grad
            candidate = project_simplex(a + v)
        else:
            candidate = project_simplex(a - lr * grad)

        # Backtracking if objective increases too much
        c_new = fast_convolve(candidate)
        val = c_new.max()

        if val > best_obj * 1.5:
            # Too much increase, shrink step
            if use_momentum:
                v *= 0.5
                candidate = project_simplex(a + v)
            else:
                candidate = project_simplex(a - lr * 0.1 * grad)
            c_new = fast_convolve(candidate)
            val = c_new.max()

        a = candidate
        obj = val

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            stall_count = 0
        else:
            no_improve += 1
            stall_count += 1

        # Adaptive hyperparameters
        if no_improve > 15:
            T *= 1.03
            lr *= 0.995
        if no_improve > 50:
            T *= 1.1
            lr *= 0.95
        if no_improve > 100:
            T = T_init * (1.0 + 0.1 * np.sin(it * 0.01))  # Oscillate temperature
            lr = lr_init * 0.8
        if no_improve > 40:
            top_k_peaks = min(top_k_peaks + 1, 50)  # Consider more peaks

        # Random restart if stuck
        if no_improve > 300:
            rng = np.random.default_rng((it * 7919) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - 300) / 500.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            if use_momentum:
                v = np.zeros(n)
            no_improve = 0
            stall_count = 0
            top_k_peaks = 10

    return best_a, best_obj


def peak_asymmetrization(a0, time_limit, n_peaks=5):
    """
    Directly reduce convolution peaks by making central pairs more asymmetric.
    For peak at k*, pairs (j, k*-j) with j < k*-j should have different values.
    We push mass from the "middle" element toward the "edge" element.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_convolve(a)

        # Find top peaks
        if n_peaks >= len(c):
            peak_indices = np.arange(len(c))
        else:
            peak_indices = np.argpartition(c, -n_peaks)[-n_peaks:]

        # For each peak, compute the asymmetrization direction
        direction = np.zeros(n)
        for k_star in peak_indices:
            k_star = int(k_star)
            # Active pairs: j and k_star-j both in [0, n-1]
            j_min = max(0, k_star - n + 1)
            j_max = min(n - 1, k_star)

            for j in range(j_min, j_max + 1):
                j2 = k_star - j
                if j == j2:
                    # Self-pair: a[j]^2 contributes. Can't asymmetrize alone.
                    continue
                if j > j2:
                    continue  # Handle each pair once
                # Pair (j, j2): product is a[j]*a[j2]
                # To reduce: make them more different
                # Direction: increase the larger, decrease the smaller
                # (this reduces the product while keeping sum constant)
                if a[j] > a[j2]:
                    direction[j] += 1.0  # Increase a[j]
                    direction[j2] -= 1.0  # Decrease a[j2]
                elif a[j2] > a[j]:
                    direction[j2] += 1.0
                    direction[j] -= 1.0

        # Normalize direction
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break

        direction /= dir_norm

        # Line search
        best_alpha = 0.0
        best_val = best_obj
        for alpha in [0.001, 0.005, 0.01, 0.02, 0.05, 0.01, 0.005]:
            candidate = project_simplex(a + alpha * direction)
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = project_simplex(a + best_alpha * direction)
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()

            # Refine with smaller steps
            for scale in [0.3, 0.1]:
                candidate = project_simplex(a + best_alpha * scale * direction)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-12:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
        else:
            # No improvement from asymmetrization, try perturbations
            if it % 10 == 0:
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-12:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()

        # Break if no progress in a while
        if it > 200 and not (best_obj < fast_convolve(a0).max() - 1e-8):
            break

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle for global moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp *= 1.3
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.1, 4.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.2, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_peak_polish(a0, time_limit):
    """
    Fine coordinate descent: for each top peak, try small adjustments
    to individually reduce it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_convolve(a)
        k_peak = int(np.argmax(c))

        # Variables affecting this peak
        j_min = max(0, k_peak - n + 1)
        j_max = min(n - 1, k_peak)

        # Try adjusting each affected variable
        improved = False
        for j in range(j_min, j_max + 1):
            if a[j] < 1e-12:
                continue
            # Try reducing a[j] and redistributing to a random other variable
            for _ in range(3):
                rng = np.random.default_rng((it * 1000 + j * 7) % (2**31))
                # Move small amount from j to a random position
                target = int(rng.integers(0, n))
                if target == j:
                    continue
                delta = a[j] * rng.uniform(0.01, 0.1)
                candidate = a.copy()
                candidate[j] -= delta
                candidate[target] += delta
                candidate = project_simplex(candidate)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-13:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0

    return best_a, best_obj


def multi_n_search(parent, time_budget, rng):
    """Try a few different sequence lengths to see if a different n helps."""
    best_a = None
    best_score = float('inf')

    n_parent = len(parent)
    n_candidates = [n_parent]
    for n_try in [n_parent - 128, n_parent + 128, 512, 1024]:
        if n_try >= 128 and n_try <= 2048 and n_try != n_parent:
            n_candidates.append(n_try)

    time_per_n = time_budget / len(n_candidates)

    for n_try in n_candidates:
        if time.monotonic() > time_budget:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        # Quick optimization
        a_opt, _ = smooth_gradient_descent(start, time_limit=time_per_n * 0.7,
                                            T_init=0.002, lr_init=0.02)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  n={n_try}: score={score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search optimized for the
    2*n*max(conv)/(sum)^2 objective.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - start_time
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_default")

    n_best = len(parent)

    # === Phase 1: Quick multi-n exploration (150s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t1 = time.monotonic()

    # Try parent length with quick optimization
    a_opt, _ = smooth_gradient_descent(parent, time_limit=40,
                                        T_init=0.001, lr_init=0.03, top_k_peaks=15)
    try_seq(a_opt, "quick_sg_parent")

    # Try peak asymmetrization
    remaining = 150 - (time.monotonic() - t1)
    if remaining > 20 and best_seq is not None:
        a_opt, _ = peak_asymmetrization(best_seq, time_limit=min(remaining, 50))
        try_seq(a_opt, "asym_phase1")

    # Try different n values
    remaining = 150 - (time.monotonic() - t1)
    if remaining > 30:
        a_opt, _ = multi_n_search(best_seq, time_limit=min(remaining, 60), rng=rng)
        if a_opt is not None:
            try_seq(a_opt, "multi_n")
            if len(a_opt) != n_best:
                # Resize back to original n for continued optimization
                best_seq = np.interp(np.linspace(0, 1, n_best),
                                      np.linspace(0, 1, len(a_opt)), a_opt)
                best_seq = np.maximum(best_seq, 0)
                best_seq = best_seq / best_seq.sum()
                try_seq(best_seq, "multi_n_resized")

    print(f"  Phase 1 done: best={best_score:.8f} (t={time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep smoothed gradient (500s) ===
    print("\n=== Phase 2: Deep Smoothed Gradient ===")
    t2 = time.monotonic()

    if best_seq is not None:
        # Run with different initial temperatures
        configs = [
            {'T_init': 0.0005, 'lr_init': 0.02, 'top_k': 10},
            {'T_init': 0.001, 'lr_init': 0.03, 'top_k': 20},
            {'T_init': 0.002, 'lr_init': 0.02, 'top_k': 30},
            {'T_init': 0.0003, 'lr_init': 0.01, 'top_k': 5},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 700
            if remaining < 30:
                break
            t_alloc = min(remaining, 120)
            print(f"  SG config {ci} (T={cfg['T_init']}, k={cfg['top_k']}), t={t_alloc:.0f}s")

            # Start from current best with small perturbation
            if ci > 0:
                perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            else:
                perturbed = best_seq.copy()

            a_opt, _ = smooth_gradient_descent(perturbed, time_limit=t_alloc,
                                                T_init=cfg['T_init'],
                                                lr_init=cfg['lr_init'],
                                                top_k_peaks=cfg['top_k'])
            try_seq(a_opt, f"sg2_{ci}")

    print(f"  Phase 2 done: best={best_score:.8f} (t={time.monotonic()-t2:.0f}s)")

    # === Phase 3: Peak asymmetrization + FW-LP (400s) ===
    print("\n=== Phase 3: Asymmetrization + FW-LP ===")
    t3 = time.monotonic()

    if best_seq is not None:
        # Peak asymmetrization (longer run)
        remaining = deadline - time.monotonic() - 300
        if remaining > 30:
            t_alloc = min(remaining, 150)
            print(f"  Peak asym, t={t_alloc:.0f}s")
            a_opt, _ = peak_asymmetrization(best_seq, time_limit=t_alloc, n_peaks=8)
            try_seq(a_opt, "asym3")

        # FW-LP
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 120)
            print(f"  FW-LP, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw3")

        # Another round of smoothed gradient from FW result
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 80)
            a_opt, _ = smooth_gradient_descent(best_seq, time_limit=t_alloc,
                                                T_init=0.0005, lr_init=0.015,
                                                top_k_peaks=15)
            try_seq(a_opt, "sg3b")

    print(f"  Phase 3 done: best={best_score:.8f} (t={time.monotonic()-t3:.0f}s)")

    # === Phase 4: Intensive local refinement (400s) ===
    print("\n=== Phase 4: Local Refinement ===")
    t4 = time.monotonic()

    if best_seq is not None:
        # Coordinate peak polish
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Coord polish, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_peak_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord4")

        # Very fine smoothed gradient (low T, small lr)
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Fine SG, t={t_alloc:.0f}s")
            a_opt, _ = smooth_gradient_descent(best_seq, time_limit=t_alloc,
                                                T_init=0.0001, lr_init=0.005,
                                                top_k_peaks=5)
            try_seq(a_opt, "fine4")

        # Alternating: asym + fine SG
        remaining = deadline - time.monotonic() - 50
        if remaining > 30:
            for round_i in range(3):
                remaining = deadline - time.monotonic() - 30
                if remaining < 15:
                    break

                a_opt, _ = peak_asymmetrization(best_seq, time_limit=min(remaining, 40),
                                                 n_peaks=10)
                try_seq(a_opt, f"alt_asym_{round_i}")

                remaining = deadline - time.monotonic() - 20
                if remaining < 10:
                    break
                a_opt, _ = smooth_gradient_descent(best_seq, time_limit=min(remaining, 40),
                                                    T_init=0.0002, lr_init=0.008,
                                                    top_k_peaks=8)
                try_seq(a_opt, f"alt_sg_{round_i}")

    print(f"  Phase 4 done: best={best_score:.8f} (t={time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final polish with micro-steps (200s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        # Ultra-fine gradient with momentum
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 80)
            a_opt, _ = smooth_gradient_descent(best_seq, time_limit=t_alloc,
                                                T_init=0.00005, lr_init=0.003,
                                                top_k_peaks=3)
            try_seq(a_opt, "ultra_fine")

        # Tiny perturbations with quick evaluation
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            n_perturb = int(remaining / 0.5)  # ~0.5s per perturbation
            n_perturb = max(10, min(n_perturb, 500))
            print(f"  Micro-perturbations: {n_perturb} attempts")
            for pi in range(n_perturb):
                if time.monotonic() > deadline - 10:
                    break
                # Geometric noise scale: start larger, get smaller
                scale = 0.005 * (0.995 ** pi)
                if scale < 0.0001:
                    scale = 0.0001
                noise = rng.standard_normal(n_best) * scale
                candidate = project_simplex(best_seq + noise)
                score = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = candidate.copy()
                    print(f"    Micro-{pi}: {score:.8f}")

        # One final coordinate polish
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_opt, _ = coordinate_peak_polish(best_seq, time_limit=min(remaining, 15))
            try_seq(a_opt, "final_coord")

    print(f"  Phase 5 done: best={best_score:.8f} (t={time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, elapsed={elapsed:.0f}s")
    return [float(x) for x in best_seq]
