"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max L-BFGS optimization
with softmax reparameterization, annealed temperature, structured asymmetric
initializations, and local polishing via coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def smooth_max_and_grad(a, alpha):
    """Compute smooth max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    scaled = alpha * b
    scaled -= scaled.max()
    e = np.exp(scaled)
    e_sum = e.sum()
    if e_sum < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = e / e_sum
    smooth_max_val = (1.0 / alpha) * (np.log(e_sum) + b.max())
    
    # Gradient: grad_a[i] = 2 * sum_k a[k-i] * w[k]
    # Vectorized computation
    n_b = len(b)  # 2n-1
    # For each i in [0, n-1], we need sum over k where k-i in [0, n-1]
    # i.e., k in [i, i+n-1], clipped to [0, 2n-2]
    # grad_a[i] = 2 * dot(a[0..n-1], w[i..i+n-1]) (with appropriate clipping)
    
    # Efficient computation using matrix multiplication
    # W[i, j] = w[i+j] if i+j < 2n-1 else 0
    # grad_a = 2 * W @ a
    
    i_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    k_idx = i_idx + j_idx
    valid = k_idx < n_b
    # Clip for safe indexing
    k_clipped = np.minimum(k_idx, n_b - 1)
    w_mat = np.where(valid, w[k_clipped], 0.0)
    grad_a = 2.0 * (w_mat @ a)
    
    return smooth_max_val, grad_a


def objective_and_grad_z(z, alpha, n):
    """Objective and gradient for optimization over z (where a = softmax(z))."""
    a = softmax(z)
    val, grad_a = smooth_max_and_grad(a, alpha)
    # d(val)/dz[j] = a[j] * (grad_a[j] - sum_i grad_a[i] * a[i])
    grad_z = a * (grad_a - np.dot(grad_a, a))
    obj = 2.0 * n * val
    return obj, 2.0 * n * grad_z

        a = np.ones(n) / n
    else:
        a = a / s
    
    # Convert to z parameterization
    z = np.log(a + 1e-30)
    z -= z.mean()  # centering for numerical stability
    
    t0 = time.monotonic()
    best_obj = float('inf')
    best_z = z.copy()
    
    if alpha_end is None:
        alpha_end = alpha_start * 20
    
    # Anneal alpha from alpha_start to alpha_end
    n_anneal = 5
    
    for stage in range(n_anneal):
        if time.monotonic() - t0 > time_limit:
            break
        frac = stage / max(n_anneal - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        stage_time = min(time_limit - (time.monotonic() - t0), 
                        time_limit / n_anneal)
        
        if verbose:
            print(f"    L-BFGS stage {stage}: alpha={alpha:.1f}, t_remaining={time_limit-(time.monotonic()-t0):.1f}s")
        
        result = minimize(
            fun=lambda zz: objective_and_grad_z(zz, alpha, n)[0],
            x0=z,
            jac=lambda zz: objective_and_grad_z(zz, alpha, n)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                     'ftol': 1e-12, 'gtol': 1e-8, 'iprint': -1}
        )
        
        z = result.x
        a_test = softmax(z)
        obj = 2.0 * n * fast_convolve(a_test).max()
        
        if obj < best_obj:
            best_obj = obj
            best_z = z.copy()
        
        if verbose:
            print(f"    Stage {stage} done: obj={obj:.10f}, success={result.success}")
    
    best_a = softmax(best_z)
    return best_a, best_obj


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_full(a, conv, alpha):
    """Compute full softmax-weighted subgradient (all peaks)."""
    n = len(a)
    scaled = alpha * conv
    scaled -= scaled.max()
    e = np.exp(scaled)
    ws = e.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = e / ws
    
    # grad_a[i] = 2 * sum_k a[k-i] * w[k]
    n_b = len(conv)
    i_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    k_idx = i_idx + j_idx
    valid = k_idx < n_b
    k_clipped = np.minimum(k_idx, n_b - 1)
    w_mat = np.where(valid, w[k_clipped], 0.0)
    grad_a = 2.0 * (w_mat @ a)
    return grad_a


def mu_weights_optimize(f0, time_limit, alpha=20.0, lr=0.003,
                        momentum=0.85, verbose=False, label=""):
    """
    Multiplicative-weights subgradient optimization with Nesterov momentum,
    using full softmax weights (all peaks).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_alpha = alpha

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        sg = compute_subgradient_full(f_nest, conv_nest, cur_alpha)

        # Update momentum
        v = momentum * v + (1 - momentum) * sg

        # Multiplicative weights update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_lr *= 0.9995
        if cur_lr < 0.00003:
            cur_lr = 0.00003
        cur_alpha *= 1.003

        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.003 + 0.002 * min(no_improve // 400, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_alpha = alpha * 0.5
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MU iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue

            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def pair_exchange_search(f0, time_limit, max_pairs=500, verbose=False):
    """Pair-exchange local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        top_peaks = np.argpartition(conv, -min(20, 2 * n - 1))[-min(20, 2 * n - 1):]

        trials = min(max_pairs, 300)
        for _ in range(trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            old_i = f[i]
            old_j = f[j]
            if old_i < 1e-12:
                continue
            for frac in [0.2, 0.5, 0.8]:
                amt = old_i * frac
                f[i] = old_i - amt
                f[j] = old_j + amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] = old_j
                    break

        if not improved:
            # Try peak-directed transfers
            for k in top_peaks[:5]:
                if time.monotonic() - t0 > time_limit:
                    break
                k = int(k)
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                contribs = f[idx] * f[k - idx]
                hi = idx[np.argmax(contribs)]
                lo = idx[np.argmin(contribs)]
                if hi == lo or f[hi] < 1e-12:
                    continue
                for frac in [0.1, 0.3]:
                    amt = f[hi] * frac
                    f[hi] -= amt
                    f[lo] += amt
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[hi] += amt
                        f[lo] -= amt
                        break

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=6):
    """Generate structured initial sequences including asymmetric ones."""
    inits = []
    inits.append(np.ones(n) / n)

    # Hann window
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    inits.append(np.maximum(hann, 1e-10))
    inits[-1] /= inits[-1].sum()

    # Asymmetric ramp (increasing)
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())

    # Asymmetric exponential decay
    decay = np.exp(-np.linspace(0, 3, n))
    inits.append(decay / decay.sum())

    # Asymmetric exponential growth
    growth = np.exp(np.linspace(0, 2, n))
    inits.append(growth / growth.sum())

    # Gaussian (asymmetric center)
    center = n * (0.3 + 0.4 * rng.random())
    sigma = n / (3.0 + rng.random() * 3.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Blackman-Harris window
    bh = (0.35875 - 0.48829 * np.cos(2*np.pi*x) + 0.14128 * np.cos(4*np.pi*x) 
          - 0.01168 * np.cos(6*np.pi*x))
    inits.append(np.maximum(bh, 1e-10))
    inits[-1] /= inits[-1].sum()

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    inits.append(np.maximum(r, 1e-10))
    inits[-1] /= inits[-1].sum()

    return inits[:count]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max L-BFGS optimization
with FFT-accelerated gradients, annealed temperature, asymmetric structured
initializations, multiplicative-weights subgradient, and local polishing.
Key: soft-max reparameterization makes the non-smooth max objective smooth,
enabling second-order (L-BFGS) convergence. Gradient computed via FFT-based
cross-correlation in O(n log n).
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import fftconvolve


# --- Global context for L-BFGS callbacks (avoids lambdas/closures) ---
_ANNEAL_ALPHA = 50.0
_ANNEAL_N = 768


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def smooth_max_and_grad(a, alpha):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Uses FFT-based cross-correlation for O(n log n) gradient.
    """
    n = len(a)
    b = fast_convolve(a)
    scaled = alpha * b
    scaled -= scaled.max()
    e = np.exp(scaled)
    e_sum = e.sum()
    if e_sum < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = e / e_sum
    smooth_max_val = (1.0 / alpha) * (np.log(e_sum) + b.max())

    # grad_a[i] = 2 * sum_j a[j] * w[i+j]
    # Computed via FFT: c[i] = conv(a, w[::-1], 'full')[2n-2-i]
    w_flipped = w[::-1]
    conv_result = fftconvolve(a, w_flipped, mode='full')
    c = conv_result[2 * n - 2 - np.arange(n)]
    grad_a = 2.0 * c

    return smooth_max_val, grad_a


def _smooth_obj(zz):
    """Objective for L-BFGS (uses global _ANNEAL_ALPHA, _ANNEAL_N)."""
    a = softmax_fn(zz)
    val, _ = smooth_max_and_grad(a, _ANNEAL_ALPHA)
    return 2.0 * _ANNEAL_N * val


def _smooth_jac(zz):
    """Gradient for L-BFGS (uses global _ANNEAL_ALPHA, _ANNEAL_N)."""
    a = softmax_fn(zz)
    _, grad_a = smooth_max_and_grad(a, _ANNEAL_ALPHA)
    grad_z = a * (grad_a - np.dot(grad_a, a))
    return 2.0 * _ANNEAL_N * grad_z


def lbfgs_smooth_max(a0, time_limit, alpha_start=30.0, alpha_end=300.0,
                     max_iter=150, n_anneal=5, verbose=False, label=""):
    """
    Optimize using L-BFGS-B on softmax-reparameterized problem with annealed
    smooth-max temperature.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = np.log(a + 1e-30)
    z -= z.mean()

    global _ANNEAL_ALPHA, _ANNEAL_N
    _ANNEAL_N = n

    t0 = time.monotonic()
    best_obj = float('inf')
    best_z = z.copy()

    for stage in range(n_anneal):
        if time.monotonic() - t0 > time_limit:
            break
        frac = stage / max(n_anneal - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        _ANNEAL_ALPHA = alpha
        stage_time = min(time_limit - (time.monotonic() - t0), time_limit / n_anneal)

        if verbose:
            print(f"    L-BFGS stage {stage}: alpha={alpha:.1f}, remaining={time_limit-(time.monotonic()-t0):.1f}s")

        try:
            result = minimize(
                fun=_smooth_obj,
                x0=z,
                jac=_smooth_jac,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                         'ftol': 1e-14, 'gtol': 1e-9, 'gtol_x': 1e-9,
                         'iprint': -1, 'maxls': 50}
            )
            z = result.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS stage {stage} failed: {e}")
            continue

        a_test = softmax_fn(z)
        obj = 2.0 * n * fast_convolve(a_test).max()

        if obj < best_obj:
            best_obj = obj
            best_z = z.copy()

        if verbose:
            print(f"    Stage {stage} done: obj={obj:.10f}, success={result.success}")

    best_a = softmax_fn(best_z)
    return best_a, best_obj


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_full(a, conv, alpha):
    """Full softmax-weighted subgradient using all peaks (FFT-accelerated)."""
    n = len(a)
    scaled = alpha * conv
    scaled -= scaled.max()
    e = np.exp(scaled)
    ws = e.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = e / ws
    w_flipped = w[::-1]
    conv_result = fftconvolve(a, w_flipped, mode='full')
    c = conv_result[2 * n - 2 - np.arange(n)]
    return 2.0 * c


def mu_weights_optimize(f0, time_limit, alpha=20.0, lr=0.003,
                        momentum=0.85, verbose=False, label=""):
    """
    Multiplicative-weights subgradient with Nesterov momentum,
    using full softmax weights and FFT-accelerated gradients.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_alpha = alpha

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        sg = compute_subgradient_full(f_nest, conv_nest, cur_alpha)

        # Update momentum
        v = momentum * v + (1 - momentum) * sg

        # Multiplicative weights update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_lr *= 0.9995
        if cur_lr < 0.00003:
            cur_lr = 0.00003
        cur_alpha *= 1.004

        # Restart if stuck
        if no_improve > 150:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.004 + 0.003 * min(no_improve // 300, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_alpha = alpha * 0.5
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MU iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    outer_count = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue

            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        outer_count += 1
        if not improved:
            break

    return best_f, best_obj


def pair_exchange_search(f0, time_limit, max_pairs=300, verbose=False):
    """Pair-exchange local search with peak-directed transfers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        k_peaks = min(20, 2 * n - 1)
        top_peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]

        for _ in range(max_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            old_i = f[i]
            old_j = f[j]
            if old_i < 1e-12:
                continue
            for frac in [0.2, 0.5, 0.8]:
                amt = old_i * frac
                f[i] = old_i - amt
                f[j] = old_j + amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] = old_j
                    break

        # Peak-directed transfers
        for k in top_peaks[:8]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            hi = idx[np.argmax(contribs)]
            lo = idx[np.argmin(contribs)]
            if hi == lo or f[hi] < 1e-12:
                continue
            for frac in [0.1, 0.3, 0.5]:
                amt = f[hi] * frac
                f[hi] -= amt
                f[lo] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[hi] += amt
                    f[lo] -= amt
                    break

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=6):
    """Generate diverse structured initial sequences including asymmetric ones."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    h = np.maximum(hann, 1e-10)
    inits.append(h / h.sum())

    # Asymmetric ramp (increasing)
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())

    # Asymmetric exponential decay
    decay = np.exp(-np.linspace(0, 3.0, n))
    inits.append(decay / decay.sum())

    # Asymmetric exponential growth
    growth = np.exp(np.linspace(0, 2.5, n))
    inits.append(growth / growth.sum())

    # Blackman-Harris window
    bh = (0.35875 - 0.48829 * np.cos(2 * np.pi * x)
          + 0.14128 * np.cos(4 * np.pi * x) - 0.01168 * np.cos(6 * np.pi * x))
    bh = np.maximum(bh, 1e-10)
    inits.append(bh / bh.sum())

    # Asymmetric Gaussian
    center = n * (0.25 + 0.5 * rng.random())
    sigma = n / (3.0 + rng.random() * 4.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    klen = max(3, n // 50)
    kernel = np.ones(klen) / klen
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    return inits[:count]

            {"alpha": 8.0, "lr": 0.006, "mom": 0.80},
        ]
        for p, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 350
            if rem < 15:
                break
            t_alloc = min(rem * 0.25, 100)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.01
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init)

            print(f"  MU {p}: alpha={cfg['alpha']}, lr={cfg['lr']}, t={t_alloc:.0f}s")
            a_opt, _ = mu_weights_optimize(f_init, time_limit=t_alloc,
                                            alpha=cfg["alpha"], lr=cfg["lr"],
                                            momentum=cfg["mom"])
            try_seq(a_opt, f"mu_{p}")

    # === Phase 4: L-BFGS deep refinement ===
    print(f"\n=== Phase 4: L-BFGS Deep Refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(rem * 0.5, 150)
            print(f"  t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_smooth_max(best_seq, time_limit=t_alloc,
                                         alpha_start=50.0, alpha_end=500.0,
                                         max_iter=200, n_anneal=5, verbose=True,
                                         label="deep")
            try_seq(a_opt, "lbfgs_deep")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        base_n = len(best_seq)
        targets = sorted(set([base_n - 128, base_n + 128, base_n + 256, base_n + 384]))
        targets = [t for t in targets if 128 <= t <= 4000]
        for target_n in targets:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 70)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_smooth_max(f_interp, time_limit=t_alloc,
                                         alpha_start=25.0, alpha_end=250.0,
                                         max_iter=100, n_anneal=4, verbose=False,
                                         label=f"len{target_n}")
            try_seq(a_opt, f"len_{target_n}")
            # Quick MU polish on the interpolated result
            rem = deadline - time.monotonic() - 120
            if rem > 15 and a_opt is not None:
                t_mu = min(rem * 0.4, 30)
                a_opt2, _ = mu_weights_optimize(a_opt, time_limit=t_mu,
                                                 alpha=25.0, lr=0.003, momentum=0.9)
                try_seq(a_opt2, f"mu_len_{target_n}")

    # === Phase 6: Local search ===
    print(f"\n=== Phase 6: Local Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_pex = min(rem * 0.4, 50)
            print(f"  Pair exchange t={t_pex:.0f}s")
            a_opt, _ = pair_exchange_search(best_seq, time_limit=t_pex)
            try_seq(a_opt, "pex")

        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_cd = min(rem * 0.6, 50)
            print(f"  Coord descent t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 7: Final fine L-BFGS ===
    print(f"\n=== Phase 7: Final L-BFGS (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_final = min(rem, 35)
            print(f"  t={t_final:.0f}s")
            a_opt, _ = lbfgs_smooth_max(best_seq, time_limit=t_final,
                                         alpha_start=80.0, alpha_end=800.0,
                                         max_iter=80, n_anneal=3, verbose=False,
                                         label="final")
            try_seq(a_opt, "final_lbfgs")

        # One more pair exchange
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = pair_exchange_search(best_seq, time_limit=min(rem, 10))
            try_seq(a_opt, "pex_final")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
