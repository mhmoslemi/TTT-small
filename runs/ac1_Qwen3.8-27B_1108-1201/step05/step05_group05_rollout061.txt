"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient descent with
log-sum-exp approximation, adaptive temperature, softmax reparameterization,
multiple initializations, and palindromic symmetry.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 1 << (2*n - 2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax(g):
    g_shifted = g - g.max()
    e = np.exp(g_shifted)
    return e / e.sum()

def logsumexp_conv(f, T):
    conv = fast_convolve(f)
    scaled = T * conv
    scaled -= scaled.max()
    return (scaled.max() + np.log(np.sum(np.exp(scaled)))) / T

def smooth_grad_f(f, T):
    """Compute gradient of log-sum-exp of conv w.r.t. f."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = T * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Term B: (w * f)[0:n]
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    term_b = np.fft.irfft(W * F, fft_size)[:n]
    
    # Term A: (w * f_rev)[n-1 : 2n-1]
    f_rev = f[::-1]
    Fr = np.fft.rfft(f_rev, fft_size)
    term_a = np.fft.irfft(W * Fr, fft_size)[n-1:2*n-1]
    
    return term_a + term_b

def smooth_grad_g(g, T):
    """Gradient of smooth objective w.r.t. g (softmax parameterization)."""
    f = softmax(g)
    grad_f = smooth_grad_f(f, T)
    # Chain rule for softmax: dF/dg[i] = f[i] * (grad_f[i] - f @ grad_f)
    return f * (grad_f - f @ grad_f)

def smooth_obj_g(g, T):
    f = softmax(g)
    return logsumexp_conv(f, T)

def smooth_optimize(f0, time_limit, T_initial=50, T_final=500, verbose=False):
    """Adam-based smooth optimization in log-space."""
    n = len(f0)
    f0 = np.maximum(f0, 1e-15)
    f0 = f0 / f0.sum()
    g = np.log(f0)
    
    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    t = 0
    
    best_f = f0.copy()
    best_obj = fast_convolve(f0).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_initial * (T_final / T_initial) ** progress
        
        obj = smooth_obj_g(g, T)
        grad = smooth_grad_g(g, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        g -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate true objective
        f_new = softmax(g)
        true_obj = fast_convolve(f_new).max()
        
        if true_obj < best_obj - 1e-14:
            best_obj = true_obj
            best_f = f_new.copy()
            no_improve = 0
            # Decay learning rate slightly
            lr *= 0.9999
        else:
            no_improve += 1
            if no_improve > 200:
                # Restart with perturbation
                g = np.log(np.maximum(best_f + np.random.standard_normal(n) * 0.002, 1e-15))
                m = np.zeros(n)
                v = np.zeros(n)
                lr = 0.01
                t = 0
                no_improve = 0
        
        if verbose and t % 100 == 0:
            print(f"    Smooth iter {t}: true_obj={best_obj:.10f}, T={T:.1f}, lr={lr:.6f}")
    
    return best_f, best_obj

def lp_fw_optimize(f0, time_limit, verbose=False, top_k=30):
    """LP-Frank-Wolfe (simplified from previous)."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Select active constraints
        k_act = min(top_k, 2*n-1)
        k_sorted = np.argsort(conv)[::-1][:k_act]
        
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in k_sorted, g >= 0
        K = len(k_sorted)
        A_ub = np.zeros((K, n))
        for row, k in enumerate(k_sorted):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
        b_ub = np.full(K, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                no_improve += 1
                if no_improve > 50:
                    break
                continue
        except:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search
        best_alpha = 0
        best_alpha_obj = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_try = f + alpha * (g - f)
            f_try = project_simplex(f_try)
            obj_try = fast_convolve(f_try).max()
            if obj_try < best_alpha_obj:
                best_alpha_obj = obj_try
                best_alpha = alpha
        
        if best_alpha > 0:
            f = f + best_alpha * (g - f)
            f = project_simplex(f)
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 60:
            f = project_simplex(best_f + np.random.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj

In the frequency domain: FFT(a ⊛ b) = FFT(a) * conj(FFT(b)).

So (a ⊛ b)[i] = IFFT(FFT(a) * conj(FFT(b)))[i].

With zero-padding, this gives the linear cross-correlation.

I want A[i] = sum_{m=0}^{n-1} w[m+i] * f[m].

Let me set a = w, b = f. Then (a ⊛ b)[i] = sum_j a[j] * b[j-i] = sum_j w[j] * f[j-i].

For j in [0, 2n-2] (valid w) and j-i in [0, n-1] (valid f):
j-i in [0, n-1] → j in [i, i+n-1]
j in [0, 2n-2] (always satisfied for i in [0,n-1] and j in [i, i+n-1] since i+n-1 <= 2n-2)

So (a ⊛ b)[i] = sum_{j=i}^{i+n-1} w[j] * f[j-i] = sum_{m=0}^{n-1} w[m+i] * f[m] = A[i]. ✓!

So the gradient is:
grad_f = 2 * IFFT(rfft(w, size) * conj(rfft(f, size)), size)[:n]

where size = next power of 2 >= (2n-1) + n - 1 = 3n-2.

For n=768: size = 4096 (since 3*768-2 = 2302, next power of 2 is 4096).

This is very efficient! One FFT of w (length 4096), one FFT of f (length 4096), one pointwise multiply, one inverse FFT. Total: 3 FFTs.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient descent with
log-sum-exp approximation, adaptive temperature, softmax reparameterization,
LP-Frank-Wolfe, palindromic symmetry, and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def safe_softmax(g):
    """Numerically stable softmax."""
    g_shifted = g - g.max()
    e = np.exp(g_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(g)) / len(g)
    return e / s


def smooth_obj(f, T):
    """Log-sum-exp approximation of max(conv(f))."""
    conv = fast_convolve(f)
    scaled = T * conv
    scaled -= scaled.max()
    return (scaled.max() + np.log(np.sum(np.exp(scaled)))) / T


def smooth_grad_f(f, T):
    """Gradient of log-sum-exp(conv) w.r.t. f. Returns array of length n."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = T * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # grad_f[i] = 2 * sum_m w[m+i] * f[m] = 2 * (w cross-correlate f)[i]
    size = 1 << (2 * n - 1 + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)
    corr = np.fft.irfft(W * np.conj(F), size)[:n]
    return 2.0 * corr


def smooth_grad_g(g, T):
    """Gradient of smooth objective w.r.t. g (log-space, softmax param)."""
    f = safe_softmax(g)
    grad_f = smooth_grad_f(f, T)
    return f * (grad_f - np.dot(f, grad_f))

def lp_fw_step(f, conv, M, top_k, n):
    """One LP-FW step: solve LP for direction, return new f."""
    k_sorted = np.argsort(conv)[::-1][:min(top_k, 2 * n - 1)]
    K = len(k_sorted)
    A_ub = np.zeros((K, n))
    for row, k in enumerate(k_sorted):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
    b_ub = np.full(K, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 10})
        if not res.success:
            return None, M
        g = res.x
    except Exception:
        return None, M
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, M
    g = g / g_sum
    if np.max(np.abs(g - f)) < 1e-12:
        return None, M
    return g, M

                M = conv.max()
                no_improve = 0

        best_alpha = 0.0

        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

def palindromic_smooth(f0, time_limit, T_init=10.0, T_end=1500.0,
                       lr_init=0.02, verbose=False):
    """Smooth optimization with palindromic constraint."""
    n = len(f0)
    if n % 2 != 0:
        n = n - 1
        f0 = f0[:n]
    half = n // 2
    f0 = np.maximum(f0, 1e-15)
    f0 = f0 / f0.sum()
    # Average with reverse to get palindromic start
    f0_pal = (f0 + f0[::-1]) / 2.0
    f0_pal = f0_pal / f0_pal.sum()
    half_vals = f0_pal[:half]
    g = np.log(np.maximum(half_vals, 1e-15))

    m = np.zeros(half)
    v = np.zeros(half)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    t_step = 0

    best_f = f0_pal.copy()
    best_obj = fast_convolve(f0_pal).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t_step += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_end / T_init) ** progress

        half_sm = safe_softmax(g)
        full = np.concatenate([half_sm, half_sm[::-1]])
        full = full / full.sum()

        conv = fast_convolve(full)
        scaled = T * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Gradient w.r.t. full f
        size = 1 << (2 * n - 1 + n - 1 - 1).bit_length()
        W = np.fft.rfft(w, size)
        Ff = np.fft.rfft(full, size)
        grad_full = 2.0 * np.fft.irfft(W * np.conj(Ff), size)[:n]

        # Map to half: grad_half[i] = grad_full[i] + grad_full[n-1-i]
        grad_half = grad_full[:half] + grad_full[n - 1: n - 1 - half: -1]
        # Softmax chain rule for half
        dot = np.dot(half_sm, grad_half)
        grad_g = half_sm * (grad_half - dot)

        m = beta1 * m + (1 - beta1) * grad_g
        v = beta2 * v + (1 - beta2) * grad_g ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        g = g - lr * m_hat / (np.sqrt(v_hat) + eps)

        half_new = safe_softmax(g)
        full_new = np.concatenate([half_new, half_new[::-1]])
        full_new = full_new / full_new.sum()
        true_obj = fast_convolve(full_new).max()

        if true_obj < best_obj - 1e-15:
            best_obj = true_obj
            best_f = full_new.copy()
            no_improve = 0
            lr *= 0.99995
        else:
            no_improve += 1
            if no_improve > 250:
                rng = np.random.default_rng(t_step * 11 + 7)
                hf = np.maximum(best_f[:half] + rng.standard_normal(half) * 0.003, 1e-15)
                g = np.log(hf / hf.sum())
                m = np.zeros(half)
                v = np.zeros(half)
                t_step = 0
                lr = max(lr_init * 0.5, 0.004)
                no_improve = 0

        if verbose and t_step % 200 == 0:
            print(f"    Pal-Adam iter {t_step}: obj={best_obj:.10f}, T={T:.1f}")

    return best_f, best_obj


def coord_desc_polish(f0, time_limit, num_multipliers=8):
    """Coordinate descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    if num_multipliers <= 8:
        multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    else:
        multipliers = np.array([0.4, 0.6, 0.75, 0.9, 1.0, 1.1, 1.3, 1.5, 1.8, 2.2, 2.8, 3.5])

    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or 1.0 - new_val <= 1e-12:
                    continue
                scale = (1.0 - new_val) / (1.0 - old_val)
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / (1.0 - old_val)
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate a probability sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def subgradient_polish(f0, time_limit, lr=0.001, mom=0.93, top_k=30, temp=0.0001):
    """Subgradient method with softmax-weighted active peaks."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    v = np.zeros(n)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        co = conv_new.max()
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        f = f_new
        cur_lr *= 0.9999
        cur_temp *= 1.003
        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj

    def try_seq(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = evaluate_sequence(a_norm.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and p.sum() > 0.01:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, initial score={evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent, using uniform n=512")

    try_seq(parent, "parent")
    n_parent = len(parent)

    # === Phase 1: Smooth gradient descent (Adam) on parent ===
    print(f"\n=== Phase 1: Smooth Adam (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t1 = min(400, rem * 0.25)
        print(f"  Time budget: {t1:.0f}s")
        a_opt, obj_opt = smooth_adam(parent, time_limit=t1, T_init=5.0,
                                     T_end=3000.0, lr_init=0.04, verbose=True)
        try_seq(a_opt, "smooth_adam_p1")

    # === Phase 2: L-BFGS-B refinement ===
    print(f"\n=== Phase 2: L-BFGS-B (n={n_parent}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic()
    if rem > 60:
        t2 = min(150, rem * 0.12)
        print(f"  Time budget: {t2:.0f}s")
        a_opt, obj_opt = lbfbs_smooth(base, time_limit=t2, T=500.0)
        try_seq(a_opt, "lbfbs")

    # === Phase 3: Palindromic smooth optimization ===
    print(f"\n=== Phase 3: Palindromic Smooth (n={n_parent}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic()
    if rem > 80:
        t3 = min(250, rem * 0.18)
        print(f"  Time budget: {t3:.0f}s")
        a_opt, obj_opt = palindromic_smooth(base, time_limit=t3, T_init=5.0,
                                            T_end=2000.0, lr_init=0.025, verbose=True)
        try_seq(a_opt, "palindromic_smooth")

    # === Phase 4: LP-FW on best ===
    print(f"\n=== Phase 4: LP-FW (n={len(best_seq) if best_seq is not None else n_parent}) ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic()
    if rem > 80:
        t4 = min(300, rem * 0.2)
        print(f"  Time budget: {t4:.0f}s")
        a_opt, obj_opt = lp_fw_optimize(base, time_limit=t4, top_k=40, verbose=True)
        try_seq(a_opt, "lpfw")

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Multi-length exploration ===")
    base = best_seq if best_seq is not None else parent
    target_lengths = [512, 640, 896, 1024]
    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        if target_n == len(base):
            continue
        f_interp = interpolate_seq(base, target_n)
        t_alloc = min(120, rem * 0.15)
        print(f"  Smooth Adam n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_adam(f_interp, time_limit=t_alloc, T_init=5.0,
                               T_end=1500.0, lr_init=0.03)
        try_seq(a_opt, f"smooth_n{target_n}")
        # Quick LP-FW
        rem = deadline - time.monotonic()
        if rem > 30:
            t_lpfw = min(60, rem * 0.1)
            a_opt2, _ = lp_fw_optimize(a_opt, time_limit=t_lpfw, top_k=30)
            try_seq(a_opt2, f"lpfw_n{target_n}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    base = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic()
    if rem > 40:
        t6 = min(200, rem * 0.2)
        print(f"  Time budget: {t6:.0f}s")
        a_opt, _ = coord_desc_polish(base, time_limit=t6)
        try_seq(a_opt, "coord_desc")

        # Second pass
        rem = deadline - time.monotonic()
        if rem > 30:
            t6b = min(100, rem * 0.15)
            a_opt2, _ = coord_desc_polish(a_opt, time_limit=t6b)
            try_seq(a_opt2, "coord_desc2")

    # === Phase 7: Subgradient polish ===
    print(f"\n=== Phase 7: Subgradient ===")
    base = best_seq if best_seq is not None else parent
    n_best = len(base)
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(150, rem * 0.15)
        print(f"  Time budget: {t7:.0f}s")
        a_opt, _ = subgradient_polish(base, time_limit=t7, lr=0.0008,
                                      mom=0.94, top_k=max(20, n_best // 6),
                                      temp=0.00005)
        try_seq(a_opt, "sg_polish")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    base = best_seq if best_seq is not None else parent
    n_best = len(base)
    rem = deadline - time.monotonic()

    # Short smooth run
    if rem > 40:
        t8 = min(100, rem * 0.2)
        a_opt, _ = smooth_adam(base, time_limit=t8, T_init=100.0,
                               T_end=5000.0, lr_init=0.01)
        try_seq(a_opt, "final_smooth")

    # Perturbation + re-optimize
    for i in range(3):
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        noise_scale = 0.002 * (i + 1)
        f_pert = project_simplex(base + rng.standard_normal(n_best) * noise_scale)
        t_pert = min(60, rem * 0.15)
        a_opt, _ = smooth_adam(f_pert, time_limit=t_pert, T_init=50.0,
                               T_end=2000.0, lr_init=0.015)
        try_seq(a_opt, f"perturb_{i}")

    # Final SG
    rem = deadline - time.monotonic()
    if rem > 15:
        a_opt, _ = subgradient_polish(best_seq, time_limit=min(rem * 0.3, 50),
                                      lr=0.0005, mom=0.95,
                                      top_k=max(10, len(best_seq) // 10),
                                      temp=0.00002)
        try_seq(a_opt, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = evaluate_sequence(best_seq.tolist())

    # Final verification
    fs = evaluate_sequence(best_seq.tolist())
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
