"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed L-BFGS-B
optimization with softmax parameterization, adaptive temperature continuation,
multi-length exploration, and local search polishing.
Key innovations: exact smooth gradient via FFT cross-correlation, continuation
schedule on temperature, and diverse initialization across lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def lse_objective_and_grad(z, T):
    """
    Log-sum-exp smoothed objective and gradient via softmax parameterization.
    z: unconstrained parameter vector
    T: inverse temperature (higher = sharper focus on max)
    Returns (objective_value, gradient_wrt_z)
    """
    n = len(z)
    # Softmax
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()

    # Convolution via FFT
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:conv_size]

    # Log-sum-exp
    m = conv.max()
    shifted = T * (conv - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(conv_size)
        w[int(np.argmax(conv))] = 1.0
        w_sum = 1.0
    else:
        w /= w_sum

    obj = 2.0 * n * (m + np.log(w_sum) / T)

    # Gradient wrt a: g[i] = 2 * sum_k w[k] * a[k-i]
    # Use FFT cross-correlation: cross[i] = sum_j w[i+j] * a[j]
    N = 1 << (max(3 * n - 2, conv_size + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, N)
    A_full = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(W) * A_full, N)[:n]
    g = 2.0 * cross

    # Gradient wrt z: dz_i = a_i * (g_i - dot(a, g))
    mean_g = np.dot(a, g)
    dz = a * (g - mean_g)

    return obj, dz


def lse_objective_only(z, T):
    """Objective only (for scipy if grad not used)."""
    obj, _ = lse_objective_and_grad(z, T)
    return obj


def optimize_with_continuation(a_init, time_limit, T_start=50.0, T_end=2000.0,
                                n_steps=8, max_iter=200):
    """
    L-BFGS-B optimization with temperature continuation.
    a_init: initial probability vector (sum=1, a>=0)
    Returns optimized probability vector.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()

    # Convert to log-space for softmax parameterization
    z = np.log(a + 1e-30)

    best_a = a.copy()
    best_obj = float('inf')

    # Temperature schedule (geometric)
    temps = np.geomspace(T_start, T_end, n_steps)

    t0 = time.monotonic()
    for t_i, T in enumerate(temps):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        # Local function for scipy
        def fun_and_grad(z_local):
            return lse_objective_and_grad(z_local, T)

        # Use scipy minimize with L-BFGS-B
        max_iter_this = min(max_iter, max(20, int(100 * remaining / time_limit * n_steps)))
        result = scipy_minimize(
            fun_and_grad, z, method='L-BFGS-B',
            jac=True, options={'maxiter': max_iter_this, 'maxfun': max_iter_this * 3,
                                'ftol': 1e-12, 'gtol': 1e-8}
        )

        if result.success or result.fun < best_obj:
            z_best = result.x
            a_new = np.exp(z_best - z_best.max())
            a_new = a_new / a_new.sum()
            # Evaluate actual objective
            conv = fast_convolve(a_new)
            actual_obj = 2.0 * n * conv.max()
            if actual_obj < best_obj:
                best_obj = actual_obj
                best_a = a_new.copy()
            a = a_new
            z = z_best

        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

    return best_a


def optimize_momentum(a_init, time_limit, lr=0.01, momentum=0.9,
                       T_start=50.0, T_end=3000.0, n_steps=10):
    """
    Projected gradient descent with Nesterov momentum and log-sum-exp smoothing.
    Works directly on the simplex (no parameterization needed).
    """
    n = len(a_init)
    f = np.maximum(a_init, 1e-15)
    f = f / f.sum()

    best_f = f.copy()
    best_obj = fast_convolve(f).max() * 2.0 * n

    v = np.zeros(n)  # momentum
    t0 = time.monotonic()

    # Temperature schedule
    temps = np.geomspace(T_start, T_end, n_steps)
    temp_idx = 0
    T = T_start

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1

        # Update temperature
        if it % max(1, it // n_steps + 1) == 0 and temp_idx < len(temps) - 1:
            temp_idx += 1
            T = temps[temp_idx]

        # Nesterov lookahead
        f_nest = f + momentum * v
        f_nest = np.maximum(f_nest, 1e-15)
        f_nest = f_nest / f_nest.sum()

        # Compute gradient via log-sum-exp
        conv_size = 2 * n - 1
        fft_size = 1 << (conv_size - 1).bit_length()
        A = np.fft.rfft(f_nest, fft_size)
        conv = np.fft.irfft(A * A, fft_size)[:conv_size]

        m = conv.max()
        shifted = T * (conv - m)
        shifted = np.clip(shifted, -500.0, 500.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(conv_size)
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Cross-correlation for gradient
        N = 1 << (max(3 * n - 2, conv_size + n - 1) - 1).bit_length()
        W = np.fft.rfft(w, N)
        A_full = np.fft.rf if hasattr(np.fft, 'rf') else np.fft.rfft
        A_full = np.fft.rfft(f_nest, N)
        cross = np.fft.irfft(np.conj(W) * A_full, N)[:n]
        grad = 2.0 * cross

        # Project gradient onto simplex tangent (subtract mean)
        grad = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - lr * grad

        # New point
        f_new = f + v
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max() * 2.0 * n

        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()

        # Backtracking if objective gets too bad
        if obj_new > best_obj * 1.5:
            v *= 0.5
            f_new = f + v
            f_new = np.maximum(f_new, 1e-15)
            f_new = f_new / f_new.sum()

        f = f_new

        # Adaptive learning rate
        lr *= 0.9995
        if lr < 0.0001:
            lr = 0.0001

    return best_f


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                    improved = True

        if not improved:
            break

    return best_f


def pair_exchange_polish(f0, time_limit, n_trials_per_s=200):
    """Pair exchange local search."""
    n = len(f0)
    f = np.maximum(f0, 1e-15).copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        # Focus on top peaks
        top_k = np.argpartition(conv, -min(20, 2 * n - 1))[-min(20, 2 * n - 1):]
        improved = False

        for _ in range(n_trials_per_s):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick a peak and transfer mass from high contributor to low
            k = int(rng.choice(top_k))
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je - js + 1 < 2:
                continue
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            hi_local = np.argmax(contribs)
            lo_local = np.argmin(contribs)
            hi = idx[hi_local]
            lo = idx[lo_local]
            if hi == lo or f[hi] < 1e-10:
                continue

            for frac in [0.1, 0.3, 0.5]:
                amt = f[hi] * frac
                f[hi] -= amt
                f[lo] += amt
                f = np.maximum(f, 0)
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[hi] += amt
                    f[lo] -= amt
                    break

        if not improved:
            # Try random pairs
            for _ in range(n_trials_per_s):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or f[i] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[i] * frac
                    f[i] -= amt
                    f[j] += amt
                    f = np.maximum(f, 0)
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-13:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[i] += amt
                        f[j] -= amt
                        break
            if not improved:
                break

    return best_f


def structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / (n - 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    # Gaussian (varying widths)
    for sigma_frac in [0.2, 0.3, 0.4]:
        center = (n - 1) / 2.0
        sigma = n * sigma_frac
        gauss = np.exp(-0.5 * ((x * (n - 1) - center) / sigma) ** 2)
        inits.append(gauss / gauss.sum())

    # Bimodal
    for d in [0.15, 0.25]:
        centers = [(n - 1) * d, (n - 1) * (1 - d)]
        bimol = np.zeros(n)
        for c in centers:
            bimol += np.exp(-0.5 * ((x * (n - 1) - c) / (n * 0.1)) ** 2)
        bimol = np.maximum(bimol, 1e-10)
        inits.append(bimol / bimol.sum())

    # Power law (concentration toward center)
    for p in [0.5, 1.0, 2.0]:
        dist = np.abs(x - 0.5)
        pw = (0.5 - dist) ** p + 1e-10
        inits.append(pw / pw.sum())

    # Cosine-based (various frequencies)
    for freq in [1, 2, 3]:
        cos_pat = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x)
        cos_pat = np.maximum(cos_pat, 1e-10)
        inits.append(cos_pat / cos_pat.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    return inits


def interpolate_to_length(f, new_n):
    """Interpolate a probability vector to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 1e-15)
    f_interp = f_interp / f_interp.sum()
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: L-BFGS-B with log-sum-exp smoothing and
    temperature continuation, momentum-based gradient descent, and local
    search polishing. Explores multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=512, score={best_score:.6f}")

    # === Phase 1: L-BFGS-B with continuation on multiple candidates ===
    print("\n=== Phase 1: L-BFGS-B Optimization ===")

    # Build candidate list
    candidates = []
    labels = []

    # Parent
    candidates.append(parent.copy())
    labels.append("parent")

    # Interpolated parent to other lengths
    for n_try in [256, 512, 1024]:
        if n_try != len(parent):
            candidates.append(interpolate_to_length(parent, n_try))
            labels.append(f"parent_n{n_try}")

    # Structured initializations at a few lengths
    for n_try in [384, 512, 768, 1024]:
        inits = structured_inits(n_try, rng)
        for i, init in enumerate(inits[:4]):
            candidates.append(init)
            labels.append(f"struct_{n_try}_{i}")

    # Budget per candidate
    n_cands = len(candidates)
    phase1_deadline = deadline - 600.0  # Reserve 600s for later phases
    t_per_cand = max(5.0, (phase1_deadline - time.monotonic()) / n_cands)
    t_per_cand = min(t_per_cand, 40.0)

    for idx, cand in enumerate(candidates):
        remaining = phase1_deadline - time.monotonic()
        if remaining < 5:
            break
        t_alloc = min(t_per_cand, remaining - 2)
        if t_alloc < 3:
            break
        n_cand = len(cand)
        print(f"  Cand {idx} ({labels[idx]}): n={n_cand}, t={t_alloc:.0f}s")

        a_opt = optimize_with_continuation(cand, time_limit=t_alloc,
                                            T_start=30.0, T_end=1500.0,
                                            n_steps=5, max_iter=150)
        try_seq(a_opt, labels[idx])

    # === Phase 2: Deep optimization on best candidate ===
    print(f"\n=== Phase 2: Deep Optimization (best={best_score:.6f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        phase2_deadline = deadline - 350.0

        # Multiple L-BFGS-B runs with different temperature schedules
        configs = [
            {"T_s": 20, "T_e": 5000, "steps": 8, "label": "deep_wide"},
            {"T_s": 50, "T_e": 3000, "steps": 6, "label": "deep_mid"},
            {"T_s": 10, "T_e": 8000, "steps": 10, "label": "deep_sharp"},
        ]

        for cfg in configs:
            remaining = phase2_deadline - time.monotonic()
            if remaining < 10:
                break
            t_alloc = min(remaining * 0.35, 120)

            # Slightly perturb the best
            noise = rng.standard_normal(n_best) * 0.003
            f_pert = np.maximum(best_seq + noise, 1e-10)
            f_pert = f_pert / f_pert.sum()

            a_opt = optimize_with_continuation(f_pert, time_limit=t_alloc,
                                                T_start=cfg["T_s"], T_end=cfg["T_e"],
                                                n_steps=cfg["steps"], max_iter=300)
            try_seq(a_opt, cfg["label"])

        # Momentum-based gradient descent (different optimization landscape)
        remaining = phase2_deadline - time.monotonic()
        if remaining > 20:
            t_alloc = min(remaining * 0.4, 150)
            print(f"  Momentum GD: t={t_alloc:.0f}s")
            a_opt = optimize_momentum(best_seq, time_limit=t_alloc,
                                       lr=0.02, momentum=0.85,
                                       T_start=20.0, T_end=5000.0, n_steps=12)
            try_seq(a_opt, "momentum")

        # Multi-start from best
        for start_i in range(3):
            remaining = phase2_deadline - time.monotonic()
            if remaining < 15:
                break
            t_alloc = min(remaining * 0.25, 90)
            scale = [0.002, 0.005, 0.01][start_i]
            f_pert = np.maximum(best_seq + rng.standard_normal(n_best) * scale, 1e-10)
            f_pert = f_pert / f_pert.sum()
            a_opt = optimize_with_continuation(f_pert, time_limit=t_alloc,
                                                T_start=30, T_end=3000,
                                                n_steps=5, max_iter=200)
            try_seq(a_opt, f"restart_{start_i}")

    # === Phase 3: Try different lengths from best shape ===
    print(f"\n=== Phase 3: Length Exploration (best={best_score:.6f}) ===")

    if best_seq is not None:
        phase3_deadline = deadline - 150.0
        target_lengths = [640, 896, 1152, 1280]
        # Avoid the length we already have
        target_lengths = [tl for tl in target_lengths if tl != len(best_seq)]

        for tl in target_lengths:
            remaining = phase3_deadline - time.monotonic()
            if remaining < 15:
                break
            f_interp = interpolate_to_length(best_seq, tl)
            t_alloc = min(remaining * 0.3, 80)
            print(f"  n={tl}, t={t_alloc:.0f}s")
            a_opt = optimize_with_continuation(f_interp, time_limit=t_alloc,
                                                T_start=30, T_end=2000,
                                                n_steps=5, max_iter=200)
            try_seq(a_opt, f"len_{tl}")

    # === Phase 4: Local search polishing ===
    print(f"\n=== Phase 4: Local Search (best={best_score:.6f}) ===")

    if best_seq is not None:
        phase4_deadline = deadline - 60.0

        # Coordinate descent
        remaining = phase4_deadline - time.monotonic()
        if remaining > 15:
            t_cd = min(remaining * 0.4, 100)
            print(f"  Coord descent: t={t_cd:.0f}s")
            a_opt = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

        # Pair exchange
        remaining = phase4_deadline - time.monotonic()
        if remaining > 10:
            t_pe = min(remaining * 0.4, 80)
            print(f"  Pair exchange: t={t_pe:.0f}s")
            a_opt = pair_exchange_polish(best_seq, time_limit=t_pe)
            try_seq(a_opt, "pair_exch")

        # Another round of CD after PE
        remaining = phase4_deadline - time.monotonic()
        if remaining > 10:
            t_cd2 = min(remaining * 0.5, 60)
            print(f"  Coord descent 2: t={t_cd2:.0f}s")
            a_opt = coordinate_descent_polish(best_seq, time_limit=t_cd2)
            try_seq(a_opt, "cd2")

    # === Phase 5: Final fine-tuning ===
    print(f"\n=== Phase 5: Final Fine-tuning (best={best_score:.6f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very sharp L-BFGS-B (low T start, focusing on true max)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(remaining * 0.4, 60)
            print(f"  Sharp L-BFGS-B: t={t_alloc:.0f}s")
            a_opt = optimize_with_continuation(best_seq, time_limit=t_alloc,
                                                T_start=500, T_end=10000,
                                                n_steps=3, max_iter=300)
            try_seq(a_opt, "sharp_lbf")

        # Fine momentum
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_alloc = min(remaining * 0.3, 40)
            print(f"  Fine momentum: t={t_alloc:.0f}s")
            a_opt = optimize_momentum(best_seq, time_limit=t_alloc,
                                       lr=0.005, momentum=0.9,
                                       T_start=500.0, T_end=8000.0, n_steps=4)
            try_seq(a_opt, "fine_mom")

        # Final pair exchange
        remaining = deadline - time.monotonic()
        if remaining > 5:
            t_alloc = min(remaining * 0.4, 30)
            print(f"  Final PE: t={t_alloc:.0f}s")
            a_opt = pair_exchange_polish(best_seq, time_limit=t_alloc,
                                          n_trials_per_s=300)
            try_seq(a_opt, "final_pe")

        # Final CD
        remaining = deadline - time.monotonic()
        if remaining > 5:
            t_alloc = min(remaining * 0.5, 25)
            print(f"  Final CD: t={t_alloc:.0f}s")
            a_opt = coordinate_descent_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
