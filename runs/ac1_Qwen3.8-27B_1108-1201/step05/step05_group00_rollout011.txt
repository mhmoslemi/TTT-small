"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid FISTA + peak-peeling approach.
Key differences from previous approach:
- FISTA (accelerated proximal gradient) instead of plain GD for smoothed objective
- CG-based QP solver for game-iterate subproblem (faster than active set for large n)

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]. p has length 2n-1, a has length n."""
    return np.correlate(p, a, mode='valid')


def softmax_safe(x, T):
    """Numerically stable softmax with temperature."""
    x = x / T
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def fista_smoothed(a0, time_limit, T_init=0.001, T_min=0.0001, alpha_decay=0.9995):
    """
    FISTA (accelerated proximal gradient) on the smoothed objective.
    Smooths max with softmax at temperature T, anneals T over time.
    Uses backtracking line search for step size.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_T = T

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    eta = 0.5  # Initial step size (safe: L <= 2 for normalized p)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # FISTA variables
        z = a.copy()
        t_k = 1.0
        a_prev = a.copy()

        # Inner FISTA iterations (batch)
        for inner in range(50):
            if time.monotonic() - t0 > time_limit - 0.05:
                break

            # Momentum step
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = z + (t_k - 1.0) / t_new * (z - a_prev)
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys < 1e-15:
                y = z
            else:
                y = y / ys

            # Compute gradient at y
            conv_y = fast_convolve(y)
            p = softmax_safe(conv_y, T)
            grad = 2.0 * hankel_mv(p, y)

            # Backtracking line search
            a_candidate = y
            f_y = np.dot(y, hankel_mv(p, y))  # y^T M y
            improved = False
            for bt in range(20):
                a_candidate = project_simplex(y - eta * grad)
                conv_c = fast_convolve(a_candidate)
                p_c = softmax_safe(conv_c, T)
                f_c = np.dot(a_candidate, hankel_mv(p_c, a_candidate))
                # Descent condition (approximate)
                if f_c <= f_y - 0.5 * eta * np.dot(grad, y - a_candidate):
                    improved = True
                    break
                eta *= 0.5
            if not improved:
                a_candidate = project_simplex(y - eta * grad)

            a_prev = a
            a = a_candidate
            z = a
            t_k = t_new

        # Evaluate
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            best_T = T
            no_improve = 0
        else:
            no_improve += 1

        # Adapt temperature and step size
        if no_improve > 30:
            T = max(T_min, T * 0.85)
            eta = min(eta * 1.05, 1.0)
        if no_improve > 100:
            T = max(T_min, T * 0.7)
        if no_improve > 5:
            eta = min(eta * 1.02, 1.5)

    return best_a, best_obj


def fsgd_smoothed(a0, time_limit, T_init=0.001, lr_init=0.05):
    """
    Fast Smoothed Gradient Descent with momentum (similar to previous but with
    adaptive top-k weighting and better step size scheduling).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    momentum = np.zeros(n)
    mom_coeff = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        # Nesterov-like momentum
        momentum = mom_coeff * momentum - lr * grad
        a = project_simplex(a + momentum)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive scheduling
        if no_improve > 30:
            T *= 0.9
            lr *= 0.98

        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 17) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            momentum = np.zeros(n)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def eg_peak_peel(a0, time_limit, top_k=10, decay=0.7):
    """
    Exponentiated Gradient with focused peak peeling.
    Uses the top-k peaks with geometric decay weighting.
    EG update: a[i] *= exp(-eta * grad[i]), then normalize.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = 0.01

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        conv = fast_convolve(a)
        # Get top-k peaks
        k = min(top_k, len(conv))
        peak_idx = np.argpartition(-conv, k)[:k]
        peak_idx = peak_idx[np.argsort(-conv[peak_idx])]

        # Weighted subgradient
        grad = np.zeros(n)
        for rank, ki in enumerate(peak_idx):
            w = decay ** rank
            if w < 0.01:
                break
            ki = int(ki)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w * 2.0 * a[ki - idx_arr]

        # EG update (multiplicative weights)
        log_update = -eta * grad
        log_update -= log_update.max()  # For numerical stability
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-30:
            break
        a_new = a_new / s_new

        # Line search: try a few eta values
        best_eta = eta
        best_val = fast_convolve(a_new).max()
        for trial_eta in [eta * 0.5, eta * 2.0, eta * 0.25, eta * 4.0]:
            a_trial = a * np.exp(-trial_eta * grad)
            a_trial = a_trial / a_trial.sum()
            val = fast_convolve(a_trial).max()
            if val < best_val:
                best_val = val
                best_eta = trial_eta
                a_new = a_trial

        a = a_new
        eta = best_eta * 0.95  # Slight decay

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            rng = np.random.default_rng((it * 31627 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            eta = 0.01
            no_improve = 0

    return best_a, best_obj


def game_fista(a0, time_limit, T_init=0.001):
    """
    Game-theoretic iterate where the primal QP is solved using FISTA
    (proximal gradient with acceleration) instead of active set.
    This is faster and more robust for large n.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Dual update: p = softmax(conv(a)/T)
        conv = fast_convolve(a)
        p = softmax_safe(conv, T)

        # Primal: solve min a'^T M(p) a' s.t. a' in simplex using FISTA
        # M(p) x = hankel_mv(p, x)
        # grad of (1/2) x^T M x = M x = hankel_mv(p, x)
        # L = 2*||M||_2 <= 2 (since p is probability vector)
        eta_qp = 0.4
        z = a.copy()
        t_k = 1.0
        x = a.copy()
        x_prev = a.copy()

        for _ in range(30):
            if time.monotonic() - t0 > time_limit - 0.05:
                break
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = z + (t_k - 1.0) / t_new * (z - x_prev)
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys > 1e-15:
                y = y / ys
            else:
                y = z

            # Backtracking for QP step
            q_grad = hankel_mv(p, y)
            best_x = y
            best_f = np.dot(y, q_grad)
            for bt in range(15):
                x_candidate = project_simplex(y - eta_qp * q_grad)
                f_c = np.dot(x_candidate, hankel_mv(p, x_candidate))
                if f_c <= best_f - 0.5 * eta_qp * np.dot(q_grad, y - x_candidate) + 1e-15:
                    best_x = x_candidate
                    break
                eta_qp *= 0.5

            x_prev = x
            x = best_x
            z = x
            t_k = t_new

        a = x
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt T
        if no_improve > 40:
            T *= 0.85
        if no_improve > 200:
            T = max(0.00005, T * 0.7)
        if no_improve > 500:
            rng = np.random.default_rng((it * 104729 + 11) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            no_improve = 0

    return best_a, best_obj


def block_coordinate_polish(a0, time_limit, block_size=None):
    """
    Block coordinate descent: partition a into blocks, optimize each block
    while fixing the rest. Uses FISTA within each block.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    if block_size is None:
        block_size = max(8, n // 32)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        improved = False
        for block_start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit - 0.05:
                break
            block_end = min(block_start + block_size, n)

            # Precompute contributions from fixed blocks
            a_fixed = a.copy()
            a_fixed[block_start:block_end] = 0
            conv_fixed = fast_convolve(a_fixed)

            # Optimize the block: min max_k (conv_fixed[k] + 2*a_fixedconj[k] + a_block^T H a_block)
            # where a_fixedconj[k] = sum_{i in block, j not in block} a[i]*a[j] + symmetric
            # This is complex; instead use gradient on the full objective restricted to block
            a_block = a[block_start:block_end].copy()
            a_block_sum = a_block.sum()

            for bl in range(20):
                if time.monotonic() - t0 > time_limit - 0.05:
                    break
                # Full gradient for the block variables
                conv_full = fast_convolve(a)
                # For block variable a[i], derivative of max conv is 2*sum_k p[k]*a[k-i]
                T_block = max(0.001, conv_full.max() * 0.01)
                p = softmax_safe(conv_full, T_block)
                grad_full = 2.0 * hankel_mv(p, a)
                grad_block = grad_full[block_start:block_end]

                # Simple gradient descent on block with simplex constraint
                lr_block = 0.1
                a_block_new = a_block - lr_block * grad_block
                # Keep block sum constant (to maintain total sum=1)
                if a_block_new.sum() > 1e-10:
                    a_block_new = np.maximum(a_block_new, 0)
                    a_block_new = a_block_new / a_block_new.sum() * a_block_sum
                else:
                    a_block_new = np.maximum(a_block, 0) / max(a_block.sum(), 1e-15) * a_block_sum

                a[block_start:block_end] = a_block_new
                a = np.maximum(a, 0)
                a = a / a.sum()

            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                improved = True

        if not improved and it > 100:
            break

    return best_a, best_obj


def structured_restarts(a0, n_restarts, time_per_restart, rng):
    """
    Generate structured perturbations based on the convolution structure.
    Instead of random noise, perturb based on where the peaks are.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    results = []
    conv = fast_convolve(a)
    peaks = np.argpartition(-conv, min(20, len(conv)))[:min(20, len(conv))]
    peaks = peaks[np.argsort(-conv[peaks])]

    for r in range(n_restarts):
        if r < len(peaks):
            # Perturb based on a peak: shift mass away from contributing positions
            ki = int(peaks[r])
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            affected = np.arange(js, je + 1)
            perturbation = np.zeros(n)
            # Reduce at positions that contribute most to this peak
            contrib = a[ki - affected]
            perturbation[affected] = -0.005 * contrib / max(contrib.max(), 1e-10)
            # Add elsewhere (away from peaks)
            other = np.setdiff1d(np.arange(n), affected)
            if len(other) > 0:
                perturbation[other] = 0.002 / len(other)
        else:
            # Pure random for diversity
            perturbation = rng.standard_normal(n) * 0.003

        a_pert = project_simplex(a + perturbation)
        results.append(a_pert)

    return results


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_step(f, M, m_exp=1.0):
    """Single FW-LP step."""
    n = len(f)
    g = solve_lp_oracle(f, M * m_exp)
    if g is None:
        return None, m_exp
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None, m_exp
    g = g / g_sum

    # Line search
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha > 1e-10:
        result = (1.0 - best_alpha) * f + best_alpha * g
        result = np.maximum(result, 0.0)
        result = result / result.sum()
        return result, 1.0
    return None, min(m_exp * 1.3, 5.0)

        if n_try == n_parent and parent is not None:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(remaining, 50)
        print(f"  FISTA n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fista_smoothed(start, time_limit=t_alloc, T_init=0.001, T_min=0.0001)
        try_seq(a_opt, f"fista_n{n_try}")

        # Quick game iterate
        remaining = deadline - time.monotonic() - 1200
        if remaining > 20:
            a_opt2, _ = game_fista(a_opt, time_limit=min(remaining, 25), T_init=0.001)
            try_seq(a_opt2, f"game_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.8f} (elapsed {time.monotonic()-t1_start:.0f}s)")

    # === Phase 2: Deep FISTA + Game iterate on best (~500s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t2_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Long FISTA
        remaining = deadline - time.monotonic() - 900
        if remaining > 30:
            t_alloc = min(remaining, 250)
            print(f"  Deep FISTA n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fista_smoothed(best_seq, time_limit=t_alloc, T_init=0.0005, T_min=0.00005)
            try_seq(a_opt, "deep_fista")
            if a_opt.sum() > 1e-10:
                best_for_phase2 = a_opt / a_opt.sum()
            else:
                best_for_phase2 = best_seq

        # Long game iterate
        remaining = deadline - time.monotonic() - 700
        if remaining > 30:
            t_alloc = min(remaining, 200)
            print(f"  Deep Game FISTA n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = game_fista(best_for_phase2, time_limit=t_alloc, T_init=0.0005)
            try_seq(a_opt, "deep_game")

        # FSGD with momentum
        remaining = deadline - time.monotonic() - 500
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  FSGD n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fsgd_smoothed(best_for_phase2, time_limit=t_alloc, T_init=0.001, lr_init=0.03)
            try_seq(a_opt, "fsgd")

    print(f"  Phase 2 done: best={best_score:.8f} (elapsed {time.monotonic()-t2_start:.0f}s)")

    # === Phase 3: EG Peak Peeling + FW-LP (~400s) ===
    print("\n=== Phase 3: EG Peak Peel + FW-LP ===")
    t3_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # EG peak peeling
        remaining = deadline - time.monotonic() - 350
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  EG Peak Peel top_k=5, t={t_alloc:.0f}s")
            a_opt, _ = eg_peak_peel(best_seq, time_limit=t_alloc, top_k=5, decay=0.6)
            try_seq(a_opt, "eg_k5")

        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  EG Peak Peel top_k=15, t={t_alloc:.0f}s")
            a_opt, _ = eg_peak_peel(best_seq, time_limit=t_alloc, top_k=15, decay=0.8)
            try_seq(a_opt, "eg_k15")

        # FW-LP for global structure
        remaining = deadline - time.monotonic() - 150
        if remaining > 30:
            t_alloc = min(remaining, 120)
            print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
            # Run multiple FW-LP iterations
            f = best_seq.copy()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            for _ in range(t_alloc // 10):
                if time.monotonic() > deadline - 150:
                    break
                result, m_exp = fw_lp_step(f, M, m_exp)
                if result is not None:
                    f = result
                    conv = fast_convolve(f)
                    M = conv.max()
                    try_seq(f, "fwlp_iter")
                else:
                    if m_exp > 4:
                        break

    print(f"  Phase 3 done: best={best_score:.8f} (elapsed {time.monotonic()-t3_start:.0f}s)")

    # === Phase 4: Block coordinate + structured restarts (~300s) ===
    print("\n=== Phase 4: Block Coord + Structured Restarts ===")
    t4_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Block coordinate polish
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Block Coord t={t_alloc:.0f}s")
            a_opt, _ = block_coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "block_coord")

        # Structured restarts with quick optimization
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            restarts = structured_restarts(best_seq, 5, 0, rng)
            for ri, a_r in enumerate(restarts):
                remaining = deadline - time.monotonic() - 80
                if remaining < 10:
                    break
                t_alloc = min(remaining, 30)
                a_opt, _ = fista_smoothed(a_r, time_limit=t_alloc, T_init=0.001, T_min=0.0001)
                try_seq(a_opt, f"restart_{ri}")

    print(f"  Phase 4 done: best={best_score:.8f} (elapsed {time.monotonic()-t4_start:.0f}s)")

    # === Phase 5: Final refinement (~200s) ===
    print("\n=== Phase 5: Final Refinement ===")
    t5_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine FISTA
        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 60)
            print(f"  Fine FISTA t={t_alloc:.0f}s")
            a_opt, _ = fista_smoothed(best_seq, time_limit=t_alloc, T_init=0.0001, T_min=0.00001)
            try_seq(a_opt, "fine_fista")

        # Fine game iterate
        remaining = deadline - time.monotonic() - 50
        if remaining > 15:
            t_alloc = min(remaining, 40)
            print(f"  Fine Game t={t_alloc:.0f}s")
            a_opt, _ = game_fista(best_seq, time_limit=t_alloc, T_init=0.0002)
            try_seq(a_opt, "fine_game")

        # Tiny perturbations + FISTA
        remaining = deadline - time.monotonic() - 20
        if remaining > 15:
            for pi in range(4):
                remaining = deadline - time.monotonic() - 10
                if remaining < 5:
                    break
                noise_scale = 0.0005 + pi * 0.0003
                a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
                t_alloc = min(remaining, 8)
                a_opt, _ = fista_smoothed(a_pert, time_limit=t_alloc, T_init=0.0002, T_min=0.00005)
                try_seq(a_opt, f"pert_{pi}")

        # Final single EG step with very small eta
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_opt, _ = eg_peak_peel(best_seq, time_limit=min(remaining, 10), top_k=3, decay=0.5)
            try_seq(a_opt, "final_eg")

    print(f"  Phase 5 done: best={best_score:.8f} (elapsed {time.monotonic()-t5_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic()-t_start:.0f}s")
    return [float(x) for x in best_seq]
