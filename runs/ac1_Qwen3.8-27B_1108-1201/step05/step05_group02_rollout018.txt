    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        contribution = np.zeros(n)
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            contribution[idx[mask]] += a[kg[mask]]

        # Normalize contribution
        cmax = contribution.max()
        if cmax > 1e-15:
            contribution /= cmax

        a_new = a * np.exp(-alpha * contribution)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = best_a.copy()
        else:
            a_new /= s_new

        conv = fast_convolve(a_new)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Mult it={it}: obj={best_obj:.10f} alpha={alpha:.5f}")
        else:
            no_improve += 1
            if no_improve > 200:
                a = best_a.copy()
                alpha = alpha_init
                no_improve = 0
                it = 0
                t0 = time.monotonic()
                continue

        a = a_new
        alpha *= alpha_decay

    return best_a, best_obj


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def lp_fw_optimize(a0, time_limit, verbose=False, max_iters=None):
    """LP-based Frank-Wolfe with min-peak LP oracle."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0) if 'f0' in dir() else len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if max_iters is not None and it > max_iters:
            break

        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        else:
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW it={it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.002 * min(no_improve // 40, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        top_k = min(3, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Sample a subset of coordinates for efficiency
            candidates = np.arange(js, je + 1)
            if len(candidates) > 50:
                rng = np.random.default_rng(passes * 13 + peak_k)
                candidates = rng.choice(candidates, size=50, replace=False)

            for j in candidates:
                j = int(j)
                old_val = f[j]
                if old_val < 1e-13:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj
                multipliers = [0.5, 0.7, 0.85, 1.15, 1.4, 1.7]

                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                    if best_j_obj < best_obj - 1e-13:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_window(n, wtype='uniform'):
    """Create various analytical window shapes normalized to sum=1."""
    x = np.arange(n, dtype=np.float64)
    xc = (x - (n - 1) / 2.0) / ((n - 1) / 2.0 + 1e-10)

    if wtype == 'uniform':
        w = np.ones(n)
    elif wtype == 'hann':
        w = 0.5 * (1 + np.cos(2 * np.pi * xc))
    elif wtype == 'hamming':
        w = 0.54 + 0.46 * np.cos(2 * np.pi * xc)
    elif wtype == 'blackman':
        w = 0.42 + 0.5 * np.cos(2 * np.pi * xc) + 0.08 * np.cos(4 * np.pi * xc)
    elif wtype == 'blackman_harris':
        w = (0.35875
             - 0.48829 * np.cos(2 * np.pi * xc)
             + 0.14128 * np.cos(4 * np.pi * xc)
             - 0.01168 * np.cos(6 * np.pi * xc))
    elif wtype == 'parabolic':
        w = 1 - xc * xc
    elif wtype == 'gaussian':
        w = np.exp(-4 * xc * xc)
    elif wtype == 'arcsine':
        w = 1.0 / np.sqrt(np.maximum(1 - xc * xc, 1e-10))
    elif wtype == 'u_shape':
        w = 0.3 + 0.7 * xc * xc
    elif wtype == 'superflat':
        #attempt to flatten spectrum more
        w = 0.5 + 0.3 * np.cos(2 * np.pi * xc) + 0.15 * np.cos(4 * np.pi * xc) + 0.05 * np.cos(6 * np.pi * xc)
    else:
        w = np.ones(n)

    w = np.maximum(w, 0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    return w


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining Adam smoothed descent,
    LP-FW, multiplicative updates, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 0: Try diverse analytical initializations ===
    print("\n=== Phase 0: Analytical initializations ===")
    window_types = ['hann', 'hamming', 'blackman_harris', 'parabolic',
                    'gaussian', 'u_shape', 'superflat', 'arcsine']

    for wt in window_types:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        w = make_window(base_n, wt)
        score = try_seq(w, f"init_{wt}")
        # Quick Adam refinement (short)
        if score < best_score + 0.05:
            a_opt, _ = adam_smoothed_descend(w, time_limit=min(30, rem * 0.1),
                                             lr=0.005, top_k=60, temp_init=0.001)
            try_seq(a_opt, f"adam_{wt}")

    # Use the best so far as the main working sequence
    main_seq = best_seq.copy() if best_seq is not None else parent
    main_n = len(main_seq)

    # === Phase 1: Adam smoothed descent (main workhorse) ===
    print(f"\n=== Phase 1: Adam smoothed descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 500)
    if t1 > 30:
        print(f"  n={main_n}, t={t1:.0f}s")
        a_opt, _ = adam_smoothed_descend(main_seq, time_limit=t1,
                                         lr=0.003, top_k=80,
                                         temp_init=0.0005, temp_mult=1.002,
                                         verbose=True)
        try_seq(a_opt, "adam_main")

    # Second Adam run with different params
    rem = deadline - time.monotonic()
    t1b = min(rem * 0.2, 250)
    if t1b > 30:
        a_opt, _ = adam_smoothed_descend(best_seq, time_limit=t1b,
                                         lr=0.001, top_k=30,
                                         temp_init=0.0002, temp_mult=1.004,
                                         verbose=False)
        try_seq(a_opt, "adam_fine")

    # === Phase 2: LP-FW ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.2, 300)
    if t2 > 30:
        print(f"  n={len(best_seq)}, t={t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        try_seq(a_opt, "lpfw_main")

    # === Phase 3: Multiplicative updates ===
    print(f"\n=== Phase 3: Multiplicative updates (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200)
    if t3 > 20:
        a_opt, _ = multiplicative_update(best_seq, time_limit=t3,
                                         alpha_init=0.015, top_k=15,
                                         alpha_decay=0.9995)
        try_seq(a_opt, "mult_update")

    # === Phase 4: Dimension exploration ===
    print(f"\n=== Phase 4: Dimension exploration (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    candidate_lengths = set()
    for delta in [-128, -64, 0, 64, 128, 256, -256]:
        ln = len(base) + delta
        if 384 <= ln <= 2048:
            candidate_lengths.add(ln)
    candidate_lengths = sorted(candidate_lengths)

    for target_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.25, 180)
        if t_alloc < 20:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = adam_smoothed_descend(f_interp, time_limit=t_alloc,
                                         lr=0.003, top_k=60,
                                         temp_init=0.0005, temp_mult=1.002)
        try_seq(a_opt, f"dim_{target_n}")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.2, 200)
    if t5 > 20:
        print(f"  n={len(best_seq)}, t={t5:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t5)
        try_seq(a_opt, "cd_polish")

    # === Phase 6: Final combined polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    n_best = len(best_seq)

    # Quick Adam
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        a_opt, _ = adam_smoothed_descend(best_seq, time_limit=min(rem, 80),
                                         lr=0.0008, top_k=20,
                                         temp_init=0.0001, temp_mult=1.003)
        try_seq(a_opt, "adam_final")

    # LP-FW polish
    rem = deadline - time.monotonic() - 20
    if rem > 15:
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 60))
        try_seq(a_opt, "lpfw_final")

    # Perturbation-based restarts
    for p in range(8):
        rem = deadline - time.monotonic() - 15
        if rem < 12:
            break
        noise_scale = 0.0005 + 0.0003 * p
        noise = rng.standard_normal(n_best) * noise_scale
        f_pert = project_simplex(best_seq + noise)
        a_opt, _ = adam_smoothed_descend(f_pert, time_limit=min(rem, 30),
                                         lr=0.002, top_k=40,
                                         temp_init=0.0003, temp_mult=1.002)
        try_seq(a_opt, f"perturb_{p}")

    # Final LP-FW on best
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 40))
        try_seq(a_opt, "lpfw_very_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
