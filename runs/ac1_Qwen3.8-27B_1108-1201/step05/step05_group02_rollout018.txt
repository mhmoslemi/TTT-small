def adam_softmax_optimize(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, top_k=20):

    a = a0.copy()
    a = a / a.sum()
    
    # Initialize x such that softmax(x) = a
    # x_i = log(a_i) works if a_i > 0
    # For zeros, use a small value
    a_safe = np.maximum(a, 1e-10)
    x = np.log(a_safe) - np.log(a_safe).mean()  # Center for numerical stability
    
    v = np.zeros(n)  # First moment
    m = np.zeros(n)  # Second moment
    t = 0
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    def softmax(x):
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        return exp_x / exp_x.sum()
    
    t0 = time.time()
    while time.time() - t0 < time_limit:
        t += 1
        a_current = softmax(x)
        conv = fast_convolve(a_current)
        # Compute subgradient using top-k peaks with softmax weighting
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / (pv.max() * 0.01 + 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Subgradient of max conv wrt a
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                kg = k - j
                if 0 <= kg < n:
                    sg[j] += 2.0 * w[i] * a_current[kg]
        
        # Gradient through softmax: grad_x = softmax(x) * (sg - sum(softmax(x) * sg))
        px = softmax(x)
        grad_x = px * (sg - np.dot(px, sg))
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = softmax(x)
        conv_new = fast_convolve(a_new)
        obj = conv_new.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_flatten_step(a, softness=1.0):
    """One step of spectral flattening using rfft."""
    n = len(a)
    A = np.fft.rfft(a)
    n_bins = len(A)
    if n_bins <= 1:
        return a.copy()
    dc = A[0].copy()
    non_dc = A[1:]
    non_dc_energy = np.sum(np.abs(non_dc) ** 2)
    if non_dc_energy < 1e-30:
        return a.copy()
    target_mag = np.sqrt(non_dc_energy / len(non_dc))
    mags = np.abs(non_dc)
    phases = np.angle(non_dc)
    if softness >= 1.0:
        new_mags = np.full(len(non_dc), target_mag)
    else:
        new_mags = mags ** (1.0 - softness) * target_mag ** softness
    zero_mask = mags < 1e-15
    new_mags[zero_mask] = target_mag * 0.01
    new_A = np.zeros(n_bins, dtype=complex)
    new_A[0] = dc
    new_A[1:] = new_mags * np.exp(1j * phases)
    a_new = np.fft.irfft(new_A, n)
    return a_new


def pocs_spectral_optimize(a0, time_limit, verbose=False, softness_start=0.5, softness_end=1.0):
    """POCS: alternate between non-negativity projection and spectral flattening."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    total_iters = 200

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Anneal softness from soft_start to soft_end
        progress = min(1.0, it / total_iters)
        softness = softness_start + (softness_end - softness_start) * progress

        # Spectral flatten
        a_new = spectral_flatten_step(a, softness)
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new

        conv = fast_convolve(a_new)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and it % 100 == 0:
                print(f"    POCs iter {it}: obj={best_obj:.10f}, softness={softness:.3f}")
        else:
            no_improve += 1
            a = a_new

        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 31337)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
            if verbose:
                print(f"    POCs restart at iter {it}")

    return best_a, best_obj


def softmax_func(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        t += 1
        a_current = softmax_func(x)
        conv = fast_convolve(a_current)

        # Compute subgradient with softmax-weighted peaks
        k_actual = min(cur_top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        max_pv = pv.max()
        if max_pv < 1e-15:
            continue
        temp = max_pv * 0.05
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient of max conv wrt a
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * a_current[kg[mask]]

        # Gradient through softmax: d(softmax)/dx = softmax * (e_i - softmax)
        # So grad_x = softmax(x) * (sg - sum(softmax(x) * sg))
        px = a_current
        grad_x = px * (sg - np.dot(px, sg))

        # Adam update
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad_x
        v_vec = beta2 * v_vec + (1.0 - beta2) * grad_x ** 2
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        x = x - cur_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_func(x)
        conv_new = fast_convolve(a_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and t % 500 == 0:
                print(f"    Adam iter {t}: obj={best_obj:.10f}, lr={cur_lr:.6f}")
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 200:
            cur_lr *= 0.7
            cur_top_k = max(10, cur_top_k // 2)
            no_improve = 0

        if no_improve > 800:
            # Restart
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            a_pert = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            x = np.log(np.maximum(a_pert, 1e-12))
            x -= x.mean()
            m_vec = np.zeros(n)
            v_vec = np.zeros(n)
            t = 0
            cur_lr = lr * 0.5
            no_improve = 0

    return best_a, best_obj


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with min-peak LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        else:
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.003 * min(no_improve // 50, 4)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def peak_targeting_cd(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break

                old_val = f[j]
                if old_val < 1e-13:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj
                multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.4, 1.8]

                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.8:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 5:
                rng = np.random.default_rng(passes * 104729)
                for _ in range(min(30, n // 8)):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.3, 0.6, 1.5, 2.0]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.8:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break
                    if improved:
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.003, momentum=0.94, top_k=40, temp=0.00005):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0008)
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining spectral POCS, Adam softmax, LP-FW,
    subgradient, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Spectral POCS (novel) ===
    print(f"\n=== Phase 1: Spectral POCS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(350, rem * 0.25)
    if t1 > 30:
        print(f"  n={len(best_seq)}, t={t1:.0f}s")
        a_opt, _ = pocs_spectral_optimize(best_seq, time_limit=t1, verbose=True)
        try_seq(a_opt, "pocs_main")

        # Second POCs run with different softness schedule
        rem = deadline - time.monotonic()
        t1b = min(200, rem * 0.12)
        if t1b > 20 and best_seq is not None:
            a_opt, _ = pocs_spectral_optimize(best_seq, time_limit=t1b,
                                              softness_start=0.3, softness_end=0.8)
            try_seq(a_opt, "pocs_soft")

    # === Phase 2: Adam Softmax (novel) ===
    print(f"\n=== Phase 2: Adam Softmax (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(300, rem * 0.2)
        if t2 > 30:
            print(f"  n={len(best_seq)}, t={t2:.0f}s")
            a_opt, _ = adam_softmax_optimize(best_seq, time_limit=t2,
                                             lr=0.015, top_k=30, verbose=True)
            try_seq(a_opt, "adam_main")

            # Second Adam with different params
            rem = deadline - time.monotonic()
            t2b = min(200, rem * 0.12)
            if t2b > 20:
                a_opt, _ = adam_softmax_optimize(best_seq, time_limit=t2b,
                                                 lr=0.008, beta1=0.85, beta2=0.998,
                                                 top_k=50)
                try_seq(a_opt, "adam_fine")

    # === Phase 3: LP-FW ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(250, rem * 0.18)
        if t3 > 30:
            print(f"  n={len(best_seq)}, t={t3:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 4: Subgradient ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t4 = min(250, rem * 0.18)
        if t4 > 30:
            print(f"  n={n_best}, t={t4:.0f}s")

                                            lr=0.0025, momentum=0.94,
                                            top_k=max(30, n_best // 8),
                                            temp=0.00004)
            try_seq(a_opt, "sg")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        base_n = len(base)
        candidate_lengths = set()

        for target_n in candidate_lengths[:5]:

                break

            if rem > 40:

                                                top_k=max(20, target_n // 10),

    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Peak CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t6 = min(200, rem * 0.2)
        if t6 > 25:
            print(f"  n={len(best_seq)}, t={t6:.0f}s")
            a_opt, _ = peak_targeting_cd(best_seq, time_limit=t6)
            try_seq(a_opt, "peak_cd")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick POCs
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = pocs_spectral_optimize(best_seq, time_limit=min(rem, 60),
                                              softness_start=0.7, softness_end=1.0)
            try_seq(a_opt, "pocs_polish")

        # Quick Adam
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = adam_softmax_optimize(best_seq, time_limit=min(rem, 50),
                                             lr=0.005, top_k=20)
            try_seq(a_opt, "adam_polish")

        # Quick LP-FW
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 45))
            try_seq(a_opt, "lpfw_polish")

        # Perturbations with quick optimization
        for p in range(5):
            rem = deadline - time.monotonic() - 15
            if rem < 12:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.0008 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = pocs_spectral_optimize(f_pert, time_limit=min(rem, 20),
                                              softness_start=0.5, softness_end=0.9)

    # Fallback
    if best_seq is None:

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):

    else:

        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
