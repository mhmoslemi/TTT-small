"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative weight descent (fast, O(n*k) per iter)
2. Adam optimizer on softmax parameterization of LSE-smoothed objective

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def pressure_matrix(f, w, n_peak_targets):
    """
    Compute the 'pressure' on each element of f given peak weights w.
    pressure_m = sum_k w[k] * f[k-m] (with valid indices only)
    This is the gradient of sum_k w[k]*b[k] w.r.t. f[m] divided by 2.
    """
    n = len(f)
    # We want: p[m] = sum_{k: 0<=k-m<n} w[k] * f[k-m]
    # This is a cross-correlation: p = corr(w, f) but with specific alignment
    # Actually p[m] = sum_k w[k] * f[k-m] where k-m in [0, n-1] and k in [0, 2n-2]
    # = sum_j w[m+j] * f[j] where j in [0, n-1] and m+j in [0, 2n-2]
    # = cross-correlation of w and f at lag m
    
    # Use FFT for efficiency
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    # Cross-correlation: ifft(conj(W) * F) gives sum_k w[k+m] * f[k]... 
    # Actually let's be more careful.
    # p[m] = sum_j f[j] * w[m+j] for j=0..n-1, m+j valid
    # This is like a cross-correlation.
    # corr[m] = sum_j f[j] * w[m+j]
    # = ifft(fft(f) * conj(fft(w))) ... let me just use direct for small n_peak targets
    
    # For efficiency, use the fact that we only have a few nonzero w entries
    pressure = np.zeros(n)
    for k in n_peak_targets:
        wk = w[k]
        if wk < 1e-18:
            continue
        k = int(k)
        # p[m] += w[k] * f[k-m] for m where 0 <= k-m < n, i.e., max(0,k-n+1) <= m <= min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            pressure[ms] += wk * f[k - ms]
    return pressure

def multiplicative_descent(f0, time_limit, n_peaks=50, lr=0.01, beta=None):
    """
    Multiplicative weight descent for peak reduction.
    a_m <- a_m * exp(-lr * pressure_m)
    where pressure_m = sum_k w[k] * a[k-m] (gradient contribution to peaks)
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_max = conv.max()
        
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        # Compute peak weights
        if beta is not None:
            bc = beta * conv
            bc_max = bc.max()
            exp_bc = np.exp(bc - bc_max)
            w = exp_bc / exp_bc.sum()
            # Use only top entries for efficiency
            idx = np.argsort(w)[::-1][:n_peaks]
        else:
            idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
            w = np.zeros(len(conv))
            # Exponential weighting based on peak height
            pv = conv[idx]
            w[idx] = np.exp(pv - pv.max())
            w /= w.sum()
        
        # Compute pressure
        pressure = pressure_matrix(f, w, idx)
        
        # Multiplicative update
        f = f * np.exp(-lr * pressure)
        f = np.maximum(f, 1e-30)  # Avoid zeros
        f = f / f.sum()
        
        # Adaptive lr
        if no_improve > 50:
            lr *= 0.95
            if lr < 1e-5:
                break
        elif no_improve == 0 and lr < 0.05:
            lr *= 1.02
    
    return best_f, best_obj

def softmax_grad_descent(f0, time_limit, beta_start=10.0, beta_end=5000.0, 
                         n_peaks=200, lr_init=0.1, use_adam=True):
    """
    Gradient descent on softmax parameterization with LSE-smoothed objective.
    Variables: theta (length n), a = softmax(theta)
    Objective: LSE(beta * conv(a,a)) = (1/beta) * log(sum(exp(beta * b_k)))
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    # Initialize theta from f
    theta = np.log(f + 1e-15) - np.log(f + 1e-15).max()
    
    beta = beta_start
    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    it = 0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Current a from theta
        log_f = theta - theta.max()
        f = np.exp(log_f)
        f = f / f.sum()
        
        conv = fast_convolve(f)
        cur_max = conv.max()
        
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Gradient w.r.t. a: grad_a[m] = 2 * sum_k w[k] * a[k-m]
        # (factor of 2 because b_k = sum a_i * a_{k-i}, and d/dam gives two terms)
        # Actually d(b_k)/d(a_m) = a[k-m] + a[k-m] = 2*a[k-m] (when both indices valid)
        # But more carefully: b_k = sum_i a_i * a_{k-i}
        # d(b_k)/d(a_m) = a_{k-m} (from i=m) + a_{k-m} (from k-i=m, i=k-m)
        # = 2 * a_{k-m} if 0 <= k-m < n and m != k-m
        # = a_{k-m} if k-m = m (i.e., k = 2m)
        
        # For simplicity, compute the gradient numerically or use the pressure approach
        # The gradient of LSE w.r.t. a is: grad_a = sum_k w[k] * grad(b_k)/grad(a)
        # = sum_k w[k] * 2 * a[k-m] (approximately, ignoring the diagonal)
        
        # Gradient w.r.t. theta: grad_theta = (I - f*f^T) @ grad_a
        # Since a = softmax(theta), da/dtheta = diag(a) - a*a^T
        
        # Compute grad_a
        n_conv = len(conv)
        idx = np.argsort(w)[::-1][:n_peaks]
        grad_a = np.zeros(n)
        for k in idx:
            wk = w[k]
            if wk < 1e-18:
                continue
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                ms = np.arange(lo, hi + 1)
                # d(b_k)/d(a_m) = a[k-m] + (a[k-m] if k-m != m else 0)
                # = 2*a[k-m] - a[k-m]*delta(k-2m) ... let's just use 2*a[k-m]
                grad_a[ms] += 2.0 * wk * f[k - ms]
        
        # Gradient w.r.t. theta
        grad_theta = grad_a * f - f * np.dot(grad_a, f)
        
        # Adam update
        adam_t += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta**2
        m_hat = m / (1 - beta1**adam_t)
        v_hat = v / (1 - beta2**adam_t)
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Schedule
        if beta < beta_end:
            beta *= 1.005
        if no_improve > 50:
            lr *= 0.95
        if no_improve > 200:
            lr = max(lr * 0.8, 0.001)
    
    return best_f, best_obj

def peak_shave_coordinate(f0, time_limit, n_trials=5000):
    """
    Coordinate descent: try moving mass from one position to another.
    Accept if max(conv) decreases.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < n_trials:
        # Try a random coordinate move
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Move delta from i to j
        delta = f[i] * rng.uniform(0.05, 0.3)
        if delta > f[i]:
            delta = f[i] * 0.1
        
        f_new = f.copy()
        f_new[i] -= delta
        f_new[j] += delta
        
        if f_new[i] < 0:
            continue
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:

def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_pressure(f, w, peak_indices):
    """
    Compute pressure_m = sum_k w[k] * f[k-m] for valid indices.
    Only iterates over peak_indices for efficiency.
    """
    n = len(f)
    pressure = np.zeros(n)
    for k in peak_indices:
        wk = w[k]
        if wk < 1e-18:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            pressure[ms] += wk * f[k - ms]
    return pressure


def spectral_shave(f0, time_limit, n_iter_max=50000, lam=0.5, alpha=1.0):
    """
    Spectral peak shaving: iteratively flatten the auto-convolution in frequency domain.
    a_new = IFFT(FFT(a) / (1 + lam*b)^{alpha/2})[:n], then project to simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()

    X = np.fft.rfft(f, fft_size)
    X_abs = np.abs(X)
    X_phase = X / np.maximum(X_abs, 1e-30)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lam = lam

    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1
        # Compute full circular auto-convolution
        P = X_abs ** 2
        b_full = np.fft.irfft(P, fft_size)

        # Damp peaks
        damp = 1.0 + cur_lam * b_full
        damp = np.maximum(damp, 1e-10)
        scale = 1.0 / np.sqrt(damp) ** alpha

        # Update spectrum
        X = X_phase * X_abs * scale
        # Numerical cleanup: remove tiny imaginary parts
        f_new = np.fft.irfft(X, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        f = f_new
        X = np.fft.rfft(f, fft_size)
        X_abs = np.abs(X)
        X_phase = X / np.maximum(X_abs, 1e-30)

        # Adapt lambda
        if no_improve > 100:
            cur_lam *= 0.97
            if cur_lam < 0.01:
                break

    return best_f, best_obj


def multiplicative_descent(f0, time_limit, n_peaks=50, lr=0.01, use_lse=False, lse_beta=100.0):
    """
    Multiplicative weight descent: a_m <- a_m * exp(-lr * pressure_m)
    where pressure_m = sum_k w[k] * a[k-m]
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_max = conv.max()

        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        # Compute peak weights
        if use_lse:
            bc = lse_beta * conv
            bc_max = bc.max()
            exp_bc = np.exp(bc - bc_max)
            w = exp_bc / exp_bc.sum()
            idx = np.argsort(w)[::-1][:n_peaks]
        else:
            idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
            w = np.zeros(len(conv))
            pv = conv[idx]
            w[idx] = np.exp(pv - pv.max())
            w /= w.sum()

        # Compute pressure
        pressure = compute_pressure(f, w, idx)

        # Multiplicative update
        f = f * np.exp(-cur_lr * pressure)
        f = np.maximum(f, 1e-30)
        f = f / f.sum()

        # Adaptive learning rate
        if no_improve > 50:
            cur_lr *= 0.96
            if cur_lr < 1e-6:
                break
        elif no_improve == 0 and cur_lr < 0.05:
            cur_lr *= 1.01

    return best_f, best_obj


def softmax_adam_optimize(f0, time_limit, beta_start=10.0, beta_end=5000.0,
                          n_peaks=200, lr_init=0.1, beta1=0.9, beta2=0.999):
    """
    Adam optimizer on softmax parameterization with LSE-smoothed objective.
    Uses correct gradient: dL/dtheta_m = 2*a_m*(pressure_m - C)
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    # Initialize theta from f
    theta = np.log(f + 1e-15) - np.log(f + 1e-15).max()

    beta = beta_start
    t0 = time.monotonic()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    it = 0

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    lr = lr_init
    eps = 1e-8

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Current a from theta
        log_f = theta - theta.max()
        f = np.exp(log_f)
        f = f / f.sum()

        conv = fast_convolve(f)
        cur_max = conv.max()

        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                break

        # LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Select top peaks for efficiency
        idx = np.argsort(w)[::-1][:n_peaks]

        # Compute pressure and C
        pressure = compute_pressure(f, w, idx)
        C = np.dot(w[idx], conv[idx])

        # Gradient: dL/dtheta_m = 2*a_m*(pressure_m - C)
        grad_theta = 2.0 * f * (pressure - C)

        # Adam update
        adam_t += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** adam_t)
        v_hat = v / (1 - beta2 ** adam_t)
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Schedule
        if beta < beta_end:
            beta *= 1.003
        if no_improve > 40:
            lr *= 0.97
        if no_improve > 150:
            lr = max(lr * 0.9, 0.005)
            if no_improve > 250:
                # Restart from best
                f_best = best_f.copy()
                theta = np.log(f_best + 1e-15) - np.log(f_best + 1e-15).max()
                m = np.zeros(n)
                v = np.zeros(n)
                adam_t = 0
                lr = lr_init * 0.5
                no_improve = 0

    return best_f, best_obj
