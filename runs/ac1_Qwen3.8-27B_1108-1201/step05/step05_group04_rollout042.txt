import numpy as np
import time

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_obj(a, beta):
    """LSE approximation to max of auto-convolution."""
    conv = fast_convolve(a)
    return np.logsumexp(beta * conv) / beta, conv

def compute_lse_gradient(a, beta):
    """
    Compute gradient of LSE-smoothed max auto-convolution.
    Uses FFT for O(n log n) computation.
    Returns (grad, conv, weights).
    """
    n = len(a)
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    
    # Auto-convolution
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:conv_size]
    
    # Softmax weights
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Gradient via cross-correlation
    # grad[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # = 2 * (convolve(w, a[::-1]))[i + n - 1]
    a_rev = a[::-1]
    corr_len = conv_size + n - 1  # = 3n - 2
    corr_fft = 1 << (corr_len - 1).bit_length()
    
    Wf = np.fft.rfft(w, corr_fft)
    Af = np.fft.rfft(a_rev, corr_fft)
    corr = np.fft.irfft(Wf * Af, corr_fft)[:corr_len]
    
    grad = 2.0 * corr[n-1:2*n-1]
    
    return grad, conv, w

def nesterov_lse(a0, time_limit, beta_start=10.0, beta_end=10000.0, verbose=False):
    """LSE-smoothed Nesterov accelerated gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    # Nesterov state
    y = a.copy()
    t_nest = 1.0
    
    # Adaptive step size
    step = 1.0 / (4.0 * beta + 2.0)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        grad, conv_y, w = compute_lse_gradient(y, beta)
        
        # Gradient step
        z = y - step * grad
        x_new = project_simplex(z)
        
        # Check improvement
        conv_new = fast_convolve(x_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        y = x_new + ((t_nest - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_s = y.sum()
        if y_s > 1e-15:
            y /= y_s
        a = x_new
        t_nest = t_new
        
        # Adapt beta
        if no_improve >= 20 and beta < beta_end:
            beta *= 1.3
            step = 1.0 / (4.0 * beta + 2.0)
            t_nest = 1.0
            y = best_a.copy()
            no_improve = 0

            t_nest = 1.0

def multiplicative_weight_step(a, beta, eta):
    """Mirror descent step using multiplicative weights."""
    n = len(a)
    conv = fast_convolve(a)
    bc = beta * conv
    bc -= bc.max()
    exp_bc = np.exp(bc)
    w = exp_bc / exp_bc.sum()
    
    # Gradient (same as LSE gradient but without the 1/beta factor)
    a_rev = a[::-1]
    conv_size = 2 * n - 1
    corr_len = conv_size + n - 1
    corr_fft = 1 << (corr_len - 1).bit_length()
    Wf = np.fft.rfft(w, corr_fft)
    Af = np.fft.rfft(a_rev, corr_fft)
    corr = np.fft.irfft(Wf * Af, corr_fft)[:corr_len]
    grad = 2.0 * corr[n-1:2*n-1]
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

def interpolate_to_length(f, new_n):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based LSE gradient (O(n log n) per iteration) - key speedup
2. Nesterov accelerated gradient descent with adaptive step
3. Multiplicative weight (mirror descent) alternation
4. Directed peak-reduction moves
5. Multi-length exploration with adaptive time allocation
Key improvement: FFT-based gradient via cross-correlation allows 10x more iterations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, beta):
    """
    Compute gradient of LSE-smoothed max auto-convolution via FFT.
    grad[i] = 2 * sum_j w[i+j] * a[j]
    Returns (grad, conv, weights).
    """
    n = len(a)
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()

    # Auto-convolution
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:conv_size]

    # Softmax weights (numerically stable)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w_sum = exp_bc.sum()
    if w_sum < 1e-300:
        w = np.ones(conv_size) / conv_size
    else:
        w = exp_bc / w_sum

    # Gradient via cross-correlation using FFT
    # grad[i] = 2 * (convolve(w, a_reversed))[i + n - 1]
    a_rev = a[::-1]
    corr_len = conv_size + n - 1  # 3n - 2
    corr_fft = 1 << (corr_len - 1).bit_length()

    Wf = np.fft.rfft(w, corr_fft)
    Af = np.fft.rfft(a_rev, corr_fft)
    corr = np.fft.irfft(Wf * Af, corr_fft)[:corr_len]

    grad = 2.0 * corr[n - 1:2 * n - 1]

    return grad, conv, w


def nesterov_lse(a0, time_limit, beta_start=10.0, beta_end=10000.0, verbose=False):
    """LSE-smoothed Nesterov accelerated gradient descent with FFT gradients."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    # Nesterov state
    y = a.copy()
    t_nest = 1.0
    step = 1.0 / (4.0 * beta + 2.0)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute gradient at y (Nesterov lookahead point)
        grad, conv_y, w = compute_lse_gradient(y, beta)

        # Gradient step
        z = y - step * grad
        x_new = project_simplex(z)

        # Evaluate
        conv_new = fast_convolve(x_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        y = x_new + ((t_nest - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_s = y.sum()
        if y_s > 1e-15:
            y /= y_s
        a = x_new
        t_nest = t_new

        # Adapt beta when stuck
        if no_improve >= 30 and beta < beta_end:
            beta = min(beta * 1.5, beta_end)
            step = 1.0 / (4.0 * beta + 2.0)
            t_nest = 1.0
            y = best_a.copy()
            no_improve = 0

        if no_improve > 500:
            rng = np.random.default_rng((it * 7919 + 13) % 999983)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            y = a.copy()
            t_nest = 1.0
            no_improve = 0
            beta = max(beta_start, beta * 0.6)
            step = 1.0 / (4.0 * beta + 2.0)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Nest it={it}, obj={best_obj:.12f}, beta={beta:.1f}, t={elapsed:.0f}s")

    return best_a, best_obj


def multiplicative_optimize(a0, time_limit, beta_start=200.0, eta_start=0.1, verbose=False):
    """Optimize using multiplicative weight (mirror descent) updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    beta = beta_start
    eta = eta_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute gradient
        grad, conv_cur, w = compute_lse_gradient(a, beta)

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        a_s = a_new.sum()
        if a_s > 1e-15:
            a_new /= a_s
        else:
            a_new = a.copy()

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = min(eta * 1.005, 0.5)
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                eta *= 0.92
            if no_improve > 200:
                a = best_a.copy()
                eta = eta_start
                no_improve = 0

        # Slowly increase beta
        if it % 50 == 0 and beta < 10000:
            beta *= 1.05

        if verbose and it % 500 == 0:
            print(f"    MultWt it={it}, obj={best_obj:.12f}, beta={beta:.1f}, eta={eta:.4f}")

    return best_a, best_obj


def peak_targeting(a0, time_limit, n_peaks=10, verbose=False):
    """Directly target and reduce the top peaks using weighted gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv_size = 2 * n - 1

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    curr_n_peaks = n_peaks

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)

        # Select top peaks with sharp weighting
        k_top = min(curr_n_peaks, conv_size)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peak_idx]
        pv_max = pv.max()

        # Exponential weighting emphasizing the highest
        scale = pv / (pv_max + 1e-30)
        w_peaks = np.exp(5.0 * (scale - 1.0))  # sharpest for the max
        w = np.zeros(conv_size)
        w[peak_idx] = w_peaks
        w_sum = w.sum()
        if w_sum < 1e-30:
            w[peak_idx[np.argmax(pv)]] = 1.0
        else:
            w /= w_sum

        # Compute weighted gradient (same FFT correlation)
        a_rev = a[::-1]
        corr_len = conv_size + n - 1
        corr_fft = 1 << (corr_len - 1).bit_length()
        Wf = np.fft.rfft(w, corr_fft)
        Af = np.fft.rfft(a_rev, corr_fft)
        corr = np.fft.irfft(Wf * Af, corr_fft)[:corr_len]
        grad = 2.0 * corr[n - 1:2 * n - 1]

        # Try multiple step sizes
        best_trial_obj = best_obj
        best_trial = None
        for lr in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_trial = project_simplex(a - lr * grad)
            conv_trial = fast_convolve(a_trial)
            obj_trial = conv_trial.max()
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_trial = a_trial

        if best_trial is not None:
            a = best_trial
            if best_trial_obj < best_obj - 1e-15:
                best_obj = best_trial_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                curr_n_peaks = min(curr_n_peaks + 3, 50)
            if no_improve > 200:
                rng = np.random.default_rng((it * 31627 + 7) % 999983)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                curr_n_peaks = n_peaks
                no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"    PeakTgt it={it}, obj={best_obj:.12f}, n_peaks={curr_n_peaks}")

    return best_a, best_obj


def symmetric_nesterov(a0, time_limit, beta_start=20.0, beta_end=20000.0):
    """Nesterov LSE but restricted to symmetric sequences for exploration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    # Symmetrize
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0.0)
    a_sym /= a_sym.sum()

    conv = fast_convolve(a_sym)
    best_obj = conv.max()
    best_a = a_sym.copy()

    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    y = a_sym.copy()
    t_nest = 1.0
    step = 1.0 / (4.0 * beta + 2.0)

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv_y, w = compute_lse_gradient(y, beta)

        # Symmetrize gradient
        grad_sym = (grad + grad[::-1]) / 2.0

        z = y - step * grad_sym
        x_new = project_simplex(z)
        # Symmetrize
        x_new = (x_new + x_new[::-1]) / 2.0
        x_new = np.maximum(x_new, 0.0)
        x_s = x_new.sum()
        if x_s > 1e-15:
            x_new /= x_s

        conv_new = fast_convolve(x_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        y = x_new + ((t_nest - 1.0) / t_new) * (x_new - a_sym)
        y = (y + y[::-1]) / 2.0
        y = np.maximum(y, 0.0)
        y_s = y.sum()
        if y_s > 1e-15:
            y /= y_s
        a_sym = x_new
        t_nest = t_new

        if no_improve >= 40 and beta < beta_end:
            beta = min(beta * 1.4, beta_end)
            step = 1.0 / (4.0 * beta + 2.0)
            t_nest = 1.0
            y = best_a.copy()
            no_improve = 0

        if no_improve > 400:
            rng = np.random.default_rng((it * 104729 + 3) % 999983)
            pert = rng.standard_normal(n)
            pert = (pert + pert[::-1]) / 2.0
            a_sym = project_simplex(best_a + pert * 0.003)
            a_sym = (a_sym + a_sym[::-1]) / 2.0
            a_sym = np.maximum(a_sym, 0.0)
            a_sym /= a_sym.sum()
            y = a_sym.copy()
            t_nest = 1.0
            no_improve = 0

    return best_a, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    f_new_s = f_new.sum()
    if f_new_s > 1e-15:
        f_new /= f_new_s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def cosine_init(n, freq=1.0, skew=0.0):
    """Create a cosine-bell initialization with optional asymmetry."""
    x = np.arange(n) / (n - 1)
    # Cosine bell
    base = 0.5 * (1 + np.cos(2 * np.pi * x))
    # Add skew for asymmetry
    if skew != 0.0:
        base = base * (1 + skew * (2 * x - 1))
    base = np.maximum(base, 0.0)
    s = base.sum()
    if s > 1e-15:
        base /= s
    else:
        base = np.ones(n) / n
    return base
