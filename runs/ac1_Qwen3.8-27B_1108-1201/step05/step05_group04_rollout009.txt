"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated exact subgradient descent
2. LSE-smoothed gradient with FFT-based weighted gradient computation
3. LP-based Frank-Wolfe for global structure
4. Multi-start random perturbations to escape local minima
5. Multi-length exploration with interpolation
6. Coordinate-wise peak reduction as a final polish
Key innovations: FFT-based weighted gradient, Nesterov momentum,
aggressive beta scheduling, and coordinate descent polish.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weighted_grad_fft(f, w):
    """Compute gradient of sum_k w[k]*(f*f)[k] using FFT cross-correlation."""
    n = len(f)
    conv_len = 2 * n - 1
    # Need N >= (2n-1) + n - 1 = 3n-2
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    corr = np.fft.irfft(W * np.conj(F), N)
    return 2.0 * corr[:n]


def compute_subgrad_at_peak(f, k):
    """Compute exact subgradient of (f*f)[k] w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[lo:hi + 1] = 2.0 * f[k - idx]
    return grad


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def nesterov_subgrad(f0, time_limit, lr_init=None, momentum=0.93):
    """Nesterov-accelerated subgradient descent on exact peak subgradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if lr_init is None:
        lr_init = 1.5 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    no_improve = 0
    it = 0
    cur_lr = lr_init
    cur_mom = momentum

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov lookahead
        y = f + cur_mom * v

        conv_y = fast_convolve(y)
        k_star = int(np.argmax(conv_y))
        grad = compute_subgrad_at_peak(y, k_star)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad /= grad_norm

        v = cur_mom * v - cur_lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if no_improve == 0 and it % 50 == 0:
                cur_lr = min(cur_lr * 1.001, 3.0 / n)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                cur_lr *= 0.97
                cur_mom = min(cur_mom + 0.001, 0.98)
            if no_improve > 200:
                f = best_f.copy()
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.0001)
                no_improve = 0

    return best_f, best_obj


def lse_gradient_descent(f0, time_limit, beta_start=30.0, beta_end=30000.0):
    """LSE-smoothed gradient descent with FFT-based gradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 1.0 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_weighted_grad_fft(f, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if it % 200 == 0:
                lr = min(lr * 1.001, 2.5 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.96
            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                beta = min(beta * 2.0, beta_end)
                lr = max(lr * 0.7, 0.0005)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.005

    return best_f, best_obj


def fw_lp(f0, time_limit, max_iters=None):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        if max_iters is not None and it >= max_iters:
            break
        it += 1
        A_cache = build_lp_matrix(f)
        g = solve_lp_direction(f, M * m_exp, A_cache)
        if g is None:
            m_exp = min(m_exp * 1.2, 5.0)
            continue

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 15:
            break

    return best_f, best_obj


def coordinate_polish(f0, time_limit, n_coords_per_iter=200):
    """Coordinate-wise optimization: for each coordinate, find best value."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))
        current_max = conv[k_star]

        # Pick coordinates to try: those with large |subgradient| at peak
        subgrad = compute_subgrad_at_peak(f, k_star)
        # Also check neighboring peaks
        top_peaks = np.argpartition(conv, -10)[-10:]
        combined_grad = np.zeros(n)
        for pk in top_peaks:
            combined_grad += compute_subgrad_at_peak(f, int(pk))

        # Select top coordinates by gradient magnitude
        n_try = min(n_coords_per_iter, n)
        if n_try < n:
            idx = np.argpartition(np.abs(combined_grad), -n_try)[-n_try:]
        else:
            idx = np.arange(n)
        rng.shuffle(idx)

        improved = False
        for j in idx:
            if time.monotonic() - t0 > time_limit:
                break
            # Try decreasing f[j] (since subgradient suggests we should decrease)
            # Binary search for the maximum decrease that doesn't increase the max
            lo = 0.0
            hi = f[j] * 0.5
            for _ in range(15):
                mid = (lo + hi) / 2.0
                f[j] = f[j] - mid
                if f[j] < 0:
                    f[j] = 0
                conv_test = fast_convolve(f)
                if conv_test.max() < current_max - 1e-15:
                    lo = mid
                    improved = True
                else:
                    hi = mid
            # Restore and apply best
            f[j] = f[j] + lo - lo  # already applied
            # Actually, let's redo this more carefully
            f[j] = best_f[j]  # reset to best
            # Recompute from scratch
            f[j] = best_f[j] - lo
            if f[j] < 0:

        conv_new = fast_convolve(f)
        if conv_new.max() < best_obj - 1e-14:

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    # Use a smooth interpolation
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def make_perturbed_init(f0, scale=0.02, seed=0):

    f_new = np.maximum(f0 + perturbation, 0.0)

            print(f"  FW-LP pass 2 n={len(current)}, t={t_alloc:.0f}s", flush=True)
            f_opt, _ = fw_lp(current, time_limit=t_alloc, max_iters=8)
            try_sequence(f_opt, best_tracker)

    # === Phase 5: Multi-length exploration (~250s) ===
    print("\n=== Phase 5: Multi-Length Exploration ===", flush=True)
    t5_end = min(deadline - 150, time.monotonic() + 250)
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current)

    offsets = [32, -32, 64, -64, 128, -128, 16, -16]
    tried_lengths = set([n_best])
    for off in offsets:
        n_try = n_best + off
        if n_try < 200 or n_try > 1500 or n_try in tried_lengths:
            continue
        t_remaining = t5_end - time.monotonic()
        if t_remaining < 40:
            break
        tried_lengths.add(n_try)
        t_alloc = min(60, t_remaining * 0.4)
        if t_alloc < 15:
            break

        f_init = interpolate_to_length(current, n_try)
        print(f"  LSE n={n_try}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = lse_gradient_descent(f_init, time_limit=t_alloc,
                                         beta_start=20.0, beta_end=10000.0)
        try_sequence(f_opt, best_tracker)
        # Quick Nesterov
        t_alloc2 = min(20, t5_end - time.monotonic() - 10)
        if t_alloc2 > 10:
            f_opt2, _ = nesterov_subgrad(f_opt, time_limit=t_alloc2,
                                          lr_init=2.0 / n_try, momentum=0.93)
            try_sequence(f_opt2, best_tracker)

    # === Phase 6: Final intensive polish (~150s) ===
    print("\n=== Phase 6: Final Polish ===", flush=True)
    t6_end = min(deadline - 30, time.monotonic() + 150)
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    # High-beta LSE
    t_alloc = min(80, t6_end - time.monotonic() - 20)
    if t_alloc > 15:
        print(f"  High-beta LSE n={n_cur}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = lse_gradient_descent(current, time_limit=t_alloc,
                                         beta_start=500.0, beta_end=100000.0)
        try_sequence(f_opt, best_tracker)

    # Nesterov with small learning rate
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(50, t6_end - time.monotonic() - 10)
    if t_alloc > 10:
        print(f"  Nesterov fine-tune n={len(current)}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = nesterov_subgrad(current, time_limit=t_alloc,
                                     lr_init=0.5 / len(current), momentum=0.95)
        try_sequence(f_opt, best_tracker)

    # Coordinate polish
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(40, t6_end - time.monotonic() - 5)
    if t_alloc > 10:
        print(f"  Coordinate polish n={len(current)}, t={t_alloc:.0f}s", flush=True)
        f_opt, _ = coordinate_polish(current, time_limit=t_alloc, n_coords_per_iter=100)
        try_sequence(f_opt, best_tracker)

    # === Phase 7: Final alternating micro-iterations ===
    print("\n=== Phase 7: Final Micro-Iterations ===", flush=True)
    current = best_tracker[1] if best_tracker[1] is not None else parent
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Alternate between very short LSE and Nesterov
        for cycle in range(6):
            t_rem = deadline - time.monotonic()
            if t_rem < 15:
                break
            t_alloc = min(25, t_rem * 0.4)
            if t_alloc < 5:
                break
            if cycle % 2 == 0:
                f_opt, _ = lse_gradient_descent(current, time_limit=t_alloc,
                                                 beta_start=1000.0, beta_end=50000.0)
            else:
                f_opt, _ = nesterov_subgrad(current, time_limit=t_alloc,
                                             lr_init=0.3 / len(current), momentum=0.96)
            try_sequence(f_opt, best_tracker)
            current = best_tracker[1] if best_tracker[1] is not None else current

    # === Final verification and return ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        # Scale to maximize the original score (the evaluator clips to [0,1000]
        # and uses sum^2 in denominator, so scaling doesn't matter)
        # But we should keep values reasonable
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}", flush=True)
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using parent fallback", flush=True)
    return [float(x) for x in parent]
