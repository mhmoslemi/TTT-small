from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_of_max_conv(f, conv, n, top_k, temperature):
    """Compute subgradient of max(conv) using softmax-weighted top-k peaks."""
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    pv_range = pv.max() - pv.min()
    if pv_range < 1e-15:
        w = np.ones(len(peaks)) / len(peaks)
    else:
        scaled = (pv - pv.max()) / (temperature * pv_range + 1e-15)
        w = np.exp(scaled)
        w /= w.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w[i] * f[k - indices]
    return sg

def interpolate_to(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def raised_cosine_init(n, order=1):
    """Raised cosine window of given order."""
    j = np.arange(n)
    t = j / (n - 1)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = a ** order
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def gaussian_init(n, sigma_factor=0.3):
    """Gaussian-shaped initialization."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def hamming_init(n):
    """Hamming window initialization."""
    j = np.arange(n)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * j / (n - 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def blackman_init(n):
    """Blackman window initialization."""
    j = np.arange(n)
    M = n - 1
    a = 0.42 - 0.5 * np.cos(2 * np.pi * j / M) + 0.08 * np.cos(4 * np.pi * j / M)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
