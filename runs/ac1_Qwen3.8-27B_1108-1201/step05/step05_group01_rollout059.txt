```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Griffin-Lim spectral flattening,
projected gradient descent with correct autoconvolution gradients, and LP-Frank-Wolfe
polishing across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def griffin_lim_nonneg(a_init, n_iter=2000, target_type='flat', omega=1.0, seed=0):
    """
    Griffin-Lim algorithm with non-negativity and support constraints.
    Alternates between spectral shaping and time-domain projection.
    """
    n = len(a_init)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(n_iter):
        # Compute FFT
        A = np.fft.fft(a, fft_size)
        
        # Target magnitude
        if target_type == 'flat':
            # A[0] = sum(a) = 1, target all |A[k]| = 1
            target_mag = np.ones(fft_size)
        elif target_type == 'tapered':
            # Slightly tapered target to help convergence
            k = np.arange(fft_size)
            # Symmetric taper
            target_mag = 1.0 + 0.05 * np.cos(2 * np.pi * k / fft_size)
        else:
            target_mag = np.ones(fft_size)
        
        # Update magnitude, keep phase
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        A_new = target_mag * phase
        # Force DC component
        A_new[0] = A[0]  # Keep DC (real and positive since a >= 0)
        
        # Inverse FFT
        a_full = np.fft.ifft(A_new).real
        
        # Project: keep first n samples, clip to non-negative
        a_candidate = a_full[:n]
        a_candidate = np.maximum(a_candidate, 0.0)
        
        # Over-relaxation
        if omega != 1.0:
            a_candidate = omega * a_candidate + (1 - omega) * a
            a_candidate = np.maximum(a_candidate, 0.0)
        
        # Renormalize
        s = a_candidate.sum()
        if s < 1e-15:
            break
        a_new = a_candidate / s
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def gradient_refine(a_init, time_limit, lr0=0.001, decay=0.999, n_peaks=5):
    """
    Projected gradient descent minimizing max(conv(a,a)) with sum(a)=1, a>=0.
    Uses the exact subgradient at the active peaks.
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = lr0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            lr = lr0 * 0.5
            no_improve = 0
            conv = fast_convolve(a)
            obj = conv.max()
        
        # Find top n_peaks lags
        k_actual = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in peaks:
            k = int(k)
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                grad[indices] += 2.0 * a[k - indices]
        
        # Scale gradient: since all peaks contribute equally to the max,
        # we use the average (or just sum, since we're using it as a direction)
        # Actually for minimizing max, the subgradient is the average of active gradients
        # But for simplicity, let's just use the sum (direction is the same up to scale)
        
        # Projected gradient step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        # Line search (simple: try a few step sizes)
        best_lr = lr
        best_val = obj
        for lr_try in [lr * 0.5, lr, lr * 2, lr * 0.1]:
            a_try = a - lr_try * grad
            a_try = project_simplex(a_try)
            val_try = fast_convolve(a_try).max()
            if val_try < best_val:
                best_val = val_try
                best_lr = lr_try
        
        a = a - best_lr * grad
        a = project_simplex(a)
        
        lr *= decay
    
    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with proper step sizes."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

        f = f / s
    n = len(f)

    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 60:
                break
            m_exp = min(m_exp * 1.04, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 2.0/(it+2)]
        
        for alpha in alphas:

        if best_alpha > 1e-10:

            f_new = np.maximum(f_new, 0.0)

            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new
            
            if M < best_obj - 1e-14:

            else:

        else:

            m_exp = 1.0

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Griffin-Lim spectral flattening
(POCS-style alternating projections), exact subgradient descent for the max-convolution,
LP-Frank-Wolfe polishing, and multi-length exploration with random-phase initializations.
Key innovation: treats the problem as finding a non-negative sequence with flat autoconvolution
spectrum via iterative spectral shaping + time-domain projection.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def griffin_lim_nonneg(a_init, n_iter=2000, target_type='flat', omega=1.0, seed=0):
    """
    Griffin-Lim algorithm with non-negativity and support constraints.
    Alternates between spectral magnitude shaping and time-domain projection.
    """
    n = len(a_init)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    a = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for it in range(n_iter):
        # Compute FFT
        A = np.fft.fft(a, fft_size)

        # Target magnitude
        if target_type == 'flat':
            target_mag = np.ones(fft_size)
        elif target_type == 'tapered':
            k = np.arange(fft_size)
            target_mag = 1.0 + 0.05 * np.cos(2.0 * np.pi * k / fft_size)
        elif target_type == 'flat_dc0':
            target_mag = np.ones(fft_size)
            target_mag[0] = 0.0  # will be overridden anyway
        else:
            target_mag = np.ones(fft_size)

        # Update magnitude, keep phase
        mag = np.abs(A)
        safe_mag = np.maximum(mag, 1e-30)
        phase = A / safe_mag
        A_new = target_mag * phase
        # Force DC component (A[0] = sum(a) = 1, real and positive)
        A_new[0] = 1.0 + 0.0j

        # Inverse FFT
        a_full = np.fft.ifft(A_new).real

        # Project: keep first n samples, clip to non-negative
        a_candidate = a_full[:n]
        a_candidate = np.maximum(a_candidate, 0.0)

        # Over-relaxation
        if abs(omega - 1.0) > 1e-6:
            a_candidate = omega * a_candidate + (1.0 - omega) * a
            a_candidate = np.maximum(a_candidate, 0.0)

        # Renormalize
        s = a_candidate.sum()
        if s < 1e-15:
            break
        a_new = a_candidate / s

        # Periodically check improvement (not every iteration for speed)
        if it % 50 == 0 or it == n_iter - 1:
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def random_phase_init(n, seed=0):
    """
    Generate initialization from flat magnitude spectrum with random phases.
    This gives a sequence that approximately has flat autoconvolution spectrum.
    """
    rng = np.random.default_rng(seed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Random phases with conjugate symmetry for real output
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    # Enforce conjugate symmetry: A[k] = conj(A[fft_size-k])
    A = np.exp(1j * phases)
    A[0] = 1.0 + 0.0j  # DC
    if fft_size % 2 == 0:
        A[fft_size // 2] = np.abs(A[fft_size // 2]) + 0.0j
    # Enforce conjugate symmetry
    for k in range(1, fft_size // 2 + 1):
        A[fft_size - k] = np.conj(A[k])

    a_full = np.fft.ifft(A).real
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def gradient_refine(a_init, time_limit, lr0=0.001, decay=0.999, n_peaks=5):
    """
    Projected subgradient descent minimizing max(conv(a,a)) with sum(a)=1, a>=0.
    Uses exact subgradient: d conv[k]/d a[m] = 2*a[k-m] if 0<=k-m<n.
    """
    a = np.maximum(a_init.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    lr = lr0
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            scale = 0.005 * min(1 + no_improve // 300, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            lr = max(lr0 * 0.3, 0.0001)
            no_improve = 0
            continue

        # Find top n_peaks lags (with softmax weighting for smooth subgradient)
        k_actual = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        # Softmax weighting
        temp = 0.01 * obj + 1e-10
        pv = conv[peaks] / temp
        pv -= pv.max()
        w = np.exp(pv)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                grad[indices] += w[i] * 2.0 * a[k - indices]

        # Projected gradient step with line search
        best_lr = lr
        best_val = obj
        for lr_try in [lr * 0.5, lr, lr * 2.0, lr * 0.1]:
            a_try = a - lr_try * grad
            a_try = project_simplex(a_try)
            val_try = fast_convolve(a_try).max()
            if val_try < best_val:
                best_val = val_try
                best_lr = lr_try

        a = a - best_lr * grad
        a = project_simplex(a)
        lr *= decay

    return best_a, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.6f}")

    # === Phase 1: Griffin-Lim at multiple lengths with diverse inits ===
    print(f"\n=== Phase 1: Griffin-Lim multi-length (best={best_score:.6f}) ===")

    target_lengths = [768, 1024, 1280, 1536, 1800, 2048]

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 500
        if rem < 10:
            break

        inits = []
        # Random phase inits (key innovation)
        for rs in [0, 1, 2, 3, 4]:
            inits.append(random_phase_init(target_n, seed=seed + rs * 100))
        # Cosine modulated
        for freq in [1, 2, 3]:
            inits.append(cosine_modulated_init(target_n, freq, 0.4))
        # Gaussian window
        inits.append(gaussian_window_init(target_n, target_n / 4.0))
        # Interpolated parent
        inits.append(interpolate_sequence(parent, target_n))
        # Random uniform
        inits.append(rng.random(target_n))

        best_gl_obj = float('inf')
        best_gl_a = None

        for i_init, a_init in enumerate(inits):
            rem = deadline - time.monotonic() - 450
            if rem < 3:
                break

            a_opt, obj = griffin_lim_nonneg(a_init, n_iter=800, target_type='flat',
                                            omega=1.0, seed=seed + i_init)
            if obj < best_gl_obj:
                best_gl_obj = obj
                best_gl_a = a_opt.copy()

            # Second stage: over-relaxed refinement
            a_opt2, obj2 = griffin_lim_nonneg(a_opt, n_iter=400, target_type='flat',
                                              omega=1.4, seed=seed + 200 + i_init)
            if obj2 < best_gl_obj:
                best_gl_obj = obj2
                best_gl_a = a_opt2.copy()

        if best_gl_a is not None:
            try_seq(best_gl_a, f"gl_n{target_n}")

    # === Phase 2: Further Griffin-Lim refinement on best ===
    print(f"\n=== Phase 2: GL refinement (best={best_score:.6f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        for tt in ['flat', 'tapered']:
            rem = deadline - time.monotonic() - 400
            if rem < 10:
                break
            a_opt, obj = griffin_lim_nonneg(best_seq, n_iter=1500, target_type=tt,
                                            omega=1.0)
            try_seq(a_opt, f"gl_{tt}_refine")

            # Over-relaxed
            a_opt2, obj2 = griffin_lim_nonneg(a_opt, n_iter=800, target_type=tt,
                                              omega=1.5)
            try_seq(a_opt2, f"gl_{tt}_overrel")

    # === Phase 3: Gradient refinement ===
    print(f"\n=== Phase 3: Gradient refinement (best={best_score:.6f}) ===")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            t_grad = min(rem * 0.5, 120)
            print(f"  Gradient refine t={t_grad:.0f}s, n={len(best_seq)}")
            a_opt, _ = gradient_refine(best_seq, time_limit=t_grad,
                                        lr0=0.002, decay=0.9995, n_peaks=5)
            try_seq(a_opt, "grad_refine")

        # Fine gradient
        rem = deadline - time.monotonic() - 300
        if rem > 15:
            a_opt, _ = gradient_refine(a_opt, time_limit=min(rem, 80),
                                        lr0=0.0005, decay=0.999, n_peaks=8)
            try_seq(a_opt, "grad_fine")

    # === Phase 4: LP-FW polish (if n is manageable) ===
    print(f"\n=== Phase 4: LP-FW polish (best={best_score:.6f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        if n_best <= 1024:
            rem = deadline - time.monotonic() - 200
            if rem > 30:
                t_lp = min(rem * 0.5, 100)
                print(f"  LP-FW t={t_lp:.0f}s, n={n_best}")
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp, verbose=True)
                try_seq(a_opt, "lpfw_polish")

    # === Phase 5: Interpolate to longer lengths + optimize ===
    print(f"\n=== Phase 5: Longer lengths (best={best_score:.6f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        for ext_factor in [2, 1.5]:
            ext_n = int(n_best * ext_factor)
            if ext_n > 3000 or ext_n <= n_best:
                continue
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break

            f_interp = interpolate_sequence(best_seq, ext_n)
            a_opt, obj = griffin_lim_nonneg(f_interp, n_iter=1000, target_type='flat',
                                            omega=1.0)
            try_seq(a_opt, f"gl_ext_n{ext_n}")

            rem = deadline - time.monotonic() - 120
            if rem > 15:
                a_opt2, _ = gradient_refine(a_opt, time_limit=min(rem, 60),
                                             lr0=0.001, decay=0.999, n_peaks=5)
                try_seq(a_opt2, f"grad_ext_n{ext_n}")

    # === Phase 6: Multi-start with random phase + GL + gradient ===
    print(f"\n=== Phase 6: Multi-start (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(6):
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break

            if p == 0:
                f_init = random_phase_init(n_best, seed=seed + 500)
            elif p == 1:
                f_init = 0.7 * best_seq + 0.3 * random_phase_init(n_best, seed=seed + 501)
                f_init = project_simplex(f_init)
            elif p == 2:
                f_init = 0.8 * best_seq + 0.2 * cosine_modulated_init(n_best, 3, 0.3)
                f_init = project_simplex(f_init)
            elif p == 3:
                noise = rng.standard_normal(n_best) * 0.01
                f_init = project_simplex(best_seq + noise)
            elif p == 4:
                f_init = random_phase_init(n_best, seed=seed + 503)
            else:
                f_init = best_seq[::-1].copy()
                f_init = project_simplex(f_init)

            a_opt, obj = griffin_lim_nonneg(f_init, n_iter=600, target_type='flat',
                                            omega=1.0, seed=seed + 600 + p)
            try_seq(a_opt, f"multi_gl_{p}")

            rem = deadline - time.monotonic() - 60
            if rem > 10:
                a_opt2, _ = gradient_refine(a_opt, time_limit=min(rem, 40),
                                             lr0=0.001, decay=0.999, n_peaks=5)
                try_seq(a_opt2, f"multi_grad_{p}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Quick GL
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = griffin_lim_nonneg(best_seq, n_iter=500, target_type='flat',
                                          omega=1.3)
            try_seq(a_opt, "final_gl")

        # Fine gradient
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = gradient_refine(best_seq, time_limit=min(rem, 50),
                                        lr0=0.0003, decay=0.9995, n_peaks=10)
            try_seq(a_opt, "final_grad")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
