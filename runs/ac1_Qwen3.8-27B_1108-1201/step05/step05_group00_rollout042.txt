```python
"""
Hybrid search for low-peak self-convolutions of nonnegative sequences.

Minimizes:
    2 * n * max(conv(a, a)) / (sum(a))^2

using:
- the provided parent as a high-quality anchor,
- active-peak smooth (log-sum-exp) L-BFGS optimization in log-softmax coordinates,
- fast exact pair-transfer coordinate descent for local peak shaving,
- spectral phase-retrieval / box-projection searches for diverse global moves,
- multi-length exploration and final polishing.
"""

import time
import numpy as np
from scipy.optimize import minimize

_BEST_STATE = {
    "a": None,
    "score": float("inf"),
}

_SMOOTH_STATE = {
    "mode": "all",
    "peak_infos": [],
    "T": 1.0,
    "n": 1,
}


def normalize_nonnegative(v):
    """Clip to nonnegative values and normalize to sum 1."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or v.size == 0:
        return np.ones(1, dtype=np.float64)

    v = np.clip(v, 0.0, None)
    s = float(v.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(v.size, dtype=np.float64) / float(v.size)

    out = v / s
    if not np.all(np.isfinite(out)):
        out = np.ones(v.size, dtype=np.float64) / float(v.size)
    return out


def fast_convolve(a):
    """Linear self-convolution via rFFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(A * A, fft_size)[:size]
    return out


def actual_score(a):
    """Internal scale-invariant evaluation, matching evaluate_sequence on normalized input."""
    a = normalize_nonnegative(a)
    n = a.size
    c = fast_convolve(a)
    m = float(np.max(c)) if c.size else 0.0
    if m < 0.0:
        m = 0.0
    return float(2.0 * n * m)


def record_candidate(a, label=""):
    """Evaluate and store a candidate if it improves the global best."""
    a = normalize_nonnegative(a)
    if a.size == 0:
        return
    score = actual_score(a)
    if not np.isfinite(score):
        return
    if _BEST_STATE["a"] is None or score < _BEST_STATE["score"] - 1e-12:
        _BEST_STATE["a"] = a.copy()
        _BEST_STATE["score"] = float(score)
        print(f"NEW BEST {label}: {score:.9f}  n={a.size}")


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / float(n)

    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_normalize(a, n):
    """Linearly resample a normalized sequence to length n."""
    a = normalize_nonnegative(a)
    if a.size == n:
        return a.copy()
    x_in = np.linspace(0.0, 1.0, a.size)
    x_out = np.linspace(0.0, 1.0, n)
    b = np.interp(x_out, x_in, a)
    return normalize_nonnegative(b)


def safe_maxfun(remaining, n, cap):
    """Conservative maxfun estimate based on remaining time and dimension."""
    rem = max(5.0, float(remaining))
    rate = 120.0 * (768.0 / max(1.0, float(n)))
    val = int(rem * rate)
    if val < 80:
        val = 80
    if val > cap:
        val = cap
    return max(80, val)


def top_peaks_of(a, K):
    """Return indices of top K convolution peaks and the convolution."""
    a = normalize_nonnegative(a)
    c = fast_convolve(a)
    K = int(min(K, c.size))
    if K <= 0:
        return np.array([], dtype=np.int64), c
    idx = np.argpartition(c, -K)[-K:]
    idx = idx[np.argsort(c[idx])[::-1]]
    return idx.astype(np.int64, copy=False), c


def build_peak_infos(peaks, n):
    """Precompute slices needed to differentiate selected convolution coefficients."""
    infos = []
    for k in peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            arr = np.arange(js, je + 1, dtype=np.int64)
            target_indices = (k - arr).astype(np.int64, copy=False)
            infos.append((k, int(js), int(je), target_indices))
    return infos


def softmax_from_y(y):
    """Stable softmax on a recentered vector."""
    y = np.asarray(y, dtype=np.float64)
    if y.size == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(y)):
        return np.ones(y.size, dtype=np.float64) / float(y.size)

    y = y - float(np.mean(y))
    m = float(np.max(y))
    e = np.exp(y - m)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(y.size, dtype=np.float64) / float(y.size)
    return e / s


def smooth_fun_grad(y):
    """
    Objective and gradient for a smoothed log-sum-exp surrogate.

    State is read from _SMOOTH_STATE:
      mode='all':   use all convolution peaks
      mode='fixed': use a fixed active set of peaks
    """
    state = _SMOOTH_STATE
    T = float(state["T"])
    n = int(state["n"])
    if n <= 0:
        return (float("inf"), np.array([], dtype=np.float64))

    T = max(T, 1e-12)
    a = softmax_from_y(y)
    c = fast_convolve(a)
    mode = state["mode"]

    if mode == "fixed":
        infos = state["peak_infos"]
        if not infos:
            return (float("inf"), np.zeros(n, dtype=np.float64))

        cvals = np.empty(len(infos), dtype=np.float64)
        for i, info in enumerate(infos):
            cvals[i] = c[info[0]]

        cmax = float(np.max(cvals))
        scaled = (cvals - cmax) / T
        expv = np.exp(scaled)
        denom = float(expv.sum())
        if not np.isfinite(denom) or denom <= 0.0:
            return (float("inf"), np.zeros(n, dtype=np.float64))

        obj = 2.0 * n * T * (cmax + np.log(denom))
        p = expv / denom

        h = np.zeros(n, dtype=np.float64)
        for info_i in range(len(infos)):
            w = float(p[info_i])
            if w < 1e-15:
                continue
            _, js, je, ti = infos[info_i]
            h[js:je + 1] += (2.0 * w) * a[ti]
    else:
        cvals = c
        cmax = float(np.max(cvals))
        scaled = (cvals - cmax) / T
        expv = np.exp(scaled)
        denom = float(expv.sum())
        if not np.isfinite(denom) or denom <= 0.0:
            return (float("inf"), np.zeros(n, dtype=np.float64))

        obj = 2.0 * n * T * (cmax + np.log(denom))
        p = expv / denom
        h = 2.0 * np.correlate(p, a, "valid")

    grad_a = 2.0 * n * h
    grad_y = a * (grad_a - float(np.dot(a, grad_a)))

    if not np.all(np.isfinite(grad_y)):
        return (float("inf"), np.zeros(n, dtype=np.float64))

    # Remove the flat direction coming from softmax scale invariance.
    grad_y = grad_y - float(np.mean(grad_y))
    return (float(obj), grad_y)


def smooth_optimize(a0, peaks, T, maxiter=80, maxfun=1000, mode="fixed"):
    """Run L-BFGS on the log-softmax parameterization using a smooth surrogate."""
    a = normalize_nonnegative(a0)
    n = a.size
    if n == 0:
        return a

    if mode == "all":
        _SMOOTH_STATE["mode"] = "all"
        _SMOOTH_STATE["peak_infos"] = []
    else:
        if peaks is None or len(peaks) == 0:
            _, c = top_peaks_of(a, 50)
            peaks = np.argsort(c)[-50:]
        peaks = np.asarray(peaks, dtype=np.int64)
        peaks = np.unique(peaks)
        infos = build_peak_infos(peaks, n)
        _SMOOTH_STATE["mode"] = "fixed"
        _SMOOTH_STATE["peak_infos"] = infos

    _SMOOTH_STATE["n"] = int(n)
    _SMOOTH_STATE["T"] = float(max(1e-12, T))

    a_safe = np.maximum(a, 1e-10)
    a_safe = a_safe / float(a_safe.sum())
    y0 = np.log(a_safe)
    y0 = y0 - float(np.mean(y0))

    maxiter = int(max(10, maxiter))
    maxfun = int(max(80, maxfun))

    try:
        res = minimize(
            smooth_fun_grad,
            y0,
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter,
                "maxfun": maxfun,
                "ftol": 1e-12,
                "gtol": 1e-7,
                "maxls": 20,
            },
        )
        y1 = res.x
    except Exception:
        return a

    a2 = softmax_from_y(y1)
    return normalize_nonnegative(a2)


def pair_grid_max(c, d, deltas, p, q):
    """Exact maximum of the updated convolution for a grid of pair moves."""
    L = len(c)
    cand = c[None, :] + 2.0 * (deltas[:, None] * d[None, :])
    dd = deltas * deltas

    pos = 2 * p
    if 0 <= pos < L:
        cand[:, pos] += dd
    pos = 2 * q
    if 0 <= pos < L:
        cand[:, pos] += dd
    pos = p + q
    if 0 <= pos < L:
        cand[:, pos] -= 2.0 * dd

    return cand.max(axis=1)


def try_pair_move(a, c, p, q, n, grid_size=21, current_max=None):
    """
    Try transferring mass between coordinates p and q.

    Returns (delta, d, candidate_max).
    If no improving move is found, delta=0 and d=None.
    """
    p = int(p)
    q = int(q)
    if p == q:
        return (0.0, None, float(current_max) if current_max is not None else float(c.max()))

    if current_max is None:
        current_max = float(c.max())

    ap = float(a[p])
    aq = float(a[q])
    span = ap + aq
    eps = 1e-16 * max(1.0, span)

    lo = -ap + eps
    hi = aq - eps
    if hi <= lo:
        return (0.0, None, float(current_max))

    L = 2 * n - 1
    d = np.zeros(L, dtype=np.float64)
    d[p:p + n] = a
    d[q:q + n] -= a

    grid_size = max(7, int(grid_size))
    deltas = np.linspace(lo, hi, grid_size + 1)
    maxes = pair_grid_max(c, d, deltas, p, q)
    idx = int(np.argmin(maxes))
    best_delta = float(deltas[idx])
    best_max = float(maxes[idx])

    step = (hi - lo) / float(max(1, grid_size))
    lo2 = max(lo, best_delta - 2.0 * step)
    hi2 = min(hi, best_delta + 2.0 * step)
    if hi2 > lo2:
        deltas2 = np.linspace(lo2, hi2, grid_size)
        maxes2 = pair_grid_max(c, d, deltas2, p, q)
        idx2 = int(np.argmin(maxes2))
        if float(maxes2[idx2]) < best_max:
            best_delta = float(deltas2[idx2])
            best_max = float(maxes2[idx2])

    if best_max < float(current_max) - 1e-14:
        return (best_delta, d, best_max)
    return (0.0, None, best_max)


def apply_and_check_pair(a, c, p, q, delta, d, old_max, n):
    """Apply a pair move, update c analytically, and keep only if it improves."""
    old_p = float(a[p])
    old_q = float(a[q])

    a[p] = old_p + delta
    a[q] = old_q - delta

    clipped = False
    if a[p] < 0.0:
        if a[p] < -1e-12:
            a[p] = old_p
            a[q] = old_q
            return False, float(c.max())
        a[p] = 0.0
        clipped = True
    if a[q] < 0.0:
        if a[q] < -1e-12:
            a[p] = old_p
            a[q] = old_q
            return False, float(c.max())
        a[q] = 0.0
        clipped = True

    L = len(c)
    c += 2.0 * delta * d

    pos = 2 * p
    if pos < L:
        c[pos] += delta * delta
    pos = 2 * q
    if pos < L:
        c[pos] += delta * delta
    pos = p + q
    if pos < L:
        c[pos] -= 2.0 * delta * delta

    if clipped:
        a[:] = normalize_nonnegative(a)
        c = fast_convolve(a)

    new_max = float(c.max())
    if new_max < float(old_max) - 1e-14:
        return True, new_max

    a[p] = old_p
    a[q] = old_q
    c = fast_convolve(a)
    return False, float(c.max())


def pair_descent(a0, time_limit=60.0, top_k=30, grid_size=21, seed=12345):
    """
    Fast local coordinate descent by exact pair-transfer line searches.

    Pair moves are biased toward the currently largest convolution peaks,
    especially symmetric pairs whose sum is the peak index.
    """
    a = normalize_nonnegative(a0)
    n = a.size
    if n == 0:
        return a

    c = fast_convolve(a)
    current_max = float(c.max())
    best_a = a.copy()
    best_score = float(2.0 * n * current_max)

    deadline = time.monotonic() + max(1.0, float(time_limit))
    rng = np.random.default_rng(seed)

    attempts = 0
    no_improve = 0
    accepts = 0
    peak_refresh = 30

    K_init = min(int(top_k), len(c))
    if K_init <= 0:
        return best_a
    top_peaks = np.argpartition(c, -K_init)[-K_init:]
    top_peaks = top_peaks[np.argsort(c[top_peaks])[::-1]]

    while time.monotonic() < deadline:
        if attempts % peak_refresh == 0:
            c = fast_convolve(a)
            current_max = float(c.max())
            K = min(int(top_k), len(c))
            if K > 0:
                top_peaks = np.argpartition(c, -K)[-K:]
                top_peaks = top_peaks[np.argsort(c[top_peaks])[::-1]]

        # Work on the largest peaks only; enough to be fast yet aggressive.
        for peak in top_peaks[:6]:
            if time.monotonic() >= deadline:
                break

            k = int(peak)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je - js + 1 < 2:
                continue

            arr_i = np.arange(js, je + 1, dtype=np.int64)
            arr_q = k - arr_i
            valid = arr_i != arr_q
            if not np.any(valid):
                continue

            idx_valid = arr_i[valid]
            q_valid = arr_q[valid]
            diffs = a[q_valid] - a[idx_valid]
            absdiff = np.abs(diffs)

            candidates = []
            if idx_valid.size > 0:
                order = np.argsort(absdiff)[::-1]
                for o in order[:2]:
                    candidates.append((int(idx_valid[o]), int(q_valid[o])))
                for _ in range(2):
                    o = int(rng.integers(idx_valid.size))
                    candidates.append((int(idx_valid[o]), int(q_valid[o])))

                # One ordinary affected-pair move for the same peak.
                length = je - js + 1
                o1 = int(rng.integers(length))
                o2 = int(rng.integers(length))
                p1 = js + o1
                q1 = js + o2
                if p1 != q1:
                    candidates.append((int(p1), int(q1)))

            for p, q in candidates:
                if time.monotonic() >= deadline:
                    break
                if p == q or p < 0 or p >= n or q < 0 or q >= n:
                    continue

                delta, d, cand_max = try_pair_move(a, c, p, q, n, grid_size, current_max)
                attempts += 1
                if delta != 0.0 and d is not None and cand_max < current_max - 1e-13:
                    keep, new_max = apply_and_check_pair(a, c, p, q, delta, d, current_max, n)
                    if keep:
                        current_max = new_max
                        no_improve = 0
                        accepts += 1
                        score = float(2.0 * n * current_max)
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                        if accepts % 200 == 0:
                            a = normalize_nonnegative(a)
                            c = fast_convolve(a)
                            current_max = float(c.max())
                    else:
                        no_improve += 1
                else:
                    no_improve += 1

        # A few global random pair moves to escape narrow basins.
        for _ in range(2):
            if time.monotonic() >= deadline:
                break
            p = int(rng.integers(n))
            q = int(rng.integers(n))
            if p == q:
                continue
            delta, d, cand_max = try_pair_move(a, c, p, q, n, grid_size, current_max)
            attempts += 1
            if delta != 0.0 and d is not None and cand_max < current_max - 1e-13:
                keep, new_max = apply_and_check_pair(a, c, p, q, delta, d, current_max, n)
                if keep:
                    current_max = new_max
                    no_improve = 0
                    accepts += 1
                    score = float(2.0 * n * current_max)
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                    if accepts % 200 == 0:
                        a = normalize_nonnegative(a)
                        c = fast_convolve(a)
                        current_max = float(c.max())
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 10000:
            a = best_a.copy()
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            c = fast_convolve(a)
            current_max = float(c.max())
            no_improve = 0
            accepts = 0
            K = min(int(top_k), len(c))
            if K > 0:
                top_peaks = np.argpartition(c, -K)[-K:]
                top_peaks = top_peaks[np.argsort(c[top_peaks])[::-1]]

    final_score = actual_score(best_a)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    return best_a


def project_box_sum(v, M):
    """
    Project v onto {x: 0 <= x <= M, sum(x)=1}.

    Returns None if infeasible.
    """
    v = np.asarray(v, dtype=np.float64)
    L = v.size
    if L <= 0 or M <= 0.0:
        return None
    if L * M < 1.0 - 1e-10:
        return None

    mn = float(np.min(v))
    mx = float(np.max(v))
    lo = mn - M - 1e-12
    hi = mx + 1e-12

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, 0.0, M)
        s = float(x.sum())
        if s > 1.0:
            lo = mid
        else:
            hi = mid

    t = 0.5 * (lo + hi)
    x = np.clip(v - t, 0.0, M)

    for _ in range(20):
        diff = 1.0 - float(x.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            mask = x < M - 1e-14
            if not np.any(mask):
                break
            room = np.zeros_like(x)
            room[mask] = M - x[mask]
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            add = min(diff, tot)
            x += room * (add / tot)
        else:
            mask = x > 1e-14
            if not np.any(mask):
                break
            room = x.copy()
            room[~mask] = 0.0
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            sub = min(-diff, tot)
            x -= room * (sub / tot)

    s = float(x.sum())
    if s > 0.0 and abs(s - 1.0) < 1e-8:
        ratio = 1.0 / s
        if float(np.max(x)) * ratio <= M + 1e-10:
            x *= ratio

    return x


def spectral_project_search(a0, target_M, time_limit=30.0, max_iters=3000, seed=1, blend=1.0):
    """
    Different projection / phase-retrieval search.

    Alternates:
      a -> c = conv(a) -> project c to [0, target_M] and sum 1
      -> frequency amplitude update -> inverse transform -> nonnegative support.
    """
    a = normalize_nonnegative(a0)
    n = a.size
    if n == 0:
        return a

    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()

    best_a = a.copy()
    best_score = actual_score(a)
    if not np.isfinite(best_score):
        return best_a

    current_M = best_score / (2.0 * n)
    target = float(target_M)
    if target >= current_M:
        target = current_M * 0.98
    if target < 1.0 / L:
        target = 1.0 / L + 1e-12

    deadline = time.monotonic() + max(1.0, float(time_limit))
    rng = np.random.default_rng(seed)
    no_improve = 0
    it = 0
    blend = float(max(0.0, min(1.0, blend)))

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        padded = np.concatenate((a, np.zeros(N - n, dtype=np.float64)))
        A = np.fft.fft(padded, N)
        C = A * np.conjugate(A)
        c = np.fft.ifft(C).real[:L]

        x = project_box_sum(c, target)
        if x is None:
            target = min(target * 1.02, current_M * 0.98)
            target = max(target, 1.0 / L + 1e-12)
            continue

        spec = np.fft.fft(x, N)
        amp = np.sqrt(np.maximum(np.real(spec), 0.0))

        absA = np.abs(A)
        phase = np.ones(N, dtype=np.complex128)
        mask = absA > 1e-12
        phase[mask] = A[mask] / absA[mask]
        Anew = amp * phase
        anew = np.fft.ifft(Anew).real[:n]

        anew = np.maximum(anew, 0.0)
        s = float(anew.sum())
        if s < 1e-15:
            break
        anew = anew / s

        if blend < 1.0:
            anew = (1.0 - blend) * a + blend * anew
            anew = normalize_nonnegative(anew)

        score = actual_score(anew)
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = anew.copy()
            no_improve = 0
            best_M = best_score / (2.0 * n)
            target = max(target * 0.998, best_M * 0.99)
            target = max(target, 1.0 / L + 1e-12)
        else:
            no_improve += 1
            if no_improve % 25 == 0:
                target = min(best_score * 0.99 / (2.0 * n), current_M * 0.98)
                target = max(target, 1.0 / L + 1e-12)
                a = best_a.copy()
                a = project_simplex(a + rng.standard_normal(n) * 0.001)

        a = anew
        if np.isfinite(score) and score > best_score * 1.2 + 1e-8:
            a = best_a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search entry point.

    Returns a list of nonnegative floats minimizing the evaluator.
    """
    _BEST_STATE["a"] = None
    _BEST_STATE["score"] = float("inf")

    _SMOOTH_STATE["mode"] = "all"
    _SMOOTH_STATE["peak_infos"] = []
    _SMOOTH_STATE["T"] = 1.0
    _SMOOTH_STATE["n"] = 1

    start = time.monotonic()
    deadline = start + max(60.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(seed)

    # Load the provided parent if available.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and float(p.sum()) > 1e-12:
            parent = p / float(p.sum())
    except Exception:
        parent = None

    if parent is not None:
        record_candidate(parent, "parent")

    if _BEST_STATE["a"] is None:
        fallback = np.ones(768, dtype=np.float64) / 768.0
        record_candidate(fallback, "uniform_fallback")

    # ---------------------------------------------------------------
    # Phase 1: quick polish of the parent / current best.
    # ---------------------------------------------------------------
    if time.monotonic() < deadline - 400:
        a = _BEST_STATE["a"].copy()
        peaks, c = top_peaks_of(a, 80)
        M = float(np.max(c))
        maxfun = safe_maxfun(deadline - time.monotonic() - 350, a.size, 1500)
        if maxfun >= 100:
            a1 = smooth_optimize(a, peaks, M * 0.10,
                                 maxiter=max(30, maxfun // 20),
                                 maxfun=maxfun, mode="fixed")
            record_candidate(a1, "init_fixed_smooth")

    if time.monotonic() < deadline - 350:
        rem = min(30.0, deadline - time.monotonic() - 320)
        if rem > 5.0:
            a1 = pair_descent(_BEST_STATE["a"], time_limit=rem,
                              top_k=30, grid_size=21, seed=11)
            record_candidate(a1, "init_pair")

    # ---------------------------------------------------------------
    # Phase 2: a few cheap structured alternative starts.
    # ---------------------------------------------------------------
    if time.monotonic() < deadline - 280:
        n0 = _BEST_STATE["a"].size
        x = np.linspace(0.0, 1.0, n0)
        starts = []

        edge = 1.0 / (np.minimum(np.arange(n0), n0 - 1 - np.arange(n0)) + 1.0)
        starts.append(edge / float(edge.sum()))

        g = (np.exp(-0.5 * ((x - 0.25) / 0.15) ** 2) +
             np.exp(-0.5 * ((x - 0.75) / 0.15) ** 2))
        starts.append(g / float(g.sum()))

        c1 = 1.0 + 0.6 * np.cos(np.pi * np.arange(n0))
        c1 = np.maximum(c1, 0.1)
        starts.append(c1 / float(c1.sum()))

        c2 = 1.0 + 0.3 * np.cos(2.0 * np.pi * x)
        c2 = np.maximum(c2, 0.1)
        starts.append(c2 / float(c2.sum()))

        starts.append(rng.dirichlet(np.ones(n0) * 0.4))

        for i, s in enumerate(starts):
            if time.monotonic() > deadline - 240:
                break
            peaks, conv = top_peaks_of(s, 60)
            M = float(np.max(conv))
            maxfun = safe_maxfun(deadline - time.monotonic() - 220, s.size, 800)
            if maxfun < 100:
                break
            a1 = smooth_optimize(s, peaks, M * 0.15,
                                 maxiter=max(20, maxfun // 20),
                                 maxfun=maxfun, mode="fixed")
            record_candidate(a1, f"alt_start_{i}")

    # ---------------------------------------------------------------
    # Phase 3: multi-length exploration.
    # ---------------------------------------------------------------
    base = _BEST_STATE["a"].copy()
    n_base = base.size
    n_candidates = []
    for nn in [n_base,
               int(round(n_base * 1.25)),
               int(round(n_base * 1.50)),
               int(n_base * 2),
               512, 1024, 1536]:
        if 256 <= nn <= 2048 and nn not in n_candidates:
            n_candidates.append(nn)

    for nn in n_candidates:
        if time.monotonic() > deadline - 200:
            break

        if nn == n_base:
            a_start = base.copy()
        else:
            a_start = interpolate_normalize(base, nn)

        peaks, conv = top_peaks_of(a_start, 60)
        M = float(np.max(conv))
        maxfun = safe_maxfun(deadline - time.monotonic() - 170, nn, 1000)
        a_cur = a_start
        if maxfun >= 100:
            a1 = smooth_optimize(a_start, peaks, M * 0.12,
                                 maxiter=max(25, maxfun // 20),
                                 maxfun=maxfun, mode="fixed")
            record_candidate(a1, f"n{nn}_smooth")
            a_cur = a1

        rem = deadline - time.monotonic() - 150
        if rem > 8.0 and a_cur.size == nn:
            a2 = pair_descent(a_cur, time_limit=min(12.0, rem),
                              top_k=25, grid_size=17, seed=1000 + nn)
            record_candidate(a2, f"n{nn}_pair")

    # ---------------------------------------------------------------
    # Phase 4: deeper all-peak smooth optimization.
    # ---------------------------------------------------------------
    if time.monotonic() < deadline - 180:
        a = _BEST_STATE["a"].copy()
        for Tfac in [(0.03, "T03"), (0.08, "T08"), (0.15, "T15")]:
            if time.monotonic() > deadline - 140:
                break
            a = _BEST_STATE["a"].copy()
            _, conv = top_peaks_of(a, 1)
            M = float(np.max(conv))
            maxfun = safe_maxfun(deadline - time.monotonic() - 120, a.size, 1600)
            if maxfun < 100:
                break
            a1 = smooth_optimize(a, None, M * Tfac[0],
                                 maxiter=max(30, maxfun // 20),
                                 maxfun=maxfun, mode="all")
            record_candidate(a1, f"all_{Tfac[1]}")

    # ---------------------------------------------------------------
    # Phase 5: active-peak smooth iterations with occasional restarts.
    # ---------------------------------------------------------------
    no_improve_round = 0
    tfac_cycle = [0.03, 0.05, 0.08, 0.12, 0.20]
    for r in range(120):
        if time.monotonic() > deadline - 140:
            break

        a = _BEST_STATE["a"].copy()
        peaks, conv = top_peaks_of(a, 100)
        M = float(np.max(conv))
        Tfac = tfac_cycle[r % len(tfac_cycle)]
        maxfun = safe_maxfun(deadline - time.monotonic() - 110, a.size, 1300)
        if maxfun < 100:
            break

        a1 = smooth_optimize(a, peaks, M * Tfac,
                             maxiter=max(30, maxfun // 20),
                             maxfun=maxfun, mode="fixed")
        s1 = actual_score(a1)
        if np.isfinite(s1) and s1 < _BEST_STATE["score"] - 1e-10:
            record_candidate(a1, f"active_{r}")
            no_improve_round = 0
        else:
            no_improve_round += 1
            if no_improve_round >= 2:
                a_best = _BEST_STATE["a"].copy()
                noise = rng.standard_normal(a_best.size) * 0.0015
                a_pert = project_simplex(a_best + noise)
                record_candidate(a_pert, f"perturb_{r}")

                a_cur = _BEST_STATE["a"].copy()
                peaks2, conv2 = top_peaks_of(a_cur, 100)
                M2 = float(np.max(conv2))
                mx2 = safe_maxfun(deadline - time.monotonic() - 90, a_cur.size, 900)
                if mx2 >= 100:
                    a2 = smooth_optimize(a_cur, peaks2, M2 * 0.20,
                                         maxiter=max(20, mx2 // 20),
                                         maxfun=mx2, mode="fixed")
                    record_candidate(a2, f"perturb_smooth_{r}")
                no_improve_round = 0

    # ---------------------------------------------------------------
    # Phase 6: spectral box-projection / phase retrieval starters.
    # ---------------------------------------------------------------
    if time.monotonic() < deadline - 140:
        a = _BEST_STATE["a"].copy()
        n = a.size
        current_M = _BEST_STATE["score"] / (2.0 * n)
        desired_M = 1.5027 / (2.0 * n)

        targets = []
        for t in [desired_M,
                  current_M * 0.999,
                  current_M * 0.997,
                  current_M * 0.995,
                  current_M * 0.99]:
            if t > 1.0 / (2 * n - 1) and t > 0.0 and t not in targets:
                targets.append(t)

        for ti, target in enumerate(targets):
            if time.monotonic() > deadline - 110:
                break
            rem = deadline - time.monotonic() - 100
            if rem < 12.0:
                break
            a1 = spectral_project_search(
                a,
                target_M=target,
                time_limit=min(35.0, rem),
                max_iters=5000,
                seed=1000 + 13 * ti,
                blend=1.0,
            )
            record_candidate(a1, f"sk_{ti}")
            a = _BEST_STATE["a"].copy()

    # ---------------------------------------------------------------
    # Phase 7: long exact pair-transfer polishing.
    # ---------------------------------------------------------------
    if time.monotonic() < deadline - 90:
        rem = deadline - time.monotonic() - 60
        a = _BEST_STATE["a"].copy()
        a1 = pair_descent(a, time_limit=max(10.0, min(180.0, rem * 0.60)),
                          top_k=50, grid_size=21, seed=777)
        record_candidate(a1, "pair_long_1")

    if time.monotonic() < deadline - 60:
        rem = deadline - time.monotonic() - 35
        a = _BEST_STATE["a"].copy()
        a1 = pair_descent(a, time_limit=max(5.0, min(120.0, rem)),
                          top_k=40, grid_size=19, seed=888)
        record_candidate(a1, "pair_long_2")

    # ---------------------------------------------------------------
    # Phase 8: final micro-polish.
    # ---------------------------------------------------------------
    if time.monotonic() < deadline - 40:
        rem = deadline - time.monotonic() - 20
        a = _BEST_STATE["a"].copy()
        a1 = pair_descent(a, time_limit=max(5.0, min(40.0, rem)),
                          top_k=40, grid_size=17, seed=999)
        record_candidate(a1, "final_pair")

    if time.monotonic() < deadline - 20:
        a = _BEST_STATE["a"].copy()
        peaks, conv = top_peaks_of(a, 80)
        M = float(np.max(conv))
        maxfun = safe_maxfun(deadline - time.monotonic() - 10, a.size, 900)
        if maxfun >= 90:
            a1 = smooth_optimize(a, peaks, M * 0.03,
                                 maxiter=max(20, maxfun // 20),
                                 maxfun=maxfun, mode="fixed")
            record_candidate(a1, "final_smooth")

    # ---------------------------------------------------------------
    # Finalization.
    # ---------------------------------------------------------------
    final_a = _BEST_STATE["a"]
    if final_a is None:
        final_a = np.ones(768, dtype=np.float64) / 768.0

    final_a = np.asarray(final_a, dtype=np.float64)
    final_a = np.clip(final_a, 0.0, 1000.0)
    s = float(final_a.sum())
    if not np.isfinite(s) or s < 0.01:
        final_a = np.ones(final_a.size if final_a.size > 0 else 768, dtype=np.float64) / 768.0
    else:
        final_a = final_a / s

    try:
        final_score = float(evaluate_sequence([float(x) for x in final_a]))
        print(f"FINAL internal score: {_BEST_STATE['score']:.9f}, evaluator score: {final_score:.9f}, n={final_a.size}")
    except Exception as exc:
        print("Final evaluator check failed:", exc)

    return [float(x) for x in final_a]
```
