"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with log-sum-exp smoothing.
Uses multiple initializations and adaptive temperature schedule.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_and_grad(z, n, T):
    """Compute smoothed objective and gradient w.r.t. z."""
    p = softmax_fn(z - z.mean())
    conv = fft_convolve(p)
    
    # Log-sum-exp with temperature
    scaled = conv / T
    max_scaled = np.max(scaled)
    log_L = max_scaled + np.log(np.sum(np.exp(scaled - max_scaled)))
    obj = 2.0 * n * T * log_L
    
    # Softmax weights
    w = np.exp(scaled - max_scaled)
    w /= w.sum()
    
    # Hankel product: h[i] = sum_k w[k] * p[k-i]
    # This is np.correlate(w, p, 'full')[:n]
    h = np.correlate(w, p, mode='full')[:n]
    
    # w @ conv
    w_conv = np.dot(w, conv)
    
    # Gradient
    grad = 4.0 * n * (h - p * w_conv)
    
    return obj, grad


def lbfgs_smooth(z0, n, T, max_iter=300):
    """Single L-BFGS-B step at fixed temperature T."""
    def fun(z):
        obj, _ = obj_and_grad(z, n, T)
        return obj
    
    def grad(z):
        _, g = obj_and_grad(z, n, T)
        return g
    
    try:
        res = minimize(fun, z0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10, 'maxls': 40})
        return res.x, res.fun, res.nit
    except Exception as e:
        print(f"  L-BFGS-B error: {e}")
        return z0, np.inf, 0


def multi_scale_lBFGS(p0, n, time_budget, T_factor=0.5, min_T_frac=1e-6):
    """
    Multi-scale L-BFGS-B optimization.
    Starts from p0 (probability vector), converts to z = log(p0),
    and runs L-BFGS-B with decreasing temperature T.
    """
    p = np.maximum(p0, 1e-15)
    p = p / p.sum()
    
    # Initial temperature
    conv0 = fft_convolve(p)
    T = conv0.max() * T_factor
    
    # Initial z
    z = np.log(p)
    z = z - z.mean()
    
    best_p = p.copy()
    best_exact = 2.0 * n * conv0.max()
    
    t0 = time.monotonic()
    stage = 0
    
    while True:
        elapsed = time.monotonic() - t0
        if elapsed > time_budget:
            break
        
        remaining = time_budget - elapsed
        if remaining < 3.0:
            break
        
        # Estimate how many iterations we can afford
        # Each iteration is roughly 5ms for n=768
        max_iter = min(1000, max(50, int(remaining * 100)))
        
        z, fval, nit = lbfgs_smooth(z, n, T, max_iter=max_iter)
        
        # Convert back to p
        p_new = softmax_fn(z - z.mean())
        conv_new = fft_convolve(p_new)
        exact_obj = 2.0 * n * conv_new.max()
        
        stage += 1
        print(f"    Stage {stage}: T={T:.2e}, nit={nit}, exact_obj={exact_obj:.8f}")
        
        if exact_obj < best_exact - 1e-15:
            best_exact = exact_obj
            best_p = p_new.copy()
        
        # Check if T is too small
        if T < conv_new.max() * min_T_frac:
            break
        
        T *= 0.5
    
    return best_p, best_exact


def exact_objective(p, n):
    """Compute the exact objective value."""
    conv = fft_convolve(p)
    return 2.0 * n * conv.max()


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    
    # 1. Parent (if available)
    if parent is not None:
        inits.append(('parent', parent.copy()))
        
        # 2. Tilted parent
        for alpha in [0.1, 0.3, 0.5, 1.0]:
            x = np.arange(n, dtype=np.float64)
            tilt = 1.0 + alpha * (2.0 * x / (n - 1) - 1.0)
            p = parent * tilt
            p = np.maximum(p, 1e-10)
            p = p / p.sum()
            inits.append((f'tilted_a{alpha}', p))
        
        # 3. Sine-modulated parent
        for freq in [1, 2, 3]:
            x = np.arange(n, dtype=np.float64)
            mod = 1.0 + 0.3 * np.sin(2.0 * np.pi * freq * x / n)
            p = parent * mod
            p = np.maximum(p, 1e-10)
            p = p / p.sum()
            inits.append((f'sine_f{freq}', p))
    
    # 4. Pure ramp
    x = np.arange(n, dtype=np.float64)
    p = x + 1.0
    p = p / p.sum()
    inits.append(('ramp', p))
    
    # 5. Quadratic
    p = (x + 1.0) ** 2
    p = p / p.sum()
    inits.append(('quad', p))
    
    # 6. Cosine window (Hann)
    p = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    p = np.maximum(p, 1e-10)
    p = p / p.sum()
    inits.append(('hann', p))
    
    # 7. Exponential ramp
    for rate in [0.5, 1.0, 2.0]:
        p = np.exp(rate * x / n)
        p = p / p.sum()
        inits.append((f'exp_r{rate}', p))
    
    # 8. Random smooth
    for _ in range(3):
        freqs = rng.choice(range(1, 10), size=5, replace=False)
        phases = rng.uniform(0, 2*np.pi, size=5)
        amps = rng.uniform(0.1, 1.0, size=5)
        p = np.ones(n)
        for f, ph, a in zip(freqs, phases, amps):
            p += a * np.sin(2*np.pi*f*x/n + ph)
        p = np.maximum(p, 0.01)
        p = p / p.sum()
        inits.append(('rand_smooth', p))
    
    # 9. "U-shaped" (large at edges, small in middle)
    center = 0.5 * (n - 1)
    p = np.ones(n)
    p *= (np.abs(x - center) / center + 0.1)
    p = p / p.sum()
    inits.append(('u_shaped', p))
    
    return inits

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_p = p_new.copy()
        
        p = best_p  # Continue from best
    
    return best_p, best_obj
