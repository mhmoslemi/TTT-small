"""
Multi-phase optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated subgradient descent, LP Frank-Wolfe,
peak shaving, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, peak_weights, n):
    """Compute subgradient of max weighted convolution peaks.
    peak_weights: dict mapping lag -> weight
    """
    sg = np.zeros(n)
    for k, w in peak_weights.items():
        if w < 1e-15:
            continue
        # sg[j] += 2*w*f[k-j] for 0<=k-j<n
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        for j in range(js, je):
            idx = k - j
            if 0 <= idx < n:
                sg[j] += 2.0 * w * f[idx]
    return sg

def nesterov_subgradient(f0, time_limit, lr=0.01, momentum=None, top_k=20, temp=0.001, verbose=False):
    """Nesterov-accelerated projected subgradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # Nesterov momentum
    if momentum is None:
        momentum = 0.9
    beta = momentum
    
    prev_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find top peaks with softmax weights
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws
        
        # Build peak weight dict
        peak_weights = {}
        for i, pk in enumerate(peaks):
            if w[i] > 1e-12:
                peak_weights[int(pk)] = w[i]
        
        # Compute subgradient
        sg = compute_subgradient(f, peak_weights, n)
        
        # Nesterov step
        y = f + beta * (f - prev_f)
        y = y - cur_lr * sg
        f_new = project_simplex(y)
        
        prev_f = f
        f = f_new
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    NSG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            
        # Adaptive learning rate
        cur_lr *= 0.9999
        cur_temp *= 1.002
        
        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng(hash((it, n)) % 2**31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            prev_f = f.copy()
            cur_lr = max(lr * 0.3, 0.001)
            no_improve = 0
            conv = fast_convolve(f)
    
    return best_f, best_obj

"""
Multi-phase optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated subgradient descent with adaptive peak weighting,
LP-based Frank-Wolfe, peak shaving coordinate descent, and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, peak_indices, peak_weights):
    """Compute subgradient of sum_i w_i * b[k_i] where b=convolve(f,f).
    grad b[k] at position j = 2*f[k-j] if 0<=k-j<n, else 0.
    """
    n = len(f)
    sg = np.zeros(n)
    for i, k in enumerate(peak_indices):
        w = peak_weights[i]
        if w < 1e-15:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        for j in range(js, je):
            idx = k - j
            if 0 <= idx < n:
                sg[j] += 2.0 * w * f[idx]
    return sg


def compute_subgradient_fft(f, peak_indices, peak_weights):
    """FFT-based subgradient computation for efficiency."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    w_full = np.zeros(fft_size)
    for i, k in enumerate(peak_indices):
        w_full[int(k)] = peak_weights[i]
    
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_full, fft_size)
    
    # sg[j] = 2 * sum_k w[k] * f[k-j]
    # This is cross-correlation: sg = 2 * correlate(w_full, f)
    # In FFT: correlate(a,b) = ifft(conj(FFT(a)) * FFT(b))
    # But we need sum_k w[k]*f[k-j], which is (w * f_rev)[j+n-1] or similar.
    # Let's use: sum_k w[k]*f[k-j] = ifft(FFT(w) * conj(FFT(f)))... 
    # Actually: (w * f)[m] = sum_k w[k]*f[m-k]. We want sum_k w[k]*f[k-j] = (w * f)[0-j+?]
    # Let f_rev[m] = f[-m] circularly. Then sum_k w[k]*f[k-j] = sum_k w[k]*f_rev[j-k] = (w*f_rev)[j]
    
    f_rev_padded = np.zeros(fft_size)
    f_rev_padded[:n] = f[::-1]
    # Circular reversal in frequency domain:
    # f_rev_dft[k] = conj(F[k]) if we define f_rev[m] = f[-m mod fft_size]
    # But our f_rev is only defined for m in [0, n-1] as f[n-1-m]
    # This is not a simple circular reversal.
    
    # Let's just use the direct method for correctness
    return compute_subgradient(f, peak_indices, peak_weights)


def nesterov_sg(f0, time_limit, lr_init=0.005, top_k=30, temp_init=0.0005,
                restart_interval=600, verbose=False):
    """Nesterov-accelerated projected subgradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr_init
    cur_temp = temp_init
    beta = 0.9
    prev_f = f.copy()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = compute_subgradient(f, peaks, w)
        
        # Nesterov momentum step
        y = f + beta * (f - prev_f)
        y = y - cur_lr * sg
        f_new = project_simplex(y)
        
        prev_f = f.copy()
        f = f_new
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    NSG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        cur_lr *= 0.99992
        cur_temp *= 1.003
        beta = min(0.98, 0.9 + 0.08 * it / max(it, 1000))
        
        if no_improve > restart_interval:
            rng = np.random.default_rng((it * 7919 + hash(n)) % (2**31))
            noise_scale = 0.001 + 0.001 * min(no_improve // restart_interval, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            prev_f = f.copy()
            cur_lr = max(lr_init * 0.4, 0.0005)
            cur_temp = temp_init * 0.5
            no_improve = 0
            conv = fast_convolve(f)
    
    return best_f, best_obj


def sg_momentum(f0, time_limit, lr=0.003, momentum=0.93, top_k=40, temp=0.00008,
                verbose=False):
    """Momentum-based subgradient descent (simpler, more robust)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    dec_counter = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        dec_counter += 1
        
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = compute_subgradient(f, peaks, w)
        
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new
        
        if dec_counter % 500 == 0:
            cur_lr *= 0.97
            cur_temp *= 1.05
        
        if no_improve > 800:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
    
    return best_f, best_obj


def solve_lp_min_peak(f, n):
    """LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

            no_improve += 1

            continue

def peak_shaving_cd(f0, time_limit, verbose=False):
    """Coordinate descent that directly targets peak reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        top_k = min(5, len(conv))
        peak_positions = np.argsort(conv)[-top_k:]
        
        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            for j in range(js, je + 1):
                if f[j] < 1e-12:
                    continue
                
                old_val = f[j]
                best_j_val = old_val
                best_j_obj = best_obj
                
                for mult in [0.3, 0.5, 0.7, 1.4, 1.8, 2.5]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val
                
                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                    
                    if best_j_obj < best_obj - 1e-13:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                rng = np.random.default_rng((passes * 104729 + 7) % (2**31))
                for _ in range(min(30, n // 8)):
                    j = int(rng.integers(0, n))
                    if f[j] < 1e-10:
                        continue
                    for mult in [0.4, 0.7, 1.5, 2.0]:
                        new_val = f[j] * mult
                        if new_val < 0 or new_val > 0.5:
                            continue
                        denom = 1.0 - f[j]
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-13:

                            best_obj = val

                    if improved:

        else:

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t = min(rem, 100)
            a_opt, _ = nesterov_sg(best_seq, time_limit=t,
                                   lr_init=0.001, top_k=40, temp_init=0.0001)
            try_seq(a_opt, "polish_nsg")
        
        rem = deadline - time.monotonic() - 25
        if rem > 15:
            t = min(rem, 60)
            a_opt, _ = peak_shaving_cd(best_seq, time_limit=t)
            try_seq(a_opt, "polish_cd")
        
        # Perturbation-based exploration
        for p in range(8):
            rem = deadline - time.monotonic() - 15
            if rem < 15:
                break
            noise_scale = 0.0008 + 0.0004 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t = min(rem, 30)
            a_opt, _ = sg_momentum(f_pert, time_limit=t,
                                   lr=0.0008, momentum=0.95,
                                   top_k=max(15, n_best // 12),
                                   temp=0.00002)
            try_seq(a_opt, f"perturb_{p}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
