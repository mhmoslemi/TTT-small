def compute_subgradient(a, active_peaks, p):
    """Compute subgradient of max_k conv(a,a)[k] at a.
    
    a: the sequence (sum = 1, non-negative)
    active_peaks: indices where conv(a,a)[k] is near maximum
    p: probability distribution over active peaks
    """
    n = len(a)
    grad = np.zeros(n)
    for k, pk in zip(active_peaks, p):
        # grad[m] += 2 * pk * a[k-m] for valid m
        # k-m in [0, n-1] means m in [k-n+1, k]
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            grad[m_lo:m_hi+1] += 2 * pk * a[k - np.arange(m_lo, m_hi + 1)]
    return grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with soft-subgradient,
multi-length exploration, coordinate descent polish, and adaptive restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if not len(rho):
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, beta, top_k=50):
    """Compute soft-subgradient of max_k conv(a,a)[k]."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights over top-k peaks
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = beta * pv
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-12:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            grad[m_lo:m_hi+1] += 2.0 * weights[i] * a[k - np.arange(m_lo, m_hi + 1)]
    return grad, conv

    t_k = 1.0
    lr = lr_start
    beta = beta_start
    
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:

        grad, conv_y = compute_subgrad(y, beta, top_k)

        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)

def cosine_init(n, freq, amp):

    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f = np.interp(new_x, old_x, a)
    f = np.maximum(f, 0.0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f

    cur_beta = float(beta)

    t0 = time.monotonic()
    it = 0
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv_max = compute_smooth_grad(a, cur_beta, top_k)

        # Decreasing step size
        step = cur_lr / np.sqrt(1.0 + 0.01 * it)
        a_new = project_simplex(a - step * grad)
        a = a_new
        all_iters.append(a.copy())

        # Keep memory manageable
        if len(all_iters) > 5000:
            all_iters = all_iters[1000:]

        conv = fast_convolve(a)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        # Also check averaged solution
        if it % 200 == 0:
            a_avg = np.mean(all_iters, axis=0)
            a_avg = project_simplex(a_avg)
            conv_avg = fast_convolve(a_avg)
            obj_avg = conv_avg.max()
            if obj_avg < best_obj - 1e-14:
                best_obj = obj_avg
                best_a = a_avg.copy()

        # Adapt beta
        if it % 500 == 0:
            cur_beta = min(cur_beta * 1.2, 8000.0)

        # Restart
        if it % 2000 == 0:
            a = project_simplex(0.7 * best_a + 0.3 * a)
            all_iters = [a.copy()]
            cur_lr = max(cur_lr * 0.8, 1e-5)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG-Avg it={it}, obj={best_obj:.10f}, lr={cur_lr:.2e}, time={elapsed:.1f}s")

    return best_a, best_obj


def coord_dir_eval(a, j, x, denom):
    """Evaluate objective when coordinate j is set to x, others scaled."""
    n = len(a)
    at = a.copy()
    at[j] = x
    rest_sum = 1.0 - x
    if rest_sum <= 0:
        return float('inf')
    mask = np.ones(n, dtype=bool)
    mask[j] = False
    at[mask] = at[mask] * (rest_sum / denom)
    return fast_convolve(at).max()


def coordinate_descent(a0, time_limit, max_rounds=50):
    """Coordinate descent with golden section search per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    no_round_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit and no_round_improve < 4:
        # Randomize coordinate order for diversity
        perm = rng.permutation(n)
        round_improved = False

        for j in perm:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            # Golden section search
            lo = 0.0
            hi = min(0.9999, max(2.0 * old_val, denom * 0.5))

            c1 = lo + inv_gr * (hi - lo)
            c2 = hi - inv_gr * (hi - lo)
            f1 = coord_dir_eval(a, j, c1, denom)
            f2 = coord_dir_eval(a, j, c2, denom)

            for _ in range(18):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = coord_dir_eval(a, j, c1, denom)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = hi - inv_gr * (hi - lo)
                    f2 = coord_dir_eval(a, j, c2, denom)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)

            if opt_val < best_obj - 1e-12:
                at = a.copy()
                at[j] = opt_x
                rest_sum = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                at[mask] = at[mask] * (rest_sum / denom)
                at_sum = at.sum()
                if at_sum > 1e-15:
                    at /= at_sum
                a = at
                new_obj = fast_convolve(a).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_a = a.copy()
                round_improved = True

        if round_improved:
            no_round_improve = 0
        else:
            no_round_improve += 1

    return best_a, best_obj


def cosine_init(n, freq, amp):
    """Generate cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gaussian_window_init(n, center_frac=0.5, width_frac=0.3):
    """Generate a Gaussian-windowed initialization."""
    j = np.arange(n)
    center = center_frac * n
    width = width_frac * n
    a = np.exp(-0.5 * ((j - center) / width) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f = np.interp(new_x, old_x, a)
    f = np.maximum(f, 0.0)
    ss = f.sum()
    if ss > 1e-15:
        f /= ss
    return f


def try_and_track(a, best_score_ref, label):
    """Evaluate sequence and return (score, is_new_best)."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return float('inf'), False
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    return score, (np.isfinite(score) and score < best_score_ref[0])

            return float('inf')
        a = a / s

            parent = p / p.sum()
