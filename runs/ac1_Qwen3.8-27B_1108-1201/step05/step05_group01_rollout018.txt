"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using enhanced LP-Frank-Wolfe with
dual-based line search, multi-length exploration, and coordinate descent polish.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def solve_lp_direction(f, M, n, warm_start=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    # First try a grid

        if v < best_val:
            best_val = v

        lo = max(0.0, best_alpha - 0.05 * (grid[1] - grid[0]))

def coordinate_descent(f0, time_limit):
    """Cycle through coordinates, optimizing each with golden section."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_pass_improve = 0
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR

    while time.monotonic() - t0 < time_limit:
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            def eval_at(x):
                ft = f.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = f[mask] * (ft2 / denom)
                return fast_convolve(ft).max()

            # Golden section search
            lo = 0.0
            hi = min(0.999, 3.0 * old_val)
            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)
            f1 = eval_at(c1)
            f2 = eval_at(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = obj(c2)  # BUG: should be eval_at(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = fast_convolve(f).max()

            if opt_val < cur_val - 1e-13:
                ft = f.copy()
                ft[j] = opt_x
                ft2_val = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = f[mask] * (ft2_val / denom)
                f = ft / ft.sum()
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_f = f.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def subgradient_step(f, conv, top_k=30, temp=0.001):
    """Compute a subgradient-based direction (cheaper than LP)."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the convolution matrix F where (Fg)[k] = sum_j f[k-j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A


def solve_lp_direction(f, M, n, A_ub=None):
    """Solve LP: max sum(g) s.t. A_ub @ g <= M, g >= 0."""
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    if A_ub is None:
        A_ub = build_conv_matrix(f, n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def quadratic_line_search(f, g, conv_f):
    """Line search using the quadratic structure of the objective."""
    n = len(f)
    d = g - f
    conv_d = fast_convolve(d)
    cross = fast_cross_convolve(f, d)
    
    def obj(alpha):
        return (conv_f + 2.0 * alpha * cross + alpha * alpha * conv_d).max()
    
    best_alpha = 0.0
    best_val = obj(0.0)
    
    # Coarse grid
    for a in np.linspace(0, 1, 25):
        v = obj(a)
        if v < best_val:
            best_val = v
            best_alpha = a
    
    # Golden section refinement
    if best_alpha > 0 and best_alpha < 1:
        inv_gr = 2.0 / (1.0 + np.sqrt(5.0))
        lo = max(0.0, best_alpha - 0.05)
        hi = min(1.0, best_alpha + 0.05)
        c1 = lo + inv_gr * (hi - lo)
        c2 = hi - inv_gr * (hi - lo)
        f1 = obj(c1)
        f2 = obj(c2)
        for _ in range(20):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c2 = hi - inv_gr * (hi - lo)
                f2 = obj(c2)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c1 = lo + inv_gr * (hi - lo)
                f1 = obj(c1)
        opt = (lo + hi) / 2.0
        vo = obj(opt)
        if vo < best_val:
            best_alpha = opt
            best_val = vo
    
    return best_alpha

def coord_descent(f0, time_limit):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_pass_improve = 0
    inv_gr = 2.0 / (1.0 + np.sqrt(5.0))

    while time.monotonic() - t0 < time_limit:
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Golden section search
            lo = 0.0
            hi = min(0.999, 2.5 * old_val)
            c1 = lo + inv_gr * (hi - lo)
            c2 = hi - inv_gr * (hi - lo)

            def eval_at(x):
                ft = f.copy()
                ft[j] = x
                r = 1.0 - x
                if r < 0:
                    return float('inf')
                ft[ft != 0]  # just to ensure it's an array
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = f[mask] * (r / denom)
                return fast_convolve(ft).max()

            f1 = eval_at(c1)
            f2 = eval_at(c2)

            for _ in range(12):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c2 = hi - inv_gr * (hi - lo)
                    f2 = eval_at(c2)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)

            if opt_val < best_obj - 1e-13:
                ft = f.copy()
                ft[j] = opt_x
                r = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = f[mask] * (r / denom)
                f = ft / ft.sum()
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_f = f.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_f, best_obj

- Smarter restart strategies with structured perturbations
- Final coordinate descent for local polish
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build convolution matrix F where (F@x)[k] = sum_j f[k-j]*x[j] for valid j."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, n, A_ub):
    """Solve: max 1^T g s.t. A_ub @ g <= M, g >= 0. Returns g or None."""
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={"presolve": True, "time_limit": 8.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def quadratic_line_search(conv_f, d, max_alpha=1.0):
    """
    Find alpha in [0, max_alpha] minimizing max_k (conv_f + 2a*cross + a^2*conv_d)[k]
    where cross = conv(f,d), conv_d = conv(d,d).
    We pass d = g - f, and precompute what we need.
    """
    cross = fast_cross_convolve(f_global_ref[0], d)
    conv_d = fast_convolve(d)

    def obj(alpha):
        return (conv_f + 2.0 * alpha * cross + alpha * alpha * conv_d).max()

    # Coarse grid
    n_grid = 41
    alphas = np.linspace(0.0, max_alpha, n_grid)
    vals = np.empty(n_grid)
    for i in range(n_grid):
        vals[i] = obj(alphas[i])
    best_idx = int(np.argmin(vals))
    best_alpha = alphas[best_idx]
    best_val = vals[best_idx]

    if best_idx == 0:
        return 0.0
    if best_idx == n_grid - 1:
        return max_alpha

    # Golden section in [alphas[best_idx-1], alphas[best_idx+1]]
    lo = alphas[best_idx - 1]
    hi = alphas[best_idx + 1]
    inv_gr = 2.0 / (1.0 + np.sqrt(5.0))
    c1 = hi - inv_gr * (hi - lo)
    c2 = lo + inv_gr * (hi - lo)
    f1 = obj(c1)
    f2 = obj(c2)

    for _ in range(25):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c2 = lo + inv_gr * (hi - lo)
            f2 = obj(c2)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c1 = hi - inv_gr * (hi - lo)
            f1 = obj(c1)

    opt = (lo + hi) / 2.0
    vo = obj(opt)
    if vo < best_val:
        best_alpha = opt
        best_val = vo

    return best_alpha


# Global reference for line search (avoids passing f around)
f_global_ref = [None]


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with quadratic line search and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    global f_global_ref
    f_global_ref[0] = f

    A_ub = build_conv_matrix(f, n)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        g = solve_lp(f, M * m_exp, n, A_ub)
        if g is None:
            m_exp *= 1.05
            if m_exp > 4.0:
                # Fallback: use subgradient direction
                peaks = np.argsort(conv)[-3:]
                grad = np.zeros(n)
                for k in peaks:
                    m_lo = max(0, int(k) - (n - 1))
                    m_hi = min(n - 1, int(k))
                    if m_lo <= m_hi:
                        grad[m_lo:m_hi + 1] += f[int(k) - np.arange(m_lo, m_hi + 1)]
                j_min = int(np.argmin(grad))
                g = np.zeros(n)
                g[j_min] = 1.0
                m_exp = 1.0
            else:
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.02, 2.0)
            continue

        f_global_ref[0] = f
        alpha = quadratic_line_search(conv, g - f)

        if alpha > 1e-9:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            st = f_new.sum()
            if st < 1e-15:
                no_improve += 1
                continue
            f_new = f_new / st

            A_ub = build_conv_matrix(f_new, n)
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new
            f_global_ref[0] = f

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.0f}s")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.04, 4.0)

        if no_improve > 35:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.004 + 0.004 * min(no_improve // 35, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            A_ub = build_conv_matrix(f, n)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            f_global_ref[0] = f

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coord_descent(f0, time_limit, n_steps_per_pass=None):
    """Coordinate descent with golden section per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_pass_improve = 0
    inv_gr = 2.0 / (1.0 + np.sqrt(5.0))

    # Randomize coordinate order for diversity
    rng = np.random.default_rng(12345)
    coord_order = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        rng.shuffle(coord_order)

        for j in coord_order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            lo = 0.0
            hi = min(0.999, 3.0 * old_val)
            c1 = lo + inv_gr * (hi - lo)
            c2 = hi - inv_gr * (hi - lo)

            def eval_cd(x):
                ft = f.copy()
                ft[j] = x
                r = 1.0 - x
                if r < 1e-10:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = f[mask] * (r / denom)
                return fast_convolve(ft).max()

            f1 = eval_cd(c1)
            f2 = eval_cd(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c2 = lo + inv_gr * (hi - lo)
                    f2 = eval_cd(c2)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c1 = hi - inv_gr * (hi - lo)
                    f1 = eval_cd(c1)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = fast_convolve(f).max()

            if opt_val < cur_val - 1e-13:
                ft = f.copy()
                ft[j] = opt_x
                r = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = f[mask] * (r / denom)
                fsum = ft.sum()
                if fsum < 1e-15:
                    continue
                f = ft / fsum
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_f = f.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_f, best_obj


def cosine_modulated_init(n, freq, amp):
    """Generate a cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
