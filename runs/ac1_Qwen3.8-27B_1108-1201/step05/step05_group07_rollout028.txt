Key innovations: exact subgradient from all near-max lags, deflation steps that

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_precise_subgradient(a, conv, rel_tol, abs_tol):
    """
    Compute subgradient of max_k conv[k] using all lags within tolerance of max.
    d(conv[k])/da[j] = 2*a[k-j] if 0<=k-j<n else 0.
    """
    n = len(a)
    max_val = conv.max()
    threshold = max_val * (1.0 - rel_tol) - abs_tol
    active_mask = conv >= threshold
    active_lags = np.where(active_mask)[0]
    
    if len(active_lags) == 0:
        active_lags = np.array([int(np.argmax(conv))])
    
    sg = np.zeros(n)
    for k in active_lags:
        k = int(k)
        # d(conv[k])/da[j] = 2*a[k-j] for j where 0 <= k-j < n
        # i.e., j in [max(0, k-n+1), min(n-1, k)]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * a[k - idx]
    
    sg /= len(active_lags)
    return sg, active_lags, max_val


def compute_topk_subgradient(a, conv, top_k, temperature):
    """Softmax-weighted subgradient over top-k peaks."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg

def deflation_optimize(f0, time_limit, n_deflation=10, verbose=False):
    """
    Iterative deflation: identify the top peaks and explicitly reduce
    their contributing coordinates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        max_val = conv.max()
        
        # Get top peaks
        top_peaks = np.argpartition(conv, -n_deflation)[-n_deflation:]
        top_peaks = top_peaks[np.argsort(conv[top_peaks])[::-1]]

        # Compute which coordinates contribute most to the top peaks
        contrib = np.zeros(n)
        for k in top_peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += f[k - idx]

        # Find the "most responsible" coordinates
        top_coords = np.argsort(contrib)[::-1][:n // 4]
        
        # Deflate: reduce top-contributing coords, boost others
        boost_amount = 0.01
        total_deflate = 0
        for j in top_coords:
            if f[j] > 1e-10:
                amt = min(f[j] * 0.05, boost_amount / n * 10)
                f[j] -= amt
                total_deflate += amt
        
        # Distribute to non-top coords
        non_top = np.setdiff1d(np.arange(n), top_coords)
        if len(non_top) > 0 and total_deflate > 0:
            # Distribute proportional to current values
            weights = f[non_top] + 1e-10
            weights /= weights.sum()
            f[non_top] += total_deflate * weights
        
        f = np.maximum(f, 0)
        fs = f.sum()
        if fs > 1e-15:
            f /= fs
        
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
    
    return best_f, best_obj

                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def bimodal_init(n, rng, alpha=0.3):
    """
    Create a bimodal initialization: high at edges, low in center.
    This structure minimizes the center cross-correlation.
    """
    x = np.linspace(0, 1, n)
    # Front half: decaying from 1 to alpha
    # Back half: rising from alpha to 1
    mid = n // 2
    f = np.ones(n)
    f[:mid] = 1.0 - (1.0 - alpha) * np.linspace(0, 1, mid)
    f[mid:] = alpha + (1.0 - alpha) * np.linspace(0, 1, n - mid)
    # Add some randomness
    f += rng.standard_normal(n) * 0.05
    f = np.maximum(f, 1e-10)
    return f / f.sum()


def structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Bimodal (edge-peaked)
    inits.append(bimodal_init(n, rng, alpha=0.2 + rng.random() * 0.4))
    
    # Asymmetric: front-heavy
    x = np.linspace(0, 1, n)
    front_heavy = np.exp(-2.0 * x) + 0.1
    inits.append(front_heavy / front_heavy.sum())
    
    # Asymmetric: back-heavy
    back_heavy = np.exp(-2.0 * (1 - x)) + 0.1
    inits.append(back_heavy / back_heavy.sum())
    
    # Gaussian center
    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 3.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())
    
    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())
    
    # "Comb" pattern: alternating high/low
    comb = np.ones(n)
    comb[0::2] = 1.0 + rng.random() * 0.8
    comb[1::2] = 0.2 + rng.random() * 0.3
    inits.append(comb / comb.sum())
    
    return inits[:count]


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def pair_exchange_search(f0, time_limit, verbose=False):
    """
    Pair-exchange local search: transfer mass between coordinate pairs
    to reduce the convolution peak. Uses gradient guidance for pair selection.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        # Get gradient for guidance
        sg, active_lags, _ = compute_precise_subgradient(f, conv, rel_tol=0.01, abs_tol=1e-10)
        
        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
        
        # Try guided pairs: from high-grad to low-grad coordinates
        n_trials = min(300, n)
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick i from high-gradient coords, j from low-gradient coords
            top_grad = np.argsort(-sg)[:n // 2]
            bot_grad = np.argsort(sg)[:n // 2]
            i = top_grad[rng.integers(0, len(top_grad))]
            j = bot_grad[rng.integers(0, len(bot_grad))]
            i = int(i)
            j = int(j)
            
            if i == j or f[i] < 1e-10:
                continue
            
            old_i = f[i]
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = old_i * frac
                f[i] = old_i - amt
                f[j] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] -= amt
                    break
        
        # Also try random pairs
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or f[i] < 1e-10:
                continue
            old_i = f[i]
            for frac in [0.1, 0.2, 0.4]:
                amt = old_i * frac
                f[i] = old_i - amt
                f[j] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] -= amt
                    break
        
        if not improved:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining multiplicative-weights subgradient with
    precise active-set gradients, deflation, coordinate descent, and multi-length
    exploration. Exploits bimodal structure to minimize the autocorrelation peak.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = bimodal_init(768, rng)
        print("No parent found, using bimodal init n=768")

    # === Phase 1: Quick optimization on parent with multiple configs ===
    print("\n=== Phase 1: Quick MU Opt on Parent ===")
    
    n_parent = len(parent)
    configs = [
        {"top_k": max(20, n_parent // 8), "temp": 0.00003, "lr": 0.003,
         "mom": 0.87, "precise": True, "rel_tol": 0.002,
         "defl_int": 100, "defl_str": 0.8},
        {"top_k": max(30, n_parent // 5), "temp": 0.0001, "lr": 0.004,
         "mom": 0.80, "precise": False, "rel_tol": 0.01,
         "defl_int": 150, "defl_str": 1.0},
        {"top_k": max(10, n_parent // 12), "temp": 0.00001, "lr": 0.002,
         "mom": 0.92, "precise": True, "rel_tol": 0.001,
         "defl_int": 50, "defl_str": 1.5},
    ]

    rem = deadline - time.monotonic() - 400
    t_each = max(20, rem / len(configs))
    
    for idx, cfg in enumerate(configs):
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_alloc = min(t_each, rem * 0.4)
        print(f"  Config {idx}: t={t_alloc:.0f}s")
        a_opt, _ = mu_weights_optimize(
            parent, time_limit=t_alloc,
            top_k=cfg["top_k"], temperature=cfg["temp"],
            lr=cfg["lr"], momentum=cfg["mom"],
            rel_tol=cfg["rel_tol"], use_precise=cfg["precise"],
            deflation_interval=cfg["defl_int"],
            deflation_strength=cfg["defl_str"])
        try_seq(a_opt, f"p1_mu_{idx}")

    # === Phase 2: Structured inits ===
    print(f"\n=== Phase 2: Structured Inits (best={best_score:.6f}) ===")
    
    target_lengths = [n_parent]
    # Add a couple of other lengths
    if n_parent < 1024:
        target_lengths.append(1024)
    if n_parent < 2048:
        target_lengths.append(1536)
    
    for n_try in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        inits = structured_inits(n_try, rng, count=4)
        t_each = max(10, rem / (len(inits) * 2))
        
        for idx, init in enumerate(inits):
            rem = deadline - time.monotonic() - 150
            if rem < 10:
                break
            t_alloc = min(t_each, rem * 0.3)
            a_opt, _ = mu_weights_optimize(
                init, time_limit=t_alloc,
                top_k=max(20, n_try // 8), temperature=0.00004,
                lr=0.003, momentum=0.85,
                rel_tol=0.002, use_precise=True,
                deflation_interval=100, deflation_strength=0.8)
            try_seq(a_opt, f"p2_init_{n_try}_{idx}")

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        deep_configs = [
            # Very precise, slow convergence
            {"lr": 0.001, "mom": 0.93, "rel_tol": 0.0005, "defl_int": 80, "defl_str": 1.2},
            # Moderate
            {"lr": 0.002, "mom": 0.88, "rel_tol": 0.002, "defl_int": 120, "defl_str": 0.8},
            # Aggressive with restarts
            {"lr": 0.004, "mom": 0.82, "rel_tol": 0.005, "defl_int": 60, "defl_str": 1.5},
            # Different: use softmax subgradient
            {"lr": 0.003, "mom": 0.85, "rel_tol": 0.002, "defl_int": 100, "defl_str": 1.0},
        ]
        
        for idx, cfg in enumerate(deep_configs):
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            t_alloc = min(rem * 0.25, 180)
            
            if idx == 0:
                f_init = best_seq.copy()
            elif idx == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif idx == 2:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                # Symmetrize-reverse mix (exploits asymmetry)
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init)
            
            print(f"  Deep {idx}: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = mu_weights_optimize(
                f_init, time_limit=t_alloc,
                top_k=max(20, n_best // 6), temperature=0.00003,
                lr=cfg["lr"], momentum=cfg["mom"],
                rel_tol=cfg["rel_tol"], use_precise=(idx < 3),
                deflation_interval=cfg["defl_int"],
                deflation_strength=cfg["defl_str"])
            try_seq(a_opt, f"deep_{idx}")

    # === Phase 4: Deflation + Local Search ===
    print(f"\n=== Phase 4: Deflation & Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_defl = min(rem * 0.3, 60)
            print(f"  Deflation: t={t_defl:.0f}s")
            a_opt, _ = deflation_optimize(best_seq, time_limit=t_defl)
            try_seq(a_opt, "deflation")

            rem = deadline - time.monotonic() - 50
            if rem > 15:
                t_pex = min(rem * 0.5, 80)
                print(f"  Pair exchange: t={t_pex:.0f}s")
                a_opt, _ = pair_exchange_search(best_seq, time_limit=t_pex)
                try_seq(a_opt, "pair_exch")

            rem = deadline - time.monotonic() - 40
            if rem > 15:
                t_cd = min(rem * 0.7, 80)
                print(f"  Coord descent: t={t_cd:.0f}s")
                a_opt, _ = gradient_coordinate_descent(best_seq, time_limit=t_cd)
                try_seq(a_opt, "coord_desc")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Try nearby lengths
        candidates_n = set()
        candidates_n.add(n_best // 2)
        candidates_n.add(n_best)
        candidates_n.add(n_best * 3 // 2)
        candidates_n.add(n_best * 2)
        # Add standard sizes
        for s in [512, 768, 1024, 1536, 2048]:
            candidates_n.add(s)
        candidates_n = sorted([x for x in candidates_n if 256 <= x <= 4096])
        
        for target_n in candidates_n:
            if target_n == n_best:
                continue
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.25, 100)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = mu_weights_optimize(
                f_interp, time_limit=t_alloc,
                top_k=max(20, target_n // 8), temperature=0.00004,
                lr=0.003, momentum=0.87,
                rel_tol=0.002, use_precise=True,
                deflation_interval=100, deflation_strength=0.8)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Ultra-fine subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = mu_weights_optimize(
                best_seq, time_limit=min(rem, 50),
                top_k=n_best, temperature=0.000001,
                lr=0.0003, momentum=0.96,
                rel_tol=0.0001, use_precise=True,
                deflation_interval=50, deflation_strength=0.5)
            try_seq(a_opt, "sg_fine")

        # Pair exchange on fine result
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            a_opt, _ = pair_exchange_search(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "pex_fine")

        # Small perturbations + quick optimize
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            for scale in [0.001, 0.002, 0.004]:
                rem = deadline - time.monotonic() - 8
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(best_seq + noise)
                t_pert = min(rem, 10)
                a_opt, _ = mu_weights_optimize(
                    f_pert, time_limit=t_pert,
                    top_k=max(20, n_best // 5), temperature=0.00002,

        # Coordinate descent as last resort
        rem = deadline - time.monotonic() - 5
        if rem > 5:

    # Fallback

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
