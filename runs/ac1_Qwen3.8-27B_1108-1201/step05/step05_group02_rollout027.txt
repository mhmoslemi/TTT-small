def targeted_cd(f0, time_limit, verbose=False):

def lp_min_peak(f, n):
    """LP: min t s.t. (f*g)_k <= t, g>=0, sum(g)=1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a multi-strategy approach:
1. Adaptive subgradient descent with momentum, softmax peak weighting, and mirror descent variants
2. LP-based Frank-Wolfe for stronger directional guarantees
3. Peak-targeted coordinate descent for local polish
4. Multi-length interpolation and exploration
Key innovations: alternating sharp/broad peak targeting, adaptive restart with increasing noise,
and hybrid mirror/projection steps for better convergence.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_step(f, sg, lr):
    """Mirror descent step with negative entropy (multiplicative update on simplex)."""
    exp_arg = -lr * sg
    exp_arg -= np.max(exp_arg)
    w = np.exp(exp_arg)
    f_new = f * w
    s = f_new.sum()
    if s < 1e-30:
        return np.ones(len(f)) / len(f)
    return f_new / s


def compute_sg(f, conv, peaks, weights):
    """Compute weighted subgradient of max-peak objective at current f."""
    n = len(f)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        kg = k - idx
        mask = (kg >= 0) & (kg < n)
        if mask.any():
            sg[idx[mask]] += 2.0 * weights[i] * f[kg[mask]]
    return sg


def adaptive_sg(f0, time_limit, lr=0.003, momentum=0.92, top_k=50,
                temp=5e-6, use_mirror=False, mirror_every=3,
                restart_thresh=250, verbose=False):
    """
    Adaptive subgradient with momentum, softmax peak weighting,
    optional mirror descent steps, and periodic perturbative restarts.
    """
    f = np.maximum(f0, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    cur_mom = momentum

    while time.monotonic() - t0 < time_limit:
        it += 1

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        sg = compute_sg(f, conv, peaks, w)

        if use_mirror and it % mirror_every == 0:
            f_new = mirror_step(f, sg, cur_lr * 0.6)
        else:
            v = cur_mom * v - cur_lr * sg
            f_new = project_simplex(f + v)

        f_new = np.maximum(f_new, 0)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / fs

        conv = fast_convolve(f_new)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        cur_lr *= 0.99993
        cur_temp *= 1.005
        cur_mom = min(0.9995, momentum + 0.00003 * it)

        if no_improve > restart_thresh:
            rng = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.001 + 0.0015 * min(no_improve // restart_thresh, 6)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0004)
            cur_temp = temp * 0.4
            no_improve = 0

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}, lr={cur_lr:.6f}, temp={cur_temp:.2e}")

    return best_f, best_obj


def lp_min_peak(f, n):
    """
    LP: min t s.t. (f*g)_k <= t for all k, g >= 0, sum(g) = 1.
    Returns (g_normalized, t) or (None, None).
    """
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

    M = best_obj

    no_improve = 0
    it = 0

        it += 1

        g, lp_t = lp_min_peak(f, n)
        if g is None:
            break
        gs = g.sum()
        g = g / gs if gs > 1e-15 else np.ones(n) / n
        g = np.maximum(g, 0)
        gs = g.sum()
        g = g / gs if gs > 1e-15 else np.ones(n) / n

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            # Bisection refinement
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            f = f_new
            M = best_val
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 35:
            rng = np.random.default_rng(it * 104729 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")

    return best_f, best_obj


def targeted_cd(f0, time_limit, verbose=False):
    """
    Coordinate descent that targets coordinates contributing to the top peaks.
    For each top peak, adjusts the highest-contributing coordinates.
    """
    f = np.maximum(f0, 0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        top_k = min(4, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Find top contributing coordinates
            contribs = []
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    contribs.append((f[j] * f[partner], j))
            if not contribs:
                continue
            contribs.sort(reverse=True)

            for contrib, j in contribs[:8]:
                if f[j] < 3e-4:
                    continue

                best_j_obj = best_obj
                best_j_val = f[j]

                for mult in [0.5, 0.65, 0.8, 1.25, 1.5]:
                    new_val = f[j] * mult
                    if new_val < 0 or new_val > 0.4:
                        continue
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != f[j]:
                    f[j] = best_j_val
                    f = f / f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 6:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Cubic-like interpolation of sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Adaptive SG (long run, moderate peaks)
    2. LP-FW (strong directional pulls)
    3. Adaptive SG (sharp + broad variants)
    4. Multi-length exploration
    5. Coordinate descent polish
    6. Final fine-tuning with perturbations
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # --- Get parent sequence ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Long adaptive subgradient (workhorse) ===
    print(f"\n=== Phase 1: Adaptive SG main (n={base_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(550, rem * 0.38)
    if t1 > 60:
        print(f"  Time budget: {t1:.0f}s")
        a_opt, _ = adaptive_sg(parent, time_limit=t1, lr=0.0035,
                               momentum=0.92, top_k=50, temp=4e-6,
                               use_mirror=False, verbose=True)
        try_seq(a_opt, "sg_main")

    # === Phase 2: LP-based Frank-Wolfe ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(350, rem * 0.22)
    if t2 > 30 and best_seq is not None:
        print(f"  n={len(best_seq)}, time budget: {t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        try_seq(a_opt, "lpfw")

    # === Phase 3: Adaptive SG variants ===
    print(f"\n=== Phase 3: SG variants (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()

        # Sharp targeting (top_k=1)
        t3a = min(200, rem * 0.15)
        if t3a > 20:
            print(f"  Sharp SG (top_k=1): {t3a:.0f}s")
            a_opt, _ = adaptive_sg(best_seq, time_limit=t3a, lr=0.002,
                                   momentum=0.93, top_k=1, temp=1e-7,
                                   use_mirror=False)
            try_seq(a_opt, "sg_sharp")

        # Broad targeting
        rem = deadline - time.monotonic()
        t3b = min(200, rem * 0.15)
        if t3b > 20:
            k_broad = max(30, n_best // 6)
            print(f"  Broad SG (top_k={k_broad}): {t3b:.0f}s")
            a_opt, _ = adaptive_sg(best_seq, time_limit=t3b, lr=0.002,
                                   momentum=0.91, top_k=k_broad, temp=2e-5,
                                   use_mirror=False)
            try_seq(a_opt, "sg_broad")

        # Mirror descent variant
        rem = deadline - time.monotonic()
        t3c = min(150, rem * 0.12)
        if t3c > 20:
            print(f"  Mirror SG: {t3c:.0f}s")
            a_opt, _ = adaptive_sg(best_seq, time_limit=t3c, lr=0.0025,
                                   momentum=0.92, top_k=40, temp=3e-6,
                                   use_mirror=True, mirror_every=2)
            try_seq(a_opt, "sg_mirror")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    candidate_lengths = []
    for delta in [-96, -48, 48, 96, 192]:
        ln = len(base) + delta
        if 256 <= ln <= 4096:
            candidate_lengths.append(ln)

    for target_n in candidate_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_alloc = min(rem * 0.2, 150)
        if t_alloc < 20:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  SG n={target_n}: {t_alloc:.0f}s")
        a_opt, _ = adaptive_sg(f_interp, time_limit=t_alloc, lr=0.003,
                               momentum=0.92, top_k=max(30, target_n // 15),
                               temp=5e-6)
        try_seq(a_opt, f"sg_n{target_n}")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n=== Phase 5: CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.15, 180)
        if t_cd > 15:
            print(f"  n={len(best_seq)}, time: {t_cd:.0f}s")
            a_opt, _ = targeted_cd(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 6: Final polish with perturbations ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine SG on best
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t = min(rem * 0.4, 80)
            a_opt, _ = adaptive_sg(best_seq, time_limit=t, lr=0.0008,
                                   momentum=0.95, top_k=25, temp=1.5e-6,
                                   restart_thresh=150)
            try_seq(a_opt, "sg_fine")

        # Perturbed restarts
        for p in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 12:
                break
            noise_scale = 0.0008 + 0.0008 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t = min(rem * 0.6, 35)
            if t < 8:
                break
            a_opt, _ = adaptive_sg(f_pert, time_limit=t, lr=0.0006,
                                   momentum=0.95, top_k=20, temp=1e-6,
                                   restart_thresh=100)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
