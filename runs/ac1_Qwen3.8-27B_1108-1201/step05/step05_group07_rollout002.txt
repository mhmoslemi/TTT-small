"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max gradient optimization
with L-BFGS-B in log-space parameterization, adaptive temperature annealing,
peak-shaving local search, multi-length exploration, and structured restarts.
Key innovations over previous approach: proper (not sub) gradients via FFT-based
cross-correlation, softmax parameterization for non-negativity, and targeted
peak-shaving moves that directly reduce the dominant convolution lag.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import expit


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max_value(conv, t):
    """Smooth approximation to max via log-sum-exp."""
    m = conv.max()
    x = t * (conv - m)
    return m + np.log(np.exp(x).sum()) / t


def smooth_max_weights(conv, t):
    """Softmax weights for smooth max."""
    m = conv.max()
    x = t * (conv - m)
    e = np.exp(x)
    return e / e.sum()


def grad_smooth_max_wrt_a(a, conv, t):
    """
    Gradient of smooth_max(conv(a,a)) w.r.t. a.
    Returns vector g where g[j] = d(smooth_max)/d(a[j]).
    Uses FFT cross-correlation for efficiency.
    """
    n = len(a)
    p = smooth_max_weights(conv, t)
    # g[j] = 2 * sum_k p[k] * a[k-j] (valid indices only)
    # This is 2 * cross_correlation(p, a)[j]
    # Need N >= len(p) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    N = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(P * np.conj(A), N)
    return 2.0 * c[:n]


def logspace_objective_and_grad(x, n, temperature):
    """
    Objective and gradient in log-space (softmax) parameterization.
    a = softmax(x), objective = 2*n*smooth_max(conv(a,a))
    """
    # Compute softmax
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    a = e / e.sum()

    conv = fast_convolve(a)
    sm = smooth_max_value(conv, temperature)
    obj = 2.0 * n * sm

    # Gradient of smooth_max w.r.t. a
    g = grad_smooth_max_wrt_a(a, conv, temperature)

    # Chain rule: da_i/dx_j = a_i*(delta_ij - a_j)
    # df/dx_j = 2n * sum_i g_i * a_i * (delta_ij - a_j)
    #         = 2n * a_j * (g_j - sum_i g_i*a_i)
    G = np.dot(g, a)
    grad_x = 2.0 * n * a * (g - G)

    return obj, grad_x


def lbfgs_smooth_max(f0, time_limit, t_start=50.0, t_end=5000.0, max_iter=2000):
    """
    L-BFGS-B optimization in log-space with annealed temperature.
    f0 is the initial sequence (will be normalized to sum=1).
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    # Initialize in log-space
    x0 = np.log(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max() * 2.0 * n
    t0 = time.monotonic()

    # Anneal temperature in stages
    n_stages = 4
    t_values = np.geomspace(t_start, t_end, n_stages)
    x_cur = x0.copy()

    for stage, temp in enumerate(t_values):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        stage_time = min(rem * (0.5 if stage < n_stages - 1 else 0.6), rem - 2)
        if stage_time < 3:
            stage_time = max(rem * 0.4, 1)

        iters_done = 0

        def obj_grad(x):
            nonlocal iters_done
            iters_done += 1
            o, g = logspace_objective_and_grad(x, n, temp)
            return o, g

        try:
            res = minimize(
                obj_grad, x_cur, method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter, 'gtol': 1e-8,
                         'maxfun': max_iter * 2}
            )
            x_cur = res.x
        except Exception:
            pass

        # Extract and evaluate
        x_shifted = x_cur - x_cur.max()
        e = np.exp(x_shifted)
        a = e / e.sum()
        a = np.maximum(a, 0)
        a = a / a.sum()

        conv = fast_convolve(a)
        obj = conv.max() * 2.0 * n
        if obj < best_obj:
            best_obj = obj
            best_f = a.copy()

    return best_f, best_obj / (2.0 * n)


def peak_shaving_search(f0, time_limit, verbose=False):
    """
    Targeted peak-shaving: identify the dominant convolution lag and
    redistribute mass to reduce it while not increasing other lags.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        cur_max = conv.max()
        peak_lag = int(np.argmax(conv))

        # Get contributors to the peak
        js = max(0, peak_lag - n + 1)
        je = min(n - 1, peak_lag)
        idx = np.arange(js, je + 1)
        partners = peak_lag - idx
        contribs = f[idx] * f[partners]

        # Find the pair that contributes most to the peak
        # Try to reduce it by moving mass
        order = np.argsort(contribs)[::-1]

        improved = False
        for oi in order[:min(20, len(order))]:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(idx[oi])
            j = int(partners[oi])
            if f[i] < 1e-12 or f[j] < 1e-12:
                continue

            # Try moving a small fraction from the larger to a "safe" index
            # Safe indices are those far from any peak
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = min(f[i], f[j]) * frac
                # Try moving from i to a random non-peak-contributing index
                # or from i to j if that helps (balanced pair)
                for target in [j] + list(np.random.choice(n, size=3, replace=False)):
                    if target == i:
                        continue
                    f_trial = f.copy()
                    f_trial[i] -= amt
                    f_trial[target] += amt
                    f_trial = np.maximum(f_trial, 0)
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f_trial.copy()
                        f = f_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random perturbation + local optimization
            rng = np.random.default_rng(int(time.monotonic() * 1e6))
            for _ in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or f[i] < 1e-12:
                    continue
                amt = f[i] * rng.uniform(0.02, 0.15)
                f_trial = f.copy()
                f_trial[i] -= amt
                f_trial[j] += amt
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    improved = True
            if not improved:
                break

    return best_f, best_obj


def multiplicative_weights_refine(f0, time_limit, top_k=50, temperature=0.00005,
                                   lr=0.003, momentum=0.85):
    """
    Multiplicative weights subgradient optimization (kept from previous approach
    as a secondary method, with some improvements).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        k_actual = min(cur_top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]

        v = momentum * v + (1 - momentum) * sg

        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        if cur_lr < 0.00003:
            cur_lr = 0.00003
        cur_temp *= 1.01

        if no_improve > 200:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.003
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 0)
            f = f / f.sum()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 1e-10)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    print(f"\n=== Phase 6: Combined Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()

        # L-BFGS-B fine-tune
        if rem > 60:
            t6 = min(rem * 0.2, 150)
            print(f"  L-BFGS fine: {t6:.0f}s")
            a_opt, _ = lbfgs_smooth_max(best_seq, time_limit=t6,
                                         t_start=500.0, t_end=10000.0)
            try_seq(a_opt, "lbfgs_fine")

        # Peak shaving again
        rem = deadline - time.monotonic()
        if rem > 40:
            t6b = min(rem * 0.2, 100)
            print(f"  Peak shave 2: {t6b:.0f}s")
            a_opt, _ = peak_shaving_search(best_seq, time_limit=t6b)
            try_seq(a_opt, "peak_shave2")

        # MU with different params
        rem = deadline - time.monotonic()
        if rem > 30:
            t6c = min(rem * 0.2, 100)
            print(f"  MU variant: {t6c:.0f}s")
            a_opt, _ = multiplicative_weights_refine(
                best_seq, time_limit=t6c,
                top_k=max(10, n_best // 10), temperature=0.00001,
                lr=0.001, momentum=0.92)
            try_seq(a_opt, "mu_fine")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()

        # Very fine L-BFGS
        if rem > 30:
            t7 = min(rem * 0.3, 60)
            a_opt, _ = lbfgs_smooth_max(best_seq, time_limit=t7,
                                         t_start=2000.0, t_end=20000.0)
            try_seq(a_opt, "final_lbfgs")

        # Final peak shaving
        rem = deadline - time.monotonic()
        if rem > 15:
            t7b = min(rem * 0.5, 30)
            a_opt, _ = peak_shaving_search(best_seq, time_limit=t7b)
            try_seq(a_opt, "final_peak")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
