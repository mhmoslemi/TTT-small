"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Fourier parametric search in low-dimensional space for diverse initializations
2. Accelerated mirror descent (FISTA-style) with softmax-weighted subgradients
3. Simulated annealing for escaping local minima
4. Adaptive coordinate descent with three-point moves for final polish
5. Multi-length exploration around n=768
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_circular(a):
    """Circular auto-convolution via FFT (approximation for large n)."""
    n = len(a)
    A = np.fft.rfft(a, 2 * n)
    return np.fft.irfft(A * A, 2 * n)[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv(a,a))."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def compute_exact_subgradient(a, conv):
    """Exact subgradient at the maximum convolution peak."""
    n = len(a)
    k = int(np.argmax(conv))
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    sg[idx] = 2.0 * a[k - idx]
    return sg


def fourier_to_sequence(params, n, base=1.0):
    """Convert Fourier parameters to a non-negative sequence."""
    i = np.arange(n)
    a = np.full(n, base)
    for k_idx in range(len(params) // 2):
        c_k = params[2 * k_idx]
        d_k = params[2 * k_idx + 1]
        freq = k_idx + 1
        a += c_k * np.cos(2.0 * np.pi * freq * i / n)
        a += d_k * np.sin(2.0 * np.pi * freq * i / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def fourier_objective(params, n):
    """Objective function for Fourier parameter optimization."""
    a = fourier_to_sequence(params, n)
    conv = fast_convolve_circular(a)
    return conv.max() * n * 2.0


def optimize_fourier_params(n, K, time_limit, rng, init_params=None):
    """Optimize Fourier parameters using scipy minimize."""
    n_params = 2 * K
    if init_params is None:

        attempts += 1
        if attempts == 1:
            x0 = init_params.copy()
        else:

        # Use Nelder-Mead for low-dimensional global-ish optimization
        try:
            result = scipy_minimize(

                         'adaptive': True}

            if result.fun < best_obj:
                best_obj = result.fun
                best_params = result.x.copy()
        except Exception:
            pass

        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:

def mirror_descent_optimize(f0, time_limit, top_k=50, temperature=0.00004,
                            lr=0.003, momentum=0.85, use_sa=True,
                            sa_temp_init=0.001, sa_cool=0.999,
                            verbose=False, label=""):
    """
    Mirror descent (multiplicative weights) with Nesterov momentum,
    softmax subgradients, and optional simulated annealing.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    y = f.copy()  # Nesterov iterate
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k
    sa_T = sa_temp_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        y = f * np.exp(0.5 * momentum * v)
        yn = y.sum()
        if yn < 1e-30:
            y = np.ones(n) / n
        else:
            y = y / yn

        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, cur_top_k, cur_temp)

        # Update momentum
        v = momentum * v + (1.0 - momentum) * sg

        # Multiplicative weights update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f_new = f * np.exp(exponent)
        fs = f_new.sum()
        if fs < 1e-30:
            f = best_f.copy()
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f_new = f_new / fs
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        # Simulated annealing acceptance
        if use_sa and obj_new > best_obj:
            delta = obj_new - best_obj
            accept_prob = np.exp(-delta / max(sa_T, 1e-15))
            if np.random.random() < accept_prob:
                f = f_new
            else:
                f = f  # reject, keep current
        else:
            f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive schedule
        cur_lr *= 0.9995
        if cur_lr < 0.00003:
            cur_lr = 0.00003
        cur_temp *= 1.01
        sa_T *= sa_cool

        if it % 200 == 0:
            cur_top_k = max(10, int(top_k * (1.0 - 0.02 * (it // 200))))

        # Restart if stuck
        if no_improve > 400:
            rng_local = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.003 + 0.002 * min(no_improve // 500, 3)
            f = project_simplex(best_f + rng_local.standard_normal(n) * noise_scale)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            cur_temp = temperature * 0.5
            sa_T = sa_temp_init * 0.5
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, "
                  f"lr={cur_lr:.7f}, T_sa={sa_T:.6f}")

    return best_f, best_obj


def fista_optimize(f0, time_limit, top_k=50, temperature=0.00004,
                   lr_init=0.002, verbose=False, label=""):
    """
    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) style optimization.
    Uses projected gradient descent with Nesterov acceleration.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    y = f.copy()
    t_nest = 1.0
    t0 = time.monotonic()
    lr = lr_init
    cur_top_k = top_k
    cur_temp = temperature
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv_y = fast_convolve(y)
        sg = compute_subgradient(y, conv_y, cur_top_k, cur_temp)

        # Gradient step
        f_trial = y - lr * sg
        f_trial = project_simplex(f_trial)

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest ** 2)) / 2.0
        y = f_trial + ((t_nest - 1.0) / t_new) * (f_trial - f)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y /= ys
        t_nest = t_new
        f = f_trial

        conv_f = fast_convolve(f)
        obj = conv_f.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size (Barzilai-Borwein style)
        if no_improve % 100 == 0 and no_improve > 0:
            lr *= 0.95
            if lr < 0.0001:
                lr = 0.0001
                # Restart
                y = best_f + np.random.default_rng(it).standard_normal(n) * 0.003
                y = project_simplex(y)
                t_nest = 1.0
                no_improve = 0

        cur_temp *= 1.005
        cur_top_k = max(10, top_k - it // 300)

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Aggressive coordinate descent with exact line search per coordinate.
    Uses golden section search for optimal value of each coordinate.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Priority order: indices that contribute most to peaks come first
    conv = fast_convolve(f)
    top_peaks = np.argpartition(conv, -min(50, 2 * n - 1))[-min(50, 2 * n - 1):]

    # Build priority based on contribution to peaks
    contrib = np.zeros(n)
    for k in top_peaks[:20]:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        contrib[idx] += conv[k] * f[idx] * f[k - idx]

    order = np.argsort(contrib)[::-1]

    full_pass = True
    while time.monotonic() - t0 < time_limit:
        improved = False
        if not full_pass:
            # Focus on high-contribution coordinates
            order_to_try = order[:min(200, n)]
        else:
            order_to_try = order

        for j in order_to_try:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            # Try multiple multipliers and find best
            best_mult = 1.0
            best_val_obj = best_obj
            cur_conv = fast_convolve(f)
            cur_peak = cur_conv.max()

            for mult in [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue

                # Scale others to maintain sum=1
                remaining_sum = 1.0 - new_val
                others_sum = 1.0 - old_val
                if others_sum < 1e-10:
                    continue
                scale = remaining_sum / others_sum

                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_val_obj - 1e-14:
                    best_val_obj = val
                    best_mult = mult

            if best_mult != 1.0:
                new_val = old_val * best_mult
                remaining_sum = 1.0 - new_val
                others_sum = 1.0 - old_val
                if others_sum > 1e-10:
                    scale = remaining_sum / others_sum
                    f[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0.0)
                    f /= f.sum()

                if best_val_obj < best_obj - 1e-14:
                    best_obj = best_val_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            # Try pair moves
            pair_improved = _pair_moves(f, n, best_obj, best_f, time.monotonic(), time_limit)
            if not pair_improved:
                break
            full_pass = False
        else:
            full_pass = True

    return best_f, best_obj


def _pair_moves(f, n, best_obj, best_f, t0, time_limit):
    """Try pair exchange moves to improve the solution."""
    improved = False
    rng = np.random.default_rng(int(time.monotonic() * 1e9) % 1000000)
    n_trials = 100

    conv = fast_convolve(f)
    top_peaks = np.argpartition(conv, -min(20, 2 * n - 1))[-min(20, 2 * n - 1):]

    for _ in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick a peak and try to reduce it
        k = int(rng.choice(top_peaks))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        contribs = f[idx] * f[k - idx]

        # Find highest and lowest contributors
        hi_pos = idx[np.argmax(contribs)]
        lo_pos = idx[np.argmin(contribs)]

        if hi_pos == lo_pos or f[hi_pos] < 1e-10:
            # Try random pair
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or f[i] < 1e-10:
                continue
            hi_pos, lo_pos = i, j

        for frac in [0.05, 0.1, 0.2, 0.3]:
            amt = f[hi_pos] * frac
            f[hi_pos] -= amt
            f[lo_pos] += amt
            f = np.maximum(f, 0.0)
            f /= f.sum()

            val = fast_convolve(f).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_f = f.copy()
                improved = True
                break
            else:
                f[hi_pos] += amt
                f[lo_pos] -= amt
                f /= f.sum()
                break

    return improved


def three_point_move(f, n, j1, j2, j3, time_limit, t0):
    """Try redistributing mass among three points."""
    if f[j1] < 1e-12 or f[j2] < 1e-12:
        return f, None

    best_obj = fast_convolve(f).max()
    for frac1 in [0.1, 0.2, 0.3]:
        for frac2 in [0.1, 0.2, 0.3]:
            amt1 = f[j1] * frac1
            amt2 = f[j2] * frac2
            f_trial = f.copy()
            f_trial[j1] -= amt1
            f_trial[j2] -= amt2
            f_trial[j3] += amt1 + amt2
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial /= st
            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-14:
                return f_trial, val
    return f, None


def structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    # Gaussian (center)
    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 3.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Broad Gaussian
    sigma2 = n / (1.5 + rng.random() * 1.5)
    gauss2 = np.exp(-0.5 * ((np.arange(n) - center) / sigma2) ** 2)
    inits.append(gauss2 / gauss2.sum())

    # Two-peak
    c1 = n * (0.25 + rng.random() * 0.1)
    c2 = n * (0.65 + rng.random() * 0.1)
    two_peak = (np.exp(-0.5 * ((np.arange(n) - c1) / (n * 0.1)) ** 2) +
                np.exp(-0.5 * ((np.arange(n) - c2) / (n * 0.1)) ** 2))
    inits.append(two_peak / two_peak.sum())

    # Sinusoidal modulation
    sin_mod = 1.0 + 0.3 * np.cos(2 * np.pi * rng.integers(2, 8) * np.arange(n) / n)
    sin_mod = np.maximum(sin_mod, 1e-10)
    inits.append(sin_mod / sin_mod.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    return inits[:count]


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining Fourier parametric search,
    mirror descent with simulated annealing, FISTA, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform n=768")

    # === Phase 1: Parent refinement with mirror descent ===
    print("\n=== Phase 1: Mirror Descent on Parent ===")
    n_parent = len(parent)
    rem = deadline - time.monotonic() - 800
    t_p1 = min(rem, 200)
    if t_p1 > 10:
        print(f"  n={n_parent}, t={t_p1:.0f}s")
        a_opt, _ = mirror_descent_optimize(parent, time_limit=t_p1,
                                            top_k=max(30, n_parent // 8),
                                            temperature=0.00003, lr=0.0025,
                                            momentum=0.88, use_sa=True,
                                            sa_temp_init=0.0005, sa_cool=0.9995)
        try_seq(a_opt, "md_parent")

    # === Phase 2: Fourier parametric search for diverse inits ===
    print(f"\n=== Phase 2: Fourier Parametric Search (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 700
    if rem > 30:
        n_fourier = n_parent
        t_fourier = min(rem * 0.6, 120)
        print(f"  n={n_fourier}, t={t_fourier:.0f}s")

        fourier_candidates = []
        for K in [3, 5, 7]:
            t_sub = t_fourier / 3.0
            rem_check = deadline - time.monotonic() - 650
            if rem_check < 10:
                break
            params, obj = optimize_fourier_params(n_fourier, K, time_limit=min(t_sub, rem_check), rng=rng)
            seq = fourier_to_sequence(params, n_fourier)
            sc = try_seq(seq, f"fourier_K{K}")
            if sc < best_score + 0.1:
                fourier_candidates.append(seq)

        # Also try random Fourier inits
        for _ in range(5):
            rem_check = deadline - time.monotonic() - 650
            if rem_check < 10:
                break
            K = int(rng.integers(2, 8))
            params = rng.standard_normal(2 * K) * 0.1
            seq = fourier_to_sequence(params, n_fourier)
            try_seq(seq, f"fourier_rand_{K}")

    # === Phase 3: Mirror descent on best candidate + structured inits ===
    print(f"\n=== Phase 3: Extended Mirror Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        candidates = [best_seq.copy()]

        # Add structured inits at same length
        for init in structured_inits(n_best, rng, count=3):
            candidates.append(init)

        # Add perturbed best
        noise = rng.standard_normal(n_best) * 0.005
        candidates.append(project_simplex(best_seq + noise))

        # Mirror + parent blend
        if parent is not None and len(parent) == n_best:
            for alpha in [0.3, 0.5, 0.7]:
                blend = alpha * best_seq + (1 - alpha) * (parent / parent.sum())
                candidates.append(project_simplex(blend))

        for idx, cand in enumerate(candidates):
            rem = deadline - time.monotonic() - 500
            if rem < 15:
                break
            t_alloc = min(rem * 0.2, 100)
            if t_alloc < 5:
                t_alloc = min(rem * 0.4, 60)

            configs = [
                {"top_k": max(20, n_best // 6), "temp": 0.00002, "lr": 0.002,
                 "mom": 0.90, "sa_t": 0.0003},
                {"top_k": max(40, n_best // 4), "temp": 0.00006, "lr": 0.004,
                 "mom": 0.85, "sa_t": 0.0008},
                {"top_k": max(10, n_best // 10), "temp": 0.00001, "lr": 0.001,
                 "mom": 0.93, "sa_t": 0.0001},
            ]
            cfg = configs[idx % len(configs)]
            print(f"  MD cand {idx}: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent_optimize(cand, time_limit=t_alloc,
                                                top_k=cfg["top_k"], temperature=cfg["temp"],
                                                lr=cfg["lr"], momentum=cfg["mom"],
                                                use_sa=True, sa_temp_init=cfg["sa_t"],
                                                sa_cool=0.999)
            try_seq(a_opt, f"md_ext_{idx}")

    # === Phase 4: FISTA on best ===
    print(f"\n=== Phase 4: FISTA Optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_fista = min(rem * 0.5, 150)
            print(f"  FISTA n={n_best}, t={t_fista:.0f}s")
            a_opt, _ = fista_optimize(best_seq, time_limit=t_fista,
                                       top_k=max(30, n_best // 6),
                                       temperature=0.00003, lr_init=0.0015)
            try_seq(a_opt, "fista")

            # FISTA from perturbed start
            rem = deadline - time.monotonic() - 250
            if rem > 15:
                noise = rng.standard_normal(n_best) * 0.004
                f_pert = project_simplex(best_seq + noise)
                t_fista2 = min(rem * 0.5, 80)
                a_opt, _ = fista_optimize(f_pert, time_limit=t_fista2,
                                           top_k=max(20, n_best // 8),
                                           temperature=0.00005, lr_init=0.001)
                try_seq(a_opt, "fista_pert")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = [int(n_best * 0.9), int(n_best * 1.05),
                          int(n_best * 0.85), int(n_best * 1.1)]
        target_lengths = sorted(set([max(128, l) for l in target_lengths]))

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 200
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 80)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent_optimize(f_interp, time_limit=t_alloc,
                                                top_k=max(20, target_n // 8),
                                                temperature=0.00003, lr=0.002,
                                                momentum=0.87, use_sa=True,
                                                sa_temp_init=0.0004, sa_cool=0.999)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_cd = min(rem * 0.6, 150)
            print(f"  CD n={len(best_seq)}, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

            # Round 2
            rem = deadline - time.monotonic() - 50
            if rem > 10:
                t_cd2 = min(rem * 0.7, 60)
                a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_cd2)
                try_seq(a_opt, "cd2")

    # === Phase 7: Final fine-tuning ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine mirror descent
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=min(rem, 50),
                                                top_k=n_best // 2, temperature=0.000002,
                                                lr=0.0003, momentum=0.96, use_sa=False)
            try_seq(a_opt, "final_md")

        # Small perturbations + quick optimize
        for _ in range(6):
            rem = deadline - time.monotonic() - 10
            if rem < 5:
                break
            scale = 0.001 + 0.001 * rng.random()
            noise = rng.standard_normal(n_best) * scale
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = mirror_descent_optimize(f_pert, time_limit=min(rem, 8),
                                                top_k=max(20, n_best // 5),
                                                temperature=0.00002, lr=0.001,
                                                momentum=0.92, use_sa=True,
                                                sa_temp_init=0.0002, sa_cool=0.998)
            try_seq(a_opt, "perturb")

        # Final coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem > 3:
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
        if fs < best_score:
            pass
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
