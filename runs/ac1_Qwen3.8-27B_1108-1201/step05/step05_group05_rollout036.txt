"""

Key insight: optimal sequences are NON-palindromic - asymmetry reduces the

import numpy as np
from scipy.special import logsumexp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def cross_corr(w, a):
    """Compute (w star a)[j] = sum_k w[k]*a[k-j] for j=0..n-1."""
    n = len(a)
    size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)
    return corr[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def stable_softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    ez = np.exp(z_shifted)
    return ez / ez.sum()


def smooth_obj_grad(z, M):
    """Compute smooth objective value and gradient in z-space."""
    a = stable_softmax(z)
    n = len(a)
    conv = fast_convolve(a)
    scaled = M * conv
    smooth_max_val = logsumexp(scaled) / M
    obj = 2.0 * n * smooth_max_val
    w = np.exp(scaled - logsumexp(scaled))
    grad_a = 2.0 * cross_corr(w, a)
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    return obj, 2.0 * n * grad_z

                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-12:
                    continue
                a_trial /= s
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_asymmetric_init(n, shape='linear', seed=42):
    """Create asymmetric initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    if shape == 'linear':
        a = x + 1.0
    elif shape == 'quad':
        a = (x + 1.0) ** 2
    elif shape == 'exp':
        t = rng.uniform(0.5, 2.0)
        a = np.exp(t * x / n)
    elif shape == 'cubic':
        a = (x + 1.0) ** 3
    elif shape == 'sqrt':
        a = np.sqrt(x + 1.0)
    elif shape == 'power':
        p = rng.uniform(1.5, 4.0)
        a = (x + 1.0) ** p
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining smooth gradient (log-sum-exp continuation),
    subgradient polishing, coordinate descent, and multi-length exploration.
    Exploits the key insight that non-palindromic sequences achieve lower scores.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_asymmetric_init(768, 'linear')
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default init: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Smooth optimization on parent ===
    print(f"\n=== Phase 1: Smooth Opt on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(remaining_s(rem), 250)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_opt, _ = smooth_optimize(parent, time_limit=t1, n=n_parent,
                                   M_start=8.0, M_end=1500.0)
        try_seq(a_opt, "smooth_parent")

    # === Phase 2: Smooth opt with different initializations ===
    print(f"\n=== Phase 2: Diverse Initializations ===")
    inits = [
        ("linear", make_asymmetric_init(n_parent, 'linear', seed)),
        ("quad", make_asymmetric_init(n_parent, 'quad', seed + 1)),
        ("exp", make_asymmetric_init(n_parent, 'exp', seed + 2)),
        ("cubic", make_asymmetric_init(n_parent, 'cubic', seed + 3)),
        ("blend_parent_lin", project_simplex(0.6 * parent + 0.4 * make_asymmetric_init(n_parent, 'linear', seed + 4))),
        ("blend_parent_quad", project_simplex(0.6 * parent + 0.4 * make_asymmetric_init(n_parent, 'quad', seed + 5))),
    ]

    for name, f_init in inits:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.2, 120)
        print(f"  Init [{name}], t={t_alloc:.0f}s", end="")
        a_opt, _ = smooth_optimize(f_init, time_limit=t_alloc, n=n_parent,
                                   M_start=6.0, M_end=800.0)
        scr = float(evaluate_sequence(a_opt.tolist()))
        print(f" -> score={scr:.6f}")
        try_seq(a_opt, f"init_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-Length (best={best_score:.6f}) ===")
    for target_n in [700, 800, 850, 900, 1024]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(rem * 0.25, 150)
        print(f"  n={target_n}, t={t_alloc:.0f}s", end="")
        a_opt, _ = smooth_optimize(f_interp, time_limit=t_alloc, n=target_n,
                                   M_start=8.0, M_end=1200.0)
        scr = float(evaluate_sequence(a_opt.tolist()))
        print(f" -> score={scr:.6f}")
        try_seq(a_opt, f"n{target_n}")

    # === Phase 4: Subgradient polishing ===
    print(f"\n=== Phase 4: Subgradient Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_best = len(best_seq)
        t_sg = min(rem * 0.35, 300)
        print(f"  SG t={t_sg:.0f}s, n={n_best}")
        a_opt, _ = subgradient_polish(best_seq, time_limit=t_sg,
                                      lr_init=0.015, top_k=max(15, n_best // 20))
        try_seq(a_opt, "sg")

        # Refine with smaller top_k
        rem = deadline - time.monotonic()
        if rem > 30:
            t_sg2 = min(rem * 0.25, 150)
            a_opt2, _ = subgradient_polish(a_opt, time_limit=t_sg2,
                                           lr_init=0.008, top_k=max(8, n_best // 40))
            try_seq(a_opt2, "sg_fine")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_cd = min(rem * 0.35, 250)
        print(f"  CD t={t_cd:.0f}s")
        a_opt, _ = coord_descent(best_seq, time_limit=t_cd)
        try_seq(a_opt, "cd")

        rem = deadline - time.monotonic()
        if rem > 20:
            t_cd2 = min(rem * 0.3, 120)
            a_opt2, _ = coord_descent(a_opt, time_limit=t_cd2)
            try_seq(a_opt2, "cd2")

    # === Phase 6: Final smooth opt on best ===
    print(f"\n=== Phase 6: Final Smooth Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        n_best = len(best_seq)
        t_final = min(rem * 0.4, 200)
        print(f"  Final smooth t={t_final:.0f}s")
        a_opt, _ = smooth_optimize(best_seq, time_limit=t_final, n=n_best,
                                   M_start=50.0, M_end=5000.0)
        try_seq(a_opt, "final_smooth")

        # Subgradient after
        rem = deadline - time.monotonic()
        if rem > 20:
            a_opt2, _ = subgradient_polish(a_opt, time_limit=min(rem * 0.3, 80),
                                           lr_init=0.005, top_k=max(5, n_best // 50))
            try_seq(a_opt2, "final_sg")

    # === Phase 7: Perturbation + re-optimize ===
    print(f"\n=== Phase 7: Perturbations (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for i in range(6):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            scale = 0.003 * (i + 1)
            noise = rng.standard_normal(n_best) * scale
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.25, 60)
            a_opt, _ = smooth_optimize(f_pert, time_limit=t_pert, n=n_best,
                                       M_start=20.0, M_end=2000.0)
            try_seq(a_opt, f"perturb_{i}")

    # Fallback
    if best_seq is None:
        best_seq = make_asymmetric_init(512, 'linear')
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def remaining_s(rem):
    """Helper to avoid closure issues (just returns rem)."""
    return rem
