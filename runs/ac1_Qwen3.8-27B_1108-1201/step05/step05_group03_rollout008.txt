    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_cd(f0, time_limit, verbose=False):
    """Fast coordinate descent minimizing max(conv(f,f)) with sum(f)=1."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    no_improve = 0
    pass_num = 0

    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        pass_improved = False

        if pass_num % 7 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx]
            s_i = f[i]
            denom = 1.0 - s_i
            if denom < 1e-12:
                continue

            b = f / denom
            b[i] = 0.0
            conv_b = fast_convolve(b)

            # Build correction arrays (precompute for vectorization)
            # k ranges from i to i+n-1, j = k-i ranges from 0 to n-1
            b_slice = b[0:n]  # same as b since b has length n
            conv_b_slice = conv_b[i:i+n]
            conv_b_2i = conv_b[2 * i] if 2 * i < 2 * n - 1 else 0.0

            def obj_at_t(t):
                factor = (1.0 - t) * (1.0 - t)
                result = factor * conv_b.copy()
                affected = 2.0 * t * (1.0 - t) * b_slice + factor * conv_b_slice
                if 2 * i < 2 * n - 1:
                    affected[i] = t * t + factor * conv_b_2i
                result[i:i + n] = affected
                return result.max()

            # Grid search
            grid = np.linspace(0.0, 1.0, 17)
            grid_vals = np.array([obj_at_t(t) for t in grid])
            best_gi = int(np.argmin(grid_vals))
            best_t = grid[best_gi]
            best_val = grid_vals[best_gi]

            # Golden section refinement
            if 0 < best_gi < 16:
                lo = grid[best_gi - 1]
                hi = grid[best_gi + 1]
            elif best_gi == 0:
                lo, hi = 0.0, grid[1]
            else:
                lo, hi = grid[-2], 1.0

            gr = (np.sqrt(5) - 1) / 2
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = obj_at_t(c1)
            f2 = obj_at_t(c2)

            for _ in range(25):
                if hi - lo < 1e-10:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = obj_at_t(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = obj_at_t(c2)

            golden_t = (lo + hi) / 2.0
            golden_val = obj_at_t(golden_t)
            if golden_val < best_val:
                best_t = golden_t
                best_val = golden_val

            # Also check t=0 explicitly (might be optimal to zero out)
            zero_val = obj_at_t(0.0)
            if zero_val < best_val:
                best_t = 0.0
                best_val = zero_val

            current_val = obj_at_t(s_i)
            if best_val < current_val - 1e-15:
                f = b * (1.0 - best_t)
                f[i] = best_t
                f = np.maximum(f, 0.0)
                pass_improved = True

        if pass_improved:
            cur_obj = fast_convolve(f).max()
            if cur_obj < best_obj - 1e-15:
                best_obj = cur_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and pass_num % 100 == 0:
                    print(f"    CD pass {pass_num}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= 20:
            scale = 0.003 + 0.002 * min(no_improve // 10, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            no_improve = 0

    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, min(j + n, 2 * n - 1))
        A_ub[rows, j] = f[:len(rows)]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        g = lp_direction(f, M, n)
        if g is None:
            g = lp_direction(f, M * 1.01, n)
            if g is None:
                no_improve += 1
                if no_improve > 50:
                    break
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def block_cd(f0, time_limit, block_size=4, verbose=False):
    """Block coordinate descent: optimize small blocks of coordinates jointly."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    num_blocks = (n + block_size - 1) // block_size
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        block_order = rng.permutation(num_blocks)
        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            bs = end - start
            block_sum = f[start:end].sum()
            if block_sum < 1e-12:
                continue

            # Try optimizing the block with random directions
            for _ in range(5):
                pert = rng.standard_normal(bs) * 0.1 * block_sum
                trial = np.maximum(f[start:end] + pert, 0)
                ts = trial.sum()
                if ts < 1e-12:
                    continue
                trial = trial * (block_sum / ts)
                f_trial = f.copy()
                f_trial[start:end] = trial
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: Fast CD, LP-FW, multi-length, block CD, final polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if len(a) == 0 or a.sum() < 0.01:
            return
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Fast Coordinate Descent on parent ===
    print(f"\n=== Phase 1: Fast CD (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 600
    t1 = min(rem, 500)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_opt, _ = fast_cd(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "cd_p1")

    # === Phase 2: LP-FW on best ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 450
        t2 = min(rem, 200)
        if t2 > 20:
            print(f"  Time budget: {t2:.0f}s, n={len(best_seq)}")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
            try_seq(a_opt, "lpfw_p2")

    # === Phase 3: Fast CD again on updated best ===
    print(f"\n=== Phase 3: Fast CD round 2 (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        t3 = min(rem, 300)
        if t3 > 30:
            print(f"  Time budget: {t3:.0f}s, n={len(best_seq)}")
            a_opt, _ = fast_cd(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "cd_p3")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent

    for target_n in [512, 1024, 1536, 2048]:
        if target_n == len(base):
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.35, 100)
        if t_alloc > 15:
            print(f"  CD n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = fast_cd(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"cd_n{target_n}")

            # Quick LP-FW on the CD result
            rem = deadline - time.monotonic() - 150
            if rem > 15:
                t_lpfw = min(rem * 0.4, 60)
                if t_lpfw > 10:
                    a_opt2, _ = lp_fw_optimize(a_opt, time_limit=t_lpfw)
                    try_seq(a_opt2, f"lpfw_n{target_n}")

    # === Phase 5: Block coordinate descent ===
    print(f"\n=== Phase 5: Block CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_blk = min(rem * 0.7, 150)
            n_blk = len(best_seq)
            bs = max(3, min(8, n_blk // 100))
            print(f"  Time: {t_blk:.0f}s, block_size={bs}, n={n_blk}")
            a_opt, _ = block_cd(best_seq, time_limit=t_blk, block_size=bs)
            try_seq(a_opt, "block")

    # === Phase 6: Diverse restarts with fast CD ===
    print(f"\n=== Phase 6: Diverse restarts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(6):
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            t_alloc = min(rem / max(1, 6 - p), 80)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                rng = np.random.default_rng(200 + p)
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            elif p == 2:
                f_init = best_seq[::-1].copy()
                f_init = f_init / f_init.sum()
            elif p == 3:
                # Cosine-weighted
                x = np.linspace(0, 1, n_best)
                cw = (np.cos(np.pi * x) ** 2)
                f_init = cw / cw.sum()
            elif p == 4:
                # Bimodal
                centers = np.array([0.25, 0.75])
                sig = 0.1
                f_init = np.sum([np.exp(-((np.linspace(0, 1, n_best) - c) ** 2) / (2 * sig * sig)) for c in centers], axis=0)
                f_init = f_init / f_init.sum()
            else:
                # Mix of best and uniform
                f_init = 0.6 * best_seq + 0.4 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()

            a_opt, _ = fast_cd(f_init, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"diverse_{p}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_final_cd = min(rem * 0.5, 100)
            print(f"  Final CD: t={t_final_cd:.0f}s")
            a_opt, _ = fast_cd(best_seq, time_limit=t_final_cd, verbose=True)
            try_seq(a_opt, "cd_final")

        rem = deadline - time.monotonic() - 10
        if rem > 8:
            t_final_lpfw = min(rem * 0.6, 50)
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final_lpfw)
            try_seq(a_opt, "lpfw_final")

        # Tiny perturbations
        for trial in range(5):
            rem = deadline - time.monotonic() - 5
            if rem < 6:
                break
            rng = np.random.default_rng(800 + trial)
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.003)
            a_opt, _ = fast_cd(f_pert, time_limit=min(rem, 15))
            try_seq(a_opt, f"pert_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
