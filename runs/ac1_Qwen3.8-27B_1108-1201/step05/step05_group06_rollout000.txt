"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal descent directions (inspired by SOTA),
combined with coordinate bisection polishing and multi-length exploration.
The LP direction finding provides much better gradients than subgradient methods,
enabling faster convergence to near-optimal solutions.
"""

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Full cross-convolution via FFT."""
    n_f, n_g = len(f), len(g)
    size = n_f + n_g - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def build_conv_matrix(f):
    """Build matrix A where (A @ g)[k] = (f*g)(k) for all k in 0..2n-2."""
    n = len(f)
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n))
    for k in range(conv_len):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = f[k - je:k - js + 1][::-1]
    return A


def solve_direction_lp(f, M, solver_timeout=30):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)(k) <= M for all k.
    Returns normalized direction g/sum(g) or None if failed."""
    n = len(f)
    conv_len = 2 * n - 1

    A = build_conv_matrix(f)

    g = cp.Variable(n)
    obj = cp.Maximize(cp.sum(g))
    constraints = [g >= 0, A @ g <= M]

    prob = cp.Problem(obj, constraints)

    solvers_to_try = [cp.SCS, cp.CLARABEL, cp.ECOS, cp.GLPK]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver, max_iter=50000, verbose=False)
            if g.value is not None and np.all(np.isfinite(g.value)):
                val = np.maximum(g.value, 0.0)
                S = val.sum()
                if S > 1.0 + 1e-8:
                    return val
                return None
        except Exception:
            continue
    return None


def lp_optimize(f0, time_limit, verbose=True, min_improvement=1e-10):
    """LP-based optimization with adaptive step sizes and line search."""
    f = f0.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_M = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        direction = solve_direction_lp(f, M)

        if direction is None:
            if verbose:
                print(f"    LP: no improvement at iter {it}, M={M:.10f}")
            stall_count += 1
            if stall_count > 3:
                break
            continue

        S = direction.sum()
        g_norm = direction / S

        # Line search over step sizes
        best_t = 0.0
        best_obj = M
        step_candidates = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in step_candidates:
            f_trial = (1.0 - t) * f + t * g_norm
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial /= st
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - min_improvement:
                best_obj = obj
                best_t = t

        if best_t == 0.0:
            stall_count += 1
            if verbose and stall_count % 5 == 0:
                print(f"    LP stall at iter {it}, M={M:.10f}, S={S:.6f}")
            if stall_count > 5:
                break
            continue

        # Apply update
        f = (1.0 - best_t) * f + best_t * g_norm
        f = np.maximum(f, 0.0)
        st = f.sum()
        if st < 1e-15:
            break
        f /= st

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            stall_count = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: M={M:.10f}, S={S:.4f}, t={best_t:.2f}, "
                  f"elapsed={elapsed:.1f}s")

    return best_f, best_M


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_bisection(a0, time_limit, verbose=False):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < 1000:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0

            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-13:
                    hi = mid
                elif deriv < -1e-13:
                    lo = mid
                else:
                    break

            x_new = (lo + hi) / 2.0

            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check

            conv = fast_convolve(a)
            obj = conv.max()

            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            else:
                break

    return best_a, best_obj


def subgradient_phase(a0, time_limit, lr=0.008, mom=0.92, top_k=30,
                      temp=0.0005, decay=0.9998):
    """Projected subgradient descent with momentum (fallback method)."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        cur_lr *= decay
        cur_temp *= 1.003

        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 5)
            noise_scale = 0.002 + 0.002 * min(no_improve // 100, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric and renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym
