    f = f0.copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    
    for phase in range(num_phases):
        beta = beta_start * (beta_end / beta_start) ** (phase / num_phases)
        
        # Use L-BFGS-B on the LSE objective
        def objective(a):
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15: return best_obj
            a = a / s
            conv = fast_convolve(a)
            val = beta * conv
            val -= val.max()
            return val.max() + np.log(np.exp(val).sum())
        
        def gradient(a):
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15: return np.zeros(n)
            a = a / s
            conv = fast_convolve(a)
            val = beta * conv
            val -= val.max()

            w /= w.sum()
            grad = beta * np.correlate(w, a, mode='valid')

            grad = grad - a * grad.sum()
            return grad

            best_f = f.copy()
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_f, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-scale LSE gradient

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.irfft(A * A, fft_size)[:size]
    return B


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Scale to handle target_sum
    v = v.copy()
    # Standard algorithm for target_sum=1, then scale
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_and_grad(a, beta):
    """Compute LSE(beta * conv(a,a)) and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    val = beta * conv
    val_max = val.max()
    exp_val = np.exp(val - val_max)
    lse_val = val_max + np.log(exp_val.sum())
    w = exp_val / exp_val.sum()
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return lse_val, grad


def lse_optimize(f0, time_limit, beta_start=5.0, beta_end=50000.0,
                 num_betas=30, maxiter_per_beta=50, verbose=False, label=""):
    """
    Multi-scale LSE optimization using L-BFGS-B at each beta level.
    Beta is annealed from beta_start to beta_end.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    
    # Geometric schedule for beta
    betas = np.geomspace(beta_start, beta_end, num_betas)
    
    for bi, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        def objective(x):
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-15:
                return 1e10
            x = x / s
            val, _ = lse_and_grad(x, beta)
            return val
        
        def gradient(x):
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-15:
                return np.zeros(n)
            x = x / s
            _, grad = lse_and_grad(x, beta)
            # Adjust for the normalization constraint (sum=1)
            # The gradient of the constrained problem
            grad = grad - x * (grad @ x) / max(x.sum()**2, 1e-30)
            return grad
        
        # Use L-BFGS-B
        try:
            res = minimize(objective, f, jac=gradient, method='L-BFGS-B',
                         bounds=[(0, None)] * n,
                         options={'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 3,
                                  'maxls': 20})
            f_new = res.x
        except Exception:
            f_new = f
        
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            f_new = best_f.copy()
        else:
            f_new = f_new / s
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
        
        f = f_new
        
        if verbose and bi % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE beta={beta:.1f}, obj={best_obj:.10f}, t={elapsed:.1f}s")
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, np.maximum(f, 0))
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)

    hann = 0.5 * (1 - np.cos(2 * np.pi * x))

    inits.append(hann / hann.sum())
    
    # Gaussian
    center = n / 2.0
    sigma = n * (0.15 + rng.random() * 0.2)

    inits.append(gauss / gauss.sum())

    c1, c2 = n * 0.25, n * 0.75
    s = n * 0.1
    bi = np.exp(-0.5 * ((np.arange(n) - c1) / s) ** 2) + np.exp(-0.5 * ((np.arange(n) - c2) / s) ** 2)
    bi = np.maximum(bi, 1e-10)

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))

    # Sinc-like

    sinc = np.abs(sinc)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-scale log-sum-exp (LSE)
gradient optimization with L-BFGS-B on a softmax parameterization, combined
with exact coordinate descent, peak-directed mass redistribution, and
multi-length exploration. The LSE smooth approximation to max is annealed
from coarse to fine, providing proper (non-sub)gradients for fast convergence.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Full linear auto-convolution via real FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_and_grad(a, beta):
    """
    Compute LSE(beta * conv(a,a)) and gradient wrt a.
    Returns (lse_value, grad_a) where grad_a has length n.
    """
    n = len(a)
    conv = fast_convolve(a)
    val = beta * conv
    val_max = val.max()
    exp_val = np.exp(val - val_max)
    lse_val = val_max + np.log(exp_val.sum())
    w = exp_val / exp_val.sum()
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    return lse_val, grad_a


def lse_softmax_objective(z, beta):
    """Objective and gradient for LSE with softmax parameterization a = softmax(z)."""
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a_sum = a.sum()
    if a_sum < 1e-30:
        return 1e10, np.zeros(len(z))
    a = a / a_sum
    n = len(a)

    conv = fast_convolve(a)
    val = beta * conv
    val_max = val.max()
    exp_val = np.exp(val - val_max)
    exp_sum = exp_val.sum()
    if exp_sum < 1e-30:
        return 1e10, np.zeros(len(z))
    lse_val = val_max + np.log(exp_sum)
    w = exp_val / exp_sum
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return lse_val, grad_z


def lse_optimize(f0, time_limit, beta_start=5.0, beta_end=50000.0,
                 num_betas=25, maxiter_per_beta=40, verbose=False, label=""):
    """
    Multi-scale LSE optimization using L-BFGS-B on softmax parameterization.
    Beta is annealed geometrically from beta_start to beta_end.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    # Initialize z from f (log of f, clipped)
    z = np.log(np.maximum(f, 1e-30))
    z -= z.mean()  # center for numerical stability

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    betas = np.geomspace(beta_start, beta_end, num_betas)

    for bi, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit - 3:
            break

        def obj_fun(zv):
            val, _ = lse_softmax_objective(zv, beta)
            return val

        def grad_fun(zv):
            _, g = lse_softmax_objective(zv, beta)
            return g

        try:
            res = minimize(obj_fun, z, jac=grad_fun, method='L-BFGS-B',
                           options={'maxiter': maxiter_per_beta,
                                    'maxfun': maxiter_per_beta * 4,
                                    'maxls': 20,
                                    'ftol': 1e-10,
                                    'gtol': 1e-7})
            z = res.x
        except Exception:
            pass

        # Convert back to f
        z_shifted = z - z.max()
        f_new = np.exp(z_shifted)
        f_sum = f_new.sum()
        if f_sum < 1e-30:
            f_new = best_f.copy()
        else:
            f_new = f_new / f_sum

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()

        f = f_new

        if verbose and bi % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE [{label}] beta={beta:.1f}, obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_f, best_obj


def lse_projected_grad(f0, time_limit, beta_start=10.0, beta_end=80000.0,
                       num_iters=3000, lr_start=0.01, lr_end=0.0005,
                       verbose=False, label=""):
    """
    LSE optimization with projected gradient (no parameterization).
    Uses adaptive learning rate and momentum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    momentum = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    epsilon = 1e-8

    for it in range(num_iters):
        if time.monotonic() - t0 > time_limit - 2:
            break
        step += 1
        # Adaptive beta
        frac = min(1.0, it / (num_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac

        # Compute gradient
        lse_val, grad_a = compute_lse_and_grad(f, beta)
        # Normalize gradient for stable updates
        grad_norm = np.sqrt(np.mean(grad_a ** 2))
        if grad_norm > 1e-15:
            grad_a = grad_a / grad_norm

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_a
        v = beta2 * v + (1 - beta2) * grad_a ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        # Adaptive learning rate
        lr = lr_start * (lr_end / lr_start) ** (it / max(num_iters, 1))
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # Project to simplex
        f_new = project_simplex(f_new, 1.0)

        f = f_new
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PG [{label}] it={it}, beta={beta:.1f}, obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_f, best_obj

    t0 = time.monotonic()

            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.5, 0.0]
        else:
            multipliers = [0.9, 0.95, 1.05, 1.1]

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]

            for mult in multipliers:
                new_val = old_val * mult if old_val > 0 else 1e-8
                if new_val < 0 or new_val > 0.8:
                    continue

                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-13:
                    f = f_trial.copy()
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break

        if not improved:
            break

    return best_f, best_obj


def peak_directed_search(f0, time_limit, verbose=False):
    """
    Peak-directed local search: identify top peaks and redistribute mass
    from high-contributing to low-contributing positions to reduce peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(98765)

    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_max = conv.max()

        if current_max < best_obj - 1e-13:
            best_obj = current_max
            best_f = f.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > 100:
            break

        # Get top peaks
        top_k = min(15, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(-conv[peak_indices])]

        improved_this_round = False

        for k in peak_indices[:6]:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]

            if contribs.sum() < 1e-15:
                continue

            # Top contributors to this peak
            n_top = min(5, len(idx))
            top_contrib = np.argsort(-contribs)[:n_top]

            for ti in top_contrib:
                if time.monotonic() - t0 > time_limit:
                    break
                src = int(idx[ti])
                src_val = f[src]
                if src_val < 1e-10:
                    continue

                # Try moving mass to positions that don't contribute to top peaks
                # Score each target: low contribution to top peaks
                target_scores = np.zeros(n)
                for pk in peak_indices[:5]:
                    pk = int(pk)
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    # Position t contributes to peak pk if a[t]*a[pk-t] is large
                    pk_t = pk - np.arange(n)
                    valid = (pk_t >= 0) & (pk_t < n)
                    contrib_at_t = np.where(valid, f[pk_t[valid]], 0.0)
                    # We want positions where a[pk-t] is small (so adding mass doesn't help)
                    # AND where a[t] is currently small
                    target_scores += f * contrib_at_t

                # Add noise for diversity
                target_scores += rng.random(n) * 0.01

                # Best targets: low score, not the source
                target_scores[src] = np.inf
                n_targets = min(8, n)
                best_targets = np.argsort(target_scores)[:n_targets]

                for tgt in best_targets:
                    if time.monotonic() - t0 > time_limit:
                        break
                    tgt = int(tgt)
                    if tgt == src:
                        continue

                    for frac in [0.05, 0.1, 0.15, 0.25, 0.4]:
                        amt = src_val * frac
                        if amt < 1e-12:
                            break
                        f_trial = f.copy()
                        f_trial[src] -= amt
                        f_trial[tgt] += amt
                        f_trial = np.maximum(f_trial, 0)
                        st = f_trial.sum()
                        if st < 1e-15:
                            break
                        f_trial /= st
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-13:
                            f = f_trial.copy()
                            best_obj = val
                            best_f = f.copy()
                            improved_this_round = True
                            break

                    if improved_this_round:
                        break

                if improved_this_round:
                    break

            if improved_this_round:
                break

        if not improved_this_round:
            # Try random perturbations
            improved_any = False
            for scale in [0.003, 0.008, 0.015]:
                for _ in range(10):
                    if time.monotonic() - t0 > time_limit:
                        break
                    noise = rng.standard_normal(n) * scale
                    f_trial = project_simplex(f + noise)
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-13:
                        f = f_trial.copy()
                        best_obj = val
                        best_f = f.copy()
                        improved_any = True
                        improved_this_round = True
                        break
                if improved_this_round:
                    break

            if not improved_this_round:
                if no_improve_count > 50:
                    break
                # Stuck, try a bigger perturbation
                noise = rng.standard_normal(n) * 0.05
                f = project_simplex(f + noise)
                no_improve_count = 0

    return best_f, best_obj


def pair_exchange_search(f0, time_limit, verbose=False):
    """
    Pair-exchange local search: transfer mass between pairs of indices.
    Focuses on pairs that affect the top peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        top_k = min(10, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        improved = False

        # Try random pairs
        num_trials = 200
        for _ in range(num_trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if f[i] < 1e-12:
                continue

            for frac in [0.1, 0.2, 0.4]:
                amt = f[i] * frac
                f[i] -= amt
                f[j] += amt
                f = np.maximum(f, 0)
                st = f.sum()
                if st > 1e-15:
                    f /= st
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] += amt
                    f[j] -= amt
                    f = np.maximum(f, 0)
                    st = f.sum()
                    if st > 1e-15:
                        f /= st
                    break

            if improved:
                break

        if not improved:
            # Systematic: try to reduce top peak
            k = int(peak_indices[0])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            hi = idx[np.argmax(contribs)]
            # Find a low position
            lo_candidates = np.argsort(f)[:10]
            for lo in lo_candidates:
                if lo == hi:
                    continue
                for frac in [0.1, 0.2, 0.3]:
                    amt = f[hi] * frac
                    f[hi] -= amt
                    f[lo] += amt
                    f = np.maximum(f, 0)
                    st = f.sum()
                    if st > 1e-15:
                        f /= st
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-13:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[hi] += amt
                        f[lo] -= amt
                        f = np.maximum(f, 0)
                        st = f.sum()
                        if st > 1e-15:
                            f /= st
                        break
                if improved:
                    break

            if not improved:
                break

    return best_f, best_f_obj if False else best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length (preserving sum)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, np.maximum(f, 0))
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    # Gaussian (asymmetric center)
    center = n * (0.3 + rng.random() * 0.4)
    sigma = n * (0.1 + rng.random() * 0.15)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    gauss = np.maximum(gauss, 1e-10)
    inits.append(gauss / gauss.sum())

    # Bimodal
    c1, c2 = n * 0.25, n * 0.75
    s1 = n * 0.08
    bi = np.exp(-0.5 * ((np.arange(n) - c1) / s1) ** 2) + np.exp(-0.5 * ((np.arange(n) - c2) / s1) ** 2)
    bi = np.maximum(bi, 1e-10)
    inits.append(bi / bi.sum())

    # Linear ramp (asymmetric)
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())

    # Quadratic bump
    quad = 1.0 - (2 * x - 1) ** 2
    quad = np.maximum(quad, 1e-10)
    inits.append(quad / quad.sum())

    # Sinc-like
    t = np.arange(n) - n / 2.0
    t_safe = np.where(t == 0, 1e-10, t)
    sinc = np.sinc(t_safe / (n * 0.25))
    sinc = np.abs(sinc)
    sinc = np.maximum(sinc, 1e-10)
    inits.append(sinc / sinc.sum())

    return inits[:count]
