"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-Frank-Wolfe with vectorized
active-set LP, multi-length exploration, Nesterov-momentum subgradient, and
coordinate descent. Key insight: optimal sequences break palindromic symmetry
(palindromic gives objective >= 2). Uses efficient FFT convolution and
adaptive constraint selection for fast per-iteration cost.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, k_sorted, n):
    """Build LP constraint matrix for active set k_sorted."""
    K = len(k_sorted)
    A_ub = np.zeros((K, n))
    for row, k in enumerate(k_sorted):
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(n - 1, k_int)
        if js <= je:
            A_ub[row, js:je + 1] = f[k_int - je:k_int - js + 1][::-1]
    return A_ub


def solve_lp(f, M, n, k_sorted):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in k_sorted, g >= 0."""
    c = -np.ones(n)
    A_ub = build_lp_matrix(f, k_sorted, n)
    b_ub = np.full(len(k_sorted), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lpfw(f0, time_limit, top_k=35, verbose=False):
    """LP-Frank-Wolfe with adaptive active set and multi-point line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    alphas = [0.5, 0.3, 0.7, 0.4, 0.6, 0.2]

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1.0))

        # Adaptive top-k: broader early, tighter later
        k_use = max(8, int(top_k * (1.4 - 0.7 * progress)))
        if k_use >= 2 * n - 1:
            k_sorted = np.arange(2 * n - 1)
        else:
            k_sorted = np.argpartition(conv, -k_use)[-k_use:]

        # Solve LP direction
        g = solve_lp(f, M, n, k_sorted)
        if g is None:
            g = solve_lp(f, M * 1.05, n, k_sorted)
            if g is None:
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Multi-point line search
        best_alpha = 0.0
        best_trial = M
        for alpha in alphas:
            f_trial = f + alpha * (g - f)
            f_trial = project_simplex(f_trial)
            c_trial = fast_convolve(f_trial).max()
            if c_trial < best_trial:
                best_trial = c_trial
                best_alpha = alpha

        if best_alpha > 0:
            f_new = project_simplex(f + best_alpha * (g - f))
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            conv = conv_new

            if M_new < best_obj - 1e-15:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                M = M_new
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
            f = f_new
        else:
            no_improve += 1

        # Random restart if stuck
        if no_improve > 40:
            rng = np.random.default_rng(it * 17 + 3)
            scale = 0.001 + 0.002 * min(no_improve // 50, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def compute_subgradient(a, conv, top_m, temperature):
    """Compute smoothed subgradient of max conv at top_m peaks."""
    n = len(a)
    m = min(top_m, len(conv))
    peaks = np.argpartition(conv, -m)[-m:]
    vals = conv[peaks]
    scaled = vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    sg = np.zeros(n)
    for idx in range(m):
        if w[idx] < 1e-11:
            continue
        k = int(peaks[idx])
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo > i_hi:
            continue
        sg[i_lo:i_hi + 1] += 2.0 * w[idx] * a[k - i_hi:k - i_lo - 1: -1]

    return sg


def sg_optimize(f0, time_limit, top_m=40, temp0=0.00004, lr0=0.003, mom=0.92):
    """Subgradient optimization with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr0
    temp = temp0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgradient(f, conv, top_m, temp)
        v = mom * v - lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        lr *= 0.9997
        temp *= 1.003

        if no_improve > 250:
            rng = np.random.default_rng(it * 31 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = max(lr0 * 0.5, 0.0004)
            temp = temp0
            no_improve = 0

    return best_f, best_obj


def coord_desc(f0, time_limit):
    """Coordinate descent with multi-scale local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0

    multipliers = [0.6, 0.8, 1.1, 1.3, 1.5, 0.7, 0.9, 1.2]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old = f[j]
            if old < 1e-12:
                continue
            best_val = old
            best_obj_j = best_obj
            denom = 1.0 - old
            if denom < 1e-10:
                continue
            for mult in multipliers:
                new_val = old * mult
                if new_val < 0 or new_val > 1.0 - 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial /= f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj_j - 1e-15:
                    best_obj_j = val
                    best_val = new_val
            if best_val != old:
                scale = (1.0 - best_val) / denom
                f[j] = best_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f /= f.sum()
                if best_obj_j < best_obj - 1e-14:
                    best_obj = best_obj_j
                    best_f = f.copy()
                    improved = True
        if not improved:
            no_improve += 1
            if no_improve > 2:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-Frank-Wolfe with vectorized LP, multi-length
    exploration, subgradient polishing with momentum, and coordinate descent.
    Exploits the insight that optimal sequences are non-palindromic.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform init")

    n_parent = len(parent)

    # === Phase 1: LP-FW on parent ===
    print(f"\n=== Phase 1: LP-FW on parent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(250, rem * 0.25)
    if t1 > 30:
        a_opt, _ = lpfw(parent, time_limit=t1, top_k=40, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-length exploration (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    base_n = len(base)

    length_offsets = [0, -5, 5, -10, 10, -3, 3, 15, -15, 20]
    tried_lengths = set()
    for dn in length_offsets:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        target_n = base_n + dn
        if target_n < 128 or target_n > 2048 or target_n in tried_lengths:
            continue
        tried_lengths.add(target_n)
        f_init = interpolate_seq(base, target_n)
        t_alloc = min(rem * 0.1, 80)
        if t_alloc < 15:
            continue
        print(f"  LP-FW n={target_n} (offset={dn:+d}), t={t_alloc:.0f}s")
        a_opt, _ = lpfw(f_init, time_limit=t_alloc, top_k=30)
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 3: Extended LP-FW on best ===
    print(f"\n=== Phase 3: Extended LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 50:
            t2 = min(rem * 0.22, 300)
            print(f"  t={t2:.0f}s, top_k=50")
            a_opt, _ = lpfw(best_seq, time_limit=t2, top_k=50, verbose=True)
            try_seq(a_opt, "lpfw_extended")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: SG Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_sg = min(rem * 0.18, 180)
            n_b = len(best_seq)
            print(f"  t={t_sg:.0f}s, n={n_b}")
            a_opt, _ = sg_optimize(best_seq, time_limit=t_sg,
                                   top_m=max(30, n_b // 8),
                                   temp0=0.00003, lr0=0.0025, mom=0.93)
            try_seq(a_opt, "sg")

    # === Phase 5: LP-FW with different top-k ===
    print(f"\n=== Phase 5: LP-FW variant params (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        for tk in [20, 60]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t3 = min(rem * 0.15, 150)
            a_opt, _ = lpfw(best_seq, time_limit=t3, top_k=tk)
            try_seq(a_opt, f"lpfw_k{tk}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.2, 150)
            print(f"  t={t_cd:.0f}s")
            a_opt, _ = coord_desc(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

            rem = deadline - time.monotonic()
            if rem > 20:
                t_cd2 = min(rem * 0.15, 80)
                a_opt2, _ = coord_desc(a_opt, time_limit=t_cd2)
                try_seq(a_opt2, "cd2")

    # === Phase 7: Diverse restarts + final polish ===
    print(f"\n=== Phase 7: Diverse restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        restart_configs = []

        # Noise-based restarts
        for scale in [0.002, 0.004, 0.006]:
            restart_configs.append((f"noise_{scale}",
                                    project_simplex(best_seq + rng.standard_normal(n_b) * scale)))

        # Smooth perturbation
        kernel = np.exp(-0.5 * (np.arange(n_b) - n_b / 2) ** 2 / (n_b * 0.3) ** 2)
        restart_configs.append(("smooth_pert",
                                project_simplex(best_seq + 0.003 * kernel * rng.standard_normal(n_b))))

        # Frequency-domain perturbation
        A = np.fft.rfft(best_seq)
        mag = np.abs(A)
        phase = np.angle(A) + rng.standard_normal(len(A)) * 0.05
        A_pert = mag * np.exp(1j * phase)
        sig_pert = np.fft.irfft(A_pert, n_b)[:n_b]
        sig_pert = np.maximum(sig_pert, 0)
        ss = sig_pert.sum()
        if ss > 1e-10:
            sig_pert /= ss
            restart_configs.append(("freq_pert", sig_pert))

        for name, f_init in restart_configs:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_r = min(rem * 0.12, 100)
            a_opt, _ = lpfw(f_init, time_limit=t_r, top_k=30)
            try_seq(a_opt, f"restart_{name}")

    # === Phase 8: Final fine-tuning ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_fine = min(rem * 0.3, 100)
            a_opt, _ = sg_optimize(best_seq, time_limit=t_fine,
                                   top_m=25, temp0=0.00001, lr0=0.001, mom=0.94)
            try_seq(a_opt, "final_sg")

        rem = deadline - time.monotonic()
        if rem > 15:
            t_final = min(rem * 0.3, 60)
            a_opt, _ = lpfw(best_seq, time_limit=t_final, top_k=80)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
