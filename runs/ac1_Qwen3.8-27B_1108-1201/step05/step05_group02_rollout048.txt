def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_structured_start(n, length_idx, rng):
    """Generate diverse structured starting sequences."""
    x = np.linspace(0, 1, n)
    x_centered = x - 0.5

    if length_idx == 0:
        # Gaussian
        sigma = 0.15
        a = np.exp(-0.5 * (x_centered / sigma) ** 2)
    elif length_idx == 1:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x_centered))
    elif length_idx == 2:
        # Triangular
        a = np.maximum(0, 1 - 2 * np.abs(x_centered))
    elif length_idx == 3:
        # Parabolic (wide)
        a = np.maximum(0, 1 - 4 * x_centered ** 2)
    elif length_idx == 4:
        # Flat-top with smooth edges
        edge = 0.1
        a = np.ones(n)
        mask_left = x < edge
        mask_right = x > 1 - edge
        a[mask_left] = 0.5 * (1 + np.cos(np.pi * x[mask_left] / edge))
        a[mask_right] = 0.5 * (1 + np.cos(np.pi * (1 - x[mask_right]) / edge))
    elif length_idx == 5:
        # Two-bump
        a = np.exp(-0.5 * ((x_centered + 0.2) / 0.12) ** 2) + np.exp(-0.5 * ((x_centered - 0.2) / 0.12) ** 2)
    elif length_idx == 6:
        # Wide Gaussian
        sigma = 0.28
        a = np.exp(-0.5 * (x_centered / sigma) ** 2)
    elif length_idx == 7:
        # Sech-like
        a = 1.0 / np.cosh(3 * x_centered / 0.3)
    elif length_idx == 8:
        # Power law decay from center
        a = (1 - 2 * np.abs(x_centered)) ** 0.5
        a = np.maximum(a, 0)
    else:
        # Mix of bumps
        a = np.exp(-0.5 * (x_centered / 0.08) ** 2) + 0.5 * np.exp(-0.5 * ((x_centered - 0.15) / 0.1) ** 2) + 0.5 * np.exp(-0.5 * ((x_centered + 0.15) / 0.1) ** 2)

    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(n) / n
    else:
        a /= ss
    return a


def generate_random_smooth(n, rng, smoothness=0.1):
    """Generate a random smooth non-negative sequence."""
    raw = rng.standard_normal(n)
    # Smooth via convolution with Gaussian kernel
    ksize = max(3, int(2 * smoothness * n) | 1)
    kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (smoothness * n / 2) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(raw, kernel, mode='same')
    a = np.maximum(smoothed, 0)
    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(n) / n
    else:
        a /= ss
    return a


def subgradient_optimize(f0, time_limit, lr=0.003, momentum=0.93, top_k=50,
                         temp_init=0.0001, temp_final=0.000001, verbose=False):
    """Subgradient descent with aggressive geometric temperature annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    max_iters_estimate = max(1000, int(time_limit * 50))
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        # Adaptive temperature: geometric schedule
        progress = min(1.0, it / max_iters_estimate)
        cur_temp = temp_init * (temp_final / temp_init) ** progress
        cur_lr = lr * (1.0 + 0.5 * (1 - progress))

        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"      SG iter {it}: obj={best_obj:.10f}, T={cur_temp:.2e}")
        else:
            no_improve += 1
            f = f_new

        if no_improve > 300:
            # Reset with perturbation
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.001 + 0.002 * min(no_improve // 100, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj

    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Find top peaks
        top_k_peaks = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k_peaks)[-top_k_peaks:]
        peak_positions = np.sort(peak_positions)[::-1]

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            current_peak_val = conv[peak_k]
            if current_peak_val < M * 0.95:
                continue  # Skip if this peak is no longer near the max

            # Find contributing coordinates
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            kg = peak_k - idx
            mask = (kg >= 0) & (kg < n)
            idx = idx[mask]
            kg = kg[mask]

            # Contribution of each coordinate to this peak
            contrib = f[idx] * f[kg]
            top_contrib_k = min(8, len(idx))
            if np.sum(contrib) < 1e-15:
                continue
            contrib_order = np.argsort(contrib)[::-1][:top_contrib_k]

            for ci in contrib_order:
                j = int(idx[ci])
                if f[j] < 1e-12:
                    continue

                # Try reducing this coordinate

                best_obj_val = best_obj

                for red_factor in [0.3, 0.5, 0.7, 0.85, 0.95]:
                    if red_factor >= best_red:
                        continue
                    new_j_val = f[j] * red_factor
                    denom = 1.0 - f[j]
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_j_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_j_val
                    mask_s = np.ones(n, dtype=bool)
                    mask_s[j] = False
                    f_trial[mask_s] *= scale

                    if ss < 1e-15:
                        continue
                    f_trial /= ss
                    val = fast_convolve(f_trial).max()
                    if val < best_obj_val - 1e-14:
                        best_obj_val = val
                        best_red = red_factor

                if best_red < 1.0:
                    new_j_val = f[j] * best_red
                    denom = 1.0 - f[j]
                    if denom > 1e-10:
                        scale = (1.0 - new_j_val) / denom
                        f[j] = new_j_val
                        mask_s = np.ones(n, dtype=bool)
                        mask_s[j] = False
                        f[mask_s] *= scale
                        ss = f.sum()
                        if ss > 1e-15:
                            f /= ss
                    if best_obj_val < best_obj - 1e-14:
                        best_obj = best_obj_val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        M = conv.max()
                        improved = True

        if not improved:

                # Try increasing low-contributing coordinates

                        denom = 1.0 - f[j]

                        scale = (1.0 - new_j_val) / denom

                        f_trial = f.copy()
                        f_trial[j] = new_j_val
                        mask_s = np.ones(n, dtype=bool)

                        if ss < 1e-15:

def lp_fw_optimize(f0, time_limit, max_iters=200, verbose=False):
    """LP-based Frank-Wolfe with min-peak formulation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1

        # Solve LP: min t s.t. (f*g)[k] <= t, g >= 0, sum(g) = 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        n_constraints = 2 * n - 1
        A_ub = np.zeros((n_constraints, n + 1))
        for j in range(n):
            A_ub[j:j + n, j] = f
        A_ub[:, -1] = -1.0
        b_ub = np.zeros(n_constraints)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if not res.success:
            # Fallback: uniform
            g = np.ones(n) / n
        else:
            g = res.x[:n]
            g = np.maximum(g, 0)
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 15:
            break

    return best_f, best_obj


def quick_local_optimize(f0, time_budget):

def cross_pollinate(f1, f2, n_target):
    """Create a new sequence by taking top contributions from both candidates."""
    if len(f1) != n_target:
        f1 = interpolate_sequence(f1, n_target)
    if len(f2) != n_target:
        f2 = interpolate_sequence(f2, n_target)

    # Blend with different weights
    a = 0.5 * f1 + 0.5 * f2
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n_target) / n_target
    return a / ss

        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_seq = a_norm.copy()
                print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
            candidates.append((score, a_norm.copy()))
        return score

    # === Get parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
            candidates.append((p_score, parent.copy()))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Diverse structured starts ===
    print(f"\n=== Phase 1: Diverse starts (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_starts = min(14, max(6, int(rem / 80)))
    per_start_time = min(70, rem / n_starts - 5)

    print(f"  Trying {n_starts} diverse starts, ~{per_start_time:.0f}s each")

    for si in range(n_starts):
        rem = deadline - time.monotonic()
        if rem < 30:
            break

        # Choose starting sequence
        if si == 0:
            # Parent itself
            start = parent.copy()
            label = f"parent"
        elif si == 1:
            # Symmetrized parent
            start = 0.5 * (parent + parent[::-1])
            start = project_simplex(start)
            label = f"sym_parent"
        elif si <= 7:
            # Structured windows
            start = generate_structured_start(base_n, si - 2, rng)
            label = f"struct_{si-2}"
        elif si <= 10:
            # Random smooth
            smoothness = 0.05 + 0.05 * (si - 8)
            start = generate_random_smooth(base_n, rng, smoothness)
            label = f"rand_smooth_{si-8}"
        else:
            # Interpolated parent to different length
            target_n = base_n + rng.integers(-200, 200)
            target_n = max(512, min(2048, target_n))
            start = interpolate_sequence(parent, target_n)
            label = f"interp_n{target_n}"

        # Quick local optimization
        t_alloc = min(per_start_time, rem - 30)
        if t_alloc < 10:
            break

        print(f"  Start {si+1}/{n_starts} [{label}], t={t_alloc:.0f}s")
        opt_f = quick_local_optimize(start, t_alloc)
        try_seq(opt_f, label)

    # === Phase 2: Best refinement ===
    print(f"\n=== Phase 2: Best refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        n_best = len(best_seq)

        # Extended subgradient
        t_sg = min(rem * 0.3, 300)
        if t_sg > 30:
            print(f"  Extended SG: n={n_best}, t={t_sg:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg,
                                            lr=0.002, momentum=0.94,
                                            top_k=max(40, n_best // 6),
                                            temp_init=0.00008,
                                            temp_final=0.0000005,
                                            verbose=True)
            try_seq(a_opt, "sg_extended")

        # Peak shaving CD
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.25, 250)
        if t_cd > 20 and best_seq is not None:
            print(f"  Peak CD: n={len(best_seq)}, t={t_cd:.0f}s")
            a_opt, _ = peak_shaving_cd(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "cd_main")

        # LP-FW polish
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.2, 180)
        if t_lp > 15 and best_seq is not None:
            print(f"  LP-FW: n={len(best_seq)}, t={t_lp:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp,
                                       max_iters=150, verbose=True)
            try_seq(a_opt, "lpfw_refine")

    # === Phase 3: Length exploration ===
    print(f"\n=== Phase 3: Length exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        n_best = len(best_seq)
        test_lengths = []
        for delta in [-128, -64, 64, 128, -256, 256]:
            ln = n_best + delta
            if 384 <= ln <= 3000:
                test_lengths.append(ln)

        for target_n in test_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.25, 120)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(f_interp, time_limit=t_alloc * 0.5,
                                            lr=0.002, momentum=0.94,
                                            top_k=max(30, target_n // 8),
                                            temp_init=0.00006, temp_final=0.000001)
            a_opt, _ = peak_shaving_cd(a_opt, time_limit=t_alloc * 0.5)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 4: Cross-pollination and final polish ===
    print(f"\n=== Phase 4: Cross-pollination + Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

    if len(candidates) >= 2 and rem > 40:
        # Sort candidates by score
        candidates.sort(key=lambda x: x[0])
        # Try blending top candidates
        n_target = len(best_seq)
        pairs_tried = 0
        for i in range(min(3, len(candidates))):
            for j in range(i + 1, min(5, len(candidates))):
                if pairs_tried >= 3:
                    break
                if time.monotonic() > deadline - 30:
                    break
                pairs_tried += 1
                blended = cross_pollinate(candidates[i][1], candidates[j][1], n_target)
                t_polish = min(30, rem - 30)
                if t_polish > 5:
                    a_opt = quick_local_optimize(blended, t_polish)
                    try_seq(a_opt, f"blend_{i}_{j}")

    # Final polish on best
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        n_best = len(best_seq)

        if rem > 20:
            print(f"  Final SG polish, t={min(rem, 60):.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=min(rem, 60),
                                            lr=0.0008, momentum=0.95,
                                            top_k=max(20, n_best // 10),
                                            temp_init=0.00002, temp_final=0.0000002)
            try_seq(a_opt, "final_sg")

        rem = deadline - time.monotonic() - 15
        if rem > 15:
            print(f"  Final CD polish, t={min(rem, 40):.0f}s")
            a_opt, _ = peak_shaving_cd(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_cd")

        # Small perturbations
        rem = deadline - time.monotonic() - 10
        for p in range(4):
            if rem < 10:
                break
            noise_scale = 0.0005 + 0.0003 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(15, rem - 10)
            if t_pert > 5:
                a_opt, _ = subgradient_optimize(f_pert, time_limit=t_pert,
                                                lr=0.0005, momentum=0.95,
                                                top_k=max(15, n_best // 12),
                                                temp_init=0.00001, temp_final=0.0000001)
                try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
