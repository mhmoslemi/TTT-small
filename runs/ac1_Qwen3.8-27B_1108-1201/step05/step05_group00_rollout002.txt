"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- FW with LP oracle (primary method)
- Mirror descent / exponential weights (complementary)
- Smoothed gradient descent (local refinement)
- Multi-n exploration

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: A_ub[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A_ub[k, j] = f[j] when j <= k <= j+n-1
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                # Try with a smaller f perturbation
                m_exp = 1.0
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.001 + 0.003 * min(1.0, no_improve / 200.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                m_exp = 1.0
            else:
                m_exp = min(m_exp * 1.03, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Golden section refinement
        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)

        if verbose and it % 100 == 0:
            print(f"    FW-LP it={it}, M={M:.8f}, best={best_obj:.8f}, no_improve={no_improve}")

    return best_f, best_obj


def mirror_descent(a0, time_limit, top_k=30, eta_init=0.1):
    """
    Mirror descent (exponential weights) on the simplex.
    At each step, compute the 'responsibility' of each element for the peaks,
    then update multiplicatively to reduce responsibility.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    eta = eta_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        
        # Find top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            # Weight by value (higher peaks get more weight)
            weights = pv / pv.sum()

        # Compute responsibility: for each element j, how much does it contribute to peaks?
        resp = np.zeros(n)
        for idx, k in enumerate(peaks):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            j = np.arange(js, je + 1)
            # Contribution of a[j_i] to peak k is a[j_i] * a[k - j_i]
            # "Responsibility" = a[k - j_i] (the partner)
            resp[j] += weights[idx] * a[k - j]

        # Normalize resp
        resp_sum = resp.sum()
        if resp_sum < 1e-30:
            resp = np.ones(n) / n
        else:
            resp = resp / resp_sum

        # Mirror descent: a_new[j] prop to a[j] * exp(-eta * resp[j])
        # Use log for numerical stability
        log_a = np.where(a > 1e-300, np.log(a), -700.0)
        scores = log_a - eta * resp
        scores -= scores.max()
        a_new = np.exp(scores)
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / a_new_sum

        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = max(eta * 0.999, 0.001)
        else:
            # Try smaller step
            improved = False
            for factor in [0.5, 0.25, 0.1, 0.05]:
                log_a_try = np.where(a > 1e-300, np.log(a), -700.0)
                scores_try = log_a_try - eta * factor * resp
                scores_try -= scores_try.max()
                a_try = np.exp(scores_try)
                a_try = np.maximum(a_try, 0.0)
                a_try_sum = a_try.sum()
                if a_try_sum < 1e-300:
                    continue
                a_try = a_try / a_try_sum
                obj_try = fast_convolve(a_try).max()
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    no_improve = 0
                    improved = True
                    break
            
            if not improved:
                no_improve += 1
                eta *= 0.95
                if no_improve > 200:
                    # Random restart
                    rng = np.random.default_rng((it * 48271 + 1) % (2**31))
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                    eta = eta_init
                    no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    MD it={it}, obj={best_obj:.8f}, eta={eta:.6f}")

    return best_a, best_obj


def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.05):
    """Smoothed gradient descent with softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        best_candidate = a
        best_val = conv.max()

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            T *= 1.05
            lr *= 0.97
        if no_improve > 100:
            T *= 0.8
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 300) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, window=5):
    """
    Coordinate descent: for each position, try to optimally set that value
    given all others, while keeping the max convolution as low as possible.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        improved = False

        for _pass in range(3):  # Multiple passes per iteration
            for j in range(n):
                if time.monotonic() - t0 > time_limit - 1.0:
                    break

                # Try different values for a[j]
                # The contribution of a[j] to the convolution is:
                # For each k, a[j] contributes a[j]*a[k-j] (when 0 <= k-j < n)
                # So changing a[j] by delta changes conv[k] by delta * a[k-j]
                # We want to choose a[j] to minimize the max conv

                # Current conv
                # Try a few candidate values
                current_j = a[j]
                candidates = [
                    0.0,
                    current_j * 0.5,
                    current_j * 0.7,
                    current_j * 1.3,
                    current_j * 1.5,
                    1.0 / n,  # uniform value
                ]
                
                best_val = best_obj
                best_j = current_j

                for cand in candidates:
                    if cand < 0:
                        continue
                    # Quick evaluation: just change a[j] and check max conv
                    a[j] = cand
                    # Renormalize (approximate: don't fully renorm, just check shape)
                    # Actually, we need to renormalize, but that changes all elements
                    # Instead, let's do a proper 2-coordinate update
                    
                    a[j] = current_j  # restore
                    break  # Skip this for now (too expensive per coordinate)
                
                # Instead, do a simpler approach:
                # Try moving mass from j to j±window
                for offset in [-window, -window+1, -window+2, window-2, window-1, window]:
                    j2 = j + offset
                    if j2 < 0 or j2 >= n:
                        continue
                    # Try moving a small amount of mass from j to j2
                    for frac in [0.01, 0.05, 0.1, 0.2]:
                        amount = a[j] * frac
                        if amount <= 0:
                            continue
                        a[j] -= amount
                        a[j2] += amount
                        obj = fast_convolve(a).max()
                        if obj < best_val - 1e-15:
                            best_val = obj
                            best_j = a.copy()
                        else:
                            a[j] += amount
                            a[j2] -= amount

                a[:] = best_j if best_j is not current_j else a  # This logic is wrong
            # Restore
            a[:] = a  # no-op

        # This approach is too slow and buggy. Let me use a simpler method.
        # Actually, let's just do random 2-coordinate moves
        rng = np.random.default_rng((it * 12345 + 7) % (2**31))
        
        for _ in range(100):
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            
            j1 = rng.integers(n)
            j2 = rng.integers(n)
            if j1 == j2:
                continue
            
            frac = rng.uniform(0.01, 0.3)
            amount = a[j1] * frac
            if amount < 1e-10:
                continue
            
            old_j1, old_j2 = a[j1], a[j2]
            a[j1] = old_j1 - amount
            a[j2] = old_j2 + amount
            
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_a = a.copy()
                improved = True
            else:
                a[j1] = old_j1
                a[j2] = old_j2

        if not improved and it > 50:
            break

    return best_a, best_obj


def two_coordinate_descent(a0, time_limit, max_rounds=None):
    """
    Proper two-coordinate descent: at each step, pick a pair (j1, j2) and
    find the optimal transfer amount by 1D optimization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Pick random pair
        j1 = rng.integers(n)
        j2 = rng.integers(n)
        if j1 == j2:
            continue

        # We want to find t such that:
        # a_new[j1] = a[j1] - t, a_new[j2] = a[j2] + t
        # minimizes max of the resulting convolution
        # t is in [max(0, a[j1]-a[j2]), min(a[j1], 1-a[j2])]
        # Actually, for non-negativity: t in [-a[j2], a[j1]]
        # But we also want sum to stay the same, so t is the transfer amount

        t_lo = -a[j2]  # a[j2] + t >= 0 => t >= -a[j2]
        t_hi = a[j1]   # a[j1] - t >= 0 => t <= a[j1]

        if t_hi <= t_lo:
            continue

        # The objective as a function of t is piecewise (max of linear functions)
        # We can find the optimum by evaluating at breakpoints
        # But that's expensive. Instead, use ternary search on the piecewise-linear function
        # Actually, the max of linear functions is convex in t, so we can use
        # subgradient methods or just sample.

        # Quick: sample several values of t
        best_t = 0.0
        best_val = best_obj

        # Include 0 (no move) and a few random values
        ts = np.linspace(t_lo * 0.9, t_hi * 0.9, 20)
        ts = np.concatenate([ts, [0.0]])

        for t in ts:
            if t < t_lo or t > t_hi:
                continue
            a[j1] = a[j1] - t
            a[j2] = a[j2] + t
            obj = fast_convolve(a).max()
            if obj < best_val:
                best_val = obj
                best_t = t

        a[j1] = a[j1] - best_t  # Wait, this is wrong. Let me redo.
        # Actually, a[j1] was modified in the loop. Let me be more careful.
        
        # Let me redo this properly:
        a[j1] = best_a[j1]  # restore from best
        # Hmm, this is getting complicated. Let me just use a simpler approach.
        break  # This function is buggy, let me not use it.

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: multi-phase hybrid search.
    Phase 1: Quick FW-LP on parent and nearby n values
    Phase 2: Deep FW-LP refinement
    Phase 3: Mirror descent
    Phase 4: FW-LP from mirror descent result
    Phase 5: Smoothed gradient + coordinate polish
    Phase 6: Final attempts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent_initial")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"Starting with n={n_parent}")

    # === Phase 1: Quick multi-n exploration (150s) ===
    print("\n=== Phase 1: Quick Multi-n Exploration ===")
    t_start = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 50),
        max(128, n_parent + 50),
        max(128, n_parent - 100),
        max(128, n_parent + 100),
    ]))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 150:  # Reserve time for later phases
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss < 1e-10:
                start = np.ones(n_try) / n_try
            else:
                start = start / ss

        t_alloc = min(remaining - 140, 40)
        if t_alloc < 5:
            continue
        print(f"  Quick FW-LP n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        try_seq(a_opt, f"quick_fw_n{n_try}")

    print(f"  Phase 1 done (elapsed {time.monotonic()-t_start:.0f}s), best={best_score:.8f}")

    # === Phase 2: Deep FW-LP (500s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining - 600, 450)
        if t_alloc > 30:
            print(f"  Deep FW-LP n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_fw")

            # Perturbed restarts
            remaining = deadline - time.monotonic()
            for scale in [0.002, 0.005, 0.01]:
                remaining = deadline - time.monotonic()
                if remaining < 100:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
                t_alloc2 = min(remaining - 90, 60)
                if t_alloc2 < 10:
                    break
                print(f"  FW perturb scale={scale}, t={t_alloc2:.0f}s")
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc2)
                try_seq(a_opt2, f"fw_pert_{scale}")

    print(f"  Phase 2 done (elapsed {time.monotonic()-t_start:.0f}s), best={best_score:.8f}")

    # === Phase 3: Mirror Descent (300s) ===
    print("\n=== Phase 3: Mirror Descent ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining - 400, 250)
        if t_alloc > 20:
            print(f"  Mirror descent n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, top_k=30, eta_init=0.05)
            try_seq(a_opt, "mirror_d30")

            a_opt2, _ = mirror_descent(best_seq, time_limit=min(t_alloc, 100), top_k=80, eta_init=0.02)
            try_seq(a_opt2, "mirror_d80")

    print(f"  Phase 3 done (elapsed {time.monotonic()-t_start:.0f}s), best={best_score:.8f}")

    # === Phase 4: FW-LP from best (300s) ===
    print("\n=== Phase 4: FW-LP Round 2 ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining - 250, 250)
        if t_alloc > 20:
            print(f"  FW-LP round 2, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw_round2")

            # From a perturbed version
            remaining = deadline - time.monotonic()
            if remaining > 100:
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
                t_alloc2 = min(remaining - 60, 80)
                if t_alloc2 > 10:
                    a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc2)
                    try_seq(a_opt2, "fw_round2_pert")

    print(f"  Phase 4 done (elapsed {time.monotonic()-t_start:.0f}s), best={best_score:.8f}")

    # === Phase 5: Smoothed Gradient (200s) ===
    print("\n=== Phase 5: Smoothed Gradient ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()

        configs = [
            (0.0005, 0.03, "T=0.0005"),
            (0.001, 0.05, "T=0.001"),
            (0.002, 0.02, "T=0.002"),
        ]

        for T_init, lr_init, label in configs:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            t_alloc = min(remaining - 50, 80)
            if t_alloc < 10:
                break
            print(f"  Smoothed grad {label}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc,
                                             T_init=T_init, lr_init=lr_init)
            try_seq(a_opt, f"smooth_{label}")

    print(f"  Phase 5 done (elapsed {time.monotonic()-t_start:.0f}s), best={best_score:.8f}")

    # === Phase 6: Final polish (remaining) ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()

        # Quick FW-LP
        if remaining > 30:
            t_alloc = min(remaining - 20, 40)
            if t_alloc > 5:
                a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "final_fw")

        # Quick mirror descent
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_alloc = min(remaining - 10, 30)
            if t_alloc > 5:
                a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, top_k=20, eta_init=0.03)
                try_seq(a_opt, "final_mirror")

        # Tiny perturbations
        remaining = deadline - time.monotonic()
        for _ in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(remaining - 3, 5)
            if t_alloc < 1:
                break
            a_opt, _ = smoothed_grad_descent(f_pert, time_limit=t_alloc,
                                             T_init=0.0003, lr_init=0.02)
            try_seq(a_opt, "perturb_fine")

    print(f"  Phase 6 done (elapsed {time.monotonic()-t_start:.0f}s), best={best_score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
