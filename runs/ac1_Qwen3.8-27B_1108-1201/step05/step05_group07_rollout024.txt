import time
from scipy.optimize import linprog

def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    Solve the min-max LP to find the steepest descent direction.
    
    minimize: t
    subject to: 2 * sum_j A[k,j] * g[j] <= t for k in active set
                sum_j g[j] = 0
    
    Variables: g[0], ..., g[n-1], t
    """
    n = len(a)
    k = min(n_constraints, len(active_indices))
    active = active_indices[:k]

    # For each active peak at position p:
    # 2 * sum_j a[p-j] * g[j] - t <= conv[p] - current_max (we want to decrease)
    # Actually, we want: 2 * sum_j a[p-j] * g[j] <= 0 for all active p
    # (We're looking for a direction that decreases all active peaks)

    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.zeros((k, n + 1))

        je = min(n - 1, p)
        for j in range(js, je + 1):

        A_ub[i, n] = -1.0

    A_eq[0, :n] = 1.0

    g = result.x[:n]
    return g

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
with multi-start, adaptive active set, and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, top_k=50):
    """Compute weighted subgradient of max(conv)."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    # Use softmax weighting
    scaled = pv / 0.001
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-10:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg

def lp_direction(a, active_peaks, num_constraints=50):
    """
    Solve min-max LP to find steepest descent direction.
    minimize t
    subject to: 2*sum_j a[p-j]*g[j] - t <= 0 for p in active_peaks
                sum g = 0
    """
    n = len(a)
    k = min(num_constraints, len(active_peaks))
    peaks = active_peaks[:k]
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = np.zeros((k, n + 1))
    b_ub = np.zeros(k)
    
    for i, p in enumerate(peaks):
        p = int(p)
        js = max(0, p - n + 1)
        je = min(n - 1, p)
        for j in range(js, je +
