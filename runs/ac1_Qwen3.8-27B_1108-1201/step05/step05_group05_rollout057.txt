```python
def exact_line_search(conv_aa, conv_ad, conv_dd, t_lo=0.0, t_hi=1.0):
    """
    Find t in [t_lo, t_hi] that minimizes max_k (conv_aa[k] + 2t*conv_ad[k] + t^2*conv_dd[k]).
    """
    m = len(conv_aa)
    
    def obj(t):
        return np.max(conv_aa + 2*t*conv_ad + t**2*conv_dd)
    
    def subgrad(t):
        vals = conv_aa + 2*t*conv_ad + t**2*conv_dd
        active = vals >= vals.max() - 1e-12
        # Subgradient is a convex combination of 2*conv_ad[k] + 2t*conv_dd[k] for active k
        # For the purpose of bisection, we just need the sign
        # If all active have same derivative, that's the subgradient
        # Otherwise, we need the full subdifferential
        derivs = 2*conv_ad[active] + 2*t*conv_dd[active]
        return derivs.min(), derivs.max()  # [min, max] of subdifferential
    
    # Bisection: find where 0 is in the subdifferential
    for _ in range(60):
        t_mid = (t_lo + t_hi) / 2
        d_min, d_max = subgrad(t_mid)
        if d_min > 0:
            t_hi = t_mid  # function is increasing, go left
        elif d_max < 0:
            t_lo = t_mid  # function is decreasing, go right
        else:
            break  # 0 is in subdifferential
        t_lo, t_hi = ... # update
    
    return (t_lo + t_hi) / 2
```

```python
def min_max_parabolas(A, B, C, t_lo=0, t_hi=1, max_iter=50):
    """
    Minimize max_k (A[k] + B[k]*t + C[k]*t^2) over t in [t_lo, t_hi].
    A, B, C are arrays of length m.
    """
    m = len(A)
    active = np.arange(m)  # start with all
    
    for _ in range(max_iter):
        # Find t that minimizes max over active set
        # The minimum of max of quadratics over a finite set occurs where:
        # (a) Two quadratics cross and their max is minimized, or
        # (b) One quadratic's vertex is the max, or
        # (c) A boundary
        
        # Get the active quadratics
        Aa = A[active]
        Ba = B[active]
        Ca = C[active]
        
        # Find the point where the "upper envelope" is minimized
        # Use bisection on the subgradient
        lo, hi = float(t_lo), float(t_hi)
        for _ in range(50):
            mid = (lo + hi) / 2.0
            vals = Aa + Ba * mid + Ca * mid * mid
            max_val = vals.max()
            act = vals >= max_val - 1e-14
            derivs = Ba[act] + 2 * Ca[act] * mid
            if derivs.min() > 1e-12:
                hi = mid
            elif derivs.max() < -1e-12:
                lo = mid
            else:
                break
        t_star = (lo + hi) / 2.0
        
        # Check if any non-active quadratic exceeds at t_star
        all_vals = A + B * t_star + C * t_star * t_star
        new_active = all_vals >= all_vals.max() - 1e-10
        if np.all(new_active[active]) and not np.any(new_active[~active]):
            # No new active constraints
            break
        active = np.where(new_active)[0]
    
    return t_star
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved Frank-Wolfe method
with exact line search, adaptive constraint selection, combinatorial 
initializations, and multi-phase polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_fw_lp(f, n, active_k):
    """Solve: min t s.t. conv(f,g)[k] <= t for k in active_k, g>=0, sum(g)=1."""
    K = len(active_k)
    # Variables: g_0, ..., g_{n-1}, t (total n+1 variables)
    # Objective: minimize t → c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: conv(f,g)[k] - t <= 0 for each active k
    # conv(f,g)[k] = sum_j f[j] * g[k-j] for valid j
    # So the row is: [f[k-n+1], ..., f[k]] (appropriately) - e_last
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    for row, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[row, j] = f[k - j]
        A_ub[row, -1] = -1.0
    b_ub[:] = 0.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]  # return g
    return None

def exact_line_search(conv_aa, conv_ad, conv_dd, t_lo=0.0, t_hi=1.0):
    """Exact line search: minimize max_k (conv_aa[k] + 2t*conv_ad[k] + t^2*conv_dd[k])."""
    # Bisection on subgradient
    for _ in range(50):
        t_mid = (t_lo + t_hi) / 2.0
        vals = conv_aa + 2.0 * t_mid * conv_ad + t_mid * t_mid * conv_dd
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        derivs = 2.0 * conv_ad[active] + 2.0 * t_mid * conv_dd[active]
        if derivs.min() > 1e-10:
            t_hi = t_mid
        elif derivs.max() < -1e-10:
            t_lo = t_mid
        else:
            break
    return (t_lo + t_hi) / 2.0
```
