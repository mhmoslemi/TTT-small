def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def optimize_pair(f, conv, i, j, top_k_indices, top_k_values):
    """Find optimal delta for pair (i,j) that minimizes max peak among top_k.
    
    f_new[i] = f[i] + delta, f_new[j] = f[j] - delta
    Constraint: delta > -f[i] (for f_new[i] >= 0) and delta < f[j] (for f_new[j] >= 0)
    
    The peak values become:
    peak_k = conv[k] + 2*delta*(f[k-i] if 0<=k-i<n else 0 - f[k-j] if 0<=k-j<n else 0) 
             + delta^2*([k=2i] - 2*[k=i+j] + [k=2j])
    """
    n = len(f)
    delta_lo = -f[i] + 1e-15
    delta_hi = f[j] - 1e-15
    
    if delta_lo >= delta_hi:
        return 0.0, top_k_values.max()
    
    # Compute linear coefficients for the top peaks
    # b_k = 2*(f[k-i] - f[k-j]) with out-of-range = 0
    b = np.zeros(len(top_k_indices))
    for idx in range(len(top_k_indices)):
        k = top_k_indices[idx]
        fi = f[k - i] if 0 <= k - i < n else 0.0
        fj = f[k - j] if 0 <= k - j < n else 0.0
        b[idx] = 2.0 * (fi - fj)
    
    # Quadratic coefficients (usually all zero except at specific k)
    c = np.zeros(len(top_k_indices))
    for idx in range(len(top_k_indices)):
        k = top_k_indices[idx]
        if k == 2 * i:
            c[idx] += 1.0
        if k == i + j:
            c[idx] -= 2.0
        if k == 2 * j:
            c[idx] += 1.0
    
    # Minimize max_k (a_k + b_k*delta + c_k*delta^2) over delta in [delta_lo, delta_hi]
    # Using ternary search (the function is convex in delta)
    
    a = top_k_values.copy()
    
    def obj(delta):
        vals = a + b * delta + c * delta * delta
        return vals.max()

    lo = delta_lo
    hi = delta_hi

        if obj(m1) < obj(m2):

        else:
            lo = m1
    
    delta_opt = (lo + hi) / 2.0

def structured_perturb(f, n_peaks=10, scale=0.01):
    """Add a structured perturbation that targets peak reduction."""
    f = f.copy()
    n = len(f)
    conv = fast_convolve(f)
    
    # Find the top peak
    k_max = np.argmax(conv)
    
    # The peak at k_max is sum_j f[j]*f[k_max-j]
    # To reduce it, decrease the largest contributors
    # The contributors are pairs (j, k_max-j)
    # Reduce the larger of each pair
    reduction = np.zeros(n)
    for j in range(max(0, k_max-n+1), min(n, k_max+1)):
        j2 = k_max - j
        if 0 <= j2 < n:
            if j != j2:
                # Reduce the larger one
                if f[j] > f[j2]:
                    reduction[j] += f[j] * f[j2]
                else:
                    reduction[j2] += f[j] * f[j2]
            else:
                reduction[j] += f[j] * f[j2]
    
    # Normalize reduction
    rmax = reduction.max()
    if rmax > 1e-15:
        reduction /= rmax
    
    # Apply perturbation: decrease high contributors, increase low ones
    low_idx = np.argmin(f)
    perturbation = -scale * reduction
    perturbation[low_idx] += scale * reduction.sum() / max(reduction.sum(), 1e-15)
    
    f_new = f + perturbation
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    
    return f_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Pairwise coordinate descent with exact 1D oracle (ternary search)
2. Active-set subgradient method with adaptive step sizes
3. Diversification via structured and random perturbations
4. Multi-n exploration with interpolation
5. Final micro-polish with high-precision settings
Key novelty: exact 1D coordinate optimization finding optimal delta for
each pair update, avoiding gradient approximation errors.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def optimize_pair_delta(f, conv, i, j, top_k_indices, top_k_values):

    Returns (optimal_delta, resulting_max_peak).

    delta_hi = min(f[j] - 1e-14, 1.0)

    if delta_lo >= delta_hi:
        return 0.0, top_k_values.max()

    # Linear coefficients: b_k = 2*(f[k-i] - f[k-j]) with out-of-range = 0
    b = np.zeros(len(top_k_indices))
    for idx in range(len(top_k_indices)):
        k = int(top_k_indices[idx])
        ki = k - i
        kj = k - j
        fi = f[ki] if 0 <= ki < n else 0.0
        fj = f[kj] if 0 <= kj < n else 0.0
        b[idx] = 2.0 * (fi - fj)

    # Quadratic coefficients (nonzero only at k=2i, i+j, 2j)
    c = np.zeros(len(top_k_indices))
    for idx in range(len(top_k_indices)):
        k = int(top_k_indices[idx])
        if k == 2 * i:
            c[idx] += 1.0
        if k == i + j:
            c[idx] -= 2.0
        if k == 2 * j:
            c[idx] += 1.0

    a = top_k_values.copy()

    # Quick check: if all b are zero, optimal delta is 0 (or check quadratic)
    if np.allclose(b, 0.0) and np.allclose(c, 0.0):
        return 0.0, a.max()

    # Ternary search for the convex function max_k(a_k + b_k*delta + c_k*delta^2)
    lo = delta_lo
    hi = delta_hi
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = (a + b * m1 + c * m1 * m1).max()
        v2 = (a + b * m2 + c * m2 * m2).max()
        if v1 < v2:
            hi = m2
        else:
            lo = m1

    delta_opt = (lo + hi) / 2.0
    peak_after = (a + b * delta_opt + c * delta_opt * delta_opt).max()
    return delta_opt, peak_after


def coordinate_descent(f0, time_limit, n_top=30, n_j_candidates=5, verbose=False):
    """Pairwise coordinate descent with exact 1D oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    sweep = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        conv = fast_convolve(f)
        current_max = conv.max()

        # Get top peaks
        k_top = min(n_top, len(conv))
        top_k_indices = np.argpartition(conv, -k_top)[-k_top:]
        sort_order = np.argsort(conv[top_k_indices])[::-1]
        top_k_indices = top_k_indices[sort_order]
        top_k_values = conv[top_k_indices]

        # Randomize processing order
        order = rng.permutation(n)

        improvements_in_sweep = 0
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Generate j candidates: complement + random
            j_candidates = [(i + n // 2) % n, (i + n // 3) % n, (i + 2 * n // 3) % n]
            if n_j_candidates > 3:
                extra = rng.integers(0, n, size=min(n_j_candidates - 3, 10))
                j_candidates.extend(extra.tolist())

            best_delta = 0.0
            best_peak_after = current_max

            for j in j_candidates:
                if j == i:
                    continue
                delta, peak_after = optimize_pair_delta(
                    f, conv, i, j, top_k_indices, top_k_values)
                if peak_after < best_peak_after - 1e-14:
                    best_peak_after = peak_after
                    best_delta = delta

            if abs(best_delta) > 1e-12:
                # Find which j gave the best result (redo to get j)
                # Actually we already found best_delta, need to re-find j
                # For efficiency, just apply with the best j found
                # (We track this implicitly - the best_delta corresponds to some j)
                # Let's redo: find best (i,j,delta) together
                best_j = -1
                for j in j_candidates:
                    if j == i:
                        continue
                    delta, peak_after = optimize_pair_delta(
                        f, conv, i, j, top_k_indices, top_k_values)
                    if peak_after < best_peak_after + 1e-15:
                        best_peak_after = peak_after
                        best_delta = delta
                        best_j = j

                if best_j >= 0 and abs(best_delta) > 1e-12:
                    f[i] += best_delta
                    f[best_j] -= best_delta
                    f[i] = max(0.0, f[i])
                    f[best_j] = max(0.0, f[best_j])
                    improvements_in_sweep += 1

        # End of sweep: check improvement
        conv = fast_convolve(f)
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
            if verbose and sweep % 10 == 0:
                print(f"    CD sweep {sweep}: obj={best_obj:.12f}, improvements={improvements_in_sweep}")

        # Early stopping if no improvements
        if improvements_in_sweep == 0 and sweep > 3:
            # Try a few more sweeps then break
            if sweep > 20:
                break

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, n_top=20, lr_init=0.001, momentum=0.92):
    """Subgradient method targeting top peaks with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Select top peaks with softmax weighting
        k_top = min(n_top, len(conv))
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        peak_vals = conv[peak_idx]

        beta_w = 3.0 + 2.0 * min(1.0, it / 200.0)
        w = np.exp(beta_w * (peak_vals - peak_vals.max()))
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peak_idx))
            w[0] = 1.0
        else:
            w /= w_sum

        # Compute subgradient: grad[m] = 2 * sum_k w_k * f[k-m]
        grad = np.zeros(n)
        for idx in range(len(peak_idx)):
            k = int(peak_idx[idx])
            wk = w[idx]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                ms = np.arange(lo, hi + 1)
                grad[ms] += 2.0 * wk * f[k - ms]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 200
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 0.005)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.95
            if no_improve > 120:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_init * 0.4, 0.0001)
                momentum = 0.90
                no_improve = 0

    return best_f, best_obj


def lse_optimize(f0, time_limit, beta_start=50.0, beta_end=3000.0, max_peaks=100):
    """Log-sum-exp smoothed optimization with top-peak restriction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 1.5 / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # LSE weights on top peaks only
        k_sel = min(max_peaks, conv_len)
        top_idx = np.argpartition(conv, -k_sel)[-k_sel:]
        top_conv = conv[top_idx].copy()

        bc = beta * top_conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w_top = exp_bc / exp_bc.sum()

        # Build gradient using only top peaks
        w_full = np.zeros(conv_len)
        w_full[top_idx] = w_top

        grad = np.zeros(n)
        for idx in range(len(top_idx)):
            k = int(top_idx[idx])
            wk = w_top[idx]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                ms = np.arange(lo, hi + 1)
                grad[ms] += 2.0 * wk * f[k - ms]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 200
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 12:
                lr *= 0.94
            if no_improve > 80:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(1.5 / n, 0.001)
                beta = min(beta * 1.8, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.005

    return best_f, best_obj


def try_sequence(seq, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(np.asarray(seq, dtype=np.float64), 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def diversify_and_optimize(base_f, n_variants, time_per_variant, rng_seed):
    """Create perturbed variants and optimize each."""
    results = []
    n = len(base_f)
    rng = np.random.default_rng(rng_seed)

    for v in range(n_variants):
        # Different perturbation strategies
        strategy = v % 4
        if strategy == 0:
            # Gaussian noise
            f_pert = project_simplex(base_f + rng.standard_normal(n) * 0.008)
        elif strategy == 1:
            # Peak-targeted: reduce top contributor, spread to bottom
            conv = fast_convolve(base_f)
            k_peak = np.argmax(conv)
            f_pert = base_f.copy()
            # Find the largest contributor to the peak
            contributions = np.zeros(n)
            for j in range(max(0, k_peak - n + 1), min(n, k_peak + 1)):
                j2 = k_peak - j
                if 0 <= j2 < n:
                    contributions[j] = f_pert[j] * f_pert[j2]
            max_contrib_idx = np.argmax(contributions)
            min_idx = np.argmin(base_f)
            f_pert[max_contrib_idx] *= 0.95
            f_pert[min_idx] += 0.05 * base_f[max_contrib_idx]
            f_pert = project_simplex(f_pert)
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = rng.integers(2, 8)
            x = np.arange(n)
            pert = np.cos(2 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
            f_pert = project_simplex(base_f + 0.005 * pert)
        else:
            # Sparse perturbation: change a few coordinates
            n_changes = rng.integers(5, 20)
            idx = rng.choice(n, size=n_changes, replace=False)
            f_pert = base_f.copy()
            for idx_i in idx:
                f_pert[idx_i] *= (1.0 + rng.uniform(-0.1, 0.1))
            f_pert = project_simplex(f_pert)

        results.append(f_pert)

    return results


def propose_candidate(seed=42, budget_s=1800):
    """
    Main multi-phase optimization:
    Phase 1: Coordinate descent with exact 1D oracle
    Phase 2: Subgradient + LSE polishing
    Phase 3: Diversification (perturb + re-optimize)
    Phase 4: Multi-n exploration
    Phase 5: Final intensive polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Coordinate Descent ===
    print(f"\n=== Phase 1: Coordinate Descent (n={n_parent}) ===")
    t1_end = deadline - 900
    t_alloc = min(350, t1_end - time.monotonic())
    if t_alloc > 30:
        print(f"  CD pass 1: t={t_alloc:.0f}s, n_top=30")
        f_cd, obj_cd = coordinate_descent(parent, time_limit=t_alloc,
                                          n_top=30, n_j_candidates=4)
        try_sequence(f_cd, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(250, t1_end - time.monotonic())
    if t_alloc > 20:
        print(f"  CD pass 2: t={t_alloc:.0f}s, n_top=50")
        f_cd2, _ = coordinate_descent(current, time_limit=t_alloc,
                                      n_top=50, n_j_candidates=5)
        try_sequence(f_cd2, best_tracker)

    # === Phase 2: Subgradient + LSE ===
    print(f"\n=== Phase 2: Subgradient + LSE ===")
    t2_end = deadline - 550
    current = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = min(150, t2_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Subgradient: t={t_alloc:.0f}s, n_top=20")
        f_sg, _ = subgradient_optimize(current, time_limit=t_alloc,
                                       n_top=20, lr_init=0.0008, momentum=0.92)
        try_sequence(f_sg, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(130, t2_end - time.monotonic())
    if t_alloc > 20:
        print(f"  LSE: t={t_alloc:.0f}s, beta_end=2000")
        f_lse, _ = lse_optimize(current, time_limit=t_alloc,
                                beta_start=30.0, beta_end=2000.0, max_peaks=80)
        try_sequence(f_lse, best_tracker)

    # === Phase 3: Diversification ===
    print(f"\n=== Phase 3: Diversification ===")
    t3_end = deadline - 300
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    n_variants = 6
    t_remaining = t3_end - time.monotonic()
    t_per_variant = min(80, max(20, t_remaining / (n_variants + 1)))

    if t_per_variant > 15:
        variants = diversify_and_optimize(current, n_variants, t_per_variant, seed + 100)
        for vi, f_var in enumerate(variants):
            t_rem = t3_end - time.monotonic()
            if t_rem < 30:
                break
            t_alloc = min(t_per_variant, t_rem * 0.45)
            if t_alloc < 15:
                break
            print(f"  Variant {vi+1}/{n_variants}: t={t_alloc:.0f}s")
            # Run CD on the variant
            f_opt, _ = coordinate_descent(f_var, time_limit=t_alloc,
                                          n_top=30, n_j_candidates=4)
            try_sequence(f_opt, best_tracker)

            # Quick subgradient follow-up
            t_sg = min(25, t3_end - time.monotonic() - 15)
            if t_sg > 10:
                f_opt2, _ = subgradient_optimize(f_opt, time_limit=t_sg,
                                                 n_top=15, lr_init=0.0005, momentum=0.93)
                try_sequence(f_opt2, best_tracker)

    # === Phase 4: Multi-n Exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration ===")
    t4_end = deadline - 150
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current)

    n_tries = [n_best + 64, n_best - 64, n_best + 128, n_best - 128,
               n_best + 32, n_best - 32, n_best + 96, n_best - 96]
    n_tries = [nt for nt in sorted(set(n_tries)) if 256 <= nt <= 1500]

    for nt in n_tries:
        t_remaining = t4_end - time.monotonic()
        if t_remaining < 40:
            break
        t_alloc = min(55, t_remaining * 0.3)

                                      n_top=25, n_j_candidates=4)
        try_sequence(f_opt, best_tracker)

        # LSE follow-up
        t_lse = min(30, t4_end - time.monotonic() - 20)
        if t_lse > 10:
            f_opt2, _ = lse_optimize(f_opt, time_limit=t_lse,
                                     beta_start=20.0, beta_end=1500.0, max_peaks=60)
            try_sequence(f_opt2, best_tracker)

    # === Phase 5: Final Intensive Polish ===
    print(f"\n=== Phase 5: Final Intensive Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent

    # High-precision CD
    t_alloc = min(100, deadline - time.monotonic() - 40)
    if t_alloc > 20:
        print(f"  Final CD (n_top=60): t={t_alloc:.0f}s")
        f_final, _ = coordinate_descent(current, time_limit=t_alloc,
                                        n_top=60, n_j_candidates=6)
        try_sequence(f_final, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(70, deadline - time.monotonic() - 25)
    if t_alloc > 15:
        print(f"  Final Subgrad: t={t_alloc:.0f}s")
        f_final2, _ = subgradient_optimize(current, time_limit=t_alloc,
                                           n_top=12, lr_init=0.0003, momentum=0.95)
        try_sequence(f_final2, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(50, deadline - time.monotonic() - 10)
    if t_alloc > 10:
        print(f"  Final LSE: t={t_alloc:.0f}s")
        f_final3, _ = lse_optimize(current, time_limit=t_alloc,
                                   beta_start=100.0, beta_end=5000.0, max_peaks=50)
        try_sequence(f_final3, best_tracker)

    # === Return best ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using parent fallback")
    return [float(x) for x in parent]
