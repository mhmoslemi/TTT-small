"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact subgradient descent on max of auto-convolution
2. Multiplicative (exponential) gradient updates for natural non-negativity
3. Frank-Wolfe with LP oracle for large steps
4. Multiple diverse initializations including antisymmetric structures
5. Adaptive learning rate with restarts
Key: Use the exact subgradient (not LSE approximation) and multiplicative updates.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # We want to project onto {x >= 0, sum(x) = target_sum}
    # Standard simplex projection scaled
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - target_sum)
    if not cond.any():
        return np.ones(n) * (target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_max_subgrad(f, conv):
    """Compute subgradient of max(conv) w.r.t. f."""
    n = len(f)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    # c[k] = sum_i a[i]*a[k-i], dc[ks]/da[j] = a[ks-j] + a[j] (if ks != 2j) or a[j] (if ks==2j)
    # But we need to be careful about bounds
    for j in range(n):
        kj = k_star - j
        if 0 <= kj < n:
            if k_star == 2 * j:
                grad[j] = f[j]
            else:
                grad[j] = 2.0 * f[kj]
    return grad

def multiplicative_grad_descent(f0, time_limit, lr_init=0.1, momentum=0.5):
    """Multiplicative gradient descent: f[j] *= exp(-lr * grad[j])"""
    f = np.maximum(f0.astype(np.float64), 1e-10)
    s = f.sum()
    f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    v = np.zeros(n)  # momentum buffer for the log-gradient
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-12:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        
        # Compute subgradient
        grad = compute_max_subgrad(f, conv)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Momentum
        v = momentum * v - lr * grad
        
        # Multiplicative update: f[j] *= exp(v[j])
        # But we need to maintain sum = 1
        # f_new[j] = f[j] * exp(v[j]) / sum(f[i]*exp(v[i]))
        log_f = np.log(f + 1e-30) + v
        log_f -= log_f.max()  # for numerical stability
        f_new = np.exp(log_f)
        f_new = f_new / f_new.sum()
        
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        
        if M_new < M:
            f = f_new
            lr = min(lr * 1.01, 2.0)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.9
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.001)
                no_improve = 0
    
    return best_f, best_obj

def euclidean_grad_descent(f0, time_limit, lr_init=0.05, momentum=0.7):
    """Euclidean projected gradient descent on max(c)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    v = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-12:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        
        grad = compute_max_subgrad(f, conv)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v, target_sum=1.0)
        
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        
        if M_new < M - 1e-14:

            if no_improve > 200:

def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative (exponential) gradient descent on exact max subgradient
2. Frank-Wolfe with LP oracle for large structural moves
3. Diverse initializations including antisymmetric and structured sequences
4. Multi-length exploration and interleaved refinement
Key insight: Use exact subgradient of max(convolution) with multiplicative updates
for natural non-negativity, combined with FW-LP for escaping local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.ones(n) * (target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_max_subgrad(f, conv):
    """Compute subgradient of max(c) w.r.t. f at the maximizing lag."""
    n = len(f)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    for j in range(n):
        kj = k_star - j
        if 0 <= kj < n:
            if k_star == 2 * j:
                grad[j] = f[j]
            else:
                grad[j] = 2.0 * f[kj]
    return grad


def compute_smoothed_grad(f, conv, beta, max_peaks=200):
    """Compute gradient of LSE(beta * conv) for smoothed optimization."""
    n = len(f)
    conv_len = len(conv)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()

    if len(w) > max_peaks:
        idx = np.argpartition(w, -max_peaks)[-max_peaks:]
        w_sparse = np.zeros(conv_len)
        w_sparse[idx] = w[idx]
        w = w_sparse

    grad = np.zeros(n)
    nonzero_w = np.where(w > 1e-15)[0]
    for k in nonzero_w:
        wk = w[k]
        k = int(k)
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                if k == 2 * j:
                    grad[j] += wk * f[j]
                else:
                    grad[j] += 2.0 * wk * f[kj]
    return grad


def multiplicative_grad_descent(f0, time_limit, lr_init=0.3, momentum=0.5, beta_mode=False, beta_start=10.0, beta_end=500.0):
    """Multiplicative gradient descent with exact max or LSE smoothed gradient."""
    f = np.maximum(f0.astype(np.float64), 1e-10)
    s = f.sum()
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    v = np.zeros(n)
    beta = beta_start

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-12:
            best_obj = M
            best_f = f.copy()
            no_improve = 0

        if beta_mode:
            grad = compute_smoothed_grad(f, conv, beta, max_peaks=150)
        else:
            grad = compute_max_subgrad(f, conv)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = momentum * v - lr * grad

        # Multiplicative update
        log_f = np.log(f + 1e-30) + v
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < M - 1e-14:
            f = f_new
            no_improve = 0
            lr = min(lr * 1.008, 5.0)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.92
            if no_improve > 150:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_init * 0.4, 0.01)
                no_improve = 0

        if beta_mode and beta < beta_end:
            beta *= 1.005

    return best_f, best_obj

def interpolate_to_length(f, new_n):

                print(f"  Cycle {cycles}: LSE smoothed t={t_alloc:.0f}s")
                f_opt, _ = euclidean_grad_descent(current, time_limit=t_alloc,
                                                  lr_init=0.02, momentum=0.75,
                                                  beta_mode=True, beta_start=30.0, beta_end=500.0)
                try_sequence(f_opt, best_tracker)
        else:
            t_alloc = min(60, remaining * 0.3)
            if t_alloc > 20:
                print(f"  Cycle {cycles}: FW-LP t={t_alloc:.0f}s")
                f_opt, _ = fw_lp(current, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)

    # === Phase 4: High-beta polishing ===
    print(f"\n=== Phase 4: High-Beta Polishing ===")
    t4_end = deadline - 80
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)

    # Very high beta LSE for fine polishing
    t_alloc = min(120, t4_end - time.monotonic())
    if t_alloc > 30:
        print(f"  High-beta LSE: n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = euclidean_grad_descent(current, time_limit=t_alloc,
                                          lr_init=0.01, momentum=0.8,
                                          beta_mode=True, beta_start=200.0, beta_end=5000.0)
        try_sequence(f_opt, best_tracker)

    # Exact max with very small steps
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(80, t4_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Exact max fine step: t={t_alloc:.0f}s")
        f_opt, _ = multiplicative_grad_descent(current, time_limit=t_alloc,
                                               lr_init=0.05, momentum=0.7)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Try nearby lengths ===
    print(f"\n=== Phase 5: Nearby Lengths ===")
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        offsets = [64, -64, 128, -128, 32, -32]
        for off in offsets:
            n_try = n_best + off
            if n_try < 128 or n_try > 1500:
                continue
            t_remaining = deadline - time.monotonic() - 15
            if t_remaining < 40:
                break
            t_alloc = min(50, t_remaining * 0.4)
            if t_alloc < 15:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = multiplicative_grad_descent(f_init, time_limit=t_alloc,
                                                   lr_init=0.2, momentum=0.5)
            try_sequence(f_opt, best_tracker)

    # === Final verification and return ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using uniform fallback")
    fallback = np.ones(768) / 768
    return [float(x) for x in fallback]
