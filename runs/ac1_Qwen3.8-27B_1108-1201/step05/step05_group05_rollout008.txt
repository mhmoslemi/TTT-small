"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
LP-Frank-Wolfe with adaptive constraints, coordinate descent, Nesterov-accelerated
subgradient, multi-length exploration, and diverse restarts.
Warm-started from previous best solution.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lpfw_optimize(f0, time_limit, top_k=40, verbose=False, m_expand=1.05,
                  no_improve_break=100, use_all_every=50):
    """LP-Frank-Wolfe with adaptive constraint selection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_conv(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Adaptive constraint selection
        active_k = None
        if it > 3:
            k_sorted = np.argsort(conv)[::-1][:top_k]
            active_k = k_sorted.tolist()
        if use_all_every and it % use_all_every == 0:
            active_k = None

        # Solve LP
        K = 2 * n - 1 if active_k is None else len(active_k)
        c_vec = -np.ones(n)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M * m_exp)

        if active_k is None:
            for j in range(n):
                A_ub[j:j+n, j] = f
        else:
            for row, k in enumerate(active_k):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    A_ub[row, j] = f[k - j]

        bounds = [(0, None)] * n
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if not res.success:
            m_exp = min(m_exp * m_expand, 5.0)
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > no_improve_break:
                break
            m_exp = min(m_exp * 1.02, 3.0)
            continue

        # Line search (simple golden section, 20 iterations)
        phi = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c_pt = hi - phi * (hi - lo)
        d_pt = lo + phi * (hi - lo)
        f_c = project_simplex(f + c_pt * (g - f))
        f_d = project_simplex(f + d_pt * (g - f))
        fc = fast_conv(f_c).max()
        fd = fast_conv(f_d).max()

        for _ in range(20):
            if fc < fd:
                hi, d_pt = d_pt, c_pt
                fd = fc
                c_pt = hi - phi * (hi - lo)
                f_c = project_simplex(f + c_pt * (g - f))
                fc = fast_conv(f_c).max()
            else:
                lo, c_pt = c_pt, d_pt
                fc = fd
                d_pt = lo + phi * (hi - lo)
                f_d = project_simplex(f + d_pt * (g - f))
                fd = fast_conv(f_d).max()

        alpha = (lo + hi) / 2.0

        if alpha > 1e-10:
            f_new = project_simplex(f + alpha * (g - f))
            conv = fast_conv(f_new)
            M_new = conv.max()

            if M_new < best_obj - 1e-15:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                M = M_new
                m_exp = 1.0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW it={it}, obj={best_obj:.10f}, t={elapsed:.0f}s")
            else:
                no_improve += 1
            f = f_new
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 4.0)

        if no_improve > 50:
            # Restart with perturbation
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.002 + 0.004 * min(no_improve // 50, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_conv(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coord_descent(f0, time_limit, n_mults=10, verbose=False):
    """Coordinate descent with efficient convolution updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_conv(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve_passes = 0

    # Multipliers to try for each coordinate
    multipliers = np.array([0.1, 0.3, 0.5, 0.7, 0.85, 1.0,
                            1.15, 1.3, 1.5, 1.8, 2.2, 3.0])

    pass_count = 0
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                # Try turning on a zero coordinate
                others_sum = 1.0
                f[j] = 0.05
                f[np.arange(n) != j] *= 0.95 / others_sum  # rough
                f = f / f.sum()
                trial_conv = fast_conv(f).max()
                if trial_conv > best_obj + 1e-15:
                    f[j] = 0.0
                    f = f / f.sum()
                else:
                    best_obj = trial_conv
                    best_f = f.copy()
                    improved = True
                continue

            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                # Normalize to be safe
                f_trial /= f_trial.sum()

                trial_conv = fast_conv(f_trial).max()
                if trial_conv < best_j_obj - 1e-16:
                    best_j_obj = trial_conv
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()

                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}")

def nesterov_sg(f0, time_limit, lr0=0.003, mom=0.95, top_k=50, temp=1e-5):
    """Nesterov-accelerated subgradient method."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    conv = fast_conv(y)
    best_obj = conv.max()
    best_f = f.copy()
    beta = 0.5

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Compute subgradient at active peaks
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(k_act)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(k_act):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[k - idx]

        # Nesterov update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        theta = (1.0 - beta) / beta_new
        beta = beta_new

        y = project_simplex(y - lr0 * sg)
        f_new = (1.0 - theta) * y + theta * f

        conv = fast_conv(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 17)
            scale = 0.003 + 0.002 * min(no_improve // 200, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            y = f.copy()
            beta = 0.5
            no_improve = 0

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving non-negativity and normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.exp(1j * rng.uniform(0, 2 * np.pi, fft_size // 2 + 1))
    sig = np.fft.irfft(spectrum, fft_size)[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
LP-Frank-Wolfe with adaptive constraints, coordinate descent, Nesterov-accelerated
subgradient, multi-length exploration, and diverse restarts. Warm-started from
previous best solution for maximum efficiency.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lpfw_optimize(f0, time_limit, top_k=45, verbose=False,
                  no_improve_break=80, use_all_every=40):
    """LP-Frank-Wolfe with adaptive constraint selection and golden section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_conv(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    phi = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptive constraint selection
        active_k = None
        if it > 3:
            k_sorted = np.argsort(conv)[::-1][:top_k]
            active_k = k_sorted.tolist()
        if use_all_every > 0 and it % use_all_every == 0:
            active_k = None

        # Build LP
        K = (2 * n - 1) if active_k is None else len(active_k)
        c_vec = -np.ones(n)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M * m_exp)

        if active_k is None:
            for j in range(n):
                A_ub[j:j + n, j] = f
        else:
            for row, k in enumerate(active_k):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    A_ub[row, j] = f[k - j]

        bounds = [(0, None)] * n
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if not res.success:
            m_exp = min(m_exp * 1.08, 5.0)
            if m_exp >= 5.0:
                m_exp = 1.0
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > no_improve_break:
                break
            m_exp = min(m_exp * 1.03, 3.0)
            continue

        # Golden section line search
        lo, hi = 0.0, 1.0
        c_pt = hi - phi * (hi - lo)
        d_pt = lo + phi * (hi - lo)
        f_c = project_simplex(f + c_pt * (g - f))
        f_d = project_simplex(f + d_pt * (g - f))
        fc = fast_conv(f_c).max()
        fd = fast_conv(f_d).max()

        for _ in range(22):
            if fc < fd:
                hi, d_pt = d_pt, c_pt
                fd = fc
                c_pt = hi - phi * (hi - lo)
                f_c = project_simplex(f + c_pt * (g - f))
                fc = fast_conv(f_c).max()
            else:
                lo, c_pt = c_pt, d_pt
                fc = fd
                d_pt = lo + phi * (hi - lo)
                f_d = project_simplex(f + d_pt * (g - f))
                fd = fast_conv(f_d).max()

        alpha = (lo + hi) / 2.0

        if alpha > 1e-10:
            f_new = project_simplex(f + alpha * (g - f))
            conv = fast_conv(f_new)
            M_new = conv.max()

            if M_new < best_obj - 1e-15:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                M = M_new
                m_exp = 1.0
                if verbose and it % 40 == 0:
                    print(f"    LP-FW it={it}, obj={best_obj:.10f}, t={time.monotonic()-t0:.0f}s")
            else:
                no_improve += 1
            f = f_new
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.04, 4.0)

        if no_improve > 40:
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.002 + 0.003 * min(no_improve // 40, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_conv(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coord_descent(f0, time_limit, verbose=False):
    """Coordinate descent with adaptive multipliers and simplex constraint."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_conv(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve_passes = 0
    pass_count = 0

    multipliers = np.array([0.2, 0.4, 0.6, 0.75, 0.9, 1.1,
                            1.25, 1.45, 1.7, 2.0, 2.5, 3.5])
    mask = np.ones(n, dtype=bool)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < -1e-15 or new_val > 1.0 - 1e-12:
                    continue
                new_val = max(0.0, new_val)
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask[j] = False
                f_trial[mask] *= scale
                mask[j] = True
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial /= ss

                trial_obj = fast_conv(f_trial).max()
                if trial_obj < best_j_obj - 1e-16:
                    best_j_obj = trial_obj
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask[j] = False
                f[mask] *= scale
                mask[j] = True
                f = f / f.sum()

                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

        if verbose and pass_count % 3 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}")

    return best_f, best_obj


def nesterov_sg(f0, time_limit, lr0=0.0025, top_k=50, temp=5e-6):
    """Nesterov-accelerated subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    conv = fast_conv(y)
    best_obj = conv.max()
    best_f = f.copy()
    beta = 0.5

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(k_act)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(k_act):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[k - idx]

        # Nesterov momentum update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        theta = (1.0 - beta) / beta_new
        beta = beta_new

        y = project_simplex(y - lr0 * sg)
        f = (1.0 - theta) * y + theta * f

        conv = fast_conv(f)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            rng = np.random.default_rng(it * 31 + 17)
            scale = 0.003 + 0.002 * min(no_improve // 150, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            y = f.copy()
            beta = 0.5
            no_improve = 0

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving non-negativity and normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum (random phases)."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.exp(1j * rng.uniform(0.0, 2.0 * np.pi, fft_size // 2 + 1))
    sig = np.fft.irfft(spectrum, fft_size)[:n]
    sig = np.maximum(sig, 0.0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: LP-FW with adaptive constraints, coordinate descent,
    Nesterov subgradient, multi-length exploration, diverse restarts, and micro
    perturbations. Warm-starts from height_sequence_1.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent from previous best
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent found, using uniform n=512, score={best_score:.6f}")

    # Phase 1: LP-FW on parent
    print(f"\n=== Phase 1: LP-FW on parent (n={len(parent)}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(350.0, rem * 0.25)
    if t1 > 30:
        a_opt, _ = lpfw_optimize(parent, time_limit=t1, top_k=50, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [lpfw_parent]: {sc:.6f}")

    # Phase 2: Coordinate descent
    print(f"\n=== Phase 2: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(220.0, rem * 0.14)
    if t2 > 20 and best_seq is not None:
        a_opt, _ = coord_descent(best_seq, time_limit=t2, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [cd]: {sc:.6f}")

    # Phase 3: Multi-length exploration
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    for target_n in [1024, 1536, 2048]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        f_init = interpolate_seq(base, target_n)
        fs = flat_spectrum_init(target_n, seed=target_n)
        f_mix = project_simplex(0.7 * f_init + 0.3 * fs)
        t_alloc = min(rem * 0.18, 220)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lpfw_optimize(f_mix, time_limit=t_alloc, top_k=40)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [len_{target_n}]: {sc:.6f}")

        rem = deadline - time.monotonic()
        if rem > 40:
            t_cd = min(rem * 0.08, 100)
            a_opt2, _ = coord_descent(a_opt, time_limit=t_cd)
            sc2 = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_opt2.copy()
                print(f"  NEW BEST [cd_len_{target_n}]: {sc2:.6f}")

    # Phase 4: Nesterov subgradient on best
    print(f"\n=== Phase 4: Nesterov SG (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(180.0, rem * 0.12)
    if t4 > 20 and best_seq is not None:
        n_best = len(best_seq)
        a_opt, _ = nesterov_sg(best_seq, time_limit=t4,
                               lr0=0.002, top_k=max(30, n_best // 4), temp=5e-6)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST [nesterov]: {sc:.6f}")

    # Phase 5: Diverse restarts
    print(f"\n=== Phase 5: Diverse Restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        restart_list = []
        restart_list.append(("noise1", project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)))
        restart_list.append(("noise2", project_simplex(best_seq + rng.standard_normal(n_best) * 0.008)))
        restart_list.append(("rev", project_simplex(0.6 * best_seq + 0.4 * best_seq[::-1])))
        restart_list.append(("flat", project_simplex(0.82 * best_seq + 0.18 * np.ones(n_best) / n_best)))
        center = n_best / 2.0
        gauss = np.exp(-3.0 * (np.arange(n_best) - center) ** 2 / (0.45 * n_best) ** 2)
        restart_list.append(("gauss", project_simplex(gauss)))
        two_pk = (np.exp(-10.0 * (np.arange(n_best) - 0.3 * n_best) ** 2 / (0.12 * n_best) ** 2) +
                  np.exp(-10.0 * (np.arange(n_best) - 0.7 * n_best) ** 2 / (0.12 * n_best) ** 2))
        restart_list.append(("two_pk", project_simplex(two_pk)))

        for i, (name, f_init) in enumerate(restart_list):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem / max(1, len(restart_list) - i), 140)
            print(f"  Restart [{name}] t={t_alloc:.0f}s")
            a_opt, _ = lpfw_optimize(f_init, time_limit=t_alloc, top_k=35)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [restart_{name}]: {sc:.6f}")

    # Phase 6: Final coordinate descent + Nesterov
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.35, 180)
            a_opt, _ = coord_descent(best_seq, time_limit=t_cd)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [final_cd]: {sc:.6f}")

        rem = deadline - time.monotonic()
        if rem > 20:
            n_best = len(best_seq)
            t_sg = min(rem * 0.3, 100)
            a_opt2, _ = nesterov_sg(best_seq, time_limit=t_sg,
                                    lr0=0.001, top_k=max(20, n_best // 5), temp=3e-6)
            sc2 = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_opt2.copy()
                print(f"  NEW BEST [final_sg]: {sc2:.6f}")

    # Phase 7: Micro perturbations
    print(f"\n=== Phase 7: Micro Perturbations (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for i in range(10):
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            noise_scale = 0.0004 * (i + 1)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_pert = min(rem * 0.12, 25)
            a_opt, _ = lpfw_optimize(f_pert, time_limit=t_pert, top_k=25)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [micro_{i}]: {sc:.6f}")

        # Tiny final CD
        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt, _ = coord_descent(best_seq, time_limit=min(rem * 0.4, 40))
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [tiny_cd]: {sc:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
