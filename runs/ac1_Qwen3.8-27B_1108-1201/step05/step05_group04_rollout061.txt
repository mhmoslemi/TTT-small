            best_a = a.copy()
        else:
            a[i] = best_a[i]
            a[j] = best_a[j]
    return best_a, best_obj


def try_sequence(a, best_tracker):
    """Evaluate and update best tracker."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    if new_n <= 0:
        return np.ones(1)
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def diverse_init(n, seed):
    """ Create diverse structured initialization. """
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n
    x = np.arange(n)
    for _ in range(rng.integers(1, 4)):
        freq = rng.integers(1, max(2, n // 30))
        amp = 0.05 + 0.15 * rng.random()
        phase = rng.random() * 2.0 * np.pi
        a += amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization: multi-phase search combining FFT-accurate LSE/FISTA,
    peak subgradient, and coordinate refinement with adaptive beta scheduling.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Diverse multi-length exploration ===
    print("\n=== Phase 1: Diverse Exploration ===")
    t1_end = deadline - 800
    candidates = []
    # Parent at same length
    candidates.append((n_parent, parent.copy()))
    # Interpolated to nearby lengths
    for off in [64, -64, 128, -128]:
        nl = n_parent + off
        if 256 <= nl <= 1400:
            candidates.append((nl, interpolate_to_length(parent, nl)))
    # Diverse inits at parent length
    for si in range(3):
        candidates.append((n_parent, diverse_init(n_parent, seed + 100 + si)))
    # Diverse at larger length
    nl_large = min(n_parent + 192, 1200)
    candidates.append((nl_large, diverse_init(nl_large, seed + 200)))

    tried = 0
    for nl, f_init in candidates:
        t_rem = t1_end - time.monotonic()
        if t_rem < 35 or tried >= 5:
            break
        t_alloc = min(130, t_rem * 0.32)
        if t_alloc < 20:
            break
        tried += 1
        print(f"  FISTA n={nl}, t={t_alloc:.0f}s")
        f_opt, _ = fista_lse(f_init, time_limit=t_alloc,
                              beta_start=8.0, beta_end=1500.0, verbose=False)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Intensive FISTA with high beta ===
    print(f"\n=== Phase 2: Intensive FISTA ===")
    t2_end = deadline - 550
    current = best_tracker[1]
    t_alloc = min(280, t2_end - time.monotonic())
    if t_alloc > 30:
        print(f"  FISTA n={len(current)}, t={t_alloc:.0f}s, beta 15->8000")
        f_opt, _ = fista_lse(current, time_limit=t_alloc,
                              beta_start=15.0, beta_end=8000.0, verbose=True)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Peak subgradient ===
    print(f"\n=== Phase 3: Peak Subgradient ===")
    t3_end = deadline - 350
    current = best_tracker[1]
    t_alloc = min(180, t3_end - time.monotonic())
    if t_alloc > 20:
        print(f"  PeakSub n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = peak_subgrad(current, time_limit=t_alloc,
                                 n_peaks=8, lr_init=0.002, mom_init=0.93)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1]
    t_alloc = min(120, t3_end - time.monotonic())
    if t_alloc > 15:
        print(f"  FISTA high-beta n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = fista_lse(current, time_limit=t_alloc,
                              beta_start=300.0, beta_end=15000.0, verbose=False)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: Alternating refinement ===
    print(f"\n=== Phase 4: Alternating Refinement ===")
    t4_end = deadline - 150
    cyc = 0
    while time.monotonic() < t4_end and cyc < 6:
        cyc += 1
        current = best_tracker[1]
        n_cur = len(current)
        rem = t4_end - time.monotonic()
        if rem < 50:
            break
        if cyc % 2 == 1:
            t_alloc = min(90, rem * 0.45)
            if t_alloc > 15:
                print(f"  Cycle {cyc}: FISTA n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = fista_lse(current, time_limit=t_alloc,
                                      beta_start=500.0, beta_end=20000.0, verbose=False)
                try_sequence(f_opt, best_tracker)
        else:
            t_alloc = min(90, rem * 0.45)
            if t_alloc > 15:
                print(f"  Cycle {cyc}: PeakSub n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = peak_subgrad(current, time_limit=t_alloc,
                                         n_peaks=5, lr_init=0.001, mom_init=0.95)
                try_sequence(f_opt, best_tracker)

    # === Phase 5: Coordinate refinement ===
    print(f"\n=== Phase 5: Coordinate Refinement ===")
    current = best_tracker[1]
    t_alloc = min(70, deadline - time.monotonic() - 10)
    if t_alloc > 10:
        print(f"  CoordRef n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = coord_refine(current, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Nearby Lengths ===")
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        for off in [32, -32, 64, -64]:
            n_try = n_best + off
            if n_try < 256 or n_try > 1500:
                continue
            t_rem = deadline - time.monotonic() - 10
            if t_rem < 25:
                break
            t_alloc = min(45, t_rem * 0.4)
            if t_alloc < 12:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = fista_lse(f_init, time_limit=t_alloc,
                                  beta_start=20.0, beta_end=3000.0, verbose=False)
            try_sequence(f_opt, best_tracker)

    # Final
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Uniform fallback")
    fb = np.ones(768) / 768
    return [float(x) for x in fb]
