        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += wk * f[k - np.arange(lo1, hi1 + 1)]
        # f[i-k] valid when 0 <= i-k < n
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            grad[lo2:hi2+1] += wk * f[np.arange(lo2, hi2 + 1) - k]
    
    return grad

def lse_value(f, beta):
    conv = fast_convolve(f)
    bc = beta * conv
    return (1.0 / beta) * (bc.max() + np.log(np.exp(bc - bc.max()).sum()))

def fista_lse(f0, time_limit, beta_start=10.0, beta_end=10000.0, 
              max_terms=300, lr_init=None, momentum_schedule='nesterov'):
    """FISTA with LSE-smoothed max on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    if lr_init is None:
        lr = 1.0 / (beta_start * n * 2)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    beta = beta_start
    t_nest = 1.0
    y = f.copy()
    v = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient at y
        grad = lse_gradient(y, beta, max_terms)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            no_improve += 100
            break
        grad /= gnorm
        
        # FISTA update
        lr = lr_init * (beta_start / beta) if lr_init else 1.0 / (beta * n * 2)
        x_new = project_simplex(y - lr * grad)
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        y = x_new + ((t_nest - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = x_new
        else:
            y = y / ys  # Keep y on simplex for stability
        t_nest = t_new
        f = x_new
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Reset momentum and try higher beta
                f = best_f.copy()
                y = f.copy()
                t_nest = 1.0
                beta = min(beta * 1.5, beta_end)
                no_improve = 0
        
        # Gradually increase beta
        if beta < beta_end:
            beta *= 1.003
    
    return best_f, best_obj

def coord_descent_peak(f0, time_limit, n_sweeps=1000):
    """Coordinate descent that directly targets the max peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))
        
        # Compute pressure for the peak
        # p[i] = gradient contribution to (f*f)[k_star]
        p = np.zeros(n)
        lo1 = max(0, k_star - n + 1)
        hi1 = min(n - 1, k_star)
        if lo1 <= hi1:
            p[lo1:hi1+1] += f[k_star - np.arange(lo1, hi1 + 1)]
        lo2 = max(0, k_star)
        hi2 = min(n - 1, k_star + n - 1)
        if lo2 <= hi2:
            p[lo2:hi2+1] += f[np.arange(lo2, hi2 + 1) - k_star]
        
        # Find source (highest pressure) and target (lowest pressure)
        # Only consider positions where f > 0 for source
        source_candidates = np.where(f > 1e-10)[0]
        if len(source_candidates) == 0:
            break
        
        p_source = p[source_candidates]
        src_local = np.argmax(p_source)
        src = source_candidates[src_local]
        
        # Target: minimize pressure, prefer positions where f is small
        target_candidates = np.where(f < 0.5 * f.max())[0]
        if len(target_candidates) == 0:
            target_candidates = np.arange(n)
        
        p_target = p[target_candidates]
        # Weight by current value to prefer moving to "empty" positions
        target_score = p_target + 10.0 * f[target_candidates]
        tgt_local = np.argmin(target_score)
        tgt = target_candidates[tgt_local]
        
        if src == tgt:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Move mass from src to tgt
        # Try different deltas
        best_delta = 0.0
        best_conv_max = conv[k_star]
        current_val = f[src]
        
        # Binary search for the largest delta that doesn't increase the peak
        deltas = current_val * np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8])
        for delta in deltas:
            if delta >= current_val:
                continue
            f_trial = f.copy()
            f_trial[src] -= delta
            f_trial[tgt] += delta
            f_trial = np.maximum(f_trial, 0)
            f_trial /= f_trial.sum()
            
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            
            if obj_trial < best_conv_max - 1e-15:
                best_conv_max = obj_trial
                best_delta = delta
        
        if best_delta > 1e-10:
            f[src] -= best_delta
            f[tgt] += best_delta
            f = np.maximum(f, 0)
            f /= f.sum()
            
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Perturb and restart
            rng = np.random.default_rng(it * 7919 + 3)
            scale = 0.001 + 0.002 * min(1.0, no_improve / 200.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            no_improve = 0
    
    return best_f, best_obj

def multiplicative_updates(f0, time_limit, beta=50.0, lr_init=0.1, max_terms=200):
    """Multiplicative (mirror descent) updates for the LSE objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)

    n = len(f)

    t0 = time.monotonic()

        it += 1
        
        # Compute gradient
        grad = lse_gradient(f, cur_beta, max_terms)
        
        # Multiplicative update: f[i] *= exp(-lr * grad[i])
        # Clip to avoid overflow
        update = np.exp(-lr * np.clip(grad, -10, 10))
        f_new = f * update
        f_new = np.maximum(f_new, 1e-15)
        f_new /= f_new.sum()
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()

            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.95
            if no_improve > 50:

                f = np.maximum(f, 1e-10)
                f /= f.sum()
                cur_beta = min(cur_beta * 1.3, 5000.0)
                lr = lr_init * 0.5
                no_improve = 0

            cur_beta *= 1.002
    
    return best_f, best_obj

def mixed_peak_lse(f0, time_limit, n_peaks_schedule=None):
    """
    Hybrid method: alternate between LSE gradient (for global structure)
    and direct max subgradient (for local peak reduction).
    Uses adaptive weighting between the two.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv_len = 2 * n - 1
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    beta = 30.0
    lr = 2.0 / n
    momentum = 0.92
    v = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Compute LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w_lse = exp_bc / exp_bc.sum()
        
        # Compute top-k peak weights
        k_top = min(max(4, n_peaks_schedule[it] if n_peaks_schedule else 10), conv_len)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        # Sharpen: weight proportional to how much above the second-highest
        pv_sorted = np.sort(pv)[::-1]
        if len(pv_sorted) > 1 and pv_sorted[0] > pv_sorted[1]:
            margin = pv_sorted[0] - pv_sorted[1]
            w_peak = np.exp((pv - pv[0]) / max(margin * 0.5, 1e-15))
        else:
            w_peak = np.ones(k_top) / k_top
        w_peak /= w_peak.sum()
        
        # Blend
        alpha_blend = min(0.7, 0.3 + 0.001 * it)
        w_full = np.zeros(conv_len)
        w_full[peaks] += w_peak
        # Add LSE contribution
        w_full += (1.0 - alpha_blend) * w_lse
        w_full /= w_full.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        # Use top max_terms for efficiency
        num_active = min(max_terms := 200, conv_len)
        active_idx = np.argpartition(w_full, -num_active)[-num_active:]
        active_idx = active_idx[w_full[active_idx] > 1e-12]
        
        for k in active_idx:
            wk = w_full[int(k)]
            if wk < 1e-15:
                continue
            k = int(k)
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            if lo1 <= hi1:
                grad[lo1:hi1+1] += wk * f[k - np.arange(lo1, hi1 + 1)]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if lo2 <= hi2:
                grad[lo2:hi2+1] += wk * f[np.arange(lo2, hi2 + 1) - k]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            no_improve += 100
            break
        grad /= gnorm
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 6:
                lr *= 0.94
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                beta = min(beta * 1.4, 5000.0)
                lr = max(2.0 / n, 0.001)
                momentum = 0.90
                no_improve = 0
        
        beta *= 1.003
    
    return best_f, best_obj
