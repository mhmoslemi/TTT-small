def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None, time_limit=30):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def lse_continuation(f0, time_limit, beta_start=10.0, beta_max=20000.0, max_peaks=300):
    """LSE continuation with very gradual beta increase and adaptive LR."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        bc_max = bc.max()

            w_sparse = w

def _compute_grad(f, w):
    """Compute gradient of weighted sum of convolution peaks."""
    n = len(f)
    conv_len = 2 * n - 1
    grad = np.zeros(n)

    # Only process non-negligible weights
    threshold = 1e-10 * w.max() if w.max() > 0 else 0
    active = np.where(w > threshold)[0]

    for k in active:
        wk = w[k]
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        ms = np.arange(lo, hi + 1)
        # Two contributions (symmetric)
        grad[ms] += wk * f[k - ms]
        # Second part: same range by symmetry for auto-conv
        ms2_lo = max(0, k - hi)
        ms2_hi = min(n - 1, k - lo)
        if ms2_lo <= ms2_hi:
            ms2 = np.arange(ms2_lo, ms2_hi + 1)
            grad[ms2] += wk * f[k - ms2]

    return grad


def peak_shaving(f0, time_limit, n_peaks=10, eta=0.002):
    """Directly shave peaks by moving mass from high-pressure to low-pressure coords."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:

    n = len(f)

    conv = fast_convolve(f)

    it = 0

        it += 1
        conv = fast_convolve(f)
        cur_max = conv.max()

        # Select top peaks
        k_sel = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        pv = conv[peaks]
        # Weight peaks by excess over mean
        mean_conv = conv.mean()
        excess = np.maximum(pv - mean_conv, 0)
        w = excess / (excess.sum() + 1e-15)

        # Build weight vector
        w_full = np.zeros(len(conv))
        w_full[peaks] = w

        grad = _compute_grad(f, w_full)
        grad -= grad.mean()

        # Find best swap direction: decrease high-grad coords, increase low-grad
        # Project to get descent direction
        desc = -grad
        desc -= desc.mean()
        desc_norm = np.linalg.norm(desc)
        if desc_norm < 1e-15:
            no_improve += 50
            break
        desc /= desc_norm

        # Try multiple step sizes
        best_alpha = 0
        best_val = cur_max
        for alpha in [eta * 0.5, eta, eta * 2, eta * 4, eta * 8]:
            f_new = np.maximum(f + alpha * desc, 0.0)
            f_new /= f_new.sum()
            val = fast_convolve(f_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0 and best_val < best_obj - 1e-14:
            f = np.maximum(f + best_alpha * desc, 0.0)
            f /= f.sum()
            best_obj = best_val
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:

                no_improve = 0

    return best_f, best_obj


def multi_scale_optimize(parent, n_target, time_limit):
    """Coarse-to-fine optimization: optimize a shorter sequence, then upsample."""
    # Start at n_target/4, optimize, then upsample
    n_coarse = max(64, n_target // 4)
    f_coarse = parent[:n_coarse].copy() if len(parent) >= n_coarse else parent.copy()
    if len(f_coarse) != n_coarse:
        f_coarse = np.interp(np.linspace(0, 1, n_coarse), np.linspace(0, 1, len(parent)), parent)
    f_coarse = np.maximum(f_coarse, 0)
    f_coarse /= f_coarse.sum()

    # Optimize coarse
    t_coarse = time_limit * 0.3
    f_coarse_opt, _ = fw_lp_extended(f_coarse, time_limit=t_coarse)

    # Upsample
    f_up = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_coarse), f_coarse_opt)
    f_up = np.maximum(f_up, 0)
    s = f_up.sum()
    if s > 1e-15:
        f_up /= s
    else:
        f_up = np.ones(n_target) / n_target

    # Refine at target length
    t_refine = time_limit * 0.7
    f_final, _ = fw_lp_extended(f_up, time_limit=t_refine)
    return f_final


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
