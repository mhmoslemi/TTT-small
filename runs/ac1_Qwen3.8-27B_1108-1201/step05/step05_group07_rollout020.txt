import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)

    Find optimal value for a[j] given all other values.

For example, a triangular distribution (high in the middle) would give b_{n-1} = sum a_i * a_{n-1-i} which is like the "energy" of the sequence (if it's symmetric). But if a is asymmetric (e.g., increasing), then a_i * a_{n-1-i} = a_i * a_{n-1-i} where one is small and the other is large, giving a smaller product.

            for k in range(k_start, k_end + 1):

                else:

            D = np.zeros(2*n-1)

                if idx != j:
                    D[k] = 2.0 * a[idx]

            C = np.zeros(2*n-1)
            if 2*j < 2*n-1:
                C[2*j] = 1.0

            inv_S = 1.0 / S

            max_vals = vals.max(axis=1)

                a = np.maximum(a, 0)

        if not improved:
            break
        
        no_improve = 0 if improved else no_improve + 1
    
    return best_a, best_obj
