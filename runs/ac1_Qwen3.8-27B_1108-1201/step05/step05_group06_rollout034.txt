    Given f (normalized) and M = max(f*f), solve:
    max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k

    #   For each k in 0..2n-2: sum_{j=max(0,k-n+1)}^{min(n-1,k)} f_{k-j} * g_j <= M

from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_constraint_matrix(f):
    """Build the constraint matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[k, j] = f[k - j]
    return A

def lp_descent(f_init, time_limit, t_step=0.1, max_iters=10000, verbose=True):
    """LP-based descent method."""
    f = f_init.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2 * n * conv.max()  # since sum=1
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        M = conv.max()
        
        # Build constraint matrix
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            A[k, js:je+1] = f[k - np.arange(js, je+1)]
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        except Exception as e:
            if verbose:
                print(f"  LP error at iter {it}: {e}")
            break
        
        if not result.success:
            if verbose:
                print(f"  LP failed at iter {it}: {result.message}")
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        g_norm = g / g_sum
        
        # Check if g is close to f (convergence)
        diff = np.abs(g_norm - f).max()
        if diff < 1e-10:
            if verbose and it % 100 == 0:
                print(f"    Converged at iter {it}, obj={best_obj:.10f}")
            break

            f_trial = np.maximum(f_trial, 0)

            obj_trial = 2 * n * conv_trial.max()

                f = f_trial
                conv = conv_trial
                best_obj = obj_trial

                improved = True

        if verbose and it % 100 == 0:
            print(f"  LP iter {it}: obj={best_obj:.10f}, max_diff={diff:.2e}")
    
    return best_f, best_obj

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build matrix A where (A@g)[k] = (f*g)[k] for linear convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = f[k - np.arange(js, je + 1)]
    return A


def lp_descent(f_init, time_limit, t_init=0.1, verbose=True):
    """
    LP-based descent: at each step, solve LP to find direction g that
    maximizes sum(g) subject to f*g <= M and g >= 0. Then take a step
    towards g.
    """
    f = f_init.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()  # Since sum=1, objective = 2*n*max(conv)
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    stagnant = 0
    t = t_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 0.5:
            break
        
        M = conv.max()
        
        # Build constraint matrix
        A = build_conv_matrix(f)
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub,
                            bounds=[(0, None)] * n, method='highs')
        except Exception:
            break
        
        if not result.success:
            break
        
        g = result.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        g_norm = g / g_sum
        
        # Check convergence: if g is very close to f
        diff = np.max(np.abs(g_norm - f))
        if diff < 1e-12:
            if verbose:
                print(f"    LP converged at iter {it}, obj={best_obj:.12f}")
            break
        
        # Backtracking line search
        improved = False
        t_trial = t
        for _ in range(12):
            f_trial = (1 - t_trial) * f + t_trial * g_norm
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                t_trial *= 0.5
                continue
            f_trial /= st
            
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            
            if obj_trial < best_obj - 1e-14:
                f = f_trial
                conv = conv_trial
                best_obj = obj_trial
                best_f = f.copy()
                improved = True
                t = max(t_trial, 0.01)  # Don't shrink t too much
                break
            t_trial *= 0.5
        
        if not improved:
            stagnant += 1
            if stagnant > 20:
                if verbose:
                    print(f"    LP stuck at iter {it}, obj={best_obj:.12f}")
                break
        else:
            stagnant = 0
        
        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: obj={best_obj:.12f}, t={t:.4f}, diff={diff:.2e}")
    
    if verbose:
        print(f"    LP finished: {it} iters, {elapsed:.1f}s, obj={best_obj:.12f}")
    
    return best_f, best_obj


def coordinate_bisection(a0, time_limit, max_passes=200, verbose=False):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        max_improvement = 0.0
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-14:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            lo = 0.0
            hi = 1.0
            x_new = x_old
            
            for _ in range(45):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                
                k_max = int(np.argmax(f_vals))
                
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                
                if deriv > 1e-13:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-13:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break
            
            if abs(x_new - x_old) < 1e-14:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue
            
            conv = fast_convolve(a)
            obj = conv.max()
            
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
                improvement = best_obj - obj  # This is actually the new objective
                # Actually improvement should be old_best - new_obj, but we already updated best_obj
                # Let me just track pass_improved
        
        if not pass_improved:
            stagnant += 1
            if stagnant > 3:
                if passes > 5:
                    rng = np.random.default_rng(passes * 47 + 11)
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
                    conv = fast_convolve(a)
                    # Reset a to the perturbed version and continue
                    stagnant = 0
                else:
                    break
        else:
            stagnant = 0
        
        if verbose and passes % 10 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.12f}")
    
    if verbose:
        print(f"    CD finished: {passes} passes, {time.monotonic()-t0:.1f}s, obj={best_obj:.12f}")
    
    return best_a, best_obj


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
LP-based descent for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses the LP formulation: max sum(g) s.t. f*g <= M, g >= 0,
combined with coordinate bisection polishing and multi-restart search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build matrix A where (A@g)[k] = (f*g)[k] for linear convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = f[k - np.arange(js, je + 1)]
    return A

"""

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build matrix A where (A@g)[k] = (f*g)[k] for linear convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = f[k - np.arange(js, je + 1)]
    return A

    """
    LP-based descent: solve max sum(g) s.t. f*g <= M, g>=0 to find direction.
    Take steps towards the normalized LP solution with backtracking.
    """
    f = f_init.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    current_obj = conv.max()
    best_obj = current_obj
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    stagnant = 0
    t = t_init
    lp_opts = {'time_limit': lp_timeout, 'presolve': True}

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        M = conv.max()
        A = build_conv_matrix(f)

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        try:
            result = linprog(c, A_ub=A, b_ub=b_ub,
                            bounds=[(0, None)] * n, method='highs',
                            options=lp_opts)
        except Exception:
            break

                print(f"    LP failed at iter {it}: {result.message}")
            break

        g = result.x
        g_sum = g.sum()

        diff = np.max(np.abs(g_norm - f))

                print(f"    LP converged at iter {it}, obj={best_obj:.12f}")
            break

        improved = False
        t_trial = t

                conv = conv_trial
                current_obj = obj_trial

        print(f"    LP done: {it} iters, {time.monotonic()-t0:.1f}s, best={best_obj:.12f}")

    return best_f, best_obj

def coordinate_bisection(a0, time_limit, max_passes=300, verbose=False):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    t0 = time.monotonic()
    passes = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(45):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))

                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-13:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-13:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue

            conv = fast_convolve(a)
            obj = conv.max()

            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            stagnant += 1
            if stagnant > 3:
                if passes > 5:
                    rng = np.random.default_rng(passes * 47 + 11)
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
                    conv = fast_convolve(a)
                    stagnant = 0
                else:
                    break
        else:
            stagnant = 0

        if verbose and passes % 20 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.12f}")

    if verbose:
        print(f"    CD done: {passes} passes, {time.monotonic()-t0:.1f}s, best={best_obj:.12f}")

    return best_a, best_obj


def peak_directed_reduction(a0, time_limit, verbose=False):
    """
    Heuristic: identify peak-contributing positions and reduce them,
    redistributing mass to low-contribution positions.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        current_obj = conv.max()

        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()

        # Find top peaks
        top_k = min(10, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        max_peak = conv[peak_indices[0]]

        # Compute contribution score for each position
        # Position i contributes to conv[k] via a[i]*a[k-i]
        contrib = np.zeros(n)
        for k in peak_indices:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                other = k - i
                if 0 <= other < n:
                    contrib[i] += a[i] * a[other]

        # Reduce high-contribution positions, increase low-contribution
        # Use softmax weighting
        top_contrib_idx = np.where(contrib > np.median(contrib))[0]
        bot_contrib_idx = np.where(contrib <= np.median(contrib))[0]

        if len(top_contrib_idx) == 0 or len(bot_contrib_idx) == 0:
            break

        # Transfer a small fraction
        total_top = a[top_contrib_idx].sum()
        total_bot = a[bot_contrib_idx].sum()

        if total_top < 1e-10:
            break

        transfer_frac = 0.02
        transfer_amount = total_top * transfer_frac

        # Proportionally reduce top
        a[top_contrib_idx] -= a[top_contrib_idx] * transfer_frac

        # Proportionally increase bottom
        if total_bot > 1e-10:
            a[bot_contrib_idx] += a[bot_contrib_idx] * (transfer_amount / total_bot)

        a = np.maximum(a, 0.0)
        s_check = a.sum()
        if s_check > 1e-15:
            a /= s_check
        else:
            a = best_a.copy()
            break

        new_conv = fast_convolve(a)
        new_obj = new_conv.max()

        if new_obj >= current_obj + 1e-8:
            # Revert and try smaller step
            a = best_a.copy()
            break

        if verbose and it % 50 == 0:
            print(f"    PD iter {it}: obj={best_obj:.12f}")

    if verbose:
        print(f"    PD done: {it} iters, {time.monotonic()-t0:.1f}s, best={best_obj:.12f}")

    return best_a, best_obj


def symmetrize(a):
    """Make sequence symmetric: a[i] = (a[i] + a[n-1-i])/2."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def subgradient_polish(a0, time_limit, lr=0.005, mom=0.93, top_k=30,
                       temp=0.0005, decay=0.9998):
    """Multi-peak subgradient descent with momentum for fine polishing."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        cur_lr *= decay
        cur_temp *= 1.003

        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 5)
            noise_scale = 0.001 + 0.001 * min(no_improve // 100, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0

    return best_a, best_obj

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_try)})")

    # === Phase 1: LP descent v1 on parent ===
    print(f"\n=== Phase 1: LP descent v1 (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 900
    if rem > 60:
        t_alloc = min(rem, 400)
        print(f"  LP v1, t={t_alloc:.0f}s")
        a_opt, obj = lp_descent(parent, time_limit=t_alloc, t_init=0.1, verbose=True)
        update_best(a_opt, "LP1_parent")

    # === Phase 2: LP descent v2 on best ===
    print(f"\n=== Phase 2: LP descent v2 (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 700
        if rem > 40:
            t_alloc = min(rem, 200)
            print(f"  LP v2, t={t_alloc:.0f}s")
            a_opt, obj = lp_descent_v2(best_a, time_limit=t_alloc, t_init=0.05, verbose=True)
            update_best(a_opt, "LP2")

    # === Phase 3: CD polish ===
    print(f"\n=== Phase 3: CD polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  CD, t={t_alloc:.0f}s")
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "CD_polish")

    # === Phase 4: Peak-directed + LP ===
    print(f"\n=== Phase 4: Peak-directed (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_alloc = min(rem, 60)
            print(f"  Peak-directed, t={t_alloc:.0f}s")
            a_opt, obj = peak_directed_reduction(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "peak_dir")

            # Then LP from there
            rem2 = deadline - time.monotonic() - 300
            if rem2 > 30:
                t_alloc2 = min(rem2, 120)
                a_opt2, obj2 = lp_descent(a_opt, time_limit=t_alloc2, t_init=0.08, verbose=True)
                update_best(a_opt2, "peak_dir_LP")

    # === Phase 5: Symmetric LP ===
    print(f"\n=== Phase 5: Symmetric (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            a_sym = symmetrize(best_a)
            t_alloc = min(rem, 100)
            print(f"  Symmetric LP, t={t_alloc:.0f}s")
            a_opt, obj = lp_descent(a_sym, time_limit=t_alloc, t_init=0.1, verbose=True)
            
            rem2 = deadline - time.monotonic() - 250
            if rem2 > 15:
                t_alloc2 = min(rem2, 50)
                a_opt, obj = coordinate_bisection(a_opt, time_limit=t_alloc2)
            update_best(a_opt, "sym_LP")

    # === Phase 6: Multi-length ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    target_lengths = [640, 896, 1024]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.35, 80)
        print(f"  n={target_n}, LP t={t_alloc:.0f}s")
        a_opt, obj = lp_descent(f_interp, time_limit=t_alloc, t_init=0.1, verbose=False)

        rem2 = deadline - time.monotonic() - 200
        if rem2 > 15:
            t_alloc2 = min(rem2 * 0.3, 30)
            a_opt, obj = coordinate_bisection(a_opt, time_limit=t_alloc2)
        update_best(a_opt, f"n{target_n}")

    # === Phase 7: Multi-restart ===
    print(f"\n=== Phase 7: Multi-restart (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        restart_configs = []

        for ns in [0.008, 0.02, 0.04]:
            r_rng = np.random.default_rng(seed + int(ns * 10000))
            f_r = project_simplex(best_a + r_rng.standard_normal(n_best) * ns)
            restart_configs.append((f"noise{int(ns*1000)}", f_r))

        restart_configs.append(("flip", project_simplex(0.7 * best_a + 0.3 * best_a[::-1])))
        restart_configs.append(("flat", project_simplex(0.85 * best_a + 0.15 * np.ones(n_best) / n_best)))

        for idx, (label, f_init) in enumerate(restart_configs):
            rem = deadline - time.monotonic() - 80 + idx * 20
            if rem < 15:
                break
            t_alloc = min(rem, 60)
            print(f"  Restart {label}, t={t_alloc:.0f}s")
            a_opt, obj = lp_descent(f_init, time_limit=t_alloc, t_init=0.1, verbose=False)

            rem2 = deadline - time.monotonic() - 80
            if rem2 > 10:
                t_alloc2 = min(rem2, 15)
                a_opt, obj = coordinate_bisection(a_opt, time_limit=t_alloc2)

            update_best(a_opt, f"rs_{label}")

    # === Phase 8: Final intensive polish ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 100)
            print(f"  Final CD, t={t_alloc:.0f}s")
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "final_CD")

            rem = deadline - time.monotonic() - 15
            if rem > 10:
                t_alloc = min(rem, 50)
                print(f"  Final LP, t={t_alloc:.0f}s")
                a_opt, obj = lp_descent(best_a, time_limit=t_alloc, t_init=0.02, verbose=True)
                update_best(a_opt, "final_LP")

            rem = deadline - time.monotonic() - 15
            if rem > 8:
                t_alloc = min(rem, 20)
                print(f"  Final SG, t={t_alloc:.0f}s")
                a_opt, obj = subgradient_polish(best_a, time_limit=t_alloc,
                                                lr=0.003, mom=0.94,
                                                top_k=max(20, len(best_a) // 6),
                                                temp=0.0002)
                update_best(a_opt, "final_SG")

        # Tiny perturbations + CD
        rem = deadline - time.monotonic() - 10
        if rem > 6 and best_a is not None:
            n_best = len(best_a)
            for p in range(5):
                rem = deadline - time.monotonic() - 8
                if rem < 4:
                    break
                rng_p = np.random.default_rng(seed + 9000 + p * 13)
                f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.0004)
                t_alloc = min(rem, 4)
                a_opt, obj = coordinate_bisection(f_pert, time_limit=t_alloc)
                update_best(a_opt, f"pert_{p}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
