"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a multi-strategy approach:
1. Multi-start exploration with structured and random initializations
2. Nesterov-accelerated subgradient descent with adaptive softmax weighting

4. Peak-flattening coordinate descent
5. Adaptive dimension tuning

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def nesterov_sg_optimize(f0, time_limit, lr=0.003, momentum=0.93, top_k=50, 
                          temp=0.00005, verbose=False):
    """Nesterov-accelerated subgradient with adaptive softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()  # Nesterov momentum variable
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        y = f + momentum * v
        
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient at y
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * y[kg[mask]]

        # Gradient step on y
        y = y - cur_lr * sg
        
        # Project
        f_new = project_simplex(y)
        
        # Update momentum
        v = f_new - f
        
        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    NSG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        f = f_new
        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0005)
            no_improve = 0

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            # Fallback: try max-sum
            c = -np.ones(n)
            A_ub = np.zeros((2 * n - 1, n))
            for j in range(n):
                A_ub[j:j + n, j] = f
            b_ub = np.full(2 * n - 1, M * 1.1)
            bounds = [(0, None)] * n
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                g = res.x
                gs = g.sum()
                g = g / gs if gs > 1e-15 else np.ones(n) / n
            else:
                g = np.ones(n) / n
        else:
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Backtracking line search
        alpha = 1.0
        best_val = fast_convolve(f).max()
        best_alpha = 0.0
        
        for _ in range(20):
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_obj - alpha * 0.1 * (M - lp_t) if lp_t is not None else val < best_obj:
                best_val = val
                best_alpha = alpha
                break
            alpha *= 0.5

        if best_alpha < 1e-10:
            # Try a small fixed step
            best_alpha = 0.1
            ft = (1 - best_alpha) * f + best_alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st > 1e-15:
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_obj:
                    best_val = val
                else:
                    best_alpha = 0.0

        if best_alpha > 1e-12:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            sn = f.sum()
            if sn < 1e-15:
                f = np.ones(n) / n
            else:
                f = f / sn
            M = fast_convolve(f).max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.003 + 0.002 * min(no_improve // 50, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj


def peak_flatten_cd(f0, time_limit, verbose=False):
    """Coordinate descent that directly targets reducing peak contributions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Find the global peak
        pk = int(np.argmax(conv))
        
        # Also find second and third peaks for diversity
        conv_sorted = np.sort(conv)[::-1]
        if len(conv) >= 3:
            threshold2 = conv_sorted[1]
            threshold3 = conv_sorted[2]
        else:
            threshold2 = threshold3 = 0

        # Get all peaks above 95% of max
        peak_thresh = best_obj * 0.95
        peak_positions = np.where(conv >= peak_thresh)[0]
        
        if len(peak_positions) == 0:
            peak_positions = np.array([pk])

        for peak_k in peak_positions[:min(5, len(peak_positions))]:
            peak_k = int(peak_k)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # For each coordinate in the peak's support, try to reduce it
            # while maintaining sum = 1
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break

                old_val = f[j]
                if old_val < 1e-14:
                    continue

                # Try reducing this coordinate and spreading to others
                best_j_obj = best_obj
                best_mult = 1.0
                
                for mult in [0.5, 0.7, 0.8, 0.9, 1.1, 1.3]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_mult = mult

                if best_mult != 1.0:
                    new_val = old_val * best_mult
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - new_val) / denom
                        f[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                        if best_j_obj < best_obj - 1e-14:
                            best_obj = best_j_obj
                            best_f = f.copy()
                            improved = True
                            conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 5:
                # Random perturbation
                rng = np.random.default_rng(passes * 7919 + 3)
                for _ in range(10):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.5, 0.8, 1.2, 1.5]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.5:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break
                    if improved:
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = int(rng.integers(0, 5))
    if choice == 0:
        # Cosine window
        f = np.cos(np.pi * np.linspace(0, 1, n)) ** 2
        f = np.maximum(f, 0)
    elif choice == 1:
        # Hamming window
        f = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
        f = np.maximum(f, 0)
    elif choice == 2:
        # Gaussian
        x = np.linspace(-3, 3, n)
        f = np.exp(-x ** 2 / 2)
    elif choice == 3:
        # Alternating with decay
        f = np.ones(n) * (1 - 0.3 * np.cos(2 * np.pi * np.arange(n) * 3 / n))
        f = np.maximum(f, 0)
    else:
        # Random smooth
        f = np.ones(n)
        freq = rng.integers(2, 8)
        f += 0.3 * np.cos(2 * np.pi * np.arange(n) * freq / n + rng.uniform(0, 2 * np.pi))
        f = np.maximum(f, 0)
    
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n
    return f

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated subgradient,
LP-based Frank-Wolfe, peak-equalization coordinate descent, and dimension tuning.
Key insight: optimal non-negative sequences have autocorrelation peaks away from lag 0,
achieving flatter autocorrelation than the uniform distribution.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def nesterov_sg_optimize(f0, time_limit, lr=0.003, momentum=0.93, top_k=50,

    n = len(f)

    v = np.zeros(n)

    best_f = f.copy()

    no_improve = 0

    cur_temp = temp
    it = 0

        y = f + momentum * v

        scaled -= scaled.max()
        w = np.exp(scaled)

                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            sn = f.sum()
            if sn < 1e-15:
                f = np.ones(n) / n
            else:
                f = f / sn
            M = fast_convolve(f).max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.003 + 0.002 * min(no_improve // 40, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj


def peak_equality_cd(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak equalization across top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        peak_thresh = best_obj * 0.97
        peak_positions = np.where(conv >= peak_thresh)[0]

        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        for peak_k in peak_positions[:min(4, len(peak_positions))]:
            peak_k = int(peak_k)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Sort by contribution to this peak
            contributions = []
            for j in range(js, je + 1):
                idx_other = peak_k - j
                if 0 <= idx_other < n:
                    contributions.append((f[j] * f[idx_other], j))
            contributions.sort(reverse=True)

            for _, j in contributions[:min(10, len(contributions))]:
                if time.monotonic() - t0 > time_limit:
                    break

                old_val = f[j]
                if old_val < 1e-14:
                    continue

                best_j_obj = best_obj
                best_mult = 1.0

                for mult in [0.5, 0.7, 0.85, 0.95, 1.1, 1.3, 1.5]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_mult = mult

                if best_mult != 1.0:
                    new_val = old_val * best_mult
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - new_val) / denom
                        f[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                        if best_j_obj < best_obj - 1e-14:
                            best_obj = best_j_obj
                            best_f = f.copy()
                            improved = True
                            conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 4:
                rng = np.random.default_rng(passes * 7919 + 3)
                for _ in range(15):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.4, 0.6, 1.4, 1.8]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.5:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break
                    if improved:
                        break
            if no_improve > 12:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0, 1, n)
        f = np.cos(np.pi * (x - 0.5)) ** 2
        f = np.maximum(f, 0)
    elif choice == 1:
        x = np.linspace(-3, 3, n)
        f = np.exp(-x ** 2 / 2)
    elif choice == 2:
        f = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) * 2.0 / n)
        f = np.maximum(f, 0)
    elif choice == 3:
        f = 1.0 + 0.4 * np.sin(2 * np.pi * np.arange(n) * 5.0 / n + rng.uniform(0, 6.28))
        f = np.maximum(f, 0.01)
    elif choice == 4:
        # Two-scale pattern
        coarse = np.ones(n // 4 + 1) * (1 + 0.3 * np.cos(2 * np.pi * np.arange(n // 4 + 1) / max(1, n // 16)))
        coarse = np.maximum(coarse, 0.1)
        f = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(coarse)), coarse)
        f = np.maximum(f, 0)
    else:
        f = np.ones(n)
        f += 0.3 * np.cos(2 * np.pi * np.arange(n) * rng.integers(2, 10) / n)
        f += 0.2 * np.cos(2 * np.pi * np.arange(n) * rng.integers(3, 15) / n + rng.uniform(0, 6.28))
        f = np.maximum(f, 0.01)

    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n
    return f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: Nesterov subgradient, LP-FW, peak-equality CD,
    structured initializations, and adaptive dimension tuning.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(832) / 832
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Quick exploration with structured inits ===
    print(f"\n=== Phase 1: Exploration (best={best_score:.8f}) ===")
    t1_total = min(250, (deadline - time.monotonic()) * 0.15)
    t_per = t1_total / 5.0

    for i in range(5):
        if time.monotonic() > deadline - 600:
            break
        f_init = structured_init(base_n, rng)
        a_opt, _ = nesterov_sg_optimize(f_init, time_limit=t_per,
                                         lr=0.005, momentum=0.91,
                                         top_k=max(40, base_n // 5),
                                         temp=0.0001, temp_growth=1.005)
        try_seq(a_opt, f"struct_{i}")

    # Also try a few different lengths
    for target_n in [base_n - 64, base_n + 64, base_n + 128]:
        if time.monotonic() > deadline - 500:
            break
        if target_n < 256 or target_n > 2048:
            continue
        f_init = interpolate_sequence(parent, target_n)
        t_alloc = min(40, (deadline - time.monotonic()) * 0.03)
        if t_alloc < 15:
            continue
        a_opt, _ = nesterov_sg_optimize(f_init, time_limit=t_alloc,
                                         lr=0.004, momentum=0.92,
                                         top_k=max(30, target_n // 6),
                                         temp=0.00008)
        try_seq(a_opt, f"expl_n{target_n}")

    # === Phase 2: Intensive Nesterov SG on best ===
    print(f"\n=== Phase 2: Intensive NSG (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.22, 380)
    if t2 > 40 and best_seq is not None:
        n_best = len(best_seq)
        print(f"  n={n_best}, t={t2:.0f}s")
        a_opt, _ = nesterov_sg_optimize(best_seq, time_limit=t2,
                                         lr=0.0018, momentum=0.94,
                                         top_k=max(50, n_best // 4),
                                         temp=0.00003, temp_growth=1.002,
                                         verbose=True)
        try_seq(a_opt, "nsg_main")

    # Second NSG run with different params
    rem = deadline - time.monotonic()
    t2b = min(rem * 0.12, 200)
    if t2b > 30 and best_seq is not None:
        n_best = len(best_seq)
        a_opt, _ = nesterov_sg_optimize(best_seq, time_limit=t2b,
                                         lr=0.001, momentum=0.95,
                                         top_k=max(30, n_best // 6),
                                         temp=0.000015, temp_growth=1.004)
        try_seq(a_opt, "nsg_fine")

    # === Phase 3: LP-FW ===
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.18, 280)
    if t3 > 30 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t3:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
        try_seq(a_opt, "lpfw")

    # === Phase 4: Peak-equality CD ===
    print(f"\n=== Phase 4: Peak CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.18, 280)
    if t4 > 25 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t4:.0f}s")
        a_opt, _ = peak_equality_cd(best_seq, time_limit=t4)
        try_seq(a_opt, "peak_cd")

    # === Phase 5: Dimension tuning ===
    print(f"\n=== Phase 5: Dimension tuning (best={best_score:.8f}) ===")
    if best_seq is not None:
        current_n = len(best_seq)
        candidate_lengths = []
        for delta in [-96, -64, -32, 32, 64, 96, 128, 192, 256]:
            ln = current_n + delta
            if 384 <= ln <= 2048 and ln != current_n:
                candidate_lengths.append(ln)
        candidate_lengths = sorted(set(candidate_lengths))

        for target_n in candidate_lengths[:4]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.15, 120)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            print(f"  NSG n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_sg_optimize(f_interp, time_limit=t_alloc,
                                              lr=0.0015, momentum=0.94,
                                              top_k=max(35, target_n // 5),
                                              temp=0.00002)
            try_seq(a_opt, f"dim_{target_n}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine NSG
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_polish = min(rem * 0.3, 80)
            a_opt, _ = nesterov_sg_optimize(best_seq, time_limit=t_polish,
                                              lr=0.0005, momentum=0.95,
                                              top_k=max(25, n_best // 7),
                                              temp=0.00001, temp_growth=1.005)
            try_seq(a_opt, "polish_nsg")

        # Quick CD
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_cd = min(rem * 0.3, 60)
            a_opt, _ = peak_equality_cd(best_seq, time_limit=t_cd)
            try_seq(a_opt, "polish_cd")

        # Tiny perturbations
        for p in range(5):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.0003 + 0.0002 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = nesterov_sg_optimize(f_pert, time_limit=min(rem, 18),
                                              lr=0.0003, momentum=0.95,
                                              top_k=max(15, n_best // 10),
                                              temp=0.000005)
            try_seq(a_opt, f"perturb_{p}")

        # One more LP-FW on the polished result
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "lpfw_polish")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
