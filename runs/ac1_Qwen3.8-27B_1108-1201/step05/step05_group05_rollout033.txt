import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_corr(w, a):
    """Cross-correlation of w (length 2n-1) with a (length n) via FFT.
    Returns array of length n where result[p] = sum_m w[p+m]*a[m]
    for m=0..n-1 (clamped to valid range)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Cross-correlation: ifft(conj(W) * A)
    corr = np.fft.irfft(np.conj(W) * A, fft_size)
    return corr[:n]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_safe(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def solve_lp_direction(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    if active_k is None:
        K = 2 * n - 1
        A_ub = np.zeros((K, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(K, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    f_interp = np.maximum(f_interp, 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy combining:
1. Soft-max smoothed gradient descent with FFT-based cross-correlation gradients
2. LP-Frank-Wolfe with adaptive constraint selection
3. Multi-scale coordinate descent
4. Frequency-domain informed initialization
5. Multi-length exploration (1024, 1536)
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Compute cross-correlation (w ⊛ a)[p] = sum_m w[p+m]*a[m] for p=0..n-1 via FFT."""
    n = len(a)
    # Need FFT size >= len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(W) * A, fft_size)
    return corr[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_safe(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def softmax_gd(f0, time_limit, verbose=False, tau_start=3.0, tau_end=800.0,
               lr_start=0.015, lr_end=0.00005, momentum_start=0.85, momentum_end=0.95):
    """Soft-max smoothed gradient descent with adaptive temperature and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    no_improve = 0
    iteration = 0
    total_budget = max(time_limit, 1.0)

    while time.monotonic() - t0 < t0 + total_budget:
        elapsed = time.monotonic() - t0
        progress = min(elapsed / total_budget, 1.0)

        # Animate temperature (geometric schedule)
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)
        # Learning rate decay
        lr = lr_start * (lr_end / lr_start) ** (progress ** 0.8)
        # Momentum increases over time
        mom = momentum_start + (momentum_end - momentum_start) * progress

        # Compute soft-max weights over the convolution
        scaled = tau * conv
        w = softmax_safe(scaled)

        # Compute gradient: grad[p] = 2 * sum_k w[k] * a[k-p] (for valid k-p)
        # = 2 * (w cross-corr a)[p]
        grad = 2.0 * fast_cross_corr(w, f)

        # Momentum update
        v = mom * v - lr * grad
        f = project_simplex(f + v)

        # Compute new objective
        conv = fast_convolve(f)
        obj = conv.max()
        iteration += 1

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and iteration % 200 == 0:
                print(f"    SoftMax-GD iter {iteration}: obj={best_obj:.10f}, tau={tau:.1f}")
        else:
            no_improve += 1

        if no_improve > 300 and iteration > 50:
            rng = np.random.default_rng(iteration * 7919 + 13)
            noise_scale = 0.001 + 0.003 * min(no_improve // 200, 4)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj


def solve_lp_direction(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_set, g >= 0."""
    c = -np.ones(n)
    if active_k is None:
        K = 2 * n - 1
        A_ub = np.zeros((K, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(K, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, top_k_constraints=60):
    """LP-based Frank-Wolfe with adaptive constraints and bisection line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptive constraint selection
        active_k = None
        if top_k_constraints > 0 and top_k_constraints < 2 * n - 1 and it > 5:
            k_sorted = np.argsort(conv)[::-1][:top_k_constraints]
            active_k = k_sorted.tolist()
        # Periodically use all constraints
        if it % 200 == 0:
            active_k = None

        g = solve_lp_direction(f, M * m_exp, n, active_k=active_k)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 100:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Bisection line search
        lo = 0.0
        hi = 1.0
        obj_hi = fast_convolve(np.maximum(g, 0)).max()
        obj_lo = M
        for _ in range(18):
            mid = (lo + hi) / 2.0
            f_mid = np.maximum(f + mid * (g - f), 0)
            f_mid = f_mid / f_mid.sum()
            obj_mid = fast_convolve(f_mid).max()
            if obj_mid < obj_lo:
                lo = mid
                obj_lo = obj_mid
            else:
                hi = mid
        alpha = lo

        if alpha > 1e-8:
            f_new = np.maximum(f + alpha * (g - f), 0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M_new = conv.max()

            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                M = M_new
                m_exp = 1.0
                if verbose and it % 30 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
            f = f_new
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 70:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(no_improve // 100, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coord_descent(f0, time_limit):
    """Coordinate descent with multiplier-based local search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 2:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def block_coord_descent(f0, time_limit, block_size=8):
    """Block coordinate descent: optimize groups of coordinates together via small LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                break
            end = min(start + block_size, n)
            block_indices = list(range(start, end))
            old_block = f[block_indices].copy()
            rest_sum = 1.0 - old_block.sum()
            if rest_sum < 1e-10:
                continue

            # Try scaling the entire block
            best_scale = 1.0
            best_obj_block = best_obj
            for scale in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                new_block_sum = old_block.sum() * scale
                if new_block_sum >= 1.0 or new_block_sum < 0:
                    continue
                rest_scale = (1.0 - new_block_sum) / rest_sum
                if rest_scale < 0:
                    continue
                f_trial = f.copy()
                f_trial[block_indices] = old_block * scale
                mask = np.ones(n, dtype=bool)
                mask[block_indices] = False
                f_trial[mask] *= rest_scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj_block - 1e-15:
                    best_obj_block = val
                    best_scale = scale

            if best_scale != 1.0:
                new_block_sum = old_block.sum() * best_scale
                rest_scale = (1.0 - new_block_sum) / rest_sum
                f[block_indices] = old_block * best_scale
                mask = np.ones(n, dtype=bool)
                mask[block_indices] = False
                f[mask] *= rest_scale
                f = f / f.sum()
                if best_obj_block < best_obj - 1e-14:
                    best_obj = best_obj_block
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0

    return best_f, best_obj

            print(f"  Time: {t7:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t7,
                                      verbose=True, top_k_constraints=0)
            try_seq(a_opt, "lpfw_full")

            rem = deadline - time.monotonic()
            if rem > 30:
                # Final coordinate descent
                t7b = min(60, rem * 0.5)
                a_opt, _ = coord_descent(a_opt, time_limit=t7b)
                try_seq(a_opt, "final_cd")

    # === Phase 8: Diverse restarts on best ===
    print(f"\n=== Phase 8: Diverse restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 80:
            restarts = [
                ("noise1", project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)),
                ("noise2", project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)),
                ("rev_blend", project_simplex(0.7 * best_seq + 0.3 * best_seq[::-1])),
                ("flat_blend", project_simplex(0.85 * best_seq + 0.15 * np.ones(n_best) / n_best)),
            ]
            for i, (name, f_init) in enumerate(restarts):
                rem = deadline - time.monotonic()
                if rem < 30:
                    break
                t_alloc = min(60, rem / max(1, len(restarts) - i))
                print(f"  Restart [{name}], Time: {t_alloc:.0f}s")
                a_opt, _ = softmax_gd(f_init, time_limit=t_alloc,
                                      tau_start=10.0, tau_end=500.0,
                                      lr_start=0.008, lr_end=0.00002)
                try_seq(a_opt, f"restart_{name}")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
