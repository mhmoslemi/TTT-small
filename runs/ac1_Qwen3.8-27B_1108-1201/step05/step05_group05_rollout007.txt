import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_grad(f, tau):
    """
    Compute gradient of (1/tau)*log(sum(exp(tau*(f*f)[k]))) w.r.t. f.
    Returns (gradient, smooth_max_value).
    """
    n = len(f)
    conv = fast_convolve(f)
    L = len(conv)  # 2n-1
    
    # Compute softmax weights
    scaled = tau * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum
    
    smooth_max_val = (scaled.max() + np.log(w_sum)) / tau
    
    # Gradient: grad[j] = 2 * sum_k w[k] * f[k-j] (for valid k-j)
    # This is 2 * cross_correlation(w, f)[j]
    
    # Method: use FFT
    # r[j] = sum_k w[k] * f[k-j] = (w * flip(f))[j] in convolution terms
    # Actually: r[j] = sum_m w[j+m] * f[m] = (w * f_reversed)[j + n - 1]
    
    # Let's use the standard cross-correlation via FFT:
    # corr[j] = sum_k w[k+j] * f[k] for j >= 0
    #         = sum_k w[k] * f[k-j] for j >= 0 (by substitution)
    
    # Using FFT: corr = ifft(conj(FFT(w)) * FFT(f)) ... but this gives circular
    # For linear cross-correlation, we need FFT size >= L + n - 1
    
    fft_size = 1 << (L + n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    
    # Circular cross-correlation: ifft(conj(W) * F)
    # This gives: sum_k w[k] * f[k-j] (circular)
    # For linear, we need fft_size >= L + n - 1
    cc = np.fft.irfft(np.conj(W) * F, fft_size)

"""
Multi-phase optimizer for minimizing 2*n*max(autoconv)/(sum)^2.
Combines smooth-max Adam gradient (fast, many iterations), LP-Frank-Wolfe
(exact KKT polishing), adaptive coordinate descent, and diverse restarts
with spectral and structural initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_grad(f, tau):
    n = len(f)
    conv = fast_convolve(f)
    L = len(conv)
    scaled = tau * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w /= w_sum
    smooth_val = (scaled.max() + np.log(w_sum)) / tau
    fft_size = 1 << (L + n - 2).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    cc = np.fft.irfft(np.conj(W) * F, fft_size)
    grad = 2.0 * cc[0:n]
    return grad, smooth_val


def adam_smooth_optimize(f0, time_limit, tau_init=5.0, tau_end=200.0,
                         lr_init=0.003, decay=0.9997, beta1=0.85,
                         beta2=0.999, epsilon=1e-8, verbose=False,
                         restart_interval=500, restart_scale=0.004):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau_init
    lr = lr_init
    best_f = f.copy()
    best_conv = fast_convolve(f)
    best_obj = best_conv.max()
    t0 = time.monotonic()
    no_improve = 0
    conv = best_conv.copy()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, _ = smooth_max_and_grad(f, tau)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        f_new = project_simplex(f_new)
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()
        f = f_new
        conv = conv_new
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and t % 100 == 0:
                print(f"      Adam t={t}: obj={best_obj:.10f}, tau={tau:.2f}")
        else:
            no_improve += 1
        tau = tau_init + (tau_end - tau_init) * min(1.0, t / (2.0 * restart_interval))
        lr = lr_init * (decay ** t)
        if no_improve > restart_interval:
            rng = np.random.default_rng(t * 7919)
            f = project_simplex(best_f + rng.standard_normal(n) * restart_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = lr_init * 0.5
            tau = tau_init * 2
            no_improve = 0
            conv = fast_convolve(f)
    return best_f, best_obj


def solve_lp_direction(f, M, n, active_k=None):
    c = -np.ones(n)
    K = len(active_k) if active_k is not None else (2 * n - 1)
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)
    if active_k is None:
        for j in range(n):
            A_ub[j:j + n, j] = f
    else:
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None


def lpfw_optimize(f0, time_limit, top_k=40, verbose=False, use_full_every=50):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % use_full_every == 0:
            active_k = None
        else:
            k_sorted = np.argsort(conv)[::-1][:top_k]
            active_k = k_sorted.tolist()
        g = solve_lp_direction(f, M * m_exp, n, active_k=active_k)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
                active_k = None
                g = solve_lp_direction(f, M, n, active_k=None)
                if g is None:
                    break
            else:
                continue
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 60:
                break
            m_exp = min(m_exp * 1.06, 4.0)
            continue
        best_alpha = 0.0
        best_alpha_obj = best_obj
        for alpha in [0.1, 0.25, 0.4, 0.5, 0.65, 0.8, 0.9, 1.0]:
            f_trial = f + alpha * (g - f)
            f_trial = np.maximum(f_trial, 0.0)
            s2 = f_trial.sum()
            if s2 < 1e-15:
                continue
            f_trial = f_trial / s2
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_alpha_obj - 1e-14:
                best_alpha_obj = obj_trial
                best_alpha = alpha
        if best_alpha > 0:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M_new = conv.max()
            f = f_new
            if M_new < best_obj - 1e-13:
                best_obj = M_new
                best_f = f.copy()
                no_improve = 0
                M = M_new
                m_exp = 1.0
                if verbose and it % 20 == 0:
                    print(f"      LP-FW it={it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)
        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(no_improve // 100, 8)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    return best_f, best_obj


def coord_desc(f0, time_limit, n_sweep=8):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-13:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 2:
                break
        else:
            no_improve_passes = 0
    return best_f, best_obj


def spectral_init(n, seed=0, chirp_order=1):
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    t = np.arange(fft_size)
    phase = 2.0 * np.pi * chirp_order * t * t / fft_size
    spectrum = np.exp(1j * phase)
    extra_phase = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum[:fft_size // 2 + 1] *= np.exp(1j * extra_phase[:fft_size // 2 + 1])
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def gaussian_init(n, center_frac=0.5, width_frac=0.3):
    x = np.arange(n)
    center = center_frac * n
    width = width_frac * n
    f = np.exp(-0.5 * ((x - center) / width) ** 2)
    f = np.maximum(f, 0)
    ss = f.sum()
    if ss > 1e-10:
        f /= ss
    else:
        f = np.ones(n) / n
    return f


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_palindromic(a, n):
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def swanneal_polish(f0, time_limit, n_sweeps=5, temp=0.002):

    t0 = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        cur_temp *= 0.7
        for _ in range(n):
            if time.monotonic() - t0 > time_limit:

            j = rng.integers(0, n)
            delta = rng.standard_normal() * cur_temp

            f_trial[j] = max(0.0, f[j] + delta)
            ss = f_trial.sum()
            if ss < 1e-10:

            f_trial /= ss
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_obj:

            if rng.random() < accept_prob:
                f = f_trial
                if obj_trial < best_obj - 1e-14:

    return best_f, best_obj

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    n_parent = len(parent)

    # Phase 1: Fast Adam smooth-max on parent
    print("=== Phase 1: Adam smooth-max on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(200, rem * 0.2)
    if t1 > 20:
        print(f"  t={t1:.0f}s")
        a_opt, _ = adam_smooth_optimize(parent, time_limit=t1, tau_init=5.0,
                                        tau_end=150.0, lr_init=0.004,
                                        decay=0.9996, restart_interval=400)
        try_seq(a_opt, "adam_parent")

    # Phase 2: LP-FW on best so far
    print("=== Phase 2: LP-FW polishing ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(250, rem * 0.25)
        if t2 > 30:
            print(f"  t={t2:.0f}s")
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t2, top_k=50,
                                     verbose=True, use_full_every=40)
            try_seq(a_opt, "lpfw")

    # Phase 3: Palindromic LP-FW
    print("=== Phase 3: Palindromic LP-FW ===")
    if best_seq is not None:
        n_best = len(best_seq)
        half_n = (n_best + 1) // 2
        pal_half = (best_seq[:half_n] + best_seq[n_best - 1: n_best - half_n - 1: -1]) / 2.0
        pal_full = project_simplex(make_palindromic(pal_half, n_best))
        pal_score = try_seq(pal_full, "pal_init")
        rem = deadline - time.monotonic()
        t3 = min(150, rem * 0.15)
        if t3 > 20:
            a_opt, _ = lpfw_optimize(pal_full, time_limit=t3, top_k=40)
            try_seq(a_opt, "lpfw_pal")

    # Phase 4: Different lengths with spectral init
    print("=== Phase 4: Multi-length spectral init ===")
    base = best_seq if best_seq is not None else parent
    for target_n in [640, 768, 896, 1024]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        f_interp = interpolate_seq(base, target_n)
        f_spec = spectral_init(target_n, seed=target_n, chirp_order=1)
        f_mix = project_simplex(0.6 * f_interp + 0.4 * f_spec)
        t_alloc = min(rem * 0.18, 120)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = adam_smooth_optimize(f_mix, time_limit=min(t_alloc * 0.5, 60),
                                        tau_init=3.0, tau_end=100.0, lr_init=0.003)
        try_seq(a_opt, f"adam_spec_n{target_n}")
        rem = deadline - time.monotonic()
        if rem > 40:
            a_opt2, _ = lpfw_optimize(a_opt, time_limit=min(rem * 0.3, 80), top_k=40)
            try_seq(a_opt2, f"lpfw_n{target_n}")

    # Phase 5: Diversified restarts on best
    print(f"=== Phase 5: Diversified restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        restarts = [
            ("noise_small", project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)),
            ("noise_med", project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)),
            ("rev_sym", project_simplex(0.7 * best_seq + 0.3 * best_seq[::-1])),
            ("gauss_blend", project_simplex(0.85 * best_seq + 0.15 * gaussian_init(n_best))),
            ("flat_blend", project_simplex(0.9 * best_seq + 0.1 * np.ones(n_best) / n_best)),
        ]
        for name, f_init in restarts:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem / max(1, len(restarts)), 120)
            print(f"  [{name}] t={t_alloc:.0f}s")
            a_opt, _ = adam_smooth_optimize(f_init, time_limit=min(t_alloc * 0.55, 60),
                                            tau_init=8.0, tau_end=200.0,
                                            lr_init=0.002, decay=0.9995)
            try_seq(a_opt, f"adam_{name}")
            rem = deadline - time.monotonic()
            if rem > 30:
                a_opt2, _ = lpfw_optimize(a_opt, time_limit=min(rem * 0.35, 60), top_k=45)
                try_seq(a_opt2, f"lpfw_{name}")

    # Phase 6: Coordinate descent
    print(f"=== Phase 6: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_cd = min(rem * 0.3, 200)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coord_desc(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")
            rem = deadline - time.monotonic()
            if rem > 30:
                a_opt2, _ = coord_desc(a_opt, time_limit=min(rem * 0.3, 100))
                try_seq(a_opt2, "cd2")

    # Phase 7: Simulated annealing polish
    print(f"=== Phase 7: Simulated Annealing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_sa = min(rem * 0.2, 100)
            a_opt, _ = swanneal_polish(best_seq, time_limit=t_sa,
                                       n_sweeps=6, temp=0.003)
            try_seq(a_opt, "sa")

    # Phase 8: Final deep LP-FW with smaller top_k for precision
    print(f"=== Phase 8: Final Deep LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_final = min(rem * 0.5, 200)
            print(f"  Final LP-FW t={t_final:.0f}s")
            a_opt, _ = lpfw_optimize(best_seq, time_limit=t_final, top_k=25,
                                     use_full_every=30, verbose=True)
            try_seq(a_opt, "final_lpfw")
            rem = deadline - time.monotonic()
            if rem > 20:
                a_opt2, _ = lpfw_optimize(a_opt, time_limit=min(rem * 0.4, 100),
                                          top_k=20, use_full_every=25)
                try_seq(a_opt2, "final_lpfw2")

    # Phase 9: Tiny final Adam + CD
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            a_opt, _ = adam_smooth_optimize(best_seq, time_limit=min(rem * 0.3, 40),
                                            tau_init=50.0, tau_end=300.0,
                                            lr_init=0.001, decay=0.999)
            try_seq(a_opt, "final_adam")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_opt2, _ = coord_desc(a_opt, time_limit=min(rem * 0.5, 30))
            try_seq(a_opt2, "final_cd")

    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
