"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with log-sum-exp smoothing
in softmax parameterization, combined with temperature annealing, multi-length
exploration, and local polishing via pair-exchange and coordinate descent.
Key innovation: replaces slow subgradient methods with a smooth approximation
that allows efficient quasi-Newton optimization.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def lse_obj_grad(z, T):
    """Log-sum-exp smoothed objective and gradient w.r.t. z (softmax params)."""
    a = softmax_fn(z)
    n = len(a)
    b = fft_convolve(a)

    # Log-sum-exp of b / T
    m = b.max()
    shifted = (b - m) / T
    sm = shifted.max()
    log_se = m + T * (sm + np.log(np.sum(np.exp(shifted - sm))))
    obj = 2.0 * n * log_se

    # Weights w_k = exp(b_k/T) / sum(exp(b_j/T))
    w = np.exp(shifted - sm)
    w /= w.sum()

    # dF/da_j = 2 * sum_k w_k * a_{k-j} = 2 * cross-correlation of w and a
    dF_da = 2.0 * np.correlate(w, a, 'full')[n - 1:2 * n - 1]

    # Gradient w.r.t. z via softmax Jacobian
    grad_z = a * dF_da - a * np.dot(a, dF_da)

    return obj, grad_z


def run_lbfgs(f_init, T, maxiter=500, tol=1e-10):
    """Run L-BFGS-B on the smoothed objective in z-space."""
    n = len(f_init)
    a = np.maximum(f_init, 1e-30)
    a = a / a.sum()
    z0 = np.log(a)

    res = minimize(
        lse_obj_grad, z0, args=(T,),
        jac=True, method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-8,
                 'maxls': 50, 'maxfun': 2000}
    )
    a_opt = softmax_fn(res.x)
    a_opt = np.maximum(a_opt, 0.0)
    a_opt /= a_opt.sum()
    return a_opt, res


def run_lbfgs_anneal(f_init, temperatures, maxiter_per_T=300, time_limit=100):

    a = f_init.copy()

        a, res = run_lbfgs(a, T, maxiter=maxiter_per_T)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def pair_exchange_search(f0, time_limit, max_rounds=50):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 1e-10)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=4):

    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)

    inits.append(hann / hann.sum())

    center = n / 2.0
    sigma = n / (4.0 + rng.random() * 4.0)

    r = np.convolve(r, kernel, mode='same')

    # === Phase 5: Multi-seed exploration ===
    print(f"\n=== Phase 5: Multi-Seed (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for s in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            for noise_scale in [0.002, 0.005, 0.01]:
                rem = deadline - time.monotonic() - 8
                if rem < 6:
                    break
                rng_local = np.random.default_rng(seed + s * 100 + int(noise_scale * 1000))
                noise = rng_local.standard_normal(n_best) * noise_scale
                f_pert = project_simplex(best_seq + noise)
                temps = [0.001, 0.0005, 0.0002, 0.0001]
                t_alloc = min(rem * 0.6, 20)
                a_opt = run_lbfgs_anneal(f_pert, temps, maxiter_per_T=100, time_limit=t_alloc)
                try_seq(a_opt, f"seed{s}_ns{noise_scale}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = evaluate_sequence(best_seq.tolist())

    # Final verification
    fs = evaluate_sequence(best_seq.tolist())
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
