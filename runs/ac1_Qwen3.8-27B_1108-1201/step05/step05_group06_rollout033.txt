"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast analytical coordinate descent
with per-coordinate quadratic model exploitation, multi-length exploration,
adaptive perturbation restarts, and block coordinate refinement.
Key innovation: O(n) per-coordinate updates (no FFT during iterations) enabling
10x more optimization passes within the time budget.

import numpy as np
import time


def fast_autoconv(a):
    """Compute full autoconvolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cd_pass_ordered(a, conv, n, conv_len, order=None):
    """
    One pass of coordinate descent over all coordinates.
    Uses analytical per-coordinate optimization (no FFT needed).
    Returns (a, conv, improved_flag).
    """
    if order is None:
        order = list(range(n))
    
    improved = False
    best_local = conv.max()
    
    for j in order:
        x_old = a[j]
        k_start = j
        k_end = j + n - 1  # always <= 2n-2 since j <= n-1
        m = n  # always n elements in range [j, j+n-1]
        
        # Build compact quadratic model:
        # For k = j+i (i=0..n-1):
        #   f(k, x) = base[i] + alpha[i]*x + beta[i]*x^2
        #   alpha[i] = 2*a[i] (for i != j), 0 (for i == j)
        #   beta[i] = 1 (for i == j), 0 (otherwise)
        #   base[i] = conv[j+i] - alpha[i]*x_old - beta[i]*x_old^2
        
        alpha_c = 2.0 * a.copy()
        alpha_c[j] = 0.0
        beta_c = np.zeros(n)
        beta_c[j] = 1.0
        
        base_c = conv[k_start:k_start + m] - alpha_c * x_old - beta_c * x_old * x_old
        
        # Constant: max of conv outside [k_start, k_end]
        C = 0.0
        if k_start > 0:
            C = max(C, conv[:k_start].max())
        if k_end + 1 < conv_len:
            C = max(C, conv[k_end + 1:].max())
        
        # Bisection to find x* minimizing max(max_k f(k,x), C)
        lo = 0.0
        hi = 1.0
        
        for _ in range(35):
            mid = (lo + hi) * 0.5
            vals = base_c + alpha_c * mid + beta_c * (mid * mid)
            max_in = vals.max()
            
            if C >= max_in - 1e-13:
                # C dominates; objective is C regardless of x. Break.
                break
            
            k_active = int(np.argmax(vals))
            deriv = alpha_c[k_active] + 2.0 * beta_c[k_active] * mid
            
            # Check for multiple active constraints
            max_val = vals[k_active]
            active_tol = 1e-9
            if max_in - max_val < active_tol:
                # Check if others are close
                active = vals >= max_in - active_tol
                if active.sum() > 1:
                    min_d = float(np.min(alpha_c[active] + 2.0 * beta_c[active] * mid))
                    max_d = float(np.max(alpha_c[active] + 2.0 * beta_c[active] * mid))
                else:
                    min_d = max_d = deriv
            else:
                min_d = max_d = deriv
            
            if min_d > 1e-10:
                hi = mid
            elif max_d < -1e-10:
                lo = mid
            else:
                break
        
        x_new = (lo + hi) * 0.5
        
        if abs(x_new - x_old) < 1e-12:
            continue
        
        # Update a[j] and conv
        a[j] = x_new
        delta = x_new - x_old
        
        # conv[k_start:k_start+m] update:
        # For i != j: conv[j+i] += 2*delta*a[i]
        # For i == j: conv[2j] += 2*x_old*delta + delta^2
        conv_update = 2.0 * delta * a.copy()
        conv_update[j] = 2.0 * x_old * delta + delta * delta
        conv[k_start:k_start + m] += conv_update
        
        cur_max = conv.max()
        if cur_max < best_local - 1e-12:
            best_local = cur_max
            improved = True
    
    return a, conv, improved

def symmetrize(a):
    """Make sequence symmetric and renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    return [float(x) for x in best_a]
