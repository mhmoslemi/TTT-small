"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe with
all-constraint LP, adaptive subgradient polishing, and fine coordinate descent.

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_full(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Using sparse matrices for efficiency.
    """
    # A_ub has shape (2n-1, n)
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_topk(f, M, n, active_k):

    K = len(active_k)
    rows = []
    cols = []
    vals = []

        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(i)
            cols.append(j)
            vals.append(f[k - j])
    
    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)
    c = -np.ones(n)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

def armijo_line_search(f, g, n, M, c0=0.5, rho=0.9, max_iter=50):
    """Armijo backtracking line search for direction g-f.
    We want to minimize max(conv) along f + t*(g-f), projected to simplex.
    """
    f0 = f.copy()
    d = g - f0
    # Initial step
    t = 1.0
    f0_conv = fast_convolve(f0)
    M0 = f0_conv.max()
    
    for _ in range(max_iter):
        f_trial = f0 + t * d
        f_trial = project_simplex(f_trial)
        conv_trial = fast_convolve(f_trial)
        M_trial = conv_trial.max()
        
        if M_trial < M0 - rho * t * (M0 - M):  # Hmm, this isn't quite right for non-smooth
            return f_trial, M_trial
        
        t *= c0
    
    # Return the best we found
    return f0, M0

def subgradient_optimize(f0, time_limit, top_k=50, base_lr=0.003, base_temp=1e-5):
    """Subgradient optimization with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)  # momentum
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = base_lr
    temp = base_temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find active peaks
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        
        # Softmax weights for active peaks
        pv = conv[peaks]
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * f[k - idx]
        
        # Gradient step with momentum
        v = 0.94 * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        M_new = conv.max()
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        lr *= 0.9999
        temp *= 1.002
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = max(base_lr * 0.7, 0.0005)
            temp = base_temp * 0.5
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_mults=7):
    """Coordinate descent with logarithmic multiplier sweep."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0
    
    # Log-spaced multipliers
    multipliers = np.logspace(np.log10(0.5), np.log10(2.0), n_mults)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                remaining = 1.0 - new_val
                if remaining <= 1e-10:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe with
sparse full-constraint LP, adaptive subgradient polishing with Hessian
approximation, fine coordinate descent, and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_full(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def solve_lp_topk(f, M, n, active_k):
    """Solve LP with only active constraints."""
    K = len(active_k)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(i)
            cols.append(j)
            vals.append(f[k - j])
    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, use_full=True, top_k=60):
    """LP-based Frank-Wolfe with full or top-k constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    step_sizes = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Solve LP
        if use_full:
            g = solve_lp_full(f, M * m_exp, n)
        else:
            k_sorted = np.argsort(conv)[::-1][:top_k]
            g = solve_lp_topk(f, M * m_exp, n, k_sorted.tolist())

        if g is None:
            m_exp *= 1.12
            if m_exp > 4.0:
                m_exp = 1.0
                continue
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 50:
                # Noise restart
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.001 * min(1 + no_improve // 30, 5)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        # Line search: try geometric step sizes
        best_t = 0.0
        best_f_new = f.copy()
        best_M_new = M

        for t_cand in step_sizes:
            f_new = f + t_cand * (g - f)
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new < 1e-15:
                continue
            f_new = f_new / s_new
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            if M_new < best_M_new:
                best_M_new = M_new
                best_f_new = f_new.copy()
                best_t = t_cand

        if best_t > 0 and best_M_new < best_obj - 1e-13:
            best_obj = best_M_new
            best_f = best_f_new.copy()
            f = best_f_new
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            m_exp = 1.0
            if verbose and it % 50 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}, t={best_t:.3f}, time={elapsed:.1f}s")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 4.0)
            # Still take the step to explore
            if best_t > 0:
                f = best_f_new
                conv = fast_convolve(f)
                M = conv.max()

        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 * min(1 + no_improve // 50, 8)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def sg_optimize(f0, time_limit, top_k=50, base_lr=0.003, base_temp=1e-5, momentum=0.94):
    """Subgradient optimization with adaptive parameters and Nesterov acceleration."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()  # Nesterov extrapolation
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = base_lr
    temp = base_temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov step
        y = f + momentum * (f - y) if it > 1 else f
        conv_y = fast_convolve(y)

        # Find active peaks with softmax weights
        k_act = min(top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -k_act)[-k_act:]
        pv = conv_y[peaks]
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient at f
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * f[k - idx]

        # Update with momentum
        v = momentum * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        M_new = conv.max()

        if M_new < best_obj - 1e-13:
            best_obj = M_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.99992
        temp *= 1.003

        if no_improve > 400:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            v = np.zeros(n)
            lr = max(base_lr * 0.6, 0.0004)
            temp = base_temp * 0.5
            no_improve = 0

    return best_f, best_obj


def coord_desc(f0, time_limit, n_mults=8):
    """Coordinate descent with log-spaced multipliers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    multipliers = np.logspace(np.log10(0.5), np.log10(2.0), n_mults)

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-13:
                continue

            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                remaining = 1.0 - new_val
                if remaining <= 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def pair_swap_descent(f0, time_limit, n_pairs_per_pass=50):
    """Pairwise mass transfer optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Focus on indices where the sequence has significant mass
    active_indices = np.argsort(f)[::-1][:min(200, n)]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(n_pairs_per_pass):
            if time.monotonic() - t0 > time_limit:
                break
            i, j = rng.choice(active_indices, size=2, replace=False)
            if f[i] < 1e-12 or f[j] < 1e-12:
                continue

            # Try moving fraction of mass from i to j
            for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
                delta = f[i] * frac
                if delta > f[j] * 0.5:
                    continue
                f_trial = f.copy()
                f_trial[i] -= delta
                f_trial[j] += delta
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f_trial.copy()
                    f = f_trial
                    improved = True
                    break
            # Also try reverse
            if not improved:
                for frac in [0.01, 0.05, 0.1]:
                    delta = f[j] * frac
                    if delta > f[i] * 0.5:
                        continue
                    f_trial = f.copy()
                    f_trial[j] -= delta
                    f_trial[i] += delta
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f_trial.copy()
                        f = f_trial
                        improved = True
                        break
        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

                                        use_full=False, top_k=60)
            try_seq(a_opt, "lpfw_topk60")

    # === Phase 3: Nesterov subgradient ===
    if best_seq is not None:
        print("\n=== Phase 3: Nesterov SG ===")
        rem = deadline - time.monotonic()
        t3 = min(180, rem * 0.15)
        if t3 > 20:
            n_best = len(best_seq)
            print(f"  Time: {t3:.0f}s, n={n_best}")
            a_opt, obj = sg_optimize(best_seq, time_limit=t3,
                                     top_k=max(30, n_best // 8),
                                     base_lr=0.002, base_temp=5e-6,
                                     momentum=0.93)
            try_seq(a_opt, "sg_nesterov")

            rem = deadline - time.monotonic()
            t3b = min(80, rem * 0.06)
            if t3b > 15:
                a_opt2, obj2 = sg_optimize(a_opt, time_limit=t3b,
                                           top_k=max(20, n_best // 12),
                                           base_lr=0.001, base_temp=2e-6,
                                           momentum=0.95)
                try_seq(a_opt2, "sg_fine")

    # === Phase 4: Coordinate descent ===
    if best_seq is not None:
        print("\n=== Phase 4: Coordinate Descent ===")
        rem = deadline - time.monotonic()
        t4 = min(180, rem * 0.15)
        if t4 > 30:
            print(f"  Time: {t4:.0f}s")
            a_opt, obj = coord_desc(best_seq, time_limit=t4, n_mults=8)
            try_seq(a_opt, "cd1")

            rem = deadline - time.monotonic()
            t4b = min(80, rem * 0.06)
            if t4b > 20:
                a_opt2, obj2 = coord_desc(a_opt, time_limit=t4b, n_mults=10)
                try_seq(a_opt2, "cd2")

    # === Phase 5: Pair swap ===
    if best_seq is not None:
        print("\n=== Phase 5: Pair Swap ===")
        rem = deadline - time.monotonic()
        t5 = min(60, rem * 0.04)
        if t5 > 10:
            a_opt, obj = pair_swap_descent(best_seq, time_limit=t5, n_pairs_per_pass=80)
            try_seq(a_opt, "pair_swap")

    # === Phase 6: Different lengths ===
    print("\n=== Phase 6: Multi-length ===")
    if best_seq is not None:
        for target_n in [640, 896, 1024]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.10, 100)
            if t_alloc < 20:
                break
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc,
                                        verbose=False, use_full=True)
            try_seq(a_opt, f"len_{target_n}")

    # === Phase 7: Final polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()

        # Final full LP-FW
        if rem > 30:
            t7 = min(rem * 0.30, 150)
            print(f"  Final LP-FW t={t7:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t7,
                                        verbose=False, use_full=True)
            try_seq(a_opt, "final_lpfw")

        # Final SG
        rem = deadline - time.monotonic()
        if rem > 20:
            t7b = min(rem * 0.25, 80)
            a_opt, obj = sg_optimize(best_seq, time_limit=t7b,
                                     top_k=max(15, n_best // 15),
                                     base_lr=0.0008, base_temp=1e-6,
                                     momentum=0.95)
            try_seq(a_opt, "final_sg")

        # Final CD
        rem = deadline - time.monotonic()
        if rem > 15:
            t7c = min(rem * 0.30, 60)
            a_opt, obj = coord_desc(best_seq, time_limit=t7c, n_mults=12)
            try_seq(a_opt, "final_cd")

        # Tiny perturbations + quick LP-FW
        rem = deadline - time.monotonic()
        for i in range(3):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.0005 * i)
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.2, 30)
            a_opt, obj = lp_fw_optimize(f_pert, time_limit=t_pert,
                                        verbose=False, use_full=False, top_k=40)
            try_seq(a_opt, f"perturb_{i}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
