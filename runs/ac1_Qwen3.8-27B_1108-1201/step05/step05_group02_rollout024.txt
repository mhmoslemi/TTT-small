"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Mirror descent with exact subgradients and Nesterov acceleration
2. Peak-targeting coordinate descent with analytical updates
3. Multi-scale dimension exploration
4. Adaptive restarts

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(f)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    w = w / ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        # subgradient at position k: for coordinate j, it's 2*f[k-j] if 0<=k-j<n
        # so sg[j] += 2 * w[i] * f[k-j] where k-j is in [0, n-1]
        # equivalently: for each valid j, k-j is in [0,n-1], so j is in [k-(n-1), k] intersect [0, n-1]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            sg[j] += 2.0 * w[i] * f[k - j]
    
    return sg

def mirror_descent_optimize(f0, time_limit, verbose=False, lr=0.01, momentum=0.9, 
                            top_k=50, temp=0.001):
    """Mirror descent with Nesterov momentum and adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_subgradient(f, conv, top_k, cur_temp)
        
        # Nesterov: use f + v to compute gradient, then update v
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        conv = conv_new
        
        # Adapt learning rate
        if no_improve > 100:
            cur_lr *= 0.95
            cur_temp *= 1.05
        if no_improve > 500:
            # Restart
            rng = np.random.default_rng(it * 31 + 7)
            noise_scale = 0.001 * min(no_improve / 100.0, 5.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            conv = fast_convolve(f)
            cur_lr = min(lr, cur_lr * 2)
            cur_temp = max(0.0001, cur_temp * 0.5)
            no_improve = 0
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}, temp={cur_temp:.6f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_obj

def coordinate_descent_optimize(f0, time_limit, verbose=False):
    """Coordinate descent with analytical peak reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        
        # Find top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        
        for pk in sorted(peak_positions, key=lambda x: -conv[x]):
            k = int(pk)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # For each contributing coordinate, try to reduce it
            # The contribution of coordinate j to peak k is f[j] * f[k-j]
            # If we reduce f[j] and compensately increase others, the peak decreases
            # but other convolution values might increase
            
            # Strategy: try reducing the largest contributors
            contribs = np.array([f[j] * f[k - j] for j in range(js, je + 1)])
            sorted_idx = np.argsort(contribs)[::-1]

                j = js + int(sorted_idx[si])
                if f[j] < 1e-10:
                    continue
                
                # Try several multiplicative factors
                for mult in [0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5]:
                    new_val = f[j] * mult
                    if new_val < 0 or new_val >= 1.0:
                        continue
                    denom = 1.0 - f[j]
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()

                        break

            no_improve += 1
            if no_improve > 20:

        else:
            no_improve = 0

            elapsed = time.monotonic() - t0
            print(f"    CD iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_vec(f, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv) efficiently."""
    n = len(f)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = (pv - pv.max()) / temperature
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for ki in range(len(peaks)):
        if w[ki] < 1e-14:
            continue
        k = int(peaks[ki])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # f[k-j] for j in [js..je] = f[k-je : k-js+1] reversed
        sg[js:je + 1] += 2.0 * w[ki] * f[k - je:k - js + 1][::-1]

    return sg


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def peak_coordinate_descent(f0, time_limit, verbose=False, max_peaks=5, max_coords=8):
    """Coordinate descent targeting the contributing coordinates of top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        # Find top peaks
        top_k = min(max_peaks, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]
        peak_positions = sorted(peak_positions, key=lambda x: -conv[x])

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(pk)
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            # Compute contributions
            contribs = np.array([f[j] * f[k - j] for j in range(js, je + 1)])
            sorted_idx = np.argsort(contribs)[::-1]

            for si in sorted_idx[:max_coords]:
                if time.monotonic() - t0 > time_limit:
                    break
                j = js + int(sorted_idx[si])
                if f[j] < 1e-12:
                    continue

                # Try multiplicative adjustments
                for mult in [0.6, 0.75, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                    new_val = f[j] * mult
                    if new_val < 0 or new_val >= 0.99:
                        continue
                    denom = 1.0 - f[j]
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale

                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        conv = fast_convolve(f)
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PCD iter {it}: best={best_obj:.10f}, elapsed={elapsed:.1f}s")

def backtracking_subgradient_optimize(f0, time_limit, verbose=False,
                                       top_k=40, initial_temp=0.0005,
                                       init_eta=0.01):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

    t0 = time.monotonic()

    cur_temp = initial_temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        sg = compute_subgradient_vec(f, conv, top_k, cur_temp)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        sg_dir = -sg / sg_norm  # normalized descent direction

        # Backtracking line search
        eta = init_eta
        found = False
        for _ in range(12):
            f_trial = project_simplex(f + eta * sg_dir)
            conv_trial = fast_convolve(f_trial)

                if obj_trial < best_obj - 1e-15:

                    best_f = f.copy()
                found = True
                no_improve = 0
                break

            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t3, verbose=True,
                                                max_peaks=5, max_coords=10)
            try_seq(a_opt, "pcd_main")

    # === Phase 4: Backtracking Subgradient ===
    print(f"\n=== Phase 4: Backtracking SG (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t4 = min(250, rem * 0.15)
        if t4 > 20:
            n_b = len(best_seq)
            print(f"  n={n_b}, Budget: {t4:.0f}s")
            a_opt, _ = backtracking_subgradient_optimize(
                best_seq, time_limit=t4, verbose=True,
                top_k=max(25, n_b // 8), initial_temp=0.0004, init_eta=0.008)
            try_seq(a_opt, "bts")

    # === Phase 5: Dimension Exploration ===
    print(f"\n=== Phase 5: Dimension Exploration (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        candidate_lengths = sorted(set([
            len(base) - 64, len(base) - 32,
            len(base) + 32, len(base) + 64, len(base) + 128
        ]))
        candidate_lengths = [ln for ln in candidate_lengths if 256 <= ln <= 2048]

        for target_n in candidate_lengths[:3]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_alloc = min(rem * 0.3, 120)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  NSG n={target_n}, Budget: {t_alloc:.0f}s")
            a_opt, _ = nesterov_subgradient_optimize(
                f_interp, time_limit=t_alloc, verbose=False,
                initial_lr=0.004, momentum=0.92,
                top_k=max(25, target_n // 6), initial_temp=0.0005,
                restart_patience=200)
            try_seq(a_opt, f"nsg_n{target_n}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Short NSG with fine params
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = nesterov_subgradient_optimize(
                best_seq, time_limit=min(rem, 100), verbose=False,
                initial_lr=0.001, momentum=0.95,
                top_k=max(15, n_best // 12), initial_temp=0.0002,
                restart_patience=150, lr_decay=0.9999)
            try_seq(a_opt, "nsg_polish")

        # Short CD
        rem = deadline - time.monotonic() - 25
        if rem > 12:
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=min(rem, 60),
                                                verbose=False, max_peaks=3, max_coords=6)
            try_seq(a_opt, "cd_polish")

        # Perturbation + short optimization
        for p in range(6):
            rem = deadline - time.monotonic() - 12
            if rem < 10:
                break
            # Structured perturbation: alternate between random and peak-targeted
            if p % 2 == 0:
                noise = rng.standard_normal(n_best) * (0.001 + 0.0004 * p)
            else:
                # Perturb coordinates that contribute to the peak
                conv_b = fast_convolve(best_seq)
                peak_k = int(np.argmax(conv_b))
                noise = np.zeros(n_best)
                js = max(0, peak_k - n_best + 1)
                je = min(n_best - 1, peak_k)
                noise[js:je + 1] = rng.standard_normal(je - js + 1) * (0.003 + 0.001 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = nesterov_subgradient_optimize(
                f_pert, time_limit=min(rem, 20), verbose=False,
                initial_lr=0.0008, momentum=0.94,
                top_k=max(10, n_best // 15), initial_temp=0.0002,
                restart_patience=100)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
