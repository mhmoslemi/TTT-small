import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None, time_limit=20):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit,
                                                'dual_feasibility_tolerance': 1e-10,
                                                'primal_feasibility_tolerance': 1e-10})
        if res.success and res.x is not None:
            return res.x, res.y
    except Exception:
        pass
    return None, None

def compute_peak_gradient(f, peaks, weights):
    """Compute gradient of sum_k w[k]*(f*f)[k] w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        k = int(peaks[idx])
        wk = weights[idx]
        if wk < 1e-18:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += wk * f[k - ms]
            # Symmetric contribution
            ms2 = np.arange(lo, hi + 1)
            grad[ms2] += wk * f[k - ms2]
    return grad


def bb_subgradient_optimize(f0, time_limit, n_peaks=50, max_iter=None):
    """Adaptive subgradient with Barzilai-Borwein step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    f_prev = f.copy()
    g_prev = None
    step_prev = None
    lr = 1.0 / n
    momentum = 0.0
    v = np.zeros(n)
    beta_momentum = 0.0

    if max_iter is None:
        max_iter = 1000000

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        conv = fast_convolve(f)

        # Select top peaks
        k_top = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        # Exponential soft weighting
        pv_shifted = pv - pv.min()
        weights = np.exp(pv_shifted - pv_shifted.max())
        weights /= weights.sum()

        grad = compute_peak_gradient(f, peaks, weights)
        gn = np.linalg.norm(grad)
        if gn < 1e-18:
            break
        grad /= gn

        # Barzilai-Borwein step size
        if g_prev is not None and step_prev is not None:
            s_vec = f - f_prev
            y_vec = grad - g_prev
            sy = np.dot(s_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            yy = np.dot(y_vec, y_vec)
            if sy > 1e-14:
                if ss > 1e-14 and yy > 1e-14:
                    lr_BB1 = ss / sy
                    lr_BB2 = sy / yy
                    lr = 0.5 * (lr_BB1 + lr_BB2)
                lr = np.clip(lr, 0.01 / n, 10.0 / n)
            else:
                lr *= 0.9

        f_prev = f.copy()
        g_prev = grad.copy()

        # Momentum
        beta_momentum = min(0.95, 0.005 * it)
        v = beta_momentum * v - lr * grad
        f_new = project_simplex(f + v)
        step_prev = f_new - f

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 1.0 / n
                momentum = 0.0
                no_improve = 0
                # Small random perturbation
                rng = np.random.default_rng(it * 7919 + 3)
                f = project_simplex(f + rng.standard_normal(n) * 0.001)

    return best_f, best_obj


def lse_optimize(f0, time_limit, beta_start=50.0, beta_end=20000.0, max_peaks=300):
    """Log-sum-exp smoothed gradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 1.5 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        # Clip to avoid overflow
        bc_clipped = np.clip(bc - bc_max, -50, 0)
        exp_bc = np.exp(bc_clipped)
        w = exp_bc / exp_bc.sum()

        # Compute gradient using top peaks for efficiency
        if max_peaks < len(w):
            sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
            sig_idx = sig_idx[w[sig_idx] > 1e-12]
        else:
            sig_idx = np.where(w > 1e-14)[0]

        grad = np.zeros(n)
        for k in sig_idx:
            wk = w[int(k)]
            if wk < 1e-15:
                continue
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            if lo <= hi:
                ms = np.arange(lo, hi + 1)
                grad[ms] += wk * f[ki - ms]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            no_improve += 100
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.94
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 80:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(1.5 / n, 0.0005)
                beta = min(beta * 2.0, beta_end)
                momentum = 0.90
                no_improve = 0

        if beta < beta_end:
            beta *= 1.006

    return best_f, best_obj


def spectral_init(n, n_iter=50, seed=42):
    """Gerchberg-Saxton style initialization for flat auto-convolution."""
    rng = np.random.default_rng(seed)
    f = rng.random(n) + 0.1
    f = np.maximum(f, 0.0)
    f /= f.sum()

    N = 1 << (2 * n - 1).bit_length()

    for i in range(n_iter):
        F = np.fft.rfft(f, N)
        # Compute spectrum of auto-convolution
        G = F * F
        # Flatten the spectrum: blend G with its mean (excluding DC)
        g_mean = np.sum(G[1:].real) / (len(G) - 1)
        alpha = 0.3  # Flattening strength
        G_new = G.copy()
        G_new[1:] = (1 - alpha) * G[1:] + alpha * g_mean
        # Take square root to get new F
        # Need to handle complex square root carefully
        phase = np.angle(G_new)
        mag = np.sqrt(np.abs(G_new))
        F_new = mag * np.exp(1j * phase)
        # Ensure DC component preserves sum
        F_new[0] = np.sqrt(max(0, G[0].real)) if G[0].real > 0 else 1.0
        if F_new[0].real < 1e-10:
            F_new[0] = 1.0

        f_new = np.fft.irfft(F_new, N)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new /= s
        f = f_new

    return f


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    # Second FW pass from best
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(200, t2_end - time.monotonic())
    if t_alloc > 20:
        print(f"  FW-LP pass 2, n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = fw_lp(current, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: BB Subgradient ===
    print(f"\n=== Phase 3: BB Subgradient ===")
    t3_end = deadline - 500
    current = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = min(250, t3_end - time.monotonic())
    if t_alloc > 30:
        print(f"  BB subgrad n={len(current)}, t={t_alloc:.0f}s, peaks=80")
        f_opt, _ = bb_subgradient_optimize(current, time_limit=t_alloc, n_peaks=80)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(200, t3_end - time.monotonic())
    if t_alloc > 20:
        print(f"  BB subgrad n={len(current)}, t={t_alloc:.0f}s, peaks=30")
        f_opt, _ = bb_subgradient_optimize(current, time_limit=t_alloc, n_peaks=30)
        try_sequence(f_opt, best_tracker)

    # === Phase 4: LSE High-Beta Polishing ===
    print(f"\n=== Phase 4: LSE Polishing ===")
    t4_end = deadline - 250
    current = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = min(200, t4_end - time.monotonic())
    if t_alloc > 30:
        print(f"  LSE n={len(current)}, t={t_alloc:.0f}s, beta 100->10000")
        f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                 beta_start=100.0, beta_end=10000.0, max_peaks=300)
        try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current
    t_alloc = min(200, t4_end - time.monotonic())
    if t_alloc > 20:
        print(f"  LSE n={len(current)}, t={t_alloc:.0f}s, beta 2000->50000")
        f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                 beta_start=2000.0, beta_end=50000.0, max_peaks=200)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Alternating refinement ===
    print(f"\n=== Phase 5: Alternating Refinement ===")
    t5_end = deadline - 100
    cycles = 0
    while time.monotonic() < t5_end and cycles < 6:
        cycles += 1
        current = best_tracker[1] if best_tracker[1] is not None else parent
        remaining = t5_end - time.monotonic()
        if remaining < 40:
            break

        if cycles % 3 == 1:
            t_alloc = min(60, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: FW n={len(current)}, t={t_alloc:.0f}s")
                f_opt, _ = fw_lp(current, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)
        elif cycles % 3 == 2:
            t_alloc = min(60, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: BB n={len(current)}, t={t_alloc:.0f}s")
                f_opt, _ = bb_subgradient_optimize(current, time_limit=t_alloc, n_peaks=20)
                try_sequence(f_opt, best_tracker)
        else:
            t_alloc = min(80, remaining * 0.4)
            if t_alloc > 15:
                print(f"  Cycle {cycles}: LSE n={len(current)}, t={t_alloc:.0f}s, high beta")
                f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                         beta_start=5000.0, beta_end=80000.0, max_peaks=150)
                try_sequence(f_opt, best_tracker)

    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Nearby Lengths ===")
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        offsets = [32, -32, 64, -64, 96, -96]
        for off in offsets:
            n_try = n_best + off
            if n_try < 200 or n_try > 1200:
                continue
            t_remaining = deadline - time.monotonic() - 10
            if t_remaining < 25:
                break
            t_alloc = min(45, t_remaining * 0.5)
            if t_alloc < 15:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize(f_init, time_limit=t_alloc,
                                     beta_start=100.0, beta_end=8000.0, max_peaks=200)
            try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using parent fallback")
    return [float(x) for x in parent]
