def fast_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_descent(f0, time_limit, lr=0.005, momentum=0.93, 
                        temp_init=0.00005, temp_growth=1.003,
                        verbose=False):
    """Subgradient descent with softmax-weighted peaks and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp_init
    it = 0
    baseline = best_obj
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Softmax weights over all peaks
        scaled = conv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        # Compute weighted subgradient: sg[j] = 2 * sum_k w[k] * a[k-j] (with bounds)
        # More efficiently: sg = 2 * (w * f) reversed... 
        # Actually, the gradient of conv[k] = sum_j f[j]*f[k-j] w.r.t. f[j] is f[k-j]
        # So weighted gradient at f[j] = sum_k w[k] * f[k-j] (where 0 <= k-j < n)
        # This is the cross-correlation of w and f, which we can compute via FFT!
        
        # sg[j] = sum_k w[k] * f[k-j] for valid k
        # = (w * f)[j] where * is cross-correlation
        # cross_correlation(w, f)[j] = sum_k w[k] * f[k-j]
        # This equals convolve(w, f_reversed)[j + n - 1] ... let me think
        
        # Actually: sum_k w[k] * f[k-j] = sum_k w[k] * f[k-j]
        # Let m = k-j, so k = m+j: sum_m w[m+j] * f[m]
        # = (w_shifted * f) where w_shifted[m] = w[m+j]
        # This is the cross-correlation of w and f at lag j.
        
        # Using FFT: cross_correlation(w, f) = ifft(fft(w) * conj(fft(f)))
        # But we need the full cross-correlation.
        
        # Alternative: the weighted gradient is:
        # grad[j] = sum_k w[k] * f[max(0, min(n-1, k-j))]  (with f out of range = 0)
        # = sum_{k=j}^{j+n-1} w[k] * f[k-j]  (if w is defined on [0, 2n-2])
        # This is the convolve of w and f (full), evaluated at specific indices.
        
        # Let's just use FFT:
        # cross_corr(w, f)[j] = sum_k w[k]*f[k-j] for k in [0, 2n-2] and k-j in [0, n-1]
        # = sum_{k=j}^{j+n-1} w[k]*f[k-j]
        # This is equivalent to: ifft(fft(w, size) * conj(fft(f, size))) gives circular cross-correlation
        # For full cross-correlation, we zero-pad.
        
        # Simpler approach: use the fact that the gradient of sum_k w[k]*conv[k] w.r.t. f is:
        # For each k, d(conv[k])/df[j] = f[k-j] (if 0 <= k-j < n)
        # So grad[j] = sum_k w[k] * f[k-j] * I(0 <= k-j < n)
        #           = sum_{m=0}^{n-1} w[j+m] * f[m]  (where m = k-j, so k = j+m)
        # This is the cross-correlation of w and f at lag j (with w zero-padded).
        
        # Compute via FFT:
        # xcorr(w, f) at lag j = sum_m w[j+m]*f[m]
        # = ifft(conj(fft(f)) * fft(w))[j] (circular, with sufficient zero-padding)
        
        fft_size = 1 << (3 * n - 1).bit_length()  # Need at least 2n-1 for full, but 3n-1 to be safe
        W = np.fft.rfft(w, fft_size)
        F = np.fft.rfft(f, fft_size)
        sg_full = np.fft.irfft(W * np.conj(F), fft_size)
        # sg_full[j] = sum_k w[k]*f[k-j] (circular, but with zero-padding it's full for j in [0, n-1])
        sg = 2.0 * sg_full[:n]
        
        # Update with momentum
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        co = conv_new.max()
        
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new
        
        # Adaptive learning rate
        cur_lr *= 0.9999
        cur_temp *= temp_growth
        
        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 2**31)
            scale = min(0.003, 0.001 * (no_improve // 200))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.001)
            cur_temp = max(temp_init, cur_temp * 0.5)
            no_improve = 0
        
        if cur_lr < 0.0001:
            break
    
    return best_f, best_obj

def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def lp_frank_wolfe(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe using min-peak LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        else:
            gs = g.sum()
            if gs < 1e-15:
                g = np.ones(n) / n
            else:
                g = g / gs
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs < 1e-15:
                g = np.ones(n) / n
            else:
                g = g / gs
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search with golden section
        best_alpha = 0.0
        best_val = M
        
        # Coarse search
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-12:
            f = np.maximum(f + best_alpha * (g - f), 0.0)
            sn = f.sum()
            if sn < 1e-15:
                f = np.ones(n) / n
            else:
                f = f / sn
            M = fast_convolve(f).max()
            
            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 104729 + 7)
            scale = 0.001 + 0.002 * min(no_improve // 30, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        conv = fast_convolve(f)
        
        # Find top peaks
        num_peaks = min(5, len(conv))
        peak_positions = np.argpartition(conv, -num_peaks)[-num_peaks:]
        
        for pk in sorted(peak_positions, key=lambda x: -conv[x]):
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            # For each coordinate in the peak's support, try to reduce it
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-12:
                    continue
                
                best_j_val = old_val
                best_j_obj = best_obj
                
                for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.8:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val
                
                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                        if best_j_obj < best_obj - 1e-14:
                            best_obj = best_j_obj
                            best_f = f.copy()
                            improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                # Random exploration
                rng = np.random.default_rng(passes * 7919)
                for _ in range(min(30, n // 5)):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.4, 0.6, 1.5, 2.0]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.8:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best
