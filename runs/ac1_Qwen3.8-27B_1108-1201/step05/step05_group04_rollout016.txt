    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    return np.fft.irfft(W * F, fft_size)[:n] * 2.0

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated gradient descent
with adaptive peak weighting, Nesterov momentum, multi-phase optimization,
and simulated annealing for escaping local minima.
Key improvement: O(n log n) gradient via FFT convolution of weight vector with sequence.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_gradient(f, w):
    """Compute gradient of sum_k w[k] * conv(f,f)[k] w.r.t. f using FFT.
    
    w has length 2n-1, f has length n.
    Returns array of length n representing the subgradient.
    Formula: grad[m] = 2 * sum_j w[m+j] * f[j]
    This equals 2 * (w * f)[n-1+m] where * is linear convolution.
    """
    n = len(f)
    w_len = len(w)
    # Full convolution length: (2n-1) + n - 1 = 3n - 2
    total_len = w_len + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f, fft_size)
    conv_full = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_full[n - 1:2 * n - 1]


def lse_optimize(f0, time_limit, beta_start=10.0, beta_end=100000.0,
                 verbose=False, patience=80):
    """LSE smoothed gradient descent with FFT-based gradient and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 3.0 / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = fft_gradient(f, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 200
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

            best_obj = obj_new
            best_f = f_new.copy()

            no_improve += 1
            if no_improve >= patience:
                f = best_f.copy()
                v = np.zeros(n)
                beta = min(beta * 1.8, beta_end)
                lr = max(lr * 0.75, 0.05 / n)
                momentum = min(momentum + 0.005, 0.97)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.003

        if verbose and it % 200 == 0:
            print(f"    LSE it={it}, beta={beta:.1f}, obj={best_obj:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def topk_peak_optimize(f0, time_limit, k_schedule=None, verbose=False):
    """Optimize by directly targeting top-k peaks with adaptive weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = 2.5 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    patience = 100

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Adaptive k: grow over iterations
        if k_schedule is not None:
            k = min(k_schedule, max(2, int(2 + (it / 3000.0) * (k_schedule - 2))))
        else:
            k = min(30, max(2, it // 200 + 2))

        # Get top-k peaks with exponential weighting
        if k >= 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            peak_vals = conv.copy()
        else:
            peaks = np.argpartition(conv, -k)[-k:]
            peak_vals = conv[peaks]

        max_val = peak_vals.max()
        if max_val > 1e-18:
            # Heavy weighting on the largest peak
            weights = np.exp(30.0 * (peak_vals - max_val) / max_val)
            w_sum = weights.sum()
            if w_sum > 1e-15:
                weights /= w_sum
            else:
                weights = np.zeros(len(peaks))
                weights[0] = 1.0
        else:
            weights = np.ones(len(peaks)) / len(peaks)

        w = np.zeros(2 * n - 1)
        w[peaks] = weights

        grad = fft_gradient(f, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 200
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                # Random perturbation to escape
                rng = np.random.default_rng((it * 7919 + 17) % 1000000)
                scale = 0.002 + 0.003 * min(1.0, no_improve / (3.0 * patience))
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                v = np.zeros(n)
                lr = min(2.5 / n * (1 + no_improve / patience), 8.0 / n)
                no_improve = 0

        if verbose and it % 300 == 0:
            print(f"    TopK it={it}, k={k}, obj={best_obj:.10f}, lr={lr:.6f}")

    return best_f, best_obj


def peak_equalize(f0, time_limit, verbose=False):
    """Try to equalize the top peaks by applying differential pressure."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = 1.5 / n
    momentum = 0.94
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        max_val = conv.max()

        # Weight peaks by how much they exceed the "target" (mean of top peaks)
        top_n = min(20, 2 * n - 1)
        top_peaks = np.argpartition(conv, -top_n)[-top_n:]
        top_vals = conv[top_peaks]
        target = top_vals.mean()

        # Positive weight on peaks above target, proportional to excess
        excess = np.maximum(top_vals - target, 0)
        if excess.max() < 1e-18:
            # All top peaks are equal, focus on the single max
            w = np.zeros(2 * n - 1)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.zeros(2 * n - 1)
            w[top_peaks] = excess
            w_sum = w.sum()
            if w_sum > 1e-18:
                w /= w_sum
            else:
                w[np.argmax(conv)] = 1.0

        grad = fft_gradient(f, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 200
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 50:
                lr *= 0.92
            if no_improve >= 300:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 1.5 / n
                no_improve = 0

    return best_f, best_obj


def anneal_optimize(f0, time_limit, T_start=0.01, T_end=0.001, verbose=False):
    """Gradient descent with simulated annealing for escaping local minima."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = 3.0 / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    beta = 50.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Annealing schedule
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = fft_gradient(f, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 200
            break
        grad /= grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        delta = obj_new - conv.max()

        # Accept if better, or with probability exp(-delta/T) if worse
        if delta < 0 or (delta > 0 and np.random.random() < np.exp(-delta / max(T, 1e-15))):
            f = f_new
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            v = 0.5 * v
            no_improve += 1

        if no_improve >= 100:
            f = best_f.copy()
            v = np.zeros(n)
            beta = min(beta * 1.5, 10000.0)
            no_improve = 0

        if beta < 10000.0:
            beta *= 1.005

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def cosine_init(n, freqs=None, amplitudes=None):
    """Create a cosine-modulated initial sequence for diversity."""
    if freqs is None:
        freqs = [1, 2, 3]
    if amplitudes is None:
        amplitudes = [0.05, 0.03, 0.01]
    base = np.ones(n) / n
    x = np.arange(n) / n
    f = base.copy()
    for fr, am in zip(freqs, amplitudes):
        f += am * np.cos(2 * np.pi * fr * x)
    f = np.maximum(f, 0)
    f /= f.sum()
    return f


def random_symmetric_init(n):
    """Random symmetric initial sequence."""
    rng = np.random.default_rng(12345)
    half = (n + 1) // 2
    h = rng.random(half) + 0.1
    if n % 2 == 0:
        f = np.concatenate([h, h[::-1]])
    else:
        f = np.concatenate([h, h[-2:0:-1]])
    f = np.maximum(f, 0)
    f /= f.sum()
    return f


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Quick multi-length exploration with LSE
    Phase 2: Intensive LSE at best length (high beta)
    Phase 3: Top-k peak optimization (multiple k values)
    Phase 4: Peak equalization
    Phase 5: Simulated annealing
    Phase 6: Final polish at best and nearby lengths
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Quick multi-length exploration ===
    print("\n=== Phase 1: Multi-length Quick Exploration ===")
    t1_end = deadline - 950

    # Try several lengths
    length_candidates = sorted(set([
        n_parent, n_parent - 64, n_parent + 64,
        n_parent - 128, n_parent + 128,
        512, 640, 768, 896, 1024
    ]))
    length_candidates = [nc for nc in length_candidates if 192 <= nc <= 1280]

    # Quick LSE at each length
    for nc in length_candidates[:8]:
        t_rem = t1_end - time.monotonic()
        if t_rem < 25:
            break
        t_alloc = min(50, t_rem * 0.3)
        if t_alloc < 15:
            break
        f_init = interpolate_to_length(parent, nc)
        print(f"  Quick LSE n={nc}, t={t_alloc:.0f}s")
        f_opt, _ = lse_optimize(f_init, time_limit=t_alloc,
                                 beta_start=10.0, beta_end=3000.0)
        try_sequence(f_opt, best_tracker)

    # Try a diverse initialization at parent length
    if time.monotonic() < t1_end - 30:
        f_div = cosine_init(n_parent, freqs=[1, 3, 5], amplitudes=[0.04, 0.02, 0.01])
        t_alloc = min(50, t1_end - time.monotonic() - 10)
        if t_alloc > 15:
            print(f"  Diverse init n={n_parent}, t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize(f_div, time_limit=t_alloc,
                                     beta_start=10.0, beta_end=5000.0)
            try_sequence(f_opt, best_tracker)

    # === Phase 2: Intensive LSE at best length ===
    print(f"\n=== Phase 2: Intensive LSE (best so far: {best_tracker[0]:.8f}) ===")
    t2_end = deadline - 550
    current = best_tracker[1] if best_tracker[1] is not None else parent

    # Long LSE run with high beta
    t_alloc = min(280, t2_end - time.monotonic())
    if t_alloc > 30:
        print(f"  Long LSE n={len(current)}, t={t_alloc:.0f}s, beta_end=200000")
        f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                 beta_start=20.0, beta_end=200000.0,
                                 patience=100)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Top-k peak optimization ===
    print(f"\n=== Phase 3: Top-k Peak Optimization ===")
    t3_end = deadline - 300
    current = best_tracker[1] if best_tracker[1] is not None else current

    for k_val in [5, 10, 20, 50]:
        t_rem = t3_end - time.monotonic()
        if t_rem < 25:
            break
        t_alloc = min(70, t_rem * 0.4)
        if t_alloc < 20:
            break
        print(f"  TopK n={len(current)}, k={k_val}, t={t_alloc:.0f}s")
        f_opt, _ = topk_peak_optimize(current, time_limit=t_alloc,
                                       k_schedule=k_val)
        try_sequence(f_opt, best_tracker)
        current = best_tracker[1] if best_tracker[1] is not None else current

    # === Phase 4: Peak equalization ===
    print(f"\n=== Phase 4: Peak Equalization ===")
    t4_end = deadline - 180
    current = best_tracker[1] if best_tracker[1] is not None else current

        f_opt, _ = peak_equalize(current, time_limit=t_alloc)

    # === Phase 5: Simulated annealing ===
    print(f"\n=== Phase 5: Simulated Annealing ===")

    print(f"\n=== Phase 6: Final Polish ===")

                                 beta_start=500.0, beta_end=500000.0,
                                 patience=50)

    if best_tracker[1] is not None:

        for off in [32, -32, 64, -64, 128, -128]:

            t_rem = deadline - time.monotonic() - 15
            if t_rem < 20:

            t_alloc = min(35, t_rem * 0.4)
            if t_alloc < 12:

            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Nearby n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize(f_init, time_limit=t_alloc,
                                     beta_start=50.0, beta_end=100000.0)
            try_sequence(f_opt, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using parent fallback")
    return [float(x) for x in parent]
