"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
with line search, coordinate descent polishing, and multi-length search.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def build_lp_matrix(a):
    """Build sparse constraint matrix for the LP.
    A[k, i] = a[k-i] if 0 <= k-i < n else 0, for k=0..2n-2, i=0..n-1
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_direction(a, M, method='highs'):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns optimal b.

    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method)
    if res.status == 0:
        return res.x
    # Fallback: return a itself (which is feasible)
    return a.copy()

    for i in range(n_steps + 1):
        t = i / n_steps
        val = (1-t)**2 * C_aa + 2*t*(1-t) * C_ag + t**2 * C_gg
        mv = val.max()
        if mv < best_val:
            best_val = mv
            best_t = t
    return best_t

def lp_optimize(a0, time_limit, verbose=False, max_iters=5000):
    """LP-based optimization."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    C_aa = fast_convolve(a)
    M = C_aa.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        
        # Solve LP for direction
        g = lp_direction(a, M * 1.0001)
        sg_ = g.sum()
        if sg_ < 1e-12:
            break
        g = g / sg_  # Normalize to sum=1
        
        # Compute cross terms for line search
        C_ag = fast_cross_convolve(a, g)
        C_gg = fast_convolve(g)
        
        # Line search
        t = line_search_lp(a, g, C_aa, C_ag, C_gg, n_steps=30)
        
        if t < 1e-6:
            # No improvement possible with this direction
            no_improve += 1
            if no_improve > 20:
                # Try with relaxed M
                M_relaxed = M * 1.01
                g = lp_direction(a, M_relaxed)
                sg_ = g.sum()
                if sg_ < 1e-12:
                    break
                g = g / sg_
                C_ag = fast_cross_convolve(a, g)
                C_gg = fast_convolve(g)
                t = line_search_lp(a, g, C_aa, C_ag, C_gg, n_steps=30)
                if t < 1e-6:
                    break
            else:
                break
        else:
            no_improve = 0
        
        # Update
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-12:
            break
        a_new /= sn
        
        C_aa_new = fast_convolve(a_new)
        M_new = C_aa_new.max()
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_a = a_new.copy()
        
        a = a_new
        C_aa = C_aa_new
        M = M_new
        
        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: M={M:.10f}, t={t:.4f}")
    
    return best_a, best_obj

def coordinate_bisection(a0, time_limit, verbose=False, max_passes=500):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    C_aa = fast_convolve(a)
    best_obj = C_aa.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-14:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            # b = a with a[j]=0, normalized
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            # b_valid[k] = b[k-j] if 0 <= k-j < n, else 0
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            # Bisection on x in [0, 1]
            lo = 0.0
            hi = 1.0
            
            for _ in range(50):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                
                k_max = int(np.argmax(f_vals))
                
                # Derivative of f_{k_max} at mid
                k = k_max
                b_kj = b[k - j] if (0 <= k - j < n) else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                
                if deriv > 1e-13:
                    hi = mid
                elif deriv < -1e-13:
                    lo = mid
                else:
                    break
            
            x_new = (lo + hi) / 2.0
            
            if abs(x_new - x_old) < 1e-14:
                continue
            
            # Apply update
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue
            
            C_aa = fast_convolve(a)
            obj = C_aa.max()
            
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            if passes > 2:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            else:
                break
    
    return best_a, best_obj

def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

def subgradient_phase(a0, time_limit, lr=0.01, mom=0.9, top_k=50, temp=0.001):
    """Projected subgradient descent with momentum."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    C_aa = fast_convolve(a)
    best_obj = C_aa.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(C_aa))
        peaks = np.argpartition(C_aa, -k_actual)[-k_actual:]
        
        pv = C_aa[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                sg[idx] += 2.0 * w[i] * a[k - idx]
        
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        
        C_aa_new = fast_convolve(a_new)
        obj_new = C_aa_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        C_aa = C_aa_new
        cur_lr *= 0.9998
        cur_temp *= 1.002
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 5)
            noise_scale = 0.002 + 0.002 * min(no_improve // 100, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            C_aa = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0
    
    return best_a, best_obj

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
with line search, accelerated coordinate descent, subgradient polishing,
multi-length exploration, and adaptive restarts.
Key innovation: solve an LP at each step to find the optimal descent direction,
which is more principled than subgradient methods.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric: a[i] = (a[i] + a[n-1-i]) / 2, then renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def build_lp_matrix(a):
    """Build sparse constraint matrix for the LP.
    A[k, i] = a[k-i] if 0 <= k-i < n else 0, for k=0..2n-2, i=0..n-1.
    This represents conv(a, b)[k] = sum_i A[k,i] * b[i].
    """
    n = len(a)
    # Column i has non-zeros at rows i, i+1, ..., i+n-1 with values a[0], a[1], ..., a[n-1]
    row_idx = np.concatenate([np.arange(i, i + n) for i in range(n)])
    col_idx = np.concatenate([np.full(n, i) for i in range(n)])
    vals = np.tile(a, n)
    A = sp.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def lp_direction(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns optimal b vector.
    """
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 2.0)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 10})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    # Fallback: return a itself
    return a.copy()


def line_search_conv(a, g, C_aa, C_ag, C_gg, n_evals=40):
    """Find best t in [0,1] to minimize max_k conv((1-t)a + t*g, (1-t)a + t*g)_k."""
    best_t = 0.0
    best_val = C_aa.max()
    for i in range(n_evals + 1):
        t = i / n_evals
        val = (1.0 - t) * (1.0 - t) * C_aa + 2.0 * t * (1.0 - t) * C_ag + t * t * C_gg
        mv = val.max()
        if mv < best_val - 1e-15:
            best_val = mv
            best_t = t
    return best_t


def lp_optimize(a0, time_limit, verbose=False, max_iters=10000):
    """LP-based optimization: iteratively find best direction via LP."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    C_aa = fast_convolve(a)
    M = C_aa.max()
    best_obj = M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1

        # Solve LP for direction
        g = lp_direction(a, M * (1.0 + 1e-6))
        sg = g.sum()
        if sg < 1e-12:
            break
        g = g / sg

        # Check if g is essentially the same as a
        if np.linalg.norm(g - a, ord=1) < 1e-8:
            no_improve += 1
            stall_count += 1
        else:
            stall_count = 0

        # Compute cross terms for line search
        C_ag = fast_cross_convolve(a, g)
        C_gg = fast_convolve(g)

        # Line search
        t = line_search_conv(a, g, C_aa, C_ag, C_gg, n_evals=50)

        if t < 1e-6:
            no_improve += 1
        else:
            no_improve = 0

        # Update
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-12:
            break
        a_new /= sn

        C_aa_new = fast_convolve(a_new)
        M_new = C_aa_new.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_a = a_new.copy()

        a = a_new
        C_aa = C_aa_new
        M = M_new

        # Stall handling
        if no_improve > 100 or stall_count > 200:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 5:
                break
            rng = np.random.default_rng(it * 31 + 7)
            noise = 0.003 * min(1.0, no_improve / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            C_aa = fast_convolve(a)
            M = C_aa.max()
            no_improve = 0
            stall_count = 0

        if verbose and it % 200 == 0:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    LP iter {it}: M={M:.10f}, best={best_obj:.10f}, t={t:.4f}, rem={rem:.0f}s")

    return best_a, best_obj


def coordinate_bisection(a0, time_limit, verbose=False, max_passes=200):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    C_aa = fast_convolve(a)
    best_obj = C_aa.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            # b = a with a[j]=0, normalized
            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_convolve(b)
            conv_len = 2 * n - 1

            # b_valid[k] = b[k-j] if 0 <= k-j < n, else 0
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            # Bisection on x in [max(0, x_old*0.5), min(1, x_old*2)]
            lo = max(0.0, x_old * 0.3)
            hi = min(1.0, x_old * 3.0 + 0.1)

            for _ in range(45):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))

                k = k_max
                b_kj = b[k - j] if (0 <= k - j < n) else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-14:
                    hi = mid
                elif deriv < -1e-14:
                    lo = mid
                else:
                    break

            x_new = (lo + hi) / 2.0

            if abs(x_new - x_old) < 1e-14:
                continue

            # Apply update
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue

            C_aa = fast_convolve(a)
            obj = C_aa.max()

            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
                if verbose and passes <= 2:
                    print(f"    CD pass {passes}, coord {j}: obj={best_obj:.10f}")

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
            else:
                break

    return best_a, best_obj


def subgradient_phase(a0, time_limit, lr=0.008, mom=0.9, top_k=50, temp=0.0005):
    """Projected subgradient descent with momentum and adaptive temperature."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    C_aa = fast_convolve(a)
    best_obj = C_aa.max()
    best_a = a.copy()

    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(C_aa))
        peaks = np.argpartition(C_aa, -k_actual)[-k_actual:]

        pv = C_aa[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)

        C_aa_new = fast_convolve(a_new)
        obj_new = C_aa_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        C_aa = C_aa_new
        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 200:
            rng = np.random.default_rng(it * 13 + 5)
            noise_scale = 0.002 + 0.003 * min(no_improve // 100, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            C_aa = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0

    return best_a, best_obj
