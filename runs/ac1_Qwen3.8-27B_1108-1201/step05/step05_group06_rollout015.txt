"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with log-sum-exp
smoothing and annealing, combined with exact coordinate bisection polishing.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_g(a, p):
    """Compute g_j = sum_k p_k * a_{k-j} for j=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    # Need convolution of a_rev (length n) and p (length 2n-1)
    # Result has length n + 2n - 2 = 3n - 2
    # We want elements at indices n-1 to 2n-2 (inclusive), which is n elements
    total_len = 3 * n - 2
    fft_size = 1 << (total_len - 1).bit_length()
    R = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(R * P, fft_size)
    return conv_full[n-1 : 2*n-1]

def softmax_obj_and_grad(x, T):
    """
    Compute objective and gradient for the softmax-parameterized problem.
    x is the log-parameter, a = softmax(x), objective = 2n * T * log(sum exp(c/T)).
    """
    n = len(x)
    a = np.exp(x - x.max())
    a /= a.sum()
    
    c = fast_convolve(a)
    max_c = c.max()
    p_unnorm = np.exp((c - max_c) / T)
    Z = p_unnorm.sum()
    p = p_unnorm / Z
    
    # Objective: 2n * T * log(sum exp(c/T)) = 2n * (max_c + T * log(Z))
    obj = 2.0 * n * (max_c + T * np.log(Z))
    
    # Gradient
    g = compute_g(a, p)
    dot = np.dot(a, g)
    grad_x = 4.0 * n * a * (g - dot)
    
    return obj, grad_x

def lbfgs_softmax(a0, T, max_iter=5000, tol=1e-12):
    """Run L-BFGS on the softmax-parameterized problem with temperature T."""
    n = len(a0)
    x0 = np.log(a0 + 1e-30)  # Convert from a to x
    
    result = minimize(
        softmax_obj_and_grad,
        x0,
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'gtol': tol, 'ftol': 1e-14, 'maxfun': 100000}
    )
    
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt /= a_opt.sum()
    
    return a_opt, result

def anneal_lbfgs(a0, T_list, max_iter=3000, verbose=False):

    a = a0.copy()
    for i, T in enumerate(T_list):
        a, result = lbfgs_softmax(a, T, max_iter=max_iter)

    return a

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coordinate_bisection(a0, time_limit, verbose=False, max_passes=200):

    a = a0.copy().astype(np.float64)

    n = len(a)

        passes += 1

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

When we fix all coordinates except `a_j` and optimize `a_j`, the max of the convolution is:

`max_k [ (1-x)^2 * B_k + 2*x*(1-x) * b_{k-j} + x^2 * 1_{k=2j} ]`

where `B = b*b` (b is a with a_j removed and rescaled), and `b_{k-j}` is the valid part of b shifted by j.

This is a max of quadratic functions in x, each of the form `(1-x)^2*B_k + 2*x*(1-x)*b_{k-j} + x^2*1_{k=2j}`.

Each individual quadratic is: `B_k - 2*B_k*x + B_k*x^2 + 2*b_{k-j}*x - 2*b_{k-j}*x^2 + x^2*1_{k=2j}`
`= B_k + 2*(b_{k-j} - B_k)*x + (B_k - 2*b_{k-j} + 1_{k=2j})*x^2`

The max of these quadratics is a convex function of x (since max of convex functions is convex). So the bisection approach is valid: we find the minimum of this convex function.

The derivative of the k-th quadratic at x is:
`2*(b_{k-j} - B_k) + 2*(B_k - 2*b_{k-j} + 1_{k=2j})*x`

At the optimum, the derivative of the active constraint (the one achieving the max) is zero. If no single constraint is active (i.e., the max is achieved at the "kink" where two constraints cross), then we need to find the crossing point.

The bisection approach handles this correctly: at each step, we find which k achieves the max, then check the derivative of that k's quadratic. If positive, move left; if negative, move right.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with log-sum-exp smoothing
and geometric annealing, combined with exact coordinate bisection polishing,
symmetric variant exploration, and multi-length search.

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_g(a, p):
    """Compute g_j = sum_k p_k * a_{k-j} for j=0..n-1 using FFT."""
    n = len(a)
    a_rev = a[::-1]
    total_len = 3 * n - 2
    fft_size = 1 << (total_len - 1).bit_length()
    R = np.fft.rfft(a_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(R * P, fft_size)
    return conv_full[n - 1:2 * n - 1]


def softmax_obj_and_grad(x, T):
    """
    Compute objective and gradient for softmax-parameterized problem.
    a = softmax(x), objective = 2n * T * log(sum exp(c/T)) where c = a*a.
    """
    n = len(x)
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    a = a / a_sum

    c = fast_convolve(a)
    max_c = c.max()
    p_unnorm = np.exp((c - max_c) / T)
    Z = p_unnorm.sum()
    p = p_unnorm / Z

    obj = 2.0 * n * (max_c + T * np.log(Z))

    g = compute_g(a, p)
    dot = np.dot(a, g)
    grad_x = 4.0 * n * a * (g - dot)

    return obj, grad_x


def lbfgs_single(a0, T, max_iter=5000, gtol=1e-10):
    """Run L-BFGS on softmax-parameterized problem at fixed temperature T."""
    n = len(a0)
    x0 = np.log(a0 + 1e-30)

    result = minimize(
        softmax_obj_and_grad,
        x0,
        method='L-BFGS-B',
        jac=True,
        args=(T,),
        options={'maxiter': max_iter, 'gtol': gtol, 'ftol': 1e-15,
                 'maxfun': 100000, 'iprint': -1}
    )

    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt /= a_opt.sum()
    return a_opt


def anneal_lbfgs(a0, T_list, max_iter=3000, verbose=False):
    """Run L-BFGS with geometric annealing over temperature list."""
    a = a0.copy()
    n = len(a)

    return a


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_bisection(a0, time_limit, verbose=False, max_passes=500, noise_scale=0.0003):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0
            x_new = x_old

def symmetrize(a):
    """Make sequence palindromic."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        T_list2 = np.geomspace(max_c * 8.0, max_c * 0.01, 30)
        a_opt = anneal_lbfgs(base4, T_list2, max_iter=3000, verbose=False)
        best_a, best_score, improved = trial_update(a_opt, best_a, best_score)
        if improved:
            print(f"  Aggressive T: {best_score:.6f}")
            rem = deadline - time.monotonic() - 350
            if rem > 10:
                t_alloc = min(rem, 80)
                a_opt, _ = coordinate_bisection(a_opt, time_limit=t_alloc)
                best_a, best_score, improved = trial_update(a_opt, best_a, best_score)
                if improved:
                    print(f"  Aggressive T + CD: {best_score:.6f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    base_a = best_a if best_a is not None else parent

    target_lengths = [640, 896, 1024, 1280]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 250
        if rem < 30:
            break
        f_interp = interpolate_to_length(base_a, target_n)
        c = fast_convolve(f_interp)
        max_c = c.max()
        T_list = np.geomspace(max_c * 8.0, max_c * 0.03, 20)

        a_opt = anneal_lbfgs(f_interp, T_list, max_iter=1500, verbose=False)
        best_a, best_score, improved = trial_update(a_opt, best_a, best_score)
        if improved:
            print(f"  n={target_n} L-BFGS: {best_score:.6f}")

        rem = deadline - time.monotonic() - 250
        if rem > 10:
            t_alloc = min(rem, 60)
            a_opt, _ = coordinate_bisection(a_opt, time_limit=t_alloc)
            best_a, best_score, improved = trial_update(a_opt, best_a, best_score)
            if improved:
                print(f"  n={target_n} CD: {best_score:.6f}")

    # === Phase 6: Perturbation + re-optimize ===
    print(f"\n=== Phase 6: Perturbation restarts (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        perturbations = [
            ("noise_001", 0.001),
            ("noise_003", 0.003),
            ("noise_005", 0.005),
            ("rev_mix", None),
        ]

        for idx, (label, scale) in enumerate(perturbations):
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break

            if scale is not None:
                rng_p = np.random.default_rng(seed + 5000 + idx * 13)
                f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * scale)
            else:
                f_pert = project_simplex(0.5 * best_a + 0.5 * best_a[::-1])

            c = fast_convolve(f_pert)
            max_c = c.max()
            T_list = np.geomspace(max_c * 5.0, max_c * 0.05, 15)

            a_opt = anneal_lbfgs(f_pert, T_list, max_iter=1500, verbose=False)
            best_a, best_score, improved = trial_update(a_opt, best_a, best_score)
            if improved:
                print(f"  {label}: {best_score:.6f}")
                rem = deadline - time.monotonic() - 150
                if rem > 10:
                    t_alloc = min(rem, 40)
                    a_opt, _ = coordinate_bisection(a_opt, time_limit=t_alloc)
                    best_a, best_score, improved = trial_update(a_opt, best_a, best_score)
                    if improved:
                        print(f"  {label} + CD: {best_score:.6f}")

    # === Phase 7: Final intensive CD ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 400)
            print(f"  Final CD, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_bisection(best_a, time_limit=t_alloc, verbose=True,
                                            noise_scale=0.0002, max_passes=1000)
            best_a, best_score, improved = trial_update(a_opt, best_a, best_score)
            if improved:
                print(f"  Final CD: {best_score:.6f}")

        # Micro perturbations + quick CD
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            n_best = len(best_a)
            for p in range(5):
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                rng_p = np.random.default_rng(seed + 9000 + p * 31)
                f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.0004)
                t_alloc = min(rem, 8)
                a_opt, _ = coordinate_bisection(f_pert, time_limit=t_alloc, max_passes=50)
                best_a, best_score, improved = trial_update(a_opt, best_a, best_score)
                if improved:
                    print(f"  Micro {p}: {best_score:.6f}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))
        print(f"Fallback to parent: {best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    print(f"{'='*50}")
    return [float(x) for x in best_a]
