"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-Adam with progressive beta annealing and restarts
2. Peak-targeting subgradient for high-beta regime
3. Coordinate descent polishing
4. Diverse restarts to escape local minima
Key: uses vectorized gradient via Toeplitz matrix-vector product,
Adam optimizer for per-coordinate adaptive step sizes.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(f, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*conv))) w.r.t. f.
    Returns (grad, conv, w) where w is the softmax weight over lags.
    """
    n = len(f)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    # grad[i] = 2 * sum_l w[i+l] * f[l]
    i_idx = np.arange(n)[:, None]
    l_idx = np.arange(n)[None, :]
    w_mat = w[i_idx + l_idx]
    grad = 2.0 * (w_mat @ f)
    return grad, conv, w


def compute_peak_grad(f, n_peaks=10):
    """Subgradient of max(conv) using top-k peaks.
    Returns (grad, conv).
    """
    n = len(f)
    conv = fast_convolve(f)
    k = min(n_peaks, len(conv))
    top_idx = np.argpartition(conv, -k)[-k:]
    # Soft weight: exp of scaled peak values
    pv = conv[top_idx]
    pv_max = pv.max()
    pv_min = pv.min()
    if pv_max - pv_min > 1e-15:
        w = np.exp((pv - pv_max) / (pv_max - pv_min + 1e-15))
    else:
        w = np.ones(k)
    w /= w.sum()
    w_full = np.zeros(len(conv))
    w_full[top_idx] = w
    i_idx = np.arange(n)[:, None]
    l_idx = np.arange(n)[None, :]
    w_mat = w_full[i_idx + l_idx]
    grad = 2.0 * (w_mat @ f)
    return grad, conv


def lse_adam(f0, time_limit, beta_start, beta_end, beta_rate,
             lr=0.01, betas=(0.9, 0.999), eps=1e-8,
             restart_iter=50, n_peaks=None, verbose=True):
    """LSE-Adam optimization with progressive beta.
    If n_peaks is set, uses peak-targeting subgradient instead of full LSE.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta = beta_start
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t += 1

        if n_peaks is not None:
            grad, conv = compute_peak_grad(f, n_peaks=n_peaks)
        else:
            grad, conv, _ = compute_lse_grad(f, beta)

        gn = np.linalg.norm(grad)
        if gn < 1e-18:
            no_improve += 100
            continue

        m = betas[0] * m + (1 - betas[0]) * grad
        v = betas[1] * v + (1 - betas[1]) * grad * grad
        m_hat = m / (1 - betas[0] ** t)
        v_hat = v / (1 - betas[1] ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= restart_iter:
                # Restart with perturbation
                rng = np.random.default_rng((it * 7919 + 13) % 999983)
                scale = 0.001 + 0.002 * min(1.0, no_improve / (restart_iter * 3))
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0
                if verbose:
                    print(f"    Restart it={it}, obj={fast_convolve(f).max():.10f}, scale={scale:.5f}")

        # Progressive beta increase
        if n_peaks is None and beta < beta_end:
            beta *= beta_rate

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_peaks=5, verbose=True):
    """Coordinate descent: adjust pairs (i, j) to reduce peak.
    For each step, find the (i, j, delta) that most reduces the current peak
    while maintaining sum(f)=1 and f>=0.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    peak_k = int(np.argmax(conv))
    best_obj = conv[peak_k]
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    # Precompute which indices affect the current peak
    # conv[peak_k] = sum_i f[i]*f[peak_k-i] (valid indices only)

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        if peak_val < best_obj - 1e-12:
            best_obj = peak_val
            best_f = f.copy()
            no_improve = 0

        # Find the most "offending" pair for the peak
        # conv[peak_k] = sum_i f[i] * f[peak_k - i]
        # For each i in valid range, the "other" index is j = peak_k - i
        # If i != j: contribution is f[i]*f[j], we can reduce by moving mass away
        # If i == j: contribution is f[i]^2, minimize by reducing f[i]

        # Strategy: find the pair (i, j) with largest f[i]*f[j] contribution
        # and transfer mass from the larger to a "safe" location
        valid_i = np.arange(max(0, peak_k - n + 1), min(n, peak_k + 1))
        valid_j = peak_k - valid_i
        contribs = f[valid_i] * f[valid_j]

        if len(contribs) == 0:
            break

        # Find the coordinate with highest individual contribution to peak
        # f[i] appears in multiple terms; total sensitivity:
        # sens[i] = sum_{k} w[k] * (f[k-i] if valid)
        # For single peak: sens[i] = f[peak_k - i] if 0 <= peak_k-i < n, else 0
        sens = np.zeros(n)
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        if lo <= hi:
            sens[hi - lo + 1 - (hi - lo) :]  # this is getting complex
            # Direct: for i in [lo, hi], sens[i] = f[peak_k - i]
            sens[lo:hi+1] = f[peak_k - np.arange(lo, hi+1)]

        # The term where i = peak_k - i (i.e., 2*i = peak_k) has double weight
        if peak_k % 2 == 0 and 0 <= peak_k // 2 < n:
            sens[peak_k // 2] *= 2.0

        # Find coordinate with highest sensitivity (most reduces peak if decreased)
        # and coordinate with lowest sensitivity (safe to increase)
        i_dec = int(np.argmax(sens))
        # For i_inc, we want low sensitivity AND we want to move mass there
        # such that it doesn't create new peaks
        i_inc_pool = np.where(sens < np.percentile(sens, 25))[0]
        if len(i_inc_pool) == 0:
            i_inc_pool = np.where(sens == sens.min())[0]

        # Try transferring mass from i_dec to several candidates
        best_delta = 0
        best_new_obj = peak_val
        best_j = i_dec

        # Binary search for optimal delta (limited by f[i_dec])
        f_dec = f[i_dec]
        if f_dec < 1e-15:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng((it * 31 + 7) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue

        for j in i_inc_pool[:min(20, len(i_inc_pool))]:
            if j == i_dec:
                continue
            # Try delta = f_dec * frac for various frac
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
                delta = f_dec * frac
                f_try = f.copy()
                f_try[i_dec] -= delta
                f_try[j] += delta
                if f_try[i_dec] < 0 or f_try[j] < 0:
                    continue
                conv_try = fast_convolve(f_try)
                obj_try = conv_try.max()
                if obj_try < best_new_obj:
                    best_new_obj = obj_try
                    best_delta = delta
                    best_j = j

        if best_delta > 0:
            f[i_dec] -= best_delta
            f[best_j] += best_delta
            f = project_simplex(f)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                # Try random direction
                rng = np.random.default_rng((it * 104729 + 3) % 999983)
                direction = rng.standard_normal(n)
                direction[i_dec] -= 1.0  # bias toward reducing peak coordinate
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    f_try = project_simplex(f + frac * direction / np.linalg.norm(direction))
                    conv_try = fast_convolve(f_try)
                    if conv_try.max() < best_obj - 1e-12:
                        f = f_try
                        best_obj = conv_try.max()
                        best_f = f.copy()
                        no_improve = 0
                        break
                if no_improve > 80:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    no_improve = 0

    return best_f, best_obj

def symmetric_init(n, freq=2.0, amplitude=0.1):
    """Symmetric sinusoidal perturbation of uniform."""
    base = np.ones(n) / n
    x = np.arange(n)
    sym = np.cos(2.0 * np.pi * freq * x / n)
    sym_full = np.concatenate([sym[:n//2], sym[n//2-1::-1]]) if n > 2 else np.ones(n)
    # Make it length n
    if len(sym_full) > n:
        sym_full = sym_full[:n]
    elif len(sym_full) < n:
        sym_full = np.concatenate([sym_full, sym_full[::-1][:(n-len(sym_full))]])
    f = base + amplitude * sym_full
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f


def diverse_init(n, seed=42):
    """Create a diverse random initialization."""
    rng = np.random.default_rng(seed)
    # Mix of uniform and peaked
    f1 = np.ones(n) / n
    f2 = np.zeros(n)
    center = n // 2
    width = n // 4
    for i in range(n):
        f2[i] = np.exp(-0.5 * ((i - center) / width) ** 2)
    f2 /= f2.sum()
    f = (1 - 0.5) * f1 + 0.5 * f2
    f += rng.standard_normal(n) * 0.02
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f

        try_sequence(f_opt, best_tracker)
        current = best_tracker[1] if best_tracker[1] is not None else current

        # CD
        remaining = deadline - 30 - time.monotonic()
        if remaining > 15:
            f_opt, _ = coordinate_descent(current, time_limit=max(10, remaining - 5),
                                          n_peaks=3, verbose=False)
            try_sequence(f_opt, best_tracker)
            current = best_tracker[1] if best_tracker[1] is not None else current

    # === Phase 6: Try different initializations for fresh starts ===
    print(f"\n=== Phase 6: Diverse restarts ===")
    if time.monotonic() < deadline - 50:
        remaining = deadline - 50 - time.monotonic()
        if remaining > 60:
            t_each = remaining / 3
            for s in [123, 456, 789]:
                if time.monotonic() > deadline - 40:
                    break
                t_alloc = min(t_each, deadline - 40 - time.monotonic())
                if t_alloc < 20:
                    break
                f_div = diverse_init(n, seed=s)
                # Quick LSE
                f_opt, _ = lse_adam(f_div, time_limit=t_alloc,
                                    beta_start=50.0, beta_end=2000.0, beta_rate=1.01,
                                    lr=0.01, restart_iter=30, verbose=False)
                try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using fallback")
    fallback = parent if parent is not None else np.ones(768) / 768
    return [float(x) for x in fallback]
