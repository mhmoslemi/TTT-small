Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exponentiated gradient (multiplicative weights) with adaptive peak targeting
2. Frequency-domain Gerchberg-Saxton for big-step escapes
3. LSE gradient descent for fine polishing
4. Multi-length exploration

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad_peaks(f, peak_positions):
    """Compute average subgradient at given peak positions."""
    n = len(f)
    g = np.zeros(n)
    for p in peak_positions:
        p = int(p)
        # f[p-m] for valid m
        m_lo = max(0, p - n + 1)
        m_hi = min(n - 1, p)
        if m_lo <= m_hi:
            ms = np.arange(m_lo, m_hi + 1)
            g[ms] += f[p - ms]
        # f[m-p] for valid m
        m2_lo = max(0, p)
        m2_hi = min(n - 1, p + n - 1)
        if m2_lo <= m2_hi:
            ms2 = np.arange(m2_lo, m2_hi + 1)
            g[ms2] += f[ms2 - p]
    g /= len(peak_positions)
    return g


def exp_gradient_descent(f0, time_limit, n_peaks=10, eta0=0.5, decay=0.001):
    """Exponentiated gradient method with adaptive peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_eta = eta0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        # Select top peaks
        k = min(n_peaks, len(conv))
        peak_idx = np.argpartition(conv, -k)[-k:]
        # Weight peaks by their value (softer selection)
        pv = conv[peak_idx]
        weights = np.exp(pv / (pv.max() * 0.1 + 1e-15))
        weights /= weights.sum()
        
        # Compute weighted subgradient
        g = np.zeros(n)
        for i, p in enumerate(peak_idx):
            p = int(p)
            # f[p-m]
            m_lo = max(0, p - n + 1)
            m_hi = min(n - 1, p)
            if m_lo <= m_hi:
                ms = np.arange(m_lo, m_hi + 1)
                g[ms] += weights[i] * f[p - ms]
            # f[m-p]
            m2_lo = max(0, p)
            m2_hi = min(n - 1, p + n - 1)
            if m2_lo <= m2_hi:
                ms2 = np.arange(m2_lo, m2_hi + 1)
                g[ms2] += weights[i] * f[ms2 - p]
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Exponentiated update
        scaled = cur_eta * g / g_norm
        # Shift for numerical stability
        scaled -= scaled.max()
        f_new = f * np.exp(scaled)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f.copy()
        else:
            f_new /= s_new
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            cur_eta = min(cur_eta * 1.01, 5.0)
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                cur_eta *= 0.97
            if no_improve > 100:
                # Restart with noise
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.005 + 0.01 * min(1.0, (no_improve - 100) / 200)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                f /= f.sum()
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
                cur_eta = max(eta0 * 0.5, 0.01)
                no_improve = 0
        
        # Increase number of peaks over time
        if it % 500 == 0:
            n_peaks = min(n_peaks + 2, 50)
    
    return best_f, best_obj


def freq_gs(f0, time_limit, n_steps=50, threshold_quantile=0.5):
    """Frequency-domain Gerchberg-Saxton for peak reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit and it < n_steps:
        it += 1
        
        F = np.fft.fft(f, fft_size)
        power = np.abs(F) ** 2
        c = np.fft.ifft(power).real
        
        # Modify c to reduce peaks
        # Use a soft threshold
        c_mean = c[:2*n-1].mean()
        c_max = c[:2*n-1].max()
        threshold = c_mean + 0.5 * (c_max - c_mean)
        
        # Soft reduction: anything above threshold gets pulled down
        c_mod = c.copy()
        above = c_mod > threshold
        c_mod[above] = threshold + (c_mod[above] - threshold) * 0.3
        
        # Ensure non-negative (c should be non-negative but numerically might not be)
        c_mod = np.maximum(c_mod, 0)
        
        # New power spectrum
        new_power = np.fft.fft(c_mod).real
        new_power = np.maximum(new_power, 0)
        
        # Update magnitude, keep phase
        new_abs = np.sqrt(new_power)
        phase = np.angle(F)
        F_new = new_abs * np.exp(1j * phase)
        
        # New f
        f_new = np.fft.ifft(F_new).real[:n]
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new > 1e-15:
            f_new /= s_new
        else:
            f_new = best_f.copy()
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        elif obj_new < best_obj + 0.1 * (best_obj - c_mean):
            # Accept if not too bad (for exploration)
            f = f_new
        else:
            # Reject, try with stronger reduction
            f = best_f.copy()
            if it > n_steps // 2:
                break
    
    return best_f, best_obj


def lse_optimize(f0, time_limit, beta_start=50.0, beta_end=10000.0, max_peaks=200):
    """Log-sum-exp smoothed gradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 3.0 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Compute gradient using top peaks only
        k = min(max_peaks, len(conv))
        sig_idx = np.argpartition(w, -k)[-k:]
        sig_idx = sig_idx[w[sig_idx] > 1e-10 * w.max()]
        
        grad = np.zeros(n)
        for ki in sig_idx:
            wi = w[ki]
            ki = int(ki)
            m_lo = max(0, ki - n + 1)
            m_hi = min(n - 1, ki)
            if m_lo <= m_hi:
                ms = np.arange(m_lo, m_hi + 1)
                grad[ms] += wi * f[ki - ms]
            m2_lo = max(0, ki)
            m2_hi = min(n - 1, ki + n - 1)
            if m2_lo <= m2_hi:
                ms2 = np.arange(m2_lo, m2_hi + 1)
                grad[ms2] += wi * f[ms2 - ki]
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad /= grad_norm
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 5.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.94
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 80:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(2.0 / n, 0.001)
                beta = min(beta * 1.5, beta_end)
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.003
    
    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_peak_subgrad(f, peaks, weights=None):
    """Compute weighted sum of subgradients at given peak positions."""
    n = len(f)
    g = np.zeros(n)
    for i, p in enumerate(peaks):
        p = int(p)
        w = weights[i] if weights is not None else 1.0
        if w < 1e-15:
            continue
        # f[p-m] for valid m: 0<=p-m<n => p-n+1<=m<=p
        m_lo = max(0, p - n + 1)
        m_hi = min(n - 1, p)
        if m_lo <= m_hi:
            ms = np.arange(m_lo, m_hi + 1)
            g[ms] += w * f[p - ms]
        # f[m-p] for valid m: 0<=m-p<n => p<=m<=p+n-1
        m2_lo = max(0, p)
        m2_hi = min(n - 1, p + n - 1)
        if m2_lo <= m2_hi:
            ms2 = np.arange(m2_lo, m2_hi + 1)
            g[ms2] += w * f[ms2 - p]
    return g

def freq_gs(f0, time_limit, n_steps=200, reduction=0.4):

    Alternates between:

    - Frequency domain: modify |F|^2 to reduce peaks in ifft(|F|^2)

    it = 0

        it += 1

        power = np.abs(F) ** 2
        c = np.fft.ifft(power).real

        # Cyclic convolution wraps around; for N >= 2n-1 the linear part is c[0:n]
        # and the "negative lag" part is c[n:N] (wrapped)

        # Let's just work with c[0:2n-1] as the auto-conv (padded)

        auto_conv[n:] = c[N - (2 * n - 1 - n):N][::-1] if False else c[N - (2*n-2):N][::-1]
        # Simpler: just use c directly, the peak is at c[0]
        c_work = c.copy()

        # Find the effective peak locations in the auto-convolution
        # c[0] is lag 0, c[1:n-1] are positive lags, c[n:N-1] are negative lags (wrapped)
        # For peak finding, consider all of c
        c_max = c.max()
        c_mean = c.mean()

        # Let's not enforce this strictly, just ensure non-negativity

def lse_optimize(f0, time_limit, beta_start=30.0, beta_end=8000.0, max_peaks=250):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 4.0 / n
    v = np.zeros(n)  # Nesterov momentum
    gamma = 0.9  # momentum coefficient
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_look = project_simplex(f - gamma * v)
        conv = fast_convolve(f_look)

        # LSE weights
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Compute gradient at top peaks only
        k = min(max_peaks, len(conv))
        sig_idx = np.argpartition(w, -k)[-k:]
        sig_mask = w[sig_idx] > 1e-10 * w.max()
        sig_idx = sig_idx[sig_mask]
        sig_w = w[sig_idx]

        g = compute_peak_subgrad(f_look, sig_idx, sig_w)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Update
        v_new = gamma * v - lr * g
        f_new = project_simplex(f + v_new)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()

            no_improve = 0
            lr = min(lr * 1.003, 8.0 / n)

                beta = min(beta * 1.4, beta_end)

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def cosine_init(n, order=1):
    """Cosine-based initialization with order parameter."""
    x = np.arange(n)
    f = np.ones(n)
    for k in range(1, order + 1):
        f += 0.3 / k * np.cos(2.0 * np.pi * k * x / n)
    f = np.maximum(f, 0.01)
    f /= f.sum()
    return f

    # Very high beta LSE for final squeeze
    t_alloc = min(100, deadline - time.monotonic() - 30)
    if t_alloc > 15:
        print(f"  Final LSE n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                 beta_start=500.0, beta_end=20000.0, max_peaks=400)
        try_sequence(f_opt, best_tracker)

    # Final EG with very small eta for micro-adjustments
    current = best_tracker[1].copy() if best_tracker[1] is not None else current
    t_alloc = min(60, deadline - time.monotonic() - 15)
    if t_alloc > 10:
        print(f"  Final EG micro n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = exp_gradient_descent(current, time_limit=t_alloc,
                                         n_peaks=30, eta0=0.05, eta_max=0.5)
        try_sequence(f_opt, best_tracker)

    # Try one nearby length with LSE
    if best_tracker[1] is not None:
        n_best = len(best_tracker[1])
        for off in [32, -32]:
            n_try = n_best + off
            if n_try < 128 or n_try > 1200:
                continue
            t_remaining = deadline - time.monotonic() - 10
            if t_remaining < 20:
                break
            t_alloc = min(40, t_remaining * 0.5)
            if t_alloc < 10:
                break
            f_init = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Final try n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize(f_init, time_limit=t_alloc,
                                     beta_start=50.0, beta_end=3000.0, max_peaks=200)
            try_sequence(f_opt, best_tracker)

    # === Final verification and return ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(448) / 448
    return [float(x) for x in fallback]
