import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_obj_gradient(a, T, num_peaks=200):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a).
    Uses top num_peaks values with softmax weighting.
    Returns (smoothed_value, gradient)
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Select top peaks
    k = min(num_peaks, len(conv))
    peak_idx = np.argpartition(conv, -k)[-k:]
    peak_val = conv[peak_idx]
    
    # Log-sum-exp
    max_pv = peak_val.max()
    log_sum_exp = max_pv + np.log(np.sum(np.exp(peak_val - max_pv)))
    # Softmax weights
    weights = np.exp(peak_val - max_pv)
    weights /= weights.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with O(n) pairwise convolution updates (no FFT per eval)
2. Log-sum-exp smoothed Adam optimizer for initial shaping
3. Peak-targeted coordinate descent
4. Multi-length exploration with interpolation
Key innovation: exact O(n) formula for conv change under pair transfers, enabling
~100K+ evaluations/second for coordinate descent phases.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_pair_delta(conv, a, i, j, di):
    """
    Compute delta_conv when a[i] += di and a[j] -= di (preserving sum).
    Formula: delta[k] = 2*di*(a[k-i] - a[k-j]) + di^2*(I(k==2i) - 2*I(k==i+j) + I(k==2j))
    where a[k-i] is 0 if k-i is out of [0, n-1].
    Returns the full new conv (conv + delta).
    """
    n = len(a)
    size = 2 * n - 1
    delta = np.zeros(size)
    
    # Linear terms: 2*di*a[k-i] for k in [i, i+n-1], -2*di*a[k-j] for k in [j, j+n-1]
    k_lo1 = i
    k_hi1 = i + n - 1
    delta[k_lo1:k_hi1 + 1] += 2.0 * di * a[0:n]
    
    k_lo2 = j
    k_hi2 = j + n - 1
    delta[k_lo2:k_hi2 + 1] -= 2.0 * di * a[0:n]
    
    # Quadratic corrections
    if 0 <= 2 * i < size:
        delta[2 * i] += di * di
    if 0 <= i + j < size:
        delta[i + j] -= 2.0 * di * di
    if 0 <= 2 * j < size:
        delta[2 * j] += di * di
    
    return conv + delta


def adam_smoothed(a0, time_limit, verbose=False):

            if f[i] < 1e-12:
                continue
            
            # Build candidate list for j
            candidates = []
            # Neighbors
            for d in [1, -1, 2, -2, 3, -3]:
                jj = (i + d) % n
                if jj != i:
                    candidates.append(jj)
            # Based on convolution structure: positions that contribute to the peak
            peak_k = int(np.argmax(conv))
            j_lo = max(0, peak_k - n + 1)
            j_hi = min(n - 1, peak_k)
            # Random positions in the peak region
            for _ in range(3):
                jj = int(rng.integers(j_lo, j_hi + 1))
                if jj != i:
                    candidates.append(jj)
            # Random
            for _ in range(2):
                jj = int(rng.integers(n))
                if jj != i:
                    candidates.append(jj)
            
            for j in candidates[:8]:
                j = int(j)
                if j == i:
                    continue
                
                # Binary search for optimal di in [0, f[i]]
                # We transfer di from i to j: f[i] -= di, f[j] += di
                lo = 0.0
                hi = f[i]
                
                # Quick check: current state
                cur_max = conv.max()
                
                # Try a few strategic values first
                best_di = 0.0
                best_val = cur_max
                
                # Midpoint and boundaries
                test_dis = [hi * 0.5, hi * 0.3, hi * 0.7, hi * 0.1, hi * 0.9]
                for di in test_dis:
                    if di <= 0 or di >= hi:
                        continue
                    new_conv = conv_pair_delta(conv, f, i, j, -di)
                    val = new_conv.max()
                    total_evals += 1
                    if val < best_val:
                        best_val = val
                        best_di = di
                
                # If we found improvement, do binary search to refine
                if best_di > 0 and best_val < cur_max - 1e-13:
                    # Binary search in [lo, hi] around best_di
                    bl = max(0.0, best_di - hi * 0.2)
                    bh = min(hi, best_di + hi * 0.2)
                    for _ in range(6):
                        mid = (bl + bh) / 2
                        if mid <= 0 or mid >= hi:
                            break
                        new_conv = conv_pair_delta(conv, f, i, j, -mid)
                        val = new_conv.max()
                        total_evals += 1
                        # Also check one more point
                        mid2 = (bl + mid) / 2 if mid > bl else bl + 0.01 * hi
                        new_conv2 = conv_pair_delta(conv, f, i, j, -min(mid2, hi))
                        val2 = new_conv2.max()
                        total_evals += 1
                        
                        if val < best_val:
                            best_val = val
                            best_di = mid
                            bh = mid
                        elif val2 < best_val:
                            best_val = val2
                            best_di = min(mid2, hi)
                            bh = min(mid2, hi)
                        else:
                            bl = mid
                    
                    # Apply best transfer
                    f[i] -= best_di
                    f[j] += best_di
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    conv = fast_convolve(f)
                    best_obj = conv.max()
                    best_f = f.copy()
                    round_improved = True
        
        if round_improved:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1
            if verbose:
                print(f"    CD round {rounds}: no improvement, best={best_obj:.10f}")
    
    if verbose:
        print(f"    CD done: rounds={rounds}, evals={total_evals}, best={best_obj:.10f}")
    return best_f, best_obj

def peak_targeted_cd(a0, time_limit, verbose=False):
    """
    Coordinate descent that specifically targets reducing the convolution peak.
    Identifies contributors to the peak and tries to reduce them.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 200:
            break
        
        conv = fast_convolve(f)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]
        
        # Also check for near-peaks
        threshold = max_val * (1.0 - 1e-4)
        peak_positions = np.where(conv >= threshold)[0]
        
        improved = False
        
        # For each peak position, try to reduce it
        for pk in peak_positions[:5]:
            pk = int(pk)
            if improved:
                break
            
            # Contributors to this peak
            j_lo = max(0, pk - n + 1)
            j_hi = min(n - 1, pk)
            contributors = np.arange(j_lo, j_hi + 1)
            
            # Compute contributions: f[j] * f[pk-j]
            contribs = f[contributors] * f[pk - contributors]
            
            # Sort by contribution (largest first)
            order = np.argsort(contribs)[::-1]
            
            # Try reducing top contributors by transferring mass away
            for oi in order[:8]:
                if improved:
                    break
                ci = int(contributors[oi])
                if f[ci] < 1e-12:
                    continue
                
                # Find target: position that doesn't contribute to any peak
                # Simple heuristic: random position far from all peaks
                for attempt in range(15):
                    j = int(rng.integers(n))
                    if j == int(ci):
                        continue
                    # Check if j would create a new peak
                    # (skip this check for speed, just try)
                    
                    di = f[ci] * 0.5  # transfer half
                    new_conv = conv_pair_delta(conv, f, int(ci), j, -di)
                    new_max = new_conv.max()
                    
                    if new_max < best_obj - 1e-13:
                        # Refine: try smaller transfers
                        best_di = di
                        best_nm = new_max
                        for frac in [0.3, 0.4, 0.6, 0.7]:
                            di2 = f[ci] * frac
                            if di2 >= f[ci]:
                                continue
                            nc2 = conv_pair_delta(conv, f, int(ci), j, -di2)
                            nm2 = nc2.max()
                            if nm2 < best_nm:
                                best_nm = nm2
                                best_di = di2
                        
                        f[ci] -= best_di
                        f[j] += best_di
                        f = np.maximum(f, 0.0)
                        f = f / f.sum()
                        conv = fast_convolve(f)
                        best_obj = conv.max()
                        best_f = f.copy()
                        improved = True
                        break
        
        if not improved:
            # Global random pair transfer
            for _ in range(30):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or f[i] < 1e-10:
                    continue
                di = f[i] * (0.2 + 0.6 * rng.random())
                new_conv = conv_pair_delta(conv, f, i, j, -di)
                if new_conv.max() < best_obj - 1e-13:
                    f[i] -= di
                    f[j] += di
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    conv = fast_convolve(f)
                    best_obj = conv.max()
                    best_f = f.copy()
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    Peak-CD iter {it}: obj={best_obj:.10f}")
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def multi_pair_cd(a0, time_limit, verbose=False):
    """
    Advanced coordinate descent: each step tries multiple pairs simultaneously,
    using a smarter candidate selection based on gradient information.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0
    it = 0
    
    # Precompute which positions are "hot" (contribute to peaks)
    def get_hot_cold(f, conv, n, num_hot=50):
        """Return indices of hot (high contribution) and cold (low) positions."""
        peak_k = int(np.argmax(conv))
        # Hot: positions that appear in many peak convolutions
        hot_count = np.zeros(n)
        for k in range(max(0, peak_k - 5), min(2*n-1, peak_k + 6)):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                hot_count[j] += conv[k] * f[j] * f[k-j]
        
        hot = np.argsort(hot_count)[::-1][:num_hot]
        cold = np.argsort(hot_count)[:num_hot]
        return hot, cold
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 500:
            break
        
        conv = fast_convolve(f)
        improved = False
        
        # Get hot and cold positions
        hot, cold = get_hot_cold(f, conv, n, num_hot=30)
        
        # Try transfers from hot to cold
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = int(hot[int(rng.integers(len(hot)))])
            j = int(cold[int(rng.integers(len(cold)))])
            if i == j or f[i] < 1e-12:
                continue
            
            # Binary search for best di
            lo = 0.0
            hi = f[i] * 0.9
            best_di = 0.0
            best_val = conv.max()
            
            # Coarse search
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
                di = f[i] * frac
                if di <= 0 or di >= f[i]:
                    continue
                new_conv = conv_pair_delta(conv, f, i, j, -di)
                val = new_conv.max()
                if val < best_val:
                    best_val = val
                    best_di = di
            
            # Fine search
            if best_di > 0:
                bl = max(0.0, best_di - f[i] * 0.15)
                bh = min(f[i] * 0.95, best_di + f[i] * 0.15)
                for _ in range(5):
                    mid = (bl + bh) / 2
                    new_conv = conv_pair_delta(conv, f, i, j, -mid)
                    val = new_conv.max()
                    if val < best_val:
                        best_val = val
                        best_di = mid
                        bh = mid
                    else:
                        bl = mid
            
            if best_di > 0 and best_val < best_obj - 1e-13:
                f[i] -= best_di
                f[j] += best_di
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                conv = fast_convolve(f)
                best_obj = conv.max()
                best_f = f.copy()
                improved = True
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
    
    if verbose:
        print(f"    MultiPair-CD done: iters={it}, best={best_obj:.10f}")
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing 2*n*max(conv)/(sum)^2.
    Uses fast O(n) pairwise coordinate descent as the primary engine,
    complemented by smoothed Adam for initial shaping and multi-length search.
    """
    global height_sequence_1
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        try:
            sc = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        best_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")
    
    print(f"\n=== Phase 1: Adam smoothing (n={len(parent)}) ===")
    rem = deadline - time.monotonic() - 900
    t1 = min(rem, 250)
    if t1 > 30:
        a_opt, _ = adam_smoothed(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "adam_p1")
    
    # Use best so far for subsequent phases
    base = best_seq.copy()
    n_base = len(base)
    
    print(f"\n=== Phase 2: Fast coordinate descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 700
    t2 = min(rem, 350)
    if t2 > 30:
        print(f"  Time: {t2:.0f}s, n={n_base}")
        a_opt, _ = coord_descent_fast(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "cd_p2")
    
    print(f"\n=== Phase 3: Peak-targeted CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    t3 = min(rem, 250)
    if t3 > 30 and best_seq is not None:
        a_opt, _ = peak_targeted_cd(best_seq, time_limit=t3, verbose=True)
        try_seq(a_opt, "peak_cd_p3")
    
    print(f"\n=== Phase 4: Multi-pair CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 350
    t4 = min(rem, 200)
    if t4 > 30 and best_seq is not None:
        a_opt, _ = multi_pair_cd(best_seq, time_limit=t4, verbose=True)
        try_seq(a_opt, "multipair_p4")
    
    print(f"\n=== Phase 5: Multi-length exploration (best={best_score:.8f}) ===")
    for target_n in [640, 896, 1024]:
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break
        if target_n == len(best_seq) if best_seq is not None else False:
            continue
        
        f_interp = interpolate_sequence(best_seq, target_n)
        
        # Quick Adam
        t_a = min(rem * 0.3, 80)
        if t_a > 15:
            a_opt, _ = adam_smoothed(f_interp, time_limit=t_a, verbose=False)
            try_seq(a_opt, f"adam_n{target_n}")
        
        # CD
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_c = min(rem * 0.5, 120)
            a_opt, _ = coord_descent_fast(a_opt if 'a_opt' in dir() else f_interp, time_limit=t_c, verbose=False)
            try_seq(a_opt, f"cd_n{target_n}")
    
    print(f"\n=== Phase 6: Intensive final CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 30 and best_seq is not None:
        t6 = min(rem, 300)
        a_opt, _ = coord_descent_fast(best_seq, time_limit=t6, verbose=True)
        try_seq(a_opt, "cd_final")
    
    print(f"\n=== Phase 7: Peak-targeted final (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 20 and best_seq is not None:

        a_opt, _ = peak_targeted_cd(best_seq, time_limit=t7, verbose=False)
        try_seq(a_opt, "peak_final")
    
    print(f"\n=== Phase 8: Multi-pair final (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 20 and best_seq is not None:
        t8 = min(rem, 100)
        a_opt, _ = multi_pair_cd(best_seq, time_limit=t8, verbose=False)

    # Final Adam polish

        try_seq(a_opt, "adam_polish")
    
    # One more CD
    rem = deadline - time.monotonic() - 5
    if rem > 10 and best_seq is not None:
        a_opt, _ = coord_descent_fast(best_seq, time_limit=min(rem, 40), verbose=False)
        try_seq(a_opt, "cd_polish")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = min(best_score, fs)
    except Exception:
        pass
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
