def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_xy(f, g):
    """Linear convolution of f and g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n, top_k=None):
    """Solve LP: min t s.t. (f*g)[k] <= t for (top_k) peak positions, g >= 0, sum(g) = 1.
    Returns (g, t) or (None, None) on failure."""
    # First compute current convolution to identify peak positions
    conv_f = fast_convolve(f)
    if top_k is not None and top_k < len(conv_f):
        peak_positions = np.argpartition(conv_f, -top_k)[-top_k:]
        # Add a margin of positions around the peaks
        all_positions = set()
        for p in peak_positions:
            for delta in range(-2, 3):
                pos = int(p) + delta
                if 0 <= pos < len(conv_f):
                    all_positions.add(pos)
        constraint_positions = sorted(all_positions)
    else:
        constraint_positions = list(range(len(conv_f)))

    n_constraints = len(constraint_positions)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Build constraint matrix: (f*g)[k] - t <= 0 for each k in constraint_positions
    A_ub = np.zeros((n_constraints, n + 1))
    for idx, k in enumerate(constraint_positions):
        # (f*g)[k] = sum_j f[k-j] * g[j] for valid j
        # g[j] contributes f[k-j] to position k when 0 <= j < n and 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A_ub[idx, js:je + 1] = f[k - np.arange(js, je + 1)]
        A_ub[idx, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_softmax(f, n, weights):
    """Solve LP: min sum_k w_k * (f*g)[k] s.t. g >= 0, sum(g) = 1.
    Returns g or None on failure."""
    c = np.zeros(n)
    # (f*g)[k] = sum_j f[k-j] * g[j], so sum_k w_k * (f*g)[k] = sum_j g[j] * sum_k w_k * f[k-j]
    # The coefficient of g[j] is sum_k w_k * f[k-j]
    for j in range(n):
        # k ranges such that 0 <= k < 2n-1 and 0 <= k-j < n

        k_max = min(2 * n - 2, j + n - 1)
        if k_min <= k_max:
            ks = np.arange(k_min, k_max + 1)
            c[j] = np.dot(weights[k_min:k_max + 1], f[ks - j])

    A_eq[0, :] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n

    res = linprog(c, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 2
            if no_improve > 60:
                break
            continue

        # Line search with golden section refinement
        best_alpha = 0.0
        best_val = M

        # Coarse search
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search around best
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 2

        if no_improve > 40:
            # Restart with perturbation
            rng = np.random.default_rng(it * 6151 + 7)
            scale = 0.001 + 0.002 * min(no_improve // 40, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

        if no_improve > 120:
            break

    return best_f, best_obj


def sg_optimize(f0, time_limit, lr=0.003, momentum=0.94, top_k=40, temp=0.00005,
                verbose=False):
    """Subgradient method with adaptive softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def peak_cd_optimize(f0, time_limit, verbose=False):
    """Coordinate descent targeting coordinates that contribute to the current peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Find top peaks
        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Get contributing coordinates sorted by contribution
            contributions = []
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    contrib = f[j] * f[partner]
                    contributions.append((contrib, j))
            contributions.sort(reverse=True)

            # Try adjusting top contributors
            for contrib_val, j in contributions[:15]:
                if time.monotonic() - t0 > time_limit:
                    break
                if contrib_val < 1e-10:
                    continue

                old_val = f[j]
                best_j_val = old_val
                best_j_obj = best_obj

                for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / max(f_trial.sum(), 1e-15)

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-15:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / max(f.sum(), 1e-15)

                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)
                        break  # Move to next peak

        if not improved:
            no_improve += 1
            if no_improve > 2:
                # Random coordinate exploration
                rng = np.random.default_rng(passes * 31337 + 3)
                n_tries = min(50, n // 5)
                for _ in range(n_tries):
                    if time.monotonic() - t0 > time_limit:
                        break
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.3, 0.6, 1.4, 2.0, 0.1]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.4:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / max(f_trial.sum(), 1e-15)
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break
                    if improved:
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization with enhanced strategies:
    - Adaptive LP-FW with constraint reduction
    - Softmax subgradient with mirror steps
    - Peak-targeting coordinate descent
    - Multi-length exploration
    - Perturbation-based restarts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: LP-FW with min-peak on parent ===
    print(f"\n=== Phase 1: LP-FW (min-peak, adaptive) on n={base_n} ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.28)
    if t1 > 30:
        print(f"  Time budget: {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True,
                                  use_min_peak=True, top_k_peaks=25)
        try_seq(a_opt, "lpfw_phase1")

    # === Phase 2: LP-FW with softmax on best ===
    print(f"\n=== Phase 2: LP-FW (softmax) ===")
    rem = deadline - time.monotonic()
    t2 = min(200, rem * 0.15)
    if t2 > 25 and best_seq is not None:
        print(f"  n={len(best_seq)}, Time budget: {t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2,
                                  use_min_peak=False, top_k_peaks=30)
        try_seq(a_opt, "lpfw_softmax")

    # === Phase 3: Try nearby lengths ===
    print(f"\n=== Phase 3: Nearby lengths ===")
    base = best_seq if best_seq is not None else parent
    candidate_lengths = []
    for delta in [-128, -64, 0, 64, 128, 256]:
        ln = len(base) + delta
        if ln >= 256 and ln <= 2048 and ln != len(base):
            candidate_lengths.append(ln)
    # Add some power-of-2 options
    for pw in [512, 768, 832, 1024, 1152]:
        if 256 <= pw <= 2048 and pw != len(base):
            candidate_lengths.append(pw)
    candidate_lengths = sorted(set(candidate_lengths))

    for target_n in candidate_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.25, 180)
        if t_alloc < 20:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc,
                                  use_min_peak=True, top_k_peaks=25)
        try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_sg = min(rem * 0.25, 300)
        if t_sg > 30:
            print(f"  n={n_best}, Time budget: {t_sg:.0f}s")
            a_opt, _ = sg_optimize(best_seq, time_limit=t_sg,
                                   lr=0.003, momentum=0.94,
                                   top_k=max(30, n_best // 8),
                                   temp=0.00004)
            try_seq(a_opt, "sg_main")

            # Second SG with different params
            rem = deadline - time.monotonic()
            t_sg2 = min(rem * 0.15, 150)
            if t_sg2 > 20:
                a_opt, _ = sg_optimize(best_seq, time_limit=t_sg2,
                                       lr=0.001, momentum=0.95,
                                       top_k=max(20, n_best // 12),
                                       temp=0.00002)
                try_seq(a_opt, "sg_fine")

    # === Phase 5: Peak-targeting coordinate descent ===
    print(f"\n=== Phase 5: Peak CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.3, 250)
        if t_cd > 25:
            print(f"  n={len(best_seq)}, Time budget: {t_cd:.0f}s")
            a_opt, _ = peak_cd_optimize(best_seq, time_limit=t_cd)
            try_seq(a_opt, "peak_cd")

    # === Phase 6: Final polish with perturbations ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick SG polish
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = sg_optimize(best_seq, time_limit=min(rem, 60),
                                   lr=0.0008, momentum=0.95,
                                   top_k=max(15, n_best // 10),
                                   temp=0.00002)
            try_seq(a_opt, "sg_polish")

        # Quick LP-FW polish
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 50),
                                      use_min_peak=True, top_k_peaks=20)
            try_seq(a_opt, "lpfw_polish")

        # Perturbation-based restarts
        for p in range(8):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise_scale = 0.0005 + 0.001 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem, 30)
            a_opt, _ = sg_optimize(f_pert, time_limit=t_pert,
                                   lr=0.0005, momentum=0.95,
                                   top_k=max(10, n_best // 15),
                                   temp=0.00001)
            try_seq(a_opt, f"perturb_{p}")

    # === Phase 7: Try a completely different approach - structured initialization ===
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        print(f"\n=== Phase 7: Structured restarts ===")
        n_best = len(best_seq)

        # Try periodic-like structures
        for trial in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break
            # Create a structure with alternating high/low values
            base_struct = np.ones(n_best)
            period = int(n_best / (3 + trial))
            for i in range(n_best):
                phase = (i % period) / period
                base_struct[i] = 1.0 + 0.5 * np.cos(2 * np.pi * phase)
            # Add random noise
            base_struct += rng.standard_normal(n_best) * 0.1
            base_struct = np.maximum(base_struct, 0)
            base_struct /= base_struct.sum()

            a_opt, _ = lp_fw_optimize(base_struct, time_limit=min(rem, 40),
                                      use_min_peak=True, top_k_peaks=20)
            try_seq(a_opt, f"structured_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
