Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth LSE-approximated optimization

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, w, n):
    """Compute gradient of LSE objective w.r.t. a.
    g[m] = 2 * sum_j a[j] * w[m+j]
    """
    # Using the fact that this is a cross-correlation
    # For efficiency with moderate n, use direct computation
    m_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    # w has length 2*n-1, and m+j ranges from 0 to 2*n-2, all valid
    W = w[m_idx + j_idx]  # (n, n) matrix
    g = 2.0 * (W * a[None, :]).sum(axis=1)
    return g


def lse_objective_and_grad(z, T, precomputed=None):
    """Compute LSE objective and gradient w.r.t. z (softmax parameterization)."""
    a = softmax(z)
    conv = fast_convolve(a)
    n = len(a)
    
    # LSE computation (numerically stable)
    max_c = conv.max()
    shifted = T * (conv - max_c)
    exp_shifted = np.exp(shifted)
    lse_val = max_c + np.log(exp_shifted.sum()) / T
    
    # Weights
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient w.r.t. a
    g = compute_lse_grad(a, w, n)
    
    # Gradient w.r.t. z (chain rule through softmax)
    ag = np.dot(a, g)
    grad_z = a * (g - ag)
    
    return lse_val, grad_z
