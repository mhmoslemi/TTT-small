Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed subgradient descent with Adam and adaptive temperature
2. Targeted coordinate descent on active lags
3. Multi-length exploration with interpolation
4. Deflation-based peak reduction

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_optimize(f0, time_limit, T_start=None, T_end=None):
    """Adam on log-sum-exp smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    if T_start is None:
        T_start = 2.0 / n
    if T_end is None:
        T_end = 0.1 / n
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.005
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / max(1, int(time_limit / 0.01)))
        T = T_start * (T_end / T_start) ** progress
        lr = base_lr * (1.0 - 0.3 * progress)
        
        conv = fast_convolve(f)
        
        # Log-sum-exp subgradient
        conv_shifted = conv - conv.max()
        weights = np.exp(conv_shifted / T)
        if weights.sum() < 1e-15:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= weights.sum()
        
        # Gradient of sum_k w_k * conv[k] w.r.t. a
        # conv[k] = sum_i a[i] * a[k-i]
        # d(conv[k])/da[i] = a[k-i] (for valid k-i)
        grad = np.zeros(n)
        for k in range(2*n - 1):
            if weights[k] < 1e-12:
                continue
            # valid i: max(0, k-n+1) <= i <= min(n-1, k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # grad[i] += w[k] * a[k-i] for i in [i_start, i_end]
            # a[k-i] for i in [i_start, i_end] is a[k-i_end] to a[k-i_start]
            seg = a_segment(f, k, i_start, i_end)  # a[k-i] for i in range
            grad[i_start:i_end+1] += weights[k] * seg
        
        # Adam update
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        
        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 7 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
            continue
        
        f = f_new
        new_M = fast_convolve(f).max()
        
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_f, best_obj

Now let me think about the coordinate descent part.

**Targeted Coordinate Descent:**

After smooth optimization converges (or plateaus), we do targeted coordinate descent:

1. Compute conv, find the top-K lags (K=5 or so)
2. For each top lag k, identify the "active" coordinates: those i where a[i] * a[k-i] is large
3. For each active coordinate i, try to move mass to a coordinate j that doesn't appear in any top lag
4. Use golden section search on the transfer amount

This is more targeted than the previous code's pairwise approach.

**Deflation Approach:**

Another idea: after each smooth optimization step, explicitly "deflate" the peak:
1. Find the lag k* with maximum conv
2. For each i contributing to conv[k*], compute the "marginal cost" of decreasing a[i]:
   - If we decrease a[i] by δ, conv[k*] decreases by a[k*-i]*δ
   - But other lags might increase
3. Choose the i that gives the best trade-off and transfer mass to a "safe" coordinate

This is similar to the LP-FW approach but more heuristic.

"""

1. Log-sum-exp smoothed subgradient descent with Adam, Nesterov restarts, adaptive temperature
2. Targeted coordinate descent on active lags with golden-section search
3. Multi-length exploration with interpolation and diverse initializations
4. Deflation-based peak reduction for final polish

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(f, conv, T):
    """Compute subgradient of log-sum-exp smoothed max of conv."""
    n = len(f)
    conv_shifted = conv / T - (conv / T).max()
    weights = np.exp(conv_shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(2 * n - 1)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        start = k - i_max
        end = k - i_min
        grad[i_min:i_max + 1] += wk * f[start:end + 1][::-1]
    return grad


def smooth_optimize(f0, time_limit, T_start=None, T_end=None, base_lr=None, verbose=False):
    """Adam optimizer on log-sum-exp smoothed max(conv)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    if T_start is None:
        T_start = 3.0 / n
    if T_end is None:
        T_end = 0.05 / n
    if base_lr is None:
        base_lr = 0.006

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    it = 0
    no_improve = 0
    total_iters = max(1, int(time_limit / 0.008))

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / (total_iters * 0.85))
        T = T_start * (T_end / T_start) ** progress
        lr = base_lr * (1.0 - 0.4 * progress)

        conv = fast_convolve(f)
        grad = compute_smooth_gradient(f, conv, T)

        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)

        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            if no_improve > 300:
                rng = np.random.default_rng(it * 13 + 7)
                scale = 0.002 + 0.004 * min(no_improve // 100, 4)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
                conv = fast_convolve(f)
                if conv.max() > best_obj * 1.1:
                    f = best_f.copy()
                    no_improve = 0
            continue

        f = f_new
        conv = fast_convolve(f)
        new_M = conv.max()

        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Smooth iter {it}: obj={best_obj:.10f}, T={T:.6f}, t={elapsed:.1f}s")
        else:
            no_improve += 1

    return best_f, best_obj


def targeted_coord_descent(f0, time_limit, top_k=3, verbose=False):
    """Targeted coordinate descent: modify coordinates in active lags."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 500:
            break

        conv = fast_convolve(f)
        current_max = conv.max()

        if current_max < best_obj - 1e-15:
            best_obj = current_max
            best_f = f.copy()
            no_improve = 0

        # Find top lags
        k_actual = min(top_k, 2 * n - 1)
        top_lags = np.argpartition(conv, -k_actual)[-k_actual:]
        top_lags = top_lags[np.argsort(conv[top_lags])[::-1]]

        # Find active coordinates (those contributing to top lags)
        active_set = set()
        for k in top_lags:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                if f[i] > 1e-6 and f[k - i] > 1e-6:
                    active_set.add(i)
                    active_set.add(k - i)

        if not active_set:
            no_improve += 1
            continue

        active_list = list(active_set)

        # For each active coordinate, try to move mass to a random non-active coordinate
        improved_this_pass = False
        for i in active_list:
            if time.monotonic() - t0 > time_limit:
                break
            if no_improve > 500:
                break

            # Find a target: a coordinate not in active set with small value
            non_active_indices = [j for j in range(n) if j not in active_set]
            if not non_active_indices:
                non_active_indices = list(range(n))

            # Try a few random targets
            n_try = min(3, len(non_active_indices))
            targets = rng.choice(non_active_indices, size=n_try, replace=False) if n_try < len(non_active_indices) else np.array(non_active_indices[:n_try])

            for j in targets:
                if i == j:
                    continue
                c = f[i] + f[j]
                if c < 1e-10:
                    continue

                # Golden section search on t = a[i] / (a[i]+a[j])
                lo, hi = 0.0, 1.0
                cur_t = f[i] / c if c > 0 else 0.5
                cur_val = current_max

                # Quick scan first
                best_t = cur_t
                best_val = current_max
                for t in [cur_t, 0.0, 1.0, 0.2, 0.5, 0.8]:
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1 - t) * c
                    val = fast_convolve(f_trial).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t

                # Golden section refinement
                if best_t > 0.01 and best_t < 0.99:
                    lo = max(0.0, best_t - 0.15)
                    hi = min(1.0, best_t + 0.15)
                    gr = (np.sqrt(5) - 1) / 2
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)

                    f1 = f.copy()
                    f1[i] = c1 * c
                    f1[j] = (1 - c1) * c
                    v1 = fast_convolve(f1).max()

                    f2 = f.copy()
                    f2[i] = c2 * c
                    f2[j] = (1 - c2) * c
                    v2 = fast_convolve(f2).max()

                    for _ in range(8):
                        if hi - lo < 1e-7:
                            break
                        if v1 < v2:
                            hi = c2
                            c2 = c1
                            v2 = v1
                            c1 = hi - gr * (hi - lo)
                            f1 = f.copy()
                            f1[i] = c1 * c
                            f1[j] = (1 - c1) * c
                            v1 = fast_convolve(f1).max()
                        else:
                            lo = c1
                            c1 = c2
                            v1 = v2
                            c2 = lo + gr * (hi - lo)
                            f2 = f.copy()
                            f2[i] = c2 * c
                            f2[j] = (1 - c2) * c
                            v2 = fast_convolve(f2).max()

                    mid_t = (lo + hi) / 2
                    f_mid = f.copy()
                    f_mid[i] = mid_t * c
                    f_mid[j] = (1 - mid_t) * c
                    v_mid = fast_convolve(f_mid).max()
                    if v_mid < best_val - 1e-14:
                        best_val = v_mid
                        best_t = mid_t

                if best_val < current_max - 1e-13:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    f = f / f.sum()
                    current_max = best_val
                    improved_this_pass = True
                    if current_max < best_obj - 1e-13:
                        best_obj = current_max
                        best_f = f.copy()

        if not improved_this_pass:
            no_improve += 1
            if no_improve % 100 == 0:
                # Perturb and restart
                rng2 = np.random.default_rng(no_improve * 31)
                f = project_simplex(best_f + rng2.standard_normal(n) * 0.002)
                no_improve = 0

    return best_f, best_obj


def deflation_optimize(f0, time_limit, verbose=False):
    """Iteratively deflate the peak lag by targeted mass transfer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_max = conv.max()
        peak_lag = int(np.argmax(conv))

        # Find dominant terms at the peak lag
        i_min = max(0, peak_lag - n + 1)
        i_max = min(n - 1, peak_lag)
        products = np.array([f[i] * f[peak_lag - i] for i in range(i_min, i_max + 1)])
        indices = np.arange(i_min, i_max + 1)

        # Sort by product size
        order = np.argsort(products)[::-1]
        top_contributors = indices[order[:min(5, len(order))]]

        # Try to reduce the peak by moving mass from dominant contributors
        improved = False
        for i in top_contributors:
            if time.monotonic() - t0 > time_limit:
                break
            partner = peak_lag - i
            if partner < 0 or partner >= n or partner == i:
                continue

            # Try moving mass from i (or partner) to a random other coordinate
            c = f[i] + f[partner]
            if c < 1e-10:
                continue

            # Find a safe target: coordinate not in any top-3 lag
            top_lags = np.argpartition(conv, -3)[-3:]
            unsafe = set()
            for k in top_lags:
                k = int(k)
                k_min = max(0, k - n + 1)
                k_max = min(n - 1, k)
                for ii in range(k_min, k_max + 1):
                    unsafe.add(ii)
                    unsafe.add(k - ii)

            safe_indices = [j for j in range(n) if j not in unsafe and j != i and j != partner]
            if not safe_indices:
                safe_indices = [j for j in range(n) if j != i and j != partner]

            if not safe_indices:
                continue

            n_try = min(5, len(safe_indices))
            targets = rng.choice(safe_indices, size=n_try, replace=False)

            for j in targets:
                # Move some mass from i to j
                c2 = f[i] + f[j]
                if c2 < 1e-10:
                    continue
                best_t = f[i] / c2
                best_val = current_max
                for t in [best_t, best_t * 0.5, best_t * 0.7, best_t * 0.9, 0.0]:
                    if t < 0 or t > 1:
                        continue
                    f_trial = f.copy()
                    f_trial[i] = t * c2
                    f_trial[j] = (1 - t) * c2
                    val = fast_convolve(f_trial).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t

                if best_val < current_max - 1e-13:
                    f[i] = best_t * c2
                    f[j] = (1 - best_t) * c2
                    f = f / f.sum()
                    current_max = best_val
                    improved = True
                    if current_max < best_obj - 1e-13:
                        best_obj = current_max
                        best_f = f.copy()
                    break

        if not improved:
            # Small perturbation
            f = project_simplex(f + rng.standard_normal(n) * 0.001 * f.mean())
            if fast_convolve(f).max() > best_obj * 1.5:
                f = best_f.copy()

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def diverse_init(n, rng):
    """Create diverse initial sequences."""
    choice = rng.integers(0, 6)
    if choice == 0:
        return np.ones(n) / n
    elif choice == 1:
        # Two peaks
        f = np.zeros(n)
        f[n // 4] = 0.5
        f[3 * n // 4] = 0.5
        return f / f.sum()
    elif choice == 2:
        # Sinusoidal
        x = np.linspace(0, 2 * np.pi, n)
        f = np.maximum(0, 1 + 0.5 * np.sin(x * 3))
        return f / f.sum()
    elif choice == 3:
        # Random with structure
        f = np.maximum(0, rng.standard_normal(n) * 0.3 + 1)
        return f / f.sum()
    elif choice == 4:
        # Geometric
        r = 0.95
        f = np.array([r ** i for i in range(n)])
        return f / f.sum()
    else:
        # Block structure
        f = np.zeros(n)
        for start in range(0, n, max(1, n // 8)):
            f[start:start + max(1, n // 16)] = rng.uniform(0.5, 1.5)
        f = np.maximum(f, 0)
        ss = f.sum()
        if ss > 0:
            f /= ss
        else:
            f = np.ones(n) / n
        return f

        rem = deadline - time.monotonic()
        t2 = rem * 0.15
        print(f"\n=== Phase 2: Coord descent (n={len(best_seq)}, t={t2:.0f}s) ===")
        if t2 > 15:
            a_opt, _ = targeted_coord_descent(best_seq, time_limit=t2, top_k=3, verbose=True)
            try_seq(a_opt, "coord_p2")

    # === Phase 3: Smooth again on best (10% time) ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = rem * 0.12
        print(f"\n=== Phase 3: Smooth again (n={len(best_seq)}, t={t3:.0f}s) ===")
        if t3 > 10:
            a_opt, _ = smooth_optimize(best_seq, time_limit=t3,
                                       T_start=0.5 / len(best_seq),
                                       T_end=0.02 / len(best_seq),
                                       base_lr=0.003, verbose=True)
            try_seq(a_opt, "smooth_p3")

    # === Phase 4: Multi-length exploration (25% time) ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    base_n = len(base)

    target_lengths = []
    # Try nearby lengths and some larger ones
    for tl in [512, 640, 896, 1024, 1280]:
        if tl != base_n:
            target_lengths.append(tl)

    # Also try the parent length if different from base
    if base_n != n_parent:
        target_lengths.insert(0, n_parent)

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.35, 120)
        if t_alloc < 10:
            break

        f_interp = interpolate_sequence(base, target_n)
        print(f"  Exploring n={target_n}, t={t_alloc:.0f}s")

        # Smooth optimization
        t_smooth = t_alloc * 0.6
        a_opt, _ = smooth_optimize(f_interp, time_limit=t_smooth, verbose=False)
        try_seq(a_opt, f"smooth_n{target_n}")

        # Quick coord descent
        rem = deadline - time.monotonic()
        if rem > 15:
            t_cd = min(rem * 0.2, 40)
            if t_cd > 5:
                a_opt2, _ = targeted_coord_descent(a_opt, time_limit=t_cd, top_k=2)
                try_seq(a_opt2, f"coord_n{target_n}")

    # === Phase 5: Diverse restarts on best (10% time) ===
    print(f"\n=== Phase 5: Diverse restarts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed)

        for p in range(6):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(rem / max(1, 6 - p), 80)
            if t_alloc < 8:
                break

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = best_seq[::-1].copy()
                f_init = f_init / f_init.sum()
            elif p == 2:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            elif p == 3:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.03)
            elif p == 4:
                # Mix with uniform
                f_init = project_simplex(0.6 * best_seq + 0.4 * np.ones(n_best) / n_best)
            else:
                f_init = diverse_init(n_best, rng)

            # Alternate between smooth and coord descent
            if p % 2 == 0:
                a_opt, _ = smooth_optimize(f_init, time_limit=t_alloc,
                                           T_start=1.0 / n_best,
                                           T_end=0.03 / n_best,
                                           base_lr=0.004, verbose=(p == 0))
            else:
                a_opt, _ = targeted_coord_descent(f_init, time_limit=t_alloc, top_k=2)
            try_seq(a_opt, f"diverse_{p}")

    # === Phase 6: Deflation + final polish (remaining time) ===
    print(f"\n=== Phase 6: Deflation + final (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t_def = min(rem * 0.4, 60)
            print(f"  Deflation: n={len(best_seq)}, t={t_def:.0f}s")
            a_opt, _ = deflation_optimize(best_seq, time_limit=t_def)
            try_seq(a_opt, "deflation")

        rem = deadline - time.monotonic()
        if rem > 10:
            t_final = min(rem * 0.4, 50)
            print(f"  Final smooth: t={t_final:.0f}s")
            a_opt, _ = smooth_optimize(best_seq, time_limit=t_final,
                                       T_start=0.3 / len(best_seq),
                                       T_end=0.01 / len(best_seq),
                                       base_lr=0.002, verbose=True)
            try_seq(a_opt, "smooth_final")

        rem = deadline - time.monotonic()
        if rem > 8:
            t_cd = min(rem * 0.5, 40)
            print(f"  Final coord: t={t_cd:.0f}s")
            a_opt, _ = targeted_coord_descent(best_seq, time_limit=t_cd, top_k=5)
            try_seq(a_opt, "coord_final")

        # Tiny perturbation + quick smooth
        rem = deadline - time.monotonic()
        if rem > 10 and best_seq is not None:
            n_best = len(best_seq)
            for trial in range(3):
                rem = deadline - time.monotonic()
                if rem < 8:
                    break
                rng_trial = np.random.default_rng(900 + trial)
                f_pert = project_simplex(best_seq + rng_trial.standard_normal(n_best) * 0.002)
                a_opt, _ = smooth_optimize(f_pert, time_limit=min(rem, 20),
                                           T_start=0.2 / n_best,
                                           T_end=0.01 / n_best,
                                           base_lr=0.002)
                try_seq(a_opt, f"perturb_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
