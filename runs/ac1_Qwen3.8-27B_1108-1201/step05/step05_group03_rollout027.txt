import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k, temperature):
    """Compute weighted subgradient of max(conv) using top-k peaks."""
    n = len(f)
    k_actual = min(top_k, 2 * n - 1)
    if k_actual == 1:
        peaks = np.array([np.argmax(conv)])
    else:
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = temperature * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def adam_optimize(f0, time_limit, verbose=False, top_k_start=100, top_k_end=10, T_start=None, T_end=None, base_lr=0.003):

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999

    nesterov = True

    while time.monotonic() - t0 < time_limit:

        # Learning rate schedule

        # Adam update

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):

    else:
        f = f / s
    n = len(f)

    M = conv.max()
    best_obj = M
    best_f = f.copy()

    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        g = lp_direction(f, M, n)
        if g is None:
            # Try with slightly relaxed constraint
            for factor in [1.001, 1.005, 1.01, 1.02, 1.05]:
                g = lp_direction(f, M * factor, n)
                if g is not None:
                    break
            if g is None:
                no_improve += 1
                if no_improve > 50:
                    break
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:

            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Golden section refinement
        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            gr = (np.sqrt(5) - 1) / 2
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)

            ft1 = np.maximum(f + c1 * (g - f), 0)
            ft1 = ft1 / max(ft1.sum(), 1e-15)
            fc1 = fast_convolve(ft1).max()
            ft2 = np.maximum(f + c2 * (g - f), 0)
            ft2 = ft2 / max(ft2.sum(), 1e-15)
            fc2 = fast_convolve(ft2).max()

            for _ in range(8):
                if hi - lo < 1e-7:

                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)

                    fc2 = fast_convolve(ft2).max()

            if val < best_val:

                best_alpha = cand

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        # Random order for diversification
        if passes % 3 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            i = order[idx]
            c = f[i]
            remaining = 1.0 - c  # sum of all other coordinates

            if remaining < 1e-15:
                continue

            # When we change f[i], the constraint is that the sum stays 1
            # So we're really choosing f[i] in [0, 1] and redistributing
            # Actually, for coordinate descent with simplex constraint:
            # Fix all other coordinates, choose f[i] to minimize the objective
            # subject to f[i] >= 0 and sum remains 1
            # This means f[i] can be at most 1 (all mass on one coordinate)
            # But practically, we want f[i] in [0, c + some_other_coord]

            # Simpler: try a few values for f[i]
            # The objective as a function of f[i] is a quadratic
            # (because conv(a,a) is quadratic in a)
            # So we can find the optimal f[i] analytically!

            # Let a = f with a[i] = t
            # conv(a,a)[k] is quadratic in t for each k
            # We want to minimize max_k conv(a,a)[k]

            # For now, try a scan
            candidates = [c]
            if c > 0:
                candidates.extend([c * 0.5, c * 0.8, c * 1.2, c * 1.5, 0.0])
                # Also try values based on neighbors
                candidates.append(remaining * 0.01)

            best_t = c
            best_val = best_obj

            for t in candidates:
                if t < 0 or t > 0.5:  # Don't put more than 50% on one coord
                    continue
                f_trial = f.copy()
                f_trial[i] = t
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_val:
                    best_val = val
                    best_t = t

            if best_val < best_obj - 1e-13:
                f[i] = best_t
                f = f / f.sum()
                best_obj = best_val
                best_f = f.copy()
                pass_improved = True

        if pass_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

    return best_f, best_obj


def pairwise_optimize(f0, time_limit, verbose=False):
    """Optimize pairs of coordinates by mass transfer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(98765)

    # Focus on pairs that are far apart (to reduce central peak)
    # and pairs that are close (to smooth local structure)
    no_improve_passes = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        if passes % 2 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            i = order[idx_i]

            # Try pairs at various distances
            for dist in [1, 2, 3, 5, 10, 20, 50, 100, n // 2]:
                j = (i + dist) % n
                if j == i:
                    continue

                c = f[i] + f[j]
                if c < 1e-12:
                    continue

                # Try transfer values
                current_ratio = f[i] / c if c > 1e-15 else 0.5
                candidates = [current_ratio, 0.0, 1.0, 0.1, 0.3, 0.5, 0.7, 0.9]
                if 0.1 < current_ratio < 0.9:
                    candidates.extend([current_ratio - 0.1, current_ratio + 0.1])

                best_t = current_ratio
                best_val = best_obj

                for t in candidates:
                    t = np.clip(t, 0.0, 1.0)
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1 - t) * c
                    val = fast_convolve(f_trial).max()
                    if val < best_val:
                        best_val = val
                        best_t = t

                if best_val < best_obj - 1e-13:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_parametric_init(n, family, rng):
    """Generate parametric initializations."""
    if family == 'uniform':
        return np.ones(n) / n
    elif family == 'u_shape':
        x = np.linspace(-1, 1, n)
        alpha = rng.uniform(0.2, 1.0)
        a = 1.0 + alpha * x ** 2
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif family == 'w_shape':
        x = np.linspace(0, 1, n)
        alpha = rng.uniform(0.3, 1.5)
        a = 1.0 + alpha * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        k = rng.integers(1, 5)
        alpha = rng.uniform(0.2, 0.8)
        a = 1.0 + alpha * np.cos(2 * np.pi * k * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif family == 'bimodal':
        centers = np.sort(rng.uniform(0.1, 0.9, 2))
        sigmas = rng.uniform(0.05, 0.2, 2)
        x = np.linspace(0, 1, n)
        a = sum(np.exp(-((x - c) ** 2) / (2 * s ** 2)) for c, s in zip(centers, sigmas))
        a += 0.01
        return a / a.sum()
    elif family == 'random':
        a = rng.standard_normal(n)
        a = np.abs(a) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k, temperature):
    """Compute weighted subgradient of max(conv) using top-k peaks with softmax."""
    n = len(f)
    k_actual = min(top_k, 2 * n - 1)
    if k_actual == 1:
        peaks = np.array([np.argmax(conv)])
    else:
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = temperature * (pv - pv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg

        top_k_start = max(20, n // 8)
    if top_k_end is None:
        top_k_end = max(5, n // 40)
    if T_start is None:
        T_start = max(0.3, 4.0 / np.sqrt(n))
    if T_end is None:
        T_end = max(0.05, 0.2 / np.sqrt(n))

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    v_pred = np.zeros(n)

    it = 0
    no_improve = 0
    max_iters = int(time_limit / 0.006)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > max_iters:
            break
        progress = min(1.0, it / max(1, max_iters))

        # Adaptive schedules
        T = T_start * (T_end / T_start) ** (progress ** 0.6)
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 + 0.3 * (1.0 - progress)) * min(1.0, it / 10.0)

        conv = fast_convolve(f)

        # Nesterov lookahead
        if use_nesterov and it > 1:
            f_look = project_simplex(f - beta1 * v_pred)
            conv_look = fast_convolve(f_look)
            sg = compute_subgradient(f_look, conv_look, top_k, T)
        else:
            sg = compute_subgradient(f, conv, top_k, T)

        # Adam update
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        v_pred = step

        f_new = project_simplex(f - step)

        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            if no_improve > 150:
                rng = np.random.default_rng(it * 41 + 13)
                scale = 0.005 * min(1.0, no_improve / 80.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                m = np.zeros(n)
                v = np.zeros(n)
                v_pred = np.zeros(n)
                no_improve = 0
            continue

        f = f_new
        new_M = fast_convolve(f).max()

        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1

    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with golden-section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        g = lp_direction(f, M, n)
        if g is None:
            found = False
            for factor in [1.001, 1.005, 1.01, 1.02, 1.05]:
                g = lp_direction(f, M * factor, n)
                if g is not None:
                    found = True
                    break
            if not found:
                no_improve += 1
                if no_improve > 40:
                    break
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Coarse line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Golden section refinement
        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            gr = (np.sqrt(5) - 1) / 2
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)

            ft1 = np.maximum(f + c1 * (g - f), 0)
            ft1 = ft1 / max(ft1.sum(), 1e-15)
            fc1 = fast_convolve(ft1).max()
            ft2 = np.maximum(f + c2 * (g - f), 0)
            ft2 = ft2 / max(ft2.sum(), 1e-15)
            fc2 = fast_convolve(ft2).max()

            for _ in range(8):
                if hi - lo < 1e-7:
                    break
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    ft1 = np.maximum(f + c1 * (g - f), 0)
                    ft1 = ft1 / max(ft1.sum(), 1e-15)
                    fc1 = fast_convolve(ft1).max()
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    ft2 = np.maximum(f + c2 * (g - f), 0)
                    ft2 = ft2 / max(ft2.sum(), 1e-15)
                    fc2 = fast_convolve(ft2).max()

            cand = (lo + hi) / 2
            ft = np.maximum(f + cand * (g - f), 0)
            ft = ft / max(ft.sum(), 1e-15)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = cand

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.002 * min(no_improve // 30, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj


def pairwise_optimize(f0, time_limit, verbose=False):
    """Optimize pairs of coordinates by mass transfer at multiple distances."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(98765)

    no_improve_passes = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        if passes % 2 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            i = order[idx_i]

            for dist in [1, 2, 3, 5, 10, 20, 40, n // 4, n // 2]:
                j = (i + dist) % n
                if j == i:
                    continue

                c = f[i] + f[j]
                if c < 1e-12:
                    continue

                current_ratio = f[i] / c if c > 1e-15 else 0.5
                candidates = [current_ratio, 0.0, 1.0, 0.1, 0.3, 0.5, 0.7, 0.9]
                if 0.1 < current_ratio < 0.9:
                    candidates.extend([current_ratio - 0.1, current_ratio + 0.1])

                best_t = current_ratio
                best_val = best_obj

                for t in candidates:
                    t = float(np.clip(t, 0.0, 1.0))
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1.0 - t) * c
                    val = fast_convolve(f_trial).max()
                    if val < best_val:
                        best_val = val
                        best_t = t

                if best_val < best_obj - 1e-13:
                    f[i] = best_t * c
                    f[j] = (1.0 - best_t) * c
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 4:
                break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent: optimize one coordinate at a time via quadratic minimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        if passes % 3 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            i = order[idx]
            c = f[i]

            # When we change f[i] to t, the rest must sum to 1-t
            # The rest is proportional to f with f[i] removed
            f_rest = f.copy()
            f_rest[i] = 0.0
            rest_sum = f_rest.sum()
            if rest_sum < 1e-15:
                continue

            # f_rest_normalized = f_rest / rest_sum
            # Trial: f_trial[i] = t, f_trial[j!=i] = (1-t) * f_rest[j] / rest_sum
            # This is a 1D optimization problem in t

            # conv(f_trial, f_trial)[k] is quadratic in t
            # We can find the optimal t by scanning or analytically
            candidates = [c, 0.0, c * 0.5, c * 0.7, c * 1.3, c * 2.0, 1.0 / (3 * n)]
            candidates = [float(np.clip(t, 0.0, 0.999)) for t in candidates]

            best_t = c
            best_val = best_obj

            for t in candidates:
                f_trial = f.copy()
                f_trial[i] = t
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_val:
                    best_val = val
                    best_t = t

            if best_val < best_obj - 1e-13:
                f[i] = best_t
                f = f / f.sum()
                best_obj = best_val
                best_f = f.copy()
                pass_improved = True

        if pass_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_parametric_init(n, family, rng):
    """Generate diverse parametric initializations."""
    if family == 'uniform':
        return np.ones(n) / n
    elif family == 'u_shape':
        x = np.linspace(-1, 1, n)
        alpha = rng.uniform(0.3, 1.5)
        a = 1.0 + alpha * x ** 2
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif family == 'w_shape':
        x = np.linspace(0, 1, n)
        alpha = rng.uniform(0.3, 1.2)
        a = 1.0 + alpha * np.cos(4.0 * np.pi * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif family == 'cosine':
        x = np.linspace(0, 1, n)
        k = int(rng.integers(1, 6))
        alpha = rng.uniform(0.2, 0.9)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + alpha * np.cos(2.0 * np.pi * k * x + phase)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif family == 'bimodal':
        centers = np.sort(rng.uniform(0.1, 0.9, 2))
        sigmas = rng.uniform(0.03, 0.15, 2)
        x = np.linspace(0, 1, n)
        a = np.zeros(n)
        for c, s in zip(centers, sigmas):
            a += np.exp(-((x - c) ** 2) / (2.0 * s ** 2))
        a += 0.01
        return a / a.sum()
    elif family == 'edges':
        x = np.linspace(0, 1, n)
        alpha = rng.uniform(1.0, 3.0)
        a = 1.0 / (1.0 + alpha * np.sin(np.pi * x) ** 2)
        return a / a.sum()
    elif family == 'random':
        a = np.abs(rng.standard_normal(n)) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def evaluate_safe(a):
    """Safe evaluation wrapper."""
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return float('inf')
    a = a / a.sum()
    try:
        sc = float(evaluate_sequence(a.tolist()))
        return sc
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: diverse inits -> Adam -> LP-FW -> pairwise -> coord descent.
    Uses coarse-to-fine multi-resolution and adaptive method switching.
    """
    total_budget = max(1.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 10.0
    best_seq = None
    best_score = float('inf')
    master_rng = np.random.default_rng(seed)

    # Helper to try a sequence
    def do_try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a = a / a.sum()
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # === Get parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = do_try(parent, "parent")
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent error: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        do_try(parent, "default_parent")

    # === Phase 1: Diverse initializations ===
    print(f"\n=== Phase 1: Diverse inits (best={best_score:.6f}, t_remaining={deadline - time.monotonic():.0f}s) ===")
    lengths = [256, 384, 512, 768, 1024, 1280]
    families = ['u_shape', 'w_shape', 'cosine', 'bimodal', 'edges', 'random']

    candidates = []

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        for family in families:
            f_init = make_parametric_init(n, family, master_rng)
            sc = do_try(f_init, f"init_{family}_{n}")
            if np.isfinite(sc):
                candidates.append((f_init, sc, f"init_{family}_{n}"))

    # Interpolate parent to various lengths
    for n in lengths:
        if n == len(parent):
            continue
        if time.monotonic() > deadline - 550:
            break
        f_interp = interpolate_sequence(parent, n)
        sc = do_try(f_interp, f"interp_{n}")
        if np.isfinite(sc):
            candidates.append((f_interp, sc, f"interp_{n}"))

    candidates.sort(key=lambda x: x[1])
    print(f"  Top candidates: {[(c[2], f'{c[1]:.6f}') for c in candidates[:5]]}")

    # === Phase 2: Adam on top candidates ===
    print(f"\n=== Phase 2: Adam optimization (best={best_score:.6f}, t_remaining={deadline - time.monotonic():.0f}s) ===")
    top_candidates = candidates[:10]
    for idx, (f_init, sc_init, label) in enumerate(top_candidates):
        if time.monotonic() > deadline - 250:
            break
        n = len(f_init)
        remaining_candidates = len(top_candidates) - idx
        t_alloc = min(80.0, (deadline - time.monotonic() - 250) / max(1, remaining_candidates))
        if t_alloc < 10:
            continue
        print(f"  Adam [{label}] n={n}, init={sc_init:.6f}, t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(f_init, time_limit=t_alloc, verbose=False)
        do_try(a_opt, f"adam_{label}")

    # === Phase 3: LP-FW on current best ===
    print(f"\n=== Phase 3: LP-FW on best (best={best_score:.6f}, t_remaining={deadline - time.monotonic():.0f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        t_lpfw = min(120.0, (deadline - time.monotonic() - 150) * 0.5)
        if t_lpfw > 20:
            print(f"  LP-FW n={len(best_seq)}, t={t_lpfw:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lpfw, verbose=True)
            do_try(a_opt, "lpfw_best")

    # === Phase 4: Multi-length deep exploration ===
    print(f"\n=== Phase 4: Multi-length deep (best={best_score:.6f}, t_remaining={deadline - time.monotonic():.0f}s) ===")
    base = best_seq if best_seq is not None else parent

    deep_lengths = [384, 512, 768, 1024, 1280, 1536]
    for target_n in deep_lengths:
        if time.monotonic() > deadline - 100:
            break
        if target_n == len(base):
            continue

        f_interp = interpolate_sequence(base, target_n)
        do_try(f_interp, f"deep_interp_{target_n}")

        t_alloc = min(50.0, (deadline - time.monotonic() - 100) * 0.3)
        if t_alloc > 15:
            print(f"  Deep Adam n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_optimize(f_interp, time_limit=t_alloc, verbose=False)
            do_try(a_opt, f"deep_adam_{target_n}")

            rem = deadline - time.monotonic() - 80
            if rem > 20:
                t_lpfw = min(rem * 0.4, 60.0)
                if t_lpfw > 15:
                    print(f"  Deep LP-FW n={target_n}, t={t_lpfw:.0f}s")
                    a_opt2, _ = lp_fw_optimize(a_opt, time_limit=t_lpfw)
                    do_try(a_opt2, f"deep_lpfw_{target_n}")

    # === Phase 5: Diverse restarts from best ===
    print(f"\n=== Phase 5: Diverse restarts (best={best_score:.6f}, t_remaining={deadline - time.monotonic():.0f}s) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        restarts = [
            ("identity", best_seq.copy()),
            ("noise_small", project_simplex(best_seq + master_rng.standard_normal(n_best) * 0.003)),
            ("reversed", best_seq[::-1].copy()),
            ("blend_uniform", project_simplex(0.85 * best_seq + 0.15 * np.ones(n_best) / n_best)),
            ("noise_med", project_simplex(best_seq + master_rng.standard_normal(n_best) * 0.01)),
            ("mirror", project_simplex(np.concatenate([best_seq[:n_best//2], best_seq[:n_best//2][::-1]])[:n_best] + master_rng.standard_normal(n_best) * 0.002)),
        ]

        for ridx, (rlabel, f_init) in enumerate(restarts):
            if time.monotonic() > deadline - 80:
                break
            t_alloc = min(50.0, (deadline - time.monotonic() - 80) / max(1, len(restarts) - ridx))
            if t_alloc < 10:
                continue
            print(f"  Restart [{rlabel}] n={n_best}, t={t_alloc:.0f}s")
            if ridx % 2 == 0:
                a_opt, _ = adam_optimize(f_init, time_limit=t_alloc)
            else:
                a_opt, _ = lp_fw_optimize(f_init, time_limit=min(t_alloc, 80))
            do_try(a_opt, f"restart_{rlabel}")

    # === Phase 6: Pairwise + Coordinate descent ===
    print(f"\n=== Phase 6: Pairwise + Coord (best={best_score:.6f}, t_remaining={deadline - time.monotonic():.0f}s) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 30:
            t_pw = min(rem * 0.5, 80.0)
            print(f"  Pairwise n={len(best_seq)}, t={t_pw:.0f}s")
            a_opt, _ = pairwise_optimize(best_seq, time_limit=t_pw)
            do_try(a_opt, "pairwise")

            rem = deadline - time.monotonic() - 30
            if rem > 20:
                t_cd = min(rem * 0.5, 50.0)
                print(f"  Coord descent n={len(best_seq)}, t={t_cd:.0f}s")
                a_opt2, _ = coordinate_descent(a_opt, time_limit=t_cd)
                do_try(a_opt2, "coord_desc")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}, t_remaining={deadline - time.monotonic():.0f}s) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Adam with different LRs
        for lr in [0.001, 0.003, 0.005]:
            rem = deadline - time.monotonic() - 10
            if rem < 15:
                break
            a_opt, _ = adam_optimize(best_seq, time_limit=min(rem * 0.3, 40.0), base_lr=lr)
            do_try(a_opt, f"final_adam_lr{lr}")

        # LP-FW final
        rem = deadline - time.monotonic() - 8
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 35.0))
            do_try(a_opt, "lpfw_final")

        # Tiny perturbations + quick Adam
        for trial in range(4):
            rem = deadline - time.monotonic() - 5
            if rem < 10:
                break
            f_pert = project_simplex(best_seq + master_rng.standard_normal(n_best) * 0.002)
            a_opt, _ = adam_optimize(f_pert, time_limit=min(rem * 0.3, 25.0), base_lr=0.002)
            do_try(a_opt, f"perturb_{trial}")

        # Final pairwise
        rem = deadline - time.monotonic() - 3
        if rem > 10:
            a_opt, _ = pairwise_optimize(best_seq, time_limit=min(rem, 20.0))
            do_try(a_opt, "pw_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = evaluate_safe(best_seq)

    # Final verification
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
