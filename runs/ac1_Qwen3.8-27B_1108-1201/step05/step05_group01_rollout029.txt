One more thought: the "away-steps" Frank-Wolfe (AFW) method is known to converge at a geometric rate (vs. 1/k for standard FW). The key difference is that in AFW, when the new vertex v is "away" from the current iterate (i.e., <x*, v> < 0 for the appropriate dual variable), we move away from it instead of toward it. This can dramatically speed up convergence.

For our problem:
- Current iterate: f (on the simplex)
- New vertex: g (from the LP)
- Standard FW: f_new = (1-γ)f + γg
- AFW: if any component of f is "away" from g (i.e., f_i > g_i and f_i > 0), we can decrease that component instead of increasing g_i.

def afw_step(f, g, gamma):
    """Away-Steps Frank-Wolfe step.
    f, g: points on the simplex
    gamma: step size
    Returns new point on the simplex.
    """
    # Standard FW step
    # f_new = (1-gamma)*f + gamma*g
    
    # AFW: find i where f_i > g_i and f_i is large
    # The "away" direction is e_i (the i-th basis vector)
    # We want to find alpha such that:
    # f_new = (1-alpha)*f - alpha_away*e_i + alpha_towards*g
    # where alpha + alpha_away + alpha_towards = 1 (approximately)
    
    # Simplified AFW:
    # If (g - f)^T (f - e_i) < (g - f)^T (f - something), use away step
    # For simplicity, use the "away-step" heuristic:
    # Find i that minimizes g_i (the vertex g has least mass where f has most)
    # Then: f_new = (1-gamma)*f + gamma*g, but also decrease f_i
    
    # Actually, the standard AFW algorithm is:
    # 1. Compute v = g - f (towards direction)
    # 2. Find i that maximizes -v_i (i.e., minimizes g_i - f_i, or equivalently where f has excess)
    # 3. Compute the away direction: w = -e_i
    # 4. Check if the away step is better: if v^T w < gamma * ||v||^2 / 2, use away step
    # 5. Away step: f_new = (1-gamma)*f + gamma * (2*f - e_i) ... no that's not right
    
    # Let me use a simpler version:
    # Standard FW: f_new = (1-gamma)*f + gamma*g
    # AFW variant: 
    #   Find i = argmax_j [f_j * (1 - g_j/f_j)] or i = argmin_j g_j
    #   Direction towards: d_t = g - f
    #   Direction away: d_a = -e_i
    #   Step: f_new = f + gamma * (d_t - 2*f_i*d_a) ... 
    #   This isn't quite right either.
    
    # Let me just use the standard FW with a twist:
    # After computing f_new = (1-gamma)*f + gamma*g
    # Find the index i where f_new[i] is smallest relative to f[i]
    # "Push" mass away from i by redistributing
    
    # Actually, the cleanest AFW implementation:
    n = len(f)
    # Towards direction
    d_t = gamma * (g - f)
    
    # Find away index: i that maximizes f_i * (f_i - g_i) (excess mass where g has less)
    diff = f - g  # positive where f has more than g
    excess = f * diff  # weighted excess
    i_away = np.argmax(excess)
    
    # Check if away step is beneficial
    # The away direction is -e_i
    # f_away = f - gamma * f_i_away * e_i_away (decrease i_away component)
    # But this leaves the simplex! Need to redistribute.
    
    # Proper AFW:
    # f_new = (1 - gamma) * f + gamma * g  (standard)
    # AWF: f_new = (1 - 2*gamma*f_i) * f + gamma * g + gamma * f_i * (something)
    
    # This is getting complicated. Let me just use standard FW with line search,
    # which is what the previous code does. The AFW improvement might be too
    # complex to implement correctly in the time available.
    return (1 - gamma) * f + gamma * g

    while time.monotonic() - t0 < time_limit:
        it += 1

        sg = compute_subgrad(f, conv, top_k, cur_temp)

        # Multiplicative update
        log_update = -cur_lr * sg
        log_update = np.clip(log_update, -10, 10)
        f_new = f * np.exp(log_update)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f_new = f_new / ss

        conv = fast_convolve(f_new)
        obj = conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            cur_lr *= 0.9
            cur_temp *= 1.1
        elif no_improve == 0:
            cur_lr = min(cur_lr * 1.01, lr0)
            cur_temp = max(cur_temp * 0.99, temp0 * 0.5)

        if no_improve > 3000:
            rng = np.random.default_rng(it * 17 + 3)
            f_new = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            f = f_new
            conv = fast_convolve(f)
            cur_lr = max(lr0 * 0.5, 0.01)
            cur_temp = max(temp0 * 0.5, 0.002)
            no_improve = 0

        f = f_new

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
                continue
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        fw_alpha = 2.0 / (it + 2.0)
        alphas = sorted(set([0.1, 0.2, 0.3, fw_alpha, 0.5, 0.7, 0.9]))

        for alpha in alphas:
            ft = (1 - alpha) * f + alpha * g
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Binary refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.2)
            hi = min(1.0, best_alpha + 0.2)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1 - mid) * f + mid * g
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = (1 - best_alpha) * f + best_alpha * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 23 + 11)
            scale = 0.002 + 0.002 * min(no_improve // 50, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, max_passes=50):
    """Coordinate descent with line search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_pass_improve = 0

    for _ in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Try a few multiplier values
            cur_obj = fast_convolve(f).max()
            best_x = old_val
            best_local = cur_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                ft = f.copy()
                ft[j] = new_val
                remaining = 1.0 - new_val
                if remaining < 0:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (remaining / denom)
                val = fast_convolve(ft).max()
                if val < best_local:
                    best_local = val
                    best_x = new_val

            if best_x != old_val:
                ft = f.copy()
                ft[j] = best_x
                remaining = 1.0 - best_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (remaining / denom)
                ft = ft / ft.sum()
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_f = f.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def smoothed_newton(f0, time_limit, tau0=0.01, max_iter=5000):
    """Optimize using smoothed max (log-sum-exp) with gradient descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    cur_tau = tau0
    cur_lr = 0.005
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1

        # Compute smoothed gradient: grad of tau * log(sum(exp(conv[k]/tau)))
        # = sum_k exp(conv[k]/tau) * dconv[k]/df / sum_k exp(conv[k]/tau)
        # For efficiency, use top-k approximation
        k_actual = min(100, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute gradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            wi = 2.0 * w[i]
            if k < n:
                grad[:n - k] += wi * f[k:]
            if k > 0:
                grad[k:] += wi * f[:n - k]

        # Gradient step with projection
        f_new = project_simplex(f - cur_lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            conv = conv_new
        else:
            no_improve += 1
            if no_improve > 50:
                cur_lr *= 0.9
                cur_tau *= 1.02
                no_improve = 0

        # Anneal tau
        if it % 100 == 0:
            cur_tau = max(cur_tau * 0.8, 0.001)

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Interpolate parent to longer lengths
    2. Nesterov subgradient for rapid initial optimization
    3. LP-Frank-Wolfe for larger improvements
    4. Mirror descent for different geometry
    5. Coordinate descent for fine-tuning
    6. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512

    # === Phase 1: Interpolate to various lengths and quick-optimize ===
    print("=== Phase 1: Multi-length interpolation ===")
    target_lengths = [1024, 1536, 1800]

    best_interp = None
    for tl in target_lengths:
        rem = deadline - time.monotonic() - 600
        if rem < 20:
            break
        f_interp = interpolate_sequence(parent, tl)
        # Quick subgradient to evaluate
        t_quick = min(rem * 0.15, 30)
        a_opt, _ = nesterov_optimize(f_interp, time_limit=t_quick,
                                      lr0=0.005, top_k=30, temp0=0.01)
        s = try_seq(a_opt, f"interp_n{tl}")
        if best_interp is None or s < best_score:
            best_interp = a_opt

    if best_interp is not None:
        n_main = len(best_interp)
    else:
        n_main = 1800
        best_interp = interpolate_sequence(parent, n_main)
        try_seq(best_interp, "interp_defaultn1800")

    print(f"  Main length: n={n_main}, current best: {best_score:.6f}")

    # === Phase 2: Extended Nesterov on best ===
    print(f"\n=== Phase 2: Nesterov optimization (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_ns = min(rem * 0.5, 300)
        print(f"  Nesterov t={t_ns:.0f}s, n={n_main}, lr0=0.006, top_k=60")
        a_opt, _ = nesterov_optimize(best_interp, time_limit=t_ns,
                                      lr0=0.006, mom0=0.87, top_k=60, temp0=0.008)
        try_seq(a_opt, "nesterov_main")

        # Second pass with different params
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            a_opt2, _ = nesterov_optimize(a_opt, time_limit=min(rem, 120),
                                           lr0=0.003, mom0=0.91, top_k=40, temp0=0.003)
            try_seq(a_opt2, "nesterov_fine")

    # === Phase 3: Mirror descent ===
    print(f"\n=== Phase 3: Mirror descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_md = min(rem * 0.4, 120)
            print(f"  Mirror descent t={t_md:.0f}s")
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_md,
                                                lr0=0.03, top_k=40, temp0=0.005)
            try_seq(a_opt, "mirror")

            # Blend with best
            if a_opt is not None:
                for alpha in [0.3, 0.5, 0.7]:
                    blended = alpha * best_seq + (1 - alpha) * a_opt
                    blended = project_simplex(blended)
                    try_seq(blended, f"blend_{alpha}")

    # === Phase 4: LP-Frank-Wolfe ===
    print(f"\n=== Phase 4: LP-Frank-Wolfe (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_lpfw = min(rem * 0.6, 200)
            print(f"  LP-FW t={t_lpfw:.0f}s, n={n_best}")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lpfw, verbose=True)
            try_seq(a_opt, "lpfw_main")

            # Continue from LP-FW result with subgradient
            rem = deadline - time.monotonic() - 150
            if rem > 20:
                a_opt2, _ = nesterov_optimize(a_opt, time_limit=min(rem, 80),
                                               lr0=0.003, mom0=0.9, top_k=50, temp0=0.004)
                try_seq(a_opt2, "post_lpfw_nesterov")

    # === Phase 5: Smoothed Newton ===
    print(f"\n=== Phase 5: Smoothed optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_sn = min(rem * 0.5, 100)
            print(f"  Smoothed t={t_sn:.0f}s")
            a_opt, _ = smoothed_newton(best_seq, time_limit=t_sn, tau0=0.02)
            try_seq(a_opt, "smoothed")

    # === Phase 6: Multi-start exploration ===
    print(f"\n=== Phase 6: Multi-start (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 80)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                f_init = 0.8 * best_seq + 0.2 * best_seq[::-1]
                f_init = project_simplex(f_init)
            elif p == 2:
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                f_init = 0.7 * best_seq + 0.3 * cosine_modulated_init(n_best, 3, 0.2)
                f_init = project_simplex(f_init)

            a_opt, _ = nesterov_optimize(f_init, time_limit=t_alloc,
                                          lr0=0.004, mom0=0.88, top_k=50, temp0=0.006)
            try_seq(a_opt, f"multistart_{p}")

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_cd = min(rem * 0.5, 120)
            print(f"  CD t={t_cd:.0f}s, n={n_best}")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, max_passes=30)
            try_seq(a_opt, "coord_descent")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick Nesterov with very small steps
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = nesterov_optimize(best_seq, time_limit=min(rem, 60),
                                          lr0=0.001, mom0=0.93, top_k=80, temp0=0.001)
            try_seq(a_opt, "final_nesterov")

        # Try different lengths from best
        rem = deadline - time.monotonic() - 25
        if rem > 15:
            for tl in [n_best * 2 // 3, int(n_best * 1.2)]:
                if tl < 100 or tl > 3000:
                    continue
                f_i = interpolate_sequence(best_seq, tl)
                a_opt, _ = nesterov_optimize(f_i, time_limit=min(rem, 20),
                                              lr0=0.003, top_k=40, temp0=0.005)
                try_seq(a_opt, f"final_interp_n{tl}")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        for _ in range(3):
            rem = deadline - time.monotonic() - 8
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = nesterov_optimize(f_pert, time_limit=min(rem, 15),
                                          lr0=0.001, mom0=0.94, top_k=60, temp0=0.001)
            try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
