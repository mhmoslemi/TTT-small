def coordinate_descent_efficient(a0, time_limit, max_passes=200):
    """Efficient coordinate descent using the fact that changing one coordinate
    only affects n entries of the convolution."""
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    for pass_num in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-13:
                continue
            
            # When we change a[j] from x_old to x_new, the convolution changes as:
            # new_conv[k] = old_conv[k] - x_old * a[k-j] + x_new * a[k-j] + (x_new^2 - x_old^2) * delta(k, 2j)
            # = old_conv[k] + (x_new - x_old) * a[k-j] + (x_new^2 - x_old^2) * delta(k, 2j)
            # where a[k-j] is the old value (with a[j] = x_old)
            # But we also need to rescale the other coordinates...
            # Actually, let's work in the unnormalized space for now.
            # If we fix sum = 1 and change a[j] from x_old to x_new, we need to
            # rescale the others: a'[m] = a[m] * (1-x_new)/(1-x_old) for m != j
            # This makes it more complex. Let's just do simple coordinate descent
            # where we only change one coordinate and accept the sum change.
            # Then renormalize.
            
            # Simpler approach: try a few candidate values for a[j]
            # and pick the best one
            candidates = [0.0, x_old * 0.5, x_old * 0.7, x_old, x_old * 1.3, x_old * 1.5, x_old * 2.0]
            
            best_cand_obj = best_obj
            best_cand_val = x_old
            
            for x_try in candidates:
                if x_try < 0 or x_try > 1.0:
                    continue
                a_trial = a.copy()
                a_trial[j] = x_try
                st = a_trial.sum()
                if st < 1e-13:
                    continue
                a_trial /= st
                
                # Update convolution efficiently
                # The change in a[j] affects conv[k] for k in [j, j+n-1]
                # But since we renormalize, it's easier to just recompute
                # For n=768, FFT is fast enough
                conv_trial = fast_convolve(a_trial)
                obj_trial = conv_trial.max()
                if obj_trial < best_cand_obj - 1e-13:
                    best_cand_obj = obj_trial
                    best_cand_val = x_try
            
            if best_cand_val != x_old:
                a[j] = best_cand_val
                st = a.sum()
                a /= st
                conv = fast_convolve(a)
                obj = conv.max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Add noise to escape
            if pass_num > 1:
                rng = np.random.default_rng(pass_num * 31)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                conv = fast_convolve(a)
            else:
                break
    
    return best_a, best_obj


def cosine_init(n, f=1):
    """Initialize with a cosine-modulated sequence."""
    i = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * f * i / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def triangle_init(n):
    """Initialize with a triangular (Bartlett) window."""
    i = np.arange(n)
    a = np.sin(np.pi * i / n)**2
    s = a.sum()
    if s > 0:
        a /= s
    return a


def gaussian_init(n, sigma_ratio=0.3):
    """Initialize with a Gaussian window."""
    i = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / (sigma_ratio * n / 2))**2)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def symmetrize(a):
    """Make sequence symmetric."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
