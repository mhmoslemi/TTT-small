def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, constraint_indices=None):
    """Build sparse constraint matrix for LP. Shape: (num_constraints, n)."""
    n = len(f)
    if constraint_indices is None:
        constraint_indices = np.arange(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, constraint_indices=None, time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in constraint_indices, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f, constraint_indices)
    if A.shape[0] == 0:
        return np.ones(n) / n
    
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, constraint_indices=None):
    n = len(f)
    if constraint_indices is None:
        constraint_indices = np.arange(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, top_k=30, time_limit=15):
    n = len(f)
    conv_len = 2 * n - 1
    conv = fast_convolve(f)
    # Select top constraints: top_k peaks + center
    peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    # Also include the center peak
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    
    A = build_lp_matrix(f, peak_idx)
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_subgradient(f, active_peaks, weights=None):
    """Compute weighted sum of gradients of (f*f)[k] for k in active_peaks."""
    n = len(f)
    if weights is None:
        weights = np.ones(len(active_peaks)) / len(active_peaks)
    grad = np.zeros(n)
    for idx, k in enumerate(active_peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad


def lse_weights(conv, beta):
    """Compute LSE softmax weights on the convolution."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total

def subgradient_optimize(f0, time_limit, beta_start=10.0, beta_end=5000.0, top_k=50):
    """Subgradient method with LSE smoothing and backtracking."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 1.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # LSE weights
        w_full = lse_weights(conv, beta)
        
        # Select top peaks for efficiency
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w_full[peak_idx]
        w_peaks_sum = w_peaks.sum()
        if w_peaks_sum > 1e-15:
            w_peaks = w_peaks / w_peaks_sum
        else:
            w_peaks = np.ones(k_top) / k_top
        
        # Compute subgradient
        grad = compute_subgradient(f, peak_idx, w_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            no_improve += 100
            break
        grad_normed = grad / gnorm
        
        # Try a few step sizes (backtracking)
        step = lr
        improved = False
        for _ in range(8):
            f_new = project_simplex(f - step * grad_normed)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                break
            step *= 0.6
        if not improved:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.9
                if no_improve > 60:
                    # Restart
                    f = best_f.copy()
                    lr = max(1.0 / np.sqrt(n), 0.001)
                    beta = min(beta * 2.0, beta_end)
                    no_improve = 0
                    # Small perturbation
                    rng = np.random.default_rng(it * 31 + 17)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            
        # Increase beta slowly
        if beta < beta_end:
            beta *= 1.005
    
    return best_f, best_obj


def freq_perturb(f, rng, strength=0.05):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    # Perturb magnitude
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive Frank-Wolfe with LP oracle (top-K constraint selection)
2. Subgradient descent with LSE smoothing and backtracking line search
3. Frequency-domain perturbation for diversification
4. Multi-length exploration and interpolation
5. Peak-pair coordinate adjustment for fine refinement
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, constraint_indices=None):
    """Build sparse constraint matrix for LP. Shape: (num_constr, n)."""
    n = len(f)
    if constraint_indices is None:
        constraint_indices = np.arange(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, top_k=30, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-K peaks, g >= 0."""
    n = len(f)
    conv = fast_convolve(f)
    peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    A = build_lp_matrix(f, peak_idx)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_subgradient(f, active_peaks, weights=None):
    """Compute weighted sum of gradients of (f*f)[k] for k in active_peaks.
    Gradient of (f*f)[k] wrt f[i] = 2*f[k-i] if i in valid range, else 0.
    """
    n = len(f)
    if weights is None:
        weights = np.ones(len(active_peaks)) / len(active_peaks)
    grad = np.zeros(n)
    for idx, k in enumerate(active_peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad


def lse_weights(conv, beta):
    """Compute LSE softmax weights on the convolution peaks."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total

def subgradient_optimize(f0, time_limit, beta_start=10.0, beta_end=5000.0, top_k=60):
    """Subgradient method with LSE smoothing and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 2.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        conv = fast_convolve(f)
        w_full = lse_weights(conv, beta)

        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w_full[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top

        grad = compute_subgradient(f, peak_idx, w_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            no_improve += 100
            break
        grad_normed = grad / gnorm

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(10):
            f_new = project_simplex(f - step * grad_normed)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.005, 10.0 / np.sqrt(n))
                break
            step *= 0.55

        if not improved:
            no_improve += 1
            if no_improve >= 12:
                lr *= 0.88
            if no_improve > 70:
                f = best_f.copy()
                lr = max(2.0 / np.sqrt(n), 0.005)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0
                rng = np.random.default_rng(it * 31 + 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)

def peak_pair_adjust(f, n_pairs=50, time_limit=30.0):
    """Try to reduce peaks by adjusting pairs (i, j) simultaneously."""
    f = np.maximum(f0 if 'f0' in dir() else f, 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Get top peaks
    top_peaks = np.argpartition(conv, -min(20, len(conv)))[-min(20, len(conv)):]

    for _ in range(n_pairs):
        if time.monotonic() - t0 > time_limit:
            break
        # Pick a random peak
        k = int(top_peaks[np.random.randint(len(top_peaks))])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo >= hi:
            continue

        # The peak (f*f)[k] = sum_{j=lo}^{hi} f[j]*f[k-j]
        # Try to reduce f[j] for the j that maximizes f[j]*f[k-j]
        vals = f[lo:hi+1] * f[k - np.arange(lo, hi+1)]
        j_max = lo + np.argmax(vals)
        j_other = k - j_max
        if j_other < 0 or j_other >= n or j_other == j_max:
            continue

        # Move mass from j_max to a random position far from the peak
        candidates = np.random.choice(n, size=min(10, n), replace=False)
        for j_dest in candidates:
            if j_dest == j_max or j_dest == j_other:
                continue
            # Check if moving mass to j_dest would increase other peaks
            delta = f[j_max] * 0.1  # move 10% of mass
            if delta < 1e-15:
                continue
            f_trial = f.copy()
            f_trial[j_max] -= delta
            f_trial[j_dest] += delta
            if f_trial[j_max] < 0:
                continue
            f_trial = np.maximum(f_trial, 0)
            f_trial /= f_trial.sum()
            conv_trial = fast_convolve(f_trial)
            if conv_trial.max() < best_obj - 1e-15:
                best_obj = conv_trial.max()
                best_f = f_trial.copy()
                f = f_trial.copy()
                break

    return best_f, best_obj


def freq_perturb(f, rng, strength=0.05):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Main multi-phase optimization:
    Phase 1: Multi-length FW-LP exploration
    Phase 2: Subgradient refinement with adaptive beta
    Phase 3: Alternating FW + Subgradient + Freq perturbation
    Phase 4: Peak-pair coordinate adjustment
    Phase 5: Final high-precision polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Multi-length FW exploration ===
    print("\n=== Phase 1: Multi-length FW exploration ===")
    elapsed = time.monotonic() - (deadline - float(budget_s) + 25.0)
    total_budget = float(budget_s) - 25.0
    t1_fraction = 0.25
    t1_end = time.monotonic() + total_budget * t1_fraction

    n_candidates = sorted(set([n_parent,
                                max(128, n_parent - 96),
                                min(1500, n_parent + 96),
                                max(128, int(n_parent * 0.7)),
                                min(1500, int(n_parent * 1.3))]), reverse=True)
    seen = set()
    n_candidates = [x for x in n_candidates if not (x in seen or seen.add(x))]

    for nc in n_candidates[:5]:
        t_remaining = t1_end - time.monotonic()
        if t_remaining < 25:
            break
        t_alloc = min(100, t_remaining * 0.35)
        if t_alloc < 12:
            break

        if nc == n_parent:
            f_init = parent.copy()
        else:
            f_init = interpolate_to_length(parent, nc)

        print(f"  FW n={nc}, t={t_alloc:.0f}s")
        f_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc, top_k=30)
        try_sequence(f_opt, best_tracker)

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Subgradient refinement ===")
    t2_end = t1_end + total_budget * 0.25
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    t_alloc = min(200, t2_end - time.monotonic())
    if t_alloc > 20:
        print(f"  Subgradient n={n_cur}, t={t_alloc:.0f}s, beta: 10->5000")
        f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                          beta_start=10.0, beta_end=5000.0, top_k=60)
        try_sequence(f_opt, best_tracker)

    # === Phase 3: Alternating optimization ===
    print(f"\n=== Phase 3: Alternating optimization ===")
    t3_end = deadline - 200
    cycles = 0
    while time.monotonic() < t3_end and cycles < 15:
        cycles += 1
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        n_cur = len(current_best)
        remaining = t3_end - time.monotonic()
        if remaining < 35:
            break

        if cycles % 4 == 1:
            # FW step
            t_alloc = min(90, remaining * 0.35)
            if t_alloc > 12:
                print(f"  Cycle {cycles}: FW n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = fw_lp_optimize(current_best, time_limit=t_alloc, top_k=35)
                try_sequence(f_opt, best_tracker)

        elif cycles % 4 == 2:
            # Subgradient step with increasing beta
            t_alloc = min(110, remaining * 0.4)
            if t_alloc > 12:
                beta_s = 30.0 + 40.0 * cycles
                print(f"  Cycle {cycles}: Subgrad n={n_cur}, t={t_alloc:.0f}s, beta_start={beta_s:.0f}")
                f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                                  beta_start=beta_s,
                                                  beta_end=10000.0, top_k=80)
                try_sequence(f_opt, best_tracker)

        elif cycles % 4 == 3:
            # Frequency perturbation + brief FW
            t_alloc = min(80, remaining * 0.3)
            if t_alloc > 12:
                print(f"  Cycle {cycles}: FreqPerturb+FW n={n_cur}, t={t_alloc:.0f}s")
                for attempt in range(3):
                    t_rem = t3_end - time.monotonic() - 15
                    if t_rem < 12:
                        break
                    strength = 0.015 + 0.01 * attempt
                    f_pert = freq_perturb(current_best, rng, strength=strength)
                    t_sub = (t_rem - 8) / 2.0
                    if t_sub > 6:
                        f_opt, _ = fw_lp_optimize(f_pert, time_limit=t_sub, top_k=25)
                        try_sequence(f_opt, best_tracker)

        else:
            # Peak pair adjustment
            t_alloc = min(50, remaining * 0.2)
            if t_alloc > 8:
                print(f"  Cycle {cycles}: PeakPair n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = peak_pair_adjust(current_best, n_pairs=200, time_limit=t_alloc)
                try_sequence(f_opt, best_tracker)

    # === Phase 4: Final high-precision polish ===
    print(f"\n=== Phase 4: Final polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    # Very high beta subgradient (focus on the top few peaks)
    t_alloc = min(100, deadline - time.monotonic() - 50)
    if t_alloc > 12:
        print(f"  High-beta subgrad n={n_cur}, t={talloc if False else t_alloc:.0f}s")
        f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                          beta_start=500.0, beta_end=30000.0, top_k=30)
        try_sequence(f_opt, best_tracker)

    # One more FW pass with more constraints
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_alloc = min(70, deadline - time.monotonic() - 20)
    if t_alloc > 12:
        print(f"  Final FW (top_k=60) n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = fw_lp_optimize(current_best, time_limit=t_alloc, top_k=60)
        try_sequence(f_opt, best_tracker)

    # Quick random perturbations of the best
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 20:
        print(f"  Final random perturbations, t={t_rem:.0f}s")
        for _ in range(10):
            if time.monotonic() > deadline - 5:
                break
            f_pert = freq_perturb(current_best, rng, strength=0.005)
            try_sequence(f_pert, best_tracker)
            # Also try small time-domain perturbations
            f_pert2 = project_simplex(current_best + rng.standard_normal(n_cur) * 0.002)
            try_sequence(f_pert2, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using uniform fallback")
    fallback = np.ones(448) / 448.0
    return [float(x) for x in fallback]
