    a = a / a.sum()
    for _ in range(n_iters):
        conv = fast_convolve(a)
        # Find top k peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        # For each position, compute how many peaks it contributes to
        contrib_count = np.zeros(len(a))
        for p in peaks:
            p = int(p)
            js = max(0, p - len(a) + 1)
            je = min(len(a) - 1, p)
            contrib_count[js:je+1] += 1
        # Update: reduce high-contribution, increase low-contribution
        a_new = a * (1 - step * contrib_count / contrib_count.max())
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return a

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using adaptive subgradient descent
with Nesterov acceleration, LP-based global moves, and coordinate descent polish.
Key improvements: cosine LR schedule, adaptive peak temperature, deflation steps,
diverse starting points, and multi-length exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, sigma_frac=0.4):
    """Create a Gaussian window of length n, normalized to sum=1."""
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / (sigma_frac * n / 2))**2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine(n):
    """Create a cosine window of length n, normalized to sum=1."""
    x = np.arange(n)
    a = np.cos(np.pi * x / (n - 1 + 1e-10))**2 + 0.01
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_random(n, seed=42):
    """Create a random non-negative sequence normalized to sum=1."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    return a


def interpolate_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def solve_lp_min_peak(f, n, time_limit=30):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': time_limit})
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g /= gs
        return g
    return None


def subgradient_optimize(f0, time_limit, lr_init=0.005, lr_end=0.0002,
                         momentum_init=0.9, momentum_end=0.98,
                         temp_init=0.0001, temp_end=0.00001,

                         deflation_every=200, verbose=False):
    """
    Subgradient descent with Nesterov momentum, cosine LR decay,
    adaptive temperature, and periodic deflation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Cosine schedule
        progress = min(1.0, it / max(1, int(time_limit * 100)))  # rough estimate
        # Use time-based progress instead
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        
        cur_lr = lr_end + 0.5 * (lr_init - lr_end) * (1 + np.cos(np.pi * frac))
        cur_mom = momentum_init + (momentum_end - momentum_init) * frac
        cur_temp = temp_init * (temp_end / temp_init) ** frac
        cur_topk = max(5, int(top_k_init + (top_k_end - top_k_init) * frac))

        k_actual = min(cur_topk, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / max(cur_temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            # d(conv[k])/d(a_j) = 2 * a[k-j] for valid k-j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                kg = k - j
                if 0 <= kg < n:
                    sg[j] += 2.0 * w[i] * f[kg]

        v = cur_mom * v - cur_lr * sg
        f_new = project_simplex(f + v)

        # Periodic deflation: reduce coordinates contributing most to peaks
        if deflation_every > 0 and it % deflation_every == 0:
            conv_check = fast_convolve(f_new)
            top_p = np.argmax(conv_check)
            tp = int(top_p)
            js = max(0, tp - n + 1)
            je = min(n - 1, tp)
            # Slightly reduce the peak-contributing coords
            f_new[js:je+1] *= 0.999
            s = f_new.sum()
            if s > 1e-15:
                f_new = f_new / s

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f} (gen obj={2*n*best_obj:.8f})")
        else:
            no_improve += 1
            f = f_new

        if no_improve > 500:
            rng = np.random.default_rng(it * 7919)
            scale = 0.001 + 0.002 * min(no_improve // 500, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, max_lp_calls=500, verbose=False):
    """LP-based Frank-Wolfe with min-peak LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_lp_calls:
        it += 1
        g = solve_lp_min_peak(f, n, time_limit=5)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f} (gen={2*n*best_obj:.8f})")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            break

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Try adjusting each contributing coordinate
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-10:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj

                for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    ss = f_trial.sum()
                    if ss < 1e-15:
                        continue
                    f_trial = f_trial / ss
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                    if best_j_obj < best_obj - 1e-13:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using adaptive subgradient descent with
Nesterov acceleration, LP-based Frank-Wolfe global moves, coordinate descent polish,
and diverse multi-length exploration. Key improvements: cosine LR schedule,
adaptive peak temperature, periodic deflation, and better time allocation.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, sigma_frac=0.35):
    """Gaussian window normalized to sum=1."""
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / (sigma_frac * n / 2.0)) ** 2)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def make_cosine(n):
    """Cosine window normalized to sum=1."""
    x = np.arange(n)
    a = np.cos(np.pi * x / (n - 1.0)) ** 2 + 0.02
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def make_random(n, seed=42):
    """Random non-negative sequence normalized to sum=1."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    return a


def interpolate_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp


def solve_lp_min_peak(f, n, lp_time_limit=5):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': lp_time_limit})
    if res.success:
        g = res.x[:n]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g /= gs
        return g
    return None


def subgradient_optimize(f0, time_limit, lr_init=0.004, lr_end=0.0001,
                         mom_init=0.90, mom_end=0.97,
                         temp_init=0.00008, temp_end=0.000005,
                         topk_init=50, topk_end=8,
                         deflation_interval=300, verbose=False):
    """
    Subgradient descent with Nesterov momentum, cosine LR decay,
    adaptive temperature, and periodic deflation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    f_look = f.copy()  # Nesterov look-ahead
    conv = fast_convolve(f_look)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)

        cur_lr = lr_end + 0.5 * (lr_init - lr_end) * (1.0 + np.cos(np.pi * frac))
        cur_mom = mom_init + (mom_end - mom_init) * frac
        cur_temp = temp_init * (temp_end / temp_init) ** frac
        cur_topk = max(5, int(topk_init + (topk_end - topk_init) * frac))

        k_actual = min(cur_topk, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / max(cur_temp, 1e-12)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient: sg_j = 2 * sum_k w_k * f_look[k-j]
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                kg = k - j
                if 0 <= kg < n:
                    sg[j] += 2.0 * w[i] * f_look[kg]

        # Nesterov update
        v = cur_mom * v - cur_lr * sg
        f_new = f + v
        f_look = project_simplex(f_new)

        # Gradient step from current f to projected look-ahead
        f_new = project_simplex(f + (f_look - f))

        # Periodic deflation
        if deflation_interval > 0 and it % deflation_interval == 0:
            conv_check = fast_convolve(f_new)
            top_p = int(np.argmax(conv_check))
            js = max(0, top_p - n + 1)
            je = min(n - 1, top_p)
            f_new[js:je + 1] *= 0.998
            ss = f_new.sum()
            if ss > 1e-15:
                f_new = f_new / ss
            else:
                f_new = f.copy()

        conv = fast_convolve(f_new)
        co = conv.max()

        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 1000 == 0:
                print(f"    SG iter {it}: conv_max={best_obj:.10f} gen_score={2.0*n*best_obj:.8f}")
        else:
            no_improve += 1
            f = f_new

        if no_improve > 600:
            rng_local = np.random.default_rng(it * 6271 + 17)
            scale = 0.001 + 0.002 * min(no_improve // 600, 4)
            f = project_simplex(best_f + rng_local.standard_normal(n) * scale)
            v = np.zeros(n)
            f_look = f.copy()
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, max_lp_calls=400, verbose=False):
    """LP-based Frank-Wolfe with min-peak LP and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_lp_calls:
        it += 1
        g = solve_lp_min_peak(f, n, lp_time_limit=5)
        if g is None:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: conv_max={best_obj:.10f} gen_score={2.0*n*best_obj:.8f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 35:
            rng_local = np.random.default_rng(it * 3301)
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent targeting peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        top_k = min(4, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in sorted(peak_positions, key=lambda x: -conv[x]):
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Try adjusting each contributing coordinate
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = f[j]
                if old_val < 1e-10:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj

                for mult in [0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    ss = f_trial.sum()
                    if ss < 1e-15:
                        continue
                    f_trial = f_trial / ss
                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                    if best_j_obj < best_obj - 1e-13:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse starts, adaptive subgradient with Nesterov,
    LP-FW global moves, coordinate descent polish, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not access parent: {e}")

    base_n = len(parent) if parent is not None else 832

    # Generate and evaluate diverse starting points
    print("\n=== Evaluating diverse starting points ===")
    starts = {}
    if parent is not None:
        starts['parent'] = parent
        if base_n != 1024:
            starts['parent_1024'] = interpolate_length(parent, 1024)
        if base_n != 768:
            starts['parent_768'] = interpolate_length(parent, 768)
        if base_n != 1056:
            starts['parent_1056'] = interpolate_length(parent, 1056)

    starts['gauss_832'] = make_gaussian(832, 0.30)
    starts['gauss_1024'] = make_gaussian(1024, 0.30)
    starts['cos_832'] = make_cosine(832)
    starts['rand_832'] = make_random(832, seed)
    starts['rand_1024'] = make_random(1024, seed + 7)

    for name, s in starts.items():
        try_seq(s, f"start_{name}")

    print(f"  Best after starts: {best_score:.8f}")

    # === Phase 1: Main subgradient optimization (longest phase) ===
    print(f"\n=== Phase 1: Main Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.40, 650)
    if t1 > 30 and best_seq is not None:
        print(f"  n={len(best_seq)}, time={t1:.0f}s")
        a_opt, _ = subgradient_optimize(
            best_seq, time_limit=t1,
            lr_init=0.003, lr_end=0.00008,
            mom_init=0.88, mom_end=0.96,
            temp_init=0.00006, temp_end=0.000003,
            topk_init=50, topk_end=8,
            deflation_interval=400, verbose=True)
        try_seq(a_opt, "sg_main")

    # === Phase 2: LP-FW for global moves ===
    print(f"\n=== Phase 2: LP-FW Global (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400)
    if t2 > 30 and best_seq is not None:
        print(f"  n={len(best_seq)}, time={t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, max_lp_calls=350, verbose=True)
        try_seq(a_opt, "lpfw_global")

    # === Phase 3: Subgradient from perturbed state (escape local min) ===
    print(f"\n=== Phase 3: SG Escape (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300)
    if t3 > 25 and best_seq is not None:
        n_b = len(best_seq)
        perturbed = project_simplex(best_seq + rng.standard_normal(n_b) * 0.003)
        a_opt, _ = subgradient_optimize(
            perturbed, time_limit=t3,
            lr_init=0.002, lr_end=0.00005,
            mom_init=0.91, mom_end=0.97,
            temp_init=0.00004, temp_end=0.000002,
            topk_init=30, topk_end=5,
            deflation_interval=500, verbose=True)
        try_seq(a_opt, "sg_escape")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: CD Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 180)
    if t4 > 15 and best_seq is not None:
        print(f"  n={len(best_seq)}, time={t4:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t4, verbose=True)
        try_seq(a_opt, "cd_polish")

    # === Phase 5: Final fine subgradient ===
    print(f"\n=== Phase 5: Final SG (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 250)
    if t5 > 15 and best_seq is not None:
        n_b = len(best_seq)
        a_opt, _ = subgradient_optimize(
            best_seq, time_limit=t5,
            lr_init=0.0008, lr_end=0.00001,
            mom_init=0.95, mom_end=0.99,
            temp_init=0.000015, temp_end=0.000001,
            topk_init=20, topk_end=4,
            deflation_interval=0, verbose=True)
        try_seq(a_opt, "sg_final")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80 and best_seq is not None:
        target_lengths = []
        n_best = len(best_seq)
        for delta in [64, -64, 128, -128, 256]:
            tn = n_best + delta
            if 256 <= tn <= 2048 and tn != n_best:
                target_lengths.append(tn)

        for tn in target_lengths[:3]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_alloc = min(rem * 0.35, 120)
            if t_alloc < 20:
                break
            f_interp = interpolate_length(best_seq, tn)
            print(f"  n={tn}, time={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(
                f_interp, time_limit=t_alloc,
                lr_init=0.0025, lr_end=0.00005,
                mom_init=0.90, mom_end=0.96,
                temp_init=0.00005, temp_end=0.000003,
                topk_init=40, topk_end=6,
                deflation_interval=300)
            try_seq(a_opt, f"len_{tn}")

    # === Final: Quick CD on best ===
    rem = deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 60))
        try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
