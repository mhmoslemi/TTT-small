import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_palindromic(a, n):
    """Create a palindromic sequence of length n from a."""
    half = (n + 1) // 2
    if len(a) >= half:
        a = a[:half]
    else:
        a = np.pad(a, (0, half - len(a)))
    return np.concatenate([a, a[-2::-1]])


def solve_lp_direction(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""
    c = -np.ones(n)
    if active_k is None:
        K = 2 * n - 1
        A_ub = np.zeros((K, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(K, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[row, j] = f[k - j]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def golden_section_line(f, g, n, max_iter=20):
    """Golden section search for best alpha mixing f and g."""
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)

    f_c = project_simplex(f + c * (g - f))
    f_d = project_simplex(f + d * (g - f))
    fc = fast_convolve(f_c).max()
    fd = fast_convolve(f_d).max()

    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            f_c = project_simplex(f + c * (g - f))
            fc = fast_convolve(f_c).max()
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            f_d = project_simplex(f + d * (g - f))
            fd = fast_convolve(f_d).max()

    return (lo + hi) / 2.0


def lp_fw_optimize(f0, time_limit, verbose=False, adaptive_constraints=True,
                   top_k_constraints=40, use_palindrome=False):
    """LP-based Frank-Wolfe with golden section and adaptive constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        active_k = None
        if adaptive_constraints and it > 5:
            k_sorted = np.argsort(conv)[::-1][:top_k_constraints]
            active_k = k_sorted.tolist()
        if it % 40 == 0:
            active_k = None

        g = solve_lp_direction(f, M * m_exp, n, active_k=active_k)
        if g is None:
            m_exp *= 1.12
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        alpha = golden_section_line(f, g, n)

        if alpha > 1e-8:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            if use_palindrome:
                f_new = project_simplex(make_palindromic(f_new, n))

            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            conv = conv_new

            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f_new.copy()
                no_improve = 0
                M = M_new
                m_exp = 1.0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
            f = f_new
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.005 * min(no_improve // 100, 6)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            if use_palindrome:
                f = project_simplex(make_palindromic(f, n))
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coord_desc_fine(f0, time_limit):
    """Coordinate descent with multi-scale local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    multipliers = [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0, 5.0]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 2:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.93, top_k=40, temp=0.00004):
    """Subgradient polishing with softmax-weighted active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        it_count += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.99985
        cur_temp *= 1.004

        if no_improve > 300:
            seed_val = (it_count * 7919 + 13) % 1000000
            rng = np.random.default_rng(seed_val)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0004)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def lopsided_init(n, alpha=1.5):
    """Create a lopsided (increasing) initial sequence."""
    x = np.arange(n) / max(n - 1, 1)
    a = (1 + alpha * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def quadratic_lopsided_init(n, alpha=2.0):
    """Create a quadratically lopsided initial sequence."""
    x = np.arange(n) / max(n - 1, 1)
    a = (1 + alpha * x * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def cheap_subgradient_step(f, conv, lr, n, top_k=30):
    """Single subgradient step using top-k active constraints."""
    k_act = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_act)[-k_act:]

    sg = np.zeros(n)
    for k in peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += f[k - idx]

    # Normalize subgradient
    sg_norm = np.linalg.norm(sg)
    if sg_norm > 1e-15:
        sg /= sg_norm

    f_new = f - lr * sg
    f_new = project_simplex(f_new)
    return f_new

    # === Phase 1: LP-FW on best initialization ===
    print(f"\n=== Phase 1: LP-FW on best init (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.3)
    if t1 > 50:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq if best_seq is not None else parent,
                                  time_limit=t1, verbose=True,
                                  adaptive_constraints=True, top_k_constraints=45)
        try_seq(a_opt, "lpfw_main")

    # === Phase 2: LP-FW on lopsided variants ===
    print(f"\n=== Phase 2: LP-FW on lopsided variants ===")
    for alpha in [1.0, 2.0, 3.0]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        f_init = lopsided_init(n_parent, alpha)
        # Blend with current best
        if best_seq is not None and len(best_seq) == n_parent:
            f_init = project_simplex(0.5 * f_init + 0.5 * best_seq)
        t_alloc = min(rem * 0.2, 150)
        print(f"  Alpha={alpha}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc,
                                  adaptive_constraints=True, top_k_constraints=35)
        try_seq(a_opt, f"lpfw_lop_{alpha}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration ===")
    for target_n in [384, 512, 640, 896, 1024]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        if target_n == n_parent and best_seq is not None:
            continue
        # Initialize by interpolation + lopsided bias
        base = best_seq if best_seq is not None else parent
        f_interp = interpolate_sequence(base, target_n)
        f_lop = lopsided_init(target_n, 1.5)
        f_mix = project_simplex(0.7 * f_interp + 0.3 * f_lop)
        t_alloc = min(rem * 0.15, 120)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_mix, time_limit=t_alloc,
                                  adaptive_constraints=True, top_k_constraints=30)
        try_seq(a_opt, f"multi_n{target_n}")

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_cd = min(rem * 0.25, 250)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coord_desc_fine(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_main")

            rem = deadline - time.monotonic()
            if rem > 20:
                t_cd2 = min(rem * 0.15, 120)
                a_opt2, _ = coord_desc_fine(a_opt, time_limit=t_cd2)
                try_seq(a_opt2, "cd_refine")

    # === Phase 5: Subgradient methods ===
    print(f"\n=== Phase 5: Subgradient Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 20:
            t_sg = min(rem * 0.2, 150)
            a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                                 lr=0.0015, mom=0.94,
                                 top_k=max(20, n_best // 5),
                                 temp=0.00003)
            try_seq(a_opt, "sg_main")

        rem = deadline - time.monotonic()
        if rem > 15:
            t_sg2 = min(rem * 0.15, 80)
            a_opt2, _ = batch_sg_optimize(best_seq, time_limit=t_sg2,
                                         top_k=max(25, n_best // 4),
                                         lr_init=0.03, decay=0.9995)
            try_seq(a_opt2, "batch_sg")

    # === Phase 6: Diverse restarts and deep optimization ===
    print(f"\n=== Phase 6: Diverse Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)

        restart_configs = []
        restart_configs.append(("noise1", project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)))
        restart_configs.append(("noise2", project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)))
        restart_configs.append(("rev_blend", project_simplex(0.6 * best_seq + 0.4 * best_seq[::-1])))
        restart_configs.append(("flat_blend", project_simplex(0.8 * best_seq + 0.2 * np.ones(n_best) / n_best)))
        restart_configs.append(("lopesque", project_simplex(0.7 * best_seq + 0.3 * lopsided_init(n_best, 2.0))))
        restart_configs.append(("sparse_peak", project_simplex(
            np.exp(-2.0 * (np.arange(n_best) / n_best - 0.7) ** 2 * 5))))

        for i, (name, f_init) in enumerate(restart_configs):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem / max(1, len(restart_configs) - i), 120)
            print(f"  Restart [{name}], t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc,
                                      verbose=False, adaptive_constraints=True,
                                      top_k_constraints=30)
            try_seq(a_opt, f"restart_{name}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 15:
            t_final = min(rem * 0.3, 100)
            print(f"  Final LP-FW t={t_final:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_final,
                                      adaptive_constraints=False)
            try_seq(a_opt, "final_lpfw_full")

        rem = deadline - time.monotonic()
        if rem > 10:
            t_cd_final = min(rem * 0.3, 80)
            a_opt, _ = coord_desc_fine(best_seq, time_limit=t_cd_final)
            try_seq(a_opt, "final_cd")

        # Tiny perturbations
        rem = deadline - time.monotonic()
        for i in range(5):
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.001 * i)
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.15, 30)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_pert,
                                      adaptive_constraints=True, top_k_constraints=25)
            try_seq(a_opt, f"perturb_{i}")

        # Final SG
        rem = deadline - time.monotonic()
        if rem > 8:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem * 0.4, 30),
                                 lr=0.0005, mom=0.95,
                                 top_k=max(10, n_best // 10),
                                 temp=0.00001)
            try_seq(a_opt, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
