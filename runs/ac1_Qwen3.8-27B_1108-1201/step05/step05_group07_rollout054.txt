    return np.fft.irfft(A * A, fft_size)[:size]

def solve_lp(f, M):
    """Solve max sum(g) s.t. g >= 0, (f*g)_k <= M for all k"""
    n = len(f)
    # A matrix: A[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k)
    # This is a banded matrix

    A = np.zeros((2*n-1, n))

        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end+1] = f[j_start:j_end+1]

    c = -np.ones(n)

    return None

Key innovation: At each step, solve a linear program to find a direction g

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_A_matrix(f):
    """Build the constraint matrix A where A[k,i] = f[k-i] if valid."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):

        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)

        A[k, i_start:i_end+1] = f[k-i_end:k-i_start+1][::-1]
    return A

    if res.success:
        return np.maximum(res.x, 0)
    return None

    no_improve = 0

        M = conv.max()
        
        g0 = solve_lp(f, M)

        s_g = g0.sum()
        if s_g < 1e-10:

        best_new_f = f.copy()

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
At each iteration, solves a linear program: max sum(g) s.t. g>=0, (f*g)_k<=M for all k.
This provides an informed descent direction that is much better than subgradient.
Combined with local search and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def build_A_matrix(f):
    """Build constraint matrix A[k,i] = f[k-i] if 0<=k-i<n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            # A[k, i] = f[k-i] for i in [i_start, i_end]
            # j = k-i ranges from k-i_end to k-i_start
            A[k, i_start:i_end + 1] = f[k - i_end:k - i_start + 1][::-1]
    return A


def solve_lp(f, M):
    """Solve max sum(g) s.t. g>=0, A@g <= M*1. Returns g or None."""
    n = len(f)
    A = build_A_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f0, time_limit, t_init=0.3, verbose=False, label=""):
    """LP-based iterative optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    t = t_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        g0 = solve_lp(f, M)
        if g0 is None:
            break

        s_g = g0.sum()
        if s_g < 1e-10:
            break

        g = g0 / s_g

        # Try blend factors
        best_t = 0
        best_new_obj = best_obj
        best_new_f = f.copy()

        for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_new = (1 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new /= sn
            obj = fast_convolve(f_new).max()
            if obj < best_new_obj - 1e-14:
                best_new_obj = obj
                best_new_f = f_new.copy()
                best_t = t_try

        if best_new_obj < best_obj - 1e-13:
            best_obj = best_new_obj
            best_f = best_new_f.copy()
            f = best_new_f
            no_improve = 0
            if best_t != 0:
                t = best_t
        else:
            no_improve += 1
            t = max(0.02, t * 0.8)
            if no_improve > 100:
                break

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} LP iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def mu_weights_optimize(f0, time_limit, top_k=50, temperature=0.00005, lr=0.003,
                        momentum=0.85, verbose=False, label=""):
    """Multiplicative-weights subgradient optimization with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1
        f_nest = f * np.exp(0.5 * momentum * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        k_actual = min(cur_top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]

        v = momentum * v + (1 - momentum) * sg
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        if cur_lr < 0.00005:
            cur_lr = 0.00005
        cur_temp *= 1.008
        if it % 500 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.95))

        if no_improve > 300:
            rng = np.random.default_rng((it * 7919 + 13) % 1000000)
            noise_scale = 0.002 + 0.001 * min(no_improve // 500, 5)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 0)
            f /= f.sum()
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0002)
            cur_temp = temperature * 0.5
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MU iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def local_search(f0, time_limit, verbose=False):
    """Local search: random pair transfers and peak-directed moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        current_max = conv.max()
        top_k = min(20, 2 * n - 1)
        top_peaks = np.argpartition(conv, -top_k)[-top_k:]

        rng = np.random.default_rng(int(time.monotonic() * 1e6) % 2000000)
        trials = 300
        for _ in range(trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            old_i = f[i]
            if old_i < 1e-10:
                continue
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = old_i * frac
                f[i] = old_i - amt
                f[j] = f[j] + amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] = f[j] - amt
                    break

        if not improved:
            # Peak-directed
            for k in top_peaks[:8]:
                if time.monotonic() - t0 > time_limit:
                    break
                k = int(k)
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                contribs = f[idx] * f[k - idx]
                hi = idx[np.argmax(contribs)]
                lo = idx[np.argmin(contribs)]
                if hi == lo or f[hi] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[hi] * frac
                    f[hi] -= amt
                    f[lo] += amt
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-13:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[hi] += amt
                        f[lo] -= amt
                        break

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate structured initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-12)
    inits.append(hann / hann.sum())

    # Raised cosine (broader)
    rc = 0.5 * (1 + np.cos(np.pi * (2 * x - 1)))
    rc = np.maximum(rc, 1e-12)
    inits.append(rc / rc.sum())

    # Trapezoidal
    L = int(n * (0.3 + rng.random() * 0.2))
    trap = np.zeros(n)
    trap[:L] = np.linspace(0, 1, L)
    trap[-L:] = np.linspace(1, 0, L)
    if L < n // 2:
        trap[L:n-L] = 1.0
    else:
        trap = np.maximum(trap[:L], trap[n-L:][::-1])
    trap = np.maximum(trap, 1e-12)
    inits.append(trap / trap.sum())

    # Gaussian
    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 3.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Watermelon (concentrated center)
    wm = np.exp(-((np.arange(n) - center) / (n * 0.1)) ** 4)
    inits.append(wm / wm.sum())

    return inits[:count]

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: At each step, solve a linear program (max sum(g) s.t. g>=0, (f*g)_k<=M)
to find an informed descent direction, then blend with current solution.
Combined with subgradient refinement, local search, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_A_matrix(f):
    """Build constraint matrix A[k,i] = f[k-i] if 0<=k-i<n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j_start = k - i_end
            j_end = k - i_start
            A[k, i_start:i_end + 1] = f[j_start:j_end + 1]
    return A


def solve_lp(f, M):
    """Solve max sum(g) s.t. g>=0, (f*g)_k <= M for all k. Returns g or None."""
    n = len(f)
    A = build_A_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f0, time_limit, t_init=0.3, verbose=False, label=""):
    """LP-based iterative optimization with adaptive blending."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    t = t_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        g0 = solve_lp(f, M)
        if g0 is None:
            break

        s_g = g0.sum()
        if s_g < 1e-10:
            break

        g = g0 / s_g

        best_t = 0
        best_new_obj = best_obj
        best_new_f = f.copy()

        for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new /= sn
            obj = fast_convolve(f_new).max()
            if obj < best_new_obj - 1e-14:
                best_new_obj = obj
                best_new_f = f_new.copy()
                best_t = t_try

        if best_new_obj < best_obj - 1e-13:
            best_obj = best_new_obj
            best_f = best_new_f.copy()
            f = best_new_f
            no_improve = 0
            if best_t > 0:
                t = best_t
        else:
            no_improve += 1
            t = max(0.02, t * 0.75)
            if no_improve > 80:
                break

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} LP iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, blend={t:.3f}")

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, top_k=50, temperature=0.00004, lr=0.003,
                         momentum=0.87, verbose=False, label=""):
    """Multiplicative-weights subgradient with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temperature
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1
        f_nest = f * np.exp(0.5 * momentum * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        k_actual = min(cur_top_k, len(conv_nest))
        peaks = np.argpartition(conv_nest, -k_actual)[-k_actual:]
        pv = conv_nest[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f_nest[k - idx]

        v = momentum * v + (1.0 - momentum) * sg
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.3, 0.0001)
            continue
        f /= fs
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9996
        if cur_lr < 0.00004:
            cur_lr = 0.00004
        cur_temp *= 1.007
        if it % 500 == 0:
            cur_top_k = max(5, int(cur_top_k * 0.93))

        if no_improve > 250:
            rng = np.random.default_rng((it * 104729 + 7) % 1000000)
            noise_scale = 0.002 + 0.001 * min(no_improve // 400, 5)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 0.0)
            f /= f.sum()
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            cur_temp = temperature * 0.4
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} SG iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def local_search(f0, time_limit, verbose=False):
    """Local search: random pair transfers and peak-directed moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        top_k = min(15, 2 * n - 1)
        top_peaks = np.argpartition(conv, -top_k)[-top_k:]

        rng = np.random.default_rng(int(time.monotonic() * 1e6) % 2000000)
        trials = 200
        for _ in range(trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            old_i = f[i]
            if old_i < 1e-10:
                continue
            for frac in [0.03, 0.08, 0.15, 0.25]:
                amt = old_i * frac
                f[i] = old_i - amt
                f[j] = f[j] + amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-14:
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[i] = old_i
                    f[j] = f[j] - amt
                    break

        if not improved:
            for pk in top_peaks[:6]:
                if time.monotonic() - t0 > time_limit:
                    break
                k = int(pk)
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                contribs = f[idx] * f[k - idx]
                hi = idx[np.argmax(contribs)]
                lo = idx[np.argmin(contribs)]
                if hi == lo or f[hi] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[hi] * frac
                    f[hi] -= amt
                    f[lo] += amt
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[hi] += amt
                        f[lo] -= amt
                        break

        if not improved:
            stall += 1
            if stall > 40:
                break
        else:
            stall = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences."""
    inits = []
    x = np.linspace(0, 1, n)
    center = n / 2.0

    inits.append(np.ones(n) / n)

    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    hann = np.maximum(hann, 1e-12)
    inits.append(hann / hann.sum())

    rc = 0.5 * (1.0 + np.cos(np.pi * (2.0 * x - 1.0)))
    rc = np.maximum(rc, 1e-12)
    inits.append(rc / rc.sum())

    L = max(2, int(n * (0.25 + rng.random() * 0.2)))
    trap = np.zeros(n)
    ramp = np.linspace(0, 1, L)
    trap[:L] = ramp
    trap[-L:] = ramp[::-1]
    if L < n // 2:
        trap[L:n - L] = 1.0
    trap = np.maximum(trap, 1e-12)
    inits.append(trap / trap.sum())

    sigma = n / (2.5 + rng.random() * 3.5)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    wm = np.exp(-((np.arange(n) - center) / (n * 0.08)) ** 4)
    inits.append(wm / wm.sum())

    # S-shaped (sigmoid)
    sharp = 8.0 + rng.random() * 10.0
    sig = 1.0 / (1.0 + np.exp(-sharp * (x - 0.3)))
    sig = sig - 1.0 / (1.0 + np.exp(-sharp * (x - 0.7)))
    sig = np.maximum(sig, 1e-12)
    inits.append(sig / sig.sum())

    return inits[:count]

            a_opt, _ = lp_optimize(best_seq, time_limit=t_lp, t_init=0.08, label="P5_lp")
            try_seq(a_opt, "lp_post_ls")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine LP
        rem = deadline - time.monotonic()
        if rem > 20:
            t_f = min(rem * 0.4, 50.0)
            a_opt, _ = lp_optimize(best_seq, time_limit=t_f, t_init=0.05, label="P6_lp")
            try_seq(a_opt, "final_lp")

        # Fine local search
        rem = deadline - time.monotonic()
        if rem > 15:
            t_f = min(rem * 0.4, 35.0)
            a_opt, _ = local_search(best_seq, time_limit=t_f)
            try_seq(a_opt, "final_ls")

        # Final LP
        rem = deadline - time.monotonic()
        if rem > 10:
            t_f = min(rem * 0.5, 25.0)
            a_opt, _ = lp_optimize(best_seq, time_limit=t_f, t_init=0.03, label="P6_lp2")
            try_seq(a_opt, "final_lp2")

        # Perturbation + quick LP
        for scale in [0.001, 0.003]:
            rem = deadline - time.monotonic()
            if rem < 12:
                break
            noise = rng.standard_normal(n_best) * scale
            f_pert = np.maximum(best_seq + noise, 0.0)
            f_pert /= f_pert.sum()
            t_f = min(rem * 0.4, 15.0)
            a_opt, _ = lp_optimize(f_pert, time_limit=t_f, t_init=0.1, label=f"pert{scale}")
            try_seq(a_opt, f"pert_{scale}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
