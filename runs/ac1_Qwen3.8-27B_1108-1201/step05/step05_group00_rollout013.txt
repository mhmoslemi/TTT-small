combined with LP-based Frank-Wolfe and active-set refinement.
Key innovations:
- Smooth p-norm objective with gradually increasing p
- Efficient gradient via cross-correlation
- Hybrid LP/p-norm oracle selection
- Structured initializations from Fourier analysis

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, weights):
    """
    Compute gradient of sum_k weights[k] * conv(a,a)[k] w.r.t. a.
    weights: array of length 2n-1
    Returns: array of length n
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # grad[m] = 2 * sum_{j=0}^{n-1} weights[m+j] * a[j]  (for m+j < 2n-1)
    # This is a cross-correlation: (weights * a)[m] = sum_j weights[m+j]*a[j]
    # = correlate(weights, a) at offset m
    # Using FFT: ifft(fft(weights, N) * conj(fft(a, N))) gives the cross-correlation
    N = 1 << (2 * n - 1 - 1).bit_length()
    # Actually we need N >= len(weights) + n - 1 = 3n - 2
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.fft(weights, N)
    A = np.fft.fft(a, N)
    corr = np.fft.ifft(W * np.conj(A)).real
    # corr[m] = sum_j weights[j] * conj(a[j-m]) for the circular version
    # For linear: corr[m] = sum_j weights[j+m] * a[j] (for m = 0, ..., n-1)
    # The FFT cross-correlation: ifft(fft(w) * conj(fft(a)))[m] = sum_j w[j] * a[j-m]
    # (circular). For linear, we need N >= len(w) + len(a) - 1.
    # ifft(fft(w) * conj(fft(a)))[m] = sum_j w[j] * a[(j-m) mod N]
    # For m in [0, n-1] and N large enough: = sum_j w[j] * a[j-m] (linear)
    # = sum_k w[k+m] * a[k] (substituting k = j-m)
    # So corr[m] = sum_k weights[k+m] * a[k] for k = 0..n-1 where k+m < 2n-1
    # This is exactly what we want!
    return 2.0 * corr[:n]


def p_norm_objective(a, p):
    """Compute ||conv(a)||_p = (sum_k conv[k]^p)^(1/p)"""
    c = fast_convolve(a)
    c = np.maximum(c, 0.0)  # conv of non-negative is non-negative
    return (np.sum(c ** p)) ** (1.0 / p)


def p_norm_grad(a, p):
    """Gradient of (sum_k conv[k]^p)^(1/p) w.r.t. a."""
    c = fast_convolve(a)
    c = np.maximum(c, 1e-30)  # avoid 0^(p-1)
    # d/d a of (sum c^p)^(1/p) = (1/p) * (sum c^p)^(1/p - 1) * p * sum c^(p-1) * dc/da
    # = (sum c^p)^(1/p - 1) * sum_k c[k]^(p-1) * dc[k]/da
    # sum_k c[k]^(p-1) * dc[k]/da = compute_grad(a, c^(p-1)) / 2
    # Wait: (sum c^p)^(1/p - 1) is a scalar.
    # And sum_k c[k]^(p-1) * dc[k]/da[m] = (1/2) * compute_grad(a, c^(p-1))[m]
    #   (because compute_grad returns 2 * sum_k weights[k] * dc[k]/da[m])
    
    val = np.sum(c ** p)
    if val < 1e-30:
        return np.zeros(len(a))
    factor = val ** (1.0 / p - 1.0)
    w = c ** (p - 1)
    grad = factor * compute_grad(a, w) / 2.0  # compute_grad gives 2x the sum
    # Wait, compute_grad(a, w) = 2 * sum_k w[k] * dc[k]/da
    # We want: sum_k w[k] * dc[k]/da = compute_grad(a, w) / 2
    # So grad = factor * compute_grad(a, w) / 2
    return grad


def adaptive_p_descent(a0, time_limit, p_start=4, p_end=100, lr_init=0.1):
    """
    Gradient descent on the p-norm of the auto-convolution.
    Gradually increases p from p_start to p_end.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = lr_init
    p = p_start
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        c = fast_convolve(a)
        current_max = c.max()
        
        # Compute p-norm gradient
        c_pos = np.maximum(c, 1e-15)
        val = np.sum(c_pos ** p)
        if val < 1e-30:
            break
        factor = val ** (1.0 / p - 1.0)
        w = c_pos ** (p - 1)
        grad = factor * compute_grad(a, w) / 2.0
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Line search
        best_alpha = 0.0
        best_val = current_max
        
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - trial_lr * grad / gnorm)
            cv = fast_convolve(candidate)
            v = cv.max()
            if v < best_val:
                best_val = v
                best_alpha = trial_lr
        
        if best_alpha > 0:
            a = project_simplex(a - best_alpha * grad / gnorm)
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adapt
        if no_improve > 30:
            p = min(p * 1.1, p_end)
            lr *= 0.9
        if no_improve > 100:
            # Restart with perturbation
            rng_local = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng_local.standard_normal(n) * 0.005)
            p = p_start
            lr = lr_init * 0.5
            no_improve = 0
        
        # Gradually increase p
        p = min(p_end, p * 1.001)
    
    return best_a, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp, g >= 0
        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp *= 1.5
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 80:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            M = fast_convolve(f).max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fourier_init(n, rng):
    """
    Generate an initialization with near-flat power spectrum.
    Start with random phases, take IFFT, project to non-negative.
    """
    N = 1 << (2 * n - 1 - 1).bit_length()
    # Random phases, constant magnitude
    mags = np.ones(N)
    phases = rng.uniform(0, 2 * np.pi, N)
    # Enforce Hermitian symmetry for real signal
    for k in range(1, N // 2):
        mags[N - k] = mags[k]
        phases[N - k] = -phases[k]
    
    spectrum = mags * np.exp(1j * phases)
    sig = np.fft.ifft(spectrum).real[:n]
    
    # Project to non-negative simplex
    sig = np.maximum(sig, 0)
    s = sig.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return sig / s


def active_set_polish(a0, time_limit, top_k=10):
    """
    Focus on reducing the top-k peaks specifically.
    For each peak, solve a local optimization to reduce it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        c = fast_convolve(a)
        
        # Find top peaks
        top_indices = np.argpartition(c, -top_k)[-top_k:]
        top_indices = top_indices[np.argsort(c[top_indices])[::-1]]
        
        improved = False
        for k_peak in top_indices:
            # Variables that affect this peak: a[j] where j and k_peak-j are both valid
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            
            if js > je:
                continue
            
            affected = list(range(js, je + 1))
            
            # Try to reduce this peak by redistributing mass among affected variables
            # Goal: minimize sum_{j in affected} a[j]*a[k_peak-j] while keeping sum(a)=1
            # This is a quadratic program on the affected variables
            
            # Simple approach: try to make the affected variables more uniform
            # (reducing the product)
            current_vals = a[affected]
            current_sum = current_vals.sum()
            
            if current_sum < 1e-10:
                continue
            
            # Perturb: move mass from high-product pairs to low-product ones
            products = np.array([a[j] * a[k_peak - j] for j in affected])
            if products.max() < 1e-15:
                continue
            
            # Move a tiny amount from the pair with max product
            max_idx_in_affected = np.argmax(products)
            j1 = affected[max_idx_in_affected]
            j2 = k_peak - j1
            
            if j1 == j2:
                continue
            if j1 >= n or j2 >= n or j1 < 0 or j2 < 0:
                continue
            
            # Try moving delta from j1 to j2 (or vice versa)
            delta = 0.001 * a[j1]
            for d in [delta, delta * 0.5, delta * 2.0]:
                candidate = a.copy()
                candidate[j1] -= d
                if candidate[j1] < 0:
                    continue
                # Put the mass where it won't increase the peak
                # Try putting it at a position with minimal contribution
                min_contrib_pos = None
                min_contrib_val = np.inf
                for pos in range(n):
                    contrib = a[pos] * a[k_peak - pos] if 0 <= k_peak - pos < n else 0
                    if pos != j1 and pos != j2:
                        # Adding d at pos: new contribution = (a[pos]+d) * a[k_peak-pos]
                        # if k_peak-pos is valid
                        if 0 <= k_peak - pos < n:
                            new_c = (a[pos] + d) * a[k_peak - pos]
                            old_c = a[pos] * a[k_peak - pos]
                            delta_c = new_c - old_c = d * a[k_peak - pos]
                            if delta_c < min_contrib_val:
                                min_contrib_val = delta_c
                                min_contrib_pos = pos
                
                if min_contrib_pos is None:
                    continue
                
                candidate[min_contrib_pos] += d
                s = candidate.sum()
                if s < 1e-10:
                    continue
                candidate = candidate / s
                
                c_new = fast_convolve(candidate)
                peak_new = c_new[k_peak]
                overall_max = c_new.max()
                
                if overall_max < best_obj - 1e-15:
                    a = candidate
                    best_obj = overall_max
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Try a global perturbation
            rng = np.random.default_rng((it * 48271 + 3) % (2**31))
            candidate = project_simplex(a + rng.standard_normal(n) * 0.002)
            c_new = fast_convolve(candidate)
            if c_new.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c_new.max()
                best_a = a.copy()
            elif it % 100 == 0:
                break
    
    return best_a, best_obj
