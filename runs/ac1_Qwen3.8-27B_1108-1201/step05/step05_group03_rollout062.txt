```python
    """Compute the gradient of the soft-max approximation of max(conv(f,f))."""
    n = len(f)
    conv = fast_convolve(f)

    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]

    w_peaks = np.exp(scaled)
    w_peaks /= w_peaks.sum()
    
    # Build full weight vector
    w = np.zeros(2*n-1)
    w[peaks] = w_peaks
    
    # Gradient via cross-correlation
    corr = np.correlate(w, f, 'full')
    grad = 2.0 * corr[n-1:2*n-1]
    
    return grad, conv
```

For each coordinate i:

2. For the current f[i] = old:

     - If k ≠ 2i: 2*(t-old)*f[k-i]
     - If k = 2i: t^2 - old^2
   - For k outside [i, i+n-1]: 0
3. Define h(t) = 2n * max_k (conv[k] + delta_k(t)) / (sum + t - old)^2
4. Minimize h(t) over t in [0, 10*sum/n] (some reasonable upper bound)
5. Use golden section search

The expensive part is computing max_k for each evaluation of h(t). But since delta_k is linear in t (except at k=2i), we can precompute:
- A[k] = 2*f[k-i] for k in [i, i+n-1], k ≠ 2i
- B = 1 (coefficient of t^2 at k=2i)
- base[k] = conv[k] + A[k]*(-old) for k in [i, i+n-1], k ≠ 2i  (the part not involving t)
  Actually: conv[k] + delta_k(t) = conv[k] + A[k]*(t-old) for k ≠ 2i
  = (conv[k] - A[k]*old) + A[k]*t
  
  For k=2i: conv[2i] + t^2 - old^2 = (conv[2i] - old^2) + t^2

- For k ≠ 2i in range: new_conv[k] = C[k] + A[k]*t where C[k] = conv[k] - A[k]*old
- For k = 2i: new_conv[2i] = C2 + t^2 where C2 = conv[2i] - old^2
- For k outside: new_conv[k] = conv[k]

The max is: max(max(C[k] + A[k]*t for k in range, k≠2i), C2 + t^2, max(conv[k] for k outside range))

The last term is a constant: M_out = max(conv[k] for k outside [i, i+n-1])

The first term is the max of linear functions of t (convex in t).
The second term is a quadratic in t (convex in t).
The overall max is convex in t.

The numerator is convex in t (for t >= 0), and the denominator is (sum + t - old)^2 which is also a function of t. The ratio of a convex function to a quadratic isn't necessarily convex, but in practice it should be unimodal.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast gradient via cross-correlation with log-sum-exp smoothing
2. Adam optimizer with adaptive temperature
3. LP-based Frank-Wolfe with active constraints (top-K only)
4. Coordinate descent with golden-section 1D optimization
5. Multi-length, multi-shape exploration
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_max_grad(f, T, top_k):
    """Compute gradient of soft-max approximation of max(conv(f,f))."""
    n = len(f)
    conv = fast_convolve(f)
    k_act = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / T
    w_peaks = np.exp(scaled)
    w_sum = w_peaks.sum()
    if w_sum < 1e-30:
        w_peaks = np.zeros(len(peaks))
        w_peaks[np.argmax(pv)] = 1.0
    else:
        w_peaks /= w_sum
    w = np.zeros(2 * n - 1)
    w[peaks] = w_peaks
    corr = np.correlate(w, f, 'full')
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, conv


def adam_optimize(f0, time_limit, verbose=False):
    """Adam optimizer on log-sum-exp smoothed objective with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_obj = 2.0 * n * fast_convolve(f).max() / s**2
    best_f = f.copy()
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.003
    T_start = 3.0 / n
    T_end = 0.3 / n
    top_k_start = max(100, n)
    top_k_end = max(20, n // 4)
    it = 0
    no_improve = 0
    max_iters = int(max(1, time_limit / 0.012))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(1.0, it / max(1, max_iters * 0.85))
        T = T_start * (T_end / T_start) ** progress
        top_k = max(10, int(top_k_start + (top_k_end - top_k_start) * progress))
        lr = base_lr * (1.0 - 0.4 * progress)
        grad, conv = compute_soft_max_grad(f, T, top_k)
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f - step)
        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            if no_improve > 200:

                m, v = np.zeros(n), np.zeros(n)

        f = f_new
        new_M = fast_convolve(f).max()
        new_obj = 2.0 * n * new_M

            best_obj = new_obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Adam it={it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
    return best_f, best_obj

    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break
        conv = fast_convolve(f)
        M = conv.max()
        # Get top-K constraints
        k_act = min(K_constraints, 2 * n - 1)
        active_convs = np.argpartition(conv, -k_act)[-k_act:]
        active_Ms = conv[active_convs]
        active_k = np.where(active_Ms >= M * 0.95)[0]
        if len(active_k) == 0:
            active_k = np.where(active_Ms == active_Ms.max())[0]
        active_convs = active_convs[active_k]

        # Build LP: max sum(g) s.t. (f*g)[k] <= M for k in active_convs, g >= 0
        n_c = len(active_convs)
        A_ub = np.zeros((n_c, n))
        for idx, k in enumerate(active_convs):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            A_ub[idx, js:je + 1] = f[js:je + 1]
        b_ub = np.full(n_c, M)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if not res.success or res.x is None:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 73 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0
            continue
        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 91 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                no_improve = 0
            continue
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 1e-8:
            f_new = np.maximum(f + best_alpha * (g - f), 0)
            f_new /= f_new.sum()
            new_M = fast_convolve(f_new).max()
            f = f_new
            if new_M < best_obj - 1e-14:
                best_obj = new_M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW it={it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 80:
            rng = np.random.default_rng(it * 43 + 17)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with golden-section 1D optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    sum_f = f.sum()
    best_obj = 2.0 * n * conv.max() / sum_f**2
    best_f = f.copy()
    best_conv = conv.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    no_improve_passes = 0
    gr = (np.sqrt(5) - 1) / 2

    while time.monotonic() - t0 < time_limit and no_improve_passes < 3:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old = f[i]
            if old < 1e-14:
                continue
            k_start = i
            k_end = min(2 * n - 1, i + n)
            k_range = np.arange(k_start, k_end)
            k_lin_mask = k_range != 2 * i
            k_lin = k_range[k_lin_mask]
            if len(k_lin) > 0:
                A = 2.0 * f[k_lin - i]
                C = best_conv[k_lin] - A * old
            else:
                A = np.array([])
                C = np.array([])
            k2i = 2 * i
            if k2i < 2 * n:
                C2 = best_conv[k2i] - old * old
            else:
                C2 = -np.inf
            mask_out = np.ones(2 * n - 1, dtype=bool)
            mask_out[k_range] = False
            M_out = np.max(best_conv[mask_out]) if np.any(mask_out) else -np.inf
            s0 = sum_f - old

            # Golden section on t in [0, hi]
            lo = 0.0
            hi = max(s0 * 0.5, 3.0 * sum_f / n)

            def obj_at(t):
                if t < 0 or s0 + t < 1e-15:
                    return np.inf
                mc = M_out
                if len(A) > 0:
                    mc = max(mc, float(np.max(C + A * t)))
                if k2i < 2 * n:
                    mc = max(mc, C2 + t * t)
                return 2.0 * n * mc / ((s0 + t) ** 2)

            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = obj_at(c1)
            f2 = obj_at(c2)
            for _ in range(35):
                if hi - lo < 1e-13 * max(1.0, sum_f):
                    break
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo)
                    f1 = obj_at(c1)
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo)
                    f2 = obj_at(c2)
            t_opt = (lo + hi) / 2.0
            obj_opt = obj_at(t_opt)
            if obj_opt < best_obj - 1e-12:
                # Update f and conv
                f[i] = t_opt
                sum_f += (t_opt - old)
                for k in k_lin:
                    best_conv[k] += 2.0 * (t_opt - old) * f[k - i]
                if k2i < 2 * n:
                    best_conv[k2i] += t_opt * t_opt - old * old
                best_obj = obj_opt
                best_f = f.copy()
                improved = True
        if improved:
            no_improve_passes = 0
            if verbose:
                print(f"    CD pass done: obj={best_obj:.10f}")
        else:
            no_improve_passes += 1
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate to new length and renormalize."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    return fi


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    g = np.exp(-(x - center) ** 2 / (2 * sigma ** 2))
    return g / g.sum()


def make_cosine(n):
    x = np.arange(n)
    c = np.cos(np.pi * (x - (n - 1) / 2.0) / (n / 2.0 + 0.5))
    c = np.maximum(c, 0)
    if c.sum() < 1e-10:
        c = np.ones(n)
    return c / c.sum()


def make_asymmetric(n, alpha=0.5):
    x = np.arange(n)
    a = np.exp(-alpha * x / (n - 1 + 1e-10) * 2)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: diverse initializations, Adam, LP-FW, coordinate descent,
    multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 0.01:
            return
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception):
        pass

    if parent is None:
        parent = np.ones(384) / 384.0
        best_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    n_p = len(parent)

    # Phase 1: Adam on parent
    print("\n=== Phase 1: Adam on parent ===")
    rem = deadline - time.monotonic() - 800
    t1 = min(rem, 300)
    if t1 > 30:
        print(f"  t={t1:.0f}s, n={n_p}")
        a_opt, _ = adam_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "adam_parent")

    # Phase 2: LP-FW on best
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 650
        t2 = min(rem, 200)
        if t2 > 20:
            K = min(60, len(best_seq) // 4)
            print(f"  t={t2:.0f}s, n={len(best_seq)}, K={K}")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, K_constraints=K, verbose=True)
            try_seq(a_opt, "lpfw_p2")

    # Phase 3: Coordinate descent
    print(f"\n=== Phase 3: Coord Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        t3 = min(rem, 250)
        if t3 > 20:
            print(f"  t={t3:.0f}s, n={len(best_seq)}")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "cd_p3")

    # Phase 4: Multi-length exploration with diverse shapes
    print(f"\n=== Phase 4: Multi-length diverse (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    base_n = len(base)

    shapes = [
        ("uniform", np.ones(base_n) / base_n),
        ("gaussian", make_gaussian(base_n, 0.25)),
        ("cosine", make_cosine(base_n)),
        ("asym", make_asymmetric(base_n, 0.8)),
        ("interp_base", base),
    ]

    for target_n in [384, 512, 768, 1024]:
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break
        for shape_name, shape in shapes:
            rem = deadline - time.monotonic() - 180
            if rem < 30:
                break
            if shape_name == "interp_base":
                f_init = interpolate_sequence(base, target_n)
            else:
                f_init = interpolate_sequence(shape, target_n)

            t_a = min(rem * 0.45, 120)
            if t_a > 15:
                a_opt, _ = adam_optimize(f_init, time_limit=t_a)
                try_seq(a_opt, f"adam_n{target_n}_{shape_name}")
                rem = deadline - time.monotonic() - 160
                if rem > 20:
                    t_c = min(rem * 0.5, 100)
                    if t_c > 10:
                        a_opt2, _ = coordinate_descent(a_opt, time_limit=t_c)
                        try_seq(a_opt2, f"cd_n{target_n}_{shape_name}")

    # Phase 5: LP-FW on best at best length
    print(f"\n=== Phase 5: LP-FW final (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            K = min(50, len(best_seq) // 5)
            t5 = min(rem, 150)
            print(f"  t={t5:.0f}s, n={len(best_seq)}, K={K}")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t5, K_constraints=K, verbose=True)
            try_seq(a_opt, "lpfw_p5")

    # Phase 6: More coordinate descent
    print(f"\n=== Phase 6: CD final (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t6 = min(rem, 120)
            print(f"  t={t6:.0f}s, n={len(best_seq)}")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t6, verbose=True)
            try_seq(a_opt, "cd_p6")

    # Phase 7: Final polish - Adam + CD alternating
    print(f"\n=== Phase 7: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_b = len(best_seq)
        for trial in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            rng = np.random.default_rng(700 + trial * 31)
            f_p = project_simplex(best_seq + rng.standard_normal(n_b) * 0.003)
            t_a = min(rem * 0.4, 40)
            if t_a > 8:
                a_opt, _ = adam_optimize(f_p, time_limit=t_a)
                try_seq(a_opt, f"polish_adam_{trial}")
            rem = deadline - time.monotonic() - 20
            if rem > 10:
                a_opt, _ = coordinate_descent(a_opt, time_limit=min(rem, 40))
                try_seq(a_opt, f"polish_cd_{trial}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
```
