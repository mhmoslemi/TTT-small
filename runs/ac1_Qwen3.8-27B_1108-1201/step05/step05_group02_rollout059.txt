"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Smoothed gradient descent with adaptive softmax temperature and Nesterov momentum
2. LP-based Frank-Wolfe refinement
3. Peak-targeting coordinate descent
4. Multi-length exploration
5. Final polish

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def compute_smoothed_grad(f, temp, top_k=None):
    """Compute smoothed gradient of max(conv(f,f)) using softmax weighting.
    
    Returns (gradient, smoothed_value, peak_positions, weights)
    """
    n = len(f)
    conv = fast_convolve(f)
    size = len(conv)
    
    if top_k is not None and top_k < size:
        # Only use top_k peaks for efficiency
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        # Smoothed value
        sv = temp * (pv.max() + np.log(ws))
        
        # Compute gradient: weighted sum of 2*f[kj] for each peak
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            grad[idx[mask]] += 2.0 * w[i] * f[kg[mask]]
        
        return grad, sv, peaks, w
    
    # Full softmax over all positions
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        # All weights are on the max
        peak = int(np.argmax(conv))
        sv = conv[peak]
        grad = np.zeros(n)
        js = max(0, peak - n + 1)
        je = min(n - 1, peak)
        idx = np.arange(js, je + 1)
        kg = peak - idx
        mask = (kg >= 0) & (kg < n)
        grad[idx[mask]] += 2.0 * f[kg[mask]]
        return grad, sv, np.array([peak]), np.array([1.0])
    
    w /= ws
    sv = temp * (conv.max() + np.log(ws))
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(size):
        if w[k] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        kg = k - idx
        mask = (kg >= 0) & (kg < n)
        grad[idx[mask]] += 2.0 * w[k] * f[kg[mask]]
    
    return grad, sv, np.array([np.argmax(conv)]), w

def smoothed_gradient_descent(f0, time_limit, lr0=0.01, temp0=0.001, temp_decay=0.9995,
                               momentum=0.9, top_k=50, verbose=True, lr_decay=0.9999):
    """Smoothed gradient descent with Nesterov momentum and adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # Nesterov variables
    y = f.copy()
    v = np.zeros(n)
    
    temp = temp0
    lr = lr0
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    it = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        y = f + momentum * v
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = f.copy()
        else:
            y = y / ys
        
        # Compute smoothed gradient at y
        grad, sv, peaks, w = compute_smoothed_grad(y, temp, top_k)
        
        # Gradient step
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Evaluate
        conv = fast_convolve(f_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 1000 == 0:
                print(f"      SGD iter {it}: obj={best_obj:.10f}, temp={temp:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1
            f = f_new
        
        # Adaptive parameters
        lr *= lr_decay
        temp *= temp_decay
        if temp < 1e-8:
            temp = 1e-8
        
        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            noise_scale = 0.001 * (1 + no_improve / 1000)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            lr = max(lr0 * 0.5, 0.002)
            temp = max(temp0 * 0.5, 0.0001)
            no_improve = 0
        
        # Reduce momentum as we converge
        if no_improve > 200:
            momentum = max(0.5, momentum * 0.999)
    
    return best_f, best_obj

def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe method."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        else:
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n
            g = np.maximum(g, 0)
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Fine line search
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                M = best_obj
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31337 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

            no_improve = 0
    
    return best_f, best_obj

def fast_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):

    old_n = len(f)

        return f.copy()

    # Simpler approach: linear interpolation with smoothing
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

        best_improvement = 0

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using an adaptive multi-method strategy:
1. Backtracking smoothed gradient descent with momentum (main workhorse)
2. LP-based Frank-Wolfe for exact direction finding
3. Peak-targeting coordinate descent for final polish
4. Multi-length exploration with adaptive interpolation
5. Iterative polishing cycles with perturbations
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    # Apply light smoothing
    if new_n > 4:
        kernel = np.array([1, 2, 3, 2, 1], dtype=float)
        kernel /= kernel.sum()
        f_interp = np.convolve(f_interp, kernel, mode='same')
        f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_smoothed_grad(y, temp, top_k):
    """Compute smoothed gradient of max(conv(y,y)) using softmax over top_k peaks."""
    n = len(y)
    conv = fast_convolve(y)
    size = len(conv)

    k_actual = min(top_k, size)
    if k_actual < size:
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    else:
        peaks = np.arange(size)

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        wi = w[i]
        if wi < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            kg = k - j
            if 0 <= kg < n:
                grad[j] += 2.0 * wi * y[kg]

    return grad


def backtracking_smoothed_gd(f0, time_limit, lr=0.1, temp_start=0.01, temp_end=1e-5,
                              top_k_start=100, top_k_end=5, momentum=0.7,
                              verbose=True, backtrack_factor=0.5, max_bt=20):
    """Smoothed gradient descent with backtracking line search and momentum.
    
    Minimizes g_tau(a) = tau * log(sum_k exp((a*a)[k]/tau)) over the simplex.
    The true objective max_k (a*a)[k] is monitored and the best is tracked.
    """
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    v = np.zeros(n)  # momentum velocity

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)

        # Adaptive temperature (log-linear schedule)
        temp = temp_start * (temp_end / temp_start) ** frac
        top_k = max(top_k_end, int(top_k_start * (1.0 - 0.8 * frac)))

        # Compute gradient
        grad = compute_smoothed_grad(f, temp, top_k)

        # Gradient step with momentum
        v = momentum * v - lr * grad
        direction = -grad  # Use pure gradient direction for line search

        # Backtracking line search on the smoothed objective
        step = lr
        f_curr = f
        grad_norm_sq = np.dot(grad, grad)

        # Compute current smoothed value
        conv_curr = fast_convolve(f_curr)
        peaks_curr = np.argpartition(conv_curr, -top_k)[-top_k:]
        pv_curr = conv_curr[peaks_curr]
        sc_curr = pv_curr / temp
        sc_curr -= sc_curr.max()
        g_curr = temp * (pv_curr.max() + np.log(np.exp(sc_curr).sum()))

        accepted = False
        for bt in range(max_bt):
            f_trial = f + step * v
            f_trial = project_simplex(f_trial)

            # Evaluate smoothed objective at trial point
            conv_trial = fast_convolve(f_trial)
            peaks_trial = np.argpartition(conv_trial, -top_k)[-top_k:]
            pv_trial = conv_trial[peaks_trial]
            sc_trial = pv_trial / temp
            sc_trial -= sc_trial.max()
            g_trial = temp * (pv_trial.max() + np.log(np.exp(sc_trial).sum()))

            # Armijo condition
            predicted_decrease = step * grad_norm_sq * 0.5
            if g_trial <= g_curr - 1e-4 * step * grad_norm_sq:
                accepted = True
                break
            step *= backtrack_factor

        if not accepted:
            # Fallback: small step
            f_trial = f + 0.01 * v
            f_trial = project_simplex(f_trial)
            accepted = True

        f_new = f_trial if accepted else f

        # Evaluate true objective
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"      BT-GD iter {it}: obj={best_obj:.10f}, temp={temp:.2e}, step={step:.4e}")
        else:
            no_improve += 1

        f = f_new

        # Restart if stuck
        if no_improve > 2000:
            rng = np.random.default_rng(it * 2654435761 % (2**31))
            noise_scale = 0.001 * (1 + no_improve / 5000)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            no_improve = 0

    return best_f, best_obj

def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={"presolve": True})
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def lp_fw_optimize(f0, time_limit, verbose=True, max_iters=None):
    """LP-based Frank-Wolfe: find best response direction via LP, line search to it."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if max_iters is not None and it >= max_iters:
            break
        it += 1

        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            break
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        # Line search with bisection
        best_alpha = 0.0
        best_val = M

        # Coarse search
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 5 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 31337 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj


def peak_coordinate_descent(f0, time_limit, verbose=True):

    f = np.maximum(f0.astype(np.float64), 0.0)

        it += 1
        improved = False

        # Top 3 peaks
        top_k = min(3, len(conv))

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Get contributing coordinates sorted by contribution
            contribs = []
            for j in range(js, je + 1):
                kg = peak_k - j
                if 0 <= kg < n:
                    contribs.append((f[j] * f[kg], j))
            contribs.sort(reverse=True)

            for _, j in contribs[:min(15, len(contribs))]:
                old_val = f[j]
                if old_val < 1e-12:

                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-14:

                if best_j_val != old_val:

                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                    if best_j_obj < best_obj - 1e-13:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 8:
                # Random exploration

                found = False
                for _ in range(min(50, n // 3)):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.5, 0.7, 1.4, 1.8]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.4:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-13:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)

                if out_sum > 1e-15:
                    out_weights = out_weights / out_sum * delta
                    f_trial += out_weights
                else:
                    f_trial += delta * f / max(f.sum(), 1e-15)

                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()

                new_conv = fast_convolve(f_trial)
                new_obj = new_conv.max()
                gain = best_obj - new_obj
                if gain > best_gain:
                    best_gain = gain
                    best_j = j
                    best_frac = frac
                    best_new_obj = new_obj
                    best_trial = f_trial.copy()

        if best_j >= 0 and best_gain > 1e-12:
            f = best_trial
            best_obj = best_new_obj
            best_f = f.copy()
            conv = fast_convolve(f)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Random perturbation
                rng = np.random.default_rng(it * 7919 + 42)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                best_obj = conv.max()
                no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing the peak factor 2*n*max(conv(a,a))/sum(a)^2.
    
    Phases:
    1. Momentum SGD with adaptive temperature (main workhorse)
    2. LP-based Frank-Wolfe (exact direction)
    3. Backtracking smoothed GD
    4. Multi-length exploration
    5. Peak coordinate descent (polish)
    6. Greedy peak reduction
    7. Final iterative polish cycles
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Momentum SGD (main workhorse) ===
    print(f"\n=== Phase 1: Momentum SGD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(450, rem * 0.3)
    if t1 > 60:
        print(f"  n={base_n}, t={t1:.0f}s")
        a_opt, _ = momentum_sgd(parent, time_limit=t1, lr0=0.08,
                                temp_start=0.008, temp_end=1e-5,
                                top_k_start=100, top_k_end=5,
                                momentum=0.9, verbose=True)
        try_seq(a_opt, "mom_sgd_1")

        # Second run with different params
        rem = deadline - time.monotonic()
        t1b = min(200, rem * 0.15)
        if t1b > 30 and best_seq is not None:
            a_opt, _ = momentum_sgd(best_seq, time_limit=t1b, lr0=0.03,
                                    temp_start=0.003, temp_end=1e-6,
                                    top_k_start=50, top_k_end=3,
                                    momentum=0.95, verbose=False)
            try_seq(a_opt, "mom_sgd_2")

    # === Phase 2: LP-based Frank-Wolfe ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(350, rem * 0.25)
        if t2 > 40:
            print(f"  n={len(best_seq)}, t={t2:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True,
                                      max_iters=200)
            try_seq(a_opt, "lpfw_1")

            # Second LP-FW run
            rem = deadline - time.monotonic()
            t2b = min(150, rem * 0.1)
            if t2b > 20 and best_seq is not None:
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2b, verbose=False,
                                          max_iters=100)
                try_seq(a_opt, "lpfw_2")

    # === Phase 3: Backtracking Smoothed GD ===
    print(f"\n=== Phase 3: Backtracking GD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(200, rem * 0.15)
        if t3 > 30:
            print(f"  n={len(best_seq)}, t={t3:.0f}s")
            a_opt, _ = backtracking_smoothed_gd(best_seq, time_limit=t3, lr=0.3,
                                                temp_start=0.005, temp_end=1e-5,
                                                top_k_start=60, top_k_end=5,
                                                momentum=0.5, verbose=True)
            try_seq(a_opt, "bt_gd")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        candidate_lengths = []
        for delta in [-128, -64, 32, 64, 128, 256, 512]:
            ln = len(base) + delta
            if 256 <= ln <= 4096:
                candidate_lengths.append(ln)
        # Add some specific lengths
        for ln in [512, 768, 1024, 1280, 1536, 2048]:
            if 256 <= ln <= 4096 and ln not in candidate_lengths:
                candidate_lengths.append(ln)

        candidate_lengths = sorted(set(candidate_lengths))

        # Sort by distance from current best
        current_n = len(base)
        candidate_lengths.sort(key=lambda x: abs(x - current_n))
        candidate_lengths = candidate_lengths[:6]

        for target_n in candidate_lengths:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            t_alloc = min(rem * 0.2, 180)
            if t_alloc < 30:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  MomSGD n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = momentum_sgd(f_interp, time_limit=t_alloc, lr0=0.05,
                                    temp_start=0.005, temp_end=1e-5,
                                    top_k_start=80, top_k_end=5,
                                    momentum=0.9, verbose=False)
            try_seq(a_opt, f"ml_sgd_n{target_n}")

            # Quick LP-FW on this length
            rem = deadline - time.monotonic()
            t_alloc2 = min(rem * 0.1, 60)
            if t_alloc2 > 15 and best_seq is not None and len(best_seq) == target_n:
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc2, verbose=False,
                                          max_iters=50)
                try_seq(a_opt, f"ml_lpfw_n{target_n}")

    # === Phase 5: Peak coordinate descent ===
    print(f"\n=== Phase 5: Peak CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t5 = min(200, rem * 0.2)
        if t5 > 20:
            print(f"  n={len(best_seq)}, t={t5:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "peak_cd")

    # === Phase 6: Greedy peak reduction ===
    print(f"\n=== Phase 6: Greedy Peak Reduction (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t6 = min(100, rem * 0.1)
        if t6 > 15:
            print(f"  n={len(best_seq)}, t={t6:.0f}s")
            a_opt, _ = greedy_peak_reduction(best_seq, time_limit=t6, verbose=False)
            try_seq(a_opt, "greedy_pr")

    # === Phase 7: Final iterative polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        for cycle in range(4):
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break

            # Short SG
            t_sub = min(rem * 0.25, 60)
            if t_sub > 10:
                a_opt, _ = momentum_sgd(best_seq, time_limit=t_sub, lr0=0.02,
                                        temp_start=0.002, temp_end=1e-6,
                                        top_k_start=40, top_k_end=3,
                                        momentum=0.95, verbose=False)
                try_seq(a_opt, f"polish_sgd_{cycle}")

            # Short LP-FW
            rem = deadline - time.monotonic() - 20
            t_sub = min(rem * 0.25, 50)
            if t_sub > 10 and best_seq is not None:
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_sub, verbose=False,
                                          max_iters=50)
                try_seq(a_opt, f"polish_lpfw_{cycle}")

            # Short CD
            rem = deadline - time.monotonic() - 15
            t_sub = min(rem * 0.25, 40)
            if t_sub > 10 and best_seq is not None:
                a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_sub,
                                                   verbose=False)
                try_seq(a_opt, f"polish_cd_{cycle}")

        # Perturbation-based exploration
        for p in range(10):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            noise_scale = 0.0003 + 0.0004 * p
            noise = rng.standard_normal(len(best_seq)) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_sub = min(rem, 15)
            a_opt, _ = momentum_sgd(f_pert, time_limit=t_sub, lr0=0.015,
                                    temp_start=0.001, temp_end=1e-6,
                                    top_k_start=30, top_k_end=3,
                                    momentum=0.95, verbose=False)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
