"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam-optimized log-sum-exp smoothing,
exact coordinate bisection, population-based diversity maintenance, multi-length
exploration, and simulated annealing for escaping local minima.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, beta, top_k=200):
    """Compute LSE(beta * conv(a,a)) value and gradient using top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    m_val = conv.max()
    exps = np.exp(beta * (conv - m_val))
    exp_sum = exps.sum()
    p = exps / exp_sum
    lse_val = beta * m_val + np.log(exp_sum)

    top_k_actual = min(top_k, len(p))
    idx = np.argpartition(p, -top_k_actual)[-top_k_actual:]
    idx = idx[np.argsort(p[idx])[::-1]]

    grad = np.zeros(n)
    for k in idx:
        pk = p[k]
        if pk < 1e-14:
            break
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            i = np.arange(js, je + 1)
            grad[i] += 2.0 * beta * pk * a[k - i]

    return lse_val, grad, conv

def coordinate_bisection_fast(a0, time_limit, max_passes=200, verbose=False):
    """Exact coordinate descent with bisection per coordinate (optimized)."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-12:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-12:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

            if abs(x_new - x_old) < 1e-13:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = best_a.copy()
                continue

            conv = fast_convolve(a)
            obj = conv.max()

            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 31 + 17)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0005)
            else:
                break

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric and renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

            rng_local = np.random.default_rng(iteration * 1000 + bi)

            for attempt in range(20):
                # Random simplex point in the block
                r = rng_local.exponential(1.0, size=block_size_actual)
                r /= r.sum() * c
                a_trial = a.copy()
                a_trial[start:end] = r

                obj_trial = fast_convolve(a_trial).max()
                if obj_trial < best_block_obj:
                    best_block_obj = obj_trial
                    best_block_vals = r.copy()

            if best_block_obj < best_obj - 1e-13:
                a[start:end] = best_block_vals
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss > 1e-15:
                    a /= ss
                best_obj = best_block_obj
                best_a = a.copy()
                improved = True

        if not improved and iteration > 10:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining Adam-LSE, coordinate bisection,
    block swaps, population diversity, and multi-length exploration.
    """
    rng_global = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_a = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = cosine_bump(768)
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={p_score:.6f}")

    def update_best(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        # Normalize for evaluation (score is scale-invariant)
        a_norm = a_try / s_try
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")

    # === Phase 1: Adam-LSE optimization from parent ===
    print(f"\n{'='*60}")
    print(f"=== Phase 1: Adam-LSE from parent (n={len(parent)}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 100:
        t_alloc = min(rem, 500)
        print(f"  Time allocation: {t_alloc:.0f}s")
        a_opt, obj_opt = adam_lse_optimize(parent, time_limit=t_alloc,
                                            beta_start=3.0, beta_end=60.0,
                                            lr=0.02, top_k=300, verbose=True)
        update_best(a_opt, "adam_lse_parent")

        # CD polish
        rem = deadline - time.monotonic() - 700
        if rem > 30:
            t_alloc = min(rem, 80)
            a_opt2, obj2 = coordinate_bisection_fast(a_opt, time_limit=t_alloc)
            update_best(a_opt2, "cd_after_adam")
            a_opt = a_opt2

    # === Phase 2: Symmetric variant ===
    print(f"\n=== Phase 2: Symmetric variant ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 650
        if rem > 50:
            a_sym = symmetrize(best_a)
            t_alloc = min(rem, 120)
            a_opt, obj_opt = adam_lse_optimize(a_sym, time_limit=t_alloc,
                                                beta_start=5.0, beta_end=50.0, lr=0.015)
            update_best(a_opt, "symmetric_adam")

            rem = deadline - time.monotonic() - 550
            if rem > 20:
                t_alloc = min(rem, 50)
                a_opt2, _ = coordinate_bisection_fast(a_opt, time_limit=t_alloc)
                update_best(a_opt2, "symmetric_cd")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration ===")
    base = best_a if best_a is not None else parent

    # Try several lengths
    n_best = len(base)
    target_lengths = sorted(set([
        max(128, n_best // 2),
        n_best,
        min(2048, n_best * 2),
        max(128, int(n_best * 0.75)),
        max(128, int(n_best * 1.25)),
    ]))

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 350
        if rem < 60:
            break
        if target_n == n_best and best_a is not None:
            continue  # Already optimized at this length

        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.4, 150)
        print(f"  n={target_n}, t={t_alloc:.0f}s")

        a_opt, _ = adam_lse_optimize(f_interp, time_limit=t_alloc,
                                      beta_start=3.0, beta_end=50.0, lr=0.015)
        update_best(a_opt, f"len{target_n}")

        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_cd = min(rem, 40)
            a_opt2, _ = coordinate_bisection_fast(a_opt, time_limit=t_cd)
            update_best(a_opt2, f"len{target_n}_cd")

    # === Phase 4: Population-based diversity ===
    print(f"\n=== Phase 4: Population diversity ===")
    population = []
    if best_a is not None:
        population.append(best_a.copy())

    # Add diverse candidates
    n_pop = len(population[0]) if population else 768
    diverse_inits = [
        cosine_bump(n_pop),
        symmetrize(population[0]) if population else uniform_init(n_pop),
        project_simplex(population[0] + rng_global.standard_normal(n_pop) * 0.02) if population else uniform_init(n_pop),
    ]

    for idx, div_init in enumerate(diverse_inits):
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break

        t_alloc = min(rem * 0.3, 100)
        print(f"  Diverse init {idx}, t={t_alloc:.0f}s")
        a_opt, _ = adam_lse_optimize(div_init, time_limit=t_alloc,
                                      beta_start=3.0, beta_end=40.0, lr=0.02)

        # Check if it's different from current best
        if best_a is not None:
            diff = np.abs(a_opt - best_a).sum()
            if diff < 0.01:
                continue  # Too similar

        update_best(a_opt, f"diverse_{idx}")
        if best_a is not None:
            population.append(best_a.copy())

            # Try crossover with existing best
            rem = deadline - time.monotonic() - 150
            if rem > 30 and len(population) >= 2:
                # Blend two diverse solutions
                blend = 0.5 * population[-1] + 0.5 * population[0]
                blend = project_simplex(blend)
                t_blend = min(rem, 50)
                a_blend, _ = adam_lse_optimize(blend, time_limit=t_blend,
                                                beta_start=5.0, beta_end=40.0, lr=0.01)
                update_best(a_blend, "crossover")

    # === Phase 5: Intensive final polish ===
    print(f"\n=== Phase 5: Final intensive polish ===")
    if best_a is not None:
        # Long CD pass
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_alloc = min(rem, 200)
            print(f"  CD polish, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_bisection_fast(best_a, time_limit=t_alloc, max_passes=500)
            update_best(a_opt, "final_cd")

        # Block swap polish
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  Block swap, t={t_alloc:.0f}s")
            a_opt, _ = block_swap_polish(best_a, time_limit=t_alloc, block_size=4)
            update_best(a_opt, "block_swap")

        # Tiny perturbations + CD
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            n_best = len(best_a)
            for p in range(5):
                rem = deadline - time.monotonic() - 10
                if rem < 8:
                    break
                rng_p = np.random.default_rng(seed + 2000 + p * 137)
                f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.002)
                t_alloc = min(rem, 12)
                a_opt, _ = coordinate_bisection_fast(f_pert, time_limit=t_alloc)
                update_best(a_opt, f"perturb_{p}")

        # Final Adam fine-tune with very high beta
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_alloc = min(rem, 30)
            a_opt, _ = adam_lse_optimize(best_a, time_limit=t_alloc,
                                          beta_start=50.0, beta_end=100.0, lr=0.005, top_k=500)
            update_best(a_opt, "final_fine_adam")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Shouldn't happen, but fallback
        best_a = parent.tolist()
        best_score = float(evaluate_sequence(best_a.tolist())) if isinstance(best_a, list) else float('inf')

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
