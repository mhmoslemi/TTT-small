"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact subgradient with Nesterov acceleration (focusing on true max)
2. Multi-scale peak reduction (coarse-to-fine)
3. Coordinate-wise exact local search
4. Multi-length exploration with interpolation
5. Smart random perturbation with acceptance criteria

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_max(a, top_k=5):
    """Compute subgradient of max(a*a) at the top-k peaks with weighting."""
    n = len(a)
    conv = fast_convolve(a)
    # Get top k peaks
    k = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k)[-k:]
    
    # Weight by proximity to max
    peak_vals = conv[peaks]
    max_val = peak_vals.max()
    # Exponential weighting: peaks close to max get more weight
    weights = np.exp(-(max_val - peak_vals) / (0.01 * max_val + 1e-15))
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        w = weights[i]
        if w < 1e-10:
            continue
        ki = int(peaks[i])
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w * a[ki - idx]
    
    return sg, conv.max()


def nesterov_optimize(f0, time_limit, verbose=False):
    """Nesterov-accelerated gradient method on exact max subgradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    y = f.copy()  # Nesterov momentum point
    beta = 0.9  # momentum
    
    it = 0
    no_improve = 0
    lr = 0.02  # learning rate
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov step: y = f + beta * (f - f_prev), but we track it differently
        # Actually, let's use the standard Nesterov:
        # y_{t+1} = (1 - alpha_t) * f_t + alpha_t * f_{t+1} where alpha_t = 2/(t+2)
        # Simpler: just use momentum SGD which is often better in practice
        
        sg, M = subgradient_at_max(f, top_k=5)
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            no_improve += 1
        else:
            sg = sg / sg_norm
        
        # Update with momentum
        if not hasattr(nesterov_optimize, '_velocity'):
            velocity = np.zeros(n)
        velocity = beta * velocity - lr * sg
        f_new = f + velocity
        
        # Project to simplex
        f_new = project_simplex(f_new)
        
        # Check improvement
        new_M = fast_convolve(f_new).max()
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 100 == 0:
                print(f"    Nesterov iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:
            f = f_new  # Still move (momentum helps escape)
            no_improve += 1
        
        # Decay learning rate
        if no_improve > 50:
            lr *= 0.95
            if no_improve > 200:
                # Restart
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                velocity = np.zeros(n)
                lr = 0.02
                no_improve = 0
                best_obj = fast_convolve(f).max()
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate-wise optimization: for each coordinate, find best value."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve_passes > 3:
            break
        
        pass_improved = False
        # Random order for diversity
        order = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = order[idx]
            if f[i] < 1e-12:
                continue
            
            # Try moving mass from i to other positions
            # The effect on convolution: changing a[i] to a[i]+delta affects
            # (a*a)[k] for k in range involving i
            # Specifically, (a*a)[i+j] gets contribution a[i]*a[j] + a[j]*a[i] = 2*a[i]*a[j]
            # and (a*a)[i-i+i] = a[i]^2 at lag 0 (well, not exactly)
            
            # Tauberian approach: try setting a[i] to 0 and redistributing
            # Or try small perturbations
            
            # Try: reduce a[i] by epsilon, add to random position j
            best_delta = 0.0
            best_j = -1
            best_val = best_obj
            
            # Quick try: a few candidate targets
            candidates = [i // 2, (i + 1) % n, (i + n // 2) % n, rng.integers(0, n)]
            if i > 0:
                candidates.append(i - 1)
            if i < n - 1:
                candidates.append(i + 1)
            
            for j in candidates:
                if j == i:
                    continue
                # Try transferring a fraction of f[i] to f[j]
                for frac in [0.1, 0.3, 0.5, 0.7]:
                    f_trial = f.copy()
                    delta = frac * f[i]
                    f_trial[i] -= delta
                    f_trial[j] += delta
                    if f_trial[i] < 0:
                        continue
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_val - 1e-13:
                        best_val = val
                        best_delta = delta
                        best_j = j
                        best_frac = frac
            
            if best_val < best_obj - 1e-12:
                f[i] -= best_delta
                f[best_j] += best_delta
                f = f / f.sum()
                best_obj = best_val
                best_f = f.copy()
                pass_improved = True
        
        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
    
    return best_f, best_obj


def peak_balancing(f0, time_limit, verbose=False):
    """
    Iteratively balance the convolution peaks.
    At each step, identify the highest peak and try to reduce it
    by adjusting coordinates that contribute to it.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 100:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        target = M * 0.995  # Try to get peaks down by 0.5%
        
        # Find the peak position
        peak_pos = int(np.argmax(conv))
        
        # Coordinates contributing to this peak:
        # (a*a)[peak_pos] = sum_j a[j] * a[j + peak_pos]
        js = max(0, peak_pos - n + 1)
        je = min(n - 1, peak_pos)
        contributing = np.arange(js, je + 1)
        
        # Find the largest contributor
        contributions = f[contributing] * f[peak_pos - contributing]
        if len(contributing) == 0:
            break
        max_contrib_idx = contributing[np.argmax(contributions)]
        
        # Try to reduce this coordinate
        f_trial = f.copy()
        # Move mass from max_contrib_idx to a random other position
        target_pos = rng.integers(0, n)
        if target_pos == max_contrib_idx:
            target_pos = (target_pos + 1) % n
        
        # How much to move? Try a few amounts
        improved = False
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
            delta = frac * f[max_contrib_idx]
            f_trial2 = f.copy()
            f_trial2[max_contrib_idx] -= delta
            f_trial2[target_pos] += delta
            f_trial2 = np.maximum(f_trial2, 0)
            st = f_trial2.sum()
            if st < 1e-15:
                continue
            f_trial2 = f_trial2 / st
            val = fast_convolve(f_trial2).max()
            if val < best_obj - 1e-13:
                f = f_trial2
                best_obj = val
                best_f = f.copy()
                no_improve = 0
                improved = True
                break
        
        if not improved:
            no_improve += 1
            # Try a different strategy: random perturbation
            if no_improve % 10 == 0:
                f_trial = f + rng.standard_normal(n) * 0.002
                f_trial = project_simplex(f_trial)
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-13:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def adam_smoothed(f0, time_limit, verbose=False):
    """Adam optimizer on log-sum-exp smoothed objective with adaptive top_k."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.99
    eps = 1e-8
    
    it = 0
    no_improve = 0
    
    # Schedule: start with broad smoothing, narrow to peak
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        
        # Temperature: from broad to sharp
        T = 0.5 / n * (1.0 - 0.7 * progress)
        top_k = max(3, int(n * (0.5 - 0.4 * progress)))
        lr = 0.003 * (1.0 - 0.3 * progress)
        
        conv = fast_convolve(f)
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Adam update
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        new_M = fast_convolve(f_new).max()
        
        if new_M < best_obj - 1e-14:
            best_obj = new_M
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}")
        else:
            f = f_new
            no_improve += 1
        
        if no_improve > 500:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
    
    return best_f, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_max(a, top_k=5):
    """Compute weighted subgradient at top-k peaks of a*a."""
    n = len(a)
    conv = fast_convolve(a)
    k = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k)[-k:]
    peak_vals = conv[peaks]
    max_val = peak_vals.max()
    weights = np.exp(-(max_val - peak_vals) / (0.01 * max_val + 1e-15))
    weights /= weights.sum()
    sg = np.zeros(n)
    for i in range(len(peaks)):
        w = weights[i]
        if w < 1e-10:
            continue
        ki = int(peaks[i])
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w * a[ki - idx]
    return sg, conv.max()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            M_new = fast_convolve(f_new).max()
            f = f_new
            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if verbose and it % 50 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining peak balancing, Adam smoothed,
    Nesterov, LP-FW, local refinement, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    # Phase 1: Peak balancing on parent
    print(f"\n=== Phase 1: Peak Balancing (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 700
    t1 = min(rem, 280)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s, n={len(best_seq)}")
        a_opt, obj = peak_balancing(best_seq, time_limit=t1, verbose=True)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "peak_bal_1")

    # Phase 2: Adam smoothed
    print(f"\n=== Phase 2: Adam Smoothed (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 550
    t2 = min(rem, 220)
    if t2 > 30:
        print(f"  Time: {t2:.0f}s, n={len(best_seq)}")
        a_opt, obj = adam_smoothed(best_seq, time_limit=t2, verbose=True)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "adam_2")

    # Phase 3: LP-FW
    print(f"\n=== Phase 3: LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    t3 = min(rem, 180)
    if t3 > 30:
        print(f"  Time: {t3:.0f}s, n={len(best_seq)}")
        a_opt, obj = lp_fw_step(best_seq, time_limit=t3, verbose=True)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "lpfw_3")

    # Phase 4: Nesterov
    print(f"\n=== Phase 4: Nesterov (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 300
    t4 = min(rem, 150)
    if t4 > 20:
        print(f"  Time: {t4:.0f}s, n={len(best_seq)}")
        a_opt, obj = nesterov_optimize(best_seq, time_limit=t4, verbose=True)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "nesterov_4")

    # Phase 5: Local refinement
    print(f"\n=== Phase 5: Local Refine (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 250
    t5 = min(rem, 200)
    if t5 > 20:
        print(f"  Time: {t5:.0f}s, n={len(best_seq)}")
        a_opt, obj = local_refine(best_seq, time_limit=t5, verbose=True)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "local_5")

    # Phase 6: Multi-length exploration
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.8f}) ===")
    for target_n in [512, 1024]:
        if target_n == len(best_seq):
            continue
        rem = deadline - time.monotonic() - 150
        if rem < 50:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(rem * 0.35, 80)
        if t_alloc > 20:
            print(f"  Adam n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_smoothed(f_interp, time_limit=t_alloc, verbose=False)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"adam_n{target_n}")
        rem = deadline - time.monotonic() - 130
        if rem > 30:
            t_pb = min(rem * 0.4, 60)
            if t_pb > 15:
                print(f"  PeakBal n={target_n}, t={t_pb:.0f}s")
                a_opt, _ = peak_balancing(a_opt, time_limit=t_pb, verbose=False)
                best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"pb_n{target_n}")

    # Phase 7: Diverse restarts
    print(f"\n=== Phase 7: Diverse Restarts (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    for p in range(4):
        rem = deadline - time.monotonic() - 100
        if rem < 25:
            break
        t_alloc = min(rem * 0.3, 60)
        if p == 0:
            f_init = best_seq.copy()
        elif p == 1:
            rng = np.random.default_rng(700 + p)
            f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
        elif p == 2:
            f_init = best_seq[::-1].copy()
            f_init = f_init / f_init.sum()
        else:
            rng = np.random.default_rng(900 + p)
            f_init = project_simplex(best_seq * (1.0 + rng.random(n_best) * 0.15))
        if p % 2 == 0:
            a_opt, _ = adam_smoothed(f_init, time_limit=t_alloc, verbose=(p == 0))
        else:
            a_opt, _ = peak_balancing(f_init, time_limit=t_alloc, verbose=(p == 0))
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"diverse_{p}")
        # Quick local refine after each
        rem = deadline - time.monotonic() - 80
        if rem > 15 and best_seq is not None:
            a_opt2, _ = local_refine(best_seq, time_limit=min(rem, 30))
            best_seq, best_score = try_and_update(a_opt2, best_seq, best_score, f"diverse_{p}_lr")

    # Phase 8: Final intensive polish
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        # Adam with very focused top_k
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = adam_smoothed(best_seq, time_limit=min(rem, 50), verbose=True)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "adam_final")
        # Local refine
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = local_refine(best_seq, time_limit=min(rem, 40))
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "local_final")
        # Peak balance
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = peak_balancing(best_seq, time_limit=min(rem, 30))
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "pb_final")
        # One more Nesterov
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = nesterov_optimize(best_seq, time_limit=min(rem, 25))
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "nest_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
