Given the current sequence a and convolution c = a*a:
1. Find the index k* of the maximum: c[k*] = max(c).
2. The "gradient" of c[k*] with respect to a[i] is: 2*a[k*-i] (for valid i).
3. To reduce c[k*], we want to decrease a[i] for i where a[k*-i] is large.
4. But we also need to maintain sum(a) = 1, so we must increase some other a[j].
5. The best j to increase is one where it causes the least increase in other peaks.

This is essentially a "swap" operation: move weight from position i (which contributes to the peak) to position j (which doesn't contribute much to any peak).

This is similar to the "sinkhorn" or "optimal transport" idea.

3. Move a small amount of weight from i to j.
4. Repeat.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np

import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_conv(f, M, timeout=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix A_ub where (A_ub @ g)[k] = (f*g)[k]
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def smoothed_max_descent(a0, time_limit, T_init=0.001):
    """
    Gradient descent using soft-max smoothing of the objective.
    Uses log-sum-exp as a smooth approximation to the max.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = 0.01
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        
        # Soft-max gradient
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient of sum_k p_k * conv_k with respect to a
        # = sum_k p_k * d(conv_k)/da
        # d(conv_k)/da[i] = a[k-i] (for valid k-i)
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            if p[k_idx] < 1e-12:
                continue
            k = int(k_idx)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += p[k_idx] * a[k - j]
        
        # Projected gradient descent
        candidate = project_simplex(a - lr * grad)
        c = fast_convolve(candidate)
        val = c.max()
        
        if val < best_obj - 1e-15:
            a = candidate
            best_obj = val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt

            T *= 1.03
            lr *= 0.98

            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = 0.01
            no_improve = 0
    
    return best_a, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_conv(f, M, timeout=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def peak_swap_descent(a0, time_limit):

    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    it = 0

    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.2:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        # Compute total impact for each position using top peaks
        top_k = min(15, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]

        total_impact = np.zeros(n)
        for k in peak_indices:
            kk = int(k)
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            total_impact[idx] += a[kk - idx]

        # Rank positions by impact
        src_rank = np.argsort(total_impact)[::-1]
        snk_rank = np.argsort(total_impact)

        # Try swaps: source (high impact) -> sink (low impact)
        improved = False
        n_src_try = min(8, n // 4)
        n_snk_try = min(8, n // 4)

        for src in src_rank[:n_src_try]:

                continue
            for snk in snk_rank[:n_snk_try]:

                    continue

                for frac in [0.02, 0.01, 0.005, 0.002]:

                        best_obj = c.max()

                    break
            if improved:

                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                stall_count = 0

    return best_a, best_obj
