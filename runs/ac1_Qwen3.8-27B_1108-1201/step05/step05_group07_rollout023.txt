def fast_convolve(a):

def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1} via bisection on threshold."""
    a = np.maximum(a, 0.0)

    if s < 1e-30:

def interpolate_seq(f, new_n):
    """Linear interpolation of sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

    return f_interp

        # Adam update
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Maintain constraints
        if use_symmetry:
            f_new = make_symmetric(f_new)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()

        # Evaluate
        _, _, conv_new = smooth_max_and_grad(f_new, beta)
        obj = conv_new.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # If stuck, try a perturbation
        if no_improve > 500:
            rng = np.random.default_rng(it * 31 + 7)
            noise_scale = 0.001 + 0.002 * min(no_improve // 1000, 3)
            f = f + rng.standard_normal(n) * noise_scale
            if use_symmetry:
                f = make_symmetric(f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            # Reset beta to be a bit larger to escape
            beta = max(beta * 5, beta_init * 0.5)

        f = f_new

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, beta={beta:.6f}, lr={lr:.6f}")

    return best_f, best_obj


def peak_targeted_search(f0, time_limit, top_k=10, verbose=False):
    """
    Targeted local search: identify the top peaks in auto-convolution and
    adjust contributing elements to reduce them.
    """
    f = np.maximum(f0.astype(np.float64).copy(), 0.0)
    s = f.sum()
    if s < 1e-30:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        cur_max = conv.max()

        # Get top peaks
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]

        improved = False
        for k in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            # Find contributing indices
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            # Contribution: a[j] * a[k-j]
            contribs = f[idx] * f[k - idx]
            total = contribs.sum()
            if total < 1e-15:
                continue

            # Try reducing the largest contributor
            max_pos = np.argmax(contribs)
            hi = idx[max_pos]
            hi_val = f[hi]
            if hi_val < 1e-10:
                continue

            # Try moving mass from hi to the smallest contributor at same peak
            min_pos = np.argmin(contribs)
            lo = idx[min_pos]

            if hi == lo:
                continue

            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                amt = hi_val * frac
                f[hi] = hi_val - amt
                f[lo] += amt
                new_conv = fast_convolve(f)
                new_max = new_conv.max()
                if new_max < best_obj - 1e-12:
                    best_obj = new_max
                    best_f = f.copy()
                    improved = True
                    break
                else:
                    f[hi] = hi_val
                    f[lo] -= amt

        if not improved:
            # Try random perturbations around peaks
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            any_improve = False
            for trial in range(100):
                if time.monotonic() - t0 > time_limit:
                    break
                k = int(peaks[rng.integers(0, len(peaks))])
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                if js > je:
                    continue
                idx = np.arange(js, je + 1)
                i = idx[rng.integers(0, len(idx))]
                j = idx[rng.integers(0, len(idx))]
                if i == j or f[i] < 1e-10:
                    continue
                amt = f[i] * rng.uniform(0.02, 0.15)
                f[i] -= amt
                f[j] += amt
                new_max = fast_convolve(f).max()
                if new_max < best_obj - 1e-12:
                    best_obj = new_max
                    best_f = f.copy()
                    any_improve = True
                else:
                    f[i] += amt
                    f[j] -= amt
            if not any_improve:
                break

    return best_f, best_obj


def coordinate_descent_fine(f0, time_limit, verbose=False):
    """Fine coordinate descent: for each coordinate, find optimal value."""
    f = np.maximum(f0.astype(np.float64).copy(), 0.0)
    s = f.sum()
    if s < 1e-30:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Work with pairs for symmetry
    pairs = []
    for j in range(n // 2):
        pairs.append((j, n - 1 - j))
    if n % 2 == 1:
        pairs.append((n // 2, n // 2))

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j, j2 in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Try several fractions of current value
            for mult in [0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                # Scale others to maintain sum=1
                rest_sum = 1.0 - old_val
                if rest_sum < 1e-10:
                    continue
                scale = (1.0 - new_val) / rest_sum
                f[j] = new_val
                f[j2] = new_val if j != j2 else new_val
                if j != j2:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    mask[j2] = False
                    f[mask] *= scale
                else:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                f = np.maximum(f, 0)
                f = f / f.sum()

                val = fast_convolve(f).max()
                if val < best_j_obj - 1e-13:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                rest_sum = 1.0 - old_val
                if rest_sum > 1e-10:
                    scale = (1.0 - best_j_val) / rest_sum
                    f[j] = best_j_val
                    f[j2] = best_j_val if j != j2 else best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    if j2 != j:
                        mask[j2] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining smoothed gradient descent with Adam,
    peak-targeted local search, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        parent_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default uniform n=768, score={parent_score:.6f}")
        best_seq = parent.copy()
        best_score = parent_score

    n_parent = len(parent)

    # === Phase 1: Smoothed gradient with Adam on parent ===
    print(f"\n=== Phase 1: Adam Smooth Gradient (n={n_parent}) ===")

    configs_p1 = [
        {"beta_init": 0.005, "beta_final": 0.0001, "lr_init": 0.02, "lr_final": 0.00005, "sym": False},
        {"beta_init": 0.002, "beta_final": 0.00005, "lr_init": 0.01, "lr_final": 0.0001, "sym": True},
        {"beta_init": 0.01, "beta_final": 0.0002, "lr_init": 0.05, "lr_final": 0.0002, "sym": False},
        {"beta_init": 0.001, "beta_final": 0.00002, "lr_init": 0.005, "lr_final": 0.00005, "sym": True},
    ]

    rem = deadline - time.monotonic()
    t_per = rem * 0.25 / len(configs_p1)
    for i, cfg in enumerate(configs_p1):
        rem = deadline - time.monotonic() - 30
        if rem < 10:
            break
        t_alloc = min(t_per, rem * 0.3)
        print(f"  P1 cfg{i}: t={t_alloc:.0f}s, beta={cfg['beta_init']}->{cfg['beta_final']}, sym={cfg['sym']}")
        a_opt, _ = adam_smooth_optimize(parent, time_limit=t_alloc,
                                          beta_init=cfg["beta_init"],
                                          beta_final=cfg["beta_final"],
                                          lr_init=cfg["lr_init"],
                                          lr_final=cfg["lr_final"],
                                          use_symmetry=cfg["sym"],
                                          label=f"p1_{i}")
        try_seq(a_opt, f"adam_{i}")

    # === Phase 2: Peak-targeted search on best ===
    print(f"\n=== Phase 2: Peak-Targeted Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.5, 120)
            print(f"  Peak search t={t_alloc:.0f}s, n={len(best_seq)}")
            a_opt, _ = peak_targeted_search(best_seq, time_limit=t_alloc, top_k=15)
            try_seq(a_opt, "peak_search")

            rem = deadline - time.monotonic() - 150
            if rem > 15:
                t_alloc = min(rem * 0.4, 80)
                print(f"  Peak search (二轮) t={t_alloc:.0f}s")
                a_opt, _ = peak_targeted_search(best_seq, time_limit=t_alloc, top_k=5)
                try_seq(a_opt, "peak_search2")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_alloc = min(rem * 0.5, 100)
            print(f"  Coord descent t={t_alloc:.0f}s, n={len(best_seq)}")
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = []
        n_best = len(best_seq)
        for dl in [-128, -64, 64, 128, 256, 512]:
            nl = n_best + dl
            if 256 <= nl <= 2048:
                target_lengths.append(nl)
        # Also add some standard lengths
        for tl in [512, 640, 896, 1024, 1280]:
            if tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths = sorted(set(target_lengths))[:5]

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            f_interp = interpolate_sequence_safe(best_seq, target_n)
            t_alloc = min(rem * 0.3, 90)
            print(f"  n={target_n} (from {len(best_seq)}), t={t_alloc:.0f}s")

            # Adam optimize on interpolated
            a_opt, _ = adam_smooth_optimize(f_interp, time_limit=t_alloc * 0.6,
                                             beta_init=0.003, beta_final=0.00005,
                                             lr_init=0.01, lr_final=0.0001,
                                             use_symmetry=False,
                                             label=f"len_{target_n}")
            try_seq(a_opt, f"len_{target_n}")

            # Refine
            rem = deadline - time.monotonic() - 80
            if rem > 10 and best_seq is not None and len(best_seq) == target_n:
                a_opt2, _ = peak_targeted_search(a_opt, time_limit=min(rem, 30), top_k=8)
                try_seq(a_opt2, f"len_{target_n}_ref")

    # === Phase 5: Fine-tuning on best ===
    print(f"\n=== Phase 5: Fine Tuning (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 60

        # Very fine Adam
        if rem > 20:
            t_alloc = min(rem * 0.3, 60)
            print(f"  Fine Adam t={t_alloc:.0f}s")
            a_opt, _ = adam_smooth_optimize(best_seq, time_limit=t_alloc,
                                             beta_init=0.0005, beta_final=0.000005,
                                             lr_init=0.002, lr_final=0.00002,
                                             use_symmetry=True, label="fine")
            try_seq(a_opt, "fine_adam")

        # Peak search on refined
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            a_opt, _ = peak_targeted_search(best_seq, time_limit=min(rem, 30), top_k=5)
            try_seq(a_opt, "fine_peak")

        # Coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "fine_cd")

        # Small perturbations + quick optimize
        rem = deadline - time.monotonic() - 20
        for pert in [0.0005, 0.001, 0.002]:
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * pert
            f_pert = best_seq + noise
            f_pert = np.maximum(f_pert, 0)
            f_pert = f_pert / f_pert.sum()
            a_opt, _ = adam_smooth_optimize(f_pert, time_limit=min(rem, 15),
                                             beta_init=0.001, beta_final=0.00002,
                                             lr_init=0.003, lr_final=0.00005,
                                             use_symmetry=False, label=f"pert_{pert}")
            try_seq(a_opt, f"pert_{pert}")
            rem = deadline - time.monotonic() - 20

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = peak_targeted_search(best_seq, time_limit=min(rem, 20), top_k=3)
            try_seq(a_opt, "final_peak")

        rem = deadline - time.monotonic() - 2
        if rem > 3:
            a_opt, _ = coordinate_descent_fine(best_seq, time_limit=min(rem, 10))
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def interpolate_sequence_safe(f, new_n):
    """Safe interpolation that handles edge cases."""
    f = np.asarray(f, dtype=np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    f = f / s
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp
