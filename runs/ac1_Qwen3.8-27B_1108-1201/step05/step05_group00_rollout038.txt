import numpy as np
from scipy.optimize import minimize
from scipy.linalg import solve_banded
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def form_Q(active_peaks, weights, n):
    """Form the weighted sum of Hankel matrices."""
    Q = np.zeros((n, n))
    for k, w in zip(active_peaks, weights):
        k = int(k)
        if k == 0:
            Q[np.diag_indices(n)] += w
        else:
            for i in range(n - k):
                Q[i, i+k] += w * 0.5
                Q[i+k, i] += w * 0.5
    return Q

def solve_qp(Q, n):
    """Solve min f^T Q f s.t. sum(f)=1, f>=0 using active set."""
    # Start with all variables active
    active = set(range(n))
    for _ in range(100):
        S = sorted(active)
        k = len(S)
        if k == 0:
            return np.ones(n) / n
        if k == 1:
            f = np.zeros(n)
            f[S[0]] = 1.0
            return f
        
        Q_SS = Q[np.ix_(S, S)]
        
        # Solve Q_SS * f_S = -lambda/2 * ones
        # Using the formula: f_S = ones / (ones^T Q^{-1} ones) * Q^{-1} ones
        try:
            Q_inv_ones = np.linalg.solve(Q_SS, np.ones(k))
        except np.linalg.LinAlgError:
            Q_inv_ones, _, _, _ = np.linalg.lstsq(Q_SS, np.ones(k), rcond=None)
        
        denom = np.sum(Q_inv_ones)
        if denom <= 1e-15:
            # Degenerate case
            f = np.zeros(n)
            f[S] = 1.0 / k
            return f
        
        f_S = Q_inv_ones / denom
        
        # Check non-negativity
        removed = False
        new_active = set()
        for i, idx in enumerate(S):
            if f_S[i] < -1e-12:
                removed = True
            else:
                new_active.add(idx)
        
        if removed:
            active = new_active
            continue
        
        # Compute f
        f = np.zeros(n)
        for i, idx in enumerate(S):
            f[idx] = max(0.0, f_S[i])
        
        # Compute lambda
        Qf = Q @ f
        lambda_val = -2.0 * np.mean(Qf[list(S)])
        
        # Check KKT for inactive variables
        added = False
        for j in range(n):
            if j not in active:
                if 2.0 * Qf[j] + lambda_val < -1e-10:
                    active.add(j)
                    added = True
        
        if not added:
            return f
    
    # Fallback
    f = np.zeros(n)
    f[list(active)] = 1.0 / len(active)
    return f

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a QP-based active set approach.
Key innovation: At each step, form a weighted sum of active autocorrelation peaks
and solve the resulting (regularized) QP exactly, then use line search to ensure
monotonic improvement of the max-peak objective.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex (v >= 0, sum(v) = 1)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def form_Q_matrix(active_k, weights, n):
    """Form B = sum_k w_k * P_k where P_k[i,j] = 1 if j = k-i (valid), else 0."""
    B = np.zeros((n, n))
    for k, w in zip(active_k, weights):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            B[i, j] += w
    return B

def solve_qp_regularized(Q, n, support=None):
    """
    Solve min f^T Q f s.t. sum(f) = 1, f >= 0 using active set method.
    Q is assumed to be SPD (regularized).
    Returns the solution f of length n.
    """
    if support is None:
        support = set(range(n))
    support = set(support)
    
    for _ in range(200):
        S = sorted(support)
        k = len(S)
        if k == 0:
            return np.ones(n) / n
        if k == 1:
            f = np.zeros(n)
            f[S[0]] = 1.0
            return f
        
        # Extract submatrix
        Q_SS = Q[np.ix_(S, S)]
        
        # Solve Q_SS * f_S = 1 (then normalize)
        try:
            x = np.linalg.solve(Q_SS, np.ones(k))
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(Q_SS, np.ones(k), rcond=None)[0]
            except:
                x = np.ones(k) / k
        
        denom = x.sum()
        if abs(denom) < 1e-15:
            f = np.zeros(n)
            f[S] = 1.0 / k
            return f
        
        f_S = x / denom
        
        # Check non-negativity
        removed = False
        new_support = set()
        for i, idx in enumerate(S):
            if f_S[i] < -1e-10:
                removed = True
            else:
                new_support.add(idx)
        
        if removed:
            support = new_support
            continue
        
        # Form full f
        f = np.zeros(n)
        for i, idx in enumerate(S):
            f[idx] = max(0.0, f_S[i])
        
        # Compute Qf
        Qf = Q @ f
        
        # Compute lambda: from 2*Qf[i] + lambda = 0 on support
        lambda_val = -2.0 * np.mean(Qf[list(S)])
        
        # Check KKT for inactive variables: 2*Qf[j] + lambda >= 0
        added = False
        for j in range(n):
            if j not in support:
                if 2.0 * Qf[j] + lambda_val < -1e-8:
                    support.add(j)
                    added = True
        
        if not added:
            return f
    
    # Fallback: uniform over support
    f = np.zeros(n)
    f[list(support)] = 1.0 / len(support)
    return f

        M = c.max()
        
        # Select active peaks
        idx = np.argpartition(c, -n_active)[-n_active:]
        peak_vals = c[idx]
        
        # Softmax weights
        scaled = (peak_vals - M) / T
        weights = np.exp(scaled)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(len(idx)) / len(idx)
        else:
            weights = weights / wsum

        B = form_Q_matrix(idx, weights, n)

        Q = B + eps_reg * np.eye(n)

        f_new = solve_qp_regularized(Q, n, support=support)

        improved = False

            cs = candidate.sum()

            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-15:
                f = candidate
                best_obj = val
                best_f = f.copy()
                improved = True
                break
            elif val < M - 1e-15:
                # Accept even if not better than best (to escape local minima)
                f = candidate
                break

        else:
            no_improve += 1

def multiplicative_update(f0, time_limit, T=0.01, n_active=20, eta=0.5):
    """
    Sinkhorn-inspired multiplicative update for reducing peak autocorrelation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        c = fast_convolve(f)
        M = c.max()
        
        # Select active peaks
        idx = np.argpartition(c, -n_active)[-n_active:]
        peak_vals = c[idx]
        weights = np.exp((peak_vals - M) / T)
        ws = weights.sum()
        if ws < 1e-15:
            weights = np.ones(len(idx)) / len(idx)
        else:
            weights = weights / ws
        
        # Compute overlap: for each position i, how much does it contribute to active peaks?
        overlap = np.zeros(n)
        for k, w in zip(idx, weights):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                overlap[i] += w * f[j]
        
        # Multiplicative update
        f_new = f * np.exp(-eta * overlap / M)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        
        # Line search
        accepted = False
        for t in [1.0, 0.5, 0.3, 0.1]:
            candidate = (1 - t) * f + t * f_new
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate /= cs
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-15:
                f = candidate
                best_obj = val
                best_f = f.copy()
                accepted = True
                break
        
        if accepted:
            no_improve = 0
        else:
            no_improve += 1
            eta *= 0.8
            if no_improve > 50:
                break
    
    return best_f, best_obj


def smoothed_grad_descent(a0, time_limit, T_init=0.0005, lr_init=0.05):
    """Smoothed gradient descent using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient: for each i, grad[i] = 2 * sum_k p[k] * f[k-i] (where valid)
        # This is equivalent to 2 * (p * f)(reversed correlation)
        # More precisely: grad[i] = sum_k p[k] * 2 * f[k-i] * [0<=k-i<n]
        # = 2 * correlate(p, a, 'valid') ... not quite
        # Let me compute it directly:
        grad = np.zeros(n)
        for k in range(len(p)):
            if p[k] < 1e-15:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * p[k] * a[j]
        
        # Projected gradient step with backtracking
        best_candidate = a
        best_val = conv.max()
        
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            T *= 1.05
            lr *= 0.97
        if no_improve > 100:
            T *= 0.8
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a QP-based active set approach with
Hankel matrix structure exploitation, multiplicative (Sinkhorn-inspired) updates,
and smoothed gradient descent for final polish. Key innovation: at each step, form
the weighted Hankel matrix from active autocorrelation peaks and solve the resulting
regularized QP exactly via active set, then use line search for guaranteed progress.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def form_hankel_B(active_k, weights, n):
    """
    Form B = sum_k w_k * P_k where P_k[i,j] = 1 if j = k-i (valid), else 0.
    B is a Hankel matrix: B[i,j] = w_{i+j} (if i+j is an active peak index).
    """
    B = np.zeros((n, n))
    for k, w in zip(active_k, weights):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            B[i, j] += w
    return B


def solve_qp_hankel(B, n, eps_reg, support=None):
    """
    Solve min f^T (B + eps*I) f s.t. sum(f)=1, f>=0 using active set.
    B is symmetric Hankel. Uses direct linear solve for the reduced system.
    """
    Q = B + eps_reg * np.eye(n)
    if support is None:
        support = list(range(n))
    else:
        support = list(support)

    for _ in range(300):
        S = list(support)
        k = len(S)
        if k == 0:
            return np.ones(n) / n
        if k == 1:
            f = np.zeros(n)
            f[S[0]] = 1.0
            return f

        Q_SS = Q[np.ix_(S, S)]
        ones = np.ones(k)

        try:
            x = np.linalg.solve(Q_SS, ones)
        except np.linalg.LinAlgError:
            try:
                x = np.linalg.lstsq(Q_SS, ones, rcond=None)[0]
            except Exception:
                x = ones / k

        denom = x.sum()
        if abs(denom) < 1e-14:
            f = np.zeros(n)
            f[S] = 1.0 / k
            return f

        f_S = x / denom

        # Check non-negativity
        removed = False
        new_support = []
        for i, idx in enumerate(S):
            if f_S[i] < -1e-10:
                removed = True
            else:
                new_support.append(idx)

        if removed:
            support = new_support
            continue

        # Build full f
        f = np.zeros(n)
        for i, idx in enumerate(S):
            f[idx] = max(0.0, f_S[i])

        # Compute Qf
        Qf = Q @ f

        # Lambda from active set
        lambda_val = -2.0 * np.mean(Qf[np.array(S)])

        # Check KKT for inactive
        added = False
        new_support_set = set(S)
        for j in range(n):
            if j not in new_support_set:
                if 2.0 * Qf[j] + lambda_val < -1e-6:
                    support = list(new_support_set) + [j]
                    added = True

        if not added:
            return f

    # Fallback
    f = np.zeros(n)
    f[support] = 1.0 / len(support)
    return f


def qp_optimize(f0, time_limit, T_init=0.005, n_active_init=20, verbose=True):
    """
    QP-based optimization: iteratively solve regularized QP on active peaks

    s = f.sum()

    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = float(fast_convolve(f).max())

    T = T_init
    n_active = n_active_init
    eps_reg = 5.0
    support = list(range(n))

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        # Compute autocorrelation
        c = fast_convolve(f)
        M = c.max()

        # Select active peaks
        na = min(n_active, len(c))
        idx = np.argpartition(c, -na)[-na:]
        peak_vals = c[idx]

        # Softmax weights
        scaled = (peak_vals - M) / T
        scaled -= scaled.max()
        weights = np.exp(scaled)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(len(idx)) / len(idx)
        else:
            weights = weights / wsum

        # Form Hankel matrix B
        B = form_hankel_B(idx, weights, n)

        # Solve regularized QP
        f_new = solve_qp_hankel(B, n, eps_reg, support=support)

        # Line search: try to improve
        improved = False
        for t in [1.0, 0.9, 0.7, 0.5, 0.4, 0.3, 0.2, 0.1]:
            candidate = (1.0 - t) * f + t * f_new
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            val = float(fast_convolve(candidate).max())
            if val < best_obj - 1e-14:
                f = candidate
                best_obj = val
                best_f = f.copy()
                improved = True
                break

        if not improved:
            # Try accepting current QP solution if it reduces current M (not best)
            for t in [0.3, 0.2, 0.1]:
                candidate = (1.0 - t) * f + t * f_new
                candidate = np.maximum(candidate, 0.0)
                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs
                val = float(fast_convolve(candidate).max())
                if val < M - 1e-8:
                    f = candidate
                    break

        if improved:
            no_improve = 0
            # Decrease regularization as we converge
            eps_reg = max(2.0, eps_reg * 0.995)
        else:
            no_improve += 1

            n_active = min(n_active + 5, 80)
        if no_improve > 40:

            n_active = max(n_active - 15, 10)
            eps_reg = max(2.0, eps_reg * 0.9)

            print(f"    QP it={it}, obj={best_obj:.8f}, T={T:.4f}, k={n_active}, eps={eps_reg:.1f}, no_imp={no_improve}")

    return best_f, best_obj
