import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def subgradient_peak(a, conv, n_peaks=1):
    """Compute subgradient at the top peaks."""
    n = len(a)
    k_actual = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    # Equal weight on top peaks
    sg = np.zeros(n)
    for k in peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * a[k - idx] / k_actual
    return sg

        return subgradient_peak(a, conv, 1)
    w /= ws
    
    sg = np.zeros(n)
    # Only iterate over significant weights
    thresh = w.max() * 1e-6
    significant = np.where(w > thresh)[0]
    for i in significant:
        k = int(i)
        if w[i] < 1e-10:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def subgrad_optimize(f0, time_limit, n_peaks=1, lr=0.005, momentum=0.9,
                      temperature=None, use_soft=False, verbose=True):
    """
    Subgradient optimization with Nesterov momentum.
    Uses additive update with simplex projection.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_npeaks = n_peaks

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f - cur_lr * v
        f_nest = project_simplex(f_nest)

        conv_nest = fast_convolve(f_nest)
        if use_soft and temperature is not None:
            sg = subgradient_softmax(f_nest, conv_nest, temperature)
        else:
            sg = subgradient_peak(f_nest, conv_nest, cur_npeaks)

        # Update momentum
        v = momentum * v + (1 - momentum) * sg

        # Additive update
        f_new = f - cur_lr * v
        f_new = project_simplex(f_new)
        f_new = np.maximum(f_new, 0)
        fs = f_new.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr *= 0.5
            continue
        f_new /= fs
        f = f_new

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive scheduling
        if no_improve < 100:
            cur_lr *= 0.9995
        elif no_improve < 300:
            cur_lr *= 0.998
        else:
            cur_lr *= 0.995

        # Gradually increase number of peaks considered
        if it % 200 == 0 and cur_npeaks < min(10, n // 10):
            cur_npeaks += 1

        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng((it * 31 + 7) % 1000000)
            noise_scale = 0.001 + 0.002 * min(no_improve // 500, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            cur_npeaks = 1
            conv = fast_convolve(f)
            no_improve = 0

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, lr={cur_nr:.7f}" if False else f"    SG iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}, lr={cur_lr:.7f}")

    return best_f, best_obj


def peak_surgery(f0, time_limit, verbose=True):
    """
    Peak-surgery local search: identifies the top peaks and proposes
    targeted moves to reduce them without increasing others.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_max = conv.max()

        # Find top peaks
        n_peaks_to_consider = min(5, len(conv))
        top_peaks = np.argpartition(conv, -n_peaks_to_consider)[-n_peaks_to_consider:]
        top_peaks = top_peaks[np.argsort(conv[top_peaks])[::-1]]

        improved = False

        # For each peak, try to reduce it
        for k in top_peaks[:3]:
            k = int(k)
            if conv[k] < best_obj * 0.99:
                continue  # Not a significant peak

            # Get contributing coordinates
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            partners = k - idx
            contribs = f[idx] * f[partners]
            total_contrib = contribs.sum()
            if total_contrib < 1e-15:
                continue

            # Try reducing the top contributors
            order = np.argsort(contribs)[::-1]
            for oi in order[:5]:
                i = idx[oi]
                j = partners[oi]
                if f[i] < 1e-10:
                    continue

                # Try transferring mass from i to a random location
                # that doesn't affect this peak
                for _ in range(20):
                    target = rng.integers(0, n)
                    if target == i or target == j:
                        continue
                    # Check if target doesn't contribute to this peak
                    # target contributes to peak k if target is in [js, je] and k-target is in [0, n-1]
                    if js <= target <= je and 0 <= k - target < n:
                        continue  # Would still affect this peak

                    for frac in [0.05, 0.1, 0.2, 0.3]:
                        amt = f[i] * frac
                        if amt < 1e-12:
                            break
                        f[i] -= amt
                        f[target] += amt
                        new_conv = fast_convolve(f)
                        new_obj = new_conv.max()
                        if new_obj < best_obj - 1e-13:
                            best_obj = new_obj
                            best_f = f.copy()
                            improved = True
                            conv = new_conv
                            break
                        else:
                            f[i] += amt
                            f[target] -= amt
                            break  # Try less aggressive

                if improved:
                    break

        if not improved:
            # Try random pair moves
            for _ in range(200):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or f[i] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[i] * frac
                    f[i] -= amt
                    f[j] += amt
                    new_obj = fast_convolve(f).max()
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_f = f.copy()
                        improved = True
                        break
                    f[i] += amt
                    f[j] -= amt
                    break

            if not improved:
                # Escalate: swap values
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    f[i], f[j] = f[j], f[i]
                    f = project_simplex(f)
                    new_obj = fast_convolve(f).max()
                    if new_obj < best_obj - 1e-13:
                        best_obj = new_obj
                        best_f = f.copy()
                        improved = True
                    else:
                        f = best_f.copy()

                if not improved:
                    # Add noise and retry
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                    conv = fast_convolve(f)
                    if conv.max() < best_obj:
                        best_obj = conv.max()
                        best_f = f.copy()
                    else:
                        f = best_f.copy()
                    # Give up if we've been stuck too long
                    if time.monotonic() - t0 > time_limit * 0.8:
                        break

    return best_f, best_obj


def simulated_annealing(f0, time_limit, t_start=0.01, t_end=0.0001, verbose=True):
    """
    Simulated annealing with peak-targeted moves.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    current_obj = conv.max()
    best_f = f.copy()
    best_obj = current_obj

    t0 = time.monotonic()
    rng = np.random.default_rng(67890)
    total_iters = 0
    accepted = 0

    while time.monotonic() - t0 < time_limit:
        # Adaptive temperature
        progress = (time.monotonic() - t0) / time_limit
        temperature = t_start * (t_end / t_start) ** progress

        # Generate move
        move_type = rng.integers(0, 3)

        if move_type == 0:
            # Single coordinate scaling
            i = rng.integers(0, n)
            old_val = f[i]
            if old_val < 1e-12:
                continue
            factor = np.exp(rng.standard_normal() * 0.3)
            new_val = old_val * factor
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            scale = (1.0 - new_val) / denom
            f_trial = f.copy()
            f_trial[i] = new_val
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            f_trial[mask] *= scale
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial /= st

        elif move_type == 1:
            # Pair transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            amt = abs(rng.standard_normal()) * 0.01
            if f[i] < amt:
                f[i], f[j] = f[j], f[i]
                i, j = j, i
                if f[i] < amt:
                    continue
            f_trial = f.copy()
            f_trial[i] -= amt
            f_trial[j] += amt

        else:
            # Peak-targeted move: reduce a coordinate that contributes to the max peak
            conv_cur = fast_convolve(f)
            k = int(np.argmax(conv_cur))
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            # Pick a contributor proportional to its contribution
            total = contribs.sum()
            if total < 1e-15:
                continue
            probs = contribs / total
            pick = rng.choice(len(idx), p=probs)
            i = idx[pick]
            j = rng.integers(0, n)
            if j == i:
                continue
            frac = abs(rng.standard_normal()) * 0.05 + 0.01
            amt = f[i] * frac
            if amt < 1e-12:
                continue
            f_trial = f.copy()
            f_trial[i] -= amt
            f_trial[j] += amt

        new_conv = fast_convolve(f_trial)
        new_obj = new_conv.max()
        total_iters += 1

        delta = new_obj - current_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(temperature, 1e-10)):
            f = f_trial
            current_obj = new_obj
            accepted += 1
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                if verbose and (total_iters % 5000 == 0):
                    print(f"    SA: iter={total_iters}, obj={best_obj:.10f}, T={temperature:.6f}")
        else:
            # Revert (f unchanged)
            pass

    return best_f, best_obj


def multiply_weights_optimize(f0, time_limit, top_k=1, temperature=None,
                               lr=0.003, momentum=0.85, use_softmax=False,
                               verbose=True):
    """
    Multiplicative weights subgradient optimization.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_top_k = top_k

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(momentum * 0.5 * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        if use_softmax and temperature is not None:
            sg = subgradient_softmax(f_nest, conv_nest, temperature)
        else:
            sg = subgradient_peak(f_nest, conv_nest, cur_top_k)

        v = momentum * v + (1 - momentum) * sg

        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.3, 0.0001)
            continue
        f = np.maximum(f, 0.0)
        f /= f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        if cur_lr < 0.00003:
            cur_lr = 0.00003

        if it % 300 == 0:
            cur_top_k = min(cur_top_k + 1, max(5, n // 20))

        if no_improve > 400:
            rng = np.random.default_rng((it * 17 + 3) % 1000000)
            noise_scale = 0.001 + 0.001 * min(no_improve // 400, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0005)
            cur_top_k = 1
            conv = fast_convolve(f)
            no_improve = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MU iter {it}, t={elapsed:.1f}s, obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Use cubic spline for smoother interpolation
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(old_x, f, bc_type='natural')
        f_interp = cs(new_x)
    except Exception:
        f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def diverse_inits(n, rng, count=5):
    """Generate diverse initial sequences."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian
    center = n / 2.0
    sigma = n / (3.0 + rng.random() * 5.0)
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Hann window
    x = np.linspace(0, 1, n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann = np.maximum(hann, 1e-10)
    inits.append(hann / hann.sum())

    # Power law taper
    center = (n - 1) / 2.0
    d = np.abs(np.arange(n) - center) / center
    alpha = 0.5 + rng.random() * 2.0
    taper = (1 - d) ** alpha
    taper = np.maximum(taper, 1e-10)
    inits.append(taper / taper.sum())

    # Random smoothed
    r = np.abs(rng.standard_normal(n))
    kernel_len = max(3, n // 50)
    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-10)
    inits.append(r / r.sum())

    # Two-lobe structure
    c1 = n * (0.25 + rng.random() * 0.1)
    c2 = n * (0.65 + rng.random() * 0.1)
    s1 = n * 0.05
    s2 = n * 0.05
    two_lobe = np.exp(-0.5 * ((np.arange(n) - c1) / s1) ** 2) + \
               np.exp(-0.5 * ((np.arange(n) - c2) / s2) ** 2)
    two_lobe = np.maximum(two_lobe, 1e-10)
    inits.append(two_lobe / two_lobe.sum())

    return inits[:count]


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining subgradient methods, peak surgery,
    simulated annealing, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent found, using uniform n=512")

    # === Phase 1: Subgradient on parent with different configs ===
    print("\n=== Phase 1: Subgradient Optimization ===")

    n_parent = len(parent)
    configs = [
        {"top_k": 1, "lr": 0.004, "mom": 0.88, "softmax": False},
        {"top_k": 3, "lr": 0.003, "mom": 0.90, "softmax": False},
        {"top_k": 5, "lr": 0.002, "mom": 0.92, "softmax": True, "temp": 0.001},
        {"top_k": 1, "lr": 0.006, "mom": 0.85, "softmax": False},
    ]

    for p, cfg in enumerate(configs):
        rem = deadline - time.monotonic() - 600
        if rem < 20:
            break
        t_alloc = min(rem * 0.25, 180)
        if t_alloc < 10:
            t_alloc = max(rem * 0.4, 15)

        if p == 0:
            f_init = parent.copy()
        elif p == 1:
            noise = rng.standard_normal(n_parent) * 0.005
            f_init = project_simplex(parent + noise)
        elif p == 2:
            f_init = 0.7 * parent + 0.3 * (np.ones(n_parent) / n_parent)
            f_init = project_simplex(f_init)
        else:
            f_init = parent.copy()

        print(f"  Config {p}: n={n_parent}, t={t_alloc:.0f}s")
        a_opt, _ = multiply_weights_optimize(f_init, time_limit=t_alloc,
                                              top_k=cfg["top_k"],
                                              temperature=cfg.get("temp"),
                                              lr=cfg["lr"],
                                              momentum=cfg["mom"],
                                              use_softmax=cfg.get("softmax", False))
        try_seq(a_opt, f"sg_{p}")

    # === Phase 2: Peak Surgery ===
    print(f"\n=== Phase 2: Peak Surgery (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_ps = min(rem * 0.5, 200)
            print(f"  Peak surgery: t={t_ps:.0f}s, n={len(best_seq)}")
            a_opt, _ = peak_surgery(best_seq, time_limit=t_ps)
            try_seq(a_opt, "psurgery")

    # === Phase 3: Simulated Annealing ===
    print(f"\n=== Phase 3: Simulated Annealing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_sa = min(rem * 0.5, 250)
            print(f"  SA: t={t_sa:.0f}s, n={len(best_seq)}")
            a_opt, _ = simulated_annealing(best_seq, time_limit=t_sa,
                                            t_start=0.005, t_end=0.00005)
            try_seq(a_opt, "sa")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [1024, 1280, 1536]

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 120)
            print(f"  n={target_n}: t={t_alloc:.0f}s")

            # Optimize the interpolated sequence
            a_opt, _ = multiply_weights_optimize(f_interp, time_limit=t_alloc,
                                                  top_k=1, temperature=None,
                                                  lr=0.004, momentum=0.88)
            try_seq(a_opt, f"len_{target_n}")

            # Quick peak surgery on the result
            rem = deadline - time.monotonic() - 150
            if rem > 10:
                a_opt2, _ = peak_surgery(a_opt, time_limit=min(rem * 0.3, 40))
                try_seq(a_opt2, f"len_ps_{target_n}")

    # === Phase 5: Diverse starts at optimal length ===
    print(f"\n=== Phase 5: Diverse Starts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        inits = diverse_inits(n_best, rng, count=4)

        for idx, init in enumerate(inits):
            rem = deadline - time.monotonic() - 100
            if rem < 10:
                break
            # Blend with current best
            f_init = 0.5 * best_seq + 0.5 * init
            f_init = project_simplex(np.maximum(f_init, 0))
            t_alloc = min(rem * 0.2, 60)

            a_opt, _ = multiply_weights_optimize(f_init, time_limit=t_alloc,
                                                  top_k=1, lr=0.005,
                                                  momentum=0.90)
            try_seq(a_opt, f"div_{idx}")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine subgradient
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = multiply_weights_optimize(best_seq, time_limit=min(rem * 0.4, 60),
                                                  top_k=1, lr=0.0005,
                                                  momentum=0.95)
            try_seq(a_opt, "fine_sg")

        # Peak surgery
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            a_opt, _ = peak_surgery(best_seq, time_limit=min(rem * 0.5, 30))
            try_seq(a_opt, "fine_ps")

        # Small perturbations + quick optimize
        for scale in [0.0005, 0.001, 0.002]:
            rem = deadline - time.monotonic() - 10
            if rem < 5:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = multiply_weights_optimize(f_pert, time_limit=min(rem * 0.4, 20),
                                                  top_k=1, lr=0.001,
                                                  momentum=0.92)
            try_seq(a_opt, f"pert_{scale}")

    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
