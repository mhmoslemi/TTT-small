    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new


def nesterov_lse(a0, time_limit, t_init=50.0, t_max=1e5, t_growth=1.002,
                  lr_init=0.02, mom_init=0.9, name=""):
    """Nesterov accelerated projected gradient on log-sum-exp smoothed max-conv."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_autoconv(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    y = a.copy()
    v = np.zeros(n)
    t_curr = t_init
    lr = lr_init
    mom = mom_init
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        y = a + mom * v
        y = project_simplex(y)
        
        # Compute gradient at y
        conv_y = fast_autoconv(y)
        tc = t_curr * conv_y
        tc -= tc.max()
        exp_tc = np.exp(tc)
        w = exp_tc / exp_tc.sum()
        grad = grad_wrt_weights(y, w)
        
        # Update
        a_new = project_simplex(y - lr * grad)
        v = a_new - a
        a = a_new
        
        # Evaluate true objective
        conv_a = fast_autoconv(a)
        obj = conv_a.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        t_curr = min(t_curr * t_growth, t_max)
        
        if no_improve > 200:
            # Small perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            y = a.copy()
            v = np.zeros(n)
            t_curr = t_init * 0.5
            lr = lr_init
            no_improve = 0
        
        if it % 10000 == 0 and name:
            elapsed = time.monotonic() - t0
            print(f"    {name} it={it}, obj={best_obj:.8f}, t={t_curr:.0f}, lr={lr:.4f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_obj


def alternating_minimax(a0, time_limit, gamma_init=10.0, gamma_growth=1.05,
                        inner_iters=20, inner_lr=0.1, name=""):
    """Alternating minimax: inner QP on a, outer dual ascent on p."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_autoconv(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    p = np.ones(2 * n - 1) / (2 * n - 1)
    gamma = gamma_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(67890)
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Inner: minimize a^T Q_p a s.t. a in simplex
        # Use projected gradient with the current p
        for s_iter in range(inner_iters):
            if time.monotonic() - t0 > time_limit:
                break
            grad = grad_wrt_weights(a, p)
            # Safe learning rate for inner loop
            inner_lr_safe = inner_lr / (1.0 + gamma * it * 1e-6)
            a_new = project_simplex(a - inner_lr_safe * grad)
            a = a_new
        
        # Outer: update p via dual ascent
        conv_a = fast_autoconv(a)
        obj = conv_a.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if it % 100 == 0 and name:
                elapsed = time.monotonic() - t0
                print(f"    {name} outer_it={it}, obj={best_obj:.8f}, gamma={gamma:.1f}, elapsed={elapsed:.0f}s")
        else:
            no_improve += 1
        
        # Dual update
        p = softmax_stable(conv_a, gamma)
        gamma *= gamma_growth
        
        if no_improve > 100:
            # Perturb and restart
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            p = np.ones(2 * n - 1) / (2 * n - 1)
            gamma = gamma_init
            no_improve = 0
    
    return best_a, best_obj
