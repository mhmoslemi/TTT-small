Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Log-sum-exp smooth gradient with Adam optimizer and τ-annealing
2. Multiplicative weights (mirror descent) refinement
3. Coordinate descent with golden section search

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def lse_grad_and_val(a, tau):
    """Compute log-sum-exp value and its gradient."""
    conv = fast_convolve(a)
    m = conv.max()
    shifted = tau * (conv - m)
    shifted -= shifted.max()  # numerical stability
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    val = m + log_sum / tau
    p = exp_shifted / exp_shifted.sum()
    grad = np.correlate(p, a, mode='valid')
    return val, grad, conv, p

def adam_lse_optimize(f0, time_limit, tau_init=10.0, tau_max=1000.0, 
                       lr=0.01, beta1=0.8, beta2=0.999, eps=1e-8,
                       tau_growth=1.001, verbose=False):
    """Adam optimizer on log-sum-exp smooth approximation with τ-annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    tau = tau_init
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        val, grad, conv, p = lse_grad_and_val(f, tau)
        
        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        m_hat_hat = m_hat / (1 - beta1 ** t)
        v_hat_hat = v_hat / (1 - beta2 ** t)
        
        # Update in dual space (before projection)
        f_new = f - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps)
        f_new = project_simplex(f_new)
        
        # Evaluate actual objective
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and t % 100 == 0:
                print(f"    Adam-LSE iter {t}: obj={best_obj:.10f}, tau={tau:.1f}")
        else:
            no_improve += 1
            f = f_new  # still update to explore
        
        # Anneal tau
        if no_improve % 50 == 0 or t % 200 == 0:
            tau = min(tau * tau_growth, tau_max)
        
        if no_improve > 500:
            # Restart with perturbation
            rng = np.random.default_rng(t * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            t = 0
            tau = max(tau_init, tau * 0.5)
            lr = max(0.001, lr * 0.7)
            no_improve = 0
        
        # Decay learning rate slowly
        if t % 1000 == 0:
            lr *= 0.95
    
    return best_f, best_obj

2. Log-sum-exp smooth gradient with Adam optimizer and tau-annealing

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_simplex_sym(v, n):
    """Project onto symmetric simplex: v is the half-sequence, project so full sums to 1."""
    if n % 2 == 1:
        # Odd: half has (n+1)/2 elements, full = [sym, reverse(sym)[:-1]]
        # sum(full) = 2*sum(sym) - sym[-1] = 1
        # So we need 2*sum(sym) - sym[-1] = 1
        # This is a simple linear constraint. Project v onto {x>=0, 2*sum(x)-x[-1]=1}
        m = len(v)
        # Sum of first m-1 elements: s1, last element: x_m
        # 2*(s1 + x_m) - x_m = 2*s1 + x_m = 1
        # Let's just project the full sequence
        full = np.zeros(n)
        full[:m] = v
        full[m:] = v[:0:-1]
        full_proj = project_simplex(full)
        # Extract symmetric part
        return full_proj[:m].copy()
    else:
        # Even: half has n/2 elements, full = [sym, reverse(sym)]
        # sum(full) = 2*sum(sym) = 1, so sum(sym) = 0.5
        m = len(v)
        full = np.zeros(n)
        full[:m] = v
        full[m:] = v[::-1]
        full_proj = project_simplex(full)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

    return f_interp

    grad = np.correlate(p, a, mode='valid')

            print(f"    Spectral-L4 iter {it}: max_conv={best_obj:.10f}, lr={cur_lr:.6f}")

    return best_f, best_obj


def adam_lse_optimize(f0, time_limit, tau_init=20.0, tau_max=5000.0,
                      lr=0.005, beta1=0.85, beta2=0.999, eps=1e-8,
                      tau_growth=1.002, verbose=False):
    """Adam optimizer on log-sum-exp smooth approximation with tau-annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    tau = tau_init

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        val, grad, conv = lse_grad_and_val(f, tau)

        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        m_hat_hat = m_hat / (1 - beta1 ** t)
        v_hat_hat = v_hat / (1 - beta2 ** t)

        f_new = project_simplex(f - lr * m_hat_hat / (np.sqrt(v_hat_hat) + eps))

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and t % 200 == 0:
                print(f"    Adam-LSE iter {t}: obj={best_obj:.10f}, tau={tau:.1f}")
        else:
            no_improve += 1

        f = f_new

        # Anneal tau: grow it to make the approximation sharper
        if t % 100 == 0:
            tau = min(tau * tau_growth ** (no_improve + 1), tau_max)

        # Restart if stuck
        if no_improve > 400:
            rng = np.random.default_rng(t * 31 + 7)
            scale = 0.002 + 0.003 * min(no_improve // 200, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            m_hat = np.zeros(n)
            v_hat = np.zeros(n)
            t = 0
            tau = max(tau_init, tau * 0.3)
            lr = max(0.001, lr * 0.6)
            no_improve = 0

        if t % 2000 == 0:
            lr *= 0.95

    return best_f, best_obj


def mul_weights_optimize(f0, time_limit, eta_init=0.05, top_k=30, verbose=False):
    """Mirror descent with multiplicative weights (negative entropy)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    eta = eta_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        # Average subgradient over top peaks
        grad = np.zeros(n)
        for k in peaks:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            kg = ki - idx
            mask = (kg >= 0) & (kg < n)
            grad[idx[mask]] += f[kg[mask]] / k_actual

        # Multiplicative update
        f_new = f * np.exp(-eta * grad)
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s_new

        conv = fast_convolve(f_new)
        obj_new = conv.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        eta *= 0.99995

        if no_improve > 200:
            rng = np.random.default_rng(it * 104729 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            eta = eta_init * 0.5
            no_improve = 0

    return best_f, best_obj


def coord_descent_golden(f0, time_limit, verbose=False):
    """Coordinate descent with golden section search on peak-contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Find top 3 peaks
        top_k = min(3, 2 * n - 1)
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            # Rank coordinates by contribution to this peak
            contribs = []
            for j in range(js, je + 1):
                contribs.append((f[j] * f[peak_k - j], j))
            contribs.sort(reverse=True)

            # Try top coordinates
            for _, j in contribs[:8]:
                if time.monotonic() - t0 > time_limit:
                    break
                if f[j] < 1e-14:
                    continue

                # Golden section search on f[j]
                fj = f[j]
                a_lo = max(0.0, fj * 0.2)
                a_hi = min(0.8, fj * 2.5 + 0.01)
                if a_hi - a_lo < 1e-12:
                    continue

                c = a_hi - inv_phi * (a_hi - a_lo)
                d = a_lo + inv_phi * (a_hi - a_lo)

                def _eval(v_val):
                    if v_val < 0 or v_val > 1:
                        return best_obj
                    denom = 1.0 - fj
                    if denom < 1e-12:
                        return best_obj
                    scale = (1.0 - v_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = v_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    return fast_convolve(f_trial).max()

                fc = _eval(c)
                fd = _eval(d)

                for _ in range(15):
                    if a_hi - a_lo < 1e-10:
                        break
                    if fc < fd:
                        a_hi = d
                        d = c
                        fd = fc
                        c = a_hi - inv_phi * (a_hi - a_lo)
                        fc = _eval(c)
                    else:
                        a_lo = c
                        c = d
                        fc = fd
                        d = a_lo + inv_phi * (a_hi - a_lo)
                        fd = _eval(d)

                best_v = (a_lo + a_hi) / 2.
                best_val = _eval(best_v)

                if best_val < best_obj - 1e-13:
                    denom = 1.0 - fj
                    if denom > 1e-12:
                        scale = (1.0 - best_v) / denom
                        f[j] = best_v
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    conv = fast_convolve(f)
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 4:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe: solve min t s.t. (f*g)[k] <= t, g>=0, sum(g)=1."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        n_constraints = 2 * n - 1
        A_ub = np.zeros((n_constraints, n + 1))
        for j in range(n):
            A_ub[j:j + n, j] = f
        A_ub[:, -1] = -1.0
        b_ub = np.zeros(n_constraints)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g = res.x[:n]
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n
        else:
            g = np.ones(n) / n

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = best_obj
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization exploiting symmetry and using
    spectral L4 pre-conditioning, Adam+LSE, multiplicative weights,
    coordinate descent, and LP-FW.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(832) / 832
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # Phase 1: Spectral L4 pre-conditioning
    print("\n=== Phase 1: Spectral L4 pre-conditioning ===")
    rem = deadline - time.monotonic()
    t1 = min(300, rem * 0.2)
    if t1 > 20:
        print(f"  n={base_n}, t={t1:.0f}s")
        a_opt, _ = spectral_l4_optimize(parent, time_limit=t1, lr=0.05, verbose=True)
        try_seq(a_opt, "spectral_l4")

    # Phase 2: Adam + Log-sum-exp with tau annealing
    print("\n=== Phase 2: Adam + LSE (tau-annealing) ===")
    start_f = best_seq if best_seq is not None else parent
    rem = deadline - time.monotonic()
    t2 = min(400, rem * 0.3)
    if t2 > 30:
        print(f"  n={len(start_f)}, t={t2:.0f}s")
        a_opt, _ = adam_lse_optimize(start_f, time_limit=t2,
                                     tau_init=30.0, tau_max=5000.0,
                                     lr=0.004, beta1=0.85, beta2=0.999,
                                     tau_growth=1.002, verbose=True)
        try_seq(a_opt, "adam_lse_main")

    # Phase 3: Second Adam+LSE with different params
    print("\n=== Phase 3: Adam + LSE (fine) ===")
    rem = deadline - time.monotonic()
    t3 = min(250, rem * 0.2)
    if t3 > 20 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t3:.0f}s")
        a_opt, _ = adam_lse_optimize(best_seq, time_limit=t3,
                                     tau_init=100.0, tau_max=8000.0,
                                     lr=0.002, beta1=0.9, beta2=0.9999,
                                     tau_growth=1.003, verbose=True)
        try_seq(a_opt, "adam_lse_fine")

    # Phase 4: Multiplicative weights
    print("\n=== Phase 4: Multiplicative weights ===")
    rem = deadline - time.monotonic()
    t4 = min(150, rem * 0.15)
    if t4 > 20 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t4:.0f}s")
        a_opt, _ = mul_weights_optimize(best_seq, time_limit=t4,
                                        eta_init=0.03, top_k=40)
        try_seq(a_opt, "mul_weights")

    # Phase 5: Coordinate descent with golden section
    print("\n=== Phase 5: Golden section CD ===")
    rem = deadline - time.monotonic()
    t5 = min(200, rem * 0.2)
    if t5 > 15 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t5:.0f}s")
        a_opt, _ = coord_descent_golden(best_seq, time_limit=t5, verbose=True)
        try_seq(a_opt, "golden_cd")

    # Phase 6: LP-FW
    print("\n=== Phase 6: LP-FW ===")
    rem = deadline - time.monotonic()
    t6 = min(120, rem * 0.12)
    if t6 > 15 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t6:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t6, verbose=True)
        try_seq(a_opt, "lp_fw")

    # Phase 7: Multi-length exploration
    print("\n=== Phase 7: Multi-length exploration ===")
    if best_seq is not None:
        base = best_seq
        candidate_lengths = []
        for delta in [-128, -64, -32, 32, 64, 128, 256]:
            ln = len(base) + delta
            if ln >= 256 and ln <= 2048:
                candidate_lengths.append(ln)
        candidate_lengths = sorted(set(candidate_lengths))

        for target_n in candidate_lengths[:3]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.3, 100)
            if t_alloc < 15:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  Adam-LSE n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_lse_optimize(f_interp, time_limit=t_alloc,
                                         tau_init=50.0, tau_max=3000.0,
                                         lr=0.003, beta1=0.85, beta2=0.999,
                                         tau_growth=1.002)
            try_seq(a_opt, f"len_{target_n}")

    # Phase 8: Final polish
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick golden CD
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_polish = min(rem * 0.4, 80)
            print(f"  CD polish t={t_polish:.0f}s")
            a_opt, _ = coord_descent_golden(best_seq, time_limit=t_polish)
            try_seq(a_opt, "cd_polish")

        # Quick Adam-LSE
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_polish = min(rem * 0.4, 60)
            print(f"  LSE polish t={t_polish:.0f}s")
            a_opt, _ = adam_lse_optimize(best_seq, time_limit=t_polish,
                                         tau_init=200.0, tau_max=10000.0,
                                         lr=0.001, beta1=0.9, beta2=0.9999,
                                         tau_growth=1.005)
            try_seq(a_opt, "lse_polish")

        # Small perturbations + quick optimize
        for p in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.001 * p)
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem, 20)
            a_opt, _ = adam_lse_optimize(f_pert, time_limit=t_pert,
                                         tau_init=100.0, tau_max=5000.0,
                                         lr=0.001, beta1=0.9, beta2=0.999,
                                         tau_growth=1.003)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(832) / 832
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(832) / 832
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
