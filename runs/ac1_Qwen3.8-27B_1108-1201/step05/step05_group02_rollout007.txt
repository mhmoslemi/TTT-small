def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        else:
            no_improve += 1

            y = f.copy()
            v = np.zeros(n)

def peak_spread_cd(f0, time_limit, verbose=False):
    """Coordinate descent that specifically targets peak reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        
        # Find top peaks
        top_k = min(5, len(c))
        peak_positions = np.argpartition(c, -top_k)[-top_k:]
        
        for pk in peak_positions:
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            
            if je < js:
                continue
            
            # For each coordinate that contributes to this peak,
            # try to reduce it
            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                
                old_val = f[j]
                if old_val < 1e-13:
                    continue
                
                # The contribution of a[j] to c[peak_k] is 2*a[j]*a[peak_k-j] (or a[j]^2 if j == peak_k-j)
                # Try reducing a[j] and redistributing to a random other coordinate
                for mult in [0.5, 0.7, 0.85]:
                    new_val = old_val * mult
                    released = old_val - new_val
                    if released < 1e-15:
                        continue
                    
                    # Add released mass to a random coordinate far from the peak
                    candidates = [m for m in range(n) if m != j and abs(m - peak_k) > n // 4]
                    if not candidates:
                        candidates = [m for m in range(n) if m != j]
                    if not candidates:
                        continue
                    
                    target = candidates[np.random.default_rng(it * 1000 + j + int(mult * 100)).integers(0, len(candidates))]
                    
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    f_trial[target] += released
                    f_trial = np.maximum(f_trial, 0)
                    st = f_trial.sum()
                    if st < 1e-15:
                        continue
                    f_trial = f_trial / st
                    
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        c = fast_convolve(f)
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj

**Idea: Semi-definite relaxation**
The problem `min t s.t. sum_j a[j]*a[k-j] <= t, a >= 0, sum(a) = 1` can be relaxed. If we let `X = a*a^T` (outer product, rank 1), then the constraint becomes `sum_j X[j, k-j] <= t`. **Idea: Fictitious play / repeated games**
Model the problem as a game where "Nash equilibrium" corresponds to the optimal solution.

"""
Minimizes 2*n*max(conv(a,a))/sum(a)**2 using:
1. LP-based Frank-Wolfe (primary)
2. Multiplicative-weights mirror descent (diversity)
3. Weighted FW with peak focusing
4. Adaptive length exploration
5. Final polish with accelerated subgradient

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_descent(f0, time_limit, lr=0.01, verbose=False):
    """Multiplicative-weights mirror descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t = time.monotonic() - t0
        frac = min(1.0, t / time_limit)
        
        # Anneal learning rate
        effective_lr = cur_lr * (1.0 - 0.7 * frac)
        
        # Find the peak and compute subgradient
        k_peak = int(np.argmax(c))
        
        # Subgradient of c[k_peak] at f: 2*f shifted
        # grad[j] = 2 * f[k_peak - j] for valid j
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        for j in range(js, je + 1):
            grad[j] = 2.0 * f[k_peak - j]
        
        # Multiplicative update: f_new[j] = f[j] * exp(-lr * grad[j])
        # But we need to ensure non-negativity and sum=1
        log_f = np.log(f + 1e-30) - effective_lr * grad
        log_f -= log_f.max()  # numerical stability
        f_new = np.exp(log_f)
        sn = f_new.sum()
        if sn < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / sn
        
        c = fast_convolve(f_new)
        co = c.max()
        
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new
        
        if no_improve > 200:
            # Try a larger step or restart
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            c = fast_convolve(f)
            no_improve = 0
            cur_lr = max(0.001, cur_lr * 0.8)
    
    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with min-peak LP and aggressive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve min-peak LP
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        n_constraints = 2 * n - 1
        A_ub = np.zeros((n_constraints, n + 1))
        for j in range(n):
            A_ub[j:j + n, j] = f
        A_ub[:, -1] = -1.0
        b_ub = np.zeros(n_constraints)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.success:
            g = res.x[:n]
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n
        else:
            # Fallback: uniform
            g = np.ones(n) / n
        
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Aggressive line search
        best_alpha = 0.0
        best_val = M + 1e-10
        
        # Coarse search
        alphas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-14:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.002 * min(no_improve // 30, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            c = fast_convolve(f)
            M = c.max()
            no_improve = 0
    
    return best_f, best_obj


def weighted_subgradient(f0, time_limit, verbose=False):
    """Subgradient with adaptive softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    momentum = 0.9
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t = time.monotonic() - t0
        frac = min(1.0, t / time_limit)
        
        # Cosine-annealed step size
        lr = 0.004 * (0.1 + 0.9 * (1 + np.cos(np.pi * frac)) / 2)
        
        # Softmax-weighted peaks
        temperature = 0.00005 + 0.0005 * frac
        k_actual = min(40, len(c))
        peaks = np.argpartition(c, -k_actual)[-k_actual:]
        pv = c[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w /= ws
        
        # Weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]
        
        # Nesterov-like update
        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        
        c = fast_convolve(f_new)
        co = c.max()
        
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        f = f_new
        momentum = min(0.95, 0.88 + 0.07 * frac)
        
        if no_improve > 300:
            rng = np.random.default_rng(int(t * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            no_improve = 0
    
    return best_f, best_obj
