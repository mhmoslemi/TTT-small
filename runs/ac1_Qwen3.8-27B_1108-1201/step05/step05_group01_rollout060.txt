import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient(a, peaks, weights):
    """Compute subgradient of max_k conv(a,a)[k]."""
    n = len(a)
    sg = np.zeros(n)
    for k, w in zip(peaks, weights):
        if w < 1e-15:
            continue
        # d conv[k]/d a[m] = 2*a[k-m] for 0 <= k-m < n
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w * a[k - indices]
    return sg

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_subgrad(f0, time_limit, lr0=0.01, mom0=0.9, top_k=20, temp0=0.01):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find significant peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        
        # Softmax weights
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute subgradient
        sg = compute_subgradient(f, peaks, w)
        
        # Nesterov update
        v = cur_mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        
        # Nesterov momentum
        t_val = (1 + np.sqrt(1 + 4 * it * it)) / 2
        beta = (t_val - 1) / t_val
        y = f_new + beta * (f_new - f)
        y = project_simplex(y)
        f = f_new
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-scale Nesterov subgradient
with closed-form subgradients, LP-Frank-Wolfe polishing, window-function
initializations (U-shaped), and adaptive multi-peak weighting.
Exploits the structure d(conv[k])/da[m] = 2*a[k-m] for efficient gradients.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peaks, weights):
    """Compute subgradient of max_k conv(a,a)[k] using closed form."""
    n = len(a)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        w = weights[i]
        if w < 1e-15:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w * a[k - indices]
    return sg


def u_shaped_window(n, sharpness=2.0):
    """Generate a U-shaped (high at edges, low in middle) window."""
    j = np.arange(n)
    x = j / max(n - 1, 1)
    w = 1.0 + sharpness * np.abs(x - 0.5) ** 2
    w = np.maximum(w, 0.01)
    w /= w.sum()
    return w


def cosine_window(n, mode=0):
    """Generate cosine-modulated windows with different modes."""
    j = np.arange(n)
    if mode == 0:
        # U-shaped: high at edges
        w = 0.5 + 0.5 * np.cos(2 * np.pi * j / (n - 1))
    elif mode == 1:
        # Double U: two peaks
        w = 0.5 + 0.5 * np.cos(4 * np.pi * j / (n - 1))
    elif mode == 2:
        # Inverted Hanning (U-shaped)
        w = 0.5 + 0.5 * np.cos(2 * np.pi * j / (n - 1))
    elif mode == 3:
        # Parabolic U
        x = j / max(n - 1, 1)
        w = 1.0 + 3.0 * (x - 0.5) ** 2
    else:
        w = np.ones(n)
    w = np.maximum(w, 0.01)
    w /= w.sum()
    return w


def nesterov_subgrad(f0, time_limit, lr0=0.008, mom0=0.88, top_k=30, temp0=0.005):
    """Nesterov-accelerated projected subgradient with adaptive multi-peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = compute_subgradient(f, peaks, w)

        v = cur_mom * v - cur_lr * sg
        f_new = project_simplex(f + v)

        t_val = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        beta = (t_val - 1.0) / t_val
        y = f_new + beta * (f_new - f)
        y = project_simplex(y)
        f = f_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:

            best_f = f.copy()
            no_improve = 0
        else:

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j]*g[k-j] <= M
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(2 * n - 1, j + n)
        A_ub[start:end, j] = f[:end - start]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coordinate_descent(f0, time_limit, passes_max=50):
    """Block coordinate descent: optimize pairs of coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_pass_improve = 0

    for _ in range(passes_max):
        if time.monotonic() - t0 > time_limit:
            break
        pass_improved = False

        for j in range(0, n, max(1, n // 200)):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Golden section on this coordinate
            lo = 0.0
            hi = min(0.999, 3.0 * old_val)
            GR = (1.0 + np.sqrt(5.0)) / 2.0
            inv_gr = 1.0 / GR

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            def eval_coord(x):
                ft = f.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] *= (ft2 / denom)
                return fast_convolve(ft).max()

            f1 = eval_coord(c1)
            f2 = eval_coord(c2)
            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_coord(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = fast_convolve(f).max()

            if opt_val < cur_val - 1e-14:
                ft = f.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] *= (ft2 / denom)
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def spectral_init(n, num_peaks=3):
    """Initialize using spectral shaping: create a spectrum with controlled peaks."""
    # Create a spectrum that's mostly flat with some structure
    freqs = np.fft.rfftfreq(n)
    # Start with flat spectrum
    B = np.ones(len(freqs))
    # Add some harmonic structure
    for k in range(1, min(num_peaks + 1, len(freqs))):
        B[k] *= 0.5
    # Compute time domain
    a = np.fft.irfft(B * n, n)
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss < 1e-10:
        a = u_shaped_window(n)
    else:
        a /= ss
    return a

        rem = deadline - time.monotonic() - 300
        if rem > 30 and best_seq is not None:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 150), verbose=False)
            try_seq(a_opt, "lpfw_768")

    print(f"After Phase 3: best={best_score:.6f}")

    # === Phase 4: Scale up to n=1024 ===
    print(f"\n=== Phase 4: Scale to n=1024 (best={best_score:.6f}) ===")
    if best_seq is not None:
        n1024 = 1024
        f1024 = interpolate_sequence(best_seq, n1024)
        try_seq(f1024, "interp_1024")

        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.5, 200)
            a_opt, _ = nesterov_subgrad(f1024, time_limit=t_alloc,
                                         lr0=0.005, mom0=0.88, top_k=40, temp0=0.003)
            try_seq(a_opt, "ns_1024")

    # Also try direct init at 1024
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 150), verbose=False)
            try_seq(a_opt, "lpfw_1024")

    print(f"After Phase 4: best={best_score:.6f}")

    # === Phase 5: Scale up to n=1536 ===
    print(f"\n=== Phase 5: Scale to n=1536 (best={best_score:.6f}) ===")
    if best_seq is not None:
        n1536 = 1536
        f1536 = interpolate_sequence(best_seq, n1536)
        try_seq(f1536, "interp_1536")

        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.6, 180)
            a_opt, _ = nesterov_subgrad(f1536, time_limit=t_alloc,
                                         lr0=0.004, mom0=0.89, top_k=45, temp0=0.002)
            try_seq(a_opt, "ns_1536")

    print(f"After Phase 5: best={best_score:.6f}")

    # === Phase 6: Coordinate descent on best ===
    print(f"\n=== Phase 6: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_cd = min(rem * 0.7, 150)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 7: Final polishing ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Fine subgradient
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=min(rem, 80),
                                         lr0=0.002, mom0=0.93, top_k=max(20, n_best // 5),
                                         temp0=0.001)
            try_seq(a_opt, "final_ns")

        # Short LP-FW
        rem = deadline - time.monotonic() - 20
        if rem > 10 and n_best <= 768:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "final_lpfw")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        for _ in range(5):
            rem = deadline - time.monotonic() - 8
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = nesterov_subgrad(f_pert, time_limit=min(rem, 15),
                                         lr0=0.001, mom0=0.94,
                                         top_k=max(15, n_best // 6), temp0=0.0005)
            try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
