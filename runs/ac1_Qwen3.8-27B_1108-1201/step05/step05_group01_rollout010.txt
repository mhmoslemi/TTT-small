    n = len(v)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP-Frank-Wolfe,
spectral refinement, multi-length exploration, and Nesterov subgradient polishing.
Key insight: non-symmetric sequences reduce center auto-correlation peak below 1/n.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1)
    cond = u * rho_arr > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cosine_init(n, freq, amp, seed=0):
    """Cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def hann_init(n):
    """Hann window initialization."""
    j = np.arange(n)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * j / n))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def antisymmetric_init(n, strength=0.5):
    """Initialize with antisymmetric component to reduce center peak."""
    j = np.arange(n)
    # Linear ramp creates antisymmetric structure around center
    ramp = (j - n / 2 + 0.5) / (n / 2)
    a = 1.0 + strength * ramp
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def mixed_init(n, frac=0.5):
    """Mix of uniform and antisymmetric."""
    uni = np.ones(n) / n
    anti = antisymmetric_init(n, 0.6)
    a = frac * uni + (1 - frac) * anti
    return a / a.sum()


def solve_active_lp(f, M, n, top_k):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k lags, g >= 0."""
    conv = fast_convolve(f)
    k_actual = min(top_k, len(conv))
    active = np.argpartition(conv, -k_actual)[-k_actual:]

    c = -np.ones(n)
    A_ub = np.zeros((len(active), n))
    for i, k_val in enumerate(active):
        k_int = int(k_val)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        js = np.arange(lo, hi + 1)
        A_ub[i, js] = f[k_int - js]
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs > 1e-15:
            return g / gs
    return None


def solve_full_lp(f, M, n):
    """Solve LP with all constraints (fallback)."""
    c = -np.ones(n)
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n))
    for k in range(n_cons):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        A_ub[k, js] = f[k - js]
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs > 1e-15:
            return g / gs
    return None


def active_lp_fw(f0, time_limit, top_k=30, verbose=False):
    """Active-set LP Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_mult = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        # Adaptive top_k: start small, grow if stuck
        cur_top_k = min(n, top_k + no_improve // 5)

        g = solve_active_lp(f, M * m_mult, n, cur_top_k)
        if g is None:
            g = solve_full_lp(f, M * m_mult, n)
        if g is None:
            m_mult = min(m_mult * 1.15, 5.0)
            no_improve += 5
            if m_mult > 4.0:
                m_mult = 1.0
            continue

        m_mult = 1.0

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 60:
                break
            continue

        # Line search with golden section refinement
        best_alpha = 0.0
        best_val = M

        # Candidate alphas
        fw_alpha = 2.0 / (it + 2.0)
        candidates = sorted(set([0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0, fw_alpha]))

        for alpha in candidates:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Golden section refinement
        if 0 < best_alpha < 1.0:
            lo = max(0.0, best_alpha - 0.2)
            hi = min(1.0, best_alpha + 0.2)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)

            def _eval_alpha(a_val):
                ft = f + a_val * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    return float('inf')
                ft = ft / st
                return fast_convolve(ft).max()

            f1 = _eval_alpha(c1)
            f2 = _eval_alpha(c2)
            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _eval_alpha(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = _eval_alpha(c2)
            mid = (lo + hi) / 2.0
            fm = _eval_alpha(mid)
            if fm < best_val:
                best_val = fm
                best_alpha = mid

        if best_alpha > 1e-12:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Perturbation restart
        if no_improve > 100:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.004 * min(no_improve // 100, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def nesterov_subgrad(f0, time_limit, lr0=0.005, mom0=0.85, top_k=50, temp0=0.0001):
    """Nesterov-accelerated subgradient with multi-peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    y = f.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        # Compute subgradient from top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute gradient via FFT (efficient)
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = rfft(f, fft_size)
        # Gradient of conv[k] wrt a: g[m] = 2 * sum_i w[i] * f[k_i - m] for valid k_i - m
        # Use FFT: gradient = 2 * ifft(A * conj(rfft(w_extended)))... 
        # Actually let's just do it in time domain for the peaks
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                idx_arr = np.arange(m_lo, m_hi + 1)
                sg[idx_arr] += 2.0 * w[idx] * f[k - idx_arr]

        # Nesterov momentum update
        v = cur_mom * v - cur_lr * sg
        y = y + v
        y = project_simplex(y)
        f_new = project_simplex(y)
        y = f_new + cur_mom * (f_new - f)
        y = project_simplex(y)
        f = f_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        cur_lr *= 0.9997
        cur_temp *= 1.003
        cur_mom = min(0.98, cur_mom * 1.0002)

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.7, 0.001)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def spectral_refine(f0, time_limit, n_steps=200):
    """Iterative spectral equalization to flatten auto-correlation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    gamma = 0.3

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        # Compute the "excess" spectrum
        target = 1.0 / (2 * n - 1)
        excess = conv - target
        # In frequency domain, modifying the power spectrum
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = rfft(f, fft_size)
        # conv ~ ifft(|A|^2), so gradient of max(conv) in freq domain
        # Approximate: reduce |A|^2 where it contributes to peaks
        # Use the excess as a guide
        excess_fft = rfft(np.concatenate([excess, np.zeros(fft_size - len(excess))]), fft_size)
        # Adjust: A_new = A * (1 - gamma * excess_fft / |A|^2) ... complex
        # Simpler: multiplicative update in time domain
        # f_new[j] proportional to f[j] * exp(-gamma * gradient_j)
        # gradient of max(conv) at point j is the subgradient
        k_actual = min(30, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        max_pv = pv.max()
        w = np.exp((pv - max_pv) * 5.0)
        w /= w.sum()

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-10:
                continue
            k = int(peaks[idx])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                idx_arr = np.arange(m_lo, m_hi + 1)
                sg[idx_arr] += 2.0 * w[idx] * f[k - idx_arr]

        # Multiplicative update (like matrix scaling)
        f_new = f * np.exp(-gamma * sg / max(sg.max(), 1e-10))
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()

        new_conv = fast_convolve(f_new)
        new_obj = new_conv.max()
        if new_obj < best_obj:
            best_obj = new_obj
            best_f = f_new.copy()
            f = f_new
            gamma *= 0.98
        else:
            gamma *= 0.85
            f = f  # keep old
            if gamma < 0.01:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(budget_s=1800, **kwargs):
    """

    Nesterov subgradient, multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return float('inf')
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
        if p.ndim == 1 and p.size > 0 and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        try_seq(parent, "uniform_512")

    # === Phase 1: LP-FW on parent ===
    print("\n=== Phase 1: Active LP-FW on parent ===")
    n_par = len(parent)
    t1 = min(200, deadline - time.monotonic() - 1000)
    if t1 > 30:
        a_opt, _ = active_lp_fw(parent, time_limit=t1, top_k=n_par // 8, verbose=True)
        try_seq(a_opt, f"lpfw_parent_n{n_par}")

    # === Phase 2: Multiple initializations ===
    print("\n=== Phase 2: Multiple initializations ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    inits = []
    inits.append(("antisym", antisymmetric_init(n_base, 0.4)))
    inits.append(("hann", hann_init(n_base)))
    inits.append(("anti0.6", antisymmetric_init(n_base, 0.6)))
    inits.append(("mix", mixed_init(n_base, 0.5)))
    inits.append(("cos1", cosine_init(n_base, 1, 0.3)))
    inits.append(("cos2", cosine_init(n_base, 2, 0.25)))
    inits.append(("cos3", cosine_init(n_base, 3, 0.2)))

    for name, a_init in inits:
        rem = deadline - time.monotonic() - 600
        if rem < 20:
            break
        s = try_seq(a_init, f"init_{name}")
        if np.isfinite(s) and s < best_score + 0.3:
            t_alloc = min(rem * 0.15, 60)
            if t_alloc > 10:
                a_opt, _ = active_lp_fw(a_init, time_limit=t_alloc, top_k=n_base // 10)
                try_seq(a_opt, f"lpfw_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = [512, 768, 1024, 1536]
        for tn in target_lengths:
            rem = deadline - time.monotonic() - 400
            if rem < 30:
                break
            if len(best_seq) == tn:
                continue
            f_interp = interpolate_sequence(best_seq, tn)
            s0 = try_seq(f_interp, f"interp_n{tn}")
            if np.isfinite(s0):
                t_alloc = min(rem * 0.3, 120)
                if t_alloc > 15:
                    print(f"  LP-FW n={tn}, t={t_alloc:.0f}s")
                    a_opt, _ = active_lp_fw(f_interp, time_limit=t_alloc,
                                             top_k=max(20, tn // 10), verbose=(tn == 768))
                    try_seq(a_opt, f"lpfw_n{tn}")

    # === Phase 4: Nesterov subgradient polish ===
    print(f"\n=== Phase 4: Nesterov polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_ns = min(rem * 0.4, 150)
            print(f"  Nesterov t={t_ns:.0f}s, n={n_best}")
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_ns,
                                         lr0=0.006, mom0=0.87,
                                         top_k=max(30, n_best // 6), temp0=0.00005)
            try_seq(a_opt, "nesterov1")

            rem = deadline - time.monotonic() - 120
            if rem > 15:
                a_opt2, _ = nesterov_subgrad(a_opt, time_limit=min(rem, 80),
                                              lr0=0.002, mom0=0.92,
                                              top_k=max(20, n_best // 8), temp0=0.00002)
                try_seq(a_opt2, "nesterov2")

    # === Phase 5: Spectral refinement ===
    print(f"\n=== Phase 5: Spectral refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            a_opt, _ = spectral_refine(best_seq, time_limit=min(rem, 80))
            try_seq(a_opt, "spectral")

    # === Phase 6: Final multi-start ===
    print(f"\n=== Phase 6: Final multi-start (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(12345)

        starts = [best_seq.copy()]
        starts.append(project_simplex(best_seq + rng.standard_normal(n_best) * 0.004))
        starts.append(0.8 * best_seq + 0.2 * antisymmetric_init(n_best, 0.5))
        starts[-1] = project_simplex(starts[-1])
        starts.append(0.85 * best_seq + 0.15 * hann_init(n_best))
        starts[-1] = project_simplex(starts[-1])
        starts.append(project_simplex(best_seq[::-1] * 0.3 + best_seq * 0.7))

        for i, f_init in enumerate(starts):
            rem = deadline - time.monotonic() - 50
            if rem < 15:
                break
            t_alloc = min(rem * 0.35, 60)
            a_opt, _ = active_lp_fw(f_init, time_limit=t_alloc, top_k=n_best // 8)
            try_seq(a_opt, f"final_{i}")

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Micro-polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=min(rem, 40),
                                         lr0=0.001, mom0=0.94,
                                         top_k=max(15, len(best_seq) // 10), temp0=0.00001)
            try_seq(a_opt, "micro")

        rem = deadline - time.monotonic() - 8
        if rem > 8:
            a_opt, _ = active_lp_fw(best_seq, time_limit=min(rem, 30), top_k=len(best_seq) // 12)
            try_seq(a_opt, "micro_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
