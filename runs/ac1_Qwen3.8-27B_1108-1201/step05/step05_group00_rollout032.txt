   The constraints are quadratic (bilinear). This is NOT a linear feasibility problem. However, if we fix all but one variable, it becomes a quadratic constraint in that variable, which is tractable.
   
   This suggests a **sequential quadratic programming** or **coordinate descent** approach:
   - Initialize a
   - For each coordinate i:
     - Find the optimal value of a[i] given all other coordinates fixed
     - The constraint (a*a)[k] <= t for all k gives quadratic constraints on a[i]
     - Also need a[i] >= 0 and the sum constraint

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')

        fg = np.convolve(f, g)
        
        violations = fg - M

def full_lp_oracle(f, M):
    """Full LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=b_ub if b_ub is not A_ub else b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def logsumexp_grad_descent(a0, time_limit, T_init=0.001, T_end=0.0001, lr_init=0.02):
    """
    Gradient descent using log-sum-exp approximation of max.
    phi_T(a) = T * log(sum_k exp((a*a)[k] / T))
    gradient = (1/T) * sum_k exp((a*a)[k]/T) * d(a*a)[k]/da
             = sum_k p_k * 2 * (shift_a)_k
    where p = softmax(conv/T) and (shift_a)_k is the relevant shift.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    total_iters_target = max(100, int(time_limit / 0.01))

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient of smooth approximation: 2 * hankel(p, a)
        grad = 2.0 * hankel_mv(p, a)

        # Momentum + Nesterov-like
        v = 0.9 * v - lr * grad
        candidate = project_simplex(a + v)

        # Check improvement
        c = fast_convolve(candidate)
        val = c.max()

        if val < best_obj - 1e-14:
            best_obj = val
            best_a = candidate.copy()
            no_improve = 0
            a = candidate
        else:
            a = candidate  # Still accept for exploration
            no_improve += 1

        # Adaptive temperature
        progress = min(1.0, it / total_iters_target)
        T = T_init * (T_end / T_init) ** progress

        if no_improve > 50:
            lr *= 0.95

            rng = np.random.default_rng((it * 31 + 7) % (2**31))

            v = np.zeros(n)
            lr = lr_init * 0.5
            T = T_init
            no_improve = 0

    return best_a, best_obj


def pairwise_descent(a0, time_limit, window=5):
    """
    Pairwise coordinate descent: for each pair (i,j) in a local window,
    try to transfer mass from one to the other.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Try all pairs in a random window
        improved = False
        rng = np.random.default_rng(it * 1000003 + 7)
        
        # Try multiple random pairs per sweep
        for _ in range(min(200, n // 4)):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < 1e-10 or a[j] < 1e-10:
                continue
            
            # Try transferring small amounts from i to j and vice versa
            for frac in [0.01, 0.05, 0.1, 0.2]:
                # Transfer from i to j
                delta = frac * a[i]
                a[j] += delta
                a[i] -= delta
                val = fast_convolve(a).max()
                if val < best_obj - 1e-15:
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break
                # Undo
                a[j] -= delta
                a[i] += delta
            
            if improved:
                break
        
        if not improved and it % 100 == 0:
            # Try larger transfers
            for _ in range(50):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                for frac in [0.3, 0.5, 0.7]:
                    delta = frac * a[i]
                    if a[i] - delta < -1e-15:
                        continue
                    a[j] += delta
                    a[i] -= delta
                    val = fast_convolve(a).max()
                    if val < best_obj - 1e-15:
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
                    a[j] -= delta
                    a[i] += delta
                if improved:
                    break
        
        if not improved and it > 500:
            break

    return best_a, best_obj


def spectral_init(n, rng):
    """
    Generate initial sequence with low autocorrelation using spectral methods.
    Based on the idea of choosing a random phase in the frequency domain
    and taking the real part.
    """
    # Random phasor approach
    phases = rng.uniform(0, 2 * np.pi, n)
    a = np.cos(phases)  # Real part of e^{i*phases}
    a = np.maximum(a, 0)  # Non-negativity
    s = a.sum()
    if s < 1e-10:
        a = rng.dirichlet(np.ones(n) * 0.5)
    else:
        a = a / s
    return a


def structured_init(n, pattern):
    """Generate structured initializations."""
    x = np.arange(n, dtype=np.float64)
    
    if pattern == 'uniform':
        return np.ones(n) / n
    elif pattern == 'bilateral':
        # Peaks at edges
        a = np.zeros(n)
        a[0] = a[n-1] = 1.0
        a = np.maximum(a + 0.1, 0.01)
        return a / a.sum()
    elif pattern == 'alternating':
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.3)
        return a / a.sum()
    elif pattern == 'gaussian':
        mu = n / 2
        sigma = n / 4
        a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        return a / a.sum()
    elif pattern == 'two_peaks':
        mu1, mu2 = n * 0.25, n * 0.75
        sigma = n * 0.1
        a = np.exp(-0.5 * ((x - mu1) / sigma) ** 2) + np.exp(-0.5 * ((x - mu2) / sigma) ** 2)
        return a / a.sum()
    elif pattern == 'random':
        return np.random.default_rng(42).dirichlet(np.ones(n) * 0.5)
    else:
        return np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Active-set FW with LP oracle for fast global optimization
- Log-sum-exp smooth gradient descent with momentum for refinement
- Pairwise coordinate descent for local polish
- Multi-n exploration with structured initializations

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]. p has length 2n-1, a has length n."""
    return np.correlate(p, a, mode='valid')


def active_set_lp(f, M, max_active=40, tol=1e-8):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for active constraints, g >= 0.
    Uses active set: start with top constraints, add violated ones.
    """
    n = len(f)
    num_constraints = 2 * n - 1

    # Initial active set: top constraints by upper bound
    # Upper bound on (f*g)[k] is f.sum() * max(f) for any g on simplex
    # Better: use the positions where f*f peaks (proxy for where f*g will peak)
    ff = fast_convolve(f)
    top_indices = np.argsort(ff)[::-1][:max_active]
    active = set(int(k) for k in top_indices[:min(10, max_active)])

    g = None
    for iteration in range(30):
        active_list = sorted(active)
        m = len(active_list)

        # Build constraint matrix
        A_ub = np.zeros((m, n))
        for i, k in enumerate(active_list):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                A_ub[i, :] = 0
            else:
                # (f*g)[k] = sum_{j=js}^{je} f[k-j] * g[j]
                A_ub[i, js:je+1] = f[k - np.arange(js, je + 1)][::-1]

        c = -np.ones(n)
        b_ub = np.full(m, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
            if res.success:
                g = np.maximum(res.x, 0.0)
            else:
                return None
        except Exception:
            return None

        if g is None or g.sum() < 1e-15:
            return None

        # Check for violated constraints
        fg = np.convolve(f, g)
        violations = fg - M
        worst_idx = int(np.argmax(violations))

        if violations[worst_idx] < tol * max(M, 1e-10):
            break  # Feasibility satisfied

        active.add(worst_idx)

        if len(active) > max_active:
            # Keep only the most violated
            active = set(int(k) for k in np.argsort(fg)[::-1][:max_active])

    return g


def full_lp_oracle(f, M, time_limit=5.0):
    """Full LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def smooth_grad_descent(a0, time_limit, T_init=0.001, T_end=0.0001, lr_init=0.02, mom=0.9):
    """
    Momentum-based gradient descent using log-sum-exp approximation of max.
    phi_T(a) = T * log(sum_k exp((a*a)[k] / T))
    gradient = 2 * hankel(softmax(conv/T), a)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    total_iters_est = max(100, int(time_limit * 150))

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        conv = fast_convolve(a)

        # Softmax weights (stable)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient
        grad = 2.0 * hankel_mv(p, a)

        # Momentum update
        v = mom * v - lr * grad
        candidate = project_simplex(a + v)

        val = fast_convolve(candidate).max()

        if val < best_obj - 1e-14:
            best_obj = val
            best_a = candidate.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = candidate

        # Adaptive temperature (exponential decay)
        progress = min(1.0, it / total_iters_est)
        T = T_init * (T_end / T_init) ** progress

        # Adaptive learning rate
        if no_improve > 30:
            lr *= 0.97
        if no_improve > 80:
            lr *= 0.9
            mom = min(0.95, mom + 0.005)
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            scale = 0.002 + 0.003 * min(1.0, (no_improve - 200) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = lr_init * 0.5
            T = T_init
            no_improve = 0

    return best_a, best_obj


def pairwise_descent(a0, time_limit, evals_per_s=5000):
    """
    Efficient pairwise coordinate descent.
    Tries to transfer mass between pairs of coordinates.
    Uses batch evaluation for speed.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

    it = 0

    active_indices = np.where(a > 1e-8)[0]

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        improved_this_round = False

        # Try random pairs
        num_pairs = min(200, max(50, n // 5))
        for _ in range(num_pairs):
            idx = rng.integers(len(active_indices), size=2)

            j = int(active_indices[idx[1]])
            if i == j or a[i] < 1e-10 or a[j] < 1e-10:
                continue

            # Try several transfer amounts
            for frac in [0.02, 0.05, 0.1, 0.15]:
                delta = frac * a[i]
                if delta < 1e-12:
                    continue
                a[i] -= delta
                a[j] += delta
                val = fast_convolve(a).max()
                if val < best_obj - 1e-15:
                    best_obj = val
                    best_a = a.copy()
                    improved_this_round = True
                    break
                # Undo
                a[i] += delta
                a[j] -= delta

            if improved_this_round:
                break

        if not improved_this_round:
            # Try moving mass to/from zero coordinates
            zero_indices = np.where(a < 1e-10)[0]
            if len(zero_indices) > 0:
                for _ in range(50):
                    i = int(rng.integers(len(active_indices)))
                    i = int(active_indices[i])
                    if a[i] < 1e-10:
                        continue
                    j = int(zero_indices[rng.integers(len(zero_indices))])
                    for frac in [0.01, 0.05, 0.1]:

                        a[i] -= delta
                        a[j] += delta

                        if val < best_obj - 1e-15:
                            best_obj = val
                            best_a = a.copy()
                            improved_this_round = True
                            break
                        a[i] += delta
                        a[j] -= delta
                    if improved_this_round:
                        break

        # Update active indices periodically
        if it % 50 == 0:
            active_indices = np.where(a > 1e-8)[0]
            if len(active_indices) == 0:
                break

        # Stopping criterion
        if not improved_this_round and it > 300:
            break

    best_a = best_a / best_a.sum()
    return best_a, best_obj


def cma_peak_reduction(a0, time_limit, target_frac=0.95):
    """
    Contour Matching Algorithm for peak reduction.
    Iteratively identifies the worst peak and adjusts the sequence to reduce it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    target = target_frac * best_obj  # Try to get below 95% of current max

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(54321)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        if peak_val < target:
            # Update target and continue
            target = 0.95 * peak_val
            best_obj = peak_val
            best_a = a.copy()
            continue

        # Find direction to reduce peak at k_peak
        # The subgradient at the peak w.r.t. a is: 2 * a shifted
        # (a*a)[k_peak] = sum_j a[j] * a[k_peak - j]
        # d/d a[i] = a[k_peak - i] + (if k_peak + i < 2n) a[k_peak + i]... 
        # Actually: d(a*a)[k]/da[i] = a[k-i] + a[k+i] (with boundary handling)
        # More precisely: d(a*a)[k]/da[i] = sum over valid j where j or k-j equals i
        
        # The subgradient of max at the unique maximizer is 2 * (the shifted version)
        # For (a*a)[k_peak] = sum_j a[j] a[k_peak - j]:
        # d/da[i] = a[k_peak - i] (from j=i term) + a[i] (from k-j=i, i.e., k_peak-j=i, j=k_peak-i)
        # But we need to be careful about double counting when j = k_peak - j
        
        # Simpler: just use the FW direction toward reducing this specific peak
        # Find g that minimizes (f*g)[k_peak] subject to sum(g)=1, g>=0
        # This is: g should put all mass where f[k_peak - j] is smallest
        
        # Actually, let's just do a simple heuristic: reduce a[i] where a[i]*a[k_peak-i] is large
        # and increase a[j] where a[j]*a[k_peak-j] is small
        
        # Compute the "contribution" of each pair
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        
        if js > je:
            break
        
        # For the peak at k_peak, the terms are a[j]*a[k_peak-j] for j in [js, je]
        contributions = np.zeros(je - js + 1)
        for idx, j in enumerate(range(js, je + 1)):
            contributions[idx] = a[j] * a[k_peak - j]
        
        # Move mass from high-contribution indices to low-contribution indices
        top_idx = np.argmax(contributions)
        bot_idx = np.argmin(contributions)
        
        j_from = js + int(top_idx)
        j_to = js + int(bot_idx)
        
        if j_from == j_to:
            # Can't do much
            a = project_simplex(a + rng.standard_normal(n) * 0.001)
            continue
        
        # Transfer mass
        for frac in [0.01, 0.03, 0.05, 0.1]:
            delta = frac * a[j_from]
            if delta < 1e-12:
                continue
            a[j_from] -= delta
            a[j_to] += delta
            new_peak = fast_convolve(a)[k_peak]
            overall_max = fast_convolve(a).max()
            if overall_max < best_obj - 1e-15:
                best_obj = overall_max
                best_a = a.copy()
                break
            a[j_from] += delta
            a[j_to] -= delta

        # If stuck, add noise
        if it % 100 == 0:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)

    return best_a, best_obj

            a_opt2, _ = fw_active_set(a_opt, time_limit=t_fw, max_active=30)
            try_seq(a_opt2, f"fw_n{n_try}")
            results_by_n[n_try] = (a_opt2, results_by_n[n_try][1])

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 2: Deep FW-LP (~500s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        
        # Main deep FW
        remaining = deadline - time.monotonic() - 600
        if remaining > 30:
            t_alloc = min(remaining, 400)
            print(f"  FW active set n={n_best}, t={t_alloc:.0f}s, max_active=50")
            a_opt, obj = fw_active_set(best_seq, time_limit=t_alloc, max_active=50)
            try_seq(a_opt, "fw_deep_50")

        # Try with larger active set
        remaining = deadline - time.monotonic() - 400
        if remaining > 20:
            t_alloc = min(remaining, 200)
            print(f"  FW active set n={n_best}, t={t_alloc:.0f}s, max_active=100")
            a_opt2, _ = fw_active_set(a_opt, time_limit=t_alloc, max_active=100)
            try_seq(a_opt2, "fw_deep_100")

        # Full LP (slower but potentially better)
        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  FW full LP n={n_best}, t={t_alloc:.0f}s")
            a_opt3, _ = fw_active_set(a_opt2, time_limit=t_alloc, max_active=10, use_full_lp=True)
            try_seq(a_opt3, "fw_full_lp")

        # Perturbed restarts
        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            for scale in [0.003, 0.01, 0.03]:
                remaining = deadline - time.monotonic() - 150
                if remaining < 15:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
                t_alloc = min(remaining, 60)
                a_opt4, _ = fw_active_set(f_pert, time_limit=t_alloc, max_active=50)
                try_seq(a_opt4, f"fw_pert_{scale}")

    # Also try best n from phase 1 if different
    if best_seq is not None and results_by_n:
        best_n_from_p1 = max(results_by_n.keys(), key=lambda k: -results_by_n[k][1])
        # Actually pick the one with best score
        best_n_from_p1 = min(results_by_n.keys(), key=lambda k: results_by_n[k][1] if np.isfinite(results_by_n[k][1]) else 999)
        if best_n_from_p1 != len(best_seq):
            remaining = deadline - time.monotonic() - 200
            if remaining > 20:
                a_p1 = results_by_n[best_n_from_p1][0]
                t_alloc = min(remaining, 80)
                print(f"  FW on best phase-1 n={best_n_from_p1}, t={t_alloc:.0f}s")
                a_opt5, _ = fw_active_set(a_p1, time_limit=t_alloc, max_active=50)
                try_seq(a_opt5, "fw_best_p1_n")

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 3: Smooth gradient refinement (~300s) ===
    print("\n=== Phase 3: Smooth Gradient Refinement ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple temperature schedules
        configs = [
            {'T_init': 0.0005, 'T_end': 0.00005, 'lr_init': 0.015, 'mom': 0.9},
            {'T_init': 0.001, 'T_end': 0.0001, 'lr_init': 0.02, 'mom': 0.92},
            {'T_init': 0.003, 'T_end': 0.0003, 'lr_init': 0.01, 'mom': 0.88},
            {'T_init': 0.0001, 'T_end': 0.00001, 'lr_init': 0.005, 'mom': 0.95},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  Smooth config {ci} (T_init={cfg['T_init']}), t={t_alloc:.0f}s")
            a_opt, _ = smooth_grad_descent(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"smooth_ref{ci}")

        # Multi-start
        remaining = deadline - time.monotonic() - 80
        if remaining > 20:
            print(f"  Multi-start, t={min(remaining, 60):.0f}s")
            a_opt, _ = multi_start_smooth(best_seq, n_starts=4,
                                         time_per_start=min(remaining, 60) / 4,
                                         T_values=[0.0005, 0.001, 0.002, 0.0003],
                                         rng=rng)
            try_seq(a_opt, "multi_start")

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 4: CMA peak reduction (~150s) ===
    print("\n=== Phase 4: CMA Peak Reduction ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 60
        if remaining > 10:
            t_alloc = min(remaining, 120)
            print(f"  CMA, t={t_alloc:.0f}s")
            a_opt, _ = cma_peak_reduction(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cma")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 5: Pairwise descent (~150s) ===
    print("\n=== Phase 5: Pairwise Descent ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 40
        if remaining > 10:
            t_alloc = min(remaining, 120)
            print(f"  Pairwise, t={t_alloc:.0f}s")
            a_opt, _ = pairwise_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "pairwise")

            # Another round with the result
            remaining = deadline - time.monotonic() - 20
            if remaining > 10 and best_seq is not None:
                a_opt2, _ = pairwise_descent(best_seq, time_limit=min(remaining, 60))
                try_seq(a_opt2, "pairwise2")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # === Phase 6: Final micro-optimization (~50s) ===
    print("\n=== Phase 6: Final Micro-Optimization ===")
    t_phase_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine smooth gradient
        remaining = deadline - time.monotonic() - 15
        if remaining > 5:
            a_opt, _ = smooth_grad_descent(best_seq, time_limit=min(remaining, 25),
                                         T_init=0.0001, T_end=0.00001, lr_init=0.005, mom=0.95)
            try_seq(a_opt, "micro_smooth")

        # Quick FW
        remaining = deadline - time.monotonic() - 10
        if remaining > 3:
            a_opt2, _ = fw_active_set(best_seq, time_limit=min(remaining, 15), max_active=60)
            try_seq(a_opt2, "micro_fw")

        # Tiny perturbations with quick polish
        remaining = deadline - time.monotonic() - 3
        if remaining > 3:
            for _ in range(4):
                remaining = deadline - time.monotonic() - 1
                if remaining < 2:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                a_opt3, _ = smooth_grad_descent(f_pert, time_limit=min(remaining, 3),
                                             T_init=0.0002, T_end=0.0001, lr_init=0.003, mom=0.9)
                try_seq(a_opt3, "micro_pert")

    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
