        top_k = min(5, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in peak_positions:
            peak_k = int(pk)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    return best_f, best_obj

                old_val = f[j]
                if old_val < 1e-13:
                    continue

                best_j_val = old_val
                best_j_obj = best_obj
                multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.35, 1.6, 2.0]

                for mult in multipliers:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.8:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()

                    val = fast_convolve(f_trial).max()
                    if val < best_j_obj - 1e-16:
                        best_j_obj = val
                        best_j_val = new_val

                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        f[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()

                    if best_j_obj < best_obj - 1e-15:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)

        if not improved:
            no_improve += 1
            if no_improve > 2:
                rng = np.random.default_rng(passes * 104729)
                n_random = min(30, n // 5)
                for _ in range(n_random):
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.4, 0.6, 1.5, 2.0]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.8:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-15:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break
                    if improved:
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_f, best_obj


def simulated_annealing(f0, time_limit, T0=0.005, T_min=0.0001, cool_rate=0.9995,
                        n_iter_per_step=1, verbose=False):
    """Simulated annealing on the probability simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    cur_obj = conv.max()
    best_obj = cur_obj
    best_f = f.copy()

    T = T0
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and T > T_min:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        for _ in range(n_iter_per_step):
            # Pick 1-3 random coordinates to perturb
            n_perturb = np.random.randint(1, 4)
            indices = np.random.choice(n, size=min(n_perturb, n), replace=False)

            f_trial = f.copy()
            for idx in indices:
                scale = np.random.uniform(0.5, 2.0)
                f_trial[idx] *= scale

            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st

            val = fast_convolve(f_trial).max()
            delta = val - cur_obj

            if delta < 0 or np.random.random() < np.exp(-delta / T):
                f = f_trial
                cur_obj = val
                if val < best_obj:
                    best_obj = val
                    best_f = f.copy()
                    if verbose and it % 500 == 0:
                        print(f"    SA iter {it}: best={best_obj:.10f}, T={T:.6f}")

        T *= cool_rate

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP-FW, subgradient, coordinate descent,
    simulated annealing, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)
    print(f"Starting optimization with n={base_n}, initial score={best_score:.8f}")

    # === Phase 1: LP-FW on parent (primary optimizer) ===
    print("\n=== Phase 1: LP-FW (standard) ===")
    rem = deadline - time.monotonic()
    t1 = min(300, rem * 0.25)
    if t1 > 30:
        print(f"  n={base_n}, t={t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_p1")

    # === Phase 2: LP-FW with momentum ===
    print("\n=== Phase 2: LP-FW (momentum) ===")
    rem = deadline - time.monotonic()
    t2 = min(200, rem * 0.18)
    if t2 > 25 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, use_momentum=True,
                                  momentum_beta=0.6)
        try_seq(a_opt, "lpfw_mom")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_sg = min(rem * 0.2, 250)
        if t_sg > 20:
            print(f"  n={n_best}, t={t_sg:.0f}s")
            a_opt, _ = sg_optimize(best_seq, time_limit=t_sg,
                                   lr=0.004, momentum=0.94,
                                   top_k=max(30, n_best // 6),
                                   temp=0.00005, verbose=True)
            try_seq(a_opt, "sg_main")

            # Second SG run with different params
            rem = deadline - time.monotonic()
            t_sg2 = min(rem * 0.15, 150)
            if t_sg2 > 15:
                a_opt, _ = sg_optimize(best_seq, time_limit=t_sg2,
                                       lr=0.001, momentum=0.96,
                                       top_k=max(50, n_best // 4),
                                       temp=0.00002)
                try_seq(a_opt, "sg_fine")

    # === Phase 4: Multi-length LP-FW ===
    print(f"\n=== Phase 4: Multi-length LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        candidate_lengths = []
        for delta in [-64, -32, 32, 64, 128]:
            ln = len(base) + delta
            if ln >= 256 and ln <= 2048:
                candidate_lengths.append(ln)
        # Also try some "round" lengths
        for ln in [512, 640, 768, 896, 1024, 1280]:
            if ln not in candidate_lengths and 256 <= ln <= 2048:
                candidate_lengths.append(ln)

        candidate_lengths = sorted(set(candidate_lengths))[:5]

        for target_n in candidate_lengths:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.25, 150)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 5: Peak-targeting coordinate descent ===
    print(f"\n=== Phase 5: Peak CD (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.25, 250)
        if t_cd > 20:
            print(f"  n={len(best_seq)}, t={t_cd:.0f}s")
            a_opt, _ = peak_targeting_cd(best_seq, time_limit=t_cd)
            try_seq(a_opt, "peak_cd")

    # === Phase 6: Simulated annealing for global exploration ===
    print(f"\n=== Phase 6: Simulated Annealing (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_sa = min(rem * 0.2, 150)
        if t_sa > 20:
            print(f"  n={len(best_seq)}, t={t_sa:.0f}s")
            a_opt, _ = simulated_annealing(best_seq, time_limit=t_sa,
                                           T0=0.003, T_min=0.00005,
                                           cool_rate=0.9997, verbose=True)
            try_seq(a_opt, "sa")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Quick LP-FW
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "lpfw_polish")

        # Quick SG
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = sg_optimize(best_seq, time_limit=min(rem, 40),
                                   lr=0.0008, momentum=0.96,
                                   top_k=max(20, n_best // 10),
                                   temp=0.00001)
            try_seq(a_opt, "sg_polish")

        # Short CD
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = peak_targeting_cd(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "cd_polish")

        # Perturbation restarts
        for p in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.0008 + 0.0004 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=min(rem, 25))
            try_seq(a_opt, f"restart_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = np.ones(512) / 512
            best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    elapsed = time.monotonic() - (deadline - float(budget_s) + 20.0)
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) [elapsed: {elapsed:.0f}s]")
    return [float(x) for x in best_seq]
