"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using multi-phase optimization:
1. LP-based Frank-Wolfe with direct g-evaluation
2. Smooth penalty gradient descent
3. Multi-length exploration
4. Coordinate descent polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_objective(f):
    """Compute the objective 2n*max(c)/S^2 for sequence f."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return float('inf')
    c = fast_convolve(f)
    return 2.0 * n * c.max() / (S * S)


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0

    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_cons)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_max_sum(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c_vec = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, evaluate_g=True):
    """LP-based Frank-Wolfe with option to evaluate g directly."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M  # with sum=1
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if it % 100 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 2:
                break

        it += 1

        g, lp_t = solve_lp_min_peak(f, n)
        if g is None:
            g = solve_lp_max_sum(f, M * 1.1, n)
            if g is not None:
                gs = g.sum()
                g = g / gs if gs > 1e-15 else np.ones(n) / n
            else:
                g = np.ones(n) / n
        else:
            gs = g.sum()
            g = g / gs if gs > 1e-15 else np.ones(n) / n

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Evaluate g directly (its own auto-convolution peak)
        best_alpha = 0.0
        best_val = 2.0 * n * M

        if evaluate_g:
            g_conv = fast_convolve(g)
            g_peak = g_conv.max()
            g_obj = 2.0 * n * g_peak
            if g_obj < best_val:
                best_val = g_obj
                best_alpha = 1.0

        # Line search over interpolation
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = 2.0 * n * fast_convolve(ft).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha

        # Refine with bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = 2.0 * n * fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-13:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = fast_convolve(f).max()

            if 2.0 * n * M < best_obj - 1e-15:
                best_obj = 2.0 * n * M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Restart when stuck
        if no_improve > 30:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.002 * min(no_improve // 20, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, best_obj

            # Project gradient onto the simplex tangent space
            grad_mean = grad.mean()
            grad_centered = grad - grad_mean  # tangent to sum(a)=1
            
            f_new = project_simplex(f + lr * grad_centered)
        else:
            # All below target, try to reduce the peak further
            target *= 0.999
            f_new = f
        
        # Check objective
        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Penalty iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        f = f_new
        lr *= decay
        
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            no_improve = 0
            lr = 0.01
            target = fast_convolve(f).max() * 0.995

    return best_f, best_obj
