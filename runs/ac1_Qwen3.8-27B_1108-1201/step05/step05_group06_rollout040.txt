"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative weights
over autocorrelation positions combined with Nesterov-accelerated projected
gradient descent, exact coordinate bisection polishing, and multi-length search.

import numpy as np
import time
from numpy.lib.stride_tricks import sliding_window_view


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, a):
    """Compute Q_p @ a where Q_p[i,j] = p[i+j]."""
    n = len(a)
    # (Q_p @ a)[i] = sum_j p[i+j] * a[j]
    windows = sliding_window_view(p, n)  # shape (n, n)
    return windows @ a


def mw_nesterov(a0, time_limit, lr=0.5, mom=0.9, eta=0.5, verbose=False):
    """Multiplicative weights + Nesterov projected gradient."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a /= s
    n = len(a)
    
    p = np.ones(2 * n - 1) / (2 * n - 1)
    v = np.zeros(n)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient
        Qa = hankel_matvec(p, a)
        grad = 2.0 * Qa
        
        # Nesterov step
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)
        
        # Evaluate
        c = fast_convolve(a_new)
        obj = c.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Update p (multiplicative weights)
        # Use log-space for numerical stability
        log_p = np.log(p + 1e-300) + eta * c
        log_p -= log_p.max()
        p = np.exp(log_p)
        ps = p.sum()
        if ps < 1e-300:
            p = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            p /= ps
        
        # Adaptive parameters
        if no_improve > 500:
            # Restart with noise
            rng = np.random.default_rng(it * 7 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            p = np.ones(2 * n - 1) / (2 * n - 1)
            lr = 0.3
            no_improve = 0
        
        # Decay learning rate
        lr *= 0.999995
        eta *= 1.00005  # increase concentration on peaks
        
        if verbose and it % 10000 == 0:
            print(f"    MW iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, eta={eta:.4f}")
    
    return best_a, best_obj


def coordinate_bisection(a0, time_limit, verbose=False, max_passes=200):
    """Exact coordinate descent with bisection."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a /= s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-15:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            b = a.copy()
            b[j] = 0.0
            b /= s_others
            
            B = fast_convolve(b)
            conv_len = 2 * n - 1
            
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            # Bisection
            lo = 0.0
            hi = 1.0
            x_new = x_old
            
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0
                
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                
                if deriv > 1e-13:
                    hi = mid
                elif deriv < -1e-13:
                    lo = mid
                else:
                    break
            
            x_new = (lo + hi) / 2.0
            
            if abs(x_new - x_old) < 1e-15:
                continue
            
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            sc = a.sum()
            if sc > 1e-15:
                a /= sc
            
            conv = fast_convolve(a)
            obj = conv.max()
            
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes > 3 and passes > 5:
                rng = np.random.default_rng(passes * 31 + 17)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            elif no_improve_passes > 1:
                break
        else:
            no_improve_passes = 0
    
    return best_a, best_obj


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    n = len(a)

    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative weights over
autocorrelation positions with Nesterov-accelerated projected gradient descent,
FFT-based Hankel matrix-vector products, exact coordinate bisection polishing,
and multi-length exploration with adaptive hyperparameters and diverse restarts.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, a):
    """Compute Q_p @ a where Q_p[i,j] = p[i+j], using FFT cross-correlation."""
    n = len(a)
    p_len = len(p)
    conv_len = n + p_len - 1
    N = 1 << (conv_len - 1).bit_length()
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, N)
    P = np.fft.rfft(p, N)
    conv = np.fft.irfft(A * P, N)
    return conv[n - 1:n - 1 + n]


def mw_nesterov(a0, time_limit, lr=0.5, mom=0.9, eta=0.5, verbose=False,
                restart_interval=800, restart_noise=0.005):
    """Multiplicative weights + Nesterov projected gradient descent."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    p = np.ones(conv_len) / conv_len
    v = np.zeros(n)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        Qa = hankel_matvec(p, a)
        grad = 2.0 * Qa

        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        c = fast_convolve(a_new)
        obj = c.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Update p via multiplicative weights
        log_p = np.log(p + 1e-300) + eta * c
        log_p -= log_p.max()
        p = np.exp(log_p)
        ps = p.sum()
        if ps < 1e-300:
            p = np.ones(conv_len) / conv_len
        else:
            p /= ps

        # Adaptively increase eta (concentrate on peaks)
        eta *= 1.00003

        # Decay learning rate slowly
        lr *= 0.999997

        # Periodic restart to escape local minima
        if no_improve > restart_interval:
            rng = np.random.default_rng(it * 13 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * restart_noise)
            v = np.zeros(n)
            p = np.ones(conv_len) / conv_len
            eta = 0.3
            lr = 0.4
            no_improve = 0

        if verbose and it % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter {it}: obj={best_obj:.10f}, lr={lr:.5f}, eta={eta:.3f}, t={elapsed:.0f}s")

    return best_a, best_obj


def coordinate_bisection(a0, time_limit, verbose=False, max_passes=300):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-15:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_convolve(b)

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0

            for _ in range(42):
                mid = (lo + hi) * 0.5
                om = 1.0 - mid

                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k = int(np.argmax(f_vals))
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-14:
                    hi = mid
                elif deriv < -1e-14:
                    lo = mid
                else:
                    break

            x_new = (lo + hi) * 0.5

            if abs(x_new - x_old) < 1e-16:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            sc = a.sum()
            if sc > 1e-15:
                a /= sc
            else:
                a = best_a.copy()
                continue

            conv = fast_convolve(a)
            obj = conv.max()

            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
                if verbose and passes <= 3:
                    print(f"    CD pass {passes}, coord {j}: obj={best_obj:.10f}")

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 3 and passes > 5:
                rng = np.random.default_rng(passes * 41 + 19)
                noise_scale = 0.001 + 0.0005 * (no_improve_passes - 3)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                no_improve_passes = 0
            elif no_improve_passes >= 1 and passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Linear interpolation of a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence palindromic: a[i] = (a[i] + a[n-1-i])/2, renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def soft_minimax_grad(a, conv, top_k, temp):
    """Compute subgradient from softmax-weighted top peaks of autocorrelation."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def subgrad_nesterov(a0, time_limit, lr=0.01, mom=0.92, top_k=50,
                     temp=0.001, verbose=False):
    """Projected Nesterov subgradient descent with multi-peak softmax."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = soft_minimax_grad(a, conv, top_k, cur_temp)
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)
        conv = fast_convolve(a_new)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        cur_lr *= 0.99995
        cur_temp *= 1.002

        if no_improve > 400:
            rng = np.random.default_rng(it * 17 + 11)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0

        if verbose and it % 10000 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}")

    return best_a, best_obj

        if s_try < 0.01:
