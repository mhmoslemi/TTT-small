def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    rho = np.nonzero(cond)[0][-1]

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Antisymmetric-aware FW-LP with precomputed constraint matrix
- Local L-BFGS on smoothed (log-sum-exp) objective for fast convergence

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_smoothed(a, p):
    """
    Gradient of T*log(sum(exp(conv/T))) w.r.t. a, given p = softmax(conv/T).
    grad[i] = 2*sum_j p[i+j]*a[j] - p[2i]*a[0] (diagonal correction)
    """
    n = len(a)
    m = len(p)
    a_rev = a[::-1]
    cross = np.convolve(p, a_rev, 'full')
    # cross[n-1+i] = sum_j p[j]*a[i+j]
    grad = 2.0 * cross[n - 1:n - 1 + n].copy()
    # Diagonal correction
    diag_mask = 2 * np.arange(n) < m
    grad[diag_mask] -= p[2 * np.arange(n)[diag_mask]] * a[0]
    return grad


def smoothed_obj_grad(a, T):
    """Evaluate smoothed objective and gradient."""
    conv = fast_convolve(a)
    scaled = conv / T
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    p = exp_s / exp_s.sum()
    obj = T * (max_s + np.log(exp_s.sum()))
    grad = compute_grad_smoothed(a, p)
    return obj, grad

def peak_coordinate_descent(a0, time_limit):
    """
    For each conv peak, try to reduce it by adjusting the contributing variables.
    Uses direct coordinate updates rather than random perturbations.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        m = len(conv)

        # Process peaks in order of magnitude
        peak_order = np.argsort(conv)[::-1]
        n_peaks = min(10, m)

        for pi in range(n_peaks):
            k = int(peak_order[pi])
            if conv[k] <= conv[0] * 0.8:  # Skip if well below current max
                break

            # Variables that affect conv[k]: j in [max(0,k-n+1), min(n-1,k)]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            affected = list(range(js, je + 1))

            if len(affected) < 2:
                continue

            # Try to redistribute mass within affected variables to reduce conv[k]
            # While keeping other peaks controlled
            best_local = a.copy()
            best_local_obj = fast_convolve(a).max()

            # Greedy: for each pair (i, j) in affected, try shifting mass from i to j
            for _ in range(3):
                rng = np.random.default_rng(epoch * 1000 + k + _)
                i = affected[rng.integers(len(affected))]
                j = affected[rng.integers(len(affected))]
                if i == j:
                    continue
                # Try small transfers
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    delta = a[i] * frac
                    if a[j] + delta > 1.0:
                        continue
                    trial = a.copy()
                    trial[i] -= delta
                    trial[j] += delta
                    if trial[i] < 0:
                        continue
                    trial = trial / max(trial.sum(), 1e-15)
                    obj = fast_convolve(trial).max()
                    if obj < best_local_obj - 1e-15:
                        best_local_obj = obj
                        best_local = trial.copy()

            if best_local_obj < fast_convolve(a).max() - 1e-15:
                a = best_local
                if a.sum() > 1e-15:
                    a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()

        # Periodic check
        if epoch % 20 == 0:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 1:
                break

    return best_a, best_obj

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
