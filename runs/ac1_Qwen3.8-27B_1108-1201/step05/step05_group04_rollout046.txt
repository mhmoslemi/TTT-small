```python
import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    return c

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (1.0 / np.arange(1, n + 1)) * (cssv - 1.0) > u
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_peak_gradient(f, peaks, weights):
    """Compute gradient of sum_j weights[j] * conv[peaks[j]] w.r.t. f.
    Uses the formula: d(conv[k])/df[m] = 2*f[|k-m|] if |k-m| < n.
    """
    n = len(f)
    grad = np.zeros(n)
    m_idx = np.arange(n)
    for k, w in zip(peaks, weights):
        k = int(k)
        diffs = np.abs(k - m_idx)
        mask = diffs < n
        grad[mask] += 2.0 * w * f[diffs[mask]]
    return grad
```

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_gradient(f, peaks_idx, weights):
    """Compute LSE gradient using the formula:
    grad[m] = 2 * sum_j w_j * f[k_j - m] * 1[valid range]
    """
    n = len(f)
    grad = np.zeros(n)
    for k, w in zip(peaks_idx, weights):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi and w > 1e-15:
            ms = np.arange(lo, hi + 1)
            grad[ms] += w * f[k - ms]
    return 2.0 * grad  # factor of 2 for the bilinear form
