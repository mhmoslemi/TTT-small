    """Nesterov accelerated gradient on the simplex with softmax subgradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    x = f.copy()
    y = f.copy()
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    
    t0 = time.monotonic()
    it = 0
    t_val = 1.0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 5000)
        T = 0.5 * (0.01 / 0.5) ** progress
        top_k = max(10, int(20 + 30 * progress))
        lr = 0.02 / (1 + 0.1 * it ** 0.5)
        
        # Nesterov momentum
        t_val = 1 + t_val
        momentum = (t_val - 2) / t_val if t_val > 2 else 0
        
        # Extrapolate
        y = x + momentum * (x - y)
        y = project_simplex(y)
        
        # Subgradient at y
        conv_y = fast_convolve(y)
        grad = compute_subgradient_fft(y, conv_y, top_k=top_k, T=T)
        
        # Gradient step
        x_new = project_simplex(y - lr * grad)
        
        M_new = fast_convolve(x_new).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = x_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Nest iter {it}: M={best_M:.10f}")
        else:
            no_improve += 1
        
        y = x
        x = x_new
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            x = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = x.copy()
            no_improve = 0
    
    return best_f, best_M

def adam_v3(f0, time_limit, verbose=False):
    """Adam with adaptive scheduling, improved subgradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    base_lr = 0.008
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 8000)
        # Cosine annealing for temperature
        T = 0.05 + 0.5 * 0.5 * (1 + np.cos(np.pi * progress))
        top_k = max(15, int(20 + 40 * progress))
        lr = base_lr / (1 + 0.05 * progress)
        
        conv = fast_convolve(f)
        grad = compute_subgradient_fft(f, conv, top_k=top_k, T=T)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        M_new = fast_convolve(f_new).max()
        
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Adam-v3 iter {it}: M={best_M:.10f}, T={T:.6f}")
        else:
            no_improve += 1
        
        f = f_new
        
        if no_improve > 800:
            rng = np.random.default_rng(it * 43 + 11)
            scale = 0.002 + 0.003 * (no_improve // 400)
            f = project_simplex(best_f + rng.standard_normal(n) * min(scale, 0.02))
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
    
    return best_f, best_M

    """Mirror descent with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)

        # Stabilize
        log_f = log_f - log_f.max()

        f_new /= f_new.sum()

                elif idx_i >= offset:

                else:

                if i == j:
                    continue

                if c < 1e-12:
                    continue

                best_val = best_obj  # Current value

                    fi = t * c
                    fj = (1 - t) * c
                    # Only modify if different from current
                    if abs(fi - f[i]) < 1e-15 and abs(fj - f[j]) < 1e-15:
                        continue
                    f[i] = fi
                    f[j] = fj
                    val = fast_convolve(f).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t
                    else:
                        # Revert
                        f[i] = fi if fi > 0 else 0  # We'll set properly below
                        f[j] = fj
                        f[i] = best_t * c
                        f[j] = (1 - best_t) * c
                
                if best_val < best_obj - 1e-14:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True
                else:
                    # Restore
                    f[i] = f[i]  # Already at best_t from above
                    # Actually let's just restore properly
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c

        else:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                break
    
    return best_f, best_obj

def triagonal_init(n, sharpness=2.0):
    """Initialize with a triangular/bell-shaped profile."""
    x = np.linspace(-1, 1, n)
    f = np.exp(-sharpness * x**2)
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 0:
        f /= s
    return f

def random_init(n, seed=42):
    """Random initialization with optional structure."""
    rng = np.random.default_rng(seed)
    r = rng.random(n)
    # Add some structure: taper at boundaries
    taper = np.linspace(0.1, 1.0, n // 4)
    if n >= 8:
        r[:n//4] *= taper
        r[-(n//4):] *= taper[::-1]
    r = np.maximum(r, 0)
    s = r.sum()
    if s > 0:
        r /= s
    return r

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam with cosine-annealed temperature and adaptive top-k
2. Nesterov accelerated gradient on the simplex
3. Mirror descent (multiplicative weights) in log-space

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=20, T=0.1):
    """Smoothed subgradient of max(conv) using top-k softmax weighting."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = (pv - pv.max()) / T
    w = np.exp(scaled)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        k = int(peaks[idx])
        wi = w[idx]
        if wi < 1e-15:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            grad[j_min:j_max + 1] += 2.0 * wi * a[k - np.arange(j_min, j_max + 1)]
    return grad


def adam_optimize(f0, time_limit, verbose=False):
    """Adam with cosine-annealed temperature and adaptive top-k."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    base_lr = 0.006
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 8000)
        T = 0.05 + 0.45 * 0.5 * (1 + np.cos(np.pi * progress))
        top_k = max(15, int(20 + 40 * progress))
        lr = base_lr * (1.0 - 0.4 * progress) / (1.0 + 0.02 * progress)

        conv = fast_convolve(f)
        grad = compute_subgradient(f, conv, top_k=top_k, T=T)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f - step)
        M_new = fast_convolve(f_new).max()

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Adam iter {it}: M={best_M:.10f}, T={T:.6f}")
        else:
            no_improve += 1
        f = f_new

        if no_improve > 1000:
            rng = np.random.default_rng(it * 43 + 11)
            scale = 0.002 + 0.002 * min(no_improve // 500, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

    return best_f, best_M


def nesterov_optimize(f0, time_limit, verbose=False):
    """Nesterov accelerated gradient on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    x = f.copy()
    y = f.copy()
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    t_val = 1.0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 5000)
        T = 0.3 * (0.02 / 0.3) ** progress
        top_k = max(10, int(15 + 35 * progress))
        lr = 0.015 / (1.0 + 0.08 * np.sqrt(it))

        t_val += 1
        momentum = (t_val - 1.0) / (t_val + 1.0) if t_val > 1 else 0.0
        y = x + momentum * (x - y)
        y = project_simplex(y)

        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, top_k=top_k, T=T)
        x_new = project_simplex(y - lr * grad)
        M_new = fast_convolve(x_new).max()

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = x_new.copy()
            no_improve = 0
            if verbose and it % 300 == 0:
                print(f"    Nest iter {it}: M={best_M:.10f}")
        else:
            no_improve += 1

        y = x
        x = x_new

        if no_improve > 400:
            rng = np.random.default_rng(it * 31 + 7)
            x = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            y = x.copy()
            no_improve = 0

    return best_f, best_M


def mirror_optimize(f0, time_limit, verbose=False):
    """Mirror descent with entropic regularization in log-space."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    log_f = np.log(f + 1e-30)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / 4000)
        T = 0.08 + 0.42 * 0.5 * (1 + np.cos(np.pi * progress))
        top_k = max(10, int(15 + 25 * progress))
        lr = 0.008 / (1.0 + 0.03 * progress)

        f_current = np.exp(log_f - log_f.max())
        f_current /= f_current.sum()
        conv = fast_convolve(f_current)
        grad = compute_subgradient(f_current, conv, top_k=top_k, T=T)

        log_f = log_f - lr * grad
        log_f = log_f - log_f.max()
        f_new = np.exp(log_f)
        f_new /= f_new.sum()

        M_new = fast_convolve(f_new).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 400 == 0:
                print(f"    Mirror iter {it}: M={best_M:.10f}")
        else:
            no_improve += 1

        log_f = np.log(f_new + 1e-30)

        if no_improve > 600:
            rng = np.random.default_rng(it * 29 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            log_f = np.log(f + 1e-30)
            no_improve = 0

    return best_f, best_M


def pairwise_optimize(f0, time_limit, verbose=False):
    """Pairwise coordinate optimization with scan-based search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)

    max_offset = min(n, 60)
    offsets = list(range(1, max_offset + 1))
    if n > 100:
        offsets += [73, 144, 233, 377]
    offsets = [o for o in offsets if o < n]

    passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False
        order = rng.permutation(n) if passes % 3 == 1 else np.arange(n)

        for idx_i in range(n):
            if time.monotonic() - t0 > time_limit - 3:
                break
            i = int(order[idx_i])
            n_offsets = min(len(offsets), 6 + (passes % 5))

            for offset in offsets[:n_offsets]:
                if idx_i + offset < n:
                    j = int(order[idx_i + offset])
                elif idx_i >= offset:
                    j = int(order[idx_i - offset])
                else:
                    continue
                if i == j:
                    continue
                c = f[i] + f[j]
                if c < 1e-12:
                    continue
                orig_i = f[i]
                orig_j = f[j]
                best_t = orig_i / c
                best_val = best_obj

                candidates = [0.0, 0.1, 0.3, best_t, 0.7, 0.9, 1.0]
                if 0.1 < best_t < 0.9:
                    candidates += [best_t - 0.05, best_t + 0.05, best_t - 0.1, best_t + 0.1]

                for t in candidates:
                    t = float(np.clip(t, 0.0, 1.0))
                    f[i] = t * c
                    f[j] = (1.0 - t) * c
                    val = fast_convolve(f).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t
                    else:
                        f[i] = orig_i
                        f[j] = orig_j

                if best_val < best_obj - 1e-14:
                    f[i] = best_t * c
                    f[j] = (1.0 - best_t) * c
                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True
                else:
                    f[i] = orig_i
                    f[j] = orig_j

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def gauss_init(n, sharpness=2.0):
    """Gaussian-shaped initialization with taper at boundaries."""
    x = np.linspace(-1, 1, n)
    f = np.exp(-sharpness * x ** 2)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    return f


def random_tapered_init(n, seed=42):
    """Random initialization with boundary taper."""
    rng = np.random.default_rng(seed)
    r = rng.random(n)
    taper_len = max(4, n // 6)
    taper = np.linspace(0.05, 1.0, taper_len)
    r[:taper_len] *= taper
    r[-taper_len:] *= taper[::-1]
    r = np.maximum(r, 0.0)
    s = r.sum()
    if s > 1e-15:
        r /= s
    return r


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-method optimization combining Adam, Nesterov, Mirror Descent,
    and Pairwise coordinate descent with multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if len(a) == 0:
            return float('inf')
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = gauss_init(512, 1.5)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default init: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Adam on parent ===
    print(f"\n=== Phase 1: Adam (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 900
    t1 = min(rem, 350)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = adam_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "adam_p1")

    # === Phase 2: Nesterov on best ===
    print(f"\n=== Phase 2: Nesterov (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 700
        t2 = min(rem, 200)
        if t2 > 30:
            print(f"  Time: {t2:.0f}s")
            a_opt, _ = nesterov_optimize(best_seq, time_limit=t2, verbose=True)
            try_seq(a_opt, "nesterov_p2")

    # === Phase 3: Mirror descent ===
    print(f"\n=== Phase 3: Mirror (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 550
        t3 = min(rem, 150)
        if t3 > 20:
            print(f"  Time: {t3:.0f}s")
            a_opt, _ = mirror_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "mirror_p3")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent

    for target_n in [384, 512, 768, 1024]:
        if target_n == len(base):
            continue
        rem = deadline - time.monotonic() - 400
        if rem < 60:
            break
        f_interp = interpolate_sequence(base, target_n)

        t_alloc = min(rem * 0.45, 130)
        if t_alloc > 20:
            print(f"  Adam n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_optimize(f_interp, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"adam_n{target_n}")

        rem = deadline - time.monotonic() - 350
        if rem > 20:
            t_nest = min(rem * 0.4, 100)
            if t_nest > 15:
                print(f"  Nesterov n={target_n}, t={t_nest:.0f}s")
                a_opt2, _ = nesterov_optimize(a_opt, time_limit=t_nest)
                try_seq(a_opt2, f"nest_n{target_n}")

    # === Phase 5: Diverse starts ===
    print(f"\n=== Phase 5: Diverse (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        diverse_inits = [
            ("gauss1", gauss_init(n_best, 1.0)),
            ("gauss3", gauss_init(n_best, 3.5)),
            ("rand1", random_tapered_init(n_best, seed=77)),
            ("rand2", random_tapered_init(n_best, seed=200)),
            ("rev", best_seq[::-1].copy() / best_seq[::-1].sum()),
            ("mix", project_simplex(0.6 * best_seq + 0.4 * gauss_init(n_best, 2.0))),
        ]

        for name, f_init in diverse_inits:
            rem = deadline - time.monotonic() - 200
            if rem < 25:
                break
            t_alloc = min(rem / max(1, len(diverse_inits)), 90)
            if t_alloc < 15:
                continue
            print(f"  {name}: t={t_alloc:.0f}s")
            a_opt, _ = adam_optimize(f_init, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"div_{name}")

    # === Phase 6: Pairwise polish ===
    print(f"\n=== Phase 6: Pairwise (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 25:
            t_pw = min(rem * 0.7, 220)
            print(f"  Time: {t_pw:.0f}s")
            a_opt, _ = pairwise_optimize(best_seq, time_limit=t_pw)
            try_seq(a_opt, "pairwise_p6")

            rem = deadline - time.monotonic() - 100
            if rem > 15:
                a_opt2, _ = adam_optimize(a_opt, time_limit=min(rem, 60))
                try_seq(a_opt2, "adam_after_pw")

    # === Phase 7: Final intensive ===
    print(f"\n=== Phase 7: Final (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            a_opt, _ = adam_optimize(best_seq, time_limit=min(rem, 80), verbose=True)
            try_seq(a_opt, "adam_final")

        rem = deadline - time.monotonic() - 25
        if rem > 15:
            a_opt, _ = nesterov_optimize(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "nest_final")

        # Micro-perturbations
        for trial in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            rng = np.random.default_rng(900 + trial * 13)
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.0015)
            a_opt, _ = adam_optimize(f_pert, time_limit=min(rem, 25))
            try_seq(a_opt, f"micro_{trial}")

        # Final pairwise
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = pairwise_optimize(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "pw_final")

    # Fallback
    if best_seq is None:
        best_seq = gauss_init(512, 1.5)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
