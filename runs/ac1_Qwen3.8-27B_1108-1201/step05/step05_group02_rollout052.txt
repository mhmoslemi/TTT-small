"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using enhanced Frank-Wolfe with
analytical quadratic line search, multi-formulation LP steps, and
adaptive restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(d, a):
    """Cross-convolution d * a via FFT. Result[k] = sum_i d[i] * a[k-i]."""
    n = len(d)
    m = len(a)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    D = np.fft.rfft(d, fft_size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(D * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_target_peak(f, n, target_k):
    """Solve LP: min (f*g)[target_k] s.t. g >= 0, sum(g) = 1."""
    c = np.zeros(n)
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    js = max(0, target_k - n + 1)
    je = min(n - 1, target_k)
    for j in range(js, je + 1):
        c[j] = f[target_k - j]

    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=None, b_ub=None, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def analytical_line_search(a, g, conv_a, cross_da, conv_d, n_iter=60):
    """
    Find t in [0,1] minimizing max_k (conv_a[k] + 2t*cross_da[k] + t^2*conv_d[k]).
    Uses ternary search since the function is convex in t.
    """
    def obj(t):
        vals = conv_a + 2.0 * t * cross_da + t * t * conv_d
        return vals.max()

    lo = 0.0
    hi = 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def quick_line_search(a, g, n_iter=40):

def fw_optimize(a0, time_limit, verbose=False, formulation="minpeak"):
    """
    Frank-Wolfe with analytical quadratic line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP for the Frank-Wolfe direction
        if formulation == "minpeak":
            g, _ = solve_lp_min_peak(a, n)
            if g is None:
                break
        elif formulation == "target":
            # Target the current peak
            peak_k = int(np.argmax(conv))
            g = solve_lp_target_peak(a, n, peak_k)
            if g is None:
                g = np.ones(n) / n
        else:
            g, _ = solve_lp_min_peak(a, n)
            if g is None:
                break

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - a)) < 1e-13:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Analytical line search
        d = g - a
        conv_a = fast_convolve(a)
        cross_da = cross_convolve(d, a)
        conv_d = fast_convolve(d)

        best_t = analytical_line_search(a, g, conv_a, cross_da, conv_d, n_iter=50)

        if best_t > 1e-12:
            a_new = a + best_t * (g - a)
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / sn

            new_obj = fast_convolve(a_new).max()
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_a = a_new.copy()
                no_improve = 0
                a = a_new
                conv = fast_convolve(a)
                if verbose and it % 10 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}, t={best_t:.4f}")

                no_improve += 1
        else:
            no_improve += 1

            if conv.max() < best_obj:
                best_obj = conv.max()

            no_improve = 0

    return best_a, best_obj


def coordinate_descent_peak(a0, time_limit, verbose=False):
    """
    Coordinate descent that directly optimizes each coordinate to minimize the peak.
    For each coordinate j, finds the optimal value by 1D search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        # Process coordinates in order of contribution to peak
        peak_k = int(np.argmax(conv))
        # Coordinates contributing to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contributing = list(range(js, je + 1))
        
        # Also include some random coordinates for exploration
        rng = np.random.default_rng(passes * 104729)
        for _ in range(min(10, n // 20)):
            contributing.append(int(rng.integers(0, n)))
        
        # Sort by value (larger first - they likely contribute more)
        contributing.sort(key=lambda j: -a[j])

        for j in contributing:
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-14:
                continue

            # Find optimal value for coordinate j
            # If we set a[j] = t, rescale others: a[i] *= (1-t)/(1-a[j]) for i != j
            old_val = a[j]
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_t = old_val
            best_obj_j = best_obj

            # Try several values
            candidates = []
            for mult in [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                t = old_val * mult
                if 0 < t < 0.8:
                    candidates.append(t)
            # Also try values around the current best
            candidates.append(max(0.01, old_val * 0.9))
            candidates.append(min(0.7, old_val * 1.1))

            for t in candidates:
                if abs(t - old_val) < 1e-10:
                    continue
                a_trial = a.copy()
                a_trial[j] = t
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= (1.0 - t) / denom
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_obj_j - 1e-14:
                    best_obj_j = val
                    best_t = t

            if best_t != old_val:
                a[j] = best_t
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= (1.0 - best_t) / denom
                a = a / a.sum()
                conv = fast_convolve(a)
                if conv.max() < best_obj - 1e-14:
                    best_obj = conv.max()
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_a, best_obj


def sg_optimize(f0, time_limit, lr=0.003, momentum=0.94, top_k=50, temp=0.00005,
                verbose=False):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        co = conv_new.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            conv = conv_new
        else:
            no_improve += 1
            f = f_new
            conv = conv_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj

Minimizes 2*n*max(conv(a,a))/sum(a)^2 using enhanced Frank-Wolfe with
analytical quadratic line search, multi-formulation LP steps, coordinate
descent, subgradient refinement, and adaptive multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(d, a):
    """Cross-convolution d * a via FFT. Result[k] = sum_i d[i] * a[k-i]."""
    n = len(d)
    m = len(a)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    D = np.fft.rfft(d, fft_size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(D * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0

    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_lp_target_peak(f, n, target_k):
    """Solve LP: min (f*g)[target_k] s.t. g >= 0, sum(g) = 1."""
    c = np.zeros(n)
    js = max(0, target_k - n + 1)
    je = min(n - 1, target_k)
    for j in range(js, je + 1):
        c[j] = f[target_k - j]

    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=None, b_ub=None, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def analytical_line_search(conv_a, cross_da, conv_d, n_iter=55):
    """
    Find t in [0,1] minimizing max_k (conv_a[k] + 2t*cross_da[k] + t^2*conv_d[k]).
    Uses ternary search since the function is convex in t.
    """
    def obj(t):
        vals = conv_a + 2.0 * t * cross_da + t * t * conv_d
        return vals.max()

    lo = 0.0
    hi = 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if formulation == "minpeak":
            g, _ = solve_lp_min_peak(a, n)
            if g is None:
                break
        elif formulation == "target":
            peak_k = int(np.argmax(conv))
            g = solve_lp_target_peak(a, n, peak_k)
            if g is None:
                g = np.ones(n) / n
        else:
            g, _ = solve_lp_min_peak(a, n)
            if g is None:
                break

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - a)) < 1e-13:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        # Analytical line search using precomputed convolutions
        d = g - a
        conv_a = fast_convolve(a)
        cross_da = cross_convolve(d, a)
        conv_d = fast_convolve(d)

        best_t = analytical_line_search(conv_a, cross_da, conv_d, n_iter=50)

        if best_t > 1e-12:
            a_new = a + best_t * (g - a)
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / sn

            new_obj = fast_convolve(a_new).max()
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_a = a_new.copy()
                no_improve = 0
                a = a_new
                conv = fast_convolve(a)
                if verbose and it % 5 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}, t={best_t:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 35:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.003 * min(no_improve // 25, 5)
            noise = rng.standard_normal(n) * scale
            a = project_simplex(best_a + noise)
            conv = fast_convolve(a)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_a = a.copy()
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_peak(a0, time_limit, verbose=False):
    """
    Coordinate descent optimizing each coordinate to minimize the peak.
    Uses multi-start per coordinate for better local search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contributing = list(range(js, je + 1))

        rng = np.random.default_rng(passes * 104729)
        for _ in range(min(15, n // 15)):
            contributing.append(int(rng.integers(0, n)))

        contributing.sort(key=lambda j: -a[j])
        # Deduplicate
        seen = set()
        unique_contributing = []
        for j in contributing:
            if j not in seen:
                seen.add(j)
                unique_contributing.append(j)
        contributing = unique_contributing

        for j in contributing:
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-14:
                continue

            old_val = a[j]
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_t = old_val
            best_obj_j = best_obj

            candidates = []
            for mult in [0.3, 0.5, 0.65, 0.8, 1.2, 1.4, 1.7, 2.2]:
                t = old_val * mult
                if 1e-6 < t < 0.9:
                    candidates.append(t)
            candidates.append(max(1e-4, old_val - 0.01))
            candidates.append(min(0.85, old_val + 0.01))

            for t in candidates:
                if abs(t - old_val) < 1e-10:
                    continue
                a_trial = a.copy()
                a_trial[j] = t
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= (1.0 - t) / denom
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_obj_j - 1e-14:
                    best_obj_j = val
                    best_t = t

            if best_t != old_val:
                a[j] = best_t
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= (1.0 - best_t) / denom
                a = a / a.sum()
                conv = fast_convolve(a)
                if conv.max() < best_obj - 1e-14:
                    best_obj = conv.max()
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 8:
                # Try random perturbations
                rng = np.random.default_rng(passes * 55501)
                for _ in range(10):
                    j = int(rng.integers(0, n))
                    if a[j] < 1e-10:
                        continue
                    new_val = a[j] * rng.uniform(0.3, 2.5)
                    if new_val >= 0.9:
                        continue
                    denom = 1.0 - a[j]
                    if denom < 1e-10:
                        continue
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] *= (1.0 - new_val) / denom
                    a_trial = a_trial / a_trial.sum()
                    val = fast_convolve(a_trial).max()
                    if val < best_obj - 1e-12:
                        a = a_trial
                        conv = fast_convolve(a)
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        break
            if no_improve > 15:
                break
        else:
            no_improve = 0

    return best_a, best_obj


def sg_optimize(f0, time_limit, lr=0.002, momentum=0.94, top_k=50, temp=0.00004,
                verbose=False):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]

        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        co = conv_new.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            conv = conv_new
        else:
            no_improve += 1
            f = f_new
            conv = conv_new

        cur_lr *= 0.9999
        cur_temp *= 1.003

        if no_improve > 250:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: Enhanced FW with analytical line search,
    coordinate descent, subgradient refinement, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Enhanced FW with min-peak LP + analytical line search ===
    print("\n=== Phase 1: Enhanced FW (min-peak, analytical LS) ===")
    rem = deadline - time.monotonic()
    t1 = min(350, rem * 0.3)
    if t1 > 30:
        print(f"  n={base_n}, t={t1:.0f}s")
        a_opt, _ = fw_optimize(parent, time_limit=t1, verbose=True, formulation="minpeak")
        try_seq(a_opt, "fw_minpeak")

    # === Phase 2: Coordinate descent (complements FW) ===
    print("\n=== Phase 2: Coordinate Descent ===")
    rem = deadline - time.monotonic()
    base = best_seq if best_seq is not None else parent
    t2 = min(200, rem * 0.18)
    if t2 > 25:
        print(f"  n={len(base)}, t={t2:.0f}s")
        a_opt, _ = coordinate_descent_peak(base, time_limit=t2, verbose=True)
        try_seq(a_opt, "coord_descent")

    # === Phase 3: FW with target-peak (exploit peak structure) ===
    print("\n=== Phase 3: FW (target-peak) ===")
    rem = deadline - time.monotonic()
    base = best_seq if best_seq is not None else parent
    t3 = min(150, rem * 0.15)
    if t3 > 20:
        print(f"  n={len(base)}, t={t3:.0f}s")
        a_opt, _ = fw_optimize(base, time_limit=t3, verbose=True, formulation="target")
        try_seq(a_opt, "fw_target")

    # === Phase 4: Subgradient refinement ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_sg = min(rem * 0.2, 250)
        if t_sg > 30:
            print(f"  n={n_best}, t={t_sg:.0f}s")
            a_opt, _ = sg_optimize(best_seq, time_limit=t_sg,
                                   lr=0.002, momentum=0.94,
                                   top_k=max(30, n_best // 6),
                                   temp=0.00004)
            try_seq(a_opt, "sg_main")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-length exploration ===")
    base = best_seq if best_seq is not None else parent
    candidate_lengths = []
    for delta in [-96, -48, 48, 96, 192]:
        ln = len(base) + delta
        if ln >= 256 and ln <= 2048:
            candidate_lengths.append(ln)
    for ln in [512, 768, 832, 896, 1024]:
        if ln not in candidate_lengths:
            candidate_lengths.append(ln)
    candidate_lengths = sorted(set(candidate_lengths))

    for target_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.25, 120)
        if t_alloc < 20:
            break
        f_interp = interpolate_sequence(base, target_n)
        print(f"  FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = fw_optimize(f_interp, time_limit=t_alloc, formulation="minpeak")
        try_seq(a_opt, f"fw_n{target_n}")

    # === Phase 6: Final polish - alternate methods ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 30
        if rem > 20:
            print("  FW polish...")
            a_opt, _ = fw_optimize(best_seq, time_limit=min(rem, 80), formulation="minpeak")
            try_seq(a_opt, "fw_polish")

        rem = deadline - time.monotonic() - 20
        if rem > 15:
            print("  CD polish...")
            a_opt, _ = coordinate_descent_peak(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "cd_polish")

        # Perturbations with quick FW
        for p in range(10):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.0008 + 0.0003 * p)
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem, 25)
            a_opt, _ = fw_optimize(f_pert, time_limit=t_pert, formulation="minpeak")
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
