def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coord_bisect(a, j, conv, B, b_valid, x_old, tol=1e-14):
    """Bisect to find optimal value of coordinate j."""
    n = len(a)
    conv_len = 2 * n - 1
    lo, hi = 0.0, 1.0
    x_new = x_old
    for _ in range(50):
        mid = (lo + hi) / 2.0
        om = 1.0 - mid
        f_vals = om * om * B + 2.0 * mid * om * b_valid
        if 2 * j < conv_len:

        B_k = B[k]

            break
        x_new = (lo + hi) / 2.0

Actually, for a non-negative sequence, `conv(a,a)[0] = a[0]^2` and `conv(a,a)[2n-2] = a[n-1]^2`. The maximum is often at the center `conv(a,a)[n-1] = sum_j a[j]*a[n-1-j]`.

If the sequence is symmetric (`a[j] = a[n-1-j]`), then `conv(a,a)[n-1] = sum_j a[j]^2 = ||a||^2`. By Cauchy-Schwarz, `||a||^2 ≥ (sum a)^2/n = 1/n`, with equality for uniform. So for symmetric sequences, `max conv ≥ 1/n`.

For non-symmetric sequences, the center value is `sum_j a[j]*a[n-1-j]` which by Cauchy-Schwarz is also ≥ `(sum_j sqrt(a[j]*a[n-1-j]))^2 / n`... hmm, not as clean.

The point is: for the uniform sequence, max conv = 1/n (at the center), giving objective = 2. To get below 2, we need to make the center peak much smaller than the uniform value while keeping other peaks from growing too much.

**Key idea**: If we make the sequence "anti-symmetric" in some sense (high at edges, low in middle, or vice versa), we can reduce the center peak. But we need to keep all other peaks low too.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast exact coordinate descent with adaptive bisection
2. Peak-focused coordinate ordering
3. Block coordinate optimization for top pairs
4. Multi-length exploration with interpolation

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def exact_coord_descent(a, time_limit, max_passes=999999, top_k_peaks=50, verbose=False):
    """
    Fast exact coordinate descent with bisection.
    Optimizes one coordinate at a time using exact 1D bisection.
    """
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    conv_len = 2 * n - 1
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    pass_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and pass_count < max_passes:
        pass_count += 1
        pass_improved = False
        total_improvement = 0.0
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            x_old = a[j]
            if x_old < 1e-15:
                continue
            
            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue
            
            # Build b = a with a[j] removed, normalized to sum 1
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            
            # B = conv(b)
            B = fast_convolve(b)
            
            # b_valid[k] = b[k-j] if 0 <= k-j < n, else 0
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]
            
            # Bisection
            lo = 0.0
            hi = 1.0
            x_new = x_old
            
            for _ in range(45):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (k == 2 * j) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                if deriv > 1e-14:
                    hi = mid
                elif deriv < -1e-14:
                    lo = mid
                else:
                    x_new = mid
                    break
                if hi - lo < 1e-14:
                    x_new = (lo + hi) / 2.0
                    break
            else:
                x_new = (lo + hi) / 2.0
            
            if abs(x_new - x_old) < 1e-14:
                continue
            
            # Apply update
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            
            conv = fast_convolve(a)
            obj = conv.max()
            improvement = best_obj - obj
            total_improvement += max(0, improvement)
            
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
        
        if not pass_improved:
            if pass_count > 3:
                # Add tiny noise
                rng = np.random.default_rng(pass_count * 31 + 7)
                noise = 0.0003 * max(1.0, 1.0 - pass_count * 0.02)
                a = project_simplex(best_a + rng.standard_normal(n) * noise)
                conv = fast_convolve(a)
            else:
                break
            no_improve = 0
        else:
            no_improve = 0
            
        if verbose and pass_count % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}, t={elapsed:.1f}s, imp={total_improvement:.2e}")
    
    return best_a, best_obj

def block_coord_descent(a, time_limit, num_blocks=50, verbose=False):
    """
    Block coordinate descent: optimize pairs of coordinates jointly.
    Focus on pairs that affect the peak convolution positions.
    """
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    conv_len = 2 * n - 1
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Identify peak positions
    peak_threshold = conv.max() * 0.95
    peak_positions = np.where(conv >= peak_threshold)[0]
    
    # For each peak position k, the affected coordinates are j and k-j
    # Build a list of important coordinate pairs
    import functools
    
    coord_importance = np.zeros(n)
    for k in peak_positions:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            coord_importance[j] += 1
    
    # Sort coordinates by importance
    sorted_coords = np.argsort(coord_importance)[::-1]
    
    # Try optimizing pairs from the most important coordinates
    tried_pairs = set()
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        # Select a pair
        if iteration <= len(sorted_coords) * (len(sorted_coords) - 1) // 2:
            # Systematic: take top coordinates in pairs
            idx = (iteration - 1) % min(num_blocks, len(sorted_coords))
            j = int(sorted_coords[idx])
            # Find partner: the coordinate most correlated with j at peak positions
            l_candidates = []
            for k in peak_positions:
                partner = k - j
                if 0 <= partner < n and partner != j:
                    l_candidates.append(partner)
            if l_candidates:
                l = int(l_candidates[0])
            else:
                l = int(sorted_coords[(idx + 1) % len(sorted_coords)])
        else:
            j = int(rng.integers(0, n))
            l = int(rng.integers(0, n))
        
        if j == l or (j, l) in tried_pairs:
            continue
        tried_pairs.add((j, l))
        tried_pairs.add((l, j))
        
        if a[j] < 1e-12 or a[l] < 1e-12:
            continue
        
        # 2D grid search for optimal (x_j, x_l)
        x_j_old = a[j]
        x_l_old = a[l]
        
        # Fix all other coordinates, normalize to sum (1 - x_j - x_l)
        s_rest = 1.0 - x_j_old - x_l_old
        if s_rest < 1e-12:
            continue
        
        # b = rest normalized
        b = a.copy()
        b[j] = 0.0
        b[l] = 0.0
        b = b / s_rest
        
        B = fast_convolve(b)
        
        # bv_j[k] = b[k-j] if valid
        bv_j = np.zeros(conv_len)
        start = j
        end = min(j + n, conv_len)
        bv_j[start:end] = b[:end - start]
        
        # bv_l[k] = b[k-l] if valid
        bv_l = np.zeros(conv_len)
        start = l
        end = min(l + n, conv_len)
        bv_l[start:end] = b[:end - start]
        
        def obj_2d(x, y):
            om = 1.0 - x - y
            if om < 0:
                return np.inf
            f_vals = om * om * B + 2.0 * om * (x * bv_j + y * bv_l)
            if 2 * j < conv_len:
                f_vals[2 * j] += x * x
            if 2 * l < conv_len:
                f_vals[2 * l] += y * y
            if j + l < conv_len:
                f_vals[j + l] += 2.0 * x * y
            return f_vals.max()
        
        # Coarse grid
        best_x, best_y = x_j_old, x_l_old
        best_val = obj_2d(x_j_old, x_l_old)
        
        # Grid over triangle x + y <= 1, x >= 0, y >= 0
        # Focus around current values
        margin = max(0.1, 0.5 * max(x_j_old, x_l_old))
        x_lo = max(0, x_j_old - margin)
        x_hi = min(1, x_j_old + margin)
        y_lo = max(0, x_l_old - margin)
        y_hi = min(1 - x_lo, x_l_old + margin)
        
        for gi in range(8):
            for gj in range(8):
                x = x_lo + (x_hi - x_lo) * gi / 7.0
                y = y_lo + (y_hi - y_lo) * gj / 7.0
                if x + y > 1.0:
                    continue
                val = obj_2d(x, y)
                if val < best_val - 1e-14:
                    best_val = val
                    best_x, best_y = x, y
        
        # Fine grid around best
        fm = max(0.01, 0.3 * max(x_hi - x_lo, y_hi - y_lo))
        for gi in range(10):
            for gj in range(10):
                x = best_x - fm + 2 * fm * gi / 9.0
                y = best_y - fm + 2 * fm * gj / 9.0
                if x < 0 or y < 0 or x + y > 1.0:
                    continue
                val = obj_2d(x, y)
                if val < best_val - 1e-14:
                    best_val = val
                    best_x, best_y = x, y
        
        # Apply
        if best_val < best_obj - 1e-13:
            a[j] = best_x
            a[l] = best_y
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            mask[l] = False
            a[mask] *= (1.0 - best_x - best_y) / s_rest
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                if verbose and iteration % 10 == 0:
                    print(f"    BC iter {iteration}: obj={best_obj:.10f}")
    
    return best_a, best_obj

            t_alloc = min(remaining * 0.2, 60)

                best_obj = obj
                best_a = a_opt
                if verbose:

        elif phase % 5 == 2:

            rng = np.random.default_rng(phase * 103 + 13)

            t_alloc = min(remaining * 0.2, 80)
            a_opt, obj = exact_coord_descent(a_noisy, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} Noise+CD: obj={best_obj:.10f}")

        elif phase % 5 == 3:
            # Symmetrize + CD
            a_sym = 0.5 * (best_a + best_a[::-1])

            ss = a_sym.sum()

                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} Sym+CD: obj={best_obj:.10f}")

        else:

            a_rev = best_a[::-1].copy()
            t_alloc = min(remaining * 0.1, 40)
            a_opt, obj = exact_coord_descent(a_rev, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def obj_2d_block(B, bv_j, bv_l, j, l, conv_len, x, y):
    """Compute max convolution value for 2D block coordinate step."""
    om = 1.0 - x - y
    if om < -1e-15:
        return np.inf
    om = max(0.0, om)
    f_vals = om * om * B + 2.0 * om * (x * bv_j + y * bv_l)
    if 2 * j < conv_len:
        f_vals[2 * j] += x * x
    if 2 * l < conv_len:
        f_vals[2 * l] += y * y
    if j + l < conv_len:
        f_vals[j + l] += 2.0 * x * y
    return f_vals.max()


def exact_coord_descent(a, time_limit, max_passes=999999, verbose=False):
    """
    Fast exact coordinate descent with bisection per coordinate.
    Normalizes to sum=1, minimizes max of self-convolution.
    """
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    pass_count = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit and pass_count < max_passes:
        pass_count += 1
        pass_improved = False
        total_imp = 0.0

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-15:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_convolve(b)

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(45):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (k == 2 * j) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                if deriv > 1e-14:
                    hi = mid
                elif deriv < -1e-14:
                    lo = mid
                else:
                    x_new = mid
                    break
                if hi - lo < 1e-14:
                    x_new = (lo + hi) / 2.0
                    break
            else:
                x_new = (lo + hi) / 2.0

            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check

            conv = fast_convolve(a)
            obj = conv.max()
            imp = best_obj - obj
            if imp > 0:
                total_imp += imp

            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            stagnation += 1
            if stagnation > 2:
                rng = np.random.default_rng(pass_count * 31 + 7)
                noise_scale = 0.0004 * max(0.5, 1.0 - 0.01 * pass_count)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                conv = fast_convolve(a)
                stagnation = 0
            elif stagnation >= 2:
                break
        else:
            stagnation = 0

        if verbose and pass_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}, t={elapsed:.1f}s, imp={total_imp:.2e}")

    return best_a, best_obj


def block_coord_descent(a, time_limit, num_pairs=80, verbose=False):
    """
    Block coordinate descent: optimize pairs of coordinates jointly via 2D grid search.
    Focuses on pairs that affect peak convolution positions.
    """
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    a = a / s
    n = len(a)
    conv_len = 2 * n - 1

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    peak_threshold = conv.max() * 0.90
    peak_positions = np.where(conv >= peak_threshold)[0]

    coord_importance = np.zeros(n)
    for k in peak_positions:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for jj in range(js, je + 1):
            coord_importance[jj] += 1.0

    sorted_coords = np.argsort(coord_importance)[::-1]
    tried_pairs = set()
    iteration = 0
    n_attempted = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        if iteration > num_pairs * 3:
            break

        if iteration <= len(sorted_coords):
            j = int(sorted_coords[iteration - 1])
            l_candidates = []
            for k in peak_positions:
                partner = k - j
                if 0 <= partner < n and partner != j:
                    l_candidates.append(partner)
            if l_candidates:
                l = int(l_candidates[(iteration - 1) % len(l_candidates)])
            else:
                l = int(sorted_coords[(iteration) % len(sorted_coords)])
        else:
            j = int(rng.integers(0, n))
            l = int(rng.integers(0, n))

        if j == l:
            continue
        pair_key = (min(j, l), max(j, l))
        if pair_key in tried_pairs:
            continue
        tried_pairs.add(pair_key)

        if a[j] < 1e-12 or a[l] < 1e-12:
            continue

        x_j_old = a[j]
        x_l_old = a[l]
        s_rest = 1.0 - x_j_old - x_l_old
        if s_rest < 1e-12:
            continue

        b = a.copy()
        b[j] = 0.0
        b[l] = 0.0
        b = b / s_rest

        B = fast_convolve(b)

        bv_j = np.zeros(conv_len)
        start_j = j
        end_j = min(j + n, conv_len)
        bv_j[start_j:end_j] = b[:end_j - start_j]

        bv_l = np.zeros(conv_len)
        start_l = l
        end_l = min(l + n, conv_len)
        bv_l[start_l:end_l] = b[:end_l - start_l]

        best_x, best_y = x_j_old, x_l_old
        best_val = obj_2d_block(B, bv_j, bv_l, j, l, conv_len, x_j_old, x_l_old)

        margin = max(0.05, 0.5 * max(x_j_old, x_l_old))
        x_lo = max(0.0, x_j_old - margin)
        x_hi = min(1.0 - 1e-6, x_j_old + margin)
        y_lo = max(0.0, x_l_old - margin)
        y_hi = min(1.0 - x_lo - 1e-6, x_l_old + margin)

        if x_hi <= x_lo or y_hi <= y_lo:
            continue

        for gi in range(6):
            for gj in range(6):
                x = x_lo + (x_hi - x_lo) * gi / 5.0
                y = y_lo + (y_hi - y_lo) * gj / 5.0
                if x + y > 1.0 - 1e-9:
                    continue
                val = obj_2d_block(B, bv_j, bv_l, j, l, conv_len, x, y)
                if val < best_val - 1e-14:
                    best_val = val
                    best_x, best_y = x, y

        fm = max(0.005, 0.25 * max(x_hi - x_lo, y_hi - y_lo))
        for gi in range(8):
            for gj in range(8):
                x = best_x - fm + 2.0 * fm * gi / 7.0
                y = best_y - fm + 2.0 * fm * gj / 7.0
                if x < 0 or y < 0 or x + y > 1.0 - 1e-9:
                    continue
                val = obj_2d_block(B, bv_j, bv_l, j, l, conv_len, x, y)
                if val < best_val - 1e-14:
                    best_val = val
                    best_x, best_y = x, y

        if best_val < best_obj - 1e-13:
            a[j] = best_x
            a[l] = best_y
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            mask[l] = False
            scale = (1.0 - best_x - best_y) / s_rest
            if scale >= 0:
                a[mask] *= scale
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                n_attempted += 1
                if verbose and n_attempted % 10 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    BC attempt {n_attempted}: obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_a, best_obj

def spectral_init(n, seed_val):
    """Create initialization via spectral shaping."""
    rng = np.random.default_rng(seed_val)
    phases = rng.uniform(0, 2 * np.pi, n)
    F = np.exp(1j * phases)
    a = np.fft.irfft(F, n)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-10:
        a /= ss
    else:
        a = np.ones(n) / n
    return a


def cosine_init(n, freq, amplitude, phase_shift=0.0):
    """Cosine-modulated initialization."""
    t = np.arange(n) / n
    a = 1.0 + amplitude * np.cos(2 * np.pi * freq * t + phase_shift)
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-15:
        a /= ss
    else:
        a = np.ones(n) / n
    return a

    n_base = len(base)

    inits = []

    a_sym = 0.5 * (base + base[::-1])
    a_sym = np.maximum(a_sym, 0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    inits.append(("sym", a_sym))

    inits.append(("cos3", cosine_init(n_base, 3, 0.3, 0.0)))
    inits.append(("cos5", cosine_init(n_base, 5, 0.25, np.pi / 4)))
    inits.append(("cos7", cosine_init(n_base, 7, 0.35, np.pi / 3)))

    a_spec = spectral_init(n_base, 12345)
    inits.append(("spec", a_spec))

    # Bimodal
    center = np.arange(n_base) - n_base / 2.0
    bimodal_w = 1.0 + 0.5 * np.exp(-center ** 2 / (2 * (n_base / 3.0) ** 2))
    a_bi = base * bimodal_w
    a_bi = np.maximum(a_bi, 0)
    ss = a_bi.sum()
    if ss > 1e-15:
        a_bi /= ss
    inits.append(("bimodal", a_bi))

    # Warped parent
    warp = 1.0 + 0.2 * np.sin(2 * np.pi * np.arange(n_base) / n_base * 2)
    a_warp = base * warp
    a_warp = np.maximum(a_warp, 0)
    ss = a_warp.sum()
    if ss > 1e-15:
        a_warp /= ss
    inits.append(("warp", a_warp))

    for label, init_a in inits:
        rem = deadline - time.monotonic() - 450
        if rem < 25:
            break
        t_alloc = min(rem * 0.25, 60)
        print(f"  Init '{label}', t={t_alloc:.0f}s")
        a_opt, _ = combined_search(init_a, time_limit=t_alloc, verbose=False)
        record_result(a_opt, label)

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score_global:.6f}) ===")
    base = best_a_global if best_a_global is not None else parent

    target_lengths = [640, 896, 1024, 1280]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 250
        if rem < 40:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.35, 120)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = combined_search(f_interp, time_limit=t_alloc, verbose=False)
        record_result(a_opt, f"n{target_n}")

    # === Phase 4: Deep optimization from best ===
    print(f"\n=== Phase 4: Deep optimize (best={best_score_global:.6f}) ===")
    if best_a_global is not None:
        n_best = len(best_a_global)

        deep_inits = [
            ("d_noise2", 0.002),
            ("d_noise1", 0.001),
            ("d_noise4", 0.004),
            ("d_noise05", 0.0005),
        ]

        for label, ns in deep_inits:
            rem = deadline - time.monotonic() - 120
            if rem < 20:
                break
            rng_r = np.random.default_rng(seed + abs(hash(label)) % 100000)
            f_init = project_simplex(best_a_global + rng_r.standard_normal(n_best) * ns)
            t_alloc = min(rem * 0.3, 80)
            print(f"  {label} (ns={ns}), t={t_alloc:.0f}s")
            a_opt, _ = combined_search(f_init, time_limit=t_alloc, verbose=False)
            record_result(a_opt, label)

    # === Phase 5: Final intensive polishing ===
    print(f"\n=== Phase 5: Final polish (best={best_score_global:.6f}) ===")
    if best_a_global is not None:
        # Long CD run
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem, 150)
            print(f"  Final long CD, t={t_alloc:.0f}s")
            a_opt, _ = exact_coord_descent(best_a_global, time_limit=t_alloc, verbose=True)
            record_result(a_opt, "final_cd")

        # Block CD polish
        rem = deadline - time.monotonic() - 40
        if rem > 15 and best_a_global is not None:
            t_alloc = min(rem, 50)
            print(f"  Final block CD, t={t_alloc:.0f}s")
            a_opt, _ = block_coord_descent(best_a_global, time_limit=t_alloc, verbose=True)
            record_result(a_opt, "final_bc")

        # Tiny perturbations + quick CD
        rem = deadline - time.monotonic() - 20
        if rem > 10 and best_a_global is not None:
            n_best = len(best_a_global)
            for p in range(5):
                rem = deadline - time.monotonic() - 10
                if rem < 8:
                    break
                rng_p = np.random.default_rng(seed + 5000 + p * 13)
                f_pert = project_simplex(best_a_global + rng_p.standard_normal(n_best) * 0.0003)
                t_alloc = min(rem, 15)
                a_opt, _ = exact_coord_descent(f_pert, time_limit=t_alloc)
                record_result(a_opt, f"micro_{p}")

        # One more combined search
        rem = deadline - time.monotonic() - 15
        if rem > 10 and best_a_global is not None:
            t_alloc = min(rem, 20)
            a_opt, _ = combined_search(best_a_global, time_limit=t_alloc)
            record_result(a_opt, "final_combined")

    # Fallback
    if best_a_global is None:
        best_a_global = parent.copy()
        best_score_global = float(evaluate_sequence(best_a_global.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a_global.tolist()))
    if np.isfinite(fs):
        best_score_global = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score_global:.6f} (n={len(best_a_global)})")
    print(f"{'='*50}")

    return [float(x) for x in best_a_global]
