def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for LP: (f*g)[k] for k=0..2n-2."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def spectral_init(n, seed=0):
    """Initialize with flat power spectrum (inverse of uniform |F|^2)."""
    rng = np.random.default_rng(seed)
    # In frequency domain, try to make |F|^2 close to uniform
    # F[k] = sum_j f[j] exp(-2pi i j k / n)
    # We want |F[k]|^2 ~ constant for k=0..n/2
    # This means F[k] ~ exp(i*theta_k) * sqrt(n) for k=1..n/2
    # f = IDFT(F)
    
    # Try random phases
    best_f = None
    best_obj = np.inf
    
    for trial in range(50):
        # Random phases in frequency domain
        nfft = n  # We'll use DFT of length n (circular), then handle linear conv
        phases = rng.uniform(0, 2 * np.pi, nfft // 2 + 1)
        F = np.zeros(nfft // 2 + 1, dtype=complex)
        F[0] = 1.0  # DC component real
        for k in range(1, nfft // 2 + 1):
            F[k] = np.exp(1j * phases[k])
        # Conjugate symmetry
        F_full = np.zeros(nfft, dtype=complex)
        F_full[:nfft//2 + 1] = F
        for k in range(1, nfft // 2 + 1):
            F_full[nfft - k] = np.conj(F[k])
        
        f = np.fft.irfft(F_full, nfft)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:

        f /= s
        obj = fast_convolve(f).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()
    
    if best_f is None:
        best_f = np.ones(n) / n
    return best_f

"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for LP: (f*g)[k] for k=0..2n-2."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def compute_conv_gradient(f, w):
    """Compute gradient of sum_k w[k] * (f*f)[k] w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    conv_len = 2 * n - 1
    for k in range(conv_len):
        wk = w[k]
        if wk < 1e-18:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        ms = np.arange(lo, hi + 1)
        # d/d f[m] of sum_j f[j] f[k-j]: f[m] appears when j=m (gives f[k-m])
        # and when k-j=m (gives f[k-m]). If m == k-m (2m=k), it's f[m]^2 so grad is 2*f[m].
        contrib = f[k - ms]
        # Check for self-term: where 2*ms == k
        self_mask = (2 * ms == k)
        if self_mask.any():
            contrib[contrib < 0] = 0  # safety
        grad[ms] += wk * (contrib + np.where(self_mask, 0.0, contrib))
    return grad


def compute_conv_gradient_fast(f, w, max_peaks=500):
    """Efficient gradient using only significant peaks."""
    n = len(f)
    conv_len = 2 * n - 1
    if conv_len <= max_peaks:
        sig_idx = np.arange(conv_len)
    else:
        sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
        sig_idx = sig_idx[w[sig_idx] > 1e-15 * w.max()]

    grad = np.zeros(n)
    for k in sig_idx:
        wk = w[k]
        if wk < 1e-18:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        ms = np.arange(lo, hi + 1)
        contrib = f[k - ms].copy()
        self_mask = (2 * ms == k)
        if self_mask.any():
            # For self-term, derivative of f[m]^2 is 2*f[m], but we already have f[k-m]=f[m]
            # The total derivative contribution is 2*f[m] for the self-term
            # Our formula: wk * (f[k-ms] + f[k-ms]) = 2*wk*f[k-ms] for all, but for self-term it should be 2*wk*f[m]
            # Actually for non-self: contribution is 2*wk*f[k-m] (from j=m and j=k-m)
            # For self (2m=k): contribution is 2*wk*f[m] (from d/dm of f[m]^2 = 2*f[m])
            # So the formula 2*wk*f[k-m] works in both cases!
            pass
        grad[ms] += 2.0 * wk * contrib
    return grad

        if g is None:

                m_exp *= 1.2

                # Perturb and retry

            continue

        g = np.maximum(g, 0.0)

            no_improve += 1
            if no_improve > 20:
                break
            m_exp = min(m_exp * 1.08, 5.0)

def lse_nesterov(f0, time_limit, beta_start=10.0, beta_end=20000.0, max_peaks=300):
    """Nesterov-accelerated LSE gradient descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 3.0 / n
    v = np.zeros(n)
    x_prev = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov extrapolation
        momentum = 0.90 + 0.09 * min(1.0, it / 200.0)
        y = f + momentum * (f - x_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = f.copy()
            ys = 1.0
        y = y / ys

        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_conv_gradient_fast(y, w, max_peaks=max_peaks)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 100
            break
        grad = grad / grad_norm

        x_prev = f.copy()
        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 8.0 / n)
        else:
            no_improve += 1
            f = f_new  # still move
            if no_improve >= 10:
                lr *= 0.94
            if no_improve > 80:
                f = best_f.copy()
                x_prev = f.copy()
                v = np.zeros(n)
                lr = max(3.0 / n, 0.001)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.003

    return best_f, best_obj

def simulated_anneal(f0, time_limit, T_start=0.005, T_end=0.0001, verbose=False):
    """Simulated annealing for final polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    total_iters = 5000
    rng = np.random.default_rng(123)

    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit - 1:
            break
        T = T_start * (T_end / T_start) ** (it / total_iters)

        # Random perturbation
        num_perturb = rng.integers(1, 5)
        idx = rng.integers(0, n, size=num_perturb)
        delta = rng.standard_normal(num_perturb) * T
        f_trial = f.copy()
        f_trial[idx] += delta
        f_trial = np.maximum(f_trial, 0.0)
        st = f_trial.sum()
        if st < 1e-15:
            continue
        f_trial /= st

        new_obj = fast_convolve(f_trial).max()
        cur_obj = fast_convolve(f).max()
        diff = new_obj - cur_obj

        if diff < 0 or rng.random() < np.exp(-diff / max(T * 0.01, 1e-10)):
            f = f_trial
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_f = f.copy()
                if verbose and it % 500 == 0:
                    print(f"    SA it={it}, obj={best_obj:.12f}, T={T:.6f}")

    return best_f, best_obj


def spectral_init(n, seed=0, n_trials=30):
    """Initialize using random phase in frequency domain."""
    rng = np.random.default_rng(seed)
    best_f = np.ones(n) / n
    best_obj = np.inf

    for trial in range(n_trials):
        # Random phases
        nfft = n
        F = np.zeros(nfft // 2 + 1, dtype=complex)
        F[0] = 1.0
        phases = rng.uniform(0, 2 * np.pi, nfft // 2)
        F[1:] = np.exp(1j * phases) * (0.5 + 0.5 * rng.random(nfft // 2))
        F_full = np.zeros(nfft, dtype=complex)
        F_full[:nfft // 2 + 1] = F
        for k in range(1, nfft // 2 + 1):
            F_full[nfft - k] = np.conj(F[k])
        f = np.fft.irfft(F_full, nfft)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:
            continue
        f /= s
        obj = fast_convolve(f).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()

    return best_f


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

        try_sequence(f_opt, best_tracker)
        f_best = best_tracker[1] if best_tracker[1] is not None else f_best
        n_best = len(f_best)

    # === Phase 4: Active-set CDS ===
    print("\n=== Phase 4: Active-Set CDS ===")
    t4_end = deadline - 250
    f_best = best_tracker[1] if best_tracker[1] is not None else f_init
    t_alloc = t4_end - time.monotonic()
    if t_alloc > 20:
        t_soft = min(300, t_alloc * 0.8)
        print(f"  CDS n={len(f_best)}, budget={t_soft:.0f}s")
        f_opt, _ = active_set_cds(f_best, time_limit=t_soft)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Annealing + Final Polish ===
    print("\n=== Phase 5: Final Polish ===")
    f_best = best_tracker[1] if best_tracker[1] is not None else f_init

    # Simulated annealing
    t_alloc = deadline - time.monotonic() - 60
    if t_alloc > 20:
        t_sa = min(150, t_alloc * 0.6)
        print(f"  Simulated Annealing n={len(f_best)}, t={t_sa:.0f}s")
        f_opt, _ = simulated_anneal(f_best, time_limit=t_sa)
        try_sequence(f_opt, best_tracker)

    # One more LSE pass with very high beta
    f_best = best_tracker[1] if best_tracker[1] is not None else f_init
    t_alloc = deadline - time.monotonic() - 20
    if t_alloc > 15:
        t_lse = min(100, t_alloc * 0.7)
        print(f"  Final LSE (high beta) n={len(f_best)}, t={t_lse:.0f}s")
        f_opt, _ = lse_nesterov(f_best, time_limit=t_lse,
                                 beta_start=500.0, beta_end=100000.0, max_peaks=200)
        try_sequence(f_opt, best_tracker)

    # One more CDS pass
    f_best = best_tracker[1] if best_tracker[1] is not None else f_init
    t_alloc = deadline - time.monotonic() - 10
    if t_alloc > 10:
        t_cds = min(80, t_alloc * 0.8)
        print(f"  Final CDS n={len(f_best)}, t={t_cds:.0f}s")
        f_opt, _ = active_set_cds(f_best, time_limit=t_cds)
        try_sequence(f_opt, best_tracker)

    # === Phase 6: Try nearby lengths ===
    print("\n=== Phase 6: Nearby Lengths ===")
    if best_tracker[1] is not None:
        n_best_final = len(best_tracker[1])
        offsets = [48, -48, 96, -96, 32, -32]
        for off in offsets:
            n_try = n_best_final + off
            if n_try < 150 or n_try > 1200:
                continue
            t_remaining = deadline - time.monotonic() - 10
            if t_remaining < 25:
                break
            t_alloc = min(60, t_remaining * 0.5)
            if t_alloc < 12:
                break
            f_try = interpolate_to_length(best_tracker[1], n_try)
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_opt, _ = lse_nesterov(f_try, time_limit=t_alloc,
                                     beta_start=50.0, beta_end=10000.0, max_peaks=200)
            try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.12f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using parent fallback")
    return [float(x) for x in parent]
