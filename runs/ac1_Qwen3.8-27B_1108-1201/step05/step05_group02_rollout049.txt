"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using smooth max approximation with
projected gradient, exact coordinate descent via golden section, LP-based
Frank-Wolfe, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_obj(a, T):
    """Log-sum-exp smooth approximation of max(conv(a,a))."""
    conv = fast_convolve(a)
    m = conv.max()
    if T < 1e-10:
        return m
    exp_vals = np.exp((conv - m) * T)
    return m + np.log(np.sum(exp_vals)) / T


def smooth_max_grad(a, T):
    """Gradient of smooth max objective w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    m = conv.max()
    if T < 1e-10:
        # Return gradient of the max (subgradient at the argmax)
        peaks = np.where(conv == m)[0]
        weights = np.zeros(len(conv))
        weights[peaks] = 1.0 / len(peaks)
    else:
        exp_vals = np.exp((conv - m) * T)
        weights = exp_vals / exp_vals.sum()
    # grad[j] = 2 * sum_k weights[k] * a[k-j] for valid k-j
    # = 2 * correlate(weights, a, 'valid')[j]
    grad = 2.0 * np.correlate(weights, a, mode='valid')
    return grad


def golden_section_minimize(func, lo, hi, iters=12):
    """Minimize a unimodal function on [lo, hi] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = lo
    b_pt = hi
    c_pt = b_pt - gr * (b_pt - a_pt)
    d_pt = a_pt + gr * (b_pt - a_pt)
    fc = func(c_pt)
    fd = func(d_pt)
    for _ in range(iters):
        if fc < fd:
            b_pt = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b_pt - gr * (b_pt - a_pt)
            fc = func(c_pt)
        else:
            a_pt = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a_pt + gr * (b_pt - a_pt)
            fd = func(d_pt)
    return (a_pt + b_pt) / 2.0


def optimize_coord_golden(a, j, iters=10):
    """Optimize coordinate j using golden section with proportional rescaling."""
    n = len(a)
    old_val = a[j]
    denom = 1.0 - old_val
    if denom < 1e-12:
        return a.copy(), a.copy()

    def obj(v):
        if v < 0.0 or v > 1.0 - 1e-12:
            return 1e10
        scale = (1.0 - v) / denom
        if scale < 0.0:
            return 1e10
        a_new = a.copy()
        a_new[j] = v
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        a_new[mask] *= scale
        if np.any(a_new < -1e-14):
            return 1e10
        conv = fast_convolve(a_new)
        return conv.max()

    # Narrow search range
    spread = max(0.005, old_val * 0.3)
    lo = max(0.0, old_val - spread)
    hi = min(1.0 - 1e-10, old_val + spread)

    opt_v = golden_section_minimize(obj, lo, hi, iters)

    scale = (1.0 - opt_v) / denom
    a_new = a.copy()
    a_new[j] = opt_v
    mask = np.ones(n, dtype=bool)
    mask[j] = False
    a_new[mask] *= scale
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), a.copy()
    a_new /= s
    return a_new, obj(opt_v)

    n = len(a)

    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1

        cur_max = conv.max()

        coords = list(important)

def projected_gradient_opt(a0, time_limit, T_start=50.0, T_end=5000.0,
                           lr=0.001, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)

    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    total_iters = 0
    cur_T = T_start
    cur_lr = lr
    no_improve = 0
    rng = np.random.default_rng(67890)

    while time.monotonic() - t0 < time_limit:
        total_iters += 1

        # Gradually increase T

        cur_lr = lr * (1.0 - 0.5 * progress)

            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            a_new = project_simplex(a - 0.5 * cur_lr * grad)
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0

        a = a_new

        # Periodic restart from best
        if no_improve > 100:
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(best_a + noise)
            no_improve = 0
            cur_lr = max(lr * 0.3, 0.0001)

        if verbose and total_iters % 5000 == 0:
            print(f"    PG iter {total_iters}: obj={2*n*best_obj:.10f}, T={cur_T:.0f}")

    return best_a, 2.0 * n * best_obj


def lp_fw_optimize(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with min-peak LP."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0 if 'f0' in dir() else a0)) / len(a0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(424242)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: min t s.t. (f*g)[k] <= t, g >= 0, sum(g) = 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        n_constraints = 2 * n - 1
        A_ub = np.zeros((n_constraints, n + 1))
        for j in range(n):
            A_ub[j:j + n, j] = f
        A_ub[:, -1] = -1.0
        b_ub = np.zeros(n_constraints)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(None, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'time_limit': max(1, time_limit - (time.monotonic() - t0) - 5)})
        except Exception:
            break

        if not res.success:
            break

        g = np.maximum(res.x[:n], 0.0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = best_val
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            M = fast_convolve(f).max()
            no_improve = 0

    return best_f, 2.0 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization using smooth projected gradient,
    exact coordinate descent, LP Frank-Wolfe, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Smooth projected gradient on parent ===
    print(f"\n=== Phase 1: Projected Gradient (n={len(parent)}, best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.25)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = projected_gradient_opt(parent, time_limit=t1,
                                           T_start=100.0, T_end=8000.0, lr=0.002)
        try_seq(a_opt, "PG_main")

    # === Phase 2: Coordinate descent refinement ===
    print(f"\n=== Phase 2: Coord Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(400, rem * 0.25)
        if t2 > 30:
            print(f"  n={len(best_seq)}, Time: {t2:.0f}s")
            a_opt, _ = coord_descent(best_seq, time_limit=t2)
            try_seq(a_opt, "CD_main")

    # === Phase 3: LP Frank-Wolfe ===
    print(f"\n=== Phase 3: LP FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        n_best = len(best_seq)
        # Only do LP if sequence isn't too large (LP is O(n^2) in size)
        if n_best <= 1200:
            t3 = min(300, rem * 0.2)
            if t3 > 30:
                print(f"  n={n_best}, Time: {t3:.0f}s")
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3)
                try_seq(a_opt, "LPFW")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        base_n = len(base)
        candidate_lengths = []
        for delta in [-64, -32, 32, 64, 128, 256, -128, -256]:
            ln = base_n + delta
            if 256 <= ln <= 4096 and ln != base_n:
                candidate_lengths.append(ln)
        # Also try some absolute lengths
        for ln in [512, 768, 1024, 1536, 2048]:
            if ln != base_n and 256 <= ln <= 4096:
                candidate_lengths.append(ln)

        candidate_lengths = sorted(set(candidate_lengths))

        for target_n in candidate_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.25, 150)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  n={target_n}, Time: {t_alloc:.0f}s")

            # Quick PG
            a_opt, _ = projected_gradient_opt(f_interp, time_limit=t_alloc * 0.5,
                                               T_start=50.0, T_end=3000.0, lr=0.002)
            try_seq(a_opt, f"PG_n{target_n}")

            rem = deadline - time.monotonic()
            if rem > 30:
                a_opt, _ = coord_descent(a_opt, time_limit=min(rem * 0.5, 80))
                try_seq(a_opt, f"CD_n{target_n}")

    # === Phase 5: Diverse restarts on best ===
    print(f"\n=== Phase 5: Diverse restarts (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for restart in range(4):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem * 0.3, 120)
            if t_alloc < 15:
                break

            # Different perturbation scales
            scales = [0.005, 0.01, 0.02, 0.003]
            noise = rng.standard_normal(n_best) * scales[restart]
            f_pert = project_simplex(best_seq + noise)

            a_opt, _ = projected_gradient_opt(f_pert, time_limit=t_alloc * 0.6,
                                               T_start=200.0, T_end=6000.0, lr=0.001)
            try_seq(a_opt, f"restart_{restart}_pg")

            rem = deadline - time.monotonic()
            if rem > 20:
                a_opt, _ = coord_descent(a_opt, time_limit=min(rem * 0.4, 60))
                try_seq(a_opt, f"restart_{restart}_cd")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_cd = min(rem * 0.5, 100)
            a_opt, _ = coord_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "final_cd")

        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_pg = min(rem * 0.5, 60)
            a_opt, _ = projected_gradient_opt(best_seq, time_limit=t_pg,
                                               T_start=3000.0, T_end=10000.0, lr=0.0005)
            try_seq(a_opt, "final_pg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
