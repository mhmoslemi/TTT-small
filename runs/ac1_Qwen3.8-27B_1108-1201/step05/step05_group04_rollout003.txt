        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 8.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 13 + 11)
            scale = 0.003 + 0.004 * min(1.0, (no_improve - 50) / 60.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def chirp_init(n, rate=1.0, amplitude=0.25):
    """Symmetric chirp-modulated uniform initialization."""
    base = np.ones(n) / n
    x = np.arange(n) / n
    chirp = np.abs(np.cos(np.pi * rate * x ** 2))
    # Make symmetric: take first half and mirror
    half = n // 2
    c_half = chirp[:half]
    chirp_sym = np.concatenate([c_half, c_half[::-1]])
    if len(chirp_sym) > n:
        chirp_sym = chirp_sym[:n]
    elif len(chirp_sym) < n:
        chirp_sym = np.concatenate([chirp_sym, np.zeros(n - len(chirp_sym))])
    f = base + amplitude * (chirp_sym - chirp_sym.mean())
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f


def cosine_init(n, freq=3.0, amplitude=0.15):
    """Symmetric cosine-modulated uniform initialization."""
    base = np.ones(n) / n
    half = n // 2
    x = np.arange(half)
    c = np.cos(2.0 * np.pi * freq * x / n)
    cos_sym = np.concatenate([c, c[::-1]])
    if len(cos_sym) > n:
        cos_sym = cos_sym[:n]
    elif len(cos_sym) < n:
        cos_sym = np.concatenate([cos_sym, np.zeros(n - len(cos_sym))])
    f = base + amplitude * cos_sym
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f


def random_smooth_init(n, seed=0, smooth=5):
    """Random smooth non-negative initialization."""
    rng = np.random.default_rng(seed)
    f = rng.standard_normal(n)
    kernel = np.ones(smooth) / smooth
    for _ in range(3):
        f = np.convolve(f, kernel, mode='same')
    f = np.abs(f)
    f /= f.sum()
    return f


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
