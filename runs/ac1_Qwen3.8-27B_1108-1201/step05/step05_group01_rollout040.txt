"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using cutting-plane LP-Frank-Wolfe with
asymmetric ramp initializations, multi-length exploration, and adaptive constraint
management. Key insight: asymmetric sequences (like ramps) achieve lower peak
autocorrelation than symmetric ones, enabling突破 of the symmetric local optimum.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def ramp_init(n, power=1.0):
    """Create a power-law ramp: a_j = (j+1)^power normalized."""
    j = np.arange(n, dtype=np.float64) + 1.0
    a = j ** power
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def cosine_init(n, freq, amp, phase=0.0):
    """Cosine-modulated uniform initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def ramp_cosine_init(n, power=1.0, freq=1.0, amp=0.3):
    """Ramp modulated by cosine envelope."""
    j = np.arange(n, dtype=np.float64) + 1.0
    base = j ** power
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = base * mod
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def build_constraint_row(f, k, n):
    """Build row of A_ub for constraint (f*g)[k] <= M."""
    row = np.zeros(n)
    m_lo = max(0, k - n + 1)
    m_hi = min(n - 1, k)
    if m_lo <= m_hi:
        row[m_lo:m_hi + 1] = f[k - m_hi:k - m_lo + 1][::-1]
    return row


def solve_lp_direction(f, M, active_lags, n, max_active=150):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active lags, g >= 0."""
    # Sort by violation and take top max_active
    if len(active_lags) > max_active:
        conv = fast_convolve(f)
        sorted_lags = sorted(active_lags, key=lambda k: -conv[k])
        active_lags = sorted_lags[:max_active]
    
    if len(active_lags) == 0:
        # If no constraints, just return uniform
        return np.ones(n) / n
    
    A_rows = [build_constraint_row(f, k, n) for k in active_lags]
    A_ub = np.array(A_rows)
    b_ub = np.full(len(active_lags), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

            hi = min(1.0, best_alpha + 0.15)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)

            m_relax = 1.0

        if no_improve > 80:
            rng = np.random.default_rng(hash(no_improve + it) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            conv = fast_convolve(f)
            M = conv.max()
            num_active = max(30, min(max_active, len(conv) // 3))
            active_lags = list(np.argpartition(conv, -num_active)[-num_active:])
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr0=0.003, mom0=0.85, top_k=20):
    """Projected subgradient with Nesterov momentum on max-conv objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # Nesterov variables
    y = f.copy()
    v = np.zeros(n)
    cur_lr = lr0
    cur_mom = mom0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 1000 == 0 and time.monotonic() - t0 > 5:
            break

        # Compute subgradient from active peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        # Softmax weighting
        scaled = (pv - pv.max()) / max(pv.max() * 0.1, 1e-12)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted gradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * w[i] * f[k - indices]

        # Nesterov update
        v = cur_mom * v - cur_lr * sg
        y_new = y + v
        y_new = project_simplex(y_new)
        f_new = project_simplex(y_new)
        
        # Update momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * cur_mom * cur_mom * it)) / 2.0
        y = f_new + (1.0 - 1.0 / beta) * (f_new - f)
        y = project_simplex(y)
        f = f_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        cur_lr *= 0.9998
        cur_mom = min(0.97, cur_mom * 1.0001)

        if no_improve > 500:
            rng = np.random.default_rng(time.monotonic_ns() % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.7, 0.0005)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent(f0, time_limit, max_passes=10):
    """Simple cyclic coordinate descent for final polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR

    passes = 0
    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Golden section on f[j]
            lo = 0.0
            hi = min(0.999, 2.5 * old_val)

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            mask = np.ones(n, dtype=bool)
            mask[j] = False

            def eval_coord(x):
                ft = f.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                ft[mask] = f[mask] * (ft2 / denom)
                return fast_convolve(ft).max()

            f1 = eval_coord(c1)
            f2 = eval_coord(c2)

            for _ in range(10):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_coord(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_coord(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = fast_convolve(f).max()

            if opt_val < cur_val - 1e-14:
                ft = f.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                ft[mask] = f[mask] * (ft2 / denom)
                ft = ft / ft.sum()
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()

        if not pass_improved:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: asymmetric init search -> cutting-plane LP-FW ->
    multi-length exploration -> subgradient refinement -> coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        # Clip to valid range
        a = np.minimum(a, 1000.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # ---- Get parent ----
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_norm = parent / parent.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    n_parent = len(parent) if parent is not None else 768

    # ---- Phase 0: Quick evaluation of structured initializations ----
    print("=== Phase 0: Structured initializations ===")
    t_start = time.monotonic()

    init_list = []
    # Try parent normalized
    if parent is not None:
        init_list.append((parent_norm, "parent"))
    
    # Ramps at parent length
    for power in [0.5, 1.0, 1.5, 2.0]:
        init_list.append((ramp_init(n_parent, power), f"ramp_p{power}_n{n_parent}"))
    
    # Ramp-cosine combinations
    for power in [0.5, 1.0]:
        for freq in [1, 2, 3]:
            for amp in [0.2, 0.4, 0.6]:
                init_list.append((ramp_cosine_init(n_parent, power, freq, amp),
                                  f"rc_p{power}_f{freq}_a{amp}"))
    
    # Cosine inits
    for freq in [1, 2, 3, 4]:
        for amp in [0.2, 0.4, 0.6]:
            init_list.append((cosine_init(n_parent, freq, amp), f"cos_f{freq}_a{amp}_n{n_parent}"))

    # Ramp at different lengths
    for n_try in [512, 768, 1024, 1536, 2048]:
        if n_try <= 4096:
            for power in [0.5, 1.0, 1.5]:
                init_list.append((ramp_init(n_try, power), f"ramp_p{power}_n{n_try}"))

    # Evaluate all initializations
    for init, label in init_list:
        if time.monotonic() > deadline - 200:
            break
        s = try_seq(init, label)

    print(f"  Phase 0 done: best={best_score:.6f}, time={time.monotonic()-t_start:.1f}s")

    # ---- Phase 1: LP-FW on best initializations ----
    print("\n=== Phase 1: LP-FW optimization ===")
    
    # Sort candidate starting points by score and try top few
    # First, optimize from the best found
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.5, 300)
            print(f"  LP-FW on best (n={len(best_seq)}), t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_cutting(best_seq, time_limit=t_alloc, max_active=150, verbose=True)
            try_seq(a_opt, "lpfw_best1")

    # Try from ramp initializations at various lengths
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            t_alloc = min(rem * 0.3, 120)
            n_use = len(best_seq)
            a_ramp = ramp_init(n_use, 1.0)
            a_opt, _ = lp_fw_cutting(a_ramp, time_limit=t_alloc, max_active=150)
            try_seq(a_opt, "lpfw_ramp")

    # ---- Phase 2: Multi-length exploration ----
    print(f"\n=== Phase 2: Multi-length (best={best_score:.6f}) ===")
    
    if best_seq is not None:
        for target_n in [1024, 1280, 1536, 2048, 2560]:
            if target_n > 3072:
                break
            rem = deadline - time.monotonic() - 300
            if rem < 25:
                break
            f_interp = interpolate_seq(best_seq, target_n)
            t_alloc = min(rem * 0.25, 100)
            print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_cutting(f_interp, time_limit=t_alloc, max_active=120)
            s = try_seq(a_opt, f"lpfw_N{target_n}")
            
            # Also try ramp at this length
            rem = deadline - time.monotonic() - 300
            if rem > 20 and np.isfinite(s):
                a_r = ramp_init(target_n, 1.0)
                t_alloc2 = min(rem * 0.15, 50)
                a_opt2, _ = lp_fw_cutting(a_r, time_limit=t_alloc2, max_active=120)
                try_seq(a_opt2, f"lpfw_ramp_N{target_n}")

    # ---- Phase 3: Subgradient refinement ----
    print(f"\n=== Phase 3: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_ns = min(rem * 0.4, 150)
            print(f"  Subgrad t={t_ns:.0f}s, n={n_best}")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_ns,
                                             lr0=0.005, mom0=0.85,
                                             top_k=max(15, n_best // 5))
            try_seq(a_opt, "subgrad")
            
            # Second pass
            rem = deadline - time.monotonic() - 150
            if rem > 15:
                a_opt2, _ = subgradient_optimize(a_opt, time_limit=min(rem, 80),
                                                  lr0=0.002, mom0=0.90,
                                                  top_k=max(10, n_best // 8))
                try_seq(a_opt2, "subgrad_fine")

    # ---- Phase 4: More LP-FW with different starts ----
    print(f"\n=== Phase 4: Multi-start LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        starts = []
        # From best
        starts.append(best_seq.copy())
        # Perturbed
        for scale in [0.005, 0.01, 0.02]:
            starts.append(project_simplex(best_seq + rng.standard_normal(n_best) * scale))
        # Blend with ramp
        starts.append(0.7 * best_seq + 0.3 * ramp_init(n_best, 1.0))
        starts[-1] = project_simplex(starts[-1])
        # Blend with cosine
        starts.append(0.8 * best_seq + 0.2 * cosine_init(n_best, 2, 0.3))
        starts[-1] = project_simplex(starts[-1])
        # Reverse blend
        starts.append(0.6 * best_seq + 0.4 * best_seq[::-1])
        starts[-1] = project_simplex(starts[-1])

        for i, f_init in enumerate(starts):
            rem = deadline - time.monotonic() - 100
            if rem < 10:
                break
            t_alloc = min(rem * 0.25, 80)
            a_opt, _ = lp_fw_cutting(f_init, time_limit=t_alloc, max_active=130)
            try_seq(a_opt, f"multi_{i}")

    # ---- Phase 5: Coordinate descent polish ----
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_cd = min(rem * 0.5, 100)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, max_passes=5)
            try_seq(a_opt, "coord_desc")

    # ---- Phase 6: Final polish ----
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Short LP-FW
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = lp_fw_cutting(best_seq, time_limit=min(rem, 50), max_active=150)
            try_seq(a_opt, "final_lpfw")
        
        # Tiny perturbations
        rem = deadline - time.monotonic() - 15
        n_best = len(best_seq)
        for _ in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            a_opt, _ = lp_fw_cutting(f_pert, time_limit=min(rem, 20), max_active=100)
            try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = ramp_init(768, 1.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: score={best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
