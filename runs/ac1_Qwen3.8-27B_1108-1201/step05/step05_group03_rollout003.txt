    candidates.append(rand)
    
    # Power law
    power = (t + 0.01) ** (-0.3)
    candidates.append(power / power.sum())
    
    # Best of all
    best_c = None
    best_v = np.inf
    for c in candidates:
        c = np.maximum(c, 0)
        s = c.sum()
        if s < 1e-15:
            continue
        c = c / s
        v = fast_convolve(c).max()
        if v < best_v:
            best_v = v
            best_c = c.copy()
    
    return best_c, best_v


def lp_solve(f, M, n, x0=None):
    """Solve the LP for FW direction with optional warm start."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    options = {'presolve': True, 'time_limit': 8}
    if x0 is not None:
        options['presolve'] = True
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options=options)
    if res.success:
        return np.maximum(res.x, 0), res
    return None, res


def line_search(f, g, M):
    """Combined coarse + golden section line search."""
    best_alpha = 0.0
    best_val = M
    
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha > 0.02 and best_alpha < 0.97:
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        gr = (np.sqrt(5) - 1) / 2
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        
        ft1 = np.maximum(f + c1 * (g - f), 0)
        ft1 = ft1 / max(ft1.sum(), 1e-15)
        fc1 = fast_convolve(ft1).max()
        ft2 = np.maximum(f + c2 * (g - f), 0)
        ft2 = ft2 / max(ft2.sum(), 1e-15)
        fc2 = fast_convolve(ft2).max()
        
        for _ in range(9):
            if hi - lo < 1e-7:
                break
            if fc1 < fc2:
                hi = c2; c2 = c1; fc2 = fc1
                c1 = hi - gr * (hi - lo)
                ft1 = np.maximum(f + c1 * (g - f), 0)
                ft1 = ft1 / max(ft1.sum(), 1e-15)
                fc1 = fast_convolve(ft1).max()
            else:
                lo = c1; c1 = c2; fc1 = fc2
                c2 = lo + gr * (hi - lo)
                ft2 = np.maximum(f + c2 * (g - f), 0)
                ft2 = ft2 / max(ft2.sum(), 1e-15)
                fc2 = fast_convolve(ft2).max()
        
        cand = (lo + hi) / 2
        ft = np.maximum(f + cand * (g - f), 0)
        ft = ft / max(ft.sum(), 1e-15)
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = cand
    
    return best_alpha, best_val


def lp_fw_optimize(f0, time_limit, verbose=False, n_restarts=10):
    """LP-based Frank-Wolfe with restarts and dual-guided perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    prev_g = None
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        g, lp_res = lp_solve(f, M, n, x0=prev_g)
        if g is None:
            g, lp_res = lp_solve(f, M * 1.01, n)
            if g is None:
                g, lp_res = lp_solve(f, M * 1.05, n)
            
            if g is None:
                no_improve += 1
                if no_improve > 40:
                    # Aggressive restart
                    if it > 200 and no_improve > 80:
                        break
                    rng = np.random.default_rng(it * 7919 + 13)
                    conv_cur = fast_convolve(f)
                    peak_idx = np.argmax(conv_cur)
                    center = min(peak_idx, n - 1)
                    w = n // (4 + (no_improve % 4))
                    mask = np.zeros(n)
                    mask[max(0, center - w):min(n, center + w)] = 1.0
                    scale = 0.003 + 0.002 * min(no_improve // 20, 4)
                    pert = rng.standard_normal(n) * scale * (0.5 + mask)
                    f = project_simplex(best_f + pert)
                    conv = fast_convolve(f)
                    M = conv.max()
                    no_improve = 0
                    prev_g = None
                    continue
                continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        g = g / g_sum
        prev_g = g.copy()
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 25:
                rng = np.random.default_rng(it * 331 + 7)
                scale = 0.004 * min(1 + no_improve / 10, 5)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                prev_g = None
            continue
        
        best_alpha, best_val = line_search(f, g, M)
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP-FW iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Major restart
            rng = np.random.default_rng(it * 104729 + 3)
            # Try different perturbation scales
            for scale_try in [0.005, 0.01, 0.02, 0.03]:
                f_try = project_simplex(best_f + rng.standard_normal(n) * scale_try)
                conv_try = fast_convolve(f_try)
                if conv_try.max() < M:
                    f = f_try
                    M = conv_try.max()
                    break
            else:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                conv = fast_convolve(f)
                M = conv.max()
            no_improve = 0
            prev_g = None
    
    return best_f, best_obj


def nesterov_subgrad(f0, time_limit, verbose=False):
    """Nesterov-accelerated subgradient with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    y = f.copy()  # Nesterov momentum point
    beta = 0.9  # momentum
    it = 0
    no_improve = 0
    
    # Temperature schedule
    T_start = 3.0 / n
    T_end = 0.3 / n
    lr_start = 0.008
    lr_end = 0.001
    
    max_iters = int(time_limit / 0.01)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / (max_iters * 0.85))
        T = T_start * (T_end / T_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        beta_t = (1 + it) ** (-0.5)  # Nesterov step size for proximal
        # Actually use constant momentum for stability
        beta_t = beta
        
        conv_y = fast_convolve(y)
        
        # Top-k weighted subgradient
        k = max(10, int(n * 0.05) * (1 - progress) + 5)
        k = min(k, 2 * n - 1)
        peaks = np.argpartition(conv_y, -k)[-k:]
        pv = conv_y[peaks]
        scaled = T * (pv - pv.max())
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[kk - idx]
        
        # Update
        y_new = project_simplex(y - lr * sg)
        f_new = f + (beta_t - 1) * (y_new - f)
        f_new = project_simplex(f_new)
        
        # Actually proper Nesterov: y = f_prev + beta_t * (f_curr - f_prev)
        # Let me use a simpler but effective scheme
        f_trial = project_simplex(f - lr * sg)
        
        new_M = fast_convolve(f_trial).max()
        
        if new_M < best_obj - 1e-14:
            y = f_trial.copy()
            f = f_trial
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Nesterov iter {it}: obj={best_obj:.10f}, T={T:.6f}")
        else:
            no_improve += 1
            if no_improve > 500:
                rng = np.random.default_rng(it * 41 + 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                y = f.copy()
                no_improve = 0
    
    return best_f, best_obj


def pairwise_optimize(f0, time_limit, verbose=False):
    """Peak-focused pairwise coordinate optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    no_improve_rounds = 0
    round_num = 0
    
    while time.monotonic() - t0 < time_limit:
        round_num += 1
        round_improved = False
        
        # Get current peaks to focus optimization
        conv_cur = fast_convolve(f)
        n_peaks = max(5, n // 10)
        peak_lags = np.argpartition(conv_cur, -n_peaks)[-n_peaks:]
        
        # Build a set of (i,j) pairs to try, focused around peak lags
        pairs = []
        for lag in peak_lags:
            lag = int(lag)
            js = max(0, lag - n + 1)
            je = min(n - 1, lag)
            for j in range(js, je + 1):
                i = lag - j
                if i != j and i >= 0 and i < n and j >= 0 and j < n:
                    pairs.append((min(i, j), max(i, j)))
        
        # Also add random pairs
        for _ in range(n):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        
        # Shuffle
        pairs = list(set(pairs))
        rng.shuffle(pairs)
        
        # Limit pairs per round
        max_pairs = min(len(pairs), 2000)
        
        for idx in range(max_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            i, j = pairs[idx]
            c = f[i] + f[j]
            if c < 1e-12:
                continue
            
            cur_obj = best_obj
            best_t = f[i] / c
            best_val = cur_obj
            
            candidates = [best_t, 0.0, 1.0, 0.1, 0.3, 0.5, 0.7, 0.9]
            if 0.1 < best_t < 0.9:
                candidates.extend([best_t - 0.05, best_t + 0.05,
                                    best_t - 0.1, best_t + 0.1])
            
            for t in candidates:
                t = np.clip(t, 0.0, 1.0)
                f_trial = f.copy()
                f_trial[i] = t * c
                f_trial[j] = (1 - t) * c
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-13:
                    best_val = val
                    best_t = t
            
            if best_val < best_obj - 1e-12:
                f[i] = best_t * c
                f[j] = (1 - best_t) * c
                best_obj = best_val
                best_f = f.copy()
                round_improved = True
        
        if round_improved:
            no_improve_rounds = 0
            if verbose and round_num % 5 == 0:
                print(f"    Pairwise round {round_num}: obj={best_obj:.10f}")
        else:
            no_improve_rounds += 1
            if no_improve_rounds >= 5:
                break
            # Perturb to escape
            rng_pert = np.random.default_rng(round_num * 7 + 3)
            f = project_simplex(best_f + rng_pert.standard_normal(n) * 0.003)
            best_obj = fast_convolve(f).max()
            best_f = f.copy()
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using FFT-based method."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    
    # Pad with zeros and resample
    fft_size = 1 << (max(old_n, new_n) * 2 - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    # Low-pass filter for downsampling or just resample
    # Simple approach: use interpolation in the "time" domain
    if new_n > old_n:
        # Upsample: insert zeros in frequency domain
        new_fft_size = 1 << (new_n * 2 - 1).bit_length()
        new_F = np.zeros(new_fft_size // 2 + 1, dtype=complex)
        copy_len = min(len(F), len(new_F))
        new_F[:copy_len] = F[:copy_len]
        x = np.fft.irfft(new_F, new_fft_size)[:new_n]
    else:
        # Downsample: keep low frequencies
        new_fft_size = 1 << (new_n * 2 - 1).bit_length()
        new_F = np.zeros(new_fft_size // 2 + 1, dtype=complex)
        copy_len = min(len(F), len(new_F))
        new_F[:copy_len] = F[:copy_len]
        x = np.fft.irfft(new_F, new_fft_size)[:new_n]
    
    x = np.maximum(x, 0)
    s = x.sum()
    if s < 1e-15:
        # Fallback to linear interpolation
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, new_n)
        x = np.interp(new_x, old_x, f)
        x = np.maximum(x, 0)
        s = x.sum()
    if s > 1e-15:
        x = x / s
    else:
        x = np.ones(new_n) / new_n
    return x

            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.001)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=min(rem, 15))
            try_seq(a_opt, f"perturb_{trial}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
