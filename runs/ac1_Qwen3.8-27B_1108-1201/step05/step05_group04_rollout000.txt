def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_gradient(f, k):
    """Exact gradient of (f*f)[k] w.r.t. f.
    Result: g[m] = 2*f[k-m] if 0<=k-m<n, else 0.
    """
    n = len(f)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * f[k - idx]
    return g


def multi_peak_gradient(f, conv, top_k=5, weight_mode='exp', temperature=5.0):
    """Gradient targeting top-k peaks with adaptive weights."""
    n = len(f)
    conv_len = len(conv)
    k_sel = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_sel)[-k_sel:]
    pv = conv[peaks]
    
    if weight_mode == 'exp':
        scaled = pv - pv.max()
        w = np.exp(scaled * temperature / max(pv.max(), 1e-15))
    elif weight_mode == 'uniform':
        w = np.ones(k_sel) / k_sel
    else:  # 'linear'
        w = np.sort(pv)[::-1]
    
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(k_sel)
        w[0] = 1.0
    else:
        w /= w_sum
    
    g = np.zeros(n)
    for i, k in enumerate(peaks):
        g += w[i] * peak_gradient(f, int(k))
    return g


def lse_gradient(f, conv, beta, max_peaks=300):
    """LSE-smoothed gradient of max(conv)."""
    n = len(f)
    conv_len = len(conv)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Only use significant weights
    threshold = w.max() * 1e-6
    active = w > threshold
    if not active.any():
        return np.zeros(n)
    
    g = np.zeros(n)
    active_idx = np.where(active)[0]
    for k in active_idx:
        wk = w[k]
        if wk < 1e-15:
            continue
        g += wk * peak_gradient(f, int(k))
    return g


def build_lp_matrix(f):
    """Build sparse constraint matrix A for LP: (A@g)[k] = (f*g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

def symmetric_init(n, freq=2.0, amplitude=0.05):
    """Create a symmetric sinusoidal perturbation of uniform."""
    base = np.ones(n) / n
    x = np.arange(n)
    sym = np.cos(2.0 * np.pi * freq * x / n)
    sym_first = sym[:n // 2]
    if n % 2 == 1:
        sym_full = np.concatenate([sym_first, [0.0], sym_first[::-1]])
    else:
        sym_full = np.concatenate([sym_first, sym_first[::-1]])
    f = base + amplitude * sym_full
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f

    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current)
    
    t_remaining = deadline - time.monotonic()
    if t_remaining > 30:
        # Very high beta LSE
        t_alloc = min(100, t_remaining * 0.5)
        if t_alloc > 15:
            print(f"  High-beta LSE t={t_alloc:.0f}s")
            f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                     beta_start=500.0, beta_end=20000.0, max_peaks=500)
            try_sequence(f_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else current
    t_remaining = deadline - time.monotonic()
    if t_remaining > 20:
        # Peak gradient with very small steps
        t_alloc = min(80, t_remaining * 0.6)
        if t_alloc > 10:
            print(f"  Fine peak grad t={t_alloc:.0f}s")
            f_opt, _ = peak_grad_optimize(current, time_limit=t_alloc,
                                           lr_init=0.0005, momentum=0.96, top_k=3, temp=15.0)
            try_sequence(f_opt, best_tracker)

    # Final quick LSE
    current = best_tracker[1] if best_tracker[1] is not None else current
    t_remaining = deadline - time.monotonic()
    if t_remaining > 15:
        t_alloc = min(60, t_remaining)
        if t_alloc > 5:
            f_opt, _ = lse_optimize(current, time_limit=t_alloc,
                                     beta_start=200.0, beta_end=8000.0, max_peaks=300)
            try_sequence(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(448) / 448
    return [float(x) for x in fallback]
