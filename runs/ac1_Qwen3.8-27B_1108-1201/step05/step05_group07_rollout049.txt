"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved multiplicative-weights
subgradient method with Adam-style per-coordinate adaptive learning rates,

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fast(a, conv, top_k, temperature):
    """Compute softmax-weighted subgradient of max(conv(a,a)) - vectorized."""
    n = len(a)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    if k_actual <= 0:
        return np.zeros(n)

    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * a[k - idx]
    return 2.0 * sg


def adam_mu_optimize(f0, time_limit, top_k_init=40, temp_init=0.00003,
                     lr_init=0.004, beta1=0.88, beta2=0.95, eps=1e-8,

    n = len(f)

    best_f = f.copy()

    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr_init
    cur_temp = temp_init
    cur_top_k = top_k_init

        # Compute subgradient

        # Adam update

        v = beta2 * v + (1.0 - beta2) * sg * sg

            no_improve = 0

        cur_top_k *= top_k_decay

def coordinate_descent_polish(f0, time_limit, max_iter=100):
    """Coordinate descent: optimize one coordinate at a time while rescaling others."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Order: try to process high-value coordinates first
    order = np.argsort(-f)

    for _ in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            old_val = f[j]
            if old_val < 1e-10:
                continue

            cur_conv = fast_convolve(f)
            cur_peak = cur_conv.max()
            best_val = old_val
            best_obj_j = cur_peak

            # Try different values for coordinate j
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st

                val = fast_convolve(f_trial).max()
                if val < best_obj_j - 1e-14:
                    best_obj_j = val
                    best_val = new_val

            if best_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    f[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_obj_j < best_obj - 1e-13:
                    best_obj = best_obj_j
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def pair_swap_search(f0, time_limit, n_trials_per_round=300, max_rounds=50):
    """Targeted pair-swap local search focused on peak-reducing moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for round_i in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        conv = fast_convolve(f)
        # Focus on top peaks
        top_k = min(20, 2 * n - 1)
        top_peaks = np.argpartition(conv, -top_k)[-top_k:]

        found = False
        for _ in range(n_trials_per_round):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick a peak to reduce
            k = int(top_peaks[rng.integers(0, len(top_peaks))])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[k - idx]
            total = contribs.sum()
            if total < 1e-15:
                continue

            # Transfer from high contributor to a random other position
            hi_local = np.argmax(contribs)
            hi = int(idx[hi_local])
            # Pick a random destination (prefer low-value positions)
            lo = int(rng.integers(0, n))
            if lo == hi:
                continue
            if f[lo] > f[hi]:
                continue

            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = f[hi] * frac
                f[hi] -= amt
                f[lo] += amt
                val = fast_convolve(f).max()
                if val < best_obj - 1e-13:
                    best_obj = val
                    best_f = f.copy()
                    found = True
                    break
                else:
                    f[hi] += amt
                    f[lo] -= amt
                    break

        if not found:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: Adam-MU subgradient with adaptive rates,
    multi-length exploration, asymmetric restarts, and local search polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent found, using uniform n=768")

    n_parent = len(parent)

    # === Phase 1: Fine-tune parent with Adam-MU ===
    print("\n=== Phase 1: Adam-MU fine-tuning of parent ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem * 0.4, 200)
        print(f"  t={t_alloc:.0f}s")
        a_opt, obj = adam_mu_optimize(
            parent, time_limit=t_alloc,
            top_k_init=max(20, n_parent // 10),
            temp_init=0.00002, lr_init=0.004,
            beta1=0.90, beta2=0.95,
            temp_decay=0.004, lr_decay=0.9996, top_k_decay=0.999
        )
        try_seq(a_opt, "adam_parent")

    # === Phase 2: Asymmetric variant ===
    print(f"\n=== Phase 2: Asymmetric variants (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    for asym_type in ["tilt", "shift", "skew"]:
        rem = deadline - time.monotonic() - 300
        if rem < 20:
            break
        t_alloc = min(rem * 0.15, 80)

        if asym_type == "tilt":
            tilt = np.linspace(0.5, 1.5, n_base)
            f_init = project_simplex(base * tilt)
        elif asym_type == "shift":
            # Circular shift by n/10
            shift_amt = n_base // 10
            f_init = np.roll(base, shift_amt)
            # Smooth the discontinuity
            kernel = np.ones(5) / 5
            f_init = np.convolve(f_init, kernel, mode='same')
            f_init = np.maximum(f_init, 0)
            f_init = f_init / f_init.sum()
        else:  # skew
            x = np.arange(n_base) / n_base
            skew = 1.0 + 0.3 * np.sin(np.pi * x)
            f_init = project_simplex(base * skew)

        print(f"  {asym_type}: t={t_alloc:.0f}s")
        a_opt, _ = adam_mu_optimize(
            f_init, time_limit=t_alloc,
            top_k_init=max(20, n_base // 8),
            temp_init=0.00003, lr_init=0.003,
            beta1=0.87, beta2=0.94,
            temp_decay=0.005, lr_decay=0.9995, top_k_decay=0.998
        )
        try_seq(a_opt, f"asym_{asym_type}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent

    target_lengths = [640, 896, 1024, 1280, 1536]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 25:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.2, 100)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = adam_mu_optimize(
            f_interp, time_limit=t_alloc,
            top_k_init=max(20, target_n // 12),
            temp_init=0.00003, lr_init=0.003,
            beta1=0.89, beta2=0.95,
            temp_decay=0.004, lr_decay=0.9996, top_k_decay=0.999
        )
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 4: Local search polishing ===
    print(f"\n=== Phase 4: Local search (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_ps = min(rem * 0.3, 60)
        print(f"  Pair swap: t={t_ps:.0f}s")
        a_opt, _ = pair_swap_search(base, time_limit=t_ps)
        try_seq(a_opt, "pair_swap")

    rem = deadline - time.monotonic() - 60
    if rem > 15:
        t_cd = min(rem * 0.5, 50)
        print(f"  Coord descent: t={t_cd:.0f}s")
        a_opt, _ = coordinate_descent_polish(base, time_limit=t_cd)
        try_seq(a_opt, "coord_desc")

    # === Phase 5: Final fine-tuning with very low temp ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    # Very fine subgradient
    rem = deadline - time.monotonic() - 40
    if rem > 15:
        t_alloc = min(rem * 0.5, 60)
        print(f"  Fine subgrad: t={t_alloc:.0f}s")
        a_opt, _ = adam_mu_optimize(
            base, time_limit=t_alloc,
            top_k_init=10,
            temp_init=0.000005, lr_init=0.001,
            beta1=0.93, beta2=0.97,
            temp_decay=0.006, lr_decay=0.9997, top_k_decay=1.0
        )
        try_seq(a_opt, "fine_sg")

    # Another pair swap round
    rem = deadline - time.monotonic() - 25
    if rem > 10:
        a_opt, _ = pair_swap_search(base, time_limit=min(rem, 30))
        try_seq(a_opt, "pex_final")

    # Short coordinate descent
    rem = deadline - time.monotonic() - 10
    if rem > 8:
        a_opt, _ = coordinate_descent_polish(base, time_limit=min(rem, 20))
        try_seq(a_opt, "cd_final")

    # === Phase 6: Random restarts if time remains ===
    rem = deadline - time.monotonic() - 5
    if rem > 15 and best_seq is not None:
        n_base = len(best_seq)
        print(f"\n=== Phase 6: Random restarts (remaining {rem:.0f}s) ===")
        for attempt in range(6):
            rem = deadline - time.monotonic() - 5
            if rem < 10:
                break
            t_alloc = min(rem * 0.3, 25)
            # Diverse perturbation
            scale = 0.002 + 0.002 * attempt
            noise = rng.standard_normal(n_base) * scale
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = adam_mu_optimize(
                f_pert, time_limit=t_alloc,
                top_k_init=max(10, n_base // 10),
                temp_init=0.00001, lr_init=0.002,
                beta1=0.90, beta2=0.95,
                temp_decay=0.005, lr_decay=0.9995, top_k_decay=0.999
            )
            try_seq(a_opt, f"restart_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
