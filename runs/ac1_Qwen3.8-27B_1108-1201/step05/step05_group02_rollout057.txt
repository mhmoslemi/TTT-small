import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def softmax_grad(f, tau, n_peaks=100):
    """Compute gradient of softmax approximation to max(conv(f))."""
    n = len(f)
    conv = fast_convolve(f)
    k_actual = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = pv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        # grad[i_coord] += 2 * w_i * f[k - i_coord] for valid k-i_coord

        valid_end = min(n - 1, k)
        if valid_start <= valid_end:
            idx = np.arange(valid_start, valid_end + 1)
            partner = k - idx
            grad[idx] += 2.0 * w[i] * f[partner]
    
    return grad, conv.max(), conv

def spectral_init(n, seed=0):
    """Create a sequence with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    # Generate random phases and uniform magnitude in frequency domain
    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.zeros(fft_size // 2 + 1, dtype=complex)
    A[0] = 1.0  # DC component
    for k in range(1, fft_size // 2):
        r = rng.uniform(0.8, 1.2)
        theta = rng.uniform(0, 2 * np.pi)
        A[k] = r * np.exp(1j * theta)
    # Conjugate symmetry
    A_full = np.zeros(fft_size, dtype=complex)
    A_full[:fft_size // 2 + 1] = A
    for k in range(fft_size // 2 + 1, fft_size):
        A_full[k] = np.conj(A_full[fft_size - k])
    
    a = np.fft.irfft(A_full)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adam-optimized gradient descent on softmax-smoothed peak
2. LP-based Frank-Wolfe with min-peak formulation
3. Targeted pairwise coordinate descent on peak-contributing coordinates
4. Multi-length exploration with interpolation
5. Spectral equalization initialization
6. Final polish with multiple methods

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def softmax_grad(f, tau, n_peaks=100):
    """Compute gradient of softmax approximation to max(conv(f))."""
    n = len(f)
    conv = fast_convolve(f)
    k_actual = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        valid_start = max(0, k - n + 1)
        valid_end = min(n - 1, k)
        if valid_start <= valid_end:
            idx = np.arange(valid_start, valid_end + 1)
            partner = k - idx
            grad[idx] += 2.0 * w[i] * f[partner]
    return grad, conv.max(), conv

    t = 0
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    tau = tau0
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, cur_max, conv = softmax_grad(f, tau, n_peaks=max(50, n // 4))
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f - step)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and t % 2000 == 0:
                print(f"    Adam iter {t}: obj={best_obj:.10f}, tau={tau:.8f}")
        else:
            f = f_new
            no_improve += 1
        tau *= tau_growth
        if no_improve > 800:
            rng = np.random.default_rng(t * 31337 + 7)
            scale = 0.001 + 0.002 * min(no_improve // 300, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            tau = max(tau0 * 0.3, 1e-6)
            no_improve = 0
    return best_f, best_obj


def solve_lp_min_peak(f, n):
    """Solve LP: min t s.t. (f*g)[k] <= t for all k, g >= 0, sum(g) = 1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_constraints = 2 * n - 1
    A_ub = np.zeros((n_constraints, n + 1))
    for j in range(n):
        A_ub[j:j + n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_constraints)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with min-peak LP."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_min_peak(f, n)
        if g is None:
            g = np.ones(n) / n
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        if best_alpha > 1e-12:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                f_new = np.ones(n) / n
            else:
                f_new = f_new / sn
            f = f_new
            M = best_val
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 80:
            rng = np.random.default_rng(it * 104729 + 3)
            scale = 0.002 + 0.001 * min(no_improve // 50, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    return best_f, best_obj


def targeted_peak_cd(f0, time_limit, verbose=False):
    """Coordinate descent targeting top peaks by adjusting contributing coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        top_k = min(6, len(conv))
        peak_positions = np.argpartition(conv, -top_k)[-top_k:]

        for pk in sorted(peak_positions.tolist(), key=lambda x: -conv[x]):
            if time.monotonic() - t0 > time_limit:
                break
            peak_k = int(pk)
            if conv[peak_k] < best_obj * 0.9:
                continue
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            # Find top contributing coordinates
            contribs = []
            for j in range(js, je + 1):
                contribs.append((f[j] * f[peak_k - j], j))
            contribs.sort(reverse=True)
            # Try adjusting top contributors
            for _, j in contribs[:8]:
                old_val = f[j]
                if old_val < 1e-10:
                    continue
                best_mult = 1.0
                best_trial = conv.max()
                for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.4, 1.7]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.5:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= scale
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < best_trial - 1e-14:
                        best_trial = val
                        best_mult = mult
                if best_mult != 1.0:
                    new_val = old_val * best_mult
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - new_val) / denom
                        f[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f[mask] *= scale
                        f = f / f.sum()
                    if best_trial < best_obj - 1e-14:
                        best_obj = best_trial
                        best_f = f.copy()
                        improved = True
                        conv = fast_convolve(f)
                        break

        if not improved:
            no_improve += 1
            if no_improve > 5:
                rng = np.random.default_rng(it * 7919 + 13)
                for _ in range(min(80, n // 3)):
                    if time.monotonic() - t0 > time_limit:
                        break
                    j = int(rng.integers(0, n))
                    old_val = f[j]
                    if old_val < 1e-10:
                        continue
                    for mult in [0.4, 0.6, 0.8, 1.3, 1.6, 2.0]:
                        new_val = old_val * mult
                        if new_val < 0 or new_val > 0.5:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = f_trial / f_trial.sum()
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            conv = fast_convolve(f)
                            improved = True
                            break
                    if improved:
                        break
            if no_improve > 50:
                break
        else:
            no_improve = 0
    return best_f, best_obj


def spectral_init(n, seed=0):
    """Create a sequence with approximately flat power spectrum for good initial peak."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    for k in range(1, fft_size // 2):
        r = rng.uniform(0.7, 1.3)
        theta = rng.uniform(0, 2 * np.pi)
        A[k] = r * np.exp(1j * theta)
        A[fft_size - k] = np.conj(A[k])
    a = np.fft.irfft(A)[:n]
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-15:
        a /= ss
    else:
        a = np.ones(n) / n
    return a


def subgradient_refine(f0, time_limit, lr=0.003, momentum=0.93, top_k=50,
                       temp=0.00005, verbose=False):
    """Subgradient method with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            kg = k - idx
            mask = (kg >= 0) & (kg < n)
            sg[idx[mask]] += 2.0 * w[i] * f[kg[mask]]
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        co = conv.max()
        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 500 == 0:
                print(f"    SG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            f = f_new
        cur_lr *= 0.9999
        cur_temp *= 1.003
        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0
    return best_f, best_obj


def evaluate_and_update(best_seq, best_score, candidate, label, deadline):
    """Evaluate candidate and update best if improved. Returns (best_seq, best_score)."""
    if time.monotonic() > deadline:
        return best_seq, best_score
    a = np.maximum(candidate, 0.0)
    if a.sum() < 1e-10:
        return best_seq, best_score
    a_norm = a / a.sum()
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining Adam GD, LP-FW, targeted CD,
    multi-length exploration, spectral init, and subgradient methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: n={len(parent)}, score={best_score:.8f}")

    base_n = len(parent)

    # === Phase 1: Adam GD on parent (aggressive) ===
    print(f"\n=== Phase 1: Adam GD aggressive (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.28)
    if t1 > 40:
        print(f"  n={base_n}, t={t1:.0f}s")
        a_opt, _ = adam_projected(parent, time_limit=t1, lr=0.004,
                                  tau0=0.00004, tau_growth=1.003, verbose=True)
        best_seq, best_score = evaluate_and_update(best_seq, best_score, a_opt, "adam1", deadline)

    # === Phase 2: LP-FW ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(300, rem * 0.22)
    if t2 > 30 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t2:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
        best_seq, best_score = evaluate_and_update(best_seq, best_score, a_opt, "lpfw", deadline)

    # === Phase 3: Targeted peak CD ===
    print(f"\n=== Phase 3: Targeted CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(250, rem * 0.17)
    if t3 > 25 and best_seq is not None:
        print(f"  n={len(best_seq)}, t={t3:.0f}s")
        a_opt, _ = targeted_peak_cd(best_seq, time_limit=t3, verbose=True)
        best_seq, best_score = evaluate_and_update(best_seq, best_score, a_opt, "cd", deadline)

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        base = best_seq
        candidate_lengths = set()
        for delta in [-96, -64, -32, 32, 64, 96, 128, 256]:
            ln = len(base) + delta
            if 256 <= ln <= 2048:
                candidate_lengths.add(ln)
        for ln in [512, 768, 1024, 1280]:
            if 256 <= ln <= 2048:
                candidate_lengths.add(ln)

        for target_n in sorted(candidate_lengths)[:5]:
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            t_alloc = min(rem * 0.22, 180)
            if t_alloc < 25:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  Adam n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_projected(f_interp, time_limit=t_alloc, lr=0.003,
                                      tau0=0.00004, tau_growth=1.003)
            best_seq, best_score = evaluate_and_update(best_seq, best_score, a_opt,
                                                       f"adam_n{target_n}", deadline)

    # === Phase 5: Subgradient refinement ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.2, 250)
        if t5 > 30:
            print(f"  n={n_best}, t={t5:.0f}s")
            a_opt, _ = subgradient_refine(best_seq, time_limit=t5, lr=0.002,
                                          momentum=0.94, top_k=max(40, n_best // 6),
                                          temp=0.00004, verbose=True)
            best_seq, best_score = evaluate_and_update(best_seq, best_score, a_opt,

    # === Phase 6: Adam fine-tune ===
    print(f"\n=== Phase 6: Adam fine (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t6 = min(rem * 0.2, 200)
        if t6 > 25:
            print(f"  n={len(best_seq)}, t={t6:.0f}s")
            a_opt, _ = adam_projected(best_seq, time_limit=t6, lr=0.0008,
                                      tau0=0.00001, tau_growth=1.005, verbose=True)
            best_seq, best_score = evaluate_and_update(best_seq, best_score, a_opt,
                                                       "adam_fine", deadline)

    # === Phase 7: Spectral initialization attempts ===
    print(f"\n=== Phase 7: Spectral init (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        spectral_lengths = []
        if best_seq is not None:
            spectral_lengths.append(len(best_seq))
        spectral_lengths.extend([512, 1024, 768])
        for sn in spectral_lengths[:3]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.3, 100)
            f_spec = spectral_init(sn, seed=seed + sn * 7)
            print(f"  Spectral n={sn}, t={t_alloc:.0f}s")
            a_opt, _ = adam_projected(f_spec, time_limit=t_alloc, lr=0.004,
                                      tau0=0.0001, tau_growth=1.002)
            best_seq, best_score = evaluate_and_update(best_seq, best_score, a_opt,
                                                       f"spectral_n{sn}", deadline)

    # === Phase 8: Final LP-FW + CD polish ===
    print(f"\n=== Phase 8: Final polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            t8a = min(rem * 0.5, 120)
            if t8a > 20:
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t8a)
                best_seq, best_score = evaluate_and_update(best_seq, best_score,
                                                           a_opt, "lpfw_final", deadline)
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t8b = min(rem * 0.5, 100)
            if t8b > 15:
                a_opt, _ = targeted_peak_cd(best_seq, time_limit=t8b)
                best_seq, best_score = evaluate_and_update(best_seq, best_score,
                                                           a_opt, "cd_final", deadline)

    # === Phase 9: Perturbation exploration ===
    print(f"\n=== Phase 9: Perturbations (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(8):
            rem = deadline - time.monotonic() - 15
            if rem < 20:
                break
            noise_scale = 0.0008 + 0.0004 * p
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_pert = min(rem * 0.25, 40)
            if t_pert < 10:
                break
            a_opt, _ = adam_projected(f_pert, time_limit=t_pert, lr=0.001,
                                      tau0=0.00001, tau_growth=1.004)
            best_seq, best_score = evaluate_and_update(best_seq, best_score,
                                                       a_opt, f"perturb_{p}", deadline)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
