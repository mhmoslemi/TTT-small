import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def compute_subgradient(f, k):
    """Compute subgradient of (f*f)[k] at f."""
    n = len(f)
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)

        if 0 <= other < n:
            if other == j:
                sg[j] += 2.0 * f[j]
            else:
                sg[j] += f[other]
    return sg

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using a multi-phase strategy:
1. Nesterov-accelerated subgradient descent with adaptive softmax peak weighting
2. LP-based Frank-Wolfe (min-peak LP)  
3. Exact coordinate descent on peak-contributing coordinates
4. Multi-length exploration via interpolation
5. Final polish with fine-grained subgradient + CD
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_subgrad_at_peak(f, k):
    """Subgradient of (f*f)[k] at point f."""
    n = len(f)
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        other = k - j
        if 0 <= other < n:
            if other == j:
                sg[j] += 2.0 * f[j]
            else:
                sg[j] += f[other]
    return sg


def sg_with_nesterov(f0, time_limit, lr_init=0.005, momentum=0.92,
                      top_k=60, temp_init=0.0001, temp_rate=1.001,
                      decay=0.99995, verbose=False):
    """
    Subgradient descent with Nesterov-like momentum.
    Uses softmax-weighted subgradients over top-k peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    # Nesterov state
    y = f.copy()  # lookahead point
    f_prev = f.copy()
    
    conv = fast_convolve(y)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        y = f + momentum * (f - f_prev)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = f.copy()
        else:
            y = y / ys
        
        # Compute convolution at y
        conv_y = fast_convolve(y)
        
        # Softmax-weighted peaks
        k_actual = min(top_k, len(conv_y))
        peaks = np.argpartition(conv_y, -k_actual)[-k_actual:]
        pv = conv_y[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Weighted subgradient at y
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            sg += w[i] * compute_subgrad_at_peak(y, int(peaks[i]))
        
        # Update: f_new = Proj_simplex(y - lr * sg)
        f_new = project_simplex(y - cur_lr * sg)
        
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f_prev = f
        f = f_new
        
        cur_lr *= decay
        cur_temp *= temp_rate
        
        if no_improve > 500:
            rng = np.random.default_rng((it * 2654435761) % (2**31))
            scale = min(0.005, 0.001 * (1 + no_improve // 200))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            f_prev = f.copy()
            y = f.copy()
            cur_lr = lr_init * max(0.3, 1.0 - no_improve / 2000.0)
            no_improve = 0
        
        if verbose and it % 2000 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f} (t={time.monotonic()-t0:.1f}s)")
    
    return best_f, best_obj


def solve_lp_min_peak(f, n):
    """LP: min t s.t. (f*g)[k] <= t for all k, g>=0, sum(g)=1."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_con = 2 * n - 1
    A_ub = np.zeros((n_con, n + 1))
    for j in range(n):
        A_ub[j:j+n, j] = f
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(n_con)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def fpw
