    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_top_peak_gradient(f, conv, top_k, beta):

    conv_len = 2 * n - 1

    if top_k < conv_len:

    else:
        peak_idx = np.arange(conv_len)
    
    # Compute softmax weights on the peaks
    pv = conv[peak_idx]
    bc = beta * pv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f):
    """Build the sparse matrix A such that A@g = f*g (cross-correlation)."""
    n = len(f)
    conv_len = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(conv_len):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(conv_len, n))

def lp_direction(f, M, A=None, time_limit=20):
    """LP: max sum(g) s.t. A@g <= M, g >= 0."""

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)

                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except:
        pass
    return None

The problem is: minimize max_k f^T B_k f subject to f in simplex.

This is a non-convex problem (max of quadratic forms). The LP linearization at point f0 is:
c_k(f0 + d) ≈ c_k(f0) + 2 * f0^T B_k d (first-order approximation)
