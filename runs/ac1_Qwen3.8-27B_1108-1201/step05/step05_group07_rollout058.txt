import time
import math

def fast_convolve(a):

    size = 2*n - 1

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def softmax_weights(conv, T, top_k=None):
    """Compute softmax weights on the convolution peaks."""
    if top_k is not None and top_k < len(conv):
        k = min(top_k, len(conv))
        top_vals = np.partition(conv, -k)[-k:]
    else:
        top_vals = conv
    max_v = top_vals.max()
    shifted = (top_vals - max_v) / T
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(w)
        w[np.argmax(top_vals)] = 1.0
    else:
        w /= ws
    return w

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    m = np.zeros(n)
    v = np.zeros(n)

            best_a = a.copy()
            no_improve = 0

            no_improve += 1

        progress = min(1.0, (time.monotonic() - t0) / time_limit)

        k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        max_pv = pv.max()
        shifted = (pv - max_pv) / T
        shifted = np.clip(shifted, -100, 100)
        w_peaks = np.exp(shifted)
        ws = w_peaks.sum()
        if ws < 1e-30:
            w_peaks = np.zeros(k)
            w_peaks[np.argmax(pv)] = 1.0
        else:
            w_peaks /= ws
        
        # Build full weight vector for FFT gradient
        w_full = np.zeros(len(conv))

        indices = 2*n - 2 - np.arange(n)
        grad = 2.0 * conv_grad[indices]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        a = a_new

            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
    
    return best_a, best_obj

        grad = compute_grad_fft(w_full, a, N_grad)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new)

        if no_improve > restart_every:
            rng = np.random.default_rng((no_improve * 7919 + t % 1000000) % 10000000)
            noise_scale = 0.003 + 0.003 * min(no_improve // 5000, 4)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{label}] iter={t}, t={elapsed:.1f}s, obj={best_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def peak_shaving_search(a0, time_limit, verbose=False):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        if max_val < best_obj - 1e-12:
            best_obj = max_val
            best_a = a.copy()

        js = max(0, int(max_idx) - n + 1)
        je = min(n - 1, int(max_idx))
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[int(max_idx) - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        top_contrib_idx = idx[np.argsort(contribs)][::-1][:min(20, len(idx))]
        improved = False

        for ci in top_contrib_idx:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[ci]
            if old_val < 1e-8:
                continue
            partners = idx[idx != ci]
            for pj in partners[:min(10, len(partners))]:
                for frac in [0.1, 0.2, 0.3]:
                    amt = old_val * frac
                    trial = a.copy()
                    trial[ci] = old_val - amt
                    trial[pj] += amt
                    val = fast_convolve(trial).max()
                    if val < best_obj - 1e-12:
                        best_obj = val
                        a = trial
                        best_a = trial.copy()
                        improved = True
                        break
                    else:
                        break
            if improved:
                break

        if not improved:
            break

    return best_a, best_obj
