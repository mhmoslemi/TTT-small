    f_d = f_d / sd if sd > 1e-15 else f
    fc = fast_convolve(f_c).max()
    fd = fast_convolve(f_d).max()
    for _ in range(25):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - phi * (hi - lo)
            f_c = np.maximum(f + c_pt * (g - f), 0.0)
            sc = f_c.sum()
            f_c = f_c / sc if sc > 1e-15 else f
            fc = fast_convolve(f_c).max()
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + phi * (hi - lo)
            f_d = np.maximum(f + d_pt * (g - f), 0.0)
            sd = f_d.sum()
            f_d = f_d / sd if sd > 1e-15 else f
            fd = fast_convolve(f_d).max()
    return (lo + hi) / 2.0

def lp_fw(f0, tlimit, do_palindrome, top_k, use_full, verbose):
    """LP Frank-Wolfe optimizer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_imp = 0
    it = 0
    # Nesterov momentum
    y = f.copy()
    momentum = 0.0
    
    while time.monotonic() - t0 < tlimit:
        it += 1
        if time.monotonic() - t0 < 0.5:
            pass
        
        # Constraint selection
        if use_full or it % 200 == 0:
            active_k = None
        else:
            k_sorted = np.argsort(conv)[::-1][:top_k]
            active_k = k_sorted.tolist()
        
        g = solve_lp_dir(y, M, n, active_k)
        if g is None:
            # Relax
            for relax in [1.01, 1.02, 1.05]:
                g = solve_lp_dir(y, M * relax, n, active_k)
                if g is not None:
                    break
            if g is None:
                no_imp += 3
                if no_imp > 150:
                    break
                continue
        
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_imp += 1
            if no_imp > 100:
                break
            continue
        
        # Nesterov update
        momentum = (it + 1.5) / (it0 + 2.5) if it > 0 else 0  # This is wrong, let me fix
        
        # Actually, let's just use simple FW with occasional momentum
        # Simple FW step
        alpha = line_search_alpha(f, g, n, M)
        
        if alpha > 1e-10:
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            f_new = f_new / sn if sn > 1e-15 else f
            if do_palindrome:
                f_new = project_simplex(make_palindromic(f_new, n))
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            conv = conv_new
            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f_new.copy()
                no_imp = 0
                M = M_new
                if verbose and it % 30 == 0:
                    print(f"    LPFW iter {it}: obj={best_obj:.10f}")
            else:
                no_imp += 1
            f = f_new
        else:
            no_imp += 1
        
        if no_imp > 60:
            # Restart with noise
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.004 * min(no_imp // 100, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            if do_palindrome:
                f = project_simplex(make_palindromic(f, n))
            conv = fast_convolve(f)
            M = conv.max()
            no_imp = 0
    
    return best_f, best_obj


def coord_desc(f0, tlimit):
    """Coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_imp_passes = 0
    mults = [0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.4, 1.7, 2.0, 2.5, 3.0]
    
    while time.monotonic() - t0 < tlimit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > tlimit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            best_val = old_val
            best_obj_j = best_obj
            for mult in mults:
                new_val = old_val * mult
                if new_val < 0 or 1.0 - new_val < 1e-10:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                ft = f.copy()
                ft[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] *= scale
                ft = ft / ft.sum()
                val = fast_convolve(ft).max()
                if val < best_obj_j - 1e-15:
                    best_obj_j = val
                    best_val = new_val
            if best_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    f[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_obj_j < best_obj - 1e-14:
                    best_obj = best_obj_j
                    best_f = f.copy()
                    improved = True
        if not improved:
            no_imp_passes += 1
            if no_imp_passes > 2:
                break
        else:
            no_imp_passes = 0
    return best_f, best_obj


def sg_polish(f0, tlimit, lr0, mom, top_k, temp0):
    """Subgradient polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_imp = 0
    cur_lr = lr0
    cur_temp = temp0
    it = 0
    
    while time.monotonic() - t0 < tlimit:
        it += 1
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        cur_lr *= 0.9999
        cur_temp *= 1.003
        if no_imp > 300:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.5, 0.0003)
            no_imp = 0
    return best_f, best_obj


def interp_seq(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    return fi


def flat_spectrum_init(n, seed=0):
    """Flat power spectrum initialization."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig[:n])
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def fourier_restart(f, n, seed):
    """Generate a restart by perturbing the DFT phases."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    # Perturb phases
    phase_pert = rng.standard_normal(fft_size // 2 + 1) * 0.3
    F_new = F * np.exp(1j * phase_pert)
    # Maintain DC component
    F_new[0] = np.abs(F_new[0])
    sig = np.fft.irfft(F_new, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = project_simplex(f)
    return sig
