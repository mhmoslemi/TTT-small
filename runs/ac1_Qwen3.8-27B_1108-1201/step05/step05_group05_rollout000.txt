    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.9, 1.1, 1.15, 1.3, 1.5, 2.0, 3.0])

    # Random pair selection for efficiency
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        improved = False
        pass_count = 0

        # Try both sequential and random pairs
        if pass_count % 3 == 0:
            # Sequential sweep (first few coordinates with all others via aggregate)
            pairs = [(j, (j + 1) % n) for j in range(0, n, 2)]
            if n % 2 == 0:
                pairs[-1] = (n - 2, n - 1)
        else:
            # Random pairs
            idx = rng.integers(0, n, size=min(200, n))
            pairs = [(int(idx[i]), int(idx[(i + 1) % len(idx)])) for i in range(0, len(idx), 2)]

        for i, j in pairs:
            if i == j:
                continue
            if time.monotonic() - t0 > time_limit:
                break

            old_i, old_j = f[i], f[j]
            if old_i < 1e-12 and old_j < 1e-12:
                continue

            best_trial_obj = best_obj
            best_trial_i, best_trial_j = old_i, old_j

            for mult in multipliers:
                new_i = old_i * mult
                new_j = old_j / mult  # preserve sum of the pair
                remaining = 1.0 - new_i - new_j
                if new_i < 0 or new_j < 0 or remaining <= 1e-10:
                    continue

                # Rescale others
                old_sum = 1.0 - old_i - old_j
                if old_sum < 1e-10:
                    continue
                scale = remaining / old_sum

                f_trial = f.copy()
                others_mask = np.ones(n, dtype=bool)
                others_mask[i] = False
                others_mask[j] = False
                f_trial[others_mask] *= scale
                f_trial[i] = new_i
                f_trial[j] = new_j
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_trial_obj - 1e-15:
                    best_trial_obj = val
                    best_trial_i = new_i
                    best_trial_j = new_j

            if best_trial_obj < best_obj - 1e-14:
                old_sum = 1.0 - old_i - old_j
                if old_sum > 1e-10:
                    remaining = 1.0 - best_trial_i - best_trial_j
                    scale = remaining / old_sum
                    others_mask = np.ones(n, dtype=bool)
                    others_mask[i] = False
                    others_mask[j] = False
                    f[others_mask] *= scale
                f[i] = best_trial_i
                f[j] = best_trial_j
                f = f / f.sum()
                best_obj = best_trial_obj
                best_f = f.copy()
                improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 2:
                break
        else:
            no_improve_passes = 0
        pass_count += 1

    return best_f, best_obj


def single_coord_descent(f0, time_limit):
    """Fast single-coordinate descent with adaptive multiplier search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve_passes = 0

    multipliers = [0.3, 0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.4, 1.7, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            for mult in multipliers:
                new_val = old_val * mult
                remaining = 1.0 - new_val
                if remaining <= 1e-10 or new_val < 0:
                    continue
                scale = remaining / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = f / f.sum()
                best_obj = best_j_obj
                best_f = f.copy()
                improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 2:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def lp_fw_short(f0, time_limit, top_k_constraints=30):
    """Short LP-FW refinement with adaptive constraints."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        active_k = None
        if it > 3:
            k_sorted = np.argsort(conv)[::-1][:top_k_constraints]
            active_k = k_sorted.tolist()

        # Solve LP
        c = -np.ones(n)
        K = len(active_k) if active_k is not None else (2 * n - 1)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M * m_exp)

        if active_k is not None:
            for row, k in enumerate(active_k):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    A_ub[row, j] = f[k - j]
        else:
            for j in range(n):
                A_ub[j:j + n, j] = f
            b_ub = np.full(2 * n - 1, M * m_exp)

        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum > 1e-15:
                g = g / g_sum
                if np.max(np.abs(g - f)) > 1e-10:
                    # Simple line search
                    for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
                        f_new = project_simplex(f + alpha * (g - f))
                        conv_new = fast_convolve(f_new)
                        M_new = conv_new.max()
                        if M_new < best_obj - 1e-14:
                            best_obj = M_new
                            best_f = f_new.copy()
                            no_improve = 0
                            f = f_new
                            conv = conv_new
                            M = M_new
                            m_exp = 1.0
                            break
        else:
            m_exp *= 1.15

        if res is None or not res.success:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 3.0)

        if no_improve > 30:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def flat_spectrum_init(n, seed=0):
    """Initialize with flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    spectrum = np.ones(fft_size // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, fft_size)
    sig = sig[:n]
    sig = np.maximum(sig, 0)
    ss = sig.sum()
    if ss < 1e-10:
        sig = np.abs(sig)
        ss = sig.sum()
    if ss > 1e-10:
        sig /= ss
    else:
        sig = np.ones(n) / n
    return sig


def structured_restart(base, n, rng, mode):
    """Generate a structured restart perturbation."""
    if mode == 0:
        return project_simplex(base + rng.standard_normal(n) * 0.002)
    elif mode == 1:
        return project_simplex(base + rng.standard_normal(n) * 0.005)
    elif mode == 2:
        reverse = base[::-1]
        return project_simplex(0.6 * base + 0.4 * reverse)
    elif mode == 3:
        pal_half = (base[:n // 2] + base[n - 1:n - n // 2 - 1: -1]) / 2.0
        return project_simplex(make_palindromic(pal_half, n))
    elif mode == 4:
        # Gaussian window perturbation
        center = np.arange(n) - n / 2.0
        gauss = np.exp(-center ** 2 / (0.3 * n) ** 2)
        gauss = gauss / gauss.sum()
        return project_simplex(0.7 * base + 0.3 * gauss)
    elif mode == 5:
        # Frequency perturbation
        fs = flat_spectrum_init(n, seed=int(rng.integers(0, 10000)))
        return project_simplex(0.8 * base + 0.2 * fs)
    else:
        return project_simplex(base + rng.standard_normal(n) * 0.003)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining Nesterov subgradient, pair-wise
    coordinate descent, LP-FW refinement, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = flat_spectrum_init(768, seed=seed)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using flat spectrum init: score={best_score:.6f}")

    n0 = len(parent)

    # === Phase 1: Quick Nesterov SG on parent ===
    print(f"\n=== Phase 1: Nesterov SG (n={n0}) ===")
    rem = deadline - time.monotonic()
    t1 = min(250, rem * 0.25)
    if t1 > 30:
        a_opt, obj1 = nesterov_sg_loop(parent, time_limit=t1, lr0=0.003,
                                       mom=0.92, top_k_start=5,
                                       temp_start=0.005, verbose=True)
        try_seq(a_opt, "nesterov_sg")

    # === Phase 2: Pair-wise coordinate descent ===
    print(f"\n=== Phase 2: Pair CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(200, rem * 0.2)
        if t2 > 30:
            a_opt, _ = pair_coord_descent(best_seq, time_limit=t2)
            try_seq(a_opt, "pair_cd")

    # === Phase 3: Single coordinate descent (fine) ===
    print(f"\n=== Phase 3: Single CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(150, rem * 0.15)
        if t3 > 30:
            a_opt, _ = single_coord_descent(best_seq, time_limit=t3)
            try_seq(a_opt, "single_cd")

    # === Phase 4: LP-FW short refinement ===
    print(f"\n=== Phase 4: LP-FW Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t4 = min(200, rem * 0.2)
        if t4 > 30:
            a_opt, _ = lp_fw_short(best_seq, time_limit=t4, top_k_constraints=35)
            try_seq(a_opt, "lpfw_short")

    # === Phase 5: Nesterov SG again (annealed) ===
    print(f"\n=== Phase 5: Annealed SG (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t5 = min(200, rem * 0.18)
        if t5 > 20:
            a_opt, _ = nesterov_sg_loop(best_seq, time_limit=t5, lr0=0.001,
                                        mom=0.95, top_k_start=10,
                                        temp_start=0.002)
            try_seq(a_opt, "annealed_sg")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        for target_n in [1024, 1280, 1536]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            # Mix with flat spectrum for diversity
            fs = flat_spectrum_init(target_n, seed=target_n)
            f_mix = project_simplex(0.75 * f_interp + 0.25 * fs)

            t_alloc = min(rem * 0.35, 120)
            print(f"  Multi-length n={target_n}, t={t_alloc:.0f}s")

            a_opt, _ = nesterov_sg_loop(f_mix, time_limit=t_alloc * 0.5,
                                        lr0=0.002, mom=0.93,
                                        top_k_start=5, temp_start=0.004)
            try_seq(a_opt, f"ml_n{target_n}_sg")

            rem = deadline - time.monotonic()
            if rem > 30:
                a_opt2, _ = single_coord_descent(a_opt, time_limit=min(rem * 0.4, 60))
                try_seq(a_opt2, f"ml_n{target_n}_cd")

    # === Phase 7: Structured restarts on best ===
    print(f"\n=== Phase 7: Structured Restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        restart_modes = [0, 1, 2, 3, 4, 5]
        for mode in restart_modes:

            if rem < 40:

            f_start = structured_restart(best_seq, n_best, rng, mode)
            t_alloc = min(rem * 0.3, 80)
            a_opt, _ = nesterov_sg_loop(f_start, time_limit=t_alloc,
                                        lr0=0.002, mom=0.93,
                                        top_k_start=5, temp_start=0.004)

            rem = deadline - time.monotonic()
            if rem > 20:
                a_opt2, _ = single_coord_descent(a_opt,
                                                 time_limit=min(rem * 0.3, 40))
                try_seq(a_opt2, f"restart_m{mode}_cd")

    # === Phase 8: Final combined polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()

        # CD
        if rem > 30:
            t = min(rem * 0.3, 80)
            a_opt, _ = single_coord_descent(best_seq, time_limit=t)
            try_seq(a_opt, "final_cd")

        # Pair CD
        rem = deadline - time.monotonic()
        if rem > 30:
            t = min(rem * 0.3, 60)
            a_opt, _ = pair_coord_descent(best_seq, time_limit=t)
            try_seq(a_opt, "final_pair_cd")

        # Short LP-FW
        rem = deadline - time.monotonic()
        if rem > 30:
            t = min(rem * 0.4, 80)
            a_opt, _ = lp_fw_short(best_seq, time_limit=t, top_k_constraints=40)
            try_seq(a_opt, "final_lpfw")

        # Final tiny SG
        rem = deadline - time.monotonic()
        if rem > 15:
            t = min(rem * 0.5, 40)
            a_opt, _ = nesterov_sg_loop(best_seq, time_limit=t, lr0=0.0005,
                                        mom=0.96, top_k_start=8,
                                        temp_start=0.001)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
