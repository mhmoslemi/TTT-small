def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def flat_conv_sequence(n):
    """Generate first n terms of the infinite flat-convolution sequence."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        s = 0.0
        if k % 2 == 1:
            m = (k - 1) // 2
            for i in range(1, m + 1):
                s += 2.0 * a[i] * a[k - i]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_gradient(f, beta, max_peaks=500):
    """Compute gradient of LSE-smoothed max convolution."""
    n = len(f)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Use only top peaks for efficiency
    if len(w) > max_peaks:
        sig_idx = np.argpartition(w, -max_peaks)[-max_peaks:]
        w_full = np.zeros(len(w))
        w_full[sig_idx] = w[sig_idx]
        w_full /= w_full.sum()
    else:
        w_full = w
    
    # Compute gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    grad = np.zeros(n)
    conv_len = len(w_full)
    for k in range(conv_len):
        wk = w_full[k]
        if wk < 1e-18:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            grad[js] += 2.0 * wk * f[k - js]
    return grad, conv
