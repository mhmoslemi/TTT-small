def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with log-sum-exp
smoothing, FFT-based gradient, adaptive continuation, pairwise coordinate
descent, and LP-Frank-Wolfe polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hard_subgradient(f, conv):
    """Compute subgradient at the argmax of conv."""
    n = len(f)
    k = int(np.argmax(conv))
    sg = np.zeros(n)
    m_lo = max(0, k - (n - 1))
    m_hi = min(n - 1, k)
    if m_lo <= m_hi:
        indices = np.arange(m_lo, m_hi + 1)
        sg[indices] = 2.0 * f[k - indices]
    return sg


def smooth_gradient_fft(f, tau, conv=None):
    """Compute gradient of (1/tau)*log(sum(exp(tau*conv))) using FFT."""
    n = len(f)
    if conv is None:
        conv = fast_convolve(f)
    
    max_val = conv.max()
    logits = tau * (conv - max_val)
    # logits <= 0, max is 0
    # For numerical stability
    p = np.exp(logits)
    p_sum = p.sum()
    if p_sum < 1e-300:
        return hard_subgradient(f, conv)
    p /= p_sum
    
    # grad[m] = 2 * sum_j f[j] * p[m+j]
    # = 2 * (f * flip(p))[2n-2-m]
    s = p[::-1]  # flip p
    conv_len = 2 * n - 1
    total_len = n + conv_len - 1  # = 3n-2
    fft_size = 1 << (total_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    S = np.fft.rfft(s, fft_size)
    result = np.fft.irfft(F * S, fft_size)
    
    # We want result[2n-2-m] for m = 0, ..., n-1
    # i.e., indices 2n-2, 2n-3, ..., n-1
    grad = 2.0 * result[n-1:2*n-1][::-1]
    
    return grad


def fista_smooth(f0, time_limit, tau_init=10.0, tau_max=10000.0, 
                  tau_growth=1.3, verbose=False):
    """FISTA with log-sum-exp continuation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    tau = tau_init
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit and tau < tau_max:
        # FISTA loop for current tau
        y = f.copy()
        t_val = 1.0
        
        # Adaptive step size
        # Lipschitz constant approx: tau * (max_conv - min_conv) * 2 + 2
        conv_y = fast_convolve(y)
        conv_range = conv_y.max() - conv_y.min()
        L = tau * conv_range * 2 + 2
        eta = 1.0 / L
        
        max_iters = 10000
        stagnant = 0
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            total_iters += 1
            conv_y = fast_convolve(y)
            g = smooth_gradient_fft(y, tau, conv_y)
            
            # FISTA step
            z = project_simplex(y - g / L)
            
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y_new = z + ((t_val - 1.0) / t_new) * (z - f)
            
            f = z
            y = project_simplex(y_new)
            t_val = t_new
            
            # Check objective
            if it % 100 == 0 or it == max_iters - 1:
                conv = fast_convolve(f)
                obj = conv.max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_f = f.copy()
                    stagnant = 0
                    if verbose and it % 1000 == 0:
                        print(f"      tau={tau:.1f}, iter={total_iters}: obj={best_obj:.10f}")
                else:
                    stagnant += 100
            
            if stagnant > 5000:
                break
        
        # Increase tau
        old_tau = tau
        tau = min(tau * tau_growth, tau_max)
        
        if verbose:
            conv = fast_convolve(best_f)
            print(f"    FISTA phase: tau {old_tau:.1f} -> {tau:.1f}, best={conv.max():.10f}")
    
    return best_f, best_obj


def fista_hard(f0, time_limit, L0=8.0, verbose=False):

    y = f.copy()

    it = 0
    no_improve = 0

        if it % 1000 == 0 and it > 100000:
            # Safety: slow down
            L *= 1.01
        
        conv_y = fast_convolve(y)

        z = project_simplex(y - g / L)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0

        f = z
        y = project_simplex(y_new)
        t_val = t_new
        
        if it % 500 == 0:
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-15:

                no_improve = 0
            else:
                no_improve += 500
            
            if verbose and it % 5000 == 0:

def pairwise_cd(f0, time_limit, verbose=False):
    """Pairwise coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    passes = 0
    no_pass_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False
        
        for i in range(0, n - 1, 2):  # Non-overlapping pairs
            if time.monotonic() - t0 > time_limit:
                break
            
            j = i + 1
            s_pair = f[i] + f[j]
            if s_pair < 1e-15:
                continue
            
            # Ternary search for optimal x = f[i], f[j] = s_pair - x
            lo = 0.0
            hi = s_pair
            x_cur = f[i]
            
            # Quick check: evaluate at a few points
            def eval_x(x):
                f[i] = x
                f[j] = s_pair - x
                c = fast_convolve(f)
                return c.max()
            
            x1 = lo + (hi - lo) / 3.0
            x2 = hi - (hi - lo) / 3.0
            
            for _ in range(15):
                f1 = eval_x(x1)
                f2 = eval_x(x2)
                if f1 < f2:
                    hi = x2
                    x2 = x1
                else:
                    lo = x1
                    x1 = x2
            
            opt_x = (lo + hi) / 2.0
            f[i] = opt_x
            f[j] = s_pair - opt_x
            
            # Check if this was an improvement
            c = fast_convolve(f)
            obj = c.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                pass_improved = True
        
        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break
    
    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
