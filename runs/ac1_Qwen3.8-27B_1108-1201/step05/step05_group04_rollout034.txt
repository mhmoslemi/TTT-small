def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    """
    Compute gradient of sum_k w[k] * (f*f)[k] w.r.t. f.
    w has length 2n-1.
    Uses the formula: grad[m] = 2 * sum_{k: m in S_k} w[k] * f[k-m]
    which is equivalent to: grad = 2 * (w * f) restricted to valid indices... 
    Actually, let's use the direct formula.

    grad = np.zeros(n)
    # For each k, the contribution to grad[m] for m in [max(0,k-n+1), min(n-1,k)] is 2*w[k]*f[k-m]
    # This is like a cross-correlation.
    # grad[m] = 2 * sum_k w[k] * f[k-m] * 1[max(0,k-n+1) <= m <= min(n-1,k)]
    # = 2 * sum_k w[k] * f[k-m] * 1[0 <= k-m < n] * 1[0 <= m < n] * 1[k >= 0] * 1[k <= 2n-2]
    # Since m is already in [0, n-1], we need: 0 <= k-m < n and 0 <= k <= 2n-2
    # i.e., m <= k <= m+n-1 and 0 <= k <= 2n-2
    # So k in [max(0, m), min(2n-2, m+n-1)]
    # And f[k-m] where k-m in [0, n-1] ✓
    
    # More efficiently: grad[m] = 2 * sum_{j=0}^{n-1} f[j] * w[m+j] (if m+j < 2n-1)
    # Because: let j = k-m, then k = m+j, and we need 0 <= j < n and 0 <= m+j <= 2n-2
    # So grad[m] = 2 * sum_{j: 0<=j<n, 0<=m+j<2n-1} f[j] * w[m+j]
    
    # This is a cross-correlation of f and w, but only for valid indices.
    # For m = 0: j from 0 to n-1, w[j] (all valid since j < n < 2n-1)
    # For m = n-1: j from 0 to n-1, w[n-1+j] (valid since n-1+j < 2n-1)
    # So for all m in [0, n-1], j ranges from 0 to n-1, and m+j ranges from m to m+n-1, all < 2n-1.
    
    # So grad[m] = 2 * sum_{j=0}^{n-1} f[j] * w[m+j]
    # This is a cross-correlation! Can be computed via FFT.
    
    w_padded = np.zeros(2 * n - 1)
    w_padded[:len(w)] = w[:2*n-1]

    # This is the "sliding dot product" of f with w starting at position m.
    # Equivalent to: if we flip f, then convolve with w, and take appropriate indices.
    # Actually, corr[m] = <f, w[m:m+n]> = sum_j f[j]*w[m+j]
    # This is the cross-correlation (f ◦ w)[m]
    
    # Using FFT: corr = ifft(fft(f, N) * conj(fft(w_padded, N))) ... hmm, let me think.
    # Actually, the cross-correlation (f ◦ w)[m] = sum_j f[j] * w[m+j]
    # = sum_j f[j] * w[m+j]
    # Let's define f_flipped[j] = f[n-1-j], then:
    # (f_flipped * w)[m+n-1] = sum_j f_flipped[j] * w[m+n-1-j] = sum_j f[n-1-j] * w[m+n-1-j]
    # Let k = n-1-j: = sum_k f[k] * w[m+n-1-(n-1-k)] = sum_k f[k] * w[m+k]
    # So (f_flipped * w)[m+n-1] = sum_k f[k] * w[m+k] = corr[m]
    
    # So corr = ifft(fft(f_flipped) * fft(w_padded))[n-1 : n-1+n]

    grad = 2.0 * corr

        self.v = np.zeros(n)  # second moment

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-Adam with Nesterov acceleration (gradient-based)
2. Coordinate descent via simplex-rescaled single-coordinate optimization
3. Systematic multi-length exploration
4. Adaptive beta scheduling and restarts
Key ideas: FFT-based gradient (exact, fast), Adam per-coordinate adaptation,
coordinate descent for local polishing, Nesterov momentum for faster convergence.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_gradient_fft(f, w):
    """
    Compute gradient of sum_k w[k]*(f*f)[k] w.r.t. f using FFT.
    grad[m] = 2 * sum_{j=0}^{n-1} f[j] * w[m+j]
    Uses cross-correlation via FFT.
    """
    n = len(f)
    w_padded = np.zeros(2 * n - 1)
    w_padded[:min(len(w), 2*n-1)] = w[:min(len(w), 2*n-1)]
    
    # Cross-correlation via FFT
    # corr[m] = sum_j f[j] * w[m+j] for m = 0,...,n-1
    # = (f_flipped * w_padded)[m+n-1]
    N = 1 << (3 * n - 2).bit_length()
    F = np.fft.rfft(f[::-1].copy(), N)
    W = np.fft.rfft(w_padded, N)
    corr_full = np.fft.irfft(F * W, N)
    corr = corr_full[n-1:n-1+n]
    
    return 2.0 * corr


def lse_weights(conv, beta):
    """Compute normalized LSE weights on convolution values."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    s = exp_bc.sum()
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[conv.argmax()] = 1.0
        return w
    return exp_bc / s


class AdamState:
    """Adam optimizer state."""
    def __init__(self, n, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
    
    def update(self, grad, lr):
        self.t += 1
        self.m = self.beta1 * self.m + (1.0 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1.0 - self.beta2) * grad * grad
        m_hat = self.m / (1.0 - self.beta1 ** self.t)
        v_hat = self.v / (1.0 - self.beta2 ** self.t)
        return -lr * m_hat / (np.sqrt(v_hat) + self.eps)


def nesterov_step(f, adam, grad, lr):
    """Nesterov-style step: compute gradient at look-ahead point."""
    look_ahead = f + adam.m / (1.0 - adam.beta1) if adam.t > 0 else f
    return look_ahead

        f = f / s

    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 80
    
    while time.monotonic() - t0 < time_limit:

        w = lse_weights(conv, beta)
        grad = compute_conv_gradient_fft(f, w)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            no_improve += 100
            break
        
        # Nesterov look-ahead (optional, can disable)
        # For now, use standard Adam
        
        step = adam.update(grad, lr)
        f_new = project_simplex(f + step)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

            f = f_new

                beta = min(beta * 3.0, beta_end)
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.008

def coordinate_descent(f0, time_limit, n_sweeps=None):

    Coordinate descent on the simplex.
    For each coordinate m, optimize: set a_m = t, scale others to maintain sum=1.
    Use ternary search over t in [0, 1].

        f = np.ones(len(f)) / len(f)
    else:
        f = f / s

    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

            if time.monotonic() - t0 > time_limit:
                break
            
            a_m = f[m]
            # Parameterize: new a_m = t, others scaled by (1-t)/(1-a_m)
            # t in [0, 1-a_m/(1-a_m)*...] actually t in [0, 1]
            # When t=0: a_m=0, others scaled by 1/(1-a_m)
            # When t=1: a_m=1, others = 0 (degenerate)
            # When t=a_m: no change
            
            def obj_at_t(t):

                    return 0.0
                scale = (1.0 - t) / (1.0 - a_m) if a_m < 1.0 else 0.0

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return np.maximum(result, 0.0)


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_gradient_fft(f, w):
    """
    Compute gradient of sum_k w[k]*(f*f)[k] w.r.t. f using FFT.
    Formula: grad[m] = 2 * sum_{j=0}^{n-1} f[j] * w[m+j]
    Uses cross-correlation via FFT for O(n log n) computation.
    """
    n = len(f)
    w_padded = np.zeros(2 * n - 1)
    lim = min(len(w), 2 * n - 1)
    w_padded[:lim] = w[:lim]

    N = 1 << (3 * n - 2).bit_length()
    F = np.fft.rfft(f[::-1].copy(), N)
    W = np.fft.rfft(w_padded, N)
    corr_full = np.fft.irfft(F * W, N)
    corr = corr_full[n - 1:n - 1 + n]
    return 2.0 * corr


def compute_grad_direct(f, w, top_k=200):
    """Direct gradient computation for sparse w (top_k peaks only)."""
    n = len(f)
    conv_len = 2 * n - 1
    if len(w) >= top_k:
        idx = np.argpartition(w, -top_k)[-top_k:]
        idx = idx[w[idx] > 1e-15 * w.max()]
    else:
        idx = np.where(w > 1e-15)[0]
    grad = np.zeros(n)
    for k in idx:
        wk = w[k]
        if wk < 1e-15:

        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            ms = np.arange(lo, hi + 1)
            grad[ms] += 2.0 * wk * f[k - ms]
    return grad

    if s < 1e-300:
        w = np.zeros(len(conv))
        w[conv.argmax()] = 1.0
        return w
    return exp_bc / s


class AdamState:
    """Adam optimizer state (top-level class, no closures)."""
    def __init__(self, n, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps

    def update(self, grad, lr):
        self.t += 1
        self.m = self.beta1 * self.m + (1.0 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1.0 - self.beta2) * grad * grad
        m_hat = self.m / (1.0 - self.beta1 ** self.t)
        v_hat = self.v / (1.0 - self.beta2 ** self.t)
        return -lr * m_hat / (np.sqrt(v_hat) + self.eps)


def lse_adam_optimize(f0, time_limit, beta_start=50.0, beta_end=200000.0, lr_init=0.1):
    """LSE-Adam optimization with adaptive beta and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    adam = AdamState(n)
    beta = beta_start
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        w = lse_weights(conv, beta)

        if beta > 1000:
            grad = compute_grad_direct(f, w, top_k=300)
        else:
            grad = compute_conv_gradient_fft(f, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = adam.update(grad, lr)
        f_new = project_simplex(f + step)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 0.8)
        else:
            no_improve += 1
            if no_improve % 8 == 0:
                lr *= 0.93
            if no_improve > 100:
                f = best_f.copy()
                adam = AdamState(n)
                lr = max(lr_init * 0.2, 0.005)
                beta = min(beta * 4.0, beta_end)
                no_improve = 0
                rng_r = np.random.default_rng(no_improve * 7919)
                f = project_simplex(f + rng_r.standard_normal(n) * 0.002)

        if beta < beta_end:
            beta *= 1.006

    return best_f, best_obj


def compute_pair_dq(f, i, j):
    """Compute linear (d) and quadratic (q) terms for pair exchange i<-j."""
    n = len(f)
    conv_len = 2 * n - 1
    d = np.zeros(conv_len)
    q = np.zeros(conv_len)

    lo1 = i
    hi1 = min(i + n - 1, conv_len - 1)
    d[lo1:hi1 + 1] += 2.0 * f[0:hi1 - lo1 + 1]

    lo2 = j
    hi2 = min(j + n - 1, conv_len - 1)
    d[lo2:hi2 + 1] -= 2.0 * f[0:hi2 - lo2 + 1]

    if 0 <= 2 * i < conv_len:
        q[2 * i] += 1.0
    if 0 <= 2 * j < conv_len:
        q[2 * j] += 1.0
    if 0 <= i + j < conv_len:
        q[i + j] -= 2.0

    return d, q


def pair_exchange_step(f, i, j, conv_cache):
    """Golden section search for optimal t in pair exchange. O(n) per eval."""
    a_lo = -f[i]
    a_hi = f[j]
    if a_hi - a_lo < 1e-15:
        return 0.0

    d, q = compute_pair_dq(f, i, j)

    def obj(t):
        return (conv_cache + t * d + t * t * q).max()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = a_lo
    b = a_hi
    c = b - gr * (b - a)
    dd = a + gr * (b - a)
    fc = obj(c)
    fd = obj(dd)

    for _ in range(30):
        if fc < fd:
            b = dd
            dd = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a = c
            c = dd
            fc = fd
            dd = a + gr * (b - a)
            fd = obj(dd)

    return (a + b) / 2.0


def coordinate_descent(f0, time_limit, n_pairs_per_sweep=200):
    """
    Coordinate descent via gradient-guided pair exchanges.
    Each step is O(n) - no FFT needed after initial convolution.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    sweep = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        conv = fast_convolve(f)

        w = lse_weights(conv, 50.0)
        grad = compute_grad_direct(f, w, top_k=100)

        # Select promising pairs
        neg_idx = np.argsort(grad)[:min(50, n)]
        pos_idx = np.argsort(grad)[::-1][:min(50, n)]

        best_t_global = 0.0
        best_obj_sweep = best_obj
        best_pair = None

        pairs_tried = 0
        for i in neg_idx:
            if pairs_tried >= n_pairs_per_sweep:
                break
            for j in pos_idx:
                if i == j:
                    continue
                if f[i] + f[j] < 1e-12:
                    continue
                t = pair_exchange_step(f, int(i), int(j), conv)
                if abs(t) > 1e-12:
                    new_conv = conv + t * 2.0 * (
                        np.roll(f, -int(i), len(f)) - np.roll(f, -int(j), len(f))
                    )
                    # Actually recompute properly
                    f_try = f.copy()
                    f_try[int(i)] += t
                    f_try[int(j)] -= t
                    f_try = np.maximum(f_try, 0.0)
                    fs = f_try.sum()
                    if fs < 1e-15:
                        continue
                    f_try /= fs
                    new_convv = fast_convolve(f_try)
                    obj_try = new_convv.max()
                    if obj_try < best_obj_sweep - 1e-12:
                        best_obj_sweep = obj_try
                        best_pair = (int(i), int(j), t)
                        best_t_global = t

                pairs_tried += 1
                if pairs_tried >= n_pairs_per_sweep:
                    break

        if best_pair is not None and best_obj_sweep < best_obj - 1e-12:
            i, j, t = best_pair
            f[i] += t
            f[j] -= t
            f = np.maximum(f, 0.0)
            f /= f.sum()
            best_obj = best_obj_sweep
            best_f = f.copy()
            stale = 0
        else:
            stale += 1
            if stale > 5:
                # Perturb and restart
                rng_r = np.random.default_rng(sweep * 31 + 7)
                f = project_simplex(best_f + rng_r.standard_normal(n) * 0.003)
                stale = 0

    return best_f, best_obj


def directed_subgrad_optimize(f0, time_limit, n_peaks=5, lr_init=0.005, momentum=0.92):
    """Direct subgradient method targeting top peaks with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        k_top = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = pv - pv.min()
        if scaled.max() > 1e-15:
            w_peaks = np.exp(scaled - scaled.max())
            w_peaks /= w_peaks.sum()
        else:
            w_peaks = np.ones(k_top) / k_top

        w_full = np.zeros(len(conv))
        w_full[peaks] = w_peaks

        grad = compute_grad_direct(f, w_full, top_k=k_top + 10)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.95
            if no_improve > 200:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.0005)
                no_improve = 0
                rng_r = np.random.default_rng(int(no_improve * 104729) % 999983)
                f = project_simplex(f + rng_r.standard_normal(n) * 0.002)

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def load_parent_sequence():
    """Load parent sequence from global variable if available."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def try_and_update(seq, best_tracker):
    """Evaluate sequence and update best tracker."""
    try:
        s = float(evaluate_sequence(seq.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(s) and s < best_tracker[0]:
        best_tracker[0] = s
        best_tracker[1] = seq.copy()
        print(f"  NEW BEST: {s:.8f} (n={len(seq)})")
    return s


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization routine:
    Phase 1: LSE-Adam at parent length
    Phase 2: Coordinate descent polishing
    Phase 3: Multi-length exploration
    Phase 4: Intensive alternating refinement
    Phase 5: Final micro-polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    rng = np.random.default_rng(seed)
    best_tracker = [float('inf'), None]

    # Load parent
    parent = load_parent_sequence()
    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
        try_and_update(parent, best_tracker)
    else:
        parent = np.ones(512) / 512
        try_and_update(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: LSE-Adam at parent length ===
    print("\n=== Phase 1: LSE-Adam ===")
    t1_end = deadline - 1100
    t_alloc = max(60, t1_end - time.monotonic())
    print(f"  LSE-Adam n={n_parent}, t={t_alloc:.0f}s")
    f_opt, _ = lse_adam_optimize(parent, time_limit=t_alloc,
                                  beta_start=50.0, beta_end=50000.0, lr_init=0.1)
    try_and_update(f_opt, best_tracker)

    # Quick coordinate descent follow-up
    remaining = t1_end - time.monotonic()
    if remaining > 30:
        print(f"  Coord descent n={n_parent}, t={remaining - 10:.0f}s")
        f_cd, _ = coordinate_descent(best_tracker[1], time_limit=remaining - 10)
        try_and_update(f_cd, best_tracker)

    # === Phase 2: Directed subgradient ===
    print("\n=== Phase 2: Directed Subgradient ===")
    t2_end = deadline - 800
    remaining = t2_end - time.monotonic()
    if remaining > 30:
        t_alloc = min(150, remaining - 10)
        print(f"  Directed subgrad n={n_parent}, t={t_alloc:.0f}s")
        f_ds, _ = directed_subgrad_optimize(best_tracker[1], time_limit=t_alloc,
                                            n_peaks=3, lr_init=0.003, momentum=0.93)
        try_and_update(f_ds, best_tracker)
        # Follow with coord descent
        remaining2 = t2_end - time.monotonic()
        if remaining2 > 20:
            f_cd2, _ = coordinate_descent(best_tracker[1], time_limit=remaining2 - 5)
            try_and_update(f_cd2, best_tracker)

    # === Phase 3: Multi-length exploration ===
    print("\n=== Phase 3: Multi-length Exploration ===")
    t3_end = deadline - 500
    n_candidates = sorted(set([
        n_parent - 64, n_parent - 32, n_parent, n_parent + 32,
        n_parent + 64, n_parent + 128, 640, 896, 1024
    ]))
    n_candidates = [nc for nc in n_candidates if 256 <= nc <= 1400]

    for nc in n_candidates:
        remaining = t3_end - time.monotonic()
        if remaining < 40:
            break
        t_alloc = min(80, remaining * 0.35)
        if t_alloc < 20:
            break

        if nc == n_parent:
            f_init = best_tracker[1].copy()
        else:
            f_init = interpolate_to_length(best_tracker[1], nc)

        print(f"  LSE-Adam n={nc}, t={t_alloc:.0f}s")
        f_opt, _ = lse_adam_optimize(f_init, time_limit=t_alloc,
                                      beta_start=30.0, beta_end=30000.0, lr_init=0.08)
        try_and_update(f_opt, best_tracker)

        # Quick coord descent
        remaining2 = t3_end - time.monotonic()
        if remaining2 > 25:
            t_cd = min(30, remaining2 - 5)
            if t_cd > 10:
                f_cd, _ = coordinate_descent(best_tracker[1], time_limit=t_cd)
                try_and_update(f_cd, best_tracker)

    # === Phase 4: Intensive alternating refinement ===
    print("\n=== Phase 4: Intensive Alternating ===")
    t4_end = deadline - 200
    cycles = 0
    while time.monotonic() < t4_end and cycles < 12:
        cycles += 1
        current = best_tracker[1].copy()
        n_cur = len(current)
        remaining = t4_end - time.monotonic()
        if remaining < 50:
            break

        if cycles % 3 == 1:
            t_alloc = min(120, remaining * 0.5)
            if t_alloc > 20:
                print(f"  Cycle {cycles}: LSE-Adam n={n_cur}, t={t_alloc:.0f}s, high-beta")
                f_opt, _ = lse_adam_optimize(current, time_limit=t_alloc,
                                              beta_start=200.0, beta_end=200000.0, lr_init=0.05)
                try_and_update(f_opt, best_tracker)
        elif cycles % 3 == 2:
            t_alloc = min(150, remaining * 0.5)
            if t_alloc > 20:
                print(f"  Cycle {cycles}: Coord descent n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = coordinate_descent(current, time_limit=t_alloc)
                try_and_update(f_opt, best_tracker)
        else:
            t_alloc = min(100, remaining * 0.4)
            if t_alloc > 20:
                print(f"  Cycle {cycles}: Directed subgrad n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = directed_subgrad_optimize(current, time_limit=t_alloc,
                                                     n_peaks=5, lr_init=0.002, momentum=0.94)
                try_and_update(f_opt, best_tracker)

    # === Phase 5: Final micro-polish ===
    print("\n=== Phase 5: Final Micro-Polish ===")
    current = best_tracker[1].copy()
    n_cur = len(current)

    # Ultra-high beta LSE
    remaining = deadline - time.monotonic() - 50
    if remaining > 30:
        t_alloc = min(150, remaining)
        print(f"  Ultra-high beta LSE n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = lse_adam_optimize(current, time_limit=t_alloc,
                                      beta_start=500.0, beta_end=500000.0, lr_init=0.03)
        try_and_update(f_opt, best_tracker)

    # Final coordinate descent
    current = best_tracker[1].copy()
    remaining = deadline - time.monotonic() - 20
    if remaining > 20:
        t_alloc = min(100, remaining)
        print(f"  Final coord descent n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = coordinate_descent(current, time_limit=t_alloc)
        try_and_update(f_opt, best_tracker)

    # Final directed subgradient with very few peaks
    current = best_tracker[1].copy()
    remaining = deadline - time.monotonic() - 10
    if remaining > 15:
        t_alloc = min(60, remaining)
        print(f"  Final directed (n_peaks=2) t={t_alloc:.0f}s")
        f_opt, _ = directed_subgrad_optimize(current, time_limit=t_alloc,
                                             n_peaks=2, lr_init=0.001, momentum=0.95)
        try_and_update(f_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using parent fallback")
    if parent is not None:
        return [float(x) for x in parent]
    return [float(x) for x in np.ones(512) / 512]
