import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP: rows are lags, columns are g indices."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def solve_lp(f, M, n, warm_start=None):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0, sum(g) <= 1 (normalized)."""
    c = -np.ones(n)
    A_ub = build_lp_matrix(f, n)
    b_ub = np.full(2*n-1, M)
    # Add sum(g) <= 1 constraint
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        g = solve_lp(f, M, n)
        if g is None:
            # Relax M
            M *= 1.1
            if it > 100:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = project_simplex(ft)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Refine with bisection
            lo = max(0, best_alpha - 0.1)
            hi = min(1, best_alpha + 0.1)
            for _ in range(10):
                mid = (lo + hi) / 2
                ft = project_simplex((1 - mid) * f + mid * g)
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
            
            f = project_simplex((1 - best_alpha) * f + best_alpha * g)
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, k):
    """Subgradient of max conv at peak index k. sg[m] = 2*a[k-m] if valid."""
    n = len(a)
    sg = np.zeros(n)
    m_start = max(0, k - n + 1)
    m_end = min(k, n - 1)
    if m_start <= m_end:
        indices = np.arange(m_start, m_end + 1)
        sg[indices] = 2.0 * a[k - indices]
    return sg


def compute_multi_subgradient(a, peak_indices, peak_values):
    """Weighted combination of subgradients at multiple peaks."""
    n = len(a)
    total_w = peak_values.sum()
    if total_w < 1e-30:
        return np.zeros(n)
    weights = peak_values / total_w
    sg = np.zeros(n)
    for i, k in enumerate(peak_indices):
        if weights[i] < 1e-10:
            continue
        sg += weights[i] * compute_subgradient(a, int(k))
    return sg


def build_sparse_lp(f, n):
    """Build sparse constraint matrix for LP: A[g,j] = f[g-j] if valid."""
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            if k < 2 * n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_oracle(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    from scipy.sparse import csr_matrix
    A = build_sparse_lp(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def alternating_optimize(a0, time_limit, fg_time_ratio=0.7):

    cycle = 0

        if fg_time > 1.0:

def generate_msequence_init(n, rng):

def generate_periodic_init(n, period, rng):

    a = np.tile(base, n // period + 1)[:n]

            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    n_parent = len(parent) if parent is not None else 768

    # === Phase 1: Quick multi-n exploration (~250s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = []
    for n_try in [n_parent, n_parent // 2, n_parent * 2, 256, 384, 512, 640, 768, 896, 1024]:
        if 128 <= n_try <= 1536 and n_try not in candidate_ns:
            candidate_ns.append(n_try)
    candidate_ns.sort()

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 10:
            break

        # Generate inits
        inits = generate_structured_inits(n_try, rng)
        if parent is not None and n_try != n_parent:
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_parent), parent)
            px = np.maximum(px, 0)
            if px.sum() > 1e-10:
                inits.append(px / px.sum())

        best_for_n = None
        best_for_n_obj = float('inf')

        for ii, init in enumerate(inits[:4]):  # Limit to 4 inits per n
            remaining = deadline - time.monotonic() - 1100
            if remaining < 3:
                break
            t_alloc = min(remaining, 25.0)
            a_opt, obj = fg_descent(init, time_limit=t_alloc, lr_init=0.008, top_k=10)
            if obj < best_for_n_obj:
                best_for_n_obj = obj
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            try_seq(best_for_n, f"explore_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP on best solution (~400s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 800
        if remaining > 20:
            t_alloc = min(remaining, 300.0)
            print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_fw")

            # Subgradient polish after FW
            remaining = deadline - time.monotonic() - 600
            if remaining > 10:
                t_alloc = min(remaining, 80.0)
                a_opt2, obj2 = fg_descent(a_opt, time_limit=t_alloc, lr_init=0.003, top_k=20)
                try_seq(a_opt2, "fw_then_fg")

    print(f"  Phase 2 done. Best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Alternating optimization (~400s) ===
    print("\n=== Phase 3: Alternating FG+LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 400
        if remaining > 20:
            t_alloc = min(remaining, 350.0)
            print(f"  Alternating opt, t={t_alloc:.0f}s")
            a_opt, obj = alternating_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "alternating")

    print(f"  Phase 3 done. Best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Try different n near the best (~250s) ===
    print("\n=== Phase 4: Adjacent n exploration ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        for n_try in [n_best - 64, n_best + 64, n_best - 128, n_best + 128]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 10 or n_try < 128 or n_try > 2048:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            if px.sum() > 1e-10:
                px = px / px.sum()
                t_alloc = min(remaining, 40.0)
                a_opt, obj = fg_descent(px, time_limit=t_alloc, lr_init=0.005, top_k=15)
                try_seq(a_opt, f"adj_n{n_try}")

    print(f"  Phase 4 done. Best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Very fine subgradient
        remaining = deadline - time.monotonic() - 30
        if remaining > 5:
            t_alloc = min(remaining, 80.0)
            a_opt, obj = fg_descent(best_seq, time_limit=t_alloc, lr_init=0.001,
                                    mom_init=0.95, top_k=30)
            try_seq(a_opt, "fine_fg")

        # Short FW-LP
        remaining = deadline - time.monotonic() - 15
        if remaining > 5:
            t_alloc = min(remaining, 40.0)
            a_opt, obj = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_fw")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 5
        if remaining > 3:
            for _ in range(3):
                remaining = deadline - time.monotonic() - 2
                if remaining < 2:
                    break
                n_best = len(best_seq)
                noise = rng.standard_normal(n_best) * 0.002
                f_pert = project_simplex(best_seq + noise)
                a_opt, obj = fg_descent(f_pert, time_limit=min(remaining, 8.0),
                                        lr_init=0.002, top_k=20)
                try_seq(a_opt, "perturb")

    print(f"  Phase 5 done. Best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
