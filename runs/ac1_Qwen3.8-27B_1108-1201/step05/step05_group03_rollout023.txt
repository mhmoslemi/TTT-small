```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective with continuation
2. Softmax parameterization for simplex constraint
3. Multi-length exploration with interpolation
4. Coordinate descent polish with exact 1D solutions
5. Diverse restarts from parent and structured initializations
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max_and_weights(c, T):
    """Compute smooth max and softmax weights of c with temperature T."""
    m = c.max()
    shifted = T * (c - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    log_sum = m * T + np.log(np.exp(shifted).sum())
    val = log_sum / T
    w = np.exp(shifted) / np.exp(shifted).sum()
    return val, w


def obj_grad_lbfqgs(b, n, T):
    """
    Objective and gradient for L-BFGS optimization.
    b: array of length n-1 (b[0] is fixed to 0 for the softmax)
    n: length of the sequence
    T: temperature for log-sum-exp smoothing
    Returns (f, grad).
    """
    # Build full b with b[0] = 0
    b_full = np.zeros(n)
    b_full[1:] = b
    
    # Softmax to get a on the simplex
    b_shifted = b_full - b_full.max()
    a = np.exp(b_shifted)
    a /= a.sum()
    
    # Compute convolution
    c = fast_convolve(a)
    
    # Smooth max
    val, w = smooth_max_and_weights(c, T)
    f = 2.0 * n * val
    
    # Gradient wrt a: G_i = sum_k w_k * (a[k-i] + a[i]) * I(0 <= k-i < n)
    # = P_i + a_i * S_i
    # where P_i = sum_{j=0}^{n-1} w[j+i] * a[j] (cross-correlation)
    # and S_i = sum_{j=0}^{n-1} w[j+i] (sliding sum)
    
    # P via FFT cross-correlation
    fft_size = 1 << (2 * n).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    P = np.fft.irfft(W * np.conj(A), fft_size)[:n]
    
    # S via cumulative sum
    cumsum_w = np.zeros(2 * n)
    cumsum_w[:2*n-1] = np.cumsum(np.concatenate([w, [0]]))
    S = np.zeros(n)
    for i in range(n):
        S[i] = cumsum_w[i + n] - cumsum_w[i]
    
    G = P + a * S
    
    # Gradient wrt b (softmax parameterization)
    # dg/db_j = a_j * (G_j - sum_i a_i * G_i)
    aG = np.dot(a, G)
    grad_b_full = a * (G - aG)
    grad_b = grad_b_full[1:]
    
    return f, grad_b


def optimize_lbfqgs(a_init, T_schedule, time_limit, verbose=False):
    """
    Run L-BFGS with continuation over T values.
    a_init: initial sequence on the simplex
    T_schedule: list of T values (decreasing)
    Returns (best_a, best_obj, best_T).
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    # Convert to b parameterization
    b = np.log(a + 1e-30)
    b = b - b[0]  # Fix b[0] = 0
    b_params = b[1:].copy()
    
    t0 = time.monotonic()
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Run L-BFGS at this T
        result = minimize(
            obj_grad_lbfqgs,
            b_params.copy(),
            args=(n, T),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
        )
        
        if result.success or result.fun < 1e10:
            b_params = result.x
            # Convert back to a
            b_full = np.zeros(n)
            b_full[1:] = b_params
            b_shifted = b_full - b_full.max()
            a_new = np.exp(b_shifted)
            a_new /= a_new.sum()
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()
            
            obj = fast_convolve(a_new).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new.copy()
            else:
                a = a_new.copy()  # Still update for next continuation step
            
            if verbose:
                print(f"    T={T:.2f}: L-BFGS obj={result.fun:.10f}, true obj={obj:.10f}, iters={result.nit}")
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Use linear interpolation (simpler and more stable)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent(a_init, time_limit, max_rounds=20, verbose=False):
    """
    Coordinate descent with exact 1D minimization.
    For each coordinate i, fix all others and minimize the objective over a_i.
    Since the objective is max of quadratic functions, we find the 1D minimum.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            
            # Fix all coordinates except i
            # The sum constraint: a_i + sum_{j!=i} a_j = 1, so a_i = 1 - sum_{j!=i} a_j
            # But we're minimizing 2n*max(conv(a,a)) with sum(a)=1
            # When we change a_i, the sum changes unless we adjust another coordinate.
            # 
            # Actually, since the objective is scale-invariant, we don't need to fix the sum.
            # We can just minimize 2n*max(conv(a,a))/sum(a)^2 over a_i >= 0 with others fixed.
            # But this is more complex. Instead, let's fix sum=1 and adjust a_i directly.
            # When a_i changes by delta, the sum changes by delta, so we need to renormalize.
            # Let's just try: set a_i to different values, renormalize, and check objective.
            
            # Try several values of a_i
            current = a[i]
            sum_others = a.sum() - current
            
            candidates = [0.0, current * 0.5, current * 0.75, current * 1.25, current * 1.5,
                          sum_others / (n - 1), current + 0.001, current - 0.001]
            candidates = [max(0, c) for c in candidates if max(0, c) < 1.0]
            
            best_val = best_obj
            best_ai = current
            
            for ai in candidates:
                a_test = a.copy()
                a_test[i] = ai
                s = a_test.sum()
                if s < 1e-10:
                    continue
                a_test /= s
                obj = fast_convolve(a_test).max()
                if obj < best_val - 1e-14:
                    best_val = obj
                    best_ai = ai
            
            if best_ai != current:
                a[i] = best_ai
                a /= a.sum()
                improved = True
                if best_val < best_obj:
                    best_obj = best_val
                    best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective with continuation (T schedule)
2. Softmax parameterization for non-negativity and scale invariance
3. FFT-based gradient computation via cross-correlation
4. Multi-length exploration with interpolation
5. Coordinate descent with golden-section 1D minimization
6. Diverse restarts and simulated annealing perturbations
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max_and_weights(c, T):
    """Compute (1/T)*log(sum(exp(T*c))) and softmax weights stably."""
    m = c.max()
    shifted = T * (c - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    log_sum = m * T + np.log(exp_shifted.sum())
    val = log_sum / T
    w = exp_shifted / exp_shifted.sum()
    return val, w


def obj_grad_lbfqgs(b, n, T):
    """
    Objective and gradient for L-BFGS.
    b: parameter vector of length n-1 (b[0] fixed to 0 in softmax)
    n: sequence length
    T: log-sum-exp temperature
    Returns (f, grad) where f = 2n * smooth_max(conv(a,a)).
    """
    b_full = np.zeros(n)
    b_full[1:] = b

    # Softmax with clipping for stability
    b_max = b_full.max()
    b_shifted = b_full - b_max
    b_shifted = np.clip(b_shifted, -100.0, 100.0)
    a = np.exp(b_shifted)
    a_sum = a.sum()
    if a_sum < 1e-30:
        a = np.ones(n) / n
    else:
        a /= a_sum

    # Convolution
    c = fast_convolve(a)

    # Smooth max and weights
    val, w = smooth_max_and_weights(c, T)
    f = 2.0 * n * val

    # Gradient wrt a_i: G_i = sum_k w_k * (a[k-i] + a[i]) * I(0<=k-i<n)
    # = P_i + a_i * S_i
    # P_i = sum_j w[j+i] * a[j] (cross-correlation via FFT)
    # S_i = sum_j w[j+i] (sliding window sum)

    # P via FFT cross-correlation
    fft_size = 1 << (2 * n).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    P = np.fft.irfft(W * np.conj(A), fft_size)[:n]

    # S via cumulative sum of w
    w_extended = np.concatenate([w, np.zeros(n)])
    cumsum_w = np.concatenate([[0.0], np.cumsum(w_extended)])
    S = np.zeros(n)
    for i in range(n):
        S[i] = cumsum_w[i + n] - cumsum_w[i]

    G = P + a * S

    # Gradient wrt b (softmax chain rule):
    # dg/db_j = a_j * (G_j - sum_i a_i * G_i)
    aG = np.dot(a, G)
    grad_b_full = a * (G - aG)
    grad_b = grad_b_full[1:]

    # Clip gradient for numerical stability
    grad_b = np.clip(grad_b, -1e6, 1e6)

    return f, grad_b


def optimize_lbfqgs(a_init, T_schedule, time_limit, verbose=False):
    """
    Run L-BFGS with continuation over T values (decreasing smoothness).
    Returns (best_a, best_true_obj).
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    # Convert to b parameterization (b[0] = 0)
    b = np.log(np.maximum(a, 1e-15))
    b = b - b[0]
    b_params = b[1:].copy()

    t0 = time.monotonic()

    for stage, T in enumerate(T_schedule):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        try:
            result = minimize(
                obj_grad_lbfqgs,
                b_params.copy(),
                args=(n, T),
                method='L-BFGS-B',
                jac=True,
                options={
                    'maxiter': 300,
                    'ftol': 1e-14,
                    'gtol': 1e-8,
                    'maxls': 30,
                    'iprint': -1
                }
            )

            if result.x is not None and np.all(np.isfinite(result.x)):
                b_params = result.x
                # Convert back to simplex
                b_full = np.zeros(n)
                b_full[1:] = b_params
                b_max = b_full.max()
                b_shifted = np.clip(b_full - b_max, -100.0, 100.0)
                a_new = np.exp(b_shifted)
                a_new = np.maximum(a_new, 0.0)
                a_sum = a_new.sum()
                if a_sum > 1e-15:
                    a_new /= a_sum
                else:
                    a_new = best_a.copy()

                obj = fast_convolve(a_new).max()
                a = a_new
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()

                if verbose:
                    print(f"    T={T:.1f} (stage {stage}): L-BFGS iters={result.nit}, "
                          f"smooth_obj={result.fun:.6f}, true_obj={obj:.10f}, best={best_obj:.10f}")

        except Exception as e:
            if verbose:
                print(f"    T={T:.1f}: L-BFGS failed: {e}")
            continue

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent(a_init, time_limit, verbose=False):
    """
    Coordinate descent with golden-section 1D minimization.
    For each coordinate i, minimize the objective over a_i (with renormalization).
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def obj_1d(ai, idx):
        a_test = a.copy()
        a_test[idx] = ai
        st = a_test.sum()
        if st < 1e-10:
            return 1e10
        a_test /= st
        return fast_convolve(a_test).max()

    rounds = 0
    no_improve_rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            current_ai = a[i]
            a_mean = a.mean()
            a_max = max(10.0 * a_mean, current_ai * 3.0, 0.05)

            # Golden section search
            lo = 0.0
            hi = a_max
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = obj_1d(c1, i)
            f2 = obj_1d(c2, i)

            for _ in range(18):
                if hi - lo < 1e-9 * max(1.0, a_max):
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = obj_1d(c1, i)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = obj_1d(c2, i)

            best_ai = (lo + hi) / 2.0
            best_f = min(f1, f2)

            # Also try the current value
            f_curr = obj_1d(current_ai, i)
            if f_curr < best_f:
                best_ai = current_ai
                best_f = f_curr

            if abs(best_ai - current_ai) > 1e-10 * max(1.0, current_ai):
                a[i] = best_ai
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 1e-10:
                    a /= a_sum
                improved = True
                obj = fast_convolve(a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()

        if not improved:
            no_improve_rounds += 1
            if no_improve_rounds >= 2:
                break
        else:
            no_improve_rounds = 0

        if verbose and rounds % 3 == 0:
            print(f"    CD round {rounds}: obj={best_obj:.10f}")

    return best_a, best_obj


def block_refine(a_init, time_limit, block_size=16, verbose=False):
    """
    Block-wise refinement: optimize small blocks using L-BFGS.
    For each block of coordinates, use the others as fixed and optimize the block.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    num_blocks = (n + block_size - 1) // block_size
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 2 * num_blocks:
            break

        block_order = rng.permutation(num_blocks)
        improved_round = False

        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break

            start = bi * block_size
            end = min(start + block_size, n)
            bs = end - start

            # Current block values
            block_vals = a[start:end].copy()
            block_sum = block_vals.sum()
            if block_sum < 1e-12:
                continue

            # Try perturbing the block
            for trial in range(5):
                if time.monotonic() - t0 > time_limit:
                    break

                if trial == 0:
                    # Small perturbation
                    pert = rng.standard_normal(bs) * 0.01 * block_sum
                elif trial == 1:
                    # Larger perturbation
                    pert = rng.standard_normal(bs) * 0.05 * block_sum
                elif trial == 2:
                    # Try uniform
                    candidate_block = np.ones(bs) * block_sum / bs
                elif trial == 3:
                    # Try concentrating
                    candidate_block = np.zeros(bs)
                    candidate_block[bs // 2] = block_sum
                else:
                    # Random simplex point
                    candidate_block = rng.dirichlet(np.ones(bs)) * block_sum

                if trial >= 2:
                    candidate_block = np.maximum(candidate_block, 0)
                    cs = candidate_block.sum()
                    if cs > 1e-12:
                        candidate_block = candidate_block / cs * block_sum
                else:
                    a_test = a.copy()
                    candidate_block = np.maximum(block_vals + pert, 0)
                    cs = candidate_block.sum()
                    if cs < 1e-12:
                        continue
                    candidate_block = candidate_block / cs * block_sum
                    a_test[start:end] = candidate_block
                    a_test = a_test / a_test.sum()
                    obj = fast_convolve(a_test).max()
                    if obj < best_obj - 1e-14:
                        a = a_test
                        best_obj = obj
                        best_a = a.copy()
                        improved_round = True
                    continue

                a_test = a.copy()
                a_test[start:end] = candidate_block
                a_test = a_test / a_test.sum()
                obj = fast_convolve(a_test).max()
                if obj < best_obj - 1e-14:
                    a = a_test
                    best_obj = obj
                    best_a = a.copy()
                    improved_round = True

        if improved_round:
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. L-BFGS continuation on parent
    2. Multi-length exploration
    3. Diverse restarts with L-BFGS
    4. Coordinate descent polish
    5. Block refinement
    6. Final intensive polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.size == 0 or a.sum() < 0.01:
            return float('inf')
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
        return sc

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except (NameError, Exception):
        pass

    if parent is None:
        parent = np.ones(384) / 384.0
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using default parent: n={len(parent)}, score={best_score:.10f}")

    n_parent = len(parent)
    rng_global = np.random.default_rng(seed)

    # === Phase 1: L-BFGS continuation on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: L-BFGS continuation on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 700.0
    if rem > 30:
        T_start = n_parent / 2.0
        T_end = n_parent * 3.0
        n_stages = 7
        T_schedule = [T_start * (T_end / T_start) ** (i / float(n_stages - 1)) for i in range(n_stages)]
        print(f"  T schedule: {[f'{t:.1f}' for t in T_schedule]}")
        t_alloc = min(rem, 350)
        a_opt, obj = optimize_lbfqgs(parent, T_schedule, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, "lbfqgs_parent")

    # === Phase 2: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Multi-length exploration (best={best_score:.10f})")
    print(f"{'='*60}")
    base = best_seq if best_seq is not None else parent

    target_lengths = [512, 768, 1024, 1280]
    # Remove the length we already have
    target_lengths = [ln for ln in target_lengths if ln != len(base)]

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 450.0
        if rem < 60:
            break

        f_interp = interpolate_sequence(base, target_n)

        T_start = target_n / 2.0
        T_end = target_n * 3.0
        n_stages = 5
        T_schedule = [T_start * (T_end / T_start) ** (i / float(n_stages - 1)) for i in range(n_stages)]

        t_alloc = min(rem * 0.4, 180)
        if t_alloc > 20:
            print(f"  L-BFGS n={target_n}, time={t_alloc:.0f}s")
            a_opt, obj = optimize_lbfqgs(f_interp, T_schedule, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, f"lbfqgs_n{target_n}")

            # Quick coordinate descent on the result
            rem2 = deadline - time.monotonic() - 400.0
            if rem2 > 10:
                a_cd, _ = coordinate_descent(a_opt, time_limit=min(rem2, 40))
                try_seq(a_cd, f"cd_n{target_n}")

    # === Phase 3: Diverse restarts ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Diverse restarts (best={best_score:.10f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_best = len(best_seq)

        restarts = []
        # Slightly perturbed
        restarts.append(np.maximum(best_seq + rng_global.standard_normal(n_best) * 0.008, 0))
        # Reversed
        restarts.append(best_seq[::-1].copy())
        # Mixed with uniform
        restarts.append(0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best))
        # Larger perturbation
        restarts.append(np.maximum(best_seq + rng_global.standard_normal(n_best) * 0.025, 0))
        # Frequency-modulated perturbation
        freq = rng_global.integers(1, 5)
        phase = rng_global.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.02 * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
        restarts.append(np.maximum(best_seq * mod, 0))

        for r_idx, f_init in enumerate(restarts):
            rem = deadline - time.monotonic() - 300.0
            if rem < 25:
                break

            f_init = f_init / f_init.sum()

            T_start = n_best / 3.0
            T_end = n_best * 2.0
            n_stages = 5
            T_schedule = [T_start * (T_end / T_start) ** (i / float(n_stages - 1)) for i in range(n_stages)]

            t_alloc = min(rem / max(1, len(restarts) - r_idx), 120)
            if t_alloc > 15:
                a_opt, obj = optimize_lbfqgs(f_init, T_schedule, time_limit=t_alloc, verbose=(r_idx == 0))
                try_seq(a_opt, f"restart_{r_idx}")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Coordinate descent (best={best_score:.10f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150.0
        if rem > 15:
            t_cd = min(rem * 0.7, 150)
            print(f"  Time: {t_cd:.0f}s, n={len(best_seq)}")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "coord_desc")

            # L-BFGS after coordinate descent
            rem2 = deadline - time.monotonic() - 100.0
            if rem2 > 10:
                n_cd = len(a_opt)
                T_schedule = [n_cd, n_cd * 2, n_cd * 4]
                a_opt2, _ = optimize_lbfqgs(a_opt, T_schedule, time_limit=min(rem2, 60), verbose=False)
                try_seq(a_opt2, "lbfqgs_after_cd")

    # === Phase 5: Block refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Block refinement (best={best_score:.10f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80.0
        if rem > 15:
            bs = max(8, min(32, len(best_seq) // 32))
            t_blk = min(rem * 0.6, 80)
            print(f"  Block size: {bs}, Time: {t_blk:.0f}s")
            a_opt, obj = block_refine(best_seq, time_limit=t_blk, block_size=bs)
            try_seq(a_opt, "block")

    # === Phase 6: Final intensive polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish (best={best_score:.10f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_final = len(best_seq)

        # Very sharp L-BFGS
        rem = deadline - time.monotonic() - 40.0
        if rem > 10:
            T_schedule = [n_final * 2, n_final * 4, n_final * 8]
            a_opt, _ = optimize_lbfqgs(best_seq, T_schedule, time_limit=min(rem, 50), verbose=True)
            try_seq(a_opt, "final_lbfqgs_sharp")

        # Coordinate descent
        rem = deadline - time.monotonic() - 15.0
        if rem > 5:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 25))
            try_seq(a_opt, "final_cd")

        # Tiny perturbations + quick optimize
        for trial in range(3):
            rem = deadline - time.monotonic() - 5.0
            if rem < 5:
                break
            rng = np.random.default_rng(900 + trial)
            f_pert = np.maximum(best_seq + rng.standard_normal(n_final) * 0.003, 0)
            f_pert /= f_pert.sum()
            T_sched = [n_final * 3, n_final * 6]
            a_opt, _ = optimize_lbfqgs(f_pert, T_sched, time_limit=min(rem, 15))
            try_seq(a_opt, f"perturb_{trial}")

    # Fallback
    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.ones(384) / 384.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    print(f"{'='*60}")
    return [float(x) for x in best_seq]
```
