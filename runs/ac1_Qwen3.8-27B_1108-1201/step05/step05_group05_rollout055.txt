import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_max = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_max] - 1.0) / (rho_max + 1.0)
    return np.maximum(v - theta, 0.0)

def beta_shape(n, alpha, beta):
    j = np.arange(n, dtype=np.float64)
    x = (j + 0.5) / n
    a = x**alpha * (1.0 - x)**beta + 1e-15  # Avoid 0
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def linear_init(n, slope=1.0):
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + slope * j / (n - 1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def subgradient_descent(a, t_limit, initial_lr=0.005, decay=0.9999, temp_init=1e-4, temp_growth=1.002, top_k=50):
    """
    Subgradient descent on max(conv(a,a)) with simplex constraint.
    Uses softmax-weighted active set for smoother gradients.
    """
    a = simplex_project(a.copy().astype(np.float64))
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)  # Momentum
    momentum = 0.9
    lr = initial_lr
    temp = temp_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < t_limit:
        it += 1
        
        # Softmax-weighted active set
        k_act = min(top_k, 2*n-1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        scaled = pv / temp - pv.max() / temp
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for idx, k in enumerate(peaks):
            if w[idx] < 1e-12:
                continue
            # d conv[k]/d a[i] = 2*a[k-i] for valid i
            # conv[k] = sum_j a[j]*a[k-j], so d/d a[i] = 2*a[k-i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                sg[i] += 2.0 * w[idx] * a[k - i]
        
        # Momentum update
        v = momentum * v - lr * sg
        a_new = simplex_project(a + v)
        
        # Evaluate
        conv = fast_convolve(a_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        lr *= decay
        temp *= temp_growth
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(hash((it, n)) % 2**31)
            noise_scale = 0.002 + 0.002 * min(no_improve // 100, 5)
            a = simplex_project(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            lr = initial_lr * 0.7
            no_improve = 0
    
    return best_a, best_obj
