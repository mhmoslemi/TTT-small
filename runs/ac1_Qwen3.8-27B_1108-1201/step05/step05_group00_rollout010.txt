def fast_convolve(a):

def nesterov_smooth_grad(f_half0, time_limit, T_init=0.0003, lr_init=0.08, beta_init=0.9):
    """Nesterov-accelerated smooth gradient on the palindromic half-sequence."""
    m = len(f_half0)
    n = 2 * m
    
    f = np.maximum(f_half0.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(m) / m
    else:
        f = f / s
    
    full = half_to_full(f)
    best_f = f.copy()
    best_obj = fast_convolve(full).max()
    
    T = T_init
    lr = lr_init
    beta = beta_init
    
    # Nesterov momentum
    y = f.copy()
    v = np.zeros(m)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        y_full = half_to_full(y)
        conv = fast_convolve(y_full)
        
        # Logsumexp gradient
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient of logsumexp w.r.t. f_half
        # d(logsumexp)/df_half[i] = sum_k p[k] * d(conv[k])/df_half[i]
        # d(conv[k])/df_half[i] = contribution of b_i to conv[k]
        # = f[k-i] (if valid) + f[k-n+1+i] (if valid) for the FULL sequence
        # Actually, let's just compute the gradient numerically via the constraint matrix
        
        # More efficiently: grad[i] = sum_k p[k] * A[k, i] where A is the constraint matrix
        # But A depends on f (the current point), so:
        A = build_lp_matrix(y_full, m)
        grad = A.T @ p  # shape (m,)
        grad = 2.0 * grad  # factor of 2 because conv is quadratic
        
        # Nesterov update
        v = beta * v - lr * grad
        y = f + beta * v  # Nesterov lookahead
        y = project_simplex(y)
        
        # Evaluate at y and at v-step from f
        candidate = project_simplex(f + v)
        
        cand_full = half_to_full(candidate)
        cand_obj = fast_convolve(cand_full).max()
        
        if cand_obj < best_obj - 1e-14:
            best_obj = cand_obj
            best_f = candidate.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = candidate
        
        # Adapt hyperparameters
        if no_improve > 30:
            T *= 1.03
            lr *= 0.98
            beta = min(0.99, beta + 0.001)
        if no_improve > 100:
            T *= 0.8
            beta = 0.9
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 200) / 300.0)
            f = project_simplex(best_f + rng.standard_normal(m) * noise)
            y = f.copy()
            v = np.zeros(m)
            T = T_init
            lr = lr_init
            beta = beta_init
            no_improve = 0
    
    return best_f, best_obj

    m = len(f_half0)
    n = 2 * m
    
    f = np.maximum(f_half0.copy(), 0.0)
    f = f / f.sum()
    
    best_f = f.copy()
    best_obj = fast_convolve(half_to_full(f)).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for i in range(m):
            if time.monotonic() - t0 > time_limit - 0.01:
                break
            
            # Optimize coordinate i
            # Fix all other coordinates, find optimal f[i]
            # The objective is max_k (a*a)[k] where a = half_to_full(f)
            # (a*a)[k] is quadratic in f[i]
            # We want to minimize the max over k
            
            # Simple approach: try a few values of f[i]
            old_val = f[i]
            f[i] = 0.0
            # After changing f[i], we need to renormalize
            # But for exact coordinate descent, we fix the sum constraint
            
            # Alternative: work with the constraint sum(f) = 1
            # If we change f[i] to f[i] + d, we need to adjust others to maintain sum
            # Simpler: just try a few perturbations
            
            # Try shifting from coordinate i to a random other coordinate
            if m > 1:
                j = (i + 1) % m  # or random
                for d in [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02]:
                    candidate = f.copy()
                    if candidate[i] + d >= 0 and candidate[j] - d >= 0:
                        candidate[i] += d
                        candidate[j] -= d
                        cand_full = half_to_full(candidate)
                        val = fast_convolve(cand_full).max()
                        if val < best_obj - 1e-15:
                            f = candidate
                            best_obj = val
                            best_f = f.copy()
                            improved = True
                        if val >= best_obj:
                            break  # Moving too far
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0
    
    return best_f, best_obj
