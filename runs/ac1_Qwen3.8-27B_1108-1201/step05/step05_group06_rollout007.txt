- Std ≈ sqrt(2/(27n^3)) ≈ 0.27/n^{1.5}

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.max(np.nonzero(cond)[0])
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_multi_peak_subgrad(a, conv, top_k=30, power=2.0):

def coordinate_bisection_fast(a0, time_limit, max_passes=200, verbose=False):
    """
    Fast coordinate bisection: for each coordinate j, find the optimal value
    by ternary search on the 1D function f(x) = max_k conv_with_j(x)_k.
    """
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        improved_this_pass = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            # Binary search for optimal x_j in [0, 1]
            # f(x) = max_k [ (1-x)^2 * B[k] + 2x(1-x) * b_valid[k] + x^2 * delta(2j=k) ]
            # where B = conv of others (normalized to sum=1)
            b = a.copy()
            b[j] = 0.0
            b = b / s_others
            B = fast_convolve(b)
            conv_len = 2 * n - 1

            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            has_diag = (2 * j < conv_len)
            diag_idx = 2 * j

            lo = 0.0
            hi = 1.0
            # 40 iterations of bisection
            for _ in range(40):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if has_diag:
                    f_vals[diag_idx] += mid * mid
                k_max = int(np.argmax(f_vals))
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (has_diag and k == diag_idx) else 0.0
                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))
                if deriv > 1e-12:
                    hi = mid
                elif deriv < -1e-12:
                    lo = mid
                else:
                    break

            x_new = (lo + hi) / 2.0
            if abs(x_new - x_old) < 1e-14:
                continue

            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check

            conv = fast_convolve(a)
            obj = conv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                improved_this_pass = True

        if not improved_this_pass:
            if passes > 3:
                rng = np.random.default_rng(passes * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0005)
            else:
                break

        if verbose and passes % 10 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

    return best_a, best_obj


def simulated_anneal(a0, time_limit, t_init=0.005, t_final=0.00005, verbose=False):
    """Simulated annealing with block perturbations."""
    a = a0.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)

    conv = fast_convolve(a)
    cur_obj = conv.max()
    best_obj = cur_obj
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    it = 0
    accept_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = (time.monotonic() - t0) / time_limit
        T = t_init * (t_final / t_init) ** progress

        # Block perturbation
        block_size = int(rng.integers(3, 15))
        start = int(rng.integers(0, max(1, n - block_size)))
        idx = np.arange(start, start + block_size)

        a_trial = a.copy()
        a_trial[idx] += rng.standard_normal(block_size) * 0.015
        a_trial = project_simplex(a_trial)

        conv_trial = fast_convolve(a_trial)
        obj_trial = conv_trial.max()

        delta = obj_trial - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_trial
            conv = conv_trial
            cur_obj = obj_trial
            accept_count += 1
            if obj_trial < best_obj - 1e-13:
                best_obj = obj_trial
                best_a = a.copy()

        if verbose and it % 10000 == 0:
            print(f"    SA iter {it}: best={best_obj:.10f}, cur={cur_obj:.10f}, T={T:.6f}")

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetrize(a):
    """Make sequence symmetric and renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining backtracking subgradient, targeted pair
    exchange, coordinate bisection, simulated annealing, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 40.0)

    best_a = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        p_score = 2.0
        print("No parent, using uniform n=768")

    n_parent = len(parent)

    def update_best(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        a_norm = a_try / s_try
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")

    # === Phase 1: Backtracking subgradient on parent ===
    print(f"\n=== Phase 1: Backtracking SG (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 800
    if rem > 60:
        t_alloc = min(rem, 500)
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = backtracking_subgrad(parent, time_limit=t_alloc, lr_init=0.006,
                                          top_k=40, power=2.5, verbose=True)
        update_best(a_opt, "bt_sg_parent")

    # === Phase 2: Targeted pair exchange ===
    print(f"\n=== Phase 2: Targeted pair exchange (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    rem = deadline - time.monotonic() - 600
    if rem > 40:
        t_alloc = min(rem, 300)
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = targeted_pair_exchange(base, time_limit=t_alloc, verbose=True)
        update_best(a_opt, "pair_exchange")

    # === Phase 3: Coordinate bisection ===
    print(f"\n=== Phase 3: Coordinate bisection (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = coordinate_bisection_fast(base, time_limit=t_alloc,
                                               max_passes=60, verbose=True)
        update_best(a_opt, "coord_bisect")

    # === Phase 4: Simulated annealing ===
    print(f"\n=== Phase 4: Simulated annealing (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = simulated_anneal(base, time_limit=t_alloc,
                                      t_init=0.004, t_final=0.0001, verbose=True)
        update_best(a_opt, "sim_anneal")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent

    for target_n in [1024, 1536]:
        rem = deadline - time.monotonic() - 250
        if rem < 50:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_sg = min(rem * 0.5, 150)
        t_pe = min(rem * 0.3, 100)
        t_cd = min(rem * 0.2, 50)

        print(f"  n={target_n}: SG {t_sg:.0f}s, PE {t_pe:.0f}s, CD {t_cd:.0f}s")

        a_opt1, _ = backtracking_subgrad(f_interp, time_limit=t_sg, lr_init=0.005,
                                         top_k=40, power=2.0)
        update_best(a_opt1, f"sg_n{target_n}")

        base2 = best_a if best_a is not None and len(best_a) == target_n else a_opt1
        if len(base2) != target_n:
            base2 = interpolate_to_length(base2, target_n)

        rem2 = deadline - time.monotonic() - 200
        if rem2 > 20:
            a_opt2, _ = targeted_pair_exchange(base2, time_limit=min(t_pe, rem2))
            update_best(a_opt2, f"pe_n{target_n}")

        base3 = best_a if best_a is not None and len(best_a) == target_n else a_opt2
        if len(base3) != target_n:
            base3 = interpolate_to_length(base3, target_n)

        rem3 = deadline - time.monotonic() - 150
        if rem3 > 15:
            a_opt3, _ = coordinate_bisection_fast(base3, time_limit=min(t_cd, rem3),
                                                   max_passes=30)
            update_best(a_opt3, f"cd_n{target_n}")

    # === Phase 6: Symmetric variant ===
    print(f"\n=== Phase 6: Symmetric (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 40:
            a_sym = symmetrize(best_a)
            n_sym = len(a_sym)
            t_sg = min(rem * 0.5, 100)
            t_cd = min(rem * 0.4, 80)
            print(f"  Symmetric n={n_sym}: SG {t_sg:.0f}s, CD {t_cd:.0f}s")
            a_opt, _ = backtracking_subgrad(a_sym, time_limit=t_sg, lr_init=0.004,
                                            top_k=30, power=2.0)
            update_best(a_opt, "sym_sg")
            rem2 = deadline - time.monotonic() - 120
            if rem2 > 15:
                base_s = best_a if best_a is not None and len(best_a) == n_sym else a_opt
                if len(base_s) != n_sym:
                    base_s = interpolate_to_length(base_s, n_sym)
                a_opt2, _ = coordinate_bisection_fast(base_s, time_limit=min(t_cd, rem2),
                                                       max_passes=20)
                update_best(a_opt2, "sym_cd")

    # === Phase 7: Final intensive polish on best ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)

        # Multiple short CD passes
        rem = deadline - time.monotonic() - 30
        n_passes = max(1, int(rem // 40))
        for p in range(n_passes):
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            t_alloc = min(rem, 40)
            a_opt, _ = coordinate_bisection_fast(best_a, time_limit=t_alloc,
                                                 max_passes=10)
            update_best(a_opt, f"final_cd_{p}")

        # Final subgradient with very small lr
        rem = deadline - time.monotonic() - 15
        if rem > 10 and best_a is not None:
            a_opt, _ = backtracking_subgrad(best_a, time_limit=min(rem, 30),
                                            lr_init=0.002, top_k=20, power=3.0)
            update_best(a_opt, "final_sg")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        if rem > 8 and best_a is not None:
            for p in range(3):
                rem = deadline - time.monotonic() - 8
                if rem < 5:
                    break
                rng_p = np.random.default_rng(seed * 1000 + p * 7 + 1)
                f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.0008)
                a_opt, _ = coordinate_bisection_fast(f_pert, time_limit=min(rem, 10),
                                                     max_passes=5)
                update_best(a_opt, f"perturb_{p}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
