def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def get_fft_size(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()

# GS optimization
def gs_step(a, L, target_flat, blend):
    """One GS iteration."""
    n = len(a)
    a_pad = np.zeros(L)
    a_pad[:n] = a
    F = np.fft.rfft(a_pad)
    mag = np.abs(F)
    phase = np.angle(F)
    
    # Target: flat spectrum except DC
    target = target_flat * np.ones(len(mag))
    target[0] = 1.0
    
    new_mag = blend * mag + (1.0 - blend) * target
    new_mag[0] = 1.0
    
    F_new = new_mag * np.exp(1j * phase)
    a_pad_new = np.fft.irfft(F_new, L)
    
    a_new = np.maximum(a_pad_new[:n], 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s

def gs_optimize(a0, time_limit, blend=0.5, verbose=False):

# Subgradient with Nesterov
def compute_subgradient(a, conv, top_k):
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    # Use sharp weights (only the max)
    max_val = pv.max()
    # Include peaks within 1% of max
    w = np.ones(len(peaks))
    w[pv < max_val * 0.995] = 0.0
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg

def coordinate_descent(f0, time_limit, verbose=False):
    """Cycle through coordinates, trying to reduce the peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)

        f = f / s

    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_val = old_val
            best_obj_j = cur_peak
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_obj_j - 1e-14:
                    best_obj_j = val
                    best_val = new_val
            
            if best_val != old_val:
                denom = 1.0 - best_val
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    f[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f /= f.sum()
                if best_obj_j < best_obj - 1e-13:
                    best_obj = best_obj_j
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj

# Interpolation
def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def structured_inits(n, rng, count=5):
    inits = []
    inits.append(np.ones(n) / n)
    
    x = np.linspace(0, 1, n)
    # Various window functions
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    inits.append(np.maximum(hann, 1e-10) / np.maximum(hann, 1e-10).sum())
    
    # Gaussian
    center = n / 2.0
    sigma = n / 5.0
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(np.maximum(rc, 1e-10) / np.maximum(rc, 1e-10).sum())
    
    # Random
    r = np.abs(rng.standard_normal(n))
    inits.append(r / r.sum())
    
    return inits[:count]

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines Gerchberg-Saxton frequency-domain spectral shaping, Nesterov-accelerated
subgradient with adaptive scheduling, structured peak-reduction local search,
and multi-length exploration. Key innovation: geometric spectral interpolation
in the GS step for smoother convergence, and alternating GS/subgradient sweeps.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def get_fft_size(n):
    """Next power of 2 >= 2n-1."""
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def gs_step_geometric(a, L, alpha):
    """One GS iteration with geometric spectral interpolation."""
    n = len(a)
    a_pad = np.zeros(L)
    a_pad[:n] = a
    F = np.fft.rfft(a_pad)
    mag = np.abs(F)
    phase = np.angle(F)

    target = np.ones(len(mag))

    # Geometric interpolation: pulls magnitudes toward 1
    # |F_new| = |F|^(1-alpha) * 1^alpha = |F|^(1-alpha)
    # This preserves the "shape" while reducing peak-to-average ratio
    eps = 1e-15
    new_mag = np.power(np.maximum(mag, eps), 1.0 - alpha)
    new_mag[0] = 1.0  # Enforce DC = 1

    F_new = new_mag * np.exp(1j * phase)
    a_pad_new = np.fft.irfft(F_new, L)

    a_new = np.maximum(a_pad_new[:n], 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def gs_step_linear(a, L, blend):
    """One GS iteration with linear spectral blending."""
    n = len(a)
    a_pad = np.zeros(L)
    a_pad[:n] = a
    F = np.fft.rfft(a_pad)
    mag = np.abs(F)
    phase = np.angle(F)

    target = np.ones(len(mag))
    new_mag = blend * mag + (1.0 - blend) * target
    new_mag[0] = 1.0

    F_new = new_mag * np.exp(1j * phase)
    a_pad_new = np.fft.irfft(F_new, L)

    a_new = np.maximum(a_pad_new[:n], 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def gs_optimize(a0, time_limit, mode="geometric", alpha=0.05, blend=0.5,
                damping=0.9, verbose=False):
    """GS optimization with optional damping to previous iterate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    L = get_fft_size(n)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if mode == "geometric":
            a_new = gs_step_geometric(a, L, alpha)
        else:
            a_new = gs_step_linear(a, L, blend)

        # Damping: blend with previous
        a_new = (1.0 - damping) * a + damping * a_new
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a
        else:
            a_new /= sn
        a = a_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()

        if verbose and it % 100 == 0:
            print(f"    GS({mode}) iter {it}, t={time.monotonic()-t0:.1f}s, obj={best_obj:.10f}")

    return best_a, best_obj


def compute_subgradient(a, conv, top_k):
    """Compute weighted subgradient of max(conv)."""
    n = len(a)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    max_val = pv.max()

    # Weight peaks that are close to the maximum
    w = np.ones(len(peaks))
    w[pv < max_val * 0.99] = 0.0
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * a[k - idx]
    return sg


def subgradient_optimize(f0, time_limit, top_k=30, lr=0.004, momentum=0.90,
                         restart_every=250, verbose=False):
    """Subgradient with Nesterov momentum and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        f_nest = f * np.exp(0.5 * momentum * v)
        fn = f_nest.sum()
        if fn < 1e-30:
            f_nest = np.ones(n) / n
        else:
            f_nest /= fn

        conv_nest = fast_convolve(f_nest)
        sg = compute_subgradient(f_nest, conv_nest, top_k)

        v = momentum * v + (1.0 - momentum) * sg

        # Multiplicative update
        exponent = -cur_lr * v
        exponent -= exponent.max()
        f = f * np.exp(exponent)
        fs = f.sum()
        if fs < 1e-30:
            f = best_f.copy()
            v = np.zeros(n)
            cur_lr *= 0.5
            continue
        f /= fs

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        if cur_lr < 0.0001:
            cur_lr = 0.0001

        if no_improve > restart_every:
            rng_r = np.random.default_rng((it * 31 + 7) % 1000000)
            noise = rng_r.standard_normal(n) * (0.002 + 0.002 * min(no_improve // 500, 5))
            f = np.maximum(best_f + noise, 0.0)
            fsum = f.sum()
            if fsum < 1e-15:
                f = best_f.copy()
            else:
                f /= fsum
            v = np.zeros(n)
            cur_lr = lr * 0.6
            no_improve = 0

    return best_f, best_obj


def alternating_gs_sg(f0, time_limit, gs_steps=5, sg_steps=50,
                      gs_alpha=0.05, top_k=30, lr=0.003, momentum=0.88,
                      verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    else:
        f = f / s
    n = len(f)
    L = get_fft_size(n)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    v = np.zeros(n)
    it = 0

    while time.monotonic() - t0 < time_limit:
        # GS phase
        for _ in range(gs_steps):
            if time.monotonic() - t0 > time_limit:
                break
            f_new = gs_step_geometric(f, L, gs_alpha)
            f = 0.7 * f + 0.3 * f_new
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f /= fs

        # Subgradient phase
        for _ in range(sg_steps):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1

            f_nest = f * np.exp(0.5 * momentum * v)
            fn = f_nest.sum()
            if fn < 1e-30:
                f_nest = np.ones(n) / n
            else:
                f_nest /= fn

            conv_nest = fast_convolve(f_nest)
            sg = compute_subgradient(f_nest, conv_nest, top_k)
            v = momentum * v + (1.0 - momentum) * sg

            exponent = -lr * v
            exponent -= exponent.max()
            f = f * np.exp(exponent)
            fs = f.sum()
            if fs < 1e-30:
                f = best_f.copy()
                v = np.zeros(n)
                break
            f /= fs

        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()

    return best_f, best_obj


def peak_reduction_search(f0, time_limit, verbose=False):
    """
    Targeted local search: identify autocorrelation peaks,
    move mass from high-contributing positions to reduce them.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        top_peaks = np.argpartition(conv, -min(15, 2 * n - 1))[-min(15, 2 * n - 1):]
        cur_max = conv.max()

        if cur_max < best_obj:
            best_obj = cur_max
            best_f = f.copy()

        improved = False
        for pk in sorted(top_peaks, key=lambda x: -conv[int(x)])[:6]:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = f[idx] * f[pk - idx]

            if contribs.sum() < 1e-15:
                continue

            hi_pos = idx[np.argmax(contribs)]
            hi_val = f[hi_pos]
            if hi_val < 1e-10:
                continue

            # Try moving mass to positions that don't contribute to this peak
            for attempt in range(15):
                if time.monotonic() - t0 > time_limit:
                    return best_f, best_obj
                lo_pos = rng.integers(0, n)
                if lo_pos == hi_pos:
                    continue
                # Prefer positions far from this peak
                for frac in [0.05, 0.1, 0.15, 0.25]:
                    amt = hi_val * frac
                    f[hi_pos] = hi_val - amt
                    f[lo_pos] += amt
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                    else:
                        f[hi_pos] = hi_val
                        f[lo_pos] -= amt
                        break
            if improved:
                break

        if not improved:
            # Random pair exchange
            found = False
            for _ in range(20):
                if time.monotonic() - t0 > time_limit:
                    return best_f, best_obj
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or f[i] < 1e-10:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[i] * frac
                    f[i] -= amt
                    f[j] += amt
                    val = fast_convolve(f).max()
                    if val < best_obj - 1e-14:
                        best_obj = val
                        best_f = f.copy()
                        found = True
                        break
                    else:
                        f[i] += amt
                        f[j] -= amt
                        break
                if found:
                    break
            if not found:
                # Perturb and continue
                noise = rng.standard_normal(n) * 0.003
                f = np.maximum(best_f + noise, 0.0)
                f /= f.sum()

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent cycling through all coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            old_val = f[j]
            if old_val < 1e-12:
                continue

            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_val = old_val
            best_obj_j = cur_peak

            for mult in [0.5, 0.7, 0.85, 1.2, 1.4, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_obj_j - 1e-14:
                    best_obj_j = val
                    best_val = new_val

            if best_val != old_val:
                denom = 1.0 - best_val
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    f[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0.0)
                    f /= f.sum()
                if best_obj_j < best_obj - 1e-13:
                    best_obj = best_obj_j
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_inits(n, rng, count=5):
    """Generate diverse structured initial sequences."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.linspace(0, 1, n)
    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    inits.append(np.maximum(hann, 1e-10) / np.maximum(hann, 1e-10).sum())

    # Gaussian
    center = n / 2.0
    sigma = n / 4.0
    gauss = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(np.maximum(rc, 1e-10) / np.maximum(rc, 1e-10).sum())

    # Two-bump
    c1 = n * 0.3
    c2 = n * 0.7
    sig = n / 8.0
    two_bump = np.exp(-0.5 * ((np.arange(n) - c1) / sig) ** 2) + np.exp(-0.5 * ((np.arange(n) - c2) / sig) ** 2)
    inits.append(two_bump / two_bump.sum())

    return inits[:count]

            a_gs, _ = gs_optimize(init, min(25, rem * 0.4), mode="geometric", alpha=0.05)
            gs_results.append(a_gs)
            best_seq, best_score = try_and_update(a_gs, best_seq, best_score, f"gs_init_n{n_try}")

    # === Phase 2: Subgradient on best candidates (700s) ===
    print(f"\n=== Phase 2: Subgradient (best={best_score:.6f}) ===")

    # Collect and rank candidates
    all_cands = [parent] + gs_results + [best_seq]
    cand_objs = []
    for c in all_cands:
        if c is None:
            continue
        conv = fast_convolve(c)
        obj = len(c) * conv.max()
        cand_objs.append((obj, c))
    cand_objs.sort(key=lambda x: x[0])

    for idx, (co, cand) in enumerate(cand_objs[:5]):
        rem = deadline - time.monotonic() - 250
        if rem < 15:
            break
        n_c = len(cand)

        # Multiple subgradient configs
        configs = [
            (max(20, n_c // 8), 0.004, 0.90),
            (max(10, n_c // 12), 0.002, 0.93),
            (max(30, n_c // 6), 0.006, 0.85),
        ]
        for cfg_idx, (tk, lr_c, mom) in enumerate(configs):
            rem = deadline - time.monotonic() - 180
            if rem < 10:
                break
            t_alloc = min(rem * 0.25, 100)
            if t_alloc < 8:
                break
            print(f"  SG cand{idx} cfg{cfg_idx}: n={n_c}, t={t_alloc:.0f}s, top_k={tk}")
            a_opt, _ = subgradient_optimize(cand, t_alloc, top_k=tk, lr=lr_c, momentum=mom)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"sg_{idx}_{cfg_idx}")

    # === Phase 3: Alternating GS+SG (200s) ===
    print(f"\n=== Phase 3: Alternating GS+SG (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alt = min(rem * 0.5, 100)
            n_b = len(best_seq)
            print(f"  Alt GS+SG: n={n_b}, t={t_alt:.0f}s")
            a_opt, _ = alternating_gs_sg(best_seq, t_alt, gs_steps=5, sg_steps=40,
                                         gs_alpha=0.04, top_k=max(20, n_b // 8),
                                         lr=0.003, momentum=0.88)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "alt_gs_sg")

        # Another alternating with different params
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alt2 = min(rem * 0.5, 60)
            a_opt, _ = alternating_gs_sg(best_seq, t_alt2, gs_steps=10, sg_steps=20,
                                         gs_alpha=0.08, top_k=max(10, n_b // 10),
                                         lr=0.002, momentum=0.92)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "alt_gs_sg2")

    # === Phase 4: Local Search (250s) ===
    print(f"\n=== Phase 4: Local Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_pr = min(rem * 0.5, 100)
            print(f"  Peak reduction: t={t_pr:.0f}s")
            a_opt, _ = peak_reduction_search(best_seq, t_pr)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "peak_red")

        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_cd = min(rem * 0.6, 60)
            print(f"  Coord descent: t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, t_cd)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "cd")

    # === Phase 5: Multi-length exploration (120s) ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_lengths = sorted(set([base_n * 2, 1536, 2048]))
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 50
            if rem < 20 or target_n > 4096:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_total = min(rem * 0.35, 50)

            # GS init
            a_gs, _ = gs_optimize(f_interp, min(t_total * 0.3, 15), mode="geometric", alpha=0.05)
            # Subgradient refine
            a_opt, _ = subgradient_optimize(a_gs, min(t_total * 0.7, 35),
                                            top_k=max(20, target_n // 8), lr=0.003, momentum=0.88)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"len_{target_n}")

    # === Phase 6: Final Polish (60s) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine subgradient
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            a_opt, _ = subgradient_optimize(best_seq, min(rem, 30),
                                            top_k=max(10, n_best // 4), lr=0.0008, momentum=0.95)
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "fine_sg")

        # Peak reduction
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_opt, _ = peak_reduction_search(best_seq, min(rem, 20))
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "fine_pr")

        # Coordinate descent (only if n is small enough)
        rem = deadline - time.monotonic() - 3
        if rem > 3 and n_best <= 1536:
            a_opt, _ = coordinate_descent(best_seq, min(rem, 15))
            best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "fine_cd")

        # Final micro-perturbation
        rem = deadline - time.monotonic() - 1
        if rem > 2:
            for scale in [0.0005, 0.001]:
                noise = rng.standard_normal(n_best) * scale
                f_pert = np.maximum(best_seq + noise, 0.0)
                f_pert /= f_pert.sum()
                a_opt, _ = subgradient_optimize(f_pert, min(rem, 5),
                                                top_k=max(10, n_best // 6), lr=0.001, momentum=0.94)
                best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"pert_{scale}")
                rem = deadline - time.monotonic() - 1
                if rem < 2:
                    break

    # Final verification
    if best_seq is None:
        best_seq = parent.copy()
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
