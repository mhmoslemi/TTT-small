import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def project_simplex_nonneg(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    return target_sum * project_simplex(v / target_sum)

def hankel_sensitive(a, k):
    """Return the vector s[j] = a[k-j] for valid j, which is the 'conjugate' for peak k."""
    n = len(a)
    s = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        s[j] = a[k - j]
    return s

def subgradient_conv(a):
    """Compute subgradient of max_k conv(a,a)[k]. Returns (max_val, active_set, gradient)."""
    conv = fast_convolve(a)
    max_val = conv.max()
    # Active set: positions where conv is within 1e-10 of max

    active = np.where(conv >= max_val - tol)[0]
    
    # Subgradient is sum over active of 2 * hankel_conjugate

            grad[j] += 2.0 * a[int(k) - j]
    
    # Normalize subgradient by number of active peaks (averaging)
    grad /= len(active)
    return max_val, active, grad

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, x>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_conjugate(a, k):
    """Return vector c where c[j] = a[k-j] for valid j, else 0."""
    n = len(a)
    c = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        c[j] = a[k - j]
    return c


def subgradient_exact(a, top_k=10):
    """
    Compute weighted subgradient of max_k conv(a,a)[k] using top_k peaks.
    Returns (max_val, subgradient).
    """
    conv = fast_convolve(a)
    max_val = conv[0] if len(conv) == 1 else conv.max()
    
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        pv = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
    
    # Weight: exponential of (pv - max_val) / (range) to focus on near-max
    pv_max = pv.max()
    pv_min = pv.min()
    spread = pv_max - pv_min

    else:
        t = (pv - pv_min) / spread

        idx = np.arange(js, je + 1)

            no_improve = 0
    
    return best_a, best_obj


def anti_corr_optimize(a0, time_limit):
    """
    Alternating optimization exploiting the two-half anti-correlation structure.
    Splits a into [p, q] and alternates optimizing each half.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    m = n // 2
    m2 = n - m  # second half length (for odd n)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    lr = 0.01
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Optimize first half (fix second half)
        for inner in range(30):
            if time.monotonic() - t0 > time_limit - 0.05:
                break
            p = a[:m]
            q = a[m:]
            # Gradient of max(conv) w.r.t. p only
            conv = fast_convolve(a)
            max_val = conv.max()
            k_peak = int(np.argmax(conv))
            
            # Sensitivity of peak w.r.t. p[j]: 2 * a[k_peak - j] (if valid)
            grad_p = np.zeros(m)
            for j in range(m):
                # Contribution to peak from a[j]: a[j] * a[k_peak - j]
                # d/d a[j] of sum_j' a[j']*a[k-j'] = a[k-j] + (if j' = k-j: a[j])
                # Simplified: 2 * a[k_peak - j] if k_peak - j is in [m, n-1] or [0, m-1]
                kj = k_peak - j
                if 0 <= kj < n:
                    grad_p[j] = 2.0 * a[kj]
            
            # Also add regularizer to keep p spread out
            grad_p += 0.01 * p  # Small L2 to prevent concentration
            
            p_new = np.maximum(p - lr * grad_p / (np.max(np.abs(grad_p)) + 1e-15), 0.0)
            p_new_sum = p_new.sum()
            if p_new_sum < 1e-15:
                p_new = p.copy()
            else:
                # Keep total sum constant
                target_sum = p.sum() + q.sum()
                p_new = p_new * (target_sum - q.sum()) / p_new_sum
                p_new = np.maximum(p_new, 0.0)
            
            a_candidate = np.concatenate([p_new, q])
            obj = fast_convolve(a_candidate).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_candidate.copy()
                a = a_candidate
            elif obj < conv.max() - 1e-14:
                a = a_candidate
            else:
                break
        
        # Optimize second half (fix first half)
        for inner in range(30):
            if time.monotonic() - t0 > time_limit - 0.05:
                break
            p = a[:m]
            q = a[m:]
            conv = fast_convolve(a)
            max_val = conv.max()
            k_peak = int(np.argmax(conv))
            
            grad_q = np.zeros(m2)
            for j in range(m2):
                kj = k_peak - (m + j)
                if 0 <= kj < n:
                    grad_q[j] = 2.0 * a[kj]
            
            grad_q += 0.01 * q
            
            q_new = np.maximum(q - lr * grad_q / (np.max(np.abs(grad_q)) + 1e-15), 0.0)
            q_new_sum = q_new.sum()
            if q_new_sum < 1e-15:
                q_new = q.copy()
            else:
                target_sum = p.sum() + q.sum()
                q_new = q_new * (target_sum - p.sum()) / q_new_sum
                q_new = np.maximum(q_new, 0.0)
            
            a_candidate = np.concatenate([p, q_new])
            obj = fast_convolve(a_candidate).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_candidate.copy()
                a = a_candidate
            elif obj < conv.max() - 1e-14:
                a = a_candidate
            else:
                break
        
        lr *= 0.999
        
        # Periodic anti-correlation boost
        if it % 50 == 0:
            m_act = n // 2
            p = a[:m_act]
            q = a[n - m_act:]
            # Measure anti-correlation
            cross = np.dot(p, q[::-1]) if m_act == n - m_act else np.dot(p, np.flip(a[n-1:n-m_act-1:-1]))
            energy = np.dot(p, p) + np.dot(a[m_act:n-m_act] if n-2*m_act > 0 else q, a[m_act:n-m_act] if n-2*m_act > 0 else q)
            # Try to reduce cross correlation
            if cross > 0:
                boost = 0.002 * cross
                a[:m_act] -= boost * a[:m_act] * a[:m_act].sum() / (a[:m_act] + 1e-15)
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp *= 1.2
            if m_exp > 4.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def coordinate_polish(a0, time_limit):
    """
    Coordinate descent: for each top peak, try to reduce it by
    adjusting a small window of variables.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    window_size = min(20, n // 4)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        # Top 3 peaks
        top3 = np.argpartition(conv, -3)[-3:] if len(conv) > 3 else np.arange(len(conv))
        
        improved = False
        for k_peak in top3:
            k_peak = int(k_peak)
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            
            # Try shifting mass within the affected window
            for _ in range(10):
                rng = np.random.default_rng((it * 31 + k_peak + _) % (2**31))
                pert = np.zeros(n)
                # Perturb within affected range
                for j in range(js, je + 1):
                    pert[j] = rng.standard_normal() * 0.002
                candidate = project_simplex(a + pert)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved and it % 100 == 0:
            # Broader perturbation
            rng = np.random.default_rng((it * 99991 + 3) % (2**31))
            candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
            else:
                if it > 500:
                    break
    
    return best_a, best_obj


def peak_targeting(a0, time_limit):
    """
    Directly target reducing individual peaks.
    For each peak, solve a small subproblem: minimize a[j]*a[k-j] for the
    dominant terms by adjusting a[j] and a[k-j].
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Find dominant terms in this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        terms = np.array([a[j] * a[k_peak - j] for j in range(js, je + 1)])
        js_arr = np.arange(js, je + 1)
        
        # Top contributing indices
        if len(terms) > 5:
            top_idx = js_arr[np.argsort(terms)[-min(5, len(terms)):]]
        else:
            top_idx = js_arr
        
        # For each pair (j, k-j), try to reduce a[j]*a[k-j]
        # by shifting mass from the larger to a less critical position
        improved_this_round = False
        for j in top_idx:
            j = int(j)
            partner = k_peak - j
            if j == partner:
                continue
            if partner >= n or partner < 0:
                continue
            
            # Try: decrease a[j] slightly, increase a[partner] slightly (or vice versa)
            # Or: decrease both and put mass elsewhere
            for delta in [0.001, 0.005, 0.01]:
                # Option 1: shift from j to partner
                if a[j] > delta:
                    cand = a.copy()
                    cand[j] -= delta
                    cand[partner] += delta
                    cand = np.maximum(cand, 0)
                    cand = cand / cand.sum()
                    c = fast_convolve(cand)
                    if c.max() < best_obj - 1e-15:
                        best_obj = c.max()
                        best_a = cand.copy()
                        a = cand
                        improved_this_round = True
                        break
                
                # Option 2: shift from j to a random far position
                if a[j] > delta:
                    far = (j + n // 2) % n
                    if far != j and abs(far - partner) > n // 4:
                        cand = a.copy()
                        cand[j] -= delta
                        cand[far] += delta
                        cand = np.maximum(cand, 0)
                        cand = cand / cand.sum()
                        c = fast_convolve(cand)
                        if c.max() < best_obj - 1e-15:
                            best_obj = c.max()
                            best_a = cand.copy()
                            a = cand
                            improved_this_round = True
                            break
            
            if improved_this_round:
                break
        
        if not improved_this_round:
            # Try a small random perturbation
            if it % 20 == 0:
                rng = np.random.default_rng((it * 48271 + 1) % (2**31))
                cand = project_simplex(a + rng.standard_normal(n) * 0.002)
                c = fast_convolve(cand)
                if c.max() < best_obj - 1e-15:
                    best_obj = c.max()
                    best_a = cand.copy()
                    a = cand
    
    return best_a, best_obj


def multi_scale_perturb_search(a0, time_limit, n_scales=5):
    """
    Try perturbations at multiple scales and pick the best one
    after a quick local refinement.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    per_scale_time = time_limit / n_scales
    
    scales = [0.001, 0.003, 0.01, 0.03, 0.1]
    
    for scale in scales:
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        n_trials = max(3, int(per_scale_time / n * 10))
        n_trials = min(n_trials, 50)
        
        for trial in range(n_trials):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            rng = np.random.default_rng((trial * 7 + len(scales)*100 + int(scale*10000)) % (2**31))
            
            # Structured perturbation
            pert = rng.standard_normal(n) * scale
            # Add anti-correlation component
            m = n // 2
            anti = rng.standard_normal(m) * scale * 0.5
            if n % 2 == 0:
                pert[:m] += anti
                pert[m:] -= anti
            else:
                pert[:m] += anti
                pert[-m:] -= anti
            
            cand = project_simplex(a + pert)
            obj = fast_convolve(cand).max()
            
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = cand.copy()
                # Quick refinement
                a_ref, _ = exact_subgrad_descent(cand, time_limit=min(2.0, (time_limit - (time.monotonic() - t0)) * 0.3),
                                                 lr_init=0.003, top_k=15)
                obj_ref = fast_convolve(a_ref).max()
                if obj_ref < best_obj:
                    best_obj = obj_ref
                    best_a = a_ref.copy()
    
    return best_a, best_obj


def interpolate_sequence(parent, new_n):
    """Interpolate a sequence to a new length."""
    n_old = len(parent)
    if new_n == n_old:
        return parent.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, parent)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    else:
        new_seq = np.ones(new_n) / new_n
    return new_seq

                try_seq(a_opt, "tiny_perturb")
    
    print(f"  Phase 6: best={best_score:.8f}, elapsed={time.monotonic()-t_phase:.0f}s")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
