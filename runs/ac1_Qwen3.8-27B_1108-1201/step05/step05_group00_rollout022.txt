def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        if res.success and res.fun > -1e-10:

def multiplicative_weights_lp(f, M, n, n_iter=50, eta=0.1):
    """
    Approximate LP solution using multiplicative weights.
    Solves: max 1^T g s.t. H g <= M, g >= 0
    where H[k,i] = f[k-i] (the convolution matrix).
    
    MW update: g_i *= (1 - step * (H @ g)_k* / M) where k* is the most violated constraint.
    """
    g = np.ones(n) / n
    
    for _ in range(n_iter):
        # Compute H @ g = conv(f, g)
        hg = fast_convolve(f * g)  # Wait, this is conv(f, f*g) not conv(f, g)
        # Actually I need H @ g where H[k,i] = f[k-i]
        # This is conv(f, g) but with a specific alignment
        # (f*g)[k] = sum_i f[k-i] * g[i] = conv(f, g)[k] (standard convolution)
        # But fast_convolve computes conv(f, f), not conv(f, g)
        
        # Let me compute conv(f, g) directly
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        F = np.fft.rfft(f, fft_size)
        G = np.fft.rfft(g, fft_size)
        hg = np.fft.irfft(F * G, fft_size)[:2*n-1]
        
        # Find most violated constraint
        k_star = int(np.argmax(hg))
        violation = hg[k_star] / M
        
        if violation < 1.0 + 1e-8:
            break
        
        # MW update
        step = eta * violation
        # g_i *= (1 - step * H[k_star, i] / M)
        # H[k_star, i] = f[k_star - i] if 0 <= k_star - i < n
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start <= i_end:
            coeffs = f[k_star - np.arange(i_start, i_end + 1)]
            g[i_start:i_end+1] *= (1.0 - step * coeffs / M)
        
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs
    
    return g


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive M."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = float(conv.max())
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        g = solve_lp_fast(f, M * m_exp, n)
        if g is None:
            # Try MW approximation
            g = multiplicative_weights_lp(f, M * m_exp, n, n_iter=20, eta=0.15)
            g_sum = g.sum()
            if g_sum < 1e-15:
                m_exp *= 1.3
                continue
            g = g / g_sum
        else:
            g_sum = g.sum()
            if g_sum < 1e-15:
                m_exp *= 1.3
                continue
            g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.03, 2.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        # Coarse search
        for alpha in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = float(fast_convolve(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search (binary)
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = float(fast_convolve(ft).max())
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = float(conv.max())
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.002 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = float(conv.max())
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj

1. FW-LP with structured LP oracle and adaptive M relaxation
2. Multiplicative weights as fast LP approximation fallback
3. Smoothed gradient descent with adaptive temperature for local refinement

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_pair(a, b):
    """Linear convolution of a and b via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix H where H[k,i] = f[k-i] if 0<=k-i<n else 0."""
    H = np.zeros((2 * n - 1, n))
    for k in range(n):
        H[k, 0:k+1] = f[0:k+1][::-1]
    for k in range(n, 2 * n - 1):
        i_start = k - n + 1
        length = n - i_start
        H[k, i_start:n] = f[k-(n-1):k+1][::-1]
    return H


def solve_lp_structured(f, M, n):
    """Solve LP: max 1^T g s.t. H(g) <= M*1, g >= 0 using structured construction."""
    c = -np.ones(n)
    H = build_conv_matrix(f, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=H, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun > -1e-10:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def multiplicative_weights_lp(f, M, n, n_iter=30, eta=0.12):
    """Approximate LP solution using multiplicative weights."""
    g = np.ones(n) / n
    H = build_conv_matrix(f, n)

    for _ in range(n_iter):
        hg = H @ g
        k_star = int(np.argmax(hg))
        if hg[k_star] <= M * (1.0 + 1e-8):
            break
        step = eta * hg[k_star] / M
        coeffs = H[k_star, :]
        g *= (1.0 - step * coeffs)
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs
    return g


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive M."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = float(conv.max())
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        g = solve_lp_structured(f, M * m_exp, n)
        if g is None:
            g = multiplicative_weights_lp(f, M * m_exp, n, n_iter=20, eta=0.12)
            g_sum = g.sum()
            if g_sum < 1e-15:
                m_exp *= 1.2
                if m_exp > 5.0:
                    m_exp = 1.0
                continue
            g = g / g_sum
        else:
            g_sum = g.sum()
            if g_sum < 1e-15:
                m_exp *= 1.2
                if m_exp > 5.0:
                    m_exp = 1.0
                continue
            g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.03, 2.5)
            continue

        best_alpha = 0.0
        best_val = M

        for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = float(fast_convolve(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = float(fast_convolve(ft).max())
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = float(conv.max())
            m_exp = 1.0

            if M < best_obj - 1e-16:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.04, 4.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.001 + 0.003 * min(1.0, no_improve / 120.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = float(conv.max())
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def smoothed_grad_descent(a0, time_limit, T_init=0.0004, lr_init=0.04):
    """Smoothed gradient descent with softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: grad[m] = 2 * sum_i a[i] * p[i+m] = 2 * correlate(p, a, 'full')[m]
        corr = np.correlate(p, a, 'full')
        grad = 2.0 * corr[:n]

        best_candidate = a
        best_val = best_obj

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = float(c.max())
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            T *= 1.04
            lr *= 0.98
        if no_improve > 60:
            T *= 0.75
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.003 * min(1.0, (no_improve - 200) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit):
    """
    Coordinate descent focusing on peak-reduction.
    For each top peak, adjust the variables that contribute to it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    rng = np.random.default_rng(12345)

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Variables affecting this peak: j in [max(0, k_peak-n+1), min(n-1, k_peak)]
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected_size = je - js + 1

        if affected_size <= 1:
            break

        # Try to reduce the peak by redistributing among affected variables
        # Use a small random perturbation biased toward reducing the peak
        improved = False
        for attempt in range(8):
            # Random perturbation of affected variables
            pert = rng.standard_normal(affected_size) * 0.0003
            # Bias: reduce variables that have high contribution to the peak
            contributions = a[js:je+1]
            pert -= 0.5 * contributions / (contributions.sum() + 1e-15) * np.abs(pert).sum()

            candidate = a.copy()
            candidate[js:je+1] = candidate[js:je+1] + pert
            candidate = project_simplex(candidate)

            c = fast_convolve(candidate)
            val = float(c.max())
            if val < best_obj - 1e-16:
                a = candidate
                best_obj = val
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break

    return best_a, best_obj


def multiplicative_weights_descent(a0, time_limit, T_init=0.001):
    """
    Multiplicative weights / matrix multiplication game approach.
    Uses the dual (peak distribution) to update the primal multiplicatively.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init

    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Multiplicative update: a_j *= (1 - eta * grad_j)
        corr = np.correlate(p, a, 'full')
        grad = 2.0 * corr[:n]

        # Normalize gradient to [0, max]
        grad_max = float(grad.max())
        if grad_max < 1e-15:
            break
        grad_norm = grad / grad_max

        eta = 0.05
        a *= (1.0 - eta * grad_norm)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            T *= 0.8
            continue
        a = a / s

        obj = float(fast_convolve(a).max())
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            T *= 0.9
            eta *= 0.95
        if no_improve > 300:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            eta = 0.05
            no_improve = 0

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge concentration
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Two peaks
    g2 = (np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.08 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.08 * n)) ** 2))
    inits.append(g2 / g2.sum())

    # Four peaks
    g4 = (np.exp(-0.5 * ((x - 0.125 * (n - 1)) / (0.06 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.375 * (n - 1)) / (0.06 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.625 * (n - 1)) / (0.06 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.875 * (n - 1)) / (0.06 * n)) ** 2))
    inits.append(g4 / g4.sum())

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x / (n - 1) - 0.5)
    tri = np.maximum(tri, 0.05)
    inits.append(tri / tri.sum())

    # Inverse cosine (peaks at edges)
    ic = 1.0 / (np.abs(np.cos(np.pi * x / (2 * n))) + 0.1)
    inits.append(ic / ic.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase hybrid optimization at n=768.
    Phase 1: Quick exploration with diverse initializations
    Phase 2: Deep FW-LP on best candidate
    Phase 3: Smoothed gradient + multiplicative weights refinement
    Phase 4: Coordinate descent final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    N = 768

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size == N and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={N}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(N) / N

    # === Phase 1: Quick exploration (~200s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_start = time.monotonic()

    inits = generate_initializations(N, parent, rng)

    for idx, init in enumerate(inits):
        remaining = deadline - time.monotonic()
        if remaining < 200:
            break
        t_alloc = min(remaining - 200, 20)
        if t_alloc < 3:
            break
        print(f"  Init {idx} (type={'parent' if idx==len(inits)-1 and parent is not None else idx}), t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(init, time_limit=t_alloc)
        try_seq(a_opt, f"fw_init{idx}")

    print(f"  Phase 1 best: {best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Deep FW-LP (~500s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_start = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 100:
            t_alloc = min(remaining - 100, 400)
            print(f"  Deep FW-LP, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_fw")

    # Perturbed restarts
    if best_seq is not None:
        for scale in [0.002, 0.005, 0.01]:
            remaining = deadline - time.monotonic()
            if remaining < 80:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(N) * scale)
            t_alloc = min(remaining - 60, 60)
            if t_alloc < 5:
                break
            print(f"  FW perturbed (scale={scale}), t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 2 best: {best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Smoothed gradient + MW descent (~400s) ===
    print("\n=== Phase 3: Gradient Methods ===")
    t_start = time.monotonic()

    if best_seq is not None:
        # Smoothed gradient with various temperatures
        for T in [0.0003, 0.001, 0.003]:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            t_alloc = min(remaining - 50, 120)
            if t_alloc < 5:
                break
            print(f"  Smoothed grad T={T}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc, T_init=T, lr_init=0.03)
            try_seq(a_opt, f"smooth_T{T}")

        # Multiplicative weights descent
        remaining = deadline - time.monotonic()
        if remaining > 50:
            t_alloc = min(remaining - 40, 100)
            if t_alloc > 5:
                print(f"  MW descent, t={t_alloc:.0f}s")
                a_opt, _ = multiplicative_weights_descent(best_seq, time_limit=t_alloc, T_init=0.001)
                try_seq(a_opt, "mw_descent")

    print(f"  Phase 3 best: {best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Final polish (~200s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_start = time.monotonic()

    if best_seq is not None:
        # Coordinate descent polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_alloc = min(remaining - 25, 80)
            if t_alloc > 5:
                print(f"  Coord descent, t={t_alloc:.0f}s")
                a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "coord")

        # Fine smoothed gradient
        remaining = deadline - time.monotonic()
        if remaining > 25:
            t_alloc = min(remaining - 20, 50)
            if t_alloc > 3:
                print(f"  Fine smooth grad, t={t_alloc:.0f}s")
                a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc,
                                                 T_init=0.0001, lr_init=0.008)
                try_seq(a_opt, "fine_smooth")

        # Final FW-LP burst
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_alloc = min(remaining - 15, 40)
            if t_alloc > 3:
                print(f"  Final FW-LP, t={t_alloc:.0f}s")
                a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, "final_fw")

        # Tiny random perturbations with quick polish
        remaining = deadline - time.monotonic()
        if remaining > 15:
            for _ in range(5):
                remaining = deadline - time.monotonic()
                if remaining < 5:
                    break
                noise = rng.standard_normal(N) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                t_alloc = min(remaining - 3, 8)
                if t_alloc < 2:
                    break
                a_opt, _ = smoothed_grad_descent(f_pert, time_limit=t_alloc,
                                                 T_init=0.0002, lr_init=0.005)
                try_seq(a_opt, "perturb_fine")

    print(f"  Phase 4 best: {best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(N) / N
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
