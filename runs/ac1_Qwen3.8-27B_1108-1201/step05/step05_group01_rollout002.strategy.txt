import numpy as np

def evaluate_sequence(sequence):
    n = len(sequence)
    if n == 0:
        return float('inf')
    conv = np.convolve(sequence, sequence, mode="full")
    max_b = np.max(conv)
    sum_a = np.sum(sequence)
    if sum_a < 0.01 or max_b == float('inf'):
        return float('inf')
    score = float(2 * n * max_b / (sum_a ** 2))
    return score
