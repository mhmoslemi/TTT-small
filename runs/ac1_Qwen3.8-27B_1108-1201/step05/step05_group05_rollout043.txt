Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
1. Structured initialization at multiple lengths

3. Coordinate descent polishing
4. Multi-restart with diverse perturbations

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M, n, active_k=None):
    c = -np.ones(n)
    if active_k is None:
        K = 2 * n - 1
        A_ub = np.zeros((K, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(K, M)
    else:
        K = len(active_k)
        A_ub = np.zeros((K, n))
        b_ub = np.full(K, M)
        for row, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            A_ub[row, js:je+1] = f[k-je:k-js+1][::-1]
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_A_ub(f, k_values, n):
    """Build constraint matrix for (f*g)[k] <= M constraints."""
    K = len(k_values)
    A = np.zeros((K, n))
    for row, k in enumerate(k_values):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            A[row, i_min:i_max + 1] = f[k - i_max:k - i_min + 1][::-1]
    return A


def solve_lp(f, M, n, active_k=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""

    return None

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_A_ub(f, k_values, n):
    """Build constraint matrix A where A[row,i] = f[k-i] if valid."""
    K = len(k_values)

    for row, k in enumerate(k_values):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            A[row, i_min:i_max + 1] = f[k - i_max:k - i_min + 1][::-1]
    return A

    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0."""

    if res.success:

    elif window_type == 'flat_top':
        x = np.linspace(-1, 1, n)
        a1, a2, a3 = 1.0, 2.39655, 0.8121
        w = a1 - a2 * np.abs(x) + a3 * np.abs(x) ** 2
        w = np.maximum(w, 1e-10)
        return w / w.sum()
    elif window_type == 'triangular':
        x = np.linspace(-1, 1, n)
        w = 1.0 - np.abs(x)
        w = np.maximum(w, 1e-10)
        return w / w.sum()
    else:
        return np.ones(n) / n


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
