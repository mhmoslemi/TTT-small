   For i=1, k=2: 2*a[k-i] = 2*a[1]. Correct!
   For i=2, k=2: 2*a[k-i] = 2*a[0]. Let me verify: ∂conv[2]/∂a[2] = from j=0: a[2]*a[0] → ∂/∂a[2] = a[0]; from j=2: a[2]*a[0] → ∂/∂a[2] = a[0]. Total = 2*a[0]. Correct!
   
   So the formula is: **∂conv[k]/∂a[i] = 2*a[k-i]** (when 0<=i<n and 0<=k-i<n), and 0 otherwise.
   
   Great! So the gradient of the log-sum-exp objective is:
   
   L = (1/T) * log(sum_k exp(T*conv[k]))
   
   ∂L/∂a[i] = (1/T) * sum_k exp(T*conv[k]) * 2*a[k-i] * [0<=k-i<n]
   
   Let w_k = exp(T*conv[k]) / sum_j exp(T*conv[j]) (softmax weights).
   
   Then ∂L/∂a[i] = 2 * sum_k w_k * a[k-i] * [0<=k-i<n]
   
   This is a weighted sum of shifted versions of a! Specifically, it's like a "blurred" version of a where the blur kernel is w.
   
   Actually, let's define: for each k, the contribution to ∂L/∂a[i] is 2*w_k * a[k-i] (if valid). So if we define a "kernel" w_k, then:
   
   ∂L/∂a[i] = 2 * sum_k w_k * a[k-i] = 2 * (w * a)[i] (circular-like convolution, but with bounds)
   
   Wait, it's not exactly a convolution because of the boundary effects. But for a good approximation, we can think of it as:
   
   grad[i] = 2 * sum_k w_k * a[(k-i) mod n] (approximately, for large n)
   
   Or more precisely: grad[i] = 2 * sum_{k: 0<=k-i<n} w_k * a[k-i]
   
   Let m = k-i. Then k = i+m, and we need 0 <= m < n and 0 <= i+m < 2n-1 (the range of k).
   
   So grad[i] = 2 * sum_m w_{i+m} * a[m] (where m ranges over valid values)
   
   This is a cross-correlation of a with w! Specifically:
   
   grad = 2 * (a ⊛ w) where ⊛ denotes cross-correlation.
   
   Or equivalently: grad[i] = 2 * sum_m a[m] * w[i+m]
   
   This can be computed efficiently using FFT!
   
   So the full gradient computation is:
   1. Compute conv = auto_convolve(a) (via FFT)
   2. Compute w = softmax(T * conv) (element-wise)
   3. Compute grad = 2 * cross_correlate(a, w) (via FFT)
   
   This is O(n log n) per iteration, which is very fast!
   
   Now, for the optimization, I'll use Adam with this gradient, projected onto the simplex.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(f, T, top_k=None):
    """Compute gradient of log-sum-exp surrogate using FFT."""
    n = len(f)
    conv = fast_convolve(f)
    m = len(conv)
    
    if top_k is not None and top_k < m:
        # Use only top-k values
        idx = np.argpartition(conv, -top_k)[-top_k:]
        conv_used = np.zeros(m)
        conv_used[idx] = conv[idx]
        # Zero out non-top-k
        conv_max = conv[idx].max()
        shifted = T * (conv_used - conv_max)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.zeros(m)
            w[idx[np.argmax(conv[idx])]] = 1.0
    else:
        conv_max = conv.max()
        shifted = T * (conv - conv_max)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.zeros(m)
            w[np.argmax(conv)] = 1.0
    
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (valid k)
    # This is a cross-correlation: grad = 2 * (w cross-correlate a)
    # Using FFT: cross-correlation of w and a = ifft(conj(fft(w)) * fft(a))
    # But we need to be careful about the indexing.
    # 
    # grad[i] = 2 * sum_k w[k] * a[k-i] where 0<=k-i<n
    # Let m = k-i, so k = i+m
    # grad[i] = 2 * sum_m w[i+m] * a[m] (where 0<=m<n and 0<=i+m<2n-1)

    # This is like a convolution of a with a shifted/flipped w.
    # Specifically, if we define w_rev[k] = w[2n-2-k], then:
    # (a * w_rev)[i] = sum_m a[m] * w_rev[i-m] = sum_m a[m] * w[2n-2-i+m]
    # That's not quite right either.
    #
    # Let me use a different approach. Define the linear cross-correlation:
    # (w ⊛ a)[i] = sum_k w[k] * a[k-i] (with a[j]=0 outside [0,n))
    #
    # This can be computed as: ifft(fft(w, N) * conj(fft(a, N)))[:N]
    # where N >= (2n-1) + n - 1 = 3n-2 to avoid circular aliasing.
    # Actually, for linear cross-correlation of sequences of length m=2n-1 and n,
    # we need FFT size >= m + n - 1 = 3n - 2.
    
    # Actually, let me think about this differently.
    # grad[i] = 2 * sum_{k=0}^{2n-2} w[k] * a[k-i] * [0 <= k-i < n]
    # = 2 * sum_{k=max(0,i)}^{min(2n-2, i+n-1)} w[k] * a[k-i]
    # 
    # Let j = k - i, so k = i + j:
    # = 2 * sum_{j=0}^{n-1} w[i+j] * a[j] (where i+j < 2n-1)
    #
    # This is a "sliding dot product" of a with windows of w.
    # It's equivalent to: for each i, take the segment w[i:i+n] and dot with a.
    #
    # This is a cross-correlation of a with w, but with w being longer than a.
    # 
    # Using FFT: Let's pad both to size N >= n + (2n-1) - 1 = 3n-2.
    # cross_corr(i) = sum_j a[j] * w[i+j]
    # This is the i-th element of the cross-correlation of a and w.
    # 
    # In FFT: cross_correlation(a, w) = ifft(conj(fft(a, N)) * fft(w, N))
    # Wait, the cross-correlation of a and w is: (a ⊛ w)[i] = sum_j conj(a[j]) * w[i+j]
    # For real signals: = sum_j a[j] * w[i+j]
    # 
    # This is what we want (up to the factor of 2).
    #
    # But we need to be careful about the definition. The standard cross-correlation
    # (a ⊛ w)[i] = sum_j a[j] * w[i+j] gives values for i = -(n-1), ..., (2n-2)-1.
    # Actually for linear cross-correlation of a (length n) and w (length 2n-1):
    # the result has length n + (2n-1) - 1 = 3n-2.
    # 
    # The i-th element (0-indexed) of the linear cross-correlation is:
    # (a ⊛ w)[i] = sum_j a[j] * w[i+j] where both indices must be valid.
    # For i = 0: sum_j a[j] * w[j] (j from 0 to n-1, w[j] valid for j < 2n-1) ✓
    # For i = 1: sum_j a[j] * w[j+1] ✓
    # ...
    #
    # So grad[i] = 2 * (a ⊛ w)[i] for i = 0, ..., n-1.
    #
    # To compute via FFT:
    # N = next_power_of_2(3*n - 2)
    # A = fft(a, N), W = fft(w, N)
    # cross = ifft(conj(A) * W)  -- wait, I need to check the convention.
    #
    # numpy ifft: ifft(X)[n] = (1/N) * sum_k X[k] * exp(2πi*n*k/N)
    # The cross-correlation (a ⊛ w)[i] = sum_j a[j] * w[i+j]
    # = sum_j a[j] * w[i+j]
    #
    # If we compute ifft(conj(FFT(a)) * FFT(w)):
    # Let A_k = FFT(a)_k, W_k = FFT(w)_k
    # ifft(conj(A) * W)[i] = (1/N) sum_k conj(A_k) * W_k * exp(2πiik/N)
    # = (1/N) sum_k (sum_j a[j] exp(-2πij k/N)) * (sum_l w[l] exp(-2πil k/N)) * exp(2πiik/N)
    # = (1/N) sum_k sum_j sum_l a[j] w[l] exp(-2πi(j+l-i)k/N)
    # = sum_j sum_l a[j] w[l] delta(i - (j+l) mod N)  [using orthogonality, assuming no wraparound]
    # = sum_j sum_l a[j] w[l] [j+l ≡ i (mod N)]
    #
    # If N >= n + (2n-1) - 1 = 3n-2, there's no wraparound for the range we care about.
    # For i in [0, n-1]: j+l = i, so l = i-j. We need 0<=j<n and 0<=l<2n-1.
    # l = i-j, so 0 <= i-j < 2n-1, i.e., i-n+1 <= j <= i.

    # So j ranges from max(0, i-n+1) to min(i, n-1).
    #

    #
    # Hmm, that gives w[i-j], not w[i+j]. So this is actually the cross-correlation
    # with the other convention: (a ⋆ w)[i] = sum_j a[j] * w[i-j].
    #
    # Let me try the other way: ifft(FFT(a) * conj(FFT(w))):
    # = (1/N) sum_k A_k * conj(W_k) * exp(2πiik/N)
    # = sum_j sum_l a[j] w[l] delta(i ≡ j-l (mod N))
    # = sum_j sum_l a[j] w[l] [j-l ≡ i (mod N)]
    # For i in [0, n-1]: j - l = i, so j = i + l.
    # 0 <= i+l < n and 0 <= l < 2n-1.
    # l ranges from 0 to n-1-i.
    # Result: sum_l a[i+l] * w[l] = sum_m a[i+m] * w[m] (where m = l, 0<=m<n-i)
    #
    # That's sum_m a[i+m] * w[m], which is NOT what we want either.
    #
    # What we want is: sum_j a[j] * w[i+j] (for j=0,...,n-1, i+j < 2n-1)
    #
    # Let me try: ifft(FFT(w) * conj(FFT(a))):
    # = sum_l sum_j w[l] a[j] [l-j ≡ i (mod N)]
    # = sum_j sum_l w[l] a[j] [l = j+i] (no wraparound)
    # = sum_j a[j] * w[j+i]
    # 
    # Yes! That's what we want.
    #
    # So: grad[i] = 2 * ifft(FFT(w, N) * conj(FFT(a, N)))[i] for i = 0,...,n-1
    # with N >= 3n-2.

    N = 1 << (3 * n - 2).bit_length()  # next power of 2 >= 3n-2
    A = np.fft.rfft(f, N)
    W = np.fft.rfft(w, N)
    cross = np.fft.irfft(W * np.conj(A), N)[:n]
    grad = 2.0 * cross
    return grad, conv

def adam_optimize(f0, time_limit, T_start=None, T_end=None, lr_base=0.01, verbose=False):
    """Adam optimization of the log-sum-exp smoothed objective on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    if T_start is None:
        T_start = 3.0 / n
    if T_end is None:
        T_end = 0.3 / n
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    no_improve = 0
    top_k = n  # Start with all constraints
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Temperature schedule: exponential decay
        T = T_start * (T_end / T_start) ** progress
        
        # Adaptive top-k: focus more as we progress
        top_k = max(10, int(n * (1.0 - 0.7 * progress)))
        
        # Learning rate: cosine annealing
        lr = lr_base * 0.5 * (1 + np.cos(np.pi * progress)) + lr_base * 0.01
        
        # Compute gradient
        grad, conv = compute_gradient(f, T, top_k=top_k)
        
        # Adam update
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** it1)
        v_hat = v / (1 - beta2 ** it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = project_simplex(f - step)
        
        if np.max(np.abs(f_new - f)) < 1e-15:
            no_improve += 1
            if no_improve > 500:
                rng = np.random.default_rng(it * 41 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
            continue
        
        f = f_new
        new_M = fast_convolve(f).max()
        
        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                obj_val = 2 * n * best_obj
                print(f"    Adam iter {it}: raw_obj={best_obj:.10f}, score={obj_val:.8f}, T={T:.6f}")
        else:
            no_improve += 1
    
    return best_f, best_obj


def pairwise_optimize(f0, time_limit, verbose=False):
    """Optimize pairs of coordinates by mass transfer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(98765)
    
    # Use FFT-based evaluation for speed
    no_improve_passes = 0
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False
        
        # Randomize pair selection
        order = rng.permutation(n)
        
        for idx_i in range(0, n, 1):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = order[idx_i]
            # Pick a few random partners
            for _ in range(3):
                j = rng.integers(0, n)
                if i == j:
                    continue
                
                c = f[i] + f[j]
                if c < 1e-12:
                    continue
                
                # Try a few transfer values
                base = f[i] / c
                candidates = [base]
                if base > 0.05:
                    candidates.append(base - 0.05)
                if base < 0.95:
                    candidates.append(base + 0.05)
                candidates.extend([0.0, 1.0, 0.5])
                
                best_t = base
                best_val = best_obj  # Current objective
                
                for t in candidates:
                    t = np.clip(t, 0.0, 1.0)
                    f_trial = f.copy()
                    f_trial[i] = t * c
                    f_trial[j] = (1 - t) * c
                    val = fast_convolve(f_trial).max()
                    if val < best_val - 1e-14:
                        best_val = val
                        best_t = t
                
                if best_val < best_obj - 1e-13:
                    f[i] = best_t * c
                    f[j] = (1 - best_t) * c
                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True
        
        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                break
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(424242)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 200:
            break
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try increasing and decreasing f[i] by moving mass from/to other coordinates
            # Simple approach: try f[i] * (1±delta) and redistribute
            for delta in [0.01, 0.005, -0.01, -0.005, 0.02, -0.02]:
                new_val = f[i] * (1 + delta)
                if new_val < 0 or new_val > 1:
                    continue
                # Redistribute the difference among other coordinates
                diff = new_val - f[i]
                if abs(diff) < 1e-15:
                    continue
                
                # Move mass to/from a random other coordinate
                j = rng.integers(0, n)
                if j == i:
                    continue
                
                f_trial = f.copy()
                f_trial[i] = new_val
                f_trial[j] = f[j] - diff
                if f_trial[j] < 0:
                    continue
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-14:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with log-sum-exp smooth surrogate (FFT-accelerated gradients)
2. Exponential temperature schedule with cosine LR annealing
3. Active-set focus via top-k constraint selection
4. LP-based Frank-Wolfe for complementary direction finding
5. Multi-start with diverse structured initializations
6. Pairwise coordinate descent for final polish
7. Multi-length exploration
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(f, T, top_k=None):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. f.
    Uses FFT for O(n log n) computation.
    Returns (gradient, convolution_values).
    """
    n = len(f)
    conv = fast_convolve(f)
    m = len(conv)

    if top_k is not None and top_k < m and top_k > 0:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        mask = np.zeros(m, dtype=bool)
        mask[idx] = True
        conv_masked = np.where(mask, conv, -np.inf)
    else:
        conv_masked = conv.copy()

    conv_max = np.max(conv_masked)
    shifted = T * (conv_masked - conv_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(m)
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum

    # grad[i] = 2 * sum_j f[j] * w[i+j] for j=0..n-1, i+j < 2n-1
    # This is cross-correlation: ifft(FFT(w) * conj(FFT(f)))
    N = 1 << (3 * n - 2).bit_length()
    F = np.fft.rfft(f, N)
    W = np.fft.rfft(w, N)
    cross = np.fft.irfft(W * np.conj(F), N)[:n]
    grad = 2.0 * cross

    return grad, conv


def adam_optimize(f0, time_limit, T_start=None, T_end=None, lr_base=0.012, verbose=False):
    """Adam optimization of log-sum-exp smoothed objective on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    if T_start is None:
        T_start = 4.0 / n
    if T_end is None:
        T_end = 0.2 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    m_mom = np.zeros(n)
    v_mom = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-6))

        # Exponential temperature decay
        T = T_start * (T_end / T_start) ** progress

        # Adaptive top-k: start broad, narrow down
        top_k = max(20, int(n * (1.2 - 0.9 * progress)))

        # Cosine learning rate
        lr = lr_base * (0.5 * (1.0 + np.cos(np.pi * progress)) + 0.02)

        grad, conv = compute_gradient_fft(f, T, top_k=top_k)

        v_mom = beta2 * v_mom + (1.0 - beta2) * grad * grad

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j]*g[k-j] for valid j
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        rows = np.arange(j, j + n)
        A_ub[rows, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 5, 'mip_rel_gap': 0.1})
    if res is not None and res.success:
        return np.maximum(res.x, 0)
    return None

    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        g = lp_direction(f, M, n)
        if g is None:
            g = lp_direction(f, M * 1.01, n)
            if g is None:
                no_improve += 1
                if no_improve > 100:
                    break
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
                continue
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = np.maximum(f + alpha * (g - f), 0.0)

            if st < 1e-15:

            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:

                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)

            if M < best_obj - 1e-14:

                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj

                    f = f / f.sum()
                    best_obj = best_val
                    best_f = f.copy()
                    pass_improved = True

        if pass_improved:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 5:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phases: Adam (main) -> Adam (diverse) -> LP-FW -> Multi-length ->
            Pairwise -> Final polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')

    def _try(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-10:
            return
        a = a / a.sum()
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n0 = len(parent)
    print(f"\nStarting optimization with n={n0}, initial score={best_score:.8f}")

    # === Phase 1: Main Adam optimization ===
    print(f"\n=== Phase 1: Adam main (n={n0}) ===")
    rem = deadline - time.monotonic() - 600
    t1 = min(rem, 550)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s")
        a_opt, _ = adam_optimize(parent, time_limit=t1, verbose=True)
        _try(a_opt, "adam_main")

    # === Phase 2: LP-FW on current best ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        t2 = min(rem, 250)
        if t2 > 30:
            print(f"  Time: {t2:.0f}s, n={len(best_seq)}")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True)
            _try(a_opt, "lpfw")

    # === Phase 3: Diverse initializations with Adam ===
    print(f"\n=== Phase 3: Diverse Adam (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    # Build list of initializations
    inits = []
    # Reversed
    inits.append(("reversed", base[::-1].copy() / base.sum()))
    # Bump-shaped
    x = np.linspace(0, 1, n_base)
    bump = np.maximum(x**0.7 * (1.0 - x)**0.7, 1e-10)
    inits.append(("bump", bump / bump.sum()))
    # Smoothed parent
    from scipy.ndimage import gaussian_filter1d
    sm = gaussian_filter1d(base, sigma=5)
    sm = np.maximum(sm, 0)
    if sm.sum() > 1e-10:
        inits.append(("smoothed", sm / sm.sum()))
    # Random perturbation
    rng = np.random.default_rng(77777)
    pert = base + rng.standard_normal(n_base) * 0.015
    pert = np.maximum(pert, 0)
    if pert.sum() > 1e-10:
        inits.append(("perturbed", pert / pert.sum()))
    # Exponential decay shape
    exp_shape = np.exp(-3.0 * x) + np.exp(-3.0 * (1.0 - x)) * 0.5
    exp_shape = np.maximum(exp_shape, 1e-10)
    inits.append(("asym_exp", exp_shape / exp_shape.sum()))

    for name, init in inits:
        rem = deadline - time.monotonic() - 350
        if rem < 30:
            break
        t_alloc = min(rem * 0.35, 160)
        if t_alloc < 20:
            break
        print(f"  Adam [{name}], t={t_alloc:.0f}s")
        a_opt, _ = adam_optimize(init, time_limit=t_alloc, verbose=False)
        _try(a_opt, f"adam_{name}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.8f}) ===")
    for target_n in [512, 1024]:
        rem = deadline - time.monotonic() - 250
        if rem < 40:
            break
        if target_n == len(best_seq):
            continue
        f_interp = interpolate_sequence(best_seq, target_n)
        t_alloc = min(rem * 0.5, 150)
        if t_alloc > 20:
            print(f"  Adam n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_optimize(f_interp, time_limit=t_alloc, verbose=False)
            _try(a_opt, f"adam_n{target_n}")

    # === Phase 5: Pairwise optimization ===
    print(f"\n=== Phase 5: Pairwise (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_pw = min(rem * 0.7, 180)
            print(f"  Time: {t_pw:.0f}s, n={len(best_seq)}")
            a_opt, _ = pairwise_optimize(best_seq, time_limit=t_pw, verbose=True)
            _try(a_opt, "pairwise")

    # === Phase 6: Final Adam polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Adam with fine parameters
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_final = min(rem * 0.5, 70)
            print(f"  Adam final, t={t_final:.0f}s")
            a_opt, _ = adam_optimize(best_seq, time_limit=t_final,
                                     T_start=1.0 / n_best, T_end=0.1 / n_best,
                                     lr_base=0.003, verbose=True)
            _try(a_opt, "adam_final")

        # Tiny perturbations + quick Adam
        for trial in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 12:
                break
            rng = np.random.default_rng(500 + trial)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
            t_small = min(rem * 0.5, 25)
            a_opt, _ = adam_optimize(f_pert, time_limit=t_small,
                                     T_start=1.5 / n_best, T_end=0.15 / n_best,
                                     lr_base=0.005)
            _try(a_opt, f"perturb_{trial}")

        # One more pairwise pass
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = pairwise_optimize(best_seq, time_limit=min(rem, 30))
            _try(a_opt, "pw_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
