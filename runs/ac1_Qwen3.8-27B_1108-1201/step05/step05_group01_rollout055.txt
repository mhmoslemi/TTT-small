import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_toeplitz_Q(top_lags, n):
    """Build the Q matrix = sum of C_k for k in top_lags.
    C_k has 1's on the k-th super/sub diagonal, C_0 = I."""

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe,
Top-K active-set QP polishing, and Nesterov subgradient refinement.
Key innovation: alternating Top-K QP that minimizes sum of top autocorrelation
peaks via a convex QP on the simplex, providing a more precise polishing step
than pure subgradient methods.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 0.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={"presolve": True, "time_limit": 30})
        if res.success or (res.status in [1, 2] and res.x is not None):
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive step sizes."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp_direction(f, M * 1.001, n)
        if g is None:
            g = solve_lp_direction(f, M * 1.01, n)
            if g is None:
                g = solve_lp_direction(f, M * 1.05, n)
                if g is None:
                    no_improve += 5
                    continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 5e-12:
            no_improve += 1
            if no_improve > 80:
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.002 + 0.002 * min(no_improve // 80, 5)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        # Line search with fewer evaluations
        best_alpha = 0.0
        best_val = M
        fw_alpha = 2.0 / (it + 2.0)
        alphas = [0.1, 0.2, 0.4, fw_alpha, 0.6, 0.8, 1.0]
        alphas = [a for a in alphas if 0 < a <= 1.0]
        alphas = sorted(set(alphas))

        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Quick binary refinement (4 steps only)
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.2)
            hi = min(1.0, best_alpha + 0.2)
            for _ in range(5):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                break
            f_new = f_new / sn
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 5

    return best_f, best_obj


def build_Q_from_lags(lags, n):
    """Build Q matrix where Q[i,j] = 1 if |i-j| is in lags."""
    Q = np.zeros((n, n))
    for k in lags:
        k = int(k)
        if k < 0 or k >= n:
            continue
        if k == 0:
            idx = np.arange(n)
            Q[idx, idx] += 1.0
        else:
            idx = np.arange(n - k)
            Q[idx, idx + k] += 1.0
            Q[idx + k, idx] += 1.0
    return Q


def solve_qp_on_simplex(Q, n, a_init=None):
    """
    Solve min a^T Q a s.t. 1^T a = 1, a >= 0 using active set method.
    Q is n x n symmetric PSD.
    """
    if a_init is not None:
        a_init = np.maximum(a_init, 0)
        s = a_init.sum()
        if s > 1e-15:
            a_init = a_init / s
            support = set(np.where(a_init > 1e-10)[0].tolist())
        else:
            support = set(range(n))
    else:
        support = set(range(n))

    if not support:
        return np.ones(n) / n

    for it in range(200):
        S = sorted(support)
        S_arr = np.array(S)
        m = len(S_arr)

        if m == 0:
            return np.ones(n) / n
        if m == 1:
            return np.eye(1)[0] * np.ones(n) * 0 + np.array([1.0 if i == S_arr[0] else 0.0 for i in range(n)])

        QSS = Q[np.ix_(S_arr, S_arr)]

        # Regularize for numerical stability
        reg = 1e-12
        QSS_reg = QSS + reg * np.eye(m)

        one_S = np.ones(m)

        try:
            L = np.linalg.cholesky(QSS_reg)
            y = np.linalg.solve(L, one_S)
            x = np.linalg.solve(L.T, y)
        except np.linalg.LinAlgError:
            try:
                x, _, _, _ = np.linalg.lstsq(QSS_reg, one_S, rcond=None)
            except Exception:
                a = np.zeros(n)
                a[S_arr] = 1.0 / m
                return a

        denom = one_S @ x
        if denom <= 1e-15 or not np.isfinite(denom):
            a = np.zeros(n)
            a[S_arr] = 1.0 / m
            return a

        a_S = x / denom

        # Check for negative components
        if (a_S < -1e-8).any():
            worst_local = np.argmin(a_S)
            worst_global = S_arr[worst_local]
            support.discard(int(worst_global))
            if not support:
                return np.ones(n) / n
            continue

        a_S = np.maximum(a_S, 0.0)
        lam_half = 1.0 / denom

        # Check reduced costs for inactive variables
        if m < n:
            a_full = np.zeros(n)
            a_full[S_arr] = a_S
            # Q[i,S] a_S for inactive i
            inactive = np.array(sorted(set(range(n)) - support))
            if len(inactive) > 0:
                rc = Q[inactive][:, S_arr] @ a_S - lam_half
                if rc.min() < -1e-8:
                    worst = inactive[np.argmin(rc)]
                    support.add(int(worst))
                    continue

        # Optimal
        a = np.zeros(n)
        a[S_arr] = a_S
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-15:
            return np.ones(n) / n
        return a / ss

    # Fallback
    a = np.zeros(n)
    S_sorted = sorted(support)
    a[S_sorted] = 1.0 / max(len(S_sorted), 1)
    return a


def topk_qp_optimize(f0, time_limit, K=25, verbose=False):
    """
    Alternating Top-K QP: minimize sum of top-K autocorrelation peaks.
    Each step: find top K lags, build Q, solve QP, update.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        conv = fast_convolve(f)
        current_max = conv.max()

        # Find top K lags (use symmetry: only first n lags matter)
        K_eff = min(K, n)
        # Use all 2n-1 lags but exploit symmetry
        top_indices = np.argsort(conv)[::-1][:K_eff]
        # Map to unique lags (use min(k, 2n-2-k) for symmetry)
        unique_lags = set()
        for kidx in top_indices:
            k = int(kidx)
            k_eff = min(k, 2 * n - 2 - k)
            if k_eff < n:
                unique_lags.add(k_eff)

        if not unique_lags:
            unique_lags = {0}

        # Build Q
        Q = build_Q_from_lags(list(unique_lags), n)

        # Solve QP
        a_new = solve_qp_on_simplex(Q, n, a_init=f)

        conv_new = fast_convolve(a_new)
        new_max = conv_new.max()

        if new_max < best_obj - 1e-14:
            best_obj = new_max
            best_f = a_new.copy()
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    TopK-QP iter {it}: obj={best_obj:.10f}, K={K_eff}")
        else:
            f = a_new  # Still update to explore
            no_improve += 1

        f = a_new
        if no_improve > 30:
            # Restart from best with different K
            K = min(n, K + 5) if K < 100 else max(10, K - 5)
            f = best_f.copy()
            no_improve = 0

    return best_f, best_obj


def nesterov_subgrad(f0, time_limit, lr0=0.005, mom0=0.85, top_k=30, temp0=0.0001):
    """Nesterov-accelerated projected subgradient with softmax peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        # Compute subgradient from top-k peaks with softmax weighting
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * w[i] * f[k - indices]

        # Nesterov update
        v = cur_mom * v - cur_lr * sg
        y_new = project_simplex(y + v)
        f_new = y_new
        y = f_new + cur_mom * (f_new - f)
        y = project_simplex(y)
        f = f_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.005
        cur_mom = min(0.97, cur_mom * 1.0002)

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999999)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.5, 0.001)
            no_improve = 0
            conv = fast_convolve(f)

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using smooth interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_modulated_init(n, freq, amp, seed=0):
    """Generate a cosine-modulated initialization."""
    j = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n)
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / ss
    return a

                        a_opt, _ = topk_qp_optimize(a_opt, time_limit=min(80, int(rem * 0.4)), K=20)
                        try_seq(a_opt, f"topk_n{target_n}")

    # === Phase 6: Multi-start exploration ===
    print(f"\n=== Phase 6: Multi-start (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            t_alloc = min(100, int(rem * 0.25))

            if p == 0:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            elif p == 1:
                f_init = 0.8 * best_seq + 0.2 * cosine_modulated_init(n_best, 2, 0.3)
                f_init = project_simplex(f_init)
            elif p == 2:
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                f_init = 0.7 * best_seq + 0.3 * cosine_modulated_init(n_best, 1, 0.4)
                f_init = project_simplex(f_init)

            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"multi_{p}")

    # === Phase 7: Final Top-K QP polish ===
    print(f"\n=== Phase 7: Final Top-K QP polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t7 = min(150, int(rem * 0.5))
            print(f"  Time budget: {t7}s")
            a_opt, _ = topk_qp_optimize(best_seq, time_limit=t7, K=30, verbose=True)
            try_seq(a_opt, "final_topk")

    # === Phase 8: Final subgradient polish ===
    print(f"\n=== Phase 8: Final subgrad polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t8 = min(100, int(rem * 0.6))
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t8,
                                         lr0=0.002, mom0=0.93,
                                         top_k=max(15, n_best // 4),
                                         temp0=0.00003)
            try_seq(a_opt, "final_subgrad")

    # === Phase 9: Tiny perturbations ===
    print(f"\n=== Phase 9: Perturbations (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for _ in range(6):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            noise_scale = 0.001 + 0.001 * _ * 0.5
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            t_quick = min(30, int(rem * 0.4))
            if t_quick > 10:
                a_opt, _ = nesterov_subgrad(f_pert, time_limit=t_quick,
                                             lr0=0.001, mom0=0.94,
                                             top_k=max(10, n_best // 5),
                                             temp0=0.00002)
                try_seq(a_opt, f"perturb_{_}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
