import numpy as np

from scipy.optimize import minimize
from scipy.special import logsumexp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_stable(w):
    """Numerically stable softmax."""
    w_shifted = w - w.max()
    exp_w = np.exp(w_shifted)
    s = exp_w.sum()
    if s < 1e-300:
        return np.ones(len(w)) / len(w)
    return exp_w / s


def smooth_obj_and_grad_w(w, beta):
    """
    Compute smooth objective and gradient in w-space where a = softmax(w).
    Objective: 2n * logsumexp(beta * conv(a,a)) / beta
    """
    n = len(w)
    a = softmax_stable(w)
    conv = fast_convolve(a)
    beta_conv = beta * conv
    log_c = logsumexp(beta_conv)
    f_val = 2.0 * n * (log_c - np.log(beta))

    # p_k = exp(beta * conv_k - log_c)
    p = np.exp(beta_conv - log_c)
    s = 4.0 * n * p  # length 2n-1

    # grad_a[i] = sum_{j=0}^{n-1} s[i+j] * a[j]
    # Cross-correlation via FFT
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    S = np.fft.rfft(s, N)
    grad_a = np.fft.irfft(np.conj(A) * S, N)[:n]

    # Gradient w.r.t. w: grad_w = a * (grad_a - sum(a * grad_a))
    mean_g = np.dot(a, grad_a)
    grad_w = a * (grad_a - mean_g)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descent_polish(a, time_limit=30.0):
    """Coordinate descent to polish a normalized non-negative sequence."""
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    t0 = time.monotonic()

    order = np.random.default_rng(42).permutation(n)

        improved_any = False

            old_val = a[i]
            if old_val < 1e-12:

                    # Take mass from the largest other coordinate

                    mask[i] = False
                    j = np.argmax(np.where(mask, a, 0))
                    if a[j] < 1e-10:
                        continue
                    amt = a[j] * frac
                    a[i] = old_val + amt
                    a[j] -= amt
                    val = fast_convolve(a).max()
                    if val < best_obj - 1e-13:
                        best_obj = val
                        best_a = a.copy()
                        improved_any = True
                        break
                    else:
                        a[i] = old_val
                        a[j] += amt
            else:
                # Try reducing this coordinate
                for frac in [0.1, 0.3, 0.5, 0.7]:
                    amt = old_val * frac
                    a[i] = old_val - amt
                    # Distribute to others proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a[mask].sum()
                    if others_sum < 1e-15:
                        a[i] = old_val
                        continue
                    a[mask] += amt * (a[mask] / others_sum)
                    val = fast_convolve(a).max()
                    if val < best_obj - 1e-13:
                        best_obj = val
                        best_a = a.copy()
                        improved_any = True
                        break
                    else:

                        a[i] = old_val

                        mask2[i] = False
                        others_sum2 = a[mask2].sum()
                        if others_sum2 > 1e-15:

def pair_local_search(a, time_limit=30.0):
    """Pair-exchange local search."""
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        improved = False
        trials = 200
        for _ in range(trials):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-10:
                continue
            old_i, old_j = a[i], a[j]

                    improved = True

def make_structural_inits(n, rng):
    """Generate diverse structural initializations."""
    inits = []

    # 1. Uniform
    inits.append(np.ones(n) / n)

    inits.append(ramp2 / ramp2.sum())

    center2 = n * 0.75

    bump = 0.7 * np.exp(-0.5 * ((np.arange(n) - center1) / sigma) ** 2) + \
           1.3 * np.exp(-0.5 * ((np.arange(n) - center2) / sigma) ** 2)
    bump = np.maximum(bump, 1e-8)
    inits.append(bump / bump.sum())

    # 5. Cosine modulated (non-symmetric)

    freq = rng.uniform(1, 4)
    cos_mod = 1.0 + 0.4 * np.cos(2 * np.pi * freq * x + phase)
    cos_mod = np.maximum(cos_mod, 0.1)
    inits.append(cos_mod / cos_mod.sum())

    # 6. Power law (heavily weighted to one side)
    alpha = rng.uniform(0.5, 2.0)
    plaw = (x + 0.1) ** alpha
    inits.append(plaw / plaw.sum())

    # 7. S-shaped (sigmoid)
    k = rng.uniform(3, 10)
    sig = 1.0 / (1.0 + np.exp(-k * (x - 0.5)))
    sig = np.maximum(sig, 0.05)
    inits.append(sig / sig.sum())

    # 8. Random smoothed
    r = np.abs(rng.standard_normal(n)) + 0.1

    kernel = np.ones(kernel_len) / kernel_len
    r = np.convolve(r, kernel, mode='same')
    r = np.maximum(r, 1e-8)
    inits.append(r / r.sum())

    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 1e-10)
    return f_interp / f_interp.sum()

    if best_seq is not None:
        n_best = len(best_seq)
        a_safe = np.maximum(best_seq, 1e-15)
        a_safe = a_safe / a_safe.sum()
        w_best = np.log(a_safe)

        for beta in [1000000, 5000000]:
            rem = deadline - time.monotonic() - 20
            if rem < 8:
                break
            t_alloc = min(rem * 0.4, 60)
            print(f"  Fine beta={beta}, t={t_alloc:.0f}s")
            w_new = run_lbfgs(w_best, beta, max_iter=200, time_limit=t_alloc)
            a_result = softmax_stable(w_new)
            try_seq(a_result, f"fine_b{beta}")

        # Final coordinate descent
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_opt = coordinate_descent_polish(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "cd_final")

        # Final pair search
        rem = deadline - time.monotonic() - 5
        if rem > 3:
            a_opt = pair_local_search(best_seq, time_limit=min(rem, 15))
            try_seq(a_opt, "pair_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
