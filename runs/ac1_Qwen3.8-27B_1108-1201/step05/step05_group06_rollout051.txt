def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_obj(a):
    """Compute max(b) for normalized a (sum=1)."""
    b = fast_convolve(a)
    return b.max()

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_objective_and_grad(a):
    """
    Q(a) = sum(b^2) where b = a*a (full conv).
    Grad: dQ/da[j] = 4 * sum_m a[m] * b[j+m]
    """
    n = len(a)
    b = fast_convolve(a)
    Q = np.sum(b * b)
    # Gradient: grad[j] = 4 * sum_{m=0}^{n-1} a[m] * b[j+m]
    # This is the cross-correlation of a and b (first n values)
    # Compute via FFT: if we zero-pad a and b to length >= 2n-1, 
    # the cross-correlation is IFFT(conj(FFT(a)) * FFT(b))
    # Actually, grad[j] = sum_m a[m] * b[j+m]
    # Let's compute this as: for each j, grad[j] = dot(a, b[j:j+n])
    # This is a "sliding dot product" which can be done via correlation.
    # corr(a, b)[j] = sum_m a[m] * conj(b[m-j])... 
    # Actually, let's use the definition: grad[j] = sum_{m=0}^{n-1} a[m] * b[j+m]
    # = sum_{m=0}^{n-1} a[m] * b[j+m]
    # Let c = a (padded to 2n-1), d = reverse of b... 
    # Simpler: use np.convolve or direct FFT.
    # grad[j] = sum_m a[m] * b[j+m] for j = 0,...,n-1
    # This is the cross-correlation (a ⋆ b)[j] = sum_m a[m] * b[j+m]
    # = (a reversed) convolved with b, evaluated at specific points.
    # Let's compute via FFT: 
    # FFT of a (padded to L): A
    # FFT of b (padded to L): B
    # Cross-correlation: IFFT(conj(A) * B) gives sum_m a[m] * b[m+l]
    # We want sum_m a[m] * b[j+m] = cross_corr at lag j.
    # So we need IFFT(conj(A) * B)[j] for j=0,...,n-1.
    
    L = 1 << (3 * n - 1).bit_length()  # enough for conv of a (len n) and b (len 2n-1)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    cross_corr = np.fft.irfft(np.conj(A) * B, L)
    # cross_corr[l] = sum_m a[m] * b[m+l] for l = 0, 1, 2, ...
    grad = 4.0 * cross_corr[:n]
    return Q, grad

def lbfgs_smooth_opt(a0, time_limit=60.0, max_iter=5000):
    """Optimize Q(a)=sum(b^2) on the simplex using L-BFGS with reparam."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    # Reparameterize: a = softmax(z)
    # Clip to avoid overflow
    z = np.log(np.maximum(a, 1e-30))
    
    best_Q = smooth_objective_and_grad(a)[0]
    best_a = a.copy()
    t0 = time.monotonic()
    
    def obj_grad(z):
        a_z = np.exp(z - z.max())
        a_z = a_z / a_z.sum()
        Q, grad_a = smooth_objective_and_grad(a_z)
        # grad_z[i] = a_z[i] * (grad_a[i] - sum_j grad_a[j]*a_z[j])
        grad_mean = np.dot(grad_a, a_z)
        grad_z = a_z * (grad_a - grad_mean)
        return Q, grad_z
    
    # Use L-BFGS-B
    result = minimize(obj_grad, z, method='L-BFGS-B', jac=True,
                      options={'maxiter': max_iter, 'maxfun': max_iter,
                               'gtol': 1e-12, 'ftol': 1e-15})
    
    if result.success or result.fun < best_Q:
        a_result = np.exp(result.x - result.x.max())
        a_result = a_result / a_result.sum()
        Q_result = smooth_objective_and_grad(a_result)[0]
        if Q_result < best_Q:
            best_Q = Q_result
            best_a = a_result
    
    return best_a, best_Q

def lbfgs_smooth_opt_timed(a0, time_limit=60.0):
    """Time-limited L-BFGS on smooth objective."""
    # Run with increasing max_iter until time runs out
    a = a0.copy()

    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s

    def obj_grad(z_input):
        a_z = np.exp(z_input - z_input.max())
        a_z = a_z / a_z.sum()
        Q, grad_a = smooth_objective_and_grad(a_z)
        grad_mean = np.dot(grad_a, a_z)
        grad_z = a_z * (grad_a - grad_mean)
        return Q, grad_z
    
    current_z = z.copy()
    batch_iter = 200
    while time.monotonic() - t0 < time_limit:
        result = minimize(obj_grad, current_z, method='L-BFGS-B', jac=True,
                         options={'maxiter': batch_iter, 'maxfun': batch_iter,
                                  'gtol': 1e-14, 'ftol': 1e-16})
        current_z = result.x
        a_result = np.exp(current_z - current_z.max())
        a_result = a_result / a_result.sum()
        Q_result = smooth_objective_and_grad(a_result)[0]
        if Q_result < best_Q:
            best_Q = Q_result
            best_a = a_result.copy()
        if result.status == 0:  # convergence
            break
        if time.monotonic() - t0 > time_limit - 5:
            break
    
    # Also try to improve max(b) from the L-BFGS result
    return best_a, best_Q

def coordinate_bisection(a0, time_limit, verbose=False, max_bisect_iter=50):
    """Exact coordinate descent with bisection per coordinate for max(b)."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < 1000:
        passes += 1
        pass_improved = False
        max_improvement = 0.0

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            # b_others = a with a_j=0, normalized to sum=1
            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            B = fast_convolve(b)
            conv_len = 2 * n - 1

            # b_valid[k] = b[k-j] if 0 <= k-j < n, else 0
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            # f_k(x) = (1-x)^2 * B[k] + 2*x*(1-x) * b_valid[k] + x^2 * delta(k,2j)
            # We want to minimize max_k f_k(x) over x in [0, 1]
            
            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(max_bisect_iter):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                # Find the maximum
                f_max = f_vals.max()
                
                # Find which k achieves the max (and where derivative is positive)
                # derivative of f_k w.r.t. x:
                # d/dx [(1-x)^2 * B[k] + 2*x*(1-x)*b_valid[k] + x^2*delta]
                # = -2(1-x)*B[k] + 2*(1-2x)*b_valid[k] + 2x*delta
                # = 2*((1-x)*(b_valid[k]-B[k]) + x*(B[k]-2*b_valid[k]+delta) + ... )
                # Let me just compute it directly:
                # d/dx (1-x)^2 * B[k] = -2(1-x)*B[k]
                # d/dx 2x(1-x)*b_valid[k] = 2(1-2x)*b_valid[k]
                # d/dx x^2 * delta = 2x * delta
                
                deriv = -2.0 * (1.0 - mid) * B + 2.0 * (1.0 - 2.0 * mid) * b_valid
                if 2 * j < conv_len:
                    deriv[2 * j] += 2.0 * mid

                # We want to find x where max_k f_k(x) is minimized.
                # Use the bisection: if at the current max point, derivative > 0, decrease x.
                k_max = int(np.argmax(f_vals))
                if deriv[k_max] > 1e-14:
                    hi = mid
                else:
                    lo = mid

            x_new = (lo + hi) / 2.0

            if abs(x_new - x_old) < 1e-14:
                continue

            # Apply: a[j] = x_new, rescale others
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check

            conv = fast_convolve(a)
            obj = conv.max()
            improvement = best_obj - obj
            if improvement > max_improvement:
                max_improvement = improvement

            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True

        if not pass_improved:
            if passes > 3:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0003)
                conv = fast_convolve(a)
            else:
                break

    return best_a, best_obj
