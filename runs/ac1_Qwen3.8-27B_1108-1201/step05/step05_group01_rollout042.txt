"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected Nesterov subgradient
with adaptive peak weighting, exact coordinate descent polish, structured

Key improvements: exact 1D coordinate minimization, log-sum-exp smooth

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k, temperature=0.001):
    """Compute weighted subgradient of max(conv) using top-k peaks with softmax."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w[i] * f[k - indices]
    return sg


def spawn_exponential_init(n, alpha, side='right'):
    """Exponential decay initialization."""
    j = np.arange(n, dtype=np.float64)
    if side == 'right':
        a = np.exp(-alpha * j / n)
    elif side == 'left':
        a = np.exp(-alpha * (n - 1 - j) / n)
    else:
        a = np.exp(-alpha * np.abs(j - (n-1)/2.0) / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def spawn_polynomial_init(n, degree, skew):
    """Polynomial-shaped initialization."""
    j = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.ones(n)
    for d in range(1, degree + 1):
        a += np.cos(np.pi * j * d) * (skew ** d)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def spawn_block_init(n, num_blocks, heights, seed=42):
    """Block-structured initialization."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    block_size = n // num_blocks
    for i in range(num_blocks):
        start = i * block_size
        end = start + block_size if i < num_blocks - 1 else n
        a[start:end] = heights[i % len(heights)]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def subgradient_optimize(f0, time_limit, lr0=0.02, mom0=0.75, top_k=50,
                         temp0=0.0005, verbose=False):
    """Nesterov-accelerated projected subgradient with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        sg = compute_subgradient(f, conv, top_k, cur_temp)
        if np.linalg.norm(sg) < 1e-14:
            break

        # Nesterov momentum update
        v = cur_mom * v - cur_lr * sg
        y_new = project_simplex(y + v)

        # Nesterov lookahead
        beta = (1.0 + np.sqrt(1.0 + 4.0 * cur_mom * cur_mom * it)) / 2.0
        y = y_new + (cur_mom / beta) * (y_new - f)
        y = project_simplex(y)
        f = y_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameter updates
        cur_lr *= 0.9998
        cur_temp *= 1.003
        cur_mom = min(0.95, cur_mom * 1.0002)

        # Periodic restarts
        if no_improve > 200:
            restart_count += 1
            rng = np.random.default_rng(int(time.monotonic() * 100000) % (2**31))
            scale = 0.002 * min(restart_count, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            y = f.copy()
            v = np.zeros(n)
            cur_lr = max(lr0 * 0.7, 0.003)
            cur_mom = 0.8
            cur_temp = temp0 * (1 + 0.5 * restart_count)
            no_improve = 0
            conv = fast_convolve(f)

        # Verbose logging
        if verbose and it % 500 == 0:
            print(f"      SG iter {it}: obj={best_obj:.10f}, it={co:.10f}, lr={cur_lr:.6f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with golden-section 1D minimization per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    passes = 0
    no_pass_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            def eval_at(x):
                ft = f.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                return fast_convolve(ft).max()

            # Search range: [0, min(0.999, 3*old_val)]
            lo = 0.0
            hi = min(0.999, max(2.0 * old_val, old_val + 0.01))
            if hi <= lo:
                hi = lo + 0.01

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)
            f1 = eval_at(c1)
            f2 = eval_at(c2)

            for _ in range(14):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_at(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)

            cur_val = eval_at(old_val)
            if opt_val < cur_val - 1e-14:
                ft = f.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                ft = ft / ft.sum()
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()

        if verbose and passes % 5 == 0:
            print(f"      CD pass {passes}: obj={best_obj:.10f}")

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_f, best_obj


def logsumexp_optimize(f0, time_limit, T=50.0, lr0=0.01, verbose=False):
    """Optimize using log-sum-exp smoothing of the max, with BFGS-like updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    cur_lr = lr0
    cur_T = T
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Log-sum-exp weight on peaks
        scaled = cur_T * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Gradient of logsumexp(T*conv) = sum_k w[k] * d conv[k]/da
        sg = np.zeros(n)
        # Only process peaks with significant weight
        thr = 1e-8
        active = np.where(w > thr)[0]
        for k in active:
            m_lo = max(0, int(k) - (n - 1))
            m_hi = min(n - 1, int(k))
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * w[k] * f[int(k) - indices]

        # Projection onto tangent space of the simplex
        sg = sg - np.mean(sg)

        # Line search
        best_alpha = 0.0
        best_val = conv.max()
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            ft = project_simplex(f - alpha * cur_lr * sg)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = project_simplex(f - best_alpha * cur_lr * sg)
            conv = fast_convolve(f)
            co = conv.max()
            if co < best_obj - 1e-14:
                best_obj = co
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            cur_lr *= 0.999
        else:
            no_improve += 1
            cur_lr *= 0.9

        cur_T *= 1.001

        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            cur_lr = max(lr0 * 0.5, 0.001)
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"      LSE iter {it}: obj={best_obj:.10f}, T={cur_T:.1f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected Nesterov subgradient
with adaptive peak weighting, exact coordinate descent polish, structured
initializations (exponential, polynomial, block, two-hump Gaussian), log-sum-exp
smoothing, and multi-length exploration with interpolation.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k, temperature=0.001):
    """Compute weighted subgradient of max(conv) using top-k peaks with softmax."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            sg[indices] += 2.0 * w[i] * f[k - indices]
    return sg


def spawn_exponential_init(n, alpha, side='right'):
    """Exponential decay initialization."""
    j = np.arange(n, dtype=np.float64)
    if side == 'right':
        a = np.exp(-alpha * j / n)
    elif side == 'left':
        a = np.exp(-alpha * (n - 1 - j) / n)
    else:
        a = np.exp(-alpha * np.abs(j - (n - 1) / 2.0) / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def spawn_polynomial_init(n, degree, skew):
    """Polynomial-shaped initialization."""
    j = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.ones(n)
    for d in range(1, degree + 1):
        a = a + np.cos(np.pi * j * d) * (skew ** d)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def spawn_block_init(n, num_blocks, heights):
    """Block-structured initialization."""
    a = np.zeros(n)
    block_size = n // num_blocks
    for i in range(num_blocks):
        start = i * block_size
        end = start + block_size if i < num_blocks - 1 else n
        a[start:end] = heights[i % len(heights)]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def spawn_twohump_init(n, c1, c2, width):
    """Two Gaussian humps at positions c1*n and c2*n."""
    j = np.arange(n, dtype=np.float64)
    a1 = np.exp(-((j - c1 * n) / (width * n / 4.0)) ** 2)
    a2 = np.exp(-((j - c2 * n) / (width * n / 4.0)) ** 2)
    a = 0.5 * a1 + 0.5 * a2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def subgradient_optimize(f0, time_limit, lr0=0.02, mom0=0.75, top_k=50,
                         temp0=0.0005, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    no_improve = 0
    cur_lr = lr0
    cur_mom = mom0
    cur_temp = temp0
    it = 0
    restart_count = 0

        sg_norm = np.linalg.norm(sg)

        y_new = project_simplex(y + v)

        # Nesterov lookahead

        y = y_new + (cur_mom / beta) * (y_new - f)
        y = project_simplex(y)
        f = y_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.003

        if verbose and it % 500 == 0:

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with golden-section 1D minimization per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    GR = (1.0 + np.sqrt(5.0)) / 2.0
    inv_gr = 1.0 / GR
    passes = 0
    no_pass_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        pass_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            # Golden section search on a[j]
            lo = 0.0
            hi = min(0.999, max(2.0 * old_val, old_val + 0.005))
            if hi <= lo:
                hi = lo + 0.005

            c1 = lo + inv_gr * (hi - lo)
            c2 = lo + (1.0 - inv_gr) * (hi - lo)

            def eval_at(x):
                ft = f.copy()
                ft[j] = x
                ft2 = 1.0 - x
                if ft2 < 0:
                    return float('inf')
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                return fast_convolve(ft).max()

            f1 = eval_at(c1)
            f2 = eval_at(c2)

            for _ in range(14):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_gr * (hi - lo)
                    f1 = eval_at(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + (1.0 - inv_gr) * (hi - lo)
                    f2 = eval_at(c2)

            opt_x = (lo + hi) / 2.0
            opt_val = min(f1, f2)
            cur_val = fast_convolve(f).max()

            if opt_val < cur_val - 1e-14:
                ft = f.copy()
                ft[j] = opt_x
                ft2 = 1.0 - opt_x
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                ft[mask] = ft[mask] * (ft2 / denom)
                ft = ft / ft.sum()
                f = ft
                pass_improved = True
                new_obj = fast_convolve(f).max()
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_f = f.copy()

        if verbose and passes % 5 == 0:
            print(f"      CD pass {passes}: obj={best_obj:.10f}")

        if pass_improved:
            no_pass_improve = 0
        else:
            no_pass_improve += 1
            if no_pass_improve >= 3:
                break

    return best_f, best_obj

        it += 1
        conv = fast_convolve(f)

        # Log-sum-exp weights
        scaled = cur_T * conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Gradient
        sg = np.zeros(n)
        thr = 1e-8
        active = np.where(w > thr)[0]
        for k in active:
            m_lo = max(0, int(k) - (n - 1))
            m_hi = min(n - 1, int(k))
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                sg[indices] += 2.0 * w[int(k)] * f[int(k) - indices]

        sg = sg - np.mean(sg)

                best_alpha = alpha

            conv = fast_convolve(f)
            co = conv.max()

                best_obj = co
                best_f = f.copy()
