def _central_binomial(n):
    """Central binomial coefficient sequence, normalized to sum 1."""
    k = np.arange(n, dtype=np.float64)
    # For large k, use log-space to avoid overflow
    # log(C(2k,k)/4^k) = log(C(2k,k)) - k*log(4)
    # For k < 100, compute directly; for larger, use Stirling
    if n <= 500:
        a = np.array([math.comb(2*int(kk), int(kk)) / (4.0**int(kk)) for kk in range(n)], dtype=np.float64)
    else:
        # Use the approximation C(2k,k)/4^k ≈ 1/sqrt(pi*k) for large k
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for kk in range(1, n):
            # Recursive: C(2k,k)/4^k = C(2k-2,k-1)/4^(k-1) * (2k-1)/(2k)
            a[kk] = a[kk-1] * (2*kk-1) / (2*kk)
    s = np.sum(a)
    if s > 0:
        a /= s
    return a

def _smooth_objective_and_grad(a, p=4):
    """Compute ||a*a||_p^p and its gradient w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)  # length 2n-1
    obj = np.sum(c**p)
    
    # Gradient: d/da_i [sum_k c_k^p] = p * sum_k c_k^(p-1) * dc_k/da_i
    # dc_k/da_i = a[k-i]*2 if k=2i, else a[k-i] (for 0<=k-i<n)
    # So grad_i = p * sum_k c_k^(p-1) * a[k-i] * (1 + [k=2i])
    #            = p * (sum_{k:0<=k-i<n} c_k^(p-1) * a[k-i]) + p * c_{2i}^(p-1) * a_i
    
    # The first term is a cross-correlation
    cp = c**(p-1)  # length 2n-1
    # cross_corr[i] = sum_j cp[j+i] * a[j] for j=0..n-1 (valid when j+i < 2n-1)
    # This is np.correlate(cp, a, 'full') at specific indices
    
    # For efficiency, use direct computation for moderate n
    grad = np.zeros(n, dtype=np.float64)
    for i in range(n):
        # sum_{j=0}^{min(n-1, 2n-2-i)} a[j] * cp[j+i]
        maxj = min(n, 2*n-1-i)
        if maxj > 0:
            grad[i] = np.dot(a[:maxj], cp[i:i+maxj])
        grad[i] = p * (grad[i] + cp[2*i] * a[i])
    
    return obj, grad

def _gd_smooth(a, p=4, max_iter=200, tol=1e-12):
    """Projected gradient descent with momentum on ||a*a||_p^p, sum(a)=1, a>=0."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 0, None).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    a_prev = a.copy()
    momentum = 0.9
    prev_obj = np.inf
    
    for it in range(max_iter):
        obj, grad = _smooth_objective_and_grad(a, p)
        
        if abs(prev_obj - obj) < tol * max(1.0, abs(obj)):
            break
        prev_obj = obj
        
        # Momentum
        v = momentum * (a - a_prev) - (1.0 - momentum) * grad  # Hmm, this isn't standard
        # Actually, standard momentum: v = beta * v_prev - lr * grad
        # Let me use a simpler approach: just gradient with adaptive lr
        
        a_prev = a.copy()
        
        # Backtracking line search
        lr = 1.0 / (np.max(np.abs(grad)) + 1e-10)
        for _ in range(30):
            a_new = a - lr * grad
            a_new = np.clip(a_new, 0, None)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                lr *= 0.5
                continue
            a_new = a_new / s_new
            obj_new, _ = _smooth_objective_and_grad(a_new, p)
            if obj_new < obj:
                break
            lr *= 0.5
        
        # Check if we made progress
        if not np.isfinite(obj_new) or obj_new >= obj:
            # Try a smaller step
            a_new = a - 0.001 * grad
            a_new = np.clip(a_new, 0, None)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
            else:
                break
            a = a_new
            break
        
        a = a_new
    
    return a

def _compute_cross_corr(cp, a):
    """Compute (cp ⊛ a)[i] = sum_j cp[j+i] * a[j] for i = 0, ..., n-1.
    cp has length 2n-1, a has length n."""

    # We need indices i + n - 1 of the full convolution cp * a_flipped
    a_flip = a[::-1]
    L = 1
    while L < (2*n-1) + n:
        L *= 2
    cp_pad = np.zeros(L, dtype=np.float64)

    af_pad = np.zeros(L, dtype=np.float64)
