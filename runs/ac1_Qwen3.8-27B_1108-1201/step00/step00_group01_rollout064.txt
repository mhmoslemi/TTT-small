"""
Mirror-descent optimization of self-convolution peak using soft-max smoothing.
Works on the simplex (sum=1) since the objective is scale-invariant.
Uses multiplicative updates adapted via adaptive step-size and temperature annealing.
Periodic random perturbations escape local minima.
"""

import numpy as np
import time


def compute_soft_max_grad(a, T):
    """Compute the gradient of the soft-max approximation to max(conv(a,a)).
    
    Returns the gradient vector of length len(a).
    """
    n = len(a)
    conv = np.convolve(a, a)
    max_c = conv.max()
    shifted = T * (conv - max_c)
    # Clip to avoid underflow issues
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        # Fallback: just use max
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_shifted / w_sum
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # This equals 2 * np.correlate(w, a, 'full')[j] for j in [0, n-1]
    grad = 2.0 * np.correlate(w, a, mode='full')[:n]
    return grad, conv, max_c


def mirror_step(a, grad, alpha):
    """Perform a mirror descent (multiplicative) step with adaptation."""
    log_factor = np.clip(-alpha * grad, -40.0, 40.0)
    a_new = a * np.exp(log_factor)
    a_new_sum = a_new.sum()
    if a_new_sum < 1e-100 or not np.isfinite(a_new_sum):
        return None
    a_new = a_new / a_new_sum
    # Ensure no element is exactly zero (for numerical robustness)
    a_new = np.maximum(a_new, 1e-300)
    a_new = a_new / a_new.sum()
    return a_new
