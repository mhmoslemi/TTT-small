def fft_convolve(x):
    """Compute linear convolution x * x using FFT."""
    n = len(x)
    size = 2 * n - 1
    N = 1
    while N < size:
        N <<= 1
    fx = np.fft.rfft(x, N)
    c = np.fft.irfft(fx * fx, N)[:size]
    return np.maximum(c, 0.0)

    vo = np.zeros_like(x)

    b2 = 0.999

    phase = 0
    step = 0

        if step % 200 == 0 and phase < len(beta_schedule) - 1:
            phase += 1
            lr = max(0.0001, lr * 0.5)

        beta = beta_schedule[phase]
        a_cur = np.exp(x)
        s = np.sum(a_cur)

        c = fft_convolve(a_cur)
        scaled = beta * c
        cm = np.max(scaled)
        exps = np.exp(scaled - cm)
        sum_exps = np.sum(exps)
        M = (cm + np.log(sum_exps)) / beta

        # Compute g[j] = sum_i a[i] * w[i+j] for j=0..n-1
        # Using FFT: g = ifft(fft(a, N) * fft(w_rev, N))[n-1:2n-1][::-1]

        g = conv_full[n - 1:2 * n - 1][::-1].copy()

        grad_x = a_cur * grad_a

        # Adam update
        step += 1
        mo = b1 * mo + (1 - b1) * grad_x
        vo = b2 * vo + (1 - b2) * grad_x ** 2
        m_hat = mo / (1 - b1 ** step)
        v_hat = vo / (1 - b2 ** step)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
