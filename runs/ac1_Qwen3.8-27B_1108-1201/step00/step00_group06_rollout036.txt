from scipy.optimize import linprog
import time

def solve_lp(p, M):
    """Solve: max sum(q) s.t. q >= 0, (p*q)[k] <= M for all k"""
    n = len(p)
    # Variables: q_0, ..., q_{n-1}
    # Objective: maximize sum(q) => minimize -sum(q)
    c = -np.ones(n)
    
    # Constraints: (p*q)[k] <= M for k = 0, ..., 2n-2
    # (p*q)[k] = sum_j p[j] * q[k-j] (valid j)
    # For each k, the constraint is: sum_j A[k,j] * q[j] <= M
    # where A[k,j] = p[j] if 0 <= k-j <= n-1, else 0
    #           A[k,j] = p[k-j] if 0 <= k-j <= n-1, else 0
    # Wait, (p*q)[k] = sum_j p[j] * q[k-j]
    # Let m = k-j, so j = k-m
    # (p*q)[k] = sum_m p[k-m] * q[m] (valid m, k-m)
    # So the coefficient of q[m] in constraint k is p[k-m] (if valid)
    
    # A[k, m] = p[k-m] if 0 <= k-m <= n-1 and 0 <= m <= n-1, else 0
    # This is a banded matrix!
    
    # For efficiency, build the constraint matrix
    # A has shape (2n-1, n)
    # A[k, m] = p[k-m] if max(0, k-(n-1)) <= m <= min(n-1, k), else 0
    
    # Actually, let's build it as a sparse matrix or just directly
    # For n up to ~5000, the matrix is dense-ish (banded with bandwidth n)
    # Total non-zeros: sum_k (min(n, k+1) - max(0, k-n+1) + 1) 
    # = sum of (number of valid m for each k)
    # For k = 0..n-1: k+1 terms
    # For k = n..2n-2: 2n-1-k terms
    # Total = n(n+1)/2 + (n-1)n/2 = n^2
    
    # So A is n^2 elements total, which for n=5000 is 25M - might be too large for dense linprog
    
    # Alternative: use the structure. Each row has at most n non-zeros.
    # Use scipy.sparse and a suitable solver.
    
    from scipy.sparse import csr_matrix
    from scipy.sparse.linalg import ...
    
    # Actually, let's just build the dense matrix for moderate n
    # For n = 2000, A is 3999 x 2000, about 8M entries. That's manageable.
    
    A_rows = []
    for k in range(2*n - 1):
        m_lo = max(0, k - (n-1))
        m_hi = min(n-1, k)
        if m_lo > m_hi:
            row = np.zeros(n)
        else:
            row = np.zeros(n)
            # A[k, m] = p[k-m] for m in [m_lo, m_hi]
            m_idx = np.arange(m_lo, m_hi+1)
            row[m_idx] = p[k - m_idx]
        A_rows.append(row)
    
    A = np.array(A_rows)  # shape (2n-1, n)
    b = np.full(2*n-1, M)
    
    # Bounds: q >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res.x

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

This is the well-known algorithm for simplex projection.

def optimize_smooth(p_init, alpha_schedule, max_iters, learning_rate_schedule):
    p = p_init / p_init.sum()

        lr = learning_rate_schedule(it)
        p_new = p - lr * grad
        
        # Project onto simplex
        p_new = project_simplex(p_new)
        
        # Ensure non-negativity (should be guaranteed by projection)
        p = p_new
    
    return p

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

"""
Algorithm: Multi-start smooth optimization for minimizing the peak auto-convolution 
of a non-negative, sum-normalized sequence. Uses log-sum-exp smoothing of the max 
function, projected gradient descent on the probability simplex, and an optional 
LP-based polish step.
"""

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_obj_grad(p, alpha):
    """
    Compute smoothed objective and gradient for p (assumed to be on the simplex).
    Objective: (1/alpha) * log(sum_k exp(alpha * (p*p)[k]))
    This approximates max_k (p*p)[k] as alpha -> infinity.
    """
    n = len(p)
    conv = np.convolve(p, p)  # length 2n-1, all non-negative
    max_c = conv.max()
    shifted = alpha * (conv - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    value = max_c + np.log(Z) / alpha
    weights = exp_vals / Z
    
    # Compute gradient using convolutions
    # grad[i] = sum_k weights[k] * p[k-i] (valid) + sum_k weights[k] * p[i-k] (valid)
    # First term = cross-correlation = convolve(weights, p[::-1])[n-1+i]
    # Second term = convolution = convolve(weights, p)[i]
    wp = np.convolve(weights, p)  # length 3n-2
    w_pr = np.convolve(weights, p[::-1])  # length 3n-2
    
    grad = np.zeros(n)
    grad += wp[:n]  # second term: S2[i] = wp[i] for i=0..n-1
    grad += w_pr[n-1:n-1+n]  # first term: S1[i] = w_pr[n-1+i] for i=0..n-1
    
    return value, grad


def optimize_projected_gradient(p_init, n_iters=500, alpha_start=20.0, alpha_end=200.0, lr_start=1.0, lr_end=0.01):
    """
    Projected gradient descent with log-sum-exp smoothing.
    """
    p = p_init.copy()
    p = p / p.sum()
    n = len(p)
    best_p = p.copy()
    best_val = smooth_obj_grad(p, alpha_end)[0]
    
    for it in range(n_iters):
        # Anneal alpha and lr
        progress = it / max(1, n_iters - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        value, grad = smooth_obj_grad(p, alpha)
        
        # Gradient step + projection
        p_new = p - lr * grad
        p_new = project_simplex(p_new)
        
        # Check improvement (use a fixed alpha for comparison)
        check_val = smooth_obj_grad(p_new, 100.0)[0]
        if check_val < best_val:
            best_val = check_val
            best_p = p_new.copy()
        
        p = p_new
    
    return best_p, best_val
