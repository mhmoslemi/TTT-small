def logsumexp(x, axis=None):
    """Numerically stable log-sum-exp."""
    x = np.asarray(x, dtype=np.float64)
    m = np.max(x, axis=axis, keepdims=True)
    m = np.where(np.isfinite(m), m, 0.0)
    out = m + np.log(np.sum(np.exp(x - m), axis=axis, keepdims=True))
    if axis is not None:
        out = np.squeeze(out, axis=axis)
    else:
        out = float(out)
    return out


def fft_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def optimize_with_adam(a, n, max_iters, tau, lr, deadline):
    """Optimize using Adam with smooth log-sum-exp objective."""
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    S = np.sum(a)
    if S < 1e-10:
        return a, best_score
    a = a / S
    S = 1.0

    m_arr = np.zeros(a.shape, dtype=np.float64)
    v_arr = np.zeros(a.shape, dtype=np.float64)
    adam_t = 0
    prev_loss = np.inf
    stall_count = 0

    for it in range(max_iters):

def ipf_balance(a, iters=30):
    """One-shot IPF to balance the convolution while keeping a >= 0."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    target = 1.0 / (2 * n - 1)
    for _ in range(iters):
        conv = fft_convolve(a)
        excess = np.maximum(conv - target, 0.0)
        alpha = np.zeros(n, dtype=np.float64)
        np.add.at(alpha, np.arange(n), excess)
        idx2 = 2 * (n - 1) - np.arange(n)
        np.add.at(alpha, idx2, excess[idx2])
        a = np.maximum(a - 0.01 * alpha, 0.0)
        S_new = np.sum(a)
        if S_new < 1e-10:
            break
        a = a / S_new
    return a


def make_monomial(d, seed, rng):
    """Higher-order monomial pattern: k^d mod W with padding to prime n >= 2W."""
    W = 2 ** d
    n = W + d
    while n < 2 * W:
        p = n + 1
        is_prime = True
        if p < 2:
            is_prime = False
        elif p % 2 == 0:
            is_prime = (p == 2)
        else:
            for f in range(3, int(np.sqrt(p)) + 1, 2):
                if p % f == 0:
                    is_prime = False
                    break
        if is_prime:
            n = p
            break
        n += 2
    a = np.zeros(n, dtype=np.float64)
    k = np.arange(n, dtype=np.float64)
    centers = [float(W + d // 2)]
    for _ in range(5):
        j = int(np.power(float(seed), d) % W)
        centers.append(float(j))
        seed = (seed * 6364136223846793005 + 1442695040888963407) & 0xFFFFFFFFFFFFFFFF
    for c in centers:
        phase = np.exp(2j * np.pi * c / (n - 1))
        vals = np.exp(2j * np.pi * np.power(k, d, dtype=np.float64) / (n - 1) * np.power(phase, d))
        a += np.real(vals)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    return a

        sc = float(evaluate_sequence(a_ipf.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a_ipf.copy()
            print(f"  mono d={d} seed={s}: n={len(a)}, score={sc:.6f}")
        sc_orig = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc_orig) and sc_orig < best_score:
            best_score = sc_orig
            best = a.copy()
            print(f"  mono d={d} seed={s} (raw): n={len(a)}, score={sc:.6f}")

    # Phase 2: Try sparse-zero approach
    print("[Phase 2] Sparse-zero approach...")
    if best is not None and time.monotonic() < deadline - 200:
        # Try starting from best and adding zeros
        n0 = len(best)
        for n_try in [n0, n0 * 2, n0 * 3]:
            if time.monotonic() > deadline - 200:
                break
            a_z, sz_score = make_sparse_zero([], rng, n_try, iterations=5000)
            if np.isfinite(sz_score) and sz_score < best_score:
                best_score = sz_score
                best = a_z.copy()
                print(f"  sparse-zero n={n_try}: score={sz_score:.6f}")

    # Phase 3: Adam refinement on best candidate and variations
    print("[Phase 3] Adam refinement...")
    candidates = []
    if best is not None:
        candidates.append(best.copy())
    # Add some random candidates
    for _ in range(3):
        n_rand = int(rng.integers(1500, 5000))
        a_rand = np.abs(rng.standard_normal(n_rand)) + 0.1
        a_rand = a_rand / np.sum(a_rand)
        candidates.append(a_rand)
    # Add perturbed version of best
    if best is not None:
        n0 = len(best)
        if n0 < 8000:
            a_pert = best.copy()
            a_pert *= np.exp(rng.normal(0, 0.2, size=n0))
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / np.sum(a_pert)
            candidates.append(a_pert)

    for ci, a_init in enumerate(candidates):
        if time.monotonic() > deadline - 10:
            break
        n = len(a_init)
        print(f"  Adam candidate {ci}: n={n}, init_score={float(evaluate_sequence(a_init.tolist())):.6f}")
        # IPF first
        a_ipf = ipf_balance(a_init, iters=30)
        sc_ipf = float(evaluate_sequence(a_ipf.tolist()))
        if np.isfinite(sc_ipf) and sc_ipf < best_score:
            best_score = sc_ipf
            best = a_ipf.copy()
            print(f"    After IPF: {sc_ipf:.6f}")
        # Adam
        a_opt, adam_score = optimize_with_adam(a_ipf, n, max_iters=3000, tau=40.0, lr=0.003, deadline=deadline)
        if adam_score < best_score:
            best_score = adam_score
            best = a_opt.copy()
            print(f"    After Adam: {adam_score:.6f}")
        # Try with higher tau
        if time.monotonic() < deadline - 30:
            a_opt2, adam_score2 = optimize_with_adam(a_ipf, n, max_iters=1500, tau=80.0, lr=0.001, deadline=deadline)
            if adam_score2 < best_score:
                best_score = adam_score2
                best = a_opt2.copy()
                print(f"    After Adam (tau=80): {adam_score2:.6f}")

    # Final: try IPF on best with more iterations
    if best is not None and time.monotonic() < deadline - 5:
        a_final = ipf_balance(best, iters=50)
        sc_final = float(evaluate_sequence(a_final.tolist()))
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best = a_final.copy()
            print(f"  Final IPF: {sc_final:.6f}")

    print(f"\n[FINAL] Best score: {best_score:.6f}, n={len(best)}")
    return [float(x) for x in best]
