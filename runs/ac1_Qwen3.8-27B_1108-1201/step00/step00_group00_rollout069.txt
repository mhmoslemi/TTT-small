"""
Step-function minimizer for  score = 2*n*max(conv(a,a)) / (sum(a))**2 .

Strategy (scale-invariant, so we work with sum(a)=1):
  * Several good bases are prepared and the best is tracked globally.
      - the provided height_sequence_1 (normalised),
      - a U-shaped (edge-enhanced) base at a couple of lengths.
  * A fast frequency-domain (Gerchberg-Saxton style) magnitude-flattening
    pass is run to knock down large peaks quickly.
  * A gradient-descent phase minimises a soft-max surrogate
      L = logsumexp(beta * c),  c = conv(a, a)
    whose gradient is  grad_j = beta * sum_k w_k * a_{k-j},
    w_k = softmax(beta*(c - max c)).  The gradient is evaluated with two
    FFT-based convolutions and applied as a multiplicative (log-space)
    update so non-negativity is preserved automatically.
  * Simulated-annealing style perturbations escape local minima.
  * A final peak-shaving pass polishes the best sequence found.

All helpers are top-level, no closures/lambdas.  No IO.  evaluate_sequence
and height_sequence_1 are assumed to be pre-imported globals.
"""

import time
import numpy as np


def _safe_eval(a):
    """Robustly call evaluate_sequence; never raise, never return non-finite."""
    try:
        out = evaluate_sequence([float(x) for x in a])
    except Exception:
        return np.inf
    if out is None:
        return np.inf
    try:
        out = float(out)
    except Exception:
        return np.inf
    if not np.isfinite(out):
        return np.inf
    return out


def _normalize_nonneg(a, floor_frac=1e-9):
    """Clip to [0,1000], guarantee a small positive floor, normalise sum to 1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = a.sum()
    if s <= 1e-30:
        a = np.ones(n, dtype=np.float64) / n
        return a
    a = a / s
    a = np.maximum(a, floor_frac / n)
    a = a / a.sum()
    return a


def _make_ushape(n, edge_power, edge_sharp):
    """Edge-enhanced (bowl) base profile, normalised."""
    i = np.arange(n, dtype=np.float64)
    if n <= 2:
        x = i - (n - 1) / 2.0
    else:
        x = 1.0 - 2.0 * i / (n - 1)          # 1 at edges, -1 in middle
    prof = 1.0 + edge_power * np.abs(x) ** edge_sharp
    return prof / prof.sum()


def _gs_flatten(a, rng, iters, target_mag, alpha, eval_every=50):
    """Prometheus-innov: iterate FFT -> partial magnitude flattening ->
    keep first n, project non-negative, renormalise.  Returns best seen."""
    n = a.size
    fft_len = 1 << (2 * n).bit_length()
    a = _normalize_nonneg(a)
    best = a.copy()
    # initial score computed by caller; here we just track via convolution max
    for _ in range(iters):
        a_pad = np.zeros(fft_len, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        mag = np.maximum(np.abs(A), 1e-15)
        scale = (target_mag / mag) ** alpha
        A = A * scale
        a_new = np.fft.ifft(A).real[:n]
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s > 1e-30:
            a = a / s
    return _normalize_nonneg(a)


def _grad_step(a, beta, lr, rng, anneal):
    """One multiplicative gradient-descent step on logsumexp(beta*conv)."""
    n = a.size
    c = np.convolve(a, a)
    c_max = c.max()
    c_shift = c - c_max
    w = np.exp(beta * c_shift)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / wsum
    a_flip = a[::-1]
    conv = np.convolve(w, a_flip)
    grad = conv[(n - 1):(n - 1) + n]
    gscale = np.mean(np.abs(grad))
    if gscale < 1e-30:
        p = grad
    else:
        p = grad / gscale
    # multiplicative log-space update
    a = a * np.exp(-lr * p)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _improve_pipeline(a, rng, budget_s):
    """Run GS + gradient + annealing on a single base; return best (a, score)."""
    deadline = time.monotonic() + budget_s
    n = a.size
    a = _normalize_nonneg(a)
    best = a.copy()
    best_score = _safe_eval(best)

    # ---- GS pre-pass (capped, only keep improvements) ----
    gs_budget = min(120.0, max(20.0, 0.12 * budget_s))
    gs_end = time.monotonic() + gs_budget
    while time.monotonic() < gs_end:
        target_mag = 1.0 / np.sqrt(2 * n - 1)
        cand = _gs_flatten(best, rng, iters=200, target_mag=target_mag, alpha=0.35)
        sc = _safe_eval(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc
            print("    GS improved: %.6f" % best_score)
    if time.monotonic() > deadline:
        return a, best_score

    # ---- gradient + annealing phase ----
    a = best.copy()
    beta = 8.0
    lr = 0.25
    last_improve = time.monotonic()
    anneal_temp = 0.15
    it = 0
    while time.monotonic() < deadline - 5.0:
        a, _ = _grad_step(a, beta, lr, rng, anneal_temp)
        a = _normalize_nonneg(a)
        it += 1

        if it % 4 == 0:
            sc = _safe_eval(a)
            if np.isfinite(sc) and sc < best_score:
                best, best_score = a.copy(), sc
                last_improve = time.monotonic()
                if best_score < 1.6:
                    print("    grad improved: %.6f  (iter %d, beta %.1f)"
                          % (best_score, it, beta))

        # adapt hyper-parameters
        if it % 40 == 0:
            lr = max(0.03, lr * 0.97)
            beta = min(400.0, beta * 1.05)

        # simulated-annealing perturbation when stuck
        if time.monotonic() - last_improve > max(5.0, 0.02 * budget_s):
            noise = np.exp(rng.normal(0.0, 0.10, n))
            a_try = best * noise
            a_try = _normalize_nonneg(a_try)
            sc_try = _safe_eval(a_try)
            accept = (not np.isfinite(sc_try)) is False and \
                     (sc_try < best_score or
                      rng.random() < np.exp(-(sc_try - best_score) / max(anneal_temp, 1e-3)))
            if accept and np.isfinite(sc_try):
                a = a_try
                if sc_try < best_score:
                    best, best_score = a_try.copy(), sc_try
                    last_improve = time.monotonic()
            anneal_temp = max(0.02, anneal_temp * 0.9)
            last_improve = time.monotonic()

    return best, best_score


def _peak_shave(a, n_steps, deadline):
    """Polish: repeatedly shrink the contributors to the argmax lag."""
    n = a.size
    a = a.copy()
    best = a.copy()
    best_score = _safe_eval(best)
    for _ in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))
        j0 = max(0, k_star - n + 1)
        j1 = min(n - 1, k_star)
        a2 = a.copy()
        for j in range(j0, j1 + 1):
            part = a2[k_star - j]
            if part > 0:
                a2[j] *= (1.0 - 0.004 * part)
        a2 = _normalize_nonneg(a2)
        sc = _safe_eval(a2)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = a2.copy(), sc
    return best, best_score

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 10.0)

    # ---- build candidate bases ----
    bases = []
    parent = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 0 and np.isfinite(h).all() and h.sum() > 1e-30:
            parent = h / h.sum()
            bases.append(("parent-n" + str(parent.size), parent.copy()))
    except Exception:
        pass
    if parent is None:
        parent = _make_ushape(4806, edge_power=2.0, edge_sharp=1.0)
        bases.append(("default", parent.copy()))

    n0 = parent.size
    # U-shaped bases at a few lengths (larger n => better theoretical floor)
    for (ep, es), lab in [((1.6, 1.0), "U1"), ((2.5, 2.0), "U2")]:
        for L in (n0, min(2 * n0, 9600)):
            bases.append(("us-" + lab + "-" + str(L), _make_ushape(L, ep, es)))

    # de-duplicate by length+label order already distinct enough
    print("Bases to try:", [b[0] for b in bases], flush=True)

    # ---- allocate time and run pipelines ----
    n_b = len(bases)
    per = (deadline - time.monotonic()) / n_b
    min_per = 60.0
    # give more time to the parent-length bases first
    global_best = None
    global_score = np.inf
    scores_accum = []
    for idx, (lab, base) in enumerate(bases):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        alloc = min(per, remaining * 0.5 + min_per * 0.5)
        alloc = max(min_per * 0.5, min(alloc, remaining - 10.0))
        print("=== base %s : alloc %.0fs ===" % (lab, alloc), flush=True)
        try:
            a, sc = _improve_pipeline(base, rng, alloc)
        except Exception as e:
            print("    pipeline error:", e)
            continue
        if np.isfinite(sc) and sc < global_score:
            global_best, global_score = a, sc
            print(">>> new global best %.6f (base %s)" % (global_score, lab), flush=True)
        scores_accum.append((lab, sc))

    # ---- final peak-shave on the global best ----
    if global_best is None:
        global_best = bases[0][1]
        global_score = _safe_eval(global_best)
    if time.monotonic() < deadline - 3.0:
        global_best, global_score = _peak_shave(
            global_best, n_steps=2000, deadline=deadline - 2.0)
        print("after peak-shave: %.6f" % global_score, flush=True)

    final = [float(x) for x in np.clip(np.asarray(global_best, dtype=np.float64), 0.0, 1000.0)]
    fs = _safe_eval(final)
    if not np.isfinite(fs):
        final = [1.0] * 512
        fs = _safe_eval(final)
    print("FINAL score = %.6f  len = %d" % (fs, len(final)), flush=True)
    return final
