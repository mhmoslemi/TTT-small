                a = a * (n / s)
            no_improve = 0
            tau = a.std() * 2.0
            eta = 0.1
    
    return best_a, best_score


def local_search(a, rng, deadline, max_iters=20000):
    """Refine with random local perturbations."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = evaluate_sequence(a.tolist())
    best_a = a.copy()
    
    block_sizes = [1, 2, 3, 5, 8, 13, 21]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        bs = int(rng.choice(block_sizes, p=[0.2, 0.15, 0.12, 0.12, 0.1, 0.1, 0.1, 0.1] if len(block_sizes) >= 8 else None))
        if len(block_sizes) < 8:
            bs = block_sizes[rng.integers(0, len(block_sizes))]
        
        start = rng.integers(0, n - bs + 1)
        candidate = best_a.copy()
        candidate[start:start + bs] *= np.exp(rng.normal(0, 0.1, bs))
        candidate = np.clip(candidate, 0, 1000)
        
        score = evaluate_sequence(candidate.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = candidate
            if best_score < 1.51:
                print(f"  local iter={it}, score={best_score:.6f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search: multiplicative update + local refinement over multiple inits."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0, budget_s - 10)
    
    # Try to get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    except Exception:
        parent = None
    
    if parent is None or parent.size == 0 or parent.sum() < 0.01:
        parent = np.ones(256)
    
    n_parent = parent.size
    
    # We'll work with multiple lengths, preferring moderate sizes for speed
    candidate_lengths = []
    if n_parent >= 512:
        candidate_lengths.append(n_parent)
    for L in [2048, 4096, 1024, 8192]:
        if L not in candidate_lengths:
            candidate_lengths.append(L)
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: -x)
    
    global_best_a = parent.copy()
    global_best_score = evaluate_sequence(parent.tolist())
    print(f"Initial parent: n={n_parent}, score={global_best_score:.6f}")
    
    # Allocate time budget across different initials and lengths
    # Spend most time on the best-performing configuration
    
    for L in candidate_lengths:
        if time.monotonic() > deadline - 30:
            break
        
        # Create initialization of length L
        inits = []
        
        if L == n_parent:
            inits.append(("parent", parent[:L].copy() if n_parent >= L else np.resize(parent, L)))
        
        inits.append(("uniform", np.ones(L)))
        inits.append(("random", rng.random(L) + 0.01))
        
        # Bimodal (high at edges)
        x = np.linspace(0, 1, L)
        inits.append(("bimodal", (x - 0.5) ** 2 + 0.01))
        
        # Triangular (high in center) - this is the Hann-like shape
        inits.append(("triangle", np.minimum(x, 1 - x) + 0.01))
        
        # For longer runs, also try "arch" shape (inverse triangle)
        if L <= 4096:
            inits.append(("arch", 1.0 - np.minimum(x, 1 - x) + 0.01))
        
        phase_deadline = min(deadline, time.monotonic() + max(60, (budget_s - 30) / len(candidate_lengths) / 2))
        
        for init_name, a_init in inits:
            if time.monotonic() > phase_deadline - 5:
                break
            
            a_init = a_init.astype(np.float64)
            a_init = a_init * (L / a_init.sum())
            
            # Quick eval
            s0 = evaluate_sequence(a_init.tolist())
            if not np.isfinite(s0):
                continue
            
            print(f"  [{init_name}, L={L}] init score: {s0:.6f}")
            
            # Run multiplicative optimization
            time_limit = min(phase_deadline, time.monotonic() + max(30, (budget_s - 60) / 6))
            best_a, best_s = multiplicative_optimize(a_init, rng, time_limit)
            
            if best_s < global_best_score:
                global_best_score = best_s
                global_best_a = best_a.copy()
                print(f"  [{init_name}, L={L}] NEW BEST: {global_best_score:.6f}")
            
            # Quick local search if good
            if best_s < 2.0 and time.monotonic() < deadline - 30:
                ls_deadline = min(deadline, time.monotonic() + 30)
                ls_a, ls_s = local_search(best_a, rng, ls_deadline)
                if ls_s < global_best_score:
                    global_best_score = ls_s
                    global_best_a = ls_a.copy()
                    print(f"  [{init_name}, L={L}] LOCAL NEW BEST: {global_best_score:.6f}")
        
        # If we found something good, spend remaining time refining it
        if global_best_score < 1.6:
            break
    
    # Final refinement of the best found
    if time.monotonic() < deadline - 20:
        print(f"Final refinement phase, current best: {global_best_score:.6f}")
        refine_deadline = deadline
        best_a, best_s = multiplicative_optimize(global_best_a, rng, refine_deadline, max_iters=200000, eval_every=3)
        if best_s < global_best_score:
            global_best_score = best_s
            global_best_a = best_a.copy()
            print(f"  Refined: {global_best_score:.6f}")
        
        if time.monotonic() < deadline - 10:
            ls_a, ls_s = local_search(global_best_a, rng, deadline, max_iters=10000)
            if ls_s < global_best_score:
                global_best_score = ls_s
                global_best_a = ls_a.copy()
                print(f"  Local refined: {global_best_score:.6f}")
    
    print(f"FINAL: n={len(global_best_a)}, score={global_best_score:.6f}")
    return [float(x) for x in global_best_a]
