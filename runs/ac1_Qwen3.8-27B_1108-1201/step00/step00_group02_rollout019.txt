import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    nfft = 1 << (size - 1).bit_length()
    fa = fft(a, n=nfft)
    fb = fft(b, n=nfft)
    return np.real(ifft(fa * fb, n=nfft))[:size]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_for_g(f, M, active_k, n):
    m = len(active_k)
    if m == 0:
        return None
    A = np.zeros((m, n))
    for i in range(m):
        k = int(active_k[i])
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[i, j_min:j_max + 1] = f[k - j_max:k - j_min + 1][::-1]
    c_obj = -np.ones(n)
    b = np.full(m, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': 20, 'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None

def subgradient_phase(f, n, max_iters, deadline):
    """Run projected subgradient method."""
    best_M = np.inf
    best_f = f.copy()
    alpha = 0.01
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iters):

        M = np.max(c)
        if M < best_M:
            best_M = M
            best_f = f.copy()
        
        # Find top few peaks
        K = min(10, len(c))
        peak_indices = np.argpartition(c, -K)[-K:]
        
        # Compute averaged subgradient
        s = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)

        f_new = project_simplex(f - alpha * v)
        f = f_new
        
        # Decay step size

    return best_f, best_M

def lp_phase(f, n, max_iters, deadline):
    """Run LP-based constraint generation."""
    best_M = np.inf
    best_f = f.copy()

        if M < best_M:

            best_f = f.copy()

        top_idx = np.argpartition(c, -K)[-K:]
        
        g = solve_lp_for_g(f, M, top_idx, n)
        if g is None:
            break
        
        # Constraint generation
        for _ in range(10):
            fg = fft_convolve(f, g)
            viol = np.max(fg) - M
            if viol < 1e-10:
                break
            worst_k = int(np.argmax(fg))
            if worst_k not in top_idx:
                top_idx = np.append(top_idx, worst_k)
            g = solve_lp_for_g(f, M, top_idx, n)
            if g is None:
                g = f.copy()
                break
        
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        t_step = t
        for _ in range(10):
            f_new = (1 - t_step) * f + t_step * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / np.sum(f_new)
            c_new = fft_convolve(f_new, f_new)
            M_new = np.max(c_new)
            if M_new <= M * (1 + 1e-10):
                break
            t_step *= 0.5
        
        f = f_new
        if M_new < M:
            t = min(0.5, t * 1.1)
        else:
            t *= 0.5
    
    return best_f, best_M

def random_perturb_phase(f, n, rng, max_iters, deadline):
    """Random perturbation local search."""
