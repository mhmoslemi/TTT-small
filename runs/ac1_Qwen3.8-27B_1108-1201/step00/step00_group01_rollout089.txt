def get_subgradient(f, conv, top_k=5):
    n=len(f)
    M = np.max(conv)
    if top_k > 1 and 2*n-1 > top_k:
       part = np.argpartition(conv, -top_k)[-top_k:]
       vals = conv[part]
       # weights: emphasize values near M
       span = M - vals.min()
       if span > 0:
           w = np.exp(10*(vals - M)/span) # near 1 for max, small for lower
       else: w=ones
    else: w...
    grad=np.zeros(n)
    for k, w in zip(part,w):
       lo=max(0, k-n+1); hi=min(n, k+1); idx=np.arange(lo,hi)
       contrib=f[k-idx]+f[idx]
       diag=2*idx==k; contrib[diag]=2*f[idx[diag]]
       grad[lo:hi] += w*contrib
    grad /= w.sum()
    return grad, M

```python
import numpy as np
import time

"""
Multi-scale projected subgradient search for minimizing
    2*n*max(f*f) / (sum(f))**2
over non-negative real sequences.

Key ideas:
- Work on normalized simplex: sum(f) = sqrt(2n), so score = max(f*f).
- Use FFT-based self-convolution for fast internal scoring.
- Use projected subgradient steps of max(f*f), with backtracking line search.
- Break symmetry explicitly with anti-symmetric smooth perturbations.
- Run a coarse-to-fine schedule over several lengths, using the best found
  shape as a base for larger lengths.
- Keep an exact-verified best using the provided evaluate_sequence().
"""


def _fft_conv_self(x):
    """Full linear self-convolution of a 1D array using FFT."""
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(x, size)
    conv = np.fft.irfft(F * F, size)[:need]
    # FFT roundoff can create tiny negative values for non-negative input.
    # This is harmless for max, but clean it mildly.
    if conv.size:
        mx = float(np.max(conv))
        if mx > 0:
            conv[np.logical_and(conv < 0.0, conv > -1e-12 * mx)] = 0.0
    return conv


def _score_ff(x):
    """Fast internal score using FFT convolution."""
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    n = x.size
    if n == 0:
        return np.inf
    x = np.clip(x, 0.0, 1000.0)
    s = float(np.sum(x))
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    conv = _fft_conv_self(x)
    if conv.size == 0:
        return np.inf
    m = float(np.max(conv))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def _normalize_to_sum(x, target_sum):
    """Clip to valid range and scale to target_sum when possible."""
    x = np.asarray(x, dtype=np.float64).reshape(-1).copy()
    n = x.size
    if n == 0:
        return x
    x = np.clip(x, 0.0, 1000.0)
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 1e-300:
        val = target_sum / float(n)
        if val > 1000.0:
            val = 1000.0
        x.fill(max(0.0, val))
        return x
    x *= target_sum / s
    x = np.clip(x, 0.0, 1000.0)
    # If clipping caused a sum drop, scale back once if possible.
    s2 = float(np.sum(x))
    if np.isfinite(s2) and s2 > 1e-300 and s2 < target_sum:
        scale = target_sum / s2
        if scale > 1.0:
            x2 = np.clip(x * scale, 0.0, 1000.0)
            if np.sum(x2) >= s2:
                x = x2
    return x


def _resample_sequence(values, n):
    """Linearly resample a sequence to length n."""
    vals = np.asarray(values, dtype=np.float64).reshape(-1)
    vals = vals[np.isfinite(vals)]
    if vals.size == 0:
        return np.ones(max(1, int(n)), dtype=np.float64)
    vals = np.clip(vals, 0.0, 1000.0)
    n = max(1, int(n))
    if vals.size == n:
        return vals.copy()
    xp = np.linspace(0.0, vals.size - 1, vals.size)
    x = np.linspace(0.0, n - 1, n)
    return np.interp(x, xp, vals)


def _anti_smooth_unit(n, rng):
    """Smooth anti-symmetric perturbation with L2 norm 1."""
    n = max(1, int(n))
    if n <= 1:
        return np.zeros(n, dtype=np.float64)
    z = rng.standard_normal(n)
    Z = np.fft.rfft(z)
    freqs = np.fft.rfftfreq(n)
    sigma = rng.uniform(2.0, 20.0)
    mask = 1.0 / (1.0 + (freqs * sigma) ** 2)
    z = np.fft.irfft(Z * mask, n)
    z = z - z[::-1]
    z -= np.mean(z)
    norm = float(np.linalg.norm(z))
    if norm > 1e-300:
        z /= norm
    return z


def _smooth_positive(n, rng):
    """Smooth positive sequence, normalized to sum sqrt(2n)."""
    n = max(1, int(n))
    target = np.sqrt(2.0 * n)
    z = rng.standard_normal(n)
    Z = np.fft.rfft(z)
    freqs = np.fft.rfftfreq(n)
    sigma = rng.uniform(2.0, 20.0)
    mask = 1.0 / (1.0 + (freqs * sigma) ** 2)
    z = np.fft.irfft(Z * mask, n)
    std = float(np.std(z))
    if std < 1e-8:
        z = rng.standard_normal(n)
        std = 1.0
    z = np.exp(z / (std + 1e-8))
    return _normalize_to_sum(z, target)


def _random_mix(f, rng, mix):
    """Mix current f with a smooth positive random sequence."""
    f = np.asarray(f, dtype=np.float64).reshape(-1)
    n = f.size
    target = np.sqrt(2.0 * n)
    z = _smooth_positive(n, rng)
    mix = float(mix)
    mix = max(0.01, min(0.9, mix))
    return _normalize_to_sum((1.0 - mix) * f + mix * z, target)

            c = rng.uniform(0.20, 0.80)
            s = rng.uniform(0.10, 0.50)
            f = np.exp(drift * (x - 0.5) + 1.0 * np.exp(-((x - c) ** 2) / (2.0 * s * s)))

    # Tiny multiplicative noise breaks any residual exact symmetries.
    if n > 0:
        noise = 0.02 * rng.standard_normal(n)
        f = f * np.exp(noise)

    f = np.clip(f, 0.0, 1000.0)
    f = _normalize_to_sum(f, target)
    return f


def _gradient_direction(f, conv, top_k):
    """
    Subgradient heuristic for max(conv) with respect to f.
    Uses the top-k largest convolution values, weighted toward the maximum.
    """
    f = np.asarray(f, dtype=np.float64).reshape(-1)
    n = f.size
    if n == 0 or conv.size == 0:
        return np.zeros(0, dtype=np.float64), 0.0

    m = int(conv.size)
    M = float(np.max(conv))
    if not np.isfinite(M) or M <= 0.0:
        return np.zeros(n, dtype=np.float64), M

    top_k = max(1, int(top_k))
    if top_k >= m:
        part = np.arange(m)
        vals = conv
    else:
        part = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[part]

    minv = float(np.min(vals))
    span = M - minv
    if span > 1e-14:
        weights = np.exp(8.0 * (vals - M) / span)
    else:
        weights = np.ones_like(vals, dtype=np.float64)

    grad = np.zeros(n, dtype=np.float64)
    wsum = 0.0
    for kk, ww in zip(part, weights):
        ww = float(ww)
        if not np.isfinite(ww) or ww <= 1e-300:
            continue
        kk = int(kk)
        lo = kk - n + 1
        if lo < 0:
            lo = 0
        hi = kk + 1
        if hi > n:
            hi = n
        if hi <= lo:
            continue
        idx = np.arange(lo, hi, dtype=np.int64)
        contrib = f[kk - idx] + f[idx]
        diag = (2 * idx == kk)
        if np.any(diag):
            contrib = contrib.copy()
            contrib[diag] = 2.0 * f[idx[diag]]
        grad[lo:hi] += ww * contrib
        wsum += ww

    if wsum > 1e-300:
        grad /= wsum
    return grad, M


def _local_optimize(f, rng, max_time, step_scale, top_k):
    """
    Local projected subgradient optimizer on sum(f)=sqrt(2n).
    Returns (best_sequence, best_internal_score).
    """
    f = np.asarray(f, dtype=np.float64).reshape(-1)
    if f.size == 0 or not np.all(np.isfinite(f)):
        f = np.ones(128, dtype=np.float64)
    n = f.size
    target = np.sqrt(2.0 * float(n))
    f = _normalize_to_sum(f, target)

    conv = _fft_conv_self(f)
    M = float(np.max(conv)) if conv.size else 0.0
    if not np.isfinite(M) or M <= 0.0:
        return f, _score_ff(f)

    best = f.copy()
    best_M = M
    current_M = M
    scale = float(step_scale)
    end = time.monotonic() + max(0.0, float(max_time))
    stall = 0

    while time.monotonic() < end:
        grad, M_grad = _gradient_direction(f, conv, top_k)
        if not np.isfinite(M_grad):
            break
        if M_grad < current_M:
            current_M = float(M_grad)

        hdz = grad - float(np.mean(grad))
        nrm_h = float(np.linalg.norm(hdz))
        if not np.isfinite(nrm_h) or nrm_h < 1e-14:
            break

        nrm_f = float(np.linalg.norm(f))
        if not np.isfinite(nrm_f) or nrm_f < 1e-14:
            break

        alpha = scale * nrm_f / nrm_h
        tol = max(1e-10, 1e-10 * current_M)
        improved = False

        for _ in range(35):
            if time.monotonic() >= end:
                break
            cand = f - alpha * hdz
            cand = _normalize_to_sum(cand, target)
            conv_c = _fft_conv_self(cand)
            M_c = float(np.max(conv_c)) if conv_c.size else np.inf
            if np.isfinite(M_c) and M_c < current_M - tol:
                f = cand
                conv = conv_c
                current_M = M_c
                improved = True
                scale = min(10.0, scale * 1.05)
                break
            alpha *= 0.5
            if alpha <= 1e-12:
                break

        if improved:
            stall = 0
            if current_M < best_M - 1e-10:
                best = f.copy()
                best_M = current_M
        else:
            stall += 1
            scale *= 0.7
            if stall >= 30 or (stall >= 10 and scale < 1e-4):
                if rng.random() < 0.7:
                    f = _random_mix(f, rng, 0.15 + 0.15 * float(rng.random()))
                else:
                    a = _anti_smooth_unit(n, rng)
                    comp = float(rng.choice([0.05, 0.10, 0.20, 0.40, 0.70])) * np.sqrt(float(n))
                    f = _normalize_to_sum(f * (1.0 + comp * a), target)
                conv = _fft_conv_self(f)
                current_M = float(np.max(conv)) if conv.size else np.inf
                if (not np.isfinite(current_M)) or current_M > best_M * 1.25 + 1e-6:
                    f = best.copy()
                    conv = _fft_conv_self(f)
                    current_M = best_M
                scale = 0.15
                stall = 0

    return best, float(best_M)


def _eval_exact(seq):
    """Call the official evaluator safely."""
    try:
        arr = np.asarray(seq, dtype=np.float64).reshape(-1)
        arr = np.clip(arr, 0.0, 1000.0)
        return float(evaluate_sequence([float(v) for v in arr]))
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Generate a non-negative sequence with low evaluation score.
    The function returns the best exact-verified sequence found within budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    budget = max(0.0, float(budget_s) - 15.0)
    if budget <= 1.0:
        seq = np.ones(128, dtype=np.float64)
        seq = _normalize_to_sum(seq, np.sqrt(2.0 * seq.size))
        return [float(v) for v in seq]

    global_deadline = start_time + budget

    # Load the provided starting construction if available.
    try:
        original = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
    except Exception:
        original = np.ones(512, dtype=np.float64)

    if original.size == 0:
        original = np.ones(512, dtype=np.float64)
    original = original[np.isfinite(original)]
    if original.size == 0:
        original = np.ones(512, dtype=np.float64)
    original = np.clip(original, 0.0, 1000.0)

    orig_work = _normalize_to_sum(original, np.sqrt(2.0 * float(original.size)))

    best_lean_seq = orig_work.copy()
    best_lean_score = _score_ff(best_lean_seq)
    best_exact_seq = None
    best_exact_score = np.inf
    exact_calls = 0

    # Length schedule: short lengths are fast and can provide good shapes.
    standard_lengths = [64, 128, 256, 512, 1024, 2048, 4096]
    candidate_lengths = list(standard_lengths)
    if original.size not in candidate_lengths:
        candidate_lengths.append(original.size)

    lengths_set = set()
    for L in candidate_lengths:
        L = int(L)
        if 32 <= L <= 8192:
            lengths_set.add(L)
    if not lengths_set:
        lengths_set.add(512)
    lengths = sorted(lengths_set)

    weights = []
    for L in lengths:
        if L <= 128:
            w = 1.0
        elif L <= 256:
            w = 1.2
        elif L <= 512:
            w = 1.6
        elif L <= 1024:
            w = 1.7
        elif L <= 2048:
            w = 1.6
        elif L <= 4096:
            w = 1.3
        else:
            w = 1.6
        weights.append(float(w))

    final_reserve = min(400.0, max(60.0, budget * 0.18))
    stage_deadline = global_deadline - final_reserve
    if stage_deadline < start_time:
        stage_deadline = start_time + max(1.0, 0.2 * budget)

    for pos, L in enumerate(lengths):
        now = time.monotonic()
        if now >= stage_deadline:
            break

        rem_w = sum(weights[pos:])
        frac = weights[pos] / rem_w if rem_w > 0 else 1.0
        stage_end = now + max(0.5, (stage_deadline - now) * min(1.0, frac) * 0.98)
        stage_end = min(stage_end, stage_deadline)

        target = np.sqrt(2.0 * float(L))
        base_orig = _normalize_to_sum(_resample_sequence(original, L), target)
        base_best = None
        if best_lean_seq is not None and np.isfinite(best_lean_score) and best_lean_score < 2.2:
            base_best = _normalize_to_sum(_resample_sequence(best_lean_seq, L), target)

        if L <= 64:
            base_fast = 0.45
        elif L <= 128:
            base_fast = 0.7
        elif L <= 512:
            base_fast = 1.3
        elif L <= 2048:
            base_fast = 2.2
        else:
            base_fast = 1.5

        top_k = 5 if L <= 2048 else 3
        variant = 0

        while time.monotonic() < stage_end:
            remain = stage_end - time.monotonic()
            if remain < 0.35:
                break

            base_to_use = base_orig
            if base_best is not None and best_lean_score < 2.0 and (variant % 2 == 1):
                base_to_use = base_best

            cand = _make_variant(L, base_to_use, variant, rng)
            fast_t = min(base_fast, max(0.25, remain * 0.12))

            try:
                seq, sc = _local_optimize(cand, rng, fast_t, step_scale=0.2, top_k=top_k)
            except Exception:
                variant += 1
                continue

            changed = False
            if np.isfinite(sc) and sc < best_lean_score:
                best_lean_seq = seq
                best_lean_score = float(sc)
                changed = True
                print(f"[stage L={L}] lean score {best_lean_score:.8f} at "
                      f"{time.monotonic() - start_time:.1f}s", flush=True)

            # If this is promising, spend a little more on it immediately.
            if (best_lean_score < 2.00 and stage_end - time.monotonic() > 6.0
                    and changed is False and np.isfinite(sc) and sc < 2.2):
                extra = min(8.0, (stage_end - time.monotonic()) * 0.25)
                try:
                    seq2, sc2 = _local_optimize(seq, rng, extra, step_scale=0.1, top_k=top_k)
                    if np.isfinite(sc2) and sc2 < best_lean_score:
                        best_lean_seq = seq2
                        best_lean_score = float(sc2)
                        changed = True
                        print(f"[stage L={L}] extra lean score {best_lean_score:.8f} at "
                              f"{time.monotonic() - start_time:.1f}s", flush=True)
                except Exception:
                    pass

            # Exact-verify new lean bests, with rate limits for long lengths.
            if changed and time.monotonic() < global_deadline - 45:
                can_exact = False
                if L <= 2048:
                    can_exact = exact_calls < 40
                else:
                    can_exact = exact_calls < 18 and time.monotonic() < global_deadline - 90

                if can_exact:
                    ex = _eval_exact(best_lean_seq)
                    exact_calls += 1
                    if np.isfinite(ex) and ex < best_exact_score:
                        best_exact_score = float(ex)
                        best_exact_seq = best_lean_seq.copy()
                        print(f"[stage L={L}] exact score {best_exact_score:.8f} at "
                              f"{time.monotonic() - start_time:.1f}s", flush=True)

            # Refresh best-based starting point every few variants.
            if variant % 5 == 0 and best_lean_score < 2.0:
                try:
                    base_best = _normalize_to_sum(_resample_sequence(best_lean_seq, L), target)
                except Exception:
                    base_best = None

            variant += 1

    # Final refinement of the current best shape.
    now = time.monotonic()
    remaining = global_deadline - now - 30.0
    focus = best_lean_seq
    if best_exact_seq is not None and np.isfinite(best_exact_score) and best_exact_score < best_lean_score + 0.01:
        focus = best_exact_seq

    if focus is not None and remaining > 30.0:
        t = min(220.0, remaining - 30.0)
        top_k = 5 if focus.size <= 2048 else 3
        try:
            seqf, scf = _local_optimize(focus, rng, t, step_scale=0.08, top_k=top_k)
        except Exception:
            seqf, scf = None, np.inf

        if seqf is not None and np.isfinite(scf) and scf < best_lean_score:
            best_lean_seq = seqf
            best_lean_score = float(scf)
            changed = True
            print(f"[final refine] lean score {best_lean_score:.8f}", flush=True)

            if time.monotonic() < global_deadline - 45 and exact_calls < 50:
                ex = _eval_exact(best_lean_seq)
                exact_calls += 1
                if np.isfinite(ex) and ex < best_exact_score:
                    best_exact_score = float(ex)
                    best_exact_seq = best_lean_seq.copy()
                    print(f"[final refine] exact score {best_exact_score:.8f}", flush=True)

    # If still above target-ish and the original length is larger, try upsampling.
    decision_score = best_exact_score if np.isfinite(best_exact_score) else best_lean_score
    source = best_lean_seq
    if best_exact_seq is not None and np.isfinite(best_exact_score) and best_exact_score < best_lean_score + 0.01:
        source = best_exact_seq

    now = time.monotonic()
    remaining = global_deadline - now - 35.0
    if (source is not None and original.size > source.size
            and decision_score > 1.49 and remaining > 80.0):
        L = int(original.size)
        target = np.sqrt(2.0 * float(L))
        try:
            up = _normalize_to_sum(_resample_sequence(source, L), target)
            t = min(170.0, remaining - 35.0)
            seq_u, sc_u = _local_optimize(up, rng, t, step_scale=0.08, top_k=3)
        except Exception:
            seq_u, sc_u = None, np.inf

        if seq_u is not None and np.isfinite(sc_u) and sc_u < best_lean_score:
            best_lean_seq = seq_u
            best_lean_score = float(sc_u)
            print(f"[upsample L={L}] lean score {best_lean_score:.8f}", flush=True)

        if seq_u is not None and time.monotonic() < global_deadline - 45 and exact_calls < 55:
            ex = _eval_exact(best_lean_seq if np.isfinite(best_lean_score) and (
                best_exact_seq is None or best_lean_score < best_exact_score + 0.01
            ) else seq_u)
            exact_calls += 1
            if np.isfinite(ex) and ex < best_exact_score:
                best_exact_score = float(ex)
                best_exact_seq = (best_lean_seq if np.isfinite(best_lean_score) and (
                    best_exact_seq is None or best_lean_score < best_exact_score + 0.01
                ) else seq_u).copy()
                print(f"[upsample L={L}] exact score {best_exact_score:.8f}", flush=True)

    # Choose final candidate. Prefer exact-verified if it is at least as good
    # as the lean best within a small margin.
    candidate = best_lean_seq
    if best_exact_seq is not None and np.isfinite(best_exact_score) and best_exact_score < best_lean_score + 1e-7:
        candidate = best_exact_seq
    if candidate is None:
        candidate = orig_work

    final_seq = None
    final_score = np.inf

    fc = _eval_exact(candidate)
    if np.isfinite(fc):
        final_seq = candidate
        final_score = float(fc)

    if best_exact_seq is not None and np.isfinite(best_exact_score) and (
            final_seq is None or best_exact_score < final_score):
        final_seq = best_exact_seq
        final_score = float(best_exact_score)

    if final_seq is None or not np.isfinite(final_score):
        final_seq = orig_work
        final_score = _eval_exact(final_seq)

    if final_seq is None or not np.isfinite(final_score):
        final_seq = np.ones(256, dtype=np.float64)
        final_seq = _normalize_to_sum(final_seq, np.sqrt(2.0 * final_seq.size))
        final_score = _eval_exact(final_seq)

    if final_seq is None or not np.isfinite(final_score):
        final_seq = np.ones(128, dtype=np.float64)
        final_seq = _normalize_to_sum(final_seq, np.sqrt(2.0 * final_seq.size))
        return [float(v) for v in final_seq]

    final_seq = np.clip(np.asarray(final_seq, dtype=np.float64).reshape(-1), 0.0, 1000.0)
    return [float(v) for v in final_seq]
```
