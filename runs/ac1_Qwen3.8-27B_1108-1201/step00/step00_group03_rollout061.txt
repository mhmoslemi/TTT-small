from scipy.sparse import csr_matrix
import time

def solve_lp(f, M, n):
    """Solve the LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0"""
    # Build constraint matrix A where A[k, i] = f[k-i] if 0 <= k-i < n
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            rows.append(k)
            cols.append(i)
            vals.append(f[k-i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M (for each row)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return f.copy()

LP-based iterative optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses the "best response" LP at each iteration, with blending and occasional
random restarts to escape local optima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_indices(n):
    """Pre-compute (rows, cols) for the convolution constraint matrix."""
    cols = np.repeat(np.arange(n), n)
    # For each column i, rows are i, i+1, ..., i+n-1
    rows = np.empty(n * n, dtype=np.int32)
    for i in range(n):
        rows[i*n:(i+1)*n] = np.arange(i, i+n)
    return rows, cols

    """Create sparse matrix with pre-computed indices and values from f."""

    return res

def coord_descent_step(f, n, i, candidates):
    """Try changing f[i] to each candidate value, return best (value, new_conv_max)."""
    old = f[i]
    conv = np.convolve(f, f)
    conv_max = conv.max()
    total = f.sum()
    
    best_val = old
    best_score = 2 * n * conv_max / (total**2)
    
    for new_val in candidates:
        delta = new_val - old
        # Update convolution
        new_conv = conv.copy()
        # Affected positions: k where k-i is in [0,n)
        k_start = i
        k_end = i + n  # exclusive
        for k in range(k_start, min(k_end, 2*n-1)):

                new_conv[k] += 2*old*delta + delta*delta
            else:
                new_conv[k] += 2*delta*f[k-i]
        new_max = new_conv.max()
        new_total = total + delta
        if new_total > 0:
            score = 2 * n * new_max / (new_total**2)
            if score < best_score:
                best_score = score
                best_val = new_val

def coord_descent_pass(f, n):
    """One pass of coordinate descent with vectorized updates."""
    f = f.copy()
    conv = np.convolve(f, f)
    total = f.sum()
    
    for i in range(n):
        old = f[i]
        conv_max = conv.max()
        base_score = 2 * n * conv_max / (total**2) if total > 0 else float('inf')
        
        best_val = old
        best_score = base_score
        best_conv = conv
        
        for scale in [0.0, 0.5, 0.75, 1.5, 2.0]:
            new_val = old * scale if old > 0 else 0.0
            if new_val == old:
                continue
            delta = new_val - old
            
            # Compute updated convolution (vectorized)
            new_conv = conv.copy()
            k_start = i
            k_end = min(i + n, 2*n - 1)
            # For k in [k_start, k_end), the change is:
            # if k == 2i: 2*old*delta + delta^2
            # else: 2*delta*f[k-i]
            ks = np.arange(k_start, k_end)
            changes = 2 * delta * f[ks - i]
            if 2*i >= k_start and 2*i < k_end:
                changes[2*i - k_start] = 2*old*delta + delta*delta
            new_conv[ks] += changes
            
            new_total = total + delta
            if new_total > 0:
                new_max = new_conv.max()
                score = 2 * n * new_max / (new_total**2)
                if score < best_score:
                    best_score = score
                    best_val = new_val
                    best_conv = new_conv
        
        f[i] = best_val
        conv = best_conv
        total = f.sum()
    
    return f

"""
Combined LP-based iteration and coordinate descent for minimizing
2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_indices(n):
    """Pre-compute (rows, cols) for sparse convolution matrix."""
    cols = np.repeat(np.arange(n), n)
    rows = np.empty(n * n, dtype=np.int32)
    for i in range(n):
        rows[i * n:(i + 1) * n] = np.arange(i, i + n)
    return rows, cols


def lp_best_response(f, n, M, rows, cols):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns normalized g or None on failure."""
    vals = np.tile(f, n)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None


def coord_descent_pass(f, n):
    """One pass of coordinate descent with vectorized convolution updates."""
    f = f.copy()
    conv = np.convolve(f, f)
    total = f.sum()
    
    for i in range(n):
        old = f[i]
        if old == 0:
            # Try a few small positive values
            candidates = [0.0, 0.01, 0.1, 0.5]
        else:
            candidates = [0.0, old * 0.25, old * 0.5, old * 0.75, old * 1.25, old * 1.5, old * 2.0]
        
        best_val = old
        best_conv = conv
        best_total = total
        conv_max = conv.max()
        base_score = 2 * n * conv_max / (total ** 2) if total > 0 else float('inf')
        best_score = base_score
        
        for new_val in candidates:
            new_val = max(0.0, new_val)
            if new_val == old:
                continue
            delta = new_val - old
            new_total = total + delta
            if new_total <= 0:
                continue
            
            # Update convolution
            new_conv = conv.copy()
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            ks = np.arange(k_start, k_end)
            changes = 2.0 * delta * f[ks - i]
            if k_start <= 2 * i < k_end:
                changes[2 * i - k_start] = 2.0 * old * delta + delta * delta
            new_conv[ks] += changes
            
            new_max = new_conv.max()
            score = 2.0 * n * new_max / (new_total ** 2)
            if score < best_score:
                best_score = score
                best_val = new_val
                best_conv = new_conv
                best_total = new_total
        
        f[i] = best_val
        conv = best_conv
        total = best_total
    
    return f
