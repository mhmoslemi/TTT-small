```python
import numpy as np
import time

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimize the evaluation function using a one-sided power-law ramp.
    
    Key insight: For f[i] = C * ((i+1)/n)^p, the score is approximately
    2*(p!)^2*(p+1)^2/(2p+1)!, which is independent of n and C.
    The constraint is that the center of the self-convolution must exceed
    the boundary value C^2, which requires n * (p!)^2/(2p+1)! > 1.
    For n=6733, p=5 is optimal, giving score ~0.026.
    """
    try:
        n = len(height_sequence_1)
    except (NameError, TypeError):
        n = 6733
    
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    # Generate candidates with different exponents
    best = None
    best_score = float('inf')
    
    # Primary: power-law ramps f[i] = 1000 * ((i+1)/n)^p
    x = np.arange(1, n + 1, dtype=np.float64) / n
    for p in [1, 2, 3, 4, 5, 6]:
        f = 1000.0 * x**p
        f = np.clip(f, 0.0, 1000.0)
        score = evaluate_sequence(f.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = f.copy()
            print(f"p={p}: score={score:.8f}")
    
    print(f"Best ramp score: {best_score:.8f}")
    
    # Try alternative normalizations
    x2 = np.arange(0, n, dtype=np.float64) / (n - 1)
    for p in [3, 4, 5]:
        f = 1000.0 * x2**p
        f = np.clip(f, 0.0, 1000.0)
        score = evaluate_sequence(f.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = f.copy()
            print(f"alt norm, p={p}: score={score:.8f}")
    
    # Try with a small offset to avoid zero at the start
    for p in [4, 5]:
        x3 = (np.arange(1, n + 1, dtype=np.float64) + 0.5) / (n + 1)
        f = 1000.0 * x3**p
        f = np.clip(f, 0.0, 1000.0)
        score = evaluate_sequence(f.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = f.copy()
            print(f"offset, p={p}: score={score:.8f}")
    
    # Simple local refinement if time remains
    if best is not None and time.monotonic() < deadline:
        rng = np.random.default_rng(seed)
        best = best.copy()
        no_improve = 0
        
        while time.monotonic() < deadline and no_improve < 50:
            # Random block perturbation
            cand = best.copy()
            width = max(1, n // 100)
            start = int(rng.integers(0, max(1, n - width)))
            end = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.3)))
            cand[start:end] *= factor
            cand = np.clip(cand, 0.0, 1000.0)
            
            score = evaluate_sequence(cand.tolist())
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                no_improve = 0
                print(f"Refined: score={score:.8f}")
            else:
                no_improve += 1
    
    print(f"Final score: {best_score:.8f}")
    return [float(v) for v in best]
```
