    while fft_size < 3 * n:
        fft_size *= 2
    F_p = np.fft.rfft(p, fft_size)
    F_a = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(F_p * np.conj(F_a), fft_size)
    return cross[:n]


def compute_objective_and_gradient(z, tau, n):
    """Compute soft-peak objective and gradient w.r.t. z (softmax params)."""
    a = _softmax(z)
    conv = _compute_conv(a, n)

    # Soft-max probabilities
    cs = conv / tau
    cs_max = cs.max()
    exp_cs = np.exp(cs - cs_max)
    exp_sum = exp_cs.sum()
    if exp_sum < 1e-300:
        p = np.ones_like(conv) / len(conv)
    else:
        p = exp_cs / exp_sum

    # Smoothed objective
    obj = 2.0 * n * tau * (cs_max + np.log(exp_sum))

    # Gradient of soft-peak w.r.t. a
    cross = _compute_cross(p, a, n)
    grad_a = cross.copy()
    # Second-order correction: a[j]*p[2j] for the k=2j diagonal
    grad_a += a * p[0:2 * n - 1:2]

    # Chain rule through softmax: grad_z[k] = a[k]*(grad_a[k] - dot(grad_a,a))
    sum_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - sum_ga_a)

    return obj, grad_z, conv, p


def _obj_grad_wrapper(z):
    """Top-level callable for scipy minimize (no closures)."""
    result = compute_objective_and_gradient(z, _tau_current, _n_current)
    return result[0], result[1]


def _valid_sequence(a):
    """Check if sequence is valid for evaluation."""
    if not np.all(np.isfinite(a)):
        return False
    if a.min() < -1e-10:
        return False
    if a.max() > 1001:
        return False
    if a.sum() < 0.01:
        return False
    return True


def _try_update(a, best, best_score):
    """Evaluate candidate and update best if improved."""
    a_clip = np.clip(a, 0.0, 1000.0)
    if a_clip.sum() < 0.01:
        return best_score, best
    score = float(evaluate_sequence(a_clip.tolist()))
    if np.isfinite(score) and score < best_score:
        return score, a_clip.copy()
    return best_score, best


def _make_init(type_idx, n, rng, parent):
    """Generate an initialization of the given type."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if type_idx == 0:
        # Parent sequence
        a = parent.copy()
    elif type_idx == 1:
        # Raised cosine
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / max(n - 1, 1)))
    elif type_idx == 2:
        # Gaussian
        sigma = n / 4.0
        a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    elif type_idx == 3:
        # Triangular
        a = np.maximum(1.0 - np.abs(j - center) / max(center, 1.0), 0.0)
    elif type_idx == 4:
        # Random
        a = rng.random(n)
    elif type_idx == 5:
        # Edge-weighted (inverse raised cosine)
        a = 1.0 - 0.5 * (1.0 + np.cos(2.0 * np.pi * j / max(n - 1, 1)))
        a = a + 0.05
    elif type_idx == 6:
        # Low-frequency sinusoid
        a = 1.0 + 0.85 * np.cos(2.0 * np.pi * 2.0 * j / n)
        a = np.maximum(a, 0.01)
    elif type_idx == 7:
        # Multi-frequency
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * j / n) + 0.3 * np.cos(2.0 * np.pi * 7.0 * j / n)
        a = np.maximum(a, 0.01)
    elif type_idx == 8:
        # Parent with noise
        a = parent.copy()
        a = a * np.exp(rng.normal(0, 0.3, n))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)

    a_sum = a.sum()
    if a_sum < 1e-10:
        a = np.ones(n)
    else:
        a = a / a_sum
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses softmax-parameterized gradient descent with
    soft-max peak approximation and FFT-based gradients. Multiple restarts
    with diverse initializations and decreasing temperature.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0

    # Get starting sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    except:
        parent = np.ones(512, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all() or parent.sum() < 0.01:
        parent = np.ones(512, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Start: n={n}, score={best_score:.6f}", flush=True)

    # Precompute some constants
    base_tau = 1.0 / (2.0 * n)

    # Temperature multipliers (relative to base_tau)
    tau_mults = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.10, 0.07, 0.05, 0.03, 0.02]

    # Number of init types
    n_init_types = 3
    init_types = [0, 1, 5]  # parent, raised cosine, edge-weighted

    # If we have extra time, try more init types
    extra_types = [2, 6, 7, 8, 4]

    total_starts = len(tau_mults) * n_init_types
    attempt = 0

    for tau_idx, tm in enumerate(tau_mults):
        if time.monotonic() > deadline:
            break
        tau = base_tau * tm

        for init_idx in range(n_init_types):
            if time.monotonic() > deadline:
                break

            attempt += 1
            remaining = deadline - time.monotonic()
            if remaining < 3:
                break

            type_idx = init_types[init_idx]
            a_init = _make_init(type_idx, n, rng, parent)

            # Add perturbation for diversity (except first)
            if attempt > 1:
                a_init = np.maximum(a_init + rng.normal(0, 0.05, n), 1e-10)
                a_init = a_init / a_init.sum()

            # Convert to z-space
            z0 = np.log(a_init)
            z0 = z0 - z0.mean()

            # Set global params
            global _tau_current, _n_current
            _tau_current = tau
            _n_current = n

            # Determine max iterations based on time budget
            max_iter = min(200, max(30, int(remaining * 20)))

            try:
                result = minimize(
                    _obj_grad_wrapper,
                    z0,
                    method='L-BFGS-B',
                    jac=True,
                    options={'maxiter': max_iter, 'gtol': 1e-10, 'ftol': 1e-12, 'maxls': 40}
                )

                # Get the result
                a_new = _softmax(result.fun_z if hasattr(result, 'fun_z') else result.x)

                # Scale to original magnitude
                target_sum = best.sum()
                a_new = a_new * (target_sum / a_new.sum())
                a_new = np.clip(a_new, 0, 1000)

                new_score, new_best = _try_update(a_new, best, best_score)
                if new_score < best_score:
                    best = new_best
                    best_score = new_score
                    print(f"  [{attempt}] tau_x={tm:.2f}, init={init_types[init_idx]}: {best_score:.6f} (nit={result.nit})", flush=True)
            except Exception as e:
                print(f"  [{attempt}] Exception: {e}", flush=True)

    # Additional passes with more diverse initializations if time remains
    if time.monotonic() < deadline - 30:
        for type_idx in extra_types:
            if time.monotonic() > deadline:
                break
            for tm in [0.1, 0.05, 0.03]:
                if time.monotonic() > deadline:
                    break
                tau = base_tau * tm
                remaining = deadline - time.monotonic()
                if remaining < 3:
                    break

                a_init = _make_init(type_idx, n, rng, parent)
                a_init = np.maximum(a_init + rng.normal(0, 0.03, n), 1e-10)
                a_init = a_init / a_init.sum()
                z0 = np.log(a_init)
                z0 = z0 - z0.mean()

                _tau_current = tau
                _n_current = n

                try:
                    result = minimize(
                        _obj_grad_wrapper,
                        z0,
                        method='L-BFGS-B',
                        jac=True,
                        options={'maxiter': 150, 'gtol': 1e-10, 'ftol': 1e-12}
                    )
                    a_new = _softmax(result.x)
                    a_new = a_new * (best.sum() / a_new.sum())
                    a_new = np.clip(a_new, 0, 1000)

                    new_score, new_best = _try_update(a_new, best, best_score)
                    if new_score < best_score:
                        best = new_best
                        best_score = new_score
                        print(f"  [extra type={type_idx}, tau_x={tm:.2f}]: {best_score:.6f}", flush=True)
                except:
                    pass

    # Final local refinement: small perturbations around best
    if time.monotonic() < deadline - 5:
        for _ in range(200):
            if time.monotonic() > deadline:
                break
            candidate = best.copy()
            # Random block perturbation
            width = max(1, n // 50)
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.05))
            candidate = np.maximum(candidate, 0)
            new_score, new_best = _try_update(candidate, best, best_score)
            if new_score < best_score:
                best = new_best
                best_score = new_score

    final_score = float(evaluate_sequence(best.tolist()))
    print(f"FINAL: score={final_score:.6f}, n={n}", flush=True)

    return [float(x) for x in best]
