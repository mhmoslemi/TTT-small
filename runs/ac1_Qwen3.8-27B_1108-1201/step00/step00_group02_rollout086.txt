import numpy as np
import time

def _next_pow2(m):
    p=1
    while p < m:
        p <<= 1
    return p

def _normalize_array(a):
    if a is None: return None
    arr=np.asarray(a, dtype=np.float64).reshape(-1)
    if arr.size ==0: return None
    arr=np.where(np.isfinite(arr), arr, 0.0)
    arr=np.maximum(arr, 0.0)
    s=float(arr.sum())
    if s <= 0.0: return None
    return arr/s

def _safe_softmax(z):
    z=np.asarray(z, dtype=np.float64).reshape(-1)
    if z.size ==0: return np.array([])
    if not np.all(np.isfinite(z)):
        z=np.nan_to_num(z, nan=0.0, posinf=12.0, neginf=-12.0)
    z=z - float(z.mean())
    z=np.clip(z, -12.0, 12.0)
    m=float(np.max(z))
    e=np.exp(z-m)
    s=float(e.sum())
    if not np.isfinite(s) or s <=0.0:
        e=np.zeros(z.shape, dtype=np.float64)
        e[0]=1.0
        s=1.0
    return e/s

def _a_to_z(a):

    z=np.log(np.maximum(a, tiny))
    z=z - float(z.mean())
    return np.clip(z, -12.0, 12.0)

import numpy as np
import time


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _clean_clip(arr, lo, hi):
    arr = np.asarray(arr, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    return np.clip(arr, lo, hi)


def _normalize_array(a):
    if a is None:
        return None
    arr = np.asarray(a, dtype=np.float64).reshape(-1)
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 0.0:
        return None
    return arr / s


def _safe_softmax(z):
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = _clean_clip(z, -12.0, 12.0)
    z = z - float(z.mean())
    z = np.clip(z, -12.0, 12.0)
    mz = float(np.max(z))
    e = np.exp(z - mz)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.zeros(z.shape, dtype=np.float64)
        e[0] = 1.0
        s = 1.0
    return e / s


def _a_to_z(a):
    a = _normalize_array(a)
    if a is None:
        return np.zeros(1, dtype=np.float64)
    tiny = np.finfo(np.float64).tiny
    z = np.log(np.maximum(a, tiny))
    z = z - float(z.mean())
    return np.clip(z, -12.0, 12.0)


def _internal_score(a):
    a = _normalize_array(a)
    if a is None:
        return np.inf
    n = a.size
    if n < 1:
        return np.inf
    L = 2 * n - 1
    pad = _next_pow2(L)

def _centered_unit(x):
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    if x.size == 0:
        return np.array([], dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = x - float(x.mean())
    m = float(np.max(np.abs(x)))
    if m <= 1e-300:
        return np.zeros_like(x)
    return x / m


def _random_anti(n, rng):
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    m = (n + 1) // 2
    r = rng.normal(size=m)
    k = min(32, max(1, m // 8))
    if k > 1:
        r = np.convolve(r, np.ones(k) / k, mode="same")
    rmin = float(r.min())
    rmax = float(r.max())
    if rmax > rmin:
        r = -0.8 + 1.6 * (r - rmin) / (rmax - rmin)
    else:
        r.fill(0.0)
    x = np.empty(n, dtype=np.float64)
    x[:m] = r
    right = r[::-1]
    if n % 2 == 0:
        x[m:] = -right
    else:
        x[m:] = -right[:n - m]
    return np.maximum(1.0 + x, 1e-12)


def _random_smooth(n, rng):
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    r = rng.normal(size=n)
    if n > 16:
        f = np.fft.rfft(r)
        keep = max(2, n // 32)
        f[keep:] = 0.0
        r = np.fft.irfft(f, n=n)
    r = r - float(r.mean())
    rmax = float(np.abs(r).max())
    if rmax > 1e-300:
        r = 0.8 * r / rmax
    else:
        r.fill(0.0)
    return np.maximum(1.0 + r, 1e-12)


def _random_softmax_weight(n, rng):
    z = rng.normal(size=n)
    z = z - float(z.mean())
    z = np.clip(z, -6.0, 6.0)
    e = np.exp(z - float(np.max(z)))
    s = float(e.sum())
    if s <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    return e / s


def _add_start(starts, n, w):
    if starts is None or n <= 0:
        return
    arr = _normalize_array(w)
    if arr is not None and arr.size == n:
        starts.append(arr)


def _make_starts(n, rng, parent=None, desired=18):
    starts = []
    if n <= 1:
        _add_start(starts, n, np.array([1.0], dtype=np.float64))
        return starts

    if parent is not None and np.asarray(parent).size == n:
        _add_start(starts, n, parent)

    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0

    for c in (0.5, 0.35, 0.65, 0.8):
        _add_start(starts, n, 1.0 + c * u)

    f = u * (1.0 - u * u)
    fmax = float(np.max(np.abs(f)))
    if fmax > 1e-300:
        f = f / fmax
    for c in (1.0, 1.5, 2.0):
        _add_start(starts, n, 1.0 + c * f)

    for al in (0.8, 1.2, 1.6):
        _add_start(starts, n, np.exp(al * u))

    s = _centered_unit(np.cos(2.0 * np.pi * t))
    for c in (0.6, 1.0):
        _add_start(starts, n, 1.0 + c * s)

    s2 = _centered_unit(np.sin(4.0 * np.pi * t))
    _add_start(starts, n, 1.0 + 0.7 * s2)

    _add_start(starts, n, 1.0 + 0.5 * u + 0.6 * s)
    _add_start(starts, n, 1.0 + 0.5 * u - 0.6 * s)

    extra = desired - len(starts)
    if extra > 0:
        for _ in range(min(extra, 4)):
            _add_start(starts, n, _random_anti(n, rng))
        extra = desired - len(starts)
        for _ in range(min(extra, 3)):
            _add_start(starts, n, _random_smooth(n, rng))
        extra = desired - len(starts)
        for _ in range(min(extra, 2)):
            _add_start(starts, n, _random_softmax_weight(n, rng))

    if len(starts) > desired:
        starts = starts[:desired]

    if not starts:
        _add_start(starts, n, np.ones(n, dtype=np.float64))

    return starts


def _linear_sequence(n, c=0.5):
    if n <= 0:
        n = 1
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    t = np.linspace(0.0, 1.0, n)
    w = 1.0 + c * (2.0 * t - 1.0)
    w = np.maximum(w, 1e-12)
    s = float(w.sum())
    if s <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    return w / s

                used_ids.add(ident)
            break

    for rec in heat_pool:
        if len(heat_seqs) >= 1:
            break
        seq = rec[3]
        ident = id(seq)
        if ident not in used_ids:
            heat_seqs.append(seq)
            used_ids.add(ident)

    for rec in heat_pool:
        if len(heat_seqs) >= 3:
            break
        seq = rec[3]
        ident = id(seq)
        if ident not in used_ids:
            heat_seqs.append(seq)
            used_ids.add(ident)

    if not heat_seqs:
        heat_seqs = [best_seq]

    heat_results = []
    for rank, seq in enumerate(heat_seqs[:3]):
        if time.monotonic() >= deadline - 60.0:
            break
        n = seq.size
        if n > 4000:
            ht = 120.0
        elif n > 1000:
            ht = 90.0
        else:
            ht = 60.0
        end = time.monotonic() + ht
        if end > deadline:
            end = deadline
        if end - time.monotonic() < 30.0:
            break

        res_seq, res_score = _optimize(
            seq,
            end,
            seed=int_seed + 100 + rank,
            tau_start=1.2,
            tau_end=0.15,
            lr_start=0.08,
            lr_end=0.04,
            eval_interval=1.0 if n > 5000 else 0.4,
            patience=700,
            verbose=False,
        )
        heat_results.append((res_seq, float(res_score), n))
        if np.isfinite(res_score) and (best_seq is None or res_score < best_score):
            best_seq = res_seq.copy()
            best_score = float(res_score)
        print(f"[heat] n={n} heat={res_score:.6f} best={best_score:.6f}")

    long_info = []
    long_ids = set()
    _add_candidate(long_info, long_ids, best_seq, best_score)

    for rec in heat_results:
        _add_candidate(long_info, long_ids, rec[0], rec[1])

    off_sorted = []
    for i, rec in enumerate(official_candidates):
        off_sorted.append((float(rec[1]), i, rec[0]))
    off_sorted.sort()
    for rec in off_sorted[:6]:
        _add_candidate(long_info, long_ids, rec[2], rec[0])

    long_info.sort()
    long_info = long_info[:3]

    start_final = time.monotonic()
    total_final = max(0.0, deadline - start_final)

    if long_info and total_final > 30.0:
        counts = len(long_info)
        if counts == 1:
            ratios = [1.0]
        elif counts == 2:
            ratios = [0.65, 0.35]
        else:
            ratios = [0.55, 0.30, 0.15]

        for rank, rec in enumerate(long_info):
            if time.monotonic() >= deadline - 20.0:
                break
            seq = rec[3]
            score = rec[4]
            n = seq.size
            frac = ratios[rank]
            dur = frac * total_final
            end = time.monotonic() + dur
            if end > deadline:
                end = deadline
            if end - time.monotonic() < 30.0:
                break

            if score > 1.8:
                tau_s = 0.45
                lr_s = 0.05
            elif score > 1.6:
                tau_s = 0.25
                lr_s = 0.035
            else:
                tau_s = 0.12
                lr_s = 0.025

            res_seq, res_score = _optimize(
                seq,
                end,
                seed=int_seed + 700 + rank,
                tau_start=tau_s,
                tau_end=0.02,
                lr_start=lr_s,
                lr_end=0.008,
                eval_interval=1.2 if n > 6000 else 0.6,
                patience=1000,
                verbose=True,
            )
            print(f"[long] n={n} long={res_score:.6f} best={best_score:.6f}")
            if np.isfinite(res_score) and (best_seq is None or res_score < best_score):
                best_seq = res_seq.copy()
                best_score = float(res_score)

    if best_seq is not None and time.monotonic() < deadline - 20.0:
        n = best_seq.size
        res_seq, res_score = _optimize(
            best_seq,
            deadline,
            seed=int_seed + 1000,
            tau_start=0.08,
            tau_end=0.008,
            lr_start=0.02,
            lr_end=0.004,
            eval_interval=1.0 if n > 5000 else 0.4,
            patience=1500,
            verbose=False,
        )
        if np.isfinite(res_score) and (best_seq is None or res_score < best_score):
            best_seq = res_seq.copy()
            best_score = float(res_score)

    if best_seq is None:
        best_seq = _linear_sequence(main_n, 0.5)

    best_seq = np.asarray(best_seq, dtype=np.float64).reshape(-1)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s < 0.01:
        best_seq = np.ones(max(2, best_seq.size), dtype=np.float64)

    final_score = _official_score(best_seq)
    print(f"[final] n={best_seq.size} score={final_score:.6f}")

    return [float(x) for x in best_seq]
