"""
Optimizes non-negative step function to minimize peak autocorrelation ratio.
Uses FFT-accelerated gradient descent with log-sum-exp smoothing and
geometric temperature annealing. Tries multiple lengths and initializations.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    n_fft = 1
    while n_fft < n:
        n_fft <<= 1
    fa = fft(a, n_fft)
    fb = fft(b, n_fft)
    return np.real(ifft(fa * fb))[:n]


def _exact_score(a):
    """Exact score matching evaluate_sequence."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    return float(2.0 * n * float(np.max(conv)) / (s * s))


def _smoothed_grad_and_score(a, T):
    """
    Compute smoothed score and gradient using log-sum-exp.
    Returns (smoothed_score, gradient_vector).
    Assumes sum(a) = 1.
    """
    n = len(a)
    conv = _fft_convolve(a, a)
    conv = np.maximum(conv, 0.0)
    
    c_scaled = conv / T
    c_max = np.max(c_scaled)
    e = np.exp(c_scaled - c_max)
    p = e / e.sum()
    smooth_max = T * (c_max + np.log(e.sum()))
    
    # Gradient of smooth_max w.r.t. a
    p_r = p[::-1]
    grad_conv = _fft_convolve(p_r, a)
    grad = 2.0 * grad_conv[n - 1 : 2 * n - 1][::-1]
    
    return smooth_max, grad

"""
Optimizes non-negative step function heights to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated smoothed gradient descent (log-sum-exp) with geometric
temperature annealing and momentum. Tries multiple lengths and initializations
(ramp, power-ramp, sine, cosine-bell) to avoid local minima.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def _fft_convolve(a, b):
    """Full linear convolution via FFT with zero-padding to next power of 2."""
    n = len(a) + len(b) - 1
    n_fft = 1
    while n_fft < n:
        n_fft <<= 1
    fa = fft(a, n_fft)
    fb = fft(b, n_fft)
    return np.real(ifft(fa * fb))[:n]


def _exact_score(a):
    """Compute exact score matching evaluate_sequence formula."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return float(2.0 * n * max_b / (s * s))


def _smooth_grad(a, T):
    """
    Compute (smoothed_max_value, gradient) of T*log(sum exp(conv/T)) wrt a.
    Assumes len(a) = n; conv has length 2n-1.
    Gradient is computed via FFT: grad = 2 * (reverse(softmax(conv/T)) * a) [sliced].
    """
    n = len(a)
    conv = _fft_convolve(a, a)
    conv = np.maximum(conv, 0.0)

    c_scaled = conv / T
    c_max = np.max(c_scaled)
    e = np.exp(c_scaled - c_max)
    e_sum = float(e.sum())
    p = e / e_sum
    smooth_max = T * (c_max + np.log(e_sum))

    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (valid k-j in [0,n-1])
    # = 2 * (p_reversed * a)[2n-2-j]
    p_r = p[::-1]
    gc = _fft_convolve(p_r, a)
    grad = 2.0 * gc[n - 1:2 * n - 1][::-1]

    return smooth_max, grad


def _make_init(n, kind):
    """Generate initial non-negative sequence (unnormalized)."""
    if kind == 'ramp':
        return np.arange(1, n + 1, dtype=np.float64)
    elif kind == 'quad':
        return np.arange(1, n + 1, dtype=np.float64) ** 1.5
    elif kind == 'sqrt':
        return np.arange(1, n + 1, dtype=np.float64) ** 0.5
    elif kind == 'sine':
        t = np.arange(n, dtype=np.float64)
        return np.sin(np.pi * (t + 0.5) / n) ** 2
    elif kind == 'cos_bell':
        t = np.arange(n, dtype=np.float64) / n
        return 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif kind == 'asym':
        t = np.arange(n, dtype=np.float64) / n
        return t ** 2.0 + 0.001
    elif kind == 'cubic':
        t = np.arange(n, dtype=np.float64) / n
        return t ** 3.0 + 0.001
    else:
        return np.ones(n, dtype=np.float64)


def _optimize(a0, n_iter, deadline, lr_base=0.04, beta=0.92, T_start_f=8.0, T_end_f=0.003):
    """
    Run smoothed gradient descent from a0 (normalized to sum=1).
    Uses geometric T annealing and momentum.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = a0.copy()
    v = np.zeros(n, dtype=np.float64)

    T_start = T_start_f / n
    T_end = T_end_f / n

    best_a = a.copy()
    best_score = _exact_score(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, n_iter - 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_base * (1.0 - 0.6 * frac)

        _, grad = _smooth_grad(a, T)

        # Momentum update
        v *= beta
        v += (1.0 - beta) * grad

        # Gradient step
        a -= lr * v
        # Clipping and renormalization
        np.maximum(a, 0.0, out=a)
        s = a.sum()
        if s < 1e-15:
            a[:] = 1.0 / n
        else:
            a /= s

        # Periodic evaluation
        if it % 300 == 0 or it == n_iter - 1:
            sc = _exact_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score
