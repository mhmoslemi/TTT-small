from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp(f, M):
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[i] <= M for i = 0, ..., 2n-2
    # (f*g)[i] = sum_j f[j] * g[i-j] for valid j
    # => sum_j A[i,j] * g[j] <= M
    # where A[i,j] = f[j] if max(0, i-n+1) <= j <= min(n-1, i), else 0
    
    # Build sparse matrix A
    rows = []
    cols = []
    vals = []
    for i in range(2*n-1):
        j_start = max(0, i - n + 1)
        j_end = min(n-1, i)
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            vals.append(f[j])  # Note: A[i,j] = f[j]
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = M * np.ones(2*n-1)
    
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x
```

from scipy.optimize import linprog
from scipy.sparse import csr_matrix, eye
import time

def solve_lp_step(f, M):
    """Solve the LP: max sum(g) s.t. f*g <= M (element-wise), g >= 0."""
    n = len(f)
    # Build constraint matrix A where A[i,j] = f[j] if max(0,i-n+1) <= j <= min(n-1,i)
    # This gives (f*g)[i] = sum_j A[i,j]*g[j]
    
    # Build sparse matrix efficiently
    rows = []
    cols = []
    vals = []
    
    for i in range(2*n - 1):
        j_start = max(0, i - n + 1)
        j_end = min(n - 1, i)
        length = j_end - j_start + 1
        rows.extend([i] * length)
        cols.extend(range(j_start, j_end + 1))
        vals.extend(f[j_start:j_end+1])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x
```
