    S=p.sum()
    if S<=0: return p,np.inf
    p/=S
    n=len(p)
    if n > 2048 or budget_s < 60: return p, score_quick(p)
    deadline = time.monotonic()+budget_s-10
    N=next_fast_len(2*n-1)

    for it in range(200):
       if time.monotonic()>deadline: break
       g=lp_direction(p,M)
       if g is None: break
       G=float(np.sum(g))
       if not np.isfinite(G) or G<=1e-12: break
       g = g / G
       improved=False
       for tval in (0.1,0.2,0.3,0.05,0.5,0.15,0.02,0.08):
          p_new=(1.0-tval)*p + tval*g
          p_new=np.clip(p_new,0.0,1000.0)

          break

def make_param(n, s, b, g, left_scale):

"""
Search for a non-negative sequence minimizing
    2 * n * max(conv(a, a)) / sum(a)^2.

The baseline one-sided square-root geometric sequence is excellent:
its convolution is nearly flat on the first half, giving the asymptotic
constant pi/2 ~= 1.5708.  The optimizer starts from parameterized
variants of this sequence and refines the full sequence with a
smoothed log-sum-exp / Adam ascent on the negative gradient.  A small
LP refinement step is included for short sequences, but the main work
comes from the FFT-based Adam routine.
"""

import numpy as np
import time

try:
    from scipy.fft import rfft, irfft, next_fast_len
except Exception:
    from scipy.fftpack import rfft, irfft

    def next_fast_len(x):
        y = 1
        limit = int(max(1, x))
        while y < limit:
            y *= 2
        return y

try:
    from scipy.optimize import linprog
    from scipy.sparse import toeplitz, csr_matrix
    _HAS_LP = True
except Exception:
    linprog = None
    toeplitz = None
    csr_matrix = None
    _HAS_LP = False


def fft_convolve_N(a, N):
    """Linear convolution a * a using a pre-specified FFT length."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    A = rfft(a, N)
    C = irfft(A * A, N)
    C = C[:2 * n - 1]
    C[C < 0.0] = 0.0
    return C


def score_quick_N(a, N):
    """Exact objective value using FFT convolution."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    if n == 0:
        return np.inf
    S = float(np.sum(a))
    if not np.isfinite(S) or S <= 0.01:
        return np.inf
    q = fft_convolve_N(a, N)
    return float(2.0 * n * float(np.max(q)) / (S * S))


def make_param(n, s, b, g, left_scale):
    """
    Build a parameterized non-negative normalized sequence.

    p_i ~ (i+1)^(-s) * (1 - i/(n-1))^b * exp(-g*i/(n-1)),
    with the left endpoint independently scaled.
    """
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)

    x = np.arange(n, dtype=float)
    if n == 1:
        t = np.zeros(1, dtype=float)
    else:
        t = x / float(n - 1)

    raw = np.power(x + 1.0, -float(s))

    if b != 0.0:
        base = 1.0 - t
        base = np.clip(base, 0.0, 1.0)
        raw = raw * np.power(base, float(b))

    if g != 0.0:
        raw = raw * np.exp(-float(g) * t)

    if left_scale != 1.0:
        raw[0] = raw[0] * float(left_scale)

    raw = np.clip(raw, 0.0, 1000.0)
    S = float(np.sum(raw))
    if not np.isfinite(S) or S <= 0.0:
        return np.ones(n, dtype=float) / float(n)
    return raw / S


def scan_params(n, deadline):
    """Quick parameter scan over a family near the optimum."""
    N = next_fast_len(2 * n - 1)

    # Strong default: the square-root geometric sequence.
    best_p = make_param(n, 0.5, 0.0, 0.0, 1.0)
    best_score = score_quick_N(best_p, N)
    best_params = (0.5, 0.0, 0.0, 1.0)

    # Coarse scan.
    s_vals = (0.48, 0.50, 0.52, 0.55, 0.60)
    b_vals = (0.0, 0.1, 0.25, 0.5, 0.8, 1.2)
    g_vals = (0.0, 0.2, 0.5, 1.0)
    l_vals = (0.25, 0.5, 0.75, 1.0)

    for s in s_vals:
        for b in b_vals:
            for g in g_vals:
                for l in l_vals:
                    if time.monotonic() > deadline:
                        return best_p, best_score, best_params
                    p = make_param(n, float(s), float(b), float(g), float(l))
                    sc = score_quick_N(p, N)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_p = p.copy()
                        best_params = (float(s), float(b), float(g), float(l))

    # Local refinement around the best coarse point.
    sb, bb, gb, lb = best_params

    s_ref = np.linspace(max(0.0, sb - 0.03), min(0.7, sb + 0.03), 7)
    b_ref = np.linspace(max(0.0, bb - 0.2), min(1.5, bb + 0.2), 7)
    g_ref = np.linspace(max(0.0, gb - 0.25), min(1.5, gb + 0.25), 5)
    l_ref = np.linspace(0.1, 1.0, 5)

    for s in s_ref:
        for b in b_ref:
            for g in g_ref:
                for l in l_ref:
                    if time.monotonic() > deadline:
                        return best_p, best_score, best_params
                    p = make_param(n, float(s), float(b), float(g), float(l))
                    sc = score_quick_N(p, N)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_p = p.copy()
                        best_params = (float(s), float(b), float(g), float(l))

    return best_p, best_score, best_params


def softmax(z):
    """Stable softmax."""
    z = np.asarray(z, dtype=float)
    if z.size == 0:
        return np.array([], dtype=float)
    z = z - float(np.max(z))
    p = np.exp(z)
    S = float(np.sum(p))
    if not np.isfinite(S) or S <= 0.0:
        return np.ones(z.size, dtype=float) / float(z.size)
    return p / S


def gradient_from_q(p, q, tau, n, Ng, idx):
    """
    Gradient of LSE(q / tau) * tau with respect to softmax logits z.

    q_k = sum_i p_i p_{k-i}, so d q_k / d p_j = 2 p_{k-j}
    when k-j is in range.
    """
    qmax = float(np.max(q))
    if tau <= 0.0 or not np.isfinite(tau):
        w = np.zeros(len(q), dtype=float)
        w[int(np.argmax(q))] = 1.0
    else:
        w = np.exp((q - qmax) / tau)
        sw = float(np.sum(w))
        if not np.isfinite(sw) or sw <= 0.0:
            w = np.zeros(len(q), dtype=float)
            w[int(np.argmax(q))] = 1.0
        else:
            w = w / sw

    wrev = w[::-1]
    P = rfft(p, Ng)
    W = rfft(wrev, Ng)
    h = irfft(P * W, Ng)

    # C_j = sum_i p_i w_{j+i}; this is a Hankel product.
    C = h[2 * n - 2 - idx]
    grad_p = 2.0 * C

    dot = float(np.dot(grad_p, p))
    grad_z = grad_p - p * dot

    if not np.all(np.isfinite(grad_z)):
        return np.zeros(n, dtype=float)
    return grad_z


def adam_refine(p, z, deadline, rng, best_score):
    """
    Smoothed log-sum-exp minimisation under the simplex constraint
    using softmax parameterization and Adam.
    """
    n = len(p)
    Nq = next_fast_len(2 * n - 1)
    Ng = next_fast_len(3 * n - 2)
    idx = np.arange(n)

    p = np.asarray(p, dtype=float).copy()
    z = np.asarray(z, dtype=float).copy()

    best_p = p.copy()
    best_z = z.copy()
    best_sc = float(best_score)

    m = np.zeros(n, dtype=float)
    v = np.zeros(n, dtype=float)
    b1pow = 1.0
    b2pow = 1.0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    stall = 0
    stall_limit = 3500
    last_print_sc = best_sc
    start_adam = time.monotonic()
    total = max(1.0, float(deadline - start_adam))

    while time.monotonic() < deadline:
        frac = min(1.0, max(0.0, (time.monotonic() - start_adam) / total))
        if frac >= 1.0:
            break

        tau_ratio = 0.25 + 19.75 * (1.0 - frac) ** 2
        lr = 0.008 + 0.135 * (1.0 - frac) ** 0.5

        p = softmax(z)
        S = float(np.sum(p))
        if not np.isfinite(S) or S <= 0.01:
            p = np.ones(n, dtype=float) / float(n)
            z = np.zeros(n, dtype=float)
            S = 1.0

        q = fft_convolve_N(p, Nq)
        qmax = float(np.max(q))
        if not np.isfinite(qmax) or qmax <= 0.0:
            p = np.ones(n, dtype=float) / float(n)
            z = np.zeros(n, dtype=float)
            continue

        score = float(2.0 * n * qmax / (S * S))
        if np.isfinite(score) and score < best_sc:
            best_sc = score
            best_p = p.copy()
            best_z = z.copy()
            stall = 0
            if best_sc < last_print_sc - 1e-5:
                last_print_sc = best_sc
                print("Adam improve: %.10f" % best_sc)
        else:
            stall += 1
            if stall > stall_limit:
                # Multiplicative noise around the best point.
                base = np.maximum(best_p, 1e-300)
                noise = np.exp(rng.normal(0.0, 0.08, size=n))
                p_reset = base * noise
                S_reset = float(np.sum(p_reset))
                if S_reset > 0.0 and np.isfinite(S_reset):
                    z = np.log(p_reset / S_reset + 1e-300)
                else:
                    z = best_z.copy()
                z = z - float(np.mean(z))
                m.fill(0.0)
                v.fill(0.0)
                b1pow = 1.0
                b2pow = 1.0
                stall = 0
                continue

        tau = qmax * tau_ratio
        grad = gradient_from_q(p, q, tau, n, Ng, idx)

        if not np.any(grad):
            continue

        b1pow *= beta1
        b2pow *= beta2

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        mhat = m / (1.0 - b1pow)
        vhat = v / (1.0 - b2pow)

        z = z - lr * mhat / (np.sqrt(vhat) + eps)
        z = z - float(np.mean(z))

        span = float(np.max(z) - np.min(z))
        if not np.isfinite(span):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            b1pow = 1.0
            b2pow = 1.0
        elif span > 180.0:
            z = (z - float(np.mean(z))) * (180.0 / span)

    p = softmax(z)
    sc = score_quick_N(p, Nq)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_p = p.copy()

    return best_p, best_sc


def lp_direction(a, M):
    """
    Solve the LP from the stated SoTA idea:
      maximize sum(g)
      subject to (a * g)[k] <= M for all k, g >= 0.
    The a sequence is scaled by n for numerical conditioning.
    """
    if not _HAS_LP:
        return None

    n = len(a)
    scale = float(n)

    first_col = np.zeros(2 * n - 1, dtype=float)
    first_col[:n] = np.asarray(a, dtype=float) * scale
    first_row = np.zeros(n, dtype=float)
    first_row[0] = first_col[0]

    try:
        A = toeplitz(first_col, first_row).tocsr()
    except Exception:
        return None

    c_obj = -np.ones(n, dtype=float)
    b_ub = np.full(2 * n - 1, float(M) * scale, dtype=float)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs")
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None))
        except Exception:
            return None

    if res is not None and getattr(res, "success", False):
        x = np.asarray(res.x, dtype=float)
        if x.size == n and np.all(np.isfinite(x)):
            return np.clip(x, 0.0, 1000.0)
    return None


def lp_refine(p, budget_s):

    and when useful time remains.
    """
    if not _HAS_LP:
        return p, score_quick_N(p, next_fast_len(2 * len(p) - 1))

    p = np.asarray(p, dtype=float).copy()
    n = len(p)
    S = float(np.sum(p))
    if not np.isfinite(S) or S <= 0.01:
        p = np.ones(n, dtype=float) / float(n)
        S = 1.0
    p = p / S

    if n > 2048 or budget_s < 90.0:
        Nq = next_fast_len(2 * n - 1)
        return p, score_quick_N(p, Nq)

    Nq = next_fast_len(2 * n - 1)
    deadline = time.monotonic() + max(0.0, budget_s) - 10.0

    best_p = p.copy()
    best_sc = score_quick_N(p, Nq)

    for _ in range(500):
        if time.monotonic() > deadline:
            break

        q = fft_convolve_N(p, Nq)
        M = float(np.max(q))
        if not np.isfinite(M) or M <= 0.0:
            break

        g = lp_direction(p, M)
        if g is None:
            break

        G = float(np.sum(g))
        if not np.isfinite(G) or G <= 1e-12:
            break

        g = g / G
        improved = False

        for tval in (0.25, 0.1, 0.5, 0.05, 0.15, 0.35, 0.02, 0.08):
            p_new = (1.0 - tval) * p + tval * g
            p_new = np.clip(p_new, 0.0, 1000.0)
            S_new = float(np.sum(p_new))
            if not np.isfinite(S_new) or S_new <= 0.01:
                continue
            p_new = p_new / S_new
            sc = score_quick_N(p_new, Nq)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_p = p_new.copy()
                p = p_new.copy()
                improved = True
                break

        if not improved:
            break

        if best_sc < 1.50:
            break

    return best_p, best_sc


def direct_score(a):
    """Exact score using direct convolution, used only near the end."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    if n == 0:
        return np.inf
    S = float(np.sum(a))
    if not np.isfinite(S) or S <= 0.01:
        return np.inf
    q = np.convolve(a, a, mode="full")
    q = np.clip(q, 0.0, None)
    return float(2.0 * n * float(np.max(q)) / (S * S))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point.  Chooses a useful length, runs a fast parameter scan,
    then refines with FFT-based Adam.  Returns the best non-negative list.
    """
    start = time.monotonic()
    budget_s = max(60.0, float(budget_s))
    deadline = start + (budget_s - 30.0)

    rng = np.random.default_rng(int(seed))

    # Prefer the supplied construction length when it is reasonable.
    n = 4096
    try:
        L = len(height_sequence_1)
        if isinstance(L, int) and 256 <= L <= 20000:
            if budget_s > 900.0:
                n = L
            else:
                n = min(L, 2048)
    except Exception:
        pass

    if budget_s < 600.0:
        n = min(n, 1024)
    n = int(max(512, min(n, 12000)))

    Nq = next_fast_len(2 * n - 1)

    # Very strong fallback: the one-sided sqrt-geometric sequence.
    fallback_p = make_param(n, 0.5, 0.0, 0.0, 1.0)
    fallback_sc = score_quick_N(fallback_p, Nq)

    # Phase 1: cheap structured scan.
    scan_deadline = min(deadline, start + min(360.0, 0.25 * budget_s))
    best_p, best_score, best_params = scan_params(n, scan_deadline)
    print("n =", n)
    print("scan best =", best_score)
    print("fallback sqrt-geom =", fallback_sc)

    if not np.isfinite(best_score) or fallback_sc < best_score:
        best_p = fallback_p.copy()
        best_score = fallback_sc

    # Ensure positivity for logarithms.
    p_safe = np.asarray(best_p, dtype=float).copy()
    p_safe = np.clip(p_safe, 1e-300, 1000.0)
    p_safe = p_safe / float(np.sum(p_safe))
    z = np.log(p_safe)
    z = z - float(np.mean(z))
    span = float(np.max(z) - np.min(z))
    if span > 180.0:
        z = z * (180.0 / span)

    # Phase 2: full refinement.
    if time.monotonic() < deadline - 60.0:
        refined_p, refined_score = adam_refine(
            p_safe, z, deadline, rng, float(best_score)
        )
        if np.isfinite(refined_score) and refined_score < best_score:
            best_p = refined_p
            best_score = refined_score
            print("Adam best =", best_score)

    # Phase 3: optional LP polish for shorter sequences.
    if (
        _HAS_LP
        and n <= 2048
        and best_score > 1.52
        and time.monotonic() < deadline - 120.0
    ):
        lp_budget = min(420.0, max(60.0, deadline - time.monotonic() - 30.0))
        lp_p, lp_score = lp_refine(best_p, lp_budget)
        if np.isfinite(lp_score) and lp_score < best_score:
            best_p = lp_p
            best_score = lp_score
            print("LP best =", best_score)

    # Final polishing of the finite-precision sequence.
    best_p = np.asarray(best_p, dtype=float)
    best_p = np.clip(best_p, 0.0, 1000.0)
    S = float(np.sum(best_p))
    if not np.isfinite(S) or S <= 0.01:
        best_p = np.ones(n, dtype=float)
        S = float(n)
    else:
        best_p = best_p / S

    # Verify with direct convolution if there is a little margin and the
    # sequence is not enormous.  This is optional but useful for logging.
    if n <= 10000 and time.monotonic() < deadline:
        exact_sc = direct_score(best_p)
        if np.isfinite(exact_sc):
            print("direct finite score =", exact_sc)
        else:
            print("direct finite score invalid")
    else:
        print("skip direct score")

    try:
        evaluator_score = evaluate_sequence([float(x) for x in best_p])
        print("evaluate_sequence score =", evaluator_score)
    except Exception:
        pass

    return [float(x) for x in best_p]
