**Connection to design theory:**

The problem of minimizing the peak of the autocorrelation is related to the problem of finding "difference sets" in combinatorial design theory. A (v, k, λ) difference set is a subset D of a group G of order v such that every non-identity element can be written as a difference of two elements of D in exactly λ ways.

"""
Algorithm: Multiplicative weight updates to flatten the autocorrelation.
We normalize the sequence to sum=1, then iteratively reduce the peak of
the self-convolution using a pressure-based multiplicative update.
Multiple starting points and local refinement are used.
"""

import numpy as np
import time

def _fft_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    N = 1 << (2*n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2*n-1]

def _score(p):
    """Score for probability vector p (sum=1)."""
    return 2.0 * len(p) * _fft_conv(p).max()

def _multiplicative_update(p, c, eta):
    """One step of the multiplicative update."""
    n = len(p)
    mean_c = c.mean()
    e = np.maximum(c - mean_c, 0.0)
    if e.max() < 1e-15:
        return p
    # force[j] = sum_k e[k] * p[k-j] = sliding dot product
    force = np.correlate(e, p, mode='valid')
    force = np.maximum(force, 0.0)
    mf = force.mean()
    if mf < 1e-15:
        return p
    # Multiplicative update
    p_new = p * np.exp(-eta * force / mf)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-15:
        return p
    return p_new / s

def _optimize(p0, max_iters=2000, eta_init=0.4, verbose=False):
    """Run multiplicative update optimization."""
    n = len(p0)
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s
    best_p = p.copy()
    best_score = _score(p)
    eta = eta_init
    patience = 0
    max_patience = 300
    rng = np.random.default_rng(123)
    
    for it in range(max_iters):
        c = _fft_conv(p)
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            patience = 0
            eta = min(eta * 1.01, 1.0)
        else:
            patience += 1
            if patience >= max_patience:
                eta *= 0.5
                if eta < 0.01:
                    break
                # Add noise
                p = np.maximum(p + rng.normal(0, eta*0.05/n, n), 0)
                s = p.sum()
                if s < 1e-15: break
                p = p / s
                patience = 0
        
        p = _multiplicative_update(p, c, eta)
    
    return best_p, best_score

def _local_refine(p, time_budget=60.0, verbose=False):
    """Local refinement with small random perturbations."""
    n = len(p)
    best_p = p.copy()
    best_score = _score(p)
    rng = np.random.default_rng(456)
    deadline = time.monotonic() + time_budget
    
    while time.monotonic() < deadline:
        # Random perturbation
        idx = rng.integers(0, n)
        width = max(1, rng.integers(1, n//50))
        start = min(idx, n - width)
        scale = np.exp(rng.normal(0, 0.1))
        candidate = best_p.copy()
        candidate[start:start+width] *= scale
        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 1e-15: continue
        candidate = candidate / s
        sc = _score(candidate)
        if sc < best_score:
            best_score = sc
            best_p = candidate
            if verbose:
                print(f"    Refine: {best_score:.8f}")
    
    return best_p, best_score
