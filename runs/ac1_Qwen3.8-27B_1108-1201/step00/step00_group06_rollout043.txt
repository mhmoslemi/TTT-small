**Frank-Wolfe (conditional gradient) approach:**

We want to minimize F(f) = max_k (f*f)(k) subject to f >= 0, sum(f) = S.

The Frank-Wolfe algorithm:
1. Start with f_0.
2. At each step, find g = argmin_{h in feasible set} <∇F(f), h - f>. (This is the "linear oracle" step.)
3. Update f <- (1-t) f + t g.

∇F(f) is the subgradient of max_k (f*f)(k). If k* is where the max is achieved, then a subgradient is ∂(f*f)(k*)/∂f = the vector with (∂(f*f)(k*)/∂f[i]) = f[k*-i] (for valid indices).

- If f[k*-i] > 0 for some i, we want h[i] = 0 for those i.
- We want to put all mass S on the index i where f[k*-i] is minimized.

"""
Minimizes 2n*max(f*f)/(sum(f))^2 using smoothed gradient descent
with multiplicative updates and LP-based refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import fftconvolve

def safe_fft_convolve(a, b):
    """Compute linear convolution using FFT with appropriate zero-padding."""
    n = len(a) + len(b) - 1
    # Use next power of 2 for efficiency
    fft_size = 1
    while fft_size < n:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(fa * fb, fft_size)
    return result[:n]

def softmax_stable(x, T):
    """Compute softmax(x/T) with numerical stability."""
    x_scaled = x / max(T, 1e-10)
    x_scaled -= x_scaled.max()
    e = np.exp(x_scaled)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s

def compute_gradient(f, w, n, fft_size):
    """Compute the smoothed subgradient s[j] = 2 * sum_i f[i] * w[i+j]."""
    # s[j] = 2 * (flip(f) * w)[n-1+j] for j = 0, ..., n-1
    g = f[::-1]
    conv = safe_fft_convolve(g, w)
    # conv has length n + (2n-1) - 1 = 3n-1
    s = 2.0 * conv[n-1 : 2*n-1]
    return s

def optimize_with_gradient(f_init, n, seed, time_budget, fft_size):
    """Smoothed gradient descent with multiplicative updates."""
    rng = np.random.default_rng(seed)
    f = np.array(f_init, dtype=np.float64).copy()
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    deadline = time.monotonic() + time_budget
    T = 1.0  # Temperature
    lr = 0.1  # Learning rate
    best_local = f.copy()
    best_local_score = np.inf
    last_eval = 0
    
    while time.monotonic() < deadline:
        # Compute b = f * f
        b = safe_fft_convolve(f, f)
        b = b[:2*n-1]  # Ensure correct length
        
        # Compute weights
        w = softmax_stable(b, T)
        
        # Compute gradient
        grad = compute_gradient(f, w, n, fft_size)
        
        # Multiplicative update
        update = np.exp(-lr * grad)
        f = f * update
        f = np.clip(f, 1e-300, None)
        f = f / f.sum()
        
        # Gradually increase temperature
        T *= 1.0005
        
        # Occasionally evaluate and add perturbation
        if time.monotonic() - last_eval > 5.0:
            score = evaluate_sequence(f.tolist())
            if score < best_local_score:
                best_local_score = score
                best_local = f.copy()
            last_eval = time.monotonic()
            
            # Random perturbation
            if rng.random() < 0.3:
                perturbation = rng.standard_normal(n) * 0.05
                f_new = f + perturbation
                f_new = np.clip(f_new, 0, None)
                if f_new.sum() > 1e-10:
                    f_new = f_new / f_new.sum()
                    f = f_new
    
    return best_local

def lp_refine(f, n, time_budget):
    """LP-based refinement using constraint generation."""
    from scipy.optimize import linprog
    
    f = np.array(f, dtype=np.float64).copy()
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-10:
        return f
    f = f / s  # Normalize to sum = 1
    
    deadline = time.monotonic() + time_budget
    active_set = set()
    
    for iteration in range(200):
        if time.monotonic() > deadline:
            break
        
        # Compute current b and max
        b = safe_fft_convolve(f, f)[:2*n-1]
        M = b.max()
        k_star = int(b.argmax())
        
        # Add new constraint if not already present
        if k_star not in active_set:

Minimizes 2n*max(f*f)/(sum(f))^2 via smoothed multiplicative gradient descent

"""

import numpy as np

def safe_fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:n]

def softmax_stable(x, T):
    """Numerically stable softmax with temperature."""
    x = x / max(T, 1e-12)
    x -= x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s

def grad_convoluted(f, w, n):
    """Compute s[j] = 2*sum_i f[i]*w[i+j] via FFT."""
    g = f[::-1]
    conv = safe_fft_convolve(g, w)
    return 2.0 * conv[n-1 : 2*n-1]

"""
Minimizes 2n*max(f*f)/(sum(f))^2 using smoothed multiplicative gradient descent
with adaptive temperature, multiple restarts, and LP-based constraint generation refinement.
"""

import numpy as np
import time


def safe_fft_convolve(a, b):
    """Compute linear convolution using FFT with power-of-2 padding."""
    n_out = len(a) + len(b) - 1
    fft_size = 1 << (n_out - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:n_out]


def softmax_stable(x, T):
    """Numerically stable softmax with temperature parameter."""
    x_scaled = x / max(T, 1e-12)
    x_scaled -= x_scaled.max()
    e = np.exp(x_scaled)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def grad_via_fft(f, w, n):
    """Compute smoothed subgradient s[j] = 2*sum_i f[i]*w[i+j] using FFT."""
    g = f[::-1]
    conv = safe_fft_convolve(g, w)
    return 2.0 * conv[n - 1:2 * n - 1]


def run_gradient_opt(f0, n, seed, time_budget):
    """Smoothed multiplicative gradient descent with periodic restart."""
    rng = np.random.default_rng(seed)
    f = np.clip(np.array(f0, dtype=np.float64), 0, None).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    deadline = time.monotonic() + time_budget
    T = 0.3
    best_f = f.copy()
    best_M = safe_fft_convolve(f, f).max()
    iters_since_improve = 0
    max_stall = 8000
    eval_interval = 300

    while time.monotonic() < deadline:
        b = safe_fft_convolve(f, f)
        w = softmax_stable(b, T)
        grad = grad_via_fft(f, w, n)

        lr = 1.0 / (grad.max() + 1e-12)
        f = f * np.exp(-lr * grad)
        f = np.clip(f, 1e-300, None)
        f = f / f.sum()

        T *= 1.0004
        if T > 50.0:
            T = 50.0

        iters_since_improve += 1
        if iters_since_improve % eval_interval == 0:
            curr_M = safe_fft_convolve(f, f).max()
            if curr_M < best_M * (1 - 1e-6):
                best_M = curr_M
                best_f = f.copy()
                iters_since_improve = 0
            elif iters_since_improve > max_stall:
                # Restart from best with perturbation
                restart_scale = 0.05 + 0.1 * rng.random()
                f = best_f + restart_scale * rng.standard_normal(n)
                f = np.clip(f, 0, None)
                fs = f.sum()
                if fs < 1e-10:
                    f = best_f.copy()
                else:
                    f = f / fs
                T = 0.3
                iters_since_improve = 0

    return best_f


def lp_refine(f, n, time_budget, max_constraints=25):
    """Constraint-generation LP refinement."""
    from scipy.optimize import linprog

    f = np.clip(np.array(f, dtype=np.float64), 0, None).copy()
    f = f / f.sum()
    deadline = time.monotonic() + time_budget
    active = []

    for iteration in range(80):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        b = safe_fft_convolve(f, f)
        M = b.max()
        ks = int(b.argmax())
        if ks not in active:
            active.append(ks)
        if len(active) > max_constraints:
            active = active[-max_constraints:]

        A_rows = []
        for k in active:
            row = np.zeros(n)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                row[i] = f[k - i]
            A_rows.append(row)
        A = np.array(A_rows)

        c = -np.ones(n)
        try:
            res = linprog(c, A_ub=A, b_ub=np.full(len(active), M),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': max(1.0, min(remaining / 2, 15.0))})
        except Exception:
            break

        if res.status != 0 or res.x is None:
            if len(active) > 5:
                active = active[:len(active) // 2]
            continue

        g = np.clip(res.x, 0, None)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        improved = False
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            f_try = (1.0 - t) * f + t * g
            f_try = np.clip(f_try, 0, None)
            fs = f_try.sum()
            if fs < 1e-10:
                continue
            f_try = f_try / fs
            new_M = safe_fft_convolve(f_try, f_try).max()
            if new_M < M * (1 - 1e-8):
                f = f_try
                M = new_M
                improved = True
                break

        if not improved:
            if len(active) >= max_constraints:
                break
            # Remove least-violated constraint
            violations = [b[k] / M for k in active]
            min_idx = int(np.argmin(violations))
            active.pop(min_idx)
