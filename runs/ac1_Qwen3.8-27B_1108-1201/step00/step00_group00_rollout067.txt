"""
LP-based iterative optimization to minimize the autocorrelation peak of a
non-negative step function.

Objective (scale invariant): score = 2 * n * max(f*f) / (sum(f))**2.
The uniform sequence gives exactly 2.0; we want to push it below that.

Method:
  * Keep f normalized to sum = 1 (the objective is scale invariant).
  * At each step compute M = max(f*f) and solve the linear program
        max  sum(g)
        s.t. (f*g)[k] <= M   for all k,   g >= 0
    where (f*g) denotes the full linear convolution. The constraint matrix is
    a known banded sparsity pattern A[k,j] = f[k-j].
  * Normalise the LP solution g to sum 1 and relax f <- (1-t)*f + t*g.
  * Interleave fast block-perturbation local search and use several starting
    shapes / lengths to escape degeneracy and local minima.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Return 2*n*max(conv)/sum^2 faithfully to evaluate_sequence, or inf."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) with A[k,j] = f[k-j] for 0 <= k-j < n else 0."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp_for_g(f, M, time_limit_s=10.0):
    """Solve max sum(g) s.t. A g <= M*1, g >= 0. Returns g or None."""
    n = int(len(f))
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit_s, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_one_step(f, t, time_limit_s=10.0):
    """One LP-driven update. Returns updated normalised f or original f."""
    conv = np.convolve(f, f)
    M = float(conv.max())
    if M < 1e-15:
        return f
    g = solve_lp_for_g(f, M, time_limit_s)
    if g is None:
        return f
    gs = g.sum()
    if gs < 1e-10:
        return f
    g = g / gs
    # If LP gave essentially the same point, flag no progress.
    if np.linalg.norm(g - f) < 1e-8 * max(1.0, f.sum()):
        return f
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s


def make_initial(n, rng, method='perturbed_uniform'):
    """Non-negative, sum-normalised starting vector of length n."""
    if method == 'perturbed_uniform':
        a = np.ones(n) + rng.normal(0.0, 0.5, n)
    elif method == 'cosine':
        x = np.linspace(0.0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
    elif method == 'symmetric_cosine':
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n)
    elif method == 'two_peaks':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 16)
        a = (np.exp(-0.5 * ((x - n // 4) / s) ** 2) +
             np.exp(-0.5 * ((x - 3 * n // 4) / s) ** 2))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def block_perturb(f, rng, low=2, high=40):
    """Multiplicative random block perturbation, renormalised to sum 1."""
    n = len(f)
    cand = f.copy()
    w = int(rng.integers(1, max(2, min(high, n))))
    if n - w < 1:
        w = max(1, n // 10)
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.25)))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s
