    g1 = np.exp(-0.5 * ((x - 0.25) / 0.08) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75) / 0.08) ** 2)
    two = g1 + g2
    inits.append(two / two.sum())

    # Weighted flat (slight center emphasis)
    w_flat = 1.0 + 0.3 * np.cos(np.pi * x)
    inits.append(w_flat / w_flat.sum())

    # Time allocation
    time_available = max(10.0, deadline - time.monotonic())
    time_per = time_available / len(inits)
    print(f"{len(inits)} inits, ~{time_per:.1f}s each", flush=True)

    for i, a0 in enumerate(inits):
        if time.monotonic() >= deadline:
            break
        t_end = min(deadline, time.monotonic() + time_per + 1.0)
        best_a, best_max = run_mirror(a0, t_end)

        score = float(evaluate_sequence(best_a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = best_a.copy()
            print(f"  Init {i}: NEW BEST score={best_score:.6f}", flush=True)
        else:
            print(f"  Init {i}: score={score:.6f}", flush=True)

    # Extended run from best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"Extended run: {remaining:.0f}s remaining", flush=True)
        s = best_seq.sum()
        if s > 0.01:
            a_start = best_seq / s
        else:
            a_start = np.ones(n) / n
        best_a, best_max = run_mirror(a_start, deadline)
        score = float(evaluate_sequence(best_a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = best_a.copy()
            print(f"  Extended: NEW BEST score={best_score:.6f}", flush=True)
        else:
            print(f"  Extended: score={score:.6f}", flush=True)

    final = float(evaluate_sequence(best_seq.tolist()))
    print(f"FINAL: {final:.6f}", flush=True)
    return [float(x) for x in best_seq]
