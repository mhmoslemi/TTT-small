def solve_lp_direction(a, M):
    """Solve the LP to find direction b."""
    n = len(a)
    # Variables: b_0, ..., b_{n-1}
    # Objective: maximize sum(b_j) => minimize -sum(b_j)
    c = -np.ones(n)

    # (a*b)[k] = sum_j a[j] * b[k-j] (with valid indices)
    # For a fixed k: sum_j a[j] * b[k-j] = sum_i a[k-i] * b[i] (where i = k-j)
    # So the constraint is: sum_i A[k,i] * b[i] <= M where A[k,i] = a[k-i] if 0<=k-i<n else 0
    
    # Build the constraint matrix A
    # A[k, i] = a[k-i] if 0 <= k-i < n else 0
    # This is a "convolution matrix"

    # Actually for n=2535, the matrix is 5069 x 2535, which is about 12.8M entries.
    # That's manageable but might be slow to build.

    # Each row k has non-zeros at columns i where 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(n-1, k)
    
    # Let's build it as a sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    # Non-negativity: b >= 0 (handled by bounds in linprog)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return None

"""
Algorithm: Multi-phase optimization for minimum peak autocorrelation.
Phase 1: Gradient descent with subgradient of max-convolution.
Phase 2: LP-based direction finding (Frank-Wolfe style).
Phase 3: Local search with random perturbations.
"""

import numpy as np
import time
import scipy.optimize as opt
import scipy.sparse as sp

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def project_to_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / np.sum(w)

def compute_subgradient(a, tol=0.01):
    """Compute the average subgradient of max(conv(a,a)) at a."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = np.max(c)
    # Active set: indices where c[k] >= max_c - tol * max_c
    active = np.where(c >= max_c * (1 - tol))[0]
    
    # For each active k, the gradient is g_k[i] = 2*a[k-i] (if valid)
    # Average over active set
    g = np.zeros(n)
    for k in active:
        # i ranges from max(0, k-n+1) to min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)  # exclusive
        # g_k[i] = a[k-i] for i in [lo, hi)
        # k-i ranges from k-lo to k-(hi-1)
        # k-lo = k - max(0, k-n+1)
        # k-(hi-1) = k - min(n,k) = max(0, k-n)
        idx = slice(lo, hi)
        rev_idx = slice(k - (hi - 1), k - lo + 1)  # a[k-(hi-1)] to a[k-lo]
        # Actually, let me think about this differently.
        # For i in [lo, hi), k-i ranges from k-lo down to k-(hi-1)
        # So we need a[k-lo], a[k-lo-1], ..., a[k-(hi-1)]
        # That's a[k-(hi-1) : k-lo+1][::-1]
        g[idx] += a[k - np.arange(lo, hi)]  # This creates an array
    
    g /= len(active)
    return g, max_c, c

def gradient_descent_phase(a, max_iters, lr_init=0.01, momentum=0.9):
    """Phase 1: Gradient descent with momentum."""
    n = len(a)
    a = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = compute_objective(a)
    lr = lr_init
    
    for it in range(max_iters):
        g, max_c, c = compute_subgradient(a, tol=0.005)
        
        # Step
        v = momentum * v - lr * g
        a_new = a + v
        
        # Project
        a_new = project_to_simplex(a_new)
        
        # Adaptive lr: reduce if objective increases significantly
        score_new = compute_objective(a_new)
        if score_new > best_score * 1.01:
            lr *= 0.8
            v *= 0.5
        else:
            a = a_new
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                lr = min(lr * 1.05, 0.1)
        
        if it % 10000 == 0:
            print(f"  GD iter {it}: score = {best_score:.6f}, lr = {lr:.6f}")
    
    return best_a, best_score

def compute_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    n = len(a)
    
    # Build constraint matrix A (2n-1 x n)
    # A[k, i] = a[k-i] if 0 <= k-i < n else 0
    # Column i has non-zeros in rows i to i+n-1 with values a[0], a[1], ..., a[n-1]
    
    # Use sparse construction
    # For column i: rows i, i+1, ..., min(i+n-1, 2n-2)
    # Values: a[0], a[1], ..., a[min(n-1, 2n-2-i)]
    
    rows_list = []
    cols_list = []
    vals_list = []
    
    for i in range(n):
        max_row = min(i + n - 1, 2 * n - 2)
        num_vals = max_row - i + 1
        rows_list.extend(range(i, max_row + 1))
        cols_list.extend([i] * num_vals)
        vals_list.extend(a[:num_vals])
    
    A = sp.csr_matrix((vals_list, (rows_list, cols_list)), shape=(2*n-1, n))
    
    # LP: minimize -sum(b) s.t. A @ b <= M, b >= 0
    c_obj = -np.ones(n)
    result = opt.linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), 
                          bounds=[(0, None)]*n, method='highs', 
                          options={'presolve': True})
    
    if result.success:
        return result.x
    else:
        return None

def lp_phase(a, max_iters=200):
    """Phase 2: LP-based direction finding."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = compute_objective(a)
    
    for it in range(max_iters):
        c = np.convolve(a, a)
        M = np.max(c)
        
        b = compute_lp_direction(a, M)
        if b is None:
            break
        
        # Rescale b to have same sum as a
        b_sum = np.sum(b)
        if b_sum < 1e-10:
            break
        b = b * (np.sum(a) / b_sum)
        
        # Try different t values
        best_t = 0

            a_candidate = np.maximum(a_candidate, 0)
            s = np.sum(a_candidate)
            if s < 1e-10:

            a_candidate = a_candidate / s
            score = compute_objective(a_candidate)
            if score < best_score_t:
                best_score_t = score
                best_t = t
                best_candidate = a_candidate
        
        if best_t > 0:
            a = best_candidate
            if best_score_t < best_score:
                best_score = best_score_t
                best_a = a.copy()
        
        if it % 10 == 0:
            print(f"  LP iter {it}: score = {best_score:.6f}")
    
    return best_a, best_score

    n = len(a)
    c = np.convolve(a, a)
    # Softmax weights
    c_max = np.max(c)
    log_w = (c - c_max) / tau
    w = np.exp(log_w)
    w_sum = np.sum(w)
    if w_sum == 0:

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
