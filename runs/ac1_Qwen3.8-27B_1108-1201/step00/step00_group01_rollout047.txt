from scipy.optimize import minimize
from scipy.fft import fft, ifft

def generate_sequence(params, n):
    """Generate a non-negative sequence from parameters."""
    # params[0] = DC offset (mean)
    # params[1:] = Fourier coefficients (amplitudes)
    mean = params[0]
    freqs = np.arange(n)
    seq = np.full(n, mean)
    for idx, m in enumerate(range(1, len(params))):
        # m-th harmonic
        seq += np.cos(2 * np.pi * m * freqs / n) * params[idx + 1]
    return np.maximum(seq, 0.0)
