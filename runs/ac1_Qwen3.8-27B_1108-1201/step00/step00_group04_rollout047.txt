import numpy as np
from scipy.optimize import linprog
import time

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M (elementwise), g >= 0"""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) → minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: for each k = 0, ..., 2n-2: sum_i f_i * g_{k-i} <= M
    # g_j = 0 if j < 0 or j >= n
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, j] = f[i]
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g_i >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix A where A @ g = convolve(f, g)."""
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        # A[k, j] = f[k-j] if 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp_for_g(f, M):
    """Solve: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def evaluate_and_update(f, n, t):

    conv = np.convolve(f, f)
    M = conv.max()
    score = 2.0 * n * M / (np.sum(f))**2

        g = g * target / np.sum(g)
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        # Renormalize to keep sum constant
        if np.sum(f_new) > 0:
            f_new = f_new / np.sum(f_new) * target
        return f_new, score
    return f, score

    k_star = np.argmax(conv)
    M = conv[k_star]

    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = f[j]
    
    # Decrease f in the direction of grad
    # f_new[i] = f[i] - alpha * grad[i]
    # Choose alpha so that max(f_new) = 0 (i.e., the minimum component hits 0)
    
    min_ratio = np.inf
    for i in range(n):
        if grad[i] > 0:
            min_ratio = min(min_ratio, f[i] / grad[i])
    
    alpha = min_ratio * 0.5  # take half step to avoid hitting 0
    f_new = f - alpha * grad
    f_new = np.maximum(f_new, 0)
    f_new = f_new / np.sum(f_new)
    return f_new
