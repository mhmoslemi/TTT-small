def _fft_convolve(a, b):

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)
    return result[:n]

    """Compute gradient of (sum b^p)^(1/p) where b = convolve(a,a)."""
    n = len(a)
    b = _fft_convolve(a, a)

This is what I had before. Good.

Now, the gradient:
dL/da_m = (1/p) * (sum b^p)^(1/p-1) * p * sum_k b_k^(p-1) * 2 * a_{k-m} * [0 <= k-m < n]
= 2 * (sum b^p)^(1/p-1) * sum_{k: 0<=k-m<n} b_k^(p-1) * a_{k-m}

Let j = k-m (so k = j+m, and 0 <= j < n):
= 2 * (sum b^p)^(1/p-1) * sum_{j=0}^{n-1} b_{j+m}^(p-1) * a_j

Define w_k = b_k^(p-1). Then:
= 2 * (sum b^p)^(1/p-1) * sum_{j=0}^{n-1} w_{j+m} * a_j

This is 2 * norm_factor * (cross-correlation of w and a at lag m).

The cross-correlation: c_m = sum_{j=0}^{n-1} w_{j+m} * a_j for m = 0, ..., n-1.

As I computed before, c_m = convolve(a_rev, w)[m + n - 1] where a_rev = a[::-1].

"""
Minimize 2n*max(convolve(a,a))/sum(a)^2 using:
- FFT-based gradient computation with power-norm max approximation
- Adam optimizer with adaptive learning rate
- Multiple restarts with ramp, random, and given-sequence initializations
- Gradually sharpening the max approximation (increasing p)
- Periodic exact evaluation via evaluate_sequence
"""

import numpy as np
import time


def _fft_convolve(a, b):
    """Compute linear convolution using real FFT."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    result = np.fft.irfft(fa * fb, N)
    return result[:n]


def _compute_loss_grad(a, p):
    """
    Compute loss (power-norm approx of max convolution) and its gradient.
    
    Loss = (sum_k b_k^p)^(1/p) where b = full_convolve(a, a)
    Gradient computed analytically via cross-correlation.
    """
    n = len(a)
    # Full self-convolution
    b = _fft_convolve(a, a)
    b = np.maximum(b, 0.0)  # numerical safety
    
    # Weights for gradient
    w = b ** (p - 1.0)
    
    # Cross-correlation: c[m] = sum_j w[j+m] * a[j] for m=0..n-1
    # c[m] = convolve(a_rev, w)[m + n - 1]
    a_rev = a[::-1]
    c_full = _fft_convolve(a_rev, w)
    c = c_full[n - 1:2 * n - 1]
    
    # Norm factor
    sp = np.sum(b ** p)
    if sp < 1e-300:
        sp = 1e-300
    norm_factor = sp ** (1.0 / p - 1.0)
    
    grad = 2.0 * norm_factor * c
    loss = sp ** (1.0 / p)
    return loss, grad


def _optimize_adam(a_init, budget_s, seed, p_start=5.0, p_end=200.0, lr=0.05):
    """
    Run Adam-based gradient descent to minimize the smooth max approximation.
    Uses gradual sharpening of p and restarts on stagnation.
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a /= s

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    best_loss = float('inf')
    best_a = a.copy()
    start = time.monotonic()
    deadline = start + budget_s

    # Estimate steps
    _ = _compute_loss_grad(a, p_start)
    t_step = max(time.monotonic() - start, 0.001)
    max_steps = int(budget_s / t_step)
    max_steps = min(max_steps, 100000)

    no_improve = 0
    patience = 300
    restart_count = 0
    max_restarts = 10

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        frac = step / max(1, max_steps)
        p = p_start * (p_end / p_start) ** frac

        loss, grad = _compute_loss_grad(a, p)

        if not np.isfinite(loss):
            # Numerical issue, reset
            a = best_a.copy() + rng.normal(0, 0.005, n)
            a = np.maximum(a, 0)
            a /= a.sum()
            m_buf = np.zeros(n)
            v_buf = np.zeros(n)
            t = 0
            continue

        if loss < best_loss - 1e-12:
            best_loss = loss
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience and restart_count < max_restarts:
            # Perturb from best and restart
            a = best_a + rng.normal(0, 0.02, n)
            a = np.maximum(a, 0)
            a /= a.sum()
            m_buf = np.zeros(n)
            v_buf = np.zeros(n)
            t = 0
            no_improve = 0
            restart_count += 1
            continue

        if no_improve > patience * 3:
            # Give up on this initialization
            break

        t += 1
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * grad ** 2
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)

        # Adaptive LR: decay over time
        eff_lr = lr * (1.0 / (1.0 + 0.001 * step))
        a = a - eff_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project: clip to non-negative, renormalize
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
        else:
            a /= s

    return best_a


def _make_ramp(n):
    """Create a linear ramp sequence normalized to sum=1."""
    a = np.arange(1, n + 1, dtype=np.float64)
    return a / a.sum()


def _make_quadratic_ramp(n):
    """Create a quadratic ramp sequence normalized to sum=1."""
    a = np.arange(1, n + 1, dtype=np.float64) ** 2
    return a / a.sum()


def _make_bimodal(n):
    """Create a bimodal sequence (high at edges, low in middle)."""
    center = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * x / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_random(n, seed):
    """Create a random non-negative sequence normalized to sum=1."""
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.05
    return a / a.sum()
