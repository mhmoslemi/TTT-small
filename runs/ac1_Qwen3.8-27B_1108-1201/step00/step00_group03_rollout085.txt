    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    spec = fa * np.conj(fa)
    ac = np.fft.irfft(spec, fft_size)[:size]
