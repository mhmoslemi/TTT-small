    for val in [0.001, 0.1, 1.0, 10.0, 100.0, 1000.0]:
        cand = np.full(n_init, val, dtype=np.float64)
        s = float(evaluate_sequence(cand.tolist()))
        if s < best_score - 1e-12:
            best = cand
            best_score = s
            print(f"  Improved: val={val}, score={s:.10f}")
    
    # Try tapered sequences (slightly non-constant)
    for n_try in [64, 128, 256, 512]:
        for taper in [0.01, 0.05, 0.1, 0.2]:
            if time.monotonic() > deadline:
                break
            # Linear taper
            x = np.linspace(1 - taper, 1 + taper, n_try)
            cand = x / x.sum() * np.sqrt(2 * n_try)
            s = float(evaluate_sequence(cand.tolist()))
            if s < best_score - 1e-12:
                best = cand
                best_score = s
                print(f"  Improved: n={n_try}, taper={taper}, score={s:.10f}")
    
    # Try the LP-based approach from SoTA on the constant sequence
    n = len(best)
    parent = best.copy()
    conv_full = np.convolve(parent, parent)
    max_b = conv_full.max()
    sum_a = parent.sum()
    
    # LP approach: find g maximizing sum(g) s.t. conv(parent, g) <= max_b, g >= 0
    # For constant parent, the constraint conv(parent, g)[k] <= max_b means
    # sum_j parent[j]*g[k-j] <= max_b for all valid k
    # Since parent is constant (=c), this is c * sum of g over valid range <= max_b
    # The binding constraint is at the center where the full sum is involved
    # So sum(g) <= max_b / c = max_b / parent[0]
    # Maximum sum(g) = max_b / parent[0] = n*c / c = n
    # So g is proportional to ones, which is the same as parent (up to scale)
    # This confirms constant is a fixed point of the LP iteration.
    
    # Try random perturbations to escape potential local issues
    best = best.copy()
    stall_count = 0
    for i in range(5000):
        if time.monotonic() > deadline:
            break
        if stall_count > 200:

        # Various perturbation strategies
        strategy = rng.integers(0, 5)
        cand = best.copy()
        n_cand = len(cand)
        
        if strategy == 0:
            # Small random Gaussian perturbation
            noise = rng.normal(0, 0.005, n_cand)
            cand = cand + noise
            cand = np.clip(cand, 0, 1000)
        elif strategy == 1:
            # Block perturbation
            width = max(2, n_cand // 50)
            start = rng.integers(0, max(1, n_cand - width))
            cand[start:start+width] *= np.exp(rng.normal(0, 0.05))
        elif strategy == 2:
            # Symmetric perturbation
            half = n_cand // 2
            pert = rng.normal(0, 0.02, half)
            pert = np.clip(pert, -0.5, 0.5)
            cand[:half] = cand[:half] * (1 + pert)
            cand[half:] = cand[half:] * (1 + pert[::-1])

            new_n = int(rng.integers(10, min(2000, max(20, n_cand * 2))))

            mean_val = cand.mean()
            cand = 0.95 * cand + 0.05 * mean_val

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best]))
    print(f"Final result: n={len(best)}, score={final_score:.10f}")
    print(f"Best score found: {best_score:.10f}")
