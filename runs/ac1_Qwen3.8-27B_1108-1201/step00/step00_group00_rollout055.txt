def smooth_max_grad(a, beta):
    """Compute log-sum-exp smooth max of autocorrelation and its gradient."""
    n = len(a)
    # Compute linear autocorrelation via FFT
    L = 1
    target = 2 * n - 1
    while L < target:
        L *= 2
    
    A = np.fft.rfft(a, L)
    b = np.fft.irfft(A * A, L)[:2*n-1]
    
    # Log-sum-exp
    b_max = np.max(b)
    shifted = beta * (b - b_max)
    # Clip to avoid underflow (not needed for exp, but for numerical stability)
    exp_vals = np.exp(np.clip(shifted, -700, 0))
    sum_exp = np.sum(exp_vals)
    log_sum_exp = b_max + np.log(sum_exp) / beta
    
    # Softmax weights
    weights = exp_vals / sum_exp
    
    # Gradient: d(log_sum_exp)/d(a[i]) = 2 * sum_j weights[i+j] * a[j]
    # Using cross-correlation via FFT
    W = np.fft.rfft(weights, L)
    grad_corr = np.fft.irfft(W * np.conj(A), L)[:n]
    grad = 2.0 * grad_corr
    
    return log_sum_exp, grad, b

def project_simplex(a):

    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

import numpy as np
import time

"""
Minimizes 2*n*max(autocorr) / sum^2 using:
1. Fourier-based initialization with iterative amplitude constraint
2. Projected gradient descent with log-sum-exp smoothing
3. Multiple restarts and length exploration
"""

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.full(n, 1.0/n)
    return v / s

def autocorr_fft(a):
    """Compute linear autocorrelation (full convolution of a with itself) using FFT."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    A = np.fft.rfft(a, L)
    b = np.fft.irfft(A * A, L)
    return b[:2*n-1], A, L

def smooth_logsumexp_autocorr(a, beta):
    """Compute smooth max of autocorrelation and its gradient."""
    n = len(a)
    b, A, L = autocorr_fft(a)
    
    b_max = np.max(b)
    shifted = beta * (b - b_max)
    shifted = np.minimum(shifted, 0.0)  # for numerical stability
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    
    log_sum_exp = b_max + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    
    # Gradient: 2 * cross_correlation(weights, a)[:n]
    W = np.fft.rfft(weights, L)
    corr = np.fft.irfft(W * np.conj(A), L)[:n]
    grad = 2.0 * corr
    
    return log_sum_exp, grad

def optimize_pg(a0, time_budget, verbose=False):
    """Optimize using projected gradient descent with log-sum-exp."""
    n = len(a0)
    a = project_simplex(a0.copy())
    
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    start_beta = 3.0

        progress = min(1.0, it / (total_iters * 0.8))
        beta = start_beta + (end_beta - start_beta) * progress

        grad_norm = np.linalg.norm(grad) / np.sqrt(n)

        lr = min(lr, 10.0)  # cap the step

def gerchberg_saxton_init(n, iters=50, seed=42):
    """Initialize using Gerchberg-Saxton (iterative amplitude constraint)."""
    rng = np.random.default_rng(seed)
    L = 1
    while L < 2 * n:
        L *= 2
    
    # Random phases
    A = np.exp(1j * rng.uniform(0, 2*np.pi, L//2 + 1))
    
    for _ in range(iters):
        # Time domain: enforce non-negativity
        a_full = np.fft.irfft(A, L)
        a = np.abs(a_full[:n])  # take first n elements, make non-negative
        a = project_simplex(a)
        
        # Frequency domain: enforce flat magnitude
        Af = np.fft.rfft(np.concatenate([a, np.zeros(L - n)]), L)
        A = np.exp(1j * np.angle(Af))
    
    return a

def fourier_taper_init(n, seed=42):
    """Initialize using a tapered Fourier design."""
    # Create a target autocorrelation that's flat in the middle
    b = np.ones(2*n-1)
    # Taper edges with a cosine window
    edge = n // 4
    for i in range(edge):
        t = i / edge
        b[i] = 0.5 * (1 + np.cos(np.pi * t))
        b[2*n-2-i] = 0.5 * (1 + np.cos(np.pi * t))
    
    # DFT
    B = np.fft.rfft(b, 2*n)
    B = np.maximum(B.real, 0)  # take real part, clip negatives
    if B.sum() < 1e-10:

    A = np.sqrt(B)
    a = np.fft.irfft(A, 2*n)[:n]
    a = np.abs(a)

import numpy as np
import time

"""
Minimizes 2*n*max(linear_autocorrelation) / (sum)^2 using:
- Multiple initialization strategies (Fourier, Gerchberg-Saxton, random)
- Projected gradient descent with log-sum-exp smoothing of the peak
- Backtracking line search for adaptive step sizes
- Block perturbation local search for final polish
- Multiple sequence lengths explored
"""

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} using clip-normalize."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return v / s

def autocorr_and_grad(a, beta):
    """
    Compute smooth max of autocorrelation and its gradient w.r.t. a.
    Returns (smooth_value, gradient, peak_index).
    """
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    
    A = np.fft.rfft(a, L)
    b = np.fft.irfft(A * A, L)
    b = b[:2 * n - 1]
    
    # Find the (approximate) peak
    peak_idx = np.argmax(b)
    
    # Log-sum-exp for smooth max
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    
    smooth_val = b_max + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    
    # Gradient: d(smooth_val)/d(a[i]) = 2 * sum_j weights[i+j] * a[j]
    # = 2 * (cross-correlation of weights and a at lag i)
    # Using FFT: cross_corr = IFFT(FFT(weights) * conj(FFT(a)))
    W = np.fft.rfft(weights, L)
    cross = np.fft.irfft(W * np.conj(A), L)[:n]
    grad = 2.0 * cross
    
    return smooth_val, grad, peak_idx, b

    """

    a = project_simplex(a0.copy())

    best_score = evaluate_sequence(best_a.tolist())

    start_beta = 5.0
    end_beta = 200.0
    base_lr = 2.0
    max_backtrack = 10
    eval_interval = 100
    
    it = 0
    prev_smooth = None
    consecutive_worse = 0
    
    while time.monotonic() < deadline:
        # Anneal beta
        elapsed_frac = (time.monotonic() - (deadline - time_budget)) / time_budget
        beta = start_beta + (end_beta - start_beta) * min(1.0, elapsed_frac * 1.5)
        
        smooth_val, grad, peak_idx, b = autocorr_and_grad(a, beta)
        
        # Backtracking line search
        lr = base_lr
        a_new = a
        smooth_new = smooth_val
        
        for bt in range(max_backtrack):
            a_trial = a - lr * grad
            a_trial = project_simplex(a_trial)
            
            if bt < max_backtrack - 1:

                sv, _, _, _ = autocorr_and_grad(a_trial, beta)
                if sv < smooth_val * (1 - 1e-4):
                    a_new = a_trial
                    smooth_new = sv
                    break
            else:
                a_new = a_trial
                smooth_new = smooth_val
        
        a = a_new
        it += 1
        
        # Periodically evaluate the true objective
        if it % eval_interval == 0:

    # Final evaluation
