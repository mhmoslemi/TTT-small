"""
LP-based alternating minimization for low-convolution non-negative sequences.
Uses the state-of-the-art LP relaxation approach: fix f, solve LP to find
optimal spreader g, blend, iterate.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix(f, n):
    """Build (2n-1) x n sparse matrix A such that A @ g = conv(f, g)."""
    k = np.arange(2 * n - 1)
    # Number of nonzeros in each row
    lens = np.minimum(np.minimum(k + 1, n), 2 * n - 1 - k)
    row_ptr = np.concatenate(([0], np.cumsum(lens)))
    total_nnz = int(row_ptr[-1])
    col_indices = np.empty(total_nnz, dtype=np.int32)
    data = np.empty(total_nnz, dtype=np.float64)
    
    for i in range(2 * n - 1):
        start = int(row_ptr[i])
        end = int(row_ptr[i + 1])
        length = end - start
        if length == 0:
            continue
        # For row i (lag k=i):
        # conv(f,g)[i] = sum_{j=max(0,i-n+1)}^{min(i,n-1)} f[j] * g[i-j]
        j_start = max(0, i - n + 1)
        j_end = min(i, n - 1)
        # g indices: i - j_end to i - j_start
        g_start = i - j_end
        g_end = i - j_start
        col_indices[start:end] = np.arange(g_start, g_end + 1, dtype=np.int32)
        data[start:end] = f[j_start:j_end + 1]
    
    return csr_matrix((data, col_indices, row_ptr), shape=(2 * n - 1, n))

    """Compute the evaluation score without calling evaluate_sequence."""

    if S < 0.01:
        return np.inf
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    return float(2 * n * max_b / (S * S))

"""
LP-based alternating minimization for low-convolution non-negative step functions.
Uses the state-of-the-art LP relaxation: fix f, solve LP to find optimal spreader g,
blend, iterate. Includes line search and local refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix(f, n):
    """Build (2n-1) x n sparse matrix A such that A @ g = conv(f, g)."""
    k_arr = np.arange(2 * n - 1)
    lens = np.minimum(np.minimum(k_arr + 1, n), 2 * n - 1 - k_arr)
    row_ptr = np.concatenate(([0], np.cumsum(lens)))
    total_nnz = int(row_ptr[-1])
    col_indices = np.empty(total_nnz, dtype=np.int32)
    data = np.empty(total_nnz, dtype=np.float64)

    for i in range(2 * n - 1):
        start = int(row_ptr[i])
        end = int(row_ptr[i + 1])
        length = end - start
        if length == 0:
            continue
        j_start = max(0, i - n + 1)
        j_end = min(i, n - 1)
        g_start = i - j_end
        g_end = i - j_start
        col_indices[start:end] = np.arange(g_start, g_end + 1, dtype=np.int32)
        data[start:end] = f[j_start:j_end + 1]

    return csr_matrix((data, col_indices, row_ptr), shape=(2 * n - 1, n))


def quick_score(f):
    """Fast score computation (same as evaluate_sequence but on numpy array)."""
    f = np.clip(f, 0.0, 1000.0)
    n = len(f)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    b = np.convolve(f, f)
    max_b = b.max()
    return float(2.0 * n * max_b / (S * S))


def solve_spreader(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 60})
    if result.success and result.x is not None:
        return np.clip(result.x, 0.0, None)
    return None


def initialize_sequence(n, init_type, rng):
    """Create an initial sequence of length n."""
    i = np.arange(n, dtype=np.float64)
    if init_type == 'raised_cosine':
        f = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / n))
    elif init_type == 'raised_cosine_shift':
        f = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / n))
    elif init_type == 'gaussian':
        center = n / 2.0
        sigma = n / 6.0
        f = np.exp(-(i - center) ** 2 / (2.0 * sigma * sigma))
    elif init_type == 'bimodal':
        c1 = n * 0.25
        c2 = n * 0.75
        sigma = n / 16.0
        f = np.exp(-(i - c1) ** 2 / (2.0 * sigma * sigma)) + np.exp(-(i - c2) ** 2 / (2.0 * sigma * sigma))
    elif init_type == 'random':
        f = np.abs(rng.standard_normal(n)) + 0.1
    elif init_type == 'inverse_triangle':
        center = n / 2.0
        f = np.ones(n) - np.abs(i - center) / center
        f = np.clip(f, 0.05, None)
    else:
        f = np.ones(n)

    f = np.clip(f, 0.0, 1000.0)
    S = np.sum(f)
    if S < 1e-6:
        f = np.ones(n)
        S = n
    f = f / S * np.sqrt(2.0 * n)
    return f

        b = np.convolve(f, f)

        g = solve_spreader(f, M, n)
        if g is None:
            stagnation += 1

                break

        if g_sum < 1e-10:

                f_new[start:ends] += delta

                    score = quick_score(f_new)
