LP-based iterative optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def build_conv_matrix(f):
    n = len(f)
    rows = (np.arange(n, dtype=np.int64)[:, None] + np.arange(n, dtype=np.int64)[None, :]).flatten()
    cols = np.repeat(np.arange(n, dtype=np.int64), n)
    vals = np.repeat(f.astype(np.float64), n)
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M, time_limit=50):
    n = len(f)
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                          method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    candidate = np.maximum(candidate, 0.0)

    return candidate
