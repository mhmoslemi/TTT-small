def fft_convolve_self(a):
    """Compute full self-convolution of a using FFT."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2*n-1]

def fft_grad_peak(a, active_mask):
    """Compute gradient of peak: g[i] = sum_k active_mask[k] * a[k-i]"""
    n = len(a)
    N = 1
    while N < 2*n - 1 + n - 1:  # need >= (2n-1) + n - 1 = 3n-2
        N *= 2
    A = np.fft.rfft(a, N)
    M = np.fft.rfft(active_mask, N)
    result = np.fft.irfft(np.conj(A) * M, N)
    return result[:n]

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Binary search for threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho such that u[rho] > (cssv[rho] - target_sum) / (rho+1)
    cond = u * np.arange(1, n+1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    # Ensure exact sum
    s = result.sum()
    if s > 0:
        result = result * target_sum / s
    return result

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using FFT-based gradient descent
on a smooth surrogate (sum of squared convolution values plus peak penalty),
with momentum, adaptive step size, and periodic random restarts.
Tries multiple sequence lengths and initializations within the time budget.
"""

import numpy as np
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def fft_cross_corr(a, b):
    """Compute g[i] = sum_k b[k]*a[k-i] for i=0..n-1 via FFT."""
    n = len(a)
    m = len(b)
    N = 1
    while N < m + n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    result = np.fft.irfft(np.conj(A) * B, N)
    return result[:n]


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result * target_sum / s
    return result


def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2 for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def make_ramp_init(n, seed=0):
    """Create an asymmetric ramp initialization."""
    rng = np.random.default_rng(seed)
    a = np.arange(1, n + 1, dtype=np.float64)
    a += rng.random(n) * 0.5
    a = np.maximum(a, 0.1)
    return a / a.sum()


def make_random_init(n, seed=0):
    """Create a random initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1
    return a / a.sum()


def make_gapped_init(n, seed=0):
    """Create a gapped pattern initialization (inspired by Barker-like sequences)."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    pattern_len = max(3, n // 50)
    for i in range(n):
        pos_in_pattern = i % pattern_len
        # Create an asymmetric pattern: high in first half, low in second
        frac = pos_in_pattern / pattern_len
        if frac < 0.6:
            a[i] = 1.0 + 0.3 * rng.random()
        else:
            a[i] = 0.1 + 0.3 * rng.random()
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_phase_init(n, seed=0):
    """Create a multi-phase initialization with asymmetry."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    freq1 = 2.0 * np.pi * 7.0 / n
    freq2 = 2.0 * np.pi * 13.0 / n
    phase1 = rng.uniform(0, 2 * np.pi)
    phase2 = rng.uniform(0, 2 * np.pi)
    idx = np.arange(n)
    a = 1.0 + 0.4 * np.cos(freq1 * idx + phase1) + 0.3 * np.cos(freq2 * idx + phase2)
    a = np.maximum(a, 0.0)
    a += 0.1 * rng.random(n)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def optimize_single(n, a_init, time_budget, seed=0, verbose=False):
    """
    Optimize a sequence of length n to minimize 2*n*max(conv)/sum^2.
    Uses gradient descent on sum(c^2) + lambda*max(c)^2 with momentum.
    """
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0).copy()
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)

    deadline = time.monotonic() + time_budget
    momentum = np.zeros(n)
    beta = 0.7

    it = 0
    no_improve = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve_self(a)
        peak = c.max()
        score = 2.0 * n * peak  # sum=1

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters based on progress
        progress = min(1.0, it / 5000.0)
        eta = 0.02 * (1.0 - 0.9 * progress) + 0.0005
        lam = 0.5 + 2.0 * progress  # increase peak penalty over time
        beta = 0.8 - 0.2 * progress

        # Compute smooth gradient: d/da sum(c^2) ≈ 4*(c cross-corr a)
        g_smooth = fft_cross_corr(a, c)

        # Compute peak gradient: d/da max(c)^2 ≈ 4*max(c)*a[peak_pos - i]
        peak_pos = int(np.argmax(c))
        # a[peak_pos - i] for i=0..n-1 (valid where 0<=peak_pos-i<n)
        # This is the cross-correlation of a with a delta at peak_pos
        # Which is just a shifted version of a
        g_peak = np.zeros(n)
        lo = max(0, peak_pos - (n - 1))
        hi = min(n, peak_pos + 1)
        if hi > lo:
            # g_peak[i] = a[peak_pos - i] if 0 <= peak_pos-i < n
            # i ranges from max(0, peak_pos-(n-1)) to min(n-1, peak_pos)
            g_peak[lo:hi] = a[peak_pos - (hi - 1):peak_pos - lo + 1][::-1]

        # Combined gradient
        g = 4.0 * g_smooth + lam * 4.0 * peak * g_peak

        # Mean-center to preserve sum
        g_mean = g.mean()
        g = g - g_mean

        # Momentum update
        momentum = beta * momentum - eta * g
        a_new = a + momentum

        # Project onto simplex {x>=0, sum=1}
        a_new = project_simplex(a_new, 1.0)
        a_new = np.clip(a_new, 0.0, 1000.0)
        a = a_new

        # Periodic random perturbation to escape local minima
        if no_improve > 2000:
            for _ in range(5):
                pert_size = 0.05 / np.sqrt(n)
                pert = rng.standard_normal(n) * pert_size
                a_test = project_simplex(a + pert, 1.0)
                c_test = fft_convolve_self(a_test)
                score_test = 2.0 * n * c_test.max()
                if score_test < best_score:
                    a = a_test
                    best_score = score_test
                    best_a = a.copy()
                    break
            no_improve = 0
            momentum *= 0.5

    # Final check with actual evaluator
    final_score = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_score):
        if verbose:
            print(f"  n={n}, iters={it}, eval_score={final_score:.6f}")
        return best_a.tolist(), final_score
    return None, np.inf
