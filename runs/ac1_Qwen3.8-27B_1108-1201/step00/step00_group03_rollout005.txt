```python
"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a linear program to find the best direction g at each step, then performs
a line search for the optimal convex combination. Combines LP iteration with
random restarts and gradient-based refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_objective(a):
    """Compute the objective: 2*n*max_b/sum_a^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    b = np.convolve(a, a)
    M = np.max(b)
    return 2.0 * n * M / (S * S)


def build_conv_matrix(f):
    """Build matrix A where A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M):
    """Solve: max sum(g) s.t. A_f @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def gradient_step(f, lr=0.01, beta=50.0):
    """One gradient step on smoothed objective (logsumexp approximation of max)."""
    n = len(f)
    b = np.convolve(f, f)
    S = np.sum(f)

    # Softmax weights over b
    b_max = np.max(b)
    w = np.exp(beta * (b - b_max))
    w /= np.sum(w)

    # Gradient of sum_m w[m]*b[m] w.r.t. a[j] = 2 * sum_m w[m]*a[m-j]
    grad = np.zeros(n)
    for j in range(n):
        # sum over valid m where 0 <= m-j < n
        m_start = j
        m_end = j + n
        if m_end > 2 * n - 1:
            m_end = 2 * n - 1
        if m_start < m_end:
            grad[j] = 2.0 * np.dot(w[m_start:m_end], f[:m_end - m_start])

    # Project gradient onto sum=constant hyperplane
    grad -= np.mean(grad)

    f_new = f - lr * grad
    f_new = np.clip(f_new, 0, 1000)
    return f_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: LP-based iteration with line search, random restarts,
    and gradient refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    # Start from height_sequence_1
    f = np.array(height_sequence_1, dtype=np.float64)
    f = np.clip(f, 0, 1000)
    n = len(f)

    best_f = f.copy()
    best_obj = compute_objective(f)
    print(f"Initial objective: {best_obj:.6f} (n={n})", flush=True)

    # Time-tracked LP iteration
    no_improve_count = 0
    it_count = 0

    while time.monotonic() < deadline:
        it_start = time.monotonic()
        it_count += 1

        b = np.convolve(f, f)
        M = np.max(b)
        S = np.sum(f)

        # Solve LP with slightly relaxed constraint for more freedom
        M_lp = M * 1.1
        g = solve_lp(f, M_lp)
        if g is None or np.sum(g) < 1e-10:
            # Fallback: gradient step
            f = gradient_step(f, lr=0.005, beta=100.0)
            continue

        # Rescale g to have same sum as f
        S_g = np.sum(g)
        g = g * (S / S_g)

        # Line search for optimal t
        best_t = 0.0
        best_local_obj = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8]:
            candidate = (1.0 - t) * f + t * g
            candidate = np.clip(candidate, 0, 1000)
            obj = compute_objective(candidate)
            if obj < best_local_obj:
                best_local_obj = obj
                best_t = t

        if best_t > 0 and best_local_obj < best_obj:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0, 1000)
            best_f = f.copy()
            best_obj = best_local_obj
            no_improve_count = 0
            elapsed = time.monotonic() - it_start
            if it_count % 20 == 1:
                print(f"  Iter {it_count}: obj={best_obj:.6f}, t={best_t:.2f}, "
                      f"M={M:.2f}, time={elapsed:.1f}s", flush=True)
        else:
            no_improve_count += 1
            # Try gradient step
            f_grad = gradient_step(f, lr=0.01, beta=50.0)
            obj_grad = compute_objective(f_grad)
            if obj_grad < best_obj:
                f = f_grad
                best_f = f.copy()
                best_obj = obj_grad
                no_improve_count = 0
            elif no_improve_count > 10:
                # Random restart / perturbation
                strength = 0.05 + 0.05 * min(no_improve_count // 10, 5)
                f = f * np.exp(rng.normal(0, strength, n))
                f = np.clip(f, 0, 1000)
                no_improve_count = 0

        # On occasional iterations, try a fresh random initial point
        if it_count % 200 == 0:
            # Try a structured random starting point
            f_rand = np.abs(rng.standard_normal(n)) + 0.1
            obj_rand = compute_objective(f_rand)
            if obj_rand < best_obj:
                print(f"  Found better random start: {obj_rand:.6f}", flush=True)
                best_f = f_rand.copy()
                best_obj = obj_rand
            f = best_f.copy()  # reset to best

    # Final gradient refinement
    f = best_f.copy()
    for lr in [0.01, 0.005, 0.002]:
        for _ in range(50):
            if time.monotonic() > deadline:
                break
            f_new = gradient_step(f, lr=lr, beta=100.0)
            obj_new = compute_objective(f_new)
            if obj_new < best_obj:
                f = f_new
                best_f = f.copy()
                best_obj = obj_new
                print(f"  Gradient refinement: obj={best_obj:.6f}", flush=True)

    final_obj = compute_objective(best_f)
    print(f"Final objective: {final_obj:.6f}", flush=True)
    return [float(x) for x in best_f]
```
