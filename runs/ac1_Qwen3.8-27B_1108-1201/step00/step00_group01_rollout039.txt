def compute_convolution(a):
    """Compute auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:

    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)

    c = compute_convolution(a)
