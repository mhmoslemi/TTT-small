def square_conv(a, M=None):

"""
Search for non-negative step-function heights that minimize

    2*n*max(conv(a,a)) / (sum(a)**2).

The optimizer combines:
  - fast FFT-based evaluation of the convolution,
  - soft-max / natural-gradient descent on the simplex,
  - damped Fourier-domain flattening steps,
  - random multiplicative / transfer perturbations,
  - a small pool of diverse initializations and adaptive racing.

def next_pow2(m):
    p = 1
    while p < int(m):
        p <<= 1
    return p


def normalize_array(a):
    arr = np.asarray(a, dtype=float)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return np.array([1.0], dtype=float)

    arr = arr.copy()
    arr[~np.isfinite(arr)] = 0.0
    arr = np.maximum(arr, 0.0)

    s = float(np.sum(arr))
    if s <= 1e-300:
        return np.ones(arr.size, dtype=float) / float(arr.size)

    arr = arr / s
    return arr


def square_conv(a, m=None):
    a = np.asarray(a, dtype=float)
    n = a.size
    if n == 0:
        return np.array([], dtype=float)

    L = 2 * n - 1

    if m is None:
        # Direct convolution is faster for small-to-moderate n.
        if n <= 256 or (long(L) * long(n)) <= 2_000_000:
            c = np.convolve(a, a)
            c = np.asarray(c, dtype=float)
            c[~np.isfinite(c)] = 0.0
            c = np.maximum(c, 0.0)
            return c

        m = next_pow2(L)

    if m < L:
        m = next_pow2(L)

    A = np.fft.rfft(a, m)
    c = np.fft.irfft(A * A, m)[:L]
    c = np.real(c).astype(float, copy=False)
    c[~np.isfinite(c)] = 0.0
    c = np.maximum(c, 0.0)
    return c

def random_perturb(a, rng, sigma):

def one_peak_gradient(a, k):
    a = np.asarray(a, dtype=float)
    n = int(a.size)
    g = np.zeros(n, dtype=float)
    k = int(k)

    j_start = max(0, k - (n - 1))
    j_end = min(k, n - 1)

    for j in range(j_start, j_end + 1):
        g[j] = 2.0 * float(a[k - j])

    g[~np.isfinite(g)] = 0.0
    return np.maximum(g, 0.0)


def gradient_from_weights(a, w):
    a = np.asarray(a, dtype=float)
    w = np.asarray(w, dtype=float)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=float)

    L = 2 * n - 1

    if n <= 200:
        g = np.zeros(n, dtype=float)
        for m in range(n):
            am = float(a[m])
            if am == 0.0:
                continue
            j_max = min(n - 1, L - 1 - m)
            if j_max >= 0:
                g[:j_max + 1] += am * w[m:m + j_max + 1]
        g *= 2.0
        g[~np.isfinite(g)] = 0.0
        return np.maximum(g, 0.0)

    m = next_pow2(n + L - 1)
    w_rev = w[::-1]
    A = np.fft.rfft(a, m)
    W = np.fft.rfft(w_rev, m)
    conv = np.fft.irfft(A * W, m).real

    idx = (L - 1) - np.arange(n, dtype=int)
    g = 2.0 * conv[idx]
    g = np.asarray(g, dtype=float)
    g[~np.isfinite(g)] = 0.0
    return np.maximum(g, 0.0)

    top = generate_top_candidates(10)

    if _BEST_SEQ is None:
        fb = normalize_array(np.array([1.0, 2.0, 2.0]))
        update_best(fb.tolist())

    print(f"after initialization best={_BEST_SCORE:.10f}", flush=True)

    active = []
    for _, a in top:
        active.append(a.copy())

    gb = None
    if _BEST_SEQ is not None:
        gb = np.asarray(_BEST_SEQ, dtype=float).copy()
        found = False
        for a in active:
            if a.size == gb.size:
                found = True
                break
        if not found:
            active.append(gb.copy())

    if base is not None:
        found = False
        for a in active:
            if a.size == base.size:
                found = True
                break
        if not found:
            active.append(base.copy())

    if gb is not None and time.monotonic() < deadline:
        targets = [256, 512, 1024, 2048, 4096]
        for target_n in targets:
            if len(active) >= 12:
                break
            if int(target_n) == int(gb.size):
                continue
            v = resize_sequence(gb, target_n)
            active.append(v)
            if time.monotonic() > deadline - 600.0:
                break

    rng_master = np.random.default_rng(int(seed))
    last_report = time.monotonic()

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem <= 30.0:
            break

        if len(active) == 0:
            active.append(normalize_array(np.ones(256)))

        scored = []
        for i, a in enumerate(active):
            sc = fast_score(a)
            scored.append((i, sc))
        scored.sort(key=scored_key)

        K = min(3, len(scored))
        slice_time = min(180.0, max(15.0, rem / (len(scored) + 1) * 1.2))

        for t in range(K):
            if time.monotonic() >= deadline - 60.0:
                break
            idx = scored[t][0]
            a = active[idx]
            s = int(rng_master.integers(0, 1000000000))
            improved, _ = local_improve(a, slice_time, seed=s)
            active[idx] = improved

        if len(active) > 8:
            scored = []
            for i, a in enumerate(active):
                scored.append((i, fast_score(a)))
            scored.sort(key=scored_key)

            new_active = []
            for i in range(min(8, len(scored))):
                new_active.append(active[scored[i][0]])

            if _BEST_SEQ is not None:
                gb2 = np.asarray(_BEST_SEQ, dtype=float)
                has = False
                for a in new_active:
                    if a.size == gb2.size:
                        has = True
                        break
                if not has:
                    if len(new_active) < 8:
                        new_active.append(gb2.copy())
                    else:
                        new_active[-1] = gb2.copy()

            active = new_active

        now = time.monotonic()
        if now - last_report > 30.0:
            print(f"t={now - start:.1f}s current best exact={_BEST_SCORE:.10f}", flush=True)
            last_report = now

    if time.monotonic() < deadline - 5.0:
        if _BEST_SEQ is not None:
            a = np.asarray(_BEST_SEQ, dtype=float)
            remaining = max(0.0, deadline - time.monotonic() - 5.0)
            a, _ = local_improve(a, remaining, seed=int(rng_master.integers(0, 1000000000)))
            if time.monotonic() < deadline:
                update_best(a.tolist())
        else:
            fb = normalize_array(np.array([1.0, 2.0, 2.0]))
            update_best(fb.tolist())

    if _BEST_SEQ is None or _BEST_SCORE > 1.921:
        fb = normalize_array(np.array([1.0, 2.0, 2.0]))
        if _BEST_SEQ is None:
            _BEST_SEQ = fb.tolist()
        else:
            fb2 = fb.tolist()
            update_best(fb2)
            if fb2 is not None and len(fb2) > 0:
                _BEST_SEQ = fb2

    print(f"final best exact={_BEST_SCORE:.10f}", flush=True)
    return [float(x) for x in _BEST_SEQ]
