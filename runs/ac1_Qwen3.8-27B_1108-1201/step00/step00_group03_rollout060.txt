def build_conv_matrix(a):
    """Build the matrix A such that A @ b = convolve(a, b)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    return A

"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using LP-based iterative refinement
combined with multiplicative subgradient polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def build_conv_matrix(a):
    """Build matrix A where A @ b = np.convolve(a, b)."""
    n = len(a)
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        # For column j, A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        A[j:j+n, j] = a
    return A

def evaluate_fast(a):
    """Fast evaluation without the overhead of evaluate_sequence."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    return float(2 * n * max_b / (s * s))

def lp_step(a, M):
    """Solve LP: max sum(b) s.t. convolve(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def multiplicative_step(a, lr):
    """One step of multiplicative subgradient descent."""
    n = len(a)
    conv = np.convolve(a, a)
    M = conv.max()
    # Find active constraints (near the max)
    threshold = M * 0.999
    S = np.where(conv >= threshold)[0]
    
    # Compute subgradient: g_i = 2/|S| * sum_{k in S} a[k-i]
    # This is equivalent to: g = 2/|S| * (restricted convolution of a with indicator of S)
    # conv of a with a, restricted to S... 
    # Actually, g_i = 2/|S| * sum_{k in S} a[k-i]
    # Let's compute this directly
    g = np.zeros(n)
    for k in S:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += a[j]
    g *= 2.0 / len(S)
    
    # Multiplicative update
    a = a * np.exp(-lr * g)
    a = np.clip(a, 0, None)
    return a

= `sum_{k=m}^{min(2n-2, m+n-1)} p[k] * a[k-m]`

This is exactly what we want: `g[m] = sum_k p[k] * a[k-m]` (with a[k-m] = 0 outside [0, n-1]).

So we need N >= (2n-1) + (n-1) = 3n-2 to avoid wraparound contamination. Let me use N = next_power_of_2(3n).

"""
Minimizes 2*n*max(convolve(a,a))/sum(a)^2 using LP-based block coordinate
descent combined with FFT-accelerated soft-subgradient polishing.
Tries multiple sequence lengths and initializations within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog


def build_conv_matrix(a):
    """Build dense matrix A (shape 2n-1 x n) such that A @ b = np.convolve(a, b)."""
    n = a.shape[0]
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def eval_fast(a):
    """Fast local evaluation of the objective."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    m = np.convolve(a, a).max()
    return 2.0 * n * m / (s * s)


def lp_best_response(a, M):
    """Solve LP: max sum(b) s.t. convolve(a,b) <= M element-wise, b >= 0."""
    n = a.shape[0]
    A = build_conv_matrix(a)
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if res.status == 0:
        return res.x
    return None


def fft_cross_corr_weighted(p, a):
    """
    Compute g[i] = sum_k p[k] * a[k-i] for i=0..n-1 using FFT.
    p has length 2n-1, a has length n.
    """
    n = a.shape[0]
    N = 1
    while N < 3 * n:
        N <<= 1
    P = np.zeros(N)
    P[:2 * n - 1] = p
    A = np.zeros(N)
    A[:n] = a
    corr = np.real(np.fft.ifft(np.fft.fft(P) * np.conj(np.fft.fft(A))))
    return corr[:n]


def soft_subgrad_step(a, lr, tau=5.0):
    """One step of multiplicative soft-subgradient descent using FFT."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    # Softmax weights (temperature tau)
    conv_shifted = tau * (conv - conv.max())
    p = np.exp(conv_shifted)
    p_sum = p.sum()
    if p_sum == 0:
        return a
    p /= p_sum
    # g[i] = 2 * sum_k p[k] * a[k-i]
    g = 2.0 * fft_cross_corr_weighted(p, a)
    # Multiplicative update
    a_new = a * np.exp(-lr * g)
    a_new = np.clip(a_new, 0.0, None)
    return a_new


def hard_subgrad_step(a, lr, frac=0.999):
    """One step using hard active set (fast when active set is small)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()
    if M <= 0:
        return a
    active = conv >= M * frac
    active_indices = np.where(active)[0]
    if len(active_indices) == 0:
        return a
    g = np.zeros(n)
    for k in active_indices:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        segment = a[max(0, k - n + 1):min(k, n - 1) + 1][::-1]
        length = i_hi - i_lo + 1
        g[i_lo:i_lo + length] += segment[:length]
    g = 2.0 * g / len(active_indices)
    a_new = a * np.exp(-lr * g)
    a_new = np.clip(a_new, 0.0, None)
    return a_new


def get_initializations(n, rng):
    """Generate several initial sequences of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x / n)))
    # Cosine squared
    inits.append(np.cos(np.pi * x / (2 * n)) ** 2)
    # Uniform
    inits.append(np.ones(n))
    # Triangular
    inits.append(np.clip(1.0 - np.abs(x - mid) / (mid + 1), 0, None))
    # Gaussian
    sigma = n / 5.0
    inits.append(np.exp(-(x - mid) ** 2 / (2 * sigma ** 2)))
    # Flat with tapered edges
    taper = n // 8
    a = np.ones(n)
    if taper > 0:
        a[:taper] = np.linspace(0, 1, taper, endpoint=False)
        a[-taper:] = np.linspace(1, 0, taper, endpoint=False)
    inits.append(a)
    # Random smooth
    noise = np.exp(rng.normal(0, 0.3, n))
    # Smooth it
    kernel = np.exp(-np.arange(201) ** 2 / 200)
    kernel /= kernel.sum()
    inits.append(np.convolve(noise, kernel, mode='same'))

    # Given sequence if length matches
    try:
        given = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
        if len(given) == n:
            inits.append(given.copy())
    except Exception:
        pass

    return inits


def run_lp_iterations(a, max_iters, deadline, best_score_ref):
    """Run LP-based block coordinate iterations. Returns (a, best_score)."""
    best_score = np.inf
    best_a = a.copy()
    n = a.shape[0]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        score = eval_fast(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

        conv = np.convolve(a, a)
        M = conv.max()
        if M <= 0:
            break

        b = lp_best_response(a, M)
        if b is None:
            break

        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        sum_a = a.sum()
        b_norm = b * (sum_a / sum_b)

        # Update with step size
        t = 0.5
        a_new = (1 - t) * a + t * b_norm
        a_new = np.clip(a_new, 0.0, None)

        # Check improvement
        score_new = eval_fast(a_new)
        if score_new < score:
            a = a_new
        else:
            # Try smaller step
            t = 0.25
            a_new = (1 - t) * a + t * b_norm
            a_new = np.clip(a_new, 0.0, None)
            score_new = eval_fast(a_new)
            if score_new < score:
                a = a_new
            else:
                # Try even smaller
                t = 0.1
                a_new = (1 - t) * a + t * b_norm
                a_new = np.clip(a_new, 0.0, None)
                score_new = eval_fast(a_new)
                if score_new < score:
                    a = a_new
                else:
                    break  # Stuck

    return best_a, best_score


def run_gradient_iterations(a, max_iters, deadline):
    """Run gradient-based polishing iterations. Returns (a, best_score)."""
    best_score = np.inf
    best_a = a.copy()
    lr = 0.3
    tau = 5.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        score = eval_fast(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

        # Decide between soft and hard subgradient
        if it % 3 == 0:
            a = soft_subgrad_step(a, lr, tau)
        else:
            a = hard_subgrad_step(a, lr)

        if a.max() <= 0:
            break

        # Adaptive learning rate
        score_new = eval_fast(a)
        if score_new > score * 1.1:
            lr *= 0.7
        elif score_new < score * 0.99:
            lr = min(lr * 1.05, 2.0)

        # Occasionally increase tau for sharper focus
        if it % 50 == 0:
            tau = min(tau * 1.5, 50.0)

    return best_a, best_score
