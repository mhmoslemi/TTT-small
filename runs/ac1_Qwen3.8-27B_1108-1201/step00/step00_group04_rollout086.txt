Coordinate descent with O(n) vectorized convolution updates to minimize
2n * max(convolve(a,a)) / sum(a)^2 over non-negative sequences.
Uses multiple random restarts and adaptive step sizes.

def _compute_outside_max(b, i, n):
    """Compute max of b outside the region [i, i+n-1]."""
    if i == 0:
        return b[n:].max()
    elif i + n >= 2 * n - 1:
        return b[:i].max()
    else:
        return max(b[:i].max(), b[i + n:].max())


def _coordinate_step(a, b, i, n, sum_a, score, rng):

    Try to improve coordinate i. Returns (new_a, new_b, new_sum_a, new_score, improved).

    C = _compute_outside_max(b, i, n)

        candidates = np.concatenate([[0.0], candidates, [v * 0.9, v * 1.1]])

            best_overall = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            if best_overall.ndim != 1 or best_overall.size == 0:
                best_overall = np.ones(256)
        except Exception:
            best_overall = np.ones(256)

    return [float(x) for x in best_overall]
