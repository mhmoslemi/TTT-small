    for k in K_star:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        seg = a[k - i_end : k - i_start + 1]
        grad[i_start : i_end + 1] += seg[::-1]
    grad /= len(K_star)
    return grad


def get_active_set(c, rel_tol=0.001):
    """Get lags where convolution is within rel_tol of max."""
    c_max = c.max()
    if c_max <= 0:
        return np.array([0])
    threshold = c_max * (1.0 - rel_tol)
    return np.where(c >= threshold)[0]


def mirror_fista(a0, time_limit, eta_init=0.3, active_tol=0.001):
    """Mirror descent with Nesterov acceleration for min max_k (a*a)[k] s.t. a in simplex."""
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()
    t_param = 1.0
    eta = eta_init
    best_a = a.copy()
    best_conv = fast_convolve(a).max()
    best_obj = 2.0 * n * best_conv
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Compute subgradient at y
        c_y = fast_convolve(y)
        K_star = get_active_set(c_y, rel_tol=active_tol)
        grad = hankel_grad(y, K_star)
        
        # Mirror descent step: a_new = a * exp(-eta * grad) / Z
        log_w = np.log(a + 1e-30) - eta * grad
        log_w -= log_w.max()
        a_new = np.exp(log_w)
        a_new = a_new / a_new.sum()
        
        # Nesterov momentum
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        beta = (t_param - 1.0) / t_curr
        t_param = t_curr
        
        y_new = a_new + beta * (a_new - a)
        a = a_new
        y = y_new
        
        # Evaluate
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            best_conv = conv.max()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive step size
        if no_improve > 50:
            eta *= 0.9
        elif no_improve > 200:
            eta = eta_init * 0.8
            # Perturb
            rng = np.random.default_rng(it * 104729 + 7)
            y = a + rng.standard_normal(n) * 0.001
            y = np.maximum(y, 0)
            y = y / y.sum()
            a = a.copy()
            no_improve = 0
        elif no_improve > 500:
            # Larger restart
            rng = np.random.default_rng(it * 7919 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            y = a.copy()
            t_param = 1.0
            eta = eta_init
            no_improve = 0
    
    return best_a, best_obj


def eucl_fista(a0, time_limit, eta_init=0.4, active_tol=0.001):
    """Euclidean FISTA (Nesterov accelerated projected gradient) for min max_k (a*a)[k]."""
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()
    t_param = 1.0
    eta = eta_init
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        c_y = fast_convolve(y)
        K_star = get_active_set(c_y, rel_tol=active_tol)
        grad = hankel_grad(y, K_star)
        
        # Backtracking line search
        a_trial = project_simplex(y - eta * grad)
        obj_trial = 2.0 * n * fast_convolve(a_trial).max()
        obj_y = 2.0 * n * c_y.max()
        
        # Simple backtracking
        while obj_trial > obj_y + 1e-15 and eta > 0.01:
            eta *= 0.5
            a_trial = project_simplex(y - eta * grad)
            obj_trial = 2.0 * n * fast_convolve(a_trial).max()
        
        a_new = a_trial
        eta = min(eta * 1.05, 0.8)  # slight increase if no backtrack was needed
        
        # Nesterov
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        beta = (t_param - 1.0) / t_curr
        t_param = t_curr
        y = a_new + beta * (a_new - a)
        a = a_new
        
        if obj_trial < best_obj - 1e-15:
            best_obj = obj_trial
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            eta = eta_init
        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 17)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            y = a.copy()
            t_param = 1.0
            no_improve = 0
    
    return best_a, best_obj


def kkt_solve(a0, n_iters=100, active_tol=0.001):
    """
    Direct KKT solver: for a given active set, solve Q@a = c*1, sum(a)=1, a>=0.
    Q is a Hankel matrix with Q[i,j] = 1/|K*| if i+j in K*, else 0.
    """
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    for _ in range(n_iters):
        c = fast_convolve(a)
        K_star = get_active_set(c, rel_tol=active_tol)
        K_list = list(K_star)
        m = len(K_list)
        
        # Q[i,j] = 1/m if i+j in K_star, else 0
        # Solve: Q @ a = c_val * 1, sum(a) = 1, a >= 0
        # This is: find a minimizing ||Q@a - 1||^2 s.t. sum(a)=1, a>=0
        # (normalizing c_val = 1)
        
        # Build Q as a Hankel matrix (sparse-ish)
        # Q[i,j] depends only on i+j
        # Q[i,j] = 1/m if (i+j) in K_star
        
        # For the linear system Q*a = 1 (trying interior solution):
        # sum_j Q[i,j] * a[j] = 1 for all i
        # sum_j a[j] = 1
        
        # Let's solve the (n+1) x (n+1) system:
        # [Q  -1] [a]   [0]
        # [1'  0] [c] = [1]
        
        # Build Q
        Q = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                if i + j in K_star:  # This is slow!
                    Q[i, j] = 1.0 / m
        
        # This O(n^2) construction is too slow for n=768 in a loop.
        # Instead, build Q as a Hankel matrix using the first row.
        # Q[i,j] = q[i+j] where q[k] = 1/m if k in K_star, else 0
        q = np.zeros(2 * n - 1)
        q[K_star] = 1.0 / m
        # Q is a Hankel matrix: Q[i,j] = q[i+j]
        # We can compute Q @ a using the structure:
        # (Q@a)[i] = sum_j q[i+j] * a[j]
        
        # For the linear system, let's just use Q @ a = c*1
        # and solve numerically
        
        # Actually, for n=768, building Q explicitly is 768^2 = 590K bytes, which is fine.
        # Construction:
        Q = np.zeros((n, n))
        for k in K_star:
            # Q[i,j] = 1/m when i+j = k
            # i ranges from max(0, k-n+1) to min(k, n-1)
            # j = k - i
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                Q[i, k - i] = 1.0 / m
        
        # Solve [Q -1; 1' 0] [a; c] = [0; 1]
        A_aug = np.zeros((n + 1, n + 1))
        A_aug[:n, :n] = Q
        A_aug[:n, n] = -1.0
        A_aug[n, :n] = 1.0
        b_aug = np.zeros(n + 1)
        b_aug[n] = 1.0
        
        try:
            sol = np.linalg.solve(A_aug, b_aug)
            a_candidate = sol[:n]
            if np.all(a_candidate >= -1e-10):
                a_candidate = np.maximum(a_candidate, 0)
                a_candidate /= a_candidate.sum()
                obj = 2.0 * n * fast_convolve(a_candidate).max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = a_candidate.copy()
                    a = a_candidate.copy()
                else:
                    # KKT with this active set doesn't improve
                    break
            else:
                break
        except np.linalg.LinAlgError:
            break
    
    return best_a, best_obj


def kkt_solve_hankel(a0, n_iters=200, active_tol=0.001):
    """
    KKT solver exploiting Hankel structure. Solves Q@a = c*1 where Q is Hankel.
    Uses FFT-based matrix-vector products for speed.
    """
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    for _ in range(n_iters):
        c = fast_convolve(a)
        K_star = get_active_set(c, rel_tol=active_tol)
        m = len(K_star)
        
        # q[k] = 1/m if k in K_star, else 0
        q = np.zeros(2 * n - 1)
        q[K_star] = 1.0 / m
        
        # Q @ a where Q[i,j] = q[i+j]
        # (Q@a)[i] = sum_j q[i+j] * a[j]
        # This is a correlation-like operation
        # (Q@a) = cross-correlation of q and a
        # Compute via FFT:
        # (Q@a)[i] = sum_j q[i+j] * a[j] for i = 0,...,n-1
        # Let's define: r[i] = sum_j q[i+j] * a[j]
        # This is like a "upper triangular" convolution
        # We can compute it as: flip q, convolve with a, then extract
        # Actually: sum_j q[i+j] a[j] = (q * flip(a))[i + n - 1]... let me think
        # 
        # Alternative: just compute directly since n is small
        Qa = np.zeros(n)
        for i in range(n):
            # Qa[i] = sum_j q[i+j] * a[j] for j=0,...,n-1
            # i+j ranges from i to i+n-1
            # q has nonzero entries only at K_star positions
            Qa[i] = np.dot(q[i:i+n], a)
        
        # Now solve: Qa = c_val * 1, sum(a) = 1
        # Using the above Qa computation, we can use an iterative solver
        # Or just build Q and solve directly
        
        # For n <= 1024, direct solve is fine
        Q = np.zeros((n, n))
        for k in K_star:
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = k - idx_i
            Q[idx_i, idx_j] = 1.0 / m
        
        A_aug = np.zeros((n + 1, n + 1))
        A_aug[:n, :n] = Q
        A_aug[:n, n] = -1.0
        A_aug[n, :n] = 1.0
        b_aug = np.zeros(n + 1)
        b_aug[n] = 1.0
        
        try:
            sol = np.linalg.solve(A_aug, b_aug)
            a_candidate = sol[:n]
            if np.all(a_candidate >= -1e-8):
                a_candidate = np.maximum(a_candidate, 0)
                a_candidate /= a_candidate.sum()
                obj = 2.0 * n * fast_convolve(a_candidate).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_candidate
                    a = a_candidate
                else:
                    break
            else:
                # Try NNLS approach: minimize ||Qa - 1||^2 s.t. a>=0, sum(a)=1
                # Use projected gradient
                a_pgc = a.copy()
                for _inner in range(50):
                    grad_pp = 2 * (Q @ a_pgc - 1)
                    a_pgc = project_simplex(a_pgc - 0.01 * grad_pp)
                obj = 2.0 * n * fast_convolve(a_pgc).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_pgc
                    a = a_pgc
                else:
                    break
        except np.linalg.LinAlgError:
            break
    
    return best_a, best_obj


def try_seq(a, label, best_ref):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref
