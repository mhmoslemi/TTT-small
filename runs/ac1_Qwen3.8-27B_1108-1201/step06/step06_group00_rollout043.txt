def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def build_constraint_matrix(f, constraint_indices):
    """Build constraint matrix A where A[k,j] = f[k-j] (0 if out of bounds)."""
    n = len(f)
    kc = len(constraint_indices)
    # A has shape (kc, n)
    # A[k_idx, j] = f[k - j] if 0 <= k-j < n, else 0
    rows = np.zeros((kc, n), dtype=np.float64)
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows[idx, js] = f[k - js]
    return rows

def solve_lp_direction(f, M, conv, constraint_indices, time_limit=10):
    """Solve LP: max sum(g) s.t. A*g <= b, g >= 0.
    Constraint: for each k in constraint_indices, sum_j f[k-j]*g[j] <= (M + conv[k])/2
    """
    n = len(f)
    A_full = build_constraint_matrix(f, constraint_indices)
    b = (M + conv[constraint_indices]) / 2.0
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_full, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def coordinate_refine(f, time_limit=30):
    """Coordinate descent: for each coordinate, find optimal value."""
    f = np.maximum(f, 0.0)
    f /= f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    order = np.random.permutation(n)
    while time.monotonic() - t0 < time_limit:
        for idx in order:
            j = int(idx)
            # Current value
            v = f[j]
            # Try reducing v: move mass to other coordinates
            # The objective changes as:
            # c[k] changes by -dv * f[k-j] for each k (where we decrease f[j] by dv)
            # and +dv * f[k-j'] for coordinates we increase
            
            # Simple approach: try a few values of f[j]
            # For each candidate value v_new, compute the resulting max
            candidates = [v * 0.5, v * 0.75, v * 1.25, v * 1.5, 0.0, v + 0.001, v - 0.001]
            candidates = [max(0, c) for c in candidates]
            
            best_v = v
            best_v_obj = best_obj
            
            for v_cand in candidates:
                if v_cand == v:
                    continue
                f_trial = f.copy()
                delta = v_cand - v
                f_trial[j] = v_cand
                # Distribute the delta evenly to other coordinates
                if abs(delta) > 1e-15:
                    # Add delta/(n-1) to all other coordinates
                    f_trial[np.arange(n) != j] += delta / (n - 1)
                    f_trial = np.maximum(f_trial, 0)
                    s = f_trial.sum()
                    if s < 1e-15:
                        continue
                    f_trial /= s
                    obj = fast_convolve(f_trial).max()
                    if obj < best_v_obj - 1e-12:
                        best_v_obj = obj
                        best_v = v_cand
            
            if best_v != v:
                f[j] = best_v
                delta = best_v - v
                f[np.arange(n) != j] += delta / (n - 1)
                f = np.maximum(f, 0)
                f /= f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                if M < best_obj:
                    best_obj = M
                    best_f = f.copy()
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with corrected LP oracle (proper linearization RHS)
2. Full constraint set for better direction computation
3. Exact line search via adaptive grid + bisection
4. Multi-length exploration with frequency-domain restarts
5. Active-peak coordinate refinement for final polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, constraint_indices):
    """Build dense constraint matrix A where A[k_idx, j] = f[k-j] (0 if out of bounds)."""
    n = len(f)
    kc = len(constraint_indices)
    rows = np.zeros((kc, n), dtype=np.float64)
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows[idx, js] = f[k - js]
    return rows


def solve_lp_direction(f, M, conv, constraint_indices, time_limit=8.0):
    """
    Solve LP: max sum(g) s.t. A*g <= b, g >= 0.
    Correct linearization: (Q_k f)^T g <= (M + c_f[k])/2 for each constraint k.
    """
    n = len(f)
    A = build_constraint_matrix(f, constraint_indices)
    # Correct RHS from linearization of c[k](a) <= M at point f:
    # c_f[k] + 2*(Q_k f)^T*(g-f) <= M  =>  (Q_k f)^T * g <= (M + c_f[k])/2
    b = (M + conv[constraint_indices]) / 2.0
    c_vec = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    # Fallback: solve with just top-peak constraint
    try:
        k_max = int(np.argmax(conv))
        idx = np.array([k_max])
        A_small = build_constraint_matrix(f, idx)
        b_small = np.array([(M + conv[k_max]) / 2.0])
        res = linprog(c_vec, A_ub=A_small, b_ub=b_small, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 3.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def line_search(f, g, M_init, n_evals_max=40):
    """Adaptive line search: coarse grid then local refinement."""
    def obj_at(alpha):
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return float('inf')
        ft /= s
        return fast_convolve(ft).max()

    best_alpha = 0.0
    best_val = M_init
    n_used = 0

    # Coarse grid
    for i in range(1, 21):
        alpha = i / 20.0
        val = obj_at(alpha)
        n_used += 1
        if val < best_val:
            best_val = val
            best_alpha = alpha
        if n_used >= n_evals_max:
            return best_alpha, best_val

    # Fine refinement around best
    lo = max(0.0, best_alpha - 0.05)
    hi = min(1.0, best_alpha + 0.05)
    for i in range(1, 21):
        alpha = lo + (hi - lo) * i / 20.0
        if alpha <= lo or alpha >= hi:
            continue
        val = obj_at(alpha)
        n_used += 1
        if val < best_val:
            best_val = val
            best_alpha = alpha
        if n_used >= n_evals_max:
            break

    return best_alpha, best_val

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]

        k_top = min(100, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks /= w_sum
        else:
            w_peaks = np.ones(k_top) / k_top

        grad = np.zeros(n)
        for idx, k in enumerate(peak_idx):
            k = int(k)
            wi = w_peaks[idx]
            if wi < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += wi * 2.0 * f[k - i_range]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_n = grad / gnorm

        step = lr
        improved = False
        for _ in range(12):
            f_new = project_simplex(f - step * grad_n)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
