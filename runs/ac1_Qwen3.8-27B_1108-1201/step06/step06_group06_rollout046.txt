                a_new = a_new / a_new.sum()
                obj = fast_convolve(a_new).max()
                if obj < best_obj:
                    best_obj = obj
                a = a_new
                theta = res.x
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
        except Exception as e:
            print(f"  L-BFGS error at T={T}: {e}")
    
    return best_a, best_obj
```

This is a clean approach. The softmax parametrization ensures the simplex constraint is satisfied automatically, and L-BFGS-B provides fast convergence.

However, there's a subtlety: the smoothed objective depends on p, which depends on a. So as we optimize, p changes. The continuation method (decreasing T) handles this: at each T, we optimize for the current p (frozen), then update p and optimize again at a lower T.

This is similar to the "mirror descent" or "follow-the-regularized-leader" approach.

Let me also think about the **temperature schedule**. Starting at T=0.1 means we're weighting all lags roughly equally. As T decreases, we focus more on the peak lags. At T=0.0001, we're essentially optimizing the max directly.

The number of T steps and the rates of decrease are important hyperparameters. Let me make this adaptive.

Now, let me also think about **multi-n exploration**. The previous code tries n = 256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 2048. I should try similar values but with the new optimization method.

One important consideration: for larger n, the L-BFGS-B optimization is more expensive (more variables). So I should allocate less time to larger n values.

Let me now put together the full solution:

```python
"""
Optimized search for minimum peak self-convolution step function.
Uses L-BFGS-B in log-space (softmax parametrization) with temperature
continuation, combined with multi-n exploration and local polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_smoothed(a, p):
    """Compute gradient of sum_k p[k]*(a*a)[k] with respect to a.
    grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
    """
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(a, p[j:j + n])
    return grad


def softmax_lbfgs(a0, time_limit, T_schedule=None):
    """Optimize using L-BFGS-B in log-space with temperature continuation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = np.where(a > 1e-15, a, 1e-15)
    a = a / a.sum()
    n = len(a)
    
    theta = np.log(a)
    
    if T_schedule is None:
        T_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for i, T in enumerate(T_schedule):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        # Compute p at current a
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        def func_theta(theta_vec):
            logits = theta_vec - theta_vec.max()
            a_vec = np.exp(logits)
            a_vec = a_vec / a_vec.sum()
            return np.dot(p, fast_convolve(a_vec))
        
        def grad_theta(theta_vec):
            logits = theta_vec - theta_vec.max()
            a_vec = np.exp(logits)
            a_vec = a_vec / a_vec.sum()
            g = compute_gradient_smoothed(a_vec, p)
            mean_g = np.dot(g, a_vec)
            return a_vec * (g - mean_g)
        
        # Allocate time proportional to remaining
        maxiter = max(50, int(200 * remaining / time_limit))
        
        try:
            res = minimize(
                func_theta, theta,
                jac=grad_theta,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': maxiter * 5}
            )
            if res.fun < 1e6:
                logits = res.x - res.x.max()
                a_new = np.exp(logits)
                a_new = a_new / a_new.sum()
                obj = fast_convolve(a_new).max()
                a = a_new
                theta = res.x.copy()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                    print(f"    T={T}: obj={obj:.8f} (improved)")
                else:
                    print(f"    T={T}: obj={obj:.8f}")
        except Exception as e:
            print(f"    L-BFGS error at T={T}: {e}")
            break
    
    return best_a, best_obj


def local_search(a0, time_limit, seed=42):
    """Local search with small perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        improved = False
        for scale in [0.0001, 0.0003, 0.001, 0.003, 0.01]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        if not improved and it > 500:
            break
    
    return best_a, best_obj
