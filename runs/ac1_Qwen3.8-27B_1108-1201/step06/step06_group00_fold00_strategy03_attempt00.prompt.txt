def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:

    # Check each element in the list for validity
    for x in sequence:

        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    sequence = [min(1000.0, x) for x in sequence]

    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, constraint_indices=None):
    """Build sparse constraint matrix for LP. Shape: (num_constr, n)."""
    n = len(f)
    if constraint_indices is None:
        constraint_indices = np.arange(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, top_k=30, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-K peaks, g >= 0."""
    n = len(f)
    conv = fast_convolve(f)
    peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    A = build_lp_matrix(f, peak_idx)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_subgradient(f, active_peaks, weights=None):
    """Compute weighted sum of gradients of (f*f)[k] for k in active_peaks.
    Gradient of (f*f)[k] wrt f[i] = 2*f[k-i] if i in valid range, else 0.
    """
    n = len(f)
    if weights is None:
        weights = np.ones(len(active_peaks)) / len(active_peaks)
    grad = np.zeros(n)
    for idx, k in enumerate(active_peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad


def lse_weights(conv, beta):
    """Compute LSE softmax weights on the convolution peaks."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total

def subgradient_optimize(f0, time_limit, beta_start=10.0, beta_end=5000.0, top_k=60):
    """Subgradient method with LSE smoothing and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 2.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        conv = fast_convolve(f)
        w_full = lse_weights(conv, beta)

        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_peaks = w_full[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top

        grad = compute_subgradient(f, peak_idx, w_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            no_improve += 100
            break
        grad_normed = grad / gnorm

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(10):
            f_new = project_simplex(f - step * grad_normed)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.005, 10.0 / np.sqrt(n))
                break
            step *= 0.55

        if not improved:
            no_improve += 1
            if no_improve >= 12:
                lr *= 0.88
            if no_improve > 70:
                f = best_f.copy()
                lr = max(2.0 / np.sqrt(n), 0.005)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0
                rng = np.random.default_rng(it * 31 + 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)

        if beta < beta_end:
            beta *= 1.004

    return best_f, best_obj


def peak_pair_adjust(f, n_pairs=50, time_limit=30.0):
    """Try to reduce peaks by adjusting pairs (i, j) simultaneously."""
    f = np.maximum(f0 if 'f0' in dir() else f, 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Get top peaks
    top_peaks = np.argpartition(conv, -min(20, len(conv)))[-min(20, len(conv)):]

    for _ in range(n_pairs):
        if time.monotonic() - t0 > time_limit:
            break
        # Pick a random peak
        k = int(top_peaks[np.random.randint(len(top_peaks))])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo >= hi:
            continue

        # The peak (f*f)[k] = sum_{j=lo}^{hi} f[j]*f[k-j]
        # Try to reduce f[j] for the j that maximizes f[j]*f[k-j]
        vals = f[lo:hi+1] * f[k - np.arange(lo, hi+1)]
        j_max = lo + np.argmax(vals)
        j_other = k - j_max
        if j_other < 0 or j_other >= n or j_other == j_max:
            continue

        # Move mass from j_max to a random position far from the peak
        candidates = np.random.choice(n, size=min(10, n), replace=False)
        for j_dest in candidates:
            if j_dest == j_max or j_dest == j_other:
                continue
            # Check if moving mass to j_dest would increase other peaks
            delta = f[j_max] * 0.1  # move 10% of mass
            if delta < 1e-15:
                continue
            f_trial = f.copy()
            f_trial[j_max] -= delta
            f_trial[j_dest] += delta
            if f_trial[j_max] < 0:
                continue
            f_trial = np.maximum(f_trial, 0)
            f_trial /= f_trial.sum()
            conv_trial = fast_convolve(f_trial)
            if conv_trial.max() < best_obj - 1e-15:
                best_obj = conv_trial.max()
                best_f = f_trial.copy()
                f = f_trial.copy()
                break

    return best_f, best_obj


def freq_perturb(f, rng, strength=0.05):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
