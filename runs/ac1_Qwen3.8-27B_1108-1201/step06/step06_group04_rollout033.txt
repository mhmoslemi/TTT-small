    return np.correlate(w, a, mode='valid')

    Q is n x n symmetric PSD (or at least numerically stable).

    a = np.maximum(a_init.copy(), 0.0)

    if s < 1e-15:

    else:
        a = a / s

    if not support:

            break
        if len(support) == 0:
            return np.ones(n) / n
        if len(support) == 1:
            result = np.zeros(n)
            result[list(support)[0]] = 1.0
            return result

        k = len(S)

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_simplex_sym(v, n_full):
    """Project v (length n_full/2) onto simplex, then expand to symmetric sequence."""
    n_half = len(v)
    v_proj = project_simplex(v)
    a = np.zeros(n_full)
    a[:n_half] = v_proj
    a[n_half:] = v_proj[::-1] if n_full % 2 == 0 else v_proj[:-1][::-1]
    if n_full % 2 == 1:
        a[n_half - 1] = v_proj[n_half - 1] / 2.0  # center element
        a[n_half + 1:] = v_proj[:n_half - 1][::-1]
        a[:n_half - 1] = v_proj[:n_half - 1]
        # Fix: for odd n, the middle element appears once
        a = np.zeros(n_full)
        half = (n_full - 1) // 2
        a[:half] = v_proj[:half]
        a[half] = v_proj[half]
        a[half + 1:] = v_proj[:half][::-1]
    else:
        half = n_full // 2
        a[:half] = v_proj[:half]
        a[half:] = v_proj[:half][::-1]
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def hankel_mv(w, a):
    """Compute W@a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    n = len(a)
    return np.correlate(w, a, mode='valid')


def fista_smoothed(a0, T, time_limit, lr=None, momentum_init=0.9, verbose=False):
    """
    FISTA on smoothed objective f_T(a) = T * log(sum exp(conv_a[k]/T))
    subject to a on simplex.
    Returns best a found.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    # Estimate step size
    conv = fast_convolve(a)
    # L ≈ 2 * max row sum of "effective" matrix; use empirical bound
    # For smoothed objective, grad = 2*hankel_mv(p, a) where p = softmax(conv/T)
    # Lipschitz const ~ 2 * max_i sum_j |w[i+j]| for the smoothed weights
    # Use a conservative step
    if lr is None:
        # Estimate: the Hessian of a^T H_k a has eigenvalues bounded by 2*max|w|*n
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        row_sums = np.array([np.sum(p[i:i+n]) for i in range(n)])
        L_est = 2.0 * np.max(row_sums)
        lr = 1.0 / max(L_est, 1e-10)
        lr = min(lr, 1.0 / n * 2.0)  # cap for stability

    v = np.zeros(n)
    t_k = 1.0
    y = a.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        # Nesterov lookahead
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = a + beta * v

        # Compute smoothed gradient
        conv_y = fast_convolve(y)
        scaled = conv_y / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        grad = 2.0 * hankel_mv(p, y)

        # Gradient step
        z = y - lr * grad
        a_new = project_simplex(z)

        # Update
        v = a_new - y
        a = a_new
        t_k = t_new

        if obj < best_obj - 1e-14:

            no_improve = 0
        else:

def primal_dual_exp(a0, time_limit, eta_init=0.5, T_smooth=0.01,
                    use_symmetric=False, verbose=False):

    Primal: FISTA on smoothed objective with current w.

    s = a.sum()
    if s < 1e-15:

        a = a / s
    n = len(a)
    n_conv = 2 * n - 1

    w = np.ones(n_conv) / n_conv
    eta = eta_init

    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        # Compute convolution
        conv = fast_convolve(a)
        cmax = conv.max()

        if cmax < best_obj - 1e-14:
            best_obj = cmax
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        scaled = conv / cmax  # in [0, 1]
        w = w * np.exp(eta * (scaled - 1.0))  # shift so max gets weight 1
        w = w / w.sum()

        qp_time = min(3.0, remaining * 0.4)

            else:
                a_candidate = a_qp
        else:
            a_candidate = a_qp

            a_new2 = (1.0 - alpha2) * a + alpha2 * a_candidate
            a_new2 = np.maximum(a_new2, 0.0)
            sn2 = a_new2.sum()
            if sn2 > 1e-15:
                a_new2 = a_new2 / sn2
            if fast_convolve(a_new2).max() < best_obj:
                a_new = a_new2
            else:
                a_new = a

        a = a_new

        # Adaptive eta
        if no_improve > 20:
            eta *= 1.1
        if no_improve > 60:
            eta = max(eta * 0.7, 0.1)
        if no_improve > 150:
            # Restart with random perturbation
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            w = np.ones(n_conv) / n_conv
            eta = eta_init
            no_improve = 0

        if verbose and it % 30 == 0:
            print(f"      PD_exp it={it}, obj={best_obj:.10f}, eta={eta:.4f}, no_imp={no_improve}")

    return best_a, best_obj


def peak_shave(a0, time_limit, verbose=False):
    """
    Direct peak shaving: find the peak lag, try to reduce it by
    moving mass from contributing positions to non-contributing ones.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        # Positions contributing to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        non_contrib = np.setdiff1d(np.arange(n), contrib)

        if len(contrib) == 0:
            break

        improved = False

        # Try shifting mass from largest contributor to random non-contributors
        for _ in range(min(10, len(contrib))):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            # Pick a contributor (weighted by value)
            probs = a[contrib] / a[contrib].sum()
            idx_from = contrib[np.random.choice(len(contrib), p=probs)]

            if len(non_contrib) > 0:
                idx_to = np.random.choice(non_contrib)
            else:
                # All positions contribute; try redistributing among contributors
                idx_to = np.random.choice(contrib)
                if idx_to == idx_from:
                    continue

            shift = a[idx_from] * np.random.uniform(0.02, 0.15)
            if shift < 1e-12:
                continue

            a_trial = a.copy()
            a_trial[idx_from] -= shift
            a_trial[idx_to] += shift
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()

            obj = fast_convolve(a_trial).max()
            if obj < best_obj - 1e-14:
                a = a_trial
                best_a = a.copy()
                best_obj = obj
                improved = True

        # Also try: reduce all contributors slightly, boost non-contributors
        if not improved and len(non_contrib) > 0:
            total_contrib = a[contrib].sum()
            frac = np.random.uniform(0.01, 0.05)
            a_trial = a.copy()
            a_trial[contrib] *= (1.0 - frac)
            redistribution = total_contrib * frac
            nc_weights = a[non_contrib]
            if nc_weights.sum() > 1e-15:
                a_trial[non_contrib] += redistribution * nc_weights / nc_weights.sum()
            else:
                a_trial[non_contrib] += redistribution / len(non_contrib)
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            obj = fast_convolve(a_trial).max()
            if obj < best_obj - 1e-14:
                a = a_trial
                best_a = a.copy()
                best_obj = obj
                improved = True

        if not improved and it > 500:
            break
        if not improved and it % 100 == 0:
            # Bigger perturbation
            rng = np.random.default_rng((it * 31 + k_peak) % (2**31))
            for _ in range(5):
                delta = rng.standard_normal(n) * 0.003
                a_trial = project_simplex(a + delta)
                obj = fast_convolve(a_trial).max()
                if obj < best_obj - 1e-14:
                    a = a_trial
                    best_a = a.copy()
                    best_obj = obj
                    improved = True
                    break

    return best_a, best_obj


def continuation_fista(a0, time_limit, T_start=0.5, T_end=0.001, n_stages=8, verbose=False):
    """
    Continuation method: gradually decrease T in log-sum-exp smoothing.
    At each stage, run FISTA to convergence.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    T_values = np.geomspace(T_start, T_end, n_stages)

    for stage, T in enumerate(T_values):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        stage_time = remaining * (0.6 + 0.4 * stage / max(n_stages - 1, 1)) / (n_stages - stage)
        stage_time = min(stage_time, remaining - 1.0)

        a_new, obj = fista_smoothed(a, T, time_limit=stage_time)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new

        if verbose:
            print(f"      Continuation stage {stage}, T={T:.6f}, obj={best_obj:.10f}")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Pairwise coordinate descent: try redistributing mass between pairs (i,j).
    Focus on pairs where i is a peak contributor and j is not.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Identify peak contributors
        conv = fast_convolve(a)
        top_k = 3
        peak_lags = np.argsort(conv)[-top_k:]

        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            contrib = np.arange(js, je + 1)
            non_contrib = np.setdiff1d(np.arange(n), contrib)

            # Try a few random pairs
            for _ in range(20):
                if time.monotonic() - t0 > time_limit - 0.3:
                    break
                if len(contrib) < 2:
                    break
                i = rng.integers(0, len(contrib))
                i = contrib[i]
                if len(non_contrib) > 0:
                    j = non_contrib[rng.integers(0, len(non_contrib))]
                else:
                    j = contrib[rng.integers(0, len(contrib))]
                    if j == i:
                        continue

                # Try several shift fractions
                for frac in [0.05, 0.1, 0.2, 0.03]:
                    shift = a[i] * frac
                    if shift < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= shift
                    a_trial[j] += shift
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    obj = fast_convolve(a_trial).max()
                    if obj < best_obj - 1e-14:
                        a = a_trial
                        best_a = a.copy()
                        best_obj = obj
                        break

    return best_a, best_obj


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Symmetric triangular
    tri = 1.0 - 2.0 * np.abs(x - (n - 1) / 2.0) / (n - 1)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    # Cosine window
    cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    inits.append(cos_w / cos_w.sum())

    # Two-peak Gaussian
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.08 * n)) ** 2)
    inits.append((g1 + g2) / (g1 + g2).sum())

    # Chirp magnitude (flat spectrum)
    phase = np.pi * x * x / n
    inits.append(np.abs(np.exp(1j * phase)) / n)  # This is just uniform...
    # Better: use a real chirp-like sequence
    chirp = np.cos(np.pi * x * x / n)
    chirp = np.abs(chirp) + 0.01
    inits.append(chirp / chirp.sum())

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Edge-heavy
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())

    # Alternating (high frequency)
    alt = np.ones(n) * (1.0 + 0.5 * np.cos(2 * np.pi * x / 2.0))
    inits.append(alt / alt.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Multi-n exploration (~250s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        max(128, n_parent // 2),
        n_parent,
        n_parent + n_parent // 4,
        n_parent * 2 if n_parent * 2 <= 3000 else n_parent * 3 // 2,
        512, 768, 1024, 1536,
    ]))
    # Filter to reasonable range
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 3000]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 20:
            break

        # Resample parent to this length
        if n_try == n_parent and parent is not None:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(remaining, 30)
        print(f"  Exploration n={n_try}, t={t_alloc:.0f}s")

        # Continuation FISTA
        a_opt, _ = continuation_fista(start, time_limit=t_alloc * 0.6,
                                       T_start=0.3, T_end=0.002, n_stages=5)
        try_seq(a_opt, f"cont_n{n_try}")

        # Quick primal-dual
        remaining = deadline - time.monotonic() - 950
        if remaining > 10:
            a_opt2, _ = primal_dual_exp(a_opt, time_limit=min(remaining, 15),
                                        eta_init=0.3, T_smooth=0.01)
            try_seq(a_opt2, f"pd_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} ({time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep primal-dual on best (~600s) ===
    print("\n=== Phase 2: Deep Primal-Dual ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple configurations
        configs = [
            {'eta_init': 0.3, 'T_smooth': 0.005, 't_frac': 0.25},
            {'eta_init': 0.5, 'T_smooth': 0.01, 't_frac': 0.25},
            {'eta_init': 0.2, 'T_smooth': 0.002, 't_frac': 0.2},
            {'eta_init': 0.8, 'T_smooth': 0.02, 't_frac': 0.2},
            {'eta_init': 0.4, 'T_smooth': 0.008, 't_frac': 0.2},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 350
            if remaining < 15:
                break
            t_alloc = min(remaining, int(budget_s * cfg['t_frac']))
            print(f"  PD config {ci} (eta={cfg['eta_init']}, T={cfg['T_smooth']}), t={t_alloc:.0f}s")
            a_opt, _ = primal_dual_exp(best_seq, time_limit=t_alloc,
                                        eta_init=cfg['eta_init'],
                                        T_smooth=cfg['T_smooth'])
            try_seq(a_opt, f"pd_deep_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} ({time.monotonic()-t2:.0f}s)")

    # === Phase 3: Continuation + FISTA refinement (~300s) ===
    print("\n=== Phase 3: Continuation FISTA ===")
    t3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Different T ranges
        cont_configs = [
            {'T_start': 0.1, 'T_end': 0.001, 'n_stages': 6},
            {'T_start': 0.05, 'T_end': 0.0005, 'n_stages': 5},
            {'T_start': 0.2, 'T_end': 0.002, 'n_stages': 8},
        ]

        for ci, cfg in enumerate(cont_configs):
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  Continuation {ci} (T: {cfg['T_start']}->{cfg['T_end']}), t={t_alloc:.0f}s")
            a_opt, _ = continuation_fista(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"cont_deep_{ci}")

        # Also try from slightly perturbed start
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            for pi in range(3):
                remaining = deadline - time.monotonic() - 120
                if remaining < 15:
                    break
                noise = rng.standard_normal(n_best) * 0.005
                perturbed = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 40)
                a_opt, _ = continuation_fista(perturbed, time_limit=t_alloc,
                                               T_start=0.05, T_end=0.001, n_stages=5)
                try_seq(a_opt, f"cont_pert_{pi}")

    print(f"  Phase 3 done. Best: {best_score:.10f} ({time.monotonic()-t3:.0f}s)")

    # === Phase 4: Peak shaving + coordinate descent (~250s) ===
    print("\n=== Phase 4: Peak Shaving & Coordinate Descent ===")
    t4 = time.monotonic()

    if best_seq is not None:
        # Peak shaving
        remaining = deadline - time.monotonic() - 150
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print(f"  Peak shave, t={t_alloc:.0f}s")
            a_opt, _ = peak_shave(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_shave")

        # Coordinate descent
        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")

        # More peak shaving on the result
        remaining = deadline - time.monotonic() - 70
        if remaining > 10 and best_seq is not None:
            t_alloc = min(remaining, 40)
            a_opt, _ = peak_shave(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_shave2")

    print(f"  Phase 4 done. Best: {best_score:.10f} ({time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine continuation
        remaining = deadline - time.monotonic() - 30
        if remaining > 15:
            t_alloc = min(remaining, 50)
            print(f"  Fine continuation, t={t_alloc:.0f}s")
            a_opt, _ = continuation_fista(best_seq, time_limit=t_alloc,
                                           T_start=0.01, T_end=0.0001, n_stages=5)
            try_seq(a_opt, "fine_cont")

        # Tiny perturbations + quick FISTA
        for _ in range(4):
            remaining = deadline - time.monotonic() - 15
            if remaining < 5:
                break
            noise_scale = 0.001 + 0.0005 * _
            pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_alloc = min(remaining, 15)
            a_opt, _ = fista_smoothed(pert, T=0.001, time_limit=t_alloc)
            try_seq(a_opt, f"tiny_pert_{_}")

        # One more peak shave
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            a_opt, _ = peak_shave(best_seq, time_limit=min(remaining, 20))
            try_seq(a_opt, "final_peak")

    print(f"  Phase 5 done. Best: {best_score:.10f} ({time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
