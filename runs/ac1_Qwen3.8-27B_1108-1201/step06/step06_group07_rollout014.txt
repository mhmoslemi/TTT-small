**Approach 6: Frank-Wolfe (Conditional Gradient) method**

Since we're minimizing a convex function over the simplex, Frank-Wolfe is applicable:

Key insight: for non-negative a, this is a convex problem (max of convex quadratics).

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_max(a):
    """Compute max of auto-convolution and its location."""
    conv = fast_convolve(a)
    k_peak = int(np.argmax(conv))
    return conv.max(), k_peak, conv


def gradient_of_conv_at_k(a, k_peak):
    """
    Compute the gradient of c_k = sum_i a_i * a_{k-i} with respect to a.
    Result: grad[j] = 2 * a[k-j] if 0 <= k-j <= n-1, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    # j ranges from max(0, k-(n-1)) to min(n-1, k)
    j_start = max(0, k_peak - (n - 1))
    j_end = min(n - 1, k_peak)
    if j_start <= j_end:
        # grad[j] = 2 * a[k_peak - j] for j in [j_start, j_end]
        # k_peak - j ranges from k_peak - j_end to k_peak - j_start
        indices = np.arange(j_start, j_end + 1)
        grad[indices] = 2.0 * a[k_peak - indices]
    return grad
