"""

- Nesterov-accelerated multiplicative (mirror) descent with multi-peak subgradient

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_row(f, k, n):
    """Return the row of the convolution matrix for peak k."""
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    row = np.zeros(n)
    for j in range(js, je + 1):
        row[j] = f[k - j]
    return row


def compute_multi_peak_subgradient(a, n_peaks=8, temperature=0.05):
    """Compute weighted subgradient from top peaks."""
    n = len(a)
    conv = fast_convolve(a)
    if n_peaks < len(conv):
        peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
    else:
        peaks = np.arange(len(conv))
    pv = conv[peaks]
    pv_max = pv.max()
    w = np.exp((pv - pv_max) / (temperature * 0.1))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(peaks)) / len(peaks)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]
    return grad, conv


def multiplicative_nesterov(a0, time_limit, n_peaks=8, T_init=0.0004,
                            lr_init=0.015, mom_init=0.9, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)

        a = a / s
    n = len(a)

    best_a = a.copy()
    v = np.zeros(n)

    lr = lr_init
    mom = mom_init

    it = 0

        it += 1

            break

        grad, conv = compute_multi_peak_subgradient(a, n_peaks, T)
        obj = conv.max()

        if a_look.sum() < 1e-15:
            a_look = a
        grad_look, _ = compute_multi_peak_subgradient(a_look, n_peaks, T)

        # Backtracking line search with multiplicative update
        best_candidate = a
        best_val = obj
        eta = lr
        for _bt in range(8):
            # Multiplicative update (mirror descent step)
            log_ratio = -eta * grad_look
            log_ratio -= log_ratio.max()  # stability
            mult = a * np.exp(log_ratio)

            if ms < 1e-15:
                eta *= 0.5
                continue
            candidate = mult / ms
            val = fast_convolve(candidate).max()

                break

                break

                candidate = project_simplex(a + frac * (-lr * grad_look))
                val = fast_convolve(candidate).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_candidate = candidate
                    break

        # Update momentum
        v = mom * v - lr * grad_look
        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0

            lr = lr_init

def block_coordinate_descent(a0, time_limit, block_size=16, verbose=False):

    rng = np.random.default_rng(12345)

        it += 1

            break

        conv = fast_convolve(a)

        n_peaks = min(6, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved_this_pass = False
        # Shuffle block order
        block_order = rng.permutation(n_blocks)

        for bi in block_order:
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            start_idx = bi * actual_bs

                continue
            block = a[start_idx:end_idx].copy()
            block_sum = block.sum()
            if block_sum < 1e-12:
                continue
            block_norm = block / block_sum

            # For each peak, compute the "cost" of each position in the block
            cost = np.zeros(end_idx - start_idx)

            # Redistribute: move mass from high-cost to low-cost positions
            # Try exponential redistribution
            T_block = 0.1 * block.max() + 1e-10
            new_block = np.exp(-cost / T_block)
            new_block = np.maximum(new_block, 1e-10)

                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    improved_this_pass = True
                    break

        if not improved_this_pass:
            # Perturbation

                rng2 = np.random.default_rng(it * 31)
                idx1 = rng2.integers(0, n)

                        delta = a[idx1] * frac

                        candidate[idx1] -= delta
                        candidate[idx2] += delta

                        val = fast_convolve(candidate).max()

def coordinate_polish(a0, time_limit, verbose=False):
    """Fine-grained coordinate descent for final polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()

    while time.monotonic() - t0 < time_limit:

        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Find the most "offending" pair
            contribs = []
            for j in range(js, je + 1):
                contribs.append((a[j] * a[k - j], j, k - j))
            contribs.sort(reverse=True)

            for (_, j1, j2) in contribs[:4]:

                    continue

                src = j1 if a[j1] >= a[j2] else j2
                if a[src] < 1e-8:
                    continue

                # Target: position where it contributes least to this peak
                best_target = -1
                best_gain = 0
                for _tr in range(min(20, n)):

                    if tgt == src:
                        continue

                    old_c = a[src] * a[k - src] if 0 <= k - src < n else 0

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using cubic-like interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):

    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Asymmetric (slightly skewed)
    w = np.exp(-0.3 * (x / (n - 1) - 0.4) ** 2 * 4)
    starts.append(w / w.sum())

    # Two-peak
    g1 = np.exp(-0.5 * ((x - 0.35 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.65 * n) / (0.1 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n) * 0.5))

    return starts

    lengths = sorted(set([n_parent, max(256, n_parent - 128), n_parent + 128,
                          512, 640, 768, 896, 1024]))
    lengths = [l for l in lengths if 128 <= l <= 2048]

    for n_try in lengths:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 10:
            break

        starts = generate_starts(n_try, parent, rng)
        for si, start in enumerate(starts[:3]):
            remaining = deadline - time.monotonic() - 1150
            if remaining < 8:
                break
            t_alloc = min(remaining, 30)
            a_opt, _ = multiplicative_nesterov(start, time_limit=t_alloc,
                                               n_peaks=6, T_init=0.0003,
                                               lr_init=0.02)
            try_seq(a_opt, f"nest_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep optimization (800s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Long Nesterov run
        remaining = deadline - time.monotonic() - 700
        if remaining > 30:
            t_alloc = min(remaining, 300)
            print(f"  Deep Nesterov n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_nesterov(best_seq, time_limit=t_alloc,
                                               n_peaks=10, T_init=0.0002,
                                               lr_init=0.012, mom_init=0.92,
                                               verbose=True)
            try_seq(a_opt, "deep_nesterov")

        # FW runs
        for fw_i in range(3):
            remaining = deadline - time.monotonic() - 400
            if remaining < 20:
                break
            pert_scale = [0.001, 0.005, 0.01][fw_i]
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * pert_scale)
            t_alloc = min(remaining, 120)
            print(f"  FW run {fw_i}, pert={pert_scale}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_deep_{fw_i}")

        # Another Nesterov with different params
        remaining = deadline - time.monotonic() - 250
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print(f"  Nesterov v2, t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_nesterov(best_seq, time_limit=t_alloc,
                                               n_peaks=16, T_init=0.001,
                                               lr_init=0.008, mom_init=0.85)
            try_seq(a_opt, "nesterov_v2")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Block coordinate descent (300s) ===
    print("\n=== Phase 3: Block CD ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        for bs in [16, 32]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break
            t_alloc = min(remaining, 120)
            print(f"  Block CD bs={bs}, t={t_alloc:.0f}s")
            a_opt, _ = block_coordinate_descent(best_seq, time_limit=t_alloc,
                                                block_size=bs, verbose=True)
            try_seq(a_opt, f"bcd_{bs}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Coordinate polish (200s) ===
    print("\n=== Phase 4: Coordinate Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 150)
            print(f"  Coord polish, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "coord_polish")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final micro-polish (remaining) ===
    print("\n=== Phase 5: Final Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        polish_rounds = 6
        for ri in range(polish_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if ri % 2 == 0:
                noise = rng.standard_normal(n_best) * 0.001
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 20)
                a_opt, _ = multiplicative_nesterov(start, time_limit=t_alloc,
                                                   n_peaks=8, T_init=0.0001,
                                                   lr_init=0.005, mom_init=0.9)
                try_seq(a_opt, f"polish_nest_{ri}")
            else:
                noise = rng.standard_normal(n_best) * 0.0008
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 15)
                a_opt, _ = fw_optimize(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{ri}")

        # Final coordinate polish
        remaining = deadline - time.monotonic()
        if remaining > 8:
            t_alloc = min(remaining, 20)
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_coord")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
