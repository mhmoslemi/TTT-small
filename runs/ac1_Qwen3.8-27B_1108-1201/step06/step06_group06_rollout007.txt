def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:

"""
Game-theoretic QP-based minimax optimization for minimum peak self-convolution.
Uses multiplicative weights (Hedge) for the dual maximizer and a QP subproblem
solved via projected gradient for the primal minimizer. Combined with exact line
search, n-adaptation, structured restarts, and local search polish.
"""

import numpy as np
from scipy.optimize import minimize as sp_minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, v):
    """Compute P @ v where P[i,j] = p[i+j] (Hankel matrix, zero-padded)."""
    n = len(v)
    n_p = len(p)
    result = np.zeros(n)
    for i in range(n):
        end = min(n, n_p - i)
        if end > 0:
            result[i] = np.dot(v[:end], p[i:i + end])
        else:
            result[i] = 0.0
    return result


def solve_qp_pgradient(p, a_init, n_inner=50, lr_init=0.1):
    """
    Solve: min a^T P a s.t. sum(a)=1, a>=0
    where P[i,j] = p[i+j] (Hankel matrix).
    Uses projected gradient with backtracking.
    """
    a = a_init.copy()
    n = len(a)
    lr = lr_init
    best_a = a.copy()
    best_val = float(a @ hankel_matvec(p, a))

    for it in range(n_inner):
        grad = 2.0 * hankel_matvec(p, a)
        # Try decreasing learning rates
       Improvement_found = False
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            candidate = project_simplex(a - lr_try * grad)
            val = candidate @ hankel_matvec(p, candidate)
            if val < best_val - 1e-15:
                a = candidate
                best_val = val
                best_a = a.copy()
                Improvement_found = True
                break
        if not Improvement_found:
            lr *= 0.5
            if lr < 1e-8:
                break
        else:
            lr = min(lr * 1.05, 1.0)

    return best_a


def solve_qp_active_set(p, a_init, tol=1e-10, max_iter=200):
    """
    Solve: min a^T P a s.t. sum(a)=1, a>=0
    using active set method. P[i,j] = p[i+j].
    """
    n = len(a_init)
    a = a_init.copy()
    support = np.arange(n)
    
    for it in range(max_iter):
        s = len(support)
        if s == 0:
            return a_init.copy()
        
        # Build P submatrix on support
        # P_sub[i,j] = p[support[i] + support[j]]
        P_sub = np.zeros((s, s))
        pi = support[:, None] + support[None, :]
        mask = pi < len(p)
        P_sub = np.where(mask, p[pi], 0.0)
        
        # Add regularization for numerical stability
        P_sub += 1e-12 * np.eye(s)
        
        # Solve: minimize 0.5 * a_sub^T P_sub a_sub s.t. sum(a_sub) = 1, a_sub >= 0
        # KKT: P_sub @ a_sub + lam = 0 (for active variables)
        # So a_sub = -lam * P_sub^{-1} @ 1
        # sum(a_sub) = -lam * 1^T P_sub^{-1} 1 = 1
        # lam = -1 / (1^T P_sub^{-1} 1)
        
        try:
            ones = np.ones(s)
            sol = np.linalg.solve(P_sub, ones)
            lam = -1.0 / np.dot(ones, sol)
            a_sub = lam * sol
            
            # Check non-negativity
            if np.min(a_sub) >= -tol:
                a_sub = np.maximum(a_sub, 0.0)
                a_sub = a_sub / a_sub.sum()
                a = np.zeros(n)
                a[support] = a_sub
                return a
            else:
                # Remove the most negative component
                worst = np.argmin(a_sub)
                support = np.delete(support, worst)
        except np.linalg.LinAlgError:
            # Fallback to projected gradient
            return solve_qp_pgradient(p, a_init, n_inner=50)
    
    return a


def game_qp_iteration(a_init, time_limit, T_init=0.01, eta_init=1.0, 
                      n_qp_inner=30, verbose=False):
    """
    Game-theoretic iteration:
    - Dual (maximizer): multiplicative weights on convolution peaks
    - Primal (minimizer): solve QP min a^T P(a) a s.t. simplex
    """
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    T = T_init
    eta = eta_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Compute dual distribution p
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Solve primal QP
        a_qp = solve_qp_active_set(p, a)
        if a_qp is None or not np.isfinite(a_qp).all():
            a_qp = solve_qp_pgradient(p, a, n_inner=n_qp_inner)
        
        a_qp = np.maximum(a_qp, 0.0)
        a_qp = a_qp / a_qp.sum()
        
        # Line search between a and a_qp
        best_alpha = 0.0
        best_line_val = conv.max()
        alphas = np.linspace(0.1, 1.0, 20)
        for alpha in alphas:
            a_trial = (1.0 - alpha) * a + alpha * a_qp
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            val = fast_convolve(a_trial).max()
            if val < best_line_val:
                best_line_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            a = (1.0 - best_alpha) * a + best_alpha * a_qp
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adapt temperature and learning rate
        if no_improve > 20:
            T *= 1.1
            eta *= 0.95
        if no_improve > 80:
            T *= 0.5
            eta = eta_init
        if no_improve > 200:
            rng = np.random.default_rng(it * 7919 + 13)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            eta = eta_init
            no_improve = 0
    
    return best_a, best_obj


def top_k_subgradient(a0, time_limit, K=3, lr_init=0.01):
    """
    Subgradient descent targeting top-K peaks of the convolution.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init
    it = 0
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        # Get top K peak locations
        top_k = np.argsort(conv)[::-1][:K]
        top_k = sorted(top_k)
        
        # Compute combined gradient
        grad = np.zeros(n)
        for k in top_k:
            weight = conv[k] / conv.max()
            for j in range(n):
                g = 0.0
                if 0 <= k - j < n:
                    g += a[k - j]
                if 0 <= j - k < n:
                    g += a[j - k]
                grad[j] += weight * g
        
        # Line search
        improved = False
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - lr_try * grad)
            val = fast_convolve(candidate).max()
            if val < best_obj - 1e-15:
                a = candidate
                best_obj = val
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.9
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

    print(f"  Phase 3 best: {best_score:.8f} (t={time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 4: Deep game-QP on best ===
    print("\n=== Phase 4: Deep Game QP ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    if remaining > 80:
        t_alloc = min(remaining * 0.25, 300)
        print(f"  Deep game QP, n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = game_qp_iteration(best_seq, time_limit=t_alloc,
                                      T_init=0.001, eta_init=0.5, n_qp_inner=30)
        best_seq, best_score = try_seq(a_opt, "deep_game", (best_seq, best_score))
    
    # Perturbed restarts
    for scale in [0.001, 0.003, 0.008, 0.02]:
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc2 = min(remaining * 0.12, 80)
        a_opt2, _ = game_qp_iteration(f_pert, time_limit=t_alloc2,
                                       T_init=0.003, eta_init=1.0, n_qp_inner=20)
        best_seq, best_score = try_seq(a_opt2, f"pert_{scale}", (best_seq, best_score))
    
    print(f"  Phase 4 best: {best_score:.8f} (t={time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 5: Local search polish ===
    print("\n=== Phase 5: Local Search ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    if remaining > 40:
        t_alloc = min(remaining * 0.30, 150)
        print(f"  Local search, t={t_alloc:.0f}s")
        a_opt, _ = locality_search(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "local1", (best_seq, best_score))
    
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc2 = min(remaining * 0.20, 100)
        a_opt2, _ = locality_search(best_seq, time_limit=t_alloc2, seed=99999)
        best_seq, best_score = try_seq(a_opt2, "local2", (best_seq, best_score))
    
    # Fine subgradient
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc3 = min(remaining * 0.20, 80)
        a_opt3, _ = top_k_subgradient(best_seq, time_limit=t_alloc3, K=10, lr_init=0.001)
        best_seq, best_score = try_seq(a_opt3, "fine_topk", (best_seq, best_score))
    
    print(f"  Phase 5 best: {best_score:.8f} (t={time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 6: Final micro-optimization ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()
    
    # Very fine local search
    if remaining > 20:
        t_alloc = min(remaining * 0.3, 60)
        a_opt, _ = locality_search(best_seq, time_limit=t_alloc, seed=77777)
        best_seq, best_score = try_seq(a_opt, "micro_local", (best_seq, best_score))
    
    # Tiny perturbations + quick game step
    remaining = deadline - time.monotonic()
    for _ in range(3):
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        noise = rng.standard_normal(n_best) * 0.0003
        f_pert = project_simplex(best_seq + noise)
        t_alloc2 = min(remaining * 0.2, 15)
        a_opt2, _ = game_qp_iteration(f_pert, time_limit=t_alloc2,
                                       T_init=0.001, eta_init=0.3, n_qp_inner=15)
        best_seq, best_score = try_seq(a_opt2, "micro_game", (best_seq, best_score))
    
    # Coordinate refinement
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_local = best_seq.copy()
        t_alloc3 = min(remaining * 0.3, 20)
        a_local, _ = locality_search(a_local, time_limit=t_alloc3, seed=43210)
        best_seq, best_score = try_seq(a_local, "coord_final", (best_seq, best_score))
    
    print(f"  Phase 6 best: {best_score:.8f} (t={time.monotonic()-t_phase:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
