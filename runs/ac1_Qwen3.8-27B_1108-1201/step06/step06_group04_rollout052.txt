import numpy as np

from scipy.optimize import linprog
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(w, a):
    return np.correlate(w, a, mode='valid')

    a = a / a.sum()

        if time.monotonic() - t0 > time_limit - 0.1:

        cmax = conv.max()
        
        # Build LP
        # Variables: d (n), t (1)

        # Subject to:

        #   sum(d) = 0
        #   -a[i] <= d[i] <= delta

            i_start = max(0, k - n + 1)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    return np.correlate(w, a, mode='valid')


def lp_direction(a, delta):
    """Solve LP to find descent direction considering all peaks.
    Returns (d, t_opt) or (None, None) on failure."""
    n = len(a)
    conv = fast_convolve(a)
    cmax = conv.max()
    
    num_vars = n + 1
    c = np.zeros(num_vars)
    c[-1] = 1.0
    
    num_constr = (2 * n - 1) + n + n
    A_ub = np.zeros((num_constr, num_vars))
    b_ub = np.zeros(num_constr)
    
    row = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[row, i] = 2.0 * a[k - i]
        A_ub[row, n] = -1.0
        b_ub[row] = -conv[k]
        row += 1
    
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = delta
        row += 1
    
    for i in range(n):
        A_ub[row, i] = -1.0
        b_ub[row] = a[i]
        row += 1
    
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None

def multiplicative_weights(a0, time_limit, eta_init=0.1, verbose=False):
    """Mirror descent with exponential weights update on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = eta_init
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Subgradient: grad[i] = 2*a[k_peak - i] (if valid)
        grad = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Exponential weights update
        log_weights = np.log(a + 1e-300) - eta * grad
        log_weights -= log_weights.max()
        w = np.exp(log_weights)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        a_new = w / w_sum
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adapt eta
        if no_improve > 20:
            eta *= 0.8
        if no_improve > 80:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            eta = eta_init * 2
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    MW it={it}, obj={best_obj:.10f}, eta={eta:.6f}")
    
    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, lr_init=0.005, momentum_init=0.9, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        y = a + mom * v
        
        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))
        
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]
        
        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        v = v_new
        a = project_simplex(a)
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        else:
            no_improve += 1
        
        lr *= 0.9998
        mom = min(0.99, mom + 0.00005)
        
        if no_improve > 100:
            lr *= 0.8
            mom *= 0.95
        if no_improve > 300:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))

            v = np.zeros(n)
            lr = lr_init * 0.5

        if verbose and it % 500 == 0:

def topk_qp_optimize(a0, time_limit, topk=10, max_iters=50, verbose=False):
    """Solve QP with top-k peak weighting, then blend."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        k_use = min(topk, len(conv))
        peaks = np.argpartition(conv, -k_use)[-k_use:]
        w = np.zeros(len(conv))
        w[peaks] = 1.0 / k_use
        
        # Build Hankel matrix and solve QP
        W = np.array([[w[i + j] for j in range(n)] for i in range(n)])
        W_reg = W + 1e-12 * np.eye(n)
        
        try:
            x = np.linalg.solve(W_reg, np.ones(n))
        except np.linalg.LinAlgError:
            x = np.linalg.lstsq(W_reg, np.ones(n), rcond=None)[0]
        
        x_sum = x.sum()
        if abs(x_sum) < 1e-14:
            a_qp = np.ones(n) / n
        else:
            a_qp = np.maximum(x / x_sum, 0.0)
            a_qp = a_qp / a_qp.sum()
        
        # Blend
        for alpha in [0.3, 0.5, 0.7]:
            a_new = (1 - alpha) * a + alpha * a_qp
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            obj = fast_convolve(a_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                break
    
    return best_a, best_obj

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")
    
    # === Phase 1: n-exploration with LP (~300s) ===
    print("\n=== Phase 1: n-exploration ===")
    t1 = time.monotonic()
    
    candidate_ns = sorted(set([
        n_parent,
        max(256, n_parent - 128),
        min(2048, n_parent + 128),
        512, 640, 768, 896, 1024,
    ]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]
    
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1100
        if remaining < 20:
            break
        
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()
        
        t_alloc = min(remaining, 45)
        print(f"  LP n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = lp_optimize(start, time_limit=t_alloc, delta_init=0.02)
        try_seq(a_opt, f"lp_n{n_try}")
        
        remaining = deadline - time.monotonic() - 1050
        if remaining > 10:
            a_opt2, _ = lp_optimize(a_opt, time_limit=min(remaining, 20), delta_init=0.01)
            try_seq(a_opt2, f"lp2_n{n_try}")
    
    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")
    
    # === Phase 2: Deep LP optimization (~800s) ===
    print("\n=== Phase 2: Deep LP ===")
    t2 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Multiple LP runs with different delta settings
        delta_configs = [0.01, 0.02, 0.03, 0.05, 0.015, 0.025]
        
        for di, delta_c in enumerate(delta_configs):
            remaining = deadline - time.monotonic() - 400
            if remaining < 15:
                break
            t_alloc = min(remaining, 130)
            print(f"  LP deep delta={delta_c}, t={t_alloc:.0f}s")
            a_opt, _ = lp_optimize(best_seq, time_limit=t_alloc, delta_init=delta_c)
            try_seq(a_opt, f"lp_deep_d{di}")
        
        # Top-k QP interleaved
        remaining = deadline - time.monotonic() - 200
        if remaining > 15:
            t_alloc = min(remaining, 60)
            print(f"  TopK QP, t={t_alloc:.0f}s")
            a_opt, _ = topk_qp_optimize(best_seq, time_limit=t_alloc, topk=8)
            try_seq(a_opt, "topk_qp")
    
    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")
    
    # === Phase 3: Multiplicative weights + Nesterov (~300s) ===
    print("\n=== Phase 3: MW & Nesterov ===")
    t3 = time.monotonic()
    
    if best_seq is not None:
        # Multiplicative weights
        for eta_c in [0.05, 0.1, 0.2, 0.03]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 10:
                break
            t_alloc = min(remaining, 60)
            print(f"  MW eta={eta_c}, t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_weights(best_seq, time_limit=t_alloc, eta_init=eta_c)
            try_seq(a_opt, f"mw_e{eta_c}")
        
        # Nesterov
        for lr_i in [0.003, 0.008, 0.001, 0.005]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            t_alloc = min(remaining, 50)
            print(f"  Nesterov lr={lr_i}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc, lr_init=lr_i)
            try_seq(a_opt, f"nesterov_lr{lr_i}")
    
    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")
    
    # === Phase 4: Fine LP + perturbations (~200s) ===
    print("\n=== Phase 4: Fine LP & Polish ===")
    t4 = time.monotonic()
    
    if best_seq is not None:
        # Very fine LP
        for delta_c in [0.005, 0.008, 0.003]:
            remaining = deadline - time.monotonic() - 80
            if remaining < 10:
                break
            t_alloc = min(remaining, 40)
            print(f"  Fine LP delta={delta_c}, t={t_alloc:.0f}s")
            a_opt, _ = lp_optimize(best_seq, time_limit=t_alloc, delta_init=delta_c)
            try_seq(a_opt, f"fine_lp_{delta_c}")
        
        # Perturb + refine
        for _ in range(4):
            remaining = deadline - time.monotonic() - 20
            if remaining < 10:
                break
            noise = 0.001 + 0.001 * _
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * noise)
            t_alloc = min(remaining, 25)
            a_opt, _ = lp_optimize(f_pert, time_limit=t_alloc, delta_init=0.01)
            try_seq(a_opt, f"perturb_lp_{_}")
        
        # Final Nesterov polish
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            t_alloc = min(remaining, 30)
            a_opt, _ = nesterov_subgrad(best_seq, time_limit=t_alloc, lr_init=0.002)
            try_seq(a_opt, "final_nesterov")
    
    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
