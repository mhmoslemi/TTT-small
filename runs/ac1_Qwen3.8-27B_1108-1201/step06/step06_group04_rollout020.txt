import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def solve_qp_hankel(w, a_init, n, max_iters=300, time_limit=5.0):
    """

    W is Hankel: W_{i,j} = w[i+j] where w has length 2n-1.

    a = np.maximum(a_init.copy(), 0.0)

    if s < 1e-15:

    else:
        a = a / s

    active = set(np.where(a > 1e-10)[0])
    if not active:
        active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:

            a[list(active)[0]] = 1.0
            return a

        act = sorted(active)

                else:

def primal_dual_optimize(a0, time_limit, T_init=0.01, use_topk=False, topk=15,
                         blend_init=0.5, verbose=False):

    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    T = T_init
    K = min(topk, 2 * n - 1)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    blend = blend_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Compute weights with adaptive K
        if use_topk:
            k_use = min(K, len(conv))
            peaks = np.argpartition(conv, -k_use)[-k_use:]
            w = np.zeros(len(conv))
            w[peaks] = 1.0 / k_use
        else:
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            w = w / w.sum()

        # Solve QP
        qp_time = min(4.0, remaining * 0.25)
        a_qp = solve_qp_hankel(w, a, n, max_iters=200, time_limit=qp_time)

        # Backtracking line search
        best_candidate = a_qp
        best_cand_obj = fast_convolve(a_qp).max()

        # Try different blends
        for alpha in [blend, blend * 0.5, blend * 0.25, 0.7, 0.3, 0.1]:
            candidate = (1.0 - alpha) * a + alpha * a_qp
            candidate = np.maximum(candidate, 0.0)
            s_cand = candidate.sum()
            if s_cand < 1e-15:
                continue
            candidate = candidate / s_cand
            obj_cand = fast_convolve(candidate).max()
            if obj_cand < best_cand_obj:
                best_cand_obj = obj_cand
                best_candidate = candidate

        a = best_candidate
        obj = best_cand_obj

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve == 10:
            blend *= 0.7
        if no_improve > 15:
            T *= 1.2
            K = max(5, int(K * 0.8))
        if no_improve > 30:
            T *= 0.6
            K = min(30, K + 5)
        if no_improve > 80:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))

            K = min(topk, 2 * n - 1)
            blend = blend_init
            no_improve = 0

        if verbose and it % 30 == 0:
            print(f"    pd it={it}, obj={best_obj:.10f}, T={T:.6f}, K={K}, no_imp={no_improve}")

    return best_a, best_obj


def nesterov_optimize(a0, time_limit, lr_init=0.004, momentum_init=0.9,
                      K=5, verbose=False):
    """
    Nesterov-accelerated subgradient with multi-peak gradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break

        y = a + mom * v
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys

        conv_y = fast_convolve(y)

        # Multi-peak gradient
        k_use = min(K, len(conv_y))
        peaks = np.argpartition(conv_y, -k_use)[-k_use:]
        # Weight peaks by their values
        peak_vals = conv_y[peaks]
        weights = peak_vals / peak_vals.sum()

        grad = np.zeros(n)
        for w_k, k_p in zip(weights, peaks):
            js = max(0, k_p - n + 1)
            je = min(n - 1, k_p)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w_k * y[k_p - idx]

        v_new = project_simplex(y - lr * grad) - y
        a = y + v_new
        v = v_new
        a = project_simplex(a)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0

        if no_improve > 80:
            lr *= 0.85
            mom *= 0.97
        if no_improve > 250:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))

            mom = momentum_init
            no_improve = 0

            print(f"    nest it={it}, obj={best_obj:.10f}, lr={lr:.6f}")

    return best_a, best_obj

def coordinate_descent(a0, time_limit, verbose=False):
    """
    Pair-swap coordinate descent with golden section search.
    Transfers mass between pairs of coordinates to reduce max conv.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    no_improve = 0

    # Golden section ratio
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def eval_swap(a_base, i, j, delta):
        """Evaluate max conv after transferring delta from i to j."""
        a_new = a_base.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < -1e-15 or a_new[j] < -1e-15:
            return float('inf')
        a_new = np.maximum(a_new, 0.0)
        return fast_convolve(a_new).max()

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Select pair: bias towards pairs where one is in a peak-contributing position
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = set(range(js, je + 1))

        # With 50% probability, pick one from contrib, one from non-contrib
        if len(contrib) < n and rng.random() < 0.5:
            i = rng.choice(list(contrib))
            non_c = np.array([x for x in range(n) if x not in contrib])
            if len(non_c) > 0:
                j = rng.choice(non_c)
            else:
                j = rng.integers(0, n)
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

        # Golden section search in delta
        lo = max(-a[j] + 1e-12, -0.5)
        hi = min(a[i] - 1e-12, 0.5)
        if lo >= hi:
            continue

        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = eval_swap(a, i, j, c)
        fd = eval_swap(a, i, j, d)

        for _ in range(20):
            if time.monotonic() - t0 > time_limit - 0.05:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
                fc = eval_swap(a, i, j, c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
                fd = eval_swap(a, i, j, d)

        best_delta = (lo + hi) / 2.0
        best_val = min(fc, fd)
        # Also check 0
        if best_obj < best_val:
            best_val = best_obj
            best_delta = 0.0

        if best_val < best_obj - 1e-13:
            a[i] -= best_delta
            a[j] += best_delta
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            new_obj = fast_convolve(a).max()
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 200:
            # Try larger random perturbations
            idx1 = rng.integers(0, n)
            idx2 = rng.integers(0, n)
            if idx1 != idx2:
                shift = 0.01 + 0.05 * rng.random()
                a[idx1] -= shift
                a[idx2] += shift
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj > best_obj * 1.05:
                    a = best_a.copy()
                no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)

    # Edge concentration (various strengths)
    x = np.arange(n, dtype=np.float64)
    for power in [0.5, 1.0, 2.0, 4.0]:
        edge = (np.minimum(x, n - 1 - x) + 0.5) ** power
        inits.append(edge / edge.sum())

    # Center concentration
    for width in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (width * n)) ** 2)
        inits.append(g / g.sum())

    # Two peaks
    for offset in [0.2, 0.25, 0.3]:
        g2 = (np.exp(-0.5 * ((x - offset * (n - 1)) / (0.08 * n)) ** 2) +
              np.exp(-0.5 * ((x - (1 - offset) * (n - 1)) / (0.08 * n)) ** 2))
        inits.append(g2 / g2.sum())

    # Cosine
    c1 = 0.3 + 0.7 * np.cos(2 * np.pi * x / n)
    c1 = np.maximum(c1, 0.01)
    inits.append(c1 / c1.sum())

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    if parent is not None and len(parent) > 1:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for auto-correlation minimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Quick n-exploration with multiple inits (~250s) ===
    print("\n=== Phase 1: n-exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 32),
        n_parent + 32,
        max(128, n_parent - 64),
        n_parent + 64,
        512, 640, 768, 896, 1024,
    ]))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1100
        if remaining < 20:
            break

        inits = generate_inits(n_try, parent, rng)
        best_for_n = None
        best_obj_n = float('inf')

        for init in inits[:4]:  # Use first 4 initializations
            t_alloc = min(remaining / max(1, len(inits[:4])), 15)
            if t_alloc < 3:
                break
            a_opt, obj = smoothed_optimize(init, time_limit=t_alloc, T_init=0.005, lr_init=0.03)
            if obj < best_obj_n:
                best_obj_n = obj
                best_for_n = a_opt

        if best_for_n is not None:
            try_seq(best_for_n, f"explore_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep primal-dual QP (~700s) ===
    print("\n=== Phase 2: Deep Primal-Dual QP ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {'T_init': 0.005, 'use_topk': False, 'topk': 20, 'blend_init': 0.5},
            {'T_init': 0.01, 'use_topk': False, 'topk': 30, 'blend_init': 0.4},
            {'T_init': 0.02, 'use_topk': True, 'topk': 10, 'blend_init': 0.6},
            {'T_init': 0.003, 'use_topk': True, 'topk': 5, 'blend_init': 0.3},
            {'T_init': 0.008, 'use_topk': True, 'topk': 15, 'blend_init': 0.5},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 500
            if remaining < 15:
                break
            t_alloc = min(remaining, 130)
            print(f"  PD config {ci} (T={cfg['T_init']}, topk={cfg.get('topk','SF')}, "
                  f"blend={cfg['blend_init']}), t={t_alloc:.0f}s")
            a_opt, _ = primal_dual_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"pd_deep_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Nesterov + Smoothed gradient (~350s) ===
    print("\n=== Phase 3: Nesterov & Smoothed Gradient ===")
    t3 = time.monotonic()

    if best_seq is not None:
        for lr_i in [0.003, 0.006, 0.001, 0.01]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 10:
                break
            t_alloc = min(remaining, 80)
            K_val = 5 if lr_i <= 0.006 else 3
            print(f"  Nesterov lr={lr_i}, K={K_val}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_optimize(best_seq, time_limit=t_alloc, lr_init=lr_i, K=K_val)
            try_seq(a_opt, f"nest_lr{lr_i}_K{K_val}")

        for T_val, lr_val in [(0.0005, 0.015), (0.002, 0.01), (0.001, 0.025)]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            t_alloc = min(remaining, 70)
            print(f"  Smoothed T={T_val}, lr={lr_val}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_optimize(best_seq, time_limit=t_alloc, T_init=T_val, lr_init=lr_val)
            try_seq(a_opt, f"smooth_T{T_val}")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Coordinate descent polishing (~300s) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 250)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")

            # Another round from the result
            remaining = deadline - time.monotonic() - 50
            if remaining > 15 and best_seq is not None:
                t_alloc2 = min(remaining, 80)
                print(f"  Coord descent round 2, t={t_alloc2:.0f}s")
                a_opt2, _ = coordinate_descent(best_seq, time_limit=t_alloc2)
                try_seq(a_opt2, "coord_desc_2")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final n-tuning and polish (~150s) ===
    print("\n=== Phase 5: Final Tuning ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Try interpolating to nearby n values
        for n_try in [n_best - 8, n_best + 8, n_best - 16, n_best + 16]:
            if n_try < 64:
                continue
            remaining = deadline - time.monotonic() - 30
            if remaining < 10:
                break
            interp = np.interp(np.linspace(0, 1, n_try),
                               np.linspace(0, 1, n_best), best_seq)
            interp = np.maximum(interp, 0)
            interp = interp / interp.sum()
            t_alloc = min(remaining, 20)
            a_opt, _ = smoothed_optimize(interp, time_limit=t_alloc, T_init=0.001, lr_init=0.01)
            try_seq(a_opt, f"interp_n{n_try}")

        # Final Nesterov polish
        remaining = deadline - time.monotonic() - 10
        if remaining > 10 and best_seq is not None:
            t_alloc = min(remaining, 30)
            a_opt, _ = nesterov_optimize(best_seq, time_limit=t_alloc, lr_init=0.001, K=3)
            try_seq(a_opt, "final_nest")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
