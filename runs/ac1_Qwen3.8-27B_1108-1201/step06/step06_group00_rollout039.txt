def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP: (f*g)[k] <= M for k in constraint_indices."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return None, None, None
    A = np.zeros((len(constraint_indices), n))
    # Actually let's use dense for simplicity if n is not too large
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            A[idx, js] = f[k - js]
    return A

def solve_lp_direction_cvxpy(f, M, constraint_indices, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in constraint_indices, g >= 0.
    Returns the optimizer g (normalized to sum 1) or None on failure."""
    n = len(f)
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0]
    
    # Add convolution constraints
    for k in constraint_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            constraint = cp.sum(f[k - js] * g[js]) <= M
            constraints.append(constraint)
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-6)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            x = np.maximum(g.value, 0)
            s = x.sum()
            if s > 1e-15:
                return x / s
    except Exception:
        pass
    return None

        beta = 50.0

        s = x.sum()
        if s < 1e-15:
            return np.zeros(n)

def subgradient_refine(f0, time_limit, beta=100.0, top_k=30):
    """Subgradient descent with LSE smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()

    t0 = time.monotonic()
    lr = 0.5 / np.sqrt(n)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        conv = fast_convolve(f)
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        
        # LSE weights
        top_vals = conv[peak_idx]
        bc = beta * top_vals
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Gradient
        grad = np.zeros(n)
        for idx, k in enumerate(peak_idx):
            k = int(k)
            wi = w[idx]
            if wi < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += wi * 2.0 * f[k - i_range]
        
        # Projected gradient step
        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.01, 5.0 / np.sqrt(n))
        else:
            f = f_new  # Still take the step (helps with convergence)
            no_improve += 1
            if no_improve > 20:
                f = best_f.copy()
                lr *= 0.7
                no_improve = 0
            if no_improve > 100:
                break
        
        # Slowly increase beta
        beta *= 1.002
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, n_rounds=1000):
    """Coordinate descent: try adjusting pairs of coordinates to reduce peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()

        k = int(peak_idx[rng.integers(len(peak_idx))])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        
        if hi - lo + 1 < 2:
            continue
        
        # Find the pair (i, j) with i+j=k that contributes most
        i_vals = np.arange(lo, hi + 1)
        contributions = f[i_vals] * f[k - i_vals]
        j_max = i_vals[np.argmax(contributions)]
        j_other = k - j_max
        
        if j_other < 0 or j_other >= n or j_other == j_max:
            continue

        for _ in range(5):
            j_dest = rng.integers(n)
            if j_dest == j_max or j_dest == j_other:

            f_trial = f.copy()
            # Try moving a small fraction
            delta = f[j_max] * 0.05
            f_trial[j_max] -= delta

                f = f_trial.copy()

def interpolate_to_length(f, new_n):

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_refine(f0, time_limit, beta_start=30.0, beta_end=5000.0, top_k=30):
    """Subgradient descent with LSE smoothing and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    beta = beta_start
    lr = 1.0 / np.sqrt(n)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(f)
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        
        # LSE weights
        top_vals = conv[peak_idx]
        bc = beta * top_vals
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        # Gradient of weighted sum of peaks
        grad = np.zeros(n)
        for idx, k in enumerate(peak_idx):
            k = int(k)
            wi = w[idx]
            if wi < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += wi * 2.0 * f[k - i_range]
        
        # Projected gradient step with backtracking
        improved = False
        step = lr
        for _ in range(8):
            f_new = project_simplex(f - step * grad)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-16:
                f = f_new
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                improved = True
                break
            step *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve > 15:
                f = best_f.copy()
                lr *= 0.8
                no_improve = 0
            if no_improve > 100:
                break
        
        # Adapt learning rate
        if improved:
            lr = min(lr * 1.005, 5.0 / np.sqrt(n))
        
        # Increase beta gradually
        beta = min(beta * 1.003, beta_end)
    
    return best_f, best_obj

        if center not in peak_idx:

            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                js = np.arange(j_start, j_end + 1)
                rows.extend([idx] * len(js))
                cols.extend(js.tolist())
                vals.extend((f[k - js]).tolist())
        
        if not rows:
            break
        
        A_sparse = csr_matrix((vals, (rows, cols)), shape=(len(peak_idx), n))
        c = -np.ones(n)
        b_ub = np.full(len(peak_idx), M)
        bounds = [(0.0, None)] * n
        
        try:
            lp_time = min(5, max(1, it_rem * 0.03))
            res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                gs = g.sum()
                if gs > 1e-15:
                    g /= gs
                else:
                    g = None
            else:
                g = None
        except Exception:
            g = None
        
        if g is None:
            # Fallback: subgradient direction
            w = np.ones(len(peak_idx)) / len(peak_idx)
            grad = np.zeros(n)
            for idx, k in enumerate(peak_idx):
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                i_range = np.arange(lo, hi + 1)
                grad[i_range] += w[idx] * 2.0 * f[k - i_range]
            g = project_simplex(f - grad / max(np.linalg.norm(grad), 1e-15))
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-13:
            no_improve += 1
            if no_improve > 25:
                rng = np.random.default_rng(int(time.time() * 1000) % 100000)
                f_pert = project_simplex(f + rng.standard_normal(n) * 0.005)
                conv_pert = fast_convolve(f_pert)
                if conv_pert.max() < best_obj:
                    best_obj = conv_pert.max()
                    best_f = f_pert.copy()
                    f = f_pert
                    conv = conv_pert
                    M = conv.max()
                no_improve = 0
            continue
        
        # Line search
        best_alpha = 0.1
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        f = (1.0 - best_alpha) * f + best_alpha * g
        f = np.maximum(f, 0.0)
        f /= f.sum()
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, n_rounds=20000, seed=42):
    """Coordinate descent: move mass to reduce the largest peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    # Get top peaks
    k_top = min(20, 2 * n - 1)
    peak_idx = np.argpartition(conv, -k_top)[-k_top:]
    
    for it in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Pick a peak to reduce (focus on the largest)
        peak_sorted = peak_idx[np.argsort(conv[peak_idx])[::-1]]
        # With probability 0.5, target the largest; otherwise random
        if rng.random() < 0.5:
            k = int(peak_sorted[0])
        else:
            k = int(peak_sorted[rng.integers(len(peak_sorted))])
        
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi - lo + 1 < 2:
            continue
        
        # Find the pair that contributes most to this peak
        i_vals = np.arange(lo, hi + 1)
        contributions = f[i_vals] * f[k - i_vals]
        j_max = i_vals[np.argmax(contributions)]
        j_other = k - j_max
        
        if j_other < 0 or j_other >= n or j_other == j_max:
            continue
        if f[j_max] < 1e-10:
            continue
        
        # Try moving a small fraction of mass from j_max to a random position
        moved = False
        for _ in range(3):
            j_dest = int(rng.integers(n))
            if j_dest == j_max or j_dest == j_other:
                continue
            
            # Try different fractions
            for frac in [0.02, 0.05, 0.1, 0.03]:
                delta = f[j_max] * frac
                if delta < 1e-12:
                    continue
                f_trial = f.copy()
                f_trial[j_max] -= delta
                f_trial[j_dest] += delta
                f_trial = np.maximum(f_trial, 0.0)
                f_trial /= f_trial.sum()
                
                c_trial = fast_convolve(f_trial)
                if c_trial.max() < best_obj - 1e-16:
                    best_obj = c_trial.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    conv = c_trial
                    # Update peak list
                    peak_idx = np.argpartition(conv, -k_top)[-k_top:]
                    moved = True
                    break
            if moved:
                break
    
    return best_f, best_obj

def try_sequence(a, best_tracker):
    """Evaluate and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    if a.sum() < 0.01:
        return float('inf')
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score

def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

def make_structured_init(n, structure='ramp', alpha=0.3):
    """Create structured initializations."""
    if structure == 'uniform':
        return np.ones(n) / n
    elif structure == 'left_heavy':
        f = np.ones(n)
        f[n // 2:] = alpha
        return f / f.sum()
    elif structure == 'ramp':
        f = np.linspace(1.0, alpha, n)
        return f / f.sum()
    elif structure == 'exponential':
        f = np.exp(-np.linspace(0, -np.log(alpha), n))
        return f / f.sum()
    elif structure == 'two_block':
        f = np.ones(n)
        f[:n // 3] = 1.0
        f[n // 3:] = alpha
        return f / f.sum()
    else:
        return np.ones(n) / n

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining Newton peak equalization,
    subgradient descent, FW with LP oracle, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, budget_s - 15.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={score:.8f}")
            best_tracker = [score, parent.copy()]
    except Exception:
        print("No parent, using initialization")
        parent = np.ones(768) / 768.0
        try_sequence(parent, best_tracker)
    
    n_parent = len(parent)
    total_time = budget_s - 15.0
    
    # === Phase 1: Quick exploration with different initializations ===
    print(f"\n=== Phase 1: Exploration ({total_time*0.1:.0f}s) ===")
    t1_end = time.monotonic() + total_time * 0.1
    
    # Try parent with Newton
    t_alloc = min(40, t1_end - time.monotonic())
    if t_alloc > 10:
        print(f"  Newton on parent, n={n_parent}, t={t_alloc:.0f}s")
        f_opt, _ = newton_peak_equalize(parent, time_limit=t_alloc, K=6)
        try_sequence(f_opt, best_tracker)
    
    # Try a couple of different lengths
    for nc in [n_parent + 64, n_parent - 64]:
        if nc < 256 or nc > 2048:
            continue
        t_rem = t1_end - time.monotonic()
        if t_rem < 10:
            break
        f_init = interpolate_to_length(parent, nc)
        print(f"  Newton n={nc}, t={min(30, t_rem-2):.0f}s")
        f_opt, _ = newton_peak_equalize(f_init, time_limit=min(30, t_rem - 2), K=6)
        try_sequence(f_opt, best_tracker)
    
    # === Phase 2: Intensive Newton refinement ===
    print(f"\n=== Phase 2: Newton Refinement ({total_time*0.30:.0f}s) ===")
    t2_end = time.monotonic() + total_time * 0.30
    current = best_tracker[1] if best_tracker[1] is not None else parent
    
    # Multiple Newton passes with varying K
    for K in [4, 6, 8, 10, 12]:
        t_rem = t2_end - time.monotonic()
        if t_rem < 10:
            break
        t_alloc = min(50, t_rem * 0.25)
        print(f"  Newton K={K}, n={len(current)}, t={t_alloc:.0f}s")
        f_opt, _ = newton_peak_equalize(current, time_limit=t_alloc, K=K)
        try_sequence(f_opt, best_tracker)
        current = best_tracker[1] if best_tracker[1] is not None else current
    
    # === Phase 3: Alternating Newton + Subgradient + FW ===
    print(f"\n=== Phase 3: Alternating ({total_time*0.35:.0f}s) ===")
    t3_end = time.monotonic() + total_time * 0.35
    cycles = 0
    
    while time.monotonic() < t3_end and cycles < 30:
        cycles += 1
        current = best_tracker[1] if best_tracker[1] is not None else parent
        remaining = t3_end - time.monotonic()
        if remaining < 15:
            break
        n_cur = len(current)
        
        method = cycles % 6
        
        if method == 0:
            # Newton with small K
            t_alloc = min(45, remaining * 0.3)
            if t_alloc > 8:
                print(f"  [{cycles}] Newton K=5, t={t_alloc:.0f}s")

        try_sequence(f_opt, best_tracker)

        rng2 = np.random.default_rng(42)
        for _ in range(30):
