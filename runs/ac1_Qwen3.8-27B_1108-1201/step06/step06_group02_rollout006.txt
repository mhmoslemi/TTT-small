def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if it % 50 == 0:
                    sc = float(evaluate_sequence(best_f.tolist()))
                    if np.isfinite(sc):
                        best_score = min(best_score, sc)
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 60:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            scale = 0.003 + 0.005 * min(1.0, (no_improve - 60) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    sc = float(evaluate_sequence(best_f.tolist()))
    if not np.isfinite(sc):
        sc = float('inf')
    return best_f, sc


def dual_guided_optimize(a0, time_limit):
    """
    Water-filling based on KKT conditions:
    At optimum, for all i with a_i > 0: (lambda * a)_i = constant.
    Use this to guide mass redistribution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = float(evaluate_sequence(best_a.tolist()))

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fast_convolve(a)
        peak = conv.max()

        # Find active peaks (within 1% of max)
        threshold = peak * 0.99
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Uniform weight on active peaks
        lam = np.zeros(len(conv))
        lam[active] = 1.0 / len(active)

        # Compute correlation: c_i = sum_k lam_k * a_{k-i}
        # Same FFT trick as gradient
        lam_rev = lam[::-1]
        size = n + len(lam_rev) - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        L = np.fft.rfft(lam_rev, fft_size)
        conv_al = np.fft.irfft(A * L, fft_size)[:size]
        c = conv_al[n - 1:2 * n - 1][::-1]

        # Water-filling: on support of a, c should be constant
        support = a > 1e-10
        if support.sum() == 0:
            break
        c_mean = c[support].mean()

        # Compute adjustment direction
        # Where c > c_mean on support: reduce a
        # Where c < c_mean on support: increase a
        adjustment = np.zeros(n)
        adjustment[support] = -(c[support] - c_mean)

        # Apply with step size
        for step in [0.01, 0.005, 0.001, 0.0005]:
            candidate = a + step * adjustment
            candidate = project_simplex(candidate)
            val = fast_convolve(candidate).max()
            if val < peak - 1e-15:
                a = candidate
                if peak < best_obj - 1e-14 and val < best_obj:
                    best_obj = val
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                break
        else:
            no_improve += 1
            # Try a smaller perturbation
            rng = np.random.default_rng((it * 12345) % (2**31))
            candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
            val = fast_convolve(candidate).max()
            if val < best_obj:
                best_obj = val
                best_a = candidate.copy()
                a = candidate
                no_improve = 0

        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            no_improve = 0

    sc = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(sc):
        sc = float('inf')
    return best_a, sc


def targeted_peak_reduction(a0, time_limit):
    """Reduce peaks by targeted mass transfers."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = float(evaluate_sequence(best_a.tolist()))

    t0 = time.monotonic()
    stale = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        if len(affected) == 0:
            break

        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        j_min_cand = outside[np.argmin(a[outside])]

        improved = False
        for delta in [0.0003, 0.0007, 0.001, 0.003, 0.007, 0.015]:
            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min_cand] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate / candidate.sum()
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                rng = np.random.default_rng((stale * 4567 + 3) % (2**31))
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                elif stale > 800:
                    break

    sc = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(sc):
        sc = float('inf')
    return best_a, sc


def multi_peak_subgrad(a0, time_limit, top_k=20, lr_init=0.003, mom_init=0.93):
    """Multi-peak subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = float(evaluate_sequence(best_a.tolist()))

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fast_convolve(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if it % 500 == 0:
                sc = float(evaluate_sequence(best_a.tolist()))
                if np.isfinite(sc):
                    best_score = min(best_score, sc)
        else:
            no_improve += 1

        lr *= 0.9998
        mom = min(0.97, mom + 0.00005)

        if no_improve > 300:
            rng = np.random.default_rng((it * 48271) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 300) / 500.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0

    sc = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(sc):
        sc = float('inf')
    return best_a, sc


def interpolate_sequence(parent, new_n):
    """Interpolate parent sequence to new length."""
    if new_n == len(parent):
        return parent.copy()
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, parent)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(new_n) / new_n
    return result


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search with adaptive n exploration.
    Phase 1: Quick multi-n exploration
    Phase 2: Deep smoothed gradient on best n
    Phase 3: FW-LP refinement
    Phase 4: Dual-guided + multi-peak subgradient
    Phase 5: Targeted reduction + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    state_score = float('inf')
    state_seq = None

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state_score = ps
                state_seq = parent.copy()
                print("Parent: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        state_seq = parent.copy()
        state_score = float(evaluate_sequence(state_seq.tolist()))
        print("No parent, using uniform n=768, score={:.8f}".format(state_score))

    def try_update(a, label):
        nonlocal state_score, state_seq
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < state_score:
            state_score = score
            state_seq = a_norm.copy()
            print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))

    # === Phase 1: Quick Multi-n Exploration (~150s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    n_parent = len(parent)
    candidate_ns = sorted(set([n_parent,
                               max(256, n_parent - 128),
                               max(256, n_parent - 64),
                               n_parent + 64,
                               n_parent + 128,
                               n_parent + 256,
                               1024, 1280]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1300
        if remaining < 20:
            break

        start = interpolate_sequence(parent, n_try)
        t_alloc = min(remaining, 25)
        print("  Smoothed grad n={}, t={:.0f}s".format(n_try, t_alloc))
        a_opt, _ = smoothed_grad_optimizer(start, time_limit=t_alloc,
                                           T_init=0.003, T_min=0.0003,
                                           lr_init=0.04, momentum_init=0.9)
        try_update(a_opt, "sg_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state_score, time.monotonic() - t_phase))

    # === Phase 2: Deep Smoothed Gradient (~500s) ===
    print("\n=== Phase 2: Deep Smoothed Gradient ===")
    t_phase = time.monotonic()
    n_best = len(state_seq)

    remaining = deadline - time.monotonic() - 800
    if remaining > 50:
        t_alloc = min(remaining, 400)
        print("  Deep SG n={}, t={:.0f}s".format(n_best, t_alloc))
        a_opt, _ = smoothed_grad_optimizer(state_seq, time_limit=t_alloc,
                                           T_init=0.002, T_min=0.0001,
                                           lr_init=0.05, momentum_init=0.92)
        try_update(a_opt, "deep_sg")

    # Second pass with different params
    remaining = deadline - time.monotonic() - 600
    if remaining > 30:
        t_alloc = min(remaining, 150)
        print("  Deep SG (low T) n={}, t={:.0f}s".format(n_best, t_alloc))
        a_opt, _ = smoothed_grad_optimizer(state_seq, time_limit=t_alloc,
                                           T_init=0.0005, T_min=0.00005,
                                           lr_init=0.03, momentum_init=0.95)
        try_update(a_opt, "deep_sg_lowT")

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state_score, time.monotonic() - t_phase))

    # === Phase 3: FW-LP (~350s) ===
    print("\n=== Phase 3: FW-LP ===")
    t_phase = time.monotonic()

    remaining = deadline - time.monotonic() - 400
    if remaining > 30:
        t_alloc = min(remaining, 250)
        print("  FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
        a_opt, _ = fw_lp_optimize(state_seq, time_limit=t_alloc)
        try_update(a_opt, "fw_lp")

    # Perturbed FW restart
    remaining = deadline - time.monotonic() - 200
    if remaining > 20:
        t_alloc = min(remaining, 100)
        f_pert = project_simplex(state_seq + rng.standard_normal(n_best) * 0.005)
        print("  FW-LP perturbed, t={:.0f}s".format(t_alloc))
        a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
        try_update(a_opt, "fw_pert")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state_score, time.monotonic() - t_phase))

    # === Phase 4: Dual-guided + Multi-peak SG (~300s) ===
    print("\n=== Phase 4: Dual-guided + Multi-peak ===")
    t_phase = time.monotonic()

    remaining = deadline - time.monotonic() - 200
    if remaining > 20:
        t_alloc = min(remaining, 120)
        print("  Dual-guided, t={:.0f}s".format(t_alloc))
        a_opt, _ = dual_guided_optimize(state_seq, time_limit=t_alloc)
        try_update(a_opt, "dual")

    # Multi-peak subgradient with different configs
    for top_k, lr, mom in [(10, 0.002, 0.94), (30, 0.003, 0.92), (50, 0.001, 0.95)]:
        remaining = deadline - time.monotonic() - 100
        if remaining < 15:
            break
        t_alloc = min(remaining, 50)
        print("  Multi-peak SG k={}, lr={}, t={:.0f}s".format(top_k, lr, t_alloc))
        a_opt, _ = multi_peak_subgrad(state_seq, time_limit=t_alloc,
                                      top_k=top_k, lr_init=lr, mom_init=mom)
        try_update(a_opt, "mpsg_k{}".format(top_k))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state_score, time.monotonic() - t_phase))

    # === Phase 5: Final Polish (~200s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    # Targeted peak reduction
    remaining = deadline - time.monotonic() - 80
    if remaining > 15:
        t_alloc = min(remaining, 80)
        print("  Targeted peak reduction, t={:.0f}s".format(t_alloc))
        a_opt, _ = targeted_peak_reduction(state_seq, time_limit=t_alloc)
        try_update(a_opt, "targeted")

    # Very fine smoothed gradient
    remaining = deadline - time.monotonic() - 40
    if remaining > 10:
        t_alloc = min(remaining, 40)
        print("  Fine SG (T=0.0001), t={:.0f}s".format(t_alloc))
        a_opt, _ = smoothed_grad_optimizer(state_seq, time_limit=t_alloc,
                                           T_init=0.0001, T_min=0.00001,
                                           lr_init=0.01, momentum_init=0.97)
        try_update(a_opt, "fine_sg")

    # Try nearby n values with quick optimization
    for delta in [-32, -16, 16, 32]:
        remaining = deadline - time.monotonic() - 15
        if remaining < 10:
            break
        n_try = n_best + delta
        if n_try < 128 or n_try > 2048:
            continue
        start = interpolate_sequence(state_seq, n_try)
        t_alloc = min(remaining, 15)
        a_opt, _ = smoothed_grad_optimizer(start, time_limit=t_alloc,
                                           T_init=0.001, T_min=0.0001,
                                           lr_init=0.03, momentum_init=0.92)
        try_update(a_opt, "n_+{}".format(delta))

    # Tiny perturbations + quick polish
    for _ in range(8):
        remaining = deadline - time.monotonic() - 5
        if remaining < 5:
            break
        for scale in [0.0003, 0.001, 0.003]:
            remaining = deadline - time.monotonic() - 3
            if remaining < 3:
                break
            noise = rng.standard_normal(n_best) * scale
            f_pert = project_simplex(state_seq + noise)
            t_alloc = min(remaining, 5)
            a_opt, _ = smoothed_grad_optimizer(f_pert, time_limit=t_alloc,
                                               T_init=0.0002, T_min=0.00005,
                                               lr_init=0.005, momentum_init=0.95)
            try_update(a_opt, "pert_{}".format(scale))

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state_score, time.monotonic() - t_phase))

    # Final verification
    if state_seq is not None:
        final_score = float(evaluate_sequence(state_seq.tolist()))
        if np.isfinite(final_score):
            state_score = min(state_score, final_score)

    print("\nFINAL: score={:.8f}, n={}".format(state_score, len(state_seq)))
    return [float(x) for x in state_seq]
