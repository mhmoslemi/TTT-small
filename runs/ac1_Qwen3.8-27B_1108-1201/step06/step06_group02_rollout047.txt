"""
Optimized step function search using spectral inversion, adaptive subgradient,
and peak equalization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def isim_step(a, n_iter=50, alpha=1.0):
    """
    Iterative Spectral Inversion Method for flat autocorrelation.
    Works on the circular version for speed, but applies to linear.
    """
    n = len(a)
    a = a.copy()
    for _ in range(n_iter):
        # Compute DFT
        A = np.fft.fft(a)
        # Target: flat magnitude (except DC)
        target_mag = np.ones(n)
        target_mag[0] = np.sqrt(n) * a.sum()  # DC component
        # Actually, for flat autocorrelation, we want |A[k]|^2 = constant for k != 0
        # |A[0]|^2 = (sum a)^2 is fixed
        # Target: |A[k]| = (sum a) / sqrt(n) for k != 0
        target_mag[1:] = a.sum() / np.sqrt(n)
        
        # Update: preserve phase, set magnitude to target
        phase = A / np.abs(A + 1e-30)
        A_new = phase * target_mag
        # Blend
        A_new = (1 - alpha) * A + alpha * A_new
        a_new = np.real(np.fft.ifft(A_new))
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    return a

def subgradient_descent(a0, time_limit, peak_weight_temp, lr_init, momentum_init):
    """
    Subgradient descent on the simplex with exponential peak weighting.
    Uses momentum and adaptive learning rate.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)  # momentum
    lr = lr_init
    mom = momentum_init
    peak_temp = peak_weight_temp
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Exponentially weighted peaks
        scaled = conv / peak_temp
        scaled -= scaled.max()
        weights = np.exp(scaled)
        total_w = weights.sum()
        if total_w > 0:
            weights /= total_w
        else:
            weights = np.zeros(len(conv))
            weights[np.argmax(conv)] = 1.0
        
        # Subgradient: weighted sum of gradients of each peak
        # grad_j of (f*f)(k) = 2*f(k-j) if 0 <= k-j <= n-1
        sg = np.zeros(n)
        for k in range(len(conv)):
            if weights[k] < 1e-10:
                continue
            # grad at lag k: 2 * f(k-j) for valid j
            # This is 2 * reverse of f, shifted by k
            # f(k-j) for j=0..n-1: this is f[k], f[k-1], ..., f[k-(n-1)]
            # = reverse of f, starting from index k
            # More precisely: for j in range(n): grad[j] += 2*weights[k]*f[k-j] if 0 <= k-j <= n-1
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # j ranges from lo to hi, and f[k-j] where k-j ranges from k-lo to k-hi
            # f[k-j] for j=lo..hi = f[k-lo], f[k-lo-1], ..., f[k-hi]
            # = f[k-lo : k-hi-1 : -1] = f[hi:lo:-1] reversed... 
            # Actually: j = lo, lo+1, ..., hi
            # k-j = k-lo, k-lo-1, ..., k-hi
            # So f[k-j] = f[k-lo], f[k-lo-1], ..., f[k-hi]
            # This is f[hi:lo-1:-1]... no
            # f[k-lo], f[k-lo-1], ..., f[k-hi] is f[k-lo : k-hi-1 : -1]
            # = f[hi:lo:-1] (since k-hi = ... let me just index directly)
            
            idx = np.arange(lo, hi + 1)
            f_vals = a[k - idx]  # f[k-j] for j in idx
            sg[idx] += 2.0 * weights[k] * f_vals
        
        # Update with momentum
        v = mom * v - lr * sg
        a = project_simplex(a + v)
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        lr *= 0.9998
        if no_improve > 50:
            lr *= 0.95
        if no_improve > 100:
            peak_temp *= 1.1  # broaden the peak weighting
        if no_improve > 50:
            peak_temp *= 0.9  # narrow it
        
        # Restart if stuck
        if no_improve > 300:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.005 * min(1.0, (no_improve - 300) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.5
            peak_temp = peak_weight_temp
            no_improve = 0
    
    return best_a, best_obj


def peak_equalization(a0, time_limit):
    """
    Explicitly try to equalize the top peaks of the convolution.
    Uses LP to find a direction that reduces the max peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP: find direction d (sum d = 0, a + d >= 0) 
        # that minimizes max_k (conv + 2*(d*a)_k)
        # Simplified: minimize (d*a) at the peak lag
        k_peak = int(np.argmax(conv))
        
        # Direction that reduces peak: we want d such that (d*a)[k_peak] < 0
        # (d*a)[k_peak] = sum_j d(j) * a(k-j)
        # To minimize this, put mass in d where a(k-j) is large, with negative sign
        # But we need sum(d) = 0 and a + d >= 0
        
        # Simple approach: reduce a[j] where a(k_peak-j) is large, 
        # and increase a[j'] where a(k_peak-j') is small
        affected = np.arange(max(0, k_peak - n + 1), min(n, k_peak + 1))
        contributions = a[affected] * a[k_peak - affected]
        
        # Find the coordinate to reduce
        j_reduce = affected[np.argmax(contributions)]
        
        # Find a coordinate to increase (one that contributes little to the peak)
        # We want to increase a[j_incr] where a[k_peak - j_incr] is small
        all_contribs = np.zeros(n)
        valid = (k_peak - np.arange(n) >= 0) & (k_peak - np.arange(n) < n)
        all_contribs[valid] = a[k_peak - np.arange(n)[valid]]
        
        # Exclude j_reduce
        all_contribs[j_reduce] = np.inf
        j_increase = int(np.argmin(all_contribs))
        
        # Try different step sizes
        improved = False
        for delta in [0.0001, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            if a[j_reduce] < delta:
                continue
            candidate = a.copy()
            candidate[j_reduce] -= delta
            candidate[j_increase] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate / candidate.sum()
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-12:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Random restart
                rng = np.random.default_rng(it * 7919 + 13)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.003)
                c = fast_convolve(candidate)
                if c.max() < best_obj:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    no_improve = 0
                elif no_improve > 500:
                    break
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """
    Coordinate descent: for each coordinate, find the best value
    (given all others) by 1D optimization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    converged = False
    
    while time.monotonic() - t0 < time_limit and not converged:
        converged = True
        n_passes = 0
        
        while time.monotonic() - t0 < time_limit - 1 and not converged:
            n_passes += 1
            for j in range(n):
                if time.monotonic() - t0 > time_limit - 1:
                    break
                
                # Try changing coordinate j
                # Keep others fixed, adjust a[j] and renormalize
                # Actually, let's try: set a[j] to new value, and scale all others proportionally
                # to maintain sum = 1
                
                old_val = a[j]
                if old_val < 1e-10:
                    # Try setting it to a small positive value
                    for new_val in [0.001, 0.002, 0.005]:
                        candidate = a.copy()
                        candidate[j] = new_val
                        rest = 1.0 - new_val
                        if rest < 0:
                            continue
                        candidate[np.arange(n) != j] *= rest / (1.0 - old_val) if (1.0 - old_val) > 1e-10 else 1.0
                        candidate = np.maximum(candidate, 0.0)
                        cs = candidate.sum()
                        if cs < 1e-15:
                            continue
                        candidate /= cs
                        c = fast_convolve(candidate)
                        if c.max() < best_obj - 1e-12:
                            a = candidate
                            best_obj = c.max()
                            best_a = a.copy()
                            converged = False
                            break
                    continue
                
                # Try different values for a[j]
                # Sample around current value
                fractions = [0.5, 0.7, 1.3, 2.0, 0.3, 3.0]
                for frac in fractions:
                    new_val = old_val * frac
                    rest = 1.0 - new_val
                    if rest < 0 or new_val < 0:
                        continue
                    candidate = a.copy()
                    candidate[j] = new_val
                    others_sum = 1.0 - old_val
                    if others_sum > 1e-10:
                        candidate[np.arange(n) != j] *= rest / others_sum
                    candidate = np.maximum(candidate, 0.0)
                    cs = candidate.sum()
                    if cs < 1e-15:
                        continue
                    candidate /= cs
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-12:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        converged = False
                        break
    
    return best_a, best_obj


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle (similar to before but with improvements)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break
        
        # Solve LP oracle
        g = _solve_lp_oracle(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 30:
                # Perturb and retry
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.01, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.005 * min(1.0, (no_improve - 60) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj


def _solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def spectral_init(n, rng, method='random_phase'):
    """
    Generate initial sequence from spectral construction.
    """
    if method == 'random_phase':
        # Random phases with flat magnitude
        phases = rng.uniform(0, 2 * np.pi, n)
        mags = np.ones(n)
        A = mags * np.exp(1j * phases)
        a = np.real(np.fft.ifft(A))
    elif method == 'lowpass':
        mags = np.zeros(n)
        mags[:n // 4] = 1.0
        mags[n - n // 4:] = 1.0
        mags[n // 2 - 1] = 1.0
        mags[n // 2] = 1.0
        mags[n // 2 + 1] = 1.0
        phases = rng.uniform(0, 2 * np.pi, n)
        A = mags * np.exp(1j * phases)
        a = np.real(np.fft.ifft(A))
    elif method == 'smooth':
        # Smooth spectrum
        k = np.arange(n)
        center = n // 2
        mags = np.exp(-0.5 * ((k - center) / (n / 6)) ** 2)
        mags[n - center:n - center + 1] = mags[:center + 1][::-1][:len(mags[n - center:])]
        # Make it symmetric
        mags = np.maximum(mags, 0)
        phases = rng.uniform(0, 2 * np.pi, n)
        A = mags * np.exp(1j * phases)
        a = np.real(np.fft.ifft(A))
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)
        return a / a.sum()
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_and_update(a, state, label):
    """Evaluate and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining spectral methods, subgradient descent,
    and coordinate descent to minimize the autocorrelation objective.
    """

            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(384) / 384

    print("\n=== Phase 1: Diverse Initialization ===")
    t_phase = time.monotonic()
    
    n_values = sorted(set([len(parent), 512, 640, 768, 896, 1024]))
    n_values = [n for n in n_values if 128 <= n <= 2048]

            break

        inits.append(edge_w / edge_w.sum())

            h = np.maximum(h, 0.01)

        # Quick ISIM on each, then score
        for i_init, init in enumerate(inits):
            if time.monotonic() > deadline - 1200:

    # Score all candidates quickly
    for a_cand, n_c, label in candidates:
        if time.monotonic() > deadline - 1200:

        try_and_update(a_cand, state, "p1_" + label)
