        A = np.fft.rfft(a, fft_size)
        power = np.abs(A)**2

        # Target: uniform power spectrum
        target_power = power.sum() / len(power)
        
        # Adjust amplitudes

        ratio = np.sqrt(target_power / (power + 1e-15))

        if s < 1e-15:

        a_new = a_new / s

    return a

    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    i = np.nonzero(cond)[0][-1]
    theta = (cssv[i] - 1.0) / (i + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec_view(p, n):
    """Create a zero-copy Hankel matrix view where H[i,j] = p[i+j]."""
    from numpy.lib.stride_tricks import as_strided
    return as_strided(p, shape=(n, n), strides=(p.strides[0], p.strides[0]))


def hankel_solve_fast(p, n):
    """Solve W*x = 1 where W[i,j] = p[i+j]. Returns x."""
    n = len(p) if p.size < 2 * 10000 else int(np.sqrt(p.size))
    # p should have length >= 2n-1
    n_actual = int((len(p) + 1) / 2)
    W = hankel_matvec_view(p, n_actual)
    try:
        x = np.linalg.solve(W + 1e-12 * np.eye(n_actual), np.ones(n_actual))
    except np.linalg.LinAlgError:
        x = np.linalg.lstsq(W, np.ones(n_actual), rcond=None)[0]
    return x


def score_normalized(a):
    """Compute the objective for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fft_convolve(a)
    return 2.0 * n * conv.max()


def compute_weights(conv, T, topk=None):
    """Compute weight distribution from convolution values."""
    if topk is not None and topk < len(conv):
        k = min(topk, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        p = np.zeros(len(conv))
        p[peaks] = 1.0 / k
    else:
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(len(conv)) / len(conv)
        else:
            p = p / p_sum
    return p


def hankel_fp_iterate(a0, time_limit, T_init=0.01, blend=0.4, topk=None,
                      gap=0.05, verbose=False):
    """
    Fixed-point iteration: solve W_p * x = 1, clip, normalize, blend.
    W_p is Hankel with entries W[i,j] = p[i+j].
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = score_normalized(a)

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    T = T_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fft_convolve(a)
        p = compute_weights(conv, T, topk)
        p_full = p  # p has length 2n-1

        # Solve W*x = 1
        try:
            W = hankel_matvec_view(p_full, n)
            x = np.linalg.solve(W + 1e-10 * np.eye(n), np.ones(n))
        except (np.linalg.LinAlgError, Exception):
            try:
                W = hankel_matvec_view(p_full, n)
                x = np.linalg.lstsq(W, np.ones(n), rcond=None)[0]
            except Exception:
                break

        x = np.maximum(x, 0.0)
        sx = x.sum()
        if sx < 1e-15:
            # Fallback: use uniform
            x = np.ones(n) / n
        else:
            x = x / sx

        a_new = (1.0 - blend) * a + blend * x
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = x
        else:
            a_new = a_new / s_new

        obj = score_normalized(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt temperature and blend
        if no_improve > 20:
            T *= 1.2
        if no_improve > 60:
            T *= 0.7
            blend = min(0.8, blend * 1.1)
        if no_improve > 150:
            rng = np.random.default_rng((it * 7919 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            T = T_init
            blend = 0.4
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    hankel_fp it={it}, obj={best_obj:.10f}, T={T:.6f}, no_imp={no_improve}")

    return best_a, best_obj


def spectral_flatten_iterate(a0, time_limit, alpha=0.3, n_compress=1.0, verbose=False):
    """
    Spectral power flattening: in FFT domain, compress the dynamic range
    of |A|^2, then project back to simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    fft_size = 1 << (2 * n - 2 + 1).bit_length()

    best_a = a.copy()
    best_obj = score_normalized(a)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        power_mean = power.sum() / len(power)

        # Compress: ratio = (target/current)^alpha
        ratio = np.power(power_mean / (power + 1e-30), alpha * n_compress)
        ratio = np.clip(ratio, 0.1, 10.0)

        A_new = A * ratio
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj = score_normalized(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt
        if no_improve > 30:
            alpha *= 1.3
            if alpha > 2.0:
                alpha = 0.1
        if no_improve > 100:
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            alpha = 0.3
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    spectral it={it}, obj={best_obj:.10f}, alpha={alpha:.4f}")

    return best_a, best_obj


def subgrad_line_search(a0, time_limit, lr_init=0.05, verbose=False):
    """
    Projected subgradient with backtracking line search.
    Uses the dominant peak gradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = score_normalized(a)
    lr = lr_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fft_convolve(a)
        k_peak = int(np.argmax(conv))

        # Gradient of conv[k_peak] w.r.t. a
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_peak - idx]

        # Backtracking line search
        accepted = False
        for shr in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            trial_lr = lr * shr
            a_try = project_simplex(a - trial_lr * grad * n / (lr_init * n))
            obj_try = score_normalized(a_try)
            if obj_try < best_obj - 1e-14:
                a = a_try
                best_obj = obj_try
                best_a = a.try if False else a.copy()
                best_a = a.copy()
                no_improve = 0
                accepted = True
                break

        if not accepted:
            no_improve += 1
            lr *= 0.95
            if no_improve > 50:
                rng = np.random.default_rng((it * 31 + 11) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                lr = lr_init
                no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    subgrad it={it}, obj={best_obj:.10f}, lr={lr:.6f}")

    return best_a, best_obj


def smoothed_subgrad(a0, time_limit, T_init=0.005, lr_init=0.02, verbose=False):
    """
    Smoothed gradient using log-sum-exp surrogate for the max.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = score_normalized(a)
    T = T_init
    lr = lr_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fft_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient of log-sum-exp: sum_k p[k] * grad(conv[k])
        # grad(conv[k])[i] = 2*a[k-i] (valid)
        # So gradient = 2 * sum_k p[k] * a[k-i] = 2 * (p_valid * a)
        # This is the 'valid' cross-correlation of p and a
        grad = np.zeros(n)
        for i in range(n):
            js = max(0, i - (len(p) - 1 - n + 1))
            je = min(n - 1, len(p) - 1 - i)
            # conv[k] = sum_j a[j]*a[k-j], grad w.r.t a[i] = 2*sum_{k: valid} a[k-i]
            # Simplified: gradient[i] = 2 * sum_k p[k] * a[k-i] (where k-i valid)
            lo = max(0, i)
            hi = min(n, i + len(p))
            # p[k] * a[k-i] for valid k, k-i
            k_range = np.arange(max(0, i), min(len(p), i + n))
            valid = (k_range - i) >= 0
            k_range = k_range[valid]
            ai_range = k_range - i
            grad[i] = 2.0 * np.dot(p[k_range], a[ai_range])

        # Line search
        best_trial = a
        best_val = score_normalized(a)
        for shr in [1.0, 0.5, 0.25, 0.1]:
            a_try = project_simplex(a - lr * shr * grad)
            val_try = score_normalized(a_try)
            if val_try < best_val:
                best_val = val_try
                best_trial = a_try

        a = best_trial
        if best_val < best_obj - 1e-12:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T *= 1.15
            lr *= 0.9
        if no_improve > 80:
            T *= 0.6
        if no_improve > 200:
            rng = np.random.default_rng((it * 99991 + 5) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            lr = lr_init
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    smoothed it={it}, obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")

    return best_a, best_obj


def lp_sequential(a0, time_limit, gap=0.03, verbose=False):
    """
    LP-based sequential minimax:
    At each step, linearize the quadratic constraints at the current point
    and solve the resulting LP.
    """
    from scipy.optimize import linprog
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = score_normalized(a)

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fft_convolve(a)
        c_max = conv.max()

        # Active set
        S = np.where(conv >= c_max * (1.0 - gap))[0]
        if len(S) == 0:
            S = np.array([np.argmax(conv)])

        # Build LP:
        # Variables: (a_0, ..., a_{n-1}, c)
        # Minimize: c
        # Subject to: 2*v_k^T a - c <= conv[k]  for k in S
        #             sum(a) = 1
        #             a >= 0, c >= 0
        # where v_k[i] = a[k-i] if 0 <= k-i < n

        nS = len(S)
        A_ub = np.zeros((nS, n + 1))
        b_ub = np.zeros(nS)
        for ii, k in enumerate(S):
            k_int = int(k)
            v_k = np.zeros(n)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            for i in range(js, je + 1):
                v_k[i] = a[k_int - i]
            A_ub[ii, :n] = 2.0 * v_k
            A_ub[ii, n] = -1.0
            b_ub[ii] = conv[k]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * (n + 1)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(0.1, time_limit - (time.monotonic() - t0) - 1.0)})
        except Exception:
            break

        if not res.success:
            break

        a_new = res.x[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        obj = score_normalized(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    LP it={it}, obj={best_obj:.10f}, |S|={nS}")

        a = a_new

        # Adapt gap
        if it % 10 == 0:
            gap = max(0.01, gap * 0.9)

    return best_a, best_obj


def peak_equalize(a0, time_limit, verbose=False):
    """
    Coordinate moves to equalize the top peaks.
    For each peak, try shifting mass from contributing positions to non-contributing ones.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = score_normalized(a)

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fft_convolve(a)
        c_max = conv.max()
        # Find top peaks
        n_peaks = min(5, len(conv))
        top_k = np.argpartition(conv, -n_peaks)[-n_peaks:]
        top_k = top_k[np.argsort(conv[top_k])[::-1]]

        improved = False
        for k_peak in top_k:
            k = int(k_peak)
            # Contributing positions: i where 0 <= k-i < n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contrib = np.arange(js, je + 1)
            non_contrib = np.setdiff1d(np.arange(n), contrib)

            if len(contrib) == 0 or len(non_contrib) == 0:
                continue

            # Find the largest contributor
            idx_max = contrib[np.argmax(a[contrib])]
            val = a[idx_max]
            if val < 1e-12:
                continue

            # Try shifting fractions
            for frac in [0.05, 0.1, 0.2, 0.3]:
                shift = val * frac
                if shift < 1e-12:
                    continue

                # Distribute to non-contrib positions
                nc_vals = a[non_contrib]
                nc_sum = nc_vals.sum()
                if nc_sum < 1e-15:
                    delta = np.zeros(n)
                    delta[idx_max] -= shift
                    delta[non_contrib] += shift / len(non_contrib)
                else:
                    delta = np.zeros(n)
                    delta[idx_max] -= shift
                    delta[non_contrib] += shift * nc_vals / nc_sum

                candidate = project_simplex(a + delta)
                obj_c = score_normalized(candidate)
                if obj_c < best_obj - 1e-12:
                    a = candidate
                    best_obj = obj_c
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it % 50 == 0:
            rng = np.random.default_rng((it * 48271 + 13) % (2**31))
            for _ in range(5):
                delta = rng.standard_normal(n) * 0.005
                candidate = project_simplex(a + delta)
                obj_c = score_normalized(candidate)
                if obj_c < best_obj - 1e-12:
                    a = candidate
                    best_obj = obj_c
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and it > 200:
            break

    return best_a, best_obj


def resample_sequence(parent, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(parent)
    if old_n == new_n:
        return parent.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, parent)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return new_seq / s


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())

    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.08 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Cosine
    c1 = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    c1 = np.maximum(c1, 0.01)
    inits.append(c1 / c1.sum())

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - 0.5 * (n - 1)) / (n - 1)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Spectral init: flat spectrum
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
    A[0] = 1.0
    angles = rng.uniform(0, 2 * np.pi, A.shape)
    A[1:] = np.exp(1j * angles) * 0.5
    a_spec = np.fft.irfft(A, fft_size)[:n]
    a_spec = np.maximum(a_spec, 0.0)
    s = a_spec.sum()
    if s > 1e-10:
        inits.append(a_spec / s)

    if parent is not None:
        px = resample_sequence(parent, n)
        inits.append(px)
        inits.append(project_simplex(px + rng.standard_normal(n) * 0.1))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = try_seq(parent, "parent")
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        try_seq(parent, "uniform512")

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, best={best_score:.10f}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: Multi-n exploration (~300s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 128) if n_parent > 256 else n_parent,
        n_parent + 128,
        n_parent * 2 if n_parent * 2 <= 3072 else n_parent,
        512, 768, 1024,
    ]))
    # Remove duplicates and sort
    candidate_ns = sorted(set(candidate_ns))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 30:
            break

        # Generate inits
        inits = generate_diverse_inits(n_try, rng, parent)

        for ii, init in enumerate(inits[:4]):  # Try up to 4 inits per n
            remaining = deadline - time.monotonic() - 1200
            if remaining < 15:
                break
            t_alloc = min(remaining, 30)
            print(f"  n={n_try}, init={ii}, t={t_alloc:.0f}s")

            a_opt, _ = hankel_fp_iterate(init, time_limit=t_alloc, T_init=0.01, blend=0.5)
            try_seq(a_opt, f"pf_n{n_try}_i{ii}")

            # Also try spectral
            remaining = deadline - time.monotonic() - 1200
            if remaining > 10:
                a_opt2, _ = spectral_flatten_iterate(a_opt, time_limit=min(remaining, 15), alpha=0.3)
                try_seq(a_opt2, f"sp_n{n_try}_i{ii}")
