"""
Minimize  score(a) = 2*n*max(conv(a,a)) / (sum a)^2  (scale-invariant, a>=0).

Key idea (different from the previous O(n^3) active-set QP):
  * conv(a,a) is computed with FFT (O(n log n)).
  * The outer problem is min_{a in simplex} max_k a^T B_k a. Because it is
    non-convex, we optimize the smooth log-sum-exp surrogate F_T(a) with
    accelerated projected gradient (FISTA) + a temperature continuation that
    anneals T to 0. Each step is O(n log n), so the smoothed problem is solved
    to high precision (the previous code's O(n^3) inner solve was time-limited
    and under-converged).
  * A Nesterov sub-gradient polish on the true (non-smooth) max, averaging the
    gradient over the (near-)tied maximal lags, shaves the last bits.
  * A fast, exact cyclical 1-D coordinate-descent polish (using the incremental
    structure of the auto-convolution, O(n) per accepted 1-D line) and a
    last-long polish finalize the per-n optimum.
  * Multi-n exploration: the per-n optimum decreases toward the ~1.5 limit, so
    we warm-start every new n by up-scaling the current global best structure.
"""

import time
import math
import numpy as np


# -----------------------------------------------------------------------------
# Core fast operations
# -----------------------------------------------------------------------------

def _next_pow2(m):
    return 1 << (max(1, int(m)) - 1).bit_length()


def fast_conv(a):
    """Linear auto-convolution a * a via FFT, real, length 2n-1."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[: 2 * n - 1]


def hankel_p_a(p, a):
    """Vector v with v[i] = sum_k p[k] a[k-i]  (p length 2n-1, a length n).
    This is the building block of grad F_T: grad = 2*hankel_p_a(p, a)."""
    n = a.size
    need = 3 * n - 2
    L = _next_pow2(need)
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p[::-1], L)
    full = np.fft.irfft(A * P, L)[:need]
    return full[n - 1: 2 * n - 1][::-1]


def conv_grad_active(a, k):
    """Gradient of (a*a)[k] w.r.t. a:  (grad)[i] = 2 a[k-i] (0<=k-i<n) else 0."""
    n = a.size
    v = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    i = np.arange(lo, hi + 1)
    v[i] = 2.0 * a[k - i]
    return v


def project_simplex(v):
    """Euclidean projection of v onto {x>=0, sum x = 1}."""
    n = v.size
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng = np.arange(1, n + 1)
    cond = u * rng > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return out / s


def local_score(a):
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return float(2.0 * a.size * fast_conv(a).max() / (s * s))


def interp_up(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    x = np.linspace(0.0, 1.0, a.size)
    y = np.linspace(0.0, 1.0, n)
    out = np.interp(y, x, a)
    out = np.clip(out, 0.0, None)
    ss = out.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return out / ss


def update_best(ctx, a):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15 or not np.isfinite(s):
        return
    a = a / s
    sc = local_score(a)
    if np.isfinite(sc) and sc < ctx['score']:
        ctx['score'] = sc
        ctx['seq'] = a.copy()
        print("    NEW BEST n=%d score=%.10f" % (a.size, sc))

def fista_stage(a, T, nmax, end_time):
    """Annealed smooth objective F_T = T*logsumexp((a*a)/T): FISTA w/ backtracking."""
    a = np.asarray(a, dtype=np.float64)
    a = a / a.sum()
    n = a.size
    y = a.copy()
    z = a.copy()
    t = 1.0
    bestF = float('inf')
    bestz = a.copy()
    for it in range(nmax):
        if time.monotonic() >= end_time - 0.05:
            break
        conv = fast_conv(y)
        cm = conv.max()
        ex = np.exp((conv - cm) / T)
        p = ex / ex.sum()
        grad = 2.0 * hankel_p_a(p, y)
        L = 4.0 * n + 8.0 / T
        zc = None
        Fy = cm + T * math.log(ex.sum())
        for _ in range(5):
            cand = project_simplex(y - (1.0 / L) * grad)
            convc = fast_conv(cand)
            cmc = convc.max()
            exc = np.exp((convc - cmc) / T)
            Fc = cmc + T * math.log(exc.sum())
            diff = 0.5 * L * float(np.sum((y - cand) ** 2))
            if Fc <= Fy + float(np.dot(grad, (y - cand))) + diff + 1e-10:
                zc = cand
                break
            L *= 2.0
        if zc is None:
            zc = project_simplex(y - (1.0 / L) * grad)
        tn = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t * t))
        ynew = zc + ((t - 1.0) / tn) * (zc - z)
        ynew = project_simplex(ynew)
        z = zc
        y = ynew
        t = tn
        if it % 4 == 0:
            convz = fast_conv(z)
            FzT = convz.max() + T * math.log(float(np.exp((convz - convz.max()) / T).sum()))
            if FzT < bestF:
                bestF = FzT
                bestz = z.copy()
    return bestz


def nesterov_polish(a, end_time, lr_init):
    """Nesterov sub-gradient on the true max, averaging grad over near-tied lags."""
    a = np.asarray(a, dtype=np.float64)
    a = a / a.sum()
    n = a.size
    if lr_init is None:
        lr_init = max(5e-5, 0.7 / n)
    v = np.zeros(n)
    mom = 0.9
    lr = lr_init
    best = a.copy()
    bests = local_score(a)
    noimp = 0
    it = 0
    while time.monotonic() < end_time - 0.05:
        it += 1
        y = a + mom * v
        conv = fast_conv(y)
        m = conv.max()
        tie = np.where(conv >= m * (1.0 - 1e-4))[0]
        if tie.size == 0:
            tie = np.array([int(np.argmax(conv))])
        g = np.zeros(n)
        for k in tie:
            g += conv_grad_active(y, int(k))
        g /= tie.size
        a_new = project_simplex(y - lr * g)
        v = a_new - y
        a = a_new
        s = local_score(a)
        if s < bests - 1e-14:
            bests = s
            best = a.copy()
            noimp = 0
        else:
            noimp += 1
        lr *= 0.9995
        mom = min(0.999, mom + 5e-5)
        if noimp > 2000:
            v = np.zeros(n)
            lr = max(5e-5, 0.5 / n)
            mom = 0.9
            noimp = 0
    return best


def _coord_eval(num0, L, S, j, t):
    newS = S + t
    if newS <= 1e-12:
        return 1e18
    num = num0 + t * L
    num[2 * j] = num0[2 * j] + t * L[2 * j] + t * t
    return float(num.max()) / (newS * newS)


def coord_polish(a, end_time, max_sweeps):

    a = np.asarray(a, dtype=np.float64).copy()

            c2 = lo + gr * (hi - lo)
            f1 = _coord_eval(num0, L, S, j, c1)

        a = fista_stage(a, T, its, st_end)
        update_best(ctx, a)
    if intensity == 'deep':
        nest_end = min(end_time - 2.0, time.monotonic() + max(0.5, 0.1 * (end_time - time.monotonic())))
        a = nesterov_polish(a, nest_end, None)
        update_best(ctx, a)
    a = coord_polish(a, end_time - 0.1, max_sweeps=(200 if intensity == 'deep' else 10))
    update_best(ctx, a)
    return a

def propose_candidate(seed=42, budget_s=1800):
    rng = np.random.default_rng(seed)
    total = max(5.0, float(budget_s))
    deadline = time.monotonic() + total - 20.0
    print("budget=%.0fs deadline_in=%.0fs" % (total, total - 20))

    best = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, None)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best = p / p.sum()
            print("parent loaded n=%d score=%.10f" % (p.size, local_score(best)))
    except Exception:
        best = None
    if best is None:
        best = np.ones(1024) / 1024.0

    pn = best.size
    ctx = {'score': local_score(best), 'seq': best.copy()}
    print("initial best n=%d score=%.10f" % (pn, ctx['score']))

    # Lengths to explore (order-of-a-few-thousand is the sweet spot).
    ns = [pn, 1024, 1280, 1600, 2048, 2560, 3200]
    ns = sorted(set([int(x) for x in ns if 256 <= int(x) <= 4096]))
    ns = ns[:7]

    # ---- Phase A: shallow screen over several n (~40% of time) ----
    tA_end = time.monotonic() + 0.40 * (deadline - time.monotonic())
    print("=== Phase A: screen n=%s (until %.0fs left) ===" %
          (ns, tA_end - time.monotonic()))
    per = max(1.0, (tA_end - time.monotonic()) / float(len(ns)))
    for n in ns:
        if time.monotonic() >= tA_end - 0.5:
            break
        init = interp_up(ctx['seq'], n)
        end_i = min(tA_end - 0.5, time.monotonic() + per)
        print("  screen n=%d (slice %.0fs)" % (n, end_i - time.monotonic()))
        a_local = workhorse_n(init, 'shallow', end_i, ctx)
        ctx_local_seq = a_local  # noqa

    # ---- Phase B: deep workhorse on the two most promising n (~45%) ----
    tB_end = time.monotonic() + 0.45 * (deadline - time.monotonic())
    n_of_best = ctx['seq'].size
    deep_set = sorted(set([n_of_best, max(1024, n_of_best),
                            min(3200, n_of_best * 2)]))
    deep_set = [x for x in deep_set if 256 <= x <= 4096][:3]
    print("=== Phase B: deep n=%s (until %.0fs left) ===" %
          (deep_set, tB_end - time.monotonic()))
    per = max(1.0, (tB_end - time.monotonic()) / float(max(1, len(deep_set))))
    for n in deep_set:
        if time.monotonic() >= tB_end - 0.5:
            break
        init = interp_up(ctx['seq'], n)
        end_i = min(tB_end - 0.5, time.monotonic() + per)
        print("  deep n=%d (slice %.0fs)" % (n, end_i - time.monotonic()))
        workhorse_n(init, 'deep', end_i, ctx)

    # ---- Phase C: final long polish on the global best ----
    print("=== Phase C: final polish (remaining %.0fs) ===" %
          (deadline - time.monotonic()))
    nb = ctx['seq'].size
    a = interp_up(ctx['seq'], nb)
    a = nesterov_polish(a, min(deadline - 3.0, time.monotonic() + 40.0), None)
    update_best(ctx, a)
    a = fista_stage(a, 0.002, 300, min(deadline - 3.0, time.monotonic() + 20.0))
    update_best(ctx, a)
    a = coord_polish(a, deadline - 0.5, max_sweeps=5000)
    update_best(ctx, a)

    # ---- Final verification against the real evaluator ----
    out = ctx['seq']
    out = np.clip(out, 0.0, None)
    outs = out.sum()
    if outs < 0.01:
        out = np.ones(256)
    out = out / out.sum()
    try:
        real = float(evaluate_sequence([float(x) for x in out]))
    except Exception:
        real = ctx['score']
    print("FINAL local=%.10f real=%s n=%d" % (ctx['score'],
          ("%.10f" % real) if np.isfinite(real) else str(real), out.size))
    return [float(x) for x in out]
