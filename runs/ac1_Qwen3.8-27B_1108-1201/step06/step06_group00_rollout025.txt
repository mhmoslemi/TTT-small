def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_objective(conv, beta):
    """Compute LSE_β of convolution values."""
    bc = beta * conv
    bc_max = bc.max()
    return bc_max + np.log(np.exp(bc - bc_max).sum()) / beta

                          verbose=True):
    """
    Nesterov-accelerated gradient descent on LSE_β(conv(f)) over the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    momentum = momentum_start
    lr = 1.0 / (2.0 * np.max(a_max) if (a_max := f.max()) > 0 else 1.0)
    lr = min(lr, 0.5 / np.sqrt(n))
    
    t0 = time.monotonic()
    no_improve = 0
    iteration = 0
    
    x = f.copy()  # current position
    v = np.zeros(n)  # velocity (for momentum)
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # Compute gradient at current point
        conv = fast_convolve(x)
        grad = lse_gradient(conv, beta, x)
        
        # Nesterov-style update
        v = momentum * v + (1.0 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        
        # Backtracking line search
        step = lr
        f_current = conv.max()
        improved = False
        for _ in range(8):
            x_new = project_simplex(x - step * v)
            conv_new = fast_convolve(x_new)
            obj_new = conv_new.max()
            if obj_new < f_current - 1e-15 * max(1.0, f_current):
                x = x_new
                f_current = obj_new
                improved = True
                break
            step *= 0.5
        
        if improved:
            conv = fast_convolve(x)
            if conv.max() < best_obj - 1e-14:
                best_obj = conv.max()
                best_f = x.copy()
                no_improve = 0
                lr = min(lr * 1.01, 2.0 / np.sqrt(n))
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.8
                x = best_f.copy()
                v = np.zeros(n)
                no_improve = 0
            if no_improve > 50:
                # Restart with perturbation
                rng = np.random.default_rng(iteration * 13 + 7)
                x = project_simplex(best_f + rng.standard_normal(n) * 0.01 * best_f.max())
                v = np.zeros(n)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0
        
        # Anneal beta and momentum
        progress = min(1.0, iteration / max(1, time_limit / 0.1))
        beta = beta_start * (beta_end / beta_start) ** progress
        momentum = momentum_start + (momentum_end - momentum_start) * progress
        
        if verbose and iteration % 1000 == 0:
            print(f"    it={iteration}, obj={best_obj:.8f}, beta={beta:.1f}, lr={lr:.6e}")
    
    return best_f, best_obj


def peak_shaving(f, time_limit, n_shaves=5000, verbose=False):
    """
    Iteratively reduce the maximum peak by moving mass from the main contributor
    to the peak to a position that minimizes overall peak increase.
    """
    f = np.maximum(f.copy(), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for _ in range(n_shaves):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        k_max = int(np.argmax(conv))
        
        # Find the main contributor to peak k_max
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo >= hi:
            idx = np.arange(lo, hi + 1)
        else:
            continue
        if len(idx) == 1:
            continue
        
        contributions = f[idx] * f[k_max - idx]
        i_star = idx[np.argmax(contributions)]
        
        # Try multiple candidate destinations
        # Score: for destination j, the increase in peak is ~ 2*delta*max_k a[k-j]
        # But more precisely, we want to minimize the resulting max peak.
        # Use a heuristic: prefer positions where the local "environment" is flat.
        
        delta = 0.005 * f[i_star]
        if delta < 1e-15:
            # Try to zero out i_star
            delta = f[i_star]
        
        # Sample some candidate destinations
        n_cand = min(50, n)
        candidates = np.random.choice(n, size=n_cand, replace=False)
        
        best_j = -1
        best_trial_obj = conv.max()
        
        for j in candidates:
            if j == i_star:
                continue
            f_trial = f.copy()
            f_trial[i_star] -= delta
            f_trial[i_star] = max(0, f_trial[i_star])
            actual_delta = f[i_star] - f_trial[i_star]
            f_trial[j] += actual_delta
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial /= st
            conv_trial = fast_convolve(f_trial)
            obj_trial = conv_trial.max()
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_j = j
        
        if best_j >= 0 and best_trial_obj < best_obj - 1e-15:
            f = f.copy()
            f[i_star] -= delta
            f[i_star] = max(0.0, f[i_star])
            actual_delta = 0  # Recompute
            f_trial = f.copy()
            f_trial[i_star] = max(0.0, f[i_star] - delta)
            actual_delta = f[best_f is not None and i_star] - 0  # This is getting buggy
            # Just redo it cleanly:
            f = best_f.copy() if False else f  # Keep current f
            f[i_star] = max(0.0, f[i_star] - delta)
            actual_delta = delta - max(0.0, f[i_star])
            f[best_j] += actual_delta
            f = np.maximum(f, 0.0)
            f /= f.sum()
            conv = fast_convolve(f)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_f = f.copy()
        else:
            # No improvement, try a smaller delta or random perturbation
            if _ % 100 == 0:
                rng_pert = np.random.default_rng(_ + 999)
                f_pert = project_simplex(f + rng_pert.standard_normal(n) * 0.003)
                conv_pert = fast_convolve(f_pert)
                if conv_pert.max() < best_obj:
                    best_obj = conv_pert.max()
                    best_f = f_pert.copy()
                    f = f_pert.copy()
    
    return best_f, best_obj


def pair_coordinate_descent(f, time_limit, n_iters=10000, verbose=False):
    """
    Try to reduce peaks by moving mass between pairs of coordinates.
    For each pair (i,j), find the optimal δ that minimizes the max peak.
    """
    f = np.maximum(f.copy(), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    # Precompute top peaks
    top_k = min(20, 2 * n - 1)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        k_max = int(peak_idx[0])  # Actually argmax
        
        # Find main contributors to the top few peaks
        for k in peak_idx[:3]:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            if len(idx) < 2:
                continue
            contributions = f[idx] * f[k - idx]
            # Try to reduce the top contributor
            i_star = idx[np.argmax(contributions)]
            
            # Pick a random destination
            j = rng.integers(0, n)
            while j == i_star:
                j = rng.integers(0, n)
            
            # Binary search for optimal δ
            lo_d = 0.0
            hi_d = f[i_star] * 0.5
            best_delta = 0.0
            best_val = conv.max()
            
            for _ in range(8):
                mid_d = (lo_d + hi_d) / 2.0
                f_trial = f.copy()
                f_trial[i_star] -= mid_d
                f_trial[j] += mid_d
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-15:
                    break
                f_trial /= st
                conv_trial = fast_convolve(f_trial)
                val = conv_trial.max()
                if val < best_val:
                    best_val = val
                    best_delta = mid_d
                    lo_d = mid_d
                else:
                    hi_d = mid_d
            
            if best_delta > 1e-10 and best_val < best_obj - 1e-15:
                f[i_star] -= best_delta
                f[j] += best_delta
                f = np.maximum(f, 0.0)
                f /= f.sum()
                conv = fast_convolve(f)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
    
    return best_f, best_obj


def gen_cosine_window(n, power):
    """Generate a cosine-powered window."""
    i = np.arange(n)
    x = np.cos(np.pi * (i + 0.5) / n)
    return np.maximum(x, 0) ** power


def gen_raised_cosine(n):
    """Raised cosine window."""
    i = np.arange(n)
    return 0.5 * (1 + np.cos(2 * np.pi * (i + 0.5) / n))


def gen_gaussian_window(n, sigma_frac):
    """Gaussian window centered at n/2."""
    i = np.arange(n)
    c = (n - 1) / 2.0
    return np.exp(-0.5 * ((i - c) / (sigma_frac * n)) ** 2)


def gen_flat_spectrum(n, rng):
    """Generate a signal with approximately flat |DFT|^2 in the passband."""
    # Create a random phase DFT with flat magnitude
    N = 1 << (2 * n - 1).bit_length()
    A_mag = np.ones(N // 2 + 1)
    A_mag[0] = 0  # DC
    # Taper the edges to reduce time-domain leakage
    taper = np.ones(N // 2 + 1)
    edge = N // 16
    taper[:edge] = np.linspace(0.1, 1, edge)
    taper[-edge:] = np.linspace(1, 0.1, edge)
    A_mag *= taper
    A_phase = rng.uniform(-np.pi, np.pi, N // 2 + 1)
    A = A_mag * np.exp(1j * A_phase)
    A[0] = 0
    # Enforce conjugate symmetry
    A_full = np.zeros(N, dtype=complex)
    A_full[:N // 2 + 1] = A
    A_full[N // 2 + 1:] = np.conj(A[1:N // 2][::-1])
    a = np.fft.ifft(A_full).real
    a = a[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def subgradient_refine(f0, time_limit, beta=2000.0, top_k=10):
    """
    Focused subgradient descent on the top few peaks (very high beta).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    lr = 0.1 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(f)
        
        # Focus on top peaks
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        
        # Compute gradient focused on top peaks
        # Use softmax with very high beta for near-max behavior
        bc = beta * conv[peak_idx]
        bc_max = bc.max()
        w = np.exp(bc - bc_max) / np.exp(bc - bc_max).sum()
        
        grad = np.zeros(n)
        for idx_i, k in enumerate(peak_idx):
            k = int(k)
            wi = w[idx_i]
            if wi < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += wi * 2.0 * f[k - i_range]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_normed = grad / gnorm
        
        # Backtracking
        step = lr
        improved = False
        for _ in range(10):
            f_new = project_simplex(f - step * grad_normed)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.02, 1.0)
                break
            step *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.7
            if no_improve > 40:
                f = best_f.copy()
                lr = 0.1 / np.sqrt(n)
                no_improve = 0
                # Small perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                f = project_simplex(f + rng.standard_normal(n) * 0.005)
    
    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Diverse initializations (30s)
    Phase 2: Nesterov LSE gradient descent (1000s)
    Phase 3: Peak shaving + coordinate descent (400s)
    Phase 4: High-beta subgradient refinement (200s)
    Phase 5: Final polish (70s)
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            score_p = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={score_p:.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass
    
    if parent is None:
        parent = gen_cosine_window(768, 1.0)
        parent /= parent.sum()
        try_sequence(parent, best_tracker)
    
    n_parent = len(parent)
    
    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse initializations ===")
    t1_end = time.monotonic() + 30.0

    lengths_to_try = sorted(set([
        n_parent,
        max(128, n_parent - 64),
        min(2048, n_parent + 64),
        max(128, int(n_parent * 0.75)),
        min(2048, int(n_parent * 1.25)),
        512, 640, 768, 896, 1024, 1280

    for nc in lengths_to_try[:8]:
        if time.monotonic() > t1_end:
            break
        # Generate multiple shapes for this length
        for power in [0.5, 1.0, 1.5, 2.0, 3.0]:
            f_init = gen_cosine_window(nc, power)
            f_init /= f_init.sum()
            initializations.append(f_init)
        
        f_init = gen_raised_cosine(nc)
        f_init /= f_init.sum()
        initializations.append(f_init)
        
        for sigma in [0.2, 0.3, 0.4]:
            f_init = gen_gaussian_window(nc, sigma)
            f_init /= f_init.sum()
            initializations.append(f_init)
        
        if nc == n_parent:
            initializations.append(parent.copy())
            initializations.append(interpolate_to_length(parent, nc))
        
        # Flat spectrum
        f_init = gen_flat_spectrum(nc, rng)
        initializations.append(f_init)
    
    # Evaluate all initializations quickly
    print(f"  Evaluating {len(initializations)} initializations...")
    for f_init in initializations:
        if time.monotonic() > t1_end:
            break
        try_sequence(f_init, best_tracker)
    
    # Pick top candidates for detailed optimization
    candidates = []
    seen_close = []
    for f_init in initializations:
        conv = fast_convolve(f_init)
        obj = 2 * len(f_init) * conv.max()  # With sum=1
        candidates.append((obj, f_init))
    candidates.sort(key=lambda x: x[0])
    
    # Deduplicate (keep diverse set)
    top_candidates = []
    for obj, f in candidates:
        diverse = True
        for _, f2 in top_candidates:
            if len(f) == len(f2):
                cos_sim = np.dot(f, f2) / (np.linalg.norm(f) * np.linalg.norm(f2) + 1e-15)
                if cos_sim > 0.999:
                    diverse = False
                    break
        if diverse:
            top_candidates.append((obj, f))
        if len(top_candidates) >= 6:

    print(f"  Top candidates: {len(top_candidates)}")
    for i, (obj, f) in enumerate(top_candidates):

        print(f"  Final coord descent: n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = pair_coordinate_descent(current_best, time_limit=t_alloc, n_iters=5000)
        try_sequence(f_opt, best_tracker)
    
    # Tiny random perturbations
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    n_cur = len(current_best)
    t_rem = deadline - time.monotonic()
    if t_rem > 5:
        print(f"  Final perturbations: t={t_rem:.0f}s")
        for _ in range(20):
            if time.monotonic() > deadline - 3:
                break
            f_pert = project_simplex(current_best + rng.standard_normal(n_cur) * 0.002)
            try_sequence(f_pert, best_tracker)
            current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    
    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]
    
    # Fallback
    print("WARNING: Using parent as fallback")
    fallback = np.maximum(parent, 0.0)
    return [float(x) for x in fallback]
