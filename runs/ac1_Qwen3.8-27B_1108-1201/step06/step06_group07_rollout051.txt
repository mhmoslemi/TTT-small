    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def coordinate_descent(a, time_limit=60.0, verbose=False):
    """
    Coordinate descent: optimize one coordinate at a time via golden section search.
    Minimizes 2n*max(a*a)/(sum a)^2.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    n = len(a)

    t0 = time.monotonic()
    best_score = compute_score(a)
    best_a = a.copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate

    for pass_num in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        pass_improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_x = a[i]
            s_rest = s - old_x

            # Set a[i] = 0 and compute base convolution
            a[i] = 0.0
            c_base = fast_convolve(a)

            # Affected range: k in [i, min(2n-2, i+n-1)]
            k_start = i
            k_end = min(2 * n - 2, i + n - 1)

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(c_base[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(c_base[k_end + 1:].max()))

            # Affected values
            idx = np.arange(k_start, k_end + 1)
            a_shift = a[idx - i]  # a[k-i] for k in affected range (a[i]=0)
            c_base_aff = c_base[idx].copy()

            # Self term at k = 2i
            k_self = 2 * i
            has_self = (k_self >= k_start and k_self <= k_end)
            self_pos = k_self - k_start if has_self else -1

            def eval_f(x):
                S_new = s_rest + x
                if S_new < 1e-15:
                    return 1e10
                c_aff = c_base_aff + 2.0 * x * a_shift
                if has_self:
                    c_aff[self_pos] = c_base_aff[self_pos] + x * x
                total_max = max(float(c_aff.max()), unaff_max)
                return 2.0 * n * total_max / (S_new * S_new)

            # Golden section search in [0, x_max]
            x_max = max(2.0 * old_x if old_x > 0 else s_rest / n * 2.0, s_rest * 0.5, 1e-8)
            x_max = min(x_max, s_rest * 3.0 + 1e-6)

            lo = 0.0
            hi = x_max
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = eval_f(c1)
            f2 = eval_f(c2)

            for _ in range(45):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = eval_f(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_f(c2)

            x_opt = (lo + hi) / 2.0
            f_opt = eval_f(x_opt)
            f_zero = eval_f(0.0)
            f_old = eval_f(old_x)

            best_x = x_opt
            best_f = f_opt
            if f_zero < best_f:
                best_x = 0.0
                best_f = f_zero
            if f_old < best_f:
                best_x = old_x
                best_f = f_old

            a[i] = best_x
            s = s_rest + best_x

            if abs(best_x - old_x) > 1e-14:
                pass_improved = True

        # Check improvement
        if pass_num % 3 == 0:
            sc = compute_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
            if verbose:
                print(f"    CD pass {pass_num}: score={sc:.10f}")

        if not pass_improved:
            break

    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a


def mirror_descent_momentum(a0, time_limit=60.0, eta_init=0.03, momentum=0.7):
    """Mirror descent with momentum and adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Compute weights on peaks (smoothed)
        tau = 0.002 * max_val + 1e-12
        w = np.exp((conv - max_val) / tau)
        w = w / w.sum()

        # Gradient: grad[i] = sum_k w[k] * 2 * a[k-i] * 1_{valid}
        grad = np.zeros(n)
        for k in range(len(conv)):
            wk = w[k]
            if wk < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            grad[js:je + 1] += wk * 2.0 * a[k - je:k - js + 1][::-1]

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        v_centered = v - v.mean()

        # Multiplicative update
        a_new = a * np.exp(-eta * v_centered)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Line search
        best_candidate = a
        best_val = max_val
        for alpha in [0.3, 0.5, 0.7, 1.0, 1.3]:
            cand = (1.0 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            eta *= 0.92
        if eta < 1e-5:
            eta = eta_init * 0.5
            momentum = min(momentum + 0.05, 0.9)
            no_improve = 0

        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            eta = eta_init
            momentum = 0.7
            no_improve = 0

    return best_a


def pairwise_exchange(a, time_limit=30.0):
    """Try pairwise exchanges to escape local minima."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try moving fraction from i to j (and vice versa)
        best_local = (0.0, 0.0)  # (f_i, f_j) fractions to move
        best_f = best_score

        for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
            # Move from i to j
            if a[i] > frac * a[i] * 2:
                a_try = a.copy()
                a_try[i] -= frac * a[i]
                a_try[j] += frac * a[i]
                f = compute_score(a_try)
                if f < best_f:
                    best_f = f
                    best_local = (frac, 0.0)

            # Move from j to i
            if a[j] > frac * a[j] * 2:
                a_try = a.copy()
                a_try[j] -= frac * a[j]
                a_try[i] += frac * a[j]
                f = compute_score(a_try)
                if f < best_f:
                    best_f = f
                    best_local = (0.0, frac)

        if best_local[0] > 0 or best_local[1] > 0:
            a[i] -= best_local[0] * a[i]
            a[j] += best_local[0] * a[i] if best_local[0] > 0 else 0
            a[j] -= best_local[1] * a[j]
            a[i] += best_local[1] * a[j] if best_local[1] > 0 else 0
            a = np.maximum(a, 0)
            best_a = a.copy()
            best_score = best_f

    return best_a


def local_search(a0, time_limit=60.0):
    """Local search: perturb + coordinate descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    t0 = time.monotonic()
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Random perturbation
        scale = rng.uniform(0.001, 0.02)
        a_pert = project_simplex(a + rng.standard_normal(n) * scale)

        # Quick coordinate descent (limited time)
        t_cd = min(remaining * 0.4, 10.0)
        a_opt = coordinate_descent(a_pert, time_limit=t_cd)
        sc = compute_score(a_opt)

        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            a = a_opt.copy()

    return best_a


def interpolate_sequence(src, n_target):
    """Interpolate a sequence to a target length."""
    if len(src) == n_target:
        return src.copy()
    x_src = np.linspace(0, 1, len(src))
    x_tgt = np.linspace(0, 1, n_target)
    result = np.interp(x_tgt, x_src, src)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: coordinate descent + mirror descent + multi-n + local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        # Clamp to valid range
        a_norm = np.clip(a_norm, 0.0, 1000.0)
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"\nStarting optimization. Parent n={n_parent}, best={best_score:.8f}")

    # === Phase 1: Coordinate descent on parent (250s) ===
    print("\n=== Phase 1: Coordinate Descent on parent ===")
    t1 = time.monotonic()
    remaining = deadline - time.monotonic()
    t_alloc = min(250.0, remaining - 300.0)
    if t_alloc > 10:
        print(f"  CD on n={n_parent}, t={t_alloc:.0f}s")
        a_opt = coordinate_descent(parent, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"cd_parent_n{n_parent}")
    print(f"  P1 done: best={best_score:.8f}, t={time.monotonic()-t1:.0f}s")

    # === Phase 2: Mirror descent with momentum (200s) ===
    print("\n=== Phase 2: Mirror Descent + Momentum ===")
    t2 = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(200.0, remaining - 250.0)
        if t_alloc > 10:
            print(f"  MD t={t_alloc:.0f}s")
            a_opt = mirror_descent_momentum(best_seq, time_limit=t_alloc, eta_init=0.02, momentum=0.7)
            try_seq(a_opt, "mirror_mom")
            # Follow up with CD
            remaining = deadline - time.monotonic()
            t_alloc2 = min(100.0, remaining - 200.0)
            if t_alloc2 > 5:
                a_opt2 = coordinate_descent(a_opt, time_limit=t_alloc2)
                try_seq(a_opt2, "mirror_cd")
    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: More coordinate descent (200s) ===
    print("\n=== Phase 3: Extended Coordinate Descent ===")
    t3 = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(200.0, remaining - 200.0)
        if t_alloc > 10:
            print(f"  Extended CD t={t_alloc:.0f}s")
            a_opt = coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "ext_cd")
    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Multi-n exploration (350s) ===
    print("\n=== Phase 4: Multi-n Exploration ===")
    t4 = time.monotonic()
    if best_seq is not None:
        ns_to_try = [512, 640, 768, 896, 1024, 1280]
        if n_parent not in ns_to_try:
            ns_to_try.append(n_parent)
        ns_to_try = sorted(set([n for n in ns_to_try if 384 <= n <= 1536]))

        for n_try in ns_to_try:
            remaining = deadline - time.monotonic()
            if remaining < 100:
                break
            if n_try == len(best_seq):
                continue

            t_alloc = min(50.0, remaining / max(1, len(ns_to_try) - ns_to_try.index(n_try)) - 20.0)
            if t_alloc < 5:
                continue

            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            a_interp = interpolate_sequence(best_seq, n_try)
            a_opt = coordinate_descent(a_interp, time_limit=t_alloc)
            try_seq(a_opt, f"multi_n{n_try}")

            # Quick mirror descent follow-up
            remaining = deadline - time.monotonic()
            if remaining > 50:
                t_md = min(15.0, remaining - 60.0)
                if t_md > 3:
                    a_md = mirror_descent_momentum(a_opt, time_limit=t_md, eta_init=0.02)
                    try_seq(a_md, f"multi_n{n_try}_md")
                    # CD polish
                    remaining = deadline - time.monotonic()
                    if remaining > 30:
                        a_cd = coordinate_descent(a_md, time_limit=min(10, remaining - 40))
                        try_seq(a_cd, f"multi_n{n_try}_md_cd")

    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Local search with restarts (300s) ===
    print("\n=== Phase 5: Local Search / Restarts ===")
    t5 = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_local = min(remaining / 30.0, 8)
        print(f"  {int(n_local)} restarts, total t={remaining:.0f}s")
        for k in range(int(n_local)):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            t_alloc = min(30.0, remaining - 20.0)
            if t_alloc < 5:
                break
            a_opt = local_search(best_seq, time_limit=t_alloc)
            try_seq(a_opt, f"restart_{k}")
    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Final polish (remaining) ===
    print("\n=== Phase 6: Final Polish ===")
    t6 = time.monotonic()
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_alloc = min(remaining - 5, 60.0)
            print(f"  Final CD t={t_alloc:.0f}s")
            a_opt = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_cd")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_alloc = min(remaining - 3, 20.0)
            a_opt = mirror_descent_momentum(best_seq, time_limit=t_alloc, eta_init=0.01, momentum=0.8)
            try_seq(a_opt, "final_md")

        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_opt = coordinate_descent(best_seq, time_limit=min(remaining - 2, 10.0))
            try_seq(a_opt, "final_cd2")

    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_a = np.clip(best_seq, 0.0, 1000.0)
    fs = float(evaluate_sequence(final_a.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
        best_seq = final_a

    print(f"\nFINAL RESULT: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
