def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    conv = fast_convolve(a)
    s = a.sum()

def fast_convolve(a):

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    if s < 1e-15:

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def symmetrize(a):
    return (a + a[::-1]) / 2.0

def make_symmetric(a):
    a = symmetrize(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_lp_matrix(n):

    # A_ub[k, j] = a[j] if k-j is in [0, n-1], i.e., j <= k <= j+n-1

    rows = []
    cols = []

def solve_lp_oracle_dense(a, M, max_time=20):
    """Solve: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': max_time, 'dual_feasibility_tolerance': 1e-8})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def fw_lp_optimize(a0, time_limit, symmetry=True, initial_M_frac=1.0, m_step=1.02):

                ft = make_symmetric(ft)
            else:
                ft = np.maximum(ft, 0.0)

            ft = np.maximum(ft, 0.0)

            if sf < 1e-15:

            ft = ft / sf

                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * g
            if symmetry:
                a = make_symmetric(a)
            else:
                a = np.maximum(a, 0.0)
                a = a / a.sum()
            
            conv = fast_convolve(a)

def active_set_newton(a0, time_limit, symmetry=True):
    """
    Newton-type local optimization on the active peak system.
    Uses KKT conditions: at optimum, gradient of weighted active peaks
    is constant on the support.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    if symmetry:
        a = make_symmetric(a)
    else:
        a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < 5000:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Find active peaks (within tolerance of max)
        tol = max(0.01 * M, M * 0.001)
        active_peaks = np.where(conv >= M - tol)[0]

        if len(active_peaks) < 2:
            # Only one peak: use simple gradient
            k_peak = int(active_peaks[0]) if len(active_peaks) > 0 else n - 1
            grad = np.zeros(n)
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] = a[k_peak - idx_arr]
            if symmetry:
                grad = make_symmetric(grad)
            
            # Gradient step
            best_candidate = a
            best_val = M
            for step in [0.01, 0.05, 0.1, 0.2]:
                candidate = a - step * grad
                if symmetry:
                    candidate = make_symmetric(candidate)
                else:
                    candidate = project_simplex(candidate)
                val = fast_convolve(candidate).max()
                if val < best_val:
                    best_val = val
                    best_candidate = candidate
            a = best_candidate
            if best_val < best_obj:
                best_obj = best_val
                best_a = a.copy()
            continue

        # Compute weighted gradient using top peaks
        # Weight by distance from max (closer peaks get more weight)
        pv = conv[active_peaks]
        weights = 1.0 / (1.0 + (M - pv) / (tol + 1e-15))
        weights = weights / weights.sum()

        # Gradient: sum_k w_k * d(a*a)_k / da
        grad = np.zeros(n)
        for i, k in enumerate(active_peaks):
            if weights[i] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += weights[i] * a[k - idx_arr]

        if symmetry:
            grad = make_symmetric(grad)

        # Project gradient to be orthogonal to ones (for simplex constraint)
        grad = grad - grad.mean()

        # Line search
        best_candidate = a
        best_val = M
        for step in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            candidate = a - step * grad
            if symmetry:
                candidate = make_symmetric(candidate)
            else:
                candidate = project_simplex(candidate)
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
        else:
            # Try smaller peak tolerance
            tol *= 0.5
            if tol < M * 1e-6:
                break

    return best_a, best_obj

def spectral_init(n, rng, symmetry=True):
    """
    Create initial sequence with prescribed spectral properties.
    Uses random phase in frequency domain to create "flat" auto-correlation.
    """
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Random PSD with some structure
    n_freq = fft_size // 2 + 1
    # Use a bandpass filter to avoid low-frequency dominance
    freq = np.arange(n_freq)
    band_frac = 0.3 + 0.4 * rng.random()
    low = int(n_freq * (1 - band_frac))
    high = int(n_freq * (1 + band_frac))
    
    psd = np.zeros(n_freq)
    mask = (freq >= low) & (freq <= high)
    psd[mask] = rng.standard_normal(mask.sum()) ** 2 + 0.1
    psd[0] = 0  # Zero DC to avoid bias
    
    # Create signal from PSD
    A = np.zeros(n_freq, dtype=complex)
    A[mask] = np.sqrt(psd[mask]) * np.exp(1j * 2 * np.pi * rng.random(mask.sum()))
    
    a_full = np.fft.irfft(A, fft_size)[:n]
    
    # Make non-negative: shift and clip
    a_full = a_full - a_full.min()
    s = a_full.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a_full / s
    
    if symmetry:
        a = make_symmetric(a)
    return a

def random_init(n, rng, symmetry=True):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, size=n)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    if symmetry:
        a = make_symmetric(a)
    return a

def interp_to_length(a, n_target):
    """Interpolate sequence to target length, preserving symmetry."""
    if len(a) == n_target:
        return a.copy()
    x_orig = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, a)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    new_seq = new_seq / s
    return make_symmetric(new_seq)

        print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, symmetry=True,
                                   initial_M_frac=0.95)
        try_update(a_opt, "deep_fw")

    # Newton refinement
    remaining = deadline - time.monotonic() - 200
    if remaining > 20:
        t_alloc = min(remaining, 100)
        print(f"  Newton n={n_best}, t={t_alloc:.0f}s")
        a_new, _ = active_set_newton(a_opt, time_limit=t_alloc, symmetry=True)
        try_update(a_new, "deep_newton")

    # Perturbation restarts
    for scale in [0.002, 0.005, 0.01, 0.02]:
        remaining = deadline - time.monotonic() - 100
        if remaining < 20:
            break
        f_pert = best_seq + rng.standard_normal(len(best_seq)) * scale
        f_pert = make_symmetric(f_pert)
        t_alloc = min(remaining, 80)
        a_p, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, symmetry=True)
        try_update(a_p, f"pert_{scale}")
        
        remaining = deadline - time.monotonic() - 60
        if remaining > 10:
            a_np, _ = active_set_newton(a_p, time_limit=min(remaining, 20), symmetry=True)
            try_update(a_np, f"pert_{scale}_new")

    print(f"  Phase 3 best: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Fine-tuning with smaller n variations (~200s) ===
    print("\n=== Phase 4: Fine-tuning ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Try nearby n values
    for delta in [-64, -32, 32, 64]:
        n_try = n_best + delta
        if n_try < 128 or n_try > 3000:
            continue
        remaining = deadline - time.monotonic() - 50
        if remaining < 15:
            break
        interp_a = interp_to_length(best_seq, n_try)
        t_alloc = min(remaining, 40)
        a_opt, _ = fw_lp_optimize(interp_a, time_limit=t_alloc, symmetry=True)
        try_update(a_opt, f"n{n_try}")
        
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            a_new, _ = active_set_newton(a_opt, time_limit=min(remaining, 15), symmetry=True)
            try_update(a_new, f"n{n_try}_new")

    print(f"  Phase 4 best: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Final polish (~100s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Very small perturbations + Newton
    for _ in range(5):
        remaining = deadline - time.monotonic() - 10
        if remaining < 8:
            break
        noise = rng.standard_normal(n_best) * 0.001
        f_pert = make_symmetric(best_seq + noise)
        t_alloc = min(remaining, 15)
        a_p, _ = active_set_newton(f_pert, time_limit=t_alloc, symmetry=True)
        try_update(a_p, "polish")

    # Final FW
    remaining = deadline - time.monotonic() - 5
    if remaining > 10:
        a_f, _ = fw_lp_optimize(best_seq, time_limit=min(remaining, 30), symmetry=True)
        try_update(a_f, "final_fw")

    # Final symmetrization check
    try_update(make_symmetric(best_seq), "final_sym")

    print(f"  Phase 5 best: {best_score:.8f} (t={time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = get_score(best_seq)
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
