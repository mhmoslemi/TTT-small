def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_smoothed_grad(a, p):

import numpy as np

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_fft(a, p):
    """
    Compute 2 * cross_correlation(p, a) at lags 0..n-1 via FFT.
    p has length 2n-1, a has length n. Result has length n.
    Result[i] = 2 * sum_j p[j+i] * a[j]
    """
    n = len(a)
    conv_len = len(p) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a[::-1], fft_size)
    conv = np.fft.irfft(P * A, fft_size)[:conv_len]
    return 2.0 * conv[n - 1: 2 * n - 1]


def hankel_mv_indicator(C, a):

    Compute (H*a)[i] = sum_j C[i+j]*a[j] where C is indicator of active lags.
    C has length 2n-1, a has length n. Result has length n.
    Same as cross_correlation(C, a) at lags 0..n-1.
    """
    return smooth_grad_fft(a, C) / 2.0

def frank_wolfe_qp(C_active, a0, time_limit, verbose=False):
    """
    Frank-Wolfe algorithm for min a^T H a s.t. a in simplex,
    where H is the Hankel matrix with indicator C_active on its anti-diagonals.
    Handles non-convex H (indefinite case) via local search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        # Compute gradient: g = 2 * H * a = 2 * cross_correlation(C, a)
        grad = hankel_mv_indicator(C_active, a)

        # FW vertex: put all mass on argmin gradient
        s_idx = int(np.argmin(grad))
        direction = np.zeros(n)
        direction[s_idx] = 1.0
        direction[s_idx] -= a[s_idx]

        # Line search along FW direction
        best_t = 0.0
        best_val = fast_autocorr(a).max()
        for t in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
            candidate = a + t * direction
            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()
            if cs < 1e-15:
                break
            candidate = candidate / cs
            cand_r = fast_autocorr(candidate).max()
            if cand_r < best_val - 1e-15:
                best_t = t
                best_val = cand_r
                best_candidate = candidate

        if best_t > 0 and best_val < best_obj - 1e-15:
            a = best_candidate
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Try perturbating the active set
                rng = np.random.default_rng((it * 104729) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                best_obj = fast_autocorr(a).max()
                no_improve = 0

    return best_a, best_obj * 2 * n


def active_set_balancing(a0, time_limit, K=10, verbose=False):
    """
    Iterative active-set peak balancing:
    1. Identify top-K peaks in autocorrelation.
    2. Solve QP (via FW) to minimize sum of those peaks.
    3. Check if any new peak exceeds; if so, add to active set.
    4. Iterate until stable.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        r = fast_autocorr(a)
        # Get top-K lags
        k_use = min(K, len(r))
        top_k = np.argpartition(r, -k_use)[-k_use:]
        
        # Create indicator
        C = np.zeros(2 * n - 1)
        C[top_k] = 1.0

        # Solve QP via FW
        qp_time = min(remaining * 0.6, 30.0)
        a_new, _ = frank_wolfe_qp(C, a, time_limit=qp_time)

        new_obj = fast_autocorr(a_new).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try a blend
            for alpha in [0.3, 0.5, 0.7]:
                blend = (1 - alpha) * a + alpha * a_new
                blend = np.maximum(blend, 0)
                blend = blend / blend.sum()
                blend_obj = fast_autocorr(blend).max()
                if blend_obj < best_obj - 1e-14:
                    best_obj = blend_obj
                    best_a = blend.copy()
                    a = blend.copy()
                    break
            else:
                # No improvement, try with more peaks
                K = min(K * 2, 2 * n - 1)
                if K > 50:
                    break

    return best_a, best_obj * 2 * n

                candidate[i] = new_val
                if removed > 1e-15:
                    # Add to others proportionally
                    others = np.ones(n)
                    others[i] = 0
                    others = others * a
                    others_sum = others.sum()
                    if others_sum > 1e-15:
                        candidate += removed * others / others_sum
                    else:
                        candidate[i] = old_val  # can't remove
                        continue
                candidate = np.maximum(candidate, 0)
                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs
                cand_r = fast_autocorr(candidate).max()
                if best_r is None or cand_r < best_r:
                    best_r = cand_r
                    best_candidate = candidate

            if best_r is not None and best_r < best_obj - 1e-14:
                a = best_candidate
                best_obj = best_r
                best_a = a.copy()
                improved_epoch = True

        if not improved_epoch:
            no_improve += 1
        else:
            no_improve = 0

    return best_a, best_obj * 2 * n


def pairwise_swap(a0, time_limit, verbose=False):
    """
    Try pairwise mass transfers: move epsilon from i to j.
    Use top contributing pairs for efficiency.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Get current peaks to guide where to move mass
        r = fast_autocorr(a)
        k_peak = int(np.argmax(r))

        # Find contributing indices for the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        non_contrib = np.setdiff1d(np.arange(n), contrib)

        # Try moving mass from a contributing index to a non-contributing one
        improved = False
        for _ in range(min(50, n)):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            if len(contrib) == 0 or len(non_contrib) == 0:
                break

            i = contrib[rng.integers(len(contrib))]
            j = non_contrib[rng.integers(len(non_contrib))]

            if a[i] < 1e-10:
                continue

            for eps in [0.001, 0.003, 0.01, 0.02, 0.05]:
                if eps > a[i] * 0.5:
                    continue
                candidate = a.copy()
                candidate[i] -= eps
                candidate[j] += eps
                candidate = np.maximum(candidate, 0)
                cs = candidate.sum()
                candidate = candidate / cs
                cand_r = fast_autocorr(candidate).max()
                if cand_r < best_obj - 1e-14:
                    a = candidate
                    best_obj = cand_r
                    best_a = a.copy()
                    improved = True
                    # Update contrib/non_contrib
                    if a[j] > 1e-10:
                        non_contrib = np.setdiff1d(non_contrib, [j])
                    break

        if not improved and it % 200 == 0:
            # Random perturbation
            idx = rng.integers(n)
            jdx = rng.integers(n)
            if idx != jdx and a[idx] > 1e-5:
                candidate = a.copy()
                move = a[idx] * 0.1
                candidate[idx] -= move
                candidate[jdx] += move
                candidate = np.maximum(candidate, 0)
                candidate = candidate / candidate.sum()
                if fast_autocorr(candidate).max() < best_obj - 1e-14:
                    a = candidate
                    best_obj = fast_autocorr(a).max()
                    best_a = a.copy()

    return best_a, best_obj * 2 * n


def spectral_balance(a0, time_limit, T=0.0005, verbose=False):
    """
    Spectral approach: work in the DFT domain to flatten the autocorrelation.
    The autocorrelation is IDFT(|DFT(a_padded)|^2). We try to equalize it
    by adjusting the power spectrum.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_autocorr(a).max()
    t0 = time.monotonic()
    lr = 0.1
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Compute current autocorrelation
        r = fast_autocorr(a)
        r_max = r.max()

        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2  # Power spectrum (length fft_size//2 + 1)

        # Target: flat autocorrelation means flat power spectrum
        # r[k] = IDFT(P)[k], so we want all r[k] equal
        # Equivalently, P should be flat (constant)
        # But we can't freely change P; it's determined by a.
        # Instead, use the gradient approach in the time domain but guided by spectrum.

        # Compute which lags need reduction
        r_scaled = r / T
        r_scaled -= r_scaled.max()
        p_weights = np.exp(r_scaled)
        p_weights = p_weights / p_weights.sum()

        # Gradient in time domain (same as before)
        grad = smooth_grad_fft(a, p_weights)

        # Step
        direction = project_simplex(a - lr * grad) - a
        candidate = a + direction
        candidate = np.maximum(candidate, 0)
        candidate = candidate / candidate.sum()
        cand_obj = fast_autocorr(candidate).max()

        if cand_obj < best_obj - 1e-15:
            a = candidate
            best_obj = cand_obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.05, 2.0)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 100:
                # Restart with perturbation
                rng = np.random.default_rng((it * 31 + 7) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                lr = 0.1
                no_improve = 0

    return best_a, best_obj * 2 * n


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    # Cosine
    c = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    # Two peaks
    g = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
        np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g / g.sum())
    # Random and structured
    r = rng.dirichlet(np.ones(n) * 1.0)
    inits.append(r)

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())
        # Perturbed parent
        for scale in [0.001, 0.003, 0.01]:
            px2 = parent.copy()
            px2 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            px2 = np.maximum(px2 + rng.standard_normal(n) * scale, 0)
            if px2.sum() > 1e-10:
                inits.append(px2 / px2.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for autocorrelation minimization.
    Phases:
    1. Quick multi-n exploration with short SGD runs.
    2. Deep smoothed gradient descent on best n.
    3. Active-set QP balancing (Frank-Wolfe on top-K peaks).
    4. Coordinate descent + pairwise swap polish.
    5. Final verification.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        # Internal score computation
        r = fast_autocorr(a_norm)
        score = 2.0 * len(a_norm) * r.max()
        if score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_norm)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = 2.0 * len(parent) * fast_autocorr(parent).max()
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384
        try_seq(parent, "uniform384")

    print(f"Starting: current_best={best_score:.10f}")

    # === Phase 1: Quick n-exploration (~250s) ===
    print("\n=== Phase 1: Quick n-exploration ===")
    t1 = time.monotonic()

    n_parent = len(parent)
    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 128),
        n_parent + 128,
        max(128, n_parent // 2),
        min(2048, n_parent * 2),
        512, 768, 1024,
    ]))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:
            break

        # Generate init from parent
        start = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_parent), parent)
        start = np.maximum(start, 0)
        if start.sum() > 1e-10:
            start = start / start.sum()
        else:
            start = np.ones(n_try) / n_try

        t_alloc = min(remaining, 25)
        print(f"  Phase1 n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = smooth_grad_descent(start, time_limit=t_alloc,
                                        T_start=0.002, T_end=0.0002,
                                        lr_init=0.5)
        try_seq(a_opt, f"p1_n{n_try}")
        t_opt, _ = smooth_grad_descent(a_opt, time_limit=min(remaining, 15),
                                        T_start=0.0005, T_end=0.00005,
                                        lr_init=0.3)
        try_seq(t_opt, f"p1b_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep smoothed gradient descent (~700s) ===
    print("\n=== Phase 2: Deep SGD ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Long run with careful annealing
        remaining = deadline - time.monotonic() - 500
        t_alloc = min(remaining, 300)
        print(f"  Deep SGD n={n_best}, T: 0.001->0.00003, t={t_alloc:.0f}s")
        a_opt, _ = smooth_grad_descent(best_seq, time_limit=t_alloc,
                                        T_start=0.001, T_end=0.00003,
                                        lr_init=1.0)
        try_seq(a_opt, "deep_sgd_1")

        # Continue from there with lower T
        remaining = deadline - time.monotonic() - 400
        if remaining > 20:
            t_alloc = min(remaining, 200)
            print(f"  Deep SGD cont, T: 0.0002->0.00001, t={t_alloc:.0f}s")
            a_opt2, _ = smooth_grad_descent(a_opt, time_limit=t_alloc,
                                             T_start=0.0002, T_end=0.00001,
                                             lr_init=0.5)
            try_seq(a_opt2, "deep_sgd_2")

        # Multiple restarts from best with different T
        for T_s in [0.0008, 0.0004, 0.00015]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  SGD restart T={T_s}, t={t_alloc:.0f}s")
            a_r, _ = smooth_grad_descent(best_seq, time_limit=t_alloc,
                                          T_start=T_s, T_end=T_s * 0.1,
                                          lr_init=0.7)
            try_seq(a_r, f"sgd_T{T_s}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Active-set QP balancing (~350s) ===
    print("\n=== Phase 3: Active-set QP ===")
    t3 = time.monotonic()

    if best_seq is not None:
        for K in [5, 10, 20]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  AS-QP K={K}, t={t_alloc:.0f}s")
            a_q, _ = active_set_balancing(best_seq, time_limit=t_alloc, K=K)
            try_seq(a_q, f"asqp_K{K}")

        # Direct FW on a combined indicator
        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            n_b = len(best_seq)
            r_b = fast_autocorr(best_seq)
            K_direct = 15
            top_k = np.argpartition(r_b, -K_direct)[-K_direct:]
            C = np.zeros(2 * n_b - 1)
            C[top_k] = 1.0
            t_alloc = min(remaining, 60)
            print(f"  Direct FW on top-{K_direct} peaks, t={t_alloc:.0f}s")
            a_fw, _ = frank_wolfe_qp(C, best_seq, time_limit=t_alloc)
            try_seq(a_fw, "direct_fw")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Coordinate descent + pairwise swap (~200s) ===
    print("\n=== Phase 4: Coord Descent & Pairwise Swap ===")
    t4 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_cd, "coord_descent")

        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            t_alloc = min(remaining, 60)
            print(f"  Pairwise swap, t={t_alloc:.0f}s")
            a_ps, _ = pairwise_swap(best_seq, time_limit=t_alloc)
            try_seq(a_ps, "pairwise_swap")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final polish (~100s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        # Very fine SGD
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  Fine SGD T=0.00005, t={t_alloc:.0f}s")
            a_f, _ = smooth_grad_descent(best_seq, time_limit=t_alloc,
                                          T_start=0.00005, T_end=0.00001,
                                          lr_init=0.2)
            try_seq(a_f, "fine_sgd")

        # Another pairwise swap round
        remaining = deadline - time.monotonic() - 15
        if remaining > 5:
            t_alloc = min(remaining, 30)
            a_ps2, _ = pairwise_swap(best_seq, time_limit=t_alloc)
            try_seq(a_ps2, "final_swap")

        # Tiny perturbation + quick refine
        remaining = deadline - time.monotonic() - 10
        if remaining > 5:
            n_b = len(best_seq)
            for noise_scale in [0.0003, 0.0008]:
                pert = project_simplex(best_seq + rng.standard_normal(n_b) * noise_scale)
                t_alloc = min(remaining, 10)
                a_p, _ = smooth_grad_descent(pert, time_limit=t_alloc,
                                              T_start=0.0001, T_end=0.00002,
                                              lr_init=0.3)
                try_seq(a_p, f"perturb_{noise_scale}")
                remaining = deadline - time.monotonic() - 5
                if remaining < 3:
                    break

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Final verification with evaluate_sequence
    if best_seq is None:
        best_seq = np.ones(384) / 384

    # Verify with the actual evaluator
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"\nVerified score: {final_score:.10f} (n={len(best_seq)})")

    # One last check: try rounding to avoid floating point issues
    # Actually, don't round - the evaluator handles floats fine.

    return [float(x) for x in best_seq]
