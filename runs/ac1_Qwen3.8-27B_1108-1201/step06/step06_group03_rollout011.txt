"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_subgradient(a, T, top_k=30):
    """Compute smoothed subgradient of max_k conv(a,a)[k]."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(len(conv))
    pv = conv[peaks]
    pv_max = pv.max()
    w = np.exp((pv - pv_max) / T)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(peaks)) / len(peaks)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += w[i] * 2.0 * a[k - idx]
    return grad / T, conv.max()


def entropy_mirror_descent(a0, time_limit, eta0=0.05, T=0.001, top_k=20):
    """Entropy mirror descent (multiplicative weights) for fast convergence."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        grad, obj = smoothed_subgradient(a, T, top_k)

        eta = eta0 / np.sqrt(1 + it * 0.01)
        log_update = -eta * grad
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        new_obj = fast_convolve(a_new).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        a = a_new

            T = 0.001
            eta0 = max(eta0, 0.01)
            stagnant = 0
        if stagnant > 500:
            break

    return best_a, best_obj


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M, top_k_constraints=None):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def subgradient_with_annealing(a0, time_limit, T_init=0.0005, lr_init=0.02):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)

    T = T_init
    lr = lr_init

    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9
    temp = 1.0

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        current_obj = conv.max()

        top_k = max(10, n // 40)
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        pv = conv[peaks]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.5))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w[i] * 2.0 * a[k - idx]

        v = mom * v - lr * grad
        best_candidate = a
        best_val = current_obj

        for frac in [1.0, 0.5, 0.25, 0.1]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = fast_convolve(a).max()

        # Simulated annealing: occasionally accept worse moves
        if it % 100 == 0 and temp > 0.01:
            rng = np.random.default_rng((it * 48271) % (2**31))
            perturb = project_simplex(a + rng.standard_normal(n) * 0.005)
            p_obj = fast_convolve(perturb).max()
            delta = p_obj - obj
            if delta < 0 or np.exp(-delta / (temp * best_obj)) > rng.random():
                a = perturb
                obj = p_obj
            temp *= 0.95

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 30:

            lr *= 0.96
        if stagnant > 100:

            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            temp = max(temp, 0.5)
            stagnant = 0

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit):
    """Coordinate descent targeting top peaks with mass redistribution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_scale = 0.003

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            pairs = []
            for j in range(js, je + 1):
                pairs.append((a[j] * a[k - j], j, k - j))
            pairs.sort(reverse=True)

            for (_, j1, j2) in pairs[:6]:
                if a[j1] < 1e-8 and a[j2] < 1e-8:
                    continue

                src = j1 if a[j1] >= a[j2] else j2
                if a[src] < 1e-8:
                    continue

                # Try multiple target positions
                for delta_frac in [step_scale, step_scale * 0.5, step_scale * 2.0]:
                    delta = a[src] * delta_frac
                    if delta < 1e-12:
                        continue

                    # Target: position far from the peak contributors
                    rng = np.random.default_rng((it * 9973 + k * 31 + src * 7) % (2**31))
                    # Prefer positions that don't contribute to current peak
                    candidates_targets = []
                    for _ in range(8):
                        jt = int(rng.integers(0, n))
                        if abs(jt - j1) > n // 4 or abs(jt - j2) > n // 4:
                            candidates_targets.append(jt)

                    for j_target in candidates_targets[:4]:
                        if j_target == src:
                            continue
                        candidate = a.copy()
                        candidate[src] -= delta
                        candidate[j_target] += delta
                        if candidate[src] < -1e-12:
                            continue
                        candidate = np.maximum(candidate, 0)
                        csum = candidate.sum()
                        if csum < 1e-15:
                            continue
                        candidate = candidate / csum
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-15:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()

                        break

                    break

                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    if parent is not None:
        starts.append(resample_sequence(parent, n))
    starts.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)
    cos_t = 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    starts.append(cos_t / cos_t.sum())

    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.25 * n)) ** 2)
    starts.append(g / g.sum())

    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Asymmetric ramp
    ramp = np.linspace(0.1, 1.0, n)
    starts.append(ramp / ramp.sum())

    # Reverse ramp
    rev = np.linspace(1.0, 0.1, n)
    starts.append(rev / rev.sum())

    for _ in range(2):
        starts.append(rng.dirichlet(np.ones(n) * 0.5))

    return starts

                remaining = deadline - time.monotonic() - 250
                if remaining < 20:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
                t_alloc2 = min(remaining, 50)
                a_opt2, _ = fw_optimize(f_pert, time_limit=t_alloc2)
                try_seq(a_opt2, f"fw_pert_{ci}_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Subgradient with simulated annealing ===
    print("\n=== Phase 3: Subgradient + Annealing ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        configs = [
            {'T_init': 0.0003, 'lr_init': 0.015},
            {'T_init': 0.001, 'lr_init': 0.02},
            {'T_init': 0.003, 'lr_init': 0.01},
        ]
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, 100)
            print(f"  Subgrad+anneal config {ci}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_with_annealing(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"subgrad_anneal_{ci}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak coordinate descent ===
    print("\n=== Phase 4: Peak Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 200)
            print(f"  Peak CD, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_cd")

            # Run again from result
            remaining = deadline - time.monotonic() - 60
            if remaining > 20 and best_seq is not None:
                t_alloc2 = min(remaining, 80)
                a_opt2, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc2)
                try_seq(a_opt2, "peak_cd2")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polish - alternate methods ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        for ri in range(12):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if ri % 3 == 0:
                noise = rng.standard_normal(n_best) * 0.0005
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 20)
                a_opt, _ = fw_optimize(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{ri}")
            elif ri % 3 == 1:
                noise = rng.standard_normal(n_best) * 0.0004
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 15)
                a_opt, _ = entropy_mirror_descent(start, time_limit=t_alloc,
                                                  eta0=0.02, T=0.0005, top_k=15)
                try_seq(a_opt, f"polish_emd_{ri}")
            else:
                t_alloc = min(remaining, 15)
                a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_cd_{ri}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
