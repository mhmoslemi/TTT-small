    rho = np.where(cond.astype(bool))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(cond[:rho[-1]+1] * 1.0 - theta, 0.0)

def fast_convolve(a):
    """Efficiently computes the full convolution using FFT."""
    n = len(a)
    size = 2*n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fft_size = 1 << (fft_size-1).bit_length()   # Ensure power of two
    A = np.fft.rfft(a, fft_size).flatten()
    conv = np.fft.irfft(A * A, fft_size).real[:size]
    return np.clip(conv, 0, 1000)   # Simple bound matching the evaluator

def solve_lp_oracle(f, M):
    """Uses linear programming to find the optimal perturbation direction: maximize sum(g) subject to slow_convolve(f,g) <= M and g>=0."""
    n = len(f)
    if n == 0:
        return None
    n_rows = 2*n - 1
    # Build the Toeplitz matrix that represents the convolution for all lags
    A = np.zeros((n_rows, n))
    for k in range(n_rows):
        for j in range(n):
            if 0 <= k - j < n:
                A[k, j] = f[j]
    # Coefficients for maximization (-1 for each variable)
    c = -np.ones(n)
    # Constant vector for all inequalities (saturating bound)
    b_ub = np.ones(n_rows) * M
    bounds = [(0, None)] * n
    # Solve the LP
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, options={'presolve': True}, method='simplex')
        if res.success and res.status == 0:
            g = res.x
        elif res.status == 3:
            g = None
        elif res.status == 4:
            g = project_simplex(res.x)
        else:
            g = None
    except np.linalg.LinAlgError:
        try:
            # Fallback to dense method
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
            if res.success == 0:  # Not necessarily success, can be other status
                g = res.x
            else:
                g = res.x
        except:
            g = None
    except Exception as e:
        print(f"LP Oracle Error: {e}")
        return None
        
    if np.isfinite(g) and g.shape[0] == n:  # Actually, if we got an array then shape (1,) for 1D
        # Break the potential infinity values and enforce bounds
        if np.sum(g) < np.finfo(float).eps * n:
            g = project_simplex(g)  # Fallback to uniform distribution
        return g
    return None

def evaluate_sequence(seq):
    """Wrapper to avoid modifying the objective and to ensure we catch errors."""
    n = len(seq)
    b_sequence = np.convolve(seq, seq)  # Using standard convolve will produce 2*n -1 elements, but wait...
    # Correction: we are now using our own lpc_optimize which uses the full convolution matrix
    # So the objective function is defined as above
    n_val = len(b_sequence)
    b_val = b_sequence
    b_sum = sum(seq)
    if b_sum < np.finfo(float).eps * n:
        b_sum = np.sum([(1 if x==1 else 0) for x in seq])  # Fall back to integer counts
    if b_sum == 0:
        return float('inf')
    # Protect against non-finite values
    if not np.isfinite(b_sum) or not np.isfinite(any(seq)) or not np.isfinite(b_sequence).all():
        print("Invalid sequence computed in evaluate_sequence!")
        return float('inf')
    try:
        # The maximum of the convolution must be considered
        max_b = max(b_sequence) if n_val > 0 else 0.0
        return (2 * n * max_b / (b_sum ** 2))
    except Exception as e:
        print(f"Evaluation error: {e}")
        return float('inf')
