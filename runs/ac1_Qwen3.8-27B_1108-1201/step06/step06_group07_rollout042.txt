import numpy as np
from scipy.fft import rfft, irfft
import time

def auto_conv(a):
    """Compute linear auto-convolution using FFT."""
    n = len(a)
    size = 1 << (2*n - 2).bit_length()
    A = rfft(a, size)
    return irfft(A * A, size)[:2*n-1]

def cross_corr_fft(p, a):
    """Compute cross-correlation: result[i] = sum_j p[i+j] * a[j] for i in [0, len(a)-1]."""
    n = len(a)
    # Use the relation: cross_corr = conv(p, a[::-1])[n-1 : 2*n-1]
    size = 1 << (len(p) + n - 2).bit_length()
    P = rfft(p, size)
    A = rfft(a[::-1], size)
    prod = irfft(P * A, size)
    return prod[n-1:2*n-1]

def objective_and_grad(a, beta):
    """Compute smooth objective and its gradient.
    Objective: (2n/beta) * logsumexp(beta * conv(a,a))
    """
    n = len(a)
    conv = auto_conv(a)
    
    # Softmax weights (numerically stable)
    bconv = beta * conv
    bconv_shifted = bconv - bconv.max()
    p = np.exp(bconv_shifted)
    p_sum = p.sum()
    p = p / p_sum
    
    obj = (2.0 * n / beta) * (np.log(p_sum) + bconv.max())
    
    # Gradient
    cross = cross_corr_fft(p, a)  # cross[i] = sum_k p[k+i] * a[k]
    
    # Window sum: ws[i] = sum_{k=i}^{i+n-1} p[k]
    p_ext = np.concatenate([p, np.zeros(n-1)])
    ws = np.array([np.sum(p_ext[i:i+n]) for i in range(n)])  # slow!
    # Better: use cumsum
    p_cum = np.concatenate([[0.0], np.cumsum(p)])
    upper = np.minimum(np.arange(n) + n, len(p))
    ws = p_cum[upper] - p_cum[np.arange(n)]
    
    grad = 2.0 * n * (cross + a * ws)
    return obj, grad, conv

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def projected_gradient_descent(a0, time_limit, beta_init=10.0, beta_max=500.0, lr_init=0.01):
    """Projected gradient descent with adaptive beta."""
    a = a0.copy()
    a = project_simplex(a)
    n = len(a)
    
    best_a = a.copy()
    best_obj = auto_conv(a).max()  # Use actual max, not smooth
    
    beta = beta_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        obj, grad, conv = objective_and_grad(a, beta)
        
        # Projected gradient step
        a_new = project_simplex(a - lr * grad)
        
        # Line search
        best_t = 0.0
        best_val = conv.max()
        for t in [0.5, 0.7, 0.85, 1.0]:
            cand = a + t * (a_new - a)
            cand = project_simplex(cand)
            val = auto_conv(cand).max()
            if val < best_val:
                best_val = val
                best_t = t
        
        if best_t > 0:
            a = a + best_t * (a_new - a)
            a = project_simplex(a)
        
        current_obj = auto_conv(a).max()
        if current_obj < best_obj - 1e-15:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive beta: increase to make approximation sharper
        if no_improve % 10 == 0 and beta < beta_max:
            beta *= 1.5
        
        # Adaptive learning rate
        if no_improve > 20:
            lr *= 0.8
        if lr < 1e-6:
            lr = lr_init
            no_improve = 0
        
        if no_improve > 100:
            # Restart
            rng = np.random.default_rng(it * 7919 + 13)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            beta = beta_init
            lr = lr_init
            no_improve = 0
        
        if it % 50 == 0:
            print(f"  PGD it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.2e}")
    
    return best_a, best_obj

def smooth_obj_and_grad(x, beta):
    n = len(x)
    x_sh = x - x.max()
    a = np.exp(x_sh)
    a_sum = a.sum()
    a = a / a_sum
    
    conv = auto_conv(a)
    bconv = beta * conv
    bconv_sh = bconv - bconv.max()
    p = np.exp(bconv_sh)
    p_sum = p.sum()
    p_n = p / p_sum
    
    obj = (2.0 * n / beta) * (np.log(p_sum) + bconv.max())
    
    cross = cross_corr_fft(p_n, a)
    p_cum = np.concatenate([[0.0], np.cumsum(p_n)])
    upper = np.minimum(np.arange(n) + n, len(p_n))
    ws = p_cum[upper] - p_cum[np.arange(n)]
    
    grad_a = 2.0 * n * (cross + a * ws)
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lfgs_optimize(a0, time_limit, beta=50.0, max_iter=500):
    """Optimize using L-BFGS-B in softmax space."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    # Initialize x from a
    x = np.log(a)
    x = x - x.mean()  # Remove flat direction
    
    t0 = time.monotonic()
    
    def fun_and_grad(x):
        if time.monotonic() - t0 > time_limit:
            raise TimeoutError()
        obj, grad = smooth_obj_and_grad(x, beta)
        return obj, grad
    
    try:
        res = minimize(fun_and_grad, x, method='L-BFGS-B', jac=True,
                      options={'maxiter': max_iter, 'maxfun': max_iter*2,
                               'ftol': 1e-14, 'gtol': 1e-10,
                               'maxls': 50})
        if res.success or res.fun < float('inf'):
            x_opt = res.x
            a_opt = np.exp(x_opt - x_opt.max())
            a_opt = a_opt / a_opt.sum()
            return a_opt
    except (TimeoutError, Exception) as e:
        pass
    
    return a

def peak_shave_refine(a0, time_limit):
    """Iteratively shave the highest peak."""
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = auto_conv(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = auto_conv(a)
        k_peak = int(np.argmax(conv))
        
        # Find positions contributing to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        
        # Compute contribution of each position
        contrib = np.zeros(n)
        for j in range(js, je+1):
            contrib[j] = a[j] * a[k_peak - j]
        
        total = contrib.sum()
        if total < 1e-15:
            break
        
        # Reduce high-contrib positions, add to low-contrib
        frac = 0.05
        delta = np.zeros(n)
        for j in range(n):
            if contrib[j] > 0:
                delta[j] = -frac * a[j] * contrib[j] / total
        
        # Add mass to positions outside the peak span
        outside = np.array([j for j in range(n) if j < js or j > je])
        if len(outside) == 0:
            outside = np.array([j for j in range(n) if j != js and j != je])
        
        if len(outside) > 0:
            total_move = -delta.sum()
            if total_move > 0:
                ext_vals = a[outside]
                ext_sum = ext_vals.sum()
                if ext_sum > 1e-15:
                    add = total_move * ext_vals / ext_sum
                else:
                    add = np.full(len(outside), total_move / len(outside))
                for idx, j in enumerate(outside):
                    delta[j] += add[idx]
        
        a_new = a + delta
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        new_obj = auto_conv(a_new).max()
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            frac *= 0.9
        
        if no_improve > 50:
            break
    
    return best_a, best_obj

def subgradient_refine(a0, time_limit, top_k=10):
    """Subgradient method with line search for final refinement."""
    a = a0.copy()
    a = project_simplex(a)
    n = len(a)
    best_a = a.copy()
    best_obj = auto_conv(a).max()
    lr = 0.01
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = auto_conv(a)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        weights = pv / pv.sum()
        
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = weights[idx]
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n-1, k)
            # grad contribution: sum over j in [js, je] of w * (a[j]*a[k-j] derivative)
            # ∂conv[k]/∂a[i] = (a[k-i] + a[i]) for i in [max(0,k-n+1), min(n-1,k)]
            for i in range(js, je+1):
                grad[i] += w * (a[k-i] + a[i])
        
        # Projected gradient step
        a_new = project_simplex(a - lr * grad)
        
        # Line search
        best_t = 0.0
        best_val = conv.max()
        for t in [0.3, 0.5, 0.7, 1.0]:
            cand = a + t * (a_new - a)
            cand = project_simplex(cand)
            val = auto_conv(cand).max()
            if val < best_val:
                best_val = val
                best_t = t
        
        if best_t > 0:
            a = project_simplex(a + best_t * (a_new - a))
        
        obj = auto_conv(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            lr *= 0.9
        if lr < 1e-5:
            lr = 0.01
            no_improve = 0
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 31831 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            no_improve = 0
            lr = 0.01
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Smooth max (log-sum-exp) with adaptive temperature
- L-BFGS-B optimization in softmax parameterization (top-level, no closures)
- FFT-based gradient computation for O(n log n) per iteration
- Multi-n exploration with interpolation
- Peak shaving refinement
- Subgradient with line search
- Diversified restarts with structured perturbations
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

# Global state
_best_seq = None
_best_score = float('inf')
_current_beta = 50.0
_current_time_limit = 0.0
_current_start_time = 0.0


def auto_conv(a):
    """Compute linear auto-convolution using FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, size)
    result = irfft(A * A, size)[:2 * n - 1]
    return np.maximum(result, 0.0)


def cross_corr_fft(p, a):
    """Compute cross-correlation: result[i] = sum_j p[i+j] * a[j] for i in [0, len(a)-1]."""
    n = len(a)
    size = 1 << (len(p) + n - 2).bit_length()
    P = rfft(p, size)
    A = rfft(a[::-1], size)
    prod = irfft(P * A, size)
    return prod[n - 1:2 * n - 1]


def compute_grad_from_weights(p, a):
    """
    Compute gradient given weights p over convolution lags.
    grad[i] = 2n * sum_k p[k] * (a[k-i] + a[i]) * indicator(0<=k-i<=n-1)
    """
    n = len(a)
    cross = cross_corr_fft(p, a)
    p_cum = np.concatenate([[0.0], np.cumsum(p)])
    upper = np.minimum(np.arange(n) + n, len(p))
    ws = p_cum[upper] - p_cum[np.arange(n)]
    return 2.0 * n * (cross + a * ws)


def smooth_obj_and_grad(x, beta):
    """Compute smooth objective and gradient in softmax parameterization."""
    n = len(x)
    x_sh = x - x.max()
    a = np.exp(x_sh)
    a_sum = a.sum()
    a = a / a_sum

    conv = auto_conv(a)
    bconv = beta * conv
    bconv_sh = bconv - bconv.max()
    p = np.exp(bconv_sh)
    p_sum = p.sum()
    p_n = p / p_sum

    obj = (2.0 * n / beta) * (np.log(p_sum) + bconv.max())

    cross = cross_corr_fft(p_n, a)
    p_cum = np.concatenate([[0.0], np.cumsum(p_n)])
    upper = np.minimum(np.arange(n) + n, len(p_n))
    ws = p_cum[upper] - p_cum[np.arange(n)]

    grad_a = 2.0 * n * (cross + a * ws)
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x


def _fun_and_grad(x):
    """Top-level function for scipy.optimize.minimize (no closure)."""
    if time.monotonic() > _current_time_limit:
        raise RuntimeError("Time limit exceeded")
    return smooth_obj_and_grad(x, _current_beta)


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_x(x):
    """Convert x to probability vector via softmax."""
    x_sh = x - x.max()
    a = np.exp(x_sh)
    s = a.sum()
    return a / s


def lbfgs_optimize(a0, time_limit, beta=50.0, max_iter=500):
    """Optimize using L-BFGS-B in softmax parameterization space."""
    global _current_beta, _current_time_limit, _current_start_time
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()

    x = np.log(a)
    x = x - x.mean()

    _current_beta = beta
    _current_start_time = time.monotonic()
    _current_time_limit = _current_start_time + time_limit

    try:
        res = minimize(_fun_and_grad, x, method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                'ftol': 1e-15, 'gtol': 1e-12,
                                'maxls': 40, 'iprint': -1})
        x_opt = res.x
        a_opt = softmax_from_x(x_opt)
        return a_opt
    except (RuntimeError, Exception):
        # Time limit or other error: return current best from the vector
        # We don't have access to the in-progress state, so return a0
        return a0


def peak_shave_refine(a0, time_limit):
    """Iteratively shave the highest convolution peak."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0, auto_conv(a0).max()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = auto_conv(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    frac = 0.05

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = auto_conv(a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        contrib = np.zeros(n)
        for j in range(js, je + 1):
            contrib[j] = a[j] * a[k_peak - j]

        total = contrib.sum()
        if total < 1e-15:
            break

        delta = np.zeros(n)
        for j in range(n):
            if contrib[j] > 0:
                delta[j] = -frac * a[j] * contrib[j] / total

        outside = np.array([j for j in range(n) if j < js or j > je])
        if len(outside) == 0:
            outside = np.array([j for j in range(n) if j != js and j != je])

        if len(outside) > 0:
            total_move = -delta.sum()
            if total_move > 0:
                ext_vals = a[outside]
                ext_sum = ext_vals.sum()
                if ext_sum > 1e-15:
                    add = total_move * ext_vals / ext_sum
                else:
                    add = np.full(len(outside), total_move / len(outside))
                for idx, j in enumerate(outside):
                    delta[j] += add[idx]

        a_new = a + delta
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        new_obj = auto_conv(a_new).max()
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            frac *= 0.95

        if no_improve > 60:
            frac = 0.05
            rng = np.random.default_rng(it * 9973 + 5)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_a, best_obj


def subgradient_refine(a0, time_limit, top_k=10):
    """Subgradient method with FFT-based gradient and line search."""
    a = project_simplex(a0.astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = auto_conv(a).max()
    lr = 0.005

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = auto_conv(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pv = conv[peaks]
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
        weights = pv / pv.sum()

        # Build weight vector p (length 2n-1)
        p = np.zeros(2 * n - 1)
        p[peaks] = weights

        grad = compute_grad_from_weights(p, a)

        a_new = project_simplex(a - lr * grad)
        direction = a_new - a

        # Line search
        best_t = 0.0
        best_val = conv.max()
        for t in [0.3, 0.5, 0.7, 1.0]:
            cand = project_simplex(a + t * direction)
            val = auto_conv(cand).max()
            if val < best_val:
                best_val = val
                best_t = t

        if best_t > 0:
            a = project_simplex(a + best_t * direction)

        obj = auto_conv(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            lr *= 0.85
        if lr < 1e-6:
            lr = 0.005
            no_improve = 0

        if no_improve > 80:
            rng = np.random.default_rng(it * 31831 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
            lr = 0.005

    return best_a, best_obj

def structured_perturb(a0, rng, mode=0):
    """Generate structured perturbations of a0."""
    n = len(a0)
    x = np.arange(n)
    if mode == 0:
        # Sinusoidal perturbation
        freq = rng.integers(1, n // 4 + 1)
        amp = rng.uniform(0.005, 0.03)
        pert = a0 + amp * np.sin(2 * np.pi * freq * x / n)
    elif mode == 1:
        # Bump perturbation (Gaussian bump)
        center = rng.integers(n // 4, 3 * n // 4)
        width = rng.uniform(n * 0.05, n * 0.2)
        amp = rng.uniform(0.01, 0.05) * rng.choice([-1, 1])
        pert = a0 + amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    elif mode == 2:
        # Edge emphasis
        edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.5
        amp = rng.uniform(0.01, 0.04)
        pert = a0 + amp * (edge_w - edge_w.mean())
    elif mode == 3:
        # Anti-symmetric perturbation
        t = (x - (n - 1) / 2.0) / (n - 1)
        amp = rng.uniform(0.01, 0.03)
        pert = a0 + amp * np.tanh(3 * t)
    else:
        # Random
        pert = a0 + rng.standard_normal(n) * 0.01
    return project_simplex(pert)
