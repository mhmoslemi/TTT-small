def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, k_star):
    """Compute subgradient of max(conv) at a when max is at index k_star.
    g[j] = 2*a[k_star - j] if 0 <= k_star - j < n, else 0."""
    n = len(a)
    g = np.zeros(n)

    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    if j_min <= j_max:
        g[j_min:j_max+1] = 2.0 * a[k_star - j_max : k_star - j_min - 1 : -1]
    return g

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, k_star):
    """Subgradient of max(conv) at a when max is at index k_star."""
    n = len(a)
    g = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    if j_min <= j_max:
        g[j_min:j_max+1] = 2.0 * a[k_star - j_max : k_star - j_min - 1 : -1]
    return g

def compute_weighted_subgradient(a, k_indices, weights):

def spectral_equalize_step(a, n_smooth=3, alpha=0.5):
    """One spectral equalization step."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    mag = np.abs(A)
    phase = np.angle(A)
    if n_smooth > 1 and len(mag) > n_smooth:
        kernel = np.ones(n_smooth) / n_smooth
        mag_smooth = np.convolve(mag, kernel, mode='same')
        mag_smooth = np.maximum(mag_smooth, 1e-15)
        mag_new = (mag ** (1.0 - alpha)) * (mag_smooth ** alpha)
    else:
        mag_new = mag.copy()
    A_new = mag_new * phase
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    return np.maximum(a_new, 0.0)

                no_improve_sweeps = 0
        
        sweep += 1
        if sweep > 500:
            break
    
    return best_f, best_obj

def intensive_local_search(a, time_limit, seed=42):
    """Intensive local search with diverse move types."""
    f = np.maximum(a.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    conv = fast_convolve(f)
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        
        improved = False
        
        # Move 1: Tiny random perturbation
        for scale in [0.0001, 0.0003, 0.0008, 0.002]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(f + pert)
            new_conv = fast_convolve(candidate)
            if new_conv.max() < best_obj - 1e-17:
                f = candidate
                best_obj = new_conv.max()
                best_f = f.copy()
                conv = new_conv
                improved = True
                break
        
        # Move 2: Coordinate adjustment
        if not improved and it % 2 == 0:
            j = int(rng.integers(0, n))
            new_f, new_val, imp = coordinate_descent_line_search(f, j, conv)
            if imp:
                f = new_f
                conv = fast_convolve(f)
                if new_val < best_obj - 1e-17:
                    best_obj = new_val
                    best_f = f.copy()
                improved = True
        
        # Move 3: Pair transfer
        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    candidate = f.copy()
                    amount = candidate[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    if candidate[i] < -1e-15:
                        continue
                    candidate[i] = max(0.0, candidate[i])
                    candidate = candidate / candidate.sum()
                    new_conv = fast_convolve(candidate)
                    if new_conv.max() < best_obj - 1e-17:
                        f = candidate
                        best_obj = new_conv.max()
                        best_f = f.copy()
                        conv = new_conv
                        improved = True
                        break
        
        # Move 4: Spectral nudge
        if not improved and it % 8 == 0:
            f_spec = spectral_equalize_step(f, 2, 0.3)
            ss = f_spec.sum()
            if ss > 1e-15:
                f_spec /= ss
                for blend in [0.05, 0.1, 0.2]:
                    candidate = (1 - blend) * f + blend * f_spec
                    candidate = project_simplex(candidate)
                    new_conv = fast_convolve(candidate)
                    if new_conv.max() < best_obj - 1e-17:
                        f = candidate
                        best_obj = new_conv.max()
                        best_f = f.copy()
                        conv = new_conv
                        improved = True
                        break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 500:
                pert = rng.standard_normal(n) * 0.005
                f = project_simplex(best_f + pert)
                conv = fast_convolve(f)
                stall = 0
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_f = f.copy()
        
        if stall > 5000:
            break
    
    return best_f, best_obj

def try_seq(a, label, best_ref):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref

def fw_lp_single_step(f, M, n):
    """Single FW-LP step: solve LP oracle and line search."""
    # Build constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    
    if len(rows) == 0:
        return None, 0.0, M
    
    rows = np.array(rows)
    cols = np.array(cols)
    vals = np.array(vals)
    
    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[rows, cols] = vals
    A_mat = A_mat.tocsr()
    
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.fun is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                g = g / s
                # Line search
                best_alpha = 0.0
                best_val = M
                for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    ft = (1 - alpha) * f + alpha * g
                    ft = np.maximum(ft, 0)
                    st = ft.sum()
                    if st < 1e-15:
                        continue
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = alpha
                return g, best_alpha, best_val
    except Exception:
        pass
    return None, 0.0, M

def fw_lp_optimize(a, time_limit, max_LP_iters=50):
    """FW-LP optimization (limited iterations for speed)."""
    f = np.maximum(a.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = M
    m_exp = 1.0
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < max_LP_iters:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        g, alpha, val = fw_lp_single_step(f, M * m_exp, n)
        
        if g is None:
            m_exp = min(m_exp * 1.5, 5.0)
            continue
        
        if alpha > 1e-10 and val < M - 1e-15:
            f = (1 - alpha) * f + alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            if M < best_obj:
                best_obj = M
                best_f = f.copy()
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.3, 5.0)
            if no_improve > 20:
                break
    
    return best_f, best_obj

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: FW-LP bootstrap -> RFG line search -> spectral -> coord descent -> local search.
    Focuses on precise optimization near the parent solution.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    t_start = time.monotonic()
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        n_default = 768
        parent = np.ones(n_default) / n_default
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Uniform init: score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # === Phase 1: Quick FW-LP to establish good baseline ===
    print(f"\n=== Phase 1: FW-LP Bootstrap (n={n_parent}) ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    t_alloc = min(remaining * 0.15, 200)
    a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, max_LP_iters=30)
    best_seq, best_score = try_seq(a_opt, "fw_boot", (best_seq, best_score))
    print(f"  Phase 1: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 2: RFG line search (main workhorse) ===
    print(f"\n=== Phase 2: RFG Line Search ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    # Long RFG run
    t_alloc = min(remaining * 0.25, 350)
    print(f"  RFG (lr=0.01) n={n_best}, t={t_alloc:.0f}s")
    a_opt, _ = rfg_line_search(best_seq, time_limit=t_alloc, lr_init=0.01)
    best_seq, best_score = try_seq(a_opt, "rfg_main", (best_seq, best_score))
    
    # Smoother RFG
    remaining = deadline - time.monotonic()
    if remaining > 50:
        t_alloc = min(remaining * 0.15, 150)
        print(f"  RFG smoothed (T=0.001), t={t_alloc:.0f}s")
        a_opt, _ = rfg_line_search(best_seq, time_limit=t_alloc, lr_init=0.005,
                                   use_smoothed=True, T_init=0.001)
        best_seq, best_score = try_seq(a_opt, "rfg_smooth", (best_seq, best_score))
    
    # Slower RFG for fine tuning
    remaining = deadline - time.monotonic()
    if remaining > 40:
        t_alloc = min(remaining * 0.10, 100)
        print(f"  RFG slow (lr=0.003), t={t_alloc:.0f}s")
        a_opt, _ = rfg_line_search(best_seq, time_limit=t_alloc, lr_init=0.003)
        best_seq, best_score = try_seq(a_opt, "rfg_slow", (best_seq, best_score))
    
    print(f"  Phase 2: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 3: Spectral equalization ===
    print(f"\n=== Phase 3: Spectral Equalization ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    
    if remaining > 40:
        t_alloc = min(remaining * 0.12, 120)
        print(f"  Spectral (smooth=3, alpha=0.4), t={t_alloc:.0f}s")
        a_opt, _ = spectral_loop(best_seq, time_limit=t_alloc, n_smooth=3, alpha=0.4)
        best_seq, best_score = try_seq(a_opt, "spec_main", (best_seq, best_score))
    
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.08, 60)
        print(f"  Spectral (smooth=2, alpha=0.3), t={t_alloc:.0f}s")
        a_opt, _ = spectral_loop(best_seq, time_limit=t_alloc, n_smooth=2, alpha=0.3)
        best_seq, best_score = try_seq(a_opt, "spec_fine", (best_seq, best_score))
    
    print(f"  Phase 3: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate Descent ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    
    if remaining > 30:
        t_alloc = min(remaining * 0.15, 150)
        print(f"  Coord descent, t={t_alloc:.0f}s")
        a_opt, _ = coord_descent_refine(best_seq, time_limit=t_alloc, seed=123)
        best_seq, best_score = try_seq(a_opt, "coord1", (best_seq, best_score))
    
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.10, 80)
        a_opt, _ = coord_descent_refine(best_seq, time_limit=t_alloc, seed=456)
        best_seq, best_score = try_seq(a_opt, "coord2", (best_seq, best_score))
    
    print(f"  Phase 4: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 5: Multi-n exploration (brief) ===
    print(f"\n=== Phase 5: Multi-n Check ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    
    for n_try in [n_best - 64, n_best + 64, n_best // 2 * 2 // 128 * 128]:
        if n_try < 256 or n_try > 1536 or n_try == n_best:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        
        start = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_best), best_seq)
        start = np.maximum(start, 0)
        ss = start.sum()
        start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try
        
        t_alloc = min(remaining * 0.08, 40)
        a_opt, _ = rfg_line_search(start, time_limit=t_alloc, lr_init=0.008)
        best_seq, best_score = try_seq(a_opt, f"n_try_{n_try}", (best_seq, best_score))
    
    print(f"  Phase 5: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 6: Intensive local search (final polish) ===
    print(f"\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic()
    n_best = len(best_seq)
    
    # Intensive local search
    if remaining > 30:
        t_alloc = min(remaining * 0.25, 200)
        print(f"  Intensive local search, t={t_alloc:.0f}s")
        a_opt, _ = intensive_local_search(best_seq, time_limit=t_alloc, seed=777)
        best_seq, best_score = try_seq(a_opt, "local1", (best_seq, best_score))
    
    # More coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.15, 80)
        a_opt, _ = coord_descent_refine(best_seq, time_limit=t_alloc, seed=888)
        best_seq, best_score = try_seq(a_opt, "coord_final", (best_seq, best_score))
    
    # Tiny RFG
    remaining = deadline - time.monotonic()
    if remaining > 15:
        t_alloc = min(remaining * 0.10, 50)
        a_opt, _ = rfg_line_search(best_seq, time_limit=t_alloc, lr_init=0.001)
        best_seq, best_score = try_seq(a_opt, "rfg_tiny", (best_seq, best_score))
    
    # Another local search
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt, _ = intensive_local_search(best_seq, time_limit=min(remaining * 0.3, 30), seed=999)
        best_seq, best_score = try_seq(a_opt, "local2", (best_seq, best_score))
    
    # Final micro-perturbations
    remaining = deadline - time.monotonic()
    if remaining > 5:
        for _ in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 2:
                break
            noise = rng.standard_normal(n_best) * 0.0002
            f_pert = project_simplex(best_seq + noise)
            # Quick eval
            conv = fast_convolve(f_pert)
            if conv.max() < fast_convolve(best_seq).max() - 1e-17:
                a_opt, _ = rfg_line_search(f_pert, time_limit=min(remaining * 0.3, 5), lr_init=0.001)
                best_seq, best_score = try_seq(a_opt, "micro", (best_seq, best_score))
    
    print(f"  Phase 6: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
