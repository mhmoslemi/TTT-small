def smoothed_gradient_descent(a0, time_limit, T_init=0.001, T_min=0.0001, lr_init=0.01):
    """Smoothed gradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        # Compute smooth max gradient
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        grad = 2.0 * np.correlate(p, a, mode='valid')
        
        # Gradient step with backtracking
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature and learning rate
        if no_improve > 20:
            T *= 1.05
            lr *= 0.95
        if no_improve > 100:
            T = max(T * 0.7, T_min)
        if no_improve > 300:
            # Restart with perturbation
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

def simulated_annealing(a0, time_limit, T0=0.01, T_end=0.0001):
    """Simulated annealing with structured peak-targeting moves."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    T = T0
    t0 = time.monotonic()
    total_time = time_limit
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = elapsed / total_time
        T = T0 * (T_end / T0) ** progress  # Exponential cooling
        
        # Generate a move
        rng = np.random.default_rng()
        move_type = rng.integers(0, 3)
        
        if move_type == 0:
            # Move mass between two random coordinates
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            amount = a[i] * rng.uniform(0.05, 0.5)
            a_new = a.copy()
            a_new[i] -= amount
            a_new[j] += amount
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
        elif move_type == 1:
            # Peak-targeting: move mass away from peak-contributing positions
            conv = fast_convolve(a)
            k_peak = int(np.argmax(conv))
            # Positions contributing to peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            i = rng.integers(js, je + 1)
            # Move to a position that doesn't contribute
            non_contrib = [j for j in range(n) if j < js or j > je]
            if non_contrib:
                j = rng.choice(non_contrib)
                amount = a[i] * rng.uniform(0.1, 0.4)
                a_new = a.copy()
                a_new[i] -= amount
                a_new[j] += amount
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
            else:
                continue
                
        else:
            # Block noise: add Gaussian noise to a random block
            block_size = rng.integers(max(2, n//50), max(3, n//10))
            start = rng.integers(0, n - block_size)
            a_new = a.copy()
            a_new[start:start+block_size] += rng.standard_normal(block_size) * (a[start:start+block_size].mean() * 0.1)
            a_new = project_simplex(a_new)
        
        obj_new = fast_convolve(a_new).max()
        obj_cur = fast_convolve(a).max()
        
        # Metropolis acceptance
        if obj_new < obj_cur or rng.random() < np.exp(-(obj_new - obj_cur) / max(T, 1e-10)):
            a = a_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
    
    return best_a, best_obj

Actually, let me think about this more carefully. If we only change a[i] and a[j], keeping everything else fixed:

The terms that involve a[i] or a[j] are:
- `a[i] * a[k-i]` (when m=i, valid if 0≤k-i<n)

- Cross terms: `a[i]*a[j]` (when k=i+j, m=i, k-m=j) and `a[j]*a[i]` (when k=i+j, m=j, k-m=i)

So for a given k:
- If none of i, j, k-i, k-j are in [0,n-1] in the right way, conv(k) doesn't change.
- Otherwise, conv(k) is a quadratic in (a[i], a[j]).

With the constraint a[i] + a[j] = s, we parameterize as a[i] = α*s, a[j] = (1-α)*s.

Then conv(k) becomes a quadratic in α: `c0 + c1*α + c2*α^2` for some coefficients.

"""
Hybrid optimization for minimizing 2*n*max(auto_corr)/(sum)^2.
Uses smoothed gradient descent, simulated annealing, and block coordinate descent.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    """Compute 2*n*max_conv for sum-normalized a."""
    conv = fast_convolve(a)
    return conv.max()

def grad_smooth_max(a, T):
    """Gradient of T*log(sum(exp(conv/T))) w.r.t. a."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    # grad[i] = 2 * sum_j p[i+j] * a[j]
    return 2.0 * np.correlate(p, a, mode='valid')

def smoothed_grad_desc(a0, time_limit, T_init=0.0005, T_decay=0.999, lr_init=0.02):
    """Smoothed gradient descent with adaptive temperature."""
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = objective(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        grad = grad_smooth_max(a, T)
        
        # Line search
        best_candidate = a
        best_val = objective(a)
        for trial in [lr, lr*0.5, lr*0.25, lr*0.1]:
            cand = project_simplex(a - trial * grad)
            obj = objective(cand)

def sim_anneal(a0, time_limit, T0=0.005, T1=0.0001):
    """Simulated annealing with peak-targeting moves."""
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = objective(a)
    cur_obj = best_obj
    t0 = time.monotonic()
    total = time_limit
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = elapsed / total
        T = T0 * (T1/T0)**frac
        
        rng = np.random.default_rng()
        move = rng.integers(0, 4)
        
        if move == 0:
            # Random 2-swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j: continue
            amt = min(a[i], a[j]) * rng.uniform(0.1, 0.6)
            if amt < 1e-12: continue
            a_new = a.copy()
            if a[i] >= a[j]:
                a_new[i] -= amt
                a_new[j] += amt
            else:
                a_new[j] -= amt
                a_new[i] += amt
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15: continue
            a_new /= s
            
        elif move == 1:
            # Peak-targeting move
            conv = fast_convolve(a)
            k_peak = int(np.argmax(conv))
            js = max(0, k_peak - n + 1)
            je = min(n-1, k_peak)
            # Pick a contributing position
            i = rng.integers(js, je+1)
            # Pick a non-contributing position
            non_contrib = np.array([j for j in range(n) if j < js or j > je])
            if len(non_contrib) == 0: continue
            j = non_contrib[rng.integers(len(non_contrib))]
            amt = a[i] * rng.uniform(0.05, 0.3)
            if amt < 1e-12: continue
            a_new = a.copy()
            a_new[i] -= amt
            a_new[j] += amt
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15: continue
            a_new /= s
            
        elif move == 2:

            bs = rng.integers(2, max(3, n//20))
            start = rng.integers(0, n - bs)
            a_new = a.copy()
            block_mean = a_new[start:start+bs].mean()

"""
Hybrid optimization for minimizing 2*n*max(auto_corr)/(sum)^2 using:
- Smoothed gradient descent (log-sum-exp) with adaptive temperature
- Simulated annealing with peak-targeting structured moves
- Block coordinate descent with golden-section 2D optimization
- Multi-n exploration and perturbation-based restarts
- ADMM-style proximal splitting for accelerated convergence
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective(a):
    """Compute max(auto-correlation) for sum-normalized a."""
    conv = fast_convolve(a)
    return conv.max()


def grad_smooth_max(a, T):
    """Gradient of T*log(sum(exp(conv/T))) w.r.t. a, computed via FFT correlation."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    return 2.0 * np.correlate(p, a, mode='valid')


def smoothed_grad_desc(a0, time_limit, T_init=0.0005, lr_init=0.02, restart_every=200):
    """Smoothed gradient descent with adaptive temperature and restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = objective(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        grad = grad_smooth_max(a, T)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break

        # Adaptive learning rate based on gradient magnitude
        eff_lr = lr * 0.5 / max(norm_g, 0.1)

        # Backtracking line search
        best_candidate = a
        best_val = best_obj
        cur_val = objective(a)
        for trial in [eff_lr, eff_lr * 0.5, eff_lr * 0.25, eff_lr * 0.1, eff_lr * 0.05]:
            cand = project_simplex(a - trial * grad)
            val = objective(cand)
            if val < cur_val - 1e-15:
                best_val = val
                best_candidate = cand
                break

        a = best_candidate
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive scheduling
        if no_improve > 30:
            T *= 1.08
            lr *= 0.95
        if no_improve > restart_every:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))

            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0
            best_obj = objective(a)

    return best_a, best_obj


def admm_smooth(a0, time_limit, T_init=0.001, rho=10.0):
    """ADMM-style proximal splitting for the smoothed problem."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    z = a.copy()  # simplex variable
    u = np.zeros(n)  # dual variable
    T = T_init
    best_a = a.copy()
    best_obj = objective(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # a-update: minimize f(a) + (rho/2)||a - z + u||^2
        # Use a few gradient steps
        a_new = a.copy()
        for _ in range(3):
            conv = fast_convolve(a_new)
            scaled = conv / T
            scaled -= scaled.max()
            p = np.exp(scaled)
            p /= p.sum()
            grad_f = 2.0 * np.correlate(p, a_new, mode='valid')
            grad_q = rho * (a_new - z + u)
            step = 1.0 / (rho + 2.0 * np.max(p) * 2.0 * np.max(a_new) + 1e-10)
            a_new = a_new - step * (grad_f + grad_q)
            a_new = np.maximum(a_new, 0.0)

        # z-update: projection onto simplex
        z_new = project_simplex(a_new + u)

        # u-update
        u = u + a_new - z_new

        a = a_new
        # Check the simplex-feasible point
        a_feas = z_new
        obj = objective(a_feas)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_feas.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            T *= 1.05
            rho *= 1.1
        if no_improve > 200:

            z = a.copy()
            u = np.zeros(n)
            T = T_init
            rho = 10.0
            no_improve = 0

            pts = rng.choice(n, size=n_pts, replace=False)
            a_new = a.copy()
            mass = a_new[pts].sum()
            if mass < 1e-12:
                continue
            new_vals = rng.dirichlet(np.ones(n_pts) * 2.0) * mass
            a_new[pts] = new_vals
            a_new = project_simplex(a_new)

        else:
            # Global structured noise (sine-modulated)
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            noise = np.sin(np.linspace(0, freq * 2 * np.pi, n) + phase) * 0.01 * a
            a_new = project_simplex(a + noise)

        new_obj = objective(a_new)

        if new_obj <= cur_obj or (T > 1e-12 and rng.random() < np.exp(-(new_obj - cur_obj) / T)):
            a = a_new
            cur_obj = new_obj
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a.copy()

    return best_a, best_obj


def block_coord_descent(a0, time_limit, n_peaks=5, n_iters_gs=10):
    """Block coordinate descent with golden-section 2D optimization on peak-contributing pairs."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = objective(a)
    t0 = time.monotonic()
    stall = 0
    it = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        top_k = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]

        improved = False

        for k_pk in peaks:
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            js = max(0, int(k_pk) - n + 1)
            je = min(n - 1, int(k_pk))
            affected = list(range(js, je + 1))

            if len(affected) < 2:
                continue

            if len(affected) > 15:
                rng_s = np.random.default_rng((it * 13 + int(k_pk)) % (2**31))
                affected = list(rng_s.choice(affected, size=15, replace=False))

            for ii in range(len(affected)):
                for jj in range(ii + 1, len(affected)):
                    i = affected[ii]
                    j = affected[jj]
                    s_val = a[i] + a[j]
                    if s_val < 1e-10:
                        continue

                    # Golden section search
                    lo, hi = 0.0, 1.0
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)

                    a[i] = c1 * s_val
                    a[j] = (1.0 - c1) * s_val
                    f1 = objective(a)
                    a[i] = c2 * s_val
                    a[j] = (1.0 - c2) * s_val
                    f2 = objective(a)

                    for _ in range(n_iters_gs):
                        if f1 < f2:
                            hi = c2
                            c2 = c1
                            f2 = f1
                            c1 = hi - gr * (hi - lo)
                            a[i] = c1 * s_val
                            a[j] = (1.0 - c1) * s_val
                            f1 = objective(a)
                        else:
                            lo = c1
                            c1 = c2
                            f1 = f2
                            c2 = lo + gr * (hi - lo)
                            a[i] = c2 * s_val
                            a[j] = (1.0 - c2) * s_val
                            f2 = objective(a)

                    best_alpha = (lo + hi) / 2.0
                    a[i] = best_alpha * s_val
                    a[j] = (1.0 - best_alpha) * s_val
                    obj = objective(a)

                    if obj < best_obj - 1e-13:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a = best_a.copy()
                        break
                if improved:
                    break

            if improved:
                stall = 0
                break
            else:
                stall += 1

        if not improved:
            stall += 1

        if stall > 30:
            rng = np.random.default_rng((it * 37 + 3) % (2**31))
            noise_scale = 0.001 + 0.002 * min(1.0, stall / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            stall = 0

        if stall > 500:
            break

    return best_a, best_obj


def diverse_restart_search(a0, time_limit, n_restarts=5):
    """Try several perturbed restarts with different strategies."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = objective(a)
    t0 = time.monotonic()

    t_per = time_limit / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit - t_per * 0.5:
            break

        rng = np.random.default_rng((r * 104729 + 42) % (2**31))
        scale = 0.005 + 0.01 * r
        a_pert = project_simplex(a + rng.standard_normal(n) * scale)

        t_alloc = min(t_per * 0.5, time_limit - (time.monotonic() - t0) - 1.0)
        if t_alloc < 2:
            continue

        a_opt, obj = smoothed_grad_desc(a_pert, time_limit=t_alloc, T_init=0.0003, lr_init=0.03)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt.copy()

        t_alloc2 = min(t_per * 0.4, time_limit - (time.monotonic() - t0) - 1.0)
        if t_alloc2 < 2:
            continue

        a_opt2, obj2 = block_coord_descent(best_a, time_limit=t_alloc2, n_peaks=4, n_iters_gs=8)
        if obj2 < best_obj - 1e-14:
            best_obj = obj2
            best_a = a_opt2.copy()

    return best_a, best_obj
