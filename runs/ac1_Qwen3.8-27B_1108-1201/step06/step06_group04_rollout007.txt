import cvxpy as cp

def solve_qp_cvxpy(w, n):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 where W_{i,j}=w[i+j]"""
    a = cp.Variable(n)
    # Build W as a Hankel matrix
    # W_{i,j} = w[i+j]
    # For CVXPY, we can use quad_form with a PSD matrix
    W = np.array([[w[i+j] for j in range(n)] for i in range(n)])
    W = (W + W.T) / 2  # Symmetrize
    # Check PSD
    # If not PSD, add regularization
    obj = cp.Minimize(cp.quad_form(a, cp.psd_wrap(W)))
    constraints = [cp.sum(a) == 1, a >= 0]
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS)  # or ECOS, or CLARABEL
    return a.value
```

def qp_projected_grad(w, a_init, n, max_iters=1000):
    """Solve min a^T W a s.t. sum(a)=1, a>=0 using projected gradient."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    step_size = 1.0 / (2 * np.max(np.abs(w)))  # Lipschitz constant bound
    
    for it in range(max_iters):
        # Gradient: 2 * W @ a = 2 * correlate(w, a, 'valid')
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # Projected gradient step
        a_new = project_simplex(a - step_size * grad)
        if np.linalg.norm(a_new - a) < 1e-12:
            break
        a = a_new
    return a
```

def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    omega = np.arange(1, n + 1)
    cond = u * omega > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, v):
    """Compute W @ v where W_{i,j} = w[i+j]. w has length 2n-1, v has length n."""
    return np.correlate(w, v, mode='valid')


def qp_nesterov(w, a_init, n, max_iters=2000, step_size=0.45, time_limit=2.0):
    """
    Solve min a^T W a s.t. a in simplex using Nesterov-accelerated projected gradient.
    W is Hankel: W_{i,j} = w[i+j].
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    mom = 0.9
    best_a = a.copy()
    best_obj = float(a @ hankel_mv(w, a))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov lookahead
        y = a + mom * v

        # Gradient at y: 2 * W @ y
        grad = 2.0 * hankel_mv(w, y)

        # Projected gradient step
        y_new = project_simplex(y - step_size * grad)
        v = y_new - y
        a = y_new

        # Evaluate
        obj = float(a @ hankel_mv(w, a))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Update momentum (polyak)
        mom = (1.0 + it + 1) / (it + 3)

    return best_a, best_obj


def qp_active_set(w, a_init, n, max_iters=500, time_limit=3.0):
    """

        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n
        if len(active) == 1:

            return a_out

        act = np.array(active)
        k = len(act)

        W_AA = (W_AA + W_AA.T) / 2.0 + 1e-14 * np.eye(k)

        try:
            x = np.linalg.solve(W_AA, np.ones(k))
        except np.linalg.LinAlgError:
            try:

            except Exception:
                x = np.ones(k) / k

        x_sum = x.sum()
        if abs(x_sum) < 1e-14:

            x_sum = 1.0

        a_act = x / x_sum

        # Check for negative components
        min_idx_local = -1
        min_val = float('inf')
        for i in range(k):
            if a_act[i] < 0 and a_act[i] < min_val:
                min_val = a_act[i]
                min_idx_local = act[i]

        if min_idx_local >= 0:
            active.remove(min_idx_local)
            a_new = np.zeros(n)
            for i, idx in enumerate(act):

                    a_new[idx] = max(0.0, a_act[i])
            s2 = a_new.sum()
            if s2 > 1e-15:
                a = a_new / s2
            continue

        # Form solution
        a = np.zeros(n)
        for i, idx in enumerate(act):
            a[idx] = max(0.0, a_act[i])

        # KKT check
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa
        mu = np.mean([grad[idx] for idx in act])

        best_delta = None

        for i in range(i_min, i_max + 1):
            j = k_peak - i
            if i == j:
                continue  # Can't unbalance a self-pair easily
            if i > j:
                continue  # Avoid duplicate pairs (each pair considered once)
            ai = a[i]
            aj = a[j]
            if ai < 1e-12 and aj < 1e-12:
                continue

            # Direction: push toward larger
            if ai > aj:
                delta_i = 1.0
                delta_j = -1.0
            elif aj > ai:
                delta_i = -1.0
                delta_j = 1.0
            else:
                # Equal: try both directions
                for sign in [1, -1]:
                    for frac in [0.01, 0.05, 0.1, 0.2]:
                        shift = min(ai, aj) * frac * sign
                        if abs(shift) < 1e-14:
                            continue
                        d = np.zeros(n)
                        d[i] = shift
                        d[j] = -shift
                        candidate = project_simplex(a + d)
                        c = fast_conv(candidate).max()
                        if c < cmax - best_improvement - 1e-14:
                            best_improvement = cmax - c
                            best_delta = d
                continue

            for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                shift = min(ai, aj) * frac * step_frac * 10
                if shift > min(ai, aj) * 0.9:
                    shift = min(ai, aj) * 0.9
                if shift < 1e-14:
                    continue
                d = np.zeros(n)
                d[i] = delta_i * shift
                d[j] = delta_j * shift
                candidate = project_simplex(a + d)
                c = fast_conv(candidate).max()
                if c < cmax - best_improvement - 1e-14:
                    best_improvement = cmax - c
                    best_delta = d

        if best_delta is not None:
            a = project_simplex(a + best_delta)
            improved = True
        else:
            break

    return a


def primal_dual_iterate(a, n, T, dual_mode, max_inner=200, time_limit=5.0,
                        use_as=False, topk=10):
    """
    One iteration of primal-dual:
    1. Compute dual weights from conv(a,a)
    2. Solve primal QP
    3. Blend
    Returns (a_new, obj).
    """
    conv = fast_conv(a)
    cmax = conv.max()

    # Compute dual weights
    if dual_mode == 'softmax':
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
    elif dual_mode == 'topk':
        k_use = min(topk, len(conv))
        peaks = np.argpartition(conv, -k_use)[-k_use:]
        w = np.zeros(len(conv))
        w[peaks] = 1.0 / k_use
    elif dual_mode == 'power':
        w = np.maximum(conv, 0) ** 2
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(len(conv)) / len(conv)
        else:
            w = w / w_sum
    else:
        w = np.ones(len(conv)) / len(conv)

    # Solve QP
    if use_as:
        a_qp, _ = qp_active_set(w, a, n, max_iters=200, time_limit=min(2.0, time_limit * 0.5))
    else:
        a_qp, _ = qp_nesterov(w, a, n, max_iters=max_inner, step_size=0.45,
                               time_limit=min(time_limit, time_limit * 0.8))

    # Blend
    a_new = 0.5 * a + 0.5 * a_qp
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = a_qp
    else:
        a_new = a_new / s

    obj = fast_conv(a_new).max()
    return a_new, obj


def multi_strategy_search(a, n, time_limit, rng):
    """
    Main search loop combining primal-dual with peak unbalancing and SA.
    """
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_conv(a).max()
    T = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        it += 1

        # Choose strategy based on iteration
        if it % 30 == 0:
            # Peak unbalancing
            a_new = peak_unbalance(a, n, n_steps=10, step_frac=0.05)
            obj = fast_conv(a_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        elif it % 10 == 0:
            # Perturbation for escape
            noise_scale = 0.002 + 0.003 * min(no_improve / 100.0, 1.0)
            a_new = project_simplex(a + rng.standard_normal(n) * noise_scale)
            obj = fast_conv(a_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        else:
            # Primal-dual step
            # Cycle through dual modes
            mode_idx = (it // 5) % 3
            if mode_idx == 0:
                dual_mode = 'softmax'
            elif mode_idx == 1:
                dual_mode = 'topk'
            else:
                dual_mode = 'power'

            a_new, obj = primal_dual_iterate(a, n, T, dual_mode, max_inner=150,
                                             time_limit=min(1.0, remaining * 0.3),
                                             use_as=(it % 10 == 5), topk=8)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new

        # Adapt temperature
        if no_improve > 5:
            T *= 1.2
        elif no_improve > 20:
            T *= 0.7
        if no_improve > 60:
            T = 0.01
            # Bigger perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            no_improve = 0

    return best_a, best_obj


def fine_polish(a, n, time_limit, rng):
    """Fine-grained polishing with very small steps."""
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_conv(a).max()

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Very small primal-dual step
        a_new, obj = primal_dual_iterate(a, n, 0.005, 'softmax', max_inner=300,
                                         time_limit=min(0.5, remaining * 0.4),
                                         use_as=False)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Peak unbalancing
            a_new = peak_unbalance(a, n, n_steps=5, step_frac=0.02)
            obj = fast_conv(a_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
            else:
                # Tiny perturbation
                a_new = project_simplex(a + rng.standard_normal(n) * 0.001)
                obj = fast_conv(a_new).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a_new.copy()
                    a = a_new

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.08 * n)) ** 2)
    inits.append(g2 / g2.sum())
    # Cosine
    c1 = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    c1 = np.maximum(c1, 0.01)
    inits.append(c1 / c1.sum())
    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - 0.5 * (n - 1)) / (n - 1)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())
        # Perturbed parent
        px_pert = np.maximum(px + rng.standard_normal(n) * 0.01, 0)
        if px_pert.sum() > 1e-10:
            inits.append(px_pert / px_pert.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for auto-correlation minimization.
    Uses primal-dual QP, peak unbalancing, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Quick n-exploration (150s) ===
    print("\n=== Phase 1: Quick n-exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(256, n_parent - 128),
        n_parent + 128,
        max(256, n_parent // 2),
        min(2048, n_parent * 2),
        512, 768, 1024,
    ]))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1100
        if remaining < 20:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(remaining, 35)
        print(f"  Exploring n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = multi_strategy_search(start, n_try, time_limit=t_alloc, rng=rng)
        try_seq(a_opt, f"explore_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep search on best n (900s) ===
    print("\n=== Phase 2: Deep multi-strategy search ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        print(f"  Deep search on n={n_best}")

        # Run multiple rounds with different seeds/time allocations
        for round_i in range(4):
            remaining = deadline - time.monotonic() - 400
            if remaining < 20:
                break
            t_alloc = min(remaining, 200)
            print(f"  Round {round_i}, t={t_alloc:.0f}s")
            a_opt, _ = multi_strategy_search(best_seq, n_best, time_limit=t_alloc, rng=rng)
            try_seq(a_opt, f"deep_r{round_i}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Polish (300s) ===
    print("\n=== Phase 3: Fine polishing ===")
    t3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        t_alloc = min(deadline - time.monotonic() - 100, 250)
        print(f"  Polish t={t_alloc:.0f}s")
        a_opt, _ = fine_polish(best_seq, n_best, time_limit=t_alloc, rng=rng)
        try_seq(a_opt, "polish")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Alternative starting points (100s) ===
    print("\n=== Phase 4: Alternative starts ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        inits = generate_inits(n_best, best_seq, rng)
        for ii, init in enumerate(inits[:4]):
            remaining = deadline - time.monotonic() - 50
            if remaining < 10:
                break
            t_alloc = min(remaining, 25)
            a_opt, _ = multi_strategy_search(init, n_best, time_limit=t_alloc, rng=rng)
            try_seq(a_opt, f"alt_start_{ii}")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final fine-tuning (50s) ===
    print("\n=== Phase 5: Final tuning ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            t_alloc = min(remaining, 40)
            a_opt, _ = fine_polish(best_seq, n_best, time_limit=t_alloc, rng=rng)
            try_seq(a_opt, "final")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
