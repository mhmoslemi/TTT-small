Let me also think about the LP more carefully. The LP is:

This is a linear program with (2n-1) constraints and n variables. For n=1024, that's 2047 constraints and 1024 variables. HiGHS should handle this in a fraction of a second.

The key parameter is M. If M is too large, the LP is easy (the constraint is not binding) and g might be close to f. If M is too small, the LP might be infeasible or give a very different g.

Let me also add a **dual approach**: instead of the LP, we can use the dual variables to get a "certificate" of optimality. The dual of the LP is:
minimize M * sum(λ_k)
subject to sum_k λ_k * f[k-i] >= 1 for all i

If at the optimal solution, the complementary slackness conditions hold (λ_k > 0 only where (f*f)[k] = M), then we're at a local optimum.

This gives us a way to check convergence: if the dual solution has λ_k > 0 only at the current peaks, and the primal solution is g = f, then we've converged.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- SCA (Successive Convex Approximation) with LP oracle
- Structured initializations (parabolic, edge-heavy, cosine)
- Adaptive M with binary search
- Mirror descent with natural gradient
- Multi-n exploration
- Diversified restarts

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_sparsity(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_lp(f, M, n, row_idx, col_idx):
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def sca_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    row_idx, col_idx = build_sparsity(n)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        g_raw = solve_lp(f, M, n, row_idx, col_idx)
        
        if g_raw is None:
            M *= 1.02
            no_improve += 1
            if no_improve > 30:
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue
        
        g = g_raw / g_raw.sum()
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = project_simplex(f + rng.standard_normal(n) * 0.005)
                M = fast_convolve(f).max() * 1.01
                no_improve = 0
            continue
        
        # Line search along direction from f to g
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        
        for alpha in np.linspace(0.05, 1.0, 20):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()
        
        if best_val < best_obj - 1e-12:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.998
        
        if no_improve > 40:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.003 + 0.005 * min(1.0, (no_improve - 40) / 50.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.01
            no_improve = 0
    
    return best_f, best_obj

def mirror_descent(a0, time_limit, eta_init=0.05, top_k=10):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        pw = pv / pv.sum()
        
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = pw[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            indices = np.arange(js, je + 1)
            grad[indices] += w * 2.0 * a[k - indices]
        
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        best_candidate = a
        best_val = max_val
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand
        
        a = best_candidate
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 15:
            eta *= 0.9
        if eta < 1e-5:
            eta = eta_init * 0.3
            no_improve = 0
        
        if no_improve > 100:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            eta = eta_init
            no_improve = 0
    
    return best_a, best_obj

def structured_inits(n, parent, rng):
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Parent (resampled)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Parabolic (zero at edges)
    par = (x + 1) * (n - x)
    inits.append(par / par.sum())
    
    # Inverse parabolic (peak at edges)
    inv_par = 1.0 / ((x + 1) * (n - x) + 1e-10)
    inits.append(inv_par / inv_par.sum())
    
    # Cosine (zero at edges)
    cos = np.cos(np.pi * x / (n - 1)) ** 2 + 1e-10
    inits.append(cos / cos.sum())
    
    # Sine (zero at edges, peak at center)
    sine = np.sin(np.pi * x / (n - 1)) ** 2 + 1e-10
    inits.append(sine / sine.sum())
    
    # Two-peak
    c1, c2 = 0.2 * (n - 1), 0.8 * (n - 1)
    wp = 0.1 * n
    g2 = np.exp(-0.5 * ((x - c1) / wp) ** 2) + np.exp(-0.5 * ((x - c2) / wp) ** 2)
    inits.append(g2 / g2.sum())
    
    # Edge concentration
    edge = 1.0 / (np.minimum(x + 1, n - x) ** 0.5 + 1e-10)
    inits.append(edge / edge.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5)
    inits.append(rng.dirichlet(np.ones(n) * 2.0))
    
    return inits
