from scipy.optimize import linprog
import time

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def logsumexp_grad(a, K=50, T=0.01):
    """
    Compute gradient of log-sum-exp of top K autocorrelation values.
    This approximates the gradient of max(a*a).
    """
    n = len(a)
    conv = fast_convolve(a)
    # Zero-pad to make it symmetric (conv is already 2n-1 long)
    # Top K peaks
    if K >= len(conv):
        indices = np.arange(len(conv))
    else:
        indices = np.argpartition(conv, -K)[-K:]
    
    # Softmax weights
    vals = conv[indices]
    scaled = (vals - vals.max()) / T
    p = np.exp(scaled)
    p = p / p.sum()
    
    # Gradient: sum_k p_k * d(a*a)_k / da
    # d(a*a)_k / da_j = a_{k-j} (if 0 <= k-j < n)
    grad = np.zeros(n)
    for idx, k in enumerate(indices):
        if p[idx] < 1e-15:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        grad[idx_arr] += p[idx] * a[k - idx_arr]
    
    return 2.0 * grad, conv

    Gradient descent on the log-sum-exp smoothed objective.
    Uses adaptive temperature and top-K selection.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    T = T_init
    K = K_init
    lr = lr_init
    
    m = np.zeros(n)  # momentum
    beta = 0.9
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        grad, conv = logsumexp_grad(a, K, T)
        obj = conv.max()
        
        # Adam-style update
        m = beta * m + (1 - beta) * grad
        
        # Backtracking line search
        best_candidate = a
        best_val = obj
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a - scale * lr * m)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val:

                best_candidate = candidate

            best_a = a.copy()
            no_improve = 0

            T = T_init
            K = K_init
            lr = lr_init

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid KKT active-set + 
log-sum-exp gradient + Frank-Wolfe approach with multi-n exploration.
Key innovations:

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, K=50, T=0.01):
    n = len(a)
    conv = fast_convolve(a)
    if K >= len(conv):
        indices = np.arange(len(conv))
    else:
        indices = np.argpartition(conv, -K)[-K:]
    vals = conv[indices]
    scaled = (vals - vals.max()) / T
    p = np.exp(scaled)
    p = p / p.sum()
    grad = np.zeros(n)
    for idx, k in enumerate(indices):
        if p[idx] < 1e-15:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        grad[idx_arr] += p[idx] * a[k - idx_arr]
    return 2.0 * grad, conv

                    break
        if not improved:
            alpha_decay *= 0.5
            eps *= 0.85
            if eps < 0.0005:
                eps = eps_init
                rng = np.random.default_rng(it * 7919 + 13)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(a)
                M = conv.max()
        if M * (1.0 - eps) - best_obj < 1e-13:
            eps = max(0.0001, eps * 0.7)
    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, m_tolerance=1.0):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, float('inf')
    f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        c_vec = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'max_time': min(20, remaining - 2)})
            if res.success:
                g = np.maximum(res.x, 0.0)
            else:
                g = None
        except Exception:
            g = None
        if g is None:
            m_exp *= 1.2
            if m_exp > 4.0:
                m_exp = 1.0
            no_improve += 1
            continue
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)
        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    return best_f, best_obj


def pgd_quick(a0, time_limit, top_k=20, lr_init=0.03, mom_init=0.92):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        conv = fast_convolve(a)
        obj = conv.max()
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]
        best_candidate = a
        best_val = obj
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.125]:
            v_new = mom * v - trial_lr * sg
            candidate = project_simplex(a + v_new)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                v = v_new
                break
        a = best_candidate
        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.92
        if no_improve > 30:
            lr *= 0.94
            mom = min(0.97, mom + 0.002)
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 150) / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            no_improve = 0
    return best_a, best_obj


def smooth_lowpass_perturb(a, rng, alpha=0.3):
    n = len(a)
    noise = rng.standard_normal(n)
    freqs = np.fft.rfftfreq(n)
    N = len(freqs)
    mask = np.exp(-freqs ** 2 / (2 * alpha ** 2 / n))
    noise_fft = np.fft.rfft(noise)
    noise_fft *= mask
    noise = np.fft.irfft(noise_fft, n)
    noise /= (np.std(noise) + 1e-15)
    return project_simplex(a + noise * 0.01)


def interp_sequence(parent, n_target):
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def try_and_update(a, best_state, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def propose_candidate(seed=42, budget_s=1800):

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps):
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)
    print(f"Starting best: {best_score:.8f}, n={len(best_seq)}")

    print("\n=== Phase 1: Init & Symmetry ===")
