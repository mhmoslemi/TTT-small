def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_oracle(f, M, max_time=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c_vec = -np.ones(n)
    # Build constraint matrix: A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # For constraint k: (f*g)[k] = sum_j f[k-j]*g[j] <= M
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in constraints k = j, j+1, ..., j+n-1
        # (f*g)[k] includes f[k-j]*g[j] when 0 <= k-j < n
        # So for fixed j, k-j ranges from 0 to n-1, so k ranges from j to j+n-1
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max_time})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_random(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def active_set_refine(a0, time_limit, tol=1e-10, max_iter=500):
    """Active-set KKT refinement."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        support = np.where(a > 1e-8)[0]
        if len(support) == 0:
            break
        
        # Active peaks: within 0.1% of max
        threshold = M * (1.0 - 0.001)
        active_peaks = np.where(conv >= threshold)[0]
        
        if len(active_peaks) <= 1:
            break
        
        m = len(support)
        p = len(active_peaks)
        
        # KKT: For j in support, sum_k mu_k * a[k-j] = nu
        # Build system: for each j in support, for each active peak k:
        #   a[k-j] (if 0 <= k-j < n)
        # System: A @ mu = nu * ones, where A[i, k_idx] = a[active_peaks[k_idx] - support[i]]
        # Plus: sum(mu) = 1 (normalization)
        
        A_sys = np.zeros((m, p))
        for i, j in enumerate(support):
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = a[idx]
        
        # We want A_sys @ mu = nu * ones(m), sum(mu) = 1, mu >= 0
        # Augmented system: [A_sys, -ones] @ [mu; nu] = 0, ones(mu) = 1
        # Use SVD to find the direction
        try:
            # Solve least squares: minimize ||A_sys @ mu - nu*1||^2 + (sum(mu) - 1)^2
            # This is: [A_sys, -1; 1^T, 0] @ [mu; nu] = [0; 1]
            A_aug = np.zeros((m + 1, p + 1))
            A_aug[:m, :p] = A_sys
            A_aug[:m, p] = -1.0
            A_aug[m, :p] = 1.0
            b_aug = np.zeros(m + 1)
            b_aug[m] = 1.0
            
            sol, residuals, rank, sv = np.linalg.lstsq(A_aug, b_aug, rcond=None)
            mu_vals = np.maximum(sol[:p], 0)
            mu_sum = mu_vals.sum()
            if mu_sum < 1e-15:
                break
            mu_vals /= mu_sum
        except np.linalg.LinAlgError:
            break
        
        # Gradient step using mu-weighted peaks
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu_vals[k_idx] < 1e-15:
                continue
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= mu_vals[k_idx] * a[k - idx_arr]
        
        # Take a step
        best_candidate = a
        best_val = M
        for step in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            candidate = project_simplex(a + step * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate
        
        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
        
        if abs(M - best_val) < 1e-14:
            break
    
    return best_a, best_obj

def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)
