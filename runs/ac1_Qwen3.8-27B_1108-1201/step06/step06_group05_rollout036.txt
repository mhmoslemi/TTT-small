def project_simplex(v):
    n = len(v)

            best_x = x
    
    return best_x, best_max

"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines: adaptive softmax gradient descent, exact coordinate descent,
LP-based Frank-Wolfe global moves, structured diversification,
and multi-length exploration to escape local minima.
"""
import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based full auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto sum(v)=1, v>=0."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(a, p):
    """Gradient of sum_k p[k]*(a*a)[k] w.r.t. a. Returns 2*correlate(p,a,'valid')."""
    return 2.0 * np.correlate(p, a, mode='valid')


def softmax_gradient(a, T):
    """Compute smoothed gradient using softmax-weighted peaks."""
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    grad = compute_grad(a, p)
    obj = conv.max()
    return grad, obj, conv, p


def solve_lp_oracle(f, M, max_time=20):
    """LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': max_time})
        if res.success:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def line_search_grid(a, d, conv_a, n_steps=15):
    """Find best step in [0,1] along direction d (assumed sum(d)=0)."""
    n = len(a)
    conv_d = fast_convolve(d)
    conv_ad = np.convolve(a, d)
    
    # Along line: conv(t) = conv_a + 2t*conv_ad + t^2*conv_d
    # We need a+t*d >= 0
    t_max = 1.0
    neg_mask = d < -1e-15
    if np.any(neg_mask):
        t_candidates = -a[neg_mask] / d[neg_mask]
        t_max = min(t_max, t_candidates.min())
    
    t_max = max(0.0, t_max)
    if t_max < 1e-10:
        return 0.0, conv_a.max()
    
    ts = np.linspace(0, t_max, n_steps)
    best_t = 0.0
    best_val = conv_a.max()
    
    for t in ts:
        if t <= 0:
            continue
        conv_t = conv_a + 2 * t * conv_ad + t * t * conv_d
        val = conv_t.max()
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine around best_t
    if best_t > 0:
        lo = max(0.0, best_t / n_steps * 0.5)
        hi = min(t_max, best_t + best_t / n_steps * 0.5)
        for t in np.linspace(lo, hi, 20):
            conv_t = conv_a + 2 * t * conv_ad + t * t * conv_d
            val = conv_t.max()
            if val < best_val:
                best_val = val
                best_t = t
    
    return best_t, best_val


def coordinate_descent(a, time_limit, n_trials=12, verbose_every=50):
    """Coordinate descent: optimize each coordinate in turn."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                trials = np.concatenate([[0.0], np.linspace(0.001, 0.05, n_trials - 1)])
            else:
                spread = max(old_val * 0.8, 0.005)
                lo = max(0.0, old_val - spread)
                hi = min(1.0, old_val + spread)
                trials = np.linspace(lo, hi, n_trials)
            
            best_x = old_val
            best_ratio = best_obj  # current peak (since sum=1)
            
            k_start = i
            k_end = i + n - 1
            conv_len = 2 * n - 1
            
            for x in trials:
                x = float(max(0.0, min(1.0, x)))
                sum_temp = 1.0 + x - old_val
                if sum_temp < 1e-10:
                    continue
                if abs(x - old_val) < 1e-14:
                    new_max = conv.max()
                else:
                    delta_vals = 2.0 * (x - old_val) * a
                    if 0 <= i < n:
                        delta_vals[i] = x * x - old_val * old_val
                    # new conv in range [k_start, k_end]
                    new_seg = conv[k_start:k_end + 1] + delta_vals
                    if k_start > 0:
                        outside = np.max(conv[:k_start])
                    else:
                        outside = 0.0
                    if k_end + 1 < conv_len:
                        outside2 = np.max(conv[k_end + 1:])
                        outside = max(outside, outside2)
                    new_max = max(outside, np.max(new_seg))
                
                ratio_val = new_max / (sum_temp * sum_temp)
                if ratio_val < best_ratio - 1e-15:
                    best_ratio = ratio_val
                    best_x = x
            
            if abs(best_x - old_val) > 1e-12:
                a[i] = best_x
                s = a.sum()
                if s < 1e-10:
                    a = np.ones(n) / n
                    conv = fast_convolve(a)
                else:
                    a = a / s
                    conv = fast_convolve(a)
                improved = True
        
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            break
        
        if verbose_every and passes % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD pass={passes}, obj={best_obj:.10f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_obj


def softmax_pgd(a0, time_limit, T_init=0.002, lr_init=0.05, mom=0.9):
    """Softmax-smoothed projected gradient with momentum."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        grad, obj, conv, p = softmax_gradient(a, T)
        
        # Project gradient to tangent space
        grad_tan = grad - grad.mean()
        
        # Momentum
        v = mom * v - lr * grad_tan
        
        # Line search with a few trial steps
        best_candidate = a
        best_val = obj
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            cand = project_simplex(a + scale * v)
            val = fast_convolve(cand).max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = cand
        
        a = best_candidate
        
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.85
        
        # Adapt temperature and lr
        if no_improve > 15:
            T *= 1.15
            lr *= 0.97
        if no_improve % 40 == 0 and no_improve > 0:
            # Reset momentum
            v *= 0.5
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 104729 + 31)
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + noise)
            v = np.zeros(n)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj


def fw_lp_step(a, time_limit, m_factor=1.0):
    """Frank-Wolfe with LP oracle."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_obj = M
    
    t0 = time.monotonic()
    no_improve = 0
    m_exp = m_factor
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        g = solve_lp_oracle(a, M * m_exp, max_time=15)
        if g is None:
            m_exp *= 1.5
            if m_exp > 10:
                m_exp = 1.0
            no_improve += 1
            continue
        
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            m_exp = min(m_exp * 1.05, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            M = fast_convolve(a).max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 40:
            break
    
    return best_a, best_obj


def structured_diversify(a, rng, mode=0):
    """Add structured perturbation to escape local minima."""
    n = len(a)
    if mode == 0:
        # Random noise
        pert = rng.standard_normal(n) * 0.01
    elif mode == 1:
        # Sinusoidal perturbation
        k = rng.integers(1, max(2, n // 4))
        phase = rng.random() * 2 * np.pi
        pert = 0.005 * np.sin(2 * np.pi * k * np.arange(n) / n + phase)
    elif mode == 2:
        # Block perturbation
        block_size = max(2, n // 8)
        start = rng.integers(0, n - block_size)
        pert = np.zeros(n)
        pert[start:start + block_size] = 0.01 * rng.standard_normal(block_size)
    else:
        # Multi-scale perturbation
        pert = np.zeros(n)
        for scale in [n // 4, n // 8, n // 16]:
            if scale > 1:
                coarse = rng.standard_normal(scale)
                pert += np.interp(np.linspace(0, n - 1, n),
                                  np.linspace(0, n - 1, scale), coarse) * (1.0 / scale)
        pert *= 0.005
    
    return project_simplex(a + pert)


def interp_to_length(a, n_target):
    """Interpolate sequence to target length."""
    if len(a) == n_target:
        return a.copy()
    x_orig = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, a)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def try_update(a, best_a, best_score):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_a, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"    NEW BEST: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization: multi-phase hybrid search.
    Phase 1: Quick gradient convergence (150s)
    Phase 2: Coordinate descent refinement (350s)
    Phase 3: LP-based global moves (250s)
    Phase 4: Multi-n exploration with diversification (400s)
    Phase 5: Deep coordinate descent + gradient (350s)
    Phase 6: Final polishing (200s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        parent = parent / parent.sum()
    except Exception:
        parent = np.ones(768) / 768
    
    best_a = parent.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Start: n={len(parent)}, score={best_score:.8f}")
    
    # === Phase 1: Quick gradient convergence ===
    print("\n=== Phase 1: Gradient Convergence ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic() - 1350
    t_alloc = min(remaining, 150)
    if t_alloc > 5:
        a_opt, obj = softmax_pgd(best_a, time_limit=t_alloc, T_init=0.003, lr_init=0.04)
        best_a, best_score = try_update(a_opt, best_a, best_score)
    
    remaining = deadline - time.monotonic() - 1300
    t_alloc = min(remaining, 50)
    if t_alloc > 5:
        a_opt, obj = coordinate_descent(best_a, time_limit=t_alloc, n_trials=10)
        best_a, best_score = try_update(a_opt, best_a, best_score)
    
    print(f"  Phase 1 done: score={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    
    # === Phase 2: Coordinate descent refinement ===
    print("\n=== Phase 2: Coordinate Descent ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic() - 1100
    t_alloc = min(remaining, 350)
    if t_alloc > 10:
        a_opt, obj = coordinate_descent(best_a, time_limit=t_alloc * 0.6, n_trials=15)
        best_a, best_score = try_update(a_opt, best_a, best_score)
        
        remaining = deadline - time.monotonic() - 1050
        t_alloc2 = min(remaining, t_alloc * 0.4)
        if t_alloc2 > 5:
            a_opt2, obj2 = coordinate_descent(best_a, time_limit=t_alloc2, n_trials=20)
            best_a, best_score = try_update(a_opt2, best_a, best_score)
    
    print(f"  Phase 2 done: score={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    
    # === Phase 3: LP-based global moves ===
    print("\n=== Phase 3: LP Global Moves ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic() - 900
    t_alloc = min(remaining, 250)
    if t_alloc > 10:
        a_opt, obj = fw_lp_step(best_a, time_limit=t_alloc * 0.7)
        best_a, best_score = try_update(a_opt, best_a, best_score)
        
        # Coordinate descent after FW
        remaining = deadline - time.monotonic() - 850
        t_alloc2 = min(remaining, t_alloc * 0.3)
        if t_alloc2 > 5:
            a_opt2, obj2 = coordinate_descent(best_a, time_limit=t_alloc2, n_trials=12)
            best_a, best_score = try_update(a_opt2, best_a, best_score)
    
    print(f"  Phase 3 done: score={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    
    # === Phase 4: Multi-n exploration with diversification ===
    print("\n=== Phase 4: Multi-n + Diversification ===")
    t0 = time.monotonic()
    n_best = len(best_a)
    
    # Try several lengths
    n_candidates = set()
    for n_try in [n_best // 2, n_best, n_best * 2, 384, 512, 640, 768, 896, 1024, 1280]:
        if 128 <= n_try <= 2048:
            n_candidates.add(n_try)
    
    n_list = sorted(n_candidates)
    for n_try in n_list:
        remaining = deadline - time.monotonic() - 450
        if remaining < 30:
            break
        
        if n_try == n_best:
            start = best_a.copy()
        else:
            start = interp_to_length(best_a, n_try)

        t_alloc = min(remaining, 80)

        best_a, best_score = try_update(a_opt, best_a, best_score)
        
        remaining = deadline - time.monotonic() - 400
        if remaining > 15:
            a_opt2, _ = softmax_pgd(a_opt, time_limit=min(remaining, t_alloc * 0.3),
                                     T_init=0.002, lr_init=0.03)
            best_a, best_score = try_update(a_opt2, best_a, best_score)
        
        remaining = deadline - time.monotonic() - 380
        if remaining > 10:
            a_opt3, _ = coordinate_descent(a_opt2, time_limit=min(remaining, t_alloc * 0.2), n_trials=15)
            best_a, best_score = try_update(a_opt3, best_a, best_score)
    
    # Diversification restarts
    n_best = len(best_a)
    for mode in range(4):
        remaining = deadline - time.monotonic() - 300
        if remaining < 20:
            break
        a_div = structured_diversify(best_a, rng, mode=mode)
        t_alloc = min(remaining, 60)
        a_opt, _ = coordinate_descent(a_div, time_limit=t_alloc * 0.6, n_trials=12)
        best_a, best_score = try_update(a_opt, best_a, best_score)
        
        remaining = deadline - time.monotonic() - 280
        if remaining > 10:
            a_opt2, _ = softmax_pgd(a_opt, time_limit=min(remaining, t_alloc * 0.4),
                                     T_init=0.001, lr_init=0.02)
            best_a, best_score = try_update(a_opt2, best_a, best_score)
    
    print(f"  Phase 4 done: score={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    
    # === Phase 5: Deep refinement ===
    print("\n=== Phase 5: Deep Refinement ===")
    t0 = time.monotonic()
    n_best = len(best_a)
    
    # Extended coordinate descent
    remaining = deadline - time.monotonic() - 250
    t_alloc = min(remaining, 200)
    if t_alloc > 10:
        a_opt, _ = coordinate_descent(best_a, time_limit=t_alloc, n_trials=20)
        best_a, best_score = try_update(a_opt, best_a, best_score)
    
    # Gradient refinement with small T
    remaining = deadline - time.monotonic() - 200
    t_alloc = min(remaining, 100)
    if t_alloc > 5:
        a_opt2, _ = softmax_pgd(best_a, time_limit=t_alloc, T_init=0.0005, lr_init=0.01, mom=0.93)
        best_a, best_score = try_update(a_opt2, best_a, best_score)
    
    # Another CD pass
    remaining = deadline - time.monotonic() - 150
    t_alloc = min(remaining, 80)
    if t_alloc > 5:
        a_opt3, _ = coordinate_descent(best_a, time_limit=t_alloc, n_trials=25)
        best_a, best_score = try_update(a_opt3, best_a, best_score)
    
    print(f"  Phase 5 done: score={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    
    # === Phase 6: Final polishing ===
    print("\n=== Phase 6: Final Polish ===")
    t0 = time.monotonic()
    n_best = len(best_a)
    
    # Tiny perturbations + quick CD
    for _ in range(8):
        remaining = deadline - time.monotonic() - 20
        if remaining < 8:
            break
        noise = rng.standard_normal(n_best) * 0.002
        a_pert = project_simplex(best_a + noise)
        t_alloc = min(remaining, 20)
        a_opt, _ = coordinate_descent(a_pert, time_limit=t_alloc, n_trials=8)
        best_a, best_score = try_update(a_opt, best_a, best_score)
    
    # Final CD with fine grid
    remaining = deadline - time.monotonic() - 10
    t_alloc = min(remaining, 30)
    if t_alloc > 3:
        a_final, _ = coordinate_descent(best_a, time_limit=t_alloc, n_trials=30)
        best_a, best_score = try_update(a_final, best_a, best_score)
    
    # Symmetry check
    sym = (best_a + best_a[::-1]) / 2.0
    sym = sym / sym.sum()
    best_a, best_score = try_update(sym, best_a, best_score)
    
    print(f"  Phase 6 done: score={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    
    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_a)}")
    return [float(x) for x in best_a]
