"""
Hybrid evolutionary + adaptive subgradient optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based fast convolution, diverse initialization, evolutionary search
for global exploration, and QP-based peak shaving for local refinement.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')

def compute_subgradient(a, top_k=10):
    """Compute weighted subgradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    # Pick top peaks

        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        # Weight by rank
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 2
        weights /= weights.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        k = int(peaks[i])
        w = weights[i]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        sg[idx_arr] += w * 2.0 * a[k - idx_arr]
    return sg, conv

def adaptive_subgradient(a0, time_limit, T_init=0.001, lr_init=0.02, top_k=20):

    best_obj = fast_convolve(a).max()

    it = 0

        a_new = project_simplex(a + v)

        elif obj_new < current_obj + T * 0.1:

            if np.random.random() < 0.3:
                no_improve += 1
            else:
                no_improve += 1
        else:

            a_shaved = qp_peak_shave(a, best_obj)
            obj_shaved = fast_convolve(a_shaved).max()

            no_improve = 0

    return best_a, best_obj

def multi_scale_optimize(a0, time_limit):

            a_up = a_up / a_up.sum()
            obj_up = fast_convolve(a_up).max()
            if obj_up < best_obj:

def parametric_family_search(n, time_limit):
    """Search over parametric families for good initializations."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    families = []
    
    # Super-Gaussian: max(0, 1 - (|x-center|/(n/2*sigma))^p)
    for sigma in [0.3, 0.5, 0.7, 1.0, 1.5]:
        for p in [1, 2, 3, 4, 6, 8]:
            h = 1.0 - (np.abs(x - center) / (n / 2.0 * sigma)) ** p
            h = np.maximum(h, 0)
            if h.sum() > 1e-10:
                families.append(h / h.sum())
    
    # Cosine window with variable width
    for width in [0.3, 0.5, 0.7, 0.9, 1.0]:
        h = np.cos(np.pi * (x - center) / (n * width))
        h = np.where(np.abs(x - center) < n * width / 2, h, 0)
        h = np.maximum(h, 0)
        if h.sum() > 1e-10:
            families.append(h / h.sum())
    
    # Flat-top with smooth edges
    for flat_frac in [0.3, 0.5, 0.7]:
        for edge_frac in [0.1, 0.2, 0.3]:
            h = np.zeros(n)
            flat_start = int(center * (1 - flat_frac))
            flat_end = int(center * (1 + flat_frac))
            edge_start = max(0, flat_start - int(n * edge_frac))
            edge_end = min(n - 1, flat_end + int(n * edge_frac))
            h[flat_start:flat_end+1] = 1.0
            if edge_start < flat_start:
                t = np.linspace(1, 0, flat_start - edge_start + 1)
                h[edge_start:flat_start] = 0.5 * (1 + np.cos(np.pi * t))
            if edge_end > flat_end:
                t = np.linspace(1, 0, edge_end - flat_end)
                h[flat_end+1:edge_end+1] = 0.5 * (1 + np.cos(np.pi * t))
            if h.sum() > 1e-10:
                families.append(h / h.sum())
    
    # Two-plateau (bimodal)
    for gap in [0.1, 0.2, 0.3]:
        for height_ratio in [0.3, 0.5, 0.7]:
            h = np.zeros(n)
            peak1 = int(n * 0.25)
            peak2 = int(n * 0.75)
            width1 = int(n * 0.15)
            width2 = int(n * 0.15 * height_ratio)
            h = np.maximum(h + np.exp(-0.5 * ((x - peak1) / width1) ** 2), 0)
            h = np.maximum(h + height_ratio * np.exp(-0.5 * ((x - peak2) / width2) ** 2), 0)
            if h.sum() > 1e-10:
                families.append(h / h.sum())
    
    # Evaluate all
    best_a = None
    best_obj = np.inf
    for f in families:
        obj = fast_convolve(f).max()
        if obj < best_obj:
            best_obj = obj
            best_a = f.copy()
    
    if best_a is not None:
        # Quick refine
        remaining = time_limit - (time.monotonic() - time.monotonic())
        # Just do a quick gradient step
        t0 = time.monotonic()
        if time.limit - (time.monotonic() - t0) > 5:
            pass
    
    return best_a, best_obj

def evolutionary_search(all_inits, time_limit, population_size=8):
    """Simple evolutionary search over the population."""
    rng = np.random.default_rng(42)
    
    # Evaluate initial population
    scored = []
    for a in all_inits:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        obj = fast_convolve(a).max()
        scored.append((obj, a.copy()))
    
    scored.sort(key=lambda x: x[0])
    
    # Keep top population_size
    pop = [a for _, a in scored[:population_size]]
    best_obj = scored[0][0]
    best_a = scored[0][1].copy()
    
    t0 = time.monotonic()
    generation = 0
    
    while time.monotonic() - t0 < time_limit and len(pop) >= 2:
        generation += 1
        if len(pop) < 2:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Create offspring
        offspring = []
        
        # Crossover
        for _ in range(len(pop) // 2):
            i = rng.integers(0, len(pop))
            j = rng.integers(0, len(pop))
            if i == j:
                j = (j + 1) % len(pop)
            alpha = rng.uniform(0.2, 0.8)
            child = alpha * pop[i] + (1 - alpha) * pop[j]
            child = project_simplex(child)
            offspring.append(child)
        
        # Mutation
        for a in pop[:len(pop)//2]:
            noise = rng.standard_normal(len(a)) * 0.01
            mutant = project_simplex(a + noise)
            offspring.append(mutant)
        
        # Evaluate offspring and select
        for child in offspring:
            obj = fast_convolve(child).max()
            if obj < best_obj:
                best_obj = obj
                best_a = child.copy()
            scored.append((obj, child))
        
        scored.sort(key=lambda x: x[0])
        pop = [a for _, a in scored[:population_size]]
        
        # Brief local refinement on best
        local_time = min(10, remaining * 0.3)
        if local_time > 3:
            a_ref, obj_ref = adaptive_subgradient(best_a, local_time, T_init=0.001, lr_init=0.01, top_k=10)
            if obj_ref < best_obj:
                best_obj = obj_ref
                best_a = a_ref
                pop[0] = best_a
        
    return best_a, best_obj
