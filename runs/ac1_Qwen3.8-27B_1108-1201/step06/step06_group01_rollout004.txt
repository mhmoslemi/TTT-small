"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FW-LP with sparse banded constraints and dual-guided perturbations
2. Adaptive softmax-smoothed gradient with Nesterov momentum  
3. Gradient-informed coordinate descent with exact line search
4. Multi-n exploration with structured initializations
5. Peak-aware simulated annealing for final polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    return np.correlate(p, a, mode='valid')


def solve_lp_oracle(f, M, time_limit=15):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_exp, time_limit=min(15, remaining - 1))
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                m_exp = 1.0
                g = np.ones(n) / n
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0

            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def smoothed_gradient_descent(a0, time_limit, T_init=0.001, lr_init=0.02):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    T = T_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    t_nest = 1.0

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Softmax smoothing
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        # Nesterov update
        alpha_nest = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta_nest = (t_nest - 1.0) / alpha_nest
        t_nest += 1.0

        v = beta_nest * v - lr * grad
        a_new = project_simplex(a + v)

        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        for trial in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = project_simplex(a - trial * lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature
        if no_improve > 50:
            T *= 1.1
        if no_improve > 150:
            T = max(T_init, T * 0.5)
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        affected = np.arange(js, je + 1)

        improved = False

        # Try moving mass from affected indices to non-affected indices
        non_affected = np.setdiff1d(np.arange(n), affected)
        if len(non_affected) == 0 or len(affected) < 2:
            break

        best_gain = 0
        best_move = None

        # Sample pairs
        rng = np.random.default_rng(it * 104729 + 13)
        n_sample = min(len(affected), 20)
        n_dst = min(len(non_affected), 20)
        src_sample = rng.choice(affected, size=n_sample, replace=False)
        dst_sample = rng.choice(non_affected, size=n_dst, replace=False)

        for src in src_sample:
            if a[src] < 1e-8:
                continue
            for dst in dst_sample:
                for frac in [0.05, 0.1, 0.2, 0.5]:
                    delta = a[src] * frac
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[src] -= delta
                    a_test[dst] += delta
                    a_test = np.maximum(a_test, 0.0)
                    s = a_test.sum()
                    if s < 1e-15:
                        continue
                    a_test = a_test / s
                    c_new = fast_convolve(a_test)
                    new_max = c_new.max()
                    gain = best_obj - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_move = (src, dst, delta)

        if best_move is not None and best_gain > 1e-14:
            src, dst, delta = best_move
            a[src] -= delta
            a[dst] += delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            c_new = fast_convolve(a)
            best_obj = c_new.max()
            best_a = a.copy()
            improved = True
            stall = 0
        else:
            stall += 1

        if stall > 30:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            scale = min(0.01, 0.002 * np.sqrt(stall / 10.0))
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            stall = 0

        if stall > 200:
            break

    return best_a, best_obj
