def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def fast_convolve(a):

    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, weights):
    """g[j] = 2 * sum_k weights[k] * a[k-j] computed via FFT."""
    n = len(a)
    L_w = len(weights)
    # g[j] = 2 * convolve(weights[::-1], a)[L_w - 1 - j]
    h = weights[::-1]

    conv_size = L_w + n - 1
    fft_size = 1 << (conv_size - 1).bit_length()

    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(H * A, fft_size)[:conv_size]
    # Extract: g[j] = 2 * conv[L_w - 1 - j] for j = 0..n-1
    g = 2.0 * conv[L_w - 1 - np.arange(n)]

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, weights):
    """
    Compute subgradient g[j] = 2 * sum_k weights[k] * a[k-j] via FFT.
    weights has length 2n-1 (the convolution length).
    """
    n = len(a)
    L_w = len(weights)

    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(H * A, fft_size)[:conv_size]
    idx = L_w - 1 - np.arange(n)
    return 2.0 * conv[idx]

def mirror_descent_step(a, g, lr):

                # Short refinement
                t_remain = time_limit - (time.monotonic() - t0)
                if t_remain > 5:
                    a_opt, obj_opt = gradient_optimize(candidate, time_limit=min(t_remain, 30))
                    if obj_opt < best_obj:
                        best_obj = obj_opt
                        best_a = a_opt.copy()
                else:
                    best_obj = val
                    best_a = candidate.copy()

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using cubic interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a, kind='cubic')
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def two_point_swap(a, i, j, delta):
    """Move delta mass from position i to position j."""
    candidate = a.copy()
    candidate[i] -= delta
    candidate[j] += delta
    candidate = np.maximum(candidate, 0)
    s = candidate.sum()
    if s < 1e-15:
        return None
    return candidate / s


def local_search(a0, time_limit):
    """
    Randomized local search: try random 2-point swaps and accept improvements.
    Uses simulated annealing-like acceptance for occasional non-improving moves.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    c = fast_convolve(a)

    t0 = time.monotonic()
    it = 0
    delta_base = 0.01
    temp = 0.001
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.005:
            break

        rng = np.random.default_rng((it * 69621 + 1) % (2**31))
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        delta = a[i] * delta_base * (0.5 + rng.random())
        direction = 1 if rng.random() > 0.5 else -1

        candidate = two_point_swap(a, i, j, direction * delta)
        if candidate is None:
            candidate = two_point_swap(a, j, i, -direction * delta)
        if candidate is None:
            continue

        val = fast_convolve(candidate).max()

        if val < best_obj - 1e-14:
            a = candidate
            best_obj = val
            best_a = a.copy()
            no_improve = 0
            temp = max(temp * 0.99, 1e-6)
        elif val < best_obj + temp * delta_base:
            # Simulated annealing: accept with probability
            if rng.random() < np.exp(-(val - best_obj) / max(temp, 1e-10)):
                a = candidate
                no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adapt
        if no_improve > 500:
            delta_base = max(delta_base * 0.9, 1e-5)
            temp *= 0.9
            if no_improve > 3000:
                # Restart
                a = project_simplex(best_a + np.random.default_rng(it).standard_normal(n) * 0.002)
                delta_base = 0.01
                temp = 0.001
                no_improve = 0

    return best_a, best_obj


def fw_lp_step(f, M, n):
    """
    Single Frank-Wolfe step using LP oracle.
    Solves: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0.
    Returns the normalized g.
    """
    from scipy.optimize import linprog

    # Build convolution matrix: (f*g)[k] = sum_j A[k,j] * g[j]
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a (2n-1) x n matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def fw_optimize(a0, time_limit, verbose=False):
    """
    Frank-Wolfe with LP oracle and adaptive M.
    Used for deeper optimization after gradient phase.
    """
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = M

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        g = fw_lp_step(f, M * 1.001, n)

        if g is None or g.sum() < 1e-15:
            g = fw_lp_step(f, M * 1.005, n)

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 15:
                rng = np.random.default_rng(it * 7919 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        if np.max(np.abs(g - f)) < 1e-12:
            stagnant += 1
            if stagnant > 20:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Line search (coarse then fine)
        best_alpha = 0.0
        best_val = M + 1e-14
        for alpha in np.linspace(0.02, 1.0, 20):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search
        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for alpha in np.linspace(lo, hi, 10):
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if verbose and it % 10 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stag={stagnant}")

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Multi-phase optimization:
    1. Quick multi-length exploration with fast gradient (200s)
    2. Deep gradient optimization on best (400s)
    3. Frank-Wolfe refinement (300s)
    4. Peak-directed coordinate descent (200s)
    5. Local search + spectral perturbation (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Quick multi-length exploration (200s) ===
    print("\n=== Phase 1: Quick Exploration (multi-length, fast gradient) ===")
    t_phase = time.monotonic()

    lengths = sorted(set([n_parent, max(256, n_parent - 100), n_parent + 100,
                          512, 640, 896, 1024, 1200]))
    lengths = [l for l in lengths if 128 <= l <= 2048]

    for n_try in lengths:
        remaining = deadline - time.monotonic() - 700
        if remaining < 20:
            break

        # Generate starts
        starts = []
        if parent is not None:
            starts.append(resample_sequence(parent, n_try))
        starts.append(np.ones(n_try) / n_try)

        # Cosine taper
        x = np.arange(n_try, dtype=np.float64)
        w = 0.5 * (1 + np.cos(np.pi * x / (n_try - 1)))
        w = np.maximum(w, 0.01)
        starts.append(w / w.sum())

        # Gaussian center
        g = np.exp(-0.5 * ((x - (n_try - 1) / 2) / (0.25 * n_try)) ** 2)
        starts.append(g / g.sum())

        # Edge concentration
        edge = 1.0 / (np.minimum(x, n_try - 1 - x) + 1.0)
        starts.append(edge / edge.sum())

        # Random
        starts.append(rng.dirichlet(np.ones(n_try) * 0.5))

        for si, start in enumerate(starts[:4]):
            remaining = deadline - time.monotonic() - 650
            if remaining < 10:
                break
            t_alloc = min(remaining, 25)
            a_opt, _ = gradient_optimize(start, time_limit=t_alloc)
            try_seq(a_opt, f"grad_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep gradient optimization on best (400s) ===
    print("\n=== Phase 2: Deep Gradient Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Multi-temperature gradient optimization
        remaining = deadline - time.monotonic() - 500

            a_opt, _ = gradient_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_gradient")

        # Mirror descent variant
        remaining = deadline - time.monotonic() - 400
        if remaining > 20:
            t_alloc = min(remaining, 80)
            a_opt, _ = gradient_optimize(best_seq, time_limit=t_alloc, use_mirror=True)
