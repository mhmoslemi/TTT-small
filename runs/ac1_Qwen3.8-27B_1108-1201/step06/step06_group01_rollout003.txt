import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M, warm_start=None):

    bounds = [(0, None)] * n
    options = {'presolve': True, 'time_limit': 20}
    try:

        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
        elif res.status == 3:  # infeasible
            return None
    except Exception:

    return None

                    hi = m2
                else:
                    lo = m1
                mid_best = (lo + hi) / 2.0
                fm = (1.0 - mid_best) * f + mid_best * g
                fm = np.maximum(fm, 0.0)
                sm = fm.sum()
                vm = fast_convolve(fm / sm).max() if sm > 1e-15 else best_val
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid_best

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            break

    return best_f, best_obj


def smoothed_nesterov(a0, time_limit, T_start=0.005, T_end=0.0002, lr_init=0.02, verbose=True):
    """Nesterov-accelerated smoothed gradient descent with temperature annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Log-linear temperature schedule
        T = T_start * (T_end / T_start) ** frac
        lr = lr_init * (1.0 - 0.3 * frac)

        # Nesterov lookahead
        alpha_t = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha_t
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        # Compute smoothed gradient
        conv = fast_convolve(y)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(len(conv)) / len(conv)
        else:
            p = p / p_sum

        # Gradient: 2 * sum_k p[k] * shifted y
        grad = np.zeros(n)
        for k in range(len(conv)):
            if p[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += p[k] * 2.0 * y[k - idx]

        # Momentum update
        v = beta * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a

        a = a_new

        # Adaptive restart
        if no_improve > 80:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 * min(1.0, np.sqrt(no_improve / 80.0))
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0

    return best_a, best_obj


def dual_informed_refine(a0, time_limit, verbose=True):
    """
    Uses LP dual information to construct better search directions.
    Alternates between LP oracle and dual-guided gradient steps.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    prev_dual = None

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Solve LP to get direction and dual info
        g = solve_lp_oracle(a, M * 1.01, warm_start=best_a)
        if g is None:
            g = solve_lp_oracle(a, M * 1.05)
        if g is None:
            # Fallback: use gradient
            peaks = np.argmax(conv)
            js = max(0, peaks - n + 1)
            je = min(n - 1, peaks)
            g = a.copy()
            g[js:je + 1] *= 0.9
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
            else:
                g = np.ones(n) / n
        else:
            gs = g.sum()
            if gs < 1e-15:
                g = np.ones(n) / n
            else:
                g = g / gs

        # Blend direction with dual-informed perturbation
        if prev_dual is not None:
            # Use dual variables to weight the gradient
            dual_weights = np.maximum(prev_dual, 0)
            dw_sum = dual_weights.sum()
            if dw_sum > 1e-15:
                dual_weights = dual_weights / dw_sum
                # Compute dual-informed gradient
                dgrad = np.zeros(n)
                for k in range(len(dual_weights)):
                    if dual_weights[k] < 1e-10:
                        continue
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    idx = np.arange(js, je + 1)
                    dgrad[idx] += dual_weights[k] * 2.0 * a[k - idx]
                # Blend: 70% LP direction, 30% dual gradient descent
                a_test = 0.7 * g + 0.3 * project_simplex(a - 0.01 * dgrad)
                g = a_test / a_test.sum()

        # Save dual info for next iteration (approximate from active constraints)
        conv_test = fast_convolve(a * (1 - 0.1) + g * 0.1) if g.sum() > 0 else conv
        active = np.where(conv > M * 0.95)[0]
        prev_dual = np.zeros(len(conv))
        prev_dual[active] = 1.0 / max(len(active), 1)

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
        else:
            # Small random perturbation
            rng = np.random.default_rng((it * 41 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)

    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=True):
    """Fine coordinate descent focusing on peak reduction."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        # Top 3 peaks
        top_k = min(3, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

        improved = False

        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            js = max(0, int(k_peak) - n + 1)
            je = min(n - 1, int(k_peak))
            affected = np.arange(js, je + 1)

            if len(affected) < 2:
                continue

            # Try transferring mass away from contributors
            best_gain = 1e-15
            best_move = None

            # Sample pairs
            if len(affected) > 20:
                rng_s = np.random.default_rng(it * 13 + k_peak)
                sample = rng_s.choice(affected, size=20, replace=False)
            else:
                sample = affected

            for ii in range(len(sample)):
                for jj in range(len(sample)):
                    if ii == jj:
                        continue
                    j_src = sample[ii]
                    j_dst = sample[jj]

                    c_src = a[int(k_peak) - j_src] if 0 <= int(k_peak) - j_src < n else 0
                    c_dst = a[int(k_peak) - j_dst] if 0 <= int(k_peak) - j_dst < n else 0

                    if c_src <= c_dst:
                        continue
                    if a[j_src] < 1e-10:
                        continue

                    for frac in [0.05, 0.15, 0.3]:
                        delta = a[j_src] * frac
                        if delta < 1e-12:
                            continue
                        a_test = a.copy()
                        a_test[j_src] -= delta
                        a_test[j_dst] += delta
                        a_test = np.maximum(a_test, 0.0)
                        s = a_test.sum()
                        if s < 1e-15:
                            continue
                        a_test = a_test / s
                        c_new = fast_convolve(a_test)
                        gain = best_obj - c_new.max()
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (j_src, j_dst, delta)

            if best_move is not None and best_gain > 1e-13:
                j_src, j_dst, delta = best_move
                a[j_src] -= delta
                a[j_dst] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                c_new = fast_convolve(a)
                best_obj = c_new.max()
                best_a = a.copy()
                improved = True
                stall = 0
                break
        else:
            stall += 1

        if not improved:
            stall += 1

        if stall > 15:
            rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))
            scale = 0.001 + 0.001 * min(1.0, stall / 30.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            stall = 0

        if stall > 100:
            break

    return best_a, best_obj


def micro_polish(a0, time_limit, verbose=True):
    """Very fine structured perturbations for final improvement."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        # Try multiple scales
        for scale in [0.0001, 0.0003, 0.0008, 0.002]:
            rng = np.random.default_rng((it * 2654435761 + int(scale * 1e6)) % (2**32))
            for attempt in range(5):
                noise = rng.standard_normal(n) * scale
                # Bias: reduce contributors to peak
                for j in range(js, je + 1):
                    partner = peak_k - j
                    if 0 <= partner < n:
                        noise[j] -= scale * 3.0 * a[partner]

                candidate = project_simplex(a + noise)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search with warm-started FW-LP,
    temperature-annealed Nesterov, dual-informed refinement, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={time.monotonic()-t0_total:.0f}s)")
        return score

    t0_total = time.monotonic()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: Multi-n Exploration (~150s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, n_parent - 128, n_parent + 128,
                               512, 640, 768, 896, 1024, 1280]))
    candidate_ns = [n for n in candidate_ns if 320 <= n <= 1400]

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1400:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        # Also try a structured initialization
        if n_try != n_parent and rng.random() < 0.3:
            # Try triangular
            tri = np.abs(np.arange(n_try) - n_try / 2.0)
            tri = 1.0 - tri / (n_try / 2.0)
            tri = np.maximum(tri, 0)
            if tri.sum() > 1e-10:
                start = tri / tri.sum()

        t_alloc = min(25, deadline - time.monotonic() - 1400)
        if t_alloc < 5:
            break
        print(f"  FW-LP n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        try_seq(a_opt, f"fw_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP on best (~500s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(400, deadline - time.monotonic() - 900)
        if t_alloc > 20:
            print(f"  Deep FW n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts
        for scale in [0.003, 0.008, 0.015]:
            t_alloc = min(50, deadline - time.monotonic() - 800)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pers{scale}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Smoothed Nesterov with annealing (~350s) ===
    print("\n=== Phase 3: Smoothed Nesterov ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        configs = [
            {'T_start': 0.003, 'T_end': 0.0002, 'lr_init': 0.015},
            {'T_start': 0.008, 'T_end': 0.0005, 'lr_init': 0.025},
            {'T_start': 0.001, 'T_end': 0.0001, 'lr_init': 0.010},
        ]
        for ci, cfg in enumerate(configs):
            t_alloc = min(110, deadline - time.monotonic() - 550)
            if t_alloc < 10:
                break
            print(f"  Nesterov T={cfg['T_start']}->{cfg['T_end']}, lr={cfg['lr_init']}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_nesterov(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"nest_{ci}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Dual-informed refinement (~250s) ===
    print("\n=== Phase 4: Dual-Informed Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(200, deadline - time.monotonic() - 300)
        if t_alloc > 20:
            print(f"  Dual-informed, t={t_alloc:.0f}s")
            a_opt, _ = dual_informed_refine(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "dual")

        # Another FW pass
        t_alloc = min(50, deadline - time.monotonic() - 200)
        if t_alloc > 10:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fw_d2")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Coordinate Polish (~150s) ===
    print("\n=== Phase 5: Coordinate Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(130, deadline - time.monotonic() - 120)
        if t_alloc > 10:
            print(f"  Coord polish, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final Micro-Polish + FW (~100s) ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(50, deadline - time.monotonic() - 50)
        if t_alloc > 5:
            print(f"  Micro polish, t={t_alloc:.0f}s")
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "micro")

        t_alloc = min(40, deadline - time.monotonic() - 20)
        if t_alloc > 5:
            print(f"  Final FW, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_fw")

        # One more micro polish on the result
        t_alloc = min(15, deadline - time.monotonic() - 5)
        if t_alloc > 3:
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "micro2")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
