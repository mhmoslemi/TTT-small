    """Raised cosine initialization."""
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    a = 0.5 * (1 + np.cos(x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_gaussian(n, sigma_frac):
    """Gaussian bump initialization."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-x**2 / (2 * sigma**2))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_uniform(n):
    return np.ones(n) / n


def make_bell(n, width_frac):
    """Bell-shaped (sinc^2-like) initialization."""
    x = (np.arange(n) - (n - 1) / 2.0) / (width_frac * n / 2.0)
    a = np.sinc(x / np.pi) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def solve_lp(f, M, tl=10):
    """LP: max 1^T g s.t. conv(f,g)[k] <= M, g >= 0. Returns g or None."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': tl})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp(f0, time_limit, m_init_factor=1.0, restart_interval=50):
    """Frank-Wolfe with LP oracle, adaptive M, and periodic restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        g = solve_lp(f, M * m_relax, tl=5)
        if g is None:
            m_relax = min(m_relax * 1.2, 4.0)
            if m_relax >= 4.0:
                m_relax = 1.0
                g = make_raised_cosine(n)
            continue

        gs = g.sum()
        if gs < 1e-15:
            g = make_raised_cosine(n)
            gs = g.sum()
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                # Perturb and restart
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                M = fast_convolve(f).max()
                m_relax = 1.0
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine search
        if 0 < best_alpha < 1.0:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 3.0)

        if no_improve > 40:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            scale = 0.002 + 0.003 * min(1.0, no_improve / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max()
            m_relax = 1.0
            no_improve = 0

    return best_f, best_obj


def exp_gradient(f0, time_limit, T_init=0.001, lr_init=0.05, decay=0.0005):
    """Exponentiated gradient (mirror descent) with smoothed subgradient."""
    a = np.maximum(f0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Smoothed weights
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Compute smoothed subgradient: g[i] = 2 * sum_k p[k] * a[k-i]
        # This is a weighted sum of correlations
        g = np.zeros(n)
        # Use FFT for efficiency
        # g[i] = 2 * sum_k p[k] * a[max(0, k-n+1):min(n-1,k)] shifted
        # Actually, g = 2 * conv_p_a where conv is cross-correlation
        # g[i] = 2 * sum_k p[k] * a[k-i] for valid k,i
        # = 2 * (p * a)[i] in the cross-correlation sense
        # Let's compute it directly using the convolution structure
        
        # p has length 2n-1, a has length n
        # g[i] = 2 * sum_{k: 0<=k-i<n, 0<=k<2n-1} p[k] * a[k-i]
        # = 2 * sum_{j: 0<=j<n, i<=k<i+n} p[i+j] * a[j]
        # Let's use FFT: cross_correlation(p, a)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        P = np.fft.rfft(p, fft_size)
        A = np.fft.rfft(a, fft_size)
        # Cross-correlation: sum_j p[i+j] * a[j] = (p * a_reversed)[i + n - 1] ... 
        # Actually, let me just compute it as a correlation
        # g[i] / 2 = sum_j a[j] * p[i+j] for j=0..n-1, i=0..n-1 (where p[i+j] is valid)
        # This is like a "valid" cross-correlation
        # corr[i] = sum_j a[j] * p[i+j] = sum_j a[j] * p_conj_shifted[i+j]
        # In FFT terms: if we define q = p, then corr = ifft(fft(p) * conj(fft(a))) ... 
        # Let me just do it directly for clarity (n is at most ~2000, so O(n^2) is fine for a few iterations)
        
        # Actually, for n up to 2000, O(n^2) per iteration might be too slow.
        # Let me use FFT.
        # We want: g[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]
        # This is the "cross-correlation" of a and p, but only for the first n values.
        # corr[i] = sum_j a[j] * conj(p[i+j]) for real signals = sum_j a[j] * p[i+j]
        # In FFT: ifft(conj(fft(a)) * fft(p))[i] gives sum_j a[j] * p[i-j] ... 
        # Hmm, let me think more carefully.
        # 
        # Standard cross-correlation: (a ★ p)[i] = sum_j a[j] * conj(p[j-i])
        # What I want: sum_j a[j] * p[i+j] = (a ★ p)[-(i)] ... this is getting confusing.
        #
        # Let me just use the direct formula with numpy:
        # g[i] = 2 * sum_{j=0}^{min(n-1, len(p)-1-i)} a[j] * p[i+j]
        #
        # For n=768, this is O(n^2) = ~600K operations per iteration, which is fine.
        
        # Direct computation (vectorized):
        # g[i] = 2 * dot(a[:n-i], p[i:i+n])
        # For efficiency, use the fact that this is a cross-correlation
        # Let's reverse p and do convolution
        p_rev = p[::-1]
        # conv(a, p_rev)[i + n - 1] = sum_j a[j] * p_rev[i+n-1-j] = sum_j a[j] * p[n-1-i+j]
        # Hmm that's not quite right either.
        #
        # Let me just use the stride trick or a loop (n is small enough):
        # Actually, for n up to 2000, a loop of n iterations each doing an inner product of length up to n is O(n^2) = 4M, which takes maybe 1-2ms. That's fine.
        
        # Better: use np.correlate
        # np.correlate(a, p, mode='full')[i] = sum_j a[j] * p[j - i + n - 1] ... 
        # This is getting confusing. Let me just do it the simple way.
        
        # g[i]/2 = sum_{j=0}^{n-1} a[j] * p[i+j] (where p has length 2n-1, so i+j < 2n-1)
        # = sum_{j=0}^{n-1} a[j] * p[i+j]
        # For i=0: sum_j a[j]*p[j] (first n elements of p)
        # For i=1: sum_j a[j]*p[j+1]
        # ...
        # For i=n-1: sum_j a[j]*p[j+n-1]
        
        # This is like a "sliding dot product". I can compute it as:
        # g_half[i] = dot(a, p[i:i+n]) for i = 0, ..., n-1
        # But p has length 2n-1, so p[i:i+n] is valid for i=0,...,n-1.
        
        # Using FFT: 
        # g_half = corr where corr[i] = sum_j a[j]*p[i+j]
        # Let's define q = p, and we want the "asymmetric correlation"
        # If I reverse a: a_rev = a[::-1], then:
        # conv(a_rev, p)[i] = sum_j a_rev[j]*p[i-j] = sum_j a[n-1-j]*p[i-j]
        # = sum_k a[k]*p[i-(n-1-k)] where k = n-1-j
        # = sum_k a[k]*p[i-n+1+k]
        # So conv(a_rev, p)[i] = sum_k a[k]*p[i-n+1+k]
        # We want g_half[i] = sum_k a[k]*p[i+k]
        # So g_half[i] = conv(a_rev, p)[i+n-1]
        
        a_rev = a[::-1]
        conv_ap = np.convolve(a_rev, p, mode='full')
        g_half = conv_ap[n-1:n-1+n]  # indices n-1 to 2n-2
        g = 2.0 * g_half
        
        # Exponentiated gradient update
        log_update = -lr * g / T  # Wait, this doesn't have the right form
        # Actually, the update should be: a_new[i] ∝ a[i] * exp(-lr * g[i])
        # But g[i] can be large, so let's normalize
        log_a = np.log(a) - lr * g
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a  # Don't update if no improvement
        
        a = a_new
        lr *= (1.0 - decay)
        T *= (1.0 + 0.001)  # Slowly increase temperature
        
        if no_improve > 50:
            T *= 1.5
            lr = max(lr * 0.5, 0.001)
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            best_obj = fast_convolve(a).max()
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def peak_shave(f0, time_limit):
    """Greedy peak shaving: repeatedly find the best single-index or pair move to reduce the max peak."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        if peak_val < best_obj * 0.999:
            best_obj = peak_val
            best_a = a.copy()
            stall = 0
            continue

        # Find best move: transfer mass from src to dst
        # Effect on peak k: Δb[k] ≈ δ * (a[k-dst] - a[k-src]) (for small δ)
        # We want a[k-src] > a[k-dst] (move from high contribution to low contribution)
        
        # For the peak at k, the contribution weight of index i is a[k-i] (if valid)
        weights = np.zeros(n)
        for i in range(n):
            j = peak_k - i
            if 0 <= j < n:
                weights[i] = a[j]
        
        # Source: high weight and high a value
        # Destination: low weight
        # Try the top candidates
        
        best_gain = 0
        best_src = -1
        best_dst = -1
        best_delta = 0
        
        # Get top 20 sources and top 20 destinations
        src_candidates = np.argsort(weights * a)[::-1][:20]  # high contribution
        dst_candidates = np.argsort(weights)[:20]  # low contribution
        
        for src in src_candidates:
            if a[src] < 1e-8:
                continue
            for dst in dst_candidates:
                if src == dst:
                    continue
                # Try moving a fraction of a[src] to dst
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = a[src] * frac
                    a_test = a.copy()
                    a_test[src] -= delta
                    a_test[dst] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test = a_test / st
                    c_new = fast_convolve(a_test)
                    new_max = c_new.max()
                    gain = peak_val - new_max  # Only look at reduction in current peak
                    # But we need overall max to decrease
                    overall_gain = best_obj - new_max
                    if overall_gain > best_gain:
                        best_gain = overall_gain
                        best_src = src
                        best_dst = dst
                        best_delta = delta
        
        if best_src >= 0 and best_gain > 1e-12:
            a[best_src] -= best_delta
            a[best_dst] += best_delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            new_max = fast_convolve(a).max()
            if new_max < best_obj:
                best_obj = new_max
                best_a = a.copy()
                stall = 0
            else:
                a = best_a.copy()  # Revert
                stall += 1
        else:
            stall += 1

        if stall > 10:
            # Try a small random perturbation
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            scale = 0.001 + 0.002 * min(1.0, stall / 50.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            stall = 0
            if fast_convolve(a).max() < best_obj:
                best_obj = fast_convolve(a).max()
                best_a = a.copy()

        if stall > 100:
            # Larger restart
            rng = np.random.default_rng((it * 48271 + 5) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            stall = 50

    return best_a, best_obj


def pair_local_search(f0, time_limit):
    """Local search trying all pairs of small moves."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Top 3 peaks
        top3 = np.argsort(conv)[::-1][:3]

        improved = False
        for pk in top3:
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            pk = int(pk)
            # Indices affecting this peak
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            
            # For each pair (i, j) in the affected range, try swapping
            # Skip if range is too large
            affected = np.arange(js, je + 1)
            if len(affected) > 40:
                rng_s = np.random.default_rng(it * 7 + pk)
                affected = rng_s.choice(affected, size=40, replace=False)
            
            for ii in range(len(affected)):
                for jj in range(ii + 1, len(affected)):
                    i, j = int(affected[ii]), int(affected[jj])
                    # Try moving mass from i to j and vice versa
                    for direction in [(i, j), (j, i)]:
                        src, dst = direction
                        if a[src] < 1e-8:
                            continue
                        for frac in [0.01, 0.03, 0.1]:
                            delta = a[src] * frac
                            a_test = a.copy()
                            a_test[src] -= delta
                            a_test[dst] += delta
                            a_test = np.maximum(a_test, 0.0)
                            st = a_test.sum()
                            if st < 1e-15:
                                continue
                            a_test = a_test / st
                            c_new = fast_convolve(a_test)
                            if c_new.max() < best_obj - 1e-14:
                                a = a_test
                                best_obj = c_new.max()
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                if improved:
                    break
            if improved:
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng((it * 2654435761) % (2**32))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                stall = 0
            if stall > 150:
                break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search:
    1. Multi-n exploration with diverse initializations
    2. Deep FW-LP optimization
    3. Exponentiated gradient
    4. Peak-targeted greedy shaving
    5. Pair local search polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    t0_total = time.monotonic()

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={time.monotonic()-t0_total:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = make_raised_cosine(512)

    n_parent = len(parent)

    # === Phase 1: Multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    # Try various n values with different initializations
    n_list = sorted(set([
        n_parent, n_parent - 128, n_parent + 128,
        384, 448, 512, 576, 640, 704, 768, 832, 896, 960, 1024, 1280
    ]))
    n_list = [n for n in n_list if 256 <= n <= 1600]

    inits = [
        ("uniform", lambda n: make_uniform(n)),
        ("raised_cos", lambda n: make_raised_cosine(n)),
        ("gauss_25", lambda n: make_gaussian(n, 0.25)),
        ("gauss_35", lambda n: make_gaussian(n, 0.35)),
        ("gauss_50", lambda n: make_gaussian(n, 0.5)),
        ("bell_50", lambda n: make_bell(n, 0.5)),
    ]

    # For each n, try a few initializations with quick FW
    for n_try in n_list:
        if time.monotonic() > deadline - 1200:
            break

        if n_try == n_parent:
            init_seq = [parent.copy()]
        else:
            # Interpolate parent to this n as one option
            init_seq = []
            if parent is not None and n_parent > 0:
                interp = np.interp(np.linspace(0, 1, n_try),
                                   np.linspace(0, 1, n_parent), parent)
                interp = np.maximum(interp, 0)
                ss = interp.sum()
                if ss > 1e-10:
                    interp = interp / ss
                init_seq.append(interp)

        # Add diverse initializations
        for name, func in inits[:3]:  # Use first 3 for speed
            init_seq.append(func(n_try))

        for init_i, init in enumerate(init_seq):
            if time.monotonic() > deadline - 1200:
                break
            t_alloc = min(20, deadline - time.monotonic() - 1200)
            if t_alloc < 3:
                break
            label = f"p1_n{n_try}_i{init_i}"
            try_seq(init, label)  # Evaluate initialization directly
            a_opt, _ = fw_lp(init, time_limit=t_alloc)
            try_seq(a_opt, label + "_fw")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Main deep run
        t_alloc = min(400, deadline - time.monotonic() - 700)
        if t_alloc > 20:
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts with different scales
        for scale in [0.003, 0.008, 0.015, 0.025]:
            t_alloc = min(50, deadline - time.monotonic() - 500)
            if t_alloc < 10:
