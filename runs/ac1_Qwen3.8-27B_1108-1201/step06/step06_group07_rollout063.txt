```
def project_simplex(v):
    n=len(v)
    if n==0: return v
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    cond = u * np.arange(1,n+1) > (cssv-1.0)
    if not np.any(cond): return np.ones(n)/n
    rho=np.nonzero(cond)[0][-1]
    theta=(cssv[rho]-1.0)/(rho+1.0)
    return np.maximum(v-theta,0.0)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a)**2).

The implementation uses:
- log-parameter L-BFGS-B local searches with smoothed objectives
  (L2 flattening of the convolution, then log-sum-exp peak penalties),
- a Successive Convex Approximation style LP oracle,
- projected subgradient/mirror-style peak shaving,
- length re-interpolation, multi-restarts, and ensemble averaging.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

_EPS = 1e-300
_LOG_BOUND = 50.0


def positive_normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    a = a / s
    return a


def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([float(a[0] * a[0])], dtype=np.float64)
    length = 2 * n - 1
    size = 1 << (length - 1).bit_length()
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)
    return c[:length]


def grad_from_weights(w, a):
    """
    Gradient of sum_k w_k * b_k where b = conv(a, a).
    Returns array of length len(a).
    """
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([2.0 * float(w[0]) * float(a[0])], dtype=np.float64)

    length = 3 * n - 2
    size = 1 << (length - 1).bit_length()
    wr = w[::-1].copy()
    W = np.fft.rfft(wr, size)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(W * A, size)
    out = 2.0 * c[n - 1:2 * n - 1][::-1]
    out = np.maximum(out, 0.0)
    return out


def exact_objective(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.inf
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    a = a / s
    b = fast_convolve(a)
    if b.size == 0:
        return np.inf
    return float(2.0 * n * float(b.max()))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_to_len(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = len(a)
    if m == n:
        return a.copy()
    if m == 0:
        return np.zeros(n, dtype=np.float64)
    if m == 1:
        return np.full(n, float(a[0]) / float(n), dtype=np.float64)
    x0 = np.linspace(0.0, 1.0, m, dtype=np.float64)
    x1 = np.linspace(0.0, 1.0, n, dtype=np.float64)
    return np.interp(x1, x0, a).astype(np.float64)


def from_logits(x, base, idx):
    n = len(x) + 1
    e = np.exp(np.clip(x, -_LOG_BOUND, _LOG_BOUND))
    s = 1.0 + float(e.sum())
    a = np.empty(n, dtype=np.float64)
    a.fill(0.0)
    a[base] = 1.0 / s
    a[idx] = e / s
    return positive_normalize(a)


def variance_fjac(x, base, idx):
    n = len(x) + 1
    e = np.exp(np.clip(x, -_LOG_BOUND, _LOG_BOUND))
    s = 1.0 + float(e.sum())
    a = np.empty(n, dtype=np.float64)
    a.fill(0.0)
    a[base] = 1.0 / s
    a[idx] = e / s
    a = np.maximum(a, 0.0)
    som = float(a.sum())
    if not np.isfinite(som) or som <= 0.0:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = a / som

    b = fast_convolve(a)
    L = len(b)
    F = float(np.dot(b, b) - 1.0 / float(L))
    J = grad_from_weights(2.0 * b, a)
    m = float(np.dot(a, J))
    g = a[idx] * (J[idx] - m)

    if not np.isfinite(F) or not np.all(np.isfinite(g)):
        F = 1e100
        g = np.zeros(len(x), dtype=np.float64)
    return F, g


def smooth_fjac(x, base, idx, beta):
    beta = float(beta)
    n = len(x) + 1
    e = np.exp(np.clip(x, -_LOG_BOUND, _LOG_BOUND))
    s = 1.0 + float(e.sum())
    a = np.empty(n, dtype=np.float64)
    a.fill(0.0)
    a[base] = 1.0 / s
    a[idx] = e / s
    a = np.maximum(a, 0.0)
    som = float(a.sum())
    if not np.isfinite(som) or som <= 0.0:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = a / som

    b = fast_convolve(a)
    L = len(b)
    invL = 1.0 / float(L)
    logL = float(np.log(float(L)))
    maxb = float(b.max())

    z = beta * (b - maxb)
    em = np.exp(z)
    sumem = float(em.sum())
    if not np.isfinite(sumem) or sumem <= 0.0:
        F = 1e100
        g = np.zeros(len(x), dtype=np.float64)
        return F, g

    F = maxb - invL + (float(np.log(sumem)) - logL) / beta
    w = em / sumem
    J = grad_from_weights(w, a)
    m = float(np.dot(a, J))
    g = a[idx] * (J[idx] - m)

    if not np.isfinite(F) or not np.all(np.isfinite(g)):
        F = 1e100
        g = np.zeros(len(x), dtype=np.float64)
    return F, g


def stage_maxiter(n, budget):
    if budget <= 0:
        return 0
    base = int((float(budget) ** 0.7) * 20.0)
    if n <= 768:
        base = int(base * 1.8)
    elif n <= 1152:
        base = int(base * 1.5)
    elif n <= 1536:
        base = int(base * 1.2)
    else:
        base = int(base * 0.9)
    return int(np.clip(base, 20, 900))


def run_minimize_stage(a, budget, kind, beta=1.0):
    a = positive_normalize(a)
    n = len(a)
    if n <= 2 or budget < 5:
        return a

    base = int(np.argmax(a))
    idx = np.array([i for i in range(n) if i != base], dtype=np.int64)
    ratio = np.maximum(a[idx], _EPS) / max(float(a[base]), _EPS)
    x0 = np.log(ratio)
    x0 = np.clip(x0, -_LOG_BOUND, _LOG_BOUND)

    maxiter = stage_maxiter(n, budget)
    if maxiter <= 0:
        return a

    bounds = [(-_LOG_BOUND, _LOG_BOUND)] * len(x0)
    options = {
        "maxiter": maxiter,
        "ftol": 1e-14,
        "gtol": 1e-12,
        "maxls": 50,
        "maxfun": max(1, maxiter * 20)
    }

    try:
        if kind == "var":
            res = minimize(
                variance_fjac,
                x0,
                args=(base, idx),
                method="L-BFGS-B",
                bounds=bounds,
                jac=True,
                options=options
            )
        else:
            res = minimize(
                smooth_fjac,
                x0,
                args=(base, idx, float(beta)),
                method="L-BFGS-B",
                bounds=bounds,
                jac=True,
                options=options
            )

        if res is None or not hasattr(res, "x") or res.x is None:
            return a
        if not np.all(np.isfinite(res.x)):
            return a
        return from_logits(res.x, base, idx)
    except Exception:
        return a


def smooth_refine(a0, time_limit, betas=(1, 4, 16, 64, 256), include_var=True):
    a = positive_normalize(a0)
    n = len(a)
    if n <= 2 or time_limit <= 0:
        return a.copy(), exact_objective(a)

    a = (1.0 - 1e-6) * a + (1e-6 / float(n))
    a = positive_normalize(a)

    best_a = a.copy()
    best_score = exact_objective(a)
    t0 = time.monotonic()

    stages = []
    if include_var:
        stages.append(("var", 1.0, 0.25 if len(betas) > 0 else 1.0))
    if len(betas) > 0:
        share = (0.75 / float(len(betas))) if include_var else (1.0 / float(len(betas)))
        for b in betas:
            stages.append(("smo", float(b), share))

    for kind, beta, share in stages:
        elapsed = time.monotonic() - t0
        if elapsed >= time_limit:
            break
        budget = min(share * float(time_limit), float(time_limit) - elapsed)
        if budget < 3.0:
            continue

        a_new = run_minimize_stage(a, budget, kind, beta)
        a_new = positive_normalize(a_new)
        sc = exact_objective(a_new)

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        if (not np.isfinite(sc)) or (sc > 10.0 * best_score):
            a = best_a.copy()
        else:
            a = a_new.copy()

    return best_a, best_score


def build_lp_matrix(n):
    if n <= 0:
        return np.array([], dtype=np.int64), np.array([], dtype=np.int64)
    k = np.arange(2 * n - 1, dtype=np.int64)
    imax = np.minimum(n - 1, k)
    imin = np.maximum(0, k - n + 1)
    lengths = imax - imin + 1
    total = int(lengths.sum())
    rows = np.repeat(k, lengths)
    starts = np.empty(2 * n - 1, dtype=np.int64)
    starts[0] = 0
    if len(lengths) > 1:
        starts[1:] = np.cumsum(lengths[:-1])
    offsets = np.arange(total, dtype=np.int64) - np.repeat(starts, lengths)
    cols = imin[rows] + offsets
    return rows.astype(np.int64), cols.astype(np.int64)


def solve_lp_oracle(f, M, rows, cols):
    n = len(f)
    rows = np.asarray(rows, dtype=np.int64)
    cols = np.asarray(cols, dtype=np.int64)
    if len(rows) == 0:
        return None
    vals = f[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(len(rows), n))
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(len(rows), float(M), dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True}
        )
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if float(g.sum()) > 1e-15:
                return g.astype(np.float64)
    except Exception:
        return None
    return None


def sca_optimize(a0, time_limit, top_cons=512):
    f = positive_normalize(a0)
    n = len(f)
    if n < 4 or time_limit < 5:
        return f.copy(), exact_objective(f)

    f = (1.0 - 1e-6) * f + (1e-6 / float(n))
    f = positive_normalize(f)

    rows_all, cols_all = build_lp_matrix(n)
    conv = fast_convolve(f)
    bestM = float(conv.max())
    bestf = f.copy()
    target = bestM * 0.998

    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(54321)
    alphas = (0.1, 0.2, 0.3, 0.5, 0.7, 1.0)

    while (time.monotonic() - t0) < time_limit:
        remaining = float(time_limit) - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        conv = fast_convolve(f)
        curM = float(conv.max())

        if rows_all.size <= 800 or n <= 600:
            rows = rows_all
            cols = cols_all
        else:
            order = np.argsort(conv)[::-1]
            m = min(int(top_cons), len(order))
            sel = order[:m]
            rows = rows_all[sel]
            cols = cols_all[sel]

        g = solve_lp_oracle(f, target, rows, cols)

        if g is None:
            if target < curM * 0.999:
                target = curM * 0.999
            else:
                target = curM * 1.005
            no_improve += 1
            if no_improve > 50:
                f = project_simplex(bestf + rng.standard_normal(n) * 0.004)
                f = positive_normalize(f + 1e-6 / float(n))
                conv = fast_convolve(f)
                target = float(conv.max()) * 0.998
                no_improve = 0
            continue

        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if gs < 1e-15:
            target = curM * 0.998
            no_improve += 1
            if no_improve > 50:
                f = project_simplex(bestf + rng.standard_normal(n) * 0.004)
                f = positive_normalize(f + 1e-6 / float(n))
                conv = fast_convolve(f)
                target = float(conv.max()) * 0.998
                no_improve = 0
            continue

        g = g / gs
        best_val = curM
        best_cand = None

        candidates = [g]
        for al in alphas:
            candidates.append((1.0 - al) * f + al * g)

        for cand in candidates:
            cand = positive_normalize(cand)
            val = float(fast_convolve(cand).max())
            if val < best_val - 1e-15:
                best_val = val
                best_cand = cand.copy()

        if best_cand is not None:
            f = best_cand
            if best_val < bestM - 1e-15:
                bestM = best_val
                bestf = f.copy()
            target = bestM * (1.0 - 0.003)
            no_improve = 0
        else:
            no_improve += 1
            new_target = max(bestM * (1.0 - 0.002), target * 0.995)
            if new_target < curM * 0.98:
                new_target = curM * 0.98
            target = new_target

        if no_improve > 60:
            f = project_simplex(bestf + rng.standard_normal(n) * 0.005)
            f = positive_normalize(f + 1e-6 / float(n))
            conv = fast_convolve(f)
            target = float(conv.max()) * 0.998
            no_improve = 0

    return bestf, float(2 * n * bestM)


def projected_subgradient(a0, time_limit, top_k=30):
    f = positive_normalize(a0)
    n = len(f)
    if n <= 2 or time_limit <= 0:
        return f.copy(), exact_objective(f)

    f = (1.0 - 1e-6) * f + (1e-6 / float(n))
    f = positive_normalize(f)

    conv = fast_convolve(f)
    bestM = float(conv.max())
    bestf = f.copy()

    t0 = time.monotonic()
    scale = 1.0
    no_improve = 0
    rng = np.random.default_rng(777)
    base_alphas = np.array([1.0, 3.0, 10.0, 30.0, 100.0, 300.0], dtype=np.float64)

    while (time.monotonic() - t0) < time_limit:
        if (time.monotonic() - t0) > time_limit:
            break

        conv = fast_convolve(f)
        curM = float(conv.max())
        L = len(conv)

        if top_k >= L:
            peaks = np.arange(L)
            wtop = np.full(L, 1.0 / float(L), dtype=np.float64)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            vals = conv[peaks]
            temp = 0.05 * curM
            if temp <= 0:
                temp = 1e-12
            z = vals - vals.max()
            wtop = np.exp(z / temp)
            wtop = wtop / wtop.sum()

        full = np.zeros(L, dtype=np.float64)
        full[peaks] = wtop
        J = grad_from_weights(full, f)
        if not np.any(J > 0.0):
            break

        best_val = curM
        best_cand = None
        for al in base_alphas * scale:
            cand = project_simplex(f - al * J)
            val = float(fast_convolve(cand).max())
            if val < best_val - 1e-15:
                best_val = val
                best_cand = cand.copy()

        if best_cand is not None and best_val < curM - 1e-15:
            f = best_cand
            if best_val < bestM:
                bestM = best_val
                bestf = f.copy()
            no_improve = 0
            scale = min(scale * 1.2, 200.0)
        else:
            no_improve += 1
            scale = max(scale * 0.8, 0.05)

        if no_improve > 70:
            f = project_simplex(bestf + rng.standard_normal(n) * 0.004)
            f = positive_normalize(f + 1e-6 / float(n))
            scale = 1.0
            no_improve = 0

    return bestf, float(2 * n * bestM)


def record_candidate(a, best_seq, best_score, store, max_store=40):
    try:
        an = positive_normalize(a)
        sc = exact_objective(an)
    except Exception:
        return best_seq, best_score, False

    improved = False
    if np.isfinite(sc) and sc < best_score:
        best_seq = an.copy()
        best_score = float(sc)
        improved = True

    if np.isfinite(sc):
        if len(store) < max_store:
            store.append((float(sc), an.copy()))
        else:
            worst_idx = 0
            worst_val = store[0][0]
            for i, (sv, _) in enumerate(store):
                if sv > worst_val:
                    worst_val = sv
                    worst_idx = i
            if sc < worst_val:
                store[worst_idx] = (float(sc), an.copy())

    return best_seq, best_score, improved


def make_ensemble(target_len, sequences, weights=None):
    seqs = [np.asarray(s, dtype=np.float64).ravel() for s in sequences if s is not None and len(s) > 0]
    if len(seqs) == 0:
        return None
    L = int(target_len)
    interpolated = [interpolate_to_len(s, L) for s in seqs]
    if weights is None:
        w = np.ones(len(interpolated), dtype=np.float64) / float(len(interpolated))
    else:
        w = np.asarray(weights, dtype=np.float64)
        w = np.maximum(w, 0.0)
        sw = float(w.sum())
        if sw <= 0.0 or not np.isfinite(sw):
            w = np.ones(len(interpolated), dtype=np.float64) / float(len(interpolated))
        else:
            w = w / sw
    out = np.zeros(L, dtype=np.float64)
    for wi, aa in zip(w, interpolated):
        out += wi * aa
    return positive_normalize(out)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    total_budget = max(120.0, float(budget_s))
    deadline = t_start + max(60.0, total_budget - 45.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf
    store = []

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and float(p.sum()) > 1e-10:
            parent = positive_normalize(p)
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(768, dtype=np.float64) / 768.0

    best_seq, best_score, _ = record_candidate(parent, best_seq, best_score, store)
    print(f"initial best: n={len(best_seq) if best_seq is not None else 0}, score={best_score:.10f}")

    # Cheap length interpolations can sometimes already improve the score.
    for n_try in (1024, 1536, 2048):
        if time.monotonic() >= deadline:
            break
        init = interpolate_to_len(parent, n_try)
        best_seq, best_score, imp = record_candidate(init, best_seq, best_score, store)
        if imp:
            print(f"interpolated n={n_try}: {best_score:.10f}")

    # First local refinement of the best initial construction.
    if time.monotonic() < deadline - 250 and best_seq is not None:
        t_alloc = min(350.0, max(20.0, deadline - time.monotonic() - 200.0))
        print(f"phase 1: smooth refine n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_opt, _ = smooth_refine(best_seq, t_alloc, betas=(1, 4, 16, 64, 256), include_var=True)
        best_seq, best_score, imp = record_candidate(a_opt, best_seq, best_score, store)
        if imp:
            print(f"phase 1 best: {best_score:.10f}")

    # Medium length sweep.
    n_sweep = (640, 896, 1024, 1280, 1536)
    for n_try in n_sweep:
        if time.monotonic() >= deadline - 160:
            break
        t_alloc = min(80.0, deadline - time.monotonic() - 120.0)
        if t_alloc < 25.0:
            break
        init = interpolate_to_len(best_seq, n_try)
        init = 0.90 * init + 0.10 / float(n_try)
        print(f"phase 2: length search n={n_try}, budget={t_alloc:.0f}s")
        a_opt, _ = smooth_refine(init, t_alloc, betas=(4, 16, 64), include_var=True)
        best_seq, best_score, imp = record_candidate(a_opt, best_seq, best_score, store)
        if imp:
            print(f"length search n={n_try}: {best_score:.10f}")

    # If time remains, try a slightly longer construction.
    if time.monotonic() < deadline - 130:
        t_alloc = min(90.0, deadline - time.monotonic() - 110.0)
        if t_alloc > 25.0:
            init = interpolate_to_len(best_seq, 2048)
            init = 0.95 * init + 0.05 / 2048.0
            print(f"phase 2b: long length n=2048, budget={t_alloc:.0f}s")
            a_opt, _ = smooth_refine(init, t_alloc, betas=(16, 64), include_var=False)
            best_seq, best_score, imp = record_candidate(a_opt, best_seq, best_score, store)
            if imp:
                print(f"long length 2048: {best_score:.10f}")

    # Deep refinement of the current best length.
    if time.monotonic() < deadline - 280:
        t_alloc = min(400.0, deadline - time.monotonic() - 170.0)
        print(f"phase 3: deep smooth refine n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_opt, _ = smooth_refine(
            best_seq,
            t_alloc,
            betas=(1, 4, 16, 64, 256, 1024),
            include_var=True
        )
        best_seq, best_score, imp = record_candidate(a_opt, best_seq, best_score, store)
        if imp:
            print(f"deep smooth refine: {best_score:.10f}")

    # SCAs / LP oracle polishing.
    if time.monotonic() < deadline - 230 and len(best_seq) <= 1600:
        t_alloc = min(250.0, deadline - time.monotonic() - 130.0)
        print(f"phase 4: SCA n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(best_seq, t_alloc)
        best_seq, best_score, imp = record_candidate(a_opt, best_seq, best_score, store)
        if imp:
            print(f"SCA: {best_score:.10f}")

    # Subgradient peak shaving.
    if time.monotonic() < deadline - 160:
        t_alloc = min(120.0, deadline - time.monotonic() - 90.0)
        print(f"phase 5: subgradient n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_opt, _ = projected_subgradient(best_seq, t_alloc, top_k=30)
        best_seq, best_score, imp = record_candidate(a_opt, best_seq, best_score, store)
        if imp:
            print(f"subgradient top30: {best_score:.10f}")

        if time.monotonic() < deadline - 90:
            t_alloc2 = min(60.0, deadline - time.monotonic() - 40.0)
            if t_alloc2 > 10.0:
                a_opt, _ = projected_subgradient(best_seq, t_alloc2, top_k=10)
                best_seq, best_score, imp = record_candidate(a_opt, best_seq, best_score, store)
                if imp:
                    print(f"subgradient top10: {best_score:.10f}")

    # Multi-restart perturbations.
    for scale in (0.004, 0.01, 0.02, 0.05):
        if time.monotonic() >= deadline - 130:
            break
        t_alloc = min(40.0, deadline - time.monotonic() - 70.0)
        if t_alloc < 15.0:
            break
        ncur = len(best_seq)
        pert = project_simplex(np.asarray(best_seq, dtype=np.float64) + rng.standard_normal(ncur) * scale)
        pert = positive_normalize(0.99 * pert + 0.01 / float(ncur))
        print(f"phase 6: restart scale={scale}, budget={t_alloc:.0f}s")
        a_opt, _ = smooth_refine(pert, t_alloc, betas=(16, 64, 256), include_var=False)
        best_seq, best_score, imp = record_candidate(a_opt, best_seq, best_score, store)
        if imp:
            print(f"restart scale={scale}: {best_score:.10f}")

    # Ensemble averaging of retained local solutions.
    if time.monotonic() < deadline - 80 and len(store) >= 2:
        seqs_sorted = [a for _, a in sorted(store)[:12]]
        n_target = int(len(best_seq))
        candidates = []
        candidates.append(make_ensemble(n_target, seqs_sorted, None))

        # Pairwise ensembles among the best retained sequences.
        for i in range(min(6, len(seqs_sorted))):
            for j in range(i + 1, min(6, len(seqs_sorted))):
                candidates.append(make_ensemble(n_target, [seqs_sorted[i], seqs_sorted[j]], None))

        print(f"phase 7: ensembles, n_target={n_target}, candidates={len(candidates)}")
        for cand in candidates:
            if time.monotonic() >= deadline - 60:
                break
            if cand is None:
                continue
            best_seq, best_score, imp = record_candidate(cand, best_seq, best_score, store)
            if imp:
                print(f"ensemble improved: {best_score:.10f}")
                if time.monotonic() < deadline - 70:
                    t_alloc = min(40.0, deadline - time.monotonic() - 50.0)
                    if t_alloc > 15.0:
                        a_opt, _ = smooth_refine(cand, t_alloc, betas=(64, 256), include_var=False)
                        best_seq, best_score, imp2 = record_candidate(a_opt, best_seq, best_score, store)
                        if imp2:
                            print(f"ensemble polished: {best_score:.10f}")

    # Final extra polishing passes if time remains.
    extra = 0
    while time.monotonic() < deadline - 70 and extra < 3:
        t_alloc = min(100.0, deadline - time.monotonic() - 60.0)
        if t_alloc < 30.0:
            break
        print(f"phase 8: final polish {extra + 1}, budget={t_alloc:.0f}s")
        a_opt, _ = smooth_refine(best_seq, t_alloc, betas=(64, 256, 1024), include_var=False)
        best_seq, best_score, imp = record_candidate(a_opt, best_seq, best_score, store)
        if imp:
            print(f"final polish {extra + 1}: {best_score:.10f}")
        else:
            break
        extra += 1

    if best_seq is None or len(best_seq) == 0 or not np.all(np.isfinite(best_seq)):
        best_seq = np.ones(768, dtype=np.float64) / 768.0
        best_score = exact_objective(best_seq)

    final_list = [float(x) for x in best_seq]

    try:
        final_score = float(evaluate_sequence(final_list))
    except Exception:
        final_score = best_score

    if not np.isfinite(final_score):
        fallback = np.ones(768, dtype=np.float64) / 768.0
        final_list = [float(x) for x in fallback]
        try:
            final_score = float(evaluate_sequence(final_list))
        except Exception:
            final_score = np.inf

    print(f"FINAL: score={final_score:.10f}, n={len(final_list)}")
    return final_list
```
