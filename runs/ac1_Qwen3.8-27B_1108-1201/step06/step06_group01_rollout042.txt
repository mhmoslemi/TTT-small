For the spectral method:

  2. Compute A(ω) = FFT(a, N)
  3. Update amplitude: |A_new(ω)| = (|A(ω)| + |P(ω)|^{1/2}) / 2 (or some interpolation)
  4. Keep the phase: A_new(ω) = |A_new(ω)| · e^{i·arg(A(ω))}
  5. a_new = IFFT(A_new)[:, :n] (take first n elements)
  6. Clip to non-negative, renormalize
  7. Repeat

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_init(n, n_iter=200, seed=42):
    """
    Gerchberg-Saxton style spectral initialization.
    Target: make the auto-convolution as flat as possible.
    
    The ideal |A(omega)|^2 for a flat convolution of length 2n-1 (padded to N)
    is the DFT of a rectangular window of length 2n-1.
    """
    rng = np.random.default_rng(seed)
    N = 1 << (2 * n - 1).bit_length()  # FFT size >= 2n-1
    
    # Target: DFT of a rectangular window of length 2n-1
    rect = np.zeros(N)
    rect[:2*n-1] = 1.0
    target = np.fft.rfft(rect)
    target_amp = np.abs(target)
    
    # Initialize with random phase
    a = rng.standard_normal(n)
    a = np.abs(a)
    a = a / a.sum()
    
    for it in range(n_iter):
        # Forward transform
        A = np.fft.rfft(a, N)
        
        # Amplitude update: interpolate between current and target
        # Target amplitude should be sqrt of the DFT of a flat sequence
        # Flat sequence of length 2n-1 has DFT = rect DFT
        # So |A|^2 should be proportional to |rect_DFT|^2
        # Thus |A| should be proportional to |rect_DFT|
        
        current_amp = np.abs(A)
        phase = A / (current_amp + 1e-30)
        
        # Update amplitude: geometric mean or linear interpolation
        alpha = 0.5  # mixing parameter
        new_amp = (1 - alpha) * current_amp + alpha * target_amp / (target_amp.max() + 1e-30) * (current_amp.max() + 1e-30)
        
        # Actually, we want |A|^2 to be proportional to |target|^2
        # So |A| should be proportional to |target|
        # But we also need the time-domain signal to be non-negative and sum to 1
        # Let's use a soft update:
        new_amp = np.sqrt((1 - alpha) * current_amp**2 + alpha * (target_amp / target_amp.max())**2 * (current_amp.max())**2)
        
        A_new = new_amp * phase
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        if np.max(np.abs(a_new - a)) < 1e-10:
            break
        a = a_new
    
    return a

def solve_lp_oracle_sparse(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses sparse constraint matrix."""
    n = len(f)
    num_constraints = 2 * n - 1
    
    # Build sparse constraint matrix
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    
    for k in range(num_constraints):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

**Approach: Direct QP with cvxpy**

We want to minimize max_k c[k] where c = a*a. This is non-convex, but we can use a cutting-plane approach:
1. Start with a feasible point a_0
2. At a_0, find k* = argmax c[k]
3. Add constraint: c[k*](a) ≤ t
4. Solve: min t s.t. c[k_i](a) ≤ t for all added k_i, sum(a)=1, a≥0
5. This is still non-convex (c[k] is not convex), but we can linearize:
   c[k](a) ≈ c[k](a_0) + ∇c[k](a_0)^T (a - a_0) (first-order Taylor)
6. The linearized problem IS a LP!
7. Repeat: add the new k* from the solution, re-solve

This is essentially the **cutting-plane method** for the non-convex problem. It should converge to a local optimum.

    a = a / a.sum()
    n = len(a)
    
    # Accumulated cuts: each cut is (gradient_vector, rhs_value)
    cuts = []  # list of (g, b) where g^T a <= b

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

        if time.monotonic() - t0 > time_limit - 5:
            break

        # g_i^T a <= b_i + t for all cuts (linearized c[k_i](a) <= t)
        # Actually, let me reformulate:
        # Minimize t
        # s.t. c[k_i](a) <= t (linearized: ∇c[k_i](a_old)^T a + c[k_i](a_old) - ∇c[k_i](a_old)^T a_old <= t)
        #      sum(a) = 1
        #      a >= 0

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Cutting-plane method with linearized auto-convolution constraints
- Spectral Gerchberg-Saxton initialization for diverse starting points
- Frank-Wolfe with sparse LP oracle for global refinement
- Pair-swap local search with efficient delta evaluation
- Multi-start with diverse n values and initializations
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_lag(a, k):
    """Compute gradient of c[k] = sum_j a[j]*a[k-j] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        grad[j] = 2.0 * a[k - j]
    return grad


def spectral_gs_init(n, n_iter=300, seed=42):
    """
    Gerchberg-Saxton spectral initialization aiming for flat auto-convolution.
    Target amplitude spectrum is derived from the DFT of a rectangular window.
    """
    rng = np.random.default_rng(seed)
    N = 1 << (2 * n - 1).bit_length()

    rect = np.zeros(N)
    rect[:2 * n - 1] = 1.0
    target = np.abs(np.fft.rfft(rect))
    target_norm = target / (target.max() + 1e-30)

    a = rng.standard_normal(n)
    a = np.abs(a) + 0.1
    a = a / a.sum()

    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        current_amp = np.abs(A)
        phase = A / (current_amp + 1e-30)

        alpha = 0.3 * (1.0 - it / float(n_iter)) + 0.1
        scale = current_amp.max() + 1e-30
        new_amp = np.sqrt((1.0 - alpha) * current_amp**2 + alpha * (target_norm * scale)**2)

        A_new = new_amp * phase
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s

        if np.max(np.abs(a_new - a)) < 1e-12:
            break
        a = a_new

    return a


def cosine_window_init(n, beta=1.0):
    """Create a cosine-tapered window as initialization."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    x = (j - center) / (center + 1e-10)
    a = np.cos(beta * np.pi * x / 2.0)**2
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def u_shaped_init(n, gamma=1.0):
    """Create a U-shaped (high at edges) initialization."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    x = (j - center) / (center + 1e-10)
    a = 1.0 + gamma * x**2
    a = a / a.sum()
    return a


def gaussian_window_init(n, sigma_frac=0.3):
    """Gaussian window centered in the sequence."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((j - center) / sigma)**2)
    a = a / a.sum()
    return a


def random_structured_init(n, seed=0, n_peaks=8):
    """Random initialization with multiple Gaussian bumps."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    for _ in range(n_peaks):
        center = rng.uniform(0, n)
        sigma = rng.uniform(n * 0.02, n * 0.15)
        amp = rng.uniform(0.5, 2.0)
        j = np.arange(n)
        a += amp * np.exp(-0.5 * ((j - center) / sigma)**2)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def solve_lp_oracle(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    num_constraints = 2 * n - 1

    rows = []
    cols = []
    vals = []

    for k in range(num_constraints):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])

    A_ub = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))

    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        lp_time = min(15, remaining * 0.25)
        g = solve_lp_oracle(f, M, time_limit=lp_time)

        if g is None:
            M2 = M * 1.05
            g = solve_lp_oracle(f, M2, time_limit=lp_time)
        if g is None:
            peak_k = int(np.argmax(conv))
            sg = compute_gradient_lag(f, peak_k)
            sg_norm = np.linalg.norm(sg)
            if sg_norm > 1e-15:
                g = project_simplex(f - 0.05 * sg / sg_norm)
            else:
                g = np.ones(n) / n

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 60:
                break
            continue

        best_alpha = 0.0
        best_val = M

        alphas = [0.01, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 104729 + 3)
            scale = 0.001 * min(10.0, no_improve / 20.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def cutting_plane_optimize(a0, time_limit, max_cuts=300, verbose=False):
    """
    Cutting-plane method: iteratively linearize the auto-convolution at the
    current point and solve the resulting LP. The accumulated cuts approximate
    the non-convex max-convolution constraint from below/above.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    cuts_grads = []
    cuts_rhs = []
    cuts_center = []

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_cuts:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        # Current true objective
        c_curr = fast_convolve(a)
        true_max = c_curr.max()
        k_worst = int(np.argmax(c_curr))

        if true_max < best_obj - 1e-14:
            best_obj = true_max
            best_a = a.copy()
            if verbose:
                print(f"    CP it{it}: obj={true_max:.10f} (best updated)")

        # Add cut for worst lag at current point
        grad = compute_gradient_lag(a, k_worst)
        c_val = c_curr[k_worst]
        center_val = np.dot(grad, a)
        rhs = c_val - center_val  # grad^T(a'-a) + rhs <= t  => grad^T a' - t <= rhs

        cuts_grads.append(grad)
        cuts_rhs.append(rhs)
        cuts_center.append(a.copy())

        # Also add cuts for near-max lags (top 3)
        top_lags = np.argpartition(c_curr, -min(3, len(c_curr)))[-min(3, len(c_curr)):]
        for k_add in top_lags:
            if k_add == k_worst:
                continue
            if c_curr[k_add] > true_max * 0.95:
                g_add = compute_gradient_lag(a, k_add)
                c_add = c_curr[k_add]
                center_add = np.dot(g_add, a)
                rhs_add = c_add - center_add
                cuts_grads.append(g_add)
                cuts_rhs.append(rhs_add)
                cuts_center.append(a.copy())

        # Solve LP: min t s.t. grad_i^T a - t <= rhs_i, sum(a)=1, a>=0, t>=0
        num_cuts = len(cuts_grads)
        if num_cuts == 0:
            a = np.ones(n) / n
            continue

        A_ub = np.zeros((num_cuts, n + 1))
        b_ub = np.zeros(num_cuts)
        for i in range(num_cuts):
            A_ub[i, :n] = cuts_grads[i]
            A_ub[i, n] = -1.0
            b_ub[i] = cuts_rhs[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0, None)] * n + [(0, None)]

        try:
            lp_time = min(10, remaining * 0.3)
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': lp_time})
            if not res.success:
                break
            a_new = np.maximum(res.x[:n], 0.0)
            t_new = res.x[n]
        except Exception:
            break

        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        true_new = c_new.max()

        if true_new < best_obj - 1e-14:
            best_obj = true_new
            best_a = a_new.copy()

        # Convergence check
        gap = true_new - t_new
        if gap < max(1e-8, true_new * 1e-6):
            if verbose:
                print(f"    CP converged at it{it}: obj={true_new:.10f}, gap={gap:.2e}")
            break

        # If gap is large, the linearization is poor. Move partway toward the LP solution.
        if gap > true_new * 0.1:
            # Blend: move 50% toward the LP solution
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            a = a_new

        # Remove old cuts that are no longer useful (keep only recent ones)
        if num_cuts > 200:
            # Keep the most recent 200 cuts
            trim = num_cuts - 200
            cuts_grads = cuts_grads[trim:]
            cuts_rhs = cuts_rhs[trim:]
            cuts_center = cuts_center[trim:]

    return best_a, best_obj


def pair_swap_local_search(a0, time_limit, verbose=False):
    """
    Greedy local search with pair-swap moves.
    At each step, find the best (i, j, delta) transfer that reduces the max convolution.
    Uses efficient delta-evaluation for the top few peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        c_curr = fast_convolve(a)
        current_max = c_curr.max()

        # Get top 5 peaks
        top_k = min(5, len(c_curr))
        peak_indices = np.argpartition(c_curr, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(c_curr[peak_indices])[::-1]]

        best_gain = 0
        best_move = None

        # For the worst peak, find good swaps
        for k_peak in peak_indices[:3]:
            if time.monotonic() - t0 > time_limit - 2.0:
                break

            peak_val = c_curr[k_peak]
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            affected = np.arange(js, je + 1)

            # Sample affected indices if too many
            if len(affected) > 20:
                rng_s = np.random.default_rng(it * 31 + k_peak)
                perm = rng_s.permutation(len(affected))
                affected = affected[perm[:20]]

            # For each affected index, try moving mass to a random other index
            rng_m = np.random.default_rng(it * 7 + k_peak + 1)
            for idx_i in affected:
                if a[idx_i] < 1e-10:
                    continue

                # Try a few delta values
                for frac in [0.05, 0.1, 0.2]:
                    delta = a[idx_i] * frac
                    if delta < 1e-12:
                        continue

                    # Pick destination: random index not in affected set
                    for _ in range(3):
                        idx_j = np.random.randint(n)
                        if idx_j == idx_i:
                            continue

                        # Quick delta estimate for this peak
                        contrib_i = a[k_peak - idx_i] if 0 <= k_peak - idx_i < n else 0
                        contrib_j = a[k_peak - idx_j] if 0 <= k_peak - idx_j < n else 0

                        # Estimated change in c[k_peak]:
                        # delta_c ≈ 2*delta*(contrib_j - contrib_i) [first order]
                        est_change = 2.0 * delta * (contrib_j - contrib_i)

                        if est_change >= -1e-15:
                            continue  # Won't help this peak

                        # Full evaluation (only for promising moves)
                        a_trial = a.copy()
                        a_trial[idx_i] -= delta
                        a_trial[idx_j] += delta
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t

                        c_trial = fast_convolve(a_trial)
                        new_max = c_trial.max()
                        gain = current_max - new_max

                        if gain > best_gain:
                            best_gain = gain
                            best_move = (idx_i, idx_j, delta)

        if best_move is not None and best_gain > 1e-12:
            i, j, delta = best_move
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            new_obj = fast_convolve(a).max()
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Perturbation when stalled
        if stall > 15:
            rng_p = np.random.default_rng((it * 48271 + 13) % (2**31))
            scale = 0.001 + 0.002 * min(3.0, stall / 30.0)
            a = project_simplex(best_a + rng_p.standard_normal(n) * scale)
            stall = 0

        if stall > 150:
            break

    return best_a, best_obj


def multi_peak_subgradient(a0, time_limit, lr_init=0.003, verbose=False):
    """
    Subgradient method with weighted multi-peak gradient and adaptive step size.
    Uses Polyak-type step size based on a known lower bound.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    # Lower bound: 2n/(2n-1) for the objective, which means max_c >= S^2/(2n-1)
    # With S=1: max_c >= 1/(2n-1)
    lb_maxc = 1.0 / (2 * n - 1)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    t_nest = 1.0
    y = a.copy()
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Nesterov lookahead
        alpha_n = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha_n
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        y = y / y.sum()
        t_nest += 1.0

        conv_y = fast_convolve(y)
        current_max = conv_y.max()

        # Find active peaks (within 5% of max)
        threshold = current_max * 0.95
        active = np.where(conv_y >= threshold)[0]

        # Weight by "excess" above the lower bound
        weights = np.maximum(conv_y[active] - lb_maxc, 0.0)
        if weights.sum() < 1e-15:
            weights = np.ones(len(active))
        weights = weights / weights.sum()

        # Compute weighted subgradient
        sg = np.zeros(n)
        for wi, ki in zip(weights, active):
            if wi < 1e-10:
                continue
            g_k = compute_gradient_lag(y, int(ki))
            sg += wi * g_k

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        # Polyak step size: (f(y) - f*) / ||g||^2
        # f(y) = current_max (max of c at y), f* >= lb_maxc
        gap = current_max - lb_maxc
        if gap < 1e-12:
            gap = 1e-12
        step = gap / (sg_norm**2 + 1e-30)
        step = min(step, lr_init * 10.0)  # Cap the step
        step = step / (1.0 + 0.01 * t_nest)  # Decay

        # Update
        a_new = project_simplex(a - step * sg)

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Restart if stuck
        if no_improve > 80:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            scale = 0.002 * min(5.0, no_improve / 80.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0

    return best_a, best_obj


def micro_polish(a0, time_limit):
    """Very fine perturbative polish using structured moves targeting the top peak."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))

        # Try several structured perturbations
        improved = False
        for attempt in range(15):
            if time.monotonic() - t0 > time_limit - 0.3:
                break

            # Create perturbation that specifically targets reducing the peak
            noise = rng.standard_normal(n) * 0.0003

            # Bias: push mass away from indices that contribute to the peak
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= 0.0005 * a[partner]

            # Also try moving mass to indices far from the peak
            far_indices = np.where(np.abs(np.arange(n) - (peak_k // 2)) > n * 0.3)[0]
            if len(far_indices) > 0:
                noise[np.random.choice(far_indices, size=min(5, len(far_indices)), replace=False)] += 0.0003

            candidate = project_simplex(a + noise)
            c_cand = fast_convolve(candidate)
            if c_cand.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c_cand.max()
                best_a = a.copy()
                improved = True
                break

        if not improved and it % 200 == 0:
            # Occasionally try a slightly larger perturbation
            noise = rng.standard_normal(n) * 0.001
            candidate = project_simplex(best_a + noise)
            c_cand = fast_convolve(candidate)
            if c_cand.max() < best_obj:
                best_obj = c_cand.max()
                best_a = candidate.copy()
                a = candidate.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining cutting-plane,
