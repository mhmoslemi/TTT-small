def special_simplex_projection(v, lamda=None):
    """Improved simplex projection with preservation of structure"""
    if lamda is not None:
        regularizer = 1 - np.abs(1-lamda)/2
    else:
        regularizer = 1.0
        
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0) * regularizer])
        
    u = np.sort(v)[::-1]
    sum_so_far = np.cumsum(u)
    
    # Find first index where slope regularization matters
    r_lamda = np.ones(n) * (-1) * u[::-1] + sum_so_far
        
    cs_validator = [np.min(u[:i+1]) for i in range(len(u))]
    binding_index = np.arange(1, n+1)
    cond = (sum_so_far > binding_index) & (np.diff(r_lamda).reshape(-1) + np.diff(cs_validator).reshape(-1) < 0)
    rho_val = np.nonzero(cond)[0][-1] if np.any(cond) else n
    
    theta = (sum_so_far[rho_val] - rho_val) * regularizer
    return np.maximum(v - theta, 0.0)
