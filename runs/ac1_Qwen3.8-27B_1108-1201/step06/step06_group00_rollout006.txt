"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP. Shape: (num_constr, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def solve_lp_direction(f, M, active_set, time_limit=10):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_set, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f, active_set)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def golden_section_line_search(f, g, f0, M, n, max_iter=40):
    """Golden section search for optimal alpha in [0,1] minimizing max conv of (1-a)*f + a*g."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = fast_convolve(np.maximum((1.0 - c) * f + c * g, 0.0)).max()
    fd = fast_convolve(np.maximum((1.0 - d) * f + d * g, 0.0)).max()
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = fast_convolve(np.maximum((1.0 - c) * f + c * g, 0.0)).max()
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = fast_convolve(np.maximum((1.0 - d) * f + d * g, 0.0)).max()
    alpha = (lo + hi) / 2.0
    return alpha


def lse_weights(conv, beta):
    """Compute LSE softmax weights on the convolution peaks."""
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_bc / total


def compute_subgradient(f, active_peaks, weights):
    """Compute weighted sum of gradients of (f*f)[k] for k in active_peaks."""
    n = len(f)
    grad = np.zeros(n)
    for idx, k in enumerate(active_peaks):
        k = int(k)
        w = weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += w * 2.0 * f[k - i_range]
    return grad


def fw_adaptive(f0, time_limit, initial_k=20, max_k=None, verbose=False):

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Dynamic active set: peaks within threshold of max
        threshold = M - 0.05 * M  # top 95% of max

            active = np.argpartition(conv, -cur_k)[-cur_k:]
        else:
            active = np.argwhere(conv >= threshold).flatten()
            if len(active) == 0:
                active = np.argpartition(conv, -cur_k)[-cur_k:]

        # LP oracle

                rng = np.random.default_rng(it * 7919 + 13)

                conv_new = fast_convolve(f_new)

                    best_f = f_new.copy()
                f = f_new.copy()

        g = np.maximum(g, 0.0)

        if gs < 1e-15:

def subgradient_optimize(f0, time_limit, beta_start=20.0, beta_end=8000.0, top_k=50):
    """Subgradient method with LSE smoothing and adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 3.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        conv = fast_convolve(f)
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        w_full = lse_weights(conv, beta)
        w_peaks = w_full[peak_idx]
        w_sum = w_peaks.sum()
        if w_sum > 1e-15:
            w_peaks = w_peaks / w_sum
        else:
            w_peaks = np.ones(k_top) / k_top

        grad = compute_subgradient(f, peak_idx, w_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_normed = grad / gnorm

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(12):
            f_new = project_simplex(f - step * grad_normed)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.003, 15.0 / np.sqrt(n))
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.85
            if no_improve > 80:
                f = best_f.copy()
                lr = max(2.0 / np.sqrt(n), 0.003)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0
                rng = np.random.default_rng(it * 31 + 17)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)

        if beta < beta_end:
            beta *= 1.006

    return best_f, best_obj


def peak_coordinate_descent(f0, time_limit):
    """Coordinate-wise refinement: for each coordinate, find optimal value
    via 1D search considering the effect on the convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Get top peaks for evaluation
    n_peaks = min(30, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(f)
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()

        # Identify top peaks
        top_peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            # Current value
            f_i = f[i]
            if f_i < 1e-12:
                continue

            # Compute convolution without f[i]'s contribution
            # conv[k] involves f[i] when: j=i (f[i]*f[k-i]) or j=k-i (f[k-i]*f[i])
            # For k=2i: f[i]^2
            # For k!=2i, 0<=k-i<n, k-i!=i: 2*f[i]*f[k-i]

            C = conv.copy()
            # Subtract contribution of f[i]
            for k in top_peaks:
                k_int = int(k)
                ki = k_int - i
                if 0 <= ki < n:
                    if k_int == 2 * i:
                        C[k_int] -= f_i * f_i
                    elif ki != i:
                        C[k_int] -= 2.0 * f_i * f[ki]
                    else:
                        C[k_int] -= f_i * f_i

            # 1D search: minimize max_k (h_k(x)) over x >= 0
            # where h_k(x) = C[k] + contribution of x to conv[k]
            # contribution: if k==2i: x^2; if 0<=k-i<n and k!=2i: 2*x*f[k-i]; else 0

            # Also need to maintain sum(f)=1, so if we change f[i] to x,
            # we need to redistribute (f_i - x) to other coordinates.
            # Simple approach: add the difference uniformly to all other coordinates.
            # But this is expensive. Instead, use the scale-invariant formulation:
            # just optimize x and then re-normalize.

            # For efficiency, only consider a small range around current value
            # Try: reduce f[i] and see if the max peak decreases
            # Parameterize: f_new[i] = f[i] * (1 - delta), delta in [-0.5, 0.5]
            # Then redistribute: f_new[j] = f[j] + delta*f[i]/(n-1) for j != i (approximate)

            # Simpler: just try a few delta values
            deltas = [0.02, 0.05, 0.1, 0.15, 0.2, -0.05, -0.1]
            best_delta = 0.0
            best_val = conv.max()

            for delta in deltas:
                new_f_i = f_i * (1.0 - delta)
                if new_f_i < 0:
                    continue
                f_trial = f.copy()
                f_trial[i] = new_f_i
                # Redistribute the difference
                redist = (f_i - new_f_i) / (n - 1)
                f_trial[np.arange(n) != i] += redist
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_val:
                    best_val = conv_trial.max()
                    best_delta = delta

            if best_delta != 0.0:
                new_f_i = f_i * (1.0 - best_delta)
                f[i] = max(0.0, new_f_i)
                redist = (f_i - f[i]) / (n - 1)
                f[np.arange(n) != i] += redist
                f = np.maximum(f, 0)
                f /= f.sum()
                improved = True

        if not improved:
            break

    return best_f, best_obj


def freq_perturb(f, rng, strength=0.03):
    """Perturb f in the frequency domain."""
    n = len(f)
    F = np.fft.rfft(f)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, n)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    t4_end = t3_end + total_budget * 0.15
    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    t_alloc = min(150, t4_end - time.monotonic())
    if t_alloc > 10:
        print(f"  CoordDesc n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = peak_coordinate_descent(current_best, time_limit=t_alloc)
        try_sequence(f_opt, best_tracker)

    # === Phase 5: Alternating refinement ===
    print(f"\n=== Phase 5: Alternating refinement ===")
    t5_end = deadline - 60
    cycles = 0
    while time.monotonic() < t5_end and cycles < 20:
        cycles += 1
        current_best = best_tracker[1] if best_tracker[1] is not None else parent
        n_cur = len(current_best)
        remaining = t5_end - time.monotonic()
        if remaining < 25:
            break

        if cycles % 3 == 1:
            t_alloc = min(60, remaining * 0.35)
            if t_alloc > 10:
                print(f"  Cycle {cycles}: FW n={n_cur}, t={t_alloc:.0f}s")
                f_opt, _ = fw_adaptive(current_best, time_limit=t_alloc, initial_k=25, max_k=120)
                try_sequence(f_opt, best_tracker)

        elif cycles % 3 == 2:
            t_alloc = min(70, remaining * 0.35)
            if t_alloc > 10:
                beta_s = 100.0 + 100.0 * cycles
                print(f"  Cycle {cycles}: Subgrad n={n_cur}, t={t_alloc:.0f}s, beta={beta_s:.0f}")
                f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                                 beta_start=beta_s, beta_end=20000.0, top_k=40)
                try_sequence(f_opt, best_tracker)

        else:
            # Freq perturb + short FW
            t_alloc = min(50, remaining * 0.3)
            if t_alloc > 10:
                print(f"  Cycle {cycles}: Perturb+FW n={n_cur}, t={t_alloc:.0f}s")
                for attempt in range(2):
                    t_rem = t5_end - time.monotonic() - 10
                    if t_rem < 10:
                        break
                    strength = 0.01 + 0.005 * attempt
                    f_pert = freq_perturb(current_best, rng, strength=strength)
                    t_sub = (t_rem - 5) / 2.0
                    if t_sub > 5:
                        f_opt, _ = fw_adaptive(f_pert, time_limit=t_sub, initial_k=15, max_k=60)
                        try_sequence(f_opt, best_tracker)

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    # High-beta subgradient focused on top peaks
    t_alloc = min(80, deadline - time.monotonic() - 30)
    if t_alloc > 10:
        print(f"  High-beta subgrad n={n_cur}, t={t_alloc:.0f}s")
        f_opt, _ = subgradient_optimize(current_best, time_limit=t_alloc,
                                         beta_start=500.0, beta_end=50000.0, top_k=20)
        try_sequence(f_opt, best_tracker)

    # One final FW with large active set
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_alloc = min(50, deadline - time.monotonic() - 10)
    if t_alloc > 10:
        print(f"  Final FW n={len(current_best)}, t={t_alloc:.0f}s")
        f_opt, _ = fw_adaptive(current_best, time_limit=t_alloc, initial_k=40, max_k=200)
        try_sequence(f_opt, best_tracker)

    # Tiny perturbations around best
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 10:
        print(f"  Micro-perturbations, t={t_rem:.0f}s")
        for _ in range(15):
            if time.monotonic() > deadline - 5:
                break
            f_pert = freq_perturb(current_best, rng, strength=0.003)
            try_sequence(f_pert, best_tracker)
            f_pert2 = project_simplex(current_best + rng.standard_normal(n_cur) * 0.001)
            try_sequence(f_pert2, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    print("WARNING: Using uniform fallback")
    fallback = np.ones(768) / 768.0
    return [float(x) for x in fallback]
