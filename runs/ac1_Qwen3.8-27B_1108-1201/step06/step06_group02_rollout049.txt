    n = len(v)
    if n == 1:

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, a):
    """Compute sum_k p[k] * grad(b_k)[j] where grad(b_k)[j] = 2*a[k-j] for valid j"""

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        result[idx] += p[k] * 2.0 * a[k - idx]
    return result

    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_gradient(p, a):
    """Compute sum_k p[k] * 2*a[k-j] for valid j (gradient of smooth max)."""
    n = len(a)
    result = np.zeros(n)
    kp = p[p > 1e-14]
    idx_p = np.where(p > 1e-14)[0]
    for k in idx_p:
        pk = p[k]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        result[idx] += pk * 2.0 * a[k - idx]
    return result


def smooth_objective_and_grad(a, T):
    """Compute T*log(sum exp(b_k/T)) and its gradient."""
    conv = fast_convolve(a)
    shifted = (conv - conv.max()) / T
    log_sum = np.log(np.sum(np.exp(shifted)))
    smooth_val = conv.max() + T * log_sum
    p = np.exp(shifted)
    p /= p.sum()
    grad = hankel_gradient(p, a)
    return smooth_val, grad, conv, p


def lp_oracle(f, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_away_step(f0, time_limit):
    """Frank-Wolfe with away-step and aggressive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    # Away-step tracking
    vertices = [f.copy()]
    coefficients = [1.0]

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = lp_oracle(f, M * m_exp, time_limit=min(15, remaining * 0.3))
        if g is None:
            m_exp = min(m_exp * 1.2, 3.0)
            if m_exp > 2.5:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.0005 + 0.003 * min(1.0, no_improve / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                vertices = [f.copy()]
                coefficients = [1.0]
            continue

        # Standard FW step
        best_alpha = 0.0
        best_val = M

        for alpha in np.linspace(0.005, 1.0, 30):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine golden-section
        if 0.01 < best_alpha < 0.99:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(15):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                f1 = (1.0 - m1) * f + m1 * g
                f1 = np.maximum(f1, 0.0) / np.maximum(f1.sum(), 1e-15)
                f2 = (1.0 - m2) * f + m2 * g
                f2 = np.maximum(f2, 0.0) / np.maximum(f2.sum(), 1e-15)
                v1 = fast_convolve(f1).max()
                v2 = fast_convolve(f2).max()
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
                best_alpha = (lo + hi) / 2.0
                ft = (1.0 - best_alpha) * f + best_alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st > 1e-15:
                    ft = ft / st
                    val = fast_convolve(ft).max()
                    if val < best_val:
                        best_val = val

        # Away-step FW
        best_away_val = best_val
        best_away_beta = 0.0
        for vi, v in enumerate(vertices):
            if np.max(np.abs(v - g)) < 1e-8:
                continue
            for beta in [0.1, 0.2, 0.3, 0.5, 0.7]:
                # F_new = (f - beta*v) / (1 - beta)
                numerator = f - beta * v
                denom = 1.0 - beta
                if denom < 0.1:
                    continue
                if np.min(numerator) < -1e-10:
                    continue
                fa = numerator / denom
                fa_sum = fa.sum()
                if fa_sum < 1e-10:
                    continue
                fa = fa / fa_sum
                val_a = fast_convolve(fa).max()
                if val_a < best_away_val:
                    best_away_val = val_a
                    best_away_beta = beta
                    best_away_vi = vi

        if best_away_beta > 0 and best_away_val < best_val:
            # Take away step
            v = vertices[best_away_vi]
            numerator = f - best_away_beta * v
            f_new = numerator / (1.0 - best_away_beta)
            f_new = np.maximum(f_new, 0.0) / f_new.sum()
            f = f_new
            # Update vertex representation (approximate)
            if len(vertices) < 20:
                vertices.append(g.copy())
                coefficients.append(0.1)
            else:
                vertices[-1] = g.copy()
        else:
            if best_alpha > 1e-10:
                f = (1.0 - best_alpha) * f + best_alpha * g
                f = np.maximum(f, 0.0)
                f = f / f.sum()
            else:
                no_improve += 1
                m_exp = min(m_exp * 1.1, 3.0)
                continue

        conv = fast_convolve(f)
        M = conv.max()
        m_exp = 1.0

        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.001 + 0.004 * min(1.0, (no_improve - 60) / 150.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
            vertices = [f.copy()]
            coefficients = [1.0]

        if no_improve > 200:
            break

    return best_f, best_obj


def fista_smooth(a0, time_limit, T_init=0.001):
    """FISTA-accelerated gradient on smooth approximation with backtracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    T = T_init
    lr = 0.02
    c_upper = fast_convolve(a).max() + T * 2

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + (t - 1.0) / t_new * (a - a_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = a.copy()

        # Gradient with backtracking
        step = lr
        smooth_val, grad, conv, p = smooth_objective_and_grad(y, T)
        accepted = False

        for _ in range(15):
            candidate = project_simplex(y - step * grad)
            cand_conv = fast_convolve(candidate)
            cand_max = cand_max = cand_conv.max()
            # Accept if smoothed value decreased (approximately)
            cand_smooth = cand_max + T * np.log(np.sum(np.exp((cand_conv - cand_max) / T)))
            if cand_smooth <= smooth_val + 1e-10:
                accepted = True
                break
            step *= 0.5
            if step < 1e-12:
                break

        if not accepted:
            candidate = project_simplex(y - step * grad)

        a = candidate
        a_prev_old = a_prev
        a_prev = a
        t = t_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive T
        if no_improve > 30:
            T = max(T * 0.95, 1e-7)
        if no_improve > 100:
            lr *= 0.8
        if no_improve > 300:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 300) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            y = a.copy()
            a_prev = a.copy()
            t = 1.0
            T = T_init * 0.5
            lr = 0.02
            no_improve = 0

    return best_a, best_obj


def block_coordinate_descent(a0, time_limit, block_size=None):
    """Block coordinate descent: optimize blocks of coordinates via LP feasibility."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    if block_size is None:
        block_size = max(32, n // 8)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        it += 1

        improved = False
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            bs = end - start
            if bs < 2:
                continue
            if time.monotonic() - t0 > time_limit - 1:
                break

            # Get current max
            conv = fast_convolve(a)
            M_cur = conv.max()

            # For this block, try to find a better sub-vector
            # Bisection on target max value
            lo = M_cur * 0.95
            hi = M_cur * 1.05
            target_sum = a[start:end].sum()

            # Try to find block values that keep all affected convolution entries <= target
            # Affected k range: [start, start+end-2] ∪ [start, n-1+end-1]... 
            # Actually k ranges from start to start + (n-1) + (bs-1) ... let's be more careful
            # (a*a)[k] involves a[j]*a[k-j] for j in [start, end-1] and k-j in [0, n-1]
            # So k ranges from start to start + n - 1, but also from (k - (end-1)) to (k - start)
            # The affected k values are those where at least one of j or k-j is in [start, end-1]
            # This is k in [start, start + n - 1] for the first part
            # and k in [0 + start, n-1 + end-1] ... 
            # Let's just use all k in [max(0, start-n+1), min(2n-2, start + n - 1 + bs - 1)]
            k_lo = max(0, start - (n - 1))
            k_hi = min(2 * n - 2, start + n - 1 + bs - 1)
            affected_ks = np.arange(k_lo, k_hi + 1)

            # Binary search on target
            best_block = a[start:end].copy()
            for _ in range(10):
                mid = (lo + hi) / 2.0
                # LP: find g >= 0, sum(g) = target_sum, (a_modified * a_modified)[k] <= mid for k in affected
                # This is nonlinear due to the quadratic nature, so use a linearization
                # Linearize: (a_new * a_new)[k] ≈ (a*a)[k] + 2*sum_{j in block} a[k-j]*(g_j - a_j) + sum_{j1,j2 in block} g_j1 * g_j2
                # For simplicity, use the current a for the cross terms
                # (a_new * a_new)[k] ≈ sum_{j not in block} a[j]*a[k-j] + 2*sum_{j in block, k-j not in block} a[k-j]*g_j + sum_{j1,j2 in block} g_j1*g_j2
                # This is still quadratic. Let's use a simpler approach:
                # Just try a few perturbations of the block

                # Alternative: just do a local search on the block
                pass

            # Simpler approach: try to move mass from high-contributing to low-contributing positions
            block_a = a[start:end].copy()
            # Compute contribution of each block element to the peak
            k_peak = int(np.argmax(conv))
            contribs = np.zeros(bs)
            for i in range(bs):
                j = start + i
                other = k_peak - j
                if 0 <= other < n:
                    contribs[i] = a[j] * a[other]
                else:
                    contribs[i] = 0

            # Move mass from high-contrib to low-contrib
            if np.max(contribs) > np.min(contribs) + 1e-10:
                i_max = int(np.argmax(contribs))
                i_min = int(np.argmin(contribs))
                for delta_frac in [0.01, 0.005, 0.002, 0.001]:
                    delta = delta_frac * block_a[i_max]
                    if delta < 1e-12:
                        continue
                    candidate = a.copy()
                    candidate[start + i_max] -= delta
                    candidate[start + i_min] += delta
                    candidate = np.maximum(candidate, 0.0)
                    candidate = candidate / candidate.sum()
                    c = fast_convolve(candidate)
                    if c.max() < best_obj - 1e-15:
                        a = candidate
                        best_obj = c.max()
                        best_a = a.copy()
                        improved = True
                        break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(it * 12345 + 7)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    no_improve = 0
                elif no_improve > 200:
                    break

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """Fine coordinate-level polishing: try small transfers between pairs."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    # Precompute which pairs to try
    # Focus on pairs that are likely to help: high-value to low-value, or symmetric positions
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        it += 1

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # The peak is at k_peak, contributed by pairs (j, k_peak-j)
        # Try to reduce a[j] for the most contributing j and increase a[j'] for least contributing
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        if len(affected) == 0:
            break

        contribs = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contribs)]
        j_min = affected[np.argmin(contribs)]

        # Also try moving to a random position outside the affected range
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) > 0:
            j_min_out = outside[np.argmin(a[outside])]
        else:
            j_min_out = j_min

        improved = False
        for target_j in [j_min, j_min_out]:
            if a[j_max] < 1e-8:
                continue
            for delta_frac in [0.0001, 0.0002, 0.0005, 0.001, 0.002, 0.005]:
                delta = delta_frac * a[j_max]
                if delta < 1e-12:
                    continue
                candidate = a.copy()
                candidate[j_max] -= delta
                candidate[target_j] += delta
                candidate = np.maximum(candidate, 0.0)
                candidate = candidate / candidate.sum()
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                # Random perturbation
                scale = 0.0001 + 0.0005 * min(1.0, no_improve / 500.0)
                candidate = project_simplex(a + rng.standard_normal(n) * scale)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    no_improve = 0
            if no_improve > 1000:
                break

    return best_a, best_obj


def multiplicative_update(a0, time_limit, eta_init=0.01):
    """Multiplicative (mirror descent with entropy) updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = np.maximum(a, 1e-15)  # avoid log(0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = eta_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        it += 1

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Subgradient at the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        subgrad = 2.0 * a[k_peak - affected]

        # Multiplicative update: a[j] *= exp(-eta * sg[j]) for affected, normalize
        a[affected] *= np.exp(-eta * subgrad)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        eta *= 0.9995
        if no_improve > 50:
            eta *= 1.5
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            eta = eta_init * 0.5
            no_improve = 0

    return best_a, best_obj


def try_and_update(a, state, label):
    """Evaluate and update best state."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """

    Phase 5: Multi-n exploration + final verification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = [float('inf'), None]

            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Block Coordinate Descent (~300s) ===
    print("\n=== Phase 3: Block Coordinate Descent ===")

    remaining = deadline - time.monotonic() - (budget_s - 30) * 0.2
    if remaining > 20:
        t_alloc = min(remaining, 250)
        print("  BCD, t={:.0f}s".format(t_alloc))
        a_opt, _ = block_coordinate_descent(state[1], time_limit=t_alloc)
        try_and_update(a_opt, state, "bcd")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Coordinate Polish + Multiplicative (~250s) ===
    print("\n=== Phase 4: Coordinate Polish ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic() - (budget_s - 30) * 0.1
    if remaining > 10:
        t_alloc = min(remaining, 120)
        print("  Coord polish, t={:.0f}s".format(t_alloc))
        a_opt, _ = coordinate_polish(state[1], time_limit=t_alloc)
        try_and_update(a_opt, state, "coord")

    remaining = deadline - time.monotonic() - (budget_s - 30) * 0.05
    if remaining > 10:
        t_alloc = min(remaining, 80)
        print("  Multiplicative, t={:.0f}s".format(t_alloc))
        a_opt, _ = multiplicative_update(state[1], time_limit=t_alloc)
        try_and_update(a_opt, state, "mult")

    # More polish
    remaining = deadline - time.monotonic() - 60
    if remaining > 10:
        t_alloc = min(remaining, 100)
        a_opt, _ = coordinate_polish(state[1], time_limit=t_alloc)
        try_and_update(a_opt, state, "coord2")

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Multi-n + Final (~remaining) ===
    print("\n=== Phase 5: Multi-n + Final ===")
    t_phase = time.monotonic()

    # Try nearby n values
    for n_try in [640, 896, 1024]:
        remaining = deadline - time.monotonic() - 30
        if remaining < 20:
            break
        # Interpolate parent to new n
        a_interp = np.interp(np.linspace(0, 1, n_try),
                             np.linspace(0, 1, n_parent), state[1])
        a_interp = np.maximum(a_interp, 0.0)
        ss = a_interp.sum()
        if ss > 1e-10:
            a_interp = a_interp / ss
        else:
            a_interp = np.ones(n_try) / n_try

        score_interp = float(evaluate_sequence(a_interp.tolist()))
        if np.isfinite(score_interp) and score_interp < state[0]:
            state = [score_interp, a_interp.copy()]
            print("  NEW BEST [interp_n{}]: {:.8f}".format(n_try, score_interp))

        t_alloc = min(remaining - 5, 40)
        if t_alloc > 5:
            a_opt, _ = fw_away_step(a_interp, time_limit=t_alloc)
            try_and_update(a_opt, state, "fw_n{}".format(n_try))

    # Final fine FW pass
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        t_alloc = min(remaining, 30)
        a_opt, _ = fw_away_step(state[1], time_limit=t_alloc)
        try_and_update(a_opt, state, "final_fw")

    # Final coordinate polish
    remaining = deadline - time.monotonic() - 5
    if remaining > 3:
        a_opt, _ = coordinate_polish(state[1], time_limit=min(remaining, 10))
        try_and_update(a_opt, state, "final_coord")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    print("Target: 1.5027 | Achieved: {:.8f} | Improvement: {:.6f}".format(
        state[0], 1.504821 - state[0]))
    return [float(x) for x in state[1]]
