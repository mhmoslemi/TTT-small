import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    # Project onto {x >= 0, sum(x) = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, T):
    """Exact gradient of T*log(sum(exp(conv/T))) with respect to a."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    shifted = conv / T - conv.max() / T
    w = np.exp(shifted)
    w = w / w.sum()
    
    # grad[j] = (1/T) * sum_k w[k] * (a[k-j] + a[j-k]) * I(valid)
    # Using FFT:
    # term1[j] = sum_u a[u] * w[j+u] for u=0..n-1
    # term2[j] = sum_u a[u] * w[j-u] for u=0..n-1
    
    fft_size = 1 << (2*(2*n-1) - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    A_rev = np.fft.rfft(a[::-1], fft_size)
    
    # term2[j] = (a * w)[j] for j=0..n-1
    conv_aw = np.fft.irfft(A * W, fft_size)
    term2 = conv_aw[:n]
    
    # term1[j] = sum_u a[u]*w[j+u]
    # = (a_rev * w)[j + n - 1]
    conv_awr = np.fft.irfft(A_rev * W, fft_size)
    term1 = conv_awr[n-1:n-1+n]
    
    return (term1 + term2) / T

    return T * np.log(np.sum(np.exp((conv - m) / T))) + m

def spectral_equalize(a0, time_limit, alpha=0.5, inner_iters=50):

    a = np.maximum(a0, 0.0)

            target = mag[0] / np.sqrt(fft_size)

            if s < 1e-15:

def try_and_update(a, best_state, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)

def interp_sequence(parent, n_target):
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s
