"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Frank-Wolfe optimization
with spectral initialization, adaptive step sizes, and multi-scale polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(u, time_limit=10.0):
    """
    Solve LP: min t s.t. (u*v)[k] <= t for all k, sum(v) = 1, v >= 0.
    Returns the optimal v (normalized to sum 1).
    """
    n = len(u)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A (num_constraints x (n+1))
    # A[k, m] = u[k-m] if 0 <= k-m < n, else 0
    # A[k, n] = -1
    A = np.zeros((num_constraints, n + 1))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for m in range(j_start, j_end + 1):
            A[k, m] = u[k - m]
        A[k, n] = -1.0
    
    b = np.zeros(num_constraints)
    
    # Equality: sum(v) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize t = x[n]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: all variables >= 0
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            v = np.maximum(res.x[:n], 0.0)
            s = v.sum()
            if s > 1e-15:
                return v / s
    except Exception:
        pass
    return None


def fw_lp_optimize(u_init, time_limit, verbose=False):
    """
    Frank-Wolfe with LP oracle (minimize max).
    """
    u = np.maximum(u_init.astype(np.float64), 0.0)
    s = u.sum()
    if s < 1e-15:
        u = np.ones(len(u)) / len(u)
    else:
        u = u / s
    n = len(u)
    
    conv = fast_convolve(u)
    M = conv.max()
    best_obj = M
    best_u = u.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Solve LP oracle
        v = solve_lp_oracle(u, time_limit=min(5.0, remaining - 1.0))
        if v is None:
            # Fallback: use the "greedy" direction (maximizing sum under current M)
            v = np.ones(n) / n
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Check if v is close to u (convergence)
        if np.max(np.abs(v - u)) < 1e-8:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        no_improve = 0
        
        # Line search
        best_alpha = 0.0
        best_val = M
        
        # Coarse search
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            u_t = (1.0 - alpha) * u + alpha * v
            u_t = np.maximum(u_t, 0.0)
            st = u_t.sum()
            if st < 1e-15:
                continue
            u_t = u_t / st
            val = fast_convolve(u_t).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine search (bisection)
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                u_t = (1.0 - mid) * u + mid * v
                u_t = np.maximum(u_t, 0.0)
                st = u_t.sum()
                if st < 1e-15:
                    break
                u_t = u_t / st
                val = fast_convolve(u_t).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-12:
            u = (1.0 - best_alpha) * u + best_alpha * v
            u = np.maximum(u, 0.0)
            u = u / u.sum()
            conv = fast_convolve(u)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_u = u.copy()
        else:
            no_improve += 1
        
        if no_improve > 80:
            # Restart with perturbation
            rng = np.random.default_rng(it * 12345 + 7)
            u = project_simplex(best_u + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(u)
            M = conv.max()
            no_improve = 0
    
    return best_u, best_obj


def subgradient_step(u, lr, top_k=50):
    """
    One step of projected subgradient descent.
    Returns updated u.
    """
    n = len(u)
    conv = fast_convolve(u)
    
    # Get top-k peaks
    k = min(top_k, len(conv))
    peak_idx = np.argpartition(conv, -k)[-k:]
    
    # Compute subgradient (average of gradients of top-k peak functions)
    grad = np.zeros(n)
    for pk in peak_idx:
        pk = int(pk)
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        # d/d u[m] of conv[u,u][pk] = u[pk-m] + u[m-pk] (if valid)
        # Actually: conv[u,u][pk] = sum_j u[j]*u[pk-j]
        # d/d u[m] = u[pk-m] (if 0 <= pk-m < n) + u[m-(pk-n+1)] ... 
        # More carefully: conv[u,u][pk] = sum_j u[j]*u[pk-j]
        # d/d u[m]: the term u[m]*u[pk-m] (if valid) and u[pk-m]*u[m] (if valid)
        # = 2*u[pk-m] if 0 <= pk-m < n, else 0
        # Wait, that's only if m and pk-m are different indices.
        # Actually: sum_j u[j]*u[pk-j]. The derivative w.r.t. u[m] is:
        # - u[m] appears as u[j] when j=m: contributes u[pk-m] (if valid)
        # - u[m] appears as u[pk-j] when pk-j=m, i.e., j=pk-m: contributes u[pk-m] (if valid)
        # Wait, that's the same thing. Let me be more careful.
        # conv[u,u][pk] = sum_j u[j] * u[pk-j]
        # The variable u[m] appears in this sum when:
        #   - j = m: term is u[m] * u[pk-m] (valid if 0 <= pk-m < n)
        #   - pk-j = m, i.e., j = pk-m: term is u[pk-m] * u[m] (valid if 0 <= pk-m < n)
        # So if both conditions give the same j (i.e., m = pk-m, i.e., pk = 2m), we have one term: u[m]^2, derivative = 2*u[m].
        # Otherwise, we have two terms, each with derivative u[pk-m].
        # Total derivative = 2 * u[pk-m] if 0 <= pk-m < n, else 0.
        # Wait, that's not right either. Let me think again.
        # 
        # f = sum_j u[j] * u[pk-j]
        # df/du[m] = sum_j [delta(j,m) * u[pk-j] + u[j] * delta(pk-j, m)]
        # = u[pk-m] (if 0 <= pk-m < n) + u[pk-m] (if 0 <= pk-m < n)
        # = 2 * u[pk-m] (if 0 <= pk-m < n)
        # 
        # Wait, the second term: delta(pk-j, m) means pk-j = m, so j = pk-m.
        # The term is u[j] * delta(pk-j, m) = u[pk-m] * 1 (when j = pk-m).
        # So yes, df/du[m] = 2 * u[pk-m] if 0 <= pk-m < n.
        
        partner = pk - m if (m is a scalar) ... 
        
    # Actually, let me just do it vectorially.
    for pk in peak_idx:
        pk = int(pk)
        # For each m in [0, n-1], the derivative is 2*u[pk-m] if 0 <= pk-m < n
        # This is equivalent to: grad += 2 * (u reversed and shifted)
        # More precisely: for m in range(n): if 0 <= pk-m < n: grad[m] += 2*u[pk-m]
        m_start = max(0, pk - n + 1)
        m_end = min(n - 1, pk)
        if m_start <= m_end:
            grad[m_start:m_end+1] += 2.0 * u[pk - m_end:pk - m_start - 1:-1]
    
    # Project subgradient step
    u_new = u - lr * grad / max(1.0, np.linalg.norm(grad))
    u_new = project_simplex(u_new)
    return u_new


def polish_coordinate(u_init, time_limit):
    """
    Coordinate descent polish: try adjusting individual elements.
    """
    u = np.maximum(u_init.astype(np.float64), 0.0)
    u = u / u.sum()
    n = len(u)
    
    best_u = u.copy()
    best_obj = fast_convolve(u).max()
    
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        # Try random pairs of coordinates
        rng = np.random.default_rng()
        for _ in range(100):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            if u[i] < 1e-10 or u[j] < 1e-10:
                continue
            
            # Try to move mass from i to j (and vice versa)
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                delta = u[i] * frac
                u_test = u.copy()
                u_test[i] -= delta
                u_test[j] += delta
                u_test = np.maximum(u_test, 0.0)
                s = u_test.sum()
                if s < 1e-15:
                    continue
                u_test = u_test / s
                obj = fast_convolve(u_test).max()
                if obj < best_obj - 1e-14:
                    u = u_test
                    best_obj = obj
                    best_u = u.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            if stall > 50:
                break
        else:
            stall = 0
    
    return best_u, best_obj


def polish_random(u_init, time_limit):
    """
    Random perturbation polish.
    """
    u = np.maximum(u_init.astype(np.float64), 0.0)
    u = u / u.sum()
    n = len(u)
    
    best_u = u.copy()
    best_obj = fast_convolve(u).max()
    
    t0 = time.monotonic()
    it = 0
    
    scales = [0.001, 0.003, 0.01, 0.005, 0.002]
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        scale = scales[it % len(scales)]
        rng = np.random.default_rng(it * 7919 + 13)
        
        # Try multiple perturbations
        for _ in range(5):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            # Structured perturbation: bias toward reducing the peak
            conv = fast_convolve(u)
            peak_k = int(np.argmax(conv))
            
            noise = rng.standard_normal(n) * scale
            # Bias: reduce elements that contribute to the peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= 2.0 * scale * u[partner]
            
            candidate = project_simplex(u + noise)
            obj = fast_convolve(candidate).max()
            if obj < best_obj - 1e-15:
                u = candidate
                best_obj = obj
                best_u = u.copy()
                break
    
    return best_u, best_obj


def spectral_shape(u_init, time_limit):
    """
    Spectral shaping: iteratively flatten the power spectrum to reduce auto-correlation peak.
    """
    u = np.maximum(u_init.astype(np.float64), 0.0)
    u = u / u.sum()
    n = len(u)
    
    best_u = u.copy()
    best_obj = fast_convolve(u).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Compute FFT
        U = np.fft.rfft(u, 2 * n)
        power = np.abs(U) ** 2
        
        # Shape the power spectrum: raise low values, lower high values
        # Use a "flattening" operation
        power_min = power.min()
        power_max = power.max()
        if power_max < 1e-15:
            break
        
        # Target: make power spectrum more uniform
        # Use a soft thresholding approach
        target = power.mean()
        alpha = 0.1  # How aggressively to flatten
        power_new = alpha * target + (1 - alpha) * power
        power_new = np.maximum(power_new, power_min * 0.5)
        
        # Recompute U with the new power (keeping phases)
        phase = np.angle(U)
        U_new = np.sqrt(power_new) * np.exp(1j * phase)
        
        # Inverse FFT
        u_new = np.fft.irfft(U_new, 2 * n)[:n]
        u_new = np.maximum(u_new, 0.0)
        s = u_new.sum()
        if s < 1e-15:
            break
        u_new = u_new / s
        
        obj = fast_convolve(u_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_u = u_new.copy()
            u = u_new
        else:
            # Try less aggressive flattening
            alpha = 0.05
            power_new = alpha * target + (1 - alpha) * power
            power_new = np.maximum(power_new, power_min * 0.5)
            U_new = np.sqrt(power_new) * np.exp(1j * phase)
            u_new = np.fft.irfft(U_new, 2 * n)[:n]
            u_new = np.maximum(u_new, 0.0)
            s = u_new.sum()
            if s < 1e-15:
                break
            u_new = u_new / s
            obj = fast_convolve(u_new).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_u = u_new.copy()
                u = u_new
    
    return best_u, best_obj
