    a = a.copy()
    n = len(a)

        conv = fast_convolve(a)
        # LSE weights
        beta_conv = beta * conv
        beta_conv -= beta_conv.max()
        w = np.exp(beta_conv)
        w /= w.sum()

        grad = np.zeros(n)
        for k in range(2*n-1):

    return a

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_grad(a, w):
    """Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i] for valid indices."""
    n = len(a)

    for k in range(2 * n - 1):
        wk = w[k]

        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        grad[i_range] += wk * a[k - i_range]
    return 2.0 * grad


def lse_weights(conv, beta):
    """Compute LSE softmax weights on the convolution peaks."""
    bc = beta * conv
    bc_max = bc.max()

        top_idx = np.argpartition(conv, -k)[-k:]
        w = np.zeros(len(conv))
        w[top_idx] = 1.0 / k
        return w
    return exp_vals / total


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend((f[k - js]).tolist())
    if not rows:
        return csr_matrix((0, n))
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def fw_lp_step(f, M, top_k=40, time_limit=10):
    """Single FW step with LP oracle."""
    n = len(f)
    conv = fast_convolve(f)
    peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    center = n - 1
    if center not in peak_idx:
        peak_idx = np.append(peak_idx, center)
    A = build_lp_matrix(f, peak_idx)
    if A.shape[0] == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def coordinate_descent(f0, time_limit, n_sweeps=50, verbose=True):
    """Coordinate descent: try adjusting individual coordinates to reduce max peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f /= f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    rng = np.random.default_rng(42)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        # Find the top peak
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        if sweep % 10 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep}: obj={best_obj:.8f}, peak={peak_val:.8f}, t={elapsed:.0f}s")

        # Try adjusting coordinates that contribute to the peak
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)

        # Get contribution of each coordinate to the peak
        contributions = np.zeros(n)
        for i in range(lo, hi + 1):
            contributions[i] = f[i] * f[k_peak - i]
        total_contrib = contributions[lo:hi+1].sum()
        if total_contrib < 1e-15:
            continue

        # Try reducing high-contribution coordinates and adding to low-contribution ones
        for _ in range(20):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick a source (high contribution) and destination (low contribution)
            src_candidates = np.where(contributions > 0.5 * total_contrib / (hi - lo + 1))[0]
            if len(src_candidates) == 0:
                src_candidates = np.arange(lo, hi + 1)
            src = int(rng.choice(src_candidates))

            # Pick destination far from peak
            dest_candidates = np.where(contributions < 0.3 * total_contrib / (hi - lo + 1))[0]
            if len(dest_candidates) == 0:
                dest_candidates = rng.choice(n, size=5, replace=False)
            dst = int(rng.choice(dest_candidates))
            if dst == src:
                continue

            # Try moving a small fraction of mass
            for frac in [0.02, 0.05, 0.1, 0.2]:
                delta = f[src] * frac
                if delta < 1e-15:
                    continue
                f_trial = f.copy()
                f_trial[src] -= delta
                f_trial[dst] += delta
                if f_trial[src] < 0:
                    continue
                f_trial = np.maximum(f_trial, 0)
                f_trial /= f_trial.sum()
                conv_trial = fast_convolve(f_trial)
                if conv_trial.max() < best_obj - 1e-14:
                    best_obj = conv_trial.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    break

    return best_f, best_obj


def active_set_subgradient(f0, time_limit, n_active=20, verbose=True):
    """Subgradient focusing on the top n_active peaks."""
    f = np.maximum(f0.astype(np.float64), 1e-20)
    f /= f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    lr = 0.1
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(f)
        # Get top peaks
        top_idx = np.argpartition(conv, -n_active)[-n_active:]
        # Weight by how close they are to the max
        peak_vals = conv[top_idx]
        max_val = peak_vals.max()
        weights = np.exp(20 * (peak_vals - max_val))
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_active) / n_active
        else:
            weights /= wsum

        # Compute gradient from top peaks only
        grad = np.zeros(n)
        for ii, k in enumerate(top_idx):
            k = int(k)
            w = weights[ii]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += w * 2.0 * f[k - i_range]

        # Projected gradient step (on simplex)
        f_new = project_simplex(f - lr * grad)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            lr = min(lr * 1.01, 0.5)
        else:
            lr = max(lr * 0.7, 0.001)
            if it % 50 == 0 and lr < 0.01:
                # Perturb
                rng = np.random.default_rng(it)
                f = project_simplex(f + rng.standard_normal(n) * 0.005)
                lr = 0.1

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AS-SD iter={it}: obj={best_obj:.8f}, lr={lr:.4f}, t={elapsed:.0f}s")

    return best_f, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f) + 1)[:-1]
    x_new = np.linspace(0, 1, new_n + 1)[:-1]
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  *** NEW BEST: {score:.8f} (n={len(a)}) ***")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main optimization: combines mirror descent, LP refinement,
    coordinate descent, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"\nStarting optimization. Parent n={n_parent}, best={best_tracker[0]:.8f}")

    # === Phase 1: Mirror descent on parent (main workhorse) ===
    print("\n=== Phase 1: Mirror descent ===")
    t_phase1_end = time.monotonic() + (deadline - time.monotonic()) * 0.35
    t_alloc = t_phase1_end - time.monotonic()
    if t_alloc > 30:
        print(f"  MD on n={n_parent}, t={t_alloc:.0f}s")
        f_opt, obj = mirror_descent(parent, time_limit=t_alloc,
                                     beta_start=3.0, beta_end=5000.0,
                                     eta_start=0.05, verbose=True, seed_offset=0)
        try_sequence(f_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    # === Phase 2: LP refinement ===
    print(f"\n=== Phase 2: LP refinement (n={n_cur}) ===")
    t_phase2_end = t_phase1_end + (deadline - t_phase1_end) * 0.25
    t_alloc = t_phase2_end - time.monotonic()
    if t_alloc > 20:
        print(f"  FW-LP, t={t_alloc:.0f}s")
        f_opt, obj = fw_optimize(current_best, time_limit=t_alloc, top_k=40, verbose=True)
        try_sequence(f_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else current_best

    # === Phase 3: Active-set subgradient ===
    print(f"\n=== Phase 3: Active-set subgradient ===")
    t_phase3_end = t_phase2_end + (deadline - t_phase2_end) * 0.2
    t_alloc = t_phase3_end - time.monotonic()
    if t_alloc > 15:
        print(f"  AS-SD, t={t_alloc:.0f}s")
        f_opt, obj = active_set_subgradient(current_best, time_limit=t_alloc,
                                             n_active=15, verbose=True)
        try_sequence(f_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else current_best

    # === Phase 4: Coordinate descent ===
    print(f"\n=== Phase 4: Coordinate descent ===")
    t_phase4_end = t_phase3_end + (deadline - t_phase3_end) * 0.15
    t_alloc = t_phase4_end - time.monotonic()
    if t_alloc > 10:
        print(f"  CD, t={t_alloc:.0f}s")
        f_opt, obj = coordinate_descent(current_best, time_limit=t_alloc,
                                         n_sweeps=100, verbose=True)
        try_sequence(f_opt, best_tracker)

    current_best = best_tracker[1] if best_tracker[1] is not None else current_best

    # === Phase 5: Alternating refinement cycles ===
    print(f"\n=== Phase 5: Alternating refinement ===")
    t_phase5_end = deadline - 120
    cycles = 0
    while time.monotonic() < t_phase5_end and cycles < 12:
        cycles += 1
        current_best = best_tracker[1] if best_tracker[1] is not None else current_best
        n_cur = len(current_best)
        remaining = t_phase5_end - time.monotonic()
        if remaining < 15:
            break

        if cycles % 3 == 1:
            # Mirror descent with higher beta
            t_alloc = min(60, remaining * 0.35)
            if t_alloc > 10:
                print(f"  Cycle {cycles}: MD (high beta), t={t_alloc:.0f}s")
                f_opt, _ = mirror_descent(current_best, time_limit=t_alloc,
                                           beta_start=50.0, beta_end=20000.0,
                                           eta_start=0.02, verbose=False,
                                           seed_offset=cycles)
                try_sequence(f_opt, best_tracker)

        elif cycles % 3 == 2:
            # LP step
            t_alloc = min(50, remaining * 0.3)
            if t_alloc > 8:
                print(f"  Cycle {cycles}: FW-LP, t={t_alloc:.0f}s")
                f_opt, _ = fw_optimize(current_best, time_limit=t_alloc,
                                        top_k=50, verbose=False)
                try_sequence(f_opt, best_tracker)

        else:
            # Coordinate descent
            t_alloc = min(40, remaining * 0.25)
            if t_alloc > 5:
                print(f"  Cycle {cycles}: CD, t={t_alloc:.0f}s")
                f_opt, _ = coordinate_descent(current_best, time_limit=t_alloc,
                                               n_sweeps=80, verbose=False)
                try_sequence(f_opt, best_tracker)

    # === Phase 6: Try other lengths ===
    print(f"\n=== Phase 6: Multi-length exploration ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_remaining = deadline - time.monotonic()
    if t_remaining > 60:
        # Try a few nearby lengths
        for n_try in [int(n_cur * 0.85), int(n_cur * 1.15), int(n_cur * 0.7), int(n_cur * 1.3)]:
            if time.monotonic() > deadline - 30:
                break
            n_try = max(128, min(2000, n_try))
            if n_try == n_cur:
                continue
            t_alloc = min(40, (deadline - time.monotonic()) * 0.4)
            if t_alloc < 10:
                break
            print(f"  Trying n={n_try}, t={t_alloc:.0f}s")
            f_interp = interpolate_to_length(current_best, n_try)
            f_opt, _ = mirror_descent(f_interp, time_limit=t_alloc,
                                        beta_start=5.0, beta_end=2000.0,
                                        eta_start=0.05, verbose=False,
                                        seed_offset=n_try)
            try_sequence(f_opt, best_tracker)

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else current_best
    t_rem = deadline - time.monotonic()
    if t_rem > 15:
        # One last MD pass with very high beta
        t_alloc = min(50, t_rem * 0.5)
        if t_alloc > 8:
            print(f"  Final MD, t={t_alloc:.0f}s")
            f_opt, _ = mirror_descent(current_best, time_limit=t_alloc,
                                       beta_start=200.0, beta_end=50000.0,
                                       eta_start=0.01, verbose=False, seed_offset=999)
            try_sequence(f_opt, best_tracker)

    # Small random perturbations of the best
    if best_tracker[1] is not None and time.monotonic() < deadline - 5:
        current_best = best_tracker[1]
        n_cur = len(current_best)
        for attempt in range(5):
            if time.monotonic() > deadline - 3:
                break
            f_pert = current_best + rng.standard_normal(n_cur) * 0.003
            f_pert = np.maximum(f_pert, 0)
            f_pert /= f_pert.sum()
            try_sequence(f_pert, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = np.maximum(best_tracker[1], 0.0)
        final_seq = np.minimum(final_seq, 1000.0)
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    print("WARNING: Using uniform fallback")
    fallback = np.ones(512) / 512.0
    return [float(x) for x in fallback]
