import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_gradient_peak(a, k):
    """
    Compute gradient of (a*a)[k] with respect to a.
    (a*a)[k] = sum_{i} a[i] * a[k-i] (valid indices)
    d/dx[j] = a[k-j] if 0 <= k-j < n
    """
    n = len(a)
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js <= je:
        idx = np.arange(js, je + 1)
        g[idx] = a[k - idx]
    return 2.0 * g  # gradient of x^T H_k x is 2*H_k*x

def solve_lp_active_set(a, active_peaks, max_t=None):
    """
    Solve the linearized problem:
    minimize t
    subject to: 2 * g_k^T x <= t + c_k for k in active_peaks
                sum(x) = 1
                x >= 0
    
    Returns (x_opt, t_opt) or (None, None) if infeasible.
    """
    n = len(a)
    conv_a = fast_convolve(a)
    
    # Build LP: variables are [x_0, ..., x_{n-1}, t]
    # Minimize: [0, ..., 0, 1] (just t)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ vars <= b_ub
    # For each active peak k: 2*g_k^T x - t <= c_k
    # => [2*g_k, -1] @ [x, t] <= c_k

    b_ub = []
    for k in active_peaks:

        row = np.zeros(n + 1)
        row[:n] = g  # 2 * g_k
        row[n] = -1.0  # -t
        A_ub.append(row)
        b_ub.append(conv_a[k])

    b_ub = np.array(b_ub)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: x >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    x_opt = result.x[:n]
    t_opt = result.x[n]
    
    return x_opt, t_opt

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP-based linearized direction finding with line search (primary)
- Nesterov-accelerated subgradient (polish)
- Multi-n exploration with adaptive time allocation
- Structured and diverse initializations

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_peak(a, k):
    """Gradient of (a*a)[k] w.r.t. a: 2*a[k-j] for valid j."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k - idx]
    return g


def solve_direction_lp(a, peaks):
    """
    Solve LP to find a descent direction.
    min t s.t. 2*g_k^T x - t <= c_k for k in peaks, sum(x)=1, x>=0
    Returns (x_opt, t_opt) or (None, None).
    """
    n = len(a)
    conv_a = fast_convolve(a)

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub_rows = []
    b_ub_vals = []
    for k in peaks:
        g = grad_peak(a, k)
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_ub_rows.append(row)
        b_ub_vals.append(conv_a[k])

    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            return None, None
        return res.x[:n], res.x[n]
    except Exception:
        return None, None


def line_search(a, direction, current_obj, n_evals=8):
    """
    Find best alpha in [0,1] to minimize max(conv(a + alpha*dir)).
    Uses multi-point evaluation.
    """
    n = len(a)
    best_alpha = 0.0
    best_val = current_obj

    # Try a range of alphas
    alphas = np.linspace(0, 1, n_evals)
    for alpha in alphas:
        if alpha == 0:
            continue
        candidate = project_simplex(a + alpha * direction)
        conv_c = fast_convolve(candidate)
        val = conv_c.max()
        if val < best_val - 1e-15:
            best_val = val
            best_alpha = alpha

    return best_alpha, best_val


def lp_direction_method(a0, time_limit, max_peaks=80, verbose=False):
    """
    LP-based direction finding with line search.
    Main optimization workhorse.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Determine active peaks: those within 99% of max, capped at max_peaks
        threshold = cmax * 0.99
        all_peaks = np.where(conv >= threshold)[0]
        if len(all_peaks) > max_peinks:
            # Take the largest ones
            peak_vals = conv[all_peaks]
            top_idx = np.argsort(peak_vals)[-max_peaks:]
            peaks = all_peaks[top_idx]
        else:
            peaks = all_peaks

        if len(peaks) < 3:
            # Ensure minimum active set
            peaks = np.argpartition(conv, -5)[-5:]

        # Solve LP for direction
        x_lp, t_lp = solve_direction_lp(a, peaks)

        if x_lp is None:
            no_improve += 1
            stagnant += 1
            if stagnant > 30:
                # Fallback: subgradient step
                k_peak = int(np.argmax(conv))
                g = grad_peak(a, k_peak)
                new_a = project_simplex(a - 0.01 * g)
                new_obj = fast_convolve(new_a).max()
                if new_obj < best_obj:
                    a = new_a
                    best_obj = new_obj
                    best_a = a.copy()
                    no_improve = 0
                    stagnant = 0
                else:
                    if no_improve > 100:
                        break
            continue

        # Line search along direction
        direction = x_lp - a
        alpha, new_obj = line_search(a, direction, cmax)

        if alpha > 0 and new_obj < cmax - 1e-15:
            a = project_simplex(a + alpha * direction)
            obj_check = fast_convolve(a).max()
            if obj_check < best_obj - 1e-14:
                best_obj = obj_check
                best_a = a.copy()
                no_improve = 0
                stagnant = 0
            else:
                no_improve += 1
                stagnant += 1
        else:
            no_improve += 1
            stagnant += 1

        # Restart mechanism
        if no_improve > 100:
            rng = np.random.default_rng((it * 2654435761) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0
            stagnant = 0

        if stagnant > 200:
            # Aggressive restart
            rng = np.random.default_rng((it * 40503 + 17) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            stagnant = 0
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    lp_dir it={it}, obj={best_obj:.10f}, peaks={len(peaks)}")

    return best_a, best_obj


def nesterov_subgrad(a0, time_limit, lr_init=0.005, momentum_init=0.9):
    """Nesterov-accelerated subgradient method."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = momentum_init
    y = a.copy()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        y = a + mom * v
        y = project_simplex(y)

        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))
        g = grad_peak(y, k_peak)

        v_new = project_simplex(y - lr * g) - y
        a = project_simplex(y + v_new)
        v = v_new

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr = max(1e-5, lr * 0.9999)
        mom = min(0.99, mom + 0.00003)

        if no_improve > 200:
            lr *= 0.7
            mom *= 0.95
        if no_improve > 500:
            rng = np.random.default_rng((it * 48271 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = momentum_init
            no_improve = 0

    return best_a, best_obj


def smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.02):
    """Smoothed gradient descent with softmax weighting of peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient: sum_k p[k] * grad_k = 2 * sum_k p[k] * a[k-j] for each j
        # = 2 * correlate(p, a) appropriately
        grad = np.zeros(n)
        for k in range(len(p)):
            if p[k] < 1e-15:
                continue
            g_k = grad_peak(a, k)
            grad += p[k] * g_k

        # Line search
        best_alpha = 0.0
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = trial_lr

        if best_alpha > 0:
            a = project_simplex(a - best_alpha * grad)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T *= 1.1
            lr *= 0.95
        if no_improve > 60:
            T *= 0.7
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def multi_peak_balance(a0, time_limit):
    """
    Iteratively balance the top peaks by solving a small QP:
    minimize max of top-m peaks, using iterative water-filling.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Get top peaks
        m = min(20, n)
        top_k = np.argpartition(conv, -m)[-m:]
        top_k = top_k[np.argsort(conv[top_k])[::-1]]

        # For each top peak, compute gradient and try to reduce it
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            g = grad_peak(a, k)
            # Try to move mass from high-gradient positions to low-gradient
            # Simple: project a - step * g
            for step in [0.005, 0.002, 0.001, 0.0005]:
                candidate = project_simplex(a - step * g)
                c_new = fast_convolve(candidate)
                if c_new.max() < best_obj - 1e-14:
                    a = candidate
                    best_obj = c_new.max()
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng((it * 7919 + 3) % (2**31))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosh window (tapered edges)
    cosh_w = 1.0 / np.cosh(0.5 * (x - 0.5 * (n - 1)) / (0.1 * n))
    inits.append(cosh_w / cosh_w.sum())

    # Blackman-like window
    bw = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    bw = np.maximum(bw, 0.001)
    inits.append(bw / bw.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
