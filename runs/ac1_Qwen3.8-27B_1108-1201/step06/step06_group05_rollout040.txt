```python
def compute_gradient(a, p):
    """Compute gradient of sum_k p[k]*c[k] where c = a*a.
    a has length n, p has length 2n-1.
    Returns gradient of length n."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(2*n - 1):
            # ∂c[k]/∂a[j] = a[k-j] if 0<=k-j<n + a[j-k] if 0<=j-k<n
            if 0 <= k - j < n:
                s += p[k] * a[k - j]
            if 0 <= j - k < n:
                s += p[k] * a[j - k]
        grad[j] = s
    return grad
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. FFT-based smoothed gradient with Adam optimizer (log-sum-exp proxy)
2. Spectral shaping for initialization and refinement
3. Frank-Wolfe with LP oracle for global improvements  
4. Active-set Newton for final polish
5. Multi-n systematic exploration
Key innovation: efficient O(N log N) gradient via cross-correlation identity,
Adam momentum for faster convergence, and spectral shaping to flatten
the auto-convolution power spectrum.
"""
import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_gradient(weights, a):
    """
    Compute gradient of sum_k weights[k]*c[k] where c = a*a.
    Uses the identity: grad = 2 * cross_correlation(a, weights)[:n]
    which is computed via FFT in O(N log N).
    weights: array of length 2n-1 (or longer, zero-padded)
    a: array of length n
    Returns: array of length n
    """
    n = len(a)
    # Need FFT size >= n + len(weights) - 1 for linear cross-correlation
    total_len = n + len(weights) - 1
    N = 1 << (total_len - 1).bit_length()
    Af = np.fft.rfft(a, N)
    Wf = np.fft.rfft(weights, N)
    cross = np.fft.irfft(Af * np.conj(Wf), N)
    return 2.0 * cross[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def logsumexp_obj_grad(a, beta):
    """
    Compute log-sum-exp approximation to max(c) and its gradient.
    Returns (objective, gradient, auto-conv).
    """
    c = fast_convolve(a)
    max_c = c.max()
    shifted = beta * (c - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    f = max_c + np.log(Z) / beta
    p = exp_s / Z
    grad = fast_gradient(p, a)
    return f, grad, c, p


def adam_smoothed(a0, time_limit, beta_init=5.0, beta_max=100.0, lr=0.005,
                  beta_growth=1.02, restart_every=200):
    """
    Adam optimizer on the log-sum-exp smoothed objective.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    beta = beta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        f, grad, c, p = logsumexp_obj_grad(a, beta)
        obj = c.max()

        # Adam update
        lr_t = lr * np.sqrt(1 - 0.999**t) / (1 - 0.9**t)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad**2
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        step = lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Try a few step sizes with backtracking
        best_candidate = a
        best_val = obj
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a - scale * step)
            c_cand = fast_convolve(candidate)
            val = c_cand.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Increase beta over time (make smoothing sharper)
        if no_improve > 20:
            beta = min(beta_max, beta * beta_growth)

        # Restart with perturbation if stuck
        if no_improve > restart_every:
            rng = np.random.default_rng(t * 31 + 7)
            noise_scale = 0.003 + 0.005 * min(1.0, (no_improve - restart_every) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            beta = beta_init
            no_improve = 0
            t = 0

    return best_a, best_obj


def spectral_shape(a0, time_limit, n_iter_max=5000):

    Spectral shaping: iteratively flatten the power spectrum to reduce
    the peak of the auto-convolution.
    
    Key idea: the auto-conv c = IFFT(|FFT(a)|^2). To reduce peak(c),
    we reduce the "excess" power spectrum at frequencies that contribute
    positively to the peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    N = 1 << (3 * n - 1).bit_length()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    alpha = 0.1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Compute current spectrum
        A = np.fft.rfft(a, N)
        P = np.abs(A)**2
        phase = np.angle(A)
        
        # Compute auto-conv
        c = np.fft.irfft(P, N)[:2*n-1]
        obj = c.max()
        
        # Compute "excess" above target
        target = obj * 0.99  # aim to reduce peak by 1%
        excess = np.maximum(c - target, 0)
        
        # Compute gradient of sum(excess) w.r.t. P
        # d/dP[m] sum_k excess[k] = FFT(excess)[m] / N (approximately)
        # More precisely: c[k] = (1/N) sum_m P[m] exp(2πimk/N)
        # So dc[k]/dP[m] = exp(2πimk/N)/N
        # d(sum_k excess[k])/dP[m] = sum_k [excess[k] > 0] * exp(2πimk/N) / N
        # = IFFT of (excess simplified) ... 
        # Actually: = FFT_padded(excess)[m] / N where excess is padded to length N
        excess_pad = np.zeros(N)
        excess_pad[:len(excess)] = (excess > 0).astype(np.float64)
        grad_P = np.fft.rfft(excess_pad) / N
        
        # Update P (reduce components that contribute to peaks)
        P_new = P - alpha * np.abs(grad_P)
        P_new = np.maximum(P_new, 0)
        
        # Ensure P is Hermitian-symmetric (it already is since we use rfft)
        
        # Reconstruct a from modified spectrum
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-16:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Reduce step size and try harder
            alpha *= 0.95
            if no_improve > 100:
                alpha = 0.1
                # Perturb
                rng = np.random.default_rng(it * 17 + 3)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                no_improve = 0
        
        # Periodically reset alpha
        if it % 500 == 0:
            alpha = 0.1
    
    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, max_lp_time=15):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        c_obj = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': max_lp_time})
            if not res.success:
                no_improve += 1
                if no_improve > 20:
                    # Perturb and restart
                    rng = np.random.default_rng(it * 7919)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                    conv = fast_convolve(f)
                    M = conv.max()
                    no_improve = 0
                continue
            g = np.maximum(res.x, 0.0)
        except Exception:
            no_improve += 1
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def active_set_refine(a0, time_limit, tol=1e-10):
    """
    Active-set Newton refinement.
    At the optimum, the active peaks (those at the max) should have
    zero gradient contribution (KKT condition).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < 5000:
        it += 1

        c = fast_convolve(a)
        M = c.max()
        support = np.where(a > 1e-10)[0]
        if len(support) == 0:
            break

        # Active peaks: within 0.1% of max
        active = np.where(c >= M * (1 - 0.001))[0]
        if len(active) == 0:
            break

        # Equal-weight subgradient from active peaks
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)
        grad = fast_gradient(p, a)

        # Gradient step with backtracking
        best_candidate = a
        best_val = M
        for step in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            candidate = project_simplex(a - step * grad)
            c_cand = fast_convolve(candidate)
            val = c_cand.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
        elif abs(M - best_val) < tol:
            break

    return best_a, best_obj


def cosine_modulated_init(n, harmonics=5, seed=42):
    """Create initial sequences with cosine modulation (better than uniform)."""
    rng = np.random.default_rng(seed)
    sequences = []
    
    # Uniform
    sequences.append(np.ones(n) / n)
    
    # Cosine-modulated
    x = np.arange(n)
    for h in range(1, harmonics + 1):
        for amp in [0.1, 0.3, 0.5]:
            a = 1.0 + amp * np.cos(2 * np.pi * h * x / n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            sequences.append(a)
    
    # Random with spectral shaping
    for _ in range(3):
        a = rng.random(n)
        a = a / a.sum()
        sequences.append(a)
    
    # Gaussian-shaped
    center = n / 2
    for sigma in [n/4, n/3, n/2, n/5]:
        a = np.exp(-0.5 * ((x - center) / sigma)**2)
        a = a / a.sum()
        sequences.append(a)
    
    # Hann window
    a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    a = a / a.sum()
    sequences.append(a)
    
    # Raised cosine variants
    for beta in [0.3, 0.5, 0.7]:
        a = np.sinc(x/(n*beta) - 1/(2*beta))  # not quite right, let me use a different approach
        # Actually use a B-spline like shape
        a = np.maximum(1 - np.abs(2*x/n - 1)**(1/beta), 0)
        a = a / a.sum()
        sequences.append(a)
    
    return sequences


def try_and_update(a, best_state, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s
