def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_nonneg_sum1(v):
    """Project onto {x >= 0, sum(x) = 1} (simplex projection)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k=0..2n-2, g >= 0."""
    n = len(f)
    num_constraints = 2 * n - 1
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    # Constraint matrix: (f*g)[k] = sum_j f_{k-j} * g_j for valid j
    # For constraint k: sum_j g_j * f[k-j] <= M (where f[k-j] is 0 if k-j not in [0,n-1])
    # Build as sparse-like structure
    A_ub = np.zeros((num_constraints, n))
    for j in range(n):
        # g_j contributes to constraints k where k-j is in [0, n-1], i.e., j <= k <= j+n-1
        # constraint k has coefficient f[k-j] for g_j
        k_start = j
        k_end = j + n - 1
        A_ub[k_start:k_end+1, j] = f
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': 60, 'primal_feasibility_tolerance': 1e-10})
        if res.success:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def kkt_active_set_solve(a0, time_limit, verbose=True):
    """
    Direct KKT-based optimization.
    At optimum: for j in support S, sum_{k in P} mu_k * a[k-j] = nu
    and (a*a)[k] = M for k in active peaks P.
    We use this to construct a better direction.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < 5000:
        it += 1
        if it % 200 == 0 and verbose:
            print(f"    KKT it={it}, obj={best_obj:.10f}")
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find active peaks (within tolerance of max)
        tol = max(1e-8, M * 1e-6)
        active_peaks = np.where(conv >= M - tol)[0]
        P = len(active_peaks)
        
        if P == 0:
            break
        
        # Support of a
        support = np.where(a > 1e-8)[0]
        S = len(support)
        
        if S == 0:
            break
        
        # Build the system: for j in support, sum_{k in active} mu_k * a[k-j] = nu
        # This is a homogeneous system A * [mu, -nu] = 0
        # where A[i, k_idx] = a[active_peaks[k_idx] - support[i]] (if valid index)
        
        A_mat = np.zeros((S, P + 1))
        for i, j in enumerate(support):
            A_mat[i, P] = -1.0  # coefficient of nu
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_mat[i, k_idx] = a[idx]
        
        # Find null space (smallest singular vector)
        try:
            try:
                U, s_vals, Vt = np.linalg.svd(A_mat)
            except np.linalg.LinAlgError:
                break
            sol = Vt[-1, :]
            mu = np.maximum(sol[:P], 0)
            nu = -sol[P]
        except:
            break
        
        if mu.sum() < 1e-15:
            # Try alternative: just use the peak at mode
            peak = int(np.argmax(conv))
            grad = np.zeros(n)
            js = max(0, peak - n + 1)
            je = min(n - 1, peak)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] = a[peak - idx_arr]
            # Move away from peak
            for step in [0.001, 0.005, 0.01, 0.05]:
                candidate = project_nonneg_sum1(a - step * grad)
                val = fast_convolve(candidate).max()
                if val < M - 1e-14:
                    a = candidate
                    if val < best_obj:
                        best_obj = val
                        best_a = a.copy()
                    break
            continue
        
        mu = mu / mu.sum()
        
        # Gradient direction: -sum_k mu_k * a[k-j] for j in support
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu[k_idx] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= mu[k_idx] * a[k - idx_arr]
        
        # Line search
        best_candidate = a
        best_val = M
        for step in [0.0005, 0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:

            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate

        else:

def pgd_subgradient(a0, time_limit, top_k=20, lr_init=0.02, verbose=False):

        pv = conv[peaks]

            js = max(0, k - n + 1)

            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]
        
        v = mom * v - lr * sg
        
        best_candidate = a
        best_val = M
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:

            noise = 0.003 + 0.003 * min(1.0, (no_improve - 200) / 200.0)
            a = project_nonneg_sum1(best_a + rng.standard_normal(n) * noise)

            mom = 0.9

    return best_a, best_obj

def softmax_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.01, verbose=False):
    """Smoothed gradient using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init
    v = np.zeros(n)
    mom = 0.92
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient: d/d(a_j) [sum_k p_k * (a*a)_k] ... simplified
        # Actually gradient of sum_k p_k * (a*a)_k with respect to a_j
        # = sum_k p_k * (a[k-j] + a[k-j] (if k-j != k-j, which is always true))
        # = 2 * sum_k p_k * a[k-j] (for the symmetric case)
        # More precisely: d(a*a)_k / da_j = a[k-j] + a[k-j] = 2*a[k-j] (when j != k-j)
        # or a[k-j] when j = k-j
        # For simplicity use: grad_j = sum_k p_k * (a[k-j] if valid) * 2
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            # For each j in [js, je], the contribution is p[k] * a[k-j]
            # But also we need to consider the "other side" of the convolution
            # (a*a)[k] = sum_j a[j]*a[k-j]

def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

import numpy as np

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    num_constraints = 2 * n - 1
    c = -np.ones(n)
    A_ub = np.zeros((num_constraints, n))
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        A_ub[k_start:k_end + 1, j] = f
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': 120,
                               'primal_feasibility_tolerance': 1e-12,
                               'dual_feasibility_tolerance': 1e-12})
        if res.success:
            g = np.maximum(res.x, 0.0)

    return None

            peak = int(np.argmax(conv))
            grad = np.zeros(n)
            js = max(0, peak - n + 1)
            je = min(n - 1, peak)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] = a[peak - idx_arr]
            for step in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.03]:
                candidate = project_simplex(a - step * grad)
                val = fast_convolve(candidate).max()
                if val < M - 1e-14:
                    a = candidate
                    if val < best_obj:
                        best_obj = val
                        best_a = a.copy()
                    break
            else:
                if it % 200 == 0:
                    rng = np.random.default_rng(it * 7 + 3)
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            continue

        # Build KKT system
        A_mat = np.zeros((S, P + 1))
        for i, j in enumerate(support):
            A_mat[i, P] = -1.0
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_mat[i, k_idx] = a[idx]

        try:
            U, s_vals, Vt = np.linalg.svd(A_mat)
            sol = Vt[-1, :]
            mu = np.maximum(sol[:P], 0)
        except:
            break

        if mu.sum() < 1e-15:
            # Fallback: weighted gradient from peaks
            pv = conv[active_peaks]
            mu = np.maximum(pv - M * 0.999, 0)
            if mu.sum() < 1e-15:
                mu = np.ones(P) / P

        mu = mu / mu.sum()

        # Compute gradient direction
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu[k_idx] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= mu[k_idx] * a[k - idx_arr]

        # Line search
        improved = False
        for step in [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.03, 0.08]:
            candidate = project_simplex(a - step * grad)
            val = fast_convolve(candidate).max()
            if val < M - 1e-15:
                a = candidate
                if val < best_obj:
                    best_obj = val
                    best_a = a.copy()
                improved = True
                break

        if not improved and it % 150 == 0:
            rng = np.random.default_rng(it * 13 + 5)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.0015)

    return best_a, best_obj


def pgd_optimize(a0, time_limit, top_k=20, lr_init=0.015, verbose=False):
    """Projected subgradient with momentum and adaptivity."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        M = conv.max()

        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 0.8
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]

        v = mom * v - lr * sg

        best_candidate = a
        best_val = M
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.92

        if no_improve > 40:
            lr *= 0.93
            mom = min(0.97, mom + 0.002)
        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = 0.9
            no_improve = 0

    return best_a, best_obj


def softmax_grad(a0, time_limit, T_init=0.001, lr_init=0.008, verbose=False):
    """Smoothed softmax-weighted gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init
    v = np.zeros(n)
    mom = 0.93
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient computation: d/da_j of sum_k p_k * (a*a)_k
        # = sum_k p_k * 2*a[k-j] (approx, ignoring diagonal)
        grad = np.zeros(n)
        for k in range(len(p)):
            if p[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += 2.0 * p[k] * a[k - idx_arr]

        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.93

        if no_improve > 25:
            T *= 1.06
            lr *= 0.95
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def interp_sequence(parent, n_target):
    """Interpolate parent to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def try_and_update(a, best_seq, best_score, label=""):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search:
    - Enhanced FW-LP with tight LP tolerances
    - KKT active-set refinement
    - Multi-n exploration
    - Diverse perturbations and restarts
    - Peak-focused local optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.10f}")

    # Symmetrize
    sym_parent = (parent + parent[::-1]) / 2.0
    sym_parent = sym_parent / sym_parent.sum()
    best_seq, best_score = try_and_update(sym_parent, best_seq, best_score, "sym_init")

    n_best = len(best_seq)

    # === Phase 1: Quick FW-LP on starting point (~100s) ===
    print("\n=== Phase 1: Initial FW-LP ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic()
    t_alloc = min(remaining * 0.08, 100)
    a_opt, M_opt = fw_lp_optimize(best_seq, time_limit=t_alloc, verbose=True)
    best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "fw_init")
    n_best = len(best_seq)
    print(f"  Phase 1 done ({time.monotonic() - t_start:.0f}s), best={best_score:.10f}")

    # === Phase 2: Multi-n exploration (~180s) ===
    print("\n=== Phase 2: Multi-n Exploration ===")
    t_start = time.monotonic()

    candidate_ns = sorted(set([n_best, max(256, n_best // 2), min(2048, n_best * 2),
                               512, 640, 768, 896, 1024, 1280]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 150:
            break

        if n_try == n_best and best_seq is not None:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        t_alloc = min(remaining * 0.07, 50)
        print(f"  n={n_try}, t={t_alloc:.0f}s")

        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc * 0.7, verbose=False)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"fw_n{n_try}")

        remaining = deadline - time.monotonic()
        if remaining > 150 and n_try == len(best_seq):
            a_opt2, _ = pgd_optimize(a_opt, time_limit=min(remaining * 0.03, 15), top_k=15)
            best_seq, best_score = try_and_update(a_opt2, best_seq, best_score, f"pgd_n{n_try}")

    n_best = len(best_seq)
    print(f"  Phase 2 done ({time.monotonic() - t_start:.0f}s), best={best_score:.10f}")

    # === Phase 3: Deep FW-LP on best configuration (~350s) ===
    print("\n=== Phase 3: Deep FW-LP ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic()
    t_alloc = min(remaining * 0.22, 350)
    print(f"  Deep FW-LP n={n_best}, t={t_alloc:.0f}s")
    a_opt, M_opt = fw_lp_optimize(best_seq, time_limit=t_alloc, verbose=True)
    best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "deep_fw")
    n_best = len(best_seq)
    print(f"  Phase 3 done ({time.monotonic() - t_start:.0f}s), best={best_score:.10f}")

    # === Phase 4: KKT refinement (~150s) ===
    print("\n=== Phase 4: KKT Refinement ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic()
    t_alloc = min(remaining * 0.1, 150)
    a_opt, _ = kkt_refine(best_seq, time_limit=t_alloc, verbose=True)
    best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "kkt_1")
    print(f"  Phase 4 done ({time.monotonic() - t_start:.0f}s), best={best_score:.10f}")

    # === Phase 5: Smoothed gradient + PGD (~120s) ===
    print("\n=== Phase 5: Smoothed Grad + PGD ===")
    t_start = time.monotonic()

    for T_init in [0.0005, 0.002, 0.004]:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        t_alloc = min(remaining * 0.06, 40)
        a_opt, _ = softmax_grad(best_seq, time_limit=t_alloc, T_init=T_init, lr_init=0.006)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"sg_T{T_int}" if False else f"sg_{T_init}")

    for top_k, lr_init in [(10, 0.004), (25, 0.003), (50, 0.002)]:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        t_alloc = min(remaining * 0.04, 25)
        a_opt, _ = pgd_optimize(best_seq, time_limit=t_alloc, top_k=top_k, lr_init=lr_init)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"pgd_k{top_k}")

    n_best = len(best_seq)
    print(f"  Phase 5 done ({time.monotonic() - t_start:.0f}s), best={best_score:.10f}")

    # === Phase 6: Perturbed FW-LP restarts (~200s) ===
    print("\n=== Phase 6: Perturbed FW-LP ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    for scale in [0.002, 0.005, 0.01, 0.02, 0.05]:
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining * 0.08, 50)
        a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"fw_p{scale}")

    # Also try KKT on perturbed
    remaining = deadline - time.monotonic()
    if remaining > 40:
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
        t_alloc = min(remaining * 0.1, 40)
        a_opt, _ = kkt_refine(f_pert, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "kkt_pert")

    n_best = len(best_seq)
    print(f"  Phase 6 done ({time.monotonic() - t_start:.0f}s), best={best_score:.10f}")

    # === Phase 7: Second deep FW-LP + KKT (~200s) ===
    print("\n=== Phase 7: Second Deep Pass ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_alloc = min(remaining * 0.25, 200)
        print(f"  Second deep FW-LP, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "deep_fw2")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.15, 60)
        a_opt, _ = kkt_refine(best_seq, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "kkt_2")

    n_best = len(best_seq)
    print(f"  Phase 7 done ({time.monotonic() - t_start:.0f}s), best={best_score:.10f}")

    # === Phase 8: Final polishing with tiny perturbations (~80s) ===
    print("\n=== Phase 8: Final Polish ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    for i in range(5):
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        noise = rng.standard_normal(n_best) * 0.0008
        f_pert = project_simplex(best_seq + noise)
        t_alloc = min(remaining * 0.12, 12)
        a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, f"polish{i}")

    # KKT final
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_opt, _ = kkt_refine(best_seq, time_limit=min(remaining * 0.4, 20), verbose=False)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "final_kkt")

    # Symmetry check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    best_seq, best_score = try_and_update(sym_final, best_seq, best_score, "final_sym")

    print(f"  Phase 8 done ({time.monotonic() - t_start:.0f}s), best={best_score:.10f}")

    # === Squeeze remaining time ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"\n=== Squeeze: {remaining:.0f}s ===")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=remaining - 5, verbose=True)
        best_seq, best_score = try_and_update(a_opt, best_seq, best_score, "squeeze")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
