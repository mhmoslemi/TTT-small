"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M, get_dual=False):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0.
    Returns (g, dual_weights) if get_dual=True, else just g."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True,
                               'dual_feasibility_tolerance': 1e-11,
                               'primal_feasibility_tolerance': 1e-11,
                               'dual_stopping_tolerance': 1e-11})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if get_dual and hasattr(res, 'ineqlin') and res.ineqlin is not None:
                duals = np.abs(res.ineqlin.marginals) if res.ineqlin.marginals is not None else np.zeros(2*n-1)
                return g, duals
            return g
    except Exception:
        pass
    return None if not get_dual else (None, None)


def convex_line_search(f, g, M_current):
    """Find optimal alpha in [0,1] minimizing max(conv((1-a)f+ag, (1-a)f+ag)).
    Uses bisection on the derivative since the function is convex in alpha."""
    n = len(f)
    fg = fast_cross_convolve(f, g)
    ff = fast_convolve(f)
    gg = fast_convolve(g)

    def obj_at_alpha(alpha):
        fa = (1.0 - alpha) * f + alpha * g
        if np.any(fa < -1e-15):
            return np.inf
        fa = np.maximum(fa, 0.0)
        s = fa.sum()
        if s < 1e-15:
            return np.inf
        fa = fa / s
        return fast_convolve(fa).max()

    # Golden section search for the 1D convex function
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at_alpha(c)
    fd = obj_at_alpha(d)

    best_alpha = c if fc < fd else d
    best_val = min(fc, fd)

    for _ in range(25):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at_alpha(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at_alpha(d)
        cur_best = c if fc < fd else d
        cur_val = min(fc, fd)
        if cur_val < best_val:
            best_val = cur_val
            best_alpha = cur_best

    return best_alpha, best_val

                    m_relax = 1.001

        g = g / g.sum()

            stagnant += 1
            if stagnant > 40:

                f = f / s

            m_relax = 1.001

def subgradient_optimize(a0, time_limit, T_init=0.0005, lr_init=0.02, top_k_frac=0.05, use_lp_dual=True):

    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9

    stagnant = 0
    it = 0
    y_prev = np.ones(2 * n - 1) / (2 * n - 1)  # Initialize with uniform weights

        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:

            # Periodically solve LP to update dual weights

            if result[0] is not None and result[1] is not None:
                duals = np.abs(result[1])
                if duals.sum() > 1e-15:
                    y_prev = duals / duals.sum()

            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        pv_max = pv.max()
        w_softmax = np.exp((pv - pv_max) / (T * 0.1))
        w_dual = y_prev[peaks]
        # Normalize both
        if w_softmax.sum() > 1e-30:
            w_softmax = w_softmax / w_softmax.sum()
        else:
            w_softmax = np.ones(len(peaks)) / len(peaks)
        if w_dual.sum() > 1e-30:
            w_dual = w_dual / w_dual.sum()
        else:
            w_dual = np.ones(len(peaks)) / len(peaks)
        # Blend: 50% softmax, 50% dual
        w = 0.5 * w_softmax + 0.5 * w_dual
        w = w / w.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            # d(conv[k])/d(a[j]) = a[k-j] + a[j'] where j' = k-j, but careful with bounds
            # grad[j] += w[i] * (a[k-j] if k-j in [0,n-1] else 0)
            # Actually for auto-conv: d/d(a[j]) sum_j' a[j']a[k-j'] = a[k-j] + a[k-j] = 2a[k-j] (if j=k/2, just a[j])
            # More carefully: conv[k] = sum_{j'} a[j'] a[k-j']
            # d/d(a[j]) = a[k-j] (from j'=j term) + a[j] * (1 if k-j is in range, contributes from j'=k-j term)
            # = a[k-j] + a[k-j] if j != k-j, or a[j] if j == k-j
            # = a[k-j] (the k-j index, if valid) + a[k-j] (from the other factor)
            # Wait: conv[k] = sum_{j'} a[j'] * a[k-j']
            # d/d(a[j]): j' = j gives a[k-j]; j' = k-j gives a[j] * delta = a[k-j] if k-j is valid
            # So d/d(a[j]) conv[k] = a[k-j] + a[k-j] = 2*a[k-j] if k-j is in [0,n-1] and j != k-j
            # If j = k-j (i.e., 2j=k), then it's just a[j] (the self-term)
            # Actually more carefully: if 2j = k, the term is a[j]^2, derivative is 2a[j]
            # If 2j != k and k-j in [0,n-1], derivative is 2*a[k-j]
            # If k-j not in [0,n-1], derivative is 0
            kj = k - idx_arr
            valid = (kj >= 0) & (kj < n)
            contrib = np.zeros(len(idx_arr))
            contrib[valid] = a[kj[valid]]
            # For the self term (2*idx = k), the contribution is a[idx] not 2*a[k-idx]
            # But since k-idx = idx when 2*idx=k, it's the same: a[k-idx] = a[idx]
            # So actually the formula 2*a[k-j] works for all j where k-j is valid
            # Wait no. Let me redo:
            # conv[k] = sum_{j'} a[j'] * a[k-j']
            # Let's write it as: for each j', the pair (j', k-j') contributes a[j']*a[k-j']
            # d/d(a[j]) of this sum:
            #   - j' = j: gives a[k-j] (from the second factor, derivative of a[k-j'] w.r.t. a[j] is 1 when j'=j means we differentiate a[k-j] w.r.t. a[j], which is 0 unless k-j=j, i.e., j=k/2)
            #   
            # I'm overcomplicating this. Let me use the standard result:
            # If h[k] = (f*f)[k] = sum_j f[j]f[k-j], then d(h[k])/d(f[i]) = f[k-i] + f[i-(k-i)] ... 
            # 
            # Actually the simplest way: h = f * f (convolution). 
            # d(h[k])/d(f[i]) = f[k-i] (treating the two f's as independent, then adding both contributions)
            # = f[k-i] + f[k-i] = 2*f[k-i] if we account for the fact that both copies of f depend on f[i]
            # But wait: h[k] = sum_j f[j] * f[k-j]
            # d/d(f[i]) h[k] = f[k-i] (from j=i in first factor) + [k-i if k-i is in [0,n-1]] * f[i] (from j=k-i in second factor)
            # = f[k-i] + f[i] * 1_{0 <= k-i <= n-1}
            # 
            # Hmm, this is only correct if i and k-i are different contributions.
            # Let me just use: d/d(f[i]) sum_j f[j]*f[k-j] = f[k-i] + (f[i] if 0<=k-i<n else 0) - correction for double counting
            # 
            # Actually the cleanest: write h[k] = sum_j f[j] g[j] where g[j] = f[k-j].
            # d/d(f[i]) = g[i] + sum_j (dg[j]/df[i]) * f[j]
            # g[i] = f[k-i]
            # dg[j]/df[i] = 1 if k-j = i, i.e., j = k-i
            # So d/d(f[i]) h[k] = f[k-i] + f[k-i] = 2*f[k-i] (if k-i is in range)
            # But this double-counts when j = k-i = i (i.e., 2i = k), giving 2*f[i] which is correct for d/d(f[i]) f[i]^2 = 2f[i].
            # 
            # So: d/d(f[i]) conv[k] = 2 * f[k-i] if 0 <= k-i < n, else 0.
            # But we also need: if k-i is not in [0,n-1], the contribution is 0.
            # 
            # Wait, I need to be more careful. conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[j]*f[k-j]
            # d/d(f[i]): i appears as f[j] when j=i (if valid) and as f[k-j] when k-j=i, i.e., j=k-i (if valid)
            # Contribution from j=i: f[k-i] (if 0<=k-i<n)
            # Contribution from j=k-i: f[i] (if 0<=k-i<n)
            # Total: if 0<=k-i<n: f[k-i] + f[i]... wait that's not right either.
            #
            # OK let me just be very explicit. conv[k] = sum_j f[j]*f[k-j] where both indices must be in [0,n-1].
            # Term involving f[i]:
            #   - f[j]*f[k-j] where j=i: gives f[i]*f[k-i] (if k-i in [0,n-1])
            #   - f[j]*f[k-j] where k-j=i, i.e., j=k-i: gives f[k-i]*f[i] (if k-i in [0,n-1])
            # If i != k-i (i.e., 2i != k): these are two different terms, each contributing f[i]*f[k-i]
            #   d/d(f[i]) of f[i]*f[k-i] = f[k-i] (from first) + f[i] from second... NO.
            #   The first term f[i]*f[k-i]: d/d(f[i]) = f[k-i]
            #   The second term f[k-i]*f[i]: d/d(f[i]) = f[k-i]
            #   Total: 2*f[k-i]
            # If i = k-i (2i=k): there's only one term f[i]*f[i] = f[i]^2
            #   d/d(f[i]) = 2*f[i] = 2*f[k-i] (since k-i=i)
            # So in all cases: d/d(f[i]) conv[k] = 2*f[k-i] if 0<=k-i<n, else 0.
            # 
            # Great! So my formula was right all along.
            grad[idx_arr[valid]] += w[i] * 2.0 * a[kj[valid]]

        # Momentum update
        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()

        # Backtracking line search
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Adapt hyperparameters
        if stagnant > 15:
            T *= 1.08
            lr *= 0.97
        if stagnant > 50:
            T *= 0.5
            lr *= 1.5
            mom = 0.93
        if stagnant > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.003 + 0.005 * min(1.0, (stagnant - 150) / 150.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            mom = 0.9
            stagnant = 0

    return best_a, best_obj


def peak_shave_coordinate_descent(a0, time_limit):
    """
    Peak-shaving: for each peak, identify contributing pairs and transfer
    mass from high-contribution pairs to low-contribution positions.
    Uses subgradient to identify "safe" target positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_scale = 0.003
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        # Get top peaks
        n_peaks = min(6, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        # Compute subgradient at the max peak to find "safe" positions
        k_max = int(peak_indices[0])
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        # Subgradient: 2*a[k-j] for j in [js, je]
        kj = k_max - np.arange(js, je + 1)
        subgrad = np.zeros(n)
        subgrad[js:je+1] = 2.0 * a[kj]
        # Safe positions are those with LOW subgradient (not contributing much to the peak)
        safe_threshold = np.percentile(subgrad, 25)
        safe_positions = np.where(subgrad <= safe_threshold)[0]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if js > je:
                continue

            # Find top contributing pairs for this peak
            contributions = []
            for j in range(js, je + 1):
                c = a[j] * a[k - j]
                contributions.append((c, j, k - j))
            contributions.sort(reverse=True)

            # Try to reduce top contributions
            for (_, j1, j2) in contributions[:8]:
                if a[j1] < 1e-8 and a[j2] < 1e-8:
                    continue

                # Identify the "excess" coordinate (the one with larger value)
                src = j1 if a[j1] >= a[j2] else j2
                if a[src] < 1e-8:
                    continue

                # Target: pick from safe positions
                if len(safe_positions) > 0:
                    # Weight by inverse a value (prefer positions with less mass)
                    target_weights = 1.0 / (a[safe_positions] + 1e-6)
                    target_weights = target_weights / target_weights.sum()
                    target_choices = rng.choice(safe_positions, size=min(5, len(safe_positions)), replace=False, p=target_weights)
                else:
                    target_choices = rng.choice(n, size=min(5, n), replace=False)

                for j_target in target_choices:
                    if j_target == src:
                        continue
                    for delta_frac in [step_scale, step_scale * 0.5, step_scale * 2.0]:
                        delta = a[src] * delta_frac
                        candidate = a.copy()
                        candidate[src] -= delta
                        candidate[j_target] += delta
                        if candidate[src] < -1e-12:
                            continue
                        candidate = np.maximum(candidate, 0)
                        cs = candidate.sum()
                        if cs < 1e-15:
                            continue
                        candidate = candidate / cs
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-15:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            step_scale *= 0.95
            if step_scale < 1e-7:
                break
            # Periodic perturbation to escape local minima
            if it % 100 == 0:
                noise_scale = max(step_scale * 3, 0.001)
                candidate = project_simplex(a + rng.standard_normal(n) * noise_scale)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 3000:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using spline-like interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []

    # Parent (resampled)
    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper (edges higher)
    x = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
    w = 0.5 - w  # Invert: higher at edges
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s > 1e-15:
        starts.append(w / s)

    # Cosine taper (center higher)
    w = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1))
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.25 * n)) ** 2)
    starts.append(g / g.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.06 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.06 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Alternating perturbation of uniform
    alt = np.ones(n) + 0.3 * np.cos(2 * np.pi * x / n)
    alt = np.maximum(alt, 0.01)
    starts.append(alt / alt.sum())

    # Fourier-based: uniform + small harmonic
    for m in [2, 3]:
        four = np.ones(n) + 0.15 * np.cos(2 * np.pi * m * x / n + 0.5)
        four = np.maximum(four, 0.01)
        starts.append(four / four.sum())

    # Random Dirichlet (concentrated)
    for _ in range(2):
        starts.append(rng.dirichlet(np.ones(n) * 0.5))

    # "Ramp" shape
    ramp = np.linspace(0.1, 1.0, n)
    starts.append(ramp / ramp.sum())

    # V-shape (high at edges, low in middle)
    vshape = np.minimum(x, n - 1 - x) / (n / 2) + 0.3
    starts.append(vshape / vshape.sum())

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Quick multi-n, multi-start exploration with FW (200s)
    2. Deep FW on best candidate with improved line search (500s)
    3. LP-dual subgradient refinement (350s)
    4. Peak-shaving coordinate descent (350s)
    5. Final polish with interleaved methods (remaining time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Quick exploration (200s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()

    # Candidate lengths - include the parent length and several others
    lengths = sorted(set([n_parent, max(256, n_parent - 128), n_parent + 128,
                          384, 512, 640, 768, 896, 1024, 1280]))
    lengths = [l for l in lengths if 192 <= l <= 2048]

    for n_try in lengths:
        remaining = deadline - time.monotonic() - 800
        if remaining < 15:
            break

        starts = generate_starts(n_try, parent, rng)

        # Pick best 3-4 starts by initial objective
        scored_starts = []
        for start in starts:
            val = fast_convolve(start).max()
            scored_starts.append((val, start))
        scored_starts.sort()
        top_starts = [s for _, s in scored_starts[:4]]

        for si, start in enumerate(top_starts):
            remaining = deadline - time.monotonic() - 750
            if remaining < 10:
                break
            t_alloc = min(remaining, 25)
            a_opt, _ = fw_optimize(start, time_limit=t_alloc)
            label = f"fw_n{n_try}_s{si}"
            try_seq(a_opt, label)

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW (500s) ===
    print("\n=== Phase 2: Deep FW ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Main deep FW
        remaining = deadline - time.monotonic() - 450
        if remaining > 30:
            t_alloc = min(remaining, 400)
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts with varied scales
        for scale in [0.0005, 0.002, 0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            print(f"  FW perturb scale={scale}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

        # Try a different n near the best
        for n_alt in [n_best + 64, n_best - 64, n_best + 128]:
            if n_alt < 128 or n_alt > 2048:
                continue
            remaining = deadline - time.monotonic() - 200
            if remaining < 20:
                break
            a_resampled = resample_sequence(best_seq, n_alt)
            t_alloc = min(remaining, 40)
            a_opt, _ = fw_optimize(a_resampled, time_limit=t_alloc)
            try_seq(a_opt, f"fw_n{n_alt}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Subgradient with LP dual (350s) ===
    print("\n=== Phase 3: LP-Dual Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {'T_init': 0.0003, 'lr_init': 0.025, 'top_k_frac': 0.03, 'use_lp_dual': True},
            {'T_init': 0.001, 'lr_init': 0.015, 'top_k_frac': 0.05, 'use_lp_dual': True},
            {'T_init': 0.002, 'lr_init': 0.01, 'top_k_frac': 0.08, 'use_lp_dual': False},
            {'T_init': 0.0005, 'lr_init': 0.02, 'top_k_frac': 0.04, 'use_lp_dual': True},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 200
            if remaining < 20:
                break
            t_alloc = min(remaining, 75)
            print(f"  Subgrad config {ci} (T={cfg['T_init']}, dual={cfg['use_lp_dual']}), t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"subgrad_{ci}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak-shaving coordinate descent (350s) ===
    print("\n=== Phase 4: Peak-Shaving CD ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 250)
            print(f"  Peak shaving, t={t_alloc:.0f}s")
            a_opt, _ = peak_shave_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_shave")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        polish_iter = 0

        while time.monotonic() < deadline - 10:
            polish_iter += 1
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            if polish_iter % 4 == 1:
                # Short FW with tiny perturbation
                noise = rng.standard_normal(n_best) * 0.0005
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 30)
                a_opt, _ = fw_optimize(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{polish_iter}")

            elif polish_iter % 4 == 2:
                # Short subgradient
                noise = rng.standard_normal(n_best) * 0.0003
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 25)
                a_opt, _ = subgradient_optimize(start, time_limit=t_alloc,
                                                T_init=0.0002, lr_init=0.008,
                                                top_k_frac=0.03, use_lp_dual=True)
                try_seq(a_opt, f"polish_sg_{polish_iter}")

            elif polish_iter % 4 == 3:
                # Short peak shaving
                t_alloc = min(remaining, 40)
                a_opt, _ = peak_shave_coordinate_descent(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_ps_{polish_iter}")

            else:
                # Direct coordinate perturbation: try random single-coordinate changes
                for _ in range(50):
                    j = int(rng.integers(0, n_best))
                    for delta in [0.0001, -0.0001, 0.00005, -0.00005]:
                        candidate = best_seq.copy()
                        candidate[j] = max(0, candidate[j] + delta)
                        # Renormalize
                        cs = candidate.sum()
                        if cs < 1e-15:
                            continue
                        candidate = candidate / cs
                        val = fast_convolve(candidate).max()
                        if val < fast_convolve(best_seq).max() - 1e-15:
                            best_seq = candidate
                            best_score = float(evaluate_sequence(best_seq.tolist()))
                            break

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
