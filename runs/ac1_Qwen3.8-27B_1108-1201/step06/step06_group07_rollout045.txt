def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):

    n = len(v)
    if n == 1:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(n):
    """Precompute sparsity pattern for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_lp_oracle(f, M, n, row_idx=None, col_idx=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    if row_idx is None or col_idx is None:
        row_idx, col_idx = build_lp_matrix(n)
    
    c = -np.ones(n)
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

Let me think about **symmetry**. The convolution `conv(a,a)[k] = conv(a,a)[2n-2-k]`. So the peaks are symmetric. The center peak (at k=n-1) is special.

For the center peak: `conv(a,a)[n-1] = sum_j a[j]*a[n-1-j]`. This is the inner product of `a` with its reversal. To minimize this while keeping `sum(a)=1` and `a>=0`, we want `a` to be "anti-symmetric" (concentrated on one side).

But the edge peaks: `conv(a,a)[0] = a[0]^2` and `conv(a,a)[2n-2] = a[n-1]^2`. If `a` is concentrated on one side, say near index 0, then `a[0]^2` is large.

So the optimal balance is: `a` should be spread out enough that no single entry is too large (to keep edge peaks small), but concentrated enough on one side that the center peak is small.

A geometric sequence `a[j] = r^j * c` with `r < 1` (concentrated near 0) has:
- `a[0] = c = (1-r)/(1-r^n)` → edge peak `c^2`
- `conv(a,a)[n-1] = c^2 * n * r^(n-1)` → center peak

For the objective to be good, we want both `c^2` and `c^2 * n * r^(n-1)` to be small. Since `c = (1-r)/(1-r^n)`, for large n and r close to 1, `c ≈ (1-r)/n * 1/(1-r) = 1/n`... wait no.

For r close to 1: `1-r^n ≈ n(1-r)`, so `c ≈ (1-r)/(n(1-r)) = 1/n`. And `r^(n-1) ≈ 1 - (n-1)(1-r)`. So center peak ≈ `c^2 * n * r^(n-1) ≈ (1/n^2) * n * (1-(n-1)(1-r)) = (1/n)(1-(n-1)(1-r))`.

For the edge peak: `c^2 ≈ 1/n^2`.

So the max is approximately `1/n` (from the center peak when r is close to 1) or `1/n^2` (from the edge, which is smaller). The objective would be `2n * 1/n = 2`.

That's the same as uniform! So geometric sequences don't help much when r is close to 1.

For r further from 1, say `r = 0.5`, `n = 1024`:

- Edge peak: `c^2 ≈ 1`
