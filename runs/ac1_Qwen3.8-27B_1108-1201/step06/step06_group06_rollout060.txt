def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max(x, T):
    """Softmax-based smooth maximum."""
    scaled = x / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w = w / w.sum()
    return np.dot(w, x), w

"""
Step function optimization minimizing 2*n*max_conv/sum^2.
Uses Nesterov-accelerated proximal gradient with FFT-based gradients,
QP subproblem solver exploiting Hankel structure, iterative peak shaving,
and multi-scale coordinate descent. Different from prior FW-LP approach.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(y, w):
    """
    Compute gradient of sum_k w[k]*(y*y)[k] w.r.t. y using FFT.
    grad[j] = 2 * sum_m y[m] * w[m+j] (cross-correlation of w and y).
    """
    n = len(y)
    conv_len = 2 * n - 1
    pad_size = 1 << (3 * n - 1).bit_length()
    W_pad = np.zeros(pad_size)
    W_pad[:conv_len] = w
    Y_pad = np.zeros(pad_size)
    Y_pad[:n] = y
    corr = np.fft.irfft(np.fft.rfft(W_pad) * np.conj(np.fft.rfft(Y_pad)), pad_size)
    return 2.0 * corr[:n]


def nesterov_grad_optimize(a0, time_limit, T_init=0.001, lr_init=0.015):

        a = np.ones(len(a)) / len(a)

        a = a / s
    n = len(a)

    T = T_init
    lr = lr_init
    no_improve = 0

    a_prev = a.copy()

            break

        beta = (beta - 1.0) / beta
        y = a + beta * (a - a_prev)

def qp_direction(y, w, n):
    """
    Compute QP direction using the Hankel Hessian structure.
    Solves: min (1/2)a^T H a + g^T a s.t. a >= 0, sum(a)=1
    using a few steps of projected gradient on the QP.
    H[i,j] = 2*w[i+j] (Hankel matrix), g = H @ y (gradient of the quadratic).
    """
    # H @ y = 2 * (w cross-correlated with y) = grad (same as before!)
    # So the QP is: min (1/2) a^T H a + (H@y)^T a s.t. simplex
    # Gradient of QP at a: H@a + H@y
    # We solve this with a few projected gradient steps

    a_qp = y.copy()
    for _ in range(3):
        # Compute H @ a_qp using FFT
        a_qp_conv = fast_convolve(a_qp)
        # H @ a_qp = 2 * cross_corr(w, a_qp)
        Ha = compute_gradient_fft(a_qp, w) / 2.0  # H@a (divide by 2 since grad was 2*H@a)
        Hy = compute_gradient_fft(y, w) / 2.0  # H@y
        qp_grad = Ha + Hy

        # Projected gradient step on QP
        a_qp = np.maximum(a_qp - 0.5 * qp_grad, 0.0)
        a_qp = a_qp / a_qp.sum()

    return a_qp

def peak_shaving(a0, time_limit):
    """
    Iterative peak shaving: identify the dominant peak lag,
    then reduce contributions from the most responsible coordinates,
    redistributing mass to non-contributing positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        # Find top peaks
        k_star = np.argmax(conv)
        peak_val = conv[k_star]

        # Responsibility: a[i] contributes a[i]*a[k_star-i] to conv[k_star]
        # (with valid indices)
        i_start = max(0, k_star - n + 1)
        i_end = min(k_star, n - 1)
        indices = np.arange(i_start, i_end + 1)
        partners = k_star - indices
        contribs = a[indices] * a[partners]
        total_contrib = contribs.sum()

        if total_contrib < 1e-15:
            break

        # Find indices with highest contribution
        # Reduce them and add to low-responsibility indices
        n_reduce = max(1, n // 50)
        top_idx = indices[np.argsort(contribs)[::-1][:n_reduce]]

        # Find positions that don't contribute to this peak
        contrib_set = set(top_idx.tolist())
        non_contrib = [i for i in range(n) if i not in contrib_set and a[i] > 0]

        if not non_contrib:
            non_contrib = list(range(n))

        # Transfer mass
        transfer_frac = 0.1
        for i in top_idx:
            amount = a[i] * transfer_frac
            a[i] -= amount
            # Distribute to random non-contributing positions
            j = non_contrib[int(rng.integers(0, len(non_contrib)))]
            a[j] += amount

        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-15:
            a = best_a.copy()
            break
        a = a / ss

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
        else:
            # Revert
            a = best_a.copy()
            # Try smaller transfer
            transfer_frac *= 0.5
            if transfer_frac < 0.001:
                # Random restart from best
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)

    return best_a, best_obj


def coordinate_descent(a0, time_limit, seed=42):
    """
    Coordinate descent: at each step, find the pair that most reduces
    the current peak and optimally transfer mass between them.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_star = np.argmax(conv)

        # Find the pair (i, j) with i+j = k_star (or i+j = k_star + offset)
        # that has the largest product a[i]*a[j]
        i_start = max(0, k_star - n + 1)
        i_end = min(k_star, n - 1)
        if i_start > i_end:
            # Try nearby lags
            for dk in range(-3, 4):
                k_try = k_star + dk
                i_start = max(0, k_try - n + 1)
                i_end = min(k_try, n - 1)
                if i_start <= i_end:
                    k_star = k_try
                    break
            else:
                break

        indices = np.arange(i_start, i_end + 1)
        partners = k_star - indices
        products = a[indices] * a[partners]

        # Pick the worst pair
        worst_local = np.argmax(products)
        i_worst = indices[worst_local]
        j_worst = partners[worst_local]

        if i_worst == j_worst:
            # Square term: reduce a[i_worst] and add to a random position
            j_target = int(rng.integers(0, n))
            for frac in [0.05, 0.1, 0.2, 0.3]:
                candidate = a.copy()
                amount = candidate[i_worst] * frac
                candidate[i_worst] -= amount
                candidate[j_target] += amount
                candidate = np.maximum(candidate, 0)
                candidate = candidate / candidate.sum()
                c = fast_convolve(candidate).max()
                if c < best_obj - 1e-15:
                    a = candidate
                    best_obj = c
                    best_a = a.copy()
                    no_improve = 0
                    break
        else:
            # Cross term: transfer from larger to smaller, or to a new position
            if a[i_worst] >= a[j_worst]:
                i_src, i_dst = i_worst, j_worst
            else:
                i_src, i_dst = j_worst, i_worst

            for frac in [0.05, 0.1, 0.2, 0.3]:
                candidate = a.copy()
                amount = candidate[i_src] * frac
                candidate[i_src] -= amount
                candidate[i_dst] += amount
                candidate = np.maximum(candidate, 0)
                candidate = candidate / candidate.sum()
                c = fast_convolve(candidate).max()
                if c < best_obj - 1e-15:
                    a = candidate
                    best_obj = c
                    best_a = a.copy()
                    no_improve = 0
                    break

            # Also try transferring to a random position
            if no_improve > 10:
                j_rand = int(rng.integers(0, n))
                for frac in [0.05, 0.1, 0.2]:
                    candidate = a.copy()
                    amount = candidate[i_src] * frac
                    candidate[i_src] -= amount
                    candidate[j_rand] += amount
                    candidate = np.maximum(candidate, 0)
                    candidate = candidate / candidate.sum()
                    c = fast_convolve(candidate).max()
                    if c < best_obj - 1e-15:
                        a = candidate
                        best_obj = c
                        best_a = a.copy()
                        no_improve = 0
                        break

        if no_improve > 200:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
        if no_improve > 1000:
            break

    return best_a, best_obj


def multi_scale_optimize(a_best, time_limit, n_base):
    """
    Multi-scale: downsample to small n, optimize, upsample back.
    This can escape local minima by exploring a coarser landscape.
    """
    n = len(a_best)
    best_a = a_best.copy()
    best_obj = fast_convolve(a_best).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    # Try several coarser scales
    for scale in [4, 2]:
        n_coarse = max(64, n_base // scale)
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        # Downsample
        coarse = np.interp(np.linspace(0, 1, n_coarse),
                           np.linspace(0, 1, n_base), a_best)
        coarse = np.maximum(coarse, 0)
        coarse = coarse / coarse.sum()

        # Optimize on coarse grid
        t_alloc = min(remaining * 0.3, 60)
        a_coarse_opt, _ = nesterov_grad_optimize(coarse, time_limit=t_alloc,
                                                  T_init=0.005, lr_init=0.05)

        # Upsample
        upsampled = np.interp(np.linspace(0, 1, n_base),
                              np.linspace(0, 1, n_coarse), a_coarse_opt)
        upsampled = np.maximum(upsampled, 0)
        upsampled = upsampled / upsampled.sum()

        # Blend with best
        for alpha in [0.3, 0.5, 0.7]:
            blend = (1 - alpha) * a_best + alpha * upsampled
            blend = project_simplex(blend)
            obj = fast_convolve(blend).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = blend.copy()

        # Continue optimization on fine grid from best blend
        remaining2 = time_limit - (time.monotonic() - t0)
        if remaining2 > 20:
            a_fine, _ = nesterov_grad_optimize(best_a, time_limit=min(remaining2 * 0.4, 40),
                                               T_init=0.001, lr_init=0.01)
            obj = fast_convolve(a_fine).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_fine.copy()

    return best_a, best_obj


def local_polish(a0, time_limit, seed=99):
    """Fine local search with multi-scale perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        improved = False

        # Multi-scale Gaussian perturbations
        for scale in [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01]:
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(a + pert)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        # Smoothed perturbations (low-pass filtered noise)
        if not improved:
            noise = rng.standard_normal(n) * 0.003
            # Low-pass filter via FFT
            fft_noise = np.fft.rfft(noise)
            freqs = np.fft.rfftfreq(n)
            cutoff = 5.0 / n
            fft_noise[freqs > cutoff] *= 0.1
            smooth_noise = np.fft.irfft(fft_noise, n)
            candidate = project_simplex(a + smooth_noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        # Pair swap
        if not improved:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.03, 0.07, 0.15, 0.3]:
                    candidate = a.copy()
                    amt = candidate[i] * frac
                    candidate[i] -= amt
                    candidate[j] += amt
                    candidate = np.maximum(candidate, 0)
                    candidate = candidate / candidate.sum()
                    c = fast_convolve(candidate).max()
                    if c < best_obj - 1e-15:
                        a = candidate
                        best_obj = c
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            # Periodic restart
            if int(time.monotonic() - t0) % 30 == 0 and int(time.monotonic() - t0) > 30:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)

        # Early termination
        if not improved and (time.monotonic() - t0) > time_limit * 0.8:
            break

    return best_a, best_obj


def try_seq(a, label, best_ref):
    """Evaluate a sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_ref
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_ref[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a_norm.copy(), score)
    return best_ref


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry: multi-phase hybrid search for minimum peak autocorrelation.
    Uses Nesterov gradient, QP directions, peak shaving, coordinate descent,
    multi-scale optimization, and local polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Quick multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = [256, 384, 512, 640, 768, 896, 1024, 1280, 1536]
    if n_parent not in candidate_ns:
        candidate_ns.append(n_parent)
    candidate_ns.sort()

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic()
        if remaining < 300:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            start = start / ss if ss > 1e-10 else np.ones(n_try) / n_try

        t_alloc = min(remaining * 0.05, 20)
        a_opt, _ = nesterov_grad_optimize(start, time_limit=t_alloc,
                                          T_init=0.002, lr_init=0.02)
        best_seq, best_score = try_seq(a_opt, f"nest_n{n_try}", (best_seq, best_score))

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep Nesterov gradient ===
    print("\n=== Phase 2: Deep Nesterov ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 100:
        t_alloc = min(remaining * 0.3, 400)
        print(f"  Nesterov n={n_best}, T=0.001, t={t_alloc:.0f}s")
        a_opt, _ = nesterov_grad_optimize(best_seq, time_limit=t_alloc,
                                          T_init=0.001, lr_init=0.012)
        best_seq, best_score = try_seq(a_opt, "nest_deep", (best_seq, best_score))

    # Restart with different T
    remaining = deadline - time.monotonic()
    if remaining > 80:
        for T_init, lr_init in [(0.003, 0.02), (0.0005, 0.008)]:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            t_alloc = min(remaining * 0.15, 100)
            a_start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
            a_opt, _ = nesterov_grad_optimize(a_start, time_limit=t_alloc,
                                              T_init=T_init, lr_init=lr_init)
            best_seq, best_score = try_seq(a_opt, f"nest_T{T_init}", (best_seq, best_score))

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: QP-based optimization ===
    print("\n=== Phase 3: QP Optimization ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 80:
        t_alloc = min(remaining * 0.25, 250)
        print(f"  QP n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = qp_optimize(best_seq, time_limit=t_alloc, T_init=0.001)
        best_seq, best_score = try_seq(a_opt, "qp_main", (best_seq, best_score))

    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_alloc = min(remaining * 0.15, 100)
        a_opt, _ = qp_optimize(best_seq, time_limit=t_alloc, T_init=0.003)
        best_seq, best_score = try_seq(a_opt, "qp_T0003", (best_seq, best_score))

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak shaving + Coordinate descent ===
    print("\n=== Phase 4: Peak Shaving + Coord Descent ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 60:
        t_alloc = min(remaining * 0.2, 100)
        print(f"  Peak shaving, t={t_alloc:.0f}s")
        a_opt, _ = peak_shaving(best_seq, time_limit=t_alloc)
        best_seq, best_score = try_seq(a_opt, "peak_shave", (best_seq, best_score))

    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_alloc = min(remaining * 0.25, 150)
        print(f"  Coordinate descent, t={t_alloc:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, seed=456)
        best_seq, best_score = try_seq(a_opt, "coord_desc", (best_seq, best_score))

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Multi-scale + alternate n ===
    print("\n=== Phase 5: Multi-scale ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 80:
        t_alloc = min(remaining * 0.2, 120)
        print(f"  Multi-scale, t={t_alloc:.0f}s")
        a_opt, _ = multi_scale_optimize(best_seq, time_limit=t_alloc, n_base=n_best)
        best_seq, best_score = try_seq(a_opt, "multiscale", (best_seq, best_score))

    # Try adjacent n values
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for n_try in [n_best + 32, n_best - 32, n_best + 64, n_best - 64]:
            if n_try < 128 or n_try > 2048:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_best), best_seq)
            start = np.maximum(start, 0)
            start = start / start.sum()
            t_alloc = min(remaining * 0.15, 40)
            a_opt, _ = nesterov_grad_optimize(start, time_limit=t_alloc,
                                              T_init=0.002, lr_init=0.015)
            best_seq, best_score = try_seq(a_opt, f"alt_n{n_try}", (best_seq, best_score))

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Fine-tuning ===
    print("\n=== Phase 6: Fine-tuning ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 40:
        t_alloc = min(remaining * 0.3, 100)
        print(f"  Fine Nesterov (T=0.0003), t={t_alloc:.0f}s")
        a_opt, _ = nesterov_grad_optimize(best_seq, time_limit=t_alloc,
                                          T_init=0.0003, lr_init=0.005)
        best_seq, best_score = try_seq(a_opt, "fine_nest", (best_seq, best_score))

    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.25, 80)
        print(f"  Local polish, t={t_alloc:.0f}s")
        a_opt, _ = local_polish(best_seq, time_limit=t_alloc, seed=77777)
        best_seq, best_score = try_seq(a_opt, "polish1", (best_seq, best_score))

    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.3, 50)
        print(f"  Fine QP (T=0.0005), t={t_alloc:.0f}s")
        a_opt, _ = qp_optimize(best_seq, time_limit=t_alloc, T_init=0.0005)
        best_seq, best_score = try_seq(a_opt, "fine_qp", (best_seq, best_score))

    print(f"  Phase 6 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 7: Final polish ===
    print("\n=== Phase 7: Final Polish ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    # Very fine local search
    if remaining > 15:
        t_alloc = min(remaining * 0.4, 50)
        a_opt, _ = local_polish(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "polish2", (best_seq, best_score))

    # Tiny Nesterov
    remaining = deadline - time.monotonic()
    if remaining > 10:
        t_alloc = min(remaining * 0.3, 30)
        a_opt, _ = nesterov_grad_optimize(best_seq, time_limit=t_alloc,
                                          T_init=0.0001, lr_init=0.002)
        best_seq, best_score = try_seq(a_opt, "ultra_fine", (best_seq, best_score))

    # Final micro-perturbations
    remaining = deadline - time.monotonic()
    if remaining > 5:
        for _ in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 2:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = nesterov_grad_optimize(f_pert, time_limit=min(remaining * 0.3, 5),
                                              T_init=0.0002, lr_init=0.003)
            best_seq, best_score = try_seq(a_opt, "micro", (best_seq, best_score))

    print(f"  Phase 7 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
