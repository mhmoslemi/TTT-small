"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines structured initialization, Frank-Wolfe with LP oracle,
log-sum-exp smoothed gradient descent, coordinate descent,
and active-set Newton polishing across multiple sequence lengths.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_u_shaped(n, alpha=0.7):
    """Create a U-shaped (bathtub) sequence normalized to sum=1."""
    j = np.arange(n)
    if n == 1:
        return np.array([1.0])
    x = j / (n - 1)
    a = 1.0 + alpha * np.cos(np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_chirp(n, alpha=1.0):
    """Create a chirp-like sequence."""
    j = np.arange(n)
    if n == 1:
        return np.array([1.0])
    x = j / (n - 1)
    a = 1.0 + alpha * np.sin(2 * np.pi * x * 2.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_multi_freq(n, freqs=(1.0, 2.0, 3.0), amps=(0.5, 0.3, 0.1)):
    """Create a multi-frequency structured sequence."""
    j = np.arange(n)
    if n == 1:
        return np.array([1.0])
    x = j / (n - 1)
    a = np.ones(n)
    for f, amp in zip(freqs, amps):
        a += amp * np.cos(2 * np.pi * f * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def hankel_product(p, a):
    """Compute M(p) @ a where p has length 2n-1 and a has length n.
    Result[j] = sum_k p[k] * a[k-j] (with valid indices only)."""
    n = len(a)
    return np.correlate(p, a, mode='valid')


def smooth_gradient(a, T):
    """Compute gradient of log-sum-exp(T * conv(a,a)) using softmax weights."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    # Gradient: sum_k p[k] * d(conv[k])/d(a[j])
    # d(conv[k])/d(a[j]) = a[k-j] + a[j-k] (if both valid)
    # = 2*a[k-j] if k=2j
    # This is hankel_product(p, a) * 2 minus diagonal correction
    grad = 2.0 * hankel_product(p, a)
    # Subtract diagonal: for k=2j, we double-counted
    diag_idx = np.arange(0, 2 * n - 1, 2)
    diag_vals = p[diag_idx] * a[:n]
    grad -= diag_vals
    return grad, conv


def lse_objective(a, T):
    """Compute log-sum-exp(T * conv(a,a)) value."""
    conv = fast_convolve(a)
    m = conv.max()
    return m + T * np.log(np.exp(conv / T - m).sum() / len(conv))


def pgd_smoothed(a0, time_limit, T_init=0.002, lr_init=0.01, mom_init=0.9):

def solve_lp_oracle_sparse(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses sparse matrix representation."""
    n = len(f)
    c = -np.ones(n)

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])

    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def active_set_polish(a0, time_limit, tol=1e-10, max_iter=500):
    """Active-set Newton polishing using KKT conditions."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        M = conv.max()
        support = np.where(a > 1e-8)[0]
        if len(support) == 0:
            break

        # Find active peaks (within 0.1% of max)
        threshold = M * (1.0 - 0.001)
        active_peaks = np.where(conv >= threshold)[0]

        if len(active_peaks) <= 1:
            break

        m = len(support)
        p = len(active_peaks)

        # Build KKT system: for j in support, sum_k mu_k * a[|k-j|] = nu
        A_sys = np.zeros((m, p + 1))
        for i, j in enumerate(support):
            A_sys[i, -1] = 1.0
            for k_idx, k in enumerate(active_peaks):
                idx1 = k - j
                idx2 = j - k
                contrib = 0.0
                if 0 <= idx1 < n:
                    contrib += a[idx1]
                if 0 <= idx2 < n and idx1 != idx2:
                    contrib += a[idx2]
                A_sys[i, k_idx] = contrib

        try:
            U, S_vals, Vt = np.linalg.svd(A_sys)
            sol = Vt[-1, :]
            mu_vals = np.maximum(sol[:p], 0)
        except np.linalg.LinAlgError:
            break

        mu_sum = mu_vals.sum()
        if mu_sum < 1e-15:
            break
        mu_vals /= mu_sum

        # Gradient step using mu-weighted peaks
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu_vals[k_idx] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= mu_vals[k_idx] * a[k - idx_arr]

        best_candidate = a
        best_val = M
        for step in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            candidate = project_simplex(a + step * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()

        if abs(M - best_val) < 1e-13:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, max_iter=10000):
    """Randomized coordinate descent: pick a coordinate, optimize it."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Pick a coordinate (bias towards edges which affect convolution more)
        j = int(rng.integers(0, n))

        # Binary search for optimal a[j]
        conv = fast_convolve(a)
        M = conv.max()

        # Try several values of a[j]
        best_val = M
        best_aj = a[j]
        # Scale around current value
        for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.3]:
            aj_new = a[j] * mult
            if aj_new < 0 or aj_new > 1.0:
                continue
            a_new = a.copy()
            a_new[j] = aj_new
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            val = fast_convolve(a_new).max()
            if val < best_val:
                best_val = val
                best_aj = aj_new

        if best_aj != a[j]:
            a[j] = best_aj
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            if best_val < best_obj:
                best_obj = best_val
                best_a = a.copy()

    return best_a, best_obj


def symmetrize(a):
    """Symmetrize a sequence."""
    return (a + a[::-1]) / 2.0


def try_and_update(a, best_state, label=""):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15 or len(a) == 0:
        return best_state
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def interp_sequence(parent, n_target):
    """Interpolate parent to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

    n_best = len(best_seq)

    # Active-set Newton
    remaining = deadline - time.monotonic() - 150
    if remaining > 20:
        t_alloc = min(remaining, 80)
        print(f"  Active-set Newton t={t_alloc:.0f}s")
        a_opt, _ = active_set_polish(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "as_newton")

    # Coordinate descent
    remaining = deadline - time.monotonic() - 100
    if remaining > 15:
        t_alloc = min(remaining, 60)
        print(f"  Coord descent t={t_alloc:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "coord")

    # Another round of smoothed gradient
    remaining = deadline - time.monotonic() - 60
    if remaining > 10:
        t_alloc = min(remaining, 40)
        a_opt, _ = pgd_smoothed(best_seq, time_limit=t_alloc, T_init=0.0003, lr_init=0.005)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "sg_fine")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Final exploitation (~200s) ===
    print("\n=== Phase 5: Final Exploitation ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Tiny perturbations + quick optimization
    for pert_scale in [0.0005, 0.001, 0.002, 0.003]:
        remaining = deadline - time.monotonic() - 30
        if remaining < 10:
            break
        a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * pert_scale)
        # Quick FW-LP
        t_alloc = min(remaining, 30)
        a_opt, _ = fw_lp_optimize(a_pert, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"pert_{pert_scale}")

        # Active-set polish
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            a_opt2, _ = active_set_polish(a_opt, time_limit=min(remaining, 15))
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"pert_as_{pert_scale}")

    # Symmetry check
    sym_final = symmetrize(best_seq).astype(np.float64)
    sym_final = sym_final / sym_final.sum()
    (best_seq, best_score) = try_and_update(sym_final, (best_seq, best_score), "final_sym")

    # One more FW-LP on the very best
    remaining = deadline - time.monotonic() - 5
    if remaining > 10:
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(remaining, 20))
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_fw")

    # Final active-set
    remaining = deadline - time.monotonic() - 2
    if remaining > 5:
        a_opt, _ = active_set_polish(best_seq, time_limit=min(remaining, 10))
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "final_as")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
