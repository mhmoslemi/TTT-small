```python
import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_matvec(w, a):
    return np.correlate(w, a, mode='valid')

def qp_projected_grad(w, a_init, n, time_limit=10.0, max_iter=5000):
    """Solve min a^T H a s.t. sum(a)=1, a>=0 using projected gradient.
    H[i,j] = w[i+j] (Hankel matrix)."""
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    t0 = time.monotonic()
    lr = 1.0
    prev_obj = fast_convolve(a, w).max()  # Actually we want a^T H a
    # Compute a^T H a:
    # a^T H a = sum_i a[i] * (H*a)[i] = sum_i a[i] * correlate(w,a,'valid')[i]
    
    def obj(a):
        return np.dot(a, hankel_matvec(w, a))
    
    best_a = a.copy()
    best_obj = obj(a)
    
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Gradient: 2*H*a
        grad = 2.0 * hankel_matvec(w, a)
        
        # Nesterov-style
        # Actually, let's just do plain projected gradient with backtracking
        # for simplicity and reliability
        
        # Backtracking line search
        step = lr
        for _ in range(20):
            a_new = project_simplex(a - step * grad)
            obj_new = obj(a_new)
            obj_curr = obj(a)
            if obj_new <= obj_curr - 1e-4 * step * np.dot(grad, a - a_new):
                break
            step *= 0.5
        
        a = a_new
        curr_obj = obj(a)
        
        if curr_obj < best_obj - 1e-14:
            best_obj = curr_obj
            best_a = a.copy()
    
    return best_a

def dual_update(conv, method='softmax', T=0.01, topk=10):
    if method == 'softmax':
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        return w / w.sum()
    elif method == 'topk':
        k = min(topk, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        w = np.zeros(len(conv))
        w[peaks] = 1.0 / k
        return w
    elif method == 'geometric':
        # Geometric mean of softmax and top-k
        w_soft = dual_update(conv, 'softmax', T)
        w_topk = dual_update(conv, 'topk', topk)
        w = np.sqrt(w_soft * w_topk) + 1e-15
        return w / w.sum()
    else:
        return np.ones(len(conv)) / len(conv)

                           topk=10, blend=0.5, verbose=False):
    """Primal-dual: alternate between dual update and primal QP solve."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()
    
    t0 = time.monotonic()
    T = T_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Dual update

        T = max(0.001, T * (0.995 if no_improve < 20 else 1.1))

        # Primal solve (time-limited)
        qp_time = min(remaining * 0.4, 5.0)
        a_qp = qp_projected_grad(w, a, n, time_limit=qp_time, max_iter=1000)
        
        # Blend
        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0.0)

        if s_new < 1e-15:

        else:
```

```python
            seg[2*p] += quad
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Projected gradient QP for primal-dual iterations (Hankel matrices)
- Coordinate descent with ternary search on 1D subproblems
- Variance minimization as smooth surrogate
- Nesterov-accelerated subgradient for peak reduction
- Multi-n exploration with adaptive time allocation
- Simulated annealing for escape from local minima
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(w, a):
    """Compute H @ a where H[i,j] = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def qp_projected_grad(w, a_init, n, time_limit=5.0, max_iter=2000):
    """
    Solve min a^T H a s.t. sum(a)=1, a>=0 using projected gradient with backtracking.
    H is Hankel: H[i,j] = w[i+j].
    """
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    def qp_obj(x):
        return np.dot(x, hankel_matvec(w, x))

    best_a = a.copy()
    best_obj = qp_obj(a)

    t0 = time.monotonic()
    lr = 1.0 / (2.0 * np.abs(hankel_matvec(w, a)).max() + 1e-10)
    lr = min(lr, 0.5)
    momentum = 0.85
    v = np.zeros(n)

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        grad = 2.0 * hankel_matvec(w, a)

        # Nesterov momentum
        v = momentum * v + grad
        # Scale gradient by momentum factor
        effective_grad = v / (1.0 + momentum) if momentum < 1.0 else v

        # Backtracking line search
        step = lr
        obj_curr = qp_obj(a)
        for _ in range(15):
            a_new = project_simplex(a - step * effective_grad)
            obj_new = qp_obj(a_new)
            if obj_new <= obj_curr - 1e-4 * step * np.dot(effective_grad, a - a_new):
                break
            step *= 0.5

        a = a_new
        obj = qp_obj(a)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        # Decay momentum slightly
        momentum = min(0.95, momentum + 0.001)

    return best_a


def dual_update(conv, method='softmax', T=0.01, topk=10):
    """Compute dual weights from convolution values."""
    n_conv = len(conv)
    if method == 'softmax':
        scaled = conv / max(T, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        s = w.sum()
        if s < 1e-30:
            return np.ones(n_conv) / n_conv
        return w / s
    elif method == 'topk':
        k = min(topk, n_conv)
        peaks = np.argpartition(conv, -k)[-k:]
        w = np.zeros(n_conv)
        w[peaks] = 1.0 / k
        return w
    elif method == 'geo_mean':
        w_soft = dual_update(conv, 'softmax', T)
        w_topk = dual_update(conv, 'topk', topk)
        w = np.sqrt(w_soft * w_topk + 1e-20)
        s = w.sum()
        if s < 1e-30:
            return np.ones(n_conv) / n_conv
        return w / s
    elif method == 'power':
        alpha = 2.0
        w = conv ** alpha
        s = w.sum()
        if s < 1e-30:
            return np.ones(n_conv) / n_conv
        return w / s
    else:
        return np.ones(n_conv) / n_conv


def primal_dual_iteration(a0, n, time_limit, T_init=0.01, method='softmax',
                          topk=10, blend=0.5, verbose=False):
    """
    Primal-dual iteration:
    - Dual: compute weights w from conv(a,a)
    - Primal: solve QP min a^T H_w a s.t. sum(a)=1, a>=0
    - Blend and update
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_conv_max = fast_convolve(a).max()

    t0 = time.monotonic()
    T = T_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        conv = fast_convolve(a)
        T_eff = max(0.001, T)
        w = dual_update(conv, method=method, T=T_eff, topk=topk)

        # Primal solve
        qp_time = min(remaining * 0.35, 4.0)
        a_qp = qp_projected_grad(w, a, n, time_limit=qp_time, max_iter=500)

        # Blend
        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new /= s_new

        conv_new = fast_convolve(a_new)
        obj = conv_new.max()

        if obj < best_conv_max - 1e-14:
            best_conv_max = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt temperature
        if no_improve > 5:
            T *= 1.1
        elif no_improve > 20:
            T *= 1.3
        elif no_improve > 40:
            T *= 0.7
        if no_improve > 80:
            rng = np.random.default_rng((it * 104729 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            T = T_init
            no_improve = 0

        if verbose and it % 20 == 0:
            print(f"      PD it={it}, conv_max={best_conv_max:.12f}, T={T:.6f}, no_imp={no_improve}")

    return best_a, best_conv_max


def subgradient_method(a0, time_limit, lr_init=0.005, verbose=False):
    """Nesterov-accelerated subgradient for min max_k conv[k]."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)

    best_a = a.copy()
    best_max = fast_convolve(a).max()

    t0 = time.monotonic()
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        # Nesterov lookahead
        y = a + momentum * v

        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))

        # Subgradient of conv[k_peak] w.r.t. a
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        # Update
        y_new = project_simplex(y - lr * grad)
        v = y_new - y
        a = y + v
        a = project_simplex(a)

        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_max - 1e-14:
            best_max = obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.005, lr_init * 3)
        else:
            no_improve += 1
            lr *= 0.9995

        momentum = min(0.99, momentum + 0.0001)

        if no_improve > 100:
            lr *= 0.7
            momentum *= 0.95
        if no_improve > 300:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init * 0.5
            momentum = 0.9
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"      SubGrad it={it}, best={best_max:.12f}, lr={lr:.6f}")

    return best_a, best_max


def variance_descent(a0, time_limit, lr_init=0.01, verbose=False):
    """Minimize ||conv||_2^2 as a smooth surrogate using projected gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)

    best_a = a.copy()
    conv = fast_convolve(a)
    best_var = np.sum(conv * conv)

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        conv = fast_convolve(a)
        # Gradient of sum_k conv[k]^2 w.r.t. a[p] = 4 * sum_k conv[k] * a[k-p]
        grad = 4.0 * np.correlate(conv, a, 'valid')

        # Backtracking
        step = lr
        var_curr = np.sum(conv * conv)
        for _ in range(15):
            a_new = project_simplex(a - step * grad)
            conv_new = fast_convolve(a_new)
            var_new = np.sum(conv_new * conv_new)
            if var_new <= var_curr - 1e-6 * step * np.dot(grad, a - a_new) + 1e-15:
                break
            step *= 0.5

        a = a_new
        conv = fast_convolve(a)
        var = np.sum(conv * conv)

        if var < best_var - 1e-14:
            best_var = var
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            lr *= 0.85
        if no_improve > 150:
            break

        if verbose and it % 100 == 0:
            print(f"      VarDesc it={it}, var={best_var:.10f}, lr={lr:.6f}")

    return best_a, best_var


def eval_conv_max_after_change(conv, a, p, x0, x, n):
    """Compute max of convolution after changing a[p] from x0 to x."""
    delta = x - x0
    if abs(delta) < 1e-15:
        return conv.max()

    k_lo = max(0, p)
    k_hi = min(2 * n - 2, p + n - 1)
    quad = x * x - x0 * x0

    # In-range: conv[k] + 2*delta*a[k-p] for k in [k_lo, k_hi]
    in_range = conv[k_lo:k_hi + 1] + 2.0 * delta * a[k_lo - p:k_hi - p + 1]
    if 2 * p >= k_lo and 2 * p <= k_hi:
        in_range[2 * p - k_lo] += quad
    max_in = in_range.max()

    # Out-of-range
    max_out = 0.0
    if k_lo > 0:
        seg = conv[:k_lo]
        max_out = max(max_out, seg.max() if len(seg) > 0 else 0.0)
    if k_hi < 2 * n - 2:
        seg = conv[k_hi + 1:]
        max_out = max(max_out, seg.max() if len(seg) > 0 else 0.0)

    # Quadratic term at 2p if outside [k_lo, k_hi]
    if 2 * p < k_lo or 2 * p > k_hi:
        max_out = max(max_out, conv[2 * p] + quad)

    return max(max_in, max_out)


def coord_descent(a0, time_limit, verbose=False):
    """Coordinate descent with ternary search on 1D subproblems."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if stall > 15:
            break

        round_improved = False

        # Randomize order occasionally to avoid cycling
        if it % 5 == 0:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            order = rng.permutation(n)
        else:
            order = range(n)

        for p in order:
            if time.monotonic() - t0 > time_limit:
                break

            x0 = a[p]

            # Ternary search bounds
            lo = 0.0
            hi = max(1.0, 3.0 * x0 + 0.005) if x0 > 0 else 0.01

            # Ternary search (f is convex in x)
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = eval_conv_max_after_change(conv, a, p, x0, m1, n)
                f2 = eval_conv_max_after_change(conv, a, p, x0, m2, n)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            x_opt = (lo + hi) / 2.0
            x_opt = max(0.0, x_opt)

            # Verify improvement
            f_opt = eval_conv_max_after_change(conv, a, p, x0, x_opt, n)
            f_curr = conv.max()

            if f_opt < f_curr - 1e-14:
                delta = x_opt - x0
                k_lo = max(0, p)
                k_hi = min(2 * n - 2, p + n - 1)
                conv[k_lo:k_hi + 1] += 2.0 * delta * a[k_lo - p:k_hi - p + 1]
                if 2 * p <= 2 * n - 2:
                    conv[2 * p] += x_opt * x_opt - x0 * x0
                a[p] = x_opt
                round_improved = True

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        conv = fast_convolve(a)
        conv_max = conv.max()

        if conv_max < best_max - 1e-14:
            best_max = conv_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if not round_improved:
                stall += 5

        if verbose and it % 50 == 0:
            print(f"      CoordDesc it={it}, best={best_max:.12f}, stall={stall}")

    return best_a, best_max


def smoothed_gradient(a0, time_limit, T_init=0.001, lr_init=0.02, verbose=False):
    """Smoothed gradient descent with log-sum-exp surrogate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)

    best_a = a.copy()
    best_max = fast_convolve(a).max()

    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        conv = fast_convolve(a)
        scaled = conv / max(T, 1e-10)
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient of log-sum-exp: sum_k p[k] * grad(conv[k])
        # grad(conv[k])[i] = 2*a[k-i] (valid)
        # Combined: grad[i] = 2 * sum_k p[k] * a[k-i] = 2 * correlate(p, a, 'valid')[i]
        grad = 2.0 * np.correlate(p, a, 'valid')

        # Line search
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate

        if best_val < best_max - 1e-14:
            best_max = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 10:
            T *= 1.15
            lr *= 0.95
        if no_improve > 40:
            T *= 0.7
        if no_improve > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_max


def annealing_descent(a0, time_limit, T_init=0.02, T_end=0.0005, verbose=False):
    """Simulated annealing with multiplicative updates for exploration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    conv = fast_convolve(a)
    best_a = a.copy()
    best_max = conv.max()
    current_max = conv.max()

    t0 = time.monotonic()
    total_time = time_limit
    seed = 12345
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / total_time)
        T = T_init * (T_end / T_init) ** frac

        rng = np.random.default_rng(seed + it * 7919)

        # Perturb a random subset
        num_pert = max(3, n // 8)
        indices = rng.choice(n, size=num_pert, replace=False)
        scale = 0.02 + 0.03 * (1.0 - frac)
        delta = rng.standard_normal(num_pert) * scale * a[indices].mean() * 10

        a_new = a.copy()
        a_new[indices] += delta
        a_new = project_simplex(a_new)

        conv_new = fast_convolve(a_new)
        new_max = conv_new.max()

        # Accept/reject
        if new_max < current_max:
            accept = True
        else:
            diff = new_max - current_max
            if T < 1e-8:
                accept = False
            else:
                accept = rng.random() < np.exp(-diff * 2 * n / T)

        if accept:
            a = a_new
            conv = conv_new
            current_max = new_max
            if new_max < best_max - 1e-14:
                best_max = new_max
                best_a = a.copy()
                if verbose and it % 100 == 0:
                    print(f"      Anneal it={it}, best={best_max:.12f}, T={T:.6f}")

    return best_a, best_max


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Cosine window
    c1 = 0.5 + 0.5 * np.cos(np.pi * (x + 0.5) / n)
    c1 = np.maximum(c1, 0.001)
    inits.append(c1 / c1.sum())
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    # Two peaks
    g2 = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / (0.12 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.7 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append(g2 / g2.sum())
    # Sine window
    s1 = np.sin(np.pi * (x + 0.5) / n)
    s1 = np.maximum(s1, 0.001)
    inits.append(s1 / s1.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: multi-phase hybrid search for auto-convolution minimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent, using uniform n=384")

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: n-exploration (300s) ===
    print("\n=== Phase 1: n-exploration ===")
    t1 = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 128),
        n_parent + 128,
        max(128, n_parent // 2),
        min(4096, n_parent * 2) if n_parent * 2 <= 4096 else n_parent,
        256, 512, 768, 1024, 1536,
    ]))
    # Remove very large n that might be too slow
    candidate_ns = [n for n in candidate_ns if 64 <= n <= 3000]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 30:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            if start.sum() > 1e-10:
                start /= start.sum()
            else:
                start = np.ones(n_try) / n_try

        # Quick optimization
        t_alloc = min(remaining, 30)
        print(f"  n={n_try}, t={t_alloc:.0f}s")

        # Variance descent (fast, smooth)
        a_v, _ = variance_descent(start, time_limit=min(t_alloc * 0.4, 10))
        try_seq(a_v, f"var_n{n_try}")

        # QP primal-dual
        a_pd, _ = primal_dual_iteration(a_v if best_seq is None or len(a_v) == n_try else start,
                                        n_try, time_limit=t_alloc * 0.5,
                                        T_init=0.01, method='softmax', topk=15, blend=0.5)
        try_seq(a_pd, f"pd_n{n_try}")

        # If this n is promising, try top-k variant
        if best_seq is not None and len(best_seq) == n_try:
            a_pd2, _ = primal_dual_iteration(a_pd, n_try, time_limit=min(remaining, 10),
                                             T_init=0.005, method='topk', topk=8, blend=0.4)
            try_seq(a_pd2, f"pd_topk_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep optimization on best n (1000s) ===
    print("\n=== Phase 2: Deep optimization ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # 2a: Variance descent fine-tuning
        remaining = deadline - time.monotonic() - 700
        if remaining > 20:
            t_alloc = min(remaining, 80)
            print(f"  VarDescent n={n_best}, t={t_alloc:.0f}s")
            a_v, _ = variance_descent(best_seq, time_limit=t_alloc, lr_init=0.005)
            try_seq(a_v, "vardeep")

        # 2b: QP primal-dual with multiple configs
        configs = [
            {'T_init': 0.005, 'method': 'softmax', 'topk': 20, 'blend': 0.5},
            {'T_init': 0.01, 'method': 'geo_mean', 'topk': 10, 'blend': 0.4},
            {'T_init': 0.003, 'method': 'topk', 'topk': 5, 'blend': 0.6},
            {'T_init': 0.02, 'method': 'power', 'topk': 10, 'blend': 0.3},
            {'T_init': 0.008, 'method': 'softmax', 'topk': 30, 'blend': 0.7},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, 150)
            print(f"  PD config {ci} (T={cfg['T_init']}, method={cfg['method']}), t={t_alloc:.0f}s")
            a_opt, _ = primal_dual_iteration(best_seq, len(best_seq), time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"pd2_{ci}")

        # 2c: Coordinate descent (new method)
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(remaining, 200)
            print(f"  CoordDescent t={t_alloc:.0f}s")
            a_cd, _ = coord_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_cd, "coord")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Subgradient + Smoothed gradient (300s) ===
    print("\n=== Phase 3: Subgradient & Smoothed ===")
    t3 = time.monotonic()

    if best_seq is not None:
        # Subgradient with different LR
        for lr_i in [0.003, 0.01, 0.001]:
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  SubGrad lr={lr_i}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_method(best_seq, time_limit=t_alloc, lr_init=lr_i)
            try_seq(a_opt, f"subgrad_{lr_i}")

        # Smoothed gradient
        remaining = deadline - time.monotonic() - 60
        if remaining > 15:
            t_alloc = min(remaining, 80)
            print(f"  SmoothedGrad T=0.0003, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_gradient(best_seq, time_limit=t_alloc, T_init=0.0003, lr_init=0.015)
            try_seq(a_opt, "smoothed")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Annealing + Coord descent combo (200s) ===
    print("\n=== Phase 4: Annealing & Coord ===")
    t4 = time.monotonic()

    if best_seq is not None:
        # Simulated annealing for exploration
        remaining = deadline - time.monotonic() - 80
        if remaining > 20:
            t_alloc = min(remaining, 80)
            print(f"  Annealing t={t_alloc:.0f}s")
            a_ann, _ = annealing_descent(best_seq, time_limit=t_alloc, T_init=0.01, T_end=0.0003)
            try_seq(a_ann, "anneal")

        # Coordinate descent again (from potentially improved starting point)
        remaining = deadline - time.monotonic() - 50
        if remaining > 15:
            t_alloc = min(remaining, 80)
            print(f"  CoordDescent2 t={t_alloc:.0f}s")
            a_cd2, _ = coord_descent(best_seq, time_limit=t_alloc)
            try_seq(a_cd2, "coord2")

        # Quick QP from best
        remaining = deadline - time.monotonic() - 40
        if remaining > 10:
            t_alloc = min(remaining, 40)
            a_opt, _ = primal_dual_iteration(best_seq, len(best_seq), time_limit=t_alloc,
                                             T_init=0.002, method='topk', topk=3, blend=0.3)
            try_seq(a_opt, "pd_fine")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final polish (150s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Tiny perturbations + quick coord descent
        for pi in range(8):
            remaining = deadline - time.monotonic() - 20
            if remaining < 10:
                break
            noise = 0.0003 + 0.0003 * pi
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise)
            talloc = min(remaining, 15)
            a_opt, _ = coord_descent(f_pert, time_limit=talloc)
            try_seq(a_opt, f"polish_{pi}")

        # Final subgradient
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            talloc = min(remaining, 30)
            a_opt, _ = subgradient_method(best_seq, time_limit=talloc, lr_init=0.001)
            try_seq(a_opt, "final_subgrad")

        # Final coord descent on best
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            talloc = min(remaining, 30)
            a_opt, _ = coord_descent(best_seq, time_limit=talloc)
            try_seq(a_opt, "final_coord")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
