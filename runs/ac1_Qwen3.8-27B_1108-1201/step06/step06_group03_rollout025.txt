def smoothed_grad_descent(a_init, time_limit, beta_start=10, beta_end=50000, 
                          lr_start=0.01, min_improve=1e-15):
    a = a_init.copy()
    a = np.maximum(a, 1e-15)  # ensure positive for log
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    n = len(a)
    total_iter = 0
    no_improve = 0
    
    # Use log parameterization
    z = np.log(a)  # a = softmax(z)
    
    while time.monotonic() - t0 < time_limit:
        total_iter += 1
        
        # Annealing: beta increases over time
        progress = min(1.0, total_iter / (time_limit * 1000))  # rough progress
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute conv
        conv_a = fast_convolve(a)
        current_max = conv_a.max()
        
        # Compute gradient of smooth approximation
        shifted = beta * conv_a
        shifted -= shifted.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        w = w / w_sum
        
        # grad_a[j] = sum_k w[k] * (a[k-j] + a[j-k])
        # Use FFT-based computation
        grad_a = compute_grad_fft(a, w)
        
        # grad_z = a * (grad_a - <grad_a, a>)
        mean_grad = np.dot(grad_a, a)
        grad_z = a * (grad_a - mean_grad)
        
        # Adaptive learning rate (backtracking)
        lr = lr_start / (1 + 0.001 * total_iter)
        improved = False
        for _ in range(20):
            z_new = z - lr * grad_z
            # Re-normalize z to avoid overflow
            z_new -= z_new.mean()
            a_new = np.exp(z_new)
            a_new /= a_new.sum()
            a_new = np.maximum(a_new, 1e-15)
            a_new /= a_new.sum()
            
            conv_new = fast_convolve(a_new).max()
            if conv_new < current_max - 1e-15:
                z = z_new
                a = a_new
                current_max = conv_new
                improved = True
                break
            lr *= 0.5
        
        if not improved:
            # Try a random perturbation
            if no_improve > 100:
                scale = min(0.01, 0.001 * (no_improve / 100))
                z += np.random.default_rng(total_iter).standard_normal(n) * scale
                z -= z.mean()
                a = np.exp(z)
                a /= a.sum()
                no_improve = 0
        
        if current_max < best_obj - min_improve:
            best_obj = current_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj
    
    z = np.log(a)
    n = len(a)
    rng = np.random.default_rng(42)
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        beta = 10 * (5000) ** progress  # anneal beta from 10 to 50000
        
        conv_a = fast_convolve(a)
        
        shifted = beta * conv_a
        shifted -= shifted.max()
        w = np.exp(shifted)
        w /= w.sum()

        mean_grad = np.dot(grad_a, a)
        grad_z = a * (grad_a - mean_grad)

        lr = 0.005
        noise_scale = T * 0.1

            a = a_new
            current_obj = conv_new
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast annealed smoothed gradient
descent with FFT-based gradient computation, simplex/momentum hybrid, 
multi-length exploration, and simulated annealing.
"""

import numpy as np
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, w):
    """
    Compute grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices.
    w has length 2n-1, a has length n, result has length n.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # p1[j] = sum_m a[m] * w[m+j] = conv(a_flip, w)[n-1+j]
    a_flip = a[::-1].copy()
    Aw = np.fft.rfft(a_flip, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_aw = np.fft.irfft(Aw * W, fft_size)
    p1 = conv_aw[n - 1:2 * n - 1]

    # p2[j] = sum_m w[m] * a[j-m] = conv(w, a)[j]
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad, fft_size)
    conv_wa = np.fft.irfft(W * A, fft_size)
    p2 = conv_wa[:n]

    return p1 + p2


def smoothed_gradient_step(a, beta):
    """
    One step of smoothed gradient descent.
    Returns new_a and the objective value.
    """
    n = len(a)
    conv_a = fast_convolve(a)

    # Softmax weights
    shifted = beta * conv_a
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(conv_a)] = 1.0
    else:
        w = w / w_sum

    # Gradient
    grad_a = compute_grad_fft(a, w)

    return grad_a, conv_a.max()


def optimize_smoothed(a_init, time_limit, beta_start=20.0, beta_end=200000.0,
                      init_lr=0.02, use_momentum=True, momentum=0.85,
                      rng_seed=42, verbose=False):
    """
    Annealed smoothed gradient descent with momentum and adaptive learning rate.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-30)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    conv_best = fast_convolve(a)
    best_obj = conv_best.max()

    rng = np.random.default_rng(rng_seed)
    t0 = time.monotonic()
    it = 0
    stagnant = 0

    v = np.zeros(n)
    mu = momentum
    lr = init_lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 0.01:
            break

        # Annealing schedule (piecewise)
        progress = min(1.0, elapsed / time_limit)
        # Use a slower ramp for the first half, faster for the second
        if progress < 0.7:
            beta_frac = (progress / 0.7) ** 1.5
        else:
            beta_frac = 0.7 ** 1.5 + (1 - 0.7 ** 1.5) * ((progress - 0.7) / 0.3) ** 0.8
        beta = beta_start * (beta_end / beta_start) ** beta_frac

        # Learning rate decay
        lr = init_lr / (1.0 + 0.5 * it / 1000)

        # Compute gradient
        grad_a, current_max = smoothed_gradient_step(a, beta)

        if use_momentum:
            v = mu * v - lr * grad_a
            step = v
        else:
            step = -lr * grad_a

        a_new = a + step
        a_new = project_simplex(a_new)

        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        # Accept if improved
        if obj_new < best_obj - 1e-15:
            a = a_new
            best_obj = obj_new
            best_a = a_new.copy()
            stagnant = 0
        elif obj_new < current_max + 1e-6:
            # Accept if not much worse (helps with momentum)
            a = a_new
            stagnant += 1
        else:
            # Reject, reduce momentum velocity
            v *= 0.5
            stagnant += 1

        # Periodic random restarts from best
        if stagnant > 80:
            scale = min(0.02, 0.003 * (stagnant / 80.0))
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            stagnant = 0
            # Reset beta slightly
            # (we'll just let the schedule continue)

        if verbose and it % 5000 == 0:
            print(f"      it={it}, beta={beta:.1f}, obj={best_obj:.10f}, "
                  f"stagnant={stagnant}, seed={rng_seed}")

    return best_a, best_obj


def sa_optimize(a_init, time_limit, T_start=0.05, T_end=0.00001,
                beta_start=20.0, beta_end=100000.0, lr=0.01,
                rng_seed=123, verbose=False):
    """
    Simulated annealing on the smoothed objective.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-30)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj

    rng = np.random.default_rng(rng_seed)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 0.01:
            break

        progress = min(1.0, elapsed / time_limit)
        T = T_start * (T_end / T_start) ** (progress ** 0.7)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.6)
        lr_eff = lr / (1.0 + progress * 5.0)

        conv_a = fast_convolve(a)
        shifted = beta * conv_a - beta * conv_a.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(conv_a)] = 1.0
        else:
            w = w / w_sum

        grad_a = compute_grad_fft(a, w)

        # Gradient step + noise
        noise_scale = T * 0.5
        a_new = a - lr_eff * grad_a + rng.standard_normal(n) * noise_scale
        a_new = project_simplex(a_new)

        conv_new = fast_convolve(a_new).max()
        delta = conv_new - current_obj

        # Accept/reject
        if delta < 0 or (T > 1e-10 and rng.random() < math.exp(-delta / max(T, 1e-15))):
            a = a_new
            current_obj = conv_new
            if current_obj < best_obj - 1e-15:
                best_obj = current_obj
                best_a = a_new.copy()

        if verbose and it % 10000 == 0:
            print(f"      SA it={it}, T={T:.6f}, beta={beta:.1f}, "
                  f"obj={best_obj:.10f}")

    return best_a, best_obj


def coordinate_polish(a_init, time_limit, rng_seed=777):
    """
    Coordinate descent: for the peak lag, try moving mass from high-contributing
    elements to low-contributing ones.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-30)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    rng = np.random.default_rng(rng_seed)
    t0 = time.monotonic()
    step_size = 0.005
    it = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))

        # For peak at lag k: conv[k] = sum_i a[i]*a[k-i]
        # Try reducing the largest product term
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        if js > je:
            break

        # Find the pair with the largest product
        best_prod = -1
        best_j = -1
        best_kk = -1
        for j in range(js, je + 1):
            prod = a[j] * a[peak_k - j]
            if prod > best_prod:
                best_prod = prod
                best_j = j
                best_kk = peak_k - j

        if best_j < 0 or a[best_j] < 1e-10:
            break

        # Try moving a small amount from best_j to a random distant location
        improved_this_iter = False
        for _ in range(3):
            target = int(rng.integers(0, n))
            if target == best_j or target == best_kk:
                continue
            # Avoid creating new peaks
            delta = a[best_j] * step_size
            candidate = a.copy()
            candidate[best_j] -= delta
            candidate[target] += delta
            if candidate[best_j] < -1e-12:
                continue
            candidate = np.maximum(candidate, 0)
            s = candidate.sum()
            if s < 1e-15:
                continue
            candidate = candidate / s

            conv_c = fast_convolve(candidate).max()
            if conv_c < best_obj - 1e-15:
                a = candidate
                best_obj = conv_c
                best_a = a.copy()
                improved_this_iter = True
                break

        if not improved_this_iter:
            stagnant += 1
            step_size *= 0.95
            if step_size < 1e-7:
                break
            if stagnant % 200 == 0:
                # Global perturbation
                a = project_simplex(best_a + rng.standard_normal(n) * step_size * 3)
        else:
            stagnant = 0

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.array([1.0])
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def score_of(a):
    """Compute the actual evaluation score."""
    a_list = [float(max(0.0, min(1000.0, x))) for x in a]
    s = np.sum(a_list)
    if s < 0.01:
        return float('inf')
    b = np.convolve(a_list, a_list)
    return float(2 * len(a_list) * max(b) / (s ** 2))


def generate_initializations(n, parent, rng):
    """Generate diverse starting points."""
    starts = []
    x = np.arange(n, dtype=np.float64)

    # Resampled parent
    if parent is not None and len(parent) > 0:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine (Bartlett-like)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Triangle
    mid = (n - 1) / 2
    tri = 1.0 - np.abs(x - mid) / (mid + 1)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / tri.sum())

    # Two triangles (V-shape)
    v = np.minimum(np.abs(x - mid), n - 1 - np.abs(x - mid))
    v = v / (mid + 1) + 0.1
    starts.append(v / v.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Gaussian centered
    g = np.exp(-0.5 * ((x - mid) / (0.25 * n)) ** 2)
    starts.append(g / g.sum())

    # Two Gaussians
    g1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Random (Dirichlet)
    starts.append(rng.dirichlet(np.ones(n) * 0.5))
    starts.append(rng.dirichlet(np.ones(n) * 0.1))

    # "Anti-symmetric" (more mass on one side)
    ramp = np.linspace(0.1, 1.0, n)
    starts.append(ramp / ramp.sum())
    ramp2 = np.linspace(1.0, 0.1, n)
    starts.append(ramp2 / ramp2.sum())

    return starts

            t_alloc = min(remaining, 150.0)
            print(f"  SA refinement: n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = sa_optimize(best_seq, time_limit=t_alloc,
                                   T_start=0.02, T_end=0.00001,
                                   beta_start=100, beta_end=200000,
                                   lr=0.008, rng_seed=seed + 6000)
            try_seq(a_opt, "sa_deep")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t2:.0f}s)")

    # === Phase 3: Multi-restart with perturbation (400s) ===
    print("\n=== Phase 3: Multi-Restart ===")
    t3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        perturb_scales = [0.005, 0.01, 0.02, 0.03, 0.05, 0.01, 0.015]

        for pi, scale in enumerate(perturb_scales):
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break
            t_alloc = min(remaining, 55.0)
            perturbed = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            # Alternate between methods
            if pi % 2 == 0:
                a_opt, _ = optimize_smoothed(perturbed, time_limit=t_alloc,
                                             beta_start=200, beta_end=300000,
                                             init_lr=0.01, momentum=0.92,
                                             rng_seed=seed + 7000 + pi * 100)
            else:
                a_opt, _ = sa_optimize(perturbed, time_limit=t_alloc,
                                       T_start=0.01, T_end=0.000005,
                                       beta_start=200, beta_end=300000,
                                       lr=0.006, rng_seed=seed + 8000 + pi * 100)
            try_seq(a_opt, f"restart_{pi}_s{scale}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t3:.0f}s)")

    # === Phase 4: Length refinement (200s) ===
    print("\n=== Phase 4: Length Refinement ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # Try nearby lengths
        nearby = [n_best - 32, n_best - 16, n_best + 16, n_best + 32,
                  n_best + 64, n_best - 64]
        nearby = [l for l in nearby if 128 <= l <= 3000]

        for n_try in nearby:
            remaining = deadline - time.monotonic() - 100
            if remaining < 10:
                break
            resampled = resample_sequence(best_seq, n_try)
            t_alloc = min(remaining, 30.0)
            a_opt, _ = optimize_smoothed(resampled, time_limit=t_alloc,
                                         beta_start=100, beta_end=200000,
                                         init_lr=0.01, rng_seed=seed + 9000 + n_try)
            try_seq(a_opt, f"len_{n_try}")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_alloc = min(remaining, remaining * 0.4)
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc,
                                         rng_seed=seed + 10000)
            try_seq(a_opt, "coord_polish")

        # Final SA with very low temperature
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc = min(remaining, remaining * 0.4)
            a_opt, _ = sa_optimize(best_seq, time_limit=t_alloc,
                                   T_start=0.001, T_end=0.0000001,
                                   beta_start=1000, beta_end=1000000,
                                   lr=0.003, rng_seed=seed + 11000)
            try_seq(a_opt, "final_sa")

        # Tiny perturb + short gradient
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_alloc = min(remaining, remaining * 0.5)
            for _ in range(3):
                remaining = deadline - time.monotonic()
                if remaining < 5:
                    break
                tiny_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
                tt = min(remaining, 15.0)
                a_opt, _ = optimize_smoothed(tiny_pert, time_limit=tt,
                                             beta_start=500, beta_end=1000000,
                                             init_lr=0.003, momentum=0.95,
                                             rng_seed=seed + 12000)
                try_seq(a_opt, "tiny_pert")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
