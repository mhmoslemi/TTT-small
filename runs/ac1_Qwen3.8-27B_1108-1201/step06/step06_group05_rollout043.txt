```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with precomputed base convolutions (O(n) per subproblem eval)
2. Projected subgradient for initial progress
3. Multi-start with diverse initializations and perturbation restarts
4. Symmetry exploitation and final polishing
"""
import numpy as np
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_from_parts(conv_max, n, sum_a):
    """Compute objective from components."""
    if sum_a < 1e-15:
        return np.inf
    return 2.0 * n * conv_max / (sum_a * sum_a)


def full_objective(a):
    """Compute full objective for a sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgrad_step(a, top_k=20):
    """Compute weighted subgradient from top-k peaks."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    order = np.argsort(pv)[::-1]
    weights = np.zeros(len(peaks))
    for rank, idx in enumerate(order):
        weights[idx] = 1.0 / (rank + 1) ** 1.5
    ws = weights.sum()
    if ws > 0:
        weights /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        sg[idx_arr] += weights[i] * a[k - idx_arr]
    return sg, conv


def pgd_refine(a0, time_limit, top_k=20, lr_init=0.02):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    lr = lr_init
    mom = 0.92
    best_a = a.copy()
    best_obj = full_objective(a)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        sg, conv = subgrad_step(a, top_k)
        obj_curr = 2.0 * n * conv.max()  # normalized: sum=1
        best_candidate = a
        best_val = obj_curr
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.125]:
            v_new = mom * v - trial_lr * sg
            candidate = project_simplex(a + v_new)
            c = fast_convolve(candidate)
            val = 2.0 * n * c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                v = v_new
                break
        a = best_candidate
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.9
        if no_improve > 30:
            lr *= 0.95
        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.003 + 0.004 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coord_descent(a_init, time_limit, verbose=False):

    Coordinate descent on the scale-invariant objective 2*n*max(conv)/(sum)^2.
    Uses precomputed base convolutions for O(n) per 1D subproblem evaluation.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    n = len(a)
    sum_a = a.sum()
    if sum_a < 1e-15:
        a = np.ones(n)
        sum_a = n

    conv = fast_convolve(a)
    best_obj = obj_from_parts(conv.max(), n, sum_a)
    best_a = a.copy()

    t0 = time.monotonic()
    cycle = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        cycle += 1
        if cycle % 50 == 0 and verbose:
            print(f"    CD cycle {cycle}, obj={best_obj:.10f}, elapsed={time.monotonic()-t0:.0f}s")
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit - 0.05:

            S = sum_a - old_val

            base = conv.copy()
            lo_k = j

            if lo_k <= hi_k:
                a_seg = a[0:hi_k - lo_k + 1]
                base[lo_k:hi_k + 1] -= 2.0 * old_val * a_seg
            # Correct k=2j: we subtracted 2*old_val*a[j]=2*old_val^2, should subtract old_val^2
            if 2 * j <= 2 * n - 2:

            shifted_a = a[0:len(shifted_base)].copy()
            # Index of k=2j within shifted range

                idx_2j = -1

            m_out = -np.inf
            if j > 0:

            if hi_k + 1 <= 2 * n - 2:

            def eval_t(t):

                denom = S + t

                m = m_out

                        if idx_2j > 0:
                            m = max(m, (shifted_base[:idx_2j] + 2.0 * t * shifted_a[:idx_2j]).max())
                        if idx_2j + 1 < len(shifted_base):
                            m = max(m, (shifted_base[idx_2j + 1:] + 2.0 * t * shifted_a[idx_2j + 1:]).max())
                        m = max(m, base[2 * j] + t * t)
                    else:
                        m = max(m, (shifted_base + 2.0 * t * shifted_a).max())
                return 2.0 * n * m / (denom * denom)

            # Find optimal t >= 0
            best_t = old_val
            best_f = eval_t(old_val)

            # Candidate evaluations
            cands = [0.0, old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5,
                     2.0 * old_val, old_val + 0.01, max(0.0, old_val - 0.01)]
            for tc in cands:
                fc = eval_t(tc)
                if fc < best_f:
                    best_f = fc
                    best_t = tc

            # Golden section refinement
            if best_t > 1e-10:
                lo = max(0.0, best_t * 0.6)
                hi = best_t * 1.5 + 0.001
            else:
                lo = 0.0
                hi = max(old_val * 1.5, 0.01)

            for _ in range(20):
                m1 = lo + 0.382 * (hi - lo)
                m2 = lo + 0.618 * (hi - lo)
                f1 = eval_t(m1)
                f2 = eval_t(m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
                tm = 0.5 * (lo + hi)
                fm = eval_t(tm)
                if fm < best_f:
                    best_f = fm
                    best_t = tm

            # Update
            a[j] = max(0.0, best_t)
            sum_a = S + a[j]
            conv = fast_convolve(a)
            curr_obj = obj_from_parts(conv.max(), n, sum_a)
            if curr_obj < best_obj - 1e-14:
                best_obj = curr_obj
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                scale = 0.002 / (1.0 + 0.15 * no_improve)
                a = np.maximum(best_a + rng.standard_normal(n) * scale, 0.0)
                sum_a = a.sum()
                conv = fast_convolve(a)
                best_obj = obj_from_parts(conv.max(), n, sum_a)
                no_improve = 0
        else:
            no_improve = 0

    return best_a, best_obj
