def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_sequence(seq, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)
