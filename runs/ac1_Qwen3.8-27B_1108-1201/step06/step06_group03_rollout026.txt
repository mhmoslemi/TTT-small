"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Smoothed log-sum-exp objective with Adam optimizer and tau annealing
- Multi-length, multi-start exploration
- SLP (successive linearization) via Frank-Wolfe with LP oracle
- Peak-targeted coordinate descent for final polish
- Exploits FFT-based gradient computation for the smoothed objective
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def logsumexp(x, tau):
    """Compute (1/tau)*log(sum(exp(tau*x))) with numerical stability."""
    m = x.max()
    return m + np.log(np.sum(np.exp(tau * (x - m)))) / tau


def smooth_obj_and_grad(z, tau):
    """Compute smoothed objective and gradient w.r.t. z (softmax parameterization)."""
    a = softmax_fn(z)
    n = len(a)
    conv = fast_convolve(a)
    
    # Softmax weights over convolution entries
    p = np.exp(tau * (conv - conv.max()))
    p = p / p.sum()
    
    # Objective value
    obj = logsumexp(conv, tau)
    
    # Gradient of weighted sum of quadratics w.r.t. a
    # g_a[j] = 2 * sum_l a[l] * p[j+l] = 2 * correlate(p, a, 'valid')[j]
    g_a = 2.0 * np.correlate(p, a, mode='valid')
    
    # Chain rule through softmax: g_z = a * (g_a - dot(a, g_a))
    g_z = a * (g_a - np.dot(a, g_a))
    
    return obj, g_z


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g)[k] = sum_j A[k,j]*g[j]."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def adam_smoothed_optimize(a0, tau_start, tau_end, time_limit, verbose=False):
    """
    Optimize using Adam on the smoothed objective with tau annealing.
    Parameterization: a = softmax(z), optimize over z unconstrained.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    # Initialize z from a
    z = np.log(a + 1e-15)
    z = z - z.mean()  # center for numerical stability
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    t = 0
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    stagnant = 0
    patience = 500
    
    while time.monotonic() - t0 < time_limit:
        # Annealing: linear from tau_start to tau_end over the time budget
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Use a smoother annealing: stay at tau_start for first 20%, then ramp
        if frac < 0.2:
            tau = tau_start
        else:
            ramp_frac = (frac - 0.2) / 0.8
            tau = tau_start + (tau_end - tau_start) * ramp_frac ** 0.7
        
        t += 1
        obj, g_z = smooth_obj_and_grad(z, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g_z
        v = beta2 * v + (1 - beta2) * g_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate actual (unsmoothed) objective
        a_new = softmax_fn(z)
        actual_obj = fast_convolve(a_new).max()
        
        if actual_obj < best_obj - 1e-12:
            best_obj = actual_obj
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Adaptive learning rate
        if stagnant > patience:
            lr *= 0.5
            stagnant = 0
            if lr < 1e-6:
                break
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0

            stagnant += 1
            if stagnant > 40:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                M = fast_convolve(f).max()
                stagnant = 0
            continue
        
        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M + 1e-14
        
        for alpha in np.linspace(0.01, 1.0, 20):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for alpha in np.linspace(lo, hi, 10):
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
        
        if stagnant > 15:
            m_relax = min(m_relax * 1.02, 1.03)
        
        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stagnant={stagnant}")
    
    return best_f, best_obj


def subgradient_optimize(a0, time_limit, T_init=0.0005, lr_init=0.01, top_k_frac=0.05):
    """Smoothed subgradient descent with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))
    
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9
    
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        
        pv = conv[peaks]
        pv_max = pv.max()
        w = np.exp((pv - pv_max) / (T * 0.1))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]
        
        v = mom * v - lr * grad
        
        best_candidate = a
        best_val = conv.max()
        
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 30:
            T *= 1.05
            lr *= 0.95
        if stagnant > 100:
            T *= 0.5
            lr *= 1.5
            mom = 0.93
        if stagnant > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.002 + 0.003 * min(1.0, (stagnant - 300) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            mom = 0.9
            stagnant = 0
    
    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit):
    """Coordinate descent targeting top peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    step_scale = 0.001
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            if js > je:
                continue
            
            # Find top contributing indices for this peak
            contribs = []
            for j in range(js, je + 1):
                contribs.append((a[j] * a[k - j], j))
            contribs.sort(reverse=True)
            
            for (_, j1) in contribs[:8]:
                if a[j1] < 1e-9:
                    continue
                
                # Try moving mass from j1 to a random distant location
                for _ in range(3):
                    j_target = int(rng.integers(0, n))
                    if abs(j_target - j1) < n // 10:
                        continue
                    
                    for delta_frac in [step_scale, step_scale * 0.5, step_scale * 2.0]:
                        delta = a[j1] * delta_frac
                        candidate = a.copy()
                        candidate[j1] -= delta
                        candidate[j_target] += delta
                        if candidate[j1] < -1e-12:
                            continue
                        candidate = np.maximum(candidate, 0)
                        candidate = candidate / candidate.sum()
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-15:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            step_scale *= 0.92
            if step_scale < 1e-7:
                break
            # Periodic perturbation
            if it % 100 == 0:
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 3000:
                break
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def nesterov_smoothed(a0, tau, time_limit, lr_init=0.05):
    """Nesterov accelerated gradient on smoothed objective (in a-space via projection)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    y = a.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init
    t = 0
    momentum = 0.9
    
    t0 = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        p = np.exp(tau * (conv_y - conv_y.max()))
        p = p / p.sum()
        g_a = 2.0 * np.correlate(p, y, mode='valid')
        
        # Gradient step with projection
        y_new = project_simplex(y - lr * g_a)
        
        # Nesterov update
        if t > 1:
            beta = 2.0 / (t + 1.5)
            a_new = beta * y_new + (1 - beta) * ((t - 1) / (t + 0.5)) * y - (t - 1) / (t + 0.5) * a
        else:
            a_new = y_new
        
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        y = a.copy()
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 100:
            lr *= 0.7
            stagnant = 0
            if lr < 1e-8:
                break
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase optimization combining smoothed gradient,
    Frank-Wolfe SLP, subgradient, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 90.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)}, M={fast_convolve(a_norm).max():.10f})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")
    
    # === Phase 1: Multi-length exploration with smoothed Adam (400s) ===
    print("\n=== Phase 1: Multi-length Smoothed Adam Exploration ===")
    t_phase = time.monotonic()
    
    n_parent = len(parent)
    # Candidate lengths: focus on a range around parent, plus some others
    candidate_lengths = []
    for nl in [n_parent - 96, n_parent - 48, n_parent, n_parent + 48, n_parent + 96,
               512, 640, 896, 1024, 1280]:
        if 256 <= nl <= 2048 and nl not in candidate_lengths:
            candidate_lengths.append(nl)
    candidate_lengths.sort()
    
    # For each length, try smoothed Adam with a few starts
    for n_try in candidate_lengths:
        remaining = deadline - time.monotonic() - 1000
        if remaining < 20:
            break
        
        # Generate starts for this length
        starts = []
        if parent is not None:
            starts.append(resample_sequence(parent, n_try))
        # Uniform
        starts.append(np.ones(n_try) / n_try)
        # Cosine taper
        x = np.arange(n_try, dtype=np.float64)
        w = 0.5 * (1 + np.cos(np.pi * x / (n_try - 1)))
        w = np.maximum(w, 0.01)
        starts.append(w / w.sum())
        # Edge concentration
        edge = 1.0 / (np.minimum(x, n_try - 1 - x) + 1.0)
        starts.append(edge / edge.sum())
        
        for si, start in enumerate(starts[:3]):
            remaining = deadline - time.monotonic() - 900
            if remaining < 15:
                break
            t_alloc = min(remaining, 50)
            print(f"  SmoothedAdam n={n_try}, start={si}, t={t_alloc:.0f}s")
            a_opt, _ = adam_smoothed_optimize(start, tau_start=5.0, tau_end=50.0,
                                              time_limit=t_alloc)
            try_seq(a_opt, f"adam_n{n_try}_s{si}")
    
    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # === Phase 2: Deep FW optimization on best (500s) ===
    print("\n=== Phase 2: Deep Frank-Wolfe ===")

    remaining = deadline - time.monotonic() - 500
    if remaining > 30:
        t_alloc = min(remaining, 350)

        try_seq(a_opt, "deep_fw")
    
    # Perturbed FW restarts
    for scale in [0.0005, 0.001, 0.002, 0.005, 0.01]:
        remaining = deadline - time.monotonic() - 350
        if remaining < 20:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc = min(remaining, 60)
        a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
        try_seq(a_opt, f"fw_pert_{scale}")
    
    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak coordinate descent + final polish (remaining) ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)

    remaining = deadline - time.monotonic() - 60

        a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)

    return [float(x) for x in best_seq]
