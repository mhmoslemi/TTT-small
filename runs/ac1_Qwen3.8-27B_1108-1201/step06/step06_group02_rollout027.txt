"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines Frank-Wolfe with LP oracle, smoothed gradient descent,
multigrid initialization, and targeted peak reduction.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_smoothed(a, p):
    """Compute gradient of smoothed objective.
    p: softmax weights over convolution values
    Returns gradient vector of length n.

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines Frank-Wolfe with LP oracle, smoothed gradient descent with adaptive temperature,
multigrid initialization, simulated annealing restarts, and targeted peak reduction.
Key improvements over baseline: better LP oracle construction, multi-scale initialization,
adaptive step sizes, and structured perturbations for escaping local minima.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f):
    """Build constraint matrix for LP oracle. A[k,j] = f[j] if j<=k<j+n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f[j]
    return A


def solve_lp_oracle(f, M, time_limit=1.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_A(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:

    except Exception:

    return None

    corr = np.correlate(p, a, mode='valid')
    # extra[i] = p[2i] * a[i] (diagonal correction)
    extra = p[2 * np.arange(n)] * a
    return 2.0 * n * (corr + extra)

    s = p.sum()
    if s < 1e-300:

    else:
        p /= s
    return p

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Adaptive T and lr
            if no_improve % 15 == 0:
                T *= 1.08
            if no_improve > 50:
                T *= 0.7
            if no_improve % 10 == 0:
                lr *= 0.97

        if no_improve > 120:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.003 + 0.006 * min(1.0, (no_improve - 120) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def multi_peak_sg(a0, time_limit, top_k=20, lr_init=0.002, mom=0.93):
    """Multi-peak subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k)[-k:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 2.0
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            kp = int(peaks[i])
            js = max(0, kp - n + 1)
            je = min(n - 1, kp)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[kp - idx_arr]

        v = mom * v - lr * sg
        a = project_simplex(a + v)

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        mom = min(0.97, mom + 0.00005)

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 150) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0

    return best_a, best_obj


def targeted_peak_reduction(a0, time_limit):
    """Iteratively reduce the highest peak by mass transfer."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)

        if len(affected) == 0:
            break

        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        j_min = outside[np.argmin(a[outside])]

        improved = False
        for delta in [0.0003, 0.0008, 0.002, 0.005, 0.01, 0.02]:
            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min] += delta
            candidate = np.maximum(candidate, 0.0)
            ss = candidate.sum()
            if ss < 1e-15:
                continue
            candidate = candidate / ss
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                elif stale > 800:
                    break

    return best_a, best_obj


def fourier_refine(a0, time_limit, n_modes=12):
    """Optimize over Fourier coefficients for structured improvement."""
    from scipy.optimize import minimize as sp_minimize
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    M = min(n_modes, n // 2)
    x = np.arange(n, dtype=np.float64)

    a_c = a - 1.0 / n
    init_coeffs = np.zeros(2 * M)
    for m in range(1, M + 1):
        init_coeffs[2 * (m - 1)] = 2.0 * np.sum(a_c * np.cos(2.0 * np.pi * m * x / n))
        init_coeffs[2 * (m - 1) + 1] = 2.0 * np.sum(a_c * np.sin(2.0 * np.pi * m * x / n))

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()

    def seq_from_coeffs(coeffs):
        base = np.ones(n) / n
        for m in range(1, M + 1):
            c = coeffs[2 * (m - 1)]
            d = coeffs[2 * (m - 1) + 1]
            base += (c * np.cos(2.0 * np.pi * m * x / n) +
                     d * np.sin(2.0 * np.pi * m * x / n)) / n
        base = np.maximum(base, 0.0)
        s = base.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return base / s

    def obj_fun(coeffs):
        seq = seq_from_coeffs(coeffs)
        return fast_convolve(seq).max()

    n_restarts = max(2, int(time_limit // 25))
    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        if r == 0:
            x0 = init_coeffs.copy()
        else:
            rng = np.random.default_rng(r * 12345 + 7)
            scale = 0.003 * (1 + r * 0.4)
            x0 = init_coeffs + rng.standard_normal(2 * M) * scale

        try:
            res = sp_minimize(obj_fun, x0, method='Nelder-Mead',
                              options={'maxiter': 5000, 'xatol': 1e-12,
                                       'fatol': 1e-14, 'maxfev': 30000})
            if res.fun < best_obj:
                best_obj = res.fun
                best_a = seq_from_coeffs(res.x)
                print("    Fourier restart {}: obj={:.10f}".format(r, best_obj))
        except Exception:
            pass

    return best_a, best_obj


def upsample(a, new_n):
    """Upsample a sequence to new length via interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def cosine_window(n, taper_frac=0.15):
    """Generate a cosine-tapered window."""
    x = np.arange(n, dtype=np.float64)
    w = np.ones(n)
    taper = int(n * taper_frac)
    if taper > 0 and taper < n // 2:
        for i in range(taper):
            w[i] = 0.5 * (1 + np.cos(np.pi * (i - taper + 1) / taper))
            w[n - 1 - i] = 0.5 * (1 + np.cos(np.pi * (i - taper + 1) / taper))
    return w / w.sum()


def plateau_window(n, plateau_frac=0.7, taper_frac=0.15):
    """Plateau window with cosine taper."""
    x = np.arange(n, dtype=np.float64)
    w = np.ones(n)
    taper = int(n * taper_frac)
    if taper > 0 and taper < n // 2:
        for i in range(taper):
            w[i] = 0.5 * (1 + np.cos(np.pi * (i - taper + 1) / taper))
            w[n - 1 - i] = 0.5 * (1 + np.cos(np.pi * (i - taper + 1) / taper))
    return w / w.sum()


def eval_score(a):
    """Compute the objective score internally."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    return float(2.0 * len(a_norm) * fast_convolve(a_norm).max())


def try_and_update(a, state, label=""):
    """Evaluate and update best state."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15 or len(a) == 0:
        return
    a_norm = a / s
    score = eval_score(a_norm)
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a_norm)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Multi-scale initialization and quick n exploration
    Phase 2: Deep FW-LP optimization
    Phase 3: Fourier-parametrized structured refinement
    Phase 4: Smoothed gradient + multi-peak subgradient
    Phase 5: Targeted peak reduction + annealing restarts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = eval_score(parent)
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = cosine_window(768)
        state = [eval_score(parent), parent.copy()]
        print("No parent, using cosine window n=768, score={:.8f}".format(state[0]))

    n_parent = len(parent)

    # === Phase 1: Multi-scale + n exploration (~150s) ===
    print("\n=== Phase 1: Multi-scale & n Exploration ===")
    t1 = time.monotonic()

    # Try several sequence lengths
    candidate_ns = sorted(set([n_parent, 512, 640, 768, 896, 1024, 1152]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 1400]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1400
        if remaining < 10:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = upsample(parent, n_try)

        t_alloc = min(remaining, 25)
        print("  FW n={}, t={:.0f}s".format(n_try, t_alloc))
        a_opt, _ = fw_optimize(start, time_limit=t_alloc)
        try_and_update(a_opt, state, "fw_n{}".format(n_try))

        # Quick smoothed gradient
        remaining = deadline - time.monotonic() - 1350
        if remaining > 8:
            a_opt2, _ = smoothed_grad_descent(a_opt, time_limit=min(remaining, 10),
                                              T_init=0.001, lr_init=0.03)
            try_and_update(a_opt2, state, "sg_n{}".format(n_try))

    print("  Phase 1 done: best={:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t1))

    # === Phase 2: Deep FW-LP on best n (~500s) ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t2 = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        remaining = deadline - time.monotonic() - 900
        if remaining > 30:
            t_alloc = min(remaining, 400)
            print("  Deep FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "deep_fw")

            # Perturbed restarts from best
            for scale in [0.004, 0.01, 0.02]:
                remaining = deadline - time.monotonic() - 650
                if remaining < 10:
                    break
                f_pert = project_simplex(state[1] + rng.standard_normal(n_best) * scale)
                t_alloc = min(remaining, 50)
                a_opt2, _ = fw_optimize(f_pert, time_limit=t_alloc)
                try_and_update(a_opt2, state, "fw_pert_{:.3f}".format(scale))

    print("  Phase 2 done: best={:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t2))

    # === Phase 3: Fourier refinement (~200s) ===
    print("\n=== Phase 3: Fourier Refinement ===")
    t3 = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 500
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print("  Fourier (12 modes), t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_modes=12)
            try_and_update(a_opt, state, "fourier_12")

        remaining = deadline - time.monotonic() - 300
        if remaining > 15:
            t_alloc = min(remaining, 80)
            print("  Fourier (20 modes), t={:.0f}s".format(t_alloc))
            a_opt, _ = fourier_refine(state[1], time_limit=t_alloc, n_modes=20)
            try_and_update(a_opt, state, "fourier_20")

    print("  Phase 3 done: best={:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t3))

    # === Phase 4: Gradient methods (~250s) ===
    print("\n=== Phase 4: Gradient Methods ===")
    t4 = time.monotonic()

    if state[1] is not None:
        # Multiple temperature settings for smoothed gradient
        configs = [
            (0.0003, 0.04, 80),
            (0.001, 0.03, 60),
            (0.003, 0.02, 50),
            (0.0001, 0.02, 40),
        ]
        for T_init, lr_init, max_t in configs:
            remaining = deadline - time.monotonic() - 150
            if remaining < 10:
                break
            t_alloc = min(remaining, max_t)
            print("  Smoothed grad T={:.4f} lr={:.3f}, t={:.0f}s".format(T_init, lr_init, t_alloc))
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=T_init, lr_init=lr_init)
            try_and_update(a_opt, state, "smooth_T{:.4f}".format(T_init))

        # Multi-peak subgradient
        sg_configs = [
            (10, 0.002, 0.94, 50),
            (30, 0.003, 0.92, 40),
            (50, 0.001, 0.95, 30),
        ]
        for top_k, lr, mom, max_t in sg_configs:
            remaining = deadline - time.monotonic() - 80
            if remaining < 8:
                break
            t_alloc = min(remaining, max_t)
            print("  Multi-peak SG k={}".format(top_k))
            a_opt, _ = multi_peak_sg(state[1], time_limit=t_alloc,
                                     top_k=top_k, lr_init=lr, mom=mom)
            try_and_update(a_opt, state, "mpsg_k{}".format(top_k))

    print("  Phase 4 done: best={:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t4))

    # === Phase 5: Targeted reduction + annealing (~200s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if state[1] is not None:
        # Targeted peak reduction
        remaining = deadline - time.monotonic() - 120
        if remaining > 10:
            t_alloc = min(remaining, 80)
            print("  Targeted peak reduction, t={:.0f}s".format(t_alloc))
            a_opt, _ = targeted_peak_reduction(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "targeted")

        # Final FW pass
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            t_alloc = min(remaining, 50)
            print("  Final FW-LP, t={:.0f}s".format(t_alloc))
            a_opt, _ = fw_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "final_fw")

        # Very fine smoothed gradient
        remaining = deadline - time.monotonic() - 40
        if remaining > 5:
            t_alloc = min(remaining, 25)
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=t_alloc,
                                             T_init=0.00005, lr_init=0.005)
            try_and_update(a_opt, state, "fine_smooth")

    print("  Phase 5 done: best={:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t5))

    # === Phase 6: Annealing restarts with remaining time ===
    print("\n=== Phase 6: Annealing Restarts ===")
    t6 = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])
        # Try larger perturbations and re-optimize
        anneal_scales = [0.05, 0.1, 0.15, 0.2, 0.3]
        for asc in anneal_scales:
            remaining = deadline - time.monotonic() - 30
            if remaining < 20:
                break
            n_restarts = max(1, int(remaining // 25))
            for r in range(n_restarts):
                remaining = deadline - time.monotonic() - 15
                if remaining < 15:
                    break
                rng2 = np.random.default_rng(hash((seed, asc, r)) % (2**31))
                perturbed = project_simplex(state[1] + rng2.standard_normal(n_best) * asc)
                t_sub = min(remaining - 10, 15)
                if t_sub < 3:
                    break
                # Quick FW
                a_opt, _ = fw_optimize(perturbed, time_limit=min(t_sub, 10))
                try_and_update(a_opt, state, "anneal_{:.1f}_r{}".format(asc, r))
                # Quick smoothed grad
                remaining2 = deadline - time.monotonic() - 10
                if remaining2 > 5:
                    a_opt2, _ = smoothed_grad_descent(a_opt, time_limit=min(remaining2, 5),
                                                      T_init=0.001, lr_init=0.02)
                    try_and_update(a_opt2, state, "anneal_{:.1f}_r{}_sg".format(asc, r))

    print("  Phase 6 done: best={:.8f} (elapsed {:.0f}s)".format(
        state[0], time.monotonic() - t6))

    # === Final: one more round of fine optimization ===
    if state[1] is not None:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_opt, _ = targeted_peak_reduction(state[1], time_limit=min(remaining, 10))
            try_and_update(a_opt, state, "final_target")
        remaining = deadline - time.monotonic()
        if remaining > 3:
            a_opt, _ = smoothed_grad_descent(state[1], time_limit=min(remaining, 5),
                                             T_init=0.00003, lr_init=0.003)
            try_and_update(a_opt, state, "final_fine")

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = eval_score(state[1])

    # Final verification with actual evaluate_sequence
    best_list = [float(max(0.0, x)) for x in state[1]]
    fs = float(evaluate_sequence(best_list))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs
        print("  Verified with evaluate_sequence: {:.8f}".format(fs))

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(max(0.0, x)) for x in state[1]]
