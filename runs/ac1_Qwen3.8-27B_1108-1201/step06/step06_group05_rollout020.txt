def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()

2. Active-set KKT refinement

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def hankel_mv(p, a):
    """Compute the vector g_j = sum_k p_k * a[k-j] using correlate."""
    return np.correlate(p, a, mode='valid')


def lse_objective(z, T, n):
    """Log-sum-exp objective and gradient. Returns (value, grad)."""
    a = softmax_safe(z)
    conv = fast_convolve(a)
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / T
    p = np.exp(shifted)
    p_sum = p.sum()
    p = p / p_sum
    obj = 2.0 * n * (max_conv + T * np.log(p_sum))
    g = 2.0 * hankel_mv(p, a)
    ga = np.dot(g, a)
    grad_z = a * (g - ga)
    return obj, grad_z


def lse_lbfgs(z0, T, n, time_limit=60):
    """Run L-BFGS on the log-sum-exp objective."""
    t0 = time.monotonic()
    counter = [0]
    
    def fun(z):
        counter[0] += 1
        obj, grad = lse_objective(z, T, n)
        return obj, grad
    
    try:
        res = minimize(fun, z0, method='L-BFGS-B', jac=True,
                       options={'maxiter': 10000, 'maxls': 50, 'iprint': -1,
                                'ftol': 1e-15, 'gtol': 1e-14, 'btol': 1e-12})
        return res.x
    except Exception:
        return z0


def z_from_a(a, eps=1e-15):
    """Convert a (sum=1, a>=0) to z (softmax parameterization)."""
    return np.log(np.maximum(a, eps))


def a_from_z(z):
    """Convert z to a (sum=1, a>0)."""
    return softmax_safe(z)


def active_set_refine(a0, time_limit=60, tol=1e-10):
    """
    Active-set refinement: identify active peaks and solve KKT system.
    """
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()

        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        threshold = M * (1.0 - 0.0001)
        active_peaks = np.where(conv >= threshold)[0]
        
        if len(active_peaks) < 2:
            break
        
        # Support of a (entries above threshold)
        support_thresh = 1e-12
        support = np.where(a > support_thresh)[0]
        if len(support) < 2:
            break
        
        # Build KKT system
        # For j in support: sum_{k in active} mu_k * a[k-j] = nu
        # We solve: A @ [mu; nu] = 0 (SVD for smallest singular vector)
        m = len(support)
        p_count = len(active_peaks)
        A_sys = np.zeros((m, p_count + 1))
        for i, j in enumerate(support):
            A_sys[i, -1] = -1.0  # -nu column
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = a[idx]
        
        try:
            U, S_vals, Vt = np.linalg.svd(A_sys)
            sol = Vt[-1, :]
            mu_vals = sol[:p_count]
        except np.linalg.LinAlgError:
            break
        
        mu_vals = np.maximum(mu_vals, 0)
        mu_sum = mu_vals.sum()
        if mu_sum < 1e-15:
            break
        mu_vals /= mu_sum
        
        # Gradient step
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu_vals[k_idx] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= mu_vals[k_idx] * a[k - idx_arr]
        
        # Projected step
        best_candidate = a
        best_val = M
        for step in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02]:

            candidate = np.maximum(candidate, 0)
            cs = candidate.sum()

            candidate = candidate / cs
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break
        
        a = best_candidate
        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
        else:
            # Try smaller steps
            improved = False
            for step in [0.0001, 0.0002, 0.00005]:

                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs

                    best_obj = val
                    best_a = candidate.copy()
                    a = candidate.copy()
                    improved = True
                    break
            if not improved:

    return best_a

def interp_toLength(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage approach:
1. Log-sum-exp smooth approximation with L-BFGS and temperature annealing
2. Active-set KKT refinement with adaptive peak selection
3. Coordinate descent and perturbation-based local search
4. Multi-n exploration for global improvements

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def hankel_mv(p, a):
    """Compute vector g_j = sum_k p_k * a[k-j] using correlate."""
    return np.correlate(p, a, mode='valid')


def lse_objective(z, T, n):
    """Log-sum-exp objective and gradient w.r.t. z. Returns (value, grad)."""
    a = softmax_safe(z)
    conv = fast_convolve(a)
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / T
    p = np.exp(shifted)
    p_sum = p.sum()
    p = p / p_sum
    obj = 2.0 * n * (max_conv + T * np.log(p_sum))
    g = 2.0 * hankel_mv(p, a)
    ga = np.dot(g, a)
    grad_z = a * (g - ga)
    return obj, grad_z


def lse_lbfgs(z0, T, n, time_limit=60):
    """Run L-BFGS on the log-sum-exp objective with time limit."""
    t0 = time.monotonic()
    call_count = [0]

    def fun(z):
        call_count[0] += 1
        if time.monotonic() - t0 > time_limit:
            raise RuntimeError("time limit")
        return lse_objective(z, T, n)

    try:
        res = minimize(fun, z0, method='L-BFGS-B', jac=True,
                       options={'maxiter': 10000, 'maxls': 50, 'iprint': -1,
                                'ftol': 1e-16, 'gtol': 1e-14, 'btol': 1e-13})
        return res.x
    except Exception:
        return z0


def z_from_a(a, eps=1e-15):
    """Convert a (sum=1, a>=0) to z (softmax parameterization)."""
    return np.log(np.maximum(a, eps))


def a_from_z(z):
    """Convert z to a (sum=1, a>0)."""
    return softmax_safe(z)


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def active_set_refine(a0, time_limit=100, tol=1e-12):
    """
    Active-set KKT refinement with adaptive peak selection.
    Tries multiple active set sizes and uses SVD for direction finding.
    """
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit and it < 1000:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 200:
                break

        # Try multiple active set thresholds
        for frac in [0.0001, 0.0005, 0.001, 0.005, 0.01, 0.02]:
            if time.monotonic() - t0 > time_limit:
                break
            threshold = M * (1.0 - frac)
            active_peaks = np.where(conv >= threshold)[0]
            if len(active_peaks) < 2 or len(active_peaks) > 100:
                continue

            support = np.where(a > 1e-10)[0]
            if len(support) < 3:
                continue

            m = len(support)
            p_count = len(active_peaks)

            # Build KKT matrix
            A_sys = np.zeros((m, p_count + 1))
            for i, j in enumerate(support):
                A_sys[i, -1] = -1.0
                for k_idx in range(p_count):
                    k = active_peaks[k_idx]
                    idx = k - j
                    if 0 <= idx < n:
                        A_sys[i, k_idx] = a[idx]

            try:
                U, S_vals, Vt = np.linalg.svd(A_sys)
                sol = Vt[-1, :]
            except np.linalg.LinAlgError:
                continue

            mu_vals = np.maximum(sol[:p_count], 0)
            mu_sum = mu_vals.sum()
            if mu_sum < 1e-14:
                continue
            mu_vals /= mu_sum

            # Compute gradient direction
            grad = np.zeros(n)
            for k_idx in range(p_count):
                if mu_vals[k_idx] < 1e-14:
                    continue
                k = active_peaks[k_idx]
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx_arr = np.arange(js, je + 1)
                grad[idx_arr] -= mu_vals[k_idx] * a[k - idx_arr]

            # Line search
            improved = False
            for step in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.0005, 0.002]:
                candidate = a + step * grad
                candidate = np.maximum(candidate, 0.0)
                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs
                val = fast_convolve(candidate).max()
                if val < M - 1e-14:
                    a = candidate
                    improved = True
                    break
            if improved:
                break

    return best_a


def subgradient_optimize(a0, time_limit=100, top_k=20):
    """Subgradient optimization with adaptive top-k and momentum."""
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    lr = 0.02
    mom = 0.9
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        # Top-k peaks
        k_use = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_use)[-k_use:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(k_use)
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.3
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        # Subgradient
        sg = np.zeros(n)
        for i in range(k_use):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]

        # Momentum step with backtracking
        v_new = mom * v - lr * sg
        best_candidate = a
        best_val = obj
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v_new)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                v = v_new
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        if no_improve > 30:
            lr *= 0.95
            mom = min(0.97, mom + 0.002)
        if no_improve > 60:
            top_k = min(50, top_k + 3)
        if no_improve > 150:
            rng_local = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.003 + 0.003 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng_local.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = 0.02
            mom = 0.9
            top_k = 20
            no_improve = 0

    return best_a


def interp_toLength(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def try_update(a, best_seq, best_score, label=""):

    a = np.maximum(a, 0.0)

    a = a / s

        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score


def fw_lp_optimize(f0, time_limit=100):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        g = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'max_time': 20})
            if res.success:
                g = np.maximum(res.x, 0.0)
        except Exception:
            pass

        if g is None or g.sum() < 1e-15:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_f

            break
        a_pert = best_seq + rng.standard_normal(n_best) * scale
        a_pert = project_simplex(a_pert)
        # Quick optimize
        z = z_from_a(a_pert)
        z = lse_lbfgs(z, 0.0005, n_best, time_limit=min(remaining, 15))
        a_opt = a_from_z(z)
        best_seq, best_score = try_update(a_opt, best_seq, best_score,
                                           f"pert_{scale}")
        # Active-set on perturbed
        remaining = deadline - time.monotonic() - 40
        if remaining > 5:
            a_opt2 = active_set_refine(a_opt, time_limit=min(remaining, 10))
            best_seq, best_score = try_update(a_opt2, best_seq, best_score,
                                               f"pert_as_{scale}")

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Final intensive refinement (200s) ===
    print("\n=== Phase 5: Final Refinement ===")
    t_start = time.monotonic()

    n_best = len(best_seq)

    # Cycle: L-BFGS -> Active-Set -> L-BFGS -> Active-Set
    for cycle in range(3):
        remaining = deadline - time.monotonic() - 30
        if remaining < 10:
            break

        z = z_from_a(best_seq)
        T_final = 0.0001 / (cycle + 1)
        z = lse_lbfgs(z, T_final, n_best, time_limit=min(remaining, 40))
        a_opt = a_from_z(z)
        best_seq, best_score = try_update(a_opt, best_seq, best_score,
                                           f"final_cycle{cycle}_lbfgs")

        remaining = deadline - time.monotonic() - 25
        if remaining < 5:
            break
        a_opt = active_set_refine(best_seq, time_limit=min(remaining, 30))
        best_seq, best_score = try_update(a_opt, best_seq, best_score,
                                           f"final_cycle{cycle}_as")

    # Final symmetry check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    best_seq, best_score = try_update(sym_final, best_seq, best_score, "final_sym")

    # One more FW if time allows
    remaining = deadline - time.monotonic() - 15
    if remaining > 10:
        a_opt = fw_lp_optimize(best_seq, time_limit=min(remaining, 20))
        best_seq, best_score = try_update(a_opt, best_seq, best_score,
                                           "final_fw")

    # Final active-set
    remaining = deadline - time.monotonic() - 10
    if remaining > 5:
        a_opt = active_set_refine(best_seq, time_limit=min(remaining, 15))
        best_seq, best_score = try_update(a_opt, best_seq, best_score,
                                           "ultimate_as")

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
