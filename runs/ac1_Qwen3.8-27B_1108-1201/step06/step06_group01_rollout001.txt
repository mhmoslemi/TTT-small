def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_toeplitz(f, n):
    """Build the Toeplitz constraint matrix for the LP."""
    # A_ub shape: (2n-1, n)
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # f[j] appears in row k=j (column 0), row k=j+1 (column 1), ..., row k=j+n-1 (column n-1)
        # Wait, A[k,j] = f[k-j] if valid
        # So for fixed j, A[k,j] = f[k-j] for k from j to j+n-1
        # That is, column j has f[0] at row j, f[1] at row j+1, ..., f[n-1] at row j+n-1
        A[j:j+n, j] = f
    return A

    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',

            g = np.maximum(res.x, 0.0)
            dual_y = res.ineqlin.marginals  # Shadow prices (dual variables)

        pass
    return None, None

            noise = rng.standard_normal(n) * 0.001

            if dual_y is not None and len(dual_y) > 0:
                peaks = np.where(conv > M * 0.95)[0]
                for k in peaks[:5]:
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    noise[js:je + 1] -= 0.0005
            
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            
            if no_improve > 60:
                break
    
    return best_f, best_obj

def dual_subgradient_descent(a0, time_limit, lr=0.01):
    """Subgradient descent using dual-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    current_obj = best_obj
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Get softmax-weighted peaks
        T = 0.001 * n  # Temperature scales with n
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Compute subgradient: grad[j] = 2 * sum_k w[k] * a[k-j] (if valid)
        # More efficiently: grad = 2 * (w * a) correlated
        # grad[j] = 2 * sum_k w[k] * a[k-j] where 0<=k-j<n
        # This is 2 * convolve(w, a, mode='valid') but we need to be careful
        # Actually: grad[j] = 2 * sum_{k: 0<=k-j<n} w[k] * a[k-j]
        # Let m = k-j, so k = j+m, and we need 0<=j+m<2n-1 and 0<=m<n
        # grad[j] = 2 * sum_{m=max(0,-j)}^{min(n-1, 2n-2-j)} w[j+m] * a[m]
        # This is just the cross-correlation of w and a, evaluated at position j
        
        # Use the Hessian-free approach:
        # grad[j] = 2 * sum_{k} w[k] * (if 0<=k-j<n then a[k-j] else 0)
        
        grad = np.zeros(n)
        # For each k, add w[k] * a[k-j] to grad[j] for valid j
        # j ranges from max(0, k-n+1) to min(n-1, k)
        for k in range(len(w)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w[k] * a[k - (js:je + 1)]
        
        grad *= 2.0
        
        # Step
        candidate = project_simplex(a - lr * grad)
        c = fast_convolve(candidate)
        val = c.max()
        
        if val < current_obj - 1e-15:
            a = candidate
            current_obj = val
            if val < best_obj - 1e-14:
                best_obj = val
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Reduce learning rate
            lr *= 0.95
        
        if no_improve > 50:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            current_obj = fast_convolve(a).max()
            no_improve = 0
            lr = min(lr * 2, 0.05)
    
    return best_a, best_obj

def coordinate_descent_polish(a0, time_limit, max_peaks=3):
    """
    Targeted pairwise transfer coordinate descent.
    For each top peak, find the best pair (j1, j2) to transfer mass,
    minimizing the resulting max convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Get top peaks
        top_k = min(max_peaks, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]
        
        improved = False
        
        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            
            kp = int(k_peak)
            # Variables affecting this peak
            js = max(0, kp - n + 1)
            je = min(n - 1, kp)
            
            if je - js + 1 < 2:
                continue
            
            # Find best transfer
            best_gain = 1e-15
            best_j1 = -1
            best_j2 = -1
            best_delta = 0.0
            
            # Sample affected indices
            affected = np.arange(js, je + 1)
            if len(affected) > 20:
                rng_sample = np.random.default_rng((it + kp) * 31)
                affected = affected[rng_sample.choice(len(affected), size=20, replace=False)]
            
            # Try transfers
            for ii in range(len(affected)):
                j1 = affected[ii]
                for jj in range(len(affected)):
                    if jj == ii:
                        continue
                    j2 = affected[jj]
                    
                    # a[j1] contributes a[j1]*a[kp-j1] to conv[kp]
                    # a[j2] contributes a[j2]*a[kp-j2] to conv[kp]
                    c1 = a[j1] * (a[kp - j1] if 0 <= kp - j1 < n else 0)
                    c2 = a[j2] * (a[kp - j2] if 0 <= kp - j2 < n else 0)
                    
                    # We want to move mass from the one with higher "partner" value
                    partner1 = a[kp - j1] if 0 <= kp - j1 < n else 0
                    partner2 = a[kp - j2] if 0 <= kp - j2 < n else 0
                    
                    # Moving from j1 to j2 reduces conv[kp] if partner1 > partner2
                    # (approximately, for small delta)
                    if partner1 > partner2 and a[j1] > 1e-8:
                        # Try moving from j1 to j2
                        for frac in [0.05, 0.1, 0.2, 0.3]:
                            delta = a[j1] * frac
                            a_test = a.copy()
                            a_test[j1] -= delta
                            a_test[j2] += delta
                            a_test = np.maximum(a_test, 0.0)
                            s = a_test.sum()
                            if s < 1e-15:
                                continue
                            a_test = a_test / s
                            c_new = fast_convolve(a_test)
                            new_max = c_new.max()
                            gain = M - new_max
                            if gain > best_gain:
                                best_gain = gain
                                best_j1 = j1
                                best_j2 = j2
                                best_delta = delta
        
            if best_j1 >= 0 and best_gain > 1e-12:
                a[best_j1] -= best_delta
                a[best_j2] += best_delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                c_new = fast_convolve(a)
                M = c_new.max()
                if M < best_obj - 1e-14:
                    best_obj = M
                    best_a = a.copy()
                improved = True
                stall = 0
        
        if not improved:
            stall += 1
        
        if stall > 30:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise_scale = 0.0005 + 0.001 * min(1.0, stall / 60.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            stall = 0
        
        if stall > 200:
            break
    
    return best_a, best_obj

def micro_polish(a0, time_limit):
    """Very fine perturbation-based polish."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.2:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Try a few small perturbations
        rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))
        found = False
        
        for scale in [0.0001, 0.0003, 0.001]:
            for _ in range(5):
                if time.monotonic() - t0 > time_limit - 0.3:
                    break
                noise = rng.standard_normal(n) * scale
                # Bias: reduce values contributing to the main peak
                peak_k = int(np.argmax(conv))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                for j in range(js, je + 1):
                    partner = peak_k - j
                    if 0 <= partner < n:
                        noise[j] -= scale * a[partner]
                
                candidate = project_simplex(a + noise)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-16:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    found = True
                    break
            if found:
                break
    
    return best_a, best_obj
