"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Multi-scale initialization (coarse to fine)
- Primal-dual QP iteration with robust active-set solver
- Log-sum-exp smoothed gradient with continuation
- Fast coordinate descent for final polish

import numpy as np
import time
import scipy.optimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]."""
    return np.correlate(w, a, mode='valid')


def solve_qp_active_set(w, a_init, n, max_iters=500, time_limit=5.0):
    """
    Solve min a^T W a s.t. sum(a)=1, a>=0 using active set.
    W is Hankel: W_{i,j} = w[i+j], w has length 2n-1.
    """
    t0 = time.monotonic()
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Start with all coordinates active
    active = set(range(n))

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) == 0:
            return np.ones(n) / n

        act = np.array(sorted(active))
        k = len(act)

        if k == 1:
            a = np.zeros(n)
            a[act[0]] = 1.0
            return a

        # Build submatrix
        W_AA = np.array([[w[act[i] + act[j]] for j in range(k)] for i in range(k)])
        W_AA += 1e-10 * np.eye(k)  # Regularization

        # Solve W_AA x = 1
        try:
            x = np.linalg.solve(W_AA, np.ones(k))
        except np.linalg.LinAlgError:
            x = np.linalg.lstsq(W_AA, np.ones(k), rcond=None)[0]

        x_sum = x.sum()
        if abs(x_sum) < 1e-14 or x_sum < 0:
            # Degenerate: use least squares or uniform
            x = np.ones(k) / k
            x_sum = 1.0

        a_act = x / x_sum

        # Check for negative components
        min_idx_in_act = np.argmin(a_act)
        if a_act[min_idx_in_act] < -1e-10:
            # Remove the most negative
            active.discard(act[min_idx_in_act])
            continue

        # Clip small negatives to zero
        a_act = np.maximum(a_act, 0.0)
        a_act = a_act / a_act.sum()

        # Form full solution
        a = np.zeros(n)
        a[act] = a_act

        # Compute gradient
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa

        # Mu: Lagrange multiplier for sum constraint
        mu = np.mean(grad[act])

        # Check KKT for inactive
        inactive = np.setdiff1d(np.arange(n), act)
        if len(inactive) > 0:
            viol = grad[inactive] - mu
            min_viol = viol.min()
            if min_viol < -1e-6:
                # Add the most violated
                active.add(inactive[np.argmin(viol)])
                continue

        # Converged
        return a

    return a


def primal_dual_qp(a0, time_limit, T_init=0.01, blend=0.5, verbose=False):
    """Primal-dual QP iteration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 1:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        conv = fast_convolve(a)
        cmax = conv.max()

        # Dual update: softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()

        # Solve QP
        qp_time = min(3.0, remaining * 0.4)
        a_qp = solve_qp_active_set(w, a, n, max_iters=200, time_limit=qp_time)

        # Blend
        a_new = (1.0 - blend) * a + blend * a_qp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a_qp
        else:
            a_new = a_new / s_new

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adapt temperature
        if no_improve > 10:
            T *= 1.2
        if no_improve > 40:
            T *= 0.7
        if no_improve > 100:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            no_improve = 0

    return best_a, best_obj


def smoothed_gradient_continuation(a0, time_limit, verbose=False):
    """
    Log-sum-exp smoothed gradient with continuation on temperature.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    # Continuation schedule: T from 1.0 down to 0.001
    T_values = np.logspace(0, -3, 30)  # 1.0, 0.8, ..., 0.001

    for T in T_values:
        if time.monotonic() - t0 > time_limit - 2:
            break
        it_T = 0
        max_it_T = 200
        lr = 0.5 / n  # Initial learning rate

        for it_T in range(max_it_T):
            if time.monotonic() - t0 > time_limit - 2:
                break

            conv = fast_convolve(a)
            scaled = conv / T
            scaled -= scaled.max()
            p = np.exp(scaled)
            p = p / p.sum()

            # Gradient of log-sum-exp: sum_k p[k] * grad b[k]
            # grad b[k][i] = 2 * a[k-i] (for valid i)
            grad = 2.0 * hankel_mv(p, a)

            # Backtracking line search
            best_candidate = a
            best_val = conv.max()
            step = lr
            for _ in range(20):
                candidate = project_simplex(a - step * grad)
                val = fast_convolve(candidate).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_candidate = candidate
                    break
                step *= 0.5

            a = best_candidate
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()

            # Slight LR decay
            lr *= 0.999

    return best_a
