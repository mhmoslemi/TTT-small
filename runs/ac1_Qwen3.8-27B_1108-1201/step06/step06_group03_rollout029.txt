    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

    n = len(a)
    if top_k >= len(conv):

import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_value(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * fast_convolve(a).max() / (s*s)

def smoothed_gradient(a, conv, top_k=10, temp=0.005):
    n = len(a)
    k = min(top_k, len(conv))
    top_idx = np.argpartition(conv, -k)[-k:]
    vals = conv[top_idx]
    w = np.exp((vals - vals.max()) / temp)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(k):
        ki = int(top_idx[i])
        if w[i] < 1e-15:
            continue
        lo = max(0, ki - (n-1))
        hi = min(n-1, ki)
        idx = np.arange(lo, hi+1)
        grad[idx] += w[i] * 2.0 * a[ki - idx]
    return grad

def project_nonneg_sum(a, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), target_sum / len(a))
    return a * (target_sum / s)

def optimize_smoothed(a0, time_limit, top_k=10, temp_init=0.005, lr_init=0.01):

    a = a0.copy().astype(np.float64)
    a = project_nonneg_sum(a)
    n = len(a)

def simulated_annealing(a0, time_limit, n_iter_max=50000):
    """
    Simulated annealing on the simplex to escape local minima.
    Uses structured moves: mass transfer, pairwise swaps, and rescaling.
    """
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)

    t0 = time.monotonic()
    T = 0.01
    T_min = 0.0001
    cool_rate = 0.9997
    rng = np.random.default_rng(12345)

    it = 0
    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Choose move type
        move_type = rng.integers(0, 3)

        if move_type == 0:
            # Mass transfer: move small amount from one coord to another
            src = int(rng.integers(n))
            tgt = int(rng.integers(n))
            if src == tgt or a[src] < 1e-10:
                continue
            scale = rng.uniform(0.001, 0.05)
            delta = a[src] * scale
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand = normalize(cand)

        elif move_type == 1:
            # Pairwise swap: swap portions of two coordinates
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            scale = rng.uniform(0.01, 0.3)
            delta = min(a[i], a[j]) * scale
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = normalize(cand)

        else:
            # Gaussian perturbation
            scale = rng.uniform(0.002, 0.01) * (T / T)
            cand = normalize(a + rng.standard_normal(n) * scale)

        cand_obj = obj_value(cand)
        delta_obj = cand_obj - obj_value(a)

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            a = cand
            if cand_obj < best_obj:
                best_obj = cand_obj
                best_a = cand.copy()

        T = max(T * cool_rate, T_min)

    return best_a, best_obj

                a_pert = normalize(best_seq + rng.standard_normal(n_best) * 0.01)
                t_alloc2 = min(remaining2, 80)
                a_opt2, _ = simulated_annealing(a_pert, time_limit=t_alloc2)
                try_seq(a_opt2, "sim_anneal2")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final Micro-Polish (remaining) ===
    print("\n=== Phase 5: Final Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Alternate very short optimizations
        polish_rounds = 8
        for ri in range(polish_rounds):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break

            if ri % 2 == 0:
                # Tiny perturb + gradient
                noise = rng.standard_normal(n_best) * 0.001
                start = normalize(best_seq + noise)
                t_alloc = min(remaining, 15)
                a_opt, _ = optimize_gradient(start, time_limit=t_alloc,
                                             top_k=15, temp_init=0.001, lr_init=0.003)
                try_seq(a_opt, f"polish_g_{ri}")
            else:
                # Tiny perturb + coord descent
                noise = rng.standard_normal(n_best) * 0.0005
                start = normalize(best_seq + noise)
                t_alloc = min(remaining, 10)
                a_opt, _ = coordinate_descent_peaks(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_c_{ri}")

        # Final coord descent
        remaining = deadline - time.monotonic()
        if remaining > 5:
            t_alloc = min(remaining, 20)
            a_opt, _ = coordinate_descent_peaks(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_coord")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
