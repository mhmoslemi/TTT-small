    n = len(a)

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient(a, peaks, weights):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)

        if len(affected) == 0:
            break

        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            outside = np.array([0])
        j_min_cand = outside[np.argmin(a[outside])]

        improved = False
        for delta in [0.0003, 0.0005, 0.001, 0.002, 0.003, 0.005, 0.01]:
            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min_cand] += delta
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0 and stale < 500:
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
            elif stale >= 500:
                break

    return best_a, best_obj


def interleave_and_optimize(a_parent, n_target, time_limit_fw, time_limit_sg):
    """Resample parent to n_target and optimize."""
    n_p = len(a_parent)
    a = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_p), a_parent)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n_target) / n_target

    # Nesterov smoothed descent
    a1, obj1 = nesterov_smoothed_descent(a, time_limit_fw, T_init=0.002, lr_init=0.04, n_top=30)

    # Subgradient top-k
    a2, obj2 = subgradient_topk_descent(a1, min(time_limit_sg, 120), lr_init=0.003, n_peaks=15)

    return a2 if obj2 < obj1 else a1


def try_and_update(a, state, label):
    """Evaluate a sequence and update state if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Quick multi-n exploration
    Phase 2: Deep Nesterov + subgradient on best n
    Phase 3: Structured perturbations + restarts
    Phase 4: Targeted peak reduction + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        state = [float(evaluate_sequence(parent.tolist())), parent.copy()]

    n_parent = len(parent)

    # === Phase 1: Quick Multi-n Exploration (~150s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, n_parent - 64, n_parent + 64, n_parent + 128, 896, 1024]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        t_alloc = min(remaining, 25)
        print("  Exploring n={}, t={:.0f}s".format(n_try, t_alloc))

        a1, obj1 = nesterov_smoothed_descent(start, time_limit=t_alloc, T_init=0.002, lr_init=0.03, n_top=20)
        try_and_update(a1, state, "expl_n{}".format(n_try))

        remaining = deadline - time.monotonic() - 1150
        if remaining > 15:
            a2, obj2 = subgradient_topk_descent(a1, min(remaining, 10), lr_init=0.002, n_peaks=10)
            try_and_update(a2, state, "sg_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Deep Optimization on Best n (~700s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])

        # Deep Nesterov with multiple temperature settings
        for T_init, lr_init, n_top in [(0.001, 0.05, 30), (0.003, 0.03, 20), (0.0005, 0.04, 50)]:
            remaining = deadline - time.monotonic() - 600
            if remaining < 30:
                break
            t_alloc = min(remaining, 150)
            print("  Nesterov T={}, lr={}, k={}, t={:.0f}s".format(T_init, lr_init, n_top, t_alloc))
            a_opt, _ = nesterov_smoothed_descent(state[1], time_limit=t_alloc,
                                                 T_init=T_init, lr_init=lr_init, n_top=n_top)
            try_and_update(a_opt, state, "nest_T{}".format(T_init))

        # Subgradient with various peak counts
        for n_peaks, lr in [(8, 0.004), (15, 0.003), (25, 0.002)]:
            remaining = deadline - time.monotonic() - 400
            if remaining < 20:
                break
            t_alloc = min(remaining, 100)
            print("  Subgrad k=peak{}, lr={}, t={:.0f}s".format(n_peaks, lr, t_alloc))
            a_opt, _ = subgradient_topk_descent(state[1], time_limit=t_alloc,
                                                lr_init=lr, n_peaks=n_peaks)
            try_and_update(a_opt, state, "sg_k{}".format(n_peaks))

        # Another Nesterov pass on the best found so far
        remaining = deadline - time.monotonic() - 300
        if remaining > 30:
            t_alloc = min(remaining, 120)
            print("  Deep Nesterov pass 2, t={:.0f}s".format(t_alloc))
            a_opt, _ = nesterov_smoothed_descent(state[1], time_limit=t_alloc,
                                                 T_init=0.0008, lr_init=0.03, n_top=40)
            try_and_update(a_opt, state, "deep_nest2")

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Structured Perturbations + Restarts (~300s) ===
    print("\n=== Phase 3: Perturbations ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])

        # Various perturbation strategies
        perturbations = [
            ("noise_0.001", 0.001),
            ("noise_0.003", 0.003),
            ("noise_0.005", 0.005),
            ("noise_0.01", 0.01),
        ]

        for label, scale in perturbations:
            remaining = deadline - time.monotonic() - 200
            if remaining < 30:
                break

            for _ in range(3):
                remaining = deadline - time.monotonic() - 180
                if remaining < 15:
                    break
                noise = rng.standard_normal(n_best) * scale
                f_pert = project_simplex(state[1] + noise)
                t_alloc = min(remaining, 40)
                a_opt, _ = nesterov_smoothed_descent(f_pert, time_limit=t_alloc,
                                                     T_init=0.001, lr_init=0.04, n_top=25)
                try_and_update(a_opt, state, label)

        # Structured perturbation: blur + sharpen
        remaining = deadline - time.monotonic() - 150
        if remaining > 30:
            # Gaussian blur
            kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
            kernel /= kernel.sum()
            blurred = np.convolve(state[1], kernel, mode='same')
            blurred = np.maximum(blurred, 0)
            bs = blurred.sum()
            if bs > 1e-10:
                blurred = blurred / bs
                t_alloc = min(remaining, 50)
                a_opt, _ = nesterov_smoothed_descent(blurred, time_limit=t_alloc,
                                                     T_init=0.001, lr_init=0.04, n_top=25)
                try_and_update(a_opt, state, "blur_refine")

        # Edge-weighted perturbation
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            x = np.arange(n_best, dtype=np.float64)
            edge_w = np.exp(-0.5 * (np.minimum(x, n_best - 1 - x) / (0.1 * n_best)) ** 2)
            perturbed = state[1] * (1 + 0.02 * edge_w)
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps > 1e-10:
                perturbed = perturbed / ps
                t_alloc = min(remaining, 40)
                a_opt, _ = nesterov_smoothed_descent(perturbed, time_limit=t_alloc,
                                                     T_init=0.001, lr_init=0.04, n_top=25)
                try_and_update(a_opt, state, "edge_perturb")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Final Polish (~200s) ===
    print("\n=== Phase 4: Final Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Targeted peak reduction
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 80)
            print("  Targeted peak reduction, t={:.0f}s".format(t_alloc))
            a_opt, _ = targeted_peak_reduction(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "targeted")

        # Fine Nesterov with very low temperature
        remaining = deadline - time.monotonic() - 50
        if remaining > 15:
            t_alloc = min(remaining, 40)
            print("  Fine Nesterov, t={:.0f}s".format(t_alloc))
            a_opt, _ = nesterov_smoothed_descent(state[1], time_limit=t_alloc,
                                                 T_init=0.0001, lr_init=0.01, n_top=50)
            try_and_update(a_opt, state, "fine_nest")

        # Subgradient with many peaks (global smoothing)
        remaining = deadline - time.monotonic() - 30
        if remaining > 15:
            t_alloc = min(remaining, 30)
            a_opt, _ = subgradient_topk_descent(state[1], time_limit=t_alloc,
                                                lr_init=0.001, n_peaks=50, decay=0.9998)
            try_and_update(a_opt, state, "sg_global")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            for _ in range(5):
                remaining = deadline - time.monotonic() - 5
                if remaining < 5:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0005
                f_pert = project_simplex(state[1] + noise)
                t_alloc = min(remaining, 5)
                a_opt, _ = nesterov_smoothed_descent(f_pert, time_limit=t_alloc,
                                                     T_init=0.0003, lr_init=0.005, n_top=30)
                try_and_update(a_opt, state, "tiny_pert")

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = parent
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(state[1].tolist()))
        if np.isfinite(fs) and fs < state[0]:
            state[0] = fs
    except Exception:
        pass

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
