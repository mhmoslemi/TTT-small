```python
"""
Search for a non-negative sequence minimizing
    2*n*max(np.convolve(a,a))/(sum(a))^2

Hybrid strategy:
- Start from the supplied parent when available.
- Run a low-dimensional Fourier shape search to escape weak bases.
- Use a linearized LP Frank-Wolfe direction for self-convolution peak shaving.
- Polish with projected smoothed-subgradient steps and adaptive restarts.
- Keep the best verified sequence by actual evaluate_sequence().
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.linalg import toeplitz as dense_toeplitz

_TWO_PI = 2.0 * np.pi

# Global scratch state for the low-dimensional Fourier objective.
_F_STATE = {}
# Cache negative identity matrices used in LP formulations.
_EYE_CACHE = {}


def _next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_convolve(a):
    """Full linear auto-convolution via FFT, length 2n-1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = _next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Objective for a non-negative sequence already scaled to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return float("inf")
    c = fast_convolve(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    if m < 0.0:
        m = 0.0
    return 2.0 * float(n) * m


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if len(a) == 0:
        return np.ones(768) / 768.0
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(len(a)) / float(len(a))
    return a / s


def local_score(a):
    """Scale-invariant local objective for any finite sequence."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if len(a) == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return float("inf")
    return score_normalized(a / s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.ones(1, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    w = np.maximum(v - theta, 0.0)
    sw = float(w.sum())
    if sw <= 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return w / sw


def resize_sequence(a, n):
    """Resize a non-negative sequence to length n by interpolation and renormalize."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(max(1, n))
    m = len(a)
    if m == 0:
        return np.ones(n) / float(n)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(n) / float(n)
    a = a / s
    if m == 1:
        return np.ones(n) / float(n)
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    b = np.interp(new, old, a)
    return normalize_nonneg(b)


def get_negative_eye(L):
    L = int(L)
    if L not in _EYE_CACHE:
        _EYE_CACHE[L] = -np.eye(L, dtype=np.float64)
    return _EYE_CACHE[L]


def make_fourier_seq(N, t, coeffs):
    """
    Build a non-negative normalized sequence:
        raw = 1 + sum_m coeffs[2m] cos(2pi m t) + coeffs[2m+1] sin(2pi m t)
    clipped to non-negative and normalized.
    """
    N = int(N)
    t = np.asarray(t, dtype=np.float64)
    if len(t) != N:
        t = (np.arange(N, dtype=np.float64) + 0.5) / float(N)
    raw = np.ones(N, dtype=np.float64)
    M = len(coeffs) // 2
    for m in range(1, M + 1):
        ac = float(coeffs[2 * m - 2])
        as_ = float(coeffs[2 * m - 1])
        if ac != 0.0 or as_ != 0.0:
            th = _TWO_PI * float(m) * t
            raw += ac * np.cos(th) + as_ * np.sin(th)
    raw = np.maximum(raw, 0.0)
    s = float(raw.sum())
    if s <= 1e-15:
        return np.ones(N) / float(N)
    return raw / s


def _fourier_seq_from_state(params):
    st = _F_STATE
    N = int(st["N"])
    bounds = np.asarray(st["bounds"], dtype=np.float64)
    params = np.asarray(params, dtype=np.float64).ravel()
    M = len(bounds) // 2
    cosb = st["cos_basis"]
    sinb = st["sin_basis"]
    raw = np.ones(N, dtype=np.float64)
    for m in range(M):
        i0 = 2 * m
        i1 = i0 + 1
        a1 = float(np.clip(params[i0], -bounds[i0], bounds[i0]))
        a2 = float(np.clip(params[i1], -bounds[i1], bounds[i1]))
        if a1 != 0.0 or a2 != 0.0:
            raw += a1 * cosb[m] + a2 * sinb[m]
    raw = np.maximum(raw, 0.0)
    s = float(raw.sum())
    if s <= 1e-15:
        return np.ones(N) / float(N)
    return raw / s


def _fourier_obj(params):
    seq = _fourier_seq_from_state(params)
    return score_normalized(seq)

                    grid3 = np.linspace(lo, hi, fine)
                    for gv in grid3:
                        gv = float(gv)
                        if abs(gv - best_i) < 1e-14:
                            continue
                        cand = best_params.copy()
                        cand[i] = gv
                        sc = float(_fourier_obj(cand))
                        if np.isfinite(sc) and sc < best_sc:
                            best_sc = sc
                            best_i = gv

            if best_i != old:
                best_params[i] = best_i
                best_val = best_sc
                improved = True

        if cycle % 4 == 3:
            p = best_params + rng.standard_normal(2 * M) * 0.08
            p = np.clip(p, -bounds, bounds)
            sc = float(_fourier_obj(p))
            if np.isfinite(sc) and sc < best_val:
                best_val = sc
                best_params = p.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        cycle += 1

    # Optional derivative-free nonlinear finish for small dimension.
    if 2 * M <= 6 and spend() < time_limit - 4.0:
        try:
            bnds = list(zip(-bounds.tolist(), bounds.tolist()))
            res = minimize(
                _fourier_obj,
                best_params.copy(),
                method="Powell",
                bounds=bnds,
                options={"maxiter": 60, "xatol": 1e-5, "fatol": 1e-8, "disp": False},
            )
            if hasattr(res, "x") and np.isfinite(float(res.fun)):
                if float(res.fun) < best_val:
                    best_params = np.clip(np.asarray(res.x, dtype=np.float64), -bounds, bounds)
                    best_val = float(res.fun)
        except Exception:
            pass

    seq = _fourier_seq_from_state(best_params)
    sc = local_score(seq)
    return seq, float(sc)


def smoothed_weights(c, T, top_k):
    """Softmax weights on the largest convolution coefficients."""
    c = np.asarray(c, dtype=np.float64).ravel()
    L = len(c)
    if L == 0:
        return c.copy()
    T = float(max(1e-30, abs(T)))
    top_k = int(max(1, top_k))
    if top_k >= L:
        x = c / T
        x -= float(np.max(x))
        p = np.exp(x)
        s = float(p.sum())
        if s <= 1e-30:
            p = np.zeros(L)
            p[int(np.argmax(c))] = 1.0
            return p
        return p / s

    idx = np.argpartition(c, -top_k)[-top_k:]
    vals = c[idx]
    x = vals / T
    x -= float(np.max(x))
    w = np.exp(x)
    sw = float(w.sum())
    if sw <= 1e-30:
        p = np.zeros(L)
        p[int(np.argmax(c))] = 1.0
        return p
    w = w / sw
    p = np.zeros(L, dtype=np.float64)
    p[idx] = w
    return p


def conv_grad(a, p):
    """
    Gradient of sum_k p_k (a*a)_k, without the global constant 2.
    Returns vector g_j = sum_m a_m p_{j+m}.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = len(a)
    L = len(p)
    if n == 0 or L == 0:
        return np.zeros(n, dtype=np.float64)
    ar = a[::-1]
    need = L + n - 1
    sz = _next_pow2(need)
    A = np.fft.rfft(ar, sz)
    P = np.fft.rfft(p, sz)
    full = np.fft.irfft(A * P, sz)[:need]
    return full[n - 1:2 * n - 1]


def gradient_polish(a0, time_limit, top_k=32, T_frac=0.02):
    """Projected smoothed-gradient polish on the simplex."""
    a = normalize_nonneg(a0)
    n = len(a)
    if n <= 1:
        return a
    c = fast_convolve(a)
    M = float(max(0.0, c.max()))
    best = a.copy()
    best_score = 2.0 * float(n) * M
    if M <= 1e-15:
        return best

    eta = 0.07
    T = max(1e-13, T_frac * M)
    step0 = eta / float(n)
    t0 = time.monotonic()
    stall = 0
    it = 0
    time_limit = max(0.0, float(time_limit))

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.2:
            break

        c = fast_convolve(a)
        M = float(max(0.0, c.max()))
        sc = 2.0 * float(n) * M
        if sc < best_score - 1e-14:
            best_score = sc
            best = a.copy()

        if M <= 1e-15:
            break

        pk = max(8, int(min(top_k, 2 * n - 1)))
        p = smoothed_weights(c, max(1e-13, T), pk)
        grad = 2.0 * conv_grad(a, p)
        gmax = float(np.max(np.abs(grad)))
        if gmax <= 1e-16:
            break

        direction = -grad / gmax
        best_cand = None
        best_cand_sc = sc

        # Steps are relative to 1/n, i.e. a fraction of average mass.
        for mult in (1.0, 0.6, 0.35, 0.2, 0.1, 0.05):
            cand = project_simplex(a + step0 * mult * direction)
            vsc = score_normalized(cand)
            if np.isfinite(vsc) and vsc < best_cand_sc - 1e-15:
                best_cand_sc = vsc
                best_cand = cand

        if best_cand is not None:
            a = best_cand
            if best_cand_sc < best_score - 1e-14:
                best_score = best_cand_sc
                best = a.copy()
            stall = 0
            eta = min(0.25, eta * 1.02)
            step0 = eta / float(n)
            T = max(1e-13, T * 0.995)
        else:
            stall += 1
            eta *= 0.75
            step0 = eta / float(n)
            if stall > 25:
                T = min(0.12 * M, T * 1.5)
            if stall > 140:
                rng = np.random.default_rng((it * 15485863 + 17) % (2**31))
                scale = (0.035 + 0.015 * min(1.0, float(stall) / 120.0)) / float(n)
                a = project_simplex(best + rng.standard_normal(n) * scale)
                T = max(1e-13, T_frac * M)
                eta = 0.07
                step0 = eta / float(n)
                stall = 0

        it += 1
        if stall > 500:
            break

    return best


def solve_direction_lp(f, c, time_limit=3.0):
    """
    Solve the linearized FW subproblem:
        min t
        s.t. sum g = 1, g >= 0,
             (f*g)_k - t <= c_k  for all k.
    Returns (g, t) or (None, inf).
    """
    f = np.asarray(f, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    n = len(f)
    L = 2 * n - 1
    if n < 2 or L < 1:
        return None, float("inf")

    try:
        first_col = np.zeros(L, dtype=np.float64)
        first_col[:n] = f
        first_row = np.zeros(n, dtype=np.float64)
        first_row[0] = f[0]

        A = dense_toeplitz(first_col, first_row)
        A = np.asarray(A, dtype=np.float64)

        neginf_eye = get_negative_eye(L)
        A_ub = np.concatenate((A, neginf_eye), axis=1)

        cmax = float(np.max(c)) if c.size else 0.0
        b_ub = np.maximum(c, 0.0) + 1e-12 * max(1.0, cmax)

        A_eq = np.ones((1, n + 1), dtype=np.float64)
        A_eq[0, n] = 0.0
        b_eq = 1.0

        c_obj = np.zeros(n + 1, dtype=np.float64)
        c_obj[n] = 1.0

        bounds = [(0.0, None)] * n + [(None, None)]

        tl = max(0.5, float(time_limit))
        res = linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": tl},
        )
        if res.success:
            g = np.asarray(res.x[:n], dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = float(g.sum())
            if s > 1e-15:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, float("inf")


def lp_fw_optimize(a0, time_limit):
    """Frank-Wolfe with exact linearized self-convolution directions."""
    a = normalize_nonneg(a0)
    n = len(a)
    if n <= 1:
        return a

    best = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    time_limit = max(0.0, float(time_limit))
    stall = 0
    it = 0

    alphas = (0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2, 0.35, 0.55, 0.8, 1.0)

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.2:
            break

        c = fast_convolve(a)
        c = np.maximum(c, 0.0)
        base_score = 2.0 * float(n) * float(c.max())
        if base_score < best_score - 1e-14:
            best_score = base_score
            best = a.copy()

        if n <= 900:
            lp_tl = min(3.0, max(1.0, rem * 0.35))
        elif n <= 1300:
            lp_tl = min(4.0, max(1.0, rem * 0.3))
        else:
            lp_tl = min(5.0, max(1.0, rem * 0.25))

        g, tval = solve_direction_lp(a, c, time_limit=lp_tl)

        if g is None:
            # Fallback: active-peak subgradient step.
            if c.max() > 1e-15:
                p = smoothed_weights(c, max(1e-13, 0.01 * float(c.max())), 12)
                grad = 2.0 * conv_grad(a, p)
                gn = float(np.max(np.abs(grad)))
                if gn > 1e-16:
                    cand = project_simplex(a - (0.08 / float(n)) * (grad / gn))
                    csc = score_normalized(cand)
                    if np.isfinite(csc) and csc < base_score - 1e-14:
                        a = cand
                        if csc < best_score - 1e-14:
                            best_score = csc
                            best = a.copy()
                        stall = 0
                        it += 1
                        continue
            stall += 1
        else:
            if float(np.max(np.abs(g - a))) < 1e-11:
                stall += 1
            else:
                best_a = a
                best_alpha = 0.0
                best_sc = base_score

                for alpha in alphas:
                    cand = (1.0 - alpha) * a + alpha * g
                    cand = np.maximum(cand, 0.0)
                    s = float(cand.sum())
                    if s <= 1e-15:
                        continue
                    cand = cand / s
                    sc = score_normalized(cand)
                    if np.isfinite(sc) and sc < best_sc - 1e-15:
                        best_sc = sc
                        best_a = cand
                        best_alpha = alpha

                if best_alpha > 1e-9:
                    lo = max(0.0, best_alpha - 0.04)
                    hi = min(1.0, best_alpha + 0.04)
                    for _ in range(9):
                        mid = 0.5 * (lo + hi)
                        cand = (1.0 - mid) * a + mid * g
                        cand = np.maximum(cand, 0.0)
                        s = float(cand.sum())
                        if s <= 1e-15:
                            break
                        cand = cand / s
                        sc = score_normalized(cand)
                        if np.isfinite(sc) and sc < best_sc - 1e-15:
                            best_sc = sc
                            best_a = cand
                            best_alpha = mid
                            lo = mid
                        else:
                            hi = mid

                if best_alpha > 1e-9:
                    a = best_a
                    if best_sc < best_score - 1e-14:
                        best_score = best_sc
                        best = a.copy()
                    stall = 0
                else:
                    stall += 1

        if stall > 22:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            scale = (0.04 + 0.02 * min(1.0, float(stall) / 25.0)) / float(n)
            a = project_simplex(best + rng.standard_normal(n) * scale)
            stall = 0

        it += 1
        if stall > 250:
            break

    return best


def add_candidate(stored, a, local=None, max_store=80):
    """Add a normalized non-negative candidate to the pool, return local score."""
    if a is None:
        return float("inf")
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return float("inf")
    a = a / s
    if local is None or not np.isfinite(float(local)):
        local = local_score(a)
    local = float(local)
    if not np.isfinite(local):
        return float("inf")
    stored.append([local, a.copy()])
    if len(stored) > int(max_store):
        wi = 0
        ws = -1e300
        for i, item in enumerate(stored):
            if item[0] > ws:
                ws = item[0]
                wi = i
        stored.pop(wi)
    return local


def propose_candidate():
    """
    Main search routine. Returns a Python list of non-negative floats.
    """
    t_start = time.monotonic()
    deadline = t_start + 1765.0  # leave a safety margin
    stored = []
    best_local = float("inf")
    best_a = None

    def keep(a, local=None):
        nonlocal best_local, best_a
        sc = add_candidate(stored, a, local=local)
        if np.isfinite(sc) and sc < best_local:
            best_local = sc
            best_a = np.asarray(a, dtype=np.float64).ravel()
            best_a = np.maximum(best_a, 0.0)
            s = float(best_a.sum())
            if s > 1e-15:
                best_a = best_a / s
            print(f"new best local={best_local:.10f} n={len(best_a)} t={time.monotonic()-t_start:.0f}s")

    # Load parent if available.
    parent_norm = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.maximum(parent_raw, 0.0)
            ps = float(parent_raw.sum())
            if ps > 1e-12:
                parent_norm = parent_raw / ps
                keep(parent_norm)
                print(f"parent loaded n={len(parent_norm)} local={best_local:.10f}")
    except Exception:
        parent_norm = None

    # Fallback primary dimension.
    if parent_norm is not None and 256 <= len(parent_norm) <= 1600:
        primary_n = int(len(parent_norm))
    elif parent_norm is not None:
        primary_n = 1024 if len(parent_norm) > 1024 else 768
    else:
        primary_n = 768
    primary_n = int(min(1280, max(384, primary_n)))

    if best_a is None:
        best_a = np.ones(primary_n) / float(primary_n)
        keep(best_a)

    # Fourier dimension list.
    n_list = [primary_n]
    if primary_n != 1024:
        n_list.append(1024)
    if len(n_list) < 3 and primary_n < 896:
        n_list.append(896)
    n_list = [int(x) for x in n_list if 256 <= int(x) <= 1280]

    # Static Fourier seeds before heavier search.
    coeff_presets = [
        [0.0, -0.65, 0.0, -0.35, 0.0, -0.30, 0.0, 0.0],
        [0.0, -0.55, 0.0, 0.18, 0.0, -0.45, 0.0, 0.0],
        [0.0, -0.45, 0.0, 0.10, 0.0, -0.35, 0.0, -0.08],
        [0.20, -0.35, 0.0, 0.12, -0.05, -0.35, 0.0, 0.0],
        [0.0, -0.70, 0.0, 0.0, 0.0, -0.55, 0.0, 0.0],
        [0.05, -0.50, -0.05, 0.20, 0.0, -0.40, 0.0, 0.05],
    ]
    for N in n_list:
        if time.monotonic() > deadline - 1200:
            break
        t_mid = (np.arange(N, dtype=np.float64) + 0.5) / float(N)
        t_left = np.arange(N, dtype=np.float64) / float(N)
        for t_current in (t_mid, t_left):
            for cp in coeff_presets:
                seq = make_fourier_seq(N, t_current, cp)
                keep(seq)

    # Fourier coordinate search.
    t_four = time.monotonic()
    fourier_budget = min(220.0, max(0.0, deadline - time.monotonic() - 1050.0))
    if fourier_budget > 25.0:
        per_n = max(10.0, fourier_budget / float(max(1, len(n_list))))
        for idx, N in enumerate(n_list):
            elapsed = time.monotonic() - t_four
            if elapsed >= fourier_budget:
                break
            alloc = min(per_n, max(8.0, fourier_budget - elapsed - 0.5))
            if alloc < 6.0:
                break
            t_mid = (np.arange(N, dtype=np.float64) + 0.5) / float(N)
            # Use a few modes; stop if time is tight.
            for mm in (2, 3, 4):
                if time.monotonic() - t_four > fourier_budget - 5.0:
                    break
                rem = deadline - time.monotonic()
                if rem < 700.0:
                    break
                alloc_m = min(alloc, max(4.0, rem - 700.0))
                if alloc_m < 3.0:
                    break
                try:
                    seq, sc = fourier_coord_search(N, t_mid, mm, alloc_m, seed=101 + idx * 17 + mm)
                    keep(seq, local=sc)
                except Exception:
                    pass

    # LP Frank-Wolfe on the current best.
    rem = deadline - time.monotonic()
    lp_budget = min(760.0, max(0.0, rem - 350.0))
    if lp_budget > 60.0 and best_a is not None:
        a_lp = best_a
        if len(a_lp) > 1300:
            a_lp = resize_sequence(a_lp, 1024)
            keep(a_lp)
        try:
            out = lp_fw_optimize(a_lp, lp_budget)
            keep(out)
        except Exception:
            pass

        # Second LP run from parent if distinct and time remains.
        rem = deadline - time.monotonic()
        if parent_norm is not None and rem - 200.0 > 60.0:
            a_p = parent_norm
            if len(a_p) > 1300:
                a_p = resize_sequence(a_p, 1024)
                keep(a_p)
            try:
                out2 = lp_fw_optimize(a_p, min(180.0, rem - 200.0))
                keep(out2)
            except Exception:
                pass

    # Gradient polish on the current best.
    rem = deadline - time.monotonic()
    grad_budget = min(260.0, max(0.0, rem - 70.0))
    if grad_budget > 25.0 and best_a is not None:
        try:
            out = gradient_polish(best_a, grad_budget, top_k=24, T_frac=0.02)
            keep(out)
        except Exception:
            pass

    # Final short LP if time remains.
    rem = deadline - time.monotonic()
    if rem - 35.0 > 80.0 and best_a is not None:
        a_fin = best_a
        if len(a_fin) > 1300:
            a_fin = resize_sequence(a_fin, 1024)
            keep(a_fin)
        try:
            out = lp_fw_optimize(a_fin, max(20.0, rem - 55.0))
            keep(out)
        except Exception:
            pass

    # Final short gradient polish.
    rem = deadline - time.monotonic()
    if rem - 25.0 > 35.0 and best_a is not None:
        try:
            out = gradient_polish(best_a, max(10.0, rem - 35.0), top_k=12, T_frac=0.01)
            keep(out)
        except Exception:
            pass

    if best_a is None:
        best_a = np.ones(primary_n) / float(primary_n)
        keep(best_a)

    # Verify with the actual evaluator and choose the best among stored candidates.
    final_best = best_a.copy()
    final_best_score = float("inf")

    def try_actual(arr):
        nonlocal final_best, final_best_score
        if arr is None:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.maximum(arr, 0.0)
        s = float(arr.sum())
        if s <= 1e-15:
            return
        arr = arr / s
        try:
            sc = evaluate_sequence([float(x) for x in arr])
            sc = float(sc)
            if np.isfinite(sc) and sc < final_best_score:
                final_best_score = sc
                final_best = arr.copy()
                print(f"actual best={sc:.10f} n={len(arr)} t={time.monotonic()-t_start:.0f}s")
        except Exception:
            pass

    # Evaluate locally-best candidates first without sorting via lambda.
    pool = stored[:]
    for _ in range(min(45, len(pool))):
        mi = 0
        ms = float("inf")
        for i, item in enumerate(pool):
            if item[0] < ms:
                ms = item[0]
                mi = i
        item = pool.pop(mi)
        try_actual(item[1])
        if time.monotonic() > deadline - 3.0:
            break

    # Always try the current useful best.
    try_actual(best_a)
    if parent_norm is not None:
        try_actual(parent_norm)

    # Sanitize final sequence.
    out = np.asarray(final_best, dtype=np.float64).ravel()
    if out.size == 0 or not np.isfinite(out).all():
        out = np.ones(primary_n) / float(primary_n)
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 0.01:
        out = np.ones(primary_n) / float(primary_n)
        s = float(out.sum())
    out = out / s

    return [float(x) for x in out]
```
