```python
def solve_lp_cvxpy(f, M, n, solver='GUROBI'):
    g = cp.Variable(n)
    # Constraints: f * g [k] <= M for all k, g >= 0
    # f * g [k] = sum_i f[i] * g[k-i] (valid indices)
    
    constraints = [g >= 0]
    for k in range(2*n-1):
        # f*g[k] = sum_{i: valid} f[i] * g[k-i]
        # valid i: max(0, k-n+1) <= i <= min(n-1, k)
        i_start = max(0, k-n+1)
        i_end = min(n-1, k)
        # f*g[k] = sum_{i=i_start}^{i_end} f[i] * g[k-i]
        # = sum_{j=k-i_end}^{k-i_start} f[k-j] * g[j]
        j_start = k - i_end
        j_end = k - i_start
        constraints.append(cp.sum(cp.multiply(f[k-j_end:k-j_start+1][::-1], g[j_start:j_end+1])) <= M)
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=solver)
    return g.value
```

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


class LPOracle:
    """Precompiled LP oracle for SCA iterations."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                rows.append(k)
                cols.append(i)
        self.row_idx = np.array(rows, dtype=np.int32)
        self.col_idx = np.array(cols, dtype=np.int32)
        self.shape = (2 * n - 1, n)

    def solve(self, f, M):
        """Solve: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
        n = self.n
        k_arr = self.row_idx
        i_arr = self.col_idx
        vals = f[k_arr - i_arr]
        A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=self.shape)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if res.success and res.x is not None:
                g = np.maximum(res.x, 0.0)
                if g.sum() > 1e-15:
                    return g
        except Exception:
            pass
        return None


def sca_optimize(f0, time_limit, oracle=None, verbose=False):
    """SCA with adaptive M using binary search strategy."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if oracle is None:
        oracle = LPOracle(n)

    conv = fast_convolve(f)
    M = conv.max()
    M_floor = M * 0.95  # Don't go below this
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g_raw = oracle.solve(f, M)

        if g_raw is None:
            # Loosen M
            M = max(M * 1.02, best_obj * 1.005)
            no_improve += 1
            if no_improve > 40:
                M = fast_convolve(f).max() * 1.01
                best_obj = M
                no_improve = 0
            continue

        g = g_raw / g_raw.sum()

        if np.max(np.abs(g - f)) < 1e-13:
            # Try tightening M
            M *= 0.998
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                M = fast_convolve(f).max() * 1.005
                no_improve = 0
            continue

        # Line search with more points
        best_val = fast_convolve(f).max()
        best_ft = f.copy()
        best_alpha = 0.0

        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()
                best_alpha = alpha

        if best_val < best_obj - 1e-14:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            # Aggressively tighten M
            M = best_val * 0.999
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.9995

        if no_improve > 50:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.003 + 0.004 * min(1.0, (no_improve - 50) / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            M = fast_convolve(f).max() * 1.005
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    SCA it={it}, obj={best_obj:.10f}, M={M:.10f}")

    return best_f, best_obj


def peak_coordinate_descent(f0, time_limit, verbose=False):
    """
    Coordinate descent targeting the active peaks.
    For each peak, identify the elements contributing most and
    adjust them to reduce the peak while maintaining non-negativity.
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        # Find top peaks
        n_peaks = min(5, len(conv))
        peak_indices = np.argpartition(conv, -n_peaks)[-n_peaks:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        improved = False
        for k in peak_indices:
            k = int(k)
            peak_val = conv[k]
            if peak_val <= best_obj * 1.001:
                continue

            # Elements contributing to this peak
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)

            # Contribution of each pair (i, k-i)
            contribs = []
            for i in range(i_start, i_end + 1):
                j = k - i
                if j < i:
                    # Already counted
                    weight = 2.0
                elif j == i:
                    weight = 1.0
                else:
                    weight = 1.0
                contribs.append((a[i] * a[j], i, j, weight))

            if not contribs:
                continue

            contribs.sort(reverse=True, key=lambda x: x[0])

            # Try reducing the top contributing elements
            for _ in range(3):
                if not contribs:
                    break
                c_val, i, j, w = contribs[0]
                if c_val < 1e-15:
                    break

                # Move epsilon from i (and j if different) to reduce peak
                # while distributing to elements that don't contribute to this peak
                eps = 0.01 * a[i]
                if j != i and j < n:
                    eps_j = 0.01 * a[j]
                else:
                    eps_j = 0

                a_new = a.copy()
                a_new[i] -= eps
                if j != i and j < n and j >= 0:
                    a_new[j] -= eps_j

                if a_new[i] < 0 or (j != i and a_new[j] < 0):
                    eps *= 0.5
                    a_new[i] = a[i] - eps
                    if j != i:
                        a_new[j] = a[j] - eps * (a[j] / (a[i] + 1e-30))
                    if np.any(a_new < -1e-15):
                        break

                # Distribute removed mass to elements far from the peak
                total_removed = (a[i] - a_new[i]) + (a[j] - a_new[j]) if j != i else (a[i] - a_new[i])
                if total_removed < 1e-15:
                    break

                # Find elements that contribute least to this peak
                outside_contrib = []
                for idx in range(n):
                    # Contribution of a[idx] to peak k: a[idx]*a[k-idx] if valid
                    j_idx = k - idx
                    if 0 <= j_idx < n:
                        contrib = a[idx] * a[j_idx]
                    else:
                        contrib = 0
                    outside_contrib.append((contrib, idx))

                outside_contrib.sort()
                # Add to the elements with lowest contribution
                add_per = total_removed / 5
                for cc, idx in outside_contrib[:5]:
                    a_new[idx] += add_per

                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                new_obj = fast_convolve(a_new).max()

                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    break
                else:
                    break  # Try next peak

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a, best_obj


def mirror_descent(a0, time_limit, eta_init=0.05, verbose=False):
    """Multiplicative weight mirror descent with peak-focused gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    eta = eta_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        max_val = conv.max()

        # Use top peaks with softmax weighting
        top_k = min(10, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        temp = 0.01 * max_val + 1e-15
        pw = np.exp((pv - pv.max()) / temp)
        pw /= pw.sum()

        # Subgradient of max(conv(a,a)) w.r.t. a
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = pw[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            for i in range(max(0, k - n + 1), min(n - 1, k) + 1):
                j = k - i
                if j == i:
                    grad[i] += w * a[j]
                else:
                    grad[i] += w * a[j]

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Blend with line search
        best_candidate = a
        best_val = max_val
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            eta *= 0.93
        if eta < 1e-5:
            eta = eta_init * 0.5
            no_improve = 0

        if no_improve > 120:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
            eta = eta_init
            no_improve = 0

    return best_a, best_obj


def subgradient_ls(a0, time_limit, top_k=8):
    """Subgradient with binary line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.03

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            weights = pv / pv.sum()

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = weights[idx]
            k = int(peaks[idx])
            for i in range(max(0, k - n + 1), min(n - 1, k) + 1):
                j = k - i
                grad[i] += w * 2.0 * a[j]

        direction = project_simplex(a - lr * grad) - a

        # Binary line search
        lo, hi = 0.0, 1.0
        best_alpha = 0.0
        best_val = conv.max()
        for _ in range(10):
            mid = (lo + hi) / 2.0
            if mid < 1e-10:
                break
            cand = a + mid * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                hi = mid
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid

        if best_alpha > 1e-10:
            a = a + best_alpha * direction
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        lr *= 0.998
        if lr < 0.003:
            lr = 0.04

        if no_improve > 60:
            rng = np.random.default_rng((it * 1299709 + 3) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
            lr = 0.04

    return best_a, best_obj


def gen_init_ramp(n, rng):
    """Linear ramp initialization."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a += rng.standard_normal(n) * 0.1 * n
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gen_init_exp(n, rng):
    """Exponential decay initialization."""
    alpha = rng.uniform(0.5, 3.0)
    a = np.exp(-alpha * np.arange(n) / n)
    a += rng.standard_normal(n) * 0.05
    a = np.maximum(a, 0.0)
    return a / a.sum()


def gen_init_bimodal(n, rng):
    """Two Gaussian bumps."""
    c1 = rng.uniform(0.1, 0.4)
    c2 = rng.uniform(0.6, 0.9)
    w1 = rng.uniform(0.03, 0.15)
    w2 = rng.uniform(0.03, 0.15)
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - c1) / w1) ** 2) + np.exp(-0.5 * ((x - c2) / w2) ** 2)
    a += rng.standard_normal(n) * 0.02
    a = np.maximum(a, 0.0)
    return a / a.sum()


def gen_init_parent_perturb(parent, rng, scale=0.02):
    """Perturb the parent sequence."""
    n = len(parent)
    a = parent + rng.standard_normal(n) * scale
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return parent.copy()
    return a / s


def gen_init_sinusoidal(n, rng):
    """Sinusoidal perturbation of uniform."""
    freq = rng.uniform(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.2, 0.7)
    x = np.arange(n)
    a = 1.0 + amp * np.sin(2 * np.pi * freq * x / n + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()
