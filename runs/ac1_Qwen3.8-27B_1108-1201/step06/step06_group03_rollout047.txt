def line_search_quad(f, g, c_f, max_iters=12):

    # But we can compute it via FFT
    size = 2 * n - 1

    F = np.fft.rfft(f, fft_size)

        c_t = (1 - t)**2 * c_f + t * (1 - t) * cross + t**2 * c_g
        return c_t.max(), c_t

    best_alpha = 0.0
    best_c_t = c_f.copy()

    gr = (np.sqrt(5.0) - 1.0) / 2.0

    d = gr * lo + (1 - gr) * hi
    c_pt = (1 - gr) * lo + gr * hi
    fd, _ = obj_at_t(d)
    fc, _ = obj_at_t(c_pt)

        if fd < fc:

            fc, _ = obj_at_t(c_pt)

    # Also check endpoints and a few extra points
    for t_check in [0.0, 0.1, 0.25, 0.5, 0.75, 1.0, best_alpha]:
        v, ct = obj_at_t(t_check)
        if v < best_val:
            best_val = v
            best_alpha = t_check
            best_c_t = ct

    return best_alpha, best_val


def fw_optimize(f0, time_limit, verbose=False, use_reduced=True, top_k=30):
    """
    Frank-Wolfe with LP oracle, golden section line search, and adaptive M.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    c_f = fast_convolve(f)
    M = c_f.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.001
    away_set = set()

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        # Solve LP oracle
        if use_reduced:
            g = solve_lp_reduced(f, M * m_relax, top_k=top_k)
        else:
            g = solve_lp_reduced(f, M * m_relax, top_k=100)

        if g is None or g.sum() < 1e-15:
            g = solve_lp_reduced(f, M * 1.005, top_k=50)

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 30:
                # Perturb and restart
                scale = 0.002 + 0.005 * min(1.0, stagnant / 100.0)
                rng = np.random.default_rng(it * 7919 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                c_f = fast_convolve(f)
                M = c_f.max()
                m_relax = 1.001
                stagnant = 0
            continue

        g = g / g.sum()

        # Check if direction is meaningful
        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 40:
                scale = 0.003 + 0.004 * min(1.0, stangant / 100.0)
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                c_f = fast_convolve(f)
                M = c_f.max()
                m_relax = 1.001
                stagnant = 0
            continue

        # Line search
        best_alpha, best_val = line_search_quad(f, g, c_f)

        if best_alpha > 1e-10:
            f_new = (1.0 - best_alpha) * f + best_alpha * g
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new = f_new / s_new

            c_new = fast_convolve(f_new)
            M_new = c_new.max()

            if M_new < best_obj - 1e-15:
                best_obj = M_new
                best_f = f_new.copy()
                stagnant = 0
            else:
                stagnant += 1

            f = f_new
            c_f = c_new
            M = M_new
            m_relax = 1.001
        else:
            stagnant += 1

        # Adaptive M relaxation
        if stagnant > 15:
            m_relax = min(m_relax * 1.03, 1.02)
        if stagnant > 50:
            m_relax = min(m_relax * 1.1, 1.05)

        # Periodic full-LP step for precision
        if it % 200 == 0 and stagnant < 10:
            g_full = solve_lp_full(f, M * 1.001)
            if g_full is not None and g_full.sum() > 1e-15:
                g_full = g_full / g_full.sum()
                if np.max(np.abs(g_full - f)) > 1e-8:
                    ba, bv = line_search_quad(f, g_full, c_f)
                    if ba > 1e-10 and bv < best_obj - 1e-14:
                        f_new = np.maximum((1 - ba) * f + ba * g_full, 0)
                        f_new /= f_new.sum()
                        c_new = fast_convolve(f_new)
                        if c_new.max() < best_obj:
                            best_obj = c_new.max()
                            best_f = f_new.copy()
                            f = f_new
                            c_f = c_new
                            M = c_new.max()

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, M={M:.10f}, stag={stagnant}")

    return best_f, best_obj


def multiplicative_descent(a0, time_limit, eta0=0.02, T0=0.005):
    """Multiplicative (mirror) descent for fast initial improvement."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    eta = eta0
    T = T0
    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        c_max = c.max()

        # Active peaks
        threshold = c_max - T
        peaks = np.where(c >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(c)])

        # Weighted subgradient via direct computation
        weights = c[peaks]
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            weights = weights / w_sum

        grad = np.zeros(n)
        for i, k in enumerate(peaks):
            wi = weights[i]
            if wi < 1e-14:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je+1] += wi * 2.0 * a[k - np.arange(js, je + 1)]

        # Multiplicative update with clipping
        update = np.exp(-eta * grad / max(grad.max(), 1e-15))
        update = np.clip(update, 0.1, 10.0)
        a_new = a * update
        a_new = np.maximum(a_new, 1e-12)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        a = a_new

        # Adapt
        if stagnant > 5:
            eta *= 0.75
        if stagnant > 20:
            T *= 0.6
        if stagnant > 100:
            break

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit, max_peaks=6, max_pairs=8):
    """Coordinate descent targeting top peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_frac = 0.01

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        c = fast_convolve(a)
        peak_indices = np.argsort(c)[::-1][:max_peaks]

        improved = False
        for ki in range(len(peak_indices)):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if js > je:
                continue

            # Top contributing pairs
            contribs = []
            for j in range(js, je + 1):
                contribs.append((a[j] * a[k - j], j, k - j))
            contribs.sort(reverse=True)

            for (val, j1, j2) in contribs[:max_pairs]:
                if a[j1] < 1e-10:
                    continue

                # Try shifting mass from j1 to a distant position
                rng = np.random.default_rng((it * 9973 + k * 31 + j1 * 7 + ki) % (2**31))
                for _attempt in range(6):
                    j_target = int(rng.integers(0, n))
                    if j_target == j1 or j_target == j2:
                        continue

                    for src in [j1, j2]:
                        if a[src] < 1e-10:
                            continue
                        for frac in [step_frac, step_frac * 0.5, step_frac * 2.0]:
                            delta = a[src] * frac
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[j_target] += delta
                            if candidate[src] < -1e-12:
                                continue
                            candidate = np.maximum(candidate, 0)
                            cs = candidate.sum()
                            if cs < 1e-15:
                                continue
                            candidate = candidate / cs
                            val_c = fast_convolve(candidate).max()
                            if val_c < best_obj - 1e-14:
                                a = candidate
                                best_obj = val_c
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            step_frac *= 0.92
            if step_frac < 0.0005:
                # Perturb
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                val_c = fast_convolve(candidate).max()
                if val_c < best_obj - 1e-14:
                    a = candidate
                    best_obj = val_c
                    best_a = a.copy()
                    step_frac = 0.005
                elif it % 500 == 0:
                    break
                elif it > 10000:
                    break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_diverse_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    x = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
    w = np.maximum(w, 0.005)
    starts.append(w / w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.15 * n)) ** 2)
    starts.append(g / g.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 2.0)
    starts.append(edge / edge.sum())

    # Bell at 1/3 and 2/3
    g1 = np.exp(-0.5 * ((x - 0.33 * n) / (0.06 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.67 * n) / (0.06 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Parabolic
    par = x * (n - 1 - x) + 1.0
    starts.append(par / par.sum())

    # Random Dirichlet
    for _ in range(3):
        starts.append(rng.dirichlet(np.ones(n) * 0.5))

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Hybrid search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Quick multiplicative descent (40s) ===
    print("\n=== Phase 1: Multiplicative Descent ===")
    t_phase = time.monotonic()
    for n_try in [n_parent]:
        t_alloc = min(deadline - time.monotonic() - 1500.0, 40.0)
        if t_alloc > 5:
            a_opt, _ = multiplicative_descent(parent, time_limit=t_alloc)
            try_seq(a_opt, "mult_descent")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Quick multi-n exploration (120s) ===
    print("\n=== Phase 2: Multi-n Exploration ===")
    t_phase = time.monotonic()
    lengths = sorted(set([n_parent, max(256, n_parent - 128), n_parent + 128,
                          512, 640, 768, 896, 1024, 1280]))
    lengths = [l for l in lengths if 256 <= l <= 2048]

    for n_try in lengths:
        remaining = deadline - time.monotonic() - 1200.0
        if remaining < 15:
            break
        starts = generate_diverse_starts(n_try, parent, rng)
        for si, start in enumerate(starts[:3]):
            remaining = deadline - time.monotonic() - 1100.0
            if remaining < 10:
                break
            t_alloc = min(remaining, 25.0)
            a_opt, _ = fw_optimize(start, time_limit=t_alloc, use_reduced=True, top_k=30)
            try_seq(a_opt, f"expl_n{n_try}_s{si}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Deep FW optimization (500s) ===
    print("\n=== Phase 3: Deep FW ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 600.0
        if remaining > 30:
            t_alloc = min(remaining, 450.0)
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, use_reduced=True, top_k=40, verbose=True)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts
        for scale in [0.001, 0.003, 0.008, 0.02]:
            remaining = deadline - time.monotonic() - 350.0
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50.0)
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc, use_reduced=True, top_k=30)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Full-LP FW for precision (200s) ===
    print("\n=== Phase 4: Full-LP FW ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 250.0
        if remaining > 20:
            t_alloc = min(remaining, 180.0)
            print(f"  Full-LP FW, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, use_reduced=False, top_k=100)
            try_seq(a_opt, "full_lp_fw")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Coordinate descent polishing (200s) ===
    print("\n=== Phase 5: Peak Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 15:
            t_alloc = min(remaining, 180.0)
            print(f"  Peak coord descent, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_coord")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final micro-polish (remaining) ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        for ri in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if ri % 3 == 0:
                noise_scale = 0.0005 * (1 + ri * 0.5)
                start = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
                t_alloc = min(remaining, 20.0)
                a_opt, _ = fw_optimize(start, time_limit=t_alloc, use_reduced=True, top_k=20)
                try_seq(a_opt, f"polish_fw_{ri}")
            elif ri % 3 == 1:
                t_alloc = min(remaining, 15.0)
                a_opt, _ = multiplicative_descent(best_seq, time_limit=t_alloc,
                                                  eta0=0.005, T0=0.002)
                try_seq(a_opt, f"polish_mult_{ri}")
            else:
                t_alloc = min(remaining, 12.0)
                a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc,
                                                   max_peaks=4, max_pairs=5)
                try_seq(a_opt, f"polish_cd_{ri}")

    print(f"  Phase 6 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
