    return np.fft.irfft(A * A, fft_size)[:size]

    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Smooth log-sum-exp gradient optimization with temperature annealing (primary)
- Frank-Wolfe with LP oracle for polishing
- Peak-targeted coordinate descent

Key improvement: exact smooth gradients (not subgradients) enable faster convergence.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def smooth_gradient_optimize(a0, time_limit, T_init=None, T_min=1e-8, T_decay=0.9995):
    """
    Smooth log-sum-exp gradient optimization with backtracking line search.
    Uses exact gradient of T*log(sum(exp(c/T))) which is a smooth approximation of max(c).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if T_init is None:
        c0 = fast_convolve(a)
        spread = c0.max() - c0.min()
        T_init = max(spread * 0.05, 1e-5)

    T = T_init
    best_a = a.copy()
    best_cmax = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    lr = 0.5
    v = np.zeros(n)  # momentum
    mom = 0.7
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        c = fast_convolve(a)
        c_max = c.max()

        if c_max < best_cmax - 1e-16:
            best_cmax = c_max
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Compute softmax weights on c
        c_shifted = c - c.max()
        w = np.exp(c_shifted / T)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(len(c)) / len(c)
        else:
            w = w / w_sum

        # Gradient: grad_a[i] = 2 * sum_m w[i+m] * a[m]
        grad_a = 2.0 * np.correlate(w, a, mode='valid')

        # Momentum update
        v = mom * v - lr * grad_a

        # Backtracking line search
        f_curr = c_max  # use max(c) as objective
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.025]:
            a_trial = project_simplex(a + frac * v)
            c_trial = fast_convolve(a_trial)
            if c_trial.max() < f_curr - 1e-16:
                a = a_trial
                improved = True
                break
            if frac < 0.03:
                a = a_trial  # take the best we found
                improved = c_trial.max() < f_curr
                break

        if not improved:
            # Fallback: small step
            a = project_simplex(a - 0.01 * lr * grad_a)

        # Adapt temperature
        if stagnant > 30:
            T = max(T * T_decay, T_min)
        if stagnant > 100:
            T = max(T * 0.95, T_min)
            lr = min(lr * 1.2, 5.0)
        if stagnant > 300:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = max(T_init * 0.1, T)
            v = np.zeros(n)
            stagnant = 0

        # Adapt learning rate
        if improved:
            lr = min(lr * 1.05, 3.0)
        else:
            lr = max(lr * 0.9, 0.01)

    return best_a, best_cmax


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle, line search, and adaptive M."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.003

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.2:
            break

        g = solve_lp_oracle(f, M * m_relax)
        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.01)
        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 15:
                rng = np.random.default_rng(it * 7919 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.003
                stagnant = 0
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-11:
            stagnant += 1
            if stagnant > 25:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M + 1e-14
        for alpha in np.linspace(0.01, 1.0, 25):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.005:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for alpha in np.linspace(lo, hi, 12):
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 10:
            m_relax = min(m_relax * 1.03, 1.04)
        if stagnant > 50:
            m_relax = min(m_relax * 1.1, 1.06)

        if verbose and it % 100 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stag={stagnant}")

    return best_f, best_obj


def peak_coord_descent(a0, time_limit):

    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_scale = 0.003

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            c_val = conv[k]
            if c_val <= best_obj * 0.95:
                continue

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Find top contributing pairs
            pairs = []
            for j in range(js, je + 1):
                prod = a[j] * a[k - j]
                pairs.append((prod, j, k - j))
            pairs.sort(reverse=True)

            for (_, j1, j2) in pairs[:6]:
                if a[j1] < 1e-10 and a[j2] < 1e-10:
                    continue

                # Try redistributing mass to random distant positions
                rng = np.random.default_rng((it * 9973 + k * 31 + j1 * 7) % (2**31))
                for _attempt in range(8):
                    j_target = int(rng.integers(0, n))
                    if abs(j_target - j1) < 3 and abs(j_target - j2) < 3:
                        continue

                    for src in [j1, j2]:
                        if a[src] < 1e-10:
                            continue
                        for delta_frac in [step_scale, step_scale * 0.5, step_scale * 2]:
                            delta = a[src] * delta_frac
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[j_target] += delta
                            if candidate[src] < -1e-14:
                                continue
                            candidate = np.maximum(candidate, 0)
                            cs = candidate.sum()
                            if cs < 1e-15:
                                continue
                            candidate = candidate / cs
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-15:
                                a = candidate
                                best_obj = val
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

            if improved:
                break

        if not improved:
            step_scale *= 0.92
            if step_scale < 1e-7:
                break
            if it % 300 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 3000:
                break

    return best_a, best_obj


def ifta_initialize(n, n_iters=500, seed=42):
    """Initialize using iterative Fourier transform for flatter convolution."""
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n) * 0.5)

    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        # Try to flatten |A|^2 (which relates to the spectrum of the autocorrelation)
        mags = np.abs(A)
        # Target: make non-DC magnitudes closer to their mean
        if len(mags) > 1:
            non_dc_mag = mags[1:]
            target = non_dc_mag.mean()
            for k in range(1, len(A)):
                if mags[k] > 1e-15:
                    blend = 0.3  # partial update to avoid oscillation
                    new_mag = (1 - blend) * mags[k] + blend * target
                    A[k] = A[k] / mags[k] * new_mag

        a = np.fft.irfft(A, fft_size)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    return a


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_diverse_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    x = np.arange(n, dtype=np.float64)

    # Parent (resampled)
    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    cos_t = 0.5 * (1 + np.cos(np.pi * x / max(n - 1, 1)))
    w = np.maximum(cos_t, 0.01)
    starts.append(w / w.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # IFTA-based
    starts.append(ifta_initialize(n, n_iters=200, seed=rng.integers(10000)))

    # Random Dirichlet (concentrated)
    starts.append(rng.dirichlet(np.ones(n) * 0.2))

    # Random Dirichlet (spread)
    starts.append(rng.dirichlet(np.ones(n) * 2.0))

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Multi-length, multi-start exploration (150s)
    2. Smooth gradient optimization - primary workhorse (900s)
    3. Frank-Wolfe polishing (300s)
    4. Peak coordinate descent + final micro-polish (250s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Multi-length, multi-start exploration ===
    print("\n=== Phase 1: Exploration ===")
    t_phase = time.monotonic()

    candidate_lengths = sorted(set([
        n_parent, max(256, n_parent - 64), n_parent + 64,
        512, 640, 768, 896, 1024, 1280
    ]))
    candidate_lengths = [l for l in candidate_lengths if 256 <= l <= 1536]

    for n_try in candidate_lengths:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:
            break

        starts = generate_diverse_starts(n_try, parent, rng)
        # Try top 3 starts with short smooth gradient
        for si, start in enumerate(starts[:3]):
            remaining = deadline - time.monotonic() - 1150
            if remaining < 8:
                break
            t_alloc = min(remaining, 25.0)
            print(f"  Exploring n={n_try}, start={si}, t={t_alloc:.0f}s")
            a_opt, _ = smooth_gradient_optimize(start, time_limit=t_alloc)
            try_seq(a_opt, f"expl_n{n_try}_s{si}")

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Smooth gradient optimization (primary) ===
    print("\n=== Phase 2: Smooth Gradient Optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Main long run
        remaining = deadline - time.monotonic() - 600
        if remaining > 60:
            t_alloc = min(remaining, 800.0)
            print(f"  Main smooth grad n={n_best}, T_init={0.001}, t={t_alloc:.0f}s")
            a_opt, _ = smooth_gradient_optimize(best_seq, time_limit=t_alloc, T_init=0.002)
            try_seq(a_opt, "main_smooth")

        # With different T schedules
        for T_init, T_decay in [(0.01, 0.999), (0.005, 0.9995), (0.0005, 0.9998)]:
            remaining = deadline - time.monotonic() - 350
            if remaining < 30:
                break
            t_alloc = min(remaining, 120.0)
            print(f"  Smooth grad T_init={T_init}, T_decay={T_decay}, t={t_alloc:.0f}s")
            a_opt, _ = smooth_gradient_optimize(best_seq, time_limit=t_alloc,
                                                 T_init=T_init, T_decay=T_decay)
            try_seq(a_opt, f"smooth_T{T_init}")

        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 200
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 80.0)
            print(f"  Smooth grad perturb={scale}, t={t_alloc:.0f}s")
            a_opt, _ = smooth_gradient_optimize(f_pert, time_limit=t_alloc, T_init=0.003)
            try_seq(a_opt, f"smooth_pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Frank-Wolfe polishing ===
    print("\n=== Phase 3: Frank-Wolfe Polishing ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Deep FW
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 250.0)
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fw")

        # FW from perturbed starts
        for scale in [0.001, 0.003, 0.008]:
            remaining = deadline - time.monotonic() - 50
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 60.0)
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak coordinate descent + micro-polish ===
    print("\n=== Phase 4: Peak Coord Descent + Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Peak coordinate descent
        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 150.0)
            print(f"  Peak coord descent, t={t_alloc:.0f}s")
            a_opt, _ = peak_coord_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_coord")

        # Alternate smooth grad + FW micro-steps
        for ri in range(6):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if ri % 2 == 0:
                t_alloc = min(remaining, 30.0)
                noise = rng.standard_normal(len(best_seq)) * 0.001
                start = project_simplex(best_seq + noise)
                a_opt, _ = smooth_gradient_optimize(start, time_limit=t_alloc, T_init=0.001)
                try_seq(a_opt, f"polish_smooth_{ri}")
            else:
                t_alloc = min(remaining, 25.0)
                a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{ri}")

        # Final peak coord
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_alloc = min(remaining, 30.0)
            a_opt, _ = peak_coord_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_peak")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
