                if abs(delta) < 1e-14:
                    continue

                new_vals = conv[ks] + 2.0 * delta * a[js]
                if 0 <= diag_offset < len(ks):
                    new_vals[diag_offset] += delta * delta

                max_affected = new_vals.max()
                new_max = max(max_affected, max_unaffected)
                obj = 2.0 * n * new_max

                if obj < best_obj:
                    best_obj = obj
                    best_t = t

            if best_t != old_ai:
                delta = best_t - old_ai
                a[i] = best_t
                new_vals = conv[ks] + 2.0 * delta * a[js]
                if 0 <= diag_offset < len(ks):
                    new_vals[diag_offset] += delta * delta
                conv[ks] = new_vals

                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if improved:
            stale = 0
        else:
            stale += 1
            if stale > 20:
                # Random perturbation to escape local minimum
                scale = 0.001 + 0.003 * min(1.0, (stale - 20) / 100.0)
                rng = np.random.default_rng(sweep * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                conv = fast_convolve(a)
                stale = 0
            elif stale > 100:
                break

    return best_a, best_score


def smoothed_subgradient(a0, time_limit, T_init=0.001, lr_init=0.02, top_k=5):
    """
    Smoothed subgradient with softmax-weighted peaks and adaptive annealing.
    Uses the top-k peaks weighted by their values.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        # Select top-k peaks with softmax weighting
        if top_k < len(conv):
            peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peak_idx = np.arange(len(conv))

        pv = conv[peak_idx]
        weights = np.exp(pv / T)
        weights /= weights.sum()

        # Compute weighted subgradient
        sg = np.zeros(n)
        for w, k in zip(weights, peak_idx):
            if w < 1e-10:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w * 2.0 * a[k - idx]

        # Try multiple step sizes
        best_candidate = None
        best_val = best_obj
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 2.0]:
            candidate = project_simplex(a - trial_lr * sg)
            c = fast_convolve(candidate)
            val = 2.0 * n * c.max()
            if val < best_val:
                best_val = val
                best_candidate = candidate

        if best_candidate is not None:
            a = best_candidate
            obj = best_val
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive parameters
        if no_improve > 15:
            T *= 1.08
            lr *= 0.96
        if no_improve > 40:
            T *= 0.7
        if no_improve > 200:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.002 + 0.004 * min(1.0, (no_improve - 200) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, max_lp_time=10):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        g = None
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                     'time_limit': min(max_lp_time, remaining * 0.3)})
            if res.success:
                g = np.maximum(res.x, 0.0)
        except Exception:
            pass

        if g is None:
            m_exp = min(m_exp * 1.5, 5.0)
            if m_exp > 4.5:
                m_exp = 1.0
            no_improve += 1
            if no_improve > 50:
                rng = np.random.default_rng(it * 7919)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 60:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = 2.0 * n * M
        for alpha in np.linspace(0.01, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = 2.0 * n * fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if 2.0 * n * M < best_obj - 1e-14:
                best_obj = 2.0 * n * M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

    return best_f, best_obj


def peak_target_optimize(a0, time_limit):
    """
    Targeted peak reduction: identify top contributing pairs for each peak
    and transfer mass to reduce them.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    stale = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        # Get top 3 peaks
        top_peaks = np.argsort(conv)[::-1][:3]

        improved = False
        for k_peak in top_peaks:
            k_peak = int(k_peak)
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            if js > je:
                continue
            affected = np.arange(js, je + 1)
            contributions = a[affected] * a[k_peak - affected]
            j_max = affected[np.argmax(contributions)]

            # Find a "safe" coordinate to receive mass
            # A safe coordinate is one that doesn't contribute to any top peak
            dangerous = set()
            for kp in top_peaks:
                kp = int(kp)
                js2 = max(0, kp - n + 1)
                je2 = min(n - 1, kp)
                dangerous.update(range(js2, je2 + 1))
                dangerous.update(range(kp - je2, kp - js2 + 1))

            safe_coords = [j for j in range(n) if j not in dangerous and j != j_max]
            if not safe_coords:
                safe_coords = [j for j in range(n) if j != j_max]

            j_min = min(safe_coords, key=lambda j: a[j]) if safe_coords else None
            if j_min is None or a[j_max] < 1e-10:
                continue

            # Try transferring mass
            for frac in [0.01, 0.02, 0.05, 0.1, 0.15]:
                delta = a[j_max] * frac
                if delta < 1e-12:
                    continue
                candidate = a.copy()
                candidate[j_max] -= delta
                candidate[j_min] += delta
                candidate = np.maximum(candidate, 0.0)
                candidate = candidate / candidate.sum()
                c = fast_convolve(candidate)
                score = 2.0 * n * c.max()
                if score < best_score - 1e-14:
                    a = candidate
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stale = 0
        else:
            stale += 1
            if stale > 50:
                rng = np.random.default_rng(it * 4567 + 3)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                stale = 0
            elif stale > 300:
                break

    return best_a, best_score


def try_and_update(a, state, label=""):
    """Evaluate and update state = [best_score, best_seq]."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        if label:
            print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search combining coordinate descent, smoothed subgradient,
    Frank-Wolfe with LP, and peak-targeted optimization across multiple n values.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.8f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)
    print("Starting optimization. Initial best: {:.8f} (n={})".format(state[0], n_parent))

    # === Phase 1: Multi-n quick exploration (~250s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([n_parent, 512, 640, 768, 896, 1024, 1280, 1536]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 2048]

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            ss = start.sum()
            if ss > 1e-10:
                start = start / ss
            else:
                start = np.ones(n_try) / n_try

        # Quick smoothed subgradient
        t_alloc = min(remaining, 30)
        a_opt, _ = smoothed_subgradient(start, time_limit=t_alloc,
                                         T_init=0.001, lr_init=0.02, top_k=5)
        try_and_update(a_opt, state, "sg_n{}".format(n_try))

    print("  Phase 1 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Coordinate descent on best (~600s) ===
    print("\n=== Phase 2: Coordinate Descent ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 600
        if remaining > 30:
            t_alloc = min(remaining, 500)
            n_best = len(state[1])
            print("  Coord descent n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = coord_descent_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "coord_best_n")

            # Also try coord descent on a few other n values
            for n_try in [n_best + 128, n_best - 128]:
                if n_try < 128 or n_try > 2048:
                    continue
                remaining = deadline - time.monotonic() - 400
                if remaining < 30:
                    break
                start = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, n_best), state[1])
                start = np.maximum(start, 0)
                start = start / start.sum()
                t_alloc = min(remaining, 60)
                print("  Coord descent n={}, t={:.0f}s".format(n_try, t_alloc))
                a_opt, _ = coord_descent_optimize(start, time_limit=t_alloc)
                try_and_update(a_opt, state, "coord_n{}".format(n_try))

    print("  Phase 2 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Smoothed subgradient + FW-LP (~400s) ===
    print("\n=== Phase 3: Subgradient + FW-LP ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        n_best = len(state[1])

        # Multiple temperature settings for smoothed subgradient
        configs = [
            (0.0005, 0.03, 3),
            (0.001, 0.02, 5),
            (0.002, 0.015, 10),
            (0.005, 0.01, 20),
        ]
        for T_init, lr_init, top_k in configs:
            remaining = deadline - time.monotonic() - 200
            if remaining < 20:
                break
            t_alloc = min(remaining, 80)
            print("  Smoothed SG T={}, lr={}, k={}, t={:.0f}s".format(
                T_init, lr_init, top_k, t_alloc))
            a_opt, _ = smoothed_subgradient(state[1], time_limit=t_alloc,
                                             T_init=T_init, lr_init=lr_init, top_k=top_k)
            try_and_update(a_opt, state, "sg_T{}_k{}".format(int(T_init*1000), top_k))

        # FW-LP
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            t_alloc = min(remaining, 100)
            print("  FW-LP n={}, t={:.0f}s".format(n_best, t_alloc))
            a_opt, _ = fw_lp_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "fw_lp")

    print("  Phase 3 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Peak targeting + coordinate descent polish (~350s) ===
    print("\n=== Phase 4: Peak Targeting + Polish ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        # Peak-targeted optimization
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print("  Peak targeting, t={:.0f}s".format(t_alloc))
            a_opt, _ = peak_target_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "peak_target")

        # More coordinate descent (finer)
        remaining = deadline - time.monotonic() - 100
        if remaining > 20:
            t_alloc = min(remaining, 100)
            print("  Coord descent polish, t={:.0f}s".format(t_alloc))
            a_opt, _ = coord_descent_optimize(state[1], time_limit=t_alloc)
            try_and_update(a_opt, state, "coord_polish")

        # Alternating: smoothed SG -> coord descent -> peak target
        for round_i in range(3):
            remaining = deadline - time.monotonic() - 50
            if remaining < 15:
                break
            t1 = min(remaining * 0.4, 30)
            t2 = min(remaining * 0.4, 30)
            a_opt, _ = smoothed_subgradient(state[1], time_limit=t1,
                                             T_init=0.0003, lr_init=0.01, top_k=3)
            try_and_update(a_opt, state, "altern_sg_{}".format(round_i))
            a_opt, _ = coord_descent_optimize(state[1], time_limit=t2)
            try_and_update(a_opt, state, "altern_cd_{}".format(round_i))

    print("  Phase 4 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Final fine-tuning with tiny perturbations (~150s) ===
    print("\n=== Phase 5: Fine Tuning ===")
    t_phase = time.monotonic()

    if state[1] is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            # Tiny random perturbations + quick coordinate descent
            n_best = len(state[1])
            n_rounds = max(1, int(remaining // 10))
            for r in range(n_rounds):
                remaining = deadline - time.monotonic() - 5
                if remaining < 8:
                    break
                noise_scale = 0.0003 + 0.0005 * (r % 5)
                rng_r = np.random.default_rng(seed + r * 1009)
                perturbed = project_simplex(state[1] + rng_r.standard_normal(n_best) * noise_scale)
                t_alloc = min(remaining, 8)
                a_opt, _ = coord_descent_optimize(perturbed, time_limit=t_alloc)
                try_and_update(a_opt, state, "fine_{}".format(r))

        # One final smoothed gradient pass with very low temperature
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_opt, _ = smoothed_subgradient(state[1], time_limit=remaining,
                                             T_init=0.0001, lr_init=0.005, top_k=2)
            try_and_update(a_opt, state, "final_sg")

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Fallback
    if state[1] is None:
        state[1] = np.ones(384) / 384
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
