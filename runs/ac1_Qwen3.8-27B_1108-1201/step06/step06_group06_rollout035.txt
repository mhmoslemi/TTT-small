"""
Smoothed gradient optimization with Nesterov momentum for minimum peak auto-convolution.
Uses log-sum-exp smoothing of the max, adaptive temperature, and local search.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_gradient(a, T):
    """Compute smoothed gradient of log-sum-exp(T * convolve(a,a))."""
    n = len(a)
    C = fast_convolve(a)
    scaled = T * C
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(C))
        w[C.argmax()] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2*T * sum_{j=0}^{n-1} w[i+j] * a[j]
    idx = np.arange(n)
    W = w[idx[:, None] + idx[None, :]]
    grad = 2.0 * T * (W @ a)
    return grad, C, w

def exact_objective(a):
    """Compute the exact objective (before dividing by 2n, just the peak)."""
    C = fast_convolve(a)
    return C.max()

                       T_decay=0.9999, min_T=100.0, seed=0):

    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    
    if T_init is None:
        C = fast_convolve(a)
        std = C.std()
        T_init = 1.0 / max(std, 1e-6)
    
    T = T_init
    lr = lr_init
    v = np.zeros(n)
    
    best_a = a.copy()
    best_obj = exact_objective(a)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        grad, C, w = smoothed_gradient(a, T)
        
        # Nesterov step
        v = momentum * v + (1 - momentum) * grad

"""
Hybrid optimization for minimum peak auto-convolution step function.
Combines smoothed gradient with Nesterov momentum, subgradient method,
LP-based Frank-Wolfe, and multi-scale local search.
Key insight: for non-negative sequences, the auto-convolution peak can be
reduced below the uniform bound by exploiting asymmetry.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, C, T):
    """Compute gradient of log-sum-exp(T*C) w.r.t. a.
    grad[i] = 2*T * sum_{j=0}^{n-1} w[i+j] * a[j]
    where w = softmax(T*C)
    """
    n = len(a)
    scaled = T * C
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(C))
        w[np.argmax(C)] = 1.0
    else:
        w /= ws
    idx = np.arange(n)
    W = w[idx[:, None] + idx[None, :]]
    grad = 2.0 * T * (W @ a)
    return grad, w


def compute_subgrad(a, C):
    """Compute subgradient at the peak of C. Returns (grad, peak_idx)."""
    n = len(a)
    k_star = int(np.argmax(C))
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    for i in range(lo, hi + 1):
        grad[i] = 2.0 * a[k_star - i]
    return grad, k_star


def smoothed_grad_optimize(a0, time_limit, T_init=None, lr_init=0.01,

    else:
        a /= s
    n = len(a)

    if T_init is None:
        C = fast_convolve(a)
        spread = C.max() - C[C > 0].min() if (C > 0).any() else 0.1
        T_init = max(1.0 / max(spread, 1e-6), 5.0)

    T = T_init
    lr = lr_init
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        if use_nesterov:
            v = momentum * v + (1.0 - momentum) * grad
            step = v
        else:
            step = grad

        best_candidate = a
        best_val = fast_convolve(a).max()

def build_lp_arrays(n):
    """Build sparse index arrays for LP oracle."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)


def fw_lp_optimize(a0, time_limit, seed=0):
    """Frank-Wolfe with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)

    row_idx, col_idx = build_lp_arrays(n)
    C = fast_convolve(a)
    M = C.max()
    best_a = a.copy()
    best_obj = M
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp, g >= 0
        c = -np.ones(n)
        vals = np.empty(len(row_idx))
        for idx in range(len(row_idx)):
            k = row_idx[idx]
            i = col_idx[idx]
            vals[idx] = a[k - i]

        A_mat = np.zeros((2 * n - 1, n))
        for idx in range(len(row_idx)):
            A_mat[row_idx[idx], col_idx[idx]] = vals[idx]

        b = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
            if res.success:
                g = np.maximum(res.x, 0.0)
                gs = g.sum()
                if gs > 1e-15:
                    g /= gs
                else:
                    g = None
            else:
                g = None
        except Exception:
            g = None

        if g is None:
            m_exp = min(m_exp * 1.5, 10.0)
            if m_exp >= 10.0:
                g = project_simplex(a + rng.standard_normal(n) * 0.005)
                m_exp = 1.0
            else:
                continue

        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
                C = fast_convolve(a)
                M = C.max()
                m_exp = 1.0
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1 - alpha) * a + alpha * g, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = np.maximum((1 - best_alpha) * a + best_alpha * g, 0)
            a /= a.sum()
            C = fast_convolve(a)
            M = C.max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 10.0)

    return best_a, best_obj

            break

    return best_a, best_obj


def game_iteration(a0, time_limit, T_start=0.001, lr_start=0.02, seed=0):
    """Game-theoretic / von Neumann minimax iteration."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    T = T_start
    lr = lr_start
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.01:
            break

        C = fast_convolve(a)
        scaled = C / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()

        # Gradient of p^T C w.r.t. a
        # Same as smoothed gradient with the softmax weights p
        idx = np.arange(n)
        W = p[idx[:, None] + idx[None, :]]
        grad = 2.0 * (W @ a)

        # Inner projected gradient steps
        a_inner = a.copy()
        for _ in range(3):
            candidate = project_simplex(a_inner - lr * grad)
            new_obj = np.dot(p, fast_convolve(candidate))
            old_obj = np.dot(p, fast_convolve(a_inner))
            if new_obj < old_obj - 1e-16:
                a_inner = candidate
            else:
                break

        a = a_inner
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            T *= 1.02
            lr *= 0.99
        if no_improve > 200:
            T = T_start
            lr = lr_start
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_a, best_obj


def try_and_update(a, label, best_seq, best_score):
    """Evaluate sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score

            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print("\n=== Phase 1: Exploration ===")
