def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""

    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0)

"""
Hybrid optimization for minimum peak auto-convolution step function.
Combines FW-LP, Nesterov-accelerated smoothed gradient, direct QP solving

Multi-n exploration with adaptive time allocation.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_fft(a, b):
    """General linear convolution via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_grad(a, w):
    """Compute gradient of sum_k w[k]*(a*a)[k] w.r.t. a.
    Uses FFT for efficiency. grad[j] = 2 * sum_m a[m] * w[m+j]
    """
    n = len(a)
    # result[j] = sum_{m=0}^{n-1} a[m] * w[m+j]
    # Use cross-correlation via FFT
    w_rev = w[::-1]
    conv = convolve_fft(a, w_rev)
    # conv has length n + (2n-1) - 1 = 3n-2
    # result[j] = conv[2n-2-j] for j=0,...,n-1
    # i.e., result = conv[2n-2], conv[2n-3], ..., conv[n-1]
    result = conv[n-1:2*n-1][::-1]
    return 2.0 * result


def build_lp_matrix(n):
    """Build sparse constraint matrix for LP oracle."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0. Returns normalized g or None."""
    A_mat = lil_matrix((2 * n - 1, n))
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        i = col_idx[idx]
        A_mat[k, i] = f[k - i]
    A_mat = A_mat.tocsr()
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 8})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None

def qp_hankel_solve(w, n, max_iter=50):
    """Solve min x^T Q x s.t. sum(x)=1, x>=0 where Q[i,j]=w[i+j].
    Uses active set method. Returns solution or None.
    """
    # Build Q (Hankel matrix)
    idx = np.arange(n)
    Q = w[np.ix_(idx[:, None] + idx[None, :])]  # Q[i,j] = w[i+j]
    
    # Start with uniform
    active = np.ones(n, dtype=bool)  # all variables active (free)
    x = np.ones(n) / n
    
    for iteration in range(max_iter):
        # Solve on active set: min x_A^T Q_AA x_A s.t. sum(x_A) = 1
        Q_aa = Q[np.ix_(active, active)]
        n_act = active.sum()
        
        try:
            # Interior solution: Q_aa @ x = -lambda/2 * 1
            # x = -lambda/2 * Q_aa^{-1} @ 1, normalized
            ones = np.ones(n_act)
            Q_inv_1 = np.linalg.solve(Q_aa + 1e-12 * np.eye(n_act), ones)
            x_act = Q_inv_1 / Q_inv_1.sum()
        except np.linalg.LinAlgError:
            # Fallback: use pseudo-inverse
            try:
                Q_inv_1 = np.linalg.pinv(Q_aa) @ ones
                x_act = Q_inv_1 / Q_inv_1.sum()
            except:
                return None
        
        # Check feasibility
        x_candidate = np.zeros(n)
        x_candidate[active] = x_act
        
        if np.min(x_act) > 1e-10:
            # Interior solution is feasible
            return np.maximum(x_candidate, 0)
        
        # Find most violated constraint
        min_idx = np.argmin(x_act)
        min_val = x_act[min_idx]
        
        if min_val > -1e-8:
            # Essentially feasible
            x_act = np.maximum(x_act, 0)
            x_candidate = np.zeros(n)
            x_candidate[active] = x_act
            x_candidate = x_candidate / x_candidate.sum()
            return x_candidate
        
        # Add constraint: fix this variable at 0
        active_indices = np.where(active)[0]
        new_fixed = active_indices[min_idx]
        active[new_fixed] = False
        
        if active.sum() < 2:
            return None
        if active.sum() <= 10:
            # Problem too small, just do uniform on active
            x_candidate = np.zeros(n)
            x_candidate[active] = 1.0 / active.sum()
            return x_candidate
    
    # Return best effort
    x_candidate = np.zeros(n)
    x_candidate[active] = 1.0 / active.sum()
    return x_candidate
