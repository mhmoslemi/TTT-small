def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        # For k >= n: second term i in [k, n-1] is empty
        # So: d/d(a[i]) = a[k-i]*1_{js<=i<=je} + a[i-k]*1_{k<=i<=n-1} (when k<n)
        
        # Contribution from first term: a[k-j] for j in [js, je]
        idx_arr = np.arange(js, je + 1)
        grad[idx_arr] += weights[idx] * a[k - idx_arr]
        
        # Contribution from second term: a[i-k] for i in [k, n-1]
        if k < n:
            idx_arr2 = np.arange(k, n)
            grad[idx_arr2] += weights[idx] * a[idx_arr2 - k]

    return grad


def nesterov_subgradient(a0, time_limit, lr_init=0.03, T_init=0.003,
                         active_frac_init=0.03, verbose=False):
    """
    Nesterov-accelerated subgradient method with adaptive parameters.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    # Nesterov state
    y = a.copy()
    prev_a = a.copy()
    t_nest = 1.0
    it = 0

    # Adaptive parameters
    T = T_init
    lr = lr_init
    active_frac = active_frac_init
    stagnant = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        conv_y = fast_convolve(y)
        subgrad = compute_subgradient(y, conv_y, active_frac, T)

        # Step size: Nesterov decay
        alpha = lr / (1.0 + 0.05 * t_nest)

        # Candidate: move in negative subgradient direction
        candidate = project_simplex(y - alpha * subgrad)

        # Nesterov extrapolation for next y
        momentum = (t_nest - 1.0) / (t_nest + 1.0)
        new_a = candidate
        y = new_a + momentum * (new_a - prev_a)
        y = np.maximum(y, 0.0)
        y_s = y.sum()
        if y_s < 1e-15:
            y = new_a
        else:
            y = y / y_s
        prev_a = new_a
        t_nest += 1.0

        # Evaluate
        conv_new = fast_convolve(new_a)
        obj = conv_new.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = new_a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Adaptive parameter updates
        if stagnant > 30:
            T *= 1.08
            active_frac = min(active_frac * 1.1, 0.5)
            lr *= 0.97
        if stagnant > 100:
            T *= 0.5
            active_frac = max(active_frac * 0.7, 0.01)
            lr = min(lr * 1.3, 0.1)
        if stagnant > 300:
            # Restart with perturbation
            rng = np.random.default_rng((it * 7919) % (2**31))
            noise_scale = 0.001 + 0.003 * min(1.0, (stagnant - 300) / 300.0)
            new_start = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = new_start
            prev_a = new_start.copy()
            a = new_start.copy()
            t_nest = 1.0
            T = T_init
            lr = lr_init
            active_frac = active_frac_init
            stagnant = 0

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NS it={it}, obj={best_obj:.10f}, T={T:.6f}, alpha={alpha:.6f}, stag={stagnant}, t={elapsed:.0f}s")

    return best_a, best_obj


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix A where (f*g) = A @ g."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and extensive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.001

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_relax)
        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.005)
        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 15:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.001 + 0.003 * min(1.0, stagnant / 40.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.001
                stagnant = 0
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 25:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M + 1e-14

        alphas_coarse = np.linspace(0.01, 1.0, 25)
        for alpha in alphas_coarse:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.01:
            lo = max(0.0, best_alpha - 0.012)
            hi = min(1.0, best_alpha + 0.012)
            alphas_fine = np.linspace(lo, hi, 12)
            for alpha in alphas_fine:
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.001

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 8:
            m_relax = min(m_relax * 1.03, 1.03)

        if verbose and it % 50 == 0:
            print(f"      FW it={it}, obj={best_obj:.10f}, stag={stagnant}")

    return best_f, best_obj


def coordinate_descent_polish(a0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: shift mass from peak contributors
    to reduce the maximum convolution peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_scale = 0.003
    n_peaks = 5

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        M = conv.max()
        n_actual_peaks = min(n_peaks, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_actual_peaks]

        improved_this_round = False

        for ki in range(n_actual_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue

            # Find top contributing pairs for this peak
            contributions = []
            for j in range(js, je + 1):
                c_val = a[j] * a[k - j]
                contributions.append((c_val, j, k - j))
            contributions.sort(reverse=True)

            for (_, j1, j2) in contributions[:4]:
                if a[j1] < 1e-10 and a[j2] < 1e-10:
                    continue

                # Try shifting mass from j1 or j2 to a less critical location
                for src in [j1, j2]:
                    if a[src] < 1e-10:
                        continue

                    # Target: a location that doesn't contribute to top peaks
                    rng = np.random.default_rng((it * 9973 + k * 31 + src * 7) % (2**31))
                    for _ in range(6):
                        j_target = int(rng.integers(0, n))
                        if j_target == src or j_target == j1 or j_target == j2:
                            continue

                        for delta_frac in [step_scale, step_scale * 0.5, step_scale * 1.5]:
                            delta = a[src] * delta_frac
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[j_target] += delta
                            if candidate[src] < -1e-14:
                                continue
                            candidate = np.maximum(candidate, 0)
                            cs = candidate.sum()
                            if cs < 1e-15:
                                continue
                            candidate = candidate / cs
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-15:
                                a = candidate
                                best_obj = val
                                best_a = a.copy()
                                improved_this_round = True
                                break
                        if improved_this_round:
                            break
                    if improved_this_round:
                        break
                if improved_this_round:
                    break

            if improved_this_round:
                break

        if not improved_this_round:
            step_scale *= 0.92
            if step_scale < 1e-7:
                break
            if it % 100 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 3000:
                break

        if verbose and it % 200 == 0:
            print(f"      CD it={it}, obj={best_obj:.10f}, step={step_scale:.6f}")

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length via interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    x = np.arange(n, dtype=np.float64)

    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper (reduces edge peaks)
    cos_t = np.cos(np.pi * x / (n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.001)
    starts.append(w / w.sum())

    # Edge concentration (reduces center peak)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    starts.append(edge / edge.sum())

    # Strong edge concentration
    edge2 = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 1.5
    starts.append(edge2 / edge2.sum())

    # Two-peak (symmetric)
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.06 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.06 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Broad two-peak
    g1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.12 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.12 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Random Dirichlet
    starts.append(rng.dirichlet(np.ones(n) * 0.2))
    starts.append(rng.dirichlet(np.ones(n) * 0.5))

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Quick multi-n exploration (250s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()

    lengths = sorted(set([n_parent, max(256, n_parent - 128), n_parent + 128,
                          384, 512, 640, 768, 896, 1024, 1280]))
    lengths = [l for l in lengths if 192 <= l <= 1536]

    for n_try in lengths:
        remaining_to_reserve = deadline - time.monotonic() - 1300
        if remaining_to_reserve < 10:
            break

        starts = generate_starts(n_try, parent, rng)
        n_starts = min(3, len(starts))

        for si in range(n_starts):
            remaining_to_reserve = deadline - time.monotonic() - 1250
            if remaining_to_reserve < 8:
                break
            t_alloc = min(remaining_to_reserve, 25)
            a_opt, _ = nesterov_subgradient(starts[si], time_limit=t_alloc,
                                            lr_init=0.04, T_init=0.005,
                                            active_frac_init=0.05)
            label = f"ns_n{n_try}_s{si}"
            try_seq(a_opt, label)

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep Nesterov subgradient on best (500s) ===
    print("\n=== Phase 2: Deep Nesterov Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {'lr_init': 0.025, 'T_init': 0.003, 'active_frac_init': 0.03},
            {'lr_init': 0.015, 'T_init': 0.001, 'active_frac_init': 0.01},
            {'lr_init': 0.04, 'T_init': 0.01, 'active_frac_init': 0.08},
        ]

        for ci, cfg in enumerate(configs):
            remaining_to_reserve = deadline - time.monotonic() - 700
            if remaining_to_reserve < 30:
                break
            t_alloc = min(remaining_to_reserve, 180)
            print(f"  NS config {ci} (lr={cfg['lr_init']}), t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgradient(best_seq, time_limit=t_alloc, verbose=True, **cfg)
            try_seq(a_opt, f"deep_ns_{ci}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: FW with LP oracle (450s) ===
    print("\n=== Phase 3: Frank-Wolfe ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Main FW run
        remaining_to_reserve = deadline - time.monotonic() - 500
        if remaining_to_reserve > 40:
            t_alloc = min(remaining_to_reserve, 350)
            print(f"  Deep FW t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts
        for scale in [0.0008, 0.002, 0.005]:
            remaining_to_reserve = deadline - time.monotonic() - 350
            if remaining_to_reserve < 25:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
            t_alloc = min(remaining_to_reserve, 60)
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Coordinate descent polish (250s) ===
    print("\n=== Phase 4: Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining_to_reserve = deadline - time.monotonic() - 200
        if remaining_to_reserve > 20:
            t_alloc = min(remaining_to_reserve, 200)
            print(f"  CD t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "coord_desc")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final multi-restart and polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Alternate short NS and CD runs with perturbations
        polish_count = 6
        for ri in range(polish_count):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            if ri % 3 == 0:
                # Short Nesterov with small perturbation
                noise = rng.standard_normal(n_best) * 0.0005
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 30)
                a_opt, _ = nesterov_subgradient(start, time_limit=t_alloc,
                                                lr_init=0.01, T_init=0.001,
                                                active_frac_init=0.01)
                try_seq(a_opt, f"polish_ns_{ri}")
            elif ri % 3 == 1:
                # Short CD
                t_alloc = min(remaining, 25)
                a_opt, _ = coordinate_descent_polish(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_cd_{ri}")
            else:
                # Short FW
                t_alloc = min(remaining, 30)
                a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{ri}")

        # Try slightly different n
        for n_delta in [32, -32, 64, -64]:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            n_try = len(best_seq) + n_delta
            if n_try < 128 or n_try > 2048:
                continue
            a_resampled = resample_sequence(best_seq, n_try)
            t_alloc = min(remaining, 25)
            a_opt, _ = nesterov_subgradient(a_resampled, time_limit=t_alloc,
                                            lr_init=0.02, T_init=0.002,
                                            active_frac_init=0.02)
            try_seq(a_opt, f"polish_n{n_try}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
