def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute LSE-smoothed max of conv(a,a) and its gradient wrt a.
    Returns (lse_value, grad_a)."""
    n = len(a)
    c = fast_convolve(a)
    
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    denom = exp_bc.sum()

    C = conv_aw[n-1 : 3*n-2]  # length n
