                        fg2 = np.convolve(f, g2, mode='full')

def fw_lp_optimize(f0, time_limit, top_k=100):
    """Frank-Wolfe with active-set LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.1:
            break

        g = solve_lp_oracle_active(f, M * m_exp, top_k=top_k, 
                                    time_limit=min(5, remaining * 0.3))
        if g is None:
            m_exp = min(m_exp * 1.2, 3.0)
            if m_exp > 2.5:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.001 + 0.005 * min(1.0, no_improve / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                continue
            m_exp = min(m_exp * 1.03, 2.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.01, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if 0.01 < best_alpha < 0.99:

                    break

                val = fast_convolve(ft).max()
                if val < best_val:

                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 3.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 60) / 150.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj

def coordinate_descent_polish(a0, time_limit):
    """Coordinate-wise descent for final polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stale = 0
    order = np.random.default_rng(42).permutation(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Try moving mass from position order[it % n] to others
        i = order[it % n]
        if a[i] < 1e-10:
            stale += 1
            if stale > n * 2:
                break
            continue

        # Find the j that benefits most from receiving mass from i
        # Moving delta from i to j changes conv by:
        # -delta * a[k-i] - delta * a[k-j] + ... at each peak
        # Simplified: try a few candidate j's
        candidates = np.array([order[(it + offset) % n] for offset in [1, -1, n//3, 2*n//3]])
        candidates = candidates[candidates != i]

        for j in candidates:
            for delta_frac in [0.01, 0.05, 0.1, 0.3]:
                delta = delta_frac * a[i]
                if delta <= 0 or a[i] - delta < 0:
                    continue
                candidate = a.copy()
                candidate[i] -= delta
                candidate[j] += delta
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                    break
            if stale == 0:
                break

        stale += 1
        if stale > n * 5:
            # Random perturbation
            rng = np.random.default_rng(it * 12345 + 99)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            stale = 0

    return best_a, best_obj


def multi_scale_optimize(parent, time_limit):
    """Coarse-to-fine optimization."""
    # Start at quarter resolution, then double
    n_parent = len(parent)
    n_coarse = max(32, n_parent // 4)
    
    # Downsample
    coarse = np.interp(np.linspace(0, 1, n_coarse), 
                       np.linspace(0, 1, n_parent), parent)
    coarse = np.maximum(coarse, 0)
    cs = coarse.sum()
    if cs > 1e-10:
        coarse = coarse / cs

    # Optimize at coarse level
    t_coarse = time_limit * 0.3
    coarse_opt, _ = fw_lp_optimize(coarse, time_limit=t_coarse, top_k=30)
    
    # Upsample to parent level
    fine = np.interp(np.linspace(0, 1, n_parent),
                     np.linspace(0, 1, n_coarse), coarse_opt)
    fine = np.maximum(fine, 0)
    fs = fine.sum()
    if fs > 1e-10:
        fine = fine / fs
    
    return fine


def fourier_init(n, mode_count=8, seed=42):
    """Generate a Fourier-based initialization."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for m in range(1, mode_count + 1):
        amp = rng.uniform(0.05, 0.2) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * m * x / n + phase)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a

"""
Improved hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous:
- Active-set LP oracle for faster Frank-Wolfe iterations
- Adaptive temperature annealing for smoothed gradient
- Multi-resolution (coarse-to-fine) optimization
- Structured initializations (Fourier, chirp-like patterns)
- Aggressive coordinate-wise polishing
- Better time management and exploration

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle_active(f, M, top_k=100, time_limit=8):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
    Using active set: only enforce top_k most likely binding constraints.
    """
    n = len(f)
    conv_len = 2 * n - 1

    if n <= 200:
        idx = np.arange(conv_len)
    else:
        ff = fast_convolve(f)
        idx = np.argpartition(ff, -top_k)[-top_k:]

    A_ub = np.zeros((len(idx), n))
    for i, k in enumerate(idx):
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        m_range = np.arange(m_min, m_max + 1)
        A_ub[i, m_range] = f[k - m_range]

    c = -np.ones(n)
    b_ub = np.full(len(idx), M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = np.maximum(res.x, 0.0)
            fg = np.convolve(f, g, mode='full')
            if fg.max() <= M + 1e-7:
                return g
            else:
                violated = np.where(fg > M)[0]
                if len(violated) > 0:
                    extra_k = violated[np.argsort(fg[violated])[::-1][:min(80, len(violated))]]
                    all_k = np.unique(np.concatenate([idx, extra_k]))
                    A_ub2 = np.zeros((len(all_k), n))
                    for i, k in enumerate(all_k):
                        m_min = max(0, k - n + 1)
                        m_max = min(n - 1, k)
                        m_range = np.arange(m_min, m_max + 1)
                        A_ub2[i, m_range] = f[k - m_range]
                    res2 = linprog(c, A_ub=A_ub2, b_ub=np.full(len(all_k), M),
                                   bounds=bounds, method='highs',
                                   options={'presolve': True, 'time_limit': time_limit})
                    if res2.success:
                        g2 = np.maximum(res2.x, 0.0)
                        fg2 = np.convolve(f, g2, mode='full')
                        if fg2.max() <= M + 1e-7:
                            return g2
        return None
    except Exception:
        return None


def fw_lp_optimize(f0, time_limit, top_k=100):
    """Frank-Wolfe with active-set LP oracle and adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.2:
            break

        g = solve_lp_oracle_active(f, M * m_exp, top_k=top_k,
                                    time_limit=min(6.0, remaining * 0.4))
        if g is None:
            m_exp = min(m_exp * 1.15, 3.0)
            if m_exp > 2.5:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 25:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.001 + 0.005 * min(1.0, no_improve / 80.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                continue
            m_exp = min(m_exp * 1.02, 2.0)
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.01, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if 0.01 < best_alpha < 0.99:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 3.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 50) / 120.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def smoothed_grad_descent(a0, time_limit, T_init=None, lr_init=None):
    """Smoothed gradient descent with adaptive temperature and learning rate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if T_init is None:
        conv0 = fast_convolve(a)
        T_init = conv0.max() / n * 5
    if lr_init is None:
        lr_init = 0.03

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * np.correlate(p, a, mode='valid')

        best_candidate = a
        best_val = conv.max()

        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 1.5]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            T *= 1.03
            lr *= 0.98
        if no_improve > 40:
            T *= 0.85
        if no_improve > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.003 * min(1.0, (no_improve - 100) / 150.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit):
    """Coordinate-wise descent for final polishing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stale = 0
    rng = np.random.default_rng(42)
    order = rng.permutation(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        i = order[it % n]
        if a[i] < 1e-12:
            stale += 1
            if stale > n * 3:
                rhs = np.random.default_rng(it + 99)
                a = project_simplex(best_a + rhs.standard_normal(n) * 0.001)
                stale = 0
            continue

        candidates = [order[(it + 1) % n], order[(it + n // 3) % n],
                      order[(it + 2 * n // 3) % n], order[(it + n - 1) % n]]
        candidates = [c for c in candidates if c != i]

        improved = False
        for j in candidates:
            for delta_frac in [0.01, 0.05, 0.1, 0.2, 0.4]:
                delta = delta_frac * a[i]
                if delta <= 0:
                    continue
                candidate = a.copy()
                candidate[i] -= delta
                candidate[j] += delta
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    stale = 0
                    break
            if improved:
                break

        if not improved:
            stale += 1
            if stale > n * 5:
                rhs = np.random.default_rng(it * 12345 + 7)
                a = project_simplex(best_a + rhs.standard_normal(n) * 0.0008)
                stale = 0

    return best_a, best_obj


def multi_scale_optimize(parent, time_limit):
    """Coarse-to-fine optimization: downsample, optimize, upsample."""
    n_parent = len(parent)
    n_coarse = max(32, n_parent // 4)

    coarse = np.interp(np.linspace(0, 1, n_coarse),
                       np.linspace(0, 1, n_parent), parent)
    coarse = np.maximum(coarse, 0)
    cs = coarse.sum()
    if cs > 1e-10:
        coarse = coarse / cs

    t_coarse = time_limit * 0.4
    coarse_opt, _ = fw_lp_optimize(coarse, time_limit=t_coarse, top_k=40)

    fine = np.interp(np.linspace(0, 1, n_parent),
                     np.linspace(0, 1, n_coarse), coarse_opt)
    fine = np.maximum(fine, 0)
    fs = fine.sum()
    if fs > 1e-10:
        fine = fine / fs

    return fine


def fourier_init(n, mode_count=8, seed=42):
    """Generate a Fourier-based initialization with random phases."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for m in range(1, mode_count + 1):
        amp = rng.uniform(0.05, 0.25) / m
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * m * x / n + phase)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def chirp_init(n, seed=42):
    """Generate a chirp-like initialization."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    f0 = rng.uniform(1, 5)
    f1 = rng.uniform(5, 20)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * (f0 * x + 0.5 * (f1 - f0) * x * x / n))
    a = np.maximum(a, 0.05)
    a = a / a.sum()
    return a

        # Tiny perturbations with quick polish
        remaining = deadline - time.monotonic() - 5
        if remaining > 3:
            for trial in range(4):
                remaining = deadline - time.monotonic() - 3
                if remaining < 2:
                    break
                noise = rng.standard_normal(len(state[1])) * 0.0004
                f_pert = project_simplex(state[1] + noise)
                a_opt, _ = smoothed_grad_descent(f_pert, time_limit=min(remaining, 4),
                                                 T_init=0.0002, lr_init=0.003)
                try_and_update(a_opt, state, "perturb_{}".format(trial))

    print("  Phase 5 best: {:.8f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # Final safety: ensure we have a valid sequence
    if state[1] is None:
        state[1] = np.ones(512) / 512
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.8f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
