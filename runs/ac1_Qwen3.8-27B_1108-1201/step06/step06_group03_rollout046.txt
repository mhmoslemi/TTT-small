"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Smooth gradient (Adam) with temperature annealing on the simplex
2. LP-based Frank-Wolfe for refinement
3. Peak-targeted coordinate descent
4. Multi-length and multi-start exploration
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, w):
    """Compute gradient of smooth max(conv) objective.
    a: length n (on simplex)
    w: length 2n-1 (softmax weights over convolution peaks)
    Returns: gradient of length n
    """
    n = len(a)
    # grad[j] = sum_m w[j+m] * a[m] for m=0..n-1 (with w[k]=0 for k>=2n-1)
    # This is the cross-correlation of w and a at lags 0..n-1
    full_corr = np.correlate(w, a, mode='full')
    return full_corr[n - 1: 2 * n - 1]


def smooth_adam_optimize(a0, time_limit, T_start=None, T_end=None, lr_init=0.01,
                         decay_start=0.5, decay_end=0.01, verbose=False):
    """Smooth gradient descent with Adam on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if T_start is None:
        T_start = 5.0 * n
    if T_end is None:
        T_end = 30.0 * n
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    t0 = time.monotonic()
    it = 0
    stagnant = 0
    stagnant_patience = 500
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.02:
            break
        it += 1
        
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        
        # Temperature schedule (geometric)
        T = T_start * (T_end / T_start) ** frac
        
        # Learning rate schedule
        current_lr = lr_init * (decay_end / decay_start) ** frac
        
        # Compute convolution
        conv = fast_convolve(a)
        
        # Softmax weights
        conv_max = conv.max()
        exp_vals = np.exp(T * (conv - conv_max))
        w = exp_vals / exp_vals.sum()
        
        # Gradient
        grad = compute_smooth_grad(a, w)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        
        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        # Evaluate
        obj = fast_convolve(a_new).max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            
        a = a_new
        
        # Restart if stagnant
        if stagnant > stagnant_patience:
            rng = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.001 + 0.003 * min(1.0, (stagnant - stagnant_patience) / 500.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            stagnant = 0
        
        if verbose and it % 5000 == 0:
            print(f"      Adam it={it}, obj={best_obj:.10f}, T={T:.1f}, lr={current_lr:.6f}, stag={stagnant}")
    
    return best_a, best_obj


def build_conv_matrix(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M):
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None
