def project_simplex(v):

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_peak_grad(a, conv, T, power=1.0):
    """
    Compute gradient of sum_k max(conv[k]-T, 0)^power * 2 w.r.t. a.
    Uses the identity: d(conv[k])/da[j] = a[k-j] (for valid indices).
    Result: grad[j] = 2 * sum_k excess[k] * a[k-j]
    """
    n = len(a)
    excess = np.maximum(conv - T, 0.0)
    if excess.max() < 1e-16:
        return np.zeros(n)
    w = np.power(excess, power)
    # grad[j] = 2 * sum_k w[k] * a[k-j] = 2 * correlate(w, a, 'valid')[j]
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def multiplicative_step(a, conv, T, lr, power=1.0):
    """
    Multiplicative weight update: a[j] *= exp(-lr * contrib[j])
    where contrib[j] = sum_k excess[k] * a[k-j]
    This is the multiplicative analog of gradient descent for non-negative variables.
    """
    n = len(a)
    excess = np.maximum(conv - T, 0.0)
    if excess.max() < 1e-16:
        return a.copy()
    w = np.power(excess, power)
    contrib = np.correlate(w, a, mode='valid')
    # Scale lr relative to the magnitude of contrib
    max_contrib = contrib.max()
    if max_contrib < 1e-16:
        return a.copy()
    scaled_lr = lr * max_contrib
    log_update = -scaled_lr * (contrib / max_contrib)
    log_update = np.clip(log_update, -3.0, 3.0)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

        t_nest += 1.0

        conv = fast_convolve(y)
        peaks = np.argpartition(conv, -max_k)[-max_k:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 1.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * y[k - idx_arr]

        v = beta * v - lr * sg
        a_new = project_simplex(a + v)
        obj = fast_convolve(a_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a

        a = a_new
        lr *= (1.0 - 0.0003)

        if no_improve > 150:
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(best_a + noise)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0

    return best_a, best_obj


def micro_polish(a0, time_limit, seed=42):
    """Very fine perturbations for the last bit of improvement."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Try multiple small perturbations
        for scale in [0.0002, 0.0005, 0.001, 0.003]:
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            noise = rng.standard_normal(n) * scale
            a_try = project_simplex(a + noise)
            obj_try = fast_convolve(a_try).max()
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try

    return best_a, best_obj


def generate_structured_init(n, seed=42):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    candidates = []

    # Uniform
    candidates.append(np.ones(n) / n)

    # Cosine-modulated uniform
    for alpha in [0.1, 0.2, 0.3, 0.5]:
        j = np.arange(n)
        a = 1.0 + alpha * np.cos(2.0 * np.pi * j / n)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
        candidates.append(a.copy())

    # Random with smoothing
    for _ in range(3):
        a = rng.random(n)
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        a = a / a.sum()
        candidates.append(a.copy())

    # Gaussian-like
    for center_frac in [0.3, 0.5, 0.7]:
        j = np.arange(n)
        center = center_frac * (n - 1)
        a = np.exp(-0.5 * ((j - center) / (n * 0.15)) ** 2)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
        candidates.append(a.copy())

    # Two-peaked
    j = np.arange(n)
    a = (np.exp(-0.5 * ((j - n * 0.2) / (n * 0.1)) ** 2) +
         np.exp(-0.5 * ((j - n * 0.8) / (n * 0.1)) ** 2))
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    candidates.append(a.copy())

    return candidates


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: multi-phase hybrid search optimizing
    2*n*max(conv(a,a))/(sum(a))^2 using multiplicative updates,
    peak equalization, local pairwise moves, and Nesterov acceleration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={time.monotonic()-t0_total:.0f}s)")
        return score

    t0_total = time.monotonic()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_default")

    n_parent = len(parent)

    # === Phase 1: Multi-n, multi-init exploration (~200s) ===
    print("\n=== Phase 1: Multi-n/Multi-init Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([
        n_parent, n_parent - 32, n_parent + 32,
        640, 704, 768, 832, 896, 960, 1024
    ]))
    candidate_ns = [n for n in candidate_ns if 384 <= n <= 1200]

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1400:
            break

        # Generate diverse inits for this n
        inits = generate_structured_init(n_try, seed=seed + n_try)
        if n_try == n_parent:
            inits.insert(0, parent.reshape(-1).copy() if len(parent) == n_try else
                         np.interp(np.linspace(0, 1, n_try),
                                   np.linspace(0, 1, n_parent), parent))

        # Quick peak equalization on each init
        best_for_n = None
        best_obj_n = float('inf')

        for ii, init in enumerate(inits[:5]):  # Limit to 5 inits per n
            if time.monotonic() > deadline - 1400:
                break
            t_alloc = min(20, deadline - time.monotonic() - 1400)
            if t_alloc < 3:
                break
            a_opt, obj = peak_equalize(init, time_limit=t_alloc, seed=seed + n_try * 100 + ii)
            if obj < best_obj_n:
                best_obj_n = obj
                best_for_n = a_opt

        if best_for_n is not None:
            try_seq(best_for_n, f"explore_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep peak equalization (~500s) ===
    print("\n=== Phase 2: Deep Peak Equalization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Main deep run
        t_alloc = min(300, deadline - time.monotonic() - 800)
        if t_alloc > 30:
            print(f"  Deep equalize n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = peak_equalize(best_seq, time_limit=t_alloc, seed=seed + 1000)
            try_seq(a_opt, "deep_eq")

        # Multiple restarts with different seeds and perturbations
        for restart_i, scale in enumerate([0.003, 0.008, 0.015, 0.03]):
            t_alloc = min(80, deadline - time.monotonic() - 600)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = peak_equalize(f_pert, time_limit=t_alloc, seed=seed + 2000 + restart_i)
            try_seq(a_opt, f"eq_restart{restart_i}_s{scale}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Nesterov + local pairwise (~400s) ===
    print("\n=== Phase 3: Nesterov + Local Pairwise ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Nesterov with different configs
        for nr, (lr_k, mk) in enumerate([(0.003, 30), (0.005, 50), (0.002, 20)]):
            t_alloc = min(80, deadline - time.monotonic() - 350)
            if t_alloc < 10:
                break
            print(f"  Nesterov config {nr}, t={t_alloc:.0f}s")

            def nesterov_with_params(a0, tl, s, lr0, mkv):
                return nesterov_optimize(a0, tl, seed=s)

            a_opt, _ = nesterov_optimize(best_seq, time_limit=t_alloc, seed=seed + 3000 + nr)
            try_seq(a_opt, f"nest_{nr}")

        # Local pairwise optimization
        t_alloc = min(100, deadline - time.monotonic() - 250)
        if t_alloc > 10:
            print(f"  Local pairwise, t={t_alloc:.0f}s")
            a_opt, _ = local_pairwise_optimize(best_seq, time_limit=t_alloc, seed=seed + 4000)
            try_seq(a_opt, "pairwise")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Combined refinement (~250s) ===
    print("\n=== Phase 4: Combined Refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Alternate between peak equalization and pairwise
        a_curr = best_seq.copy()
        for round_i in range(6):
            t_alloc = min(40, deadline - time.monotonic() - 100)
            if t_alloc < 5:
                break

            a_opt, _ = peak_equalize(a_curr, time_limit=t_alloc // 2, seed=seed + 5000 + round_i)
            try_seq(a_opt, f"refine_pe{i}_a")

            a_opt, _ = local_pairwise_optimize(a_opt, time_limit=t_alloc // 2, seed=seed + 6000 + round_i)
            try_seq(a_opt, f"refine_pe{i}_b")

            a_curr = best_seq.copy()

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Micro-polish (~150s) ===
    print("\n=== Phase 5: Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(70, deadline - time.monotonic() - 50)
        if t_alloc > 5:
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc, seed=seed + 7000)
            try_seq(a_opt, "micro1")

        # One more peak equalization pass
        t_alloc = min(40, deadline - time.monotonic() - 30)
        if t_alloc > 5:
            a_opt, _ = peak_equalize(best_seq, time_limit=t_alloc, seed=seed + 8000)
            try_seq(a_opt, "final_eq")

        # Final micro polish
        t_alloc = min(20, deadline - time.monotonic() - 10)
        if t_alloc > 3:
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc, seed=seed + 9000)
            try_seq(a_opt, "micro_final")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
