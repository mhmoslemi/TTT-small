def softmax_gradient_descent(p0, time_limit, beta_start=50, beta_end=1000, 
                              lr=0.3, momentum=0.9):
    """Smoothed gradient descent with soft-max approximation."""
    n = len(p0)
    p = p0.copy()
    p = np.maximum(p, 0)
    p = p / p.sum()
    
    best_p = p.copy()
    best_val = fast_convolve(p).max()
    
    v = np.zeros(n)  # momentum
    beta = beta_start
    
    t0 = time.mononotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute convolution
        c = fast_convolve(p)
        cur_max = c.max()
        
        if cur_max < best_val:
            best_val = cur_max
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Restart
            p = best_p.copy()
            v = np.zeros(n)
            no_improve = 0
            beta = min(beta * 2, 5000)
            continue
        
        # Compute soft-max weights
        beta_c = beta * c
        beta_max = beta_c.max()
        exp_bc = np.exp(beta_c - beta_max)
        w = exp_bc / exp_bc.sum()
        
        # Compute gradient: grad[j] = 2 * Σ_k w[k] * p[k-j] (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            # k ranges from j to min(j+n-1, 2n-2)
            k_start = j
            k_end = min(j + n, 2*n - 1)
            # p[k-j] for k in [k_start, k_end)
            idx = np.arange(k_start, k_end) - j  # = 0, 1, ..., k_end-1-j
            grad[j] = 2.0 * np.dot(w[k_start:k_end], p[idx])
        
        # Update with momentum
        v = momentum * v + grad
        p_new = p - lr * v
        
        # Project onto simplex
        p_new = project_simplex(p_new)
        
        # Check if step was good
        c_new = fast_convolve(p_new)
        if c_new.max() < cur_max + 0.01 / beta:
            p = p_new
        else:
            # Halve step size
            lr = max(lr * 0.5, 0.001)
        
        # Increase beta
        beta = min(beta * 1.01, beta_end)
    
    return best_p, best_val

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(p, w):
    """Compute gradient of sum_k w[k] * (p*p)[k] with respect to p.
    
    grad[j] = 2 * sum_k w[k] * p[k-j] for valid k-j
    This equals 2 * np.correlate(w, p, 'full')[:len(p)]
    """
    n = len(p)
    cross = np.correlate(w, p, mode='full')
    return 2.0 * cross[:n]

def freq_perturb(p, rng, strength=0.03):
    """Perturb p in the frequency domain."""
    n = len(p)
    F = np.fft.rfft(p)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    p_new = np.fft.irfft(F_new, n)
    p_new = np.maximum(p_new, 0)
    s = p_new.sum()
    if s > 1e-15:
        p_new /= s
    else:
        p_new = p.copy()
    return p_new


def interpolate_to_length(p, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(p))
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    p_new = np.maximum(p_new, 0)
    s = p_new.sum()
    if s > 1e-15:
        p_new /= s
    else:
        p_new = np.ones(new_n) / new_n
    return p_new

        k_star = int(np.argmax(c))

            if i == j or i > j:

            for _ in range(3):

                if dest == src or dest == (k_star - src):

                p_try[src] -= delta
                p_try[dest] += delta
                p_try = np.maximum(p_try, 0)
                s = p_try.sum()
                if s < 1e-15:
                    continue
                p_try /= s
                c_try = fast_convolve(p_try)
                if float(c_try.max()) < best_val - 1e-16:

                    improved = True
                    break

                best_p = p.copy()
    
    return best_p, best_val

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max smoothed gradient descent with backtracking line search
2. Multi-length exploration (interpolation from parent)
3. Adaptive beta schedule for progressive peak focusing
4. Frequency-domain perturbations for diversification
5. Targeted peak-reduction coordinate adjustments for final polish
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(p, w):
    """Gradient of sum_k w[k] * (p*p)[k] w.r.t. p.
    grad[j] = 2 * sum_k w[k] * p[k-j] (for valid k-j in [0,n-1])
    """
    n = len(p)
    cross = np.correlate(w, p, mode='full')
    return 2.0 * cross[:n]


def softmax_gradient_descent(p0, time_limit, beta_start=10.0, beta_end=5000.0,
                             lr_init=0.03, momentum=0.85, max_beta_rate=1.004,
                             verbose=False, label=""):
    """Smoothed gradient descent using soft-max approximation to max."""
    n = len(p0)
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    c = fast_convolve(p)
    best_p = p.copy()
    best_val = float(c.max())

    v = np.zeros(n)
    beta = float(beta_start)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.05:
            break

        c = fast_convolve(p)
        cur_max = float(c.max())

        if cur_max < best_val - 1e-17:
            best_val = cur_max
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            p = best_p.copy()
            v *= 0.05
            no_improve = 0
            beta = min(beta * 2.5, 20000.0)
            lr = max(lr_init * 0.4, 0.003)
            continue

        # Soft-max weights
        bc = beta * c
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.ones(len(c)) / len(c)
        else:
            w = exp_bc / total

        # Gradient
        grad = compute_gradient(p, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Momentum update
        v = momentum * v + grad

        # Backtracking line search
        alpha = 1.0
        accepted = False
        for _ in range(25):
            if alpha < 1e-10:
                break
            p_try = p - lr * alpha * v
            p_try = project_simplex(p_try)
            c_try = fast_convolve(p_try)
            new_max = float(c_try.max())
            if new_max < cur_max + 1e-15:
                p = p_try
                accepted = True
                break
            alpha *= 0.5

        if accepted:
            lr = min(lr * 1.003, 0.5)
        else:
            lr = max(lr * 0.4, 0.002)

        # Increase beta
        beta = min(beta * max_beta_rate, beta_end)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      {label} iter={it}, beta={beta:.0f}, lr={lr:.5f}, "
                  f"best={best_val:.12f}, elapsed={elapsed:.1f}s")

    return best_p, best_val


def freq_perturb(p, rng, strength=0.02):
    """Perturb p in the frequency domain for diversification."""
    n = len(p)
    F = np.fft.rfft(p)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    p_new = np.fft.irfft(F_new, n)
    p_new = np.maximum(p_new, 0)
    s = p_new.sum()
    if s > 1e-15:
        p_new /= s
    else:
        p_new = p.copy()
    return p_new


def interpolate_to_length(p, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(p))
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    p_new = np.maximum(p_new, 0)
    s = p_new.sum()
    if s > 1e-15:
        p_new /= s
    else:
        p_new = np.ones(new_n) / new_n
    return p_new


def peak_reduction_polish(p, time_limit, n_trials=200):
    """Targeted peak reduction by redistributing mass from peak contributors."""
    n = len(p)
    p = np.maximum(p.astype(np.float64), 0.0)
    p = p / p.sum()
    c = fast_convolve(p)
    best_p = p.copy()
    best_val = float(c.max())
    t0 = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(p)
        cur_max = float(c.max())
        if cur_max < best_val:
            best_val = cur_max
            best_p = p.copy()

        # Find the current peak
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi - lo < 1:
            continue

        # Get the contributing terms
        indices = np.arange(lo, hi + 1)
        partners = k_star - indices
        products = p[indices] * p[partners]
        total_contrib = products.sum()

        if total_contrib < 1e-15:
            continue

        # Find the index with highest individual product
        i_max = indices[np.argmax(products)]
        partner = k_star - i_max

        # Try to reduce by moving mass from i_max to a position
        # that minimally increases other peaks
        # Strategy: move to a position that is "far" from k_star
        delta = p[i_max] * 0.08
        if delta < 1e-15:
            continue

        # Try several candidate destinations
        # Prefer positions near the edges (1 and 2 contribute to few peaks)
        candidates = []
        candidates.append(0)
        candidates.append(n - 1)
        candidates.append(1)
        candidates.append(n - 2)
        # Also try random positions
        for _ in range(5):
            candidates.append(np.random.randint(n))

        for dest in candidates:
            if dest == i_max or dest == partner:
                continue
            p_try = p.copy()
            p_try[i_max] -= delta
            p_try[dest] += delta
            p_try = np.maximum(p_try, 0)
            s = p_try.sum()
            if s < 1e-15:
                continue
            p_try /= s
            c_try = fast_convolve(p_try)
            if float(c_try.max()) < best_val - 1e-16:
                p = p_try
                best_val = float(c_try.max())
                best_p = p.copy()
                break

    return best_p, best_val
