def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with temperature annealing
on the log-sum-exp smoothed objective, combined with entropic mirror descent
and peak equalization post-processing.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_smoothed(a, T, top_k=100):
    """
    Compute gradient of f_T(a) = T * log(sum_k exp(conv_k/T)) with respect to a.
    Uses only the top_k most significant lags for efficiency.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Compute softmax weights (for the log-sum-exp gradient)
    conv_shifted = conv / T - (conv / T).max()
    p = np.exp(conv_shifted)
    p = p / p.sum()
    
    # Only use top_k lags for the gradient
    if top_k < len(p):
        active = np.argpartition(p, -top_k)[-top_k:]
        p_masked = np.zeros_like(p)
        p_masked[active] = p[active]
        p = p_masked / p_masked.sum()  # Renormalize
    
    # Compute grad_a[m] = sum_k p[k] * (a[k-m] + a[m-k])
    n_active = np.sum(p > 1e-12)
    if n_active == 0:
        return np.zeros(n), conv
    
    grad_a = np.zeros(n)
    active_k = np.where(p > 1e-12)[0]
    
    for k in active_k:
        k = int(k)
        pk = p[k]
        
        # First term: a[k-m] for valid m
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            m_vals = np.arange(lo1, hi1 + 1)
            idx = k - m_vals
            grad_a[m_vals] += pk * a[idx]
        
        # Second term: a[m-k] for valid m
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            m_vals = np.arange(lo2, hi2 + 1)
            idx = m_vals - k
            grad_a[m_vals] += pk * a[idx]
    
    return grad_a, conv

def lbfgs_smoothing(a0, T_schedule, time_limit, maxiter_per_T=100):
    """
    Run L-BFGS with geometric temperature cooling.
    T_schedule: list of temperatures to try (in order).
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        x = np.log(a)  # log-space parameterization
        
        # Define objective and gradient for scipy
        def obj_fn(x_inner):
            a_inner = softmax(x_inner)
            grad_a, conv = compute_gradient_smoothed(a_inner, T)
            # Transform gradient to x-space
            grad_x = a_inner * grad_a - a_inner * np.sum(a_inner * grad_a)
            # Objective value
            conv_shifted = conv / T - (conv / T).max()
            val = T * (conv.max() / T + np.log(np.exp(conv_shifted).sum()))
            # Actually: f_T = T * log(sum exp(conv/T))
            # = T * (max(conv)/T + log(sum exp((conv - max(conv))/T)))
            # = max(conv) + T * log(sum exp((conv - max(conv))/T))
            val = conv.max() + T * np.log(np.exp(conv_shifted).sum())
            return val, grad_x
        
        # Run L-BFGS
        try:
            result = minimize(
                fun=lambda x_inner: obj_fn(x_inner)[0],
                jac=lambda x_inner: obj_fn(x_inner)[1],
                args=(),
                x0=x,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_T, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxfun': maxiter_per_T * 10}
            )
            a_new = softmax(result.x)
        except Exception as e:
            print(f"  L-BFGS error at T={T}: {e}")
            break
        
        a = np.maximum(a_new, 1e-15)
        a = a / a.sum()
        
        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if time.monotonic() - t0 > time_limit - 10:
            break
    
    return best_a, best_obj

def entropic_mirror_descent(a0, time_limit, lr_init=0.01, decay=0.9999, top_k=50):
    """
    Entropic mirror descent (multiplicative weight updates) in log-space.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        grad_a, conv = compute_gradient_smoothed(a, T=0.001, top_k=top_k)
        
        # Multiplicative update: a_new = a * exp(-lr * grad_a)
        # Normalize
        log_a_new = np.log(a) - lr * grad_a
        a_new = softmax(log_a_new)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Backtrack: halve learning rate
            if no_improve % 10 == 0:
                lr *= 0.9
        
        lr *= decay
        
        if no_improve > 200:
            break
    
    return best_a, best_obj

def peak_equalization(a0, time_limit, n_peaks=5, n_iter_max=5000):

    Try to equalize the top n_peaks autocorrelation values by solving
    a sequence of small QP-like subproblems.

    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

            break
        
        conv = fast_convolve(a)
        
        # Get top peaks

        peak_vals = conv[top_lags]

        if peak_vals[-1] > peak_vals[0] * 0.99:
            # Well equalized, try to reduce them all
            target = peak_vals.mean()
        else:
            target = peak_vals[0]
        
        # For each top peak, try to reduce it by moving mass
        improved = False
        for k_peak in top_lags:
            k_peak = int(k_peak)
            if conv[k_peak] <= target * 1.001:
                continue
            
            # Find the pair (j, partner) that contributes most to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            
            contributions = np.array([a[j] * a[k_peak - j] for j in range(js, je + 1)])
            
            if contributions.max() < 1e-15:
                continue
            
            # Find where to move the mass: to a position that minimally increases other peaks
            # Try moving from the top contributor to several candidate positions
            j_top = js + int(np.argmax(contributions))
            partner = k_peak - j_top
            
            # Candidate destinations: positions that are far from active peaks
            # Simple heuristic: try a few random positions
            rng = np.random.default_rng(it * 31 + k_peak + 7)
            
            best_dest = -1
            best_gain = 0
            best_delta = 0
            
            # Try small perturbations
            for scale in [0.001, 0.005, 0.01, 0.02]:
                delta = a[j_top] * scale
                if delta < 1e-12:
                    continue
                
                # Try a few destinations
                for _ in range(20):
                    dest = int(rng.randint(0, n))
                    if dest == j_top or dest == partner:
                        continue
                    
                    a_test = a.copy()
                    a_test[j_top] -= delta
                    a_test[dest] += delta
                    a_test = np.maximum(a_test, 0)
                    s = a_test.sum()
                    if s < 1e-15:
                        continue
                    a_test = a_test / s
                    c_new = fast_convolve(a_test)
                    new_max = c_new.max()
                    
                    gain = best_obj - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_dest = dest
                        best_delta = delta
            
            if best_dest >= 0 and best_gain > 1e-12:
                a[j_top] -= best_delta
                a[best_dest] += best_delta
                a = np.maximum(a, 0)
                s = a.sum()
                a = a / s
                obj = fast_convolve(a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                improved = True

                if c.max() < best_obj:
                    a = a_pert
                    best_obj = c.max()
                    best_a = a.copy()
                else:
                    if it > 500:
                        break
    
    return best_a, best_obj

    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_a(a, T):
    """
    Compute gradient and value of f_T(a) = T*log(sum_k exp(conv_k/T)).
    Key identity: d(conv_k)/da[m] = 2*a[k-m] if 0<=k-m<n, else 0.
    Therefore grad_a = 2 * cross-correlation(p, a) where p = softmax(conv/T).
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p = p / p.sum()
    # grad_a[m] = 2 * sum_j p[j+m] * a[j]
    c = np.correlate(p, a, mode='full')
    grad_a = 2.0 * c[n - 1: 2 * n - 1]
    obj = conv.max() + T * np.log(p.sum())
    return grad_a, obj, conv


def lbfgs_step(a, T, maxiter=200):
    """Single L-BFGS optimization at temperature T using softmax parameterization."""
    n = len(a)
    x0 = np.log(np.maximum(a, 1e-30))

    def fun(x):
        a_x = softmax_fn(x)
        _, obj, _ = smooth_grad_a(a_x, T)
        return obj

    def jac(x):
        a_x = softmax_fn(x)
        grad_a, _, _ = smooth_grad_a(a_x, T)
        return a_x * grad_a - a_x * np.dot(a_x, grad_a)

    result = minimize(fun, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                               'maxfun': maxiter * 10, 'iprint': -1})
    return softmax_fn(result.x)


def lbfgs_anneal(a0, T_list, time_limit, maxiter_per_T=200):
    """L-BFGS with geometric temperature cooling."""
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    n = len(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for Ti in T_list:
        if time.monotonic() - t0 > time_limit - 5:
            break
        try:
            a_new = lbfgs_step(a, Ti, maxiter=maxiter_per_T)
        except Exception:
            break
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            print(f"    T={Ti:.6f}: max_conv={obj:.10f} (improved)")
        else:
            a = best_a.copy()
            print(f"    T={Ti:.6f}: max_conv={obj:.10f} (no improvement)")
    return best_a, best_obj


def spsa_optimize(a0, time_limit, c0=0.01, alpha=0.602, gamma=0.8,
                  A0=0.1, n_restarts=3, perturb_scale=0.005):
    """
    SPSA (Simultaneous Perturbation Stochastic Approximation) optimizer.
    Works directly on the simplex without gradient computation.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 2:
            break
        if restart > 0:
            # Perturb from best
            rng_r = np.random.default_rng(restart * 1000 + 7)
            scale = perturb_scale * (1 + restart * 0.5)
            a = project_simplex(best_a + rng_r.standard_normal(n) * scale)

        rng = np.random.default_rng(42 + restart * 7919)
        A = A0
        c = c0
        a_cur = a.copy()

        for k in range(50000):
            if time.monotonic() - t0 > time_limit - 1:
                break
            # Generate Rademacher perturbation
            d = rng.integers(0, 2, size=n) * 2 - 1  # ±1

            # Evaluate at a ± c*d (projected to simplex)
            a_plus = project_simplex(a_cur + c * d)
            a_minus = project_simplex(a_cur - c * d)

            f_plus = fast_convolve(a_plus).max()
            f_minus = fast_convolve(a_minus).max()

            # SPSA update
            g = (f_plus - f_minus) / (2.0 * c) * d
            a_new = project_simplex(a_cur - A * g)

            obj = fast_convolve(a_new).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
            a_cur = a_new

            # Update gains
            k1 = k + 1
            c = c0 / (k1 + 10) ** alpha
            A = A0 / (k1 + 10) ** gamma

            # Early exit if converged
            if k > 1000 and c < 1e-7:
                break

    return best_a, best_obj


def peak_equalize(a0, time_limit, n_top=5, n_inner_iters=200):
    """
    Iteratively equalize the top autocorrelation peaks by moving mass
    from high-contributing positions to low-impact positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        conv = fast_convolve(a)
        top_lags = np.argpartition(conv, -n_top)[-n_top:]
        top_lags = top_lags[np.argsort(conv[top_lags])[::-1]]
        peak_vals = conv[top_lags]

        improved = False
        for k_peak in top_lags:
            if time.monotonic() - t0 > time_limit - 3:
                break
            k_peak = int(k_peak)
            if conv[k_peak] < peak_vals[0] * 1.005:
                continue

            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            if je < js:
                continue

            # Find top contributor to this peak
            contribs = np.array([a[j] * a[k_peak - j] for j in range(js, je + 1)])
            if contribs.max() < 1e-15:
                continue
            j_best = js + int(np.argmax(contribs))

            # Try to move mass from j_best to various destinations
            rng = np.random.default_rng(it * 31 + k_peak)
            best_dest = -1
            best_gain = 0
            best_delta = 0

            for scale in [0.002, 0.005, 0.01, 0.02]:
                delta = a[j_best] * scale
                if delta < 1e-12:
                    continue
                for _ in range(30):
                    dest = int(rng.integers(0, n))
                    if dest == j_best or dest == k_peak - j_best:
                        continue
                    a_t = a.copy()
                    a_t[j_best] -= delta
                    a_t[dest] += delta
                    a_t = np.maximum(a_t, 0)
                    s = a_t.sum()
                    if s < 1e-15:
                        continue
                    a_t = a_t / s
                    new_max = fast_convolve(a_t).max()
                    gain = best_obj - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_dest = dest
                        best_delta = delta

            if best_dest >= 0 and best_gain > 1e-12:
                a[j_best] -= best_delta
                a[best_dest] += best_delta
                a = np.maximum(a, 0)
                a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                improved = True

        if not improved:
            # Random perturbation to escape
            if it % 50 == 0:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.001 * (1 + min(it / 200.0, 3.0))
                a_pert = project_simplex(a + rng.standard_normal(n) * scale)
                if fast_convolve(a_pert).max() < best_obj:
                    a = a_pert
                    best_obj = fast_convolve(a).max()
                    best_a = a.copy()
            if it > 500:
                break

    return best_a, best_obj


def entropic_mirroring(a0, time_limit, lr_init=0.0
05, decay=0.99995, T_smooth=0.001):
    """Entropic mirror descent (multiplicative updates in log-space)."""
    a = np.maximum(a0.astype(np.float64), 1e-30)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 1:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        grad_a, _, _ = smooth_grad_a(a, T_smooth)
        # Multiplicative update
        log_a_new = np.log(a) - lr * grad_a
        a_new = softmax_fn(log_a_new)
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        lr *= decay
        if no_improve > 300:
            break
    return best_a, best_obj


def structured_init(n, seed=0):
    """Generate structured initializations for diversity."""
    rng = np.random.default_rng(seed)
    inits = []
    # Raised cosine
    j = np.arange(n)
    x = np.sin(np.pi * (j + 0.5) / n) ** 2
    inits.append(x / x.sum())
    # Triangular
    x = 1.0 - np.abs(j - (n - 1) / 2.0) / ((n - 1) / 2.0 + 1e-10)
    inits.append(np.maximum(x, 0) / np.maximum(x, 0).sum())
    # Two Gaussian bumps
    x = np.exp(-((j - n * 0.25) ** 2) / (n * 0.05)) + np.exp(-((j - n * 0.75) ** 2) / (n * 0.05))
    inits.append(x / x.sum())
    # Exponential taper
    x = np.exp(-2.0 * j / n)
    inits.append(x / x.sum())
    # Random
    x = rng.dirichlet(np.ones(n))
    inits.append(x)
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid optimization:
    Phase 1: L-BFGS annealing from parent + structured inits (multi-n)
    Phase 2: SPSA diverse exploration
    Phase 3: Fine L-BFGS refinement
    Phase 4: Peak equalization
    Phase 5: Final polish (L-BFGS + entropic mirroring)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={time.monotonic()-t0_total:.0f}s)")
        return score

    t0_total = time.monotonic()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    # === Phase 1: L-BFGS annealing with multi-n exploration (~450s) ===
    print("\n=== Phase 1: L-BFGS Annealing (multi-n) ===")
    t_phase = time.monotonic()

    n_parent = len(parent)
    candidate_ns = sorted(set([n_parent, max(256, n_parent - 128),
                                min(1200, n_parent + 128), 512, 768, 1024]))
    candidate_ns = [n for n in candidate_ns if 256 <= n <= 1200]

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1100:
            break
        # Create starting point for this n
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            s = start.sum()
            start = start / s if s > 1e-10 else np.ones(n_try) / n_try

        # Also try a structured init for this n
        starts = [start]
        for si in range(2):
            inits = structured_init(n_try, seed=seed + si * 100 + n_try)
            starts.append(inits[si % len(inits)])

        t_alloc = min(80, deadline - time.monotonic() - 1100)
        if t_alloc < 10:
            break

        T_schedule = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005]
        for si, start_a in enumerate(starts[:2]):  # Try 2 starts per n
            t_sub = t_alloc / 2
            print(f"  n={n_try}, start={si}, t={t_sub:.0f}s")
            a_opt, _ = lbfgs_anneal(start_a, T_schedule, time_limit=t_sub, maxiter_per_T=150)
            try_seq(a_opt, f"lbfgs_n{n_try}_s{si}")
            t_sub = min(40, deadline - time.monotonic() - 1100)
            if t_sub < 5:
                break

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: SPSA diverse exploration (~400s) ===
    print("\n=== Phase 2: SPSA Exploration ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        # SPSA with different settings
        spsa_configs = [
            {'c0': 0.005, 'alpha': 0.602, 'gamma': 0.8, 'A0': 0.05,
             'n_restarts': 3, 'perturb_scale': 0.003},
            {'c0': 0.01, 'alpha': 0.5, 'gamma': 0.7, 'A0': 0.1,
             'n_restarts': 2, 'perturb_scale': 0.005},
        ]
        for ci, cfg in enumerate(spsa_configs):
            t_alloc = min(150, deadline - time.monotonic() - 600)
            if t_alloc < 20:
                break
            print(f"  SPSA config {ci}, t={t_alloc:.0f}s")
            a_opt, _ = spsa_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"spsa_{ci}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Fine L-BFGS refinement (~300s) ===
    print("\n=== Phase 3: Fine L-BFGS ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        T_fine = [0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        t_alloc = min(200, deadline - time.monotonic() - 400)
        if t_alloc > 20:
            print(f"  Fine L-BFGS n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_anneal(best_seq, T_fine, time_limit=t_alloc, maxiter_per_T=200)
            try_seq(a_opt, "fine_lbfgs")

        # Entropic mirroring
        t_alloc = min(80, deadline - time.monotonic() - 350)
        if t_alloc > 10:
            print(f"  Entropic mirroring, t={t_alloc:.0f}s")
            a_opt, _ = entropic_mirroring(best_seq, time_limit=t_alloc,
                                           lr_init=0.003, T_smooth=0.0005)
            try_seq(a_opt, "entropic")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak equalization (~250s) ===
    print("\n=== Phase 4: Peak Equalization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(180, deadline - time.monotonic() - 200)
        if t_alloc > 15:
            print(f"  Peak equalization, t={t_alloc:.0f}s")
            a_opt, _ = peak_equalize(best_seq, time_limit=t_alloc, n_top=5)
            try_seq(a_opt, "peak_eq")

        # L-BFGS after peak equalization
        t_alloc = min(60, deadline - time.monotonic() - 180)
        if t_alloc > 10 and best_seq is not None:
            a_opt, _ = lbfgs_anneal(best_seq, [0.0001, 0.00005],
                                     time_limit=t_alloc, maxiter_per_T=150)
            try_seq(a_opt, "lbfgs_after_peak")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final polish (~200s) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Very fine L-BFGS
        t_alloc = min(80, deadline - time.monotonic() - 80)
        if t_alloc > 10:
            T_very_fine = [0.00003, 0.00001, 0.000005]
            print(f"  Very fine L-BFGS, t={t_alloc:.0f}s")
            a_opt, _ = lbfgs_anneal(best_seq, T_very_fine, time_limit=t_alloc, maxiter_per_T=200)
            try_seq(a_opt, "vf_lbfgs")

        # Final peak equalization
        t_alloc = min(80, deadline - time.monotonic() - 30)
        if t_alloc > 10:
            print(f"  Final peak eq, t={t_alloc:.0f}s")
            a_opt, _ = peak_equalize(best_seq, time_limit=t_alloc, n_top=3)
            try_seq(a_opt, "final_peak")

    # Final verification
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
