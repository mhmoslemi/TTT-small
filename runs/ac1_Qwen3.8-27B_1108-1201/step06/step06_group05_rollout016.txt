def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)

                # The change in conv when a[j] changes by delta = -best_s:
                # d(conv)[k] = 2*delta*a[k-j] + delta^2*delta_{k,2j} (using OLD a for the linear term)
                # = -2*best_s*old_a[k-j] + best_s^2*delta_{k,2j}
                # where old_a[k-j] = a[k-j] if k-j != j, and old_a[j] = a[j] + best_s if k-j == j

                delta = -best_s
                # Linear part: -2*best_s * old_a[k-j]
                # For k in [j, j+n-1] intersect [0, 2n-2]:
                ks_valid = np.arange(max(0, j), min(2 * n - 1, j + n - 1) + 1)
                old_vals = np.zeros(len(ks_valid))
                for idx, k in enumerate(ks_valid):
                    if k - j == j:  # k = 2j
                        old_vals[idx] = a[j] + best_s  # old a[j]
                    else:
                        old_vals[idx] = a[k - j]
                conv[ks_valid] += 2.0 * delta * old_vals
                # Quadratic part at k = 2j
                if 2 * j < 2 * n - 1:
                    conv[2 * j] += delta * delta

                new_obj = conv.max()
                if new_obj < best_obj - 1e-16:
                    best_obj = new_obj
                    improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose and pass_count % 20 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}")

    return a, best_obj


def pgd_momentum(a0, time_limit, top_k=20, lr_init=0.02, mom_init=0.9):
    """Projected gradient descent with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        obj = conv.max()

        # Subgradient from top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            order = np.argsort(pv)[::-1]
            weights = np.zeros(len(peaks))
            for rank, idx in enumerate(order):
                weights[idx] = 1.0 / (rank + 1) ** 1.5
            ws = weights.sum()
            if ws > 0:
                weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * a[k - idx_arr]

        # Momentum step with backtracking
        v_new = mom * v - lr * sg
        best_candidate = a
        best_val = obj
        for trial_lr in [1.0, 0.5, 0.25, 0.125]:
            candidate = project_simplex(a + trial_lr * v_new)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                v = v_new * trial_lr
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.9

        if no_improve > 30:
            lr *= 0.95
            mom = min(0.97, mom + 0.001)
        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 17) % (2**31 - 1))
            noise = 0.002 + 0.003 * min(1.0, (no_improve - 150) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init
            mom = mom_init
            no_improve = 0

    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with 'min t' LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Solve LP: min t s.t. (f*g)[k] <= t, sum(g)=1, g>=0
        # Variables: g_0,...,g_{n-1}, t (total n+1 variables)
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize t

        # Constraints: (f*g)[k] - t <= 0 for k = 0,...,2n-2
        n_constraints = 2 * n - 1
        A_ub = np.zeros((n_constraints, n + 1))
        for k in range(n_constraints):
            A_ub[k, n] = -1.0  # -t
            # (f*g)[k] = sum_j f_j * g_{k-j} where valid
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A_ub[k, j] = f[k - j]
        b_ub = np.zeros(n_constraints)

        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'time_limit': max(1.0, remaining * 0.3), 'mip_rel_gap': 0.1})
            if not res.success:
                no_improve += 1
                if no_improve > 20:
                    break
                continue
            g = np.maximum(res.x[:n], 0.0)
            t_star = res.x[n]
        except Exception:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Refine with bisection
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 7919 + 13)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 80.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

        if verbose and it % 20 == 0:
            print(f"    FW-LP it={it}: obj={best_obj:.10f}, t*={t_star:.6f}, M={M:.6f}")

    return best_f, best_obj


def active_set_newton(a0, time_limit, max_iter=500):
    """Newton refinement using KKT conditions."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        M = conv.max()
        support = np.where(a > 1e-10)[0]
        if len(support) < 2:
            break

        # Find active peaks (within 0.5% of max)
        threshold = M * (1.0 - 0.005)
        active_peaks = np.where(conv >= threshold)[0]

        if len(active_peaks) <= 1:
            # Single peak: use gradient step
            break

        m = len(support)
        p = len(active_peaks)

        # Build the KKT system:
        # For j in support: sum_{k in P} mu_k * a[k-j] = nu
        # We solve for mu (weights on peaks) that minimizes the gradient norm

        # Gradient: grad_j = 2 * sum_{k in P} mu_k * a[k-j]
        # We want to find mu >= 0 that makes the gradient "balanced" on the support
        # i.e., grad_j should be constant (= nu) on the support

        # A_sys[support_i, peak_k] = a[active_peaks[k] - support[i]] (if valid)
        A_sys = np.zeros((m, p))
        for i, j in enumerate(support):
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = a[idx]

        # Find mu >= 0, sum(mu) = 1 such that A_sys @ mu ≈ nu * ones
        # This is a least squares problem with constraints
        # Minimize ||A_sys @ mu - ones(m)||^2 s.t. mu >= 0, sum(mu) = 1
        # Use a simple approach: solve unconstrained LS, then project

        # Unconstrained: mu = (A_sys^T A_sys + reg*I)^{-1} A_sys^T ones
        reg = 1e-10
        try:
            ATA = A_sys.T @ A_sys + reg * np.eye(p)
            AT1 = A_sys.T @ np.ones(m)
            mu_raw = np.linalg.solve(ATA, AT1)
            mu_raw = np.maximum(mu_raw, 0)
            mu_sum = mu_raw.sum()
            if mu_sum < 1e-15:
                mu_raw = np.ones(p) / p
            else:
                mu_raw /= mu_sum
        except np.linalg.LinAlgError:
            mu_raw = np.ones(p) / p

        # Compute gradient direction
        grad = A_sys.T @ mu_raw  # length m (for support)
        grad_full = np.zeros(n)
        grad_full[support] = -grad  # negative gradient (descent direction)

        # Line search
        best_candidate = a
        best_val = M
        for step in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            candidate = project_simplex(a + step * grad_full)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
        else:
            # Try smaller steps
            for step in [0.001, 0.002, 0.003]:
                candidate = project_simplex(a + step * grad_full)
                c = fast_convolve(candidate)
                val = c.max()
                if val < best_obj - 1e-15:
                    best_obj = val
                    best_a = candidate.copy()
                    a = candidate
                    break
            else:
                # No improvement, break
                if it > 50:
                    break

    return best_a, best_obj


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]. p has length 2n-1, a has length n."""
    return np.correlate(p, a, mode='valid')


def softmax_smoothed_grad(a0, time_limit, T_init=0.001, lr_init=0.01):
    """Smoothed gradient using softmax-weighted peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    T = T_init
    lr = lr_init

    v = np.zeros(n)
    mom = 0.92
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        grad = 2.0 * hankel_mv(p, a)

        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            candidate = project_simplex(a + scale * v)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            v *= 0.9

        if no_improve > 30:
            T *= 1.1
            lr *= 0.95
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            noise = 0.003 + 0.003 * min(1.0, (no_improve - 200) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def structured_init(n, kind='cosine'):
    """Generate structured initial sequences."""
    j = np.arange(n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'cosine':
        a = np.cos(np.pi * j / (2 * (n - 1))) ** 2
    elif kind == 'sine':
        a = np.sin(np.pi * (j + 1) / (n + 1)) ** 2
    elif kind == 'triangle':
        center = (n - 1) / 2.0
        a = 1.0 - np.abs(j - center) / center
        a = np.maximum(a, 0.01)
    elif kind == 'gaussian':
        center = (n - 1) / 2.0
        sigma = center / 2.0
        a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    elif kind == 'power_up':
        a = (j + 1) ** 0.5
    elif kind == 'power_down':
        a = (n - j) ** 0.5
    elif kind == 'bimodal':
        center1 = n * 0.25
        center2 = n * 0.75
        a = np.exp(-0.5 * ((j - center1) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((j - center2) / (n * 0.1)) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def try_and_update(a, best_state, label=""):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    # Clamp to valid range
    a_norm = np.clip(a_norm, 0.0, 1000.0)
    score = float(evaluate_sequence(a_norm.tolist()))
    best_seq, best_score, best_n = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        best_n = len(a_norm)
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score, best_n)


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s
