            # Use parent if target_n ==768 and available
            seq_start = parent_sequence_initial_normalized
        else:
            # Otherwise, start with uniform distribution (the most cautious initialization)
            seq_start = np.ones(target_n) / target_n
        
        # Put into simplex space for faster convergence
        seq_start_simplex = seq_start / np.sum(seq_start)

        # Timing of this round: should be at least 180 seconds to complete SPSA
        round_start_time = time.time()
        new_sequence = optimize_with_spsa(objective_with_history, seq_start_simplex, time_limit=800, n=target_n)
        
        # Calculate the actual score for this candidate to compare (not stored in history)
        candidate_score = evaluate_sequence(new_sequence.tolist())

        if candidate_score < best_score:
            best_score = candidate_score
            best_sequence = new_sequence
            no_improve_counter = 0
            print(f"New best sequence with score {best_score:.8f} at length n={target_n}")

            no_improve_counter += 1

        # Slower decay for longer sequences (to prevent observing_chlorophyll)
        time_decay = 0.98
        latest_update_time = time.time()
    
    # Final fallback if no improvement: return parent or a uniform sequence
    fallback_available = True
    if best_sequence is None:

            best_score = float('inf')
