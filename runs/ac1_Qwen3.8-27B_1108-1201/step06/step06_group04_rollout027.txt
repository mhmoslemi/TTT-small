"""
Primal-dual optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Primal-dual game iteration with Hankel solves, (2) smoothed gradient polish,
(3) active-set coordinate descent, (4) multi-n exploration.
Key insight: The problem is a minimax game where the dual (weight distribution over lags)
guides the primal (sequence) to flatten the auto-convolution.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(w, a):
    """Compute W @ a where W_{i,j} = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def build_hankel(w, n):
    """Build n x n Hankel matrix with A(i,j) = w[i+j]."""
    idx = np.add.outer(np.arange(n), np.arange(n))
    return w[idx]


def solve_hankel(A, b):
    """Solve A x = b with fallback for ill-conditioned systems."""
    try:
        x = np.linalg.solve(A, b)
        if np.all(np.isfinite(x)):
            return x
    except np.linalg.LinAlgError:
        pass
    try:
        x, residuals, rank, sv = np.linalg.lstsq(A, b, rcond=1e-12)
        return x
    except Exception:
        return np.ones(len(b)) / len(b)


def compute_lower_bound(w, n, max_bisect=60):
    """
    Compute dual lower bound: min mu such that Hankel(w) + mu * J >= 0.
    Uses bisection on mu.
    """
    H = build_hankel(w, n)
    J_11 = n  # 1^T J 1 = n * (1^T 1) ... actually J = 1*1^T, so 1^T J 1 = (1^T 1)^2 = n^2
    # J = ones(n,n), so for any x: x^T J x = (1^T x)^2
    
    # Bisection: find min mu such that H + mu * J is PSD
    lo = 0.0
    hi = 1.0
    # First, expand hi until H + hi*J is PSD
    for _ in range(20):
        M = H + hi * np.ones((n, n))
        evals = np.linalg.eigvalsh(M)
        if evals[0] >= -1e-12:
            break
        hi *= 2
    else:
        return hi  # Give up, return upper bound
    
    for _ in range(max_bisect):
        mid = 0.5 * (lo + hi)
        M = H + mid * np.ones((n, n))
        evals = np.linalg.eigvalsh(M)
        if evals[0] >= -1e-12:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-14:
            break
    return hi


def primal_step(w, n, v_init=None):
    """
    Primal step: minimize sum_k w_k * v^T H_k v subject to 1^T v = 1.
    Solution: v = A^{-1} 1 / (1^T A^{-1} 1) where A = sum_k w_k H_k.
    With projection to non-negative.
    """
    A = build_hankel(w, n)
    # Add small regularization for numerical stability
    A_reg = A + 1e-14 * np.eye(n)
    x = solve_hankel(A_reg, np.ones(n))
    
    # The unconstrained solution might have negative entries
    # Try the raw solution first
    s = x.sum()
    if abs(s) < 1e-15:
        v = np.ones(n) / n
    else:
        v = x / s
    
    # If v has significant negative entries, project
    if v.min() < -1e-10:
        v = project_simplex(v)
    else:
        v = np.maximum(v, 0)
        s = v.sum()
        if s > 1e-15:
            v = v / s
        else:
            v = np.ones(n) / n
    
    return v


def dual_step(v, eta, momentum=0.5):
    """
    Dual step: update w using multiplicative weights.
    w_k proportional to (conv(v,v)[k])^eta
    """
    q = fast_convolve(v)
    # Symmetrize: q[k] = q[2n-2-k] is already true for auto-convolution
    # Normalize
    q = np.maximum(q, 1e-30)
    w = q ** eta
    s = w.sum()
    if s < 1e-30:
        w = np.ones(len(q)) / len(q)
    else:
        w = w / s
    return w


def primal_dual_loop(v0, n, time_limit, eta_init=3.0, eta_final=0.5,

    Main primal-dual iteration loop.
    Returns best (v, objective, lower_bound).
    """
    v = np.maximum(v0.astype(np.float64), 0.0)
    s = v.sum()
    v = v / s
    n = len(v)
    
    # Initialize dual weights from current convolution
    q = fast_convolve(v)
    w = q ** eta_init
    w = w / w.sum()
    
    best_v = v.copy()
    best_obj = fast_convolve(v).max()
    best_lb = 0.0
    best_combined = best_obj - best_lb  # Gap
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Adaptive eta: start aggressive, decay to gentle
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        eta = eta_init * (eta_final / eta_init) ** progress
        
        # === Primal step ===
        v_new = primal_step(w, n, v)
        
        # Evaluate
        obj = fast_convolve(v_new).max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_v = v_new.copy()
            no_improve = 0

            no_improve += 1
        
        v = v_new
        
        # === Dual step (every few iterations to save time) ===
        if it % 2 == 0 or no_improve > 20:

            w = w / w.sum()

        if it % 50 == 0 and remaining > 5:
            lb = compute_lower_bound(w, n, max_bisect=40)
            if lb > best_lb:

            q = fast_convolve(v)

def smoothed_gradient_descent(v0, n, time_limit, T_init=0.001, lr_init=0.05,
                              verbose=False):
    """
    Smoothed gradient descent using log-sum-exp surrogate.
    Minimizes log(sum_k exp(conv_k / T)) as a smooth approximation of max(conv).
    """
    v = np.maximum(v0.astype(np.float64), 0.0)
    v = v / v.sum()
    n = len(v)
    
    T = T_init
    lr = lr_init
    best_v = v.copy()
    best_obj = fast_convolve(v).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(v)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        
        # Gradient of smooth surrogate: 2 * correlate(p, v)
        # More precisely: grad[i] = 2 * sum_k p[k] * v[k-i] (for valid indices)
        # This is 2 * (p * v) where * is the "cross-correlation" giving a length-n result
        grad = 2.0 * hankel_mv(p, v)
        
        # Gradient of constraint (sum v = 1): just the vector 1
        # Project gradient step onto simplex tangent space
        grad_tan = grad - np.mean(grad)  # Remove component along 1
        
        # Backtracking line search
        best_candidate = v
        best_val = conv.max()
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            trial_lr = lr * shrink
            candidate = project_simplex(v - trial_lr * grad_tan)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
                break
        
        v = best_candidate
        
        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_v = v.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        if no_improve > 20:
            T *= 1.1
            lr *= 0.95
        if no_improve > 60:
            T *= 0.8
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            v = project_simplex(best_v + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    SG it={it}, obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")
    
    return best_v, best_obj


def active_peak_descent(v0, n, time_limit, verbose=False):
    """
    Targeted coordinate descent: identify active peaks and reduce them
    by shifting mass from contributing positions.
    """

    n = len(v)

    it = 0
    stuck = 0

            break
        
        conv = fast_convolve(v)
        cmax = conv.max()
        
        # Find top active lags
        n_active = min(5, 2 * n - 1)
        active_lags = np.argsort(conv)[::-1][:n_active]
        
        improved = False
        for k in active_lags:
            if conv[k] < cmax * 0.95:
                break
            # Variables contributing to lag k
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contrib = np.arange(js, je + 1)
            non_contrib = np.setdiff1d(np.arange(n), contrib)
            
            if len(contrib) < 2 or len(non_contrib) == 0:
                continue
            
            # Find the pair with largest product
            # conv[k] = sum_{i in contrib} v[i] * v[k-i]
            pairs = [(i, k - i) for i in contrib if 0 <= k - i < n]
            if not pairs:
                continue
            
            # Sort by contribution
            contributions = [(v[i] * v[k - i], i, k - i) for i, j in pairs]
            contributions.sort(reverse=True)
            
            # Try to reduce the largest contributing pair
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                i_big, j_big = contributions[0][1], contributions[0][2]
                shift = v[i_big] * frac
                
                if shift < 1e-12:
                    break
                
                # Move mass from i_big to a non-contributing position
                # Choose the non-contrib position that minimizes the increase in other peaks
                nc_vals = v[non_contrib]
                if nc_vals.sum() < 1e-15:
                    target = non_contrib[0]
                else:
                    # Move to the position with smallest "influence"
                    target = non_contrib[np.argmin(nc_vals)]
                
                delta = np.zeros(n)
                delta[i_big] -= shift
                delta[target] += shift
                
                candidate = project_simplex(v + delta)
                c_new = fast_convolve(candidate).max()
                
                if c_new < best_obj - 1e-14:
                    v = candidate
                    best_obj = c_new
                    best_v = v.copy()
                    improved = True
                    break
        
        if not improved:
            stuck += 1
            if stuck > 50:
                # Try random perturbations
                rng = np.random.default_rng((it * 104729) % (2**31))
                for _ in range(5):
                    delta = rng.standard_normal(n) * 0.001
                    candidate = project_simplex(v + delta)
                    c_new = fast_convolve(candidate).max()
                    if c_new < best_obj - 1e-14:
                        v = candidate
                        best_obj = c_new
                        best_v = v.copy()
                        stuck = 0
                        improved = True
                        break
            if stuck > 200:
                break
        else:

def nesterov_subgrad(v0, n, time_limit, lr_init=0.005, verbose=False):
    """Nesterov-accelerated subgradient method."""
    v = np.maximum(v0.astype(np.float64), 0.0)
    v = v / v.sum()
    n = len(v)
    
    velocity = np.zeros(n)
    lr = lr_init
    momentum = 0.9
    y = v.copy()
    
    best_v = v.copy()
    best_obj = fast_convolve(v).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Nesterov lookahead
        y = v + momentum * velocity
        
        # Subgradient: direction of the max peak
        conv_y = fast_convolve(y)
        k_peak = int(np.argmax(conv_y))
        
        # Gradient of conv_y[k_peak] w.r.t. v
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * y[k_peak - idx]
        
        # Update
        y_projected = project_simplex(y - lr * grad)
        velocity = y_projected - y
        v = y + velocity
        v = project_simplex(v)
        
        obj = fast_convolve(v).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_v = v.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9998
        momentum = min(0.99, momentum + 0.00003)
        
        if no_improve > 80:
            lr *= 0.7
            momentum *= 0.9
        if no_improve > 250:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            v = project_simplex(best_v + rng.standard_normal(n) * 0.002)
            velocity = np.zeros(n)
            lr = lr_init * 0.5
            momentum = 0.9
            no_improve = 0
    
    return best_v, best_obj

            w = np.ones(2 * n - 1)
            w[0] = w[-1] = 10.0
            w = w / w.sum()
            eta_init = 1.5
        else:
            # Inverse-peak dual (emphasize flat regions)
            w = 1.0 / (conv + 1e-10)
            w = w / w.sum()
            eta_init = 2.5
        
        t0 = time.monotonic()
        while time.monotonic() - t0 < t_alloc:
            # One iteration of primal-dual
            v = primal_step(w, n, v)
            obj = fast_convolve(v).max()
            
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_v = v.copy()
            
            # Dual update
            q = fast_convolve(v)
            w_new = q ** eta_init
            w_new = w_new / w_new.sum()
            w = 0.6 * w + 0.4 * w_new
            w = w / w.sum()
            
            # Slowly decay eta
            eta_init *= 0.999
        
        if verbose:
            print(f"    PD start {s}: obj={best_obj:.10f}")
    
    return best_v, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point: hybrid search for the minimum auto-convolution peak sequence.
    Uses primal-dual game iteration, smoothed gradient, active peak descent,
    and Nesterov subgradient in a multi-phase, multi-n strategy.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")
    
    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, best={best_score:.10f}")
    
    # === Phase 1: Multi-n exploration with primal-dual ===
    print("\n=== Phase 1: Multi-n Primal-Dual Exploration ===")
    t1 = time.monotonic()
    
    candidate_ns = sorted(set([
        n_parent,
        max(128, int(n_parent * 0.75)),
        n_parent + 128,
        n_parent + 256,
        max(256, n_parent * 2 if n_parent <= 1024 else n_parent),
        512, 1024, 1536, 2048,
    ]))
    # Filter out very large n
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 4096]
    candidate_ns = sorted(set(candidate_ns))
    
    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 800
        if remaining < 30:
            break
        
        # Interpolate parent to new length
        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                             np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()
        
        t_alloc = min(remaining, 60)
        print(f"  PD n={n_try}, t={t_alloc:.0f}s")
        v_opt, obj, lb = primal_dual_loop(start, n_try, time_limit=t_alloc)
        try_seq(v_opt, f"pd_n{n_try}")
        
        # Quick smoothed gradient polish
        remaining = deadline - time.monotonic() - 750
        if remaining > 10:
            v_opt2, _ = smoothed_gradient_descent(v_opt, n_try, time_limit=min(remaining, 20))
            try_seq(v_opt2, f"sg_n{n_try}")
    
    print(f"  Phase 1 done. Best: {best_score:.10f} (t={time.monotonic()-t1:.0f}s)")
    
    # === Phase 2: Deep optimization on best n ===
    print("\n=== Phase 2: Deep Optimization ===")
    t2 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # 2a: Multi-start primal-dual
        remaining = deadline - time.monotonic() - 400
        if remaining > 30:
            t_alloc = min(remaining, int(budget_s * 0.25))
            print(f"  Multi-start PD n={n_best}, t={t_alloc:.0f}s")
            v_opt, _ = multi_start_primal_dual(best_seq, n_best, time_limit=t_alloc)
            try_seq(v_opt, "pd_multi")
        
        # 2b: Smoothed gradient with different temperatures
        for T_try in [0.0005, 0.001, 0.003]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  Smoothed grad T={T_try}, t={t_alloc:.0f}s")
            v_opt, _ = smoothed_gradient_descent(best_seq, n_best, time_limit=t_alloc, T_init=T_try)
            try_seq(v_opt, f"sg_T{T_try}")
        
        # 2c: Active peak descent
        remaining = deadline - time.monotonic() - 200
        if remaining > 15:
            t_alloc = min(remaining, 80)
            print(f"  Active peak descent, t={t_alloc:.0f}s")
            v_opt, _ = active_peak_descent(best_seq, n_best, time_limit=t_alloc)
            try_seq(v_opt, "peak_desc")
    
    print(f"  Phase 2 done. Best: {best_score:.10f} (t={time.monotonic()-t2:.0f}s)")
    
    # === Phase 3: Nesterov + fine-tuning ===
    print("\n=== Phase 3: Nesterov & Fine-tuning ===")
    t3 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Nesterov with different learning rates
        for lr_try in [0.003, 0.008, 0.001, 0.015]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 10:
                break
            t_alloc = min(remaining, 60)
            print(f"  Nesterov lr={lr_try}, t={t_alloc:.0f}s")
            v_opt, _ = nesterov_subgrad(best_seq, n_best, time_limit=t_alloc, lr_init=lr_try)
            try_seq(v_opt, f"nest_lr{lr_try}")
        
        # One more primal-dual round
        remaining = deadline - time.monotonic() - 100
        if remaining > 15:
            t_alloc = min(remaining, 60)
            print(f"  PD round 2, t={t_alloc:.0f}s")
            v_opt, _, _ = primal_dual_loop(best_seq, n_best, time_limit=t_alloc,
                                           eta_init=1.5, eta_final=0.3)
            try_seq(v_opt, "pd_round2")
    
    print(f"  Phase 3 done. Best: {best_score:.10f} (t={time.monotonic()-t3:.0f}s)")
    
    # === Phase 4: Final polish with tiny perturbations ===
    print("\n=== Phase 4: Final Polish ===")
    t4 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Tiny random perturbations + quick refinement
        for trial in range(8):
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            noise = 0.0003 + 0.0002 * trial
            v_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise)
            t_alloc = min(remaining, 10)
            v_opt, _ = smoothed_gradient_descent(v_pert, n_best, time_limit=t_alloc,
                                                 T_init=0.0003, lr_init=0.003)
            try_seq(v_opt, f"polish_{trial}")
        
        # Final Nesterov
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            t_alloc = min(remaining, 20)
            v_opt, _ = nesterov_subgrad(best_seq, n_best, time_limit=t_alloc, lr_init=0.001)
            try_seq(v_opt, "final_nest")
    
    print(f"  Phase 4 done. Best: {best_score:.10f} (t={time.monotonic()-t4:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
