def fourier_rp(a, n_iter, L):
    """Fourier-domain RP-like iteration for minimizing max auto-correlation."""
    a = a.copy()
    a = a / a.sum()
    a_pad = np.zeros(L)
    a_pad[:len(a)] = a
    
    for it in range(n_iter):
        F = np.fft.fft(a_pad)
        C = np.fft.ifft(F * np.conj(F)).real  # = conv (circular, but L >= 2n so linear)
        C_lin = C[:2*len(a)-1]
        
        # Water-filling: reduce peaks in C by adjusting |F|
        C_mean = C_lin.mean()
        # Where C is high, reduce |F|^2; where low, increase
        ratio = np.sqrt(C_mean / np.maximum(C, 1e-15))
        # Apply to |F|^2: new |F|^2 = |F|^2 * ratio
        F_new_mag_sq = (np.abs(F)**2) * ratio
        F_new_mag = np.sqrt(F_new_mag_sq)
        F_new = F_new_mag * np.exp(1j * np.angle(F))
        
        a_new_pad = np.fft.ifft(F_new).real
        a_new = a_new_pad[:len(a)]
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        a = a_new
    return a

def fourier_diversify(a, rng, strength=0.3):
    """
    Create a diversified version of a using Fourier domain manipulation.
    Randomly phase-scramble the Fourier components while preserving magnitude structure.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(L)
    a_pad[:n] = a

    F = np.fft.rfft(a_pad)
    mag = np.abs(F)
    phase = np.angle(F)

    # Randomize phase in a structured way
    new_phase = phase + rng.standard_normal(len(phase)) * strength
    F_new = mag * np.exp(1j * new_phase)

    a_new_pad = np.fft.irfft(F_new, L)
    a_new = a_new_pad[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def peak_equalize(a, time_limit, verbose=False):
    """
    Iteratively equalize the top peaks by shifting mass from
    contributing positions to non-contributing ones.
    """
    a = a.copy().astype(np.float64)

    n = len(a)

            break

        cmax = conv.max()
        # Find all peaks within 2% of max
        threshold = cmax * 0.98
        active_peaks = np.where(conv >= threshold)[0]

        if len(active_peaks) == 0:
            break

        improved = False
        for k in active_peaks[:5]:  # Limit to top 5
            # Positions contributing to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contrib = np.arange(js, je + 1)
            non_contrib = np.setdiff1d(np.arange(n), contrib)

            if len(contrib) == 0 or len(non_contrib) == 0:
                continue

            # Find the position with most mass among contributors
            idx_max = contrib[np.argmax(a[contrib])]
            val = a[idx_max]
            if val < 1e-10:
                continue

            # Try shifting small amounts
            for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
                shift = val * frac
                if shift < 1e-12:
                    continue

                # Distribute to non-contrib positions weighted by their "safety"
                # (how much they contribute to other peaks - want low contribution)
                nc_contrib_sum = np.zeros(len(non_contrib))
                for j_idx, j in enumerate(non_contrib):
                    for k2 in active_peaks:
                        if max(0, k2 - n + 1) <= j <= min(n - 1, k2):
                            nc_contrib_sum[j_idx] += 1
                # Prefer positions with fewer active peak contributions
                weight = 1.0 / (nc_contrib_sum + 0.5)
                weight = weight / weight.sum()

                delta = np.zeros(n)
                delta[idx_max] -= shift
                delta[non_contrib] += shift * weight

                candidate = project_simplex(a + delta)
                c = obj_value(candidate)
                if c < best_obj - 1e-13:
                    a = candidate
                    best_obj = c

def coordinate_descent(a, time_limit, verbose=False):
    """
    Coordinate descent: at each step, pick a random coordinate pair (i,j)
    and optimize the 2D subproblem (move mass from i to j or vice versa).
    """
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = obj_value(a)
    rng = np.random.default_rng(12345)

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.02:
            break

        # Pick random pair
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        # Optimize: a[i] -> a[i] + t, a[j] -> a[j] - t, for t in [-a[i], a[j]]
        # Objective as function of t: max_k conv(a+t*ei - t*ej, a+t*ei - t*ej)[k]
        # This is complex, so use a simple 1D search
        a_i = a[i]
        a_j = a[j]

        best_t = 0.0
        best_val = best_obj

        # Sample t values
        for t in [0.001, -0.001, 0.005, -0.005, 0.01, -0.01,
                  0.02, -0.02, a_i * 0.1, -a_j * 0.1]:
            if a_i + t < 0 or a_j - t < 0:
                continue
            a[i] = a_i + t
            a[j] = a_j - t
            val = obj_value(a)
            if val < best_val:
                best_val = val
                best_t = t
            a[i] = a_i
            a[j] = a_j

        a[i] = a_i + best_t
        a[j] = a_j - best_t

        if best_val < best_obj - 1e-13:
            best_obj = best_val
            best_a = a.copy()

        if it % 5000 == 0 and verbose:
            print(f"    coord it={it}, obj={best_obj:.10f}")

    return best_a, best_obj


def generate_inits(n, parent, rng):

    inits.append(saw / saw.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())
        # Perturbed parent
        pp = parent + rng.standard_normal(len(parent)) * 0.01
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            pp_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
            pp_interp = np.maximum(pp_interp, 0)
            if pp_interp.sum() > 1e-10:
                inits.append(pp_interp / pp_interp.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search using mirror descent,
    Nesterov subgradient, peak equalization, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Fine n-exploration with mirror descent (~250s) ===
    print("\n=== Phase 1: Fine n-exploration ===")
    t1 = time.monotonic()

    # Fine grid around parent + some other values
    candidate_ns = sorted(set([
        n_parent,
        n_parent - 16, n_parent - 8, n_parent + 8, n_parent + 16,
        n_parent + 32, n_parent + 64,
        max(128, n_parent - 64),
        512, 640, 800, 896, 1024, 1280,
    ]))

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, n_parent), parent)
            start = np.maximum(start, 0)
            start = start / start.sum()

        t_alloc = min(remaining, 18)
        print(f"  Mirror n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent(start, time_limit=t_alloc, T_init=0.003, lr_init=0.08)
        try_seq(a_opt, f"mirror_n{n_try}")

        # Also try topk variant
        remaining = deadline - time.monotonic() - 1150
        if remaining > 10:
            a_opt2, _ = mirror_descent(a_opt, time_limit=min(remaining, 10),
                                       T_init=0.01, lr_init=0.05,
                                       topk_mode=True, topk=8)
            try_seq(a_opt2, f"mirror_topk_n{n_try}")

    print(f"  Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep optimization on best n (~700s) ===
    print("\n=== Phase 2: Deep optimization ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple mirror descent configs
        configs_md = [
            {'T_init': 0.001, 'lr_init': 0.12, 'topk_mode': False, 'topk': 10},
            {'T_init': 0.005, 'lr_init': 0.06, 'topk_mode': False, 'topk': 15},
            {'T_init': 0.01, 'lr_init': 0.04, 'topk_mode': True, 'topk': 5},
            {'T_init': 0.0005, 'lr_init': 0.15, 'topk_mode': False, 'topk': 20},
            {'T_init': 0.02, 'lr_init': 0.03, 'topk_mode': True, 'topk': 3},
        ]

        for ci, cfg in enumerate(configs_md):
            remaining = deadline - time.monotonic() - 500
            if remaining < 20:
                break
            t_alloc = min(remaining, 80)
            print(f"  Mirror cfg{ci} T={cfg['T_init']} lr={cfg['lr_init']} topk={cfg.get('topk_mode')}/{cfg.get('topk')}, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"md_deep_{ci}")

        # Nesterov with different settings
        configs_ne = [
            {'T_init': 0.001, 'lr_init': 0.015, 'momentum_init': 0.7},
            {'T_init': 0.005, 'lr_init': 0.008, 'momentum_init': 0.8},
            {'T_init': 0.0005, 'lr_init': 0.02, 'momentum_init': 0.6},
        ]

        for ci, cfg in enumerate(configs_ne):
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, 80)
            print(f"  Nesterov cfg{ci} T={cfg['T_init']} lr={cfg['lr_init']} mom={cfg['momentum_init']}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_smoothed(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"nesterov_{ci}")

    print(f"  Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Peak equalization + coordinate descent (~400s) ===
    print("\n=== Phase 3: Peak equalization & coordinate descent ===")
    t3 = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 200
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  Peak equalize, t={t_alloc:.0f}s")
            a_opt, _ = peak_equalize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_eq")

        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 120)
            print(f"  Coordinate descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_desc")

        # Another round of mirror descent on the improved result
        remaining = deadline - time.monotonic() - 100
        if remaining > 15:
            t_alloc = min(remaining, 100)
            print(f"  Mirror polish, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc,
                                      T_init=0.0005, lr_init=0.1)
            try_seq(a_opt, "mirror_polish")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Population-based diversification (~300s) ===
    print("\n=== Phase 4: Population-based search ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        pop_size = 5

        # Build initial population
        population = [best_seq.copy()]
        for i in range(pop_size - 1):
            # Create diverse members
            member = fourier_diversify(best_seq, rng, strength=0.1 + 0.1 * i)
            population.append(member)

        # Also add some with different n
        for n_alt in [n_best + 32, n_best - 32]:
            if n_alt > 100 and n_alt <= 2000:
                alt = np.interp(np.linspace(0, 1, n_alt),
                                np.linspace(0, 1, n_best), best_seq)
                alt = np.maximum(alt, 0)
                alt = alt / alt.sum()
                population.append(alt)

        gen = 0
        while time.monotonic() < deadline - 100:
            gen += 1
            # Select top 3
            scored = []
            for m in population:
                s = obj_value(m)
                scored.append((s, m))
            scored.sort(key=lambda x: x[0])
            top = scored[:3]

            # Mutate top members
            new_pop = [m for _, m in top[:2]]
            for s, m in top:
                # Mutation 1: Fourier diversification
                mutated = fourier_diversify(m, rng, strength=0.05 + 0.02 * gen)
                new_pop.append(mutated)
                # Mutation 2: Simplex perturbation
                rng2 = np.random.default_rng((gen * 7 + len(new_pop)) % (2**31))
                perturbed = project_simplex(m + rng2.standard_normal(len(m)) * 0.005)
                new_pop.append(perturbed)

            # Refine each new population member briefly
            refined_pop = []
            for m in new_pop:
                remaining = deadline - time.monotonic() - 80
                if remaining < 5:
                    break
                t_alloc = min(remaining, 20)
                a_opt, _ = mirror_descent(m, time_limit=t_alloc,
                                          T_init=0.001, lr_init=0.08)
                refined_pop.append(a_opt)
                try_seq(a_opt, f"pop_g{gen}")

            if refined_pop:
                population = refined_pop[:pop_size + 2]

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final fine-tuning (~100s) ===
    print("\n=== Phase 5: Final polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine mirror descent
        remaining = deadline - time.monotonic() - 30
        if remaining > 10:
            t_alloc = min(remaining, 40)
            print(f"  Final mirror T=0.0001 lr=0.05, t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent(best_seq, time_limit=t_alloc,
                                      T_init=0.0001, lr_init=0.05)
            try_seq(a_opt, "final_md")

        # Nesterov final
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            t_alloc = min(remaining, 30)
            print(f"  Final Nesterov, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_smoothed(best_seq, time_limit=t_alloc,
                                         T_init=0.0003, lr_init=0.005,
                                         momentum_init=0.8)
            try_seq(a_opt, "final_ne")

        # Tiny perturbations
        for _ in range(3):
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            noise_scale = 0.0003 * (_ + 1)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_alloc = min(remaining, 10)
            a_opt, _ = mirror_descent(f_pert, time_limit=t_alloc,
                                      T_init=0.0002, lr_init=0.03)
            try_seq(a_opt, f"tiny_pert_{_}")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
