def fast_convolve(a):

def logsumexp(x, T):
    """Compute log(sum(exp(x/T))) * T as a smooth max approximation."""
    x_max = x.max()
    if x_max > 1e10:
        return x_max
    return T * (x_max / T + np.log(np.sum(np.exp((x - x_max) / T))))


def smooth_max(x, T):
    """Smooth approximation of max(x) using log-sum-exp."""
    x_max = x.max()
    return x_max + T * np.log(np.sum(np.exp((x - x_max) / T)))


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def adam_smooth_max(a_init, T_schedule, lr, beta1, beta2, eps, time_limit):
    a = a_init.copy()
    n = len(a)
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        T = T_schedule(t)
        
        conv = fast_convolve(a)
        # Compute p
        p = np.exp(conv / T)
        p_sum = p.sum()
        if p_sum > 0:
            p = p / p_sum
        else:
            p = np.ones(2*n-1) / (2*n-1)
        
        # Gradient
        grad = 2.0 * np.correlate(p, a, mode='valid')
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        obj = conv_new.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

**Spectral method for symmetric sequences:**

If `a` is symmetric (palindromic), then `conv(a,a)` is also symmetric. The key property is that in frequency domain, `A[k] * A[L-k] = |A[k]|²` (conjugate symmetry), and for a symmetric real sequence, `A[k]` has a specific phase.

Actually, for a palindromic sequence of length n (a[j] = a[n-1-j]), the DFT is:

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max_and_grad(conv, T):
    """Return (smooth_max_value, gradient_weights p)."""
    x_max = conv.max()
    shifted = (conv - x_max) / T
    exp_shifted = np.exp(shifted)
    p_sum = exp_shifted.sum()
    if p_sum < 1e-300:
        p = np.ones(len(conv)) / len(conv)
    else:
        p = exp_shifted / p_sum
    val = x_max + T * np.log(p_sum)
    return val, p


def gradient_from_weights(p, a):
    """Compute gradient of smooth_max(conv(a,a)) given weights p."""
    # grad[i] = 2 * sum_m a[m] * p[m+i]
    return 2.0 * np.correlate(p, a, mode='valid')


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _solve_lp(f, M, time_limit):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_val_and_weights(conv, T):
    x_max = conv.max()
    shifted = (conv - x_max) / T
    shifted = np.minimum(shifted, 500.0)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    if s < 1e-300:
        p = np.ones(len(conv)) / len(conv)
    else:
        p = exp_s / s
    val = x_max + T * np.log(s)
    return val, p


def grad_from_weights(p, a):
    return 2.0 * np.correlate(p, a, mode='valid')


def adam_optimize(a_init, time_limit, T0=None, T_min=None, lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8):
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv0 = fast_convolve(a)
    if T0 is None:
        T0 = conv0.max() / 8.0
    if T_min is None:
        T_min = conv0.max() / 5000.0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = conv0.max()
    t0 = time.monotonic()
    no_improve = 0
    T = T0
    decay_rate = np.log(T_min / T0) / 8000.0

    while time.monotonic() - t0 < time_limit:
        t += 1
        T = max(T_min, T0 * np.exp(decay_rate * t))

        conv = fast_convolve(a)
        _, p = smooth_max_val_and_weights(conv, T)
        grad = grad_from_weights(p, a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        conv_new = fast_convolve(a_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 800:
            rng = np.random.default_rng(t * 7919 + 13)
            scale = 0.001 + 0.002 * min(1.0, no_improve / 3000.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            T = max(T0, T * 3.0)

    return best_a, best_obj


def solve_lp_oracle(f, M, time_limit=15):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def pairwise_coord_descent(a0, time_limit, n_peaks=8, n_pairs=40):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.2:
            break

        conv = fast_convolve(a)
        top_k = min(n_peaks, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

        improved = False
        rng = np.random.default_rng(it * 31 + 7)

        for k_peak in peak_lags:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            affected = np.arange(js, je + 1)

            if len(affected) < 2:
                continue

            # Sample pairs from affected indices
            if len(affected) > 60:
                sampled = rng.choice(affected, size=60, replace=False)
            else:
                sampled = affected

            best_gain = 0.0
            best_pair = None
            best_delta = 0.0

            for ii in range(len(sampled)):
                for jj in range(ii + 1, len(sampled)):
                    i1 = int(sampled[ii])
                    i2 = int(sampled[jj])

                    c1 = a[k_peak - i1] if 0 <= k_peak - i1 < n else 0.0
                    c2 = a[k_peak - i2] if 0 <= k_peak - i2 < n else 0.0

                    if c1 >= c2:
                        src, dst = i1, i2
                    else:
                        src, dst = i2, i1

                    if a[src] < 1e-10:
                        continue

                    for frac in [0.05, 0.15, 0.3, 0.5]:
                        delta = a[src] * frac
                        a_test = a.copy()
                        a_test[src] -= delta
                        a_test[dst] += delta
                        a_test = np.maximum(a_test, 0.0)
                        ss = a_test.sum()
                        if ss < 1e-15:
                            continue
                        a_test = a_test / ss
                        new_max = fast_convolve(a_test).max()
                        gain = best_obj - new_max
                        if gain > best_gain:
                            best_gain = gain
                            best_pair = (src, dst)
                            best_delta = delta

            if best_pair is not None and best_gain > 1e-13:
                src, dst = best_pair
                a[src] -= best_delta
                a[dst] += best_delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_obj = fast_convolve(a).max()
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1
            if stall > 50:
                scale = 0.0005 + 0.001 * min(1.0, stall / 200.0)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                conv = fast_convolve(a)
                best_obj = conv.max()
                best_a = a.copy()
                stall = 0
            if stall > 500:
                break

    return best_a, best_obj


def spectral_symmetric_refine(a0, time_limit):
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    a = (a + a[::-1]) / 2.0

    t0 = time.monotonic()
    it = 0
    damping = 0.4

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        conv = fast_convolve(a)
        target = np.ones(2 * n - 1) / (2 * n - 1)
        target[n - 1] *= 1.02

        fft_size = 1
        while fft_size < 2 * n - 1:
            fft_size *= 2
        fft_size *= 2

        A = np.fft.fft(a, fft_size)
        B = A * A
        B_target = np.fft.fft(target, fft_size)

        B_mag = np.abs(B)
        Bt_mag = np.abs(B_target)
        ratio = np.ones(fft_size)
        mask = B_mag > 1e-18
        ratio[mask] = np.sqrt(Bt_mag[mask] / B_mag[mask])

        A_new = A * ratio
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = (a_new + a_new[::-1]) / 2.0
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss > 1e-15:
            a_new = a_new / ss
        else:
            continue

        a_candidate = (1.0 - damping) * a + damping * a_new
        a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
        a_candidate = np.maximum(a_candidate, 0)
        ss = a_candidate.sum()
        if ss > 1e-15:
            a_candidate = a_candidate / ss

        c_new = fast_convolve(a_candidate)
        obj = c_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_candidate.copy()
            a = a_candidate
        elif it % 200 == 0:
            damping *= 0.95

    return best_a, best_obj
