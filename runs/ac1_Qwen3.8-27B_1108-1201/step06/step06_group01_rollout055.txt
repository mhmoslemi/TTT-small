24. **Second-order approach**:
    The objective is max_k f^T A_k f where A_k is the matrix with (A_k)_{j,m} = 1 if j+m=k.
    
    At the optimum, suppose the active set is S (indices k where (f*f)[k] = max). Then we want to minimize max_{k in S} f^T A_k f.
    
    By the maximum principle, at the optimum, there exist weights w_k >= 0, sum w_k = 1, such that the gradient of sum w_k * f^T A_k f is zero (projected onto the tangent space of the simplex).
    
    This gives us the KKT conditions:
    2 * sum_{k in S} w_k * (A_k f) = lambda * 1 + mu (componentwise, where mu_j >= 0 for f_j = 0)
    
    And for k in S: (f*f)[k] = t (the max value)
    For k not in S: (f*f)[k] <= t
    
    This is a system of equations that we can try to solve iteratively.

25. **Practical algorithm based on KKT conditions**:

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(f, conv, T):
    """Compute gradient of log-sum-exp smoothed max of convolution."""
    n = len(f)
    scaled = conv * T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p = p / p_sum
    # grad[j] = 2 * sum_m p[j+m] * f[m] = 2 * cross-correlation of p and f at lag j
    # p has length 2n-1, f has length n
    # np.correlate(p, f, mode='valid')[j] = sum_m p[j+m]*f[m], length = (2n-1)-n+1 = n
    grad = 2.0 * np.correlate(p, f, mode='valid')
    return grad


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def smoothed_nesterov(f0, time_limit, T_init=1000.0, T_max=8000.0,
                       lr_init=0.02, momentum_decay=0.9):
    """Nesterov-accelerated projected gradient on smoothed max."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    y = f.copy()
    v = np.zeros(n)
    t_nest = 1.0
    T = T_init
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        # Nesterov lookahead
        alpha = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha
        y = (1.0 + beta) * f - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = f.copy()
        else:
            y = y / ys
        t_nest += 1.0

        conv = fast_convolve(y)
        grad = smooth_gradient(y, conv, T)

        v = momentum_decay * v - lr * grad
        f_new = project_simplex(f + v)

        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            f_new = f

        f = f_new

        # Anneal temperature
        T = min(T * 1.002, T_max)
        lr *= (1.0 - 0.0001)

        # Restart if stuck
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.001 * min(1.0, no_improve / 500.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            y = f.copy()
            v = np.zeros(n)
            t_nest = 1.0
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_f, best_obj


def fw_with_lp(f0, time_limit, max_lp_calls=50):
    """Frank-Wolfe with LP oracle, limited number of LP calls."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    lp_count = 0

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        if lp_count >= max_lp_calls:
            break

        g = solve_lp_oracle(f, M * 1.01)
        lp_count += 1
        if g is None:
            if lp_count < max_lp_calls - 1:
                continue
            else:
                break

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 15:
            break

    return best_f, best_obj


def coordinate_polish(f0, time_limit):
    """Pairwise coordinate descent targeting top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(f)
        top_k = min(8, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

        improved = False
        for k_peak in peak_lags[:4]:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            affected = np.arange(js, je + 1)
            if len(affected) < 2:
                continue

            # Sample a subset for speed
            if len(affected) > 40:
                rng_s = np.random.default_rng(hash((stall, k_peak)) % (2**31))
                affected = rng_s.choice(affected, size=40, replace=False)

            best_gain = 1e-15
            best_move = None

            for ii in range(len(affected)):
                j1 = affected[ii]
                c1 = f[k_peak - j1] if 0 <= k_peak - j1 < n else 0.0
                for jj in range(ii + 1, len(affected)):
                    j2 = affected[jj]
                    c2 = f[k_peak - j2] if 0 <= k_peak - j2 < n else 0.0
                    if c1 < c2 and f[j1] > 1e-8:
                        src, dst = j1, j2
                    elif c2 >= c1 and f[j2] > 1e-8:
                        src, dst = j2, j1
                    else:
                        continue
                    for frac in [0.05, 0.15, 0.3]:
                        delta = f[src] * frac
                        if delta < 1e-10:
                            continue
                        ft = f.copy()
                        ft[src] -= delta
                        ft[dst] += delta
                        ft = np.maximum(ft, 0.0)
                        st = ft.sum()
                        if st < 1e-15:
                            continue
                        ft = ft / st
                        new_max = fast_convolve(ft).max()
                        gain = best_obj - new_max
                        if gain > best_gain:
                            best_gain = gain
                            best_move = (src, dst, frac)

            if best_move is not None:
                src, dst, frac = best_move
                delta = f[src] * frac
                f[src] -= delta
                f[dst] += delta
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                best_obj = fast_convolve(f).max()
                best_f = f.copy()
                improved = True

        if not improved:
            stall += 1
            if stall > 30:
                rng = np.random.default_rng((stall * 48271 + 1) % (2**31))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                stall = 0
            if stall > 200:
                break

    return best_f, best_obj


def make_initialization(n, method="uniform"):
    """Create structured initializations."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "raised_cosine":
        x = np.linspace(0, np.pi, n)
        f = 0.5 * (1 + np.cos(x))
        return f / f.sum()
    elif method == "gaussian":
        x = np.linspace(-3, 3, n)
        f = np.exp(-x**2 / 2)
        return f / f.sum()
    elif method == "u_shape":
        x = np.linspace(-3, 3, n)
        f = np.exp(x**2 / 4)
        return f / f.sum()
    elif method == "bimodal":
        x = np.linspace(-4, 4, n)
        f = np.exp(-(x - 2.5)**2 / 2) + np.exp(-(x + 2.5)**2 / 2) + 0.1
        return f / f.sum()
    elif method == "triangle":
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
        f = np.maximum(f, 0.01)
        return f / f.sum()
    elif method == "edge_heavy":
        x = np.linspace(0, 1, n)
        f = x**2 + (1 - x)**2 + 0.1
        return f / f.sum()
    else:
        return np.ones(n) / n

                        parent_resized = parent_resized / ps
                    else:
                        parent_resized = np.ones(n_try) / n_try
                else:
                    parent_resized = np.ones(n_try) / n_try
                structured = make_initialization(n_try, method)
                start = 0.5 * parent_resized + 0.5 * structured
                start = start / start.sum()

        print(f"  n={n_try}, init={method if n_try != n_parent else 'parent'}, t={t_alloc:.0f}s")
        a_opt, _ = smoothed_nesterov(start, time_limit=t_alloc,
                                     T_init=500.0, T_max=5000.0,
                                     lr_init=0.03, momentum_decay=0.9)
        try_seq(a_opt, f"p1_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep optimization on best n (~1000s) ===
    print("\n=== Phase 2: Deep optimization ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # 2a: Long Nesterov run
        t_alloc = min(400, deadline - time.monotonic() - 600)
        if t_alloc > 30:
            print(f"  Deep Nesterov n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_nesterov(best_seq, time_limit=t_alloc,
                                         T_init=200.0, T_max=10000.0,
                                         lr_init=0.02, momentum_decay=0.92)
            try_seq(a_opt, "p2a_nest")

        # 2b: FW with LP (limited calls)
        t_alloc = min(200, deadline - time.monotonic() - 400)
        if t_alloc > 20:
            print(f"  FW-LP n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_with_lp(best_seq, time_limit=t_alloc, max_lp_calls=40)
            try_seq(a_opt, "p2b_fw")

        # 2c: Another Nesterov with different params from best
        if best_seq is not None:
            t_alloc = min(200, deadline - time.monotonic() - 250)
            if t_alloc > 20:
                print(f"  Nesterov v2 n={len(best_seq)}, t={t_alloc:.0f}s")
                a_opt, _ = smoothed_nesterov(best_seq, time_limit=t_alloc,
                                             T_init=50.0, T_max=5000.0,
                                             lr_init=0.05, momentum_decay=0.85)
                try_seq(a_opt, "p2c_nest2")

        # 2d: LP-augmented hybrid
        if best_seq is not None:
            t_alloc = min(150, deadline - time.monotonic() - 150)
            if t_alloc > 20:
                print(f"  Hybrid LP n={len(best_seq)}, t={t_alloc:.0f}s")
                a_opt, _ = fw_with_lp(best_seq, time_limit=t_alloc, max_lp_calls=30)
                try_seq(a_opt, "p2d_hybrid")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Coordinate descent polish (~300s) ===
    print("\n=== Phase 3: Coordinate polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(200, deadline - time.monotonic() - 150)
        if t_alloc > 10:
            print(f"  Coord descent n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "p3_coord")

        # Second round of Nesterov from polished result
        if best_seq is not None:
            t_alloc = min(80, deadline - time.monotonic() - 80)
            if t_alloc > 10:
                print(f"  Nesterov v3 n={len(best_seq)}, t={t_alloc:.0f}s")
                a_opt, _ = smoothed_nesterov(best_seq, time_limit=t_alloc,
                                             T_init=10000.0, T_max=10000.0,
                                             lr_init=0.005, momentum_decay=0.95)
                try_seq(a_opt, "p3_nest3")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Final micro-refinement (~remaining time) ===
    print("\n=== Phase 4: Final refinement ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Very fine coordinate polish
        t_alloc = min(100, deadline - time.monotonic() - 30)
        if t_alloc > 10:
            print(f"  Micro coord n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "p4_micro_coord")

        # One more short Nesterov
        t_alloc = min(50, deadline - time.monotonic() - 15)
        if t_alloc > 5:
            print(f"  Final nest n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_nesterov(best_seq, time_limit=t_alloc,
                                         T_init=5000.0, T_max=5000.0,
                                         lr_init=0.002, momentum_decay=0.98)
            try_seq(a_opt, "p4_final")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
