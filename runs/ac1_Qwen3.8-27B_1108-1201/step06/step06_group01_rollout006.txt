"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using peak-weighted gradient descent
with Nesterov acceleration, exact line search, and periodic FW-LP jumps.
Key improvements over previous:
- Full softmax-weighted gradient (all peaks, temperature-annealed)
- Exact 1D line search via ternary search on convex piecewise-quadratic
- Nesterov momentum for O(1/T^2) convergence
- Adaptive temperature control
- Diverse initializations including structured sequences
- Multiple sequence length exploration

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_peak_gradient(a, conv, T, top_k=None):
    """
    Compute the gradient of the softmax-weighted peak objective.
    F(a) = sum_k w_k * (a*a)[k] where w_k = softmax(conv/T)
    grad_i = 2 * sum_k w_k * a[k-i] (with valid index handling)
    """
    n = len(a)
    size = len(conv)
    # Softmax weights
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    if top_k is not None and top_k < size:
        # Keep only top-k weights
        idx = np.argpartition(w, -top_k)[-top_k:]
        w_full = np.zeros(size)
        w_full[idx] = w[idx]
        w_full /= w_full.sum()
        w = w_full
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (valid when 0 <= k-i < n)
    g = np.zeros(n)
    for i in range(n):
        # k ranges from i to i+n-1 (so that k-i ranges from 0 to n-1)
        k_start = i
        k_end = min(i + n, size)
        if k_start >= k_end:
            continue
        g[i] = 2.0 * np.dot(w[k_start:k_end], a[:k_end - k_start])
    
    return g

def compute_peak_gradient_fast(a, conv, T):
    """
    Fast computation of gradient using convolution.
    g[i] = 2 * sum_k w[k] * a[k-i] = 2 * (w_rev * a)[i] where w_rev is reversed w padded.
    Actually: g[i] = 2 * sum_j a[j] * w[i+j]
    This is a "cross-correlation" of a with w, or equivalently a convolution with reversed w.
    """
    n = len(a)
    size = len(conv)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # g[i] = 2 * sum_j a[j] * w[i+j] for j=0..n-1 and i+j < size
    # This is the "valid" cross-correlation of a with w (or convolution with reversed w)
    # g[i] = 2 * (a * w_rev)[i] where w_rev[j] = w[size-1-j]
    # Actually, let's use FFT
    # g[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    
    # This is the "cross-correlation" of a with w, evaluated at lags 0, 1, ..., n-1
    # cross_corr[i] = sum_j a[j] * w[i+j]
    
    # Using FFT: FFT(cross_corr) = FFT(a) * conj(FFT(w_padded))
    # But we need a specific alignment...
    
    # Alternative: use np.correlate
    # w has length size = 2n-1
    # We want sum_j a[j] * w[i+j] for i = 0, ..., n-1
    # This is like sliding a over w with a specific alignment
    
    # Let's use the FFT approach:
    # g[i]/2 = sum_j a[j] * w[i+j]
    # Let's define h[k] = w[k] for k=0..size-1
    # g[i]/2 = sum_{j=0}^{n-1} a[j] * h[i+j]
    # = correlation of a with h at lag i (with a "forward" alignment)
    
    # Using FFT:
    fft_size = 1 << (n + size - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    H = np.fft.rfft(h := np.zeros(fft_size), fft_size)
    # h should be w placed at the right position
    # h[k] = w[k] for k=0..size-1
    h[:size] = w
    H = np.fft.rfft(h, fft_size)
    
    # We want: g[i] = 2 * sum_j a[j] * w[i+j]
    # In FFT terms, if we define the "cross-correlation" as:
    # (a ⋆ w)[m] = sum_j a[j] * w[j+m]
    # Then g[i] = 2 * (a ⋆ w)[i]
    
    # The FFT of (a ⋆ w) is: conj(FFT(a)) * FFT(w)
    # But we need to be careful with the alignment.
    
    # Actually, let's just use the direct formula with FFT:
    # Define: result[m] = sum_j a[j] * w[j+m]
    # This is the "cross-correlation" of a and w.
    # FFT: FFT(result) = conj(FFT(a_padded)) * FFT(w_padded)
    # where a_padded is a zero-padded to length fft_size
    # and w_padded is w zero-padded to length fft_size
    
    # But actually, the standard cross-correlation in numpy is:
    # np.correlate(a, w, mode='valid') gives sum_j a[j] * w[j+i] for i=0..len(w)-len(a)
    # which is what we want! (since len(w) = 2n-1 >= n)
    
    # Let's just use np.correlate for simplicity (it's O(n*size) = O(n^2) but n is at most ~1280)
    # Actually for n=768, size=1535, this is 768*1535 ≈ 1.18M operations per gradient eval.
    # That's fine.
    
    # g[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # = 2 * np.correlate(w, a, mode='valid')[i]... no
    # np.correlate(w, a, 'valid')[i] = sum_j w[i+j] * a[j] for j=0..len(a)-1
    # Yes! That's exactly what we want.
    
    corr = np.correlate(w, a, mode='valid')  # length = size - n + 1 = n
    return 2.0 * corr

def peak_weighted_gradient_descent(a0, time_limit, T_init=0.01, T_min=0.001, 
                                      lr_init=0.01, momentum=True):
    """
    Peak-weighted gradient descent with Nesterov acceleration and exact line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    T = T_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    t_nest = 1.0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        current_max = conv.max()
        
        # Compute gradient using peak-weighted approach
        if momentum and t_nest > 1:
            alpha_n = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            beta = (t_nest - 1.0) / alpha_n
            y = (1.0 + beta) * a - beta * y
            y = np.maximum(y, 0.0)
            ys = y.sum()
            if ys < 1e-15:
                y = a.copy()
            else:
                y = y / ys
            t_nest += 1.0
        else:
            y = a.copy()
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        grad = compute_peak_gradient_fast(y, conv_y, T)
        
        # Gradient direction (negative gradient)
        direction = -grad
        
        # Project direction onto the tangent space of the simplex
        # (subtract the mean to ensure sum(direction) = 0)
        direction = direction - direction.mean()
        
        # Line search: find best alpha
        best_alpha = lr
        best_val = current_max
        
        # Coarse search
        for trial_alpha in [lr, lr*0.5, lr*0.25, lr*2, lr*4]:
            candidate = a + trial_alpha * direction
            candidate = project_simplex(candidate)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val:
                best_val = val
                best_alpha = trial_alpha
        
        # Fine search (ternary)
        if best_val < current_max:
            lo = 0.0
            hi = best_alpha * 1.5
            # Find the minimum of f(alpha) = max_k conv(a + alpha*direction, projected)
            for _ in range(20):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                c1 = project_simplex(a + m1 * direction)
                c2 = project_simplex(a + m2 * direction)
                v1 = fast_convolve(c1).max()
                v2 = fast_convolve(c2).max()
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
                # Check if we found improvement
                mid = (lo + hi) / 2.0
                cmid = project_simplex(a + mid * direction)
                vmid = fast_convolve(cmid).max()
                if vmid < best_val:
                    best_val = vmid
                    best_alpha = mid
        
        if best_val < current_max - 1e-15:
            a_new = project_simplex(a + best_alpha * direction)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            
            # Update momentum
            v = beta * v + best_alpha * direction if momentum else best_alpha * direction
            a = a_new
            
            if best_val < best_obj - 1e-15:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
                # Anneal temperature
                T = max(T * 0.995, T_min)
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Adjust temperature
            if no_improve % 20 == 0:
                T = min(T * 1.2, T_init * 5)
        
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            if momentum:
                v = np.zeros(n)
                t_nest = 1.0
            no_improve = 0
            T = T_init
            if momentum:
                y = a.copy()
    
    return best_a, best_obj

def fw_lp_optimize_v2(f0, time_limit):
    """Improved FW-LP with better oracle and dual tracking."""
    f = np.maximum(f0.astype(np.float64), 0.0)
