"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast multi-peak subgradient with adaptive temperature
- Frank-Wolfe with sparse LP oracle and adaptive M relaxation
- KKT-condition-based coordinate polishing
- Multi-length, multi-start exploration with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_sparse(n, f):
    """Build sparse (2n-1) x n matrix A where (A @ g) = conv(f, g)."""
    rows_parts = []
    for j in range(n):
        rows_parts.append(np.arange(j, j + n))
    rows = np.concatenate(rows_parts)
    cols = np.repeat(np.arange(n), n)
    vals = np.tile(f, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M (elementwise), g >= 0."""
    n = len(f)
    A = build_conv_matrix_sparse(n, f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def subgradient_step(a, C, T, top_k, n):
    """Compute weighted subgradient from top peaks."""
    if top_k < len(C):
        peaks = np.argpartition(C, -top_k)[-top_k:]
    else:
        peaks = np.arange(len(C))
    pv = C[peaks]
    pv_max = pv.max()
    w = np.exp((pv - pv_max) / (T * 0.1))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(peaks)) / len(peaks)
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]
    return grad


def run_subgradient(a0, time_limit, T_init=0.0005, lr_init=0.02, top_k_frac=0.05):
    """Smoothed subgradient descent with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.92
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        C = fast_convolve(a)
        grad = subgradient_step(a, C, T, top_k, n)
        v = mom * v - lr * grad
        best_candidate = a
        best_val = C.max()
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        a = best_candidate
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 20:
            T *= 1.06
            lr *= 0.97
        if stagnant > 60:
            T *= 0.6
            lr *= 1.3
            mom = 0.95
        if stagnant > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            mom = 0.92
            stagnant = 0
    return best_a, best_obj


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle, adaptive M, and efficient line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    t0 = time.monotonic()
    stagnant = 0
    it = 0

        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        g = solve_lp_oracle(f, M * m_relax)
        if g is None or g.sum() < 1e-15:

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 15:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.002 + 0.005 * min(1.0, stagnant / 40.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.001
                stagnant = 0
            continue
        g = g / g.sum()
        if np.max(np.abs(g - f)) < 1e-11:
            stagnant += 1
            if stagnant > 25:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue
        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M + 1e-14
        alphas_coarse = np.linspace(0.01, 1.0, 25)
        for alpha in alphas_coarse:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 0.005:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            alphas_fine = np.linspace(lo, hi, 20)
            for alpha in alphas_fine:
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.001
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
        if stagnant > 8:
            m_relax = min(m_relax * 1.03, 1.04)
        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stag={stagnant}")
    return best_f, best_obj


def kkt_polish(a0, time_limit):
    """Polish using KKT-based mass transfer from high to low fl coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        C = fast_convolve(a)
        t = C.max()
        tol = 0.005 * t
        active = np.where(C >= t - tol)[0]
        if len(active) == 0:
            break
        F = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            F[idx_arr] += 2.0 * a[k - idx_arr]
        active_coords = np.where(a > 1e-8)[0]
        if len(active_coords) < 2:
            break
        F_active = F[active_coords]
        if F_active.std() < 1e-15:
            if it > 50:
                break
        sorted_idx = np.argsort(F_active)
        improved = False
        for trial in range(3):
            hi_pos = len(sorted_idx) - 1 - trial
            lo_pos = trial
            if hi_pos <= lo_pos:
                break
            hi = active_coords[hi_pos]
            lo = active_coords[lo_pos]
            fracs = [0.005, 0.002, 0.001, 0.0005]
            for frac in fracs:
                delta = a[hi] * frac
                if delta < 1e-12:
                    continue
                candidate = a.copy()
                candidate[hi] -= delta
                candidate[lo] += delta
                candidate = np.maximum(candidate, 0)
                candidate = candidate / candidate.sum()
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            if it % 50 == 0 and it < 200:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
                elif it > 100:
                    break
            elif it > 150:
                break
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []
    if parent is not None:
        starts.append(resample_sequence(parent, n))
    starts.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    cos_t = np.cos(np.pi * x / max(1, n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.005)
    starts.append(w / w.sum())
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.25 * n)) ** 2)
    starts.append(g / g.sum())
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())
    g1 = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.7 * (n - 1)) / (0.1 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())
    for _ in range(2):
        starts.append(rng.dirichlet(np.ones(n) * 0.2))
    # "Rippled" pattern
    ripple = 0.5 + 0.4 * np.sin(2 * np.pi * x / (n / 3.0))
    ripple = np.maximum(ripple, 0.01)
    starts.append(ripple / ripple.sum())
    return starts
