def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:

    cssv = np.cumsum(u)

    W is Hankel: W[i,j] = w[i+j].
    """
    n = len(a_init)
    a = project_simplex(a_init.copy())
    f_curr = qp_value(w, a)
    
    # Estimate Lipschitz constant
    Wa = hankel_mv(w, a)
    grad_norm = np.linalg.norm(2.0 * Wa)
    lr = 1.0 / max(grad_norm + 1e-10, 0.01)
    
    v = np.zeros(n)
    for it in range(max_iters):
        Wa = hankel_mv(w, a)
        grad = 2.0 * Wa
        
        # Backtracking line search
        step_accepted = False
        for bt in range(50):

            f_new = qp_value(w, a_new)
            # Sufficient decrease condition
            if f_new <= f_curr - 1e-4 * lr * np.dot(grad, a - a_new):
                step_accepted = True
                break

            f_new = qp_value(w, a_new)

                step_accepted = True

        a = a_new
        f_curr = f_new

        peak_vals = conv[peak_indices]
        T_peak = 0.005
        wp = np.exp((peak_vals - peak_vals.max()) / T_peak)
        wp = wp / wp.sum()
        
        # Average gradient
        grad = np.zeros(n)
        for ii in range(k):
            pi = int(peak_indices[ii])
            js = max(0, pi - n + 1)
            je = min(n - 1, pi)
            idx = np.arange(js, je + 1)
            grad[idx] += wp[ii] * 2.0 * a[pi - idx]
        
        # Nesterov update
        y = a + mom * v
        y_new = project_simplex(y - lr * grad)
        v = y_new - y
        a = y_new
        a = project_simplex(a)
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt
        lr *= 0.9997
        mom = min(0.999, mom + 0.00003)
        
        if no_improve > 100:
            lr *= 0.85
            mom *= 0.97
        if no_improve > 400:
            rng = np.random.default_rng((it * 48271 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            lr = lr_init * 0.5
            mom = mom_init
            no_improve = 0
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    x = np.arange(n, dtype=np.float64)
    # Raised cosine (edge-heavy)
    rc = 0.5 + 0.5 * np.cos(2 * np.pi * x / (n - 1))
    rc = np.maximum(rc, 0.01)
    inits.append(rc / rc.sum())
    # Inverse raised cosine (center-heavy)
    irc = 1.0 - rc
    irc = np.maximum(irc, 0.01)
    inits.append(irc / irc.sum())
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    # Two peaks
    g2 = (np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2))
    inits.append(g2 / g2.sum())
    # Linear ramp (asymmetric)
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())
    # V-shape (minimum in center)
    vshape = np.minimum(x, n - 1 - x) + 1.0
    inits.append(vshape / vshape.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for auto-correlation minimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768
    
    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")
    
    # === Phase 1: L-BFGS softmax with annealing for multiple n values (~500s) ===
    print("\n=== Phase 1: L-BFGS Softmax Annealing ===")
    t1 = time.monotonic()
    
    n_candidates = sorted(set([
        n_parent,
        max(256, n_parent - 128),
        min(2048, n_parent + 128),
        max(256, n_parent // 2),
        512, 768, 1024, 1536,
    ]))
    
    for n_try in n_candidates:
        remaining = deadline - time.monotonic() - 700
        if remaining < 30:
            break
        
        # Get initial point
        if n_try == n_parent:
            a_start = parent.copy()
        else:
            a_start = np.interp(np.linspace(0, 1, n_try),
                                np.linspace(0, 1, n_parent), parent)
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        
        t_alloc = min(remaining, 60)
        print(f"  L-BFGS anneal n={n_try}, t={t_alloc:.0f}s")
        a_opt = annealed_lbgfs(a_start, time_limit=t_alloc)
        try_seq(a_opt, f"lbgfs_n{n_try}")
        
        # Also try from a different initialization
        remaining = deadline - time.monotonic() - 650
        if remaining > 30:
            a_start2 = generate_inits(n_try, parent, rng)[1]  # Dirichlet
            t_alloc2 = min(remaining, 40)
            a_opt2 = annealed_lbgfs(a_start2, time_limit=t_alloc2,
                                    T_list=[0.03, 0.01, 0.005, 0.002])
            try_seq(a_opt2, f"lbgfs_dir_n{n_try}")
    
    print(f"  Phase 1 done. Best: {best_score:.10f} ({time.monotonic()-t1:.0f}s)")
    
    # === Phase 2: Deep primal-dual refinement (~600s) ===
    print("\n=== Phase 2: Deep Primal-Dual ===")
    t2 = time.monotonic()
    
    if best_seq is not None:
        configs = [
            {'T_init': 0.005, 'use_topk': False, 'topk': 15, 'blend': 0.4},
            {'T_init': 0.01, 'use_topk': False, 'topk': 20, 'blend': 0.5},
            {'T_init': 0.01, 'use_topk': True, 'topk': 10, 'blend': 0.5},
            {'T_init': 0.02, 'use_topk': True, 'topk': 5, 'blend': 0.6},
            {'T_init': 0.003, 'use_topk': False, 'topk': 10, 'blend': 0.3},
            {'T_init': 0.01, 'use_topk': True, 'topk': 20, 'blend': 0.4},
        ]
        
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 300
            if remaining < 20:
                break
            t_alloc = min(remaining, 120)
            print(f"  QP-PD v2 cfg{ci} (T={cfg['T_init']}, topk={cfg.get('use_topk')}, "
                  f"k={cfg['topk']}, blend={cfg['blend']}), t={t_alloc:.0f}s")
            a_opt, _ = qp_primal_dual_v2(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"qp2_cfg{ci}")
    
    print(f"  Phase 2 done. Best: {best_score:.10f} ({time.monotonic()-t2:.0f}s)")
    
    # === Phase 3: Multi-peak Nesterov subgradient (~300s) ===
    print("\n=== Phase 3: Multi-Peak Nesterov ===")
    t3 = time.monotonic()
    
    if best_seq is not None:
        for lr_i, nk in [(0.005, 8), (0.01, 12), (0.003, 5), (0.008, 15)]:
            remaining = deadline - time.monotonic() - 150
            if remaining < 15:
                break
            t_alloc = min(remaining, 80)
            print(f"  Multi-peak Nesterov lr={lr_i}, k={nk}, t={t_alloc:.0f}s")
            a_opt, _ = multi_peak_nesterov(best_seq, time_limit=t_alloc,
                                           lr_init=lr_i, n_peaks=nk)
            try_seq(a_opt, f"nesterov_lr{lr_i}_k{nk}")
    
    print(f"  Phase 3 done. Best: {best_score:.10f} ({time.monotonic()-t3:.0f}s)")
    
    # === Phase 4: L-BFGS polish on best (~200s) ===
    print("\n=== Phase 4: L-BFGS Polish ===")
    t4 = time.monotonic()
    
    if best_seq is not None:
        # Very low temperature L-BFGS for fine tuning
        for T in [0.001, 0.0005, 0.0003, 0.0002]:
            remaining = deadline - time.monotonic() - 80
            if remaining < 10:
                break
            t_alloc = min(remaining, 50)
            print(f"  L-BFGS polish T={T}, t={t_alloc:.0f}s")
            a_opt = run_lbgfs_softmax(best_seq, T, max_iter=1000, time_limit=t_alloc)
            try_seq(a_opt, f"polish_T{T}")
    
    print(f"  Phase 4 done. Best: {best_score:.10f} ({time.monotonic()-t4:.0f}s)")
    
    # === Phase 5: Final perturbations and verification (~100s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Tiny perturbations with quick L-BFGS refinement
        for _ in range(4):
            remaining = deadline - time.monotonic() - 15
            if remaining < 8:
                break
            noise_scale = 0.0005 + 0.0005 * _
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            t_alloc = min(remaining, 20)
            a_opt = run_lbgfs_softmax(f_pert, 0.0005, max_iter=500, time_limit=t_alloc)
            try_seq(a_opt, f"perturb_{_}")
        
        # One more Nesterov pass with very small lr
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            t_alloc = min(remaining, 30)
            a_opt, _ = multi_peak_nesterov(best_seq, time_limit=t_alloc,
                                           lr_init=0.002, n_peaks=20, mom_init=0.95)
            try_seq(a_opt, "final_nesterov")
        
        # Quick QP-PD
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            t_alloc = min(remaining, 25)
            a_opt, _ = qp_primal_dual_v2(best_seq, time_limit=t_alloc,
                                         T_init=0.001, use_topk=False, topk=10, blend=0.3)
            try_seq(a_opt, "final_qp")
    
    print(f"  Phase 5 done. Best: {best_score:.10f} ({time.monotonic()-t5:.0f}s)")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
