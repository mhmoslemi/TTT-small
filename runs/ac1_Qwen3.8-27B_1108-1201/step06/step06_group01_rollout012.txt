def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_oracle(f, M):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, dual_info) where dual_info contains active constraint info.
    """
    n = len(f)
    c = -np.ones(n)
    
    # Build constraint matrix: A_ub[k, j] = f[j] if j+k < n (for k < n)
    # and f[n-1-j+k] ... actually let's think more carefully.
    # (f*g)_k = sum_j f[j] * g[j+k] for valid j, j+k
    # Equivalently: (f*g)_k = sum_j f[j+k] * g[j] for valid j, j+k
    # So A_ub[k, j] = f[j+k] if 0 <= j+k < n, else 0
    
    # For k = 0: A[0, j] = f[j] for j in 0..n-1
    # For k = 1: A[1, j] = f[j+1] for j in 0..n-2, A[1, n-1] = 0
    # For k = n-1: A[n-1, j] = f[j+n-1] for j=0 only (f[n-1]), rest 0
    # For k = n: A[n, j] = f[j+n] ... only j=0 gives f[n] which doesn't exist
    #   Actually (f*g)_n = sum_j f[j]*g[j+n], only valid when j+n < n, i.e., j < 0. So (f*g)_n = 0.
    # Wait, that's not right. Let me use the standard formula.
    # (f*g)_k = sum_{j} f[j] * g[k-j] where both indices in [0, n-1]
    # = sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} f[j] * g[k-j]
    # Let i = k-j, so j = k-i:
    # = sum_{i: max(0,n-1-k) <= i <= min(n-1,k)} f[k-i] * g[i]
    # So A_ub[k, i] = f[k-i] if max(0, n-1-k) <= i <= min(n-1, k), else 0
    
    # For efficiency, build dense matrix (n is at most ~1500)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in (f*g)_k for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j] = f[0], f[1], ..., f[n-1]
        A_ub[j:j+n, j] = f
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.success:
            g = np.maximum(res.x, 0.0)
            # Extract dual info: which constraints are binding
            if res.ineqlin is not None and len(res.ineqlin) == 2 * n - 1:
                # res.ineqlin gives the slack for each inequality constraint
                # Actually in scipy, res.ineqlin might not be directly the slack
                # Let's compute it ourselves
                slacks = M - A_ub @ g
                active = slacks < 1e-8 * max(1.0, M)
                return g, active
            return g, None
    except Exception:
        pass
    return None, None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        g, active = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.2, 4.0)
            if m_exp > 3.5:
                m_exp = 1.0
                g = np.ones(n) / n
                active = None
            else:
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 5
            m_exp = min(m_exp * 1.1, 3.0)
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.02, 2.5)
            continue

        # Line search: find best alpha in [0, 1]
        best_alpha = 0.0
        best_val = M

        # Coarse search
        alphas = [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection around best_alpha
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-12:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW it={it}, obj={M:.8f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)

        # Restart with perturbation if stuck
        if no_improve > 30:
            rng = np.random.default_rng(it * 7919 + 13)
            # Informed perturbation: bias against active peaks
            noise = rng.standard_normal(n) * 0.001
            if active is not None and active.any():

            no_improve = 0

def dual_informed_subgradient(f, conv, top_k=50):
    """Compute a weighted subgradient using peaks as weights."""
    n = len(f)
    M = conv.max()
    
    # Get top-k peaks
    k_actual = min(top_k, len(conv))
    peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
    peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]
    
    # Weights: softmax-like
    peak_vals = conv[peak_indices]
    T = max(M * 0.01, 1e-10)
    weights = np.exp((peak_vals - M) / T)
    weights = weights / weights.sum()
    
    # Compute subgradient: grad of max_k (f*f)_k is 2*f conjugated with peak indicator
    sg = np.zeros(n)
    for i, k in enumerate(peak_indices):
        if weights[i] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            partner = k - j
            if 0 <= partner < n:
                sg[j] += 2.0 * weights[i] * f[partner]
    
    return sg


def nesterov_subgradient(a0, time_limit, lr_init=0.003, max_k=30):
    """Nesterov-accelerated subgradient with dual-informed gradients."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    t_nest = 1.0
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        alpha_n = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / alpha_n
        y = (1.0 + beta) * a - beta * y
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        t_nest += 1.0

        conv = fast_convolve(y)
        sg = dual_informed_subgradient(y, conv, top_k=max_k)

        v = beta * v - lr * sg
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a

        a = a_new
        lr *= 0.9998

        if no_improve > 80:
            rng = np.random.default_rng((it * 104729 + 7) % (2**31))
            noise_scale = 0.001 * min(3.0, np.sqrt(no_improve / 80.0))
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            y = a.copy()
            v = np.zeros(n)
            t_nest = 1.0
            no_improve = 0

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    """
    Targeted coordinate descent: for each peak, try pairwise mass transfers
    to reduce the peak height. Uses greedy best-improvement with early termination.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        top_k = min(8, len(conv))
        peak_lags = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = peak_lags[np.argsort(conv[peak_lags])[::-1]]

        improved = False

        for k_peak in peak_lags:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            if conv[int(k_peak)] < best_obj * 0.98:
                break  # Skip peaks that are already below best

            js = max(0, int(k_peak) - n + 1)
            je = min(n - 1, int(k_peak))
            affected = np.arange(js, je + 1)

            if len(affected) < 2:
                continue

            # Sample affected indices if too many
            if len(affected) > 25:
                rng_s = np.random.default_rng(it * 31 + int(k_peak))
                affected = rng_s.choice(affected, size=25, replace=False)

            # Find best mass transfer
            best_gain = 0
            best_src = -1
            best_dst = -1
            best_frac = 0

            for ii in range(len(affected)):
                j1 = affected[ii]
                if a[j1] < 1e-10:
                    continue
                partner1 = int(k_peak) - j1
                contrib1 = a[partner1] if 0 <= partner1 < n else 0

                for jj in range(ii + 1, len(affected)):
                    j2 = affected[jj]
                    partner2 = int(k_peak) - j2
                    contrib2 = a[partner2] if 0 <= partner2 < n else 0

                    # Move from higher contributor to lower
                    if contrib1 > contrib2 * 1.1 and a[j1] > a[j2] * 0.5:
                        src, dst = j1, j2
                    elif contrib2 > contrib1 * 1.1 and a[j2] > a[j1] * 0.5:
                        src, dst = j2, j1
                    else:
                        continue

                    for frac in [0.05, 0.1, 0.2, 0.35]:
                        delta = a[src] * frac
                        if delta < 1e-12:
                            continue
                        a_test = a.copy()
                        a_test[src] -= delta
                        a_test[dst] += delta
                        a_test = np.maximum(a_test, 0.0)
                        s = a_test.sum()
                        a_test = a_test / s
                        new_max = fast_convolve(a_test).max()
                        gain = best_obj - new_max
                        if gain > best_gain:
                            best_gain = gain
                            best_src = src
                            best_dst = dst
                            best_frac = frac

            if best_src >= 0 and best_gain > 1e-13:
                delta = a[best_src] * best_frac
                a[best_src] -= delta
                a[best_dst] += delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_obj = fast_convolve(a).max()
                best_a = a.copy()
                improved = True
                stall = 0

        if not improved:
            stall += 1

        if stall > 15:
            rng = np.random.default_rng((it * 48271 + 3) % (2**31))
            noise_scale = 0.0003 + 0.0005 * min(1.0, stall / 30.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            stall = 0

        if stall > 150:
            break

    return best_a, best_obj


def smoothed_gradient_refine(a0, time_limit, T_init=0.001, lr_init=0.02):
    """Smoothed gradient descent with softmax-weighted peaks and backtracking."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # grad of sum_k p[k] * (a*a)_k = 2 * H(p) @ a where H(p) is Hankel
        # (H(p) @ a)[j] = sum_k p[k] * a[k-j] (with bounds checking)
        grad = np.zeros(n)
        for k in range(len(p)):
            if p[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                partner = k - j
                if 0 <= partner < n:
                    grad[j] += 2.0 * p[k] * a[partner]

        # Backtracking line search
        best_candidate = a
        best_val = best_obj
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05, lr * 0.01]:
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        
        if a is not best_candidate:
            obj = best_val
        else:
            obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt parameters
        if no_improve % 20 == 0 and no_improve > 0:
            T *= 1.05
            lr *= 0.98

        if no_improve > 100:
            T = T_init
            lr = lr_init
            no_improve = 0

        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        scale = scales[si % len(scales)]
        rng = np.random.default_rng((it * 2654435761 + 1) % (2**32))

        tried = 0
        accepted = False
        while tried < 15 and time.monotonic() - t0 < time_limit - 0.1:
            tried += 1
            
            conv = fast_convolve(a)
            peak_k = int(np.argmax(conv))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)

            noise = rng.standard_normal(n) * scale
            # Structured bias: reduce values that contribute to the peak
            for j in range(js, je + 1):
                partner = peak_k - j
                if 0 <= partner < n:
                    noise[j] -= scale * 3.0 * a[partner]

            candidate = project_simplex(a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-16:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                accepted = True
                break

        if it % 200 == 0:
            si += 1

    return best_a, best_obj


def diversify_start(parent, n_target, rng):
    """Create a diverse starting point by interpolating/resampling parent."""
    n_parent = len(parent)
    if n_target == n_parent:
        return parent.copy()
    # Linear interpolation
    x_old = np.linspace(0, 1, n_parent)
    x_new = np.linspace(0, 1, n_target)
    start = np.interp(x_new, x_old, parent)
    start = np.maximum(start, 0)
    s = start.sum()
    if s > 1e-10:
        start = start / s
    else:
        start = np.ones(n_target) / n_target
    return start


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search:
    Phase 1: Multi-n exploration with FW-LP
    Phase 2: Deep FW-LP on best n
    Phase 3: Nesterov subgradient refinement
    Phase 4: Smoothed gradient
    Phase 5: Coordinate descent polish
    Phase 6: Micro-polish
    """
    rng = np.random.default_rng(seed)
    t0_total = time.monotonic()
    deadline = t0_total + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        # Clip to valid range
        a_norm = np.clip(a_norm, 0.0, 1000.0)
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t0_total
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, t={elapsed:.0f}s)")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384
        try_seq(parent, "uniform384")

    n_parent = len(parent)

    # === Phase 1: Multi-n exploration ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    t_phase = time.monotonic()

    candidate_ns = sorted(set([
        n_parent,
        max(128, n_parent - 128),
        min(2048, n_parent + 128),
        256, 320, 384, 448, 512, 576, 640, 768, 896, 1024, 1152
    ]))
    candidate_ns = [n for n in candidate_ns if 128 <= n <= 1200]

    for n_try in candidate_ns:
        if time.monotonic() > deadline - 1200:
            break

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = diversify_start(parent, n_try, rng)

        t_alloc = min(25, deadline - time.monotonic() - 1200)
        if t_alloc < 5:
            break
        print(f"  FW-LP n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc)
        try_seq(a_opt, f"fw_n{n_try}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP on best ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        t_alloc = min(350, deadline - time.monotonic() - 700)
        if t_alloc > 20:
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts
        for scale in [0.001, 0.003, 0.008, 0.015]:
            t_alloc = min(50, deadline - time.monotonic() - 600)
            if t_alloc < 10:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=t_alloc)
            try_seq(a_opt, f"fw_pers{scale}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Nesterov Subgradient ===
    print("\n=== Phase 3: Nesterov Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        configs = [
            {'lr_init': 0.003, 'max_k': 30},
            {'lr_init': 0.006, 'max_k': 50},
            {'lr_init': 0.001, 'max_k': 20},
            {'lr_init': 0.005, 'max_k': 70},
        ]
        for ci, cfg in enumerate(configs):
            t_alloc = min(100, deadline - time.monotonic() - 400)
            if t_alloc < 10:
                break
            print(f"  Nesterov lr={cfg['lr_init']}, k={cfg['max_k']}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_subgradient(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"nest_{ci}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Smoothed Gradient ===
    print("\n=== Phase 4: Smoothed Gradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        smooth_configs = [
            {'T_init': 0.0003, 'lr_init': 0.02},
            {'T_init': 0.001, 'lr_init': 0.015},
            {'T_init': 0.0001, 'lr_init': 0.01},
            {'T_init': 0.005, 'lr_init': 0.03},
        ]
        for ci, cfg in enumerate(smooth_configs):
            t_alloc = min(80, deadline - time.monotonic() - 250)
            if t_alloc < 10:
                break
            print(f"  Smooth T={cfg['T_init']}, lr={cfg['lr_init']}, t={t_alloc:.0f}s")
            a_opt, _ = smoothed_gradient_refine(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"smooth_{ci}")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Coordinate Descent ===
    print("\n=== Phase 5: Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        t_alloc = min(150, deadline - time.monotonic() - 100)
        if t_alloc > 10:
            print(f"  Coord descent, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final Polish ===
    print("\n=== Phase 6: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        # Micro polish
        t_alloc = min(60, deadline - time.monotonic() - 40)
        if t_alloc > 5:
            print(f"  Micro polish, t={t_alloc:.0f}s")
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "micro")

        # Another FW pass
        t_alloc = min(40, deadline - time.monotonic() - 25)
        if t_alloc > 5:
            print(f"  Final FW, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_fw")

        # Very fine smoothed gradient
        t_alloc = min(25, deadline - time.monotonic() - 12)
        if t_alloc > 5:
            a_opt, _ = smoothed_gradient_refine(best_seq, time_limit=t_alloc,
                                                 T_init=0.0001, lr_init=0.005)
            try_seq(a_opt, "final_smooth")

        # Final micro polish
        t_alloc = min(10, deadline - time.monotonic() - 5)
        if t_alloc > 2:
            a_opt, _ = micro_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_micro")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, total time={time.monotonic()-t0_total:.0f}s)")
    return [float(x) for x in best_seq]
