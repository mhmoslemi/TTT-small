"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed gradient descent, FW-LP, and targeted coordinate descent
with emphasis on asymmetric initializations and larger sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, a):
    return np.correlate(p, a, mode='valid')

def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.03, sparsity_every=500, sparsity_thresh=0.001):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        grad = 2.0 * hankel_mv(p, a)
        
        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate
        
        a = best_candidate
        obj = best_val
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            T *= 1.05
            lr *= 0.97
        if no_improve > 40:
            T *= 0.8
        if no_improve > 100:
            # Try sparsity
            candidate = a.copy()
            candidate[candidate < sparsity_thresh] = 0
            cs = candidate.sum()
            if cs > 0.1:
                candidate = candidate / cs
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-14:
                    best_obj = c.max()
                    best_a = candidate.copy()
                    a = candidate
                    no_improve = 0
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise_scale = 0.002 + 0.004 * min(1.0, (no_improve - 200) / 200.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            T = T_init
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

def solve_lp_oracle(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',

def generate_step_init(n, frac, direction=1):
    """Generate a step function: high on one side, low on the other."""
    if direction == 1:
        # High on left, low on right
        m = int(n * frac)
        a = np.zeros(n)
        a[:m] = 1.0
        a[m:] = 0.3  # Lower level
    else:
        # Low on left, high on right
        m = int(n * frac)
        a = np.zeros(n)
        a[:m] = 0.3
        a[m:] = 1.0
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def generate_ramp_init(n, direction=1):
    """Generate a linear ramp."""
    x = np.arange(n, dtype=np.float64)
    if direction == 1:
        a = (n - x) / n  # High on left
    else:
        a = (x + 1.0) / n  # High on right
    a = np.maximum(a, 0)
    return a / a.sum()

def generate_two_bump_init(n, gap_frac=0.2):
    """Two bumps with a gap in the middle."""
    x = np.arange(n, dtype=np.float64)
    center1 = n * (0.5 - gap_frac) / 2
    center2 = n * (0.5 + gap_frac) / 2
    width = n * 0.15
    a = np.exp(-0.5 * ((x - center1) / width) ** 2) + np.exp(-0.5 * ((x - center2) / width) ** 2)
    a = np.maximum(a, 0)
    return a / a.sum()

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed gradient descent with asymmetric initializations,
FW-LP for convergence, targeted peak reduction, and sparsity exploitation.
Explores larger sequence lengths for better theoretical lower bounds.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')


def smoothed_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.03,
                          sparsity_every=0, sparsity_thresh=0.0005,
                          restart_every=300, restart_scale=0.003):
    """Smoothed gradient descent with adaptive params and sparsity."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()
        grad = 2.0 * hankel_mv(p, a)

        best_candidate = a
        best_val = conv.max()
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            candidate = project_simplex(a - trial_lr * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            T *= 1.05
            lr *= 0.97
        if no_improve > 50:
            T *= 0.8
            lr *= 0.9

        # Sparsity exploitation
        if sparsity_every > 0 and it % sparsity_every == 0 and no_improve > 30:
            candidate = a.copy()
            candidate[candidate < sparsity_thresh] = 0.0
            cs = candidate.sum()
            if cs > 0.05:
                candidate = candidate / cs
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-14:
                    best_obj = c.max()
                    best_a = candidate.copy()
                    a = candidate
                    no_improve = 0

        # Restart from best with perturbation
        if restart_every > 0 and it % restart_every == 0 and no_improve > 50:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = rng.standard_normal(n) * restart_scale
            a = project_simplex(best_a + noise)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def solve_lp_oracle(f, M, time_limit_lp=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit_lp})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = solve_lp_oracle(f, M * m_exp, time_limit_lp=min(20, remaining - 1))
        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            if m_exp > 4.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.001 + 0.005 * min(1.0, no_improve / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            else:
                m_exp = min(m_exp * 1.05, 3.0)
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in np.linspace(0.01, 1.0, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if 0.01 < best_alpha < 0.99:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.005 * min(1.0, (no_improve - 50) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def targeted_peak_reduction(a0, time_limit):
    """Reduce the highest peak by targeted coordinate perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        if len(affected) == 0:
            break
        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]
        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        j_min_cand = outside[np.argmin(a[outside])]
        improved = False
        for delta in [0.0003, 0.001, 0.003, 0.005, 0.01]:
            if a[j_max] < delta:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min_cand] += delta
            candidate = np.maximum(candidate, 0.0)
            candidate = candidate / candidate.sum()
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break
        if improved:
            stale = 0
        else:
            stale += 1
            if stale % 20 == 0:
                rng = np.random.default_rng(it * 4567 + 3)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
                elif stale > 300:
                    break

    return best_a, best_obj


def multi_peak_subgrad(a0, time_limit, top_k=10, lr_init=0.002, mom_init=0.93):
    """Multi-peak subgradient descent with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    v = np.zeros(n)
    lr = lr_init
    mom = mom_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        conv = fast_convolve(a)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 2.5
        ws = weights.sum()
        if ws > 0:
            weights /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * sg
        a = project_simplex(a + v)
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        lr *= 0.9998
        mom = min(0.98, mom + 0.00005)

        if no_improve > 150:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31))
            noise = 0.001 + 0.003 * min(1.0, (no_improve - 150) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0

    return best_a, best_obj


def make_step_init(n, frac, low_ratio, direction):
    """Generate asymmetric step function."""
    m = max(1, int(n * frac))
    a = np.zeros(n)
    if direction == 1:
        a[:m] = 1.0
        a[m:] = low_ratio
    else:
        a[:m] = low_ratio
        a[m:] = 1.0
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        return a / s
    return np.ones(n) / n


def make_ramp_init(n, direction):
    """Generate linear ramp."""
    x = np.arange(n, dtype=np.float64)
    if direction == 1:
        a = (n - x) / n
    else:
        a = (x + 1.0) / n
    a = np.maximum(a, 0)
    return a / a.sum()


def make_bump_init(n, center_frac, width_frac):
    """Gaussian bump."""
    x = np.arange(n, dtype=np.float64)
    center = n * center_frac
    width = n * width_frac
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0)
    return a / a.sum()


def make_two_level_smooth(n, frac, transition_frac):
    """Smooth two-level: high on one side, low on other, with smooth transition."""
    x = np.arange(n, dtype=np.float64)
    m = n * frac
    w = n * transition_frac
    a = 1.0 - 0.5 * (1.0 + np.tanh((x - m) / (w / 3.0 + 1e-10)))
    a = a - 0.3 * a + 0.3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        return a / s
    return np.ones(n) / n


def evaluate_score(a):
    """Compute the score for a sequence (sum=1 normalization)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max())


def try_update(a, state, label):
    """Evaluate and update state if improved. state = [best_score, best_seq]."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.8f} (n={})".format(label, score, len(a)))
