"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Smoothed subgradient with adaptive temperature for global search
2. L-BFGS on softmax parametrization for fast local convergence
3. Frank-Wolfe with LP oracle for constraint-aware optimization
4. Peak-targeted coordinate descent for final polishing
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_oracle(f, M, n=None):
    """Solve LP: max sum(g) s.t. conv_matrix(f) @ g <= M, g >= 0."""
    if n is None:
        n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_smoothed_grad(a, conv, T):
    """Compute gradient of T*log(sum exp(c_k/T)) w.r.t. a."""
    n = len(a)
    # Compute p_k with numerical stability
    shifted = (conv - conv.max()) / T
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        # All underflowed; use top peak only
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = p / p_sum
    
    # Compute gradient: grad[j] = 2 * sum_k p[k] * a[k-j] for valid (k,j)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # a[k-j] for j in [js, je] = a[k-je], a[k-je+1], ..., a[k-js]
        # = a[k-je : k-js+1][::-1]
        lo = k - je
        hi = k - js + 1
        grad[js:je+1] += p[k] * a[lo:hi][::-1]
    return 2.0 * grad


def softmax_obj_and_grad(z, T):
    """Compute smoothed objective and gradient w.r.t. z (softmax parametrization)."""
    # Numerically stable softmax
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    
    # Smoothed objective (scale-invariant, so just track max of conv)
    # Actually, the score is 2*n*max(conv)/sum(a)^2 = 2*n*max(conv) since sum(a)=1
    # For optimization, we minimize T*log(sum exp(c_k/T))
    shifted = (conv - conv.max()) / T
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        obj = conv.max()
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        obj = conv.max() + T * np.log(p_sum)
        p = p / p_sum
    
    # Gradient w.r.t. a
    grad_a = compute_smoothed_grad(a, conv, T)
    
    # Gradient w.r.t. z: grad_z = a * (grad_a - a @ grad_a)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    
    return obj, grad_z


def lbfgs_optimize(a0, time_limit, T_initial=None, T_min=1e-6, max_iter=5000):
    """Optimize using L-BFGS-B with softmax parametrization."""
    a = np.maximum(a0, 1e-10)
    a = a / a.sum()
    n = len(a)
    
    if T_initial is None:
        conv_init = fast_convolve(a)
        T_initial = max(conv_init.max() * 0.05, 1e-6)
    
    # Initial z from log(a)
    z0 = np.log(a)
    
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    T = T_initial
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit - 5:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        # Run L-BFGS for this temperature
        time_per_run = min(remaining * 0.4, 30)
        
        def obj_grad(z):
            obj, grad = softmax_obj_and_grad(z, T)
            return obj, grad
        
        try:
            res = minimize(obj_grad, z0, method='L-BFGS-B',
                          jac=True, options={'maxiter': 200, 
                                              'maxfun': 500,
                                              'maxls': 20,
                                              'maxtime': time_per_run})
            z_opt = res.x
            a_opt = np.exp(z_opt - z_opt.max())
            a_opt = a_opt / a_opt.sum()
        except Exception:
            break
        
        score = fast_convolve(a_opt).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_opt.copy()
            z0 = z_opt
        else:

            z0 = z0 + rng.normal(0, 0.001, n)
        
        # Continue with smaller T
        T = max(T * 0.7, T_min)
    
    return best_a, fast_convolve(best_a).max()

    lr = 0.005
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    momentum = 0.9
    
    t0 = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit - 1:
        it = int((time.monotonic() - t0) / 0.001)
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        # Compute p
        shifted = (conv - conv.max()) / T
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = p / p_sum
        
        grad = compute_smoothed_grad(a, conv, T)
        
        # Momentum
        v = momentum * v - lr * grad
        
        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        cur_val = conv.max()
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < cur_val - 1e-14:
                best_candidate = candidate
                best_val = val
                break
        
        a = best_candidate
        obj = fast_convolve(a).max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Adapt
        if stagnant % 30 == 0:
            T *= 0.8
            lr *= 0.9
        if stagnant > 100:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_initial
            lr = 0.005
            v = np.zeros(n)
            stagnant = 0
            if T < 1e-8:
                break
    
    return best_a, fast_convolve(best_a).max()


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive M."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.001
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Solve LP
        g = solve_lp_oracle(f, M * m_relax, n)
        
        if g is None or g.sum() < 1e-15:
            g = solve_lp_oracle(f, M * 1.005, n)
            if g is None or g.sum() < 1e-15:
                stagnant += 1
                if stagnant > 15:
                    rng = np.random.default_rng(it * 7919 + 3)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                    conv = fast_convolve(f)
                    M = conv.max()
                    m_relax = 1.001
                    stagnant = 0
                continue
        
        g = g / g.sum()
        
        if np.max(np.abs(g - f)) < 1e-10:
            stagnant += 1
            if stagnant > 20:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue
        
        # Line search: golden section
        best_alpha = 0.0
        best_val = M + 1e-14
        
        # Coarse
        for alpha in np.linspace(0.01, 1.0, 20):
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine
        if best_alpha > 0.005:
            lo = max(0.001, best_alpha - 0.02)
            hi = min(0.999, best_alpha + 0.02)
            for alpha in np.linspace(lo, hi, 10):
                ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
                m_relax = max(0.999, m_relax * 0.995)
            else:
                stagnant += 1
        else:
            stagnant += 1
        
        if stagnant > 10:
            m_relax = min(m_relax * 1.02, 1.02)
        
        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.12f}, stagnant={stagnant}")
    
    return best_f, best_obj


def peak_coordinate_descent(a0, time_limit):
    """Coordinate descent targeting top peaks."""
    a = np.maximum(a0, 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    step_scale = 0.003
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            if js > je:
                continue
            
            # Find top contributing pairs
            contribs = []
            for j in range(js, je + 1):
                contribs.append((a[j] * a[k - j], j, k - j))
            contribs.sort(reverse=True)
            
            for (_, j1, j2) in contribs[:6]:
                if a[j1] < 1e-9 or a[j2] < 1e-9:
                    continue
                
                # Try reducing both j1 and j2 slightly, adding to other positions
                for src in [j1, j2]:
                    for delta_frac in [step_scale, step_scale * 0.5, step_scale * 2]:
                        delta = a[src] * delta_frac
                        if delta < 1e-12:
                            continue
                        
                        # Try a few target positions (spread out)
                        targets = [int(t * n) for t in [0.1, 0.3, 0.5, 0.7, 0.9]]
                        # Add random targets
                        rng = np.random.default_rng((it * 9973 + k * 31 + src * 7) % (2**31))
                        targets += [int(rng.integers(0, n)) for _ in range(3)]
                        
                        for j_target in targets:
                            if j_target == src:
                                continue
                            candidate = a.copy()
                            candidate[src] -= delta
                            if candidate[src] < -1e-12:
                                continue
                            candidate[j_target] += delta
                            candidate = np.maximum(candidate, 0)
                            candidate = candidate / candidate.sum()
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-14:
                                a = candidate
                                best_obj = val
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
            
            if improved:
                break
        
        if not improved:
            step_scale *= 0.92
            if step_scale < 1e-7:
                break
            if it % 100 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-14:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 5000:
                break
    
    return best_a, fast_convolve(best_a).max()


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Quick multi-n exploration with smoothed subgradient
    Phase 2: L-BFGS refinement with softmax parametrization
    Phase 3: Frank-Wolfe with LP oracle
    Phase 4: Peak-targeted coordinate descent
    Phase 5: Final micro-polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")
    
    # === Phase 1: Quick exploration (120s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()
    
    n_parent = len(parent)
    lengths = sorted(set([n_parent, max(256, n_parent - 128), n_parent + 128, 
                          512, 640, 768, 896, 1024, 1280]))
    lengths = [l for l in lengths if 256 <= l <= 2048]
    
    for n_try in lengths:
        if time.monotonic() > deadline - 900:
            break
        
        # Different initializations
        starts = []
        # Parent resampled
        starts.append(resample_sequence(parent, n_try))
        # Uniform
        starts.append(np.ones(n_try) / n_try)
        # Cosine
        x = np.arange(n_try, dtype=np.float64)
        w = 0.5 * (1 + np.cos(np.pi * x / (n_try - 1)))
        starts.append(w / w.sum())
        # Random
        starts.append(rng.dirichlet(np.ones(n_try) * 0.5))
        
        for si, start in enumerate(starts):
            if time.monotonic() > deadline - 850:
                break
            t_alloc = min(15, deadline - time.monotonic() - 800)
            if t_alloc < 3:
                break
            try:
                a_opt, _ = smoothed_subgradient_optimize(start, time_limit=t_alloc)
                try_seq(a_opt, f"sg_n{n_try}_s{si}")
            except Exception as e:
                pass
    
    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 2: L-BFGS refinement (350s) ===
    print("\n=== Phase 2: L-BFGS Refinement ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Multiple L-BFGS runs with different temperatures
        T_inits = [0.01, 0.005, 0.002, 0.001, 0.0005]
        for Ti in T_inits:
            if time.monotonic() > deadline - 400:
                break
            t_alloc = min(60, deadline - time.monotonic() - 350)
            if t_alloc < 10:
                break
            print(f"  L-BFGS T={Ti}, t={t_alloc:.0f}s")
            try:
                a_opt, _ = lbfgs_optimize(best_seq, time_limit=t_alloc, T_initial=Ti)
                try_seq(a_opt, f"lbfgs_T{Ti}")
            except Exception as e:
                print(f"  L-BFGS error: {e}")
        
        # Also try at different n values
        for n_try in [n_best, n_best + 64, n_best - 64, 1024]:
            if n_try <= 0 or time.monotonic() > deadline - 350:
                continue
            t_alloc = min(40, deadline - time.monotonic() - 300)
            if t_alloc < 10:
                break
            start = resample_sequence(best_seq, n_try)
            print(f"  L-BFGS n={n_try}, t={t_alloc:.0f}s")
            try:
                a_opt, _ = lbfgs_optimize(start, time_limit=t_alloc, T_initial=0.01)
                try_seq(a_opt, f"lbfgs_n{n_try}")
            except Exception:
                pass
    
    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 3: Frank-Wolfe with LP (350s) ===
    print("\n=== Phase 3: Frank-Wolfe ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Main FW
        t_alloc = min(200, max(30, int((deadline - time.monotonic() - 300) * 0.5)))
        print(f"  FW n={n_best}, t={t_alloc}s")
        try:
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "fw_main")
        except Exception as e:
            print(f"  FW error: {e}")
        
        # Perturbed restarts
        for scale in [0.002, 0.005, 0.01]:
            if time.monotonic() > deadline - 300:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(50, max(10, int((deadline - time.monotonic() - 250) * 0.3)))
            if t_alloc < 5:
                break
            print(f"  FW perturb scale={scale}, t={t_alloc}s")
            try:
                a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc)
                try_seq(a_opt, f"fw_pert_{scale}")
            except Exception:
                pass
    
    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 4: Peak coordinate descent (200s) ===
    print("\n=== Phase 4: Peak Coordinate Descent ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        t_alloc = min(200, max(20, int(deadline - time.monotonic() - 100)))
        print(f"  Peak CD, t={t_alloc}s")
        try:
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_cd")
        except Exception as e:
            print(f"  Peak CD error: {e}")
    
    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")
    
    # === Phase 5: Final micro-polish ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Alternate short runs
        for ri in range(10):
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            
            if ri % 3 == 0:
                noise = rng.standard_normal(n_best) * 0.0005
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining * 0.3, 20)
                if t_alloc < 3:
                    break
                try:
                    a_opt, _ = smoothed_subgradient_optimize(start, time_limit=t_alloc)
                    try_seq(a_opt, f"polish_sg_{ri}")
                except Exception:
                    pass
            elif ri
