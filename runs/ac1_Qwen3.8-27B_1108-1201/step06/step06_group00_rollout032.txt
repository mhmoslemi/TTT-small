"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parametric diverse initialization (smooth functions, U/V shapes)
2. Linearized QP oracle (second-order Frank-Wolfe on the simplex)
3. Adaptive peak peeling (gradually tighten constraints)
4. Coordinate descent polish on active peaks
5. Spectral diversification for escaping local minima

import numpy as np
from scipy.sparse import csr_matrix
import time
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_row(f, k):
    """Build the row of the convolution matrix for position k.
    Returns (cols, vals) such that sum vals[i] * d[cols[i]] = (f*d)[k]
    """
    n = len(f)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    j_range = np.arange(lo, hi + 1)
    return j_range, f[k - j_range]

def solve_qp_step(f, M_target, peak_indices, max_iters=5000):
    """Solve the linearized QP:
    min (1/2)||d||^2
    s.t. (f*d)[k] <= (M_target - (f*f)[k])/2 for k in peak_indices
         f + d >= 0
         sum(d) = 0
    """
    n = len(f)
    c = fast_convolve(f)
    K = len(peak_indices)
    
    # Build constraint matrix
    rows = []
    cols = []
    vals = []
    rhs = np.zeros(K)
    
    for ci, k in enumerate(peak_indices):
        k = int(k)
        j_range, f_vals = build_conv_row(f, k)
        rows.extend([ci] * len(j_range))
        cols.extend(j_range.tolist())
        vals.extend(f_vals.tolist())
        rhs[ci] = (M_target - c[k]) / 2.0
    
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(K, n))
    
    # QP: min (1/2)||d||^2 s.t. A_sparse @ d <= rhs, d >= -f, sum(d) = 0
    try:
        d_var = cp.Variable(n)
        constraints = [
            A_sparse @ d_var <= rhs,
            d_var >= -f,
            cp.sum(d_var) == 0
        ]
        objective = cp.Minimize(0.5 * cp.sum_squares(d_var))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-10, reltol=1e-10)
        
        if d_var.value is not None:
            return np.array(d_var.value).flatten()
    except Exception:
        pass
    
    # Fallback: use SCS
    try:
        d_var = cp.Variable(n)
        constraints = [
            A_sparse @ d_var <= rhs,
            d_var >= -f,
            cp.sum(d_var) == 0
        ]
        objective = cp.Minimize(0.5 * cp.sum_squares(d_var))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-8)
        
        if d_var.value is not None:
            return np.array(d_var.value).flatten()
    except Exception:
        pass
    
    return np.zeros(n)

def qp_refine(f0, time_limit, start_K=5, end_K=50, verbose=False):
    """Iteratively refine using linearized QP with adaptive peak selection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_M = M
    best_f = f.copy()
    t0 = time.monotonic()
    
    it = 0
    cur_K = start_K
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            if verbose:
                print(f"    it={it}, M={M:.10f}, K={cur_K}")
        
        # Select top K peaks
        k_top = min(cur_K, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        
        # Target: slightly below current max
        M_target = M * (1 - 0.01)  # try to reduce by 1%
        
        d = solve_qp_step(f, M_target, peak_idx)
        
        if np.max(np.abs(d)) < 1e-14:
            # Try with a more aggressive target
            M_target = M * (1 - 0.05)
            d = solve_qp_step(f, M_target, peak_idx)
            if np.max(np.abs(d)) < 1e-14:
                # Stuck, increase K
                cur_K = min(cur_K + 5, end_K)
                if cur_K > end_K:
                    # Try random restart
                    rng = np.random.default_rng(it * 1009 + 7)
                    f = project_simplex(f + rng.standard_normal(n) * 0.005)
                    conv = fast_convolve(f)
                    M = conv.max()
                    cur_K = start_K
                continue
        
        # Line search
        f_new = f + d
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            continue
        f_new = f_new / s_new
        
        # Check if it improved
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        
        if M_new < M:
            f = f_new
        else:
            # Try smaller step
            for alpha in [0.5, 0.25, 0.1]:
                f_trial = f + alpha * d
                f_trial = np.maximum(f_trial, 0.0)
                s_t = f_trial.sum()
                if s_t < 1e-15:
                    continue
                f_trial = f_trial / s_t
                M_t = fast_convolve(f_trial).max()
                if M_t < M:
                    f = f_trial
                    break
            # If no improvement, increase K
            cur_K = min(cur_K + 3, end_K)
    
    return best_f, best_M

def coordinate_descent(f0, time_limit, n_passes=100):
    """Coordinate descent: try to reduce the max peak by adjusting one coordinate at a time."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_M = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    # Focus on the top peaks
    top_peaks = np.argpartition(conv, -10)[-10:]
    
    for _ in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
        
        # Get current top peaks
        top_peaks = np.argpartition(conv, -10)[-10:]
        peak_k = int(top_peaks[0])  # The worst peak
        
        # For peak at k, c[k] = sum_i f[i]*f[k-i]
        # To reduce c[k], we can decrease f[i] for i that contributes most
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        
        # Compute contributions
        i_range = np.arange(lo, hi + 1)
        contributions = f[i_range] * f[peak_k - i_range]
        
        # Try reducing the coordinate with the largest contribution
        # and moving mass to a "safe" location
        j_max = i_range[np.argmax(contributions)]
        
        # Try moving a small amount from j_max to each other position
        delta = f[j_max] * 0.05
        if delta < 1e-15:
            continue
        
        best_delta_f = None
        best_M_trial = M
        
        # Try a few candidate destinations
        candidates = np.random.choice(n, size=min(50, n), replace=False)
        for j_dest in candidates:
            if j_dest == j_max:
                continue
            f_trial = f.copy()
            f_trial[j_max] -= delta
            f_trial[j_dest] += delta
            if f_trial[j_max] < 0:
                continue
            f_trial = np.maximum(f_trial, 0)
            f_trial /= f_trial.sum()
            M_trial = fast_convolve(f_trial).max()
            if M_trial < best_M_trial:
                best_M_trial = M_trial
                best_delta_f = f_trial.copy()
        
        if best_delta_f is not None:
            f = best_delta_f
        else:
            # Try the reverse: increase j_max and decrease another
            # (sometimes increasing a coordinate reduces the peak by
            #  shifting the mass distribution)
            # Or just try random small perturbations
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            f_trial = f + rng.standard_normal(n) * 0.001
            f_trial = project_simplex(f_trial)
            M_trial = fast_convolve(f_trial).max()
            if M_trial < M:
                f = f_trial
    
    return best_f, best_M

def spectral_refine(f0, time_limit):
    """Refine by working in the frequency domain to flatten the power spectrum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    # Ensure FFT size
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    
    # Get current spectrum
    F = np.fft.rfft(f, fft_size)
    
    # The power spectrum is |F|^2. We want it flat.
    # Current autocorrelation = IFFT(|F|^2)[:2n-1]
    # We want to modify F to make |F|^2 flatter, while keeping IFFT real, non-negative, sum-preserving.
    
    for _ in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
        
        F = np.fft.rfft(f, fft_size)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Compute target: flatten the spectrum
        # The DC component must stay at 1 (sum = 1)
        # Target: |F_target|^2 = 1/fft_size * sum(|F|^2) = const
        # = sum(f^2) / fft_size (by Parseval)
        
        target_power = np.sum(f**2) / fft_size
        target_mag = np.sqrt(target_power) * np.ones(len(mag))
        target_mag[0] = 1.0  # DC must be sum(f) = 1
        
        # Blend current and target
        alpha = 0.1
        mag_new = (1 - alpha) * mag + alpha * target_mag
        mag_new[0] = 1.0
        
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s > 1e-15:
            f_new = f_new / s
        
        conv_new = fast_convolve(f_new)
        if conv_new.max() < M:
            f = f_new
        else:
            # Try smaller alpha
            alpha = 0.03
            mag_new = (1 - alpha) * mag + alpha * target_mag
            mag_new[0] = 1.0
            F_new = mag_new * np.exp(1j * phase)
            f_new = np.fft.irfft(F_new, fft_size)[:n]
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s > 1e-15:
                f_new = f_new / s
            conv_new = fast_convolve(f_new)
            if conv_new.max() < M:
                f = f_new
    
    return best_f, best_M

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

def make_parametric(n, shape='uniform', seed=42):
    """Create a parametric initial sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    
    if shape == 'uniform':
        f = np.ones(n)
    elif shape == 'gaussian':
        f = np.exp(-8 * (x - 0.5)**2)
    elif shape == 'u_shape':
        f = 1 + 0.5 * (2*x - 1)**2
    elif shape == 'v_shape':
        f = 0.5 + np.abs(2*x - 1)
    elif shape == 'sin':
        f = 1 + 0.3 * np.sin(4 * np.pi * x)
    elif shape == 'cos':
        f = 1 + 0.3 * np.cos(4 * np.pi * x)
    elif shape == 'random_smooth':
        freqs = rng.uniform(0, 5, size=5)
        phases = rng.uniform(0, 2*np.pi, size=5)
        amps = rng.uniform(0.1, 0.3, size=5)
        f = 1.0
        for freq, phase, amp in zip(freqs, phases, amps):
            f += amp * np.sin(2*np.pi*freq*x + phase)
        f = np.maximum(f, 0.1)
    elif shape == 'random':
        f = rng.random(n) + 0.1
    else:
        f = np.ones(n)
    
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 0:
        f = f / s
    return f

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_weights(conv, beta):
    """Compute softmax weights for LSE approximation of max."""
    bc = beta * (conv - conv.max())
    exp_bc = np.exp(bc)

    if total < 1e-300:

    return exp_bc / total

                best_val = v_t
                best_alpha = alpha
        
        if best_alpha > 1e-8:
            f = f + best_alpha * d
            f = np.maximum(f, 0.0)
            f = f / f.sum()
        else:
            no_improve += 2
            cur_K = min(cur_K + 3, end_K)
    
    return best_f, best_M


def lse_sd_refine(f0, time_limit, beta_start=20.0, beta_end=5000.0, top_k=40):
    """LSE-smoothed subgradient descent with adaptive step size."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    best_M = fast_convolve(f).max()
    best_f = f.copy()
    beta = beta_start
    lr = 4.0 / np.sqrt(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        grad, M = lse_subgradient(f, beta, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Project gradient to tangent space of simplex
        grad_tan = grad - np.mean(grad)
        
        step = lr
        improved = False
        for _ in range(12):
            f_new = project_simplex(f - step * grad_tan / gnorm)
            M_new = fast_convolve(f_new).max()
            if M_new < best_M - 1e-15:
                best_M = M_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                no_improve = 0
                lr = min(lr * 1.01, 20.0 / np.sqrt(n))
                break
            step *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.85
            if no_improve > 60:
                f = project_simplex(best_f + np.random.default_rng(it * 31).standard_normal(n) * 0.002)
                lr = max(4.0 / np.sqrt(n), 0.01)
                no_improve = 0
        
        beta = min(beta * 1.006, beta_end)
    
    return best_f, best_M


def spectral_flatten(f0, time_limit):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_M = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
        
        F = np.fft.rfft(f, fft_size)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: make |F|^2 closer to uniform (but keep DC fixed)
        target_power = np.sum(f ** 2) / fft_size
        target_mag = np.sqrt(target_power) * np.ones(len(mag))
        target_mag[0] = 1.0
        
        alpha = 0.05 + 0.02 * (it % 5)
        mag_new = (1.0 - alpha) * mag + alpha * target_mag
        mag_new[0] = 1.0
        
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s > 1e-15:
            f_new = f_new / s
        
        M_new = fast_convolve(f_new).max()
        if M_new < M:
            f = f_new
    
    return best_f, best_M


def coordinate_polish(f0, time_limit, n_rounds=300):
    """Coordinate descent: adjust individual coordinates to reduce peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    best_M = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for _ in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(f)
        M = c.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
        
        # Find worst peak
        k_peak = int(np.argmax(c))
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        i_range = np.arange(lo, hi + 1)
        
        # Find coordinate with largest contribution
        contribs = f[i_range] * f[k_peak - i_range]
        j_max = i_range[np.argmax(contribs)]
        
        if f[j_max] < 1e-10:
            continue
        
        # Try moving a fraction of f[j_max] to random locations
        delta = f[j_max] * 0.03
        best_trial = None
        best_M_trial = M
        
        trials = rng.choice(n, size=min(80, n), replace=False)
        for j_dest in trials:
            if j_dest == j_max:
                continue
            f_t = f.copy()
            f_t[j_max] -= delta
            f_t[j_dest] += delta
            if f_t[j_max] < -1e-15:
                continue
            f_t = np.maximum(f_t, 0)
            f_t /= f_t.sum()
            M_t = fast_convolve(f_t).max()
            if M_t < best_M_trial:
                best_M_trial = M_t
                best_trial = f_t.copy()
        
        if best_trial is not None:
            f = best_trial
        else:
            # Try tiny random perturbation
            f_t = project_simplex(f + rng.standard_normal(n) * 0.001)
            if fast_convolve(f_t).max() < M:
                f = f_t
    
    return best_f, best_M


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def make_parametric(n, shape, seed=42):
    """Create a parametric initial sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    
    if shape == 'uniform':
        f = np.ones(n)
    elif shape == 'u_shape':
        f = 1.0 + 0.7 * (2.0 * x - 1.0) ** 2
    elif shape == 'v_shape':
        f = 0.6 + 0.8 * np.abs(2.0 * x - 1.0)
    elif shape == 'sin2':
        f = 1.0 + 0.4 * np.sin(4.0 * np.pi * x)
    elif shape == 'cos2':
        f = 1.0 + 0.4 * np.cos(4.0 * np.pi * x)
    elif shape == 'gaussian':
        f = np.exp(-12.0 * (x - 0.5) ** 2) + 0.1
    elif shape == 'random_smooth':
        freqs = rng.uniform(0.5, 6.0, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amps = rng.uniform(0.15, 0.35, size=4)
        f = 1.0
        for fr, ph, am in zip(freqs, phases, amps):
            f += am * np.sin(2 * np.pi * fr * x + ph)
        f = np.maximum(f, 0.05)
    else:
        f = np.ones(n)
    
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    return f
