import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Precompute sparse matrix for LP constraints: A[k,i] = 1 if i <= k < i+n, value = f[k-i]."""
    # We need A[k, i] = f[k-i] for valid (k,i) where 0 <= k < 2n-1 and 0 <= i < n and 0 <= k-i < n
    # So for each k, valid i range is [max(0, k-n+1), min(n-1, k)]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, n, row_idx=None, col_idx=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    if row_idx is None or col_idx is None:
        row_idx, col_idx = build_lp_matrix(n)
    
    c = -np.ones(n)
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]
    
    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def spectral_smooth(a, sigma):
    """Apply Gaussian smoothing in the frequency domain to the sequence."""
    n = len(a)
    A = np.fft.rfft(a)
    freqs = np.fft.rfftfreq(n, d=1.0/(n-1) if n > 1 else 1.0)
    kernel = np.exp(-0.5 * (freqs * sigma) ** 2)
    return np.fft.irfft(A * kernel, n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Improved SCA (Successive Convex Approximation) with LP oracle
- Tilt perturbations exploiting optimal asymmetry
- Adam gradient-based polish
- Multi-n exploration with diversified restarts
- 1D tilt parameter search for local refinement

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, n, row_idx=None, col_idx=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    if row_idx is None or col_idx is None:
        row_idx, col_idx = build_lp_matrix(n)

    c = -np.ones(n)
    k_arr = row_idx
    i_arr = col_idx
    vals = f[k_arr - i_arr]

    A_sparse = csr_matrix((vals, (k_arr, i_arr)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def scalar_tilt_search(f, t_range=(-3.0, 3.0), n_points=50):
    """1D search over tilt parameter to find optimal asymmetry."""
    n = len(f)
    x = np.arange(n, dtype=np.float64)
    tilt_pos = 2.0 * x / n - 1.0

    best_f = f.copy()
    best_val = fast_convolve(f).max()

    for t in np.linspace(t_range[0], t_range[1], n_points):
        if abs(t) < 1e-10:
            continue
        candidate = f * np.exp(t * tilt_pos)
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 1e-15:
            continue
        candidate = candidate / s
        val = fast_convolve(candidate).max()
        if val < best_val:
            best_val = val
            best_f = candidate.copy()

    return best_f, best_val


def sca_optimize(f0, time_limit, row_idx=None, col_idx=None, aggressive=False):

    SCA optimization with adaptive M and tilt-based diversification.
    """

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if row_idx is None or col_idx is None:
        row_idx, col_idx = build_lp_matrix(n)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_reduce = 0.993 if aggressive else 0.996

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.3:
            break

        g_raw = solve_lp_oracle(f, M, n, row_idx, col_idx)

        if g_raw is None:
            M *= 1.03
            no_improve += 1
            if no_improve > 40:
                M = fast_convolve(f).max() * 1.005
                no_improve = 0
            continue

        g = g_raw / g_raw.sum()

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 20:
                _diversify(f, best_f, n, it, noise_scale=0.004)

                M = fast_convolve(f).max() * 1.005

        best_val = fast_convolve(f).max()
        best_ft = f.copy()

        for alpha in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:

            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:

            best_obj = best_val
            best_f = f.copy()
            M = best_val * m_reduce
            no_improve = 0
        else:
            no_improve += 1
            M *= 0.997

            f, _, M = _do_tilt(f, n)
            M = fast_convolve(f).max() * 1.005
            no_improve = 0

    return best_f, best_obj


def _diversify(f, best_f, n, it, noise_scale=0.004):
    """In-place diversification of f using various strategies."""
    rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
    strategy = rng.integers(4)
    x = np.arange(n, dtype=np.float64)

    if strategy == 0:
        # Gaussian
        f[:] = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
    elif strategy == 1:
        # Tilt
        tilt = np.exp(rng.uniform(0.5, 2.5) * (2.0 * x / n - 1.0))
        f[:] = project_simplex(best_f * tilt)
    elif strategy == 2:
        # Peak-directed
        conv = fast_convolve(best_f)
        k_peak = int(np.argmax(conv))
        delta = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        for j in range(js, je + 1):
            if best_f[j] > 1e-12:
                delta[j] -= 0.03 * best_f[j] * best_f[k_peak - j] / best_f.sum()
        outside = np.setdiff1d(np.arange(n), np.arange(js, je + 1))
        if len(outside) > 0:
            total = -delta.sum()
            delta[outside] += total / len(outside)
        f[:] = project_simplex(best_f + delta)
    else:
        # Frequency perturbation
        F = np.fft.rfft(best_f)
        F += rng.standard_normal(len(F)) * 0.005 * np.abs(F).mean()
        f[:] = np.fft.irfft(F, n)
        f[:] = np.maximum(f, 0)
        s = f.sum()
        if s > 1e-15:
            f[:] = f / s
        else:
            f[:] = best_f


def _do_tilt(f, n, t_range=(-2.0, 2.0)):
    """Apply optimal tilt to f, return (f, best_t, M)."""
    x = np.arange(n, dtype=np.float64)
    tilt_pos = 2.0 * x / n - 1.0
    best_f = f.copy()
    best_val = fast_convolve(f).max()

    for t in np.linspace(t_range[0], t_range[1], 21):
        if abs(t) < 0.05:
            continue
        candidate = f * np.exp(t * tilt_pos)
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 1e-15:
            continue
        candidate = candidate / s
        val = fast_convolve(candidate).max()
        if val < best_val:
            best_val = val
            best_f = candidate.copy()

    return best_f, 0.0, best_val


def gradient_polish(f0, time_limit):
    """Adam gradient descent on the max-peak objective with correct subgradients."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    m_t = np.zeros(n)
    v_t = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = 0.02
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    top_k = min(15, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Weighted subgradient of max-peak
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        weights = pv ** 3 / (pv ** 3).sum()

        grad = np.zeros(n)
        for idx in range(top_k):
            w = weights[idx]
            if w < 1e-10:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += 2.0 * w * f[k - j]

        # Adam update
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * grad * grad
        m_hat = m_t / (1.0 - beta1 ** it)
        v_hat = v_t / (1.0 - beta2 ** it)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        new_obj = fast_convolve(f_new).max()

        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if no_improve > 30:
            lr *= 0.85
        if lr < 5e-5:
            lr = 0.02
            no_improve = 0

        if no_improve > 120:
            x = np.arange(n, dtype=np.float64)
            rng = np.random.default_rng((it * 204829 + 7) % (2**31 - 1))
            tilt = np.exp(rng.uniform(0.5, 2.0) * (2.0 * x / n - 1.0))
            f = project_simplex(best_f * tilt + rng.standard_normal(n) * 0.003)
            m_t = np.zeros(n)
            v_t = np.zeros(n)
            lr = 0.015
            no_improve = 0

    return best_f, best_obj


def multi_start_sca(f0, time_limit, n_starts=5):
    """
    Run SCA from multiple perturbed starting points and keep the best.
    """
    n = len(f0)
    per_start_time = time_limit / n_starts
    best_f = f0.copy()
    best_obj = fast_convolve(f0).max()

    for i in range(n_starts):
        remaining = time_limit - (time.monotonic() - t0_ms)
        if remaining < 3:
            break
        t_alloc = min(per_start_time, remaining - 1)
        if t_alloc < 2:
            break

        if i == 0:
            start = f0.copy()
        else:
            x = np.arange(n, dtype=np.float64)
            rng = np.random.default_rng((i * 31 + 7) % (2**31 - 1))
            if i % 3 == 1:
                tilt = np.exp(rng.uniform(0.5, 2.5) * (2.0 * x / n - 1.0))
                start = project_simplex(best_f * tilt)
            elif i % 3 == 2:
                start = project_simplex(best_f + rng.standard_normal(n) * 0.006)
            else:
                F = np.fft.rfft(best_f)
                F += rng.standard_normal(len(F)) * 0.008 * np.abs(F).mean()
                start = np.fft.irfft(F, n)
                start = np.maximum(start, 0)
                start = start / start.sum()

        a_opt, obj = sca_optimize(start, time_limit=t_alloc, aggressive=(i > 0))
        if obj < best_obj:
            best_obj = obj
            best_f = a_opt.copy()

    return best_f, best_obj


t0_ms = None  # global for multi_start_sca timing


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: improved SCA with tilt exploitation, gradient polish,
    multi-n exploration, and diversified restarts.
    """
    global t0_ms
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(768) / 768

    # === Phase 1: Quick tilt search on parent ===
    print("\n=== Phase 1: Tilt search ===")
    t1 = time.monotonic()
    tilted, tilt_val = scalar_tilt_search(parent, t_range=(-3.0, 3.0), n_points=60)
    try_seq(tilted, "tilt_search")
    print(f"  Tilt search: {tilt_val:.10f}, t={time.monotonic()-t1:.1f}s")

    # === Phase 2: SCA on best (with parent n) ===
    print("\n=== Phase 2: SCA Phase A ===")
    t2 = time.monotonic()
    n_cur = len(best_seq) if best_seq is not None else len(parent)
    remaining = deadline - time.monotonic() - 500
    if remaining > 30:
        t_alloc = min(remaining, 200)
        print(f"  SCA n={n_cur}, t={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(best_seq if best_seq is not None else parent,
                                time_limit=t_alloc, aggressive=True)
        try_seq(a_opt, "sca_A")

    # Tilt + SCA again
    remaining = deadline - time.monotonic() - 400
    if remaining > 20:
        tilted2, _ = scalar_tilt_search(a_opt, t_range=(-2.0, 2.0), n_points=30)
        try_seq(tilted2, "tilt2")
        t_alloc = min(remaining - 5, 60)
        if t_alloc > 5:
            a_opt2, _ = sca_optimize(tilted2, time_limit=t_alloc, aggressive=True)
            try_seq(a_opt2, "sca_B")
    print(f"  P2 done: best={best_score:.8f}, t={time.monotonic()-t2:.0f}s")

    # === Phase 3: Multi-start SCA ===
    print("\n=== Phase 3: Multi-start SCA ===")
    t3 = time.monotonic()
    remaining = deadline - time.monotonic() - 350
    if remaining > 30:
        t_alloc = min(remaining, 150)
        t0_ms = time.monotonic()
        a_opt, _ = multi_start_sca(best_seq, time_limit=t_alloc, n_starts=5)
        try_seq(a_opt, "multi_start")
    print(f"  P3 done: best={best_score:.8f}, t={time.monotonic()-t3:.0f}s")

    # === Phase 4: Gradient polish ===
    print("\n=== Phase 4: Gradient polish ===")
    t4 = time.monotonic()
    remaining = deadline - time.monotonic() - 250
    if remaining > 15:
        t_alloc = min(remaining, 100)
        a_opt, _ = gradient_polish(best_seq, time_limit=t_alloc)
        try_seq(a_opt, "grad")

    # SCA after gradient
    remaining = deadline - time.monotonic() - 150
    if remaining > 15:
        t_alloc = min(remaining, 80)
        a_opt2, _ = sca_optimize(a_opt, time_limit=t_alloc, aggressive=True)
        try_seq(a_opt2, "sca_post_grad")

    # Tilt again
    remaining = deadline - time.monotonic() - 100
    if remaining > 10:
        tilted3, _ = scalar_tilt_search(a_opt2, t_range=(-1.5, 1.5), n_points=30)
        try_seq(tilted3, "tilt3")
    print(f"  P4 done: best={best_score:.8f}, t={time.monotonic()-t4:.0f}s")

    # === Phase 5: Explore other n values ===
    print("\n=== Phase 5: Multi-n ===")
    t5 = time.monotonic()
    n_best = len(best_seq)

    for n_try in [512, 640, 896, 1024, 1280, 1536]:
        if n_try == n_best:
            continue
        remaining = deadline - time.monotonic() - 100
        if remaining < 15:
            break
        start = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_best), best_seq)
        start = np.maximum(start, 0)
        start = start / start.sum()

        t_alloc = min(15, remaining - 30)
        if t_alloc < 5:
            break
        print(f"  Try n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(start, time_limit=t_alloc, aggressive=True)
        try_seq(a_opt, f"n{n_try}")
    print(f"  P5 done: best={best_score:.8f}, t={time.monotonic()-t5:.0f}s")

    # === Phase 6: Deep SCA on best ===
    print("\n=== Phase 6: Deep SCA ===")
    t6 = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 80
    if remaining > 20:
        t_alloc = min(remaining, 150)
        a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc, aggressive=True)
        try_seq(a_opt, "deep_sca")

    # Final tilt
    remaining = deadline - time.monotonic() - 40
    if remaining > 5:
        tilted4, _ = scalar_tilt_search(a_opt, t_range=(-1.0, 1.0), n_points=25)
        try_seq(tilted4, "final_tilt")
    print(f"  P6 done: best={best_score:.8f}, t={time.monotonic()-t6:.0f}s")

    # === Phase 7: Final diversified restarts ===
    print("\n=== Phase 7: Final restarts ===")
    t7 = time.monotonic()
    n_best = len(best_seq)
    x = np.arange(n_best, dtype=np.float64)

    for i in range(6):
        remaining = deadline - time.monotonic() - 20
        if remaining < 8:
            break
        t_alloc = min(15, remaining - 10)
        if t_alloc < 4:
            break

        rng_i = np.random.default_rng((1000 + i * 7) % (2**31 - 1))
        if i % 3 == 0:
            perturbed = project_simplex(best_seq + rng_i.standard_normal(n_best) * 0.003)
        elif i % 3 == 1:
            tilt = np.exp(rng_i.uniform(0.3, 2.0) * (2.0 * x / n_best - 1.0))
            perturbed = project_simplex(best_seq * tilt)
        else:
            F = np.fft.rfft(best_seq)
            F += rng_i.standard_normal(len(F)) * 0.004 * np.abs(F).mean()
            perturbed = np.fft.irfft(F, n_best)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

        print(f"  Restart {i}, t={t_alloc:.0f}s")
        a_opt, _ = sca_optimize(perturbed, time_limit=t_alloc, aggressive=True)
        # Try tilt on result
        a_tilted, _ = scalar_tilt_search(a_opt, t_range=(-1.0, 1.0), n_points=15)
        try_seq(a_tilted, f"restart_{i}")

    print(f"  P7 done: best={best_score:.8f}, t={time.monotonic()-t7:.0f}s")

    # === Phase 8: Ultimate final polish ===
    print("\n=== Phase 8: Final polish ===")
    t8 = time.monotonic()
    remaining = deadline - time.monotonic() - 5
    if remaining > 5:
        t_alloc = min(remaining, 25)
        a_opt, _ = sca_optimize(best_seq, time_limit=t_alloc, aggressive=True)
        try_seq(a_opt, "ultimate_sca")

    remaining = deadline - time.monotonic() - 3
    if remaining > 3:
        a_opt, _ = gradient_polish(best_seq, time_limit=min(remaining, 10))
        try_seq(a_opt, "ultimate_grad")
    print(f"  P8 done: best={best_score:.8f}, t={time.monotonic()-t8:.0f}s")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
