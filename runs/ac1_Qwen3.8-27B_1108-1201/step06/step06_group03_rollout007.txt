"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Softmax parameterization to remove non-negativity constraint
- L-BFGS with temperature annealing on smoothed log-sum-exp objective
- Multi-start with diverse initializations
- Peak-targeted local refinement
- Spectral (frequency-domain) initialization

import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_a(a, w):
    """
    Compute grad_a[i] = 2 * sum_{k} w[k] * a[k-i] for valid indices.
    w has length 2n-1, a has length n.
    """
    n = len(a)
    N = 2 * n - 1
    a_pad = np.zeros(N)
    a_pad[:n] = a
    fft_size = 1 << (max(N - 1, 1).bit_length())
    W = np.fft.rfft(w, fft_size)
    Ap = np.fft.rfft(a_pad, fft_size)
    cross_cor = np.fft.irfft(W * np.conj(Ap), fft_size)[:N]
    return 2.0 * cross_cor[:n]

def smoothed_obj_grad(x, T):
    """
    Compute the smoothed objective T*log(sum exp(c[k]/T)) and its gradient
    in the softmax parameterization.
    """
    a = softmax(x)
    n = len(a)
    c = fast_convolve(a)
    
    c_max = c.max()
    scaled = (c - c_max) / T
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    s = exp_scaled.sum()
    val = c_max + T * (max_scaled + np.log(s))
    
    w = exp_scaled / s
    grad_a = compute_grad_a(a, w)
    grad_a_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - grad_a_dot_a)
    
    return val, grad_x

def optimize_softmax_lbfgs(a_init, time_limit, T_schedule=None):
    """
    Optimize using softmax parameterization and L-BFGS with temperature annealing.
    """
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    a_init = a_init / a_init.sum()
    
    if T_schedule is None:
        c0 = fast_convolve(a_init)
        c_max0 = c0.max()
        # T schedule: start at c_max0, decrease by factor of 2.5 each step
        T_schedule = []
        T = c_max0 * 1.5
        for _ in range(15):
            T_schedule.append(T)
            T /= 2.5
        T_schedule.append(c_max0 / 200)
    
    x = np.log(a_init)
    best_a = a_init.copy()
    best_score = 2 * n * fast_convolve(a_init).max()
    
    t0 = time.monotonic()
    
    for i, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        remaining = time_limit - (time.monotonic() - t0)
        # Allocate time proportionally (more time for smaller T)
        iters = min(300, max(50, int(remaining * 0.3)))
        
        def func(x_eval):
            v, _ = smoothed_obj_grad(x_eval, T)
            return v
        
        def grad_func(x_eval):
            _, g = smoothed_obj_grad(x_eval, T)
            return g
        
        try:

def spectral_init(n, rng):
    """
    Generate a sequence using spectral (frequency-domain) methods.
    Try to make the power spectrum as flat as possible.
    """
    N = 2 * n - 1
    
    # Start with a random phase spectrum
    A = np.zeros(N, dtype=np.complex128)
    A[0] = 1.0  # DC component (sum = 1)
    
    # Random phases for other frequencies
    phases = rng.uniform(0, 2 * np.pi, N)
    # Constant magnitude (flat spectrum)
    mag = np.ones(N)
    mag[0] = 1.0 / np.sqrt(N)  # Scale so that sum(a) = 1
    # Actually, |A(0)| = sum(a) = 1, so mag[0] = 1
    # For other frequencies, we want |A(k)| to be small
    # By Parseval: sum |A(k)|^2 / N = sum a[j]^2
    # If a is uniform: sum a[j]^2 = 1/n
    # So sum |A(k)|^2 = N/n
    # |A(0)|^2 = 1, so sum_{k>0} |A(k)|^2 = N/n - 1 = (2n-1)/n - 1 = (n-1)/n
    # Average |A(k)|^2 for k > 0: (n-1)/(n*(N-1)) = (n-1)/(n*(2n-2)) = 1/(2n)
    # So |A(k)| ≈ 1/sqrt(2n) for k > 0
    
    target_mag_sq = 1.0 / (2 * n)
    for k in range(1, N):
        A[k] = np.sqrt(target_mag_sq) * np.exp(1j * phases[k])
    
    # Inverse DFT to get a
    a = np.fft.irfft(A, N)[:n]
    
    # Project to non-negative
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def peak_targeted_refine(a, time_limit):
    """
    Refine by targeting the top peaks of the auto-convolution.
    For each peak, try to reduce it by adjusting the contributing pairs.
    """
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    step_size = 0.001
    it = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        c = fast_convolve(a)
        
        # Find top peaks
        n_peaks = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        for k in peak_indices:
            k = int(k)
            # Contributing pairs: a[j] * a[k-j] for valid j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Find the dominant pairs
            contributions = []
            for j in range(js, je + 1):
                contributions.append((a[j] * a[k - j], j, k - j))
            contributions.sort(reverse=True)
            
            # Try to reduce the top contributions
            for contrib, j1, j2 in contributions[:8]:
                if contrib < 1e-15:
                    continue
                
                # Try shifting mass from j1 or j2 to a random location
                for _ in range(3):
                    src = j1 if a[j1] >= a[j2] else j2
                    if a[src] < 1e-10:
                        continue
                    
                    # Target: a location that doesn't contribute to this peak
                    # (i.e., k - target is not in [0, n-1], or target is far from contributing pairs)
                    for _attempt in range(5):
                        rng = np.random.default_rng((it * 1000003 + k * 7919 + src * 31) % (2**31))
                        target = int(rng.integers(0, n))
                        if target == src or target == j1 or target == j2:
                            continue
                        
                        # Check that target doesn't contribute to this peak
                        if js <= target <= je:
                            continue
                        if js <= k - target <= je:
                            continue
                        
                        # Shift mass
                        for frac in [0.1, 0.05, 0.01]:
                            delta = a[src] * frac
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[target] += delta
                            if candidate[src] < -1e-15:
                                continue
                            candidate = np.maximum(candidate, 0)
                            s = candidate.sum()
                            if s < 1e-10:
                                continue
                            candidate = candidate / s
                            
                            new_c = fast_convolve(candidate)
                            new_score = 2 * n * new_c.max()
                            if new_score < best_score - 1e-12:
                                a = candidate
                                best_score = new_score
                                best_a = a.copy()
                                improved = True
                                break
                        
                        if improved:
                            break
                
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            step_size *= 0.9
            if step_size < 1e-8:
                break
            # Periodic perturbation
            if it % 100 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                new_score = 2 * n * fast_convolve(candidate).max()
                if new_score < best_score:
                    a = candidate
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
            elif it > 500:
                break
    
    return best_a, best_score

def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s
