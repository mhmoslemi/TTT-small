    n = len(a)

    js = max(0, k - n + 1)
    je = min(n - 1, k)
    js2 = max(0, k - n + 1)
    je2 = min(n - 1, k)
    # grad[js:je+1] = 2 * a[k-je : k-js+1][::-1]
    idx = np.arange(js, je + 1)
    grad[idx] = 2.0 * a[k - idx][::-1]
    return grad

def compute_grad_weighted(a, peaks, weights):
    """Compute weighted sum of gradients at multiple peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):
        if w < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += w * 2.0 * a[k - idx][::-1]
    return grad

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def adam_optimize(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """
    Adam optimization on log-domain parameterization.
    Minimize max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    
    # Log-domain parameterization: a = exp(x) / sum(exp(x))
    x = np.log(a0)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a0.copy()
    best_obj = fast_convolve(a0).max()
    best_peak_count = 1
    
    t0 = time.monotonic()
    no_improve = 0
    top_k = 5
    peak_threshold = 0.99  # Consider peaks that are at least 99% of max
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Current a
        a = np.exp(x - np.max(x))
        a = a / a.sum()
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find significant peaks
        threshold = peak_threshold * max_val
        peak_mask = conv >= threshold
        peaks = np.where(peak_mask)[0]
        
        if len(peaks) == 0:
            peaks = np.array([np.argmax(conv)])
        
        # Weight peaks by their value
        pv = conv[peaks]
        weights = pv ** 4  # Heavy-weight the largest peaks
        weights = weights / weights.sum()
        
        # Compute gradient (in a-space)
        grad_a = compute_grad_weighted(a, peaks, weights)
        
        # Convert to x-space gradient: a_i = exp(x_i) / Z, da_i/dx_j = a_i * (delta_ij - a_j)
        # So grad_x = a * (grad_a - <grad_a, a>)
        grad_x = a * (grad_a - np.dot(grad_a, a))
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * grad_x ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Update best
        new_a = np.exp(x - np.max(x))
        new_a = new_a / new_a.sum()
        new_obj = fast_convolve(new_a).max()
        
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = new_a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt parameters
        if no_improve > 50:
            # Try different peak count
            if top_k < 3:
                top_k += 1
            elif top_k < 10:
                top_k += 2
            # Adjust learning rate
            if no_improve > 100:
                lr *= 0.7
            if no_improve > 200:
                # Perturb
                rng = np.random.default_rng((t * 7919 + 13) % (2**31 - 1))
                x = np.log(np.maximum(best_a + rng.standard_normal(n) * 0.01, 1e-15))
                lr = 0.01
                no_improve = 0
            
        # Anneal peak_threshold: start focused on max, then broaden
        progress = min(1.0, t / max(1, int(time_limit * 10)))
        peak_threshold = 0.99 - 0.2 * progress  # From 0.99 to 0.79
        
        if t % 200 == 0:
            print(f"    Adam t={t}, obj={best_obj:.10f}, lr={lr:.6f}, peaks={len(peaks)}")
    
    return best_a, best_obj

def direct_subgradient(a0, time_limit, top_k=10):
    """Direct subgradient with adaptive step size and line search."""
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = 0.05
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Top peaks
        top_k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -top_k_actual)[-top_k_actual:]
        pv = conv[peaks]
        weights = pv ** 2
        weights = weights / weights.sum()
        
        grad = compute_grad_weighted(a, peaks, weights)
        
        # Direction: project (a - lr * grad) onto simplex
        direction = project_simplex(a - lr * grad) - a
        
        # Line search
        best_alpha = 0.0
        best_val = max_val
        for alpha in [0.1, 0.25, 0.5, 0.75, 1.0]:
            cand = a + alpha * direction
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            a = a + best_alpha * direction
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.05, 0.2)
            else:
                no_improve += 1
                lr *= 0.9
        else:
            no_improve += 1
            lr *= 0.8
        
        if lr < 0.001:
            lr = 0.05
            no_improve = 0
        
        if no_improve > 60:
            rng = np.random.default_rng((it * 48271 + 1) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            lr = 0.05
            no_improve = 0
    
    return best_a, best_obj

def peak_shave_refined(a0, time_limit):
    """
    Refined peak shaving: identify the peak, compute the exact contribution
    of each pair (i, j) with i+j = peak_pos, and redistribute.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]
        
        # All pairs (i, j) with i + j = k_peak, 0 <= i, j < n
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        
        # Contribution of each i: a[i] * a[k_peak - i]
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k_peak - i_vals
        contrib = a[i_vals] * a[j_vals]
        total = contrib.sum()
        
        if total < 1e-15:
            break
        
        # Strategy: reduce the largest contributing pairs
        # and add mass to positions that don't contribute to this peak
        in_peak = set(i_vals.tolist()) | set(j_vals.tolist())
        out_peak = [j for j in range(n) if j not in in_peak]
        
        if len(out_peak) == 0:
            out_peak = list(range(n))
        
        # Try reducing the top contributing pairs
        order = np.argsort(contrib)[::-1]
        
        best_new_obj = peak_val
        best_new_a = None
        
        for frac in [0.05, 0.1, 0.15, 0.2]:
            # Remove mass from top contributors
            delta = np.zeros(n)
            removed = 0.0
            for idx in order:
                i = i_vals[idx]
                j = j_vals[idx]
                c = contrib[idx]
                # Reducing a[i] by d reduces the pair (i,j) contribution by d*a[j]
                # We want to reduce total by frac * total
                d = min(a[i] * 0.5, (frac * total - removed) / max(a[j], 1e-15))
                d = min(d, a[i] * 0.5)
                if d <= 0:
                    break
                delta[i] -= d
                removed += d * a[j]
                if removed >= frac * total:
                    break
            
            # Add to out_peak positions proportionally
            if removed > 0 and len(out_peak) > 0:
                # Distribute to reduce other peaks
                for op in out_peak:
                    delta[op] += removed / len(out_peak)
            
            a_new = a + delta
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            new_obj = fast_convolve(a_new).max()
            
            if new_obj < best_new_obj:
                best_new_obj = new_obj
                best_new_a = a_new.copy()
        
        if best_new_a is not None and best_new_obj < best_obj - 1e-12:
            best_obj = best_new_obj
            best_a = best_new_a
            a = best_new_a
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 40:
            rng = np.random.default_rng((it * 9973 + 5) % (2**31 - 1))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
    
    return best_a, best_obj

def hybrid_optimize(a0, time_limit):
    """
    Hybrid: alternating between Adam, subgradient, and peak shaving.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    # Cycle through methods
    methods = [('adam', 30), ('subgrad', 20), ('peak', 10)]
    method_idx = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        name, duration = methods[method_idx % len(methods)]
        duration = min(duration, remaining - 1)
        if duration < 2:
            break
        
        if name == 'adam':
            a_opt, obj = adam_optimize(best_a, duration, lr=0.005)
        elif name == 'subgrad':
            a_opt, obj = direct_subgradient(best_a, duration, top_k=5)
        else:
            a_opt, obj = peak_shave_refined(best_a, duration)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_opt.copy()
        
        method_idx += 1
        print(f"    Hybrid {name}: obj={best_obj:.10f} (remaining={remaining:.0f}s)")
    
    return best_a, best_obj

def generate_inits_nonsymmetric(n, rng):
    """Generate strongly non-symmetric initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Exponential decay from left (mass concentrated at left)
    for alpha in [0.01, 0.02, 0.05, 0.1, 0.2]:
        a = np.exp(-alpha * x)
        inits.append(a / a.sum())
    
    # Exponential decay from right (mass concentrated at right)
    for alpha in [0.01, 0.02, 0.05, 0.1, 0.2]:
        a = np.exp(-alpha * (n - 1 - x))
        inits.append(a / a.sum())
    
    # Power law from left
    for beta in [0.5, 1.0, 1.5, 2.0]:
        a = (x + 1) ** (-beta)
        inits.append(a / a.sum())
    
    # Power law from right
    for beta in [0.5, 1.0, 1.5, 2.0]:
        a = (n - x) ** (-beta)
        inits.append(a / a.sum())
    
    # Linear ramp (left-heavy)
    a = (n - x).astype(float)
    inits.append(a / a.sum())
    
    # Linear ramp (right-heavy)
    a = (x + 1).astype(float)
    inits.append(a / a.sum())
    
    # Quadratic (left-heavy)
    a = (n - 1 - x) ** 2 + 1
    inits.append(a / a.sum())
    
    # Two-sided but asymmetric
    for center_frac in [0.25, 0.35, 0.65, 0.75]:
        c = center_frac * (n - 1)
        a = np.exp(-0.5 * ((x - c) / (0.15 * n)) ** 2)
        inits.append(a / a.sum())
    
    # Random non-symmetric
    for _ in range(3):
        a = rng.dirichlet(np.ones(n) * 0.5)
        # Make it more non-symmetric
        a = a * (1 + 0.5 * np.sin(np.pi * x / n))
        a = np.maximum(a, 0)
        a = a / a.sum()
        inits.append(a)
    
    # Sigmoid-shaped (asymmetric S-curve)
    for mid in [0.3, 0.5, 0.7]:
        a = 1.0 / (1.0 + np.exp(-5.0 * (x / (n - 1) - mid)))
        a = np.maximum(a, 1e-6)
        inits.append(a / a.sum())
    
    return inits

def generate_inits_symmetric(n, rng):
    """Generate symmetric and near-symmetric initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Gaussian (centered)
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4]:
        a = np.exp(-0.5 * ((x - (n-1)/2) / (sigma_frac * n)) ** 2)
        inits.append(a / a.sum())
    
    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.5
    inits.append(edge_w / edge_w.sum())
    
    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 1.0))
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    return inits
