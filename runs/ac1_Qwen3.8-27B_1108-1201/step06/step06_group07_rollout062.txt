**New idea: Use the "phase" of the Fourier transform**

The auto-convolution of a is related to the power spectrum of the Fourier transform of a. Specifically:

For linear convolution, it's a bit different, but the peak of the linear convolution is related to the "peak-to-average" ratio of the power spectrum.

This is the "peak-to-average power ratio" (PAPR) minimization problem, which is well-studied in signal processing!

Standard approaches for PAPR reduction:
1. **Schwartz's algorithm**: Iteratively clip the time-domain peaks and go back to frequency domain
2. **BLCAPE (Block Linear Complex Amplitude)**: Similar to above but with complex amplitudes
3. **SL-RAFT (Selected Mapping)**: Try different phase modifications

But these are for complex sequences. For real non-negative sequences, the problem is harder.

However, the key insight is: in the frequency domain, we want |FFT(a)| to be as flat as possible. Since a is real and non-negative, FFT(a) has Hermitian symmetry, and its magnitude is constrained.

**Approach: "Frequency-domain equalization"**

1. Start with a (real, non-negative, normalized)
2. Compute A = FFT(a) (zero-padded to 2n-1 for linear convolution)
3. The "peak" in time domain corresponds to the inverse FFT of |A|^2
4. To reduce the peak, we want to make |A| flatter
5. Update: modify |A| to be flatter, then invert back to time domain
6. Project back to non-negative (clip negative values)

This is similar to the "successive over-relaxation" method for PAPR reduction.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
from scipy.special import softmax
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_matrix_apply(a, f):
    """Compute (f * a) using FFT (linear convolution)."""
    return fast_convolve_2(f, a)


def fast_convolve_2(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_lp_sparse(n, f):
    """Build sparse LP constraint matrix for the SCA subproblem."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_sca_lp(f, M, n, A_sparse=None):
    """Solve: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    if A_sparse is None:
        A_sparse = build_lp_sparse(n, f)
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

    M = conv.max() if M_init is None else M_init
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_decrease_rate = 0.997
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:

        # Recompute A_sparse periodically (f changes)

        g_raw = solve_sca_lp(f, M, n, A_sparse)
        
        if g_raw is None:
            M *= 1.02
            no_improve += 1
            if no_improve > 100:
                M = fast_convolve(f).max() * 1.005
                no_improve = 0

        g = g_raw / g_raw.sum()
        
        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 50:

                # Fourier perturbation

                f = project_simplex(f + 0.005 * pert)

            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:

            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:

                best_ft = ft.copy()

            best_f = f.copy()
            M = best_val * 0.998

def dual_qp_method(a0, time_limit, verbose=False):
    """
    Dual method: alternate between solving inner QP and updating dual.
    Inner QP: min b^T C b s.t. sum(b)=1, b>=0 where C[j,i]=lambda[j+i]
    Dual update: lambda = (b*b) / sum(b*b) (or concentrated on peaks)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # Initialize lambda as uniform on peaks
    conv = fast_convolve(a)
    top_k = min(20, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    lam = np.zeros(len(conv))
    lam[peaks] = 1.0 / len(peaks)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        # Build C matrix: C[j,i] = lambda[j+i] (for valid j+i < 2n-1)
        # C is n x n
        C = np.zeros((n, n))
        for j in range(n):
            valid_i = np.arange(n)
            valid_k = j + valid_i
            valid_mask = valid_k < (2 * n - 1)
            C[j, valid_i[valid_mask]] = lam[valid_k[valid_mask]]
        
        # Solve: min b^T C b s.t. 1^T b = 1, b >= 0
        # First try the unconstrained solution (if C is PSD)
        try:
            C_reg = C + 1e-6 * np.eye(n)  # Regularize for numerical stability
            # KKT: 2*C*b + 1*mu = 0, sum(b) = 1
            # => b = -0.5 * mu * C^{-1} * 1
            # => sum(b) = -0.5 * mu * 1^T C^{-1} 1 = 1
            # => mu = -2 / (1^T C^{-1} 1)
            ones = np.ones(n)
            Ci1 = np.linalg.solve(C_reg, ones)
            mu = -2.0 / (ones @ Ci1)
            b_unconstrained = -0.5 * mu * Ci1
            
            if np.all(b_unconstrained >= -1e-10):
                b = np.maximum(b_unconstrained, 0)
                b = b / b.sum()
            else:
                # Need to solve with constraints
                # Use a simple projected gradient method

def fourier_perturb(a, rng, scale=0.01):
    """Apply a Fourier-based structured perturbation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    pert = np.zeros(n)
    num_freqs = rng.integers(1, 4)
    for _ in range(num_freqs):
        freq = rng.integers(1, max(2, n // 3))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.5, 1.0)
        pert += amp * np.cos(2 * np.pi * freq * x / n + phase)
    pert /= (np.max(np.abs(pert)) + 1e-15)
    return project_simplex(a + scale * pert)


def natural_perturb(a, rng, scale=0.01):
    """Perturb in log-space (natural gradient)."""
    n = len(a)
    w = np.log(np.maximum(a, 1e-30))
    w += rng.standard_normal(n) * scale
    a_new = np.exp(w - w.max())
    a_new = a_new / a_new.sum()
    return a_new


def ils_search(best_a, time_limit, seed=42):
    """
    Iterated Local Search: perturb + SCA, keep best.
    Uses structured perturbations (Fourier + natural).
    """
    rng = np.random.default_rng(seed)
    n = len(best_a)
    current_best = best_a.copy()
    best_score = fast_convolve(best_a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        pert_type = rng.integers(0, 4)
        scale = rng.uniform(0.002, 0.02)
        
        if pert_type == 0:
            a_pert = natural_perturb(current_best, rng, scale)
        elif pert_type == 1:
            a_pert = fourier_perturb(current_best, rng, scale)
        elif pert_type == 2:
            # Swap-based: move mass from a random position to another
            j = rng.integers(0, n)
            i = rng.integers(0, n)
            if i != j:
                a_pert = current_best.copy()
                delta = a_pert[j] * rng.uniform(0.1, 0.5)
                a_pert[j] -= delta
                a_pert[i] += delta
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
            else:
                a_pert = natural_perturb(current_best, rng, scale)
        else:

                a_pert = natural_perturb(current_best, rng, scale)
            else:
                a_pert = current_best.copy()
                delta = np.sum(a_pert[j_start:j_start+block_size]) * rng.uniform(0.1, 0.4)

                a_pert[j_start:j_start+block_size] *= (1 - frac)

                a_pert = np.maximum(a_pert, 0)

        # Run SCA on perturbed

        if t_alloc < 10:
            break
        a_opt, obj = sca_optimize(a_pert, time_limit=t_alloc)
        score = fast_convolve(a_opt).max()
        
        if score < best_score - 1e-14:
            best_score = score
            current_best = a_opt.copy()
            
    return current_best, best_score


def generate_inits(n, parent, rng):
    """Generate diverse initializations for a given n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Edges heavy
    edge_w = 1.0 / (np.minimum(x.astype(float), (n - 1 - x).astype(float)) + 1.0) ** 1.2
    inits.append(edge_w / edge_w.sum())
    
    # Cosine window
    cosw = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    inits.append(cosw / cosw.sum())
    
    # Two-peak Gaussian
    g2 = np.exp(-0.5 * ((x - 0.15 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.85 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())
    
    # Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 5.0))
    
    # Sine-modulated
    sine = 1.0 + 0.3 * np.sin(4 * np.pi * x / n)
    inits.append(sine / sine.sum())
    
    return inits

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve_2(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_sparse(n, f):
    """Build sparse matrix for LP: A[g] = (f*g) where A is (2n-1) x n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_sca_lp(f, M, n, A_sparse=None, lp_timeout=20):
    """Solve: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    if A_sparse is None:
        A_sparse = build_lp_sparse(n, f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': lp_timeout})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def fourier_perturb(a, rng, scale=0.01):
    """Apply a Fourier-based structured perturbation."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    pert = np.zeros(n)
    num_freqs = int(rng.integers(1, 4))
    for _ in range(num_freqs):
        freq = int(rng.integers(1, max(2, n // 3)))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.5, 1.0))
        pert += amp * np.cos(2.0 * np.pi * freq * x / n + phase)
    mx = np.max(np.abs(pert))
    if mx > 1e-15:
        pert /= mx
    return project_simplex(a + scale * pert)


def natural_perturb(a, rng, scale=0.01):
    """Perturb in log-space (natural gradient direction)."""
    n = len(a)
    w = np.log(np.maximum(a, 1e-30))
    w += rng.standard_normal(n) * scale
    a_new = np.exp(w - w.max())
    a_new = a_new / a_new.sum()
    return a_new


def subgradient_step(a, eta, top_k=10):
    """One subgradient step with natural gradient (log-space update)."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        psum = pv.sum()
        if psum < 1e-15:
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            weights = pv / psum
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]
    # Natural gradient: update in log-space
    wlog = np.log(np.maximum(a, 1e-30))
    wlog -= eta * grad
    a_new = np.exp(wlog - wlog.max())
    a_new = a_new / a_new.sum()
    return a_new


def sca_optimize(f0, time_limit, M_init=None, lp_timeout=20, min_improve=1e-15):
    """SCA optimization with adaptive M and LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    A_sparse = build_lp_sparse(n, f)

    conv = fast_convolve(f)
    M = conv.max() if M_init is None else M_init
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_rate = 0.9975

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Rebuild A_sparse periodically
        if it % 30 == 1:
            A_sparse = build_lp_sparse(n, f)

        g_raw = solve_sca_lp(f, M, n, A_sparse, lp_timeout)

        if g_raw is None:
            M *= 1.015
            no_improve += 1
            if no_improve > 80:
                M = fast_convolve(f).max() * 1.003
                no_improve = 0
            continue

        g = g_raw / g_raw.sum()

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng((it * 7919 + 13) % (2**31 - 1))
                # Try different perturbation strategies
                if no_improve % 3 == 0:
                    f = natural_perturb(f, rng, 0.004)
                elif no_improve % 3 == 1:
                    f = fourier_perturb(f, rng, 0.004)
                else:
                    f = project_simplex(f + rng.standard_normal(n) * 0.003)
                M = fast_convolve(f).max() * 1.003
                no_improve = 0
            continue

        # Line search
        best_val = fast_convolve(f).max()
        best_ft = f.copy()

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_ft = ft.copy()

        if best_val < best_obj - min_improve:
            f = best_ft
            best_obj = best_val
            best_f = f.copy()
            M = best_val * 0.9985
            no_improve = 0
        else:
            no_improve += 1
            M *= m_rate

        # Escape mechanism
        if no_improve > 60:
            rng = np.random.default_rng((it * 31831 + 7) % (2**31 - 1))
            scale = 0.003 + 0.004 * min(1.0, (no_improve - 60) / 80.0)
            # Alternate between perturbation types
            if (no_improve // 10) % 2 == 0:
                f = natural_perturb(best_f, rng, scale)
            else:
                f = fourier_perturb(best_f, rng, scale)
            M = fast_convolve(f).max() * 1.003
            no_improve = 0

    return best_f, best_obj


def subgradient_polish(a0, time_limit, eta_init=0.03, top_k=10):
    """Subgradient method with natural gradient and line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = eta_init
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it_left = time_limit - (time.monotonic() - t0)
        if it_left < 0.5:
            break

        conv = fast_convolve(a)
        cur_max = conv.max()

        # Compute subgradient
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            psum = pv.sum()
            if psum < 1e-15:
                weights = np.ones(len(peaks)) / len(peaks)
            else:
                weights = pv / psum

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = weights[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        # Natural gradient step in log-space
        wlog = np.log(np.maximum(a, 1e-30))
        wlog -= eta * grad
        a_new = np.exp(wlog - wlog.max())
        a_new = a_new / a_new.sum()

        # Line search
        best_candidate = a
        best_val = cur_max
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            eta *= 0.9
            if eta < eta_init * 0.05:
                eta = eta_init
                no_improve = 0

        if no_improve > 200:
            rng = np.random.default_rng((int(time.monotonic()*1000) + no_improve) % (2**31 - 1))
            a = natural_perturb(best_a, rng, 0.003)
            eta = eta_init
            no_improve = 0

    return best_a, best_obj


def ils_search(base_a, time_limit, seed):
    """Iterated Local Search with structured perturbations."""
    rng = np.random.default_rng(seed)
    n = len(base_a)
    current_best = base_a.copy()
    best_score = fast_convolve(base_a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        # Choose perturbation
        pert_type = int(rng.integers(0, 5))
        scale = float(rng.uniform(0.002, 0.015))

        if pert_type == 0:
            a_pert = natural_perturb(current_best, rng, scale)
        elif pert_type == 1:
            a_pert = fourier_perturb(current_best, rng, scale)
        elif pert_type == 2:
            # Multi-frequency perturbation
            x = np.arange(n, dtype=np.float64)
            pert = np.zeros(n)
            for _ in range(3):
                freq = int(rng.integers(1, max(2, n // 4)))
                phase = float(rng.uniform(0, 2 * np.pi))
                pert += np.cos(2.0 * np.pi * freq * x / n + phase)
            pert /= (np.max(np.abs(pert)) + 1e-15)
            a_pert = project_simplex(current_best + scale * pert)
        elif pert_type == 3:
            # Block redistribution
            block = int(rng.integers(max(2, n // 20), max(3, n // 8)))
            j0 = int(rng.integers(0, max(1, n - 2 * block)))
            i0 = int(rng.integers(0, max(1, n - block)))
            if i0 + block > n:
                i0 = n - block
            if abs(i0 - j0) < block:
                a_pert = natural_perturb(current_best, rng, scale)
            else:
                a_pert = current_best.copy()
                src_mass = np.sum(a_pert[j0:j0 + block])
                delta = src_mass * float(rng.uniform(0.05, 0.3))
                frac = delta / max(src_mass, 1e-15)
                a_pert[j0:j0 + block] *= (1.0 - frac)
                a_pert[i0:i0 + block] += delta / block
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
        else:
            # Antisymmetric-ish perturbation (reduces center peak)
            x = np.arange(n, dtype=np.float64)
            t = x / max(1, n - 1)
            pert = 0.5 * np.sin(2.0 * np.pi * t) + 0.3 * np.sin(4.0 * np.pi * t)
            pert *= float(rng.uniform(-1, 1))
            a_pert = project_simplex(current_best + scale * pert)

        t_alloc = min(remaining - 25, 100)
        if t_alloc < 8:
            break

        a_opt, obj_opt = sca_optimize(a_pert, time_limit=t_alloc, lp_timeout=15)
        score = fast_convolve(a_opt).max()

        if score < best_score - 1e-14:
            best_score = score
            current_best = a_opt.copy()

    return current_best, best_score


def mirror_descent_fast(a0, time_limit, eta_init=0.03, top_k=12):
    """Fast multiplicative weight mirror descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = eta_init
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        conv = fast_convolve(a)
        cur_max = conv.max()

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pw = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            pw = softmax(pv / (0.003 * cur_max + 1e-10))

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            w = pw[idx]
            if w < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += w * 2.0 * a[k - je:k - js + 1][::-1]

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Line search
        best_candidate = a
        best_val = cur_max
        for alpha in [0.3, 0.5, 0.7, 1.0]:
            cand = (1 - alpha) * a + alpha * a_new
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            val = fast_convolve(cand).max()
            if val < best_val:
                best_val = val
                best_candidate = cand

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20:
            eta *= 0.93
        if eta < 5e-5:
            eta = eta_init
            no_improve = 0

        if no_improve > 120:
            rng = np.random.default_rng((int(time.monotonic() * 1000) + 7) % (2**31 - 1))
            a = natural_perturb(best_a, rng, 0.003)
            eta = eta_init * 0.7
            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for given n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    inits.append(np.ones(n) / n)

    # Edge-heavy
    edge_w = 1.0 / (np.minimum(x, n - 1 - x).astype(float) + 1.0) ** 1.3
    inits.append(edge_w / edge_w.sum())

    # Cosine window
    cosw = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(cosw / cosw.sum())

    # Two-peak
    g2 = np.exp(-0.5 * ((x - 0.18 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.82 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Dirichlet concentrated
    inits.append(rng.dirichlet(np.ones(n) * 0.4))

    # Sinusoidal modulation
    sine = 1.0 + 0.3 * np.sin(4.0 * np.pi * x / n) + 0.1 * np.sin(8.0 * np.pi * x / n)
    inits.append(sine / sine.sum())

    return inits


def try_update(a, best_seq, best_score):
    """Try to update best. Returns (best_seq, best_score, improved)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score, False
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        return a_norm.copy(), score, True
    return best_seq, best_score, False
