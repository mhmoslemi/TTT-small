    # This is handled by bounds
    bounds = [(-a[j], None) for j in range(n)] + [(None, None)]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]

        delta = res.x[:n]
        t_new = res.x[-1]
        return delta, t_new
    
    return None, None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""
    return np.correlate(p, a, mode='valid')


def lp_direction_step(a, P, max_time=5.0):
    """
    LP-based direction finding: minimize t s.t. linearized peak constraints.
    Returns (delta, t_new) or (None, None).
    """
    n = len(a)
    conv = fast_convolve(a)
    t0 = time.monotonic()

    # Variables: delta (n), t (1) -> n+1 total
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Peak constraints: 2*sum_j delta_j * a_{k-j} - t <= -conv[k] for k in P
    nP = len(P)
    A_ub = np.zeros((nP, n + 1))
    b_ub = np.zeros(nP)
    for i, k in enumerate(P):
        A_ub[i, -1] = -1.0
        # For each j where k-j is in [0, n):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[i, j] = 2.0 * a[k - j]
        b_ub[i] = -conv[k]

    bounds = [(-a[j], None) for j in range(n)] + [(None, None)]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(0.1, max_time - (time.monotonic() - t0))})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None, None


def fw_lp_oracle(f, M, max_time=10.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(0.1, max_time)})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_direction_optimize(a0, time_limit, peak_rel_tol=0.003, max_lp_calls=500):

    a = np.maximum(a0.astype(np.float64), 0.0)

    no_improve = 0
    it = 0

        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        threshold = M * (1.0 - peak_rel_tol)

                peak_rel_tol *= 1.5
                continue
            break

        # Line search on the step

        best_val = M
        for alpha in [0.3, 0.5, 0.7, 0.9, 1.0, 1.2, 1.5]:

            sc = candidate.sum()
            if sc < 1e-15:
                continue
            candidate = candidate / sc
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-16:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-10:
            no_improve += 1
            if no_improve > 20:
                # Perturb and restart

                noise_scale = 0.001 + 0.003 * min(1.0, no_improve / 50.0)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                no_improve = 0
                peak_rel_tol = 0.003
                continue

            continue

        a = (a + best_alpha * delta)
        a = np.maximum(a, 0.0)
        a = a / a.sum()

            best_obj = best_val
            best_a = a.copy()
            no_improve = 0

        else:

def kkt_newton_polish(a0, time_limit, max_iterations=200):
    """
    Newton-type polish using the KKT system.
    Solves: peak equalization + stationarity + normalization simultaneously.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(max_iterations):
        if time.monotonic() - t0 > time_limit - 1:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Active peaks
        threshold = M * (1.0 - 0.001)
        P = np.where(conv >= threshold)[0]
        # Support
        S = np.where(a > 1e-10)[0]

        if len(P) < 2 or len(S) < 2:
            break

        nP = len(P)
        nS = len(S)

        # Build the KKT system
        # Unknowns: a_S (nS), mu_P (nP), t (1), nu (1)
        # Equations:
        #   Peak: conv(a,a)_k = t for k in P  (nP equations)
        #   Stat: 2*sum_{k in P} mu_k * a_{k-j} = nu for j in S  (nS equations)
        #   Norm: sum(a_S) = 1
        #   MuNorm: sum(mu_P) = 1
        N = nS + nP + 2

        # Initial guess
        mu = np.ones(nP) / nP
        t_init = conv[P].mean()
        # Estimate nu from current a
        nu_est = np.zeros(nS)
        for i, j in enumerate(S):
            nu_est[i] = 2.0 * sum(mu[k_idx] * a[k - j] for k_idx, k in enumerate(P) if 0 <= k - j < n)
        nu_init = nu_est.mean()

        x = np.zeros(N)
        x[:nS] = a[S]
        x[nS:nS + nP] = mu
        x[nS + nP] = t_init
        x[nS + nP + 1] = nu_init

        # Newton iterations for the fixed active set
        converged = False
        for nit in range(30):
            a_S = x[:nS]
            mu_P = x[nS:nS + nP]
            t_val = x[nS + nP]
            nu_val = x[nS + nP + 1]

            a_S_pos = np.maximum(a_S, 0.0)
            a_full = np.zeros(n)
            a_full[S] = a_S_pos

            # Residuals
            conv_full = fast_convolve(a_full)
            F_peak = conv_full[P] - t_val

            F_stat = np.zeros(nS)
            for i, j in enumerate(S):
                val = 0.0
                for k_idx, k in enumerate(P):
                    idx = k - j
                    if 0 <= idx < n:
                        val += mu_P[k_idx] * a_full[idx]
                F_stat[i] = 2.0 * val - nu_val

            F = np.concatenate([F_peak, F_stat, [a_S_pos.sum() - 1.0, mu_P.sum() - 1.0]])

            if np.max(np.abs(F)) < 1e-13:
                converged = True
                break

            # Jacobian (sparse structure)
            J = np.zeros((N, N))

            # d(F_peak)/d(a_S): J[i, j] = 2*a_full[P[i] - S[j]] if valid
            for i, k in enumerate(P):
                for j_idx, j in enumerate(S):
                    idx = k - j
                    if 0 <= idx < n:
                        J[i, j_idx] = 2.0 * a_full[idx]
            # d(F_peak)/dt = -1
            for i in range(nP):
                J[i, nS + nP] = -1.0

            # d(F_stat)/d(a_S): J[i, j] = 2*mu_P[idx] where idx = S[i]+S[j] in P
            P_set = set(P.tolist())
            P_to_idx = {p: i for i, p in enumerate(P)}
            for i, j in enumerate(S):
                for m_idx, m in enumerate(S):
                    s_val = j + m
                    if s_val in P_to_idx:
                        J[nP + i, m_idx] = 2.0 * mu_P[P_to_idx[s_val]]
            # d(F_stat)/d(mu_P): J[i, k_idx] = 2*a_full[P[k_idx] - S[i]]
            for i, j in enumerate(S):
                for k_idx, k in enumerate(P):
                    idx = k - j
                    if 0 <= idx < n:
                        J[nP + i, nS + k_idx] = 2.0 * a_full[idx]
            # d(F_stat)/d(nu) = -1
            for i in range(nS):
                J[nP + i, nS + nP + 1] = -1.0

            # d(norm)/d(a_S) = 1
            for j in range(nS):
                J[nP + nS, j] = 1.0
            # d(mu_norm)/d(mu_P) = 1
            for k in range(nP):
                J[nP + nS + 1, nS + k] = 1.0

            # Solve
            try:
                delta_x = np.linalg.solve(J, -F)
            except np.linalg.LinAlgError:
                try:
                    delta_x = np.linalg.lstsq(J, -F, rcond=None)[0]
                except Exception:
                    break

            # Damped step
            step_taken = False
            for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
                x_trial = x + alpha * delta_x
                a_S_t = np.maximum(x_trial[:nS], 0.0)
                if a_S_t.sum() < 1e-10:
                    continue
                a_t = np.zeros(n)
                a_t[S] = a_S_t
                a_t = a_t / a_t.sum()
                new_M = fast_convolve(a_t).max()
                if new_M <= M + 1e-12:
                    x = x_trial
                    step_taken = True
                    break

            if not step_taken:
                break

        # Update a from the Newton solution
        if converged or True:  # always try to use the result
            a_S_new = np.maximum(x[:nS], 0.0)
            a_new = np.zeros(n)
            a_new[S] = a_S_new
            sn = a_new.sum()
            if sn > 1e-10:
                a_new = a_new / sn
                new_M = fast_convolve(a_new).max()
                if new_M < best_obj - 1e-14:
                    best_obj = new_M
                    best_a = a_new.copy()
                    a = a_new
                elif new_M < M:
                    a = a_new

        if best_obj < M * (1.0 - 1e-8):
            # Made progress, continue
            continue
        else:
            # No significant progress, break
            break

    return best_a, best_obj


def softmax_grad_optimize(a0, time_limit, T_init=0.001, lr_init=0.01):
    """Sm
