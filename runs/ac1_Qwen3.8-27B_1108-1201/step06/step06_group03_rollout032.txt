import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def symmetrize(a):
    """Make sequence symmetric: a[j] = a[n-1-j]."""
    return (a + a[::-1]) / 2.0

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix_rows(f, n, peak_indices):
    """Build convolution matrix rows only for specified peak indices.
    Returns (2*len(peak_indices)) x n matrix where row for peak k
    represents the constraint (f*g)[k] <= M."""
    # Actually, we need the constraint f*g[k] <= M.
    # (f*g)[k] = sum_j f[j] * g[k-j] (with valid indices)
    # = sum_j g[j] * f[k-j] (swapping)
    # So the row for peak k is: [f[k], f[k-1], ..., f[k-(n-1)]] (with zeros where out of range)
    
    rows = []
    for k in peak_indices:
        row = np.zeros(n)
        for j in range(n):
            if 0 <= k - j < n:
                row[j] = f[k - j]
        rows.append(row)
    return np.array(rows)

def solve_lp_oracle(f, M, peak_indices=None, all_constrained=False):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in peak_indices, g >= 0."""
    n = len(f)
    
    if peak_indices is None and not all_constrained:
        # Use all constraints
        peak_indices = list(range(2 * n - 1))
    elif peak_indices is not None:
        peak_indices = list(peak_indices)
    else:
        peak_indices = list(range(2 * n - 1))
    
    # Build constraint matrix
    A_rows = []
    for k in peak_indices:
        row = np.zeros(n)
        for j in range(n):
            kk = k - j
            if 0 <= kk < n:
                row[j] = f[kk]
        A_rows.append(row)
    A = np.array(A_rows)
    
    c = -np.ones(n)
    b_ub = np.full(len(peak_indices), M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def symmetrize(a):
    return (a + a[::-1]) / 2.0


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, n):
    """Build (2n-1) x n matrix A where (A @ g) = f * g (full linear convolution)."""
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n))
    A[mask] = f[diff[mask]]
    return A


def build_conv_matrix_subset(f, n, peak_indices):
    """Build constraint matrix only for specified peak indices."""
    k_arr = np.asarray(peak_indices)
    j_idx = np.arange(n)[None, :]
    diff = k_arr[:, None] - j_idx
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((len(peak_indices), n))
    A[mask] = f[diff[mask]]
    return A


def solve_lp_subset(f, M, peak_indices):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in peak_indices, g >= 0."""
    n = len(f)
    A = build_conv_matrix_subset(f, n, peak_indices)
    c = -np.ones(n)
    b_ub = np.full(len(peak_indices), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_full(f, M):
    """Solve LP with all constraints."""
    n = len(f)
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_line_search(f, g, max_iters=40):
    """Adaptive line search for FW step. Returns best alpha."""
    best_alpha = 0.0
    best_val = fast_convolve(f).max() + 1e-15

    # Coarse: 20 points
    alphas = np.linspace(0.005, 1.0, 20)
    for alpha in alphas:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            continue
        ft = ft / s
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    # Fine: 15 points around best
    if best_alpha > 0.005:
        lo = max(0.0, best_alpha - 0.02)
        hi = min(1.0, best_alpha + 0.02)
        for alpha in np.linspace(lo, hi, 15):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            s = ft.sum()
            if s < 1e-15:
                continue
            ft = ft / s
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

    return best_alpha, best_val


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle, constraint generation, and symmetry."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = symmetrize(f)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Get top peaks for constraint generation
        top_k = min(20, 2 * n - 1)
        peak_indices = np.argsort(conv)[::-1][:top_k]

        # Try constraint generation: start with top peaks, add violations
        g = solve_lp_subset(f, M * 1.001, peak_indices)

        if g is None or g.sum() < 1e-15:
            # Fall back to full LP
            g = solve_lp_full(f, M * 1.001)

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 30:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.002 + 0.003 * min(1.0, stagnant / 60.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                f = symmetrize(f)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Check for violated constraints and add them
        fg = fast_convolve(f)  # not needed, we need f*g
        # Actually compute f*g
        fgn = np.convolve(f, g[:n])
        violations = np.where(fgn > M * 1.001)[0]
        if len(violations) > 0:
            # Add violated constraints
            all_peaks = np.unique(np.concatenate([peak_indices, violations[:10]]))
            g = solve_lp_subset(f, M * 1.001, all_peaks)
            if g is None or g.sum() < 1e-15:
                g = solve_lp_full(f, M * 1.001)

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            continue

        # Symmetrize the LP solution
        g_sym = symmetrize(g)
        g_sym = np.maximum(g_sym, 0.0)
        gs = g_sym.sum()
        if gs < 1e-15:
            stagnant += 1
            continue
        g_sym = g_sym / gs

        # Check direction
        if np.max(np.abs(g_sym - f)) < 1e-12:
            stagnant += 1
            if stagnant > 40:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                f = symmetrize(f)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Line search
        best_alpha, best_val = fw_line_search(f, g_sym)

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g_sym
            f = symmetrize(f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:

                best_f = f.copy()

def peak_coordinate_descent(a0, time_limit):
    """Coordinate descent targeting top peaks with mass transfer."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = symmetrize(a)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_scale = 0.003
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_peaks = min(5, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved_this_round = False

        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if js > je:
                continue

            # Find top contributing positions
            contrib = {}
            for j in range(js, je + 1):
                val = a[j] * a[k - j]
                if val > 1e-15:
                    contrib[j] = val
                    if j != k - j:
                        contrib[k - j] = max(contrib.get(k - j, 0), val)

            if not contrib:
                continue

            # Sort by contribution
            sorted_contrib = sorted(contrib.items(), key=lambda x: -x[1])

            for (j_src, _) in sorted_contrib[:8]:
                if a[j_src] < 1e-7:
                    continue

                # Try moving mass to a position that doesn't contribute to this peak
                # and doesn't create new peaks
                for _attempt in range(3):
                    j_target = int(rng.integers(0, n))
                    # Check j_target doesn't contribute to peak k
                    if j_target >= js and j_target <= je and (k - j_target) >= 0 and (k - j_target) < n:
                        continue
                    if j_target == j_src:
                        continue

                    for delta_frac in [step_scale, step_scale * 0.5, step_scale * 1.5]:
                        delta = a[j_src] * delta_frac
                        candidate = a.copy()
                        candidate[j_src] -= delta
                        # Also move the symmetric partner
                        j_src_sym = n - 1 - j_src
                        j_target_sym = n - 1 - j_target
                        if j_src_sym != j_src:
                            candidate[j_src_sym] -= delta
                        if j_target_sym != j_target:
                            candidate[j_target_sym] += delta
                        candidate[j_target] += delta

                        candidate = np.maximum(candidate, 0)
                        candidate = symmetrize(candidate)
                        candidate = project_simplex(candidate)
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-15:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved_this_round = True
                            break
                    if improved_this_round:
                        break

            if improved_this_round:
                break

        if not improved_this_round:
            step_scale *= 0.92
            if step_scale < 1e-7:
                break
            if it % 500 == 0:
                noise = rng.standard_normal(n) * step_scale * 3

    return best_a, best_obj


def resample_sequence(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def generate_starts(n, parent, rng):

    x = np.arange(n, dtype=np.float64)

        p = symmetrize(p)
        p = project_simplex(p)
        starts.append(p)

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper (symmetric)
    cos_t = 0.5 * (1 + np.cos(2 * np.pi * x / n))

    starts.append(cos_t / cos_t.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)

    g2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.06 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Triangular (inverted)
    tri = 1.0 - 2.0 * np.abs(x - (n - 1) / 2) / (n - 1)
    tri = np.maximum(tri, 0.01)
    starts.append(tri / tri.sum())

    # Random Dirichlet (symmetrized)
    for _ in range(2):
        r = rng.dirichlet(np.ones(n) * 0.5)
        r = symmetrize(r)
        r = project_simplex(r)
        starts.append(r)

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak coordinate descent (200s) ===
    print("\n=== Phase 4: Peak Coordinate Descent ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_alloc = min(remaining, 150)
            print(f"  Peak coord descent, t={t_alloc:.0f}s")
            a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_coord")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final alternating polish (remaining) ===
    print("\n=== Phase 5: Final Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Alternate FW and subgrad with tiny perturbations
        for ri in range(8):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if ri % 3 == 0:
                noise = rng.standard_normal(n_best) * 0.0005
                start = project_simplex(best_seq + noise)
                start = symmetrize(start)
                start = project_simplex(start)
                t_alloc = min(remaining, 30)
                a_opt, _ = fw_optimize(start, time_limit=t_alloc)
                try_seq(a_opt, f"polish_fw_{ri}")
            elif ri % 3 == 1:
                noise = rng.standard_normal(n_best) * 0.0003
                start = project_simplex(best_seq + noise)
                start = symmetrize(start)
                start = project_simplex(start)
                t_alloc = min(remaining, 20)
                a_opt, _ = subgradient_optimize(start, time_limit=t_alloc,
                                                 T_init=0.0002, lr_init=0.003,
                                                 top_k_frac=0.02)
                try_seq(a_opt, f"polish_sg_{ri}")
            else:
                t_alloc = min(remaining, 20)
                a_opt, _ = peak_coordinate_descent(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_cd_{ri}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
