The inner problem is: min a^T M(p) a subject to a >= 0, sum(a) = 1.

This is a QP with a specific structure (M(p) is a Hankel matrix). I can solve it using:
1. Projected gradient (what the previous code does)
2. Active set method
3. Interior point method
4. The KKT conditions directly

For the KKT approach:
- At the optimum, for i in the active set S (where a[i] > 0): (2 M(p) a)[i] + lambda = 0 (for the sum constraint) and (2 M(p) a)[i] + lambda - mu[i] = 0 (for the non-negativity, with mu[i] = 0).
- For i not in S: (2 M(p) a)[i] + lambda >= 0.

This is essentially a **semidefinite** or **quadratic** program that can be solved with active set methods.

A simple approach: 
1. Start with S = {all indices}.
2. Solve the equality-constrained QP: min a^T M a s.t. sum(a) = 1, a[i] = 0 for i not in S.
   This gives a[i] = (M_S^{-1} 1)_i / (1^T M_S^{-1} 1) for i in S.
3. Check if all a[i] in S are positive. If not, remove the most negative one from S.
4. Repeat until all are positive.

This is the active set method for QP. For n=768, each step involves inverting a matrix of size |S|, which could be expensive. But for the game-theoretic method, we might not need exact solutions - a few projected gradient steps should suffice.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def score_norm(a):
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max()


def build_lp_indices(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_oracle(f, M, n, row_idx, col_idx):
    c_vec = -np.ones(n)
    nnz = len(row_idx)
    vals = np.empty(nnz)
    for idx in range(nnz):
        k = row_idx[idx]
        i = col_idx[idx]
        jit = k - i
        if 0 <= jit < n:
            vals[idx] = f[jit]
        else:
            vals[idx] = 0.0
    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_mat = A_mat.tocsr()
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_mat, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-7})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, seed=42):

    s = f.sum()

    n = len(f)
    row_idx, col_idx = build_lp_indices(n)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:

        if g is None:

                g = project_simplex(f + rng.standard_normal(n) * 0.005)
                m_relax = 1.0

                m_relax = 1.0

            continue

        best_alpha = 0.0

        mid = (lo + hi) / 2.0

def game_iterate(a0, time_limit, T_init=0.001, inner_steps=8, lr_init=0.02, seed=42):
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    p = np.ones(2 * n - 1) / (2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Inner: projected gradient on a^T M(p) a
        a_inner = a.copy()
        for _ in range(inner_steps):
            grad = np.zeros(n)
            for j in range(n):
                start = 0
                end = n
                pseg = p[j:j + n]
                grad[j] = 2.0 * np.dot(a_inner, pseg)
            candidate = project_simplex(a_inner - lr * grad)
            new_obj = np.dot(p, fast_convolve(candidate))
            old_obj = np.dot(p, fast_convolve(a_inner))
            if new_obj < old_obj - 1e-16:
                a_inner = candidate
            else:
                lr *= 0.5
                candidate = project_simplex(a_inner - lr * grad)
                new_obj2 = np.dot(p, fast_convolve(candidate))
                if new_obj2 < old_obj - 1e-16:
                    a_inner = candidate
                break
        a = a_inner
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.05, lr_init * 2.0)
        else:
            no_improve += 1
            lr *= 0.97

        if no_improve > 50:
            T *= 1.05
            lr = lr_init * 0.8
        if no_improve > 150:
            T *= 0.7
            lr = lr_init
        if no_improve > 400:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit, seed=42):

            break
        improved = False

        # Random pair move
        for _ in range(20):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.4]:
                candidate = a.copy()
                amt = candidate[i] * frac
                if amt < 1e-15:

                ss = candidate.sum()
                if ss < 1e-10:

                candidate = candidate / ss
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    break

        # Random block perturbation
        if not improved:
            block_size = int(rng.integers(3, min(20, n // 4)))
            start = int(rng.integers(0, n - block_size))
            pert = rng.standard_normal(block_size) * 0.005

                if c.max() < best_obj - 1e-15:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True

        if not improved and it > 5000:
            break

                                              T_init=0.0008, lr_init=0.02, seed=n_try)
            best_seq, best_score = try_seq(a_opt2, f"sg_n{n_try}", best_seq, best_score)

    print(f"  Phase 1 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW-LP on best n ===
    print("\n=== Phase 2: Deep FW-LP ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 150:
        t_alloc = min(remaining * 0.30, 400)
        print(f"  Deep FW-LP n={n_best}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, seed=12345)
        best_seq, best_score = try_seq(a_opt, "deep_fw", best_seq, best_score)

    # Perturbed restarts
    for scale in [0.002, 0.005, 0.01, 0.02]:
        remaining = deadline - time.monotonic()
        if remaining < 100:
            break
        n_best = len(best_seq)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        t_alloc2 = min(remaining * 0.10, 60)
        a_opt2, _ = fw_lp_optimize(f_pert, time_limit=t_alloc2, seed=int(scale * 10000))
        best_seq, best_score = try_seq(a_opt2, f"fw_pert_{scale}", best_seq, best_score)

    print(f"  Phase 2 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Game-theoretic refinement ===
    print("\n=== Phase 3: Game- theoretic ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 100:
        t_alloc = min(remaining * 0.20, 150)
        print(f"  Game iter, t={t_alloc:.0f}s")
        a_opt, _ = game_iterate(best_seq, time_limit=t_alloc, T_init=0.0008,
                                inner_steps=10, lr_init=0.015, seed=777)
        best_seq, best_score = try_seq(a_opt, "game", best_seq, best_score)

    remaining = deadline - time.monotonic()
    if remaining > 80:
        n_best = len(best_seq)
        for T_init in [0.0005, 0.0015, 0.003]:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
            a_opt, _ = game_iterate(a_pert, time_limit=min(remaining * 0.20, 50),
                                    T_init=T_init, inner_steps=6, lr_init=0.01, seed=int(T_init * 1e6))
            best_seq, best_score = try_seq(a_opt, f"game_T{T_init}", best_seq, best_score)

    # Smoothed gradient with fine T
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_alloc = min(remaining * 0.15, 60)
        a_opt, _ = smoothed_grad_descent(best_seq, time_limit=t_alloc,
                                         T_init=0.0003, lr_init=0.01, seed=555)
        best_seq, best_score = try_seq(a_opt, "smooth_fine", best_seq, best_score)

    print(f"  Phase 3 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Alternate n and structured inits ===
    print("\n=== Phase 4: Alternate n & structures ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    # Try neighboring n values
    neighbor_ns = [n_best - 32, n_best + 32, n_best - 64, n_best + 64,
                   n_best // 2, n_best * 2]
    neighbor_ns = sorted(set([n for n in neighbor_ns if 128 <= n <= 4096]))
    for n_try in neighbor_ns:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        if n_try == n_best:
            continue
        x_new = np.linspace(0, 1, n_try)
        x_old = np.linspace(0, 1, n_best)
        start = np.interp(x_new, x_old, best_seq)
        start = np.maximum(start, 0)
        ss = start.sum()
        if ss > 1e-10:
            start = start / ss
        t_alloc = min(remaining * 0.10, 40)
        a_opt, _ = fw_lp_optimize(start, time_limit=t_alloc, seed=n_try + 100)
        best_seq, best_score = try_seq(a_opt, f"alt_n{n_try}", best_seq, best_score)

    # Structured initializations blended with best
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_best = len(best_seq)
        x = np.arange(n_best, dtype=np.float64)
        structures = []
        # Cosine window
        structures.append(np.cos(np.pi * x / (n_best - 1 + 1e-10))**2)
        # Exponential
        structures.append(np.exp(-0.01 * x))
        # Two-peak
        g1 = np.exp(-0.5 * ((x - 0.3 * n_best) / (0.1 * n_best))**2)
        g2 = np.exp(-0.5 * ((x - 0.7 * n_best) / (0.15 * n_best))**2)
        structures.append(g1 + g2)
        # Triangle
        t_struct = 1.0 - np.abs(x - n_best / 2) / (n_best / 2)
        structures.append(np.maximum(t_struct, 0.0))

        for si, struct in enumerate(structures):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            for alpha in [0.2, 0.5, 0.8]:
                blend = (1 - alpha) * best_seq + alpha * struct / struct.sum()
                blend = project_simplex(blend)
                t_alloc = min(remaining * 0.06, 15)
                a_opt, _ = fw_lp_optimize(blend, time_limit=t_alloc, seed=si * 100 + int(alpha * 10))
                best_seq, best_score = try_seq(a_opt, f"struct{si}_a{alpha}", best_seq, best_score)

    print(f"  Phase 4 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Intensive coordinate descent + final polish ===
    print("\n=== Phase 5: Coordinate Descent & Polish ===")
    t_phase = time.monotonic()
    n_best = len(best_seq)
    remaining = deadline - time.monotonic()

    if remaining > 40:
        t_alloc = min(remaining * 0.25, 100)
        print(f"  Coord descent, t={t_alloc:.0f}s")
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_alloc, seed=99999)
        best_seq, best_score = try_seq(a_opt, "coord", best_seq, best_score)

    # Very fine game iteration
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.20, 50)
        a_opt, _ = game_iterate(best_seq, time_limit=t_alloc, T_init=0.0003,
                                inner_steps=5, lr_init=0.005, seed=31415)
        best_seq, best_score = try_seq(a_opt, "game_fine", best_seq, best_score)

    # Final FW-LP
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_alloc = min(remaining * 0.25, 30)
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, seed=27182)
        best_seq, best_score = try_seq(a_opt, "final_fw", best_seq, best_score)

    # Micro perturbations + smoothed grad
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for _ in range(5):
            remaining = deadline - time.monotonic()
            if remaining < 4:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smoothed_grad_descent(f_pert, time_limit=min(remaining * 0.15, 5),
                                             T_init=0.0002, lr_init=0.003, seed=int(_ * 1000))
            best_seq, best_score = try_seq(a_opt, f"micro_{_}", best_seq, best_score)

    # Another coord descent pass
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt, _ = coordinate_descent(best_seq, time_limit=min(remaining * 0.3, 20), seed=42424)
        best_seq, best_score = try_seq(a_opt, "coord2", best_seq, best_score)

    print(f"  Phase 5 best: {best_score:.8f} (elapsed {time.monotonic()-t_phase:.0f}s)")

    # === Phase 6: Final deep optimization on absolute best ===
    print("\n=== Phase 6: Final Deep Run ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_final = len(best_seq)
        t_alloc = min(remaining * 0.6, remaining - 10)
        print(f"  Final FW-LP n={n_final}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc, seed=16180)
        best_seq, best_score = try_seq(a_opt, "final_deep", best_seq, best_score)

        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_alloc2 = min(remaining * 0.5, remaining - 5)
            a_opt2, _ = game_iterate(best_seq, time_limit=t_alloc2, T_init=0.0002,
                                     inner_steps=8, lr_init=0.005, seed=57721)
            best_seq, best_score = try_seq(a_opt2, "final_game", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
