"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a homotopy-based smooth optimization
approach with L-BFGS in log-softmax space, followed by exact subgradient polishing
and peak-targeted coordinate descent.

Key ideas:
1. Reparameterize a = softmax(x) to handle non-negativity and normalization
2. Use log-sum-exp smooth approximation of max with gradually increasing T
3. L-BFGS for the smooth subproblem
4. Exact subgradient for the final non-smooth problem
5. Coordinate descent for peak reduction
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def logsumexp_conv_objective(x, T, a_cache=None):
    """
    Compute the log-sum-exp smooth approximation of max(conv(softmax(x)))
    and its gradient with respect to x.
    """
    # Compute softmax
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    n = len(a)
    
    # Compute convolution
    conv = fast_convolve(a)
    
    # Log-sum-exp
    Tconv = T * conv
    Tconv_max = Tconv.max()
    shifted = Tconv - Tconv_max
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    f = Tconv_max / T + np.log(sum_exp) / T
    
    # Weights
    weights = exp_shifted / sum_exp
    
    # Gradient of f w.r.t. a: grad_a[j] = 2 * sum_k weights[j+k] * a[k]
    # This is the cross-correlation of weights and a
    grad_a = 2.0 * np.correlate(weights, a, mode='full')[:n]
    
    # Chain rule: grad_x = a * (grad_a - dot(grad_a, a))
    Sa = np.dot(grad_a, a)
    grad_x = a * (grad_a - Sa)
    
    return f, grad_x


def run_smooth_optimization(a0, T_levels, time_limit, verbose=False):
    """
    Run L-BFGS optimization with gradually increasing T.
    a0 is the starting point (on the simplex).
    """
    x0 = np.log(a0 + 1e-30)  # Inverse of softmax (approximately)
    # Actually, if a = softmax(x), then x = log(a) + constant. 
    # We can just use x0 = log(a0) as a starting point.
    
    best_a = a0.copy()
    best_score = fast_convolve(a0).max()
    
    t0 = time.monotonic()
    
    for T in T_levels:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # Define the objective for scipy
        def obj_and_grad(x):
            f, g = logsumexp_conv_objective(x, T)
            return f, g
        
        # Run L-BFGS
        try:
            res = minimize(obj_and_grad, x0, method='L-BFGS-B',
                          jac=True, options={'maxiter': 500, 'maxfun': 2000,
                                             'gtol': 1e-8, 'maxls': 50})
            if res.success or res.fun < obj_and_grad(x0)[0]:
                x_new = res.x
                a_new = np.exp(x_new - x_new.max())
                a_new = a_new / a_new.sum()
                score = fast_convolve(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                x0 = x_new
        except Exception as e:
            if verbose:
                print(f"  L-BFGS failed at T={T}: {e}")
            # Try from current x0 with smaller maxiter
            pass
        
        if verbose:
            print(f"  T={T:.1f}, score={best_score:.10f}, time={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score


def exact_subgradient_optimize(a0, time_limit, verbose=False):
    """
    Exact subgradient descent with backtracking line search.
    Minimizes max(conv(a)) subject to a >= 0, sum(a) = 1.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    lr = 0.01
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find the (or a) peak
        peak_idx = int(np.argmax(conv))
        
        # Subgradient
        grad = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                grad[j] = 2.0 * a[k]
        
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(20):
            candidate = project_simplex(a - alpha * grad)
            cand_val = fast_convolve(candidate).max()
            if cand_val < max_val - 1e-12:
                a = candidate
                improved = True
                if cand_val < best_obj - 1e-15:
                    best_obj = cand_val
                    best_a = a.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                break
            alpha *= 0.5
        
        if not improved:
            stagnant += 1
            # Adaptive: if stuck, try a different peak or perturb
            if stagnant > 50:
                # Try the second worst peak
                conv_sorted = np.sort(conv)[::-1]
                if len(conv_sorted) > 1 and conv_sorted[1] > conv_sorted[0] * 0.9:
                    # Multiple peaks, average subgradient
                    top_k = np.argsort(conv)[::-1][:min(5, n)]
                    for ki in top_k:
                        for j in range(n):
                            k = ki - j
                            if 0 <= k < n:
                                grad[j] += 2.0 * a[k]
                    grad /= len(top_k)
                    # Retry with this averaged gradient
                    alpha = lr * 0.5
                    for _ in range(20):
                        candidate = project_simplex(a - alpha * grad)
                        cand_val = fast_convolve(candidate).max()
                        if cand_val < max_val - 1e-12:
                            a = candidate
                            if cand_val < best_obj - 1e-15:
                                best_obj = cand_val
                                best_a = a.copy()
                            stagnant = 0
                            break
                        alpha *= 0.5
        
        if stagnant > 100:
            # Perturb and restart
            rng = np.random.default_rng(it * 7919)
            scale = 0.001 + 0.002 * min(1.0, (stagnant - 100) / 100.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            stagnant = 0
            lr = 0.01
    
    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent targeting the single worst peak.
    For the peak at position k*, try to reduce it by moving mass
    from high-contributing coordinates to low-contributing ones.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    step = 0.001
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        conv = fast_convolve(a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]
        
        # Find contributing coordinates
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j]
        # For j in [max(0, peak_idx-n+1), min(n-1, peak_idx)]
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        
        # Compute contributions
        contributions = []
        for j in range(j_start, j_end + 1):
            k = peak_idx - j
            contributions.append((a[j] * a[k], j, k))
        contributions.sort(reverse=True)
        
        # Try to move mass from top contributors to non-contributing positions
        improved = False
        for contrib_val, j, k in contributions[:10]:
            if contrib_val < 1e-12:
                break
            
            # Target positions that don't contribute to this peak
            for _ in range(3):
                rng = np.random.default_rng((it * 9973 + j * 31 + peak_idx) % (2**31))
                target = int(rng.integers(0, n))
                # Check if target doesn't contribute to the peak
                if j_start <= target <= j_end:
                    continue
                
                # Try moving a small fraction of a[j] or a[k] to target
                for src in [j, k]:
                    if a[src] < 1e-10:
                        continue
                    for frac in [step, step * 0.5, step * 2]:
                        delta = a[src] * frac
                        if delta < 1e-14:
                            continue
                        candidate = a.copy()
                        candidate[src] -= delta
                        candidate[target] += delta
                        if candidate[src] < -1e-12:
                            continue
                        candidate = np.maximum(candidate, 0)
                        candidate = candidate / candidate.sum()
                        val = fast_convolve(candidate).max()
                        if val < best_obj - 1e-15:
                            a = candidate
                            best_obj = val
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            step *= 0.95
            if step < 1e-8:
                # Try a global perturbation
                rng = np.random.default_rng(it * 48271)
                cand = project_simplex(a + rng.standard_normal(n) * 0.002)
                val = fast_convolve(cand).max()
                if val < best_obj - 1e-15:
                    a = cand
                    best_obj = val
                    best_a = a.copy()
                    step = 0.001
                elif it > 5000:
                    break
            if verbose and it % 1000 == 0:
                print(f"    CoordDescent it={it}, obj={best_obj:.10f}, step={step:.2e}")
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s
