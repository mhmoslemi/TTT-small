import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def hankel_mv(p, a):
    """Compute sum_k p[k] * a[k-j] for all j."""
    n = len(a)
    result = np.zeros(n)
    for k in range(len(p)):
        if p[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        result[js:je+1] += p[k] * a[k - (np.arange(js, je+1))]
    return result

def subgrad_from_peaks(a, top_k=20):

    return sg, conv

def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': 20})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def kkt_newton(a0, time_limit, peak_tol=1e-4, sup_tol=1e-10):
    """
    KKT Newton method for the auto-convolution peak minimization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    outer_it = 0
    
    print(f"    KKT-Newton starting, n={n}, obj={best_obj:.10f}")
    
    while time.monotonic() - t0 < time_limit:
        outer_it += 1
        if outer_it % 50 == 0:
            print(f"    KKT it={outer_it}, obj={best_obj:.10f}, elapsed={time.monotonic()-t0:.0f}s")
        
        conv = fast_convolve(a)
        t = conv.max()
        
        # Identify active peaks
        peak_threshold = t * (1.0 - peak_tol)
        P_list = np.where(conv >= peak_threshold)[0]
        if len(P_list) == 0:
            P_list = np.array([np.argmax(conv)])
        p = len(P_list)
        P_set = set(P_list.tolist())
        
        # Identify support
        S_list = np.where(a > sup_tol)[0]
        if len(S_list) == 0:
            S_list = np.array([np.argmax(a)])
        s_cnt = len(S_list)
        S_set = set(S_list.tolist())
        
        if p > 100 or s_cnt > 800:
            # Too many active constraints, use fewer
            # Keep top 50 peaks
            if p > 50:
                P_list = np.argpartition(conv, -50)[-50:]
                P_list = np.sort(P_list)
                p = len(P_list)
                P_set = set(P_list.tolist())
            # Keep top 700 support
            if s_cnt > 700:
                S_list = np.argpartition(a, -700)[-700:]
                S_list = np.sort(S_list)
                s_cnt = len(S_list)
                S_set = set(S_list.tolist())
        
        # Initialize mu and lambda
        mu = np.ones(p) / p
        # lambda = average of sum_k mu[k]*a[k-j] over j in S
        lam = 0.0
        for j in S_list:
            val = 0.0
            for idx, k in enumerate(P_list):
                diff = k - j
                if 0 <= diff < n:
                    val += mu[idx] * a[diff]
            lam += val
        lam /= s_cnt
        
        # Newton iteration
        newton_success = False
        t_val = t
        
        for newton_it in range(100):
            x_a = a[S_list].copy()  # values of a on support
            x_mu = mu.copy()
            
            # Compute residuals
            # R_k: (a*a)[k] - t_val for k in P
            R_P = np.zeros(p)
            for ki in range(p):
                k = P_list[ki]
                val = 0.0
                for j in S_list:
                    diff = k - j
                    if 0 <= diff < n:
                        val += a[j] * a[diff]
                R_P[ki] = val - t_val
            
            # R_S: sum_k mu[k]*a[k-j] - lambda for j in S
            R_S = np.zeros(s_cnt)
            for si in range(s_cnt):
                j = S_list[si]
                val = 0.0
                for ki in range(p):
                    k = P_list[ki]
                    diff = k - j
                    if 0 <= diff < n:
                        val += mu[ki] * a[diff]
                R_S[si] = val - lam
            
            R_3 = a[S_list].sum() - 1.0
            R_4 = mu.sum() - 1.0
            
            F = np.concatenate([R_P, R_S, [R_3, R_4]])
            F_norm = np.max(np.abs(F))
            
            if F_norm < 1e-12:
                newton_success = True
                break
            
            # Compute Jacobian
            dim = s_cnt + p + 2
            J = np.zeros((dim, dim))
            
            # Block (0, :s_cnt): A_block [p, s_cnt]
            for ki in range(p):
                k = P_list[ki]
                for si in range(s_cnt):
                    j = S_list[si]
                    diff = k - j
                    if 0 <= diff < n:
                        J[ki, si] = 2.0 * a[diff]
            
            # Block (0, s_cnt : 2*s_cnt): zeros (dR_P/dmu = 0)
            # Block (0, 2*s_cnt): -1 (dR_P/dt)
            J[:p, 2 * s_cnt] = -1.0
            # Block (0, 2*s_cnt+1): 0 (dR_P/dlambda)
            
            # Block (p : p+s_cnt, :s_cnt): C_block
            for si in range(s_cnt):
                j = S_list[si]
                for si2 in range(s_cnt):
                    j2 = S_list[si2]
                    k_val = j + j2
                    if k_val in P_set:
                        ki = np.searchsorted(P_list, k_val)
                        if ki < p and P_list[ki] == k_val:
                            J[p + si, si2] = mu[ki]
            
            # Block (p : p+s_cnt, s_cnt : 2*s_cnt): D_block
            for si in range(s_cnt):
                j = S_list[si]
                for ki in range(p):
                    k = P_list[ki]
                    diff = k - j
                    if 0 <= diff < n:
                        J[p + si, s_cnt + ki] = a[diff]
            
            # Block (p : p+s_cnt, 2*s_cnt+1): -1 (dR_S/dlambda)
            J[p:p+s_cnt, 2 * s_cnt + 1] = -1.0
            
            # Row s_cnt + p: sum(a[S]) - 1
            J[s_cnt + p, :s_cnt] = 1.0
            # Row s_cnt + p + 1: sum(mu) - 1
            J[s_cnt + p + 1, s_cnt:2*s_cnt] = 1.0
            
            # Solve J dx = -F
            try:
                # Add small regularization for stability
                reg = 1e-10 * np.eye(dim)
                dx = np.linalg.solve(J + reg, -F)
            except np.linalg.LinAlgError:
                try:
                    dx = np.linalg.lstsq(J + reg, -F, rcond=None)[0]
                except:
                    break
            
            # Line search
            step = 1.0
            for _ in range(20):
                # Compute new x
                new_a_S = x_a + step * dx[:s_cnt]
                new_mu = x_mu + step * dx[s_cnt:2*s_cnt]
                new_t = t_val + step * dx[2*s_cnt]
                new_lam = lam + step * dx[2*s_cnt + 1]
                
                # Check if step improves
                # Build new a
                new_a = np.zeros(n)
                new_a[S_list] = np.maximum(new_a_S, 0.0)
                ns = new_a.sum()
                if ns < 1e-15:
                    step *= 0.5
                    continue
                new_a = new_a / ns
                new_conv = fast_convolve(new_a).max()
                
                # Improvement check
                if new_conv < best_obj * (1.0 + 1e-10) and new_t >= 0:
                    a = new_a
                    mu = np.maximum(new_mu, 0.0)
                    mu_sum = mu.sum()
                    if mu_sum > 1e-15:
                        mu = mu / mu_sum
                    t_val = new_t if new_t > 0 else best_obj
                    lam = new_lam
                    newton_success = True
                    break
                else:
                    step *= 0.5
            
            if step < 1e-15:
                break
        
        # Update best
        conv = fast_convolve(a)
        obj = conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
        
        # Check convergence
        if not newton_success:
            # Try relaxing active set
            peak_tol *= 2
            if peak_tol > 0.1:
                peak_tol = 0.01
                # Perturb and restart
                rng = np.random.default_rng(outer_it * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                best_obj = fast_convolve(a).max()
                best_a = a.copy()
        
        # Check if we're at a KKT point
        conv = fast_convolve(a)
        t = conv.max()
        # Check dual feasibility and complementary slackness approximately
        if best_obj < t * (1.0 + 1e-10):
            # We're close to optimum for this active set
            break
    
    return best_a, best_obj

def try_and_update(a, best_state, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()

        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

def interp_sequence(parent, n_target):
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute sum_k p[k] * a[k-j] for all j (valid indices only)."""
    n = len(a)
    result = np.zeros(n)
    for k in range(len(p)):
        if p[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        result[js:je + 1] += p[k] * a[k - idx]
    return result


def subgrad_from_peaks(a, top_k=20):
    """Compute subgradient using top-k peaks of auto-convolution."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        weights = np.ones(len(peaks)) / len(peaks)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        k = int(peaks[i])
        w = weights[i]
        if w < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w * a[k - idx]
    return sg, conv

def solve_lp_oracle(f, M, timeout=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def kkt_newton(a0, time_limit, init_peak_tol=0.005):
    """
    KKT Newton method for minimizing max(auto-convolution(a,a)).
    Solves the KKT system:
      (a*a)[k] = t for k in P (active peaks)
      sum_k mu[k]*a[k-j] = lambda for j in S (support)
      sum(a) = 1, sum(mu) = 1
    using damped Newton with line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s0
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    outer_it = 0
    peak_tol = init_peak_tol
    max_s = min(n, 600)
    max_p = min(2 * n - 1, 60)

    print(f"    KKT-Newton start: n={n}, obj={best_obj:.10f}")

    while time.monotonic() - t0 < time_limit:
        outer_it += 1
        if outer_it % 20 == 0:
            rem = time_limit - (time.monotonic() - t0)
            print(f"    KKT it={outer_it}, obj={best_obj:.10f}, rem={rem:.0f}s")

        conv = fast_convolve(a)
        t_max = conv.max()

        # Identify active peaks
        threshold = t_max * (1.0 - peak_tol)
        P_list = np.where(conv >= threshold)[0]
        if len(P_list) > max_p:
            P_list = np.sort(np.argpartition(conv, -max_p)[-max_p:])
        elif len(P_list) < 2:
            P_list = np.array([int(np.argmax(conv))])
        p = len(P_list)
        P_set = set(P_list.tolist())

        # Identify support
        sup_threshold = max(1e-12, a.max() * 1e-6)
        S_list = np.where(a > sup_threshold)[0]
        if len(S_list) > max_s:
            S_list = np.sort(np.argpartition(a, -max_s)[-max_s:])
        elif len(S_list) < 1:
            S_list = np.array([int(np.argmax(a))])
        s_cnt = len(S_list)

        # Build index mappings
        mu_lookup = np.full(2 * n - 1, -1, dtype=int)
        for ki in range(p):
            mu_lookup[P_list[ki]] = ki

        # Initialize mu uniformly
        mu = np.ones(p) / p
        # Initialize lambda
        lam = 0.0
        for si in range(s_cnt):
            j = S_list[si]
            val = 0.0
            for ki in range(p):
                diff = P_list[ki] - j
                if 0 <= diff < n:
                    val += mu[ki] * a[diff]
            lam += val
        lam /= s_cnt
        t_val = t_max

        # Newton iteration on the KKT system
        newton_ok = False
        for nit in range(60):
            # Compute residuals
            # R_P[ki] = (a*a)[P_list[ki]] - t_val
            R_P = np.zeros(p)
            for ki in range(p):
                k = P_list[ki]
                js_start = max(0, k - n + 1)
                js_end = min(n - 1, k)
                if js_start > js_end:
                    R_P[ki] = -t_val
                    continue
                js_arr = np.arange(js_start, js_end + 1)
                R_P[ki] = np.dot(a[js_arr], a[k - js_arr]) - t_val

            # R_S[si] = sum_k mu[ki]*a[P[ki]-S[si]] - lam
            R_S = np.zeros(s_cnt)
            for si in range(s_cnt):
                j = S_list[si]
                val = 0.0
                for ki in range(p):
                    diff = P_list[ki] - j
                    if 0 <= diff < n:
                        val += mu[ki] * a[diff]
                R_S[si] = val - lam

            R_3 = a[S_list].sum() - 1.0
            R_4 = mu.sum() - 1.0

            dim = s_cnt + p + 2
            F = np.zeros(dim)
            F[:p] = R_P
            F[p:p + s_cnt] = R_S
            F[p + s_cnt] = R_3
            F[p + s_cnt + 1] = R_4

            F_norm = np.max(np.abs(F))
            if F_norm < 1e-13:
                newton_ok = True
                break

            # Build Jacobian (vectorized)
            J = np.zeros((dim, dim))

            # A_block[p, s_cnt]: dR_P[ki]/da[S[si]] = 2*a[P[ki]-S[si]] if valid
            diffs_A = P_list[:, None].astype(np.int64) - S_list[None, :].astype(np.int64)
            valid_A = (diffs_A >= 0) & (diffs_A < n)
            J[:p, :s_cnt] = np.where(valid_A, 2.0 * a[diffs_A[valid_A]], 0.0)
            # Need to handle the case where diffs_A is out of bounds carefully
            J[:p, :s_cnt] = np.zeros((p, s_cnt))
            idx_valid = np.where(valid_A)
            J[:p, :s_cnt][idx_valid] = 2.0 * a[diffs_A[idx_valid]]

            # dR_P/dt = -1
            J[:p, p + s_cnt] = -1.0

            # C_block[s_cnt, s_cnt]: dR_S[si]/da[S[si2]] = mu[S[si]+S[si2]] if in P
            sums_C = S_list[:, None].astype(np.int64) + S_list[None, :].astype(np.int64)
            mu_idx_C = mu_lookup[sums_C]
            valid_C = mu_idx_C >= 0
            J[p:p + s_cnt, :s_cnt] = np.zeros((s_cnt, s_cnt))
            idx_valid_C = np.where(valid_C)
            J[p:p + s_cnt, :s_cnt][idx_valid_C] = mu[mu_idx_C[idx_valid_C]]

            # D_block[s_cnt, p]: dR_S[si]/dmu[P[ki]] = a[P[ki]-S[si]] if valid
            diffs_D = P_list[None, :].astype(np.int64) - S_list[:, None].astype(np.int64)
            valid_D = (diffs_D >= 0) & (diffs_D < n)
            J[p:p + s_cnt, s_cnt:s_cnt + p] = np.zeros((s_cnt, p))
            idx_valid_D = np.where(valid_D)
            J[p:p + s_cnt, s_cnt:s_cnt + p][idx_valid_D] = a[diffs_D[idx_valid_D]]

            # dR_S/dlambda = -1
            J[p:p + s_cnt, p + s_cnt + 1] = -1.0

            # Row p+s_cnt: sum(a_S) = 1
            J[p + s_cnt, :s_cnt] = 1.0
            # Row p+s_cnt+1: sum(mu) = 1
            J[p + s_cnt + 1, s_cnt:s_cnt + p] = 1.0

            # Solve J dx = -F with regularization
            try:
                cond = np.linalg.cond(J)
                reg = 0.0 if cond < 1e12 else 1e-8 * np.eye(dim)
                dx = np.linalg.solve(J + reg, -F)
            except np.linalg.LinAlgError:
                try:
                    dx = np.linalg.lstsq(J + 1e-8 * np.eye(dim), -F, rcond=None)[0]
                except Exception:
                    break

            np.abs(dx).sort()
            if not np.all(np.isfinite
                         (dx)):
                break

            # Damped Newton with backtracking
            step = 1.0
            improved = False
            for _bt in range(25):
                new_a_S = a[S_list] + step * dx[:s_cnt]
                new_mu = mu + step * dx[s_cnt:s_cnt + p]
                new_t = t_val + step * dx[p + s_cnt]
                new_lam = lam + step * dx[p + s_cnt + 1]

                # Build new a (project non-negative, normalize)
                new_a = np.zeros(n)
                new_a[S_list] = np.maximum(new_a_S, 0.0)
                ns = new_a.sum()
                if ns < 1e-10:
                    step *= 0.5
                    continue
                new_a = new_a / ns
                new_obj = fast_convolve(new_a).max()

                # Accept if objective doesn't increase too much
                if new_obj <= best_obj * (1.0 + 1e-8) and new_t > 0:
                    a = new_a
                    mu = np.maximum(new_mu, 0.0)
                    mu_sum = mu.sum()
                    if mu_sum > 1e-15:
                        mu = mu / mu_sum
                    t_val = max(new_t, 1e-10)
                    lam = new_lam
                    newton_ok = True
                    improved = True
                    break
                else:
                    step *= 0.5

            if not improved:
                break

        # Update best
        conv = fast_convolve(a)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        # Convergence check: verify KKT conditions approximately
        conv = fast_convolve(a)
        t_cur = conv.max()
        if best_obj < t_cur * (1.0 + 1e-9):
            # Check if any non-active peak exceeds t
            P_check = set(P_list.tolist())
            violations = np.max([conv[k] for k in range(2*n-1) if k not in P_check]) if len(P_check) < 2*n-1 else 0
            if violations <= t_cur * (1.0 + 1e-6):
                break

        # If Newton failed, adjust active set
        if not newton_ok:
            # Try with wider peak tolerance
            peak_tol = min(peak_tol * 1.5, 0.05)
            if peak_tol > 0.03:
                # Perturb and restart
                rng = np.random.default_rng(outer_it * 104729 + 3)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                best_obj = fast_convolve(a).max()
                best_a = a.copy()
                peak_tol = init_peak_tol

    return best_a, best_obj


def try_and_update(a, best_state, label=""):
    """Evaluate and update best state."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_state
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_state
    best_seq, best_score = best_state
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return (best_seq, best_score)


def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s


def coordinate_ascent(a0, time_limit, max_outer=200):
    """Coordinate ascent: optimize one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    for outer in range(max_outer):
        if time.monotonic() - t0 > time_limit - 1:
            break
        improved = False
        # Cycle through coordinates (random order for variety)
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            # Compute base convolution (without a[j] contribution)
            base = fast_convolve(a)
            # The contribution of a[j] to conv[k] is:
            # conv[k] includes a[j]*a[k-j] and a[k-j]*a[j] (same term if k-j=j)
            # d conv[k] / d a[j] = 2*a[k-j] if k-j != j, or 2*a[j] if k=j*2
            # Actually conv[k] = ... + a[j]*a[k-j] + a[k-j]*a[j] + ...
            # If we set a[j]=0 and compute base, then:
            # conv[k] = base[k] + 2*a[j]*a[k-j] (if k-j != j and 0<=k-j<n)
            # or conv[k] = base[k] + a[j]^2 (if k=2*j)

            a_save = a[j]
            a[j] = 0.0
            base_conv = fast_convolve(a)
            a[j] = a_save

            # For new value x = a[j]:
            # conv[k] = base_conv[k] + 2*x*a[k-j] (for k where k-j is valid and k-j != j)
            # conv[k] = base_conv[k] + x^2 (for k = 2*j)
            # We want to minimize max_k conv[k] over x >= 0

            # The peak as a function of x is:
            # max( max_{k: k-j valid, k-j!=j} [base_conv[k] + 2*x*a[k-j]],
            #       base_conv[2*j] + x^2 (if 2*j < 2n-1),
            #       ...other terms... )

            # For a grid search over x:
            # x ranges from 0 to about 1 (since sum(a)=1)
            # Let's try a few values
            best_x = a_save
            best_max = base_conv.max() + 2 * a_save * 0  # current

            # Current max for reference
            cur_conv = base_conv.copy()
            # Add back a[j] contributions
            for k in range(2 * n - 1):
                diff = k - j
                if 0 <= diff < n:
                    if diff == j:
                        cur_conv[k] += a_save * a_save
                    else:
                        cur_conv[k] += 2 * a_save * a[diff]
            cur_max = cur_conv.max()

            # Try a few x values
            candidates = [0.0, a_save * 0.5, a_save * 0.75, a_save * 1.25, a_save * 1.5,
                          a_save + 0.001, max(0.0, a_save - 0.001)]
            for x in candidates:
                if x < 0:
                    continue
                test_conv = base_conv.copy()
                for k in range(2 * n - 1):
                    diff = k - j
                    if 0 <= diff < n:
                        if diff == j:
                            test_conv[k] += x * x
                        else:
                            test_conv[k] += 2 * x * a[diff]
                new_max = test_conv.max()
                if new_max < best_max - 1e-14:
                    best_max = new_max
                    best_x = x

            if best_x < a_save - 1e-12 or best_x > a_save + 1e-12:
                a[j] = best_x
                improved = True

        if not improved:
            break

        # Renormalize
        a = np.maximum(a, 0)
        a = a / a.sum()
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search:
    1. KKT Newton with active set management
    2. Frank-Wolfe with LP oracle
    3. Multi-start perturbations
    4. Length exploration
    5. Coordinate ascent refinement
    6. Final polishing
    """
    np.random.seed(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    # === Phase 1: Quick PGD warmup (60s) ===
    print("\n=== Phase 1: Quick PGD ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic() - 1300
    t_alloc = min(remaining, 60)
    a_opt, _ = pgd_refine(best_seq, time_limit=t_alloc, top_k=20, lr_init=0.03)
    (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "pgd_warm")
    print(f"  Phase 1: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 2: KKT Newton (main workhorse, 500s) ===
    print("\n=== Phase 2: KKT Newton ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic() - 700
    t_alloc = min(remaining, 450)
    print(f"  KKT allocation: {t_alloc:.0f}s")
    a_opt, _ = kkt_newton(best_seq, time_limit=t_alloc, init_peak_tol=0.005)
    (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "kkt_main")
    print(f"  Phase 2: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # Second KKT run with tighter tolerance
    remaining = deadline - time.monotonic() - 500
    if remaining > 30:
        t_alloc = min(remaining, 150)
        a_opt, _ = kkt_newton(best_seq, time_limit=t_alloc, init_peak_tol=0.002)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "kkt_tight")
        print(f"  KKT tight: {best_score:.8f}")

    # === Phase 3: FW-LP for global improvement (300s) ===
    print("\n=== Phase 3: FW-LP ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic() - 350
    t_alloc = min(remaining, 250)
    if t_alloc > 20:
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fw_lp")

    # Shorter FW for quick improvements
    remaining = deadline - time.monotonic() - 250
    if remaining > 20:
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(remaining, 80))
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "fw_lp2")
    print(f"  Phase 3: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Multi-start perturbations (250s) ===
    print("\n=== Phase 4: Multi-Start ===")
    t_start = time.monotonic()
    n_best = len(best_seq)
    rng = np.random.default_rng(seed)

    perturbation_scales = [0.001, 0.002, 0.003, 0.005, 0.008]
    for scale in perturbation_scales:
        remaining = deadline - time.monotonic() - 100
        if remaining < 15:
            break
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
        # Quick KKT on perturbed
        t_alloc = min(remaining, 40)
        a_opt, _ = kkt_newton(f_pert, time_limit=t_alloc, init_peak_tol=0.005)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"pert_{scale}")

    # Perturbed FW
    remaining = deadline - time.monotonic() - 60
    if remaining > 30:
        for scale in [0.003, 0.007]:
            remaining = deadline - time.monotonic() - 40
            if remaining < 20:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            a_opt, _ = fw_lp_optimize(f_pert, time_limit=min(remaining, 50))
            (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"pert_fw_{scale}")

    print(f"  Phase 4: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Length exploration (150s) ===
    print("\n=== Phase 5: Length Exploration ===")
    t_start = time.monotonic()
    n_best = len(best_seq)
    n_candidates = sorted(set([
        max(128, n_best - 64), max(128, n_best - 32),
        n_best + 32, n_best + 64,
        n_best // 2, n_best * 2
    ]))
    n_candidates = [n for n in n_candidates if 128 <= n <= 3000]

    for n_try in n_candidates:
        remaining = deadline - time.monotonic() - 100
        if remaining < 15:
            break
        start = interp_sequence(best_seq, n_try)
        t_alloc = min(remaining, 30)
        a_opt, _ = pgd_refine(start, time_limit=t_alloc * 0.5, top_k=15, lr_init=0.02)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), f"len_{n_try}")
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            a_opt2, _ = kkt_newton(a_opt, time_limit=min(remaining, t_alloc * 0.5), init_peak_tol=0.005)
            (best_seq, best_score) = try_and_update(a_opt2, (best_seq, best_score), f"len_kkt_{n_try}")

    print(f"  Phase 5: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Coordinate ascent + final polish (remaining) ===
    print("\n=== Phase 6: Final Polish ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    # Coordinate ascent
    remaining = deadline - time.monotonic() - 60
    if remaining > 20:
        a_opt, _ = coordinate_ascent(best_seq, time_limit=min(remaining, 60))
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "coord")

    # Smoothed gradient fine
    remaining = deadline - time.monotonic() - 40
    if remaining > 15:
        a_opt, _ = softmax_smoothed_grad(best_seq, time_limit=min(remaining, 40), T_init=0.0005, lr_init=0.005)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "smooth_fine")

    # Final KKT
    remaining = deadline - time.monotonic() - 20
    if remaining > 10:
        a_opt, _ = kkt_newton(best_seq, time_limit=min(remaining, 30), init_peak_tol=0.001)
        (best_seq, best_score) = try_and_update(a_opt, (best_seq, best_score), "kkt_final")

    # Symmetry check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    (best_seq, best_score) = try_and_update(sym_final, (best_seq, best_score), "sym_final")

    # Tiny perturbations
    remaining = deadline - time.monotonic() - 10
    while remaining > 5:
        noise = rng.standard_normal(len(best_seq)) * 0.0005
        f_pert = project_simplex(best_seq + noise)
        (best_seq, best_score) = try_and_update(f_pert, (best_seq, best_score), "micro_pert")
        remaining = deadline - time.monotonic() - 5

    print(f"  Phase 6: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
