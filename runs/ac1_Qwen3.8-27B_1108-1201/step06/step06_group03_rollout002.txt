        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    stagnant = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Solve LP
        if exact:
            g = solve_lp_exact(f, M)
            if g is None or g.sum() < 1e-15:
                g = solve_lp_relaxed(f, M, relax=1.001)

        if g is None or g.sum() < 1e-15:
            stagnant += 1
            if stagnant > 15:
                rng = np.random.default_rng(it * 7919 + 3)
                scale = 0.001 + 0.003 * min(1.0, stagnant / 40.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        g = g / g.sum()

        # Check if direction is meaningful
        if np.max(np.abs(g - f)) < 1e-12:
            stagnant += 1
            if stagnant > 25:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                stagnant = 0
            continue

        # Line search: coarse grid
        alphas = np.linspace(0.01, 1.0, 25)
        candidates = []
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            candidates.append((alpha, val))

        if not candidates:
            stagnant += 1
            continue

        # Parabolic refinement
        best_alpha = parabolic_line_search(f, g, M, candidates)

        # Evaluate the parabolic candidate and a few neighbors
        final_candidates = [(best_alpha, None)]
        for da in [-0.01, -0.005, 0.005, 0.01]:
            aa = np.clip(best_alpha + da, 0.0, 1.0)
            final_candidates.append((aa, None))
        # Add the grid best
        grid_best = min(candidates, key=lambda x: x[1])
        final_candidates.append((grid_best[0], grid_best[1]))

        best_val = np.inf
        for (aa, vv) in final_candidates:
            if vv is not None:
                if vv < best_val:
                    best_val = vv
                    best_alpha = aa
                continue
            ft = (1.0 - aa) * f + aa * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = aa

        if best_alpha > 1e-10 and best_val < M + 1e-14:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, obj={best_obj:.10f}, stagnant={stagnant}")

    return best_f, best_obj


def subgradient_optimize(a0, time_limit, T_init=0.0004, lr_init=0.02, top_k_frac=0.04):
    """Smoothed subgradient descent with momentum and adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(5, int(n * top_k_frac))

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.93

    t0 = time.monotonic()
    stagnant = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)

        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))

        pv = conv[peaks]
        pv_max = pv.max()
        # Use temperature-scaled softmax
        w = np.exp((pv - pv_max) / (T))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / w_sum

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] += w[i] * 2.0 * a[k - idx_arr]

        v = mom * v - lr * grad
        best_candidate = a
        best_val = conv.max()

        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            candidate = project_simplex(a + frac * v)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-15:
                best_val = val
                best_candidate = candidate

        a = best_candidate
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 15:
            T *= 1.08
            lr *= 0.96
        if stagnant > 50:
            T *= 0.5
            lr *= 1.4
            mom = 0.95
        if stagnant > 150:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            noise = 0.002 + 0.003 * min(1.0, (stagnant - 150) / 150.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise)
            T = T_init
            lr = lr_init
            v = np.zeros(n)
            mom = 0.93
            stagnant = 0

    return best_a, best_obj


def peak_surgery(a0, time_limit):
    """
    Targeted peak reduction using quadratic step sizing.
    For each top peak, find the best single-coordinate move to reduce it.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    step_scale = 0.003

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        n_peaks = min(6, len(conv))
        peak_indices = np.argsort(conv)[::-1][:n_peaks]

        improved_this_round = False

        for ki in range(n_peaks):
            k = int(peak_indices[ki])
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            if js > je:
                continue

            # Find top contributing pairs
            pair_vals = []
            for j in range(js, je + 1):
                pair_vals.append((a[j] * a[k - j], j, k - j))
            pair_vals.sort(reverse=True)

            # Try reducing top pairs by moving mass
            for (_, j1, j2) in pair_vals[:8]:
                if a[j1] < 1e-10 and a[j2] < 1e-10:
                    continue

                # For each source (j1 or j2), try moving to target locations
                for src in [j1, j2]:
                    if a[src] < 1e-10:
                        continue

                    # Find "safe" targets: positions far from peak-contributing positions
                    # A position p is "safe" for peak k if p and k-p are both small
                    # Simple heuristic: try positions that minimize a[p]*a[k-p]
                    target_scores = []
                    for p in range(n):
                        if p == src:
                            continue
                        # Score: how much would adding mass here increase other peaks?
                        # Approximate by current a[p] (lower is better - less interference)
                        target_scores.append((a[p], p))
                    target_scores.sort()

                    for (_, p) in target_scores[:10]:
                        # Quadratic step: find optimal delta
                        # Moving delta from src to p changes peak k by:
                        # -a[src]*a[k-src] + a[p_new]*a[k-p_new] approximately
                        # But more precisely, the change in conv[k] is:
                        # (a[src]-d)*(a[k-src]) + (a[p]+d)*(a[k-p]) - a[src]*a[k-src] - a[p]*a[k-p]
                        # = -d*a[k-src] + d*a[k-p] = d*(a[k-p] - a[k-src]) if k-src != k-p and k-p != src etc.
                        # This gets complicated with edge cases. Use empirical approach.

                        for delta_frac in [step_scale, step_scale * 0.5, step_scale * 2.0, step_scale * 0.25]:
                            delta = a[src] * delta_frac
                            if delta < 1e-12:
                                continue
                            candidate = a.copy()
                            candidate[src] -= delta
                            candidate[p] += delta
                            if candidate[src] < -1e-12:
                                continue
                            candidate = np.maximum(candidate, 0)
                            csum = candidate.sum()
                            if csum < 1e-15:
                                continue
                            candidate = candidate / csum
                            val = fast_convolve(candidate).max()
                            if val < best_obj - 1e-15:
                                a = candidate
                                best_obj = val
                                best_a = a.copy()
                                improved_this_round = True
                                break
                        if improved_this_round:
                            break
                    if improved_this_round:
                        break
                if improved_this_round:
                    break

            if improved_this_round:
                break

        if not improved_this_round:
            step_scale *= 0.92
            if step_scale < 1e-7:
                break
            # Periodic perturbation
            if it % 100 == 0:
                rng = np.random.default_rng(it * 48271)
                candidate = project_simplex(a + rng.standard_normal(n) * step_scale * 3)
                val = fast_convolve(candidate).max()
                if val < best_obj - 1e-15:
                    a = candidate
                    best_obj = val
                    best_a = a.copy()
            elif it > 3000:
                break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n == 1:
        return np.array([float(a[0])])
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def spectral_init(n, rng):
    """
    Generate an initial sequence by trying to flatten the power spectrum.
    Start with random non-negative, iteratively adjust to flatten |F|^2.
    """
    a = rng.dirichlet(np.ones(n) * 0.5)
    # A few steps of spectral flattening
    for _ in range(20):
        F = np.fft.rfft(a)
        power = np.abs(F)**2
        power_max = power.max()
        # Gradient of max power with respect to a
        # d|F[k]|^2/da = 2*Re(F[k] * conj(FFT(1 at position related to k)))
        # Simplified: just reduce the peak power by moving mass
        peak_k = np.argmax(power)
        # The contribution of a[j] to F[peak_k] is a[j]*exp(-2*pi*i*peak_k*j/n)
        # To reduce |F[peak_k]|, move mass from positions where a[j]*exp(...) is in-phase
        # to positions where it's out-of-phase
        phases = np.exp(-2j * np.pi * peak_k * np.arange(n) / n)
        # Real part of conjugate: positions where a[j]*phases[j] is positive add to power
        # Move mass from high real-part positions to low real-part positions
        real_parts = np.real(phases)
        # Sort: most positive first (sources), most negative first (targets)
        source_order = np.argsort(real_parts)[::-1]
        target_order = np.argsort(real_parts)

        moved = False
        for _ in range(5):
            src = source_order[np.random.randint(0, min(10, n))]
            tgt = target_order[np.random.randint(0, min(10, n))]
            if src == tgt:
                continue
            delta = a[src] * 0.05
            if delta < 1e-12:
                continue
            a[src] -= delta
            a[tgt] += delta
            if a[src] >= -1e-12:
                a = np.maximum(a, 0)
                a = a / a.sum()
                moved = True
            else:
                a[src] += delta
                a[tgt] -= delta
                break
        if not moved:
            break

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def generate_starts(n, parent, rng):
    """Generate diverse initializations."""
    starts = []

    if parent is not None:
        starts.append(resample_sequence(parent, n))

    # Uniform
    starts.append(np.ones(n) / n)

    # Cosine taper
    x = np.arange(n, dtype=np.float64)
    cos_t = np.cos(np.pi * x / (n - 1))
    w = 0.5 * (1 + cos_t)
    w = np.maximum(w, 0.01)
    starts.append(w / w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / (0.15 * n)) ** 2)
    starts.append(g / g.sum())

    # Edge concentration
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    starts.append(edge / edge.sum())

    # Two peaks
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.06 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.06 * n)) ** 2)
    starts.append((g1 + g2) / (g1 + g2).sum())

    # Spectral init
    starts.append(spectral_init(n, rng))

    # Random Dirichlet
    for _ in range(3):
        starts.append(rng.dirichlet(np.ones(n) * 0.3))

    # "Comb" pattern
    comb = np.ones(n) * 0.5
    comb[::2] = 1.0
    starts.append(comb / comb.sum())

    return starts


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search for minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Quick multi-n, multi-start exploration with FW (200s)
    2. Deep FW on best candidate with exact LP (450s)
    3. Subgradient refinement with multiple configs (250s)
    4. Peak surgery - targeted coordinate descent (400s)
    5. Final micro-polish with alternating methods (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(768) / 768
        try_seq(parent, "uniform_768")

    n_parent = len(parent)

    # === Phase 1: Quick exploration (200s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase = time.monotonic()

    lengths = sorted(set([n_parent, max(256, n_parent - 64), n_parent + 64,
                          512, 640, 768, 896, 1024, 1280]))
    lengths = [l for l in lengths if 192 <= l <= 2048]

    for n_try in lengths:
        remaining = deadline - time.monotonic() - 1300
        if remaining < 20:
            break

        starts = generate_starts(n_try, parent, rng)
        for si, start in enumerate(starts[:5]):
            remaining = deadline - time.monotonic() - 1280
            if remaining < 10:
                break
            t_alloc = min(remaining, 25)
            a_opt, _ = fw_optimize(start, time_limit=t_alloc)
            label = f"fw_n{n_try}_s{si}"
            try_seq(a_opt, label)

    print(f"  Phase 1 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Deep FW (450s) ===
    print("\n=== Phase 2: Deep FW ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        remaining = deadline - time.monotonic() - 1050
        if remaining > 50:
            t_alloc = min(remaining, 400)
            print(f"  Deep FW n={n_best}, t={t_alloc:.0f}s")
            a_opt, _ = fw_optimize(best_seq, time_limit=t_alloc, exact=True, verbose=True)
            try_seq(a_opt, "deep_fw")

        # Perturbed restarts at different scales
        for scale in [0.0005, 0.002, 0.005, 0.01, 0.02]:
            remaining = deadline - time.monotonic() - 700
            if remaining < 25:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(remaining, 50)
            a_opt, _ = fw_optimize(f_pert, time_limit=t_alloc, exact=True)
            try_seq(a_opt, f"fw_pert_{scale}")

    print(f"  Phase 2 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: Subgradient (250s) ===
    print("\n=== Phase 3: Subgradient ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        configs = [
            {'T_init': 0.0002, 'lr_init': 0.025, 'top_k_frac': 0.03},
            {'T_init': 0.0008, 'lr_init': 0.018, 'top_k_frac': 0.05},
            {'T_init': 0.002, 'lr_init': 0.012, 'top_k_frac': 0.08},
            {'T_init': 0.005, 'lr_init': 0.008, 'top_k_frac': 0.12},
        ]

        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 500
            if remaining < 20:
                break
            t_alloc = min(remaining, 65)
            print(f"  Subgrad cfg{ci} (T={cfg['T_init']}), t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"subgrad_{ci}")

    print(f"  Phase 3 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Peak Surgery (400s) ===
    print("\n=== Phase 4: Peak Surgery ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        remaining = deadline - time.monotonic() - 200
        if remaining > 30:
            t_alloc = min(remaining, 350)
            print(f"  Peak surgery, t={t_alloc:.0f}s")
            a_opt, _ = peak_surgery(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_surgery")

            # Second round of surgery on the result
            remaining = deadline - time.monotonic() - 150
            if remaining > 20:
                t_alloc = min(remaining, 100)
                a_opt2, _ = peak_surgery(a_opt, time_limit=t_alloc)
                try_seq(a_opt2, "peak_surgery_2")

    print(f"  Phase 4 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 5: Final micro-polish ===
    print("\n=== Phase 5: Final Micro-Polish ===")
    t_phase = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Alternating tiny FW and subgrad
        for ri in range(12):
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break

            if ri % 3 == 0:
                noise = rng.standard_normal(n_best) * 0.0004
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 20)
                a_opt, _ = fw_optimize(start, time_limit=t_alloc, exact=True)
                try_seq(a_opt, f"polish_fw_{ri}")
            elif ri % 3 == 1:
                noise = rng.standard_normal(n_best) * 0.0003
                start = project_simplex(best_seq + noise)
                t_alloc = min(remaining, 15)
                a_opt, _ = subgradient_optimize(start, time_limit=t_alloc,
                                                T_init=0.0001, lr_init=0.004,
                                                top_k_frac=0.02)
                try_seq(a_opt, f"polish_sg_{ri}")
            else:
                # Short peak surgery
                t_alloc = min(remaining, 15)
                a_opt, _ = peak_surgery(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"polish_ps_{ri}")

    print(f"  Phase 5 done: best={best_score:.10f} ({time.monotonic()-t_phase:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
