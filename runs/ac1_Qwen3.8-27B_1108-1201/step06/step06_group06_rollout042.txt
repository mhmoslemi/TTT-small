    """KL (entropy) mirror projection: minimize KL(x||a) + <v, x> s.t. x in simplex.

    shifted = v - v.max()
    exp_neg = np.exp(-shifted)
    x = a * exp_neg
    s = x.sum()
    if s < 1e-300:

    else:
        f = f / s
    n = len(f)

    if lp_oracle is None:
        lp_oracle = LPOracle(n)

    conv = fast_convolve(f)
    M = conv.max()
    best_M = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        g = lp_oracle.solve(f, M * m_exp, time_limit=min(remaining * 0.1, 4.0))
        if g is None:
            m_exp = min(m_exp * 1.5, 8.0)
            if m_exp >= 8.0:
                rng = np.random.default_rng(it * 104729 + 7)
                g = project_simplex(f + rng.standard_normal(n) * 0.005)
                m_exp = 1.0
            else:
                continue

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Refine with ternary search
        if 0 < best_alpha < 1:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(10):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.maximum((1 - m1) * f + m1 * g, 0)
                f1 = f1 / f1.sum()
                v1 = fast_convolve(f1).max()
                f2 = np.maximum((1 - m2) * f + m2 * g, 0)
                f2 = f2 / f2.sum()
                v2 = fast_convolve(f2).max()
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            mid = (lo + hi) / 2.0
            ft = np.maximum((1 - mid) * f + mid * g, 0)
            ft = ft / ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_M - 1e-15:
                best_M = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 8.0)

    return best_f, best_M


def softmax_gradient_optimize(f0, time_limit, T_start=0.002, T_end=0.00005,
                              use_mirror=False, n_inner=3):
    """Adaptive-temperature softmax gradient descent with optional mirror maps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_M = fast_convolve(f).max()

    t0 = time.monotonic()
    T = T_start
    no_improve = 0
    it = 0
    lr = 0.02

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(f)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Inner loop
        f_inner = f.copy()
        for _ in range(n_inner):
            grad = hankel_grad_fft(f_inner, p)
            if use_mirror:
                candidate = mirror_project(grad * lr, f_inner)
            else:
                candidate = project_simplex(f_inner - lr * grad)
            new_conv = fast_convolve(candidate)
            new_obj = np.dot(p, new_conv)
            old_obj = np.dot(p, conv)
            if new_obj < old_obj:
                f_inner = candidate
            else:
                break

        f = f_inner
        M = fast_convolve(f).max()
        if M < best_M - 1e-15:
            best_M = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt temperature and LR
        if no_improve > 20:
            T = max(T * 1.05, T_end)
            lr *= 0.99
        if no_improve > 80:
            T = max(T * 0.8, T_end)
        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 7) % (2**31 - 1))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            T = T_start
            lr = 0.02
            no_improve = 0

        # Gradually decrease T
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = max(T_end, T_start * (T_end / T_start) ** progress)

    return best_f, best_M


def active_set_polish(f0, time_limit):
    """Polishing using active lag identification and balanced gradient steps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_M = fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0
    T = 0.001
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Identify active lags (within 2% of max)
        threshold = M * 0.98
        active = conv >= threshold

        # Softmax weights over active lags
        active_conv = np.where(active, conv, -np.inf)
        scaled = active_conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        # Compute weighted gradient
        grad = hankel_grad_fft(f, w)

        # Try multiple step sizes
        best_candidate = f
        best_val = M
        for lr in [0.001, 0.003, 0.008, 0.015, 0.03, 0.05]:
            candidate = project_simplex(f - lr * grad)
            c = fast_convolve(candidate)
            if c.max() < best_val:
                best_val = c.max()
                best_candidate = candidate

        if best_val < M - 1e-16:
            f = best_candidate
            if M < best_M - 1e-15:
                best_M = fast_convolve(f).max()
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 15:
            T = max(T * 0.9, 0.00001)
        if no_improve > 60:
            T = max(T * 0.7, 0.00001)
        if no_improve > 150:
            rng = np.random.default_rng((it * 991 + 3) % (2**31 - 1))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            T = 0.001
            no_improve = 0

    return best_f, best_M


def coordinate_descent(f0, time_limit, seed=42):
    """Coordinate descent with exact line search on pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_M = fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        # Transfer mass from i to j (or vice versa)
        # a_i' = a_i - t, a_j' = a_j + t, t in [0, a_i]
        # Or a_i' = a_i + t, a_j' = a_j - t, t in [0, a_j]

        best_t = 0.0
        best_val = best_M

        # Try transferring from i to j
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            t = f[i] * frac
            if t <= 0:
                continue
            candidate = f.copy()
            candidate[i] -= t
            candidate[j] += t
            c = fast_convolve(candidate)
            if c.max() < best_val:
                best_val = c.max()
                best_t = t

        # Try transferring from j to i
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            t = f[j] * frac
            if t <= 0:
                continue
            candidate = f.copy()
            candidate[j] -= t
            candidate[i] += t
            c = fast_convolve(candidate)
            if c.max() < best_val:
                best_val = c.max()
                best_t = -t

        if best_t != 0.0:
            if best_t > 0:
                f[i] -= best_t
                f[j] += best_t
            else:
                f[j] -= (-best_t)
                f[i] += (-best_t)
            f = np.maximum(f, 0)
            f = f / f.sum()
            if best_val < best_M - 1e-15:
                best_M = best_val
                best_f = f.copy()

    return best_f, best_M


def local_search(f0, time_limit, seed=12345):
    """Local search with multi-scale perturbations and structured moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_M = fast_convolve(f).max()

    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        improved = False

        # Multi-scale random perturbations
        for scale in [0.0001, 0.0003, 0.001, 0.003, 0.008, 0.02]:
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            pert = rng.standard_normal(n) * scale
            candidate = project_simplex(f + pert)
            c = fast_convolve(candidate)
            if c.max() < best_M - 1e-15:
                f = candidate
                best_M = c.max()
                best_f = f.copy()
                improved = True
                break

        # Structured: sinusoidal perturbation
        if not improved and it % 5 == 0:
            freq = rng.uniform(0.5, 10)
            phase = rng.uniform(0, 2 * np.pi)
            pattern = np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            for scale in [0.0005, 0.001, 0.003]:
                candidate = project_simplex(f + pattern * scale)
                c = fast_convolve(candidate)
                if c.max() < best_M - 1e-15:
                    f = candidate
                    best_M = c.max()
                    best_f = f.copy()
                    improved = True
                    break

        # Coordinate pair moves
        if not improved and it % 3 == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                for frac in [0.02, 0.05, 0.1, 0.15]:
                    candidate = f.copy()
                    amount = f[i] * frac
                    candidate[i] -= amount
                    candidate[j] += amount
                    candidate = np.maximum(candidate, 0)
                    cs = candidate.sum()
                    if cs < 1e-10:
                        continue
                    candidate = candidate / cs
                    c = fast_convolve(candidate)
                    if c.max() < best_M - 1e-15:
                        f = candidate
                        best_M = c.max()
                        best_f = f.copy()
                        improved = True
                        break

        if not improved and it > 2000:
            break

    return best_f, best_M


def try_seq(a, label, best_seq, best_score):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return a_norm.copy(), score
    return best_seq, best_score
