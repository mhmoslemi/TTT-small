def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]."""

def compute_subgradient(a, top_k=50):
    """Compute weighted subgradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:

        pv = conv[peaks]
        # Inverse frequency weighting
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 0.5
        weights /= weights.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):

        sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        if obj < best_obj - 1e-14:

            best_a = a_new.copy()

"""
Advanced multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Linearized LP steps (local direction finding via LP)
2. Nesterov-accelerated subgradient with adaptive step size
3. Multi-scale subgradient with alternating peak focus
4. Adaptive coordinate descent with line search
5. Frank-Wolfe with LP oracle for global direction
6. Fine local search for final polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, top_k):
    """Compute weighted subgradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(peaks))
        for rank, idx in enumerate(order):
            weights[idx] = 1.0 / (rank + 1) ** 0.5
        weights /= weights.sum()

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        sg[idx_arr] += weights[i] * 2.0 * a[k - idx_arr]

    return sg, conv.max()


def nesterov_subgrad(a0, time_limit, top_k, lr_init):
    """Nesterov-accelerated subgradient descent on the simplex."""
    n = len(a0)
    x = a0.copy().astype(np.float64)
    x_prev = x.copy()
    v = np.zeros(n)
    mu = 0.92

    best_x = x.copy()
    best_obj = fast_convolve(x).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        # Nesterov momentum
        if it > 1:
            t_cur = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
            beta = (t_cur - 1.0) / t_cur
            y = x + beta * (x - x_prev)
        else:
            y = x
            x_prev = x.copy()

        sg, _ = compute_subgradient(y, top_k=top_k)

        # Adaptive step size
        lr = lr_init / (1.0 + 0.005 * it) ** 0.5

        # Gradient step with momentum
        v = mu * v - lr * sg
        x_new = project_simplex(y + v)

        obj_new = fast_convolve(x_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_x = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        x_prev = x
        x = x_new

        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 1)
            scale = 0.002 + 0.005 * min(1.0, (no_improve - 80) / 200.0)
            x = project_simplex(best_x + rng.standard_normal(n) * scale)
            x_prev = x.copy()
            v = np.zeros(n)
            no_improve = 0

    return best_x, best_obj


def linearized_lp_step(a, max_time):
    """Find best local direction via linearized LP."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()

    # Variables: delta[0..n-1], eps (total n+1)
    # Maximize eps
    # s.t. 2*(a*delta)[k] + eps <= M - conv[k] for all k
    #      sum(delta) = 0
    #      delta[j] >= -a[j]

    c = np.zeros(n + 1)
    c[-1] = -1.0

    A_ub = np.zeros((2 * n - 1 + n, n + 1))
    b_ub = np.zeros(2 * n - 1 + n)

    for k in range(2 * n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = 2.0 * a[i]
        A_ub[k, -1] = 1.0
        b_ub[k] = M - conv[k]

    for j in range(n):
        A_ub[2 * n - 1 + j, j] = -1.0
        b_ub[2 * n - 1 + j] = a[j] + 1e-12

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max_time})
        if res.success:
            delta = res.x[:n]
            eps = res.x[-1]
            return delta, eps
    except Exception:
        pass
    return None, 0.0

def solve_fw_lp(f, M):
    """Solve FW LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        it += 1
        remaining = time_limit - (time.monotonic() - t0)

        g = solve_fw_lp(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.3, 5.0)
            if m_exp > 4.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.001 + 0.004 * min(1.0, no_improve / 100.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)

                m_exp = 1.0

        best_alpha = 0.0

def multi_scale_subgrad(a0, time_limit):
    """Multi-scale subgradient with alternating peak focus."""
    n = len(a0)

def coord_descent(a0, time_limit):
    """Adaptive coordinate descent for targeted peak reduction."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        affected = np.arange(js, je + 1)
        if len(affected) == 0:
            break

        contributions = a[affected] * a[k_peak - affected]
        j_max = affected[np.argmax(contributions)]

        outside = np.setdiff1d(np.arange(n), affected)
        if len(outside) == 0:
            break
        j_min_cand = outside[np.argmin(a[outside])]

        if a[j_max] < 1e-12:
            stale += 1
            if stale > 300:
                break
            if stale % 50 == 0:
                rng = np.random.default_rng(it * 41 + 7)
                candidate = project_simplex(a + rng.standard_normal(n) * 0.001)
                c = fast_convolve(candidate)
                if c.max() < best_obj - 1e-14:
                    a = candidate
                    best_obj = c.max()
                    best_a = a.copy()
                    stale = 0
            continue

        improved = False
        max_delta = min(a[j_max] * 0.4, 0.03 / n)
        for frac in [0.15, 0.3, 0.5, 0.7, 0.85]:
            delta = max_delta * frac
            if delta < 1e-15:
                continue
            candidate = a.copy()
            candidate[j_max] -= delta
            candidate[j_min_cand] += delta
            candidate = np.maximum(candidate, 0.0)
            s = candidate.sum()
            if s < 1e-15:
                continue
            candidate = candidate / s
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                a = candidate
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                break

        if not improved:
            stale += 1
        else:
            stale = 0

    return best_a, best_obj


def fine_local_search(a0, time_limit, rng):
    """Very fine local search with tiny perturbations from best."""
    n = len(a0)
    best_a = a0.copy().astype(np.float64)
    best_obj = fast_convolve(best_a).max()

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        # Multiple tiny perturbations
        for scale in [0.0002, 0.0004, 0.0006, 0.001]:
            if time.monotonic() - t0 > time_limit - 0.3:
                break
            noise = rng.standard_normal(n) * scale
            candidate = project_simplex(best_a + noise)
            c = fast_convolve(candidate)
            if c.max() < best_obj - 1e-15:
                best_obj = c.max()
                best_a = candidate.copy()

    return best_a, best_obj

    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < state[0]:
        state[0] = score
        state[1] = a_norm.copy()
        print("  NEW BEST [{}]: {:.10f} (n={})".format(label, score, len(a)))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization pipeline:
    Phase 1: Nesterov subgradient (quick convergence)
    Phase 2: Linearized LP steps (local direction finding)
    Phase 3: Frank-Wolfe with LP oracle (global direction)
    Phase 4: Multi-scale subgradient (fine-tuning)
    Phase 5: Coordinate descent + fine local search (final polish)
    Phase 6: Explore nearby n values
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    state = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                state = [ps, parent.copy()]
                print("Parent loaded: n={}, score={:.10f}".format(len(parent), ps))
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768.0
        state = [float(evaluate_sequence(parent.tolist())), parent.copy()]
        print("Using default uniform parent, score={:.10f}".format(state[0]))

    n_parent = len(parent)

    # === Phase 1: Nesterov subgradient (~350s) ===
    print("\n=== Phase 1: Nesterov Subgradient ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic() - 1350
    t_alloc = min(remaining, 300)
    if t_alloc > 10:
        print("  n={}, t={:.0f}s".format(n_parent, t_alloc))
        a_opt, _ = nesterov_subgrad(state[1], time_limit=t_alloc, top_k=30, lr_init=0.04)
        try_and_update(a_opt, state, "nesterov_k30")

        # Also try with different top_k
        remaining = deadline - time.monotonic() - 1300
        if remaining > 30:
            a_opt2, _ = nesterov_subgrad(state[1], time_limit=min(remaining, 50),
                                         top_k=10, lr_init=0.05)
            try_and_update(a_opt2, state, "nesterov_k10")

    print("  Phase 1 best: {:.10f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 2: Linearized LP (~500s) ===
    print("\n=== Phase 2: Linearized LP Steps ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic() - 800
    t_alloc = min(remaining, 400)
    if t_alloc > 30:
        print("  t={:.0f}s".format(t_alloc))
        a_opt, _ = linearized_lp_optimize(state[1], time_limit=t_alloc, lp_time=5)
        try_and_update(a_opt, state, "lin_lp")

        # Second pass from slightly perturbed start
        remaining = deadline - time.monotonic() - 700
        if remaining > 60:
            f_pert = project_simplex(state[1] + rng.standard_normal(n_parent) * 0.003)
            t_alloc2 = min(remaining, 100)
            a_opt2, _ = linearized_lp_optimize(f_pert, time_limit=t_alloc2, lp_time=4)
            try_and_update(a_opt2, state, "lin_lp_pert")

    print("  Phase 2 best: {:.10f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 3: Frank-Wolfe (~400s) ===
    print("\n=== Phase 3: Frank-Wolfe ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic() - 400
    t_alloc = min(remaining, 300)
    if t_alloc > 30:
        print("  t={:.0f}s".format(t_alloc))
        a_opt, _ = fw_optimize(state[1], time_limit=t_alloc)
        try_and_update(a_opt, state, "fw_main")

        # Perturbed FW restarts
        for scale in [0.005, 0.015]:
            remaining = deadline - time.monotonic() - 300
            if remaining < 30:
                break
            f_pert = project_simplex(state[1] + rng.standard_normal(n_parent) * scale)
            t_alloc2 = min(remaining, 80)
            a_opt2, _ = fw_optimize(f_pert, time_limit=t_alloc2)
            try_and_update(a_opt2, state, "fw_pert_{:.3f}".format(scale))

    print("  Phase 3 best: {:.10f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 4: Multi-scale subgradient (~200s) ===
    print("\n=== Phase 4: Multi-scale Subgradient ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic() - 200
    t_alloc = min(remaining, 150)
    if t_alloc > 20:
        print("  t={:.0f}s".format(t_alloc))
        a_opt, _ = multi_scale_subgrad(state[1], time_limit=t_alloc)
        try_and_update(a_opt, state, "multi_scale")

        # Second pass with different parameters
        remaining = deadline - time.monotonic() - 150
        if remaining > 30:
            a_opt2, _ = nesterov_subgrad(state[1], time_limit=min(remaining, 60),
                                         top_k=50, lr_init=0.02)
            try_and_update(a_opt2, state, "nesterov_k50")

    print("  Phase 4 best: {:.10f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 5: Coordinate descent + fine search (~180s) ===
    print("\n=== Phase 5: Coord Descent + Fine Search ===")
    t_phase = time.monotonic()
    remaining = deadline - time.monotonic() - 80
    if remaining > 20:
        t_alloc = min(remaining, 80)
        print("  Coord descent, t={:.0f}s".format(t_alloc))
        a_opt, _ = coord_descent(state[1], time_limit=t_alloc)
        try_and_update(a_opt, state, "coord_desc")

    remaining = deadline - time.monotonic() - 50
    if remaining > 10:
        t_alloc = min(remaining, 60)
        print("  Fine local search, t={:.0f}s".format(t_alloc))
        a_opt, _ = fine_local_search(state[1], time_limit=t_alloc, rng=rng)
        try_and_update(a_opt, state, "fine_search")

    # Another coord descent pass
    remaining = deadline - time.monotonic() - 30
    if remaining > 10:
        t_alloc = min(remaining, 30)
        a_opt, _ = coord_descent(state[1], time_limit=t_alloc)
        try_and_update(a_opt, state, "coord_desc2")

    print("  Phase 5 best: {:.10f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Phase 6: Explore nearby n values (~60s) ===
    print("\n=== Phase 6: Nearby n values ===")
    t_phase = time.monotonic()
    for n_try in [n_parent - 4, n_parent - 2, n_parent + 2, n_parent + 4, n_parent + 8]:
        remaining = deadline - time.monotonic() - 20
        if remaining < 8:
            break
        if n_try < 128:
            continue
        start = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, len(state[1])), state[1])
        start = np.maximum(start, 0)
        ss = start.sum()
        if ss > 1e-10:
            start = start / ss
        t_alloc = min(remaining, 8)
        a_opt, _ = nesterov_subgrad(start, time_limit=t_alloc, top_k=15, lr_init=0.02)
        try_and_update(a_opt, state, "n{}_quick".format(n_try))

    print("  Phase 6 best: {:.10f} (elapsed {:.0f}s)".format(state[0], time.monotonic() - t_phase))

    # === Final polish ===
    if state[1] is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 3:
            a_opt, _ = fine_local_search(state[1], time_limit=min(remaining, 5), rng=rng)
            try_and_update(a_opt, state, "final_polish")

    # Fallback
    if state[1] is None:
        state[1] = np.ones(768) / 768.0
        state[0] = float(evaluate_sequence(state[1].tolist()))

    # Final verification
    fs = float(evaluate_sequence(state[1].tolist()))
    if np.isfinite(fs) and fs < state[0]:
        state[0] = fs

    print("\nFINAL SCORE: {:.10f} (n={})".format(state[0], len(state[1])))
    return [float(x) for x in state[1]]
