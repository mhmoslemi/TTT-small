"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_value(a):
    """Compute the objective value (assuming sum(a) = 1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def subgradient(a, top_k=10, temperature=0.01):
    """Compute a weighted subgradient using softmax over top peaks."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Get top peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    # Softmax weights
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # Compute subgradient
    g = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        # gradient of (a*a)[k] w.r.t. a[i] is a[k-i] + a[i-k] (with a[m]=0 for invalid m)
        for j in range(n):
            val = 0.0
            if 0 <= k - j < n:
                val += a[k - j]
            if 0 <= j - k < n:
                val += a[j - k]
            g[j] += weights[i] * val
    
    return g, conv.max()

def pairwise_descent(a0, time_limit, num_pairs_per_iter=None):
    """
    Pairwise coordinate descent: transfer mass between pairs of coordinates.
    Uses ternary search on the 1D function.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = obj_value(a)
    
    t0 = time.monotonic()
    it = 0
    stall = 0
    
    if num_pairs_per_iter is None:
        num_pairs_per_iter = min(20, n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        # Select pairs to try
        rng = np.random.default_rng((it * 12345 + 67890) % (2**31))
        
        # Priority: pairs where one has high value and the other is in a "quiet" region
        # For now, just sample random pairs
        pairs = []
        for _ in range(num_pairs_per_iter):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-10 and a[j] < a[i]:
                pairs.append((i, j))
        
        # Also add some top-value to low-value pairs
        sorted_idx = np.argsort(a)[::-1]
        for ii in range(min(5, n // 4)):
            for jj in range(n // 2, min(n // 2 + 5, n)):
                pairs.append((sorted_idx[ii], sorted_idx[jj]))
        
        improved = False
        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            if a[i] < 1e-12:
                continue
            
            # Ternary search for optimal δ (transfer from i to j)
            # δ in [0, a[i]]
            lo = 0.0
            hi = a[i]
            
            def val_at_delta(delta):
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                if a_test[i] < 0 or a_test[j] < 0:
                    return float('inf')
                return fast_convolve(a_test).max()
            
            # Ternary search
            for _ in range(15):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                v1 = val_at_delta(m1)
                v2 = val_at_delta(m2)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            
            opt_delta = (lo + hi) / 2
            if opt_delta > 1e-10:
                a[i] -= opt_delta
                a[j] += opt_delta
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                
                new_obj = obj_value(a)
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
        
        # If stalled, perturb
        if stall > 30:
            rng2 = np.random.default_rng((it * 7919 + 1) % (2**31))
            noise_scale = 0.001 * min(1.0, stall / 50.0)
            a = project_simplex(best_a + rng2.standard_normal(n) * noise_scale)
            stall = 0
        
        if stall > 200:
            break
    
    return best_a, best_obj


def simulated_annealing(a0, time_limit, T_start=0.001, T_end=0.00001, sigma=0.005):
    """
    Simulated annealing on the simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = obj_value(a)
    curr_obj = best_obj
    
    t0 = time.monotonic()
    total_time = time_limit
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Temperature schedule (geometric cooling)
        progress = (time.monotonic() - t0) / total_time
        T = T_start * (T_end / T_start) ** progress
        
        # Propose a neighbor
        rng = np.random.default_rng((it * 2654435761) % (2**32))
        noise = rng.standard_normal(n) * sigma
        a_new = project_simplex(a + noise)
        
        new_obj = obj_value(a_new)
        delta = new_obj - curr_obj
        
        # Metropolis acceptance
        if delta < 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
            a = a_new
            curr_obj = new_obj
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_a = a_new.copy()
    
    return best_a, best_obj


def momentum_subgrad(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=20):
    """
    Subgradient method with momentum and adaptive learning rate.
    Uses top-k peak weighting.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_obj = obj_value(a)
    
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break
        
        # Compute subgradient
        g, _ = subgradient(a, top_k=top_k, temperature=0.005)
        
        # Momentum update
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)
        
        new_obj = obj_value(a_new)
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Backtrack
            lr *= 0.95
        
        if no_improve > 50:
            lr = lr_init * 0.5
            no_improve = 0
        
        if it % 1000 == 0:
            lr = lr_init / (1 + it * 0.001)
    
    return best_a, best_obj


def generate_initializations(n, parent=None):
    """Generate diverse initializations for a given n."""
    inits = []
    
    # 1. Uniform
    inits.append(("uniform", np.ones(n) / n))
    
    # 2. Cosine window
    i = np.arange(n)
    cos_win = np.cos(np.pi * (i - (n-1)/2.0) / n) ** 2
    cos_win = np.maximum(cos_win, 0)
    inits.append(("cosine", cos_win / cos_win.sum()))
    
    # 3. Gaussian windows with different widths
    for sigma_frac in [0.3, 0.5, 0.7, 1.0]:
        x = (i - (n-1)/2.0) / (n * sigma_frac)
        gauss = np.exp(-0.5 * x * x)
        inits.append((f"gauss_{sigma_frac}", gauss / gauss.sum()))
    
    # 4. Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * i / n))
    inits.append(("raised_cos", rc / rc.sum()))
    
    # 5. Exponential decay from center
    dist = np.abs(i - (n-1)/2.0) / n
    for alpha in [1.0, 2.0, 3.0]:
        exp_win = np.exp(-alpha * dist)
        inits.append((f"exp_{alpha}", exp_win / exp_win.sum()))
    
    # 6. Two-hump
    hump1 = np.exp(-((i - n*0.25) / (n*0.15))**2)
    hump2 = np.exp(-((i - n*0.75) / (n*0.15))**2)
    two_hump = hump1 + hump2
    inits.append(("two_hump", two_hump / two_hump.sum()))
    
    # 7. From parent (resampled)
    if parent is not None and len(parent) > 1:
        parent_resampled = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(parent)), parent)
        parent_resampled = np.maximum(parent_resampled, 0)
        if parent_resampled.sum() > 1e-10:
            inits.append(("parent", parent_resampled / parent_resampled.sum()))
    
    return inits

- `(a*a)[0] = p^2 + q^2`
- `(a*a)[n-1] = 2pq`
- All other lags: 0

Max = max(p^2+q^2, 2pq). With p=q=1/2: max = 1/2. Objective = 2n/2 = n. Terrible.

What about a "smooth" distribution? If a is approximately Gaussian with variance σ^2, then (a*a) is approximately Gaussian with variance 2σ^2. The peak of (a*a) is approximately 1/(2πσ^2) * (1/(σ√(4π)))... this is getting complicated.

Let me just focus on the computational approach and try to get the best result.

Let me also consider: what if I use a **different n** than the parent? The parent has n=768. Maybe n=512 or n=1024 gives a better result.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse smooth initializations (cosine, Gaussian, etc.)
2. Mirror descent (excribed gradient) with vectorized subgradients
3. Momentum-based subgradient descent
4. Pairwise coordinate descent with ternary search
5. Simulated annealing
6. Multi-n exploration

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_peak(a):
    """max of auto-convolution (a is normalized to sum=1)."""
    conv = fast_convolve(a)
    return conv.max(), conv


def grad_conv_at(a, k):
    """Vectorized gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    idx = np.arange(n)
    g = np.zeros(n)
    m1 = k - idx
    mask1 = (m1 >= 0) & (m1 < n)
    g[mask1] += a[m1[mask1]]
    m2 = idx - k
    mask2 = (m2 >= 0) & (m2 < n)
    g[mask2] += a[m2[mask2]]
    return g


def weighted_subgrad(a, top_k=10, temp=0.01):
    """Compute softmax-weighted subgradient over top peaks."""
    n = len(a)
    conv = fast_convolve(a)
    
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    g = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        g += w[i] * grad_conv_at(a, int(peaks[i]))
    
    return g, conv.max()


def mirror_descent(a0, time_limit, eta_init=0.05, top_k=10, T_init=0.01):
    """Mirror descent with negative entropy mirror map."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj, _ = obj_peak(a)
    
    eta = eta_init
    T = T_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break
        
        g, peak_val = weighted_subgrad(a, top_k=top_k, temp=T)
        
        # Mirror descent update
        log_upd = -eta * g
        log_upd -= log_upd.max()
        exp_upd = np.exp(log_upd)
        a_new = a * exp_upd
        s_new = a_new.sum()
        if s_new < 1e-300:
            break
        a_new /= s_new
        
        new_obj, _ = obj_peak(a_new)
        
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 30:
            eta *= 0.9
        if no_improve > 100:
            T *= 0.8
        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 7) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            eta = eta_init
            T = T_init
            no_improve = 0
    
    return best_a, best_obj


def momentum_subgrad(a0, time_limit, lr_init=0.005, momentum=0.9, top_k=15):
    """Subgradient with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj, _ = obj_peak(a)
    
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break
        
        g, _ = weighted_subgrad(a, top_k=top_k, temp=0.005)
        
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)
        
        new_obj, _ = obj_peak(a_new)
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95
        
        if no_improve > 50:
            lr = max(lr_init * 0.1, lr * 0.5)
            no_improve = 0
    
    return best_a, best_obj


def pairwise_descent(a0, time_limit):
    """Pairwise coordinate descent with ternary search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj, _ = obj_peak(a)
    
    t0 = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        rng = np.random.default_rng((it * 12345 + 67890) % (2**31))
        
        # Sample pairs
        n_pairs = min(30, n)
        improved = False
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or a[i] < 1e-12:
                continue
            
            # Ternary search for optimal transfer δ from i to j
            lo = 0.0
            hi = a[i]
            
            for _ in range(12):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                a_t1 = a.copy()
                a_t1[i] -= m1
                a_t1[j] += m1
                v1 = fast_convolve(a_t1).max()
                
                a_t2 = a.copy()
                a_t2[i] -= m2
                a_t2[j] += m2
                v2 = fast_convolve(a_t2).max()
                
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            
            opt_d = (lo + hi) / 2.0
            if opt_d > 1e-10:
                a[i] -= opt_d
                a[j] += opt_d
                a = np.maximum(a, 0.0)
                a /= a.sum()
                new_obj, _ = obj_peak(a)
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
        
        if improved:
            stall = 0
        else:
            stall += 1
        
        if stall > 50:
            rng2 = np.random.default_rng((it * 7919) % (2**31))
            a = project_simplex(best_a + rng2.standard_normal(n) * 0.002)
            stall = 0
        if stall > 300:
            break
    
    return best_a, best_obj


def simulated_annealing(a0, time_limit, sigma=0.005, T_start=0.0005, T_end=0.000005):
    """Simulated annealing on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj, _ = obj_peak(a)
    curr_obj = best_obj
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.03:
            break
        
        progress = (time.monotonic() - t0) / time_limit
        T = T_start * (T_end / T_start) ** progress
        
        rng = np.random.default_rng((it * 2654435761) % (2**32))
        noise = rng.standard_normal(n) * sigma
        a_new = project_simplex(a + noise)
        
        new_obj, _ = obj_peak(a_new)
        delta = new_obj - curr_obj
        
        if delta < 0 or (T > 1e-18 and rng.random() < np.exp(-delta / T)):
            a = a_new
            curr_obj = new_obj
            if new_obj < best_obj - 1e-15:
                best_obj = new_obj
                best_a = a_new.copy()
    
    return best_a, best_obj


def gen_inits(n, parent=None):
    """Generate diverse initializations."""
    inits = []
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine
    c = np.cos(np.pi * (i - center) / n) ** 2
    c = np.maximum(c, 1e-10)
    inits.append(c / c.sum())
    
    # Gaussian windows
    for sf in [0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        x = (i - center) / (n * sf)
        g = np.exp(-0.5 * x * x)
        inits.append(g / g.sum())
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * i / n))
    inits.append(rc / rc.sum())
    
    # Exponential
    d = np.abs(i - center) / n
    for al in [1.0, 2.0, 4.0]:
        e = np.exp(-al * d)
        inits.append(e / e.sum())
    
    # Two-hump
    h = np.exp(-((i - n*0.25) / (n*0.12))**2) + np.exp(-((i - n*0.75) / (n*0.12))**2)
    inits.append(h / h.sum())
    
    # Four-hump
    h4 = sum(np.exp(-((i - n*f) / (n*0.08))**2) for f in [0.15, 0.35, 0.65, 0.85])
    inits.append(h4 / h4.sum())
    
    # From parent
    if parent is not None and len(parent) > 1:
        pr = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        pr = np.maximum(pr, 0)
        if pr.sum() > 1e-10:
            inits.append(pr / pr.sum())
    
    return inits
