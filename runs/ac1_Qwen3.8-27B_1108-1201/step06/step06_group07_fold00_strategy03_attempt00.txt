def fast_convolve(a):
    """Linear auto-convolution via FFT for sequences of up to 2048."""
    n = len(a)
    size = 2 * n - 1
    if size <= 4096:
        fft_size = 1 << (size - 1).bit_length()
    else:
        # For larger sequences, use safe FFT size
        fft_size = 4096
    a_pad = np.zeros(fft_size, dtype=np.float64)
    a_pad[:n] = a
    a_fft = np.fft.rfft(a_pad)
    conv_fft = a_fft * a_fft
    conv = np.fft.irfft(conv_fft)[:size]
    return conv

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    t = np.sort(v)[::-1]  # Sort descending
    u = np.cumsum(t) - 1
    rho = np.argmin(u / (np.arange(1, n+1) + 1) > 0 if True else 0
    if rho == 0:
        return np.ones(n) / n
    theta = (u[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0) / (max(v) - theta + 1e-15)
