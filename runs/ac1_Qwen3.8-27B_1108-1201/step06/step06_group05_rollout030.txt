def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_fn(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    ex = np.exp(x_shifted)
    s = ex.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return ex / s


def logsumexp_smoothed_obj_grad(a, beta):
    """
    Compute log-sum-exp smoothed objective and its gradient w.r.t. a.
    f(a) = (1/beta) * log(sum_k exp(beta * (a*a)[k]))
    grad[m] = 2 * sum_j p[j+m] * a[j]  (cross-correlation)
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    scaled = beta * (conv - max_c)
    exp_scaled = np.exp(scaled)
    logsum = max_c + np.log(exp_scaled.sum()) / beta
    p = exp_scaled / exp_scaled.sum()

    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    cross = np.fft.irfft(np.conj(P) * A, fft_size)[:n]
    grad = 2.0 * cross
    return logsum, grad

    best_score = float(evaluate_sequence(a.tolist()))
    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        conv = fast_convolve(a)
        M = conv.max()
        # Find active peaks (within tolerance of max)
        active = np.where(conv >= M * (1.0 - peak_tol))[0]
        P = active
        p_count = len(P)

        if p_count == 0:
            break

        # Build the Jacobian for active peaks: J[k_idx, m] = 2*a[k-m] if valid
        J = np.zeros((p_count, n))
        for i, k in enumerate(P):
            for m in range(n):
                idx = k - m
                if 0 <= idx < n:
                    J[i, m] = 2.0 * a[idx]

        # Residual: we want (a*a)[k] - M = 0 for k in P
        # And we want to minimize the max, so we push all active peaks down
        # Use a least-squares Newton step
        residual = M - conv[P]  # positive values we want to reduce

        # Add the sum constraint: keep sum(a) = 1
        # Solve: J^T * lambda = residual (least squares), subject to sum(Δa) = 0
        # Use the approach: find Δa that minimizes ||J Δa - residual||^2 s.t. 1^T Δa = 0

        # Project J onto the subspace sum(Δa) = 0
        # Δa = (I - (1/n)*1*1^T) @ J^T @ (J (I - (1/n)*1*1^T) J^T)^{-1} @ residual
        # This is expensive for large n. Use a simpler approach:
        # Just solve the unconstrained LSQ and then project.

        if p_count >= n:
            # Overdetermined: use least squares
            try:
                da, _, _, _ = np.linalg.lstsq(J.T, residual, rcond=None)
            except np.linalg.LinAlgError:
                break
        else:
            # Underdetermined: minimum norm solution
            try:
                # da = J^T (J J^T)^{-1} residual
                JJT = J @ J.T
                JJT += 1e-12 * np.eye(p_count)
                sol = np.linalg.solve(JJT, residual)
                da = J.T @ sol
            except np.linalg.LinAlgError:
                break

        # Project to maintain sum(a) = 1
        da -= da.mean()

        # Line search
        best_step = 0.0
        best_val = M
        for step in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_trial = a + step * da
            if (a_trial < -1e-10).any():
                continue
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            c_trial = fast_convolve(a_trial)
            val = c_trial.max()
            if val < best_val - 1e-15:
                best_val = val
                best_step = step

        if best_step > 0:
            a = np.maximum(a + best_step * da, 0.0)
            a = a / a.sum()
            if best_val < best_score - 1e-12:
                best_score = best_val
                best_a = a.copy()

        # Check convergence
        if best_step < 1e-10:
            break

        # Tighten peak tolerance as we converge
        if it % 50 == 0:
            peak_tol = max(0.001, peak_tol * 0.7)

    # Symmetrize and try
    sym_a = (best_a + best_a[::-1]) / 2.0
    sym_a = sym_a / sym_a.sum()
    sym_score = float(evaluate_sequence(sym_a.tolist()))
    if np.isfinite(sym_score) and sym_score < best_score:
        best_score = sym_score
        best_a = sym_a.copy()

    return best_a, best_score


def fw_step(f, M, n):
    """
    One Frank-Wolfe step: solve max sum(g) s.t. (f*g)[k] <= M, g >= 0.
    Uses a simplified approach with the top constraints.
    """
    # Build constraint matrix for the LP
    # (f*g)[k] = sum_j f[j] * g[k-j] <= M for all k
    # This has 2n-1 constraints and n variables.
    # For large n, this is expensive. Use a reduced version with top constraints.
    conv_f = fast_convolve(f)
    top_k = min(20, 2 * n - 1)
    peak_indices = np.argpartition(conv_f, -top_k)[-top_k:]

    # Build A_ub for only the top constraints
    A_ub = np.zeros((top_k, n))
    for i, k in enumerate(peak_indices):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[i, j] = f[idx]

    b_ub = np.full(top_k, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    from scipy.optimize import linprog
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'max_time': 15})
        if res.success:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, max_iter=100):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_obj = M
    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit - 1:
            break

        g = fw_step(f, M * 1.0, n)
        if g is None:
            break

        if np.max(np.abs(g - f)) < 1e-10:
            break

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            M = fast_convolve(f).max()
            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search:
    Phase 1: L-BFGS-B with curriculum (main workhorse)
    Phase 2: Active-set Newton polishing
    Phase 3: FW-LP for global moves
    Phase 4: Multi-n exploration and final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        p = np.maximum(p, 0.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            if np.isfinite(ps):
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default parent: n={len(parent)}, score={best_score:.10f}")

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")

    # === Phase 1: Symmetry + L-BFGS-B curriculum (~800s) ===
    print("\n=== Phase 1: L-BFGS-B Curriculum ===")
    t_start = time.monotonic()
    n0 = len(best_seq)

    # Try symmetrized version
    sym_parent = (best_seq + best_seq[::-1]) / 2.0
    sym_parent = sym_parent / sym_parent.sum()
    try_update(sym_parent, "sym_parent")

    # L-BFGS-B with full curriculum on the best so far
    remaining = deadline - time.monotonic() - 900
    if remaining > 30:
        t_alloc = min(remaining, 700)
        print(f"  L-BFGS-B full curriculum, n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = lbfgs_softmax_optimize(best_seq, time_limit=t_alloc,
                                          betas=[0.01, 0.1, 1.0, 10.0, 50.0, 200.0, 1000.0, 5000.0])
        try_update(a_opt, "lbfgs_full")

    print(f"  Phase 1 best: {best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Active-set Newton polish (~300s) ===
    print("\n=== Phase 2: Active-Set Newton ===")
    t_start = time.monotonic()

    remaining = deadline - time.monotonic() - 500
    if remaining > 30:
        t_alloc = min(remaining, 250)
        print(f"  Active-set Newton, n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = active_set_newton(best_seq, time_limit=t_alloc, max_iter=1000, peak_tol=0.01)
        try_update(a_opt, "as_newton")

        # Try with tighter tolerance
        remaining = deadline - time.monotonic() - 300
        if remaining > 20:
            a_opt2, _ = active_set_newton(best_seq, time_limit=min(remaining, 100),
                                          max_iter=500, peak_tol=0.001)
            try_update(a_opt2, "as_newton_tight")

    # FW-LP step
    remaining = deadline - time.monotonic() - 200
    if remaining > 30:
        t_alloc = min(remaining, 100)
        print(f"  FW-LP, n={len(best_seq)}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
        try_update(a_opt, "fw_lp")

    # After FW, do another Newton polish
    remaining = deadline - time.monotonic() - 150
    if remaining > 15:
        a_opt, _ = active_set_newton(best_seq, time_limit=min(remaining, 60),
                                     max_iter=300, peak_tol=0.005)
        try_update(a_opt, "as_newton2")

    print(f"  Phase 2 best: {best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 3: Multi-n exploration (~200s) ===
    print("\n=== Phase 3: Multi-n Exploration ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    def interp_sequence(parent_seq, n_target):
        if len(parent_seq) == n_target:
            return parent_seq.copy()
        x_orig = np.linspace(0, 1, len(parent_seq))
        x_new = np.linspace(0, 1, n_target)
        new_seq = np.interp(x_new, x_orig, parent_seq)
        new_seq = np.maximum(new_seq, 0)
        s = new_seq.sum()
        if s < 1e-15:
            return np.ones(n_target) / n_target
        return new_seq / s

    candidate_ns = []
    for dn in [-64, -32, 0, 32, 64, 96, 128]:
        n_try = n_best + dn
        if 256 <= n_try <= 4096:
            candidate_ns.append(n_try)

    for n_try in candidate_ns:
        remaining = deadline - time.monotonic() - 100
        if remaining < 20:
            break

        if n_try == n_best:
            start = best_seq.copy()
        else:
            start = interp_sequence(best_seq, n_try)

        # Quick L-BFGS-B
        t_alloc = min(remaining - 5, 80)
        if t_alloc < 10:
            continue
        print(f"  Multi-n: n={n_try}, t={t_alloc:.0f}s")
        a_opt, _ = lbfgs_softmax_optimize(start, time_limit=t_alloc * 0.7,
                                          betas=[1.0, 10.0, 100.0, 1000.0])
        try_update(a_opt, f"lbfgs_n{n_try}")

        # Quick Newton polish
        remaining = deadline - time.monotonic() - 80
        if remaining > 10:
            a_opt2, _ = active_set_newton(a_opt, time_limit=min(remaining, 20),
                                          max_iter=200, peak_tol=0.005)
            try_update(a_opt2, f"as_n{n_try}")

    print(f"  Phase 3 best: {best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Random restarts + final polish (~remaining time) ===
    print("\n=== Phase 4: Restarts & Final Polish ===")
    t_start = time.monotonic()
    n_best = len(best_seq)

    for restart in range(10):
        remaining = deadline - time.monotonic() - 30
        if remaining < 15:
            break

        # Perturb and optimize
        noise_scale = 0.001 + 0.002 * restart * 0.5
        f_pert = best_seq + rng.standard_normal(n_best) * noise_scale
        f_pert = np.maximum(f_pert, 0.0)
        f_pert = f_pert / f_pert.sum()

        # Symmetrize alternative
        if restart % 2 == 1:
            f_pert = (f_pert + f_pert[::-1]) / 2.0
            f_pert = f_pert / f_pert.sum()

        t_alloc = min(remaining - 5, 40)
        a_opt, _ = lbfgs_softmax_optimize(f_pert, time_limit=t_alloc,
                                          betas=[10.0, 100.0, 1000.0])
        try_update(a_opt, f"restart_{restart}")

        # Quick Newton
        remaining = deadline - time.monotonic() - 20
        if remaining > 10:
            a_opt2, _ = active_set_newton(best_seq, time_limit=min(remaining, 15),
                                          max_iter=200, peak_tol=0.003)
            try_update(a_opt2, f"as_restart_{restart}")

    # Final: one more L-BFGS-B with very high beta for precision
    remaining = deadline - time.monotonic() - 10
    if remaining > 10:
        print(f"  Final L-BFGS-B high-precision, t={remaining:.0f}s")
        a_opt, _ = lbfgs_softmax_optimize(best_seq, time_limit=remaining,
                                          betas=[500.0, 5000.0, 50000.0])
        try_update(a_opt, "final_lbfgs")

    # Final symmetry check
    sym_final = (best_seq + best_seq[::-1]) / 2.0
    sym_final = sym_final / sym_final.sum()
    try_update(sym_final, "final_sym")

    print(f"  Phase 4 best: {best_score:.10f} (elapsed {time.monotonic()-t_start:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
