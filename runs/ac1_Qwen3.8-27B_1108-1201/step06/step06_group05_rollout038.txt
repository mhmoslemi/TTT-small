```python
def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def structured_init(n, kind='gaussian', alpha=2.0):
    """Create structured initial sequences."""
    x = np.arange(n) / (n - 1) - 0.5  # centered in [-0.5, 0.5]
    if kind == 'gaussian':
        a = np.exp(-alpha * x * x * n)  # Gaussian with width ~ 1/sqrt(alpha*n)
    elif kind == 'raised_cosine':
        a = np.cos(np.pi * x) ** 2
    elif kind == 'flat_top':
        a = np.ones(n)
    elif kind == 'triangle':
        a = 1 - 2 * np.abs(x)
    elif kind == 'sinc_like':
        # Approximate a sinc function (bandsignal)
        a = np.sinc(2 * x * np.sqrt(n))
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def multi_resolution_optimize(a_init, n_final, time_limit, T_schedule_base):
    """Multi-resolution: start small, upscale, optimize at each level."""
    n = len(a_init)
    a = a_init.copy()
    t0 = time.monotonic()
    
    # Build resolution levels
    levels = []
    cur_n = max(16, n // (n // 16))  # start around 16
    while cur_n < n_final:
        cur_n = min(cur_n * 2, n_final)
        levels.append(cur_n)
    if levels[-1] != n_final:
        levels.append(n_final)
    
    print(f"  Resolution levels: {levels}")
    
    for level_n in levels:
        if time.monotonic() - t0 > time_limit - 30:
            break
        
        if level_n == n:
            pass  # already at this resolution
        else:
            # Upsample
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, level_n)
            a = np.interp(x_new, x_old, a)
            a = np.maximum(a, 0)
            a = a / a.sum()
            n = level_n
        
        # Optimize at this resolution
        time_left = time_limit - (time.monotonic() - t0) - 20

        # Scale T schedule for this level

        a = lbfgs_smoothed(a, T_sched, time_alloc, max_iter=2000)
        
    return a
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with softmax-parameterized variables and annealed smooth-max objective
2. Multi-resolution optimization (small n -> large n)
3. Diverse structured initializations (Gaussian, raised cosine, etc.)
4. Iterated local refinement with diverse perturbations

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_z(z, T, n):
    """Compute gradient of the smoothed objective w.r.t. z."""
    a = softmax(z)
    conv = fast_convolve(a)
    
    # Softmax weights
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w = w / w.sum()
    
    # grad_a[m] = sum_k w[k] * 2*a[k-m] * [0<=k-m<n]
    #           = 2 * sum_{l=0}^{n-1} w[l+m] * a[l]  (letting l = k-m)
    #           = 2 * np.correlate(w, a, mode='valid')[m]
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    
    # Chain rule: grad_z[k] = sum_m grad_a[m] * da[m]/dz[k]
    # da[m]/dz[k] = a[m] * (delta[m,k] - a[k])
    # grad_z[k] = a[k] * (grad_a[k] - sum_m grad_a[m]*a[m])
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    
    # Objective value
    fval = 2 * n * T * (scaled.max() + np.log(w.sum()))  # w.sum() = exp(scaled.max() + log(sum(exp(scaled - scaled.max()))))
    # Actually: log(sum(exp(conv/T))) = scaled.max() + log(sum(exp(scaled - scaled.max())))
    # And w = exp(scaled - scaled.max()) / sum(exp(scaled - scaled.max()))
    # So sum(exp(scaled - scaled.max())) = 1/w_normalized... 
    # Let me just compute it directly:
    log_sum_exp = scaled.max() + np.log(np.sum(np.exp(scaled - scaled.max())))
    fval = 2 * n * T * log_sum_exp
    
    return fval, grad_z


def lbfgs_anneal(a_init, n, T_values, time_per_T=30, max_iter=5000):
    """L-BFGS with annealed temperature."""
    a = np.maximum(a_init, 1e-300)
    a = a / a.sum()
    z = np.log(a)
    
    best_z = z.copy()
    best_obj = 2 * n * fast_convolve(a).max()
    
    for i, T in enumerate(T_values):
        t_start_i = time.monotonic()
        
        def obj_and_grad(z_local):
            return compute_grad_z(z_local, T, n)
        
        try:
            result = minimize(obj_and_grad, z, method='L-BFGS-B',
                            options={'maxiter': max_iter, 'gtol': 1e-9,
                                   'maxfun': max_iter * 2})
        except Exception as e:
            print(f"      L-BFGS error: {e}")
            break
        
        z = result.x
        a_new = softmax(z)
        obj_new = 2 * n * fast_convolve(a_new).max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_z = z.copy()
        
        elapsed_i = time.monotonic() - t_start_i
        print(f"      T={T:.2e}: smooth={result.fun:.6f}, true={obj_new:.8f}, best={best_obj:.8f}, iters={result.nit}, time={elapsed_i:.1f}s")
    
    a_best = softmax(best_z)
    a_best = np.maximum(a_best, 0)
    a_best = a_best / a_best.sum()
    return a_best, best_obj


def structured_init(n, kind='gaussian', alpha=3.0):
    """Create structured initial sequences."""
    if n <= 1:
        return np.ones(max(n, 1))
    x = np.arange(n) / (n - 1) - 0.5  # in [-0.5, 0.5]
    t = 2 * x  # in [-1, 1]
    
    if kind == 'gaussian':
        a = np.exp(-alpha * t * t * n / 4)
    elif kind == 'raised_cosine':
        a = (1 + np.cos(np.pi * t)) / 2
    elif kind == 'sinc_window':
        # Kaiser-like window
        a = np.sinc(t * np.sqrt(n) / 2)
        a = np.maximum(a, 0)

def upsample(a, n_target):
    """Upsample sequence to target length using linear interpolation."""
    if len(a) == n_target:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    new = np.interp(x_new, x_old, a)
    new = np.maximum(new, 0)
    s = new.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new / s
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with softmax-parameterized
variables, annealed smooth-max objective, multi-resolution optimization,
diverse structured initializations, and iterated local search.
"""
import numpy as np
from scipy.optimize import minimize
import time

# Global state to avoid closures in minimize callback
_G = {'T': 1.0, 'n': 1}


def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_z(z, T, n):
    """Compute gradient of smoothed objective 2n*T*log(sum(exp(conv/T))) w.r.t. z."""
    a = softmax(z)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()

    # grad_a[m] = 2 * sum_{l=0}^{n-1} w[l+m] * a[l] = 2 * correlate(w, a, valid)[m]
    grad_a = 2.0 * np.correlate(w, a, mode='valid')

    # Chain rule through softmax: grad_z[k] = a[k] * (grad_a[k] - dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)

    # Objective
    log_sum_exp = scaled.max() + np.log(exp_scaled.sum())
    fval = 2.0 * n * T * log_sum_exp

    return fval, grad_z


def obj_grad_wrapper(z):
    """Wrapper using global state (avoids closure)."""
    return compute_grad_z(z, _G['T'], _G['n'])


def structured_init(n, kind, alpha=3.0):
    """Create structured initial sequences."""
    if n <= 1:
        return np.ones(max(n, 1))
    t = 2.0 * (np.arange(n) / (n - 1) - 0.5)  # in [-1, 1]

    if kind == 'gaussian':
        a = np.exp(-alpha * t * t * n / 4.0)
    elif kind == 'raised_cosine':
        a = (1.0 + np.cos(np.pi * t)) / 2.0
    elif kind == 'parabolic':
        a = 1.0 - t * t
    elif kind == 'power':
        a = np.maximum(1.0 - t * t, 0.0) ** (alpha + 1.0)
    elif kind == 'sinc_window':
        a = np.sinc(t * np.sqrt(n) / 3.0)
        a = np.maximum(a, 0.0)
    elif kind == 'flat':
        a = np.ones(n)
    elif kind == 'triangle':
        a = 1.0 - np.abs(t)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def upsample(a, n_target):
    """Upsample sequence to target length."""
    if len(a) == n_target:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_target)
    new = np.interp(x_new, x_old, a)
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new / s

                                       'gtol': gtol,
                                       'maxfun': max_iter_per_T * 2})

        z = result.x
        a_new = softmax(z)
        obj_new = 2.0 * n * fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_z = z.copy()

        print(f"    T={T:.2e}: smooth={result.fun:.6f}, true={obj_new:.8f}, "
              f"best={best_obj:.8f}, iters={result.nit}")

    a_best = softmax(best_z)
    a_best = np.maximum(a_best, 0.0)
    a_best = a_best / a_best.sum()
    return a_best, best_obj


def lbfgs_time_limited(a_init, n, T_values, time_limit, max_iter_per_T=3000):
    """L-BFGS with overall time limit."""
    a = np.maximum(a_init.astype(np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)

    best_z = z.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    _G['n'] = n

    for i, T in enumerate(T_values):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 1:
            break
        if T < 1e-15:
            break

        remaining = time_limit - elapsed
        # Estimate max iterations based on remaining time
        max_iter_here = max(20, min(max_iter_per_T, int(remaining * 30)))

        _G['T'] = T
        try:
            result = minimize(obj_grad_wrapper, z, method='L-BFGS-B',
                              options={'maxiter': max_iter_here,
                                       'gtol': 1e-8,
                                       'maxfun': max_iter_here * 2})
        except Exception:
            break

        z = result.x
        a_new = softmax(z)
        obj_new = 2.0 * n * fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_z = z.copy()

    a_best = softmax(best_z)
    a_best = np.maximum(a_best, 0.0)
    a_best = a_best / a_best.sum()
    return a_best, best_obj


def subgradient_step(a, lr=0.01, top_k=20):
    """One subgradient step with momentum (for local search)."""
    n = len(a)
    conv = fast_convolve(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    order = np.argsort(pv)[::-1]
    weights = np.zeros(len(peaks))
    for rank, idx in enumerate(order):
        weights[idx] = 1.0 / (rank + 1) ** 1.5
    ws = weights.sum()
    if ws > 0:
        weights /= ws

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx_arr = np.arange(js, je + 1)
        grad[idx_arr] += weights[i] * a[k - idx_arr]

    a_new = project_simplex(a - lr * grad)
    return a_new


def iterated_local_search(a_init, n, time_limit, n_restarts=10, lr_init=0.01):
    """Iterated local search with random perturbations and subgradient steps."""
    a = np.maximum(a_init, 0.0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    rng = np.random.default_rng(12345)
    t0 = time.monotonic()

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 2:
            break

        # L-BFGS polish
        conv = fast_convolve(a)
        M = conv.max()
        T_vals = [M * 0.01, M * 0.005, M * 0.002]
        a, _ = lbfgs_time_limited(a, n, T_vals, time_limit=8, max_iter_per_T=500)

        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Perturb
        scale = 0.005 + 0.02 * (restart / max(1, n_restarts))
        noise = rng.standard_normal(n) * scale
        a = project_simplex(best_a + noise)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: L-BFGS with annealed smooth-max, multi-resolution,
    diverse structured inits, and iterated local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_a = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_score(parent)
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_a = parent.copy()
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = structured_init(512, 'gaussian', 3.0)
        best_a = parent.copy()
        best_score = evaluate_score(parent)
        print(f"Default parent: n={len(parent)}, score={best_score:.8f}")

    n_parent = len(parent)

    # Determine n values to explore
    n_targets = sorted(set([n_parent] + [256, 384, 512, 640, 768, 896, 1024, 1280]))
    n_targets = [n for n in n_targets if 128 <= n <= 2048]

    # === Phase 1: Quick L-BFGS on parent (~150s) ===
    print("\n=== Phase 1: Quick Opt on Parent ===")
    t0 = time.monotonic()

    conv = fast_convolve(parent)
    M = conv.max()
    T_vals = [M * 0.05, M * 0.02, M * 0.01, M * 0.005, M * 0.002]
    t_alloc = min(100, max(20, deadline - time.monotonic() - 700))
    if t_alloc > 10:
        a_opt, _ = lbfgs_time_limited(parent, n_parent, T_vals, time_limit=t_alloc,
                                       max_iter_per_T=3000)
        score = evaluate_score(a_opt)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  L-BFGS parent: {score:.8f}")

    # Symmetrize
    sym = (parent + parent[::-1]) / 2.0
    sym = sym / sym.sum()
    score_sym = evaluate_score(sym)
    if score_sym < best_score:
        best_score = score_sym
        best_a = sym.copy()
        print(f"  Sym parent: {score_sym:.8f}")

    print(f"  Phase 1 best: {best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    # === Phase 2: Multi-resolution from structured inits (~500s) ===
    print("\n=== Phase 2: Multi-Resolution Structured Inits ===")
    t0 = time.monotonic()

    init_configs = [
        ('gaussian', 2.0),
        ('gaussian', 4.0),
        ('gaussian', 6.0),
        ('raised_cosine', 3.0),
        ('parabolic', 3.0),
        ('power', 2.0),
        ('sinc_window', 3.0),
    ]

    for kind, alpha in init_configs:
        remaining = deadline - time.monotonic() - 250
        if remaining < 30:
            break

        for n_target in n_targets:
            remaining = deadline - time.monotonic() - 200
            if remaining < 15:
                break

            # Multiresolution: start at 32, double up to n_target
            levels = []
            cur = 32
            while cur < n_target:
                cur = min(cur * 2, n_target)
                levels.append(cur)
            if levels[-1] != n_target:
                levels.append(n_target)

            a = structured_init(levels[0], kind, alpha)
            time_per_level = max(3, min(25, (remaining - 50) / (len(levels) + 1)))

            for level_n in levels:
                remaining = deadline - time.monotonic() - 150
                if remaining < 8:
                    break
                if len(a) != level_n:
                    a = upsample(a, level_n)

                conv_l = fast_convolve(a)
                M_l = conv_l.max()
                T_l = [M_l * 0.03, M_l * 0.01, M_l * 0.004]
                a, _ = lbfgs_time_limited(a, level_n, T_l,
                                          time_limit=min(time_per_level, remaining - 30),
                                          max_iter_per_T=1500)

            score = evaluate_score(a)
            label = f"{kind}_{n_target}_{alpha}"
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"  NEW BEST [{label}]: {score:.8f}")
            else:
                print(f"  [{label}]: {score:.8f}")

    print(f"  Phase 2 best: {best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    # === Phase 3: Deep L-BFGS on best solution (~350s) ===
    print("\n=== Phase 3: Deep Optimization ===")
    t0 = time.monotonic()

    n_best = len(best_a)
    conv = fast_convolve(best_a)
    M = conv.max()
    T_vals = [M * 0.03, M * 0.01, M * 0.005, M * 0.002, M * 0.001, M * 0.0005]
    remaining = deadline - time.monotonic() - 200
    if remaining > 20:
        a_opt, _ = lbfgs_time_limited(best_a, n_best, T_vals,
                                       time_limit=min(remaining, 250),
                                       max_iter_per_T=5000)
        score = evaluate_score(a_opt)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
        print(f"  Deep L-BFGS: {score:.8f}")

    # Symmetric L-BFGS
    sym = (best_a + best_a[::-1]) / 2.0
    sym = sym / sym.sum()
    conv_s = fast_convolve(sym)
    M_s = conv_s.max()
    T_s = [M_s * 0.02, M_s * 0.01, M_s * 0.004, M_s * 0.001]
    remaining = deadline - time.monotonic() - 150
    if remaining > 15:
        a_sym, _ = lbfgs_time_limited(sym, n_best, T_s,
                                       time_limit=min(remaining, 100),
                                       max_iter_per_T=3000)
        score = evaluate_score(a_sym)
        if score < best_score:
            best_score = score
            best_a = a_sym.copy()
        print(f"  Sym L-BFGS: {score:.8f}")

    print(f"  Phase 3 best: {best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    # === Phase 4: Iterated local search with perturbations (~150s) ===
    print("\n=== Phase 4: Iterated Local Search ===")
    t0 = time.monotonic()

    n_best = len(best_a)
    remaining = deadline - time.monotonic() - 80
    if remaining > 20:
        a_opt, _ = iterated_local_search(best_a, n_best, time_limit=min(remaining, 120),
                                          n_restarts=15)
        score = evaluate_score(a_opt)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
        print(f"  ILS: {score:.8f}")

    # Targeted perturbations
    for scale in [0.003, 0.01, 0.02]:
        remaining = deadline - time.monotonic() - 50
        if remaining < 10:
            break
        noise = rng.standard_normal(n_best) * scale
        a_pert = project_simplex(best_a + noise)
        conv_p = fast_convolve(a_pert)
        M_p = conv_p.max()
        T_p = [M_p * 0.005, M_p * 0.002, M_p * 0.001]
        a_opt, _ = lbfgs_time_limited(a_pert, n_best, T_p,
                                       time_limit=min(remaining - 10, 20),
                                       max_iter_per_T=2000)
        score = evaluate_score(a_opt)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  Perturb {scale}: NEW BEST {score:.8f}")
        else:
            print(f"  Perturb {scale}: {score:.8f}")

    print(f"  Phase 4 best: {best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    # === Phase 5: Try nearby n values (~80s) ===
    print("\n=== Phase 5: Nearby n ===")
    t0 = time.monotonic()

    n_best = len(best_a)
    for dn in [-128, -64, 64, 128]:
        n_try = n_best + dn
        if n_try < 128 or n_try > 2048:
            continue
        remaining = deadline - time.monotonic() - 40
        if remaining < 10:
            break

        a_try = upsample(best_a, n_try)
        conv_t = fast_convolve(a_try)
        M_t = conv_t.max()
        T_t = [M_t * 0.02, M_t * 0.008, M_t * 0.003]
        a_opt, _ = lbfgs_time_limited(a_try, n_try, T_t,
                                       time_limit=min(remaining - 10, 20),
                                       max_iter_per_T=2000)
        score = evaluate_score(a_opt)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  n={n_try}: NEW BEST {score:.8f}")
        else:
            print(f"  n={n_try}: {score:.8f}")

    print(f"  Phase 5 best: {best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    # === Phase 6: Final ultra-fine polish (~40s) ===
    print("\n=== Phase 6: Final Polish ===")
    t0 = time.monotonic()

    n_best = len(best_a)
    conv = fast_convolve(best_a)
    M = conv.max()
    T_final = [M * 0.001, M * 0.0004, M * 0.00015, M * 0.00005]
    remaining = deadline - time.monotonic() - 5
    if remaining > 5:
        a_opt, _ = lbfgs_time_limited(best_a, n_best, T_final,
                                       time_limit=min(remaining, 30),
                                       max_iter_per_T=5000)
        score = evaluate_score(a_opt)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
        print(f"  Ultra-fine: {score:.8f}")

    # Final symmetry check
    sym = (best_a + best_a[::-1]) / 2.0
    sym = sym / sym.sum()
    score = evaluate_score(sym)
    if score < best_score:
        best_score = score
        best_a = sym.copy()
        print(f"  Final sym: {score:.8f}")

    print(f"  Phase 6 best: {best_score:.8f} ({time.monotonic()-t0:.0f}s)")

    # Fallback
    if best_a is None:
        best_a = np.ones(512) / 512
        best_score = evaluate_score(best_a)

    # Final validation
    fs = evaluate_score(best_a)
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_a)}")
    return [float(x) for x in best_a]
```
