def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:

        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """FFT-based auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mv(p, a):
    """Compute M(p) @ a where M(p)_{j,m} = p[j+m]. p has length 2n-1, a has length n."""

def solve_lp_oracle(f, M, method='highs'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    # Build the constraint matrix efficiently
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method,
                      options={'presolve': True, 'max_time': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def active_set_newton(a0, time_limit, tol=1e-12, max_iter=2000):
    """
    Newton-type local optimization using KKT conditions.
    At the optimum, for the support S of a and active peak set P:
    - sum_{k in P} mu_k * a[k-j] = nu for all j in S
    - (a*a)[k] = t for all k in P
    - sum(a) = 1, a >= 0
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        if it % 100 == 0:
            print(f"    AS-Newton it={it}, obj={best_obj:.10f}")

        conv = fast_convolve(a)
        M = conv.max()
        support = np.where(a > 1e-10)[0]
        if len(support) == 0:
            break

        # Find active peaks (within 1% of max)
        threshold = M * (1.0 - 0.001)
        active_peaks = np.where(conv >= threshold)[0]

        # If only one active peak, use simple gradient step
        if len(active_peaks) <= 1:
            break

        # Build linear system for KKT on support
        # For j in support: sum_{k in active} mu_k * a[k-j] = nu
        # We solve for mu (and nu) by least squares
        # A_system[i, k_idx] = a[active_peaks[k_idx] - support[i]] (if valid)
        # b_system[i] = nu (unknown, so we add it as a column)

        m = len(support)
        p = len(active_peaks)

        # Build matrix: A_sys is m x (p+1), last column is ones (for nu)
        A_sys = np.zeros((m, p + 1))
        for i, j in enumerate(support):
            A_sys[i, -1] = 1.0  # nu column
            for k_idx, k in enumerate(active_peaks):
                idx = k - j
                if 0 <= idx < n:
                    A_sys[i, k_idx] = a[idx]

        # We want A_sys @ [mu; nu] = 0 (homogeneous, since RHS is 0)
        # Actually, the KKT condition is: sum_k mu_k * a[k-j] = nu
        # So: sum_k mu_k * a[k-j] - nu = 0
        # This gives: A_sys @ [mu; -nu] = 0
        # We want a non-trivial solution, so we use SVD
        try:
            U, S_vals, Vt = np.linalg.svd(A_sys)
            # Smallest singular vector
            sol = Vt[-1, :]
            mu_vals = sol[:p]
            nu_val = -sol[p]
        except np.linalg.LinAlgError:
            break

        # Ensure mu >= 0 (dual feasibility)
        mu_vals = np.maximum(mu_vals, 0)
        mu_sum = mu_vals.sum()
        if mu_sum < 1e-15:
            break
        mu_vals /= mu_sum

        # Use mu as weights for a gradient step
        # Gradient direction: -sum_k mu_k * a[k-j]
        grad = np.zeros(n)
        for k_idx, k in enumerate(active_peaks):
            if mu_vals[k_idx] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_arr = np.arange(js, je + 1)
            grad[idx_arr] -= mu_vals[k_idx] * a[k - idx_arr]

        # Take a small step
        best_candidate = a
        best_val = M
        for step in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            candidate = project_simplex(a + step * grad)
            c = fast_convolve(candidate)
            val = c.max()
            if val < best_val - 1e-16:
                best_val = val
                best_candidate = candidate
                break

        a = best_candidate

        if best_val < best_obj - 1e-15:
            best_obj = best_val
            best_a = a.copy()

        # Check convergence
        if abs(M - best_val) < 1e-14:
            # Check if we should continue with tighter tolerance
            if best_obj < M * (1.0 + 1e-10):
                break

    return best_a, best_obj

def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best if improved. Returns (best_seq, best_score)."""

def interp_sequence(parent, n_target):
    """Interpolate parent sequence to target length."""
    if len(parent) == n_target:
        return parent.copy()
    x_orig = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n_target)
    new_seq = np.interp(x_new, x_orig, parent)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return new_seq / s
