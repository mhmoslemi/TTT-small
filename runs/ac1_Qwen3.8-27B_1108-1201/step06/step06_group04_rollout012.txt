import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_matvec(w, a):
    """Compute W @ a where W[i,j] = w[i+j]. w has length 2n-1, a has length n."""
    return np.correlate(w, a, mode='valid')


def hankel_matvec_fft(w, a):
    """FFT-based Hankel matrix-vector product for large n."""
    n = len(a)
    N = 1 << (len(w) + n - 1).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    result = np.fft.irfft(W * np.conj(A), N)
    return result[:n]


def objective_value(a):
    """Compute the internal objective: 2*n*max_conv for normalized a."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def nesterov_peak(a0, time_limit, K_init=5, T_init=0.005, lr_init=0.002,
                  verbose=False):
    """
    Nesterov-accelerated subgradient with adaptive top-K peak focusing.
    Minimize max_k conv(a)[k] s.t. a in simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    mom = 0.9
    T = T_init
    K = K_init

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1

        # Nesterov lookahead
        y = a + mom * v

        # Compute convolution at lookahead point
        conv_y = fast_convolve(y)

        # Select top K peaks
        K_use = min(K, len(conv_y))
        top_k_indices = np.argpartition(conv_y, -K_use)[-K_use:]

        # Compute softmax weights over top K
        top_vals = conv_y[top_k_indices]
        scaled = top_vals / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for ki, k_idx in enumerate(top_k_indices):
            k_peak = int(k_idx)
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            grad[idx] += p[ki] * 2.0 * y[k_peak - idx]

        # Nesterov update
        y_new = project_simplex(y - lr * grad)
        v = y_new - y
        a = y_new

        # Evaluate
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt hyperparameters
        if no_improve % 20 == 0:
            lr *= 0.999
            mom = min(0.995, mom + 0.0001)

        if no_improve > 50:
            T *= 1.3
            K = min(K + 2, min(n, 30))
            lr *= 0.9

        if no_improve > 150:
            T *= 0.5
            K = max(3, K - 2)
            lr = lr_init * 0.7
            mom = 0.9

        if no_improve > 400:
            rng = np.random.default_rng((it * 7919 + 13) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            lr = lr_init
            mom = 0.9
            T = T_init
            K = K_init
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    nesterov_peak it={it}, obj={best_obj:.12f}, lr={lr:.6f}, T={T:.4f}, K={K}")

    return best_a, best_obj


def cg_solve_hankel(w, n, max_iters=500, tol=1e-10):
    """
    Solve W x = 1 using conjugate gradient, where W[i,j] = w[i+j].
    Returns x = W^{-1} * 1.
    """
    b = np.ones(n)
    x = np.ones(n) / n  # initial guess
    r = b - hankel_matvec(w, x)
    p = r.copy()
    rs_old = r @ r

    for _ in range(max_iters):
        if np.sqrt(rs_old) < tol:
            break
        Ap = hankel_matvec(w, p)
        pAp = p @ Ap
        if abs(pAp) < 1e-15:
            break
        alpha = rs_old / pAp
        x += alpha * p
        r_new = r - alpha * Ap
        rs_new = r_new @ r_new
        if np.sqrt(rs_new) < tol:
            r = r_new
            break
        p = r_new + (rs_new / rs_old) * p
        r = r_new
        rs_old = rs_new

    return x


def qp_jump(a0, time_limit, T=0.01, max_iters=100, verbose=False):
    """
    QP jump: solve min a^T W a s.t. sum(a)=1, a>=0
    where W is Hankel built from softmax(conv(a)/T).
    Uses CG for the inner linear solve with active set.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    t0 = time.monotonic()
    current = a.copy()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        conv = fast_convolve(current)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(len(conv)) / len(conv)
        else:
            w = w / w_sum

        # Add regularization for numerical stability
        w_reg = w.copy()
        w_reg[0] += 1e-8
        w_reg[-1] += 1e-8

        # Active set method with CG
        active = set(np.where(current > 1e-10)[0])
        if not active:
            active = set(range(n))

        for as_iter in range(30):
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            act = sorted(active)
            k = len(act)

            if k == 1:
                current = np.zeros(n)
                current[act[0]] = 1.0
                break
            if k == n:
                # Full system - use CG
                x_full = cg_solve_hankel(w_reg, n, max_iters=200)
                x_sum = x_full.sum()
                if x_sum > 1e-10:
                    current = x_full / x_sum
                    # Check for negative components
                    neg = np.where(current < -1e-8)[0]
                    if len(neg) > 0:
                        worst = neg[np.argmin(current[neg])]
                        active.discard(worst)
                    else:
                        current = np.maximum(current, 0)
                        current = current / current.sum()
                        break
                    continue
                else:
                    break

            # Subsystem: solve W_A x = 1 for active set
            # For small active sets, use direct solve
            if k <= 200:
                W_AA = np.array([[w_reg[act[i] + act[j]] for j in range(k)] for i in range(k)])
                W_AA += 1e-12 * np.eye(k)
                try:
                    x_act = np.linalg.solve(W_AA, np.ones(k))
                except np.linalg.LinAlgError:
                    x_act = np.linalg.lstsq(W_AA, np.ones(k), rcond=None)[0]
            else:
                # For larger active sets, use perturbed CG on full system
                x_full = cg_solve_hankel(w_reg, n, max_iters=100)
                x_act = x_full[act]

            x_sum = x_act.sum()
            if abs(x_sum) < 1e-12:
                current = np.zeros(n)
                for idx in act:
                    current[idx] = 1.0 / k
                break

            x_norm = x_act / x_sum

            # Check for violations
            min_val = x_norm.min()
            if min_val < -1e-8:
                # Remove worst violating index
                worst_local = int(np.argmin(x_norm))
                worst_global = act[worst_local]
                active.discard(worst_global)
                # Update current
                current = np.zeros(n)
                for i, idx in enumerate(act):
                    if idx != worst_global:
                        current[idx] = max(0.0, x_norm[i])
                s_cur = current.sum()
                if s_cur > 1e-12:
                    current = current / s_cur
                continue
            else:
                # Check KKT for inactive variables
                current = np.zeros(n)
                for i, idx in enumerate(act):
                    current[idx] = max(0.0, x_norm[i])

                Wa = hankel_matvec(w_reg, current)
                grad = 2.0 * Wa
                mu = np.mean([grad[idx] for idx in act])

                inactive = set(range(n)) - active
                if inactive:
                    min_viol = 0.0
                    min_j = -1
                    for j in sorted(inactive):
                        viol = grad[j] - mu
                        if viol < min_viol:
                            min_viol = viol
                            min_j = j
                    if min_viol < -1e-6:
                        active.add(min_j)
                        continue

                # Converged
                break

        # Blend with previous
        alpha = 0.5
        new_a = (1.0 - alpha) * a + alpha * current
        new_a = np.maximum(new_a, 0.0)
        s_new = new_a.sum()
        if s_new > 1e-12:
            new_a = new_a / s_new
        else:
            new_a = current

        obj_old = fast_convolve(a).max()
        obj_new = fast_convolve(new_a).max()

        if obj_new < obj_old:
            a = new_a
        else:
            # Try smaller blend
            alpha2 = 0.2
            new_a2 = (1.0 - alpha2) * a + alpha2 * current
            new_a2 = np.maximum(new_a2, 0.0)
            s2 = new_a2.sum()
            if s2 > 1e-12:
                new_a2 = new_a2 / s2
                obj_new2 = fast_convolve(new_a2).max()
                if obj_new2 < obj_old:
                    a = new_a2

    return a


def coord_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: for the peak convolution coefficient,
    try shifting mass from contributing to non-contributing positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Variables contributing to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)
        non_contrib = np.setdiff1d(np.arange(n), contrib)

        if len(contrib) == 0 or len(non_contrib) == 0:
            break

        # Find best single-coordinate move
        best_improvement = 0
        best_move = None

        # Try moving mass from each contributing position to non-contributing
        for i in contrib:
            if a[i] < 1e-10:
                continue
            for shift_frac in [0.05, 0.1, 0.2, 0.3]:
                shift = a[i] * shift_frac
                if shift < 1e-10:
                    continue
                # Try putting it at each non-contrib position (sample if too many)
                if len(non_contrib) > 20:
                    candidates = np.random.default_rng(it * 1000 + i).choice(
                        non_contrib, size=20, replace=False)
                else:
                    candidates = non_contrib

                for j in candidates:
                    a_trial = a.copy()
                    a_trial[i] -= shift
                    a_trial[j] += shift
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-12:
                        continue
                    a_trial = a_trial / s_t
                    obj_t = fast_convolve(a_trial).max()
                    improvement = best_obj - obj_t
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_move = (a_trial.copy(), i, j, shift_frac)

        if best_move is not None and best_improvement > 1e-13:
            a = best_move[0]
            best_obj = best_obj - best_improvement
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Try dual-coordinate: adjust two contributing positions
                if len(contrib) >= 2:
                    rng = np.random.default_rng((it * 31 + 7) % (2**31))
                    i1, i2 = rng.choice(contrib, size=2, replace=False)
                    for delta_frac in [0.02, 0.05, 0.1]:
                        delta = (a[i1] - a[i2]) * delta_frac
                        if abs(delta) < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[i1] += delta
                        a_trial[i2] -= delta
                        a_trial = np.maximum(a_trial, 0)
                        s_t = a_trial.sum()
                        if s_t < 1e-12:
                            continue
                        a_trial = a_trial / s_t
                        obj_t = fast_convolve(a_trial).max()
                        if obj_t < best_obj - 1e-13:
                            a = a_trial
                            best_obj = obj_t
                            best_a = a.copy()
                            no_improve = 0
                            break

            if no_improve > 200:
                break

    return best_a, best_obj


def smooth_grad_descent(a0, time_limit, T_init=0.001, lr_init=0.02,
                        verbose=False):
    """Smoothed gradient descent with log-sum-exp surrogate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit - 0.05:
        it += 1
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        p = np.exp(scaled)
        p = p / p.sum()

        # Gradient of surrogate: 2 * sum_k p[k] * (grad of conv[k])
        # grad of conv[k] w.r.t. a[i] = a[k-i] (for valid i)
        # Sum_k p[k] * a[k-i] = (p * a) correlation = hankel_matvec(p, a)
        grad = 2.0 * hankel_matvec(p, a)

        # Backtracking line search
        best_candidate = a
        best_val = conv.max()
        trial_lr = lr
        for _ in range(10):
            candidate = project_simplex(a - trial_lr * grad)
            val = fast_convolve(candidate).max()
            if val < best_val - 1e-14:
                best_val = val
                best_candidate = candidate
            trial_lr *= 0.5

        a = best_candidate
        obj = best_val

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 10:
            T *= 1.2
        if no_improve > 40:
            T *= 0.6
            lr = lr_init * 0.8
        if no_improve > 150:
            rng = np.random.default_rng((it * 61 + 3) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            T = T_init
            lr = lr_init
            no_improve = 0

    return best_a, best_obj


def generate_parametric(n, param_type, rng):
    """Generate parametric initializations."""
    x = np.linspace(0, 1, n)

    if param_type == 0:  # Uniform
        a = np.ones(n)
    elif param_type == 1:  # Gaussian
        mu = rng.uniform(0.2, 0.8)
        sigma = rng.uniform(0.05, 0.3)
        a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    elif param_type == 2:  # Two Gaussians
        mu1 = rng.uniform(0.1, 0.4)
        mu2 = rng.uniform(0.6, 0.9)
        sigma = rng.uniform(0.05, 0.2)
        a = np.exp(-0.5 * ((x - mu1) / sigma) ** 2) + \
            np.exp(-0.5 * ((x - mu2) / sigma) ** 2)
    elif param_type == 3:  # Sigmoid
        center = rng.uniform(0.3, 0.7)
        sharp = rng.uniform(5, 50)
        a = 1.0 / (1.0 + np.exp(sharp * (x - center)))
    elif param_type == 4:  # Cosine window
        width = rng.uniform(0.3, 0.9)
        center = rng.uniform(0.1, 0.9 - width)
        a = np.zeros(n)
        mask = (x >= center) & (x <= center + width)
        a[mask] = 0.5 + 0.5 * np.cos(2 * np.pi * (x[mask] - center) / width)
    elif param_type == 5:  # Random smooth
        n_fourier = rng.integers(3, 10)
        freqs = rng.integers(1, 5, size=n_fourier)
        phases = rng.uniform(0, 2 * np.pi, size=n_fourier)
        amps = rng.uniform(0.1, 1.0, size=n_fourier)
        a = np.ones(n) * 0.5
        for k in range(n_fourier):
            a += amps[k] * np.cos(2 * np.pi * freqs[k] * x + phases[k])
        a = np.maximum(a, 0.01)
    elif param_type == 6:  # Edge concentration
        d = np.minimum(x, 1.0 - x)
        a = np.exp(-d / rng.uniform(0.05, 0.3))
    elif param_type == 7:  # Power law
        a = np.power(np.maximum(x, 0.01), rng.uniform(-2, 2)) + \
            np.power(np.maximum(1-x, 0.01), rng.uniform(-2, 2))
    else:
        a = rng.dirichlet(np.ones(n) * 0.3)
        return a

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(parent, new_n):
    """Interpolate a sequence to a new length."""
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, parent)
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-15:
        result = np.ones(new_n) / new_n
    else:
        result = result / s
    return result


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search for auto-correlation minimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={ps:.10f}")
            try_seq(parent, "parent")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)
    print(f"Starting: n_parent={n_parent}, current_best={best_score:.10f}")

    # === Phase 1: Exploration (~200s) ===
    print("\n=== Phase 1: Exploration ===")
    t1 = time.monotonic()

    # Try different n values
    n_candidates = sorted(set([
        n_parent,
        max(128, n_parent - 128),
        n_parent + 128,
        n_parent + 256,
        max(128, n_parent // 2),
        512, 1024, 1536, 2048,
    ]))

    quick_results = []
    for n_try in n_candidates:
        remaining = deadline - time.monotonic() - 1200
        if remaining < 20:
            break
        if n_try > 4096:
            continue

        if n_try == n_parent:
            start = parent.copy()
        else:
            start = interpolate_sequence(parent, n_try)

        # Quick Nesterov
        t_alloc = min(remaining, 25)
        a_opt, _ = nesterov_peak(start, time_limit=t_alloc, K_init=5, T_init=0.005, lr_init=0.003)
        score = try_seq(a_opt, f"explore_n{n_try}")
        quick_results.append((score, a_opt.copy(), n_try))

    # Also try parametric inits at n_parent
    for pi in range(8):
        remaining = deadline - time.monotonic() - 1100
        if remaining < 15:
            break
        a_init = generate_parametric(n_parent, pi, rng)
        t_alloc = min(remaining, 15)
        a_opt, _ = nesterov_peak(a_init, time_limit=t_alloc, K_init=5, T_init=0.01, lr_init=0.005)
        try_seq(a_opt, f"param_{pi}")

    print(f"Phase 1 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t1:.0f}s)")

    # === Phase 2: Deep optimization with QP jumps (~800s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        iter_count = 0
        while time.monotonic() < deadline - 500:
            iter_count += 1
            remaining = deadline - time.monotonic() - 500
            if remaining < 30:
                break

            # Nesterov peak optimization
            t_nest = min(remaining * 0.6, 60)
            T_val = 0.003 + 0.002 * np.sin(iter_count * 0.3)
            K_val = 3 + (iter_count % 5) * 2
            lr_val = 0.001 + 0.002 * np.exp(-iter_count * 0.05)

            a_new, _ = nesterov_peak(current, time_limit=t_nest,
                                     K_init=K_val, T_init=max(0.001, T_val),
                                     lr_init=max(0.0005, lr_val))
            score = try_seq(a_new, f"deep_nest_{iter_count}")
            if best_seq is not None and score <= best_score:
                current = best_seq.copy()
                n_best = len(current)

            # QP jump
            remaining = deadline - time.monotonic() - 500
            if remaining > 30:
                t_qp = min(remaining * 0.3, 30)
                T_qp = 0.005 + 0.01 * (iter_count % 3)
                a_qp, _ = qp_jump(current, time_limit=t_qp, T=T_qp, max_iters=50)
                # Blend
                alpha = 0.3 + 0.3 * np.random.default_rng(iter_count).random()
                a_blend = (1 - alpha) * current + alpha * a_qp
                a_blend = np.maximum(a_blend, 0)
                a_blend = a_blend / a_blend.sum()
                try_seq(a_blend, f"qp_jump_{iter_count}")
                if best_seq is not None and score <= best_score:
                    current = best_seq.copy()
                    n_best = len(current)

            # Smoothed gradient
            remaining = deadline - time.monotonic() - 500
            if remaining > 20:
                t_sg = min(remaining * 0.2, 20)
                a_sg, _ = smooth_grad_descent(current, time_limit=t_sg,
                                              T_init=0.0005, lr_init=0.01)
                try_seq(a_sg, f"smooth_{iter_count}")
                if best_seq is not None:
                    current = best_seq.copy()
                    n_best = len(current)

    print(f"Phase 2 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t2:.0f}s)")

    # === Phase 3: Fine-tuning with coordinate descent + Nesterov (~400s) ===
    print("\n=== Phase 3: Fine-tuning ===")
    t3 = time.monotonic()

    if best_seq is not None:
        # Coordinate descent
        remaining = deadline - time.monotonic() - 150
        if remaining > 20:
            t_cd = min(remaining, 100)
            print(f"  Coord descent, t={t_cd:.0f}s")
            a_cd, _ = coord_descent(best_seq, time_limit=t_cd)
            try_seq(a_cd, "coord_desc")

        # Multiple Nesterov runs with different params
        for ni, (K_n, T_n, lr_n) in enumerate([
            (3, 0.001, 0.0005), (5, 0.003, 0.001), (8, 0.005, 0.0015),
            (3, 0.0005, 0.0003), (10, 0.008, 0.002),
        ]):
            remaining = deadline - time.monotonic() - 100
            if remaining < 15:
                break
            t_alloc = min(remaining, 60)
            a_opt, _ = nesterov_peak(best_seq, time_limit=t_alloc,
                                     K_init=K_n, T_init=T_n, lr_init=lr_n)
            try_seq(a_opt, f"fine_nest_{ni}")

        # Smoothed gradient with very low temperature
        remaining = deadline - time.monotonic() - 80
        if remaining > 15:
            t_alloc = min(remaining, 50)
            a_opt, _ = smooth_grad_descent(best_seq, time_limit=t_alloc,
                                          T_init=0.0001, lr_init=0.005)
            try_seq(a_opt, "smooth_fine")

    print(f"Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Final perturbations and verification (~200s) ===
    print("\n=== Phase 4: Final Polish ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Small perturbations with quick refinement
        for pi in range(10):
            remaining = deadline - time.monotonic() - 10
            if remaining < 10:
                break
            noise_scale = 0.0003 + 0.0005 * pi
            perturbed = project_simplex(best_seq +
                                        np.random.default_rng(seed + pi).standard_normal(n_best) * noise_scale)
            t_alloc = min(remaining, 15)
            a_opt, _ = nesterov_peak(perturbed, time_limit=t_alloc,
                                     K_init=3, T_init=0.0005, lr_init=0.0005)
            try_seq(a_opt, f"perturb_{pi}")

        # One final QP jump with very low T
        remaining = deadline - time.monotonic() - 20
        if remaining > 15:
            a_opt = qp_jump(best_seq, time_limit=min(remaining, 30), T=0.001, max_iters=100)[0]
            try_seq(a_opt, "final_qp")

        # Final coordinate descent
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            a_opt, _ = coord_descent(best_seq, time_limit=min(remaining, 20))
            try_seq(a_opt, "final_cd")

    print(f"Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
