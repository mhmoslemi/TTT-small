Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Log-parametrized L-BFGS optimization of log-sum-exp smoothed surrogate
- Multi-scale temperature annealing
- Direct peak-chasing coordinate steps
- Multi-n exploration with adaptive time allocation
- Nesterov-accelerated subgradient for final polish

import numpy as np
import time
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_to_a(x):
    """Convert log-parameters to probability vector."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()


def a_to_log_params(a, floor=1e-15):
    """Convert probability vector to log-parameters."""
    return np.log(np.maximum(a, floor))


def obj_and_grad(x, T):
    """Compute smoothed objective and gradient in log-parameter space."""
    a = softmax_to_a(x)
    conv = fast_convolve(a)
    max_c = conv.max()
    scaled = (conv - max_c) / T
    p = np.exp(scaled)
    p_sum = p.sum()
    obj = T * (np.log(p_sum) + max_c / T)
    p = p / p_sum
    # Gradient of f_T w.r.t. a: 2 * correlate(p, a, 'valid')
    grad_a = 2.0 * np.correlate(p, a, 'valid')
    # Transform to x-space: grad_x = a * (grad_a - <grad_a, a>)
    mean_grad = np.dot(grad_a, a)
    grad_x = a * (grad_a - mean_grad)
    return obj, grad_x


def raw_score(a):
    """Compute the actual (non-smoothed) score for a normalized sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    conv = fast_convolve(a)
    n = len(a)
    return 2.0 * n * conv.max()


def lqbfs_smoothed(x0, T, maxiter=5000, tol=1e-10):
    """Run L-BFGS-B on the smoothed objective."""
    result = minimize(

    return result.x, result.fun

def peak_chase(a0, time_limit, verbose=False):
    """Directly chase and reduce the dominant peak."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = raw_score(a)

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Variables contributing to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.arange(js, je + 1)

        if len(contrib) == 0:
            break

        # Find the pair (i, k_peak-i) with the largest product
        best_product = 0
        best_pair = None
        for idx in contrib:
            partner = k_peak - idx
            if 0 <= partner < n:
                prod = a[idx] * a[partner]
                if prod > best_product:
                    best_product = prod
                    best_pair = (idx, partner)

        if best_pair is None or best_product < 1e-15:
            break

        i, j = best_pair
        if i == j:
            # Single variable a[i]^2 contributes; reduce a[i]
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                delta = np.zeros(n)
                shift = a[i] * frac
                delta[i] = -shift
                # Distribute to non-contributing positions
                non_contrib = np.setdiff1d(np.arange(n), contrib)
                if len(non_contrib) > 0:
                    delta[non_contrib] = shift / len(non_contrib)
                candidate = project_simplex(a + delta)
                c = raw_score(candidate)
                if c < best_obj - 1e-12:
                    a = candidate
                    best_obj = c
                    best_a = a.copy()
                    break
        else:
            # Reduce a[i]*a[j] by shifting from the larger to a non-contributing position
            if a[i] >= a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            for frac in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
                shift = a[src] * frac
                if shift < 1e-12:
                    continue
                delta = np.zeros(n)
                delta[src] = -shift
                non_contrib = np.setdiff1d(np.arange(n), contrib)
                if len(non_contrib) > 0:
                    delta[non_contrib] = shift / len(non_contrib)
                else:
                    delta[dst] += shift
                candidate = project_simplex(a + delta)
                c = raw_score(candidate)
                if c < best_obj - 1e-12:
                    a = candidate
                    best_obj = c
                    best_a = a.copy()
                    break
            else:
                # Also try shifting to dst
                for frac in [0.02, 0.05, 0.1]:
                    shift = a[src] * frac
                    delta = np.zeros(n)
                    delta[src] = -shift
                    delta[dst] = shift
                    candidate = project_simplex(a + delta)
                    c = raw_score(candidate)
                    if c < best_obj - 1e-12:
                        a = candidate
                        best_obj = c
                        best_a = a.copy()
                        break

        if it % 200 == 0 and verbose:
            print(f"    peak_chase it={it}, obj={best_obj:.10f}")

        if it > 5000:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, max_rounds=50):
    """Coordinate descent: optimize each variable one at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = raw_score(a)

    t0 = time.monotonic()
    round_i = 0

    while time.monotonic() - t0 < time_limit and round_i < max_rounds:
        round_i += 1
        improved = False
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))

        # Only optimize variables that contribute to the top peaks
        top_k = min(5, 2 * n - 1)
        top_peaks = np.argpartition(conv, -top_k)[-top_k:]

        for kp in sorted(top_peaks, key=lambda x: -conv[x]):
            if time.monotonic() - t0 > time_limit - 1.0:
                break
            js = max(0, kp - n + 1)
            je = min(n - 1, kp)
            contrib = np.arange(js, je + 1)

            for idx in contrib:
                if time.monotonic() - t0 > time_limit - 0.5:
                    break
                # Try a few perturbations of a[idx]
                base = a.copy()
                for delta_frac in [-0.02, -0.01, 0.01, 0.02]:
                    delta = np.zeros(n)
                    shift = a[idx] * delta_frac
                    delta[idx] = -shift
                    # Redistribute to others proportionally
                    others = np.delete(np.arange(n), idx)
                    others_mask = a[others] > 1e-12
                    if others_mask.sum() > 0:
                        delta[others[others_mask]] = -shift * a[others[others_mask]] / a[others[others_mask]].sum()
                    else:
                        delta[others] = -shift / len(others)
                    candidate = project_simplex(base + delta)
                    c = raw_score(candidate)
                    if c < best_obj - 1e-12:
                        a = candidate
                        best_obj = c
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            break

    return best_a, best_obj

        remaining = deadline - time.monotonic() - 100
        if remaining > 10:
            t_alloc = min(remaining, 60)
            a_opt, _ = peak_chase(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "peak_chase_2")

    print(f"  Phase 3 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t3:.0f}s)")

    # === Phase 4: Nesterov fine-tuning (~200s) ===
    print("\n=== Phase 4: Nesterov Fine-tuning ===")
    t4 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        for T_n in [0.001, 0.0005, 0.0002]:
            remaining = deadline - time.monotonic() - 80
            if remaining < 10:
                break
            t_alloc = min(remaining, 60)
            print(f"  Nesterov T={T_n}, t={t_alloc:.0f}s")
            a_opt, _ = nesterov_smooth(best_seq, T=T_n, time_limit=t_alloc,
                                       lr_init=0.02, momentum_init=0.95)
            try_seq(a_opt, f"nesterov_T{T_n}")

        # Very fine L-BFGS
        remaining = deadline - time.monotonic() - 50
        if remaining > 10:
            x0 = a_to_log_params(best_seq)
            t_alloc = min(remaining, 40)
            maxiter = int(t_alloc * 3000)
            print(f"  Fine L-BFGS T=0.00005, t={t_alloc:.0f}s")
            x_opt, _ = lqbfs_smoothed(x0, 0.00005, maxiter=maxiter)
            a_opt = softmax_to_a(x_opt)
            try_seq(a_opt, "fine_lbfgs")

    print(f"  Phase 4 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t4:.0f}s)")

    # === Phase 5: Final perturbations + verification (~80s) ===
    print("\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple tiny perturbations with quick L-BFGS refinement
        for pi in range(8):
            remaining = deadline - time.monotonic() - 10
            if remaining < 5:
                break
            noise_scale = 0.0003 * (1 + pi * 0.5)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            x0 = a_to_log_params(f_pert)
            maxiter = min(2000, int(remaining * 1000))
            x_opt, _ = lqbfs_smoothed(x0, 0.0001, maxiter=maxiter)
            a_opt = softmax_to_a(x_opt)
            try_seq(a_opt, f"perturb_{pi}")

        # Final peak chase
        remaining = deadline - time.monotonic() - 15
        if remaining > 8:
            a_opt, _ = peak_chase(best_seq, time_limit=min(remaining, 20))
            try_seq(a_opt, "final_peak")

    print(f"  Phase 5 done. Best: {best_score:.10f} (elapsed {time.monotonic()-t5:.0f}s)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
