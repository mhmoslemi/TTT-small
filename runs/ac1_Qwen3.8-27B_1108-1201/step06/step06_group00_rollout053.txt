    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, non-negative)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_weights(conv, beta, top_k=None):
    """Compute LSE softmax weights, optionally restricted to top-k peaks."""
    if top_k is not None and top_k < len(conv):
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[peak_idx]
        scaled = beta * vals
        scaled_max = scaled.max()
        exp_vals = np.exp(scaled - scaled_max)
        total = exp_vals.sum()
        if total < 1e-300:
            w = np.ones(len(vals)) / len(vals)
        else:
            w = exp_vals / total
        weights_full = np.zeros(len(conv))
        weights_full[peak_idx] = w
        return weights_full
    else:
        scaled = beta * conv
        scaled_max = scaled.max()
        exp_vals = np.exp(scaled - scaled_max)
        total = exp_vals.sum()
        if total < 1e-300:
            return np.ones(len(conv)) / len(conv)
        return exp_vals / total


def compute_gradient_topk(a, peak_idx, weights):
    """Compute weighted gradient using only top-k peak positions."""
    n = len(a)
    grad = np.zeros(n)
    for idx, k in enumerate(peak_idx):
        w = weights[idx]
        if w < 1e-15:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * w * a[k - i_vals]
    return grad


def subgradient_optimize(a0, time_limit, beta_start=5.0, beta_end=50000.0, top_k=60):
    """LSE-smoothed subgradient method with backtracking line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    beta = beta_start
    lr = 1.0 / np.sqrt(n) * 2.0
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.5:
            break

        conv = fast_convolve(a)
        cur_obj = conv.max()

        # Compute gradient using top-k peaks with LSE weights
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[peak_idx]
        scaled = beta * vals
        scaled_max = scaled.max()
        exp_vals = np.exp(scaled - scaled_max)
        total = exp_vals.sum()
        if total < 1e-300:
            w_peaks = np.ones(len(vals)) / len(vals)
        else:
            w_peaks = exp_vals / total

        grad = compute_gradient_topk(a, peak_idx, w_peaks)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Backtracking line search
        improved = False
        step = lr
        for _ in range(12):
            a_new = project_simplex(a - step * grad)
            conv_new = fast_convolve(a_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-16:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                no_improve = 0
                break
            step *= 0.5

        if improved:
            if no_improve == 0:
                lr = min(lr * 1.01, 5.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.8
            if no_improve > 60:
                # Restart from best with perturbation
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                lr = max(1.0 / np.sqrt(n), 0.01)
                no_improve = 0

        # Adaptive beta: increase slowly
        frac = (time.monotonic() - t0) / time_limit
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, beta_end)

    return best_a, best_obj


def coord_descent(a0, time_limit, top_k=15, verbose=False):
    """Smart coordinate descent: move mass from hot to cold coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Find top peaks
        k_top = min(top_k, 2 * n - 1)
        peak_idx = np.argpartition(conv, -k_top)[-k_top:]
        peak_sorted = peak_idx[np.argsort(conv[peak_idx])[::-1]]

        # Compute contribution of each coordinate to top peaks
        contrib = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                i_vals = np.arange(lo, hi + 1)
                contrib[i_vals] += a[k - i_vals]

        # Identify hot and cold coordinates
        if contrib.max() < 1e-15:
            break
        hot_thresh = np.percentile(contrib[contrib > 0], 70) if (contrib > 0).any() else 0
        cold_thresh = np.percentile(contrib[contrib > 0], 30) if (contrib > 0).any() else 0

        hot_idx = np.where(contrib > hot_thresh)[0]
        cold_pool = np.where(contrib < cold_thresh)[0]
        if len(cold_pool) == 0:
            cold_pool = np.argsort(contrib)[:n // 4]

        # Shuffle for diversity
        rng.shuffle(hot_idx)
        rng.shuffle(cold_pool)

        # Try moves from hot to cold
        improved_this_round = False
        for src in hot_idx[:25]:
            if time.monotonic() - t0 > time_limit - 1:
                break
            if a[src] < 1e-10:
                continue

            for dst in cold_pool[:25]:
                if src == dst:
                    continue

                # Try several step sizes
                for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                    delta = a[src] * frac
                    if delta < 1e-12:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= delta
                    a_trial[dst] += delta
                    if a_trial[src] < -1e-15:
                        break
                    a_trial = np.maximum(a_trial, 0.0)
                    s_trial = a_trial.sum()
                    if s_trial < 1e-15:
                        break
                    a_trial /= s_trial

                    conv_trial = fast_convolve(a_trial)
                    if conv_trial.max() < cur_obj - 1e-14:
                        a = a_trial
                        cur_obj = conv_trial.max()
                        improved_this_round = True
                        # Recompute contribution after improvement
                        contrib = np.zeros(n)
                        for k in peak_idx:
                            k = int(k)
                            lo = max(0, k - n + 1)
                            hi = min(n - 1, k)
                            if lo <= hi:
                                i_vals = np.arange(lo, hi + 1)
                                contrib[i_vals] += a[k - i_vals]
                        break

        if not improved_this_round:
            # Try focusing on the single worst peak
            conv = fast_convolve(a)
            k_worst = int(conv.argmax())
            lo = max(0, k_worst - n + 1)
            hi = min(n - 1, k_worst)
            if lo < hi:
                i_vals = np.arange(lo, hi + 1)
                contribs = a[i_vals] * a[k_worst - i_vals]
                src = i_vals[np.argmax(contribs)]

                # Try moving to several destinations
                sorted_cold = cold_pool[np.argsort(contrib[cold_pool])][:15]
                for dst in sorted_cold:
                    if dst == src:
                        continue
                    for frac in [0.01, 0.03, 0.05]:
                        delta = a[src] * frac
                        if delta < 1e-12:
                            break
                        a_trial = a.copy()
                        a_trial[src] -= delta
                        a_trial[dst] += delta
                        a_trial = np.maximum(a_trial, 0)
                        s_trial = a_trial.sum()
                        if s_trial < 1e-15:
                            break
                        a_trial /= s_trial
                        conv_trial = fast_convolve(a_trial)
                        if conv_trial.max() < best_obj - 1e-14:
                            a = a_trial
                            best_obj = conv_trial.max()
                            best_a = a.copy()
                            improved_this_round = True
                            break
                    if improved_this_round:
                        break

            if not improved_this_round:
                # Add small perturbation to escape
                pert_scale = 0.001 + 0.001 * min(stall_count / 20.0, 1.0)
                a = project_simplex(a + rng.standard_normal(n) * pert_scale)
                if stall_count > 50:
                    a = best_a.copy()
                    stall_count = 0

    return best_a, best_obj


def freq_perturb(a, rng, strength=0.03):
    """Perturb in frequency domain for diversity."""
    n = len(a)
    F = np.fft.rfft(a)
    mag = np.abs(F)
    phase = np.angle(F)
    pert = rng.standard_normal(len(mag)) * strength
    mag_new = np.maximum(mag * (1.0 + pert), 0)
    F_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(F_new, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: LSE subgradient with adaptive beta
    Phase 2: Smart coordinate descent
    Phase 3: Iterated local search (perturb + re-optimize)
    Phase 4: Final high-precision polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_score = float('inf')
    best_seq = None

    def try_seq(a):
        nonlocal best_score, best_seq
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    a = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            a = p / p.sum()
            parent_score = evaluate_sequence(a.tolist())
            print(f"Parent: n={len(a)}, score={parent_score:.10f}")
            try_seq(a)
    except Exception as e:
        print(f"Could not load parent: {e}")

    if a is None:
        a = np.ones(768) / 768.0
        try_seq(a)

    n = len(a)
    rng = np.random.default_rng(seed)
    total_time = float(budget_s) - 30.0

    # === Phase 1: Subgradient (25% of time) ===
    print(f"\n=== Phase 1: Subgradient (n={n}) ===")
    t1 = total_time * 0.25
    t1_alloc = min(t1, 450)
    print(f"  Allocating {t1_alloc:.0f}s, beta: 5 -> 20000")
    a_opt, obj = subgradient_optimize(a, time_limit=t1_alloc,
                                       beta_start=5.0, beta_end=20000.0, top_k=50)
    try_seq(a_opt)

    # Second subgradient run with higher beta
    t_rem = t1 - (time.monotonic() - (deadline - total_time))
    if t_rem > 30:
        a_cur = best_seq if best_seq is not None else a_opt
        t_alloc = min(t_rem, 150)
        print(f"  Second subgrad pass, {t_alloc:.0f}s, beta: 50 -> 50000")
        a_opt2, _ = subgradient_optimize(a_cur, time_limit=t_alloc,
                                          beta_start=50.0, beta_end=50000.0, top_k=40)
        try_seq(a_opt2)

    # === Phase 2: Coordinate Descent (35% of time) ===
    print(f"\n=== Phase 2: Coordinate Descent (n={n}) ===")
    a_cur = best_seq if best_seq is not None else a_opt
    t2 = total_time * 0.35
    t2_end = (deadline - total_time) + t1 + t2
    t2_alloc = min(t2, t2_end - time.monotonic())
    if t2_alloc > 20:
        print(f"  Allocating {t2_alloc:.0f}s")
        a_opt, obj = coord_descent(a_cur, time_limit=t2_alloc, top_k=15)
        try_seq(a_opt)

    # Second CD pass
    a_cur = best_seq if best_seq is not None else a_opt
    t_rem = t2_end - time.monotonic()
    if t_rem > 20:
        print(f"  Second CD pass, {t_rem:.0f}s")
        a_opt, _ = coord_descent(a_cur, time_limit=t_rem, top_k=10)
        try_seq(a_opt)

    # === Phase 3: Iterated Local Search (30% of time) ===
    print(f"\n=== Phase 3: Iterated Local Search ===")
    t3_end = deadline - total_time * 0.10
    a_cur = best_seq if best_seq is not None else a_opt

    ills_iter = 0
    while time.monotonic() < t3_end:
        ills_iter += 1
        remaining = t3_end - time.monotonic()
        if remaining < 40:
            break

        # Perturb
        if ills_iter <= 5:
            strength = 0.003
        elif ills_iter <= 15:
            strength = 0.005
        else:
            strength = 0.008 + 0.002 * min((ills_iter - 15) / 10.0, 1.0)

        a_pert = project_simplex(a_cur + rng.standard_normal(n) * strength)

        # Brief subgradient
        t_sub = min(remaining * 0.4, 50)
        if t_sub < 10:
            break
        a_sg, _ = subgradient_optimize(a_pert, time_limit=t_sub,
                                        beta_start=100.0, beta_end=50000.0, top_k=30)
        try_seq(a_sg)

        # Brief coordinate descent
        a_best_local = best_seq if best_seq is not None else a_sg
        t_cd = min(remaining * 0.3, 30)
        if t_cd < 5:
            break
        a_cd, _ = coord_descent(a_best_local, time_limit=t_cd, top_k=10)
        try_seq(a_cd)

        # Update current best for next iteration
        a_cur = best_seq if best_seq is not None else a_cd

        if ills_iter % 5 == 0:
            print(f"  ILS iter {ills_iter}, best={best_score:.10f}")

    # === Phase 4: Final Polish (10% of time) ===
    print(f"\n=== Phase 4: Final Polish ===")
    a_cur = best_seq if best_seq is not None else a
    t4 = deadline - time.monotonic()

    if t4 > 30:
        # Very high beta subgradient (focus on top few peaks)
        t_sub = t4 * 0.4
        print(f"  High-beta subgrad, {t_sub:.0f}s")
        a_opt, _ = subgradient_optimize(a_cur, time_limit=t_sub,
                                         beta_start=500.0, beta_end=200000.0, top_k=20)
        try_seq(a_opt)
        a_cur = best_seq if best_seq is not None else a_opt

    # Final coordinate descent with very small steps
    t_rem = deadline - time.monotonic() - 5
    if t_rem > 15:
        print(f"  Final CD, {t_rem:.0f}s")
        a_opt, _ = coord_descent(a_cur, time_limit=t_rem, top_k=8)
        try_seq(a_opt)

    # Tiny random perturbations of the best
    a_cur = best_seq if best_seq is not None else a_opt
    t_rem = deadline - time.monotonic() - 3
    if t_rem > 10:
        print(f"  Final micro-perturbations, {t_rem:.0f}s")
        for i in range(50):
            if time.monotonic() > deadline - 3:
                break
            for strength in [0.0005, 0.001, 0.002]:
                a_pert = project_simplex(a_cur + rng.standard_normal(n) * strength)
                try_seq(a_pert)
                if best_seq is not None:
                    a_cur = best_seq.copy()

    # Final verification
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        final = np.minimum(final, 1000.0)
        final_score = float(evaluate_sequence(final.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.12f}, n={len(final)}")
            return [float(x) for x in final]

    print("WARNING: Using fallback")
    fallback = np.ones(448) / 448.0
    return [float(x) for x in fallback]
